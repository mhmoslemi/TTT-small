def _c5_value(h):
    N = h.size
    dx = 2.0 / N
    return np.max(np.correlate(h, 1-h, mode='full')) * dx

def _repair(v):
    # Clip to [0,1] then adjust sum to N/2 exactly
    # Use scaling factor
    np.clip(v, 0, 1, out=v)
    N = v.size
    target_sum = N / 2.0
    cur_sum = v.sum()
    if cur_sum == 0:
        v[:] = target_sum / N
    else:
        factor = target_sum / cur_sum
        v *= factor
        np.clip(v, 0, 1, out=v)
        # Readjust if sum still off due to clipping:
        cur_sum = v.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            factor2 = target_sum / cur_sum
            v *= factor2
            np.clip(v, 0, 1, out=v)
    return v

        if elapsed >= time_limit:

        T = T0 * (1 - frac) + T_end * frac  # linear cooling

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _repair_vector(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and scale to have sum = len(v)/2 exactly."""
    np.clip(v, 0.0, 1.0, out=v)
    n = v.size
    target = n / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = target / n
        np.clip(v, 0.0, 1.0, out=v)
        return v
    factor = target / s
    v *= factor
    np.clip(v, 0.0, 1.0, out=v)
    s2 = v.sum()
    if abs(s2 - target) > 1e-12:
        factor2 = target / s2
        v *= factor2
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

            idxs = rng.choice(N, size=min(12, N), replace=False)
            delta = rng.random(len(idxs)) - 0.5
            delta -= delta.mean()  # zero‑sum perturbation
            new = cur.copy()
            new[idxs] += delta * max_eps
            new = _repair_vector(new)
            new_score = _c5_value(new)
            cur, cur_score = new, new_score
            if new_score < best_score:
                best, best_score = new.copy(), new_score
            continue

        # pick a pair of distinct indices
        i, j = rng.integers(0, N, size=2)
        while i == j:
            j = rng.integers(0, N)
        hi, hj = cur[i], cur[j]

        # feasible epsilon range that keeps both entries in [0,1]
        eps_low = max(-hi, hj - 1.0)      # lower bound
        eps_high = min(1.0 - hi, hj)      # upper bound
        if eps_low >= eps_high:
            continue
        eps = rng.uniform(eps_low, eps_high)
        # cap by max_eps magnitude
        if eps > max_eps:
            eps = max_eps
        elif eps < -max_eps:
            eps = -max_eps
        if eps == 0.0:
            continue

        new = cur.copy()
        new[i] = hi + eps
        new[j] = hj - eps
        # feasibility already guaranteed, but double‑check (tiny numeric issues)
        if not (0.0 <= new[i] <= 1.0 and 0.0 <= new[j] <= 1.0):
            continue

        new_score = _c5_value(new)
        delta_score = new_score - cur_score
        if delta_score < 0.0 or rng.random() < math.exp(-delta_score / (T + 1e-15)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
                if best_score <= target_bound:
                    # early termination if we already beat target
                    return best, best_score
    return best, best_score

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that minimises C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Obtain initial construction
    # ------------------------------------------------------------------
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        # fallback: simple half‑ones / half‑zeros vector
        n_fallback = 100
        h0 = np.concatenate([np.ones(n_fallback // 2), np.zeros(n_fallback - n_fallback // 2)],
                            dtype=np.float64)
    h0 = _repair_vector(h0)
    n0 = h0.size

    # ------------------------------------------------------------------
    # Resolution schedule (doubling until a limit)
    # ------------------------------------------------------------------
    max_n = 800                     # upper bound for speed / memory
    levels = [n0]
    while levels[-1] < max_n:
        nxt = levels[-1] * 2
        if nxt > max_n:
            nxt = max_n
        if nxt == levels[-1]:
            break
        levels.append(nxt)

    start_total = time.time()
    time_budget = budget_s * 0.95   # keep a small safety margin
    target_bound = 0.3808

    best_h = h0
    best_score = _c5_value(best_h)

    # allocate time per level (geometric decay)
    num_levels = len(levels)
    weights = np.array([0.5 ** i for i in range(num_levels)], dtype=float)
    weights /= weights.sum()
    level_times = weights * time_budget

    for lvl, N in enumerate(levels):
        # check global time
        if time.time() - start_total > budget_s * 0.99:
            break

        # interpolate previous best to current resolution
        if N != best_h.size:
            x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, N, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
        else:
            h_interp = best_h.copy()
        h_start = _repair_vector(h_interp)

        # ----------- SA on this level ----------
        lvl_time = level_times[lvl]
        # first pass from interpolated start
        cur_h, cur_score = _simulated_annealing(h_start, lvl_time, rng, target_bound)

        # try a few random restarts within remaining time of this level
        elapsed_lvl = time.time() - start_total - sum(level_times[:lvl])  # time spent so far on this level
        remaining_lvl = lvl_time - elapsed_lvl
        restarts = 0
        while remaining_lvl > 3.0 and restarts < 3:
            rand_vec = rng.random(N)
            rand_vec = _repair_vector(rand_vec)
            sub_budget = remaining_lvl * 0.5
            cand_h, cand_score = _simulated_annealing(rand_vec, sub_budget, rng, target_bound)
            if cand_score < cur_score:
                cur_h, cur_score = cand_h, cand_score
            remaining_lvl = lvl_time - (time.time() - start_total - sum(level_times[:lvl]))
            restarts += 1

        # update global best
        if cur_score < best_score:
            best_h, best_score = cur_h, cur_score
            if best_score <= target_bound:
                break

    # final repair and evaluation
    best_h = _repair_vector(best_h)
    final_score = _c5_value(best_h)

    return best_h, float(final_score), best_h.size
