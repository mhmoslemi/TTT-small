import time
import numpy as np

def upsample(arr, new_n):
    """Linear interpolation of arr (length m) to length new_n."""
    m = arr.shape[0]
    if m == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def project_onto_box_sum(v, s, lo=0.0, hi=1.0, max_iter=60):
    """Euclidean projection of v onto {x: lo<=x_i<=hi, sum x_i = s}."""
    n = v.size
    # trivial cases
    if s <= lo * n:
        return np.full_like(v, lo)
    if s >= hi * n:
        return np.full_like(v, hi)

    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo

    def f(lam):
        return np.clip(v - lam, lo, hi).sum()

    # widen interval if needed
    if f(lam_low) < s:
        lam_low = np.min(v) - hi * 10
    if f(lam_high) > s:
        lam_high = np.max(v) - lo * 10

    for _ in range(max_iter):
        lam_mid = 0.5 * (lam_low + lam_high)
        fm = f(lam_mid)
        if abs(fm - s) < 1e-12:
            break
        if fm > s:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - lam_mid, lo, hi)

def compute_c5(h):
    """Maximum overlap integral (discretized)."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    """Gradient of correlation term for a given integer shift."""
    n = h.size
    term1 = np.zeros(n, dtype=h.dtype)
    term2 = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        term1[: n - shift] = h[shift:]
        term2[shift:] = h[: n - shift]
    else:
        s = -shift
        term1[s:] = h[: n - s]
        term2[: n - s] = h[s:]
    return 1.0 - term1 - term2

def thue_morse(n):
    """Balanced Thue–Morse binary sequence (0/1)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    return seq

def random_binary_balanced(n, rng):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def pgd(h0, max_iters=5000, step0=0.8):
    """Projected gradient descent on max‑correlation objective."""
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    for it in range(1, max_iters + 1):
        # worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)

        step = step0 / np.sqrt(it)
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            eta = step * factor
            candidate = h - eta * grad
            candidate = project_onto_box_sum(candidate, target_sum)
            cand_score = compute_c5(candidate)
            if cand_score < cur_score - 1e-14:
                h = candidate
                cur_score = cand_score
                improved = True
                break
        if not improved:
            # no improvement with reduced steps; keep current h
            pass
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    target_n = 256
    target_sum = target_n / 2.0

    best_h = None
    best_score = float('inf')
    candidates = []

    # 1. Existing construction (if any)
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != target_n:
            init = upsample(init, target_n)
        init = project_onto_box_sum(init, target_sum)
        candidates.append(init)

    # 2. Structured seeds
    if (target_n & (target_n - 1)) == 0:
        candidates.append(thue_morse(target_n))

    x = np.linspace(0.0, 2.0, target_n, endpoint=False)

    tri = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    tri = project_onto_box_sum(tri, target_sum)
    candidates.append(tri)

    sin2 = np.sin(np.pi * x / 2.0) ** 2
    sin2 = project_onto_box_sum(sin2, target_sum)
    candidates.append(sin2)

    for _ in range(3):
        candidates.append(random_binary_balanced(target_n, rng))
    for _ in range(3):
        v = rng.random(target_n)
        candidates.append(project_onto_box_sum(v, target_sum))

    # 3. Optimize structured candidates
    for base_h in candidates:
        if time.time() - start_time > budget_s - 5.0:
            break
        iters = 8000 if best_h is None else 6000
        h_opt, s_opt = pgd(base_h, max_iters=iters, step0=0.8)
        if s_opt < best_score:
            best_score = s_opt
            best_h = h_opt.copy()

    # 4. Random‑restart local search
    while time.time() - start_time < budget_s - 5.0:
        v = rng.random(target_n)
        h_start = project_onto_box_sum(v, target_sum)
        h_opt, s_opt = pgd(h_start, max_iters=3000, step0=0.8)
        if s_opt < best_score:
            best_score = s_opt
            best_h = h_opt.copy()

    # 5. Final refinement on the best found vector
    if best_h is not None and time.time() - start_time < budget_s - 2.0:
        h_opt, s_opt = pgd(best_h, max_iters=8000, step0=0.8)
        if s_opt < best_score:
            best_h = h_opt
            best_score = s_opt

    # 6. Fallback if nothing succeeded
    if best_h is None:
        best_h = np.full(target_n, 0.5, dtype=float)
        best_h = project_onto_box_sum(best_h, target_sum)
        best_score = compute_c5(best_h)
    else:
        best_h = project_onto_box_sum(best_h, target_sum)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_h.size
