import time, math
import numpy as np


def _compute_c5(h: np.ndarray) -> float:
    """C5 value of a step function represented by array h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def _random_feasible(n: int) -> np.ndarray:
    """Return a random vector in [0,1] with sum exactly n/2 (up to tiny eps)."""
    target = n * 0.5
    v = np.random.rand(n)
    # scale to target sum (ignoring bounds for a moment)
    v *= target / v.sum()
    v = np.clip(v, 0.0, 1.0)
    # iterative water‑filling to fix the sum while respecting bounds
    for _ in range(10):
        cur = v.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        mask = (v > 0.0) & (v < 1.0)
        cnt = mask.sum()
        if cnt == 0:
            # all variables are at a bound – distribute the remaining diff uniformly
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
            continue
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v


def _hill_climb(start_h: np.ndarray, max_iters: int, deadline: float) -> tuple:
    """
    Simple (simulated‑annealing style) hill climbing.
    Returns best vector found and its C5 value.
    """
    n = start_h.shape[0]
    best_h = start_h.copy()
    best_val = _compute_c5(best_h)

    cur_h = start_h.copy()
    cur_val = best_val

    # temperature schedule
    temp = 0.08
    cooling = 0.9997

    for _ in range(max_iters):
        if time.time() >= deadline:
            break

        # choose move type
        if np.random.rand() < 0.6:          # delta move (preserves sum)
            i, j = np.random.randint(n, size=2)
            while i == j:
                j = np.random.randint(n)
            lo = max(-cur_h[i], cur_h[j] - 1.0)
            hi = min(1.0 - cur_h[i], cur_h[j])
            if lo >= hi:
                continue
            d = np.random.uniform(lo, hi)
            new_h = cur_h.copy()
            new_h[i] += d
            new_h[j] -= d
        else:                               # simple swap move
            i, j = np.random.randint(n, size=2)
            while i == j:
                j = np.random.randint(n)
            new_h = cur_h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]

        new_val = _compute_c5(new_h)

        if new_val < cur_val:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_val - cur_val) / max(temp, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val

        temp *= cooling
        if temp < 1e-7:
            temp = 1e-7

    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ------------------------------------------------------------------
    # 1) obtain the parent construction (if any)
    if 'initial_h_values' in globals():
        base_h = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        # uniform 0.5 works and satisfies the integral constraint automatically
        base_h = np.full(64, 0.5, dtype=np.float64)

    base_n = base_h.shape[0]

    # ------------------------------------------------------------------
    # 2) candidate grid sizes (we try the base size and one refinement)
    candidates = [base_n]
    if base_n * 2 <= 200:
        candidates.append(base_n * 2)

    # iterate larger grids first – they have more flexibility
    candidates = sorted(set(candidates), reverse=True)

    overall_best_h = None
    overall_best_val = np.inf
    overall_best_n = None

    for n in candidates:
        if time.time() >= deadline:
            break

        # ----- initialise starting vectors for this grid size -------------
        if n == base_n:
            # use the given construction directly
            start_vectors = [base_h.copy()]
        else:
            # repeat the parent pattern uniformly
            repeat = n // base_n
            start_vectors = [np.repeat(base_h, repeat).astype(np.float64)]

        # add a few random starts to diversify the search
        for _ in range(4):
            start_vectors.append(_random_feasible(n))

        # ----- run hill climbing from each start -------------------------
        for sv in start_vectors:
            if time.time() >= deadline:
                break
            # limit the number of iterations adaptively
            max_iters = 200000
            # give each start a slice of the remaining budget
            sub_deadline = deadline
            best_h, best_val = _hill_climb(sv, max_iters, sub_deadline)

            if best_val < overall_best_val:
                overall_best_h = best_h
                overall_best_val = best_val
                overall_best_n = n
                # early exit if we already beat the target
                if overall_best_val < 0.38080:
                    break
        if overall_best_val < 0.38080:
            break

    # ------------------------------------------------------------------
    # Fallback – should never happen but keep a valid answer
    if overall_best_h is None:
        overall_best_h = base_h.copy()
        overall_best_val = _compute_c5(overall_best_h)
        overall_best_n = base_n

    # Ensure the returned vector respects the constraints (tiny numeric drift possible)
    # Clip and re‑project to exact sum
    overall_best_h = np.clip(overall_best_h, 0.0, 1.0)
    target_sum = overall_best_n * 0.5
    diff = target_sum - overall_best_h.sum()
    if abs(diff) > 1e-10:
        # distribute the tiny residual uniformly over interior entries
        mask = (overall_best_h > 0.0) & (overall_best_h < 1.0)
        cnt = mask.sum()
        if cnt > 0:
            overall_best_h[mask] += diff / cnt
            overall_best_h = np.clip(overall_best_h, 0.0, 1.0)

    return overall_best_h, overall_best_val, int(overall_best_n)
