        # worst shift (the one attaining the max)
        idx_max = int(np.argmax(corr))
        k = idx_max - (n - 1)

        i1, i0 = get_swap_indices(h, k)

        # propose the swap
        h_new = h.copy()
        h_new[i1] = 0
        h_new[i0] = 1

        new_c5, _ = compute_cost_vec(h_new)

        # accept if not worse, otherwise with Metropolis probability
        if new_c5 <= cur_c5:
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        else:
            prob = np.exp(-(new_c5 - cur_c5) / (T + 1e-12))
            if np.random.rand() < prob:
                h = h_new
                cur_c5 = new_c5

        # temperature schedule (slow exponential cooling)
        iteration += 1
        T = T0 * (0.9999 ** iteration)
        if T < 1e-6:
            T = T0

    # final evaluation of the best vector found
    best_c5, _ = compute_cost_vec(best_h)

    return (best_h.astype(float), float(best_c5), n)
