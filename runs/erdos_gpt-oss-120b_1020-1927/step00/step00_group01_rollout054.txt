import numpy as np
import time
from scipy.optimize import minimize, Bounds

def _c5_score(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def _c5_grad(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    idxs = np.flatnonzero(corr == max_val)
    shifts = idxs - (n - 1)
    # accumulate gradient contributions
    grad = np.zeros_like(h)
    for s in shifts:
        # for each index, derivative = 1 - h_{i+s} - h_{i-s} (when defined)
        grad += 1.0
        if s >= 0:
            # i+s within bounds for i=0..n-1-s
            grad[:n - s] -= h[s:]
            # i-s within bounds for i=s..n-1
            grad[s:] -= h[:n - s]
        else:
            t = -s
            # i+t within bounds for i=0..n-1-t
            grad[:n - t] -= h[t:]
            # i-t within bounds for i=t..n-1
            grad[t:] -= h[:n - t]
    # average over active shifts (convex combination)
    if len(shifts) > 1:
        grad = grad / len(shifts)
    return dx * grad

def _proj_box_simplex(y, target, max_iter=60):
    # Find lambda such that sum(clip(y - lam, 0, 1)) = target.
    # Using bisection. y is 1-D array.
    lo = -1.0
    hi = 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

        free = (h > 0.0) & (h < 1.0)

            h = np.clip(h, 0.0, 1.0)
            continue
        h[free] += err / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
import time
from scipy.optimize import minimize, Bounds

def _c5_score(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def _c5_grad(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    idxs = np.flatnonzero(corr == max_val)
    shifts = idxs - (n - 1)
    grad = np.zeros_like(h)
    for s in shifts:
        grad += 1.0
        if s >= 0:
            grad[:n - s] -= h[s:]
            grad[s:] -= h[:n - s]
        else:
            t = -s
            grad[:n - t] -= h[t:]
            grad[t:] -= h[:n - t]
    if len(shifts) > 1:
        grad = grad / len(shifts)
    return dx * grad

def _proj_box_simplex(y, target, max_iter=60):
    lo = -1.0
    hi = 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def _random_feasible_h(n, rng):
    target = n / 2.0
    h = rng.random(n)
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        err = target - h.sum()
        if abs(err) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        cnt = np.count_nonzero(free)
        if cnt == 0:
            h += err / n
            h = np.clip(h, 0.0, 1.0)
            continue
        h[free] += err / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _optimize_h(h0, maxiter=200):
    n = h0.shape[0]
    bounds = Bounds(0.0, 1.0)
    cons = ({'type': 'eq', 'fun': lambda h, n=n: np.sum(h) - n/2.0},)
    res = minimize(_c5_score,
                   h0,
                   method='SLSQP',
                   jac=_c5_grad,
                   bounds=bounds,
                   constraints=cons,
                   options={'maxiter': maxiter,
                            'ftol': 1e-12,
                            'disp': False})
    return res
