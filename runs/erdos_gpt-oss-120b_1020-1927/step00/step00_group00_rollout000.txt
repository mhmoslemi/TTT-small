import time

from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper: projection onto the box [0,1] intersected with the
# hyperplane sum(x) = target.  Uses a simple binary search on the
# shift lambda such that x = clip(v + lambda,0,1) has the required sum.
# ------------------------------------------------------------
def _project_box_sum(v, target):
    low = -np.max(v)                 # makes all entries <=0 after shift
    high = 1.0 - np.min(v)           # makes all entries >=1 after shift
    for _ in range(50):
        mid = (low + high) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2.0
    x = np.clip(v + lam, 0.0, 1.0)

    # tiny correction if numerical error left a small diff
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            inc = diff / free.sum()
            x[free] += inc
            x = np.clip(x, 0.0, 1.0)
    return x

# ------------------------------------------------------------
# Smooth approximation of max_{k} C_k where
# C_k = dx * sum_i h_i (1 - h_{i+k})
# ------------------------------------------------------------
def _smooth_max_obj(h, dx, alpha):
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    m = conv.max()
    exp_shift = np.exp(alpha * (conv - m))
    return m + np.log(exp_shift.sum()) / alpha

def _smooth_max_jac(h, dx, alpha):
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    m = conv.max()
    exp_shift = np.exp(alpha * (conv - m))
    w = exp_shift / exp_shift.sum()          # weights sum to 1
    n = h.size
    grad = np.zeros_like(h)
    one_minus = 1.0 - h

    # contribution of each lag
    for lag in range(-(n - 1), n):
        wk = w[lag + n - 1] * dx
        if wk == 0.0:
            continue
        if lag >= 0:
            # first term: (1 - h_{j+lag})
            grad[: n - lag] += wk * one_minus[lag:]
            # second term: -h_{j-lag}
            grad[lag:] += wk * (-h[: n - lag])
        else:
            mlag = -lag
            # first term: (1 - h_{j-mlag})
            grad[mlag:] += wk * one_minus[: n - mlag]
            # second term: -h_{j+mlag}
            grad[: n - mlag] += wk * (-h[mlag:])
    return grad
