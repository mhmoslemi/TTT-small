    constraints = [h >= 0, h <= 1, cvxpy.sum(h) == desired_sum]

    i = rng.integers(n)
    j = rng.integers(n-1)
    if j >= i:
        j += 1
    # Determine allowed transfer from i to j
    hi = h[i]; hj = h[j]
    # If hi > hj, we can transfer from i to j up to min(hi, 1 - hj)
    # else transfer opposite direction.
    if hi > 0 and hj < 1:
        allowed = min(hi, 1 - hj)
        direction = -1  # move from i to j
    elif hi < 1 and hj > 0:
        allowed = min(1 - hi, hj)
        direction = 1  # move from j to i (swap roles)
    else:
        return None
    # Choose epsilon = rng.random() * allowed * max_move
    epsilon = rng.random() * allowed * max_move
    new_h = h.copy()
    if direction == -1:
        new_h[i] -= epsilon
        new_h[j] += epsilon
    else:
        new_h[i] += epsilon
        new_h[j] -= epsilon

def anneal(h0, time_deadline, rng):
    h = h0.copy()
    cur = evaluate_h(h)
    best_h = h.copy()
    best = cur
    n = len(h)
    it = 0
    # Temperature schedule:
    T0 = cur * 0.5 if cur>0 else 1e-3
    # We'll compute progress = (time.time() - start)/total_time used for T.
    start = time.time()
    while True:
        if time.time() > time_deadline:
            break
        # propose move
        # optionally adjust step size based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / (time_deadline - start)
        # Use max_move decreasing from 1.0 to 0.01
        max_move = max(0.01, 1.0 - frac)
        move = propose_move(h, rng, max_move)
        if move is None:
            continue
        new_h, eps = move
        new_val = evaluate_h(new_h)
        d = new_val - cur
        # update temperature
        T = T0 * (1 - frac)
        if d <= 0 or rng.random() < math.exp(-d / max(T, 1e-12)):
            h = new_h
            cur = new_val
            if cur < best:
                best = cur
                best_h = h.copy()
        it += 1
    return best_h, best

def project_to_simplex_box(v, n):
    import cvxpy as cp
    desired_sum = n / 2.0
    h = cp.Variable(n)
    objective = cp.Minimize(cp.sum_squares(h - v))

import time, math
import numpy as np
import cvxpy as cp

def _evaluate_h(h: np.ndarray) -> float:
    """Return C5 value for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def _project_simplex_box(v: np.ndarray, n: int) -> np.ndarray:
    """Projection of v onto {0≤x≤1, sum(x)=n/2} via a tiny QP."""
    target = n / 2.0
    x = cp.Variable(n)
    prob = cp.Problem(cp.Minimize(cp.sum_squares(x - v)),
                      [x >= 0, x <= 1, cp.sum(x) == target])
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    except Exception:  # fallback if solver fails
        prob.solve(solver=cp.SCS, warm_start=True, verbose=False)
    if x.value is None:
        # Very rough fallback (should rarely happen)
        proj = np.clip(v, 0, 1)
        s = proj.sum()
        if s > 0:
            proj *= target / s
        proj = np.clip(proj, 0, 1)
        diff = proj.sum() - target
        if abs(diff) > 1e-9:
            idx = np.argmax(proj) if diff > 0 else np.argmin(proj)
            proj[idx] = np.clip(proj[idx] - diff, 0, 1)
        return proj
    return np.asarray(x.value).flatten()

def _propose_move(h: np.ndarray, rng: np.random.Generator,
                  max_frac: float):
    """Return a new h after moving mass from i→j, keeping sum constant."""
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n - 1)
    if j >= i:
        j += 1
    # decide direction so that source has mass, target has room
    if h[i] > 0 and h[j] < 1:
        src, dst = i, j
    elif h[j] > 0 and h[i] < 1:
        src, dst = j, i
    else:
        return None
    allowed = min(h[src], 1.0 - h[dst])
    if allowed <= 0:
        return None
    eps = rng.random() * allowed * max_frac
    if eps <= 0:
        return None
    new_h = h.copy()
    new_h[src] -= eps
    new_h[dst] += eps
    return new_h

def _anneal(start_h: np.ndarray, deadline: float,
            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simulated‑annealing style search starting from start_h."""
    n = start_h.size
    target_sum = n / 2.0
    h = start_h.copy()
    cur_val = _evaluate_h(h)
    best_h, best_val = h.copy(), cur_val

    # temperature schedule – starts at half the current objective
    T0 = cur_val * 0.5 if cur_val > 0 else 1e-3
    # step‑size schedule – from 0.5 down to 0.001
    max_start, max_end = 0.5, 0.001
    start_time = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        T = max(T0 * (1.0 - frac), 1e-12)
        max_frac = max_start * (1.0 - frac) + max_end * frac

        nxt = _propose_move(h, rng, max_frac)
        if nxt is None:
            continue
        new_h = nxt
        new_val = _evaluate_h(new_h)
        d = new_val - cur_val
        if d <= 0 or rng.random() < math.exp(-d / T):
            h, cur_val = new_h, new_val
            if cur_val < best_val:
                best_val, best_h = cur_val, h.copy()
        # occasional re‑projection to keep numerical sum exact
        if abs(h.sum() - target_sum) > 1e-9:
            h = _project_simplex_box(h, n)
            cur_val = _evaluate_h(h)
    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # tiny safety margin

    # ---- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: simple half‑filled binary pattern
        n0 = 100
        base = np.zeros(n0, dtype=float)
        base[: n0 // 2] = 1.0
        rng.shuffle(base)

    n0 = base.size
    # Target resolution – a modest multiple of the base size.
    target_n = max(188, n0)                    # 188 works well for n0≈47
    # Interpolate linearly to the finer grid
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, target_n, endpoint=False)
    interp = np.interp(x1, x0, base)
    h_current = _project_simplex_box(interp, target_n)

    # ---- optimisation loop -----------------------------------------
    best_h, best_val = h_current.copy(), _evaluate_h(h_current)

    # Main simulated‑annealing run – uses the whole remaining budget
    h_opt, val_opt = _anneal(best_h, deadline, rng)
    if val_opt < best_val:
        best_h, best_val = h_opt, val_opt

    # a final clean projection (guarantees exact feasibility)
    best_h = _project_simplex_box(best_h, target_n)
    best_val = _evaluate_h(best_h)

    return best_h, best_val, target_n
