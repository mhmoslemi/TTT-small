def _project_to_feasible(h):
    """Clip to [0,1] and adjust to satisfy sum(h)=len(h)/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target) < 1e-12:
        return h
    # water–filling style
    if cur < target:                # need to increase some entries
        need = target - cur
        while need > 1e-12:
            free = np.where(h < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            inc = need / free.size
            max_inc = np.min(1.0 - h[free])
            inc = min(inc, max_inc)
            h[free] += inc
            need = target - h.sum()
    else:                           # need to decrease some entries
        need = cur - target
        while need > 1e-12:
            free = np.where(h > 1e-12)[0]
            if free.size == 0:
                break
            dec = need / free.size
            max_dec = np.min(h[free])
            dec = min(dec, max_dec)
            h[free] -= dec
            need = h.sum() - target
    return np.clip(h, 0.0, 1.0)

def _compute_c5(h):
