def gradient_for_lag(h, lag):
    n = h.shape[0]
    g = np.ones_like(h)
    if lag >= 0:
        # shift forward

def project_onto_constraints(v, target_sum):
    # Binary search for lambda
    low = -1.0
    high = 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target_sum:
            high = mid
        else:
            low = mid
    # after loop, choose x with close sum
    x = np.clip(v + low, 0.0, 1.0)
    # Might need small adjustment to meet sum exactly (due to precision)
    # Compute diff and distribute among non-boundary entries.
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-9:
        # Distribute diff equally among entries not at bounds
        mask = (x > 0) & (x < 1)
        if np.any(mask):
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, 0.0, 1.0)
    # Finally ensure sum close; if still off, adjust a single entry.
    sum_err = target_sum - np.sum(x)
    if abs(sum_err) > 1e-9:
        # adjust first element not at bound
        idx = np.where(x < 1)[0][0] if sum_err > 0 else np.where(x > 0)[0][0]
        x[idx] += sum_err
        x = np.clip(x, 0.0, 1.0)
    return x
```

def project_onto_constraints(v, target):

    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)

    diff = target - x.sum()

        # distribute among not at bounds
        mask = (x > 0) & (x < 1)
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def gradient_for_lag(h, lag):
    n = h.shape[0]
    g = np.ones_like(h)
    if lag >= 0:
        if lag < n:
            g[:n - lag] -= h[lag:]
            g[lag:]   -= h[:n - lag]
    else:
        lag_abs = -lag
        if lag_abs < n:
            g[lag_abs:] -= h[:n - lag_abs]
            g[:n - lag_abs] -= h[lag_abs:]
    return g

import numpy as np
import math
import time
import random

# ---------------------------------------------------------
# Helper functions (must be top‑level, no closures or lambdas)
# ---------------------------------------------------------

def _project_onto_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector v onto the set  {x | 0 <= x_i <= 1, sum(x)=target_sum }.
    Uses a simple binary search for the Lagrange multiplier.
    """
    low, high = -1.0, 1.0                     # search interval for lambda
    for _ in range(50):                       # enough for double precision
        mid = (low + high) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            high = mid
        else:
            low = mid
    x = np.clip(v + low, 0.0, 1.0)

    # tiny rounding correction (keeps feasibility)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        cnt = int(free.sum())
        if cnt:
            x[free] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
        else:
            # all at bounds – force a minimal adjustment on the first element
            idx = 0 if diff > 0 else -1
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x


def _gradient_for_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """
    Sub‑gradient of  C5 = max_k sum_i h_i (1-h_{i+k})  at the lag currently
    achieving the maximum.  For a fixed lag we have
        dC5/dh_i = 1 - h_{i+lag} - h_{i-lag}
    (missing neighbors are treated as 0).
    """
    n = h.shape[0]
    g = np.ones_like(h)

    if lag >= 0:
        if lag < n:
            g[: n - lag] -= h[lag:]          # -h_{i+lag}
            g[lag:] -= h[: n - lag]          # -h_{i-lag}
    else:                                   # lag < 0
        lag = -lag
        if lag < n:
            g[lag:] -= h[: n - lag]          # -h_{i-lag}
            g[: n - lag] -= h[lag:]          # -h_{i+lag}
    return g


def _compute_c5(h: np.ndarray) -> float:
    """
    Evaluate the objective   C5 = max_k ∫ h(x)(1-h(x+k))dx
    on the uniform grid represented by the vector h.
    """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    """

    random.seed(seed)

    n = 300                     # number of discretisation points (even → integer weight)
    target_sum = n / 2.0        # because ∫h = 1  and  dx = 2/n

    best_h = h.copy()

    start_time = time.time()
    iteration = 0
