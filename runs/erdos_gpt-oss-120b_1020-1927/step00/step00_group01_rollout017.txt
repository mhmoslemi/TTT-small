def optimize_h_start(h0, max_iter, time_limit):
    import numpy as np, time
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = h.copy()
    best_val = dx * np.max(np.correlate(h, 1 - h, mode='full'))
    start = time.time()
    alpha = 0.1
    no_improve_counter = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        # compute gradient
        corr = np.correlate(h, 1 - h, mode='full')
        maxc = np.max(corr)
        idxs = np.where(corr == maxc)[0]
        grad = np.zeros_like(h)
        offset = n - 1
        for idx in idxs:
            s = idx - offset
            if s >= 0:
                grad[:n-s] += (1 - h[s:])
                grad[s:] += -h[:n-s]
            else:
                sneg = -s
                grad[sneg:] += (1 - h[:n-sneg])
                grad[:n-sneg] += -h[sneg:]
        grad = grad * dx   # derivative of C5
        # line search
        # try current alpha
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            step = -alpha * factor * grad
            h_candidate = project_box_sum(h + step, target_sum)
            val = dx * np.max(np.correlate(h_candidate, 1 - h_candidate, mode='full'))
            if val < best_val - 1e-12:
                # accept
                h = h_candidate
                best_val = val
                best_h = h_candidate.copy()
                alpha = alpha * (1.0 + 0.1)  # increase a bit
                improved = True
                break
        if not improved:
            alpha *= 0.5
            no_improve_counter += 1
            # occasionally random perturb to escape
            if no_improve_counter % 100 == 0:
                # random small permutation
                i, j = np.random.randint(n, size=2)
                delta = (np.random.rand() - 0.5) * 0.05

                    new[i] -= delta
                    new[j] += delta
                    new = project_box_sum(new, target_sum)
                    h = new
        # check convergence maybe
        if best_val < 0.3808: # early break if target achieved
            break
    return best_h, best_val

def simulated_annealing(initial_h, time_limit):
    n = len(initial_h)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = initial_h.copy()
    # ensure feasible
    # compute initial objective
    autocorr = np.correlate(h, h, mode='full')
    min_autocorr = np.min(autocorr)
    cur_val = (np.sum(h) - min_autocorr) * dx
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    # parameters
    max_move = 0.05  # max delta per move
    T = 0.02  # initial temperature (tuned)
    cooling = 0.9995
    iteration = 0
    while time.time() - start < time_limit:
        iteration += 1
        # pick two indices i, j randomly
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # possible delta range
        hi = h[i]
        hj = h[j]
        # can move from i to j: delta max limited by hi, 1-hj
        max_delta = min(hi, 1 - hj, max_move)
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand()
        # apply
        h[i] -= delta
        h[j] += delta
        # compute objective
        autocorr = np.correlate(h, h, mode='full')
        min_autocorr = np.min(autocorr)
        new_val = (np.sum(h) - min_autocorr) * dx
        # accept?
        accept = False
        if new_val < cur_val - 1e-12:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / (T + 1e-12))
            if np.random.rand() < prob:
                accept = True
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
        # cooldown
        T *= cooling
        # occasional reset of temperature maybe adapt
        if iteration % 1000 == 0 and T < 1e-6:
            T = 0.02  # reheat to escape local min
    return best_h, best_val
```

import numpy as np
import time

# ------------------------------------------------------------
# Helper functions (all top‑level, no closures or lambdas)
# ------------------------------------------------------------

def _dx_from_n(n: int) -> float:
    """grid spacing for a given number of points."""
    return 2.0 / n


def _objective(h: np.ndarray) -> float:
    """C5 value for a step function described by the vector h."""
    n = h.shape[0]
    dx = _dx_from_n(n)
    # sum(h) is constant (= n/2) for feasible vectors,
    # C5 = max_k ∫ h(x)(1-h(x+k))dx = sum(h) - min_k autocorr(k)
    autocorr = np.correlate(h, h, mode='full')
    min_corr = np.min(autocorr)
    return (np.sum(h) - min_corr) * dx


def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project v onto the set {h : 0 ≤ h_i ≤ 1 , Σ h_i = target }.
    This is the Euclidean projection onto a capped simplex.
    """
    # start with clipping to [0,1]
    lo = -np.max(v) - 2.0   # makes v - lo ≥ 1  → all ones
    hi = np.max(v) + 2.0    # makes v - hi ≤ -1 → all zeros
    for _ in range(60):
        mid = (lo + hi) * 0.5
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _initial_vector(n: int, parent: np.ndarray | None) -> np.ndarray:
    """
    Produce an initial feasible vector of length n.
    If ``parent`` can be tiled evenly, repeat it;
    otherwise use the symmetric triangular shape.
    """
    target = n / 2.0
    if parent is not None and n % parent.size == 0:
        factor = n // parent.size
        h = np.repeat(parent, factor).astype(float)
    else:
        # triangular shape: h(x)=max(0,1-|x-1|)
        dx = _dx_from_n(n)
        x = (np.arange(n) + 0.5) * dx
        h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    # projection guarantees exact feasibility
    return _project_box_sum(h, target)


# ------------------------------------------------------------
# Simulated annealing core (keeps sum, respects bounds)
# ------------------------------------------------------------

def _simulated_annealing(start_h: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves a small amount of mass between two coordinates.
    Returns the best vector seen and its C5 value.
    """
    n = start_h.shape[0]
    target = n / 2.0
    dx = _dx_from_n(n)

    h = start_h.copy()
    cur_val = _objective(h)
    best_h = h.copy()
    best_val = cur_val

    # SA parameters
    max_move = 0.05          # maximal mass transferred in one step
    temperature = 0.02
    cooling = 0.9995
    it = 0
    start = time.time()

    while time.time() - start < time_limit:
        it += 1

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        hi, hj = h[i], h[j]
        # how much can we move from i to j while staying in [0,1]?
        move_cap = min(hi, 1.0 - hj, max_move)
        if move_cap <= 0.0:
            continue
        delta = move_cap * np.random.rand()

        # propose move
        h[i] = hi - delta
        h[j] = hj + delta

        new_val = _objective(h)

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / (temperature + 1e-16))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj

        # temperature schedule
        temperature *= cooling
        if temperature < 1e-8:
            temperature = 0.02   # reheat occasionally

    return best_h, best_val


# ------------------------------------------------------------
# A quick deterministic local refinement using the sub‑gradient
# ------------------------------------------------------------

def _local_refine(h0: np.ndarray, max_iter: int, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Gradient‑like descent on the sub‑gradient of C5.
    The step is always projected back onto the feasible set.
    """
    n = h0.shape[0]
    target = n / 2.0
    dx = _dx_from_n(n)

    h = h0.copy()
    best_h = h.copy()
    best_val = _objective(h)

    start = time.time()
    step_size = 0.08

    offset = n - 1
    for _ in range(max_iter):
        if time.time() - start >= time_limit:
            break

        # correlation h * (1‑h) needed only to locate argmax
        corr = np.correlate(h, 1.0 - h, mode='full')
        maxc = np.max(corr)
        arg = np.where(corr == maxc)[0]

        # sub‑gradient for the max‑lag(s)
        grad = np.zeros_like(h)
        for idx in arg:
            s = idx - offset
            if s >= 0:
                grad[:n - s] += (1.0 - h[s:])
                grad[s:] += -h[:n - s]
            else:
                sneg = -s
                grad[sneg:] += (1.0 - h[:n - sneg])
                grad[:n - sneg] += -h[sneg:]

        grad *= dx          # derivative of C5
        # try a few step scalings
        improved = False
        for fac in (1.0, 0.5, 0.25):
            cand = _project_box_sum(h - step_size * fac * grad, target)
            val = _objective(cand)
            if val < best_val - 1e-12:
                h = cand
                best_val = val
                best_h = cand.copy()
                step_size *= 1.07
                improved = True
                break
        if not improved:
            step_size *= 0.55

    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points).
    The routine spends the allotted time searching for a vector h
    with a low Erdős C5 value.
    """
    np.random.seed(seed)
    start_total = time.time()

    # parent construction, if present
    parent = kwargs.get('initial_h_values', None)
    if parent is None:
        try:
            parent = initial_h_values   # global injected by the harness
        except NameError:
            parent = None

    # candidate grid sizes – all ≤ 500 points (fast enough)
    n_candidates = [47, 94, 188]   # multiples of the provided 47‑point seed
    # also try a higher‑resolution even size that has no parent dependence
    n_candidates.append(200)

    best_h = None
    best_val = np.inf
    best_n = None

    # allocate time roughly equally among the grid sizes we try
    for n in n_candidates:
        elapsed = time.time() - start_total
        time_left = budget_s - elapsed
        if time_left < 4.0:
            break

        # 30 % of the remaining budget for this n (the rest is kept for later)
        time_for_n = min(time_left * 0.3, 300.0)

        # initial vector for this resolution
        h0 = _initial_vector(n, parent)

        # 1) a short simulated‑annealing phase
        h_sa, val_sa = _simulated_annealing(h0, time_for_n * 0.5)

        # 2) a deterministic refinement (cheaper, uses remaining half‑time)
        time_refine = max(0.0, time_for_n * 0.5 - (time.time() - start_total))
        h_ref, val_ref = _local_refine(h_sa, max_iter=800, time_limit=time_refine)

        # keep the best of the two
        if val_ref < best_val:
            best_val = val_ref
            best_h = h_ref.copy()
            best_n = n

        if best_val <= 0.3808:
            break

    # Fallback: if nothing succeeded, use the parent directly (or uniform 0.5)
    if best_h is None:
        if parent is not None:
            best_h = np.array(parent, copy=True)
        else:
            best_h = np.full(47, 0.5)

    # recompute the exact bound to return
    final_c5 = _objective(best_h)
