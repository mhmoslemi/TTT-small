def solve_milp(n, half, budget):
    import cvxpy as cp, numpy as np, time, math

    h = cp.Variable(n, boolean=True)
    L = cp.Variable(integer=True)
    constraints = [cp.sum(h) == half, L <= half, L >= 0]

    for k in range(n):
        length = n - k
        w = cp.Variable(length, boolean=True)
        # linking constraints:
        constraints += [w <= h[:length]]
        constraints += [w <= h[k:]]
        constraints += [w >= h[:length] + h[k:] - 1]
        constraints += [cp.sum(w) >= L]

    prob = cp.Problem(cp.Maximize(L), constraints)

        prob.solve(solver=cp.GLPK_MI, verbose=False, glpk={'msg_lev': 'GLP_MSG_OFF'}) # maybe cannot pass dict.

        return None, None
    h_val = np.round(np.array(h.value)).astype(float) # round to nearest 0 or 1
    L_val = int(round(L.value))
