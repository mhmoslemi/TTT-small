import numpy as np, time, random

def evaluate(h, dx):
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def projection(y, target_sum):
    # lambda for clipping
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = (lo+hi)/2
        x = np.clip(y - lam, 0, 1)
        s = x.sum()
        if s < target_sum:
            hi = lam
        else:
            lo = lam
    x = np.clip(y - (lo+hi)/2, 0, 1)
    # fix rounding
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff equally to those not at boundary
        mask = (x > 0 + 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0, 1)
        else:
            # adjust any index within bounds
            idx = np.argmin(np.abs(diff) + 0) # just adjust one variable
            x[idx] += diff
            x = np.clip(x, 0, 1)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    n = kwargs.get('n_points', 200) # but not reading, we set.
    dx = 2.0 / n
    target_sum = n / 2

    # Initialize h: use initial_h_values if provided else random.
    # global variable initial_h_values may be defined, we check with try/except.
    try:
        init = initial_h_values
        # upsample via interpolation
        x_old = np.linspace(0, 2, len(init), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x_new, x_old, init)
        # Enforce bounds [0,1] (should hold)
        h = np.clip(h, 0, 1)
        # Project to constraints
        h = projection(h, target_sum)
    except NameError:
        # random init with uniform [0,1]
        h = np.random.rand(n)
        h = projection(h, target_sum)

    best_h = h.copy()
    best_val = evaluate(best_h, dx)

    # SA parameters
    T0 = 0.1  # initial temperature
    Tf = 1e-4
    # schedule based on fraction of time
    # we will compute fraction = (t - start) / budget.
    # At each step compute T = T0 * (Tf/T0) ** fraction

    # iteration loop with step count maybe large
    it = 0
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget_s:
            break
        # exponential cooling

def project_onto_box_simplex(y, target_sum, lower=0.0, upper=1.0):
    # binary search lambda
    lo = -1e6
    hi = 1e6
    for _ in range(60):
        lam = (lo + hi) / 2
        x = np.clip(y - lam, lower, upper)
        s = x.sum()
        if s < target_sum:
            hi = lam
        else:
            lo = lam
    x = np.clip(y - (lo + hi) / 2, lower, upper)
    # small adjustment if needed
    diff = target_sum - x.sum()
    if abs(diff) > 1e-9:
        # distribute to non-bound variables
        mask = (x > lower + 1e-12) & (x < upper - 1e-12)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, lower, upper)
        else:
            # fallback: adjust first element within bounds
            idx = 0
            # find any index not at boundary
            for idx in range(len(x)):
                if lower <= x[idx] + diff <= upper:
                    x[idx] += diff
                    break
    return x

import numpy as np
import time
import math

def project_onto_box_simplex(y, target_sum):
    """Euclidean projection of y onto the box [0,1]^n with a fixed L1 sum."""
    lower, upper = 0.0, 1.0
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, lower, upper)
        s = x.sum()
        if s < target_sum:
            hi = lam
        else:
            lo = lam
    x = np.clip(y - (lo + hi) / 2.0, lower, upper)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > lower + 1e-12) & (x < upper - 1e-12)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, lower, upper)
        else:
            for idx in range(len(x)):
                if lower <= x[idx] + diff <= upper:
                    x[idx] += diff
                    break
    return x

def evaluate_c5(h):
    """Return max_{k} ∫ h(x)(1−h(x+k))dx for a discretised step function."""
    # full correlation of h with (1‑h) then scale by dx = 2/n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)
