def simulated_annealing(n, max_iters, time_limit):
    h = random balanced binary vector: generate array of n zeros, set first n//2 to 1, shuffle.

    compute overlap = np.correlate(h, h, mode='full')
    min_overlap = overlap.min()
    best_h = h.copy()
    best_min = min_overlap

    start = time.time()
    t_end = start + time_limit

    T = 1.0
    iter = 0
    while time.time() < t_end and iter < max_iters:
        iter += 1
        # random swap
        i = random.choice(np.where(h==1)[0])
        j = random.choice(np.where(h==0)[0])
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        # compute new min overlap
        corr_new = np.correlate(h_new, h_new, mode='full')
        min_new = corr_new.min()
        delta = min_new - min_overlap
        accept = False
        if delta > 0:
            accept = True
        else:
            # temperature schedule
            prob = np.exp(delta / T) if T > 1e-8 else 0.0
            if random.random() < prob:
                accept = True
        if accept:
            h = h_new
            min_overlap = min_new
            if min_overlap > best_min:
                best_min = min_overlap
                best_h = h.copy()
        # cooling
        # maybe linear schedule based on iteration number.
        # keep T decreasing.
        T = max(0.001, T * 0.9999)  # small cooling

    return best_h, best_min

        return np.max(corr) * (2.0 / len(arr))

        corr_full = np.correlate(h, 1 - h, mode='full')
        k_index = np.argmax(corr_full)

def project_onto_box_and_sum(v, target_sum):

    # target_sum: scalar
    # use binary search on lambda to satisfy sum_i clip(v_i - lambda, 0, 1) = target_sum

            lo = lam

import time
import random
import numpy as np

# -------------------------------------------------------------
# Helper: projection onto the simplex with box [0,1] and a given sum
def _project_box_sum(v, target):
    """Euclidean projection of v onto {0<=x<=1, sum(x)=target}."""
    lo, hi = -1.0, 1.0  # bounds guaranteed to bracket the solution
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


# -------------------------------------------------------------
# Compute the Erdős‑overlap value C₅ for a discrete step function
def _c5_value(arr):
    n = arr.size
    # correlation of h with (1-h) using linear (zero‑padded) mode
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return np.max(corr) * (2.0 / n)   # dx = 2/n


# -------------------------------------------------------------
# Gradient of the worst‑shift term
def _worst_shift_gradient(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (n - 1)                 # integer shift producing the max
    grad = np.ones_like(h)

    if shift >= 0:
        # term h_i·h_{i+shift}
        grad[: n - shift] -= h[shift:]
        # term h_{i-shift}·h_i
        grad[shift:] -= h[: n - shift]
    else:
        s = -shift
        grad[s:] -= h[: n - s]
        grad[: n - s] -= h[s:]

    return grad, shift


# -------------------------------------------------------------
# Small hill‑climb on binary vectors (exactly half ones)
def _binary_hill_climb(start, deadline):
    n = start.size
    target = n // 2
    best = start.copy()
    best_val = _c5_value(best)

    # pre‑compute index arrays once; they will be refreshed after every
    # accepted move (the vectors are tiny, overhead negligible)
    while time.time() < deadline:
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = random.choice(ones)
        j = random.choice(zeros)

        cand = best.copy()
        cand[i] = 0
        cand[j] = 1
        val = _c5_value(cand)
        if val < best_val:
            best = cand
            best_val = val
    return best, best_val


# -------------------------------------------------------------
def run(seed=42, budget_s=1100, **_):
    random.seed(seed)
    np.random.seed(seed)

    # ---------------------------------------------------------
    # initialise from the supplied construction if available
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        target_sum = n / 2.0
        h = _project_box_sum(np.clip(h, 0.0, 1.0), target_sum)
    else:
        # fall back to a modest size if no parent is given
        n = 200
        target_sum = n / 2.0
        h = np.random.rand(n)
        h = _project_box_sum(h, target_sum)

    best_h = h.copy()
    best_val = _c5_value(best_h)

    start_time = time.time()
    end_time = start_time + budget_s * 0.90   # reserve a bit for the final binary phase

    # ---------------------------------------------------------
    # continuous sub‑gradient descent / simulated annealing
    it = 0
    while time.time() < end_time:
        it += 1
        grad, _ = _worst_shift_gradient(h)

        # simple decay of step size
        step = 0.12 / (1.0 + it * 2e-4)
        h_candidate = h - step * grad
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        h_candidate = _project_box_sum(h_candidate, n / 2.0)

        val = _c5_value(h_candidate)

        # accept improvements immediately
        if val < best_val:
            best_val = val
            best_h = h_candidate.copy()
            h = h_candidate
            continue

        # occasional uphill moves (simulated annealing)
        T = max(0.001, 0.08 * (1.0 - it / 200000.0))
        if random.random() < np.exp((best_val - val) / T):
            h = h_candidate

        # occasional random jitter to escape plateaus
        if it % 4000 == 0 and random.random() < 0.15:
            jitter = np.random.normal(scale=0.04, size=n)
            h2 = np.clip(h + jitter, 0.0, 1.0)
            h2 = _project_box_sum(h2, n / 2.0)
            val2 = _c5_value(h2)
            if val2 < best_val:
                best_val = val2
                best_h = h2.copy()
                h = h2

    # ---------------------------------------------------------

    order = np.argsort(-best_h)               # descending
    binary = np.zeros_like(best_h)
