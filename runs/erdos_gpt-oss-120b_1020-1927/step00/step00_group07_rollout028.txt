import time, math, random
import numpy as np

# ---------- helper utilities ----------

def _project_onto_box_and_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Projection of v onto the intersection {0<=x<=1, sum(x)=target_sum}
    using a simple bisection on the Lagrange multiplier."""
    lo, hi = -2.0, 2.0  # wide enough bounds for the multiplier
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _gradient_of_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of f_k(h)=∑ h_i(1‑h_{i+shift}) for a given shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        # i runs 0 .. n‑shift‑1
        g[: n - shift] += 1.0 - h[shift:]
        g[shift:] -= h[: n - shift]
    else:
        s = -shift
        g[s:] += 1.0 - h[: n - s]
        g[: n - s] -= h[s:]
    return g


def _max_corr_counts(h: np.ndarray) -> float:
    """Maximum of the un‑scaled correlation counts = max_k ∑ h_i(1‑h_{i+k})."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max())


def _smooth_subgradient(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient for the max‑over‑shifts objective (average over all worst shifts)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    worst = np.flatnonzero(corr == max_val)
    g = np.zeros_like(h)
    for idx in worst:
        k = idx - (n - 1)
        g += _gradient_of_shift(h, k)
    return g / len(worst)


# ---------- local optimization routine ----------

def _optimize_from_start(
    h0: np.ndarray,
    target_sum: float,
    start_time: float,
    budget: float,
    max_iter: int = 20000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with a tiny simulated‑annealing twist."""
    n = h0.shape[0]
    h = h0.copy()
    best_h = h.copy()
    best_cnt = _max_corr_counts(h)

    for it in range(1, max_iter + 1):
        # time check
        if time.time() - start_time > budget * 0.95:
            break

        # evaluate current max‑correlation
        cur_cnt = _max_corr_counts(h)
        if cur_cnt < best_cnt:
            best_cnt = cur_cnt
            best_h = h.copy()

        # compute (averaged) sub‑gradient of the worst shift(s)
        g = _smooth_subgradient(h)

        # diminishing step size
        step = 0.6 / math.sqrt(it)
        # make a tentative move
        h_new = h - step * g
        h_new = _project_onto_box_and_sum(h_new, target_sum)

        # decide acceptance
        new_cnt = _max_corr_counts(h_new)
        if new_cnt <= cur_cnt:
            h = h_new
        else:
            # simple simulated‑annealing acceptance
            T = 0.08 / math.sqrt(it)
            if math.exp(-(new_cnt - cur_cnt) / (T + 1e-12)) > random.random():
                h = h_new

        # occasional small random jitter to escape flat regions
        if it % 500 == 0:
            jitter = (np.random.rand(n) - 0.5) * 0.02  # ±0.01
            h_jit = np.clip(h + jitter, 0.0, 1.0)
            h_jit = _project_onto_box_and_sum(h_jit, target_sum)
            if _max_corr_counts(h_jit) < cur_cnt:
                h = h_jit

    return best_h, best_cnt
