def compute_c5_bin(h):
    # h binary, dtype int or float
    n = len(h)
    S = n // 2
    conv = np.correlate(h, h, mode='full')
    min_conv = conv.min()
    dx = 2.0 / n
    c5 = (S - min_conv) * dx
    return c5, min_conv

def step_subgradient(h, step):
    n = len(h)
    dx = 2.0 / n
    # compute full correlation of h with 1-h (not scaled)
    # compute raw correlation without dx scaling to find shift
    corr = np.correlate(h, 1 - h, mode='full')
    # find shift index of max
    idx = np.argmax(corr)
    s = idx - (n - 1)   # shift s can be negative
    # compute shifted versions
    # shift_plus: h[i + s] (zero if out of range)
    # shift_minus: h[i - s]
    shift_plus = np.zeros_like(h)
    shift_minus = np.zeros_like(h)
    if s >= 0:
        shift_plus[:n - s] = h[s:]
        shift_minus[s:] = h[:n - s]
    else:
        ss = -s
        shift_plus[ss:] = h[:n - ss]
        shift_minus[:n - ss] = h[ss:]
    # gradient of g (not scaled) is -(shift_plus + shift_minus)
    grad = -(shift_plus + shift_minus)
    # step descent: update h
    h_new = h - step * grad
    # project onto [0,1] and sum constraint
    # we can clip then adjust sum using projection function
    h_new = np.clip(h_new, 0.0, 1.0)
    # project onto sum = n/2 using lambda method
    target_sum = n / 2.0
    h_new = project_box_sum(h_new, target_sum)
    return h_new

import time
import numpy as np

def _project_onto_box_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {0≤x_i≤1 , sum x_i = target}."""
    lo, hi = -1.0, 1.0          # guarantees sum(lo) > target, sum(hi) < target
    for _ in range(50):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return w

def _step_subgradient(h: np.ndarray, step: float) -> np.ndarray:
    """Perform one sub‑gradient step for the max‑overlap objective."""
    n = h.size
    # correlation of h with (1‑h) to locate the worst shift
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    s = idx - (n - 1)                     # signed shift achieving the max

    # shifted copies (zero padding outside the interval)
    plus = np.zeros_like(h)
    minus = np.zeros_like(h)
    if s >= 0:
        plus[: n - s] = h[s:]
        minus[s:] = h[: n - s]
    else:
        ss = -s
        plus[ss:] = h[: n - ss]
        minus[: n - ss] = h[ss:]

    # gradient of the active term g_s = Σ h_i (1‑h_{i+s})
    grad = 1.0 - (plus + minus)

    # descent step
    h_new = h - step * grad

    # enforce bounds and the integral constraint
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _project_onto_box_and_sum(h_new, n / 2.0)
    return h_new

    # initialise – use the supplied parent if available, otherwise a uniform start
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=np.float64)
        n = h.size
    else:
        n = 90                     # a modest size that works comfortably within the budget
        h = np.full(n, 0.5, dtype=np.float64)

    # ensure feasibility
    h = np.clip(h, 0.0, 1.0)
    h = _project_onto_box_and_sum(h, n / 2.0)

    best_h = h.copy()
    best_score = evaluate_erdos_solution(best_h)   # external helper provided by the harness

    start_time = time.time()
    time_limit = budget_s * 0.95                     # a small safety margin

    # ------------------------------------------------------------------
    # main optimisation loop – many short runs with sub‑gradient descent
    # ------------------------------------------------------------------
    while time.time() - start_time < time_limit:
        # random restart (drawn uniformly and projected onto the feasible set)
        h = _project_onto_box_and_sum(rng.random(n), n / 2.0)

        step = 0.5
        # number of iterations for this restart adapts to remaining time
        remaining = time_limit - (time.time() - start_time)
        iters = int(min(8000, max(1500, remaining * 2000)))   # empirical heuristic

        for it in range(iters):
            # one sub‑gradient step
            h = _step_subgradient(h, step)

            # occasional evaluation
            if (it & 63) == 0:               # every 64th iteration
                score = evaluate_erdos_solution(h)
                if score < best_score:
                    best_score = score
                    best_h = h.copy()
                    # stop early if we beat the target
                    if best_score < 0.3808:
                        return (best_h.tolist(), float(best_score), n)

            # simple step‑size decay
            if (it + 1) % 800 == 0:
                step *= 0.85

        # final evaluation of this restart
        score = evaluate_erdos_solution(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
            if best_score < 0.3808:
                break

    # final projection (just in case)
    best_h = _project_onto_box_and_sum(np.asarray(best_h, dtype=np.float64), n / 2.0)
    final_score = evaluate_erdos_solution(best_h)

    return (best_h.tolist(), float(final_score), n)
