def project_box_sum(v, target, low=0., high=1., tol=1e-10, max_iter=100):
    lo = np.min(v) - high
    hi = np.max(v) - low
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, low, high)
        s = x.sum()
        if np.abs(s - target) < tol:
            return x
        if s > target:
            # need increase λ to shrink sum (since clip(v - λ) smaller)
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, low, high)

import time
import math

# -------------------------------------------------------------
# Helper: projection onto the box [0,1]^n with a fixed sum.
# -------------------------------------------------------------
def _project_box_sum(v: np.ndarray, target: float, low: float = 0.0, high: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of v onto {x | low <= x_i <= high, sum(x)=target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = np.min(v) - high
    hi = np.max(v) - low
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, low, high)
        s = x.sum()
        if abs(s - target) <= 1e-12:
            return x
        if s > target:
            lo = mid
        else:
            hi = mid
    # final cut
    return np.clip(v - (lo + hi) * 0.5, low, high)


# -------------------------------------------------------------
# Gradient of the smooth approximation of max_k S_k,
# where S_k = sum_i h_i (1 - h_{i+k}) (zero padding).
# -------------------------------------------------------------
def _smooth_max_gradient(h: np.ndarray, beta: float) -> np.ndarray:
    n = h.size
    conv = np.correlate(h, 1.0 - h, mode="full")          # length 2n-1
    w = np.exp(beta * conv)
    w /= w.sum()                                         # softmax weights
    grad = np.zeros_like(h)

    offset = n - 1                                        # index shift: k = idx - offset

    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = idx - offset                                  # actual shift (may be negative)

        if k >= 0:
            # (1 - h_{j+k}) term
            grad[: n - k] += wk * (1.0 - h[k:])
            # -h_{j-k} term
            grad[k:]       -= wk * h[: n - k]
        else:
            kk = -k
            # (1 - h_{j-kk}) term
            grad[kk:]      += wk * (1.0 - h[: n - kk])
            # -h_{j+kk} term
            grad[: n - kk] -= wk * h[kk:]

    return grad

# -------------------------------------------------------------
def _c5_value(h: np.ndarray) -> float:
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / h.size)
