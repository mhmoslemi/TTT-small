def compute_c5(x):
    n = len(x)
    nfft = 1 << ((2*n-1).bit_length())  # next power of two
    ft_a = np.fft.rfft(x, nfft)
    ft_b_rev = np.fft.rfft((1 - x)[::-1], nfft)
    conv = np.fft.irfft(ft_a * ft_b_rev, nfft)
    max_val = conv[:2*n-1].max()
    return max_val * (2.0 / n)

def compute_c5(x):
    n = len(x)
    N = 2 * n
    # FFT of x
    fx = np.fft.rfft(x, N)
    # autocorrelation via inverse FFT of |fx|^2
    autocorr = np.fft.irfft(np.abs(fx)**2, N)
    # autocorr length N, contains circular autocorr. Need linear portion.
    # Linear autocorr values are autocorr[:2*n-1] (since we zero padded to length 2n).
    linear_autocorr = autocorr[:2*n-1]
    # Note that each lag k corresponds with shift (k - (n-1)). But min across all = min(linear_autocorr)
    min_c = linear_autocorr.min()
    # compute C5
    dx = 2.0 / n
    c5 = 1.0 - min_c * dx
    return c5

def compute_c5(x: np.ndarray) -> float:
    """Maximum overlap integral for a step function represented by vector x."""
    n = x.shape[0]
    N = 2 * n
    fx = np.fft.rfft(x, N)
    f1mh = np.fft.rfft(1.0 - x, N)
    corr = np.fft.irfft(fx * np.conj(f1mh), N)
    # linear correlation occupies first (2*n-1) entries
    max_val = np.max(corr[: 2 * n - 1])

def project_to_sum(v: np.ndarray, target: float, tol: float = 1e-9) -> np.ndarray:
    """Clip to [0,1] and adjust so that sum(v) == target (within tolerance)."""
    v = np.clip(v, 0.0, 1.0)
    diff = np.sum(v) - target
    if abs(diff) <= tol:
        return v
    # indices that are not on a hard bound
    free = np.where((v > 1e-12) & (v < 1.0 - 1e-12))[0]
    if free.size == 0:                     # nothing free – shift uniformly
        v -= diff / v.size
        v = np.clip(v, 0.0, 1.0)
        return project_to_sum(v, target, tol)
    adjust = diff / free.size
    v[free] -= adjust
    v = np.clip(v, 0.0, 1.0)
    return project_to_sum(v, target, tol)

    r = np.random.rand(n)
    v = r / r.sum() * target
    v = np.clip(v, 0.0, 1.0)

        base = np.asarray(initial_h_values, dtype=float)
        base_n = base.size
        base = project_to_sum(base, base_n / 2.0)
    else:
        base_n = 45
        base = np.full(base_n, 0.5)

    # Candidate grid sizes (keep them modest for speed)
    sizes = [base_n]
    if base_n * 2 <= 180:
        sizes.append(base_n * 2)
    if base_n * 4 <= 180:
        sizes.append(base_n * 4)

    best_vec = None
    best_score = float('inf')

    for n in sizes:
        target_sum = n / 2.0

        # Build a starting vector for this resolution
        if base_n == n:
            start_vec = base.copy()
        elif n % base_n == 0:
            rep = n // base_n
            start_vec = np.repeat(base, rep)
        else:
            # linear interpolation onto a uniform grid
            src_x = np.linspace(0.0, 2.0, base_n, endpoint=False)
            dst_x = np.linspace(0.0, 2.0, n, endpoint=False)
            start_vec = np.interp(dst_x, src_x, base)

        start_vec = project_to_sum(start_vec, target_sum)

        # Several optimisation restarts with diverse initialisations
        for restart in range(5):
            if time.time() - start > budget_s * 0.95:
                break

            if restart == 0:
                init = start_vec.copy()
            elif restart == 1:
                # binary start: exactly half ones
                init = np.zeros(n)
                idx = np.random.choice(n, n // 2, replace=False)
                init[idx] = 1.0
            else:
                init = random_initial(n, target_sum)

            opt = optimize_one(init, maxiter=250, ftol=1e-9)
            if opt is None:
                continue
            score = compute_c5(opt)
            if score < best_score:
                best_score = score
                best_vec = opt.copy()

        # If we already have a good vector, run a cheap simulated‑annealing
        if best_vec is not None and best_vec.shape[0] == n:
            cur = best_vec.copy()
            cur_score = best_score
            T = 0.2
            Tmin = 5e-5
            it = 0
            max_it = 60000
            while it < max_it and time.time() - start < budget_s * 0.95:
                i, j = np.random.choice(n, 2, replace=False)
                # feasible delta respecting bounds
                max_inc = min(1.0 - cur[i], cur[j])
                max_dec = min(cur[i], 1.0 - cur[j])
                if np.random.rand() < 0.5 and max_inc > 0:
                    delta = np.random.rand() * max_inc
                elif max_dec > 0:
                    delta = -np.random.rand() * max_dec
                else:
                    continue
                cur[i] += delta
                cur[j] -= delta
                new_score = compute_c5(cur)
                d = new_score - cur_score
                if d <= 0 or np.random.rand() < np.exp(-d / T):
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_vec = cur.copy()
                else:
                    cur[i] -= delta
                    cur[j] += delta
                T = max(Tmin, T * 0.9995)
                it += 1

    # Fallback: uniform half‑filled vector if optimisation failed
    if best_vec is None:
        n = base_n
        best_vec = np.full(n, 0.5)
        best_vec = project_to_sum(best_vec, n / 2.0)
        best_score = compute_c5(best_vec)

    # Final evaluation with the provided helper
    try:
        final_score = evaluate_erdos_solution(best_vec)  # pre‑imported
    except Exception:
        final_score = best_score

    return best_vec, final_score, int(best_vec.shape[0])
