def hill_climb(h, max_steps, start_time, budget_s):
    n = len(h)
    sum_h = n/2
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for step in range(max_steps):
        if time.time() - start_time > budget_s * 0.95:
            break
        # compute conv
        conv = np.convolve(cur_h, cur_h, mode='full')
        min_idx = np.argmin(conv)
        shift = min_idx - (n - 1)
        # compute partner values
        if shift >= 0:
            p = np.concatenate((cur_h[shift:], np.zeros(shift)))  # p[i] = h[i+shift] if i+shift < n else 0
        else:
            shift_abs = -shift
            p = np.concatenate((np.zeros(shift_abs), cur_h[:n-shift_abs]))
        # choose i where we can increase
        # compute scores for increase
        increase_possible = (cur_h < 1 - 1e-12)
        if not np.any(increase_possible):
            break
        # Score = p * (1 - cur_h) (increase per unit)
        inc_scores = p * (1 - cur_h)
        i = np.argmax(inc_scores)  # index to increase
        # choose j to decrease
        decrease_possible = (cur_h > 1e-12)
        if not np.any(decrease_possible):
            break
        # Score for decrease: want low p to minimize loss
        dec_scores = p * cur_h  # loss per unit delta
        # Choose j where dec_score minimal
        j = np.argmin(dec_scores)
        # But ensure i != j
        if i == j:
            # try different j maybe second smallest
            sorted_j = np.argsort(dec_scores)
            for candidate in sorted_j:
                if candidate != i:
                    j = candidate
                    break
        # compute max delta
        max_delta = min(1 - cur_h[i], cur_h[j])
        if max_delta <= 1e-12:
            # can't move target; maybe try another pair
            continue
        # Choose delta = min(step_factor * max_delta, max_delta)
        delta = max_delta * 0.5  # half of allowed
        # Update
        cur_h[i] += delta
        cur_h[j] -= delta
        # Compute new C5
        new_c5 = compute_c5(cur_h)
        # Accept if improvement (reduced C5)
        if new_c5 < cur_c5 - 1e-12:
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # Undo move
            cur_h[i] -= delta
            cur_h[j] += delta
            # maybe perform random small move else.
            # Could also accept with some probability
            # We skip.
            continue
    return best_h, best_c5

def hill_climb(initial_h, time_limit, rng):
    h = initial_h.copy()
    n = len(h)
    target_sum = n/2
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    start = time.time()
    no_improve = 0
    while time.time() - start < time_limit:
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # direction: move from i to j
        if rng.random() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        max_delta = min(h[donor], 1.0 - h[receiver])
        if max_delta < 1e-9:
            continue
        # delta size schedule
        elapsed = time.time() - start
        # Use fraction of max_delta scaled with remaining time fraction
        # and some base factor
        frac = max(0.001, (time_limit - elapsed) / time_limit)
        delta = max_delta * frac * rng.random()
        if delta < 1e-12:
            continue
        # Apply move
        h_new = h.copy()
        h_new[donor] -= delta
        h_new[receiver] += delta
        # Ensure bounds: already ensured.
        new_c5 = compute_c5(h_new)
        # Accept?
        # Use simulated annealing temperature
        T = max(1e-6, best_c5 * 0.05 * (1 - elapsed / time_limit))
        if new_c5 < cur_c5 - 1e-12 or rng.random() < math.exp(-(new_c5 - cur_c5) / T):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        # occasional perturbation if stagnation
        if no_improve > 5000:
            # restart from best with perturbation
            h = best_h + rng.normal(scale=0.05, size=n)

def _project_to_simplex_box(v: np.ndarray, target: float) -> np.ndarray:
    """Projection of v onto {x | 0<=x_i<=1, sum x_i = target}."""
    # Clamp initially to box
    x = np.clip(v, 0.0, 1.0)
    # Find lambda by bisection so that sum_i clip(x_i - lam,0,1) = target
    lo, hi = -1.0, 1.0
    # Expand bounds if necessary
    while np.sum(np.clip(x - lo, 0.0, 1.0)) > target:
        lo *= 2.0
    while np.sum(np.clip(x - hi, 0.0, 1.0)) < target:
        hi *= 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    proj = np.clip(x - lam, 0.0, 1.0)
    # Tiny correction of rounding error
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = np.where((proj > 0.0) & (proj < 1.0))[0]
        if free.size:
            proj[free] += diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
        else:
            proj[0] = np.clip(proj[0] + diff, 0.0, 1.0)
    return proj

    Search for a step function h on [0,2] that minimizes
    C5 = max_k ∫ h(x)(1−h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = budget_s * 0.94  # a safety margin

    # initialise
    h = _initial_feasible()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # statistics for acceptance schedule
    cur_h = h.copy()
    cur_c5 = best_c5
    no_improve = 0

    while time.time() - start_time < time_limit:
        # propose a mass transfer between two distinct positions
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # random direction of transfer
        if rng.random() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        max_delta = min(cur_h[donor], 1.0 - cur_h[receiver])
        if max_delta <= 1e-12:
            continue

        # step size shrinks as time runs out
        elapsed = time.time() - start_time
        remaining_frac = max(0.0, (time_limit - elapsed) / time_limit)
        delta = max_delta * remaining_frac * rng.random()
        if delta <= 1e-12:
            continue

        # apply move
        new_h = cur_h.copy()
        new_h[donor] -= delta
        new_h[receiver] += delta
        new_c5 = _c5_value(new_h)

        # simulated‑annealing acceptance
        temperature = max(1e-6, best_c5 * 0.05 * remaining_frac)
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            # occasional uphill moves early on
            if rng.random() < math.exp(-(new_c5 - cur_c5) / temperature):
                accept = True

        if accept:
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = cur_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # stagnation handling: perturb the current best
        if no_improve >= 5000:
            # add small Gaussian noise and re‑project onto feasible set
            noise = rng.normal(scale=0.04, size=n)
            cur_h = best_h + noise
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = _project_to_simplex_box(cur_h, target_sum)
            cur_c5 = _c5_value(cur_h)
            no_improve = 0

    # final verification (project once more to be safe)
    best_h = _project_to_simplex_box(best_h, target_sum)
    best_c5 = _c5_value(best_h)

    return (best_h.tolist(), float(best_c5), n)
