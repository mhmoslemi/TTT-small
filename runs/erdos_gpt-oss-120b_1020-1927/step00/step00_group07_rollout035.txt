def run(seed=42, budget_s=1100):
    import time, numpy as np, scipy.optimize as opt

    np.random.seed(seed)
    start = time.time()

    target_n = 200   # choose
    n = target_n

    dx = 2.0 / n
    target_sum = n / 2

    def project(v):
        # clip between 0 and1 then adjust via lambda shift to get sum = target_sum
        # Use binary search for lambda.
        lo, hi = -5.0, 5.0
        for _ in range(50):
            mid = (lo+hi)/2.0
            proj = np.clip(v - mid, 0.0, 1.0)
            s = proj.sum()
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        return np.clip(v - lo, 0.0, 1.0)

    best_h = None
    best_C = np.inf

    # use initial_h_values if exists
    if 'initial_h_values' in globals():
        init_h = initial_h_values.astype(np.float64)
        # If length different, interpolate
        if len(init_h) != n:
            # upsample via linear interpolation
            x_old = np.linspace(0, 1, len(init_h))
            x_new = np.linspace(0, 1, n)
            init_h = np.interp(x_new, x_old, init_h)
        init_h = np.clip(init_h, 0.0, 1.0)
        init_h = project(init_h)
        # evaluate
        C_val = (np.max(np.correlate(init_h, 1 - init_h, mode='full')) * dx)
        best_h, best_C = init_h, C_val

    # define objective
    beta = 20.0
    lambda_pen = 100.0

    def obj_grad(x):
        h = x
        sum_h = np.sum(h)
        prod = np.correlate(h, h, mode='full')
        prod_vec = prod[n-1:n-1+n]   # shift 0..n-1
        C_vec = sum_h - prod_vec   # raw (not scaled)
        # Softmax approx
        maxC = np.max(C_vec)
        exp_vals = np.exp(beta * (C_vec - maxC))
        sum_exp = np.sum(exp_vals)
        w = exp_vals / sum_exp

        approx_maxC = maxC + np.log(sum_exp) / beta

        diff = sum_h - target_sum
        obj = approx_maxC * dx + lambda_pen * diff * diff

        # Gradient
        grad = np.zeros_like(h)
        for k in range(n):
            wk = w[k]
            if wk == 0:
                continue
            if k == 0:
                pos = h
                neg = h
            else:
                pos = np.empty_like(h)
                pos[:n-k] = h[k:]
                pos[n-k:] = 0.0
                neg = np.empty_like(h)
                neg[:k] = 0.0
                neg[k:] = h[:n-k]
            grad += wk * (dx * (1.0 - pos - neg))
        grad += 2.0 * lambda_pen * diff * np.ones_like(h)
        return obj, grad

    bounds = [(0.0, 1.0)] * n

    # Number of restarts limited by time
    restart = 0
    while time.time() - start < budget_s * 0.9:
        # generate random start
        rnd = np.random.rand(n)
        rnd = project(rnd)
        # optionally use previous best as start for later restarts
        if restart % 5 == 0 and best_h is not None:
            start_vec = best_h.copy()
            # perturb a bit
            perturb = np.random.normal(scale=0.05, size=n)
            start_vec = np.clip(start_vec + perturb, 0.0, 1.0)
            start_vec = project(start_vec)
        else:
            start_vec = rnd

        # Run optimizer
        try:
            res = opt.minimize(obj_grad, start_vec, method='L-BFGS-B',
                               jac=True, bounds=bounds,
                               options={'maxfun': 1200, 'ftol': 1e-12})
        except Exception:
            restart += 1
            continue

        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project(h_opt)
        C_opt = np.max(np.correlate(h_opt, 1 - h_opt, mode='full')) * dx
        if C_opt < best_C:
            best_C = float(C_opt)
            best_h = h_opt.copy()

        restart += 1

    # In case none found, fallback to best_h else use zero array
    if best_h is None:
        best_h = np.full(n, 0.5)  # fallback
        best_C = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    return best_h, best_C, n

import time
import numpy as np
import scipy.optimize as opt

def _project_to_feasible(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the box [0,1] and the hyperplane sum(v)=target_sum."""
    lo, hi = -10.0, 10.0  # bounds for the Lagrange multiplier
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - lo, 0.0, 1.0)


def _eval_c5(h: np.ndarray) -> float:
    """Exact C5 value according to the provided evaluator."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _obj_and_grad(x: np.ndarray, n: int, beta: float,
                  lam: float, target_sum: float):
    """Smooth‑max objective (scaled by dx) plus a quadratic sum penalty."""
    dx = 2.0 / n
    sum_h = x.sum()
    # autocorrelation of h with itself (full)
    conv = np.correlate(x, x, mode="full")
    # shifts 0 .. n-1 correspond to indices n-1 .. 2n-2
    prod = conv[n - 1 : n - 1 + n]          # ∑ h_i h_{i+k}
    C_raw = sum_h - prod                    # raw (unscaled) C_k values
    # smooth max via log‑sum‑exp
    m = C_raw.max()
    exp_vals = np.exp(beta * (C_raw - m))
    s_exp = exp_vals.sum()
    w = exp_vals / s_exp                    # softmax weights (sum to 1)
    smooth_max = m + np.log(s_exp) / beta   # smooth approximation of max C_k
    # penalty for the equality constraint
    diff = sum_h - target_sum
    obj = smooth_max * dx + lam * diff * diff

    # gradient of the smooth max term
    grad = np.zeros_like(x)
    for k in range(n):
        wk = w[k]
        if wk == 0.0:
            continue
        if k == 0:
            pos = x
            neg = x
        else:
            pos = np.empty_like(x)
            pos[: n - k] = x[k:]
            pos[n - k :] = 0.0
            neg = np.empty_like(x)
            neg[:k] = 0.0
            neg[k:] = x[: n - k]
        grad += wk * (dx * (1.0 - pos - neg))
    # gradient of the sum‑penalty term
    grad += 2.0 * lam * diff * np.ones_like(x)
    return obj, grad


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C5 bound and the number of samples."""
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and initial point
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=np.float64)
        n = init.shape[0]
        # make sure the initial vector respects the constraints
        init = np.clip(init, 0.0, 1.0)
        target_sum = n / 2.0
        init = _project_to_feasible(init, target_sum)
    else:
        n = 200                     # reasonable default when no parent exists
        target_sum = n / 2.0
        init = np.random.rand(n)
        init = _project_to_feasible(init, target_sum)

    best_h = init.copy()
    best_c5 = _eval_c5(best_h)

    # optimization parameters (can be tuned)
    beta = 30.0                     # smooth‑max sharpness
    lam = 200.0                     # equality‑constraint penalty
    max_iter_per_restart = 1200    # passed to L‑BFGS‑B via maxfun
    max_restarts = 30

    # ------------------------------------------------------------------
    # Helper for the optimizer (SciPy passes *args*)
    # ------------------------------------------------------------------
    def _opt_fun(x, n, beta, lam, target_sum):
        return _obj_and_grad(x, n, beta, lam, target_sum)

    bounds = [(0.0, 1.0)] * n

    # ------------------------------------------------------------------
    # Main optimisation loop (time‑bounded)
    # ------------------------------------------------------------------
    rest = 0
    while time.time() - start_time < budget_s * 0.85 and rest < max_restarts:
        # start point: either a perturbed best or a fresh random one
        if rest % 5 == 0 and best_c5 < 0.6:
            start = best_h + np.random.normal(scale=0.03, size=n)
        else:
            start = np.random.rand(n)
        start = np.clip(start, 0.0, 1.0)
        start = _project_to_feasible(start, target_sum)

        # run a bounded L‑BFGS‑B optimisation
        try:
            res = opt.minimize(
                _opt_fun,
                start,
                args=(n, beta, lam, target_sum),
                method="L-BFGS-B",
                jac=True,
                bounds=bounds,
                options={"maxfun": max_iter_per_restart, "ftol": 1e-12},
            )
        except Exception:
            rest += 1
            continue

        h_opt = _project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
        c5_val = _eval_c5(h_opt)

        if c5_val < best_c5:
            best_c5 = float(c5_val)
            best_h = h_opt.copy()
        rest += 1

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_c5 = _eval_c5(best_h)

    return best_h, best_c5, n
