import time, math
import numpy as np

def _project_onto_sum_cap(h):
    """
    Project onto the hyperplane sum(h)=len(h)/2 with box 0<=h<=1.
    Uses a simple binary search for the Lagrange multiplier.
    """
    n = h.shape[0]
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(45):
        lam = (lo + hi) * 0.5
        x = np.clip(h + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    x = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction to hit the sum exactly
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size:
            adj = diff / free.size
            x[free] += adj
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: adjust one coordinate
            i = np.argmin(np.abs(h - 0.5))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def _c5_and_shift(h):
    """Return (C5 value, shift that attains it) for the current step function."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(conv))
    n = h.shape[0]
    shift = idx - (n - 1)          # integer shift in samples
    c5 = conv[idx] * (2.0 / n)    # convert to integral value
    return c5, shift

def _gradient_for_shift(h, shift):
    """Subgradient of the worst overlap functional for a given integer shift."""
    n = h.shape[0]
    g = np.ones(n)
    if shift >= 0:
        l = shift
        # term -h_i h_{i+l}
        g[:n - l] -= h[l:]
        # term -h_{i-l} h_i
        g[l:]     -= h[:n - l]
    else:
        d = -shift
        # term -h_i h_{i-d}
        g[d:]     -= h[:n - d]
        # term -h_{i+d} h_i
        g[:n - d] -= h[d:]
    return g

def _pair_move(h, inc_idx, dec_idx, rng):
    """Move mass from dec_idx (decrease) to inc_idx (increase) respecting bounds."""
    delta_max = min(h[dec_idx], 1.0 - h[inc_idx])
    if delta_max <= 0.0:
        return None
    # scale factor 0..0.5 for moderate steps
    delta = delta_max * rng.random() * 0.5
    new_h = h.copy()
    new_h[inc_idx] += delta
    new_h[dec_idx] -= delta
    return new_h

def _local_search(initial_h, budget):
    """Gradient‑guided simulated annealing for a fixed amount of time."""
    n = initial_h.shape[0]
    cur_h = initial_h.copy()
    cur_c5, cur_shift = _c5_and_shift(cur_h)
    best_h, best_c5 = cur_h.copy(), cur_c5
    start = time.time()
    rng = np.random.RandomState(0)
    while time.time() - start < budget:
        # gradient of the worst shift
        g = _gradient_for_shift(cur_h, cur_shift)
        inc = np.where(g < -1e-9)[0]   # we would like to increase these
        dec = np.where(g >  1e-9)[0]   # we would like to decrease these
        if inc.size and dec.size:
            i = rng.choice(inc)
            j = rng.choice(dec)
            cand = _pair_move(cur_h, i, j, rng)
        else:
            # fall back to a random feasible pair move
            i, j = rng.choice(n, size=2, replace=False)
            # decide direction randomly
            if rng.rand() < 0.5:
                cand = _pair_move(cur_h, i, j, rng)
            else:
                cand = _pair_move(cur_h, j, i, rng)
        if cand is None:
            continue
        cand_c5, cand_shift = _c5_and_shift(cand)
        # temperature schedule (linear cooling)
        elapsed = time.time() - start
        T = max(0.02 * (1.0 - elapsed / budget), 1e-6)
        if cand_c5 < cur_c5 or rng.rand() < math.exp(-(cand_c5 - cur_c5) / T):
            cur_h, cur_c5, cur_shift = cand, cand_c5, cand_shift
            if cand_c5 < best_c5:
                best_h, best_c5 = cand.copy(), cand_c5
        # occasional random restart to escape stagnation
        if rng.rand() < 0.001:
            cur_h = rng.rand(n)
            cur_h = _project_onto_sum_cap(cur_h)
            cur_c5, cur_shift = _c5_and_shift(cur_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    # ----- choose discretisation size -----
    n_target = 200                     # moderate size, fast enough for many iterations

    # ----- build an initial step function -----
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        x_base = np.linspace(0.0, 2.0, base.size, endpoint=False)
        x_target = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h0 = np.interp(x_target, x_base, base)
    else:
        h0 = np.full(n_target, 0.5)

    h0 = _project_onto_sum_cap(h0)

    # ----- run several independent searches -----
    best_h, best_c5 = h0.copy(), _c5_and_shift(h0)[0]
    total_start = time.time()
    # allocate time slices (roughly 0.2 s per restart overhead)
    time_per_restart = budget_s * 0.18
    for _ in range(4):
        elapsed = time.time() - total_start
        if elapsed + time_per_restart > budget_s * 0.97:
            break
        # random restart (except first which uses the interpolated start)
        if _ == 0:
            start_vec = h0
        else:
            start_vec = rng.rand(n_target)
            start_vec = _project_onto_sum_cap(start_vec)
        cand_h, cand_c5 = _local_search(start_vec, time_per_restart)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand_h.copy(), cand_c5

    # ----- final refinement (optional) -----
    # a short extra descent from the current best
    extra = min(5.0, budget_s - (time.time() - total_start))
    if extra > 1.0:
        best_h, best_c5 = _local_search(best_h, extra)

    return best_h, float(best_c5), int(best_h.shape[0])
