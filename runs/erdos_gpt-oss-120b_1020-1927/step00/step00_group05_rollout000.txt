import time
import random
import math
import numpy as np

def _c5_of_h(h: np.ndarray) -> float:
    """Compute the C5 value for a given (binary) h."""
    # dx = 2 / len(h)
    # correlation of h with its complement
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / h.size))

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a step function h on [0,2] (as a vector of samples) together with
    its overlap bound C5.  The vector satisfies 0≤h[i]≤1 and sum(h)*dx = 1.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ---- parameters -------------------------------------------------
    n_points = 200                     # even number, not too large
    dx = 2.0 / n_points
    target_sum = n_points // 2        # exactly half ones for binary case

    max_time = budget_s - 0.05         # safety margin
    start_time = time.time()

    # ---- helper for a random balanced binary vector -----------------
    def random_balanced():
        h = np.zeros(n_points, dtype=np.float64)
        idx = np.arange(n_points)
        np.random.shuffle(idx)
        h[idx[:target_sum]] = 1.0
        return h

    # ---- initialise -------------------------------------------------
    best_h = random_balanced()
    best_c5 = _c5_of_h(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # annealing schedule parameters
    T_start = 0.02                     # initial temperature
    # statistics for restart logic
    it_without_best = 0
    RESTART_AFTER = 200_000           # iterations without improvement

    iter_count = 0
    # main SA / hill‑climb loop
    while time.time() - start_time < max_time:
        iter_count += 1

        # linear temperature decay based on elapsed fraction
        elapsed = time.time() - start_time
        frac = elapsed / max_time
        T = max(1e-6, T_start * (1.0 - frac))

        # propose a swap of a 1 and a 0
        i, j = random.randrange(n_points), random.randrange(n_points)
        if cur_h[i] == cur_h[j]:
            continue          # need opposite values to keep the sum fixed

        # perform swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c5 = _c5_of_h(cur_h)

        delta = new_c5 - cur_c5
        accept = False
        if delta <= 0:
            accept = True
        else:
            # Metropolis acceptance
            if random.random() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = cur_h.copy()
                it_without_best = 0
            else:
                it_without_best += 1
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            it_without_best += 1

        # occasional restart if stuck
        if it_without_best > RESTART_AFTER:
            cur_h = random_balanced()
            cur_c5 = _c5_of_h(cur_h)
            it_without_best = 0

    # ensure the final solution respects the integral exactly (binary does)
    # If for any reason the best vector is not binary (e.g., accidental float),
    # project it back.
    if not np.all((best_h == 0) | (best_h == 1)):
        # simple projection: take the highest target_sum entries as 1
        flat = best_h.argsort()[::-1]
        proj = np.zeros_like(best_h)
        proj[flat[:target_sum]] = 1.0
        best_h = proj
        best_c5 = _c5_of_h(best_h)

    return best_h, best_c5, n_points
