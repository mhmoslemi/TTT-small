import numpy as np
import time

# --------------------------------------------------------------
# Helper: projection onto the capped simplex {x: sum(x)=target, 0<=x<=1}
def project_onto_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto the set {x: sum(x)=target, 0<=x<=1}."""
    # Bisection on the Lagrange multiplier λ such that
    #   w_i = clip(v_i - λ, 0, 1)  and  sum(w_i) = target
    low, high = -1.0, 1.0
    for _ in range(60):
        lam = (low + high) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(v - (low + high) * 0.5, 0.0, 1.0)

# Simple evaluation wrapper (uses the harness if available)
def evaluate_h(h: np.ndarray) -> float:
    """Return the C₅ value for vector h."""
    try:
        return evaluate_erdos_solution(h)          # provided by the harness
    except NameError:  # fallback if the helper is not present
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * (2.0 / h.size)

# Random perturbation preserving the sum constraint
def random_perturb(h: np.ndarray, scale: float, target: float) -> np.ndarray:
    n = h.size
    # zero‑mean Gaussian noise, normalised to the requested scale
    delta = np.random.randn(n)
    delta -= delta.mean()
    mx = np.max(np.abs(delta))
    if mx > 0:
        delta = delta / mx * scale
    new = h + delta
    return project_onto_capped_simplex(new, target)

# Directed step using the gradient of the worst shift
def directed_step(h: np.ndarray, step: float, target: float) -> np.ndarray:
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    s = idx - (n - 1)            # signed shift producing the maximal overlap
    grad = np.ones(n)           # ∂c/∂h_i = 1 - (h_{i+s}+h_{i-s}) where defined
    if s >= 0:
        grad[: n - s] -= h[s:]
        grad[s:]     -= h[: n - s]
    else:
        s = -s
        grad[s:]     -= h[: n - s]
        grad[: n - s] -= h[s:]
    # move opposite to the gradient while keeping zero mean (preserves sum)
    delta = -step * (grad - grad.mean())
    new = h + delta
    return project_onto_capped_simplex(new, target)

def upsample_initial(base: np.ndarray, N: int) -> np.ndarray:
    """Linearly interpolate a given base vector to length N."""
    old_n = base.size
    xs_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, N,   endpoint=False)
    return np.interp(xs_new, xs_old, base)

    # Simulated‑annealing / local search
    cur_h = h0.copy()
    cur_val = evaluate_h(cur_h)

    best_h = cur_h.copy()
    best_val = cur_val

    start = time.time()
    # cooling schedule parameters
    T0 = 0.12
    max_scale = 0.45
    min_scale = 5e-5

    # loop until the allocated budget is nearly exhausted
    while time.time() - start < budget_s * 0.95:
        elapsed = time.time() - start
        frac = elapsed / (budget_s * 0.95)  # 0 … 1
        T = max(1e-9, T0 * (1.0 - frac))
        scale = max(min_scale, max_scale * (1.0 - frac))

        # occasionally take a directed step guided by the worst shift
        if np.random.rand() < 0.18:
            cand_h = directed_step(cur_h, scale, target_sum)
        else:
            cand_h = random_perturb(cur_h, scale, target_sum)

        cand_val = evaluate_h(cand_h)

        # keep the best ever seen
        if cand_val < best_val:
            best_val = cand_val
            best_h = cand_h.copy()

        # Metropolis acceptance
        if cand_val < cur_val or np.exp(-(cand_val - cur_val) / T) > np.random.rand():
            cur_h = cand_h
            cur_val = cand_val

        # occasional restart from the best solution (helps escape plateaus)
        if frac > 0.65 and np.random.rand() < 0.006:
            cur_h = best_h.copy()
            cur_val = best_val

    # final projection to be safe and final evaluation
    best_h = project_onto_capped_simplex(best_h, target_sum)
    final_val = evaluate_h(best_h)

    return best_h.astype(np.float32), float(final_val), n
