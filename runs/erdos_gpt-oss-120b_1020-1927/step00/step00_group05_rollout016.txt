import numpy as np, cvxpy as cp, time

def compute_max_overlap(h):
    return np.max(np.correlate(h, 1 - h, mode='full'))

def compute_c5_bound(h):
    n = len(h)
    dx = 2.0 / n
    overlap = compute_max_overlap(h)
    return overlap * dx

def solve_milp(n, time_limit):
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    # w size n x (2n-1)
    w = cp.Variable((n, 2*n-1), boolean=True)
    constraints = []
    # sum constraint
    constraints.append(cp.sum(h) == n//2)
    # shift constraints
    for shift in range(-(n-1), n):
        idx = shift + n - 1
        # compute valid i
        i_start = max(0, -shift)
        i_end = min(n, n - shift)  # exclusive
        # vector of indices
        for i in range(i_start, i_end):
            j = i + shift
            # link w to h
            constraints.append(w[i, idx] <= h[i])
            constraints.append(w[i, idx] <= 1 - h[j])
            constraints.append(w[i, idx] >= h[i] - h[j])
        # sum constraint for this shift
        if i_start < i_end:
            # sum over valid i
            constraints.append(cp.sum(w[i_start:i_end, idx]) <= t)
        else:
            # no overlap, set t >= 0 but okay
            pass
    # objective
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    # solve
    prob.solve(solver=cp.GLPK_MI, glpk={'msg_lev': 'GLP_MSG_OFF', 'tm_limit': int(time_limit)} )  # maybe time limit param.

def hill_climb(start_h, time_budget):
    best_h = start_h.copy()
    best_val = compute_c5_bound(best_h)
    n = len(start_h)
    dx = 2.0 / n
    start_time = time.time()
    while time.time() - start_time < time_budget:
        # random swap
        # find indices with 1 and 0
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # create new candidate
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = compute_c5_bound(cand)
        if cand_val < best_val - 1e-9:
            best_h = cand
            best_val = cand_val
        # else maybe accept with probability?
    return best_h, best_val

def _c5_from_h(h):
    """Return C5 bound for a step function given as a numpy array."""
    n = h.shape[0]
    dx = 2.0 / n
    overlap = np.max(np.correlate(h, 1 - h, mode="full"))
    return overlap * dx

def _solve_milp_binary(n, time_limit=None):
    """Solve the exact binary formulation.
    Returns a binary numpy array of length n or None if no solution.
    """
    # variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    w = cp.Variable((n, 2 * n - 1), boolean=True)

    constraints = [cp.sum(h) == n // 2, t >= 0]

    # build constraints for each shift
    for shift in range(-(n - 1), n):
        idx = shift + n - 1
        i_start = max(0, -shift)
        i_end = min(n, n - shift)          # exclusive
        if i_start < i_end:
            # product linearisation
            for i in range(i_start, i_end):
                j = i + shift
                constraints.append(w[i, idx] <= h[i])
                constraints.append(w[i, idx] <= 1 - h[j])
                constraints.append(w[i, idx] >= h[i] - h[j])
            # overlap bound for this shift
            constraints.append(cp.sum(w[i_start:i_end, idx]) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    # try a few solvers, give time limit where supported
    for solver in (cp.CBC, cp.ECOS_BB, cp.GLPK_MI):

                solve_kw["maxSeconds"] = time_limit
            prob.solve(solver=solver, **solve_kw)
            if h.value is not None:
                break

            continue

        return None, None

    """Simple random swap hill‑climbing on a binary vector."""
    n = start_h.shape[0]
