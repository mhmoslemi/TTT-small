import time
import math
import random
import numpy as np

def _project_to_sum_bounds(v, target):
    """Project vector v onto the set 0<=x<=1 and sum(x)=target."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _score(h):
    """Return the raw max correlation value of h with (1-h)."""
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    random.seed(seed)

    # --- starting vector -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        # fallback: 80‑point half‑filled vector
        n = 80
        h = np.zeros(n, dtype=float)
        h[: n // 2] = 1.0
        rng.shuffle(h)

    n = len(h)
    target_sum = n * 0.5
    h = _project_to_sum_bounds(h, target_sum)

    dx = 2.0 / n

    # initial scores
    cur_score = _score(h)
    best_h = h.copy()
    best_score = cur_score
    best_c5 = best_score * dx

    start = time.time()
    limit = budget_s * 0.95  # leave a margin for clean exit

    # annealing parameters
    init_temp = 5.0
    init_step = 0.2

    # counters for stagnation handling
    steps_without_improve = 0
    restart_interval = 20000

    while time.time() - start < limit:
        # cooling schedules (linear)
        elapsed = time.time() - start
        prog = elapsed / limit
        T = init_temp * (1.0 - prog) + 1e-4
        max_delta = init_step * (1.0 - prog) + 1e-5

        # pick a source (decrease) and a destination (increase)
        src_candidates = np.where(h > 1e-12)[0]
        dst_candidates = np.where(h < 1.0 - 1e-12)[0]
        if src_candidates.size == 0 or dst_candidates.size == 0:
            break
        i = int(rng.choice(src_candidates))
        j = int(rng.choice(dst_candidates))
        # maximum feasible transfer from i to j
        cap = min(h[i], 1.0 - h[j])
        if cap <= 0.0:
            continue
        delta = min(max_delta, cap) * rng.random()

        # propose new vector
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        new_score = _score(h_new)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            h = h_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_h = h_new.copy()
                best_score = new_score
                best_c5 = best_score * dx
                steps_without_improve = 0
            else:
                steps_without_improve += 1
        else:
            steps_without_improve += 1

        # occasional restart to avoid being trapped
        if steps_without_improve >= restart_interval:
            # perturb the current solution slightly around the best found
            perturb = rng.uniform(-0.05, 0.05, size=n)
            h = np.clip(best_h + perturb, 0.0, 1.0)
            h = _project_to_sum_bounds(h, target_sum)
            cur_score = _score(h)
            steps_without_improve = 0

    return best_h, float(best_c5), n
