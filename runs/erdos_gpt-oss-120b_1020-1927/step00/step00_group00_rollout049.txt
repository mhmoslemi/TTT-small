def compute_gradient(h):
    n = len(h)
    # compute correlation to get k_max
    conv = np.correlate(h, 1 - h, mode='full')
    k_idx = np.argmax(conv)
    k = k_idx - (n - 1)
    # compute term1: shift h by -k (i.e., h_i+k)
    # we want for each i: term1 = 1 - h[i + k] if i + k in range else 0
    # term2 = h[i - k] if i - k in range else 0
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    if k >= 0:
        term1[:n - k] = 1 - h[k:]
        term2[k:] = h[:n - k]
    else:
        # k negative: shift other direction
        kpos = -k
        term1[kpos:] = 1 - h[:n - kpos]
        term2[:n - kpos] = h[kpos:]
    grad = (term1 - term2) * (2.0 / n)  # multiply by dx
    return grad, conv.max() * (2.0 / n), k

def compute_gradient(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(conv)
    max_val = conv[max_idx] * dx
    k = max_idx - (n - 1)  # shift (int)

    # compute term1 and term2 arrays
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    if k >= 0:
        # for i in [0, n-k-1], i+k in range
        term1[:n - k] = 1 - h[k:]
        # for i in [k, n-1], i-k in range
        term2[k:] = h[:n - k]
    else:
        k_abs = -k
        term1[k_abs:] = 1 - h[:n - k_abs]
        term2[:n - k_abs] = h[k_abs:]
    grad = (term1 - term2) * dx
    return grad, max_val, k

        diff = target_sum - np.sum(x)
        if abs(diff) < 1e-12:
            break
        # Determine free entries (not at bound)
        free = (x > 1e-12) & (x < 1 - 1e-12)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            # distribute evenly (allow small violation)
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
        else:
            delta = diff / n_free
            # adjust free entries but ensure not over bound
            adjust = np.where(free, delta, 0.0)
            x += adjust
            # clip again to keep within bounds
            x = np.clip(x, 0.0, 1.0)
    # final small correction:
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-9:
        # find an index not at bound if possible

        for i in range(n):

                idx = i
                break
        if idx is not None:
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
        else:
            # all at bound, can't adjust
            pass
    return x

# ---------- helper functions ----------

def compute_c5(h):
    """return max_k ∫ h(x)(1-h(x+k))dx approximated on the discrete grid."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx


def compute_gradient(h):
    """
    Returns (grad, current_c5, shift_k) where grad is the sub‑gradient of the
    current active overlap functional (the shift attaining the maximum).
    """
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(conv))
    cur_val = conv[max_idx] * dx
    k = max_idx - (n - 1)                     # shift corresponding to max
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    if k >= 0:
        term1[: n - k] = 1.0 - h[k:]
        term2[k:] = h[: n - k]
    else:
        kk = -k
        term1[kk:] = 1.0 - h[: n - kk]
        term2[: n - kk] = h[kk:]
    grad = (term1 - term2) * dx
    return grad, cur_val, k


def project_to_feasible(x, target_sum=None):
    """Clip to [0,1] and adjust to meet the linear sum constraint."""
    n = x.shape[0]
    if target_sum is None:
        target_sum = n / 2.0
    x = np.clip(x, 0.0, 1.0)
    for _ in range(20):
        cur = np.sum(x)
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (x > 1e-9) & (x < 1.0 - 1e-9)
        cnt = int(free.sum())
        if cnt == 0:
            # everything at a bound – distribute uniformly (tiny violation is OK)
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
        else:
            delta = diff / cnt
            x[free] += delta
            x = np.clip(x, 0.0, 1.0)
    # final tiny correction if needed
    final_diff = target_sum - np.sum(x)
    if abs(final_diff) > 1e-9:
        idx = np.where((final_diff > 0) & (x < 1 - 1e-12))[0]
        if idx.size == 0:
            idx = np.where((final_diff < 0) & (x > 1e-12))[0]
        if idx.size:
            i = int(idx[0])
            x[i] = np.clip(x[i] + final_diff, 0.0, 1.0)
    return x


def local_optimize(h, time_deadline):
    """Gradient + occasional random swap descent on a fixed grid size."""
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(best_h)
    alpha = 0.5
    stagnation = 0
    while time.time() < time_deadline:
        grad, cur, _ = compute_gradient(best_h)
        step = -alpha * grad
        cand = project_to_feasible(best_h + step, target)
        cand_score = compute_c5(cand)

        if cand_score < best_score - 1e-12:
            best_h, best_score = cand, cand_score
            alpha = min(alpha * 1.05, 1.0)
            stagnation = 0
            continue

        # random swap (preserves sum exactly)
        i, j = random.sample(range(n), 2)
        cand2 = best_h.copy()
        cand2[i], cand2[j] = cand2[j], cand2[i]
        cand2 = project_to_feasible(cand2, target)
        cand2_score = compute_c5(cand2)

        if cand2_score < best_score - 1e-12:
            best_h, best_score = cand2, cand2_score
            alpha = min(alpha * 1.05, 1.0)
            stagnation = 0
            continue

        # no improvement
        stagnation += 1
        alpha *= 0.99
        if alpha < 1e-3:
            alpha = 0.5
        if stagnation >= 3000:
            break
    return best_h, best_score


def upscale(h):
    """Double the number of sample points via linear interpolation."""
    n = h.shape[0]
    new_n = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_to_feasible(h_new, target_sum=new_n / 2.0)
    return h_new


# ---------- main entry point ----------

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # ---- obtain starting vector ----
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
    except NameError:
        # no parent: build a random balanced binary vector (half ones)
        n0 = 60
        ones = np.ones(n0 // 2)
        zeros = np.zeros(n0 - n0 // 2)
        h0 = np.concatenate([ones, zeros])
        np.random.shuffle(h0)
        h0 = h0.astype(float)
    # ensure feasibility
    h0 = project_to_feasible(h0)

    best_h = h0
    best_score = compute_c5(best_h)

    # ---- multi‑scale optimisation loop ----
    current_h = best_h
    while time.time() < deadline:
        remaining = deadline - time.time()
        # allocate time slice for this grid (at most half of remaining)
        slice_deadline = time.time() + min(remaining * 0.5, 30.0)

        # local optimisation on current grid
        opt_h, opt_score = local_optimize(current_h, slice_deadline)
        if opt_score < best_score:
            best_h, best_score = opt_h, opt_score

        # decide whether to refine the grid
        if len(opt_h) * 2 <= 200 and (deadline - time.time()) > 5.0:
            current_h = upscale(opt_h)
        else:
            # no more up‑scaling possible or not enough time left
            break

    # final tiny polishing on the finest grid used
    final_deadline = deadline
    best_h, best_score = local_optimize(best_h, final_deadline)

    # guarantee constraints (tiny numerical safety)
    best_h = project_to_feasible(best_h)

    # return as required
    return best_h, best_score, best_h.shape[0]
