def project_onto_bounds_sum(v, target):
    # v: np array
    lo = np.min(v) - 1.0 # lower bound for lambda (makes all v_i - lambda > 1)
    hi = np.max(v) # upper bound for lambda (makes all v_i - lambda < 0)
    # Actually we need to find λ such that sum Clip(v - λ) = target.
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0, 1)
        s = np.sum(x)
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi)/2.0, 0, 1)
    return x

This binary search works because sum Clip(v - λ) is decreasing in λ (as λ increases, each term decreases or is clipped). At λ -> -infty, v - λ large, Clip => 1, sum = n. At λ large, sum =0. So solution exists for target between 0 and n. Since target = n/2, doable.

def project_sum_bound(v, target):
    # ensures each element in [0,1] and sum = target
    # Use water-filling via sorting.
    # We'll implement binary search as earlier.
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

Check monotonic: For larger lam, x smaller; indeed if sum > target we need increase lam (like shift decreasing values) because we need sum lower; but here's reversed: If sum > target, we want to increase lam to decrease values (since we subtract lam), so set lo = lam. But think: We start lo low negative large => v - lam larger => after clipping to 1, many become 1; sum > target. Then we set lo = lam to increase lam. Continue. At the end lam is such that sum approx target. So logic correct: If sum > target, lam too low (we need to increase), set lo = lam. Else sum <= target, lam too high (decrease), set hi = lam. def expand_hi_to_h(hi, n):
    m = len(hi)
    base = n // m
    remainder = n % m
    segments = [base + (1 if i < remainder else 0) for i in range(m)]
    h = np.concatenate([np.full(seg, hi[i], dtype=float) for i, seg in enumerate(segments)])
    return h

Now compute C5.

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    return max_val * (2.0 / n)

def simulated_annealing(m, n, budget_sec, rng):
    target_sum = m / 2.0
    # start with random hi
    hi = random_hi(m)
    cur_h = expand_hi_to_h(hi, n)
    cur_c5 = compute_c5(cur_h)
    best_hi = hi.copy()
    best_c5 = cur_c5
    # SA parameters
    T0 = 0.1
    Tmin = 1e-5
    # approximate total iterations budget
    start = time.time()
    it = 0
    max_iter = 20000  # will break earlier if time near end
    while time.time() - start < budget_sec and it < max_iter:
        it += 1
        fraction = it / max_iter
        T = max(T0 * (1 - fraction) + Tmin * fraction, Tmin)  # linear cooling
        sigma = 0.2 * (1 - fraction) + 0.01 * fraction
        hi_new = hi + rng.normal(scale=sigma, size=m)
        hi_new = project_sum_bound(hi_new, target_sum)
        h_new = expand_hi_to_h(hi_new, n)
        c5_new = compute_c5(h_new)
        delta = c5_new - cur_c5
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            hi = hi_new
            cur_c5 = c5_new
            cur_h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_hi = hi_new.copy()
    return best_hi, best_c5

import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_sum_bound(v, target):
    # project onto [0,1] box and sum=target via binary search on lambda
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def expand_hi_to_h(hi, n):
    m = len(hi)
    base = n // m
    rem = n % m
    segments = [base + (1 if i < rem else 0) for i in range(m)]
    parts = [np.full(seg, hi[i], dtype=float) for i, seg in enumerate(segments)]
    return np.concatenate(parts)

def random_feasible_hi(m, rng):
    v = rng.random(m)
    return project_sum_bound(v, m / 2.0)

def simulated_annealing(m, n, time_budget, rng):

import numpy as np
import time
import math

def _compute_c5(h: np.ndarray) -> float:
    """C5 for a discretised step function."""
    n = h.size
    # full linear correlation of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def _project_sum_bound(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the box [0,1]^m
    with the additional linear equality sum(v) = target.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0                # wide enough interval for the multiplier
    for _ in range(60):               # enough for double precision
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _expand_hi(hi: np.ndarray, n: int) -> np.ndarray:
    """Repeat each entry of hi to obtain a length‑n vector (even n)."""
    m = hi.size
    base = n // m
    rem = n % m
    seg_lens = [base + (1 if i < rem else 0) for i in range(m)]
    parts = [np.full(l, hi[i], dtype=float) for i, l in enumerate(seg_lens)]
    return np.concatenate(parts)

def _random_feasible_hi(m: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random feasible hi (0‑1 box, sum = m/2)."""
    v = rng.random(m)
    return _project_sum_bound(v, m * 0.5)

def _simulated_annealing(m: int, n: int, time_budget: float,
                         rng: np.random.Generator):
    """Very lightweight SA on the m‑dimensional space of interval heights."""
    target_sum = m * 0.5
    hi = _random_feasible_hi(m, rng)
    cur_h = _expand_hi(hi, n)
    cur_c5 = _compute_c5(cur_h)

    best_hi = hi.copy()
    best_c5 = cur_c5

    start = time.time()
    max_iter = 50000
    for it in range(max_iter):
        if time.time() - start > time_budget:
            break

        frac = it / max_iter
        # cooling – temperature drops linearly, never reaches 0
        T = 0.2 * (1.0 - frac) + 1e-4 * frac
        # mutation strength – also decays
        sigma = 0.3 * (1.0 - frac) + 0.01 * frac

        hi_new = hi + rng.normal(scale=sigma, size=m)
        hi_new = _project_sum_bound(hi_new, target_sum)

        h_new = _expand_hi(hi_new, n)
        c5_new = _compute_c5(h_new)

        delta = c5_new - cur_c5
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            hi, cur_c5 = hi_new, c5_new
            cur_h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_hi = hi_new.copy()

    return best_hi, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with integral 1 that minimises
    max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---------- start from the supplied construction if any ----------
    best_h = None
    best_c5 = np.inf
    init = globals().get("initial_h_values")
    if init is not None:
        init_arr = np.asarray(init, dtype=float)
        best_h = init_arr.copy()
        best_c5 = _compute_c5(best_h)

    # ---------- parameters for the search ----------
    n = 200                     # number of sample points (even)
    ms = [5, 6, 7, 8, 9, 10, 12, 15, 20]   # number of intervals for the piecewise model

    remaining = len(ms)
    for m in ms:
        # time bookkeeping – stop if we are out of budget
        now = time.time()
        time_left = budget_s - (now - start)
        if time_left < 5.0:
            break

        # allocate a slice of the remaining budget to this m
        alloc = min(time_left * 0.15, max(10.0, time_left * 0.5 / remaining))
        hi_opt, c5_opt = _simulated_annealing(m, n, alloc, rng)

        if c5_opt < best_c5:
            best_h = _expand_hi(hi_opt, n)
            best_c5 = c5_opt

        remaining -= 1

    # ---------- final safety projection to respect the constraints ----------
    if best_h is None:
        # fallback – uniform 0.5 function (always feasible)
        best_h = np.full(n, 0.5, dtype=float)
        best_c5 = _compute_c5(best_h)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        target_sum = best_h.size * 0.5
        best_h = _project_sum_bound(best_h, target_sum)
        best_c5 = _compute_c5(best_h)

    return best_h, float(best_c5), int(best_h.size)
