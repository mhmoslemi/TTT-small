def project(v, target):
    # v arbitrary
    x = np.clip(v, 0.0, 1.0)
    # compute current sum
    cur = np.sum(x)
    diff = target - cur
    # We'll adjust using a water-filling method:
    # identify indices where x not at bounds (0< x <1)
    # if diff >0, we need to increase some entries up to 1.
    # if diff <0, we need to decrease some entries down to 0.
    # We'll distribute diff uniformly among free indices, respecting bounds, iteratively.
    free = np.ones_like(x, dtype=bool)
    # at bounds based on direction:
    while abs(diff) > 1e-12:
        # find indices free to adjust:
        if diff > 0:
            free_idx = np.where((x < 1 - 1e-12) & free)[0]
            if len(free_idx) == 0:
                break
            # max increase each can take:
            max_inc = 1 - x[free_idx]
            # amount per index if equal: inc = diff / len(free_idx)
            inc = diff / len(free_idx)
            # limited inc to max_inc
            inc_vec = np.minimum(inc, max_inc)
            x[free_idx] += inc_vec
            cur = np.sum(x)
            diff = target - cur
            # mark indices that hit bound now:
            free[free_idx[(x[free_idx] >= 1 - 1e-12)]] = False
        else:
            # diff <0, need to decrease
            free_idx = np.where((x > 1e-12) & free)[0]
            if len(free_idx) == 0:
                break
            max_dec = x[free_idx]
            dec = diff / len(free_idx)  # note diff negative
            dec_vec = np.maximum(dec, -max_dec)
            x[free_idx] += dec_vec
            cur = np.sum(x)
            diff = target - cur
            free[free_idx[(x[free_idx] <= 1e-12)]] = False
        # break if no change possible:
        if len(free_idx) == 0:
            break
    # final clip to guarantee bounds
    x = np.clip(x, 0.0, 1.0)
    # may still be off due to rounding; small final correction using simple approach:
    # if sum still off beyond tolerance, we adjust smallest (or largest) entries.
    cur = np.sum(x)
    diff = target - cur
    if abs(diff) > 1e-8:
        # adjust using a single element that is not at bound
        # find any index where we can add diff (if diff >0 and x<1; else if diff<0 and x>0)
        # adjust most interior element.
        if diff > 0:
            idx = np.argmax(x)  # Actually need one not at bound; find min value not at 1.

def project(v, target):
    # v: array of shape N
    # target: sum constraint.
    # Return x = clip(v - lam, 0, 1) where lam found by bisection.
    # This can handle sum within [0,N].
    low = np.min(v) - 1.0  # ensure v - low >=1 => clip to 1
    high = np.max(v)       # ensure v - high <=0 => clip to 0
    # compute sum at low and high for monotonic check
    # Actually if lambda = low, v - low >= v - (min(v)-1) = v - min(v) + 1 >=1 for all v? Since v - min(v) >=0, plus 1 >=1. So clip=1 => sum=N.
    # If lambda = high, v - high <=0 => clip=0 => sum=0.
    # So we can bisect between low and high.
    for _ in range(50):
        lam = (low + high) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    x = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    # due to numeric error, adjust small diff by scaling
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff among non-bound entries
        # find indices where 0 < x < 1
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if len(free) > 0:
            # add diff/len to each, then clip again
            add = diff / len(free)
            x[free] += add
            x = np.clip(x, 0.0, 1.0)

            # fallback: adjust any one index
            idx = np.argmin(x) if diff > 0 else np.argmax(x)
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    # final clip
    return x

import time
import random
import numpy as np

def _objective(h):
    """Return C5 value for step vector h."""
    N = h.shape[0]
    dx = 2.0 / N
    # correlation h with (1-h); np.correlate gives full cross‑correlation
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _max_corr_and_grad(h):
    """Maximum overlap (uncalibrated) and its gradient w.r.t. h."""
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    k = idx - (N - 1)                     # lag producing the maximum
    g = np.zeros_like(h)

    if k >= 0:
        # contribution from h[i] * (1 - h[i+k])
        g[: N - k] += (1.0 - h[k:])
        # contribution from -h[i] * (1 - h[i-k]) after index shift
        g[k:] -= h[: N - k]
    else:
        m = -k
        g[m:] += (1.0 - h[: N - m])
        g[: N - m] -= h[m:]

    return corr[idx], g


def _project_onto_box_and_sum(v, target):
    """
    Project vector v onto {0 <= x <= 1, sum(x) = target}.
    Uses a simple bisection on the Lagrange multiplier λ for the box‑plus‑simplex.
    """
    low = np.min(v) - 1.0          # guarantee all entries become 1
    high = np.max(v)               # guarantee all entries become 0

    for _ in range(50):
        lam = (low + high) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam

    x = np.clip(v - (low + high) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size:
            x[free] += diff / free.size
            x = np.clip(x, 0.0, 1.0)
        else:
            # adjust a single coordinate if everything is on a bound
            idx = np.argmin(x) if diff > 0 else np.argmax(x)
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)

    return x


def _optimize(start_h, max_iters, step0, time_limit, start_time):
    """Projected gradient descent with occasional random pair swaps."""
    h = start_h.copy()
    N = h.shape[0]
    target = N / 2.0
    best_h = h.copy()
    best_val = _objective(h)

    step = step0
    last_improve = 0

    for it in range(max_iters):
        # budget check
        if time.time() - start_time > time_limit * 0.99:
            break

        max_corr, grad = _max_corr_and_grad(h)

        # descent direction with zero‑sum constraint
        d = -grad
        d -= np.mean(d)

        # gradient step + projection
        cand = _project_onto_box_and_sum(h + step * d, target)
        cand_val = _objective(cand)

        if cand_val < best_val - 1e-12:
            best_val = cand_val
            best_h = cand.copy()
            last_improve = it

        if cand_val < max_corr:          # accept if we reduced the current max
            h = cand

        # occasional random pair exchange to escape flat regions
        if it % 100 == 0:
            i, j = np.random.randint(N, size=2)
            if i != j:
                # move a little mass from i to j (or reverse)
                if np.random.rand() < 0.5:
                    delta = min(h[i], 1.0 - h[j]) * np.random.rand() * 0.5
                    h_tmp = h.copy()
                    h_tmp[i] -= delta
                    h_tmp[j] += delta
                else:
                    delta = min(h[j], 1.0 - h[i]) * np.random.rand() * 0.5
                    h_tmp = h.copy()
                    h_tmp[j] -= delta
                    h_tmp[i] += delta

                val_tmp = _objective(h_tmp)
                if val_tmp < best_val:
                    best_val = val_tmp
                    best_h = h_tmp.copy()
                if val_tmp < max_corr:
                    h = h_tmp

            step *= 0.995                     # modest cooling

        # if stagnation, shrink step size
        if it - last_improve > 5000:
            step *= 0.5
            last_improve = it

    return best_h, best_val
