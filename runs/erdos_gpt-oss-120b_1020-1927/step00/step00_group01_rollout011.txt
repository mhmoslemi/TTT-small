def optimize_h(N, h_init, max_iters=5000, step0=0.2):
    h = h_init.copy()
    best_h = h.copy()
    best_val = compute_max_corr(h)
    for t in range(1, max_iters+1):
        # compute correlation and max indices
        c = np.correlate(h, 1 - h, mode='full')
        max_val = c.max()
        # early stop if improvement small maybe
        # compute gradient for max lag(s)
        # find all indices where c == max_val (within tolerance)
        # create mask
        mask = (c >= max_val - 1e-12)
        # compute lag values: indices to lag: lags = np.arange(-N+1, N)[mask]
        # For each lag, compute grad and accumulate.
        grads = np.zeros_like(h)
        lags = np.arange(-len(h)+1, len(h))
        for lag in lags[mask]:
            grads += lag_gradient(h, lag)
        grads /= mask.sum()
        # step size schedule
        step = step0 / np.sqrt(t)
        h_new = h - step * grads
        # project onto constraints
        h = project_capped_simplex(h_new, target_sum=N/2, lower=0., upper=1.)
        # compute objective
        c_new = np.correlate(h, 1 - h, mode='full')
        val_new = c_new.max()
        if val_new < best_val - 1e-12:

import time
import numpy as np

def _project_capped_simplex(v: np.ndarray, target_sum: float,
                           lower: float = 0.0, upper: float = 1.0) -> np.ndarray:
    """Project onto the intersection of the box [lower,upper]^n and the hyperplane sum=v."""
    # binary search for lambda such that sum(clip(v+lambda, lower, upper)) = target_sum
    lo = -np.max(v - lower) - 1.0
    hi = np.max(upper - v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s > target_sum:
            hi = lam
        else:
            lo = lam
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, lower, upper)


def _c5_value(h: np.ndarray) -> float:
    """Return C₅ = max_k ∫ h(x)(1‑h(x+k)) dx for the discretisation defined by h."""
    N = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / N)


def _lag_gradient(h: np.ndarray, lag: int) -> np.ndarray:
    """Gradient of the correlation term for a single lag."""
    N = h.size
    grad = np.zeros_like(h)
    if lag >= 0:
        L = lag
        if L < N:
            grad[: N - L] += 1.0 - h[L:]
            grad[L:]       -= h[: N - L]
    else:
        L = -lag
        if L < N:
            grad[L:]       += 1.0 - h[: N - L]
            grad[: N - L]  -= h[L:]
    return grad


def run(seed: int = 42, budget_s: int = 1100, **kwargs):

    start_time = time.time()

        base_N = 0

    best_h = None
    best_val = np.inf

    # ----------------------------------------------------------------------
    # Parameter space – a few grid sizes, many short optimisations.
    # ----------------------------------------------------------------------
    grid_sizes = [150, 200, 250]          # all < 1000 points – fast to evaluate
    max_restarts = 5                     # independent starts per grid size
    iters_per_restart = 4000             # sub‑gradient steps per start
    step0 = 0.8                          # initial learning rate (decays)

    for N in grid_sizes:
        if time.time() - start_time > budget_s * 0.95:
            break

        # linear interpolation from the supplied base (if any)
        def _make_initial():
            if init_base is not None:
                x_old = np.linspace(0.0, 2.0, base_N, endpoint=False)
                x_new = np.linspace(0.0, 2.0, N, endpoint=False)
                h0 = np.interp(x_new, x_old, init_base)

        for _ in range(max_restarts):

            val = _c5_value(h)
            if val < best_val:

            for t in range(1, iters_per_restart + 1):
