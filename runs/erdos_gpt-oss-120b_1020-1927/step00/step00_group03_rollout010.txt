    # find λ such that sum(min(max(y_i + λ, 0), 1)) = target.

        proj = np.clip(y + mid, 0, 1)

            low = mid
        else:

def hill_climb_sa(h, n_iter, init_temp, final_temp):
    best_h = h.copy()
    best_score = evaluate(best_h)
    cur_h = h.copy()
    cur_score = best_score
    T = init_temp

        T = max(final_temp, init_temp * (1 - step / n_iter))
        i, j = np.random.randint(n), np.random.randint(n)

            j = np.random.randint(n)

def project_to_bounds_and_sum(v, target):
    # project onto [0,1] and sum target using bisection
    lo = -np.max(v) - 1
    hi = 1 - np.min(v) + 1
    for _ in range(50):
        mid = (lo + hi) / 2
        proj = np.clip(v + mid, 0, 1)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2, 0, 1)

import time, math
import numpy as np

def _project_bounds_sum(y, target):
    """Project y onto {x | 0<=x_i<=1, sum x = target} using bisection."""
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_of(h, dx):
    """Return the C5 value for step function h (using the same formula as the harness)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _subgradient_optimize(h0, dx, target, max_iters, lr0):
    """Projected subgradient descent on max_k ⟨h, (1−h) shifted⟩."""
    n = h0.shape[0]
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_of(h, dx)

    for t in range(1, max_iters + 1):
        # correlation (full) gives values for shifts -n+1 .. n-1
        corr = np.correlate(h, 1.0 - h, mode="full")
        C = corr * dx
        idx = int(np.argmax(C))
        k = idx - (n - 1)                      # integer shift producing the max

        # gradient of the worst shift (subgradient of the max)
        grad = np.full(n, dx)                  # start with dx * 1
        if k == 0:
            grad -= 2.0 * dx * h
        elif k > 0:
            grad[: n - k] -= dx * h[k:]
            grad[k:]      -= dx * h[: n - k]
        else:  # k < 0
            kk = -k
            grad[kk:]     -= dx * h[: n - kk]
            grad[: n - kk] -= dx * h[kk:]

        lr = lr0 / math.sqrt(t)
        h_new = h - lr * grad
        h = _project_bounds_sum(h_new, target)

        if t % 100 == 0:
            val = _c5_of(h, dx)
            if val < best_val:
                best_val, best_h = val, h.copy()
    # final check
    val = _c5_of(h, dx)
    if val < best_val:
        best_val, best_h = val, h.copy()
    return best_h, best_val
