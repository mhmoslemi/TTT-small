    # initialize random binary
    h = np.zeros(N)
    ones = np.random.choice(N, N//2, replace=False)
    h[ones] = 1
    cur = compute_c5(h)

    while time.time() - start < time_budget:
        # choose swap
        i = np.random.randint(N)
        while h[i]==0:
            i = np.random.randint(N)
        j = np.random.randint(N)
        while h[j]==1:
            j = np.random.randint(N)
        # perform swap
        h[i]=0; h[j]=1
        new_val = compute_c5(h)
        d = new_val - cur
        # acceptance condition
        elapsed = time.time() - start
        fraction = elapsed / time_budget
        T = T0 * (1 - fraction)  # linear cooling
        if d < 0 or np.random.rand() < np.exp(-d / max(T,1e-9)):
            cur = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i]=1; h[j]=0

import numpy as np
import time
import math
import random

def _c5_value(h):
    """Return the overlap constant C5 for binary vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate with mode='full' gives linear correlation (no wrap)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _initial_binary(n):
    """Create a balanced binary vector of length n (half 1s)."""
    h = np.zeros(n, dtype=float)
    idx = np.random.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def _interpolated_binary(base_h, n):
    """Upsample a base vector to length n and binarise (keep exactly n/2 ones)."""
    m = len(base_h)
    x_old = np.linspace(0, 2, m, endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    cont = np.interp(x_new, x_old, base_h)          # continuous values in [0,1]
    h = np.zeros(n, dtype=float)
    # keep the n/2 largest entries as 1
    top = np.argsort(-cont)[:n // 2]
    h[top] = 1.0
    return h

def _swap_one_zero(h):
    """Perform a random 1↔0 swap and return swapped indices."""
    n = h.shape[0]
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    # execute swap
    h[i] = 0.0
    h[j] = 1.0
    return i, j

def _undo_swap(h, i, j):
    """Revert a previous swap."""
    h[i] = 1.0
    h[j] = 0.0

def _targeted_move(h, cur_val):
    """
    Reduce the current maximal overlap by targeting the worst shift.
    Returns True if a move was accepted, False otherwise.
    """
    n = h.shape[0]
    # raw correlation (without scaling) – quicker for comparisons
    raw_corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(raw_corr.argmax())
    shift = max_idx - (n - 1)          # shift can be negative
    # determine mismatching positions that contribute to the max value
    if shift >= 0:
        rng = np.arange(0, n - shift)
        mism = rng[(h[rng] == 1) & (h[rng + shift] == 0)]
    else:
        s = -shift
        rng = np.arange(s, n)
        mism = rng[(h[rng] == 1) & (h[rng - s] == 0)]
    if mism.size == 0:                 # fall back to random swap
        i, j = _swap_one_zero(h)
        new_val = _c5_value(h)
        if new_val <= cur_val:
            return True, new_val, (i, j)
        _undo_swap(h, i, j)
        return False, cur_val, None

    i = np.random.choice(mism)         # a 1 that mismatches
    zeros = np.where(h == 0)[0]
    # avoid swapping with a zero that would create a new mismatch of the same shift
    j = np.random.choice(zeros)

    # apply swap
    h[i] = 0.0
    h[j] = 1.0
    new_val = _c5_value(h)
    if new_val <= cur_val:
        return True, new_val, (i, j)
    # reject
    _undo_swap(h, i, j)
    return False, cur_val, None

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a binary step function h on [0,2] with ∫h=1
    that minimizes max_k ∫ h(x)(1-h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # ------------------------------------------------------------------
    # Determine problem size
    base_h = globals().get('initial_h_values', None)
    if base_h is None:
        # fallback: 40-point constant 0.5 (will be turned binary later)
        base_h = np.full(40, 0.5, dtype=float)
    base_h = np.asarray(base_h, dtype=float)
    base_n = base_h.shape[0]

    # aim for a modestly larger grid (3× if small, otherwise keep size)
    if base_n < 100:
        n = base_n * 3
    else:
        n = base_n
    if n % 2 == 1:
        n += 1                     # need even length for exact half‑ones

    # ------------------------------------------------------------------
    # Initial solution: up‑sample provided vector and binarise
    h = _interpolated_binary(base_h, n)
    best_h = h.copy()
    best_val = _c5_value(h)

    # keep also the plain up‑sampled (non‑binary) constant solution as a bound
    # (it is never better than binary, but guarantees we have a feasible start)
    # ------------------------------------------------------------------
    # Search parameters
    T0 = 0.03                     # initial temperature for simulated annealing
    elapsed = lambda: time.time() - start_time
    time_limit = budget_s * 0.95   # leave a safety margin

    # Main loop – simulated annealing with occasional targeted moves
    while elapsed() < time_limit:
        # temperature drops linearly
        frac = elapsed() / time_limit
        T = T0 * (1.0 - frac)

        # decide move type
        if random.random() < 0.7:    # most steps are cheap random swaps

            new_val = _c5_value(h)
            delta = new_val - best_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
            # Metropolis acceptance (using current best as reference)
            if delta <= 0 or random.random() < math.exp(-delta / max(T, 1e-12)):
                # keep the change (h already updated)
                pass
            else:
                _undo_swap(h, i, j)   # reject
        else:
            # a more directed attempt to reduce the current worst shift
            accepted, new_val, swap_info = _targeted_move(h, best_val)
            if accepted:
                best_val = new_val
                best_h = h.copy()
            # no need to revert – _targeted_move already handled accept/reject

        # occasional full random restart if we appear stuck
        if random.random() < 0.001:
            h = _initial_binary(n)
            # Evaluate freshly – may be better than current best
            cur = _c5_value(h)
            if cur < best_val:
                best_val = cur
                best_h = h.copy()

    # ------------------------------------------------------------------
    # Ensure solution satisfies constraints (binary, correct sum)
    # (tiny numerical adjustments in the unlikely event of drift)
    half = n // 2
    ones = np.where(best_h == 1)[0]
    zeros = np.where(best_h == 0)[0]
    if len(ones) > half:
        # turn excess ones to zero (choose smallest indices)
        excess = len(ones) - half
        drop = np.random.choice(ones, excess, replace=False)
        best_h[drop] = 0.0
    elif len(ones) < half:
        deficit = half - len(ones)
        gain = np.random.choice(zeros, deficit, replace=False)
        best_h[gain] = 1.0

    final_c5 = _c5_value(best_h)

    # Return as plain Python list for compatibility
    return (best_h.tolist(), final_c5, n)
