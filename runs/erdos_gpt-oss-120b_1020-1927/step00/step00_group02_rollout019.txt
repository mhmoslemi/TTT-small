def project_box_sum(v, target, lo=0.0, hi=1.0):
    # Find t such that sum of clipped(v + t) == target
    lo_t = -np.max(v) - 1.0  # ensure v + t <= lo (?), simpler large bounds
    hi_t = 1.0 - np.min(v) + 1.0
    # Actually we can start with huge bounds.
    for _ in range(60):
        mid = (lo_t + hi_t) / 2.0
        clipped = np.clip(v + mid, lo, hi)
        s = clipped.sum()
        if s < target:
            lo_t = mid
        else:
            hi_t = mid
    return np.clip(v + (lo_t + hi_t) / 2.0, lo, hi)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random
    start = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_val = np.inf
    best_n = None

    # candidate resolutions
    n_candidates = [45, 90, 180]  # maybe also 360 if time permits

    for n in n_candidates:
        if time.time() - start > budget_s * 0.9:
            break
        dx = 2.0 / n
        # initialize
        if init is not None and len(init) == n:
            h = np.array(init, dtype=float)
        elif init is not None:
            # interpolate
            x_src = np.linspace(0, 2, len(init), endpoint=False)
            x_dst = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_dst, x_src, init)
        else:
            # random initial
            h = rng.uniform(0, 1, size=n)
            # project to sum target
        # ensure within [0,1] and sum constraint
        h = project_box_sum(h, target=n/2)

        # compute initial C5
        c = np.correlate(h, 1 - h, mode='full') * dx
        cur_val = np.max(c)

        if cur_val < best_val:
            best_val, best_h, best_n = cur_val, h.copy(), n

        # Optimization parameters
        max_iter = 5000
        alpha0 = 0.8  # step size constant
        for it in range(max_iter):
            if time.time() - start > budget_s * 0.98:
                break
            # compute correlation and find worst shift
            corr = np.correlate(h, 1 - h, mode='full')
            # Derive max index and shift
            k_max = np.argmax(corr)
            shift = k_max - (n - 1)
            # compute gradient
            indices = np.arange(n)
            grad = np.ones(n)
            forward_idx = indices + shift
            mask_fw = (forward_idx >= 0) & (forward_idx < n)
            grad[mask_fw] -= h[forward_idx[mask_fw]]
            backward_idx = indices - shift
            mask_bw = (backward_idx >= 0) & (backward_idx < n)
            grad[mask_bw] -= h[backward_idx[mask_bw]]
            # step size
            step = alpha0 / np.sqrt(it + 1)
            y = h - step * grad
            # project onto constraints
            h = project_box_sum(y, target=n/2)

            # evaluate
            corr = np.correlate(h, 1 - h, mode='full') * dx
            val = np.max(corr)

import time
import numpy as np

# ------------------------------------------------------------
def _project_box_sum(v: np.ndarray, target: float,
                     lo: float = 0.0, hi: float = 1.0,
                     tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """
    Project vector v onto the set {x | lo <= x_i <= hi, sum x_i = target}.
    The projection is performed by finding a scalar shift t such that
    sum(clip(v + t, lo, hi)) = target (monotone in t) via binary search.
    """
    lo_t, hi_t = -1e6, 1e6
    for _ in range(max_iter):
        mid = (lo_t + hi_t) * 0.5
        proj = np.clip(v + mid, lo, hi)
        s = proj.sum()
        if s < target:
            lo_t = mid
        else:
            hi_t = mid
        if hi_t - lo_t < tol:
            break
    return np.clip(v + (lo_t + hi_t) * 0.5, lo, hi)


def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a step function h sampled on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)


def _gradient_worst_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Gradient of the worst (maximum) overlap term with respect to h.
    For shift s,  d/dh[i]  sum h[i] (1 - h[i+s]) = 1 - h[i+s] - h[i-s] .
    Indices outside the interval are treated as zero.
    """
    n = h.size
    grad = np.ones(n, dtype=float)
    idx = np.arange(n)

    fwd = idx + shift
    mask_fwd = (fwd >= 0) & (fwd < n)
    grad[mask_fwd] -= h[fwd[mask_fwd]]

    bwd = idx - shift
    mask_bwd = (bwd >= 0) & (bwd < n)
    grad[mask_bwd] -= h[bwd[mask_bwd]]

    return grad


def _optimize_initial(h0: np.ndarray, max_iter: int, budget: float,
                      start_time: float) -> tuple:
    """
    Perform sub‑gradient descent from an initial feasible vector h0.
    Returns the best (h, C5) pair encountered.
    """
    n = h0.size
    target_sum = n / 2.0
    h = _project_box_sum(h0, target_sum)
    best_h = h.copy()
    best_val = _c5_value(h)

    alpha0 = 0.8  # initial step size scale
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        # time guard
        if time.time() - start_time > budget:
            break

        # compute raw correlation (no scaling needed for argmax)
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(corr_raw))
        shift = k_max - (n - 1)

        # gradient of the worst shift
        grad = _gradient_worst_shift(h, shift)

        # diminishing step
        step = alpha0 / np.sqrt(it)

        # projected update
        h_candidate = _project_box_sum(h - step * grad, target_sum)

        # evaluate
        val = _c5_value(h_candidate)

        # accept if improves (else keep current)
        if val < best_val - 1e-12:
            best_val, best_h = val, h_candidate.copy()
        # simple acceptance rule to keep descent direction
        if val <= _c5_value(h):
            h = h_candidate

        # occasional small random jitter to escape flat regions
        if it % 100 == 0:
            jitter = rng.normal(scale=0.01, size=n)
            h = _project_box_sum(h + jitter, target_sum)

    return best_h, best_val

    Optimize a step function h on [0,2] to obtain a low Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_n = None

    def _initial_for_n(n):

            return init_vals.copy()
        if init_vals is not None:
            # linear interpolation from the provided sample.
            src_x = np.linspace(0.0, 2.0, init_vals.size, endpoint=False)
            dst_x = np.linspace(0.0, 2.0, n, endpoint=False)
            return np.interp(dst_x, src_x, init_vals)
        # fallback: start from a smooth ramp (integrates to 1 automatically)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        return x / 2.0  # h(x)=x/2, average =1/n*∫x/2=0.5

    # Main optimisation loop over resolutions.
    for n in candidates:

        if time.time() - start_time > budget_s * 0.93:

        h_start = _initial_for_n(n)

        # Ensure feasibility.

        max_iter = max(2000, n * 12)

        best_h = np.full(best_n, 0.5, dtype=float)
        best_val = _c5_value(best_h)

    return best_h, best_val, int(best_n)
