def project_simplex_box(v, target):
    # binary search λ
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0.0, 1.0)

def fallback_solution(init, target_sum, n, remaining_budget):
    # init as binary choice: ensure half ones
    # convert to binary
    h_int = np.zeros(n, dtype=int)
    # set half entries to 1 sorted by init values descending
    sorted_idx = np.argsort(-init)  # descending
    h_int[sorted_idx[:int(target_sum)]] = 1

    best_h = h_int.astype(float).copy()
    best_val = np.correlate(best_h, 1 - best_h, mode='full').max()

    T = 0.02
    start = time.time()
    while time.time() - start < remaining_budget * 0.95:
        # pick random swap
        ones = np.flatnonzero(h_int)
        zeros = np.flatnonzero(1 - h_int)
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_candidate = h_int.copy()
        h_candidate[i] = 0
        h_candidate[j] = 1
        cand_val = np.correlate(h_candidate.astype(float), 1 - h_candidate.astype(float), mode='full').max()
        delta = cand_val - best_val
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_int = h_candidate
            if cand_val < best_val:
                best_val = cand_val
                best_h = h_candidate.astype(float).copy()
        T *= 0.9995
        if T < 1e-6:
            T = 0.02
    # final refinement with gradient descent maybe
    # but skip for simplicity
    c5 = best_val * 2.0 / n
    return best_h, c5, n

def solve_mip_cvx(initial_bin, n, target_sum):
    import cvxpy as cp, numpy as np
    h = cp.Variable(n, boolean=True)
    # Fix first element to break symmetry
    constraints = [cp.sum(h) == target_sum, h[0] == 1]
    if initial_bin is not None:
        # warm start
        h.value = initial_bin.astype(float)
    t = cp.Variable(nonneg=True)
    # for each shift s >=0
    for s in range(0, n):
        length = n - s
        if length == 0:
            continue
        w = cp.Variable(length, boolean=True)
        i = list(range(length))
        j = [i_k + s for i_k in i]
        # linearization constraints
        constraints += [w <= h[i]]
        constraints += [w <= h[j]]
        constraints += [w >= h[i] + h[j] - 1]
        # shift constraints
        constraints.append(cp.sum(h[i]) - cp.sum(w) <= t)
        if s > 0:
            constraints.append(cp.sum(h[j]) - cp.sum(w) <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.CBC, verbose=False)
    if h.value is None:
        raise RuntimeError('MIP did not solve')
    return h.value, t.value

def simulated_annealing(start_h, n, target_sum, time_budget):
    import numpy as np, time, math
    # ensure binary vector
    h_int = (start_h > 0.5).astype(int)
    # enforce exact sum
    cur_sum = h_int.sum()
    if cur_sum > target_sum:
        # turn some 1s to 0
        excess = int(cur_sum - target_sum)
        idx_one = np.flatnonzero(h_int)
        np.random.shuffle(idx_one)
        h_int[idx_one[:excess]] = 0
    elif cur_sum < target_sum:
        deficit = int(target_sum - cur_sum)
        idx_zero = np.flatnonzero(1 - h_int)
        np.random.shuffle(idx_zero)
        h_int[idx_zero[:deficit]] = 1
    # compute initial value
    corr = np.correlate(h_int.astype(float), 1 - h_int.astype(float), mode='full')
    best_val = corr.max()
    best_h = h_int.astype(float).copy()
    cur_val = best_val
    T = 0.02
    start = time.time()
    while time.time() - start < time_budget * 0.95:
        # random swap 1 and 0
        ones = np.flatnonzero(h_int)
        zeros = np.flatnonzero(1 - h_int)
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        h_int[i] = 0
        h_int[j] = 1
        # evaluate
        corr = np.correlate(h_int.astype(float), 1 - h_int.astype(float), mode='full')
        new_val = corr.max()
        delta = new_val - cur_val
        if delta < 0 or np.random.rand() < math.exp(-delta / T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_int.astype(float).copy()
        else:
            # revert swap
            h_int[i] = 1
            h_int[j] = 0
        # temperature schedule
        T *= 0.9997
        if T < 1e-6:
            T = 0.02
    return best_h, best_val

def refine_continuous(h_start, n, target_sum, time_budget):
    import numpy as np, time
    h = h_start.copy()
    dx = 2.0 / n
    start = time.time()
    best_h = h.copy()
    best_raw = np.correlate(h, 1 - h, mode='full').max()
    lr = 0.5
    alpha = 10.0
    iter = 0
    while time.time() - start < time_budget * 0.95:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        if max_corr < best_raw:
            best_raw = max_corr
            best_h = h.copy()
        # weights
        w_raw = np.exp(alpha * (corr - max_corr))
        w_norm = w_raw / w_raw.sum()
        grad = np.zeros_like(h)
        for idx_shift, s in enumerate(range(-n+1, n)):
            w_s = w_norm[idx_shift]
            if w_s == 0:
                continue
            if s >= 0:
                l = n - s
                grad[:l] += w_s * (1 - h[s:])
                grad[s:] -= w_s * h[:l]
            else:
                s2 = -s
                l = n - s2
                grad[s2:] += w_s * (1 - h[:l])
                grad[:l] -= w_s * h[s2:]

        h_new = h - step * grad
        # project onto sum and box
        h = project_onto_sum_box(h_new, target_sum)
        iter += 1
    return best_h, best_raw

import time, math, numpy as np

def _project_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {x | 0<=x<=1, sum(x)=target} by binary search on λ."""
    lo, hi = -2.0, 2.0
    for _ in range(40):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _max_corr_raw(h: np.ndarray) -> float:
    """Raw (no dx) max correlation of h with (1-h)."""
    return np.correlate(h, 1.0 - h, mode="full").max()


def _solve_mip(initial_bin: np.ndarray, n: int, target: float):
    """Mixed‑integer linear programming formulation using cvxpy + CBC."""
    import cvxpy as cp

    h = cp.Variable(n, boolean=True)
    t = cp.Variable(nonneg=True)
    cons = [cp.sum(h) == target, h[0] == 1]          # symmetry break
    if initial_bin is not None:
        h.value = initial_bin.astype(float)          # warm start

    # w_s variables for each non‑negative shift s
    for s in range(0, n):
        length = n - s
        if length == 0:
            continue
        w = cp.Variable(length, boolean=True)
        i = list(range(length))
        j = [k + s for k in i]

        # linearisation of w = h_i ∧ h_j
        cons += [w <= h[i], w <= h[j], w >= h[i] + h[j] - 1]

        # shift s constraint
        cons.append(cp.sum(h[i]) - cp.sum(w) <= t)
        # shift –s (if s>0)
        if s > 0:
            cons.append(cp.sum(h[j]) - cp.sum(w) <= t)

    prob = cp.Problem(cp.Minimize(t), cons)
    prob.solve(solver=cp.CBC, verbose=False)
    if h.value is None:
        raise RuntimeError("MIP did not return a solution")
    return np.array(h.value).flatten(), float(t.value)


def _simulated_annealing(start_h: np.ndarray, n: int, target: float, budget: float):
    """Binary hill‑climbing with occasional uphill moves."""
    h = (start_h > 0.5).astype(int)
    cur = h.sum()
    need = int(target)
    if cur > need:                                   # fix sum if needed
        excess = cur - need
        ones = np.flatnonzero(h)
        np.random.shuffle(ones)
        h[ones[:excess]] = 0
    elif cur < need:
        deficit = need - cur
        zeros = np.flatnonzero(1 - h)
        np.random.shuffle(zeros)
        h[zeros[:deficit]] = 1

    cur_val = _max_corr_raw(h.astype(float))
    best_h, best_val = h.astype(float).copy(), cur_val
    T = 0.02
    start = time.time()
    while time.time() - start < budget * 0.95:
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = 0, 1
        new_val = _max_corr_raw(h.astype(float))
        Δ = new_val - cur_val
        if Δ < 0 or np.random.rand() < math.exp(-Δ / T):
            cur_val = new_val
            if new_val < best_val:
                best_val, best_h = new_val, h.astype(float).copy()
        else:
            # revert
            h[i], h[j] = 1, 0
        T *= 0.9997
        if T < 1e-6:
            T = 0.02
    return best_h, best_val


def _refine_continuous(h_start: np.ndarray, n: int, target: float, budget: float):
    """Projected sub‑gradient descent on the smoothed max‑correlation."""
    h = h_start.copy()
    dx = 2.0 / n
    best_h, best_raw = h.copy(), _max_corr_raw(h)
    start = time.time()
    it = 0
    while time.time() - start < budget * 0.95:
        corr = np.correlate(h, 1.0 - h, mode="full")
        mx = corr.max()
        if mx < best_raw:
            best_raw, best_h = mx, h.copy()
        # smooth weights
        w = np.exp(10.0 * (corr - mx))
        w /= w.sum()
        grad = np.zeros_like(h)
        # apply weighted gradient for each shift
        for idx, s in enumerate(range(-(n - 1), n)):
            ws = w[idx]
            if ws == 0.0:
                continue
            if s >= 0:
                l = n - s
                grad[:l] += ws * (1.0 - h[s:])
                grad[s:] -= ws * h[:l]
            else:
                s2 = -s
                l = n - s2
                grad[s2:] += ws * (1.0 - h[:l])
                grad[:l] -= ws * h[s2:]
        # step size decays slowly
        step = 0.5 / (1.0 + it * 0.001)
        h = _project_sum_box(h - step * grad, target)
        it += 1
    return best_h, best_raw


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) with an improved C5 bound."""
    np.random.seed(seed)

    # ------------------------------------------------------------------

    if init_global is not None:

        n = 70

    # build a binary start (half ones)

    except Exception:

    c5 = best_raw * (2.0 / n)
    return best_h, float(c5), int(n)
