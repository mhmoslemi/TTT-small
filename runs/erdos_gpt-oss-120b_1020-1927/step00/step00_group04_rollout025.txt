def project_bounds_sum(x, target):
    low = np.min(x) - 1.0
    high = np.max(x)
    for _ in range(60):
        lam = (low + high) / 2.0
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(x - (low+high)/2.0, 0.0, 1.0)

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    # compute correlation using FFT
    g_rev = (1 - h)[::-1]
    size = 1 << ((2*N-1).bit_length())
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(g_rev, n=size)
    conv = np.fft.irfft(H * G, n=size)[:2*N-1]
    max_corr = np.max(conv)
    return max_corr * dx

import numpy as np
import time
import random
import math

def _project_bounds_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the box [0,1] and the affine hyperplane sum(x)=target."""
    # binary search for lambda such that y = clip(x - lambda,0,1) satisfies sum(y)=target
    low = np.min(x) - 1.0
    high = np.max(x)
    for _ in range(60):
        lam = (low + high) * 0.5
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(x - (low + high) * 0.5, 0.0, 1.0)


def _full_corr(h: np.ndarray) -> np.ndarray:
    """Linear cross‑correlation of h with (1‑h) using FFT."""
    N = h.size
    g_rev = (1.0 - h)[::-1]
    size = 1 << ((2 * N - 1).bit_length())
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(g_rev, n=size)
    conv = np.fft.irfft(H * G, n=size)[: 2 * N - 1]
    return conv


def _c5_value(h: np.ndarray) -> float:
    """Return the C₅ bound for a given step vector."""
    N = h.size
    dx = 2.0 / N
    corr = _full_corr(h)
    return np.max(corr) * dx


def _subgradient_optimize(h0: np.ndarray, max_time: float) -> tuple:
    """Projected sub‑gradient descent on the max‑overlap objective."""
    start = time.time()
    N = h0.size
    target_sum = N / 2.0
    dx = 2.0 / N

    h = _project_bounds_sum(np.clip(h0, 0.0, 1.0), target_sum)
    best_h = h.copy()
    best_val = _c5_value(best_h)

    it = 0
    while time.time() - start < max_time:
        it += 1
        # correlation and worst lag
        corr = _full_corr(h)
        max_idx = int(np.argmax(corr))
        lag = max_idx - (N - 1)          # lag can be negative

        # sub‑gradient of the worst correlation term
        grad = np.zeros(N)
        if lag >= 0:
            rng = np.arange(N - lag)
            grad[rng] += 1.0 - h[rng + lag]
            grad[rng + lag] -= h[rng]
        else:
            l = -lag
            rng = np.arange(l, N)
            grad[rng] += 1.0 - h[rng - l]
            grad[rng - l] -= h[rng]

        # step size (classic diminishing)
        step = 1.0 / math.sqrt(it)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h = _project_bounds_sum(h_new, target_sum)

        cur_val = np.max(corr) * dx
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            if best_val <= 0.3808:   # desired target reached
                break
    return best_h, best_val
