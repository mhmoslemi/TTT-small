import numpy as np
import time
import math

def compute_c5(arr, dx):
    """Maximum overlap integral for a given step function."""
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return np.max(corr) * dx

def random_candidate_binary(n, rng):
    """Generate a binary (or half‑binary) feasible vector with sum = n/2."""
    if n % 2 == 0:
        half = n // 2
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, size=half, replace=False)] = 1.0
        return h
    # odd n → one entry 0.5, the rest half 1 and half 0
    half = n // 2
    half_pos = rng.integers(0, n)
    remaining = np.delete(np.arange(n), half_pos)
    h = np.zeros(n, dtype=float)
    h[half_pos] = 0.5
    h[rng.choice(remaining, size=half, replace=False)] = 1.0
    return h

def adjust_sum_to_target(arr, target):
    """Tiny repair to enforce the exact integral constraint."""
    diff = target - arr.sum()
    if abs(diff) < 1e-12:
        return arr
    # try to fix with a single element
    for i in range(len(arr)):
        new = np.clip(arr[i] + diff, 0.0, 1.0)
        if abs(new - arr[i] - diff) < 1e-12:
            arr[i] = new
            return arr
    # fallback: distribute over a couple of entries
    if diff > 0:
