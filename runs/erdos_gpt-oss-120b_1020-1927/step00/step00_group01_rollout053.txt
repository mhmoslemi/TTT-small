def random_move(h, step):
    n = len(h)
    # pick i increase
    i = np.random.randint(n)
    # pick j decrease
    j = np.random.randint(n)
    if i == j: return None
    # ensure we can increase i and decrease j
    max_inc = min(step, 1.0 - h[i])
    max_dec = min(step, h[j])
    delta = min(max_inc, max_dec)
    if delta <= 0:
        return None
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h

import time, math, random
import numpy as np

def _c5_value(h):
    """Return the C5 value for a step function represented by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate computes ∑ h[i] * (1-h[i+k]) over overlapping indices
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _project_to_constraints(h, target_sum):
    """Clip to [0,1] and adjust to make exactly target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return h
    # distribute the difference uniformly over non‑boundary entries
    free = np.where((h > 0.0) & (h < 1.0))[0]
    if free.size == 0:
        # fallback – adjust the first element (it will be clipped again)
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
        return h
    per = diff / free.size
    # apply in small pieces to avoid crossing bounds
    for i in free:
        adj = np.clip(h[i] + per, 0.0, 1.0) - h[i]
        h[i] += adj
    # a second pass in case of tiny leftover
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            per = diff / free.size
            h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h

def _initial_from_parent(parent, n_target):
    """Upsample a parent vector to length n_target preserving shape."""
    n_parent = parent.shape[0]
    if n_parent == n_target:
        return parent.astype(float).copy()
    # linear interpolation on the interval [0,2)
    x_parent = np.linspace(0.0, 2.0, n_parent, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h = np.interp(x_new, x_parent, parent)
    # enforce constraints
    target = n_target / 2.0
    h = _project_to_constraints(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h:[0,2]→[0,1] with ∫h=1
    that minimises  C5 = max_k ∫ h(x)(1-h(x+k))dx .
    Returns (h_values, c5_bound, n_points)
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    budget = float(budget_s)

    # ------------------------------------------------------------------
    # Basic parameters
    # ------------------------------------------------------------------
    # Number of sample points – a compromise between flexibility and speed.
    N = 200                       # default resolution
    # If a parent construction exists, reuse its length as a starting point.
    if "initial_h_values" in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        N = max(N, parent.shape[0])
        h_current = _initial_from_parent(parent, N)
    else:
        # simple linear decreasing start (sum already equals N/2)
        h_current = np.linspace(1.0, 0.0, N, dtype=float)
        target = N / 2.0
        h_current = h_current * (target / h_current.sum())
        h_current = np.clip(h_current, 0.0, 1.0)

    target_sum = N / 2.0
    best_h = h_current.copy()
    best_val = _c5_value(best_h)

    # ------------------------------------------------------------------
    # Simulated‑annealing / hill‑climbing core
    # ------------------------------------------------------------------
    # Parameters governing the stochastic search.
    step_start = 0.02                 # maximal mass moved in one elementary swap
    step_decay = 0.99998              # geometric decay of the step size
    temp_start = 0.015                # initial temperature for Metropolis rule

    # Helper to create a random feasible vector (used for restarts).
    def random_feasible():
        h = np.random.rand(N)
        h = _project_to_constraints(h, target_sum)
        return h

    # Main optimisation loop – runs until the allocated budget is exhausted.
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget - 0.5:          # keep a small safety margin
            break

        # Occasionally restart from a fresh random point to escape local minima.
        if random.random() < 0.15:
            h = random_feasible()
        else:
            # Start from the current best with a tiny perturbation.
            h = best_h.copy()
            # introduce a few random pairwise adjustments
            for _ in range(3):
                i, j = random.sample(range(N), 2)
                max_inc = min(step_start, 1.0 - h[i])
                max_dec = min(step_start, h[j])
                max_delta = min(max_inc, max_dec)
                if max_delta <= 0:
                    continue
                delta = random.random() * max_delta
                h[i] += delta
                h[j] -= delta

        cur_val = _c5_value(h)

        # Local stochastic refinement for the current start point.
        step = step_start
        iter_local = 0
        while True:
            elapsed = time.time() - start_time
            if elapsed >= budget - 0.5:
                break

            # Linear temperature schedule
            T = temp_start * max(0.0, 1.0 - elapsed / budget)

            # Propose a swap of a small amount of mass between two positions.
            i, j = random.sample(range(N), 2)
            # direction i receives, j loses
            max_inc = min(step, 1.0 - h[i])
            max_dec = min(step, h[j])
            max_delta = min(max_inc, max_dec)
            if max_delta <= 0.0:
                # try opposite direction
                i, j = j, i
                max_inc = min(step, 1.0 - h[i])
                max_dec = min(step, h[j])
                max_delta = min(max_inc, max_dec)
                if max_delta <= 0.0:
                    step *= step_decay
                    continue
            delta = random.random() * max_delta
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta

            new_val = _c5_value(h_new)

            # Metropolis acceptance
            if new_val < cur_val or random.random() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
                h = h_new
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best_h = h_new.copy()

            # decay the step size slowly
            step *= step_decay
            if step < 1e-6:
                step = step_start

            iter_local += 1
            if iter_local % 2000 == 0:
                # occasional early exit from the inner loop if time is low
                if time.time() - start_time >= budget - 0.5:
                    break

    # Final safety projection (very minor numerical drift may happen)
    best_h = _project_to_constraints(best_h, target_sum)
    final_val = _c5_value(best_h)

    # Return as a plain list for maximum compatibility.
    return best_h.tolist(), float(final_val), N
