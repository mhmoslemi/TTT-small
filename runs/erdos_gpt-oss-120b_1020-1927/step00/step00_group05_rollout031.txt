def evaluate(h):
    # compute correlation via np.correlate
    c = np.correlate(h, 1 - h, mode='full')
    # dx factor
    dx = 2.0 / len(h)
    return float(np.max(c) * dx)

def shape_h(w, n):
    a = 1.0/(2.0 - w)
    xs = (np.arange(n) + 0.5) * (2.0 / n)   # sample at cell centers
    h = np.empty_like(xs)
    # left ramp: xs < w
    mask_left = xs < w
    h[mask_left] = a * xs[mask_left] / w
    # plateau: w <= xs <= 2 - w
    mask_center = (xs >= w) & (xs <= 2.0 - w)
    h[mask_center] = a
    # right ramp: xs > 2 - w
    mask_right = xs > 2.0 - w
    h[mask_right] = a * (2.0 - xs[mask_right]) / w
    return h
```

import time
import numpy as np

def _evaluate(h):
    """Return C5 for a discretised step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) (full mode => shifts from -(n-1) … n-1)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def _shift(arr, k):
    """Shift array by k positions with zero padding (no wrap-around)."""
    n = arr.size
    if k > 0:
        return np.concatenate((np.zeros(k), arr[:n - k]))
    if k < 0:
        kk = -k
        return np.concatenate((arr[kk:], np.zeros(kk)))
    return arr.copy()


def _proj_sum_box(v, target):
    """Project v onto {0<=x<=1, sum(x)=target} (Euclidean projection)."""
    # clip first to the box, then adjust the sum by water‑filling (binary search on λ)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            hi = lam
        else:
            lo = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _make_shape(w, n):
    """Symmetric piecewise‑linear shape with ramp length w (0<w≤1)."""
    a = 1.0 / (2.0 - w)               # plateau height, ensures ∫h=1
    xs = (np.arange(n) + 0.5) * (2.0 / n)   # mid‑points of intervals
    h = np.empty_like(xs)

    left = xs < w
    right = xs > (2.0 - w)
    middle = (~left) & (~right)

    # rising ramp
    h[left] = a * xs[left] / w
    # plateau
    h[middle] = a
    # falling ramp
    h[right] = a * (2.0 - xs[right]) / w
    return h


def _subgradient_step(h, step):
    """Perform one projected sub‑gradient step for the max‑overlap objective."""
    n = h.size
    # compute the worst shift (index of maximal correlation)
