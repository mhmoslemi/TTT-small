def compute_corrs(h):
    n = len(h)
    res = np.empty(2*n-1)
    for s in range(-n+1, n):
        # i from max(0, -s) to min(n, n-s) - 1
        i_start = max(0, -s)
        i_end = min(n, n - s) # exclusive
        # Equivalent overlap region length = i_end - i_start
        if i_end > i_start:
            # compute sum of h[i] * (1 - h[i+s])
            product = h[i_start:i_end] * (1 - h[i_start+s:i_end+s])
            res[s + n - 1] = product.sum()
        else:
            res[s + n - 1] = 0.0
    return res

def compute_corrs_and_best(h):
    n = len(h)
    # compute correlations (f_k)
    res = np.empty(2*n-1)
    # we can compute full array via loops
    for s in range(-n+1, n):
        i_start = max(0, -s)
        i_end = min(n, n - s)
        if i_start < i_end:
            prod = h[i_start:i_end] * (1 - h[i_start + s:i_end + s])
            res[s + n - 1] = prod.sum()
        else:
            res[s + n - 1] = 0.0
    # find max
    idx = np.argmax(res)
    max_val = res[idx]
    shift = idx - (n - 1)
    return max_val, shift, res

def compute_corrs_and_best(h):
    """
    Return the maximum value of  sum_i h_i * (1-h_{i+k})
    and the shift k (integer, -n+1 .. n-1) attaining it.
    """
    n = h.shape[0]
    max_val = -1.0
    best_shift = 0
    for s in range(-n + 1, n):
        if s >= 0:
            a = h[: n - s]
            b = 1.0 - h[s:]
        else:
            k = -s
            a = h[k:]
            b = 1.0 - h[: n - k]
        val = np.dot(a, b)
        if val > max_val:
            max_val = val
            best_shift = s
    return max_val, best_shift


def evaluate_c5(h):
    """C5 value for a step function h sampled on a uniform grid."""
    max_val, _ = compute_corrs_and_best(h)
    return max_val * (2.0 / h.shape[0])


def gradient_at_shift(h, shift):
    """
    Subgradient of the worst shift term f_k(h)=sum_i h_i(1-h_{i+k})
    evaluated at the shift giving the maximum.
    """
    n = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        s = shift
        grad[: n - s] += (1.0 - h[s:])
        grad[s:] -= h[: n - s]
    else:
        s = -shift
        grad[s:] += (1.0 - h[: n - s])
        grad[: n - s] -= h[s:]
    return grad


def project_onto_box_sum(v, target_sum, lower=0.0, upper=1.0,
                         tol=1e-9, max_iter=60):

        step = step0 * (0.97 ** i)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_box_sum(h_new, target_sum)

        new_max, _ = compute_corrs_and_best(h_new)
        new_score = new_max * (2.0 / n)

        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()

        # accept if improvement, otherwise keep current (with tiny chance to jump)
        if new_score < cur_score or rng.random() < 0.01:
            h = h_new
        else:
            # occasional random shake to escape flat regions
            if rng.random() < 0.02:
                h = h + rng.normal(scale=0.05, size=n)
                h = np.clip(h, 0.0, 1.0)
                h = project_onto_box_sum(h, target_sum)

    return best_h, best_score


# ---------- Main entry point ----------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ---- initial vector -------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)  # provided by harness
    except NameError:
        # fallback: simple half‑filled binary vector (odd length gets a 0.5)
        n0 = 45
        h0 = np.zeros(n0, dtype=np.float64)
        ones = rng.choice(n0, n0 // 2, replace=False)
        h0[ones] = 1.0
        if n0 % 2 == 1:
            remaining = list(set(range(n0)) - set(ones))
            h0[rng.choice(remaining)] = 0.5

    n = h0.shape[0]
    target_sum = n / 2.0
    h0 = project_onto_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_score = evaluate_c5(best_h)

    start_time = time.time()

    # ---- first long local optimisation -----------------------------------
    time_for_first = budget_s * 0.45
    h_opt, sc_opt = local_optimize(best_h, max_iters=800,
                                   rng=rng, target_sum=target_sum,
                                   time_limit=time_for_first)
    if sc_opt < best_score:
        best_h, best_score = h_opt, sc_opt

    # ---- random‑restart phase --------------------------------------------
    while time.time() - start_time < budget_s * 0.85:
        # generate a new start: either perturb the current best or draw fresh
        if rng.random() < 0.35:
            start_vec = best_h + rng.normal(scale=0.2, size=n)
        else:
            start_vec = rng.random(n)  # uniform in [0,1)
        start_vec = np.clip(start_vec, 0.0, 1.0)
        start_vec = project_onto_box_sum(start_vec, target_sum)

        # short optimisation
        limit = max(0.0, budget_s - (time.time() - start_time) - 5.0)
        iters = 300
        h_opt, sc_opt = local_optimize(start_vec, max_iters=iters,
                                       rng=rng, target_sum=target_sum,
                                       time_limit=limit)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt

    # ---- optional up‑sampling for finer search ---------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 12.0:
        n2 = n * 2
        # linear interpolation onto the finer grid
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        xs2 = np.linspace(0.0, 2.0, n2, endpoint=False)
        h2 = np.interp(xs2, xs, best_h)
        target_sum2 = n2 / 2.0
        h2 = np.clip(h2, 0.0, 1.0)
        h2 = project_onto_box_sum(h2, target_sum2)

        h_opt, sc_opt = local_optimize(h2, max_iters=800,
                                       rng=rng, target_sum=target_sum2,
                                       time_limit=remaining)
        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt
            n = n2

    final_c5 = evaluate_c5(best_h)
    return best_h, float(final_c5), int(best_h.shape[0])
