            h0 = h0 * (target_sum / cur)
        h0 = _project_to_constraints(h0, target_sum)

        # run hill climbing within allotted slice of time
        time_left = budget_s - (time.time() - start)
        end_time = time.time() + time_left * 0.9  # keep margin
        h_opt, sc_opt = _hill_climb(h0, rng, end_time)

        if sc_opt < best_score:
            best_score, best_h, best_n = sc_opt, h_opt.copy(), n

    # final safety: if nothing succeeded, fall back to initial if present
    if best_h is None:
        if base is not None:
            best_h = base.copy()
            best_score = _score(best_h)
            best_n = base_n
        else:
            # simple uniform 0.5 solution
            best_n = 50
            best_h = np.full(best_n, 0.5)
            best_score = _score(best_h)

    return best_h, float(best_score), int(best_n)
