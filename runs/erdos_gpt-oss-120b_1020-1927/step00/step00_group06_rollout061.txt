    cur_score = compute_c5(h)

    target = N / 2.0

            continue

            src, dst = i, j
        else:
            src, dst = j, i
        max_move = min(h[src], 1.0 - h[dst])
        if max_move <= 1e-12:

        h_new[dst] += delta
        score = compute_c5(h_new)
        it += 1
        if score < cur_score or np.random.rand() < np.exp(-(score - cur_score) / max(T, 1e-9)):

import time
import numpy as np

def _project_to_constraints(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto [0,1]^n intersect {sum = target} using a simple bisection."""
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s_mid = np.clip(v + mid, 0.0, 1.0).sum()
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a given h (full correlation, zero padding)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _linear_plateau(N: int, w: float) -> np.ndarray:
    """Symmetric linear rise – flat plateau – linear fall shape."""
    dx = 2.0 / N
    x = (np.arange(N) + 0.5) * dx
    s = (2.0 - w) / 2.0                     # slope length on each side
    a = 2.0 / (2.0 + w)                     # height on the plateau
    h = np.zeros(N, dtype=np.float64)

    d = np.abs(x - 1.0)

    # plateau
    in_plateau = d <= w * 0.5
    h[in_plateau] = a

    # slopes
    if s > 0:
        in_slope = (~in_plateau) & (d < w * 0.5 + s)
        h[in_slope] = a * (1.0 - (d[in_slope] - w * 0.5) / s)

    # enforce constraints exactly
    return _project_to_constraints(h, N / 2.0)


def _quadratic(N: int, r: float) -> np.ndarray:
    """Symmetric quadratic bump: h = a * (1-(|x-1|/r)^2) on |x-1|<=r."""
    dx = 2.0 / N
    x = (np.arange(N) + 0.5) * dx
    a = 3.0 / (4.0 * r)
    d = np.abs(x - 1.0)
    h = np.where(d <= r, a * (1.0 - (d / r) ** 2), 0.0)
    return _project_to_constraints(h, N / 2.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function giving a low C5 bound.

    Returns
    -------
    h_values : np.ndarray
        The function values on a uniform grid of size ``n_points``.
    c5_bound : float
        The achieved max overlap integral.
    n_points : int
        Length of the returned vector.
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Determine problem size and start from a reasonable seed vector.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h_cur = np.asarray(initial_h_values, dtype=np.float64).copy()
        N = h_cur.shape[0]
        # enforce exact linear constraint (might be slightly off)
        h_cur = _project_to_constraints(h_cur, N / 2.0)
    else:
        N = 150                               # reasonable default
        h_cur = _linear_plateau(N, 0.0)       # pure triangle

    best_h = h_cur.copy()
    best_score = _c5_value(best_h)

    # ------------------------------------------------------------------
    # 2. Simple parametric families (plateau & quadratic) – quick sweep.
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.15:           # reserve a bit of time for this phase
        # linear plateau family
        for w in np.linspace(0.0, 2.0, 201):
            h = _linear_plateau(N, w)
            sc = _c5_value(h)
            if sc < best_score:
                best_score, best_h = sc, h.copy()
        # quadratic family
        for r in np.linspace(0.75, 1.0, 101):
            h = _quadratic(N, r)
            sc = _c5_value(h)
            if sc < best_score:
                best_score, best_h = sc, h.copy()

    # ------------------------------------------------------------------
    # 3. Stochastic hill‑climbing / simulated‑annealing on the best vector.
    # ------------------------------------------------------------------
    def _still_time(remaining):
        return time.time() - start_time < remaining

    remaining_budget = budget_s - (time.time() - start_time)
    if remaining_budget > 5.0:              # need at least a couple of seconds
        h = best_h.copy()
        cur_score = best_score
        target_sum = N / 2.0

        # annealing schedule
        T = 0.025
        decay = 0.99995
        it = 0

        while _still_time(budget_s * 0.95):
            # propose a move that preserves the total sum exactly
            i, j = np.random.randint(N, size=2)
            if i == j:
                continue
            # random direction of transfer
            if np.random.rand() < 0.5:
                src, dst = i, j
            else:
                src, dst = j, i
            max_delta = min(h[src], 1.0 - h[dst])
            if max_delta <= 1e-12:
                continue
            delta = np.random.rand() * max_delta
            h_new = h.copy()
            h_new[src] -= delta
            h_new[dst] += delta
            new_score = _c5_value(h_new)

            # accept if better or with Metropolis probability
            accept = False
            if new_score < cur_score:
                accept = True
            else:
                prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
                if np.random.rand() < prob:
                    accept = True

            if accept:
                h, cur_score = h_new, new_score
                if new_score < best_score:
                    best_score, best_h = new_score, h_new.copy()

            # cooling
            T *= decay
            it += 1

            # occasional diversification
            if it % 8000 == 0:
                if np.random.rand() < 0.3:
                    # perturb the current best a bit and re‑project
                    h = best_h.copy()
                    noise = np.random.uniform(-0.08, 0.08, size=N)
                    h = np.clip(h + noise, 0.0, 1.0)
                    h = _project_to_constraints(h, target_sum)
                    cur_score = _c5_value(h)

    # final safety projection (ensures exact sum)
    best_h = _project_to_constraints(best_h, N / 2.0)
    best_score = _c5_value(best_h)

    return best_h, float(best_score), int(N)
