import numpy as np, time

def _max_overlap(h: np.ndarray) -> float:
    """C5 value for a binary step function h on [0,2] with step size 2/len(h)."""
    n = h.size
    # full correlation of h with (1-h)
    conv = np.correlate(h, 1 - h, mode="full")
    return conv.max() * (2.0 / n)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # ---- choose number of points -------------------------------------------------
    try:
        init_vec = np.array(initial_h_values, dtype=float)
        n0 = init_vec.size
    except Exception:          # no parent construction supplied
        init_vec = None
        n0 = None

    N = 200                     # even number, modest size for speed/precision
    if N % 2:
        N += 1

    # ---- build initial binary vector --------------------------------------------
    if init_vec is not None and n0:
        # up‑sample the parent solution to the chosen resolution
        factor = N // n0 + 1
        h = np.repeat(init_vec, factor)[:N].copy()
        h = (h > 0.5).astype(float)          # force to {0,1}
        # enforce exact half‑density
        cur = int(h.sum())
        target = N // 2
        if cur > target:
            idx = np.where(h == 1)[0]
            h[np.random.choice(idx, cur - target, replace=False)] = 0
        elif cur < target:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, target - cur, replace=False)] = 1
    else:
        h = np.zeros(N)
        h[np.random.choice(N, N // 2, replace=False)] = 1.0

    best_h = h.copy()
    best_val = _max_overlap(best_h)

    # ---- optimisation parameters -------------------------------------------------
    time_limit = budget_s - 0.3          # leave a small safety margin
    stagnation = 0
    stagnation_limit = 4000
    restart_interval = 8000
    it = 0

    # ---- main loop ---------------------------------------------------------------
    while time.time() - start_time < time_limit:
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = conv.argmax()
        cur_val = conv[max_idx] * (2.0 / N)

        if cur_val < best_val - 1e-12:
