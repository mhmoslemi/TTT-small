def project_to_capped_simplex(v, S, tol=1e-9, max_iter=100):
    # Projects onto {w: sum w = S, 0 <= w <= 1}
    # uses Lagrange multiplier binary search.
    lo = np.min(v) - 1.0  # when shift results w=1 for all -> sum=n
    hi = np.max(v)        # when shift -> w=0 for all -> sum=0
    # Adjust bounds
    def sum_w(theta):
        w = v - theta
        # Using np.clip for speed - but we want sum only
        # we could avoid allocation but ok
        w = np.clip(w, 0.0, 1.0)
        return np.sum(w)
    # guarantee sum(lo) >= S, sum(hi) <= S
    # Usually lo low enough but just ensure
    while sum_w(lo) < S:
        lo -= (abs(lo) + 1.0)
    while sum_w(hi) > S:
        hi += (abs(hi) + 1.0)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = sum_w(mid)
        if s > S:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    w = np.clip(v - theta, 0.0, 1.0)
    return w

import time
import numpy as np
from scipy.special import logsumexp

def _evaluate(h: np.ndarray) -> float:
    """Return C5 for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_capped_simplex(v: np.ndarray, target_sum: float, tol: float = 1e-9) -> np.ndarray:
    """
    Euclidean projection of `v` onto the set
        { w | sum(w) = target_sum, 0 <= w_i <= 1 }.
    Uses a simple bisection on the Lagrange multiplier.
    """
    # bounds for theta: when theta is very negative -> all w_i = 1 (sum = n)
    # when theta is very positive -> all w_i = 0 (sum = 0)
    lo = np.min(v) - 1.0  # guarantees sum >= target_sum
    hi = np.max(v)        # guarantees sum <= target_sum

    def _s(theta):
        w = v - theta
        np.clip(w, 0.0, 1.0, out=w)
        return w.sum()

    # enlarge bounds if needed
    while _s(lo) < target_sum:
        lo -= max(1.0, abs(lo))
    while _s(hi) > target_sum:
        hi += max(1.0, abs(hi))

    for _ in range(40):          # sufficient for double precision
        mid = (lo + hi) * 0.5
        sm = _s(mid)
        if sm > target_sum:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
    w = v - theta
    np.clip(w, 0.0, 1.0, out=w)
    # final tiny correction to hit the sum within tolerance
    diff = target_sum - w.sum()
    if abs(diff) > tol:
        # distribute the tiny remainder uniformly among non‑saturated entries
        free = (w > 0.0) & (w < 1.0)
        if free.any():
            w[free] += diff / free.sum()
            np.clip(w, 0.0, 1.0, out=w)
    return w


def _smooth_gradient(h: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        g(h) = (1/alpha) * logsumexp(alpha * corr(h))
    where corr = np.correlate(h, 1‑h, mode='full').
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    # weights for each shift (sum to 1)
    w = np.exp(alpha * corr - logsumexp(alpha * corr))

    grad = np.ones_like(h)
    # shift range: -(n-1) .. (n-1)
    for shift in range(-(n - 1), n):
        weight = w[shift + n - 1]
        if weight == 0.0:
            continue
        if shift >= 0:
            end = n - shift
            grad[:end] -= weight * h[shift:]
            grad[shift:] -= weight * h[:end]
        else:
            s = -shift
            end = n - s
            grad[s:] -= weight * h[:end]
            grad[:end] -= weight * h[s:]
    return grad


def _max_shift_gradient(h: np.ndarray) -> (np.ndarray, int):
    """
    Sub‑gradient of the max( corr ) objective.
    Returns the gradient and the shift index producing the maximum.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)

    grad = np.ones_like(h)
    if shift >= 0:
        end = n - shift
        grad[:end] -= h[shift:]
        grad[shift:] -= h[:end]
    else:
        s = -shift
        end = n - s
        grad[s:] -= h[:end]
        grad[:end] -= h[s:]
    return grad, idx


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function with a low C5 value.

    Returns
    -------
    h_values : np.ndarray
        Optimised function values on a uniform grid over [0,2].
    c5_bound : float
        The resulting C5 (max overlap integral).
    n_points : int
        Number of discretisation points used.
    """
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------ #
    # 1) Choose grid size and build an initial candidate.
    # ------------------------------------------------------------------ #
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
        # aim for ~180 points (multiple of the parent size)
        factor = max(2, int(180 // n0))
        n = n0 * factor
        # linear interpolation onto the finer grid
        x_parent = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_fine = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_fine, x_parent, h0)
    else:
        n = 200
        h = np.full(n, 0.5)

    target_sum = n / 2.0
    h = _project_capped_simplex(h, target_sum)

    best_h = h.copy()
    best_score = _evaluate(best_h)
    cur_h = h.copy()
    cur_score = best_score

    # ------------------------------------------------------------------ #
    # 2) Simulated annealing phase (big portion of the time budget)
    # ------------------------------------------------------------------ #
    max_T = 0.12
    min_T = 1e-4
    max_step = 0.22
    min_step = 5e-4
    max_iter_without_improve = 40000
    iter_no_imp = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.90:
            break
        prog = elapsed / (budget_s * 0.90)
        T = max_T * (1.0 - prog) + min_T * prog
        step = max_step * (1.0 - prog) + min_step * prog

        # propose a simple two‑index swap with a small delta
        i, j = np.random.randint(0, n, size=2)
        while j == i:
            j = np.random.randint(0, n)
        delta = (np.random.rand() * 2.0 - 1.0) * step
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_h = _project_capped_simplex(new_h, target_sum)

        new_score = _evaluate(new_h)

        # Metropolis acceptance
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            cur_h = new_h
            cur_score = new_score
            iter_no_imp = 0 if new_score < cur_score else iter_no_imp + 1
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
                iter_no_imp = 0
        else:
            iter_no_imp += 1

        # occasional restart if stuck for long
        if iter_no_imp >= max_iter_without_improve:
            # random feasible start
            rnd = np.random.rand(n)
            rnd = rnd / rnd.sum() * target_sum
            cur_h = _project_capped_simplex(rnd, target_sum)
            cur_score = _evaluate(cur_h)
            iter_no_imp = 0

    # ------------------------------------------------------------------ #
    # 3) Local refinement using a smooth approximation of the max.
    # ------------------------------------------------------------------ #
    remaining = budget_s - (time.time() - start_time)
    if remaining > 2.0:
        # number of gradient steps limited by remaining time
        steps = int(min(5000, remaining * 20))
        alpha = 2.0          # start with a modest smoothing
        for step_id in range(steps):
            # gradually increase smoothness
            alpha = min(30.0, alpha * 1.015)

            grad = _smooth_gradient(best_h, alpha)
            eta = 0.5 * (1.0 - step_id / steps)   # decreasing step size
            cand = best_h - eta * grad
            cand = _project_capped_simplex(cand, target_sum)
            cand_score = _evaluate(cand)
            if cand_score < best_score:
                best_h = cand
                best_score = cand_score

    # ------------------------------------------------------------------ #
    # 4) Final (optional) projected sub‑gradient descent on the true max.
    # ------------------------------------------------------------------ #
    remaining = budget_s - (time.time() - start_time)
    if remaining > 1.0:
        extra_steps = int(min(2000, remaining * 10))
        for t in range(extra_steps):
            grad, _ = _max_shift_gradient(best_h)
            eta = 0.4 * (1.0 - t / extra_steps)
            cand = best_h - eta * grad
            cand = _project_capped_simplex(cand, target_sum)
            cand_score = _evaluate(cand)
            if cand_score < best_score:
                best_h = cand
                best_score = cand_score

    # ------------------------------------------------------------------ #
    # 5) Return the best solution discovered.
    # ------------------------------------------------------------------ #
    final_score = _evaluate(best_h)
    return (best_h, final_score, int(n))
