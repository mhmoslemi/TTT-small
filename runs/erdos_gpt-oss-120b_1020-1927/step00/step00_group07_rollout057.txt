import numpy as np
import time

def project_onto_simplex_box(x, target):
    # find lambda so that sum(clipping) = target
    lo, hi = -1.0, 1.0
    for _ in range(40):
        lam = (lo+hi)/2
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(x - (lo+hi)/2, 0.0, 1.0)
    return proj

def compute_rawmax(h):
    # h is np.ndarray shape (N,)
    C = np.correlate(h, 1 - h, mode='full')
    return C.max()

def smooth_max_and_grad(h, tau=0.1):
    N = h.size
    C = np.correlate(h, 1 - h, mode='full')
    Cmax = C.max()
    # compute stable softmax weights
    shifted = (C - Cmax) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    w = exp_shifted / sum_exp  # weights sum to 1
    # compute gradient
    grad = np.ones(N)  # start from 1 (since sum_k w_k = 1)
    # handle each shift
    offsets = np.arange(-N+1, N)
    # loop
    for idx, wk in enumerate(w):
        if wk == 0:
            continue
        k = offsets[idx]
        if k > 0:
            # j from 0 to N-k-1: subtract wk * h[j+k]
            grad[:N - k] -= wk * h[k:]
            # j from k to N-1: subtract wk * h[j-k]
            grad[k:] -= wk * h[:N - k]
        elif k < 0:
            kk = -k
            # j from 0 to N-kk-1: subtract wk * h[j+(-kk)]? Actually h_{j+k} where k negative => j - kk
            # For j >= kk: j-k = j+kk?
            # Subtractions:
            # term - wk * h_{j+k} (i.e., h_{j-kk}) for j >= kk
            grad[kk:] -= wk * h[:N - kk]
            # term - wk * h_{j-k} (i.e., h_{j+kk}) for j <= N-kk-1
            grad[:N - kk] -= wk * h[kk:]
        else:  # k == 0
            grad -= 2 * wk * h
    smooth = Cmax + tau * np.log(sum_exp)

import numpy as np
import time
import random

# ------------------------------------------------------------
# Helper functions (top level, no closures)
# ------------------------------------------------------------

def _project_onto_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the feasible set {0<=h_i<=1, sum(h)=target}
    by binary searching a Lagrange multiplier."""
    lo, hi = -1.0, 1.0
    # 50 iterations are enough for double precision
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def _raw_max(h: np.ndarray) -> float:
    """Maximum of the overlap integral (unnormalised, i.e. without dx)."""
    # correlation of h with (1-h) using zero padding (full mode)
    return np.max(np.correlate(h, 1.0 - h, mode='full'))


def _smooth_max_and_grad(h: np.ndarray, tau: float):
    """Smooth max (log‑sum‑exp) of the overlap vector and its gradient."""
    N = h.size
    # raw overlap vector (length 2N‑1)
    C = np.correlate(h, 1.0 - h, mode='full')
    Cmax = C.max()
    # stable log‑sum‑exp
    shifted = (C - Cmax) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    w = exp_shifted / sum_exp            # weights, sum to 1
    smooth = Cmax + tau * np.log(sum_exp)

    # gradient: for each shift k we add the contribution
    grad = np.zeros(N, dtype=np.float64)
    offsets = np.arange(-N + 1, N)      # corresponds to indices of C / w
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        k = offsets[idx]

        if k == 0:
            # (1 - 2*h) for all positions
            grad += wk * (1.0 - 2.0 * h)
            continue

        if k > 0:
            # (1 - h_{j+k}) for j = 0 .. N-1-k
            grad[: N - k] += wk * (1.0 - h[k:])
            # -h_{j-k} for j = k .. N-1
            grad[k:] -= wk * h[: N - k]
        else:               # k < 0
            kk = -k
            # (1 - h_{j+k}) for j = kk .. N-1  (j+k = j-kk)
            grad[kk:] += wk * (1.0 - h[: N - kk])
            # -h_{j-k} for j = 0 .. N-1-kk  (j-k = j+kk)
            grad[: N - kk] -= wk * h[kk:]

    return smooth, grad, C

    """
    Optimise a step function h on [0,2] (discretised) to minimise
    C5 = max_k ∫ h(x) (1-h(x+k)) dx.

    Returns:
        (h_values, c5_bound, n_points)
    """
    random.seed(seed)
    np.random.seed(seed)

    # --------------------------------------------------------
    # problem size and basic parameters
    # --------------------------------------------------------
    try:
        N = int(initial_h_values.shape[0])          # use provided construction if any

    except Exception:
        N = 80                                      # fallback size
        h0 = np.full(N, 0.5, dtype=np.float64)

    target_sum = N / 2.0
    dx = 2.0 / N

    start_time = time.time()
    time_limit = budget_s * 0.95     # keep a tiny margin for finalisation

    # --------------------------------------------------------
    # initialise feasible point (project onto box & sum)
    # --------------------------------------------------------
    h = _project_onto_box_sum(h0, target_sum)

    # best solution tracking (continuous)
    best_h = h.copy()
    best_raw = _raw_max(best_h)

    # --------------------------------------------------------
    # stage 1: smooth‑max gradient descent (≈ half the budget)
    # --------------------------------------------------------
    tau = 0.08               # smoothing parameter (start relatively large)
    step = 0.5               # learning rate
    while time.time() - start_time < time_limit * 0.5:
        smooth, grad, _ = _smooth_max_and_grad(h, tau)

        # simple back‑tracking line search
        for _ in range(10):
            cand = h - step * grad
            cand = _project_onto_box_sum(cand, target_sum)
            cand_raw = _raw_max(cand)
            if cand_raw <= best_raw:
                h = cand
                best_raw = cand_raw
                best_h = cand.copy()
                # slowly tighten the smoothing once we improve

            step *= 0.7
        else:

            cand = _project_onto_box_sum(best_h + perturb, target_sum)
            h = cand

    # --------------------------------------------------------
    # stage 2: binary hill‑climbing on the rounded version
    # --------------------------------------------------------
    # convert to binary while preserving the exact sum
    order = np.argsort(-best_h)                     # biggest values become 1
    binary = np.zeros(N, dtype=np.float64)
    binary[order[:int(target_sum)]] = 1.0

    while time.time() - start_time < time_limit:

        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]

            break
        i = np.random.choice(ones)

            cur_raw = new_raw

                # early exit if we already beat the target bound

                cur_raw = best_bin_raw

    # --------------------------------------------------------
    # choose the best of the two stages
    # --------------------------------------------------------
    if best_bin_raw < best_raw:
        final_h = best_bin_h
        final_raw = best_bin_raw
    else:
        final_h = best_h
        final_raw = best_raw

    c5_bound = final_raw * dx
    return final_h, float(c5_bound), int(N)
