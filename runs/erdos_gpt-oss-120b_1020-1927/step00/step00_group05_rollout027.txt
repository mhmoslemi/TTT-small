import numpy as np
import time
import random

# ------------------------------------------------------------
# Helper functions (top level, no closures)
# ------------------------------------------------------------

def construct_h_from_lengths(w, g, x):
    """
    Build a binary step function on the grid points x.
    w : lengths of the intervals (sum to 1)
    g : lengths of the gaps   (sum to 1)
    x : coordinate of the grid points (midpoints)
    Returns an array of 0/1 values.
    """
    h = np.zeros_like(x, dtype=float)
    pos = 0.0
    for wi, gi in zip(w, g):
        pos += gi                # gap
        start = pos
        pos += wi                # interval
        end = pos
        mask = (x >= start) & (x < end)
        h[mask] = 1.0
    return h


def compute_c5(h):
    """
    Compute the Erdős overlap value for a vector h.
    """
    n = h.shape[0]
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full").max() * dx


def dirichlet_fixed_sum(k, total, mindelta=0.0):
    """
    Sample a point from the (k‑dimensional) simplex with each coordinate
    at least mindelta and total sum = total.
    """
    if mindelta == 0.0:
        v = np.random.exponential(1.0, size=k)
        return total * v / v.sum()
    else:
        # allocate the mandatory part
        remainder = total - k * mindelta
        if remainder < 0:
            raise ValueError("mindelta too large")
        v = np.random.exponential(1.0, size=k)
        v = remainder * v / v.sum()
        return v + mindelta


def adjust_to_target(h, target):
    """
    Adjust a binary (or almost binary) vector so that its sum equals target.
    The adjustment is performed by adding/subtracting the needed amount
    from any non‑boundary entries; the function never breaks the 0‑1 bounds.
    """
    diff = target - h.sum()
    # work with a mutable copy
    h = h.copy()
    while abs(diff) > 1e-12:
        if diff > 0:          # need to increase some entry
            idx = np.where(h < 1.0 - 1e-12)[0]
            if idx.size == 0:
                break
            i = random.choice(idx)
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:                 # need to decrease some entry
            idx = np.where(h > 1e-12)[0]
            if idx.size == 0:
                break
            i = random.choice(idx)
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    return h


def random_interval_candidate(k, n, x, mindelta=0.02):
    """
    Produce a binary candidate based on k intervals and k gaps,
    each interval/gap having a minimum length `mindelta`.
    """
    # total length of intervals = 1, total length of gaps = 1
    w = dirichlet_fixed_sum(k, 1.0, mindelta)
    g = dirichlet_fixed_sum(k, 1.0, mindelta)
    h = construct_h_from_lengths(w, g, x)
    # enforce the exact integral constraint
    target = n / 2.0
    h = adjust_to_target(h, target)
    return h, w, g


def hill_climb_binary(h, time_limit):
    """
    Simple hill‑climbing on a binary vector: repeatedly try random
    swaps of a 1 and a 0 and keep the swap if it improves the objective.
    Stops when the allocated time expires.
    """
    n = h.shape[0]
    best_h = h.copy()
    best_c = compute_c5(best_h)
    start = time.perf_counter()
    while time.perf_counter() - start < time_limit:
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = random.choice(ones)
        j = random.choice(zeros)
        trial = best_h.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        c = compute_c5(trial)
        if c < best_c - 1e-12:
            best_h = trial
            best_c = c
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where
    h_values is a numpy array defining the step function,
    c5_bound is the achieved upper bound,
    n_points is the discretisation size.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ----------------------------------------------------------------
    # Determine problem size and optional starting point
    # ----------------------------------------------------------------
    if "initial_h_values" in globals():
        h_start = np.asarray(initial_h_values, dtype=float)
        n = h_start.shape[0]
        # enforce exact integral constraint on the supplied start
        target = n / 2.0
        h_start = adjust_to_target(h_start, target)
        best_h = h_start.copy()
        best_c = compute_c5(best_h)
    else:
        # default size – small enough to stay fast but fine‑grained
        n = 200
        best_h = None
        best_c = np.inf

    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx               # mid‑points of the cells
    target_sum = n / 2.0

    # ----------------------------------------------------------------
    # Random search over interval patterns
    # ----------------------------------------------------------------
    k_intervals = 5                               # C5 corresponds to 5 blocks
    start_time = time.perf_counter()
    # allocate ~60 % of the budget to the stochastic sampling phase
    random_phase_limit = budget_s * 0.60
    while time.perf_counter() - start_time < random_phase_limit:
        cand_h, _, _ = random_interval_candidate(k_intervals, n, x, mindelta=0.02)
        c_val = compute_c5(cand_h)
        if c_val < best_c:
            best_c = c_val
            best_h = cand_h.copy()

    # ----------------------------------------------------------------
    # Optional local refinement: binary hill climbing on the best vector
    # ----------------------------------------------------------------
    if best_h is not None:
        # make sure we are dealing with a strict 0/1 vector for swaps
        best_bin = (best_h >= 0.5).astype(float)
        best_bin = adjust_to_target(best_bin, target_sum)
        remaining_time = budget_s - (time.perf_counter() - start_time)
        # use at most 35 % of the total budget for this phase
        hc_time = min(remaining_time * 0.35, remaining_time - 0.01)
        best_bin, best_c = hill_climb_binary(best_bin, hc_time)
        best_h = best_bin

    final_c = compute_c5(best_h)

    return (best_h, final_c, n)
