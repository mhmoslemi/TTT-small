def project_boxed_simplex(y, S):
    # Input y array, target sum S; output x in [0,1] with sum S nearest to y.
    # Use water-filling.
    n = len(y)
    # Clip to [0,1] after shifting.
    # We need to solve for λ s.t. sum_i min(max(y_i - λ, 0), 1) = S
    # This is monotonic decreasing in λ.
    lambda_low = -np.max(y) - 1  # ensure all y_i - λ > 1 => after clipping sum = n
    lambda_high = np.max(y) + 1  # ensure all y_i - λ < 0 => sum =0
    for _ in range(50):
        lam = (lambda_low + lambda_high) / 2.0
        x = np.clip(y - lam, 0, 1)
        s = x.sum()
        if s > S:
            lambda_low = lam
        else:
            lambda_high = lam
    x = np.clip(y - (lambda_low + lambda_high) / 2.0, 0, 1)
    # due to numeric maybe the sum not exactly S; adjust small residual
    residual = S - x.sum()
    # if residual >0, add to entries not at 1 equally; else subtract from entries not at 0
    if abs(residual) > 1e-10:
        # find indices where x not at bound:
        if residual > 0:
            free = np.where(x < 1 - 1e-12)[0]
            if len(free) > 0:
                add = residual / len(free)
                x[free] = np.clip(x[free] + add, 0, 1)
        else:
            free = np.where(x > 1e-12)[0]
            if len(free) > 0:
                sub = (-residual) / len(free)
                x[free] = np.clip(x[free] - sub, 0, 1)
    # final adjust small residual.
    # Might still not exactly sum; but fine within tolerance.
    return x

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, shift):
    n = len(h)
    G = np.zeros_like(h)
    if shift >= 0:
        k = shift
        if k < n:
            G[:n - k] += 1 - h[k:]
            G[k:] -= h[:n - k]
    else:
        k = -shift
        if k < n:
            G[k:] += 1 - h[:n - k]
            G[:n - k] -= h[k:]
    return G

import numpy as np, time

def project_to_simplex_box(v, S):
    n = v.shape[0]
    lo = -np.max(v) - 2
    hi = np.max(v) + 2
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v - (lo + hi)/2.0, 0, 1)
    diff = S - proj.sum()
    if abs(diff) > 1e-10:
        free = np.where((proj > 1e-12) & (proj < 1 - 1e-12))[0]
        if free.size:
            adj = diff / free.size
            proj[free] = np.clip(proj[free] + adj, 0, 1)
    return proj

def max_corr_and_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return corr[idx], shift, corr

def gradient_for_shift(h, k):
    n = h.shape[0]
    G = np.zeros_like(h)
    if k >= 0:
        m = k
        if m < n:
            G[:n - m] += 1 - h[m:]
            G[m:] -= h[:n - m]
    else:
        m = -k
        if m < n:
            G[m:] += 1 - h[:n - m]
            G[:n - m] -= h[m:]
    return G

def evaluate_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def improve(h, S, time_deadline):
    n = h.shape[0]
    best_h = h.copy()
    best_val = evaluate_c5(h)
    alpha = 0.5
    no_improve = 0
    while time.time() < time_deadline:
        corr_val, shift, _ = max_corr_and_shift(h)
        G = gradient_for_shift(h, shift)
        step = alpha * G
        h_try = h - step
        h_try = project_to_simplex_box(h_try, S)
        new_val = evaluate_c5(h_try)
        if new_val < best_val - 1e-8:
            best_val = new_val
            best_h = h_try.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_try
        if no_improve >= 200:
            alpha *= 0.7
            no_improve = 0
        if best_val < 0.3809:
            break
    return best_h, best_val

import numpy as np
import time

def _project_to_simplex_box(v, target):
    """Project vector v onto {0<=x_i<=1, sum x_i = target}."""
    n = v.shape[0]
    lo = -np.max(v) - 2.0   # ensures all clipped to 1 → sum=n
    hi = np.max(v) + 2.0    # ensures all clipped to 0 → sum=0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size:
            adj = diff / free.size
            x[free] = np.clip(x[free] + adj, 0.0, 1.0)
    return x

def _corr_max_shift(h):
    """Return max correlation value, the corresponding shift, and the full vector."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return corr[idx], shift, corr

def _grad_shift(h, k):
    """Gradient of correlation at shift k (integer) with respect to h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if k >= 0:
        m = k
        if m < n:
            g[:n - m] += 1.0 - h[m:]
            g[m:] -= h[:n - m]
    else:
        m = -k
        if m < n:
            g[m:] += 1.0 - h[:n - m]
            g[:n - m] -= h[m:]
    return g

def _c5_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _improve(h, target_sum, deadline):
    """Gradient descent on the worst shift, respecting constraints."""
    n = h.shape[0]
    best_h = h.copy()
    best_val = _c5_value(h)
    cur_val = best_val
    alpha = 0.5
    no_imp = 0
    while time.time() < deadline:
        _, shift, _ = _corr_max_shift(h)
        g = _grad_shift(h, shift)
        step = alpha * g
        cand = h - step
        cand = _project_to_simplex_box(cand, target_sum)
        new_val = _c5_value(cand)
        if new_val < cur_val - 1e-12:
            h = cand
            cur_val = new_val
            no_imp = 0
            if new_val < best_val:
                best_val = new_val
                best_h = cand.copy()
        else:
            no_imp += 1
            alpha *= 0.7
            if alpha < 1e-4:
                alpha = 0.5
        if best_val < 0.38085:
            break
        if no_imp >= 200:
            # random small perturbation to escape possible stagnation
            pert = best_h + 0.02 * (np.random.rand(n) - 0.5)
            pert = np.clip(pert, 0.0, 1.0)
            h = _project_to_simplex_box(pert, target_sum)
            cur_val = _c5_value(h)
            no_imp = 0
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) respecting the problem constraints."""
    np.random.seed(seed)
    # initialise
    init = globals().get('initial_h_values')
    if init is not None:
        h0 = np.asarray(init, dtype=np.float64).copy()
        n = h0.size
    else:
        n = 70
        h0 = np.random.rand(n)
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _project_to_simplex_box(h0, target_sum)

    start = time.time()
    deadline = start + budget_s - 0.2   # small safety margin

    # final clean‑up and return

    final_c5 = _c5_value(best_h)
