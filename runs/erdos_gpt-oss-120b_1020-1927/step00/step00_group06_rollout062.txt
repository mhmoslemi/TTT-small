import time, numpy as np

def compute_c5_vals(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    N = 200
    # ensure even
    if N % 2 != 0:
        N += 1
    # initialize
    h = np.zeros(N, dtype=np.int8)
    ones = rng.choice(N, size=N//2, replace=False)
    h[ones] = 1
    cur_c5 = compute_c5_vals(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # parameters
    T = 0.5
    T_min = 1e-4
    alpha = 0.9995  # cooling factor
    iteration = 0
    while time.time() - start_time < budget_s:
        # propose swap
        # selected indices: i in ones, j not in ones
        # but we need dynamic ones list; we can store indices of ones and zeroes.
        idx_one = rng.choice(np.flatnonzero(h == 1))
        idx_zero = rng.choice(np.flatnonzero(h == 0))
        # swap
        h[idx_one], h[idx_zero] = h[idx_zero], h[idx_one]
        new_c5 = compute_c5_vals(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # accept
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert swap
            h[idx_one], h[idx_zero] = h[idx_zero], h[idx_one]
        # cooling
        T = max(T_min, T * alpha)
        iteration += 1
        # occasional random restart if stuck
        if iteration % 5000 == 0:
            # If not improved significantly in past period, restart
            # just random new vector
            h = np.zeros(N, dtype=np.int8)
            ones = rng.choice(N, size=N//2, replace=False)
            h[ones] = 1
            cur_c5 = compute_c5_vals(h)
    # return best
    return best_h.astype(float), best_c5, N

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    # Setup N
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        N = h.size
    else:
        N = 200
        if N % 2 != 0:
            N += 1
        h = np.zeros(N, dtype=float)
        h[rng.choice(N, size=N//2, replace=False)] = 1.0
    target_sum = N / 2.0
    # Ensure constraints by projection
    h = project_onto_hypercube_sum(h, target_sum)
    dx = 2.0 / N
    # Evaluate initial
    corr = np.correlate(h, 1 - h, mode='full')
    cur_f = np.max(corr) * dx
    best_h = h.copy()
    best_f = cur_f
    # Parameters
    T = 0.5
    T_min = 1e-6
    temp_decay = 0.9995
    max_step = 0.5
    step_decay = 0.9995
    iteration = 0
    last_improve_iter = 0
    # Main loop
    while time.time() - start < budget_s - 0.2:  # margin
        # compute correlation and worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        cur_f = np.max(corr) * dx
        if cur_f < best_f - 1e-12:
            best_f = cur_f
            best_h = h.copy()
            last_improve_iter = iteration
        # identify shift with max overlap
        max_idx = np.argmax(corr)
        s = max_idx - (N - 1)  # shift value
        # compute gradient of f_s
        g = np.zeros(N, dtype=float)
        if s >= 0:
            n = N - s
            # add +1 for i in [0,n)
            g[:n] += 1.0
            # subtract h[i+s]
            g[:n] -= h[s:]
            # subtract h[i-s] for i >= s
            g[s:] -= h[:n]
        else:
            t = -s
            n = N - t
            g[t:] += 1.0
            g[t:] -= h[:n]
            g[:n] -= h[t:]
        # step size
        step = max_step * (step_decay ** iteration)
        # propose update
        new_h = h - step * g
        # project onto feasible set
        new_h = project_onto_hypercube_sum(new_h, target_sum)
        # evaluate new

        new_f = np.max(new_corr) * dx

                T = 0.5

import time
import numpy as np

def _project_onto_hypercube_sum(v, target, lower=0.0, upper=1.0, max_iter=60):
    """Projection of v onto the set {x in [lower,upper]^n : sum(x)=target}."""
    # binary search for lambda such that sum(clip(v+lam,lower,upper)) = target
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, lower, upper)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    x = np.clip(v + (lo + hi) * 0.5, lower, upper)
    # Tiny correction to meet the sum more accurately
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        if free.any():
            adjust = diff / free.sum()
            x[free] += adjust
            x = np.clip(x, lower, upper)
        else:
            # fallback: adjust a single coordinate
            idx = np.argmin(np.abs(x - lower))
            x[idx] = np.clip(x[idx] + diff, lower, upper)
    return x

def _max_overlap_and_shift(h, dx):
    """Return (max_overlap, shift) where max_overlap = max_k ∫h(1‑h(x+k))."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    return corr[max_idx] * dx, max_idx - (len(h) - 1)

def _gradient_of_shift(h, shift):
    """Gradient ∇_h f_shift where f_shift = ∫h(1‑h(x+shift))."""
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        m = n - shift
        g[:m] += 1.0
        g[:m] -= h[shift:]          # -h_{i+shift}
        g[shift:] -= h[:m]          # -h_{i-shift}
    else:
        t = -shift
        m = n - t
        g[t:] += 1.0
        g[t:] -= h[:m]              # -h_{i-t}
        g[:m] -= h[t:]              # -h_{i+t}
    return g

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # --------------------------------------------------------------------- #
    # 1. Initialise h ------------------------------------------------------ #
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 200                     # default size (even)
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    target_sum = n / 2.0
    h = _project_onto_hypercube_sum(h, target_sum)

    dx = 2.0 / n
    cur_val, _ = _max_overlap_and_shift(h, dx)
    best_h = h.copy()
    best_val = cur_val

    # --------------------------------------------------------------------- #
    # 2. Parameters -------------------------------------------------------- #
    T = 0.5                         # initial temperature
    T_min = 1e-6
    temp_decay = 0.9995
    max_step = 0.8                  # initial step magnitude
    step_decay = 0.9995
    it = 0
    last_improve = 0
    stagnation_limit = 6000        # iterations without improvement before restart
    explore_prob = 0.15            # probability of a random swap move

    # --------------------------------------------------------------------- #
    # 3. Main optimisation loop -------------------------------------------- #
    while time.time() - start_time < budget_s - 0.2:
        # current objective and worst shift
        cur_val, worst_shift = _max_overlap_and_shift(h, dx)

        # record best
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
            last_improve = it

        # ----------------------------------------------------------------- #
        # Choose a move ----------------------------------------------------- #
        if rng.random() < explore_prob:
            # simple random swap (preserves sum exactly)
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h < 0.5)
            if ones.size and zeros.size:
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h_new = h.copy()
                h_new[i], h_new[j] = h_new[j], h_new[i]
            else:
                # fallback to gradient step if degenerate
                grad = _gradient_of_shift(h, worst_shift)
                step = max_step * (step_decay ** it)
                h_new = _project_onto_hypercube_sum(h - step * grad, target_sum)
        else:
            # gradient‑based move on the worst shift
            grad = _gradient_of_shift(h, worst_shift)
            step = max_step * (step_decay ** it)
            h_new = _project_onto_hypercube_sum(h - step * grad, target_sum)

        # ----------------------------------------------------------------- #
        # Acceptance test ---------------------------------------------------- #
        new_val, _ = _max_overlap_and_shift(h_new, dx)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            cur_val = new_val

        # ----------------------------------------------------------------- #
        # Temperature / step update ------------------------------------------ #
        T = max(T_min, T * temp_decay)
        it += 1

        # ----------------------------------------------------------------- #
        # Restart on stagnation --------------------------------------------- #
        if it - last_improve > stagnation_limit:
            # random fresh start, keep best so far
            h = np.zeros(n, dtype=float)
            h[rng.choice(n, size=n // 2, replace=False)] = 1.0
            h = _project_onto_hypercube_sum(h, target_sum)
            T = 0.5
            last_improve = it

    # --------------------------------------------------------------------- #
    # 4. Return result ------------------------------------------------------ #
    # Ensure final vector satisfies constraints (tiny numerical drift)
    best_h = _project_onto_hypercube_sum(best_h, target_sum)
    return best_h.tolist(), best_val, n
