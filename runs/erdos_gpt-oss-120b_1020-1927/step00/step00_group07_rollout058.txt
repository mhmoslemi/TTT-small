        corr = np.correlate(h, 1 - h, mode='full')

import numpy as np, time, math
from scipy.optimize import minimize

# global placeholders modified inside run()
_OPT_N = None          # total number of points
_OPT_HALF = None       # half length (n//2)
_OPT_DX = None         # step size 2/n
_OPT_T = None          # temperature for smooth max

def project_antisym(h):
    """Project vector onto the subspace h[i]+h[n-1-i]=1 (and 0<=h<=1)."""
    h = h.copy()
    n = h.shape[0]
    for i in range(n // 2):
        j = n - 1 - i
        avg = (h[i] + (1.0 - h[j])) * 0.5
        h[i] = avg
        h[j] = 1.0 - avg
    if n % 2 == 1:
        h[n // 2] = 0.5
    return h

def construct_h(g):
    """Build full h from the free half‑vector g using the antisymmetry."""
    n = _OPT_N
    half = _OPT_HALF
    h = np.empty(n, dtype=float)
    h[:half] = g
    h[half:] = 1.0 - g[::-1]
    return h

def compute_c5(h):
    """Exact C₅ value for a given discretisation."""
    dx = _OPT_DX
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def _objective_and_grad(g):
    """Smooth maximum (log‑sum‑exp) of the correlation together with its gradient."""
    n = _OPT_N
    half = _OPT_HALF
    dx = _OPT_DX
    T = _OPT_T

    h = construct_h(g)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    diff = corr - max_corr                     # ≤0
    exp_T = np.exp(T * diff)                   # stable
    sum_exp = exp_t = exp_T.sum()
    smooth = max_corr + np.log(sum_exp) / T    # log‑sum‑exp, scaled
    probs = exp_T / sum_exp

    grad_h = np.zeros_like(h)
    # shifts from -(n-1) … (n-1)
    for idx, k in enumerate(range(-(n - 1), n)):
        p = probs[idx]
        if p == 0.0:
            continue
        if k >= 0:
            grad_h[:n - k] += p * (1.0 - h[k:])
            grad_h[k:] -= p * h[:n - k]
        else:
            s = -k
            grad_h[s:] += p * (1.0 - h[:n - s])
            grad_h[:n - s] -= p * h[s:]

    # map to gradient w.r.t. g (h[i] = g[i], h[n-1-i] = 1-g[i])
    grad_g = grad_h[:half] - grad_h[n-1:n-1-half:-1]
    return smooth * dx, grad_g * dx

def random_pair_refinement(h, max_iter, delta, time_limit, start_time):
    """Simple local search: move δ from a random 1‑entry to a random 0‑entry."""
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(best_h)
    for _ in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        max_down = best_h[i]               # how much we can subtract
        max_up = 1.0 - best_h[j]           # how much we can add
        step = delta
        if step > max_down or step > max_up:
            step = min(max_down, max_up)
        if step <= 0.0:
            continue
        cand = best_h.copy()
        cand[i] -= step
        cand[j] += step
        cand = project_antisym(cand)
        val = compute_c5(cand)
        if val < best_val:
            best_h, best_val = cand, val
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h minimizing the Erdős C₅ overlap constant."""
    global _OPT_N, _OPT_HALF, _OPT_DX, _OPT_T
    np.random.seed(seed)
    start = time.time()

    # ---- problem size -------------------------------------------------
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n = int(init_h.shape[0])
    else:
        init_h = None
        n = 80                     # default small size
    if n % 2 != 0:
        n += 1                     # enforce even length for symmetry
    half = n // 2
    dx = 2.0 / n
    _OPT_N, _OPT_HALF, _OPT_DX = n, half, dx

    # ---- generate starting vectors ------------------------------------
    starters = []
    if init_h is not None:
        h0 = project_antisym(init_h)
        starters.append(h0[:half].copy())
    for _ in range(4):
        starters.append(np.random.rand(half))

    best_h, best_score = None, np.inf
    T_schedule = [4.0, 12.0, 36.0]          # increasing sharpness

    for g0 in starters:
        if time.time() - start > budget_s * 0.90:
            break
        g = g0.copy()
        for T in T_schedule:
            _OPT_T = T
            bounds = [(0.0, 1.0)] * half
            res = minimize(_objective_and_grad, g, method='L-BFGS-B',
                           jac=True, bounds=bounds,
                           options={'maxiter': 200,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            g = np.clip(res.x, 0.0, 1.0)
            if time.time() - start > budget_s * 0.93:
                break

        h_candidate = construct_h(g)
        score = compute_c5(h_candidate)
        if score < best_score:
            best_h, best_score = h_candidate, score
        if best_score < 0.38079:            # reached target early

    if best_h is None:

        best_h = construct_h(g)
        best_score = compute_c5(best_h)

    # ---- final local refinement ----------------------------------------
    # small random pair moves respecting the constraints
    best_h, best_score = random_pair_refinement(

    best_h = project_antisym(np.clip(best_h, 0.0, 1.0))
    best_score = compute_c5(best_h)

    return (best_h.tolist(), float(best_score), n)
