import numpy as np
import time
import math

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral using the discrete definition."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate uses zero padding; matches the reference evaluator.
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with integral 1 that gives a small
    max_{k} ∫ h(x)(1-h(x+k))dx  (the Erdős C5 constant).

    Returns
    -------
    (h_values, c5_bound, n_points)
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # Initialise a binary candidate.
    # If a parent construction exists, turn its largest entries into a
    # binary vector of the correct weight.
    # ----------------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        init = np.array(init, copy=True)
        n = int(init.shape[0])
        half = n // 2
        # take the `half` largest entries as ones
        idx = np.argpartition(init, -half)[-half:]
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0
    else:
        # default modest size (fast to evaluate)
        n = 70
        half = n // 2
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, size=half, replace=False)] = 1.0

    # ----------------------------------------------------------------------
    # Helper containers for the current state.
    # ----------------------------------------------------------------------
    best_h = h.copy()
    best_c = _c5_value(best_h)

    cur_h = h.copy()
    cur_c = best_c

    # maintain explicit index pools for ones / zeros
    ones = list(np.where(cur_h > 0.5)[0])
    zeros = list(np.where(cur_h <= 0.5)[0])

    # ----------------------------------------------------------------------
    # Simulated annealing configuration.
    # ----------------------------------------------------------------------
    T = 0.2                     # initial temperature
    decay = 0.9995              # exponential cooling
    target = 0.3808             # desired bound
    stagnation = 0
    iter_cnt = 0

    # ----------------------------------------------------------------------
    # Main optimisation loop.
    # ----------------------------------------------------------------------
    while True:
        # time check
        if time.time() - start_time > budget_s - 0.5:
            break

        iter_cnt += 1

        # random swap proposal
        i_one = rng.choice(ones)
        i_zero = rng.choice(zeros)

        cand = cur_h.copy()
        cand[i_one] = 0.0
        cand[i_zero] = 1.0
        cand_c = _c5_value(cand)

        delta = cand_c - cur_c
        accept = False
        if delta <= 0:
            accept = True
        else:
            prob = math.exp(-delta / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_h = cand
            cur_c = cand_c
            # update index pools
            ones.remove(i_one)
            zeros.append(i_one)
            zeros.remove(i_zero)
            ones.append(i_zero)

            # record best solution
            if cand_c < best_c:
                best_c = cand_c
                best_h = cand.copy()
                stagnation = 0
                if best_c <= target:
                    break
            else:
                stagnation += 1
        else:
            stagnation += 1

        # temperature schedule
        T *= decay

        # occasional random shake‑up if we have been stuck too long
        if stagnation > 20000:
            for _ in range(5):
                i_one = rng.choice(ones)
                i_zero = rng.choice(zeros)
                cur_h[i_one] = 0.0
                cur_h[i_zero] = 1.0
                ones.remove(i_one)
                zeros.append(i_one)
                zeros.remove(i_zero)
                ones.append(i_zero)
            cur_c = _c5_value(cur_h)
            stagnation = 0

    # ----------------------------------------------------------------------
    # Final greedy refinement: try every single swap that improves the score.
    # ----------------------------------------------------------------------
    refinement_deadline = start_time + budget_s - 0.5
    while time.time() < refinement_deadline:
        improved = False
        cur_ones = [i for i in range(n) if best_h[i] > 0.5]

                break
            for i_zero in cur_zeros:
                cand = best_h.copy()
                cand[i_one] = 0.0
                cand[i_zero] = 1.0
                cand_c = _c5_value(cand)

    # ----------------------------------------------------------------------

    except Exception:
