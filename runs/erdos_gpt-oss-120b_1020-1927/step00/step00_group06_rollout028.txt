import time

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Construct a linear ramp h(x) = x/2 on [0,2] sampled uniformly.
    This satisfies the required constraints and yields C₅ = 1/3 + O(1/n²) ≈ 0.333.
    """

    # Determine number of sample points.
