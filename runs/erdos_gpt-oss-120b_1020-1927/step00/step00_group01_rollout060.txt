def run(seed=42,budget_s=1100,**kwargs):
    import time, numpy as np, random
    random.seed(seed)
    np.random.seed(seed)
    start=time.time()
    best_h=None
    best_c5=1e9
    best_n=0
    # choose candidate ns
    candidate_ns = [60,80,100,120,140,160]
    for n in candidate_ns:
        if time.time()-start > budget_s*0.95:
            break
        # ensure even n
        if n%2!=0:
            n+=1
        S = n//2
        # create initial binary pattern (random)
        # multiple restarts
        for restart in range(5):
            if time.time()-start > budget_s*0.95:
                break
            # random binary vector with exactly S ones
            h = np.zeros(n, dtype=int)
            ones = np.random.choice(n, S, replace=False)
            h[ones] = 1
            cur_c5 = compute_c5(h.astype(float))
            # Simulated annealing parameters
            T0 = 0.2
            Tf = 0.001
            max_iter = 5000
            for it in range(max_iter):
                if time.time()-start > budget_s*0.95:
                    break
                # compute temperature schedule (exponential)
                t = it / max_iter
                T = T0 * (Tf/T0)**t
                # propose swap
                i = np.random.randint(n)
                while h[i]==0:
                    i = np.random.randint(n)
                j = np.random.randint(n)
                while h[j]==1:
                    j = np.random.randint(n)
                # compute new vector
                h_new = h.copy()
                h_new[i]=0
                h_new[j]=1
                new_c5 = compute_c5(h_new.astype(float))
                delta = new_c5 - cur_c5
                if delta <= 0 or random.random() < np.exp(-delta / T):
                    h = h_new
                    cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.astype(float)
                    best_n = n
                # optional early exit if cur_c5 already low
                if best_c5 < 0.3808:
                    break
            # break restart if good enough
            if best_c5 < 0.3808:
                break
        if best_c5 < 0.3808:
            break
    # If we haven't found binary solution, try fractional improvement using gradient descent
    if best_h is not None and best_c5 > 0.3808:
        # gradient descent refinement on real values
        # use subgradient method described earlier
        best_h = refine_fractional(best_h,start,budget_s)
        best_c5 = compute_c5(best_h)
    # Ensure constraints
    # Project to [0,1] and sum constraint
    best_h = np.clip(best_h,0,1)
    # adjust sum using projection onto bounded simplex
    best_h = project_onto_bounds_sum(best_h, n/2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)

def subgrad(h):
    n=len(h)
    corr = np.correlate(h,1-h,mode='full')
    max_val = np.max(corr)
    # pick all indices where corr==max
    max_idxs = np.flatnonzero(corr==max_val)
    grad = np.zeros(n)
    for idx in max_idxs:
        s = idx - (n-1)
        # contributions for each j
        # first: (1 - h_{j+s}) if 0 <= j+s < n
        # second: -h_{j-s} if 0 <= j-s < n
        # So we can vectorize:
        # compute term1 = shift array (1 - h shifted by -s)
        # term2 = shift array -h shifted by +s
        # Use slices
        if s >= 0:
            # term1 for j in 0..n-1-s: (1 - h[j+s])
            grad[:n-s] += 1 - h[s:]   # j from 0 to n-s-1, j+s from s..n-1
            # term2 for j in s..n-1: -h[j-s]
            grad[s:] += -h[:n-s]
        else:
            # s negative, let t = -s (>0)
            t = -s
            # term1 for j in t..n-1: (1 - h[j-s]) where j-s >=0
            grad[t:] += 1 - h[:n-t]
            # term2 for j in 0..n-t-1: -h[j+s] where j+s < n
            grad[:n-t] += -h[t:]
        # This for each max shift added
    # scale by dx
    dx = 2.0 / n
    grad = grad * dx
    return grad

def project_onto_bounds_sum(v, S):
    n = len(v)
    # if sum(v) already equals S within tolerance and all bounds satisfied, return v.
    # else binary search. Use 100 iterations.
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Due to rounding may not achieve exactly S; adjust a bit if necessary.
    # We can correct by distributing residual to some coordinates not at bounds.
    diff = S - x.sum()
    if abs(diff) > 1e-8:
        # adjust some indices not at bounds
        free = (x > 0) & (x < 1)
        # compute amount to distribute equally among free
        if free.any():
            adjust = diff / free.sum()
            x[free] += adjust
            # Clip again (should stay within bounds)
            x = np.clip(x, 0.0, 1.0)
        # else can't adjust
    return x

    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)

import numpy as np, time, random

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return (2.0 / n) * np.max(corr)

def subgrad(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    idxs = np.where(np.abs(corr - max_val) < 1e-12)[0]
    grad = np.zeros(n)
    for idx in idxs:
        s = idx - (n - 1)
        if s >= 0:
            grad[:n-s] += 1 - h[s:]
            grad[s:] += -h[:n - s]
        else:
            t = -s
            grad[t:] += 1 - h[:n - t]
            grad[:n - t] += -h[t:]
    grad = grad * (2.0 / n)  # dx factor
    return grad

def project_onto_bounds_sum(v, S):
    n = len(v)
    # Clip to bounds afterwards
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # Adjust small residual
    diff = S - x.sum()
    if abs(diff) > 1e-9:
        # distribute to non-bound entries
        free = (x > 0) & (x < 1)
        if free.any():
            adjust = diff / free.sum()
            x[free] += adjust
            x = np.clip(x, 0.0, 1.0)
    return x

def gradient_descent_continuous(h0, budget_s, start_time):
    n = len(h0)
    S = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    it = 0
    max_iter = 20000
    while it < max_iter and time.time() - start_time < budget_s * 0.7:
        g = subgrad(h)
        # step size schedule
        step = 0.5 / np.sqrt(it + 1)
        h_new = h - step * g
        h_new = project_onto_bounds_sum(h_new, S)
        val = compute_c5(h_new)
        if val < best_val - 1e-8:

            best_h = h_new.copy()
            # optionally reset iteration count? Not needed
        h = h_new
        it += 1
        # early stop if below threshold
        if best_val < 0.3808:
            break
    return best_h, best_val

def binary_hill_climb(n, budget_s, start_time):
    S = n // 2
    best_h = None
    best_val = float('inf')
    # start with multiple restarts
    for restart in range(10):
        # random binary
        h = np.zeros(n, dtype=int)
        h[np.random.choice(n, S, replace=False)] = 1
        cur_val = compute_c5(h.astype(float))
        # schedule SA
        T0 = 0.2
        Tf = 0.001
        max_iter = 8000
        for it in range(max_iter):
            if time.time() - start_time >= budget_s * 0.7:
                return best_h, best_val
            # temperature schedule
            t = it / max_iter
            T = T0 * (Tf / T0) ** t
            # propose swap
            i = np.random.randint(n)
            while h[i] == 0:
                i = np.random.randint(n)
            j = np.random.randint(n)
            while h[j] == 1:
                j = np.random.randint(n)
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1

                    best_h = h.astype(float).copy()
                    if best_val < 0.3808:

        # end restart

import time
import random
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1-h(x+k))dx  (discrete version)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (2.0 / n) * np.max(corr)


def subgrad(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k correlation w.r.t. h (full*dx factor)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    # indices where the maximal correlation is attained
    idxs = np.where(np.abs(corr - max_val) < 1e-12)[0]
    g = np.zeros(n, dtype=float)
    for idx in idxs:
        s = idx - (n - 1)                 # shift (may be negative)
        if s >= 0:
            g[: n - s] += 1.0 - h[s:]    # term (1 - h_{j+s})
            g[s:]      += -h[: n - s]    # term -h_{j-s}
        else:
            t = -s
            g[t:]      += 1.0 - h[: n - t]
            g[: n - t] += -h[t:]
    g *= (2.0 / n)                       # factor dx
    return g


def project_onto_bounds_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {0≤x≤1, Σx = target_sum}."""
    n = v.shape[0]
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny adjustment for rounding errors
    diff = target_sum - x.sum()
    if abs(diff) > 1e-9:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            adjust = diff / free.sum()
            x[free] += adjust
            x = np.clip(x, 0.0, 1.0)
    return x


def triangle_initial(n: int) -> np.ndarray:
    """A symmetric tent‑shaped start (sum = n/2 after projection)."""
    idx = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    d = np.abs(idx - center)
    h_raw = 1.0 - d / (n / 2.0)
    h_raw = np.maximum(h_raw, 0.0)
    target = n / 2.0
    h = h_raw * (target / h_raw.sum())
    return project_onto_bounds_sum(h, target)


def random_initial(n: int) -> np.ndarray:
    """Uniform random start then projection onto constraints."""
    v = np.random.rand(n)
    return project_onto_bounds_sum(v, n / 2.0)


def subgradient_descent(h0: np.ndarray, end_time: float) -> tuple:
    """Projected sub‑gradient descent with decreasing step size."""
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(best_h)
    it = 0
    while time.time() < end_time and it < 20000:
        it += 1
        g = subgrad(h)
        step = 0.5 / np.sqrt(it)            # diminishing step
        h = h - step * g
        h = project_onto_bounds_sum(h, target)
        val = compute_c5(h)
        if val < best_val - 1e-12:
            best_val = val
            best_h = h.copy()
            if best_val < 0.3808:
                break
    return best_h, best_val


def binary_hill_climb(n: int, end_time: float) -> tuple:
    """Simple swap‑based simulated annealing on binary vectors."""
    S = n // 2
    best_h = None
    best_val = float("inf")
    # temperature schedule parameters
    T0, Tf = 0.3, 0.001
    max_iter = 8000
    for _ in range(6):                                   # a few restarts
        if time.time() >= end_time:
            break
        h = np.zeros(n, dtype=int)
        h[np.random.choice(n, S, replace=False)] = 1
        cur_val = compute_c5(h.astype(float))
        for it in range(max_iter):
            if time.time() >= end_time:
                return best_h, best_val
            t = it / max_iter
            T = T0 * (Tf / T0) ** t
            # propose a swap
            i = np.random.randint(n)
            while h[i] == 0:
                i = np.random.randint(n)
            j = np.random.randint(n)
            while h[j] == 1:
                j = np.random.randint(n)
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            new_val = compute_c5(h_new.astype(float))
            delta = new_val - cur_val
            if delta <= 0.0 or random.random() < np.exp(-delta / T):
                h = h_new
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_h = h.astype(float).copy()
                    if best_val < 0.3808:
                        return best_h, best_val
    return best_h, best_val
