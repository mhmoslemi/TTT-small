import time, math, random

def _c5_value(h, dx):
    """Return the overlap bound for a step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _repair(v, target):
    """Project v onto the feasible box [0,1]^n while fixing the sum to target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    n = v.shape[0]
    i = 0
    # distribute the remaining deficit/surplus cyclically
    while abs(diff) > 1e-12 and i < 10 * n:
        if diff > 0.0:
            cap = 1.0 - v[i]
            add = min(diff, cap)
            v[i] += add
            diff -= add
        else:
            cap = v[i]
            sub = min(-diff, cap)
            v[i] -= sub
            diff += sub
        i = (i + 1) % n
    return v

def _random_feasible(n, target):
    """Generate a random feasible vector of length n with given sum."""
    v = np.random.rand(n)
    v = v / v.sum() * target
    return _repair(v, target)

def _worst_shift_and_A(h):
    """Return the shift that yields the maximal overlap and the corresponding A vector."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    s = idx - (n - 1)                     # integer shift (in samples)
    A = np.zeros(n)
    if s >= 0:
        A[: n - s] = 1.0 - h[s:]
    else:
        A[-s:] = 1.0 - h[: n + s]
    return s, A, corr
