import time
import numpy as np

def evaluate(h: np.ndarray) -> float:
    """Return the C5 value of a binary (or [0,1]) step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a binary vector of length n with exactly n/2 ones."""
    ones = n // 2
    arr = np.zeros(n, dtype=float)
    arr[:ones] = 1.0
    rng.shuffle(arr)
    return arr


def hill_climb(start_h: np.ndarray, rng: np.random.Generator, time_limit: float) -> tuple:
    """
    Simple hill‑climbing with random swaps.
    Returns (best_h, best_val) found within time_limit seconds.
    """
    n = start_h.size
    best_h = start_h.copy()
    best_val = evaluate(best_h)
    # indices of ones and zeros – recomputed after each successful swap
    ones = np.flatnonzero(best_h == 1.0)
    zeros = np.flatnonzero(best_h == 0.0)

    t0 = time.time()
    while time.time() - t0 < time_limit:
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform swap in‑place
        best_h[i], best_h[j] = best_h[j], best_h[i]
        new_val = evaluate(best_h)

        if new_val < best_val - 1e-12:          # improvement
            best_val = new_val
            # recompute index lists – cheap for small n
            ones = np.flatnonzero(best_h == 1.0)
            zeros = np.flatnonzero(best_h == 0.0)
        else:
            # revert swap
            best_h[i], best_h[j] = best_h[j], best_h[i]

    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a binary step function h on [0,2] with integral 1
    that minimizes the Erdős overlap constant C5.
    Returns (list_of_h, C5_value, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Starting vector (use parent if available)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        # force binary – any value >0.5 becomes 1, else 0
        h0 = (h0 > 0.5).astype(float)
        # ensure exact half‑ones; if rounding drifted, fix by simple re‑balancing
        ones_needed = h0.size // 2
        current_ones = int(h0.sum())
        if current_ones != ones_needed:
            idx = np.where(h0 == 1.0)[0]
            if current_ones > ones_needed:
                rng.shuffle(idx)
                h0[idx[: current_ones - ones_needed]] = 0.0
            else:
                zero_idx = np.where(h0 == 0.0)[0]
                rng.shuffle(zero_idx)
                h0[zero_idx[: ones_needed - current_ones]] = 1.0
    else:
        # fallback: 45 points, half ones, random shuffle
        n0 = 45
        h0 = random_binary_vector(n0, rng)

    best_h, best_val = hill_climb(h0, rng, time_limit=120.0)

    # ------------------------------------------------------------------
    # 2. Random‑restart hill climbs at the same size (if time permits)
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    while elapsed < budget_s * 0.5:
        remain = budget_s - elapsed
        if remain < 30:
            break
        # allocate a modest budget for each restart
        trial_time = min(30.0, remain * 0.2)

    # ------------------------------------------------------------------

    while True:

        else:

            break

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    final_val = evaluate(best_h)
    return (best_h.tolist(), final_val, int(best_h.size))
