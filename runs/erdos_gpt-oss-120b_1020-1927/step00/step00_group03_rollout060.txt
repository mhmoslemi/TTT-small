def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, h, mode='full') * dx
    # exclude zero shift (center)
    center = N-1
    # Min correlation over all k != 0
    # remove center element
    min_corr = np.min(np.concatenate((corr[:center], corr[center+1:])))
    return 1.0 - min_corr

import time
import random
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with its complement (zero‑padded)
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(vals.max())

def _random_binary(n: int) -> np.ndarray:
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h
