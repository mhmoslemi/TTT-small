def project_onto_constraints(h, target_sum):
    # Clip to [0,1] first
    h = np.clip(h, 0.0, 1.0)
    # compute current sum
    current = h.sum()
    # difference
    diff = target_sum - current
    # We'll distribute diff equally across all entries that are not at bounds (i.e., not 0 or 1 after clipping)
    # iterate a few passes
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        free = (h > 0) & (h < 1)
        n_free = free.sum()
        if n_free == 0:
            # cannot adjust further, distribute equally (may violate bounds)
            h += diff / len(h)
        else:
            delta = diff / n_free
            # apply delta to free entries
            h[free] += delta
            # clip again
            h = np.clip(h, 0.0, 1.0)
        current = h.sum()
        diff = target_sum - current
    # final clip
    h = np.clip(h, 0.0, 1.0)
    return h

    # ramp up

import time, math
import numpy as np

def _project_to_constraints(v, target):
    """
    Project vector v onto the box [0,1] and the hyperplane sum(v)=target.
    Simple iterative scheme: clip to [0,1], then correct the sum by
    distributing the excess uniformly over the non‑bound entries.
    """
    v = np.clip(v, 0.0, 1.0)
    for _ in range(15):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (v > 0.0) & (v < 1.0)
        nfree = int(free.sum())
        if nfree == 0:
            v += diff / v.size
        else:
            adj = diff / nfree
            v[free] += adj
            v = np.clip(v, 0.0, 1.0)
    return v


def _g_for_index(idx, h, shifts):
    """
    Gradient (1 - h_{i+s} - h_{i-s}) for all shifts.
    Returns a vector g of length 2*N-1 where g[s+N-1] = 1 - (h_{i+s}+h_{i-s}).
    """
    N = h.size
    g = np.empty(2 * N - 1, dtype=np.float64)
    for pos, s in enumerate(shifts):
        s_sum = 0.0
        p = idx + s
        if 0 <= p < N:
            s_sum += h[p]
        m = idx - s
        if 0 <= m < N:
            s_sum += h[m]
        g[pos] = 1.0 - s_sum
    return g


def _make_cosine(N, amp):
    """h(x)=0.5+amp*cos(pi*(x-1)), sampled at N mid‑points."""
    dx = 2.0 / N
    x = (np.arange(N) + 0.5) * dx
    h = 0.5 + amp * np.cos(np.pi * (x - 1.0))
    return _project_to_constraints(h, N / 2.0)


def _make_piecewise(N, r):
    """Triangular/plateau family described in the analysis."""
    dx = 2.0 / N
    a = (1.0 - r) / 2.0          # zero margin at each side
    t = 1.0 - r                  # flat top length (may be zero)
    x = (np.arange(N) + 0.5) * dx
    h = np.zeros(N, dtype=np.float64)

    if r < 1e-12:                # pure block of height 1
        left = a
        right = a + 1.0
        h[(x >= left) & (x < right)] = 1.0
        return _project_to_constraints(h, N / 2.0)

    # rising ramp
    inc = (x >= a) & (x < a + r)
    h[inc] = (x[inc] - a) / r

    # flat top
    flat = (x >= a + r) & (x < a + r + t)
    h[flat] = 1.0

    # falling ramp
    dec = (x >= a + r + t) & (x < a + 2 * r + t)
    h[dec] = (a + 2 * r + t - x[dec]) / r
    return _project_to_constraints(h, N / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that minimizes
    C5 = max_k ∫ h(x)(1−h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)

    # --------------------------------------------------------------
    # 1. Choose discretisation size.
    # --------------------------------------------------------------
    N = 140                     # reasonable trade‑off between speed & resolution
    target_sum = N / 2.0
    dx = 2.0 / N
    shifts = np.arange(-N + 1, N)          # integer shift indices

    # --------------------------------------------------------------
    # 2. Build a set of simple candidate shapes and keep the best.
    # --------------------------------------------------------------
    best_h = np.full(N, 0.5, dtype=np.float64)
    best_h = _project_to_constraints(best_h, target_sum)
    best_val = np.max(np.correlate(best_h, 1.0 - best_h, mode="full")) * dx

    # cosine family
    for amp in np.linspace(0.0, 0.5, 51):
        h = _make_cosine(N, amp)
        val = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
        if val < best_val:
            best_val, best_h = val, h.copy()

    # piecewise family
    for r in np.linspace(0.0, 1.0, 101):
        h = _make_piecewise(N, r)
        val = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
        if val < best_val:
            best_val, best_h = val, h.copy()

    # --------------------------------------------------------------
    # 3. Stochastic local optimisation (pairwise exchanges).
    # --------------------------------------------------------------
    h = best_h.copy()
    I = np.correlate(h, 1.0 - h, mode="full")      # raw correlation (no dx)
    cur_val = I.max() * dx

    # simulated‑annealing schedule
    T = 0.02
    Tmin = 1e-5
    decay = 0.9996

    start = time.time()
    time_limit = budget_s * 0.94       # leave a small safety margin

    while time.time() - start < time_limit:
        i, j = rng.randint(N, size=2)
        if i == j:
            continue

        # admissible delta so that both entries stay in [0,1]
        inc = min(1.0 - h[i], h[j])        # positive direction (increase i, decrease j)
        dec = min(h[i], 1.0 - h[j])        # negative direction
        if inc + dec < 1e-12:

        delta = (rng.rand() * (inc + dec)) - dec   # uniform in [-dec, inc]

        # compute gradient contributions for the two indices
        g_i = _g_for_index(i, h, shifts)
        g_j = _g_for_index(j, h, shifts)

        newI = I + delta * g_i - delta * g_j
        new_val = newI.max() * dx

        accept = False
        if new_val < cur_val - 1e-12:
            accept = True
        else:
            if T > Tmin:
                prob = math.exp(-(new_val - cur_val) / T)
                if rng.rand() < prob:
                    accept = True

        if accept:
            I = newI
            h[i] += delta
            h[j] -= delta
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_val, best_h = cur_val, h.copy()

        T = max(T * decay, Tmin)

        # occasional random perturbation to avoid dead‑ends
        if (rng.randint(50000) == 0) and (time.time() - start < time_limit):
            h = best_h.copy()
            h += (rng.rand(N) - 0.5) * 0.003
            h = _project_to_constraints(h, target_sum)
            I = np.correlate(h, 1.0 - h, mode="full")
            cur_val = I.max() * dx

    # --------------------------------------------------------------
    # 4. Final evaluation (use the provided helper if available).
    # --------------------------------------------------------------
    try:
        final_c5 = float(evaluate_erdos_solution(best_h))
    except NameError:
        final_c5 = float(best_val)

    return best_h, final_c5, N
