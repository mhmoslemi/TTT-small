    diff = target - np.sum(y)

import time, math, random
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a step function h on [0,2] (sampled) together with an upper bound
    for C_5 = max_k ∫ h(x)(1−h(x+k))dx.
    The vector h satisfies 0≤h[i]≤1 and sum(h)=n/2 (dx=2/n).
    """
    np.random.seed(seed)
    random.seed(seed)

    # ----------------------------------------------------------------------
    # 1. Starting vector
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        # fallback: a simple half‑filled binary pattern
        n0 = 42
        base = np.zeros(n0, dtype=np.float64)
        base[: n0 // 2] = 1.0
        np.random.shuffle(base)

    n0 = base.size
    # enforce binary and exact half–integral (needed for up‑sampling)
    target_half = n0 / 2
    if abs(base.sum() - target_half) > 1e-9:
        # replace by a random binary vector with the right number of ones
        base[:] = 0.0
        idx = np.random.permutation(n0)[:int(target_half)]
        base[idx] = 1.0

    # ----------------------------------------------------------------------
    # 2. Upsample to a finer grid (still binary, respects the integral)
    # ----------------------------------------------------------------------
    max_n = 800                # keep evaluation cheap
    factor = max(1, int(max_n // n0))
    n = n0 * factor
    h = np.repeat(base, factor)                # binary after repeat
    h = (h > 0.5).astype(np.float64)           # enforce exact 0/1

    # exact integral correction (should already be correct)
    target_sum = n / 2
    current_sum = h.sum()
    diff = int(round(target_sum - current_sum))
    if diff > 0:
        zeros = np.flatnonzero(h == 0)
        chosen = np.random.choice(zeros, size=diff, replace=False)
        h[chosen] = 1.0
    elif diff < 0:
        ones = np.flatnonzero(h == 1)
        chosen = np.random.choice(ones, size=-diff, replace=False)
        h[chosen] = 0.0

    dx = 2.0 / n

    def compute_c5(vec):
        """C5 for a given vector (full correlation)"""
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # ----------------------------------------------------------------------
    # 3. Greedy local improvement focusing on the worst shift
    # ----------------------------------------------------------------------
    start_time = time.time()
    time_limit = budget_s - 5.0                # safety margin
    no_improve_counter = 0
    max_no_improve = 3000                     # restart after this many stalls

    while time.time() - start_time < time_limit:
        # current correlation & worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_c5 = np.max(corr) * dx
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)             # integer shift giving the max

        # indices where a 1 contributes to that shift (bad positions)
        if shift >= 0:
            i_range = np.arange(0, n - shift)
            partner = h[shift:]
        else:
            s = -shift
            i_range = np.arange(s, n)
            partner = h[: n - s]

        bad_mask = (h[i_range] == 1) & (partner == 0)
        bad_i = i_range[bad_mask]

        if bad_i.size == 0:
            # nothing contributes – we are at optimum for this shift
            # try a random swap to explore other shifts
            ones_idx = np.flatnonzero(h == 1)
            zeros_idx = np.flatnonzero(h == 0)
            i_bad = np.random.choice(ones_idx)
            j_good = np.random.choice(zeros_idx)
        else:
            i_bad = np.random.choice(bad_i)

            # look for a zero that will NOT create a new contribution for this shift
            zeros_idx = np.flatnonzero(h == 0)
            if shift >= 0:
                cond = (zeros_idx + shift < n) & (h[zeros_idx + shift] == 1)
            else:
                cond = (zeros_idx + shift >= 0) & (h[zeros_idx + shift] == 1)
            good_zeros = zeros_idx[cond]
            if good_zeros.size:
                j_good = np.random.choice(good_zeros)
            else:
                # fallback – any zero
                j_good = np.random.choice(zeros_idx)

        # propose swap
        h_new = h.copy()
        h_new[i_bad] = 0.0
        h_new[j_good] = 1.0

        new_c5 = compute_c5(h_new)

        # accept only if we do not increase the maximal overlap
        if new_c5 <= cur_c5 + 1e-12:
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve_counter = 0
            else:
                no_improve_counter += 1
        else:
            # occasional stochastic acceptance (simulated‑annealing flavour)
            # temperature decays with the number of iterations
            T = max(1e-4, 0.02 * (1.0 - (time.time() - start_time) / time_limit))
            if random.random() < math.exp(-(new_c5 - cur_c5) / T):
                h = h_new
                cur_c5 = new_c5
                no_improve_counter += 1
            else:
                no_improve_counter += 1

        # restart if we stagnate for a while
        if no_improve_counter >= max_no_improve:
            # disturb the current best a little and continue
            h = best_h.copy()
            # perform a handful of random swaps
            for _ in range(10):
                ones_idx = np.flatnonzero(h == 1)
                zeros_idx = np.flatnonzero(h == 0)
                i1 = np.random.choice(ones_idx)
                j1 = np.random.choice(zeros_idx)
                h[i1], h[j1] = 0.0, 1.0
            no_improve_counter = 0

    # ----------------------------------------------------------------------
    # 4. Return the best discovered construction
    # ----------------------------------------------------------------------
    return best_h, best_c5, n
