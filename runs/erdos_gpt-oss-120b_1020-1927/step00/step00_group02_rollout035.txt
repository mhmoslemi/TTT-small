def evaluate(h):
    dx = 2.0/h.shape[0]
    O = np.correlate(h, 1 - h, mode='full')
    max_O = O.max()
    return max_O * dx

def local_search(N, rnd, time_limit):
    # start random binary vector
    h = np.zeros(N)
    idx = rnd.choice(N, N//2, replace=False)
    h[idx] = 1.0
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    # pre-calc
    while time.time() < time_limit:
        # propose swap
        ones = np.where(best_h==1)[0]
        zeros = np.where(best_h==0)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = rnd.choice(ones)
        j = rnd.choice(zeros)
        h_new = best_h.copy()
        h_new[i] = 0
        h_new[j] = 1
        c5_new = evaluate(h_new)
        if c5_new < best_c5 - 1e-9:
            best_h = h_new
            best_c5 = c5_new
        # else maybe accept occasionally
    return best_h, best_c5

def shift(arr, s):
    n = arr.shape[0]
    if s > 0:
        out = np.empty_like(arr)
        out[:n-s] = arr[s:]
        out[n-s:] = 0.0
    elif s < 0:
        s = -s
        out = np.empty_like(arr)
        out[s:] = arr[:n-s]
        out[:s] = 0.0
    else:
        out = arr.copy()
    return out

    best_h = h.copy()

    alpha = 0.5

    while time.time() < deadline:
        O_full = np.correlate(h, 1-h, mode='full')

def gradient_descent(N, h, M, rnd, deadline):
    # returns best solution and its raw O_max (not scaled)
    best_h = h.copy()
    # compute initial O_max raw
    O_full = np.correlate(h, 1 - h, mode='full')
    best_raw = O_full.max()
    best_iter_raw = best_raw
    best_h_iter = h.copy()
    # parameters
    alpha = 0.5
    it = 0
    no_improve = 0
    while time.time() < deadline:
        O_full = np.correlate(h, 1 - h, mode='full')
        O_max = O_full.max()
        # keep best
        if O_max < best_raw:
            best_raw = O_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # find worst shift
        idx = O_full.argmax()
        shift_val = idx - (N - 1)
        # compute gradient
        grad = 1.0 - shift_arr(h, shift_val) - shift_arr(h, -shift_val)
        # take step
        new_h = project_box_sum(h - alpha * grad, M)
        new_O_max = np.max(np.correlate(new_h, 1 - new_h, mode='full'))
        if new_O_max <= O_max:
            h = new_h
            # slightly increase alpha
            alpha = min(alpha * 1.1, 2.0)
        else:
            # reject, reduce alpha
            alpha = max(alpha * 0.5, 1e-6)
            # occasional random perturb if alpha too small
            if alpha < 1e-4:
                perturb = rnd.uniform(-0.02,0.02,size=N)
                h = project_box_sum(h + perturb, M)
                alpha = 0.5
        it += 1
        if no_improve > 2000:
            # restart from random binary
            h = np.zeros(N)
            idx2 = rnd.choice(N, int(M), replace=False)
            h[idx2] = 1.0
            alpha = 0.5
            no_improve = 0
    return best_h, best_raw

import numpy as np, time, math

def _project_box_sum(v, target, max_iter=60):
    """Euclidean projection of v onto {0<=x<=1, sum(x)=target}."""
    lo, hi = -2.0, 2.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        p = np.clip(v - mid, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _build_h(N, a1, a2, h1, h2, h3):
    """Construct symmetric 5‑step function on [0,2] with N samples."""
    dx = 2.0 / N
    a3 = 2.0 - 2.0 * a1 - 2.0 * a2          # a3 > 0  (middle interval length)
    # interval boundaries on the x‑axis
    t1 = a1
    t2 = a1 + a2
    t3 = a1 + a2 + a3
    t4 = a1 + a2 + a3 + a2
    xs = (np.arange(N) + 0.5) * dx
    h = np.empty(N, dtype=float)
    h[xs < t1] = h1
    mask = (xs >= t1) & (xs < t2)
    h[mask] = h2
    mask = (xs >= t2) & (xs < t3)
    h[mask] = h3
    mask = (xs >= t3) & (xs < t4)
    h[mask] = h2
    h[xs >= t4] = h1
    # enforce exact integral after discretisation
    return _project_box_sum(h, N * 0.5)

def _c5_of_h(h):
    """Return C5 for a discretised h (dx already accounted)."""
    dx = 2.0 / h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _refine_params(N, a1, a2, h1, h2, h3, cur_c5, rng, deadline):
    """Simple coordinate hill‑climb on the 5‑step parameters."""
    step = np.array([0.02, 0.02, 0.04, 0.04])
    for _ in range(2000):
        if time.time() >= deadline:
            break
        improved = False
        for i in range(4):
            for delta in (-step[i], step[i]):
                na1, na2, nh1, nh2 = a1, a2, h1, h2
                if i == 0:
                    na1 = a1 + delta
                elif i == 1:
                    na2 = a2 + delta
                elif i == 2:
                    nh1 = h1 + delta
                else:
                    nh2 = h2 + delta
                if na1 <= 0 or na2 <= 0 or na1 + na2 >= 1:
                    continue
                a3 = 2.0 - 2.0 * na1 - 2.0 * na2
                if a3 <= 0:
                    continue
                nh3 = (1.0 - 2.0 * na1 * nh1 - 2.0 * na2 * nh2) / a3
                if not (0.0 <= nh3 <= 1.0):
                    continue
                h = _build_h(N, na1, na2, nh1, nh2, nh3)
                c5 = _c5_of_h(h)
                if c5 + 1e-9 < cur_c5:
                    a1, a2, h1, h2, h3 = na1, na2, nh1, nh2, nh3
                    cur_c5 = c5

                    break
            if improved:
                break
        if not improved:
            step *= 0.5
