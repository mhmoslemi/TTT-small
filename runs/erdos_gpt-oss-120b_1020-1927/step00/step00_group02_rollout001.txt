def adjust_to_target_sum(h, target_sum):
    # h array values in [0,1]
    total = h.sum()
    if abs(total-target_sum) < 1e-12:
        return h
    # iterative
    while True:
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need increase some entries not at 1
            idx = np.where(h < 1)[0]
            if len(idx) == 0: break
            inc = min(diff/len(idx), 1-h[idx])
            # inc might be vector, but we can compute step = min(diff/len(idx), min(1 - h[idx]))
            step = min(diff/len(idx), np.min(1 - h[idx]))
            h[idx] += step
        else:
            # need decrease entries not at 0
            idx = np.where(h > 0)[0]
            step = min(-diff/len(idx), np.min(h[idx]))
            h[idx] -= step
        total = h.sum()
    return h
```

    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # while diff not zero, adjust iteratively
    # We'll apply adjustment in vectorized way:
    # if diff > 0: increase entries below 1
    # compute capacity = sum(1 - h[idx])
    # inc = min(diff, capacity)
    # distribute evenly among those entries
    # repeat if diff remains (due to hitting bounds)
    # analogous for diff < 0
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            capacity = np.sum(1.0 - h[idx])
            inc = min(diff, capacity)
            # allocate proportionally (even distribution)
            # distribute inc equally among idx (or proportionally to capacity)
            # We'll give each equal share:
            per = inc / idx.size
            # ensure per not exceed 1 - h[idx]

        else:

                break
            capacity = np.sum(h[idx])
            dec = min(-diff, capacity)
            per = dec / idx.size
            per = np.minimum(per, h[idx])
            h[idx] -= per
        total = h.sum()
        diff = target - total
    return h

def _compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx  (discrete version)"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / len(h))

def _adjust_to_target_sum(h, target):
    """Clip to [0,1] and modify to reach exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # simple iterative redistribution
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            cap = np.sum(1.0 - h[idx])
            take = min(diff, cap)
            per = take / idx.size
            inc = np.minimum(per, 1.0 - h[idx])
            h[idx] += inc
        else:

            if idx.size == 0:
                break
            cap = np.sum(h[idx])
            take = min(-diff, cap)
            per = take / idx.size

    if shift > 0:
        max_i = n - shift
        i = int(rng.integers(max_i))
        j = i + shift
    else:
        max_i = n + shift  # shift negative
        i = int(rng.integers(max_i))
        j = i - shift  # i + shift
    # need room to increase both
    eps_max = min(1.0 - h[i], 1.0 - h[j])
    if eps_max <= 0.0:
        return None
    # find two donors with positive mass, distinct from i,j
    donors = np.where((h > 1e-12) & (np.arange(n) != i) & (np.arange(n) != j))[0]
    if donors.size < 2:
        return None
    a, b = rng.choice(donors, size=2, replace=False)
    eps_max = min(eps_max, h[a], h[b])
    if eps_max <= 0.0:
        return None
    eps = eps_max * rng.random() * 0.5
    nh = h.copy()
    nh[i] += eps
    nh[j] += eps
    nh[a] -= eps
    nh[b] -= eps
    return nh

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initial construction ------------------------------------------------
    init_vec = globals().get("initial_h_values", None)
    if init_vec is not None:
        h0 = np.asarray(init_vec, dtype=float)
        n0 = len(h0)
    else:
        h0 = None
        n0 = 45

    # Target number of sample points (keep < 1000)
    n_target = max(45, min(300, n0 * 2))
    if h0 is not None and n_target != n0:
        # linear up‑sampling of the starter
        x_old = np.linspace(0, 2, n0, endpoint=False)
        x_new = np.linspace(0, 2, n_target, endpoint=False)
        h = np.interp(x_new, x_old, h0)
    elif h0 is not None:
        h = h0.copy()
    else:
        # random binary start
        h = np.zeros(n_target)
        h[: n_target // 2] = 1.0
        rng.shuffle(h)

    # enforce feasibility
    target_sum = n_target / 2.0
    h = _adjust_to_target_sum(h, target_sum)

    # ---- optimisation --------------------------------------------------------
    best_h = h.copy()
    best_val = _compute_c5(best_h)

    cur_h = h.copy()
    cur_val = best_val

    T0 = 0.008                     # initial temperature
    prob_target = 0.45            # chance to try a targeted move
    no_imp_iters = 0
    max_no_imp = 200_000
    iter_count = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
        # cooling schedule (linear to zero)
        T = T0 * max(0.0, 1.0 - elapsed / (budget_s * 0.9))

        # decide move type
        if rng.random() < prob_target:
            # compute current worst shift
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            shift = int(corr.argmax() - (n_target - 1))
            new_h = _targeted_move(cur_h, shift, rng)
            if new_h is None:
                new_h = _random_transfer(cur_h, rng)
        else:
            new_h = _random_transfer(cur_h, rng)

        if new_h is None:
            continue

        new_val = _compute_c5(new_h)

        delta = new_val - cur_val
        accept = False
        if delta <= 0.0:
            accept = True
        elif T > 0.0:
            if rng.random() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_h = new_h
            cur_val = new_val
            no_imp_iters = 0
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = new_h.copy()
                # early exit if we beat the target bound
                if best_val <= 0.3808:
                    break
        else:
            no_imp_iters += 1

        # occasional restart if stuck
        if no_imp_iters > max_no_imp:
            cur_h = np.zeros(n_target)
            cur_h[: n_target // 2] = 1.0
            rng.shuffle(cur_h)
            cur_h = _adjust_to_target_sum(cur_h, target_sum)
            cur_val = _compute_c5(cur_h)
            no_imp_iters = 0

        iter_count += 1

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _adjust_to_target_sum(best_h, target_sum)
    final_val = _compute_c5(best_h)

    return best_h, float(final_val), n_target
