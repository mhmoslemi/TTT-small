def mutate(h, rng, step=0.1):
    N = len(h)
    i, j = rng.choice(N, size=2, replace=False)
    # Determine how much we can transfer
    # transfer from i to j (decrease i, increase j)
    max_transfer = min(h[i], 1 - h[j])
    # also we can transfer from j to i, but we randomly choose direction
    # pick delta uniformly from [-max_transfer, max_transfer] 
    # such that resulting values stay within [0,1].
    # For direction i->j we take delta positive up to max_transfer (i decreased, j increased)
    # For direction j->i delta negative up to min(h[j], 1 - h[i]).
    # So we can sample a random direction.
    direction = rng.choice([1,-1])
    if direction == 1:
        max_delta = max_transfer
    else:
        max_delta = min(h[j], 1 - h[i])
    if max_delta <= 0:
        return h
    delta = rng.uniform(0, max_delta) * direction
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

    deficit = target - v.sum()
    # While deficit not zero and there is capacity left (v<1)

import time
import numpy as np

def _neighbor(h, rng, swaps=1):
    """Return a new feasible vector by moving mass between two positions."""
    h2 = h.copy()
    n = h2.size
    target = n / 2.0
    for _ in range(swaps):
        i, j = rng.choice(n, size=2, replace=False)
        # feasible delta to add to i and subtract from j
        low = max(-h2[i], h2[j] - 1.0)
        high = min(1.0 - h2[i], h2[j])
        if high <= low:
            continue
        delta = rng.uniform(low, high)
        h2[i] += delta
        h2[j] -= delta
    # correct tiny floating drift
    diff = target - h2.sum()
    if abs(diff) > 1e-12:
        # adjust first entry that can absorb the difference
        for idx in range(n):
            if 0.0 <= h2[idx] + diff <= 1.0:
                h2[idx] += diff
                break
    return h2

def _random_binary(n, rng):
    """Generate a binary vector with exactly n/2 ones (n assumed even)."""
    h = np.zeros(n, dtype=np.float64)
    h[:n // 2] = 1.0
    rng.shuffle(h)
    return h
