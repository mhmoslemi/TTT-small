def simulated_annealing(start_h, init_score, time_limit):
    cur = start_h.copy()
    cur_score = init_score
    best = cur.copy()
    best_score = cur_score
    n = len(cur)
    indices_one = np.flatnonzero(cur == 1)
    indices_zero = np.flatnonzero(cur == 0)
    temperature = 1.0
    # We'll use an exponential schedule:
    # compute total steps estimate: maybe 5000 iterations
    num_iter = 0
    start = time.time()
    while time.time() - start < time_limit:
        # gradually decreasing temperature
        # we can compute temperature based on progress
        progress = (time.time() - start) / time_limit
        temperature = max(0.001, 1.0 - progress)  # linear cooling from 1 to 0
        # propose swap
        if len(indices_one) == 0 or len(indices_zero) == 0:
            break
        i_one = random.choice(indices_one)
        i_zero = random.choice(indices_zero)
        # swap
        cur[i_one] = 0
        cur[i_zero] = 1
        new_score = compute_score(cur)
        delta = new_score - cur_score
        if delta < 0 or random.random() < math.exp(-delta/temperature):
            # accept
            cur_score = new_score
            # update index lists
            indices_one = np.flatnonzero(cur == 1)
            indices_zero = np.flatnonzero(cur == 0)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
        else:
            # revert
            cur[i_one] = 1

import time, random, math
import numpy as np

def _compute_score(h):
    """Return C5 bound for a (binary or real) step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) with a small C5 bound."""
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Starting point (binary) – use the provided initial_h_values if any.
    # ------------------------------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n = init.shape[0]
        # Turn the supplied vector into a binary one with exactly n/2 ones.
        # Choose the n/2 indices with largest values.
        idx = np.argsort(-init)[: n // 2]
        cur_h = np.zeros(n, dtype=int)
        cur_h[idx] = 1
    except Exception:  # no initial_h_values or something went wrong
        n = 80
        cur_h = np.zeros(n, dtype=int)
        cur_h[: n // 2] = 1
        np.random.shuffle(cur_h)

    best_h = cur_h.copy()
    best_score = _compute_score(best_h.astype(float))

    # ------------------------------------------------------------------
    # 2. Optional MILP exact optimisation (if time permits).
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    time_left = budget_s - elapsed - 5.0  # reserve a few seconds for the local search
    if time_left > 30.0:  # give the MILP a decent slice of the budget
        try:
            import cvxpy as cp

            h = cp.Variable(n, boolean=True)
            t = cp.Variable()
            constraints = [cp.sum(h) == n / 2, t >= 0]

            # product variables for unordered pairs (i<j)
            pair = {}
            for i in range(n):
                for j in range(i + 1, n):
                    z = cp.Variable(boolean=True)
                    pair[(i, j)] = z
                    constraints += [z <= h[i], z <= h[j], z >= h[i] + h[j] - 1]

            # shift constraints: for each non‑zero lag ensure overlap ≤ t
            for d in range(1, n):
                # positive lag
                idx = np.arange(0, n - d)
                constraints.append(
                    cp.sum(h[idx]) - cp.sum([pair[(i, i + d)] for i in idx]) <= t
                )
                # negative lag
                idx2 = np.arange(d, n)
                constraints.append(
                    cp.sum(h[idx2]) - cp.sum([pair[(i - d, i)] for i in idx2]) <= t
                )

            prob = cp.Problem(cp.Minimize(t), constraints)

            solver_opts = {}

            try:

            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):

                if diff != 0:

    # ------------------------------------------------------------------

        elif diff < 0:

            best_h[flip] = 0.0

    final_score = _compute_score(best_h)
    return (best_h, final_score, n)
