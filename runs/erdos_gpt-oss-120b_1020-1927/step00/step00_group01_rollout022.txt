def run(seed=42,budget_s=1100,**kwargs):
    start = time.time()
    np.random.seed(seed)
    random.seed(seed)
    n = 200
    target_sum = n//2
    # initial:
    if 'initial_h_values' in globals():
        # Use upscale and threshold:
        init = np.array(initial_h_values, dtype=float)
        # Upscale:
        x_old = np.linspace(0,2,len(init),endpoint=False)
        x_new = np.linspace(0,2,n,endpoint=False)
        up = np.interp(x_new, x_old, init)
        # threshold: pick top target_sum values:
        idx = np.argsort(up)[-target_sum:]
        h0 = np.zeros(n)
        h0[idx]=1.0
    else:
        # random
        h0 = np.zeros(n)
        h0[np.random.choice(n, target_sum, replace=False)] = 1.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h0.copy()
    cur_c5 = best_c5
    # annealing parameters:
    T0 = 0.05
    Tmin = 1e-6
    iter = 0
    while time.time() - start < budget_s * 0.95:
        # adaptive temperature:
        # linear cooling? maybe exponential based on fraction time passed.
        elapsed = time.time() - start
        fraction = elapsed / (budget_s * 0.95)
        T = max(T0 * (1 - fraction), Tmin)
        # pick swap:
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if len(ones)==0 or len(zeros)==0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap:
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_c5 = compute_c5(cur_h)
        delta = new_c5 - cur_c5
        accept = False
        if delta <= 0:
            accept = True
        else:
            if random.random() < math.exp(-delta / (T + 1e-12)):
                accept = True
        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = 1.0, 0.0
        iter += 1
        # maybe occasional random restart if stuck:
        if iter % 200000 == 0:
            # random restart with best as start for a while
            # but maintain variety.
            # If best not improved for a while maybe random change.
            pass
    # final compute via evaluate_erdos_solution if available
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except Exception:
        final_c5 = compute_c5(best_h)
    return best_h, float(final_c5), n

import cvxpy as cp

def solve_milp(N):
    # Variables
    x = cp.Variable(N, boolean=True)
    # Ensure sum x = N/2
    constraints = [cp.sum(x) == N//2]
    # B variable (real) maybe integer
    B = cp.Variable(integer=True)
    # For each ordered pair (i,j) i != j:
    # We'll need w_ij = x_i * (1 - x_j). Since both x_i and x_j are binary.
    # Introduce w_ij boolean variable.
    w = {}
    for i in range(N):
        for j in range(N):
            if i == j:
                continue
            # define w_ij
            w_ij = cp.Variable(boolean=True)
            # constraints
            constraints += [w_ij <= x[i]]
            constraints += [w_ij <= 1 - x[j]]
            constraints += [w_ij >= x[i] - x[j]]
            w[(i,j)] = w_ij

    # For each lag d from -(N-1) to N-1:
    for d in range(-N+1, N):
        # sum over i where i+d in range
        terms = []
        for i in range(N):
            j = i + d
            if 0 <= j < N:
                if d != 0:
                    terms.append(w[(i,j)])
                else:
                    # lag zero: w_i,i = x_i * (1 - x_i) = 0 always, ignore.
                    pass
        if terms:
            constraints += [cp.sum(terms) <= B]
    # objective: minimize B
    obj = cp.Minimize(B)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.CBC, warm_start=True)  # maybe specify time limit?

    # get x values
    x_val = np.round(x.value).astype(int)
    B_val = B.value

    return x_val, B_val

def solve_milp(N, time_limit=None):
    import cvxpy as cp, numpy as np, time
    x = cp.Variable(N, boolean=True)
    W = cp.Variable((N, N), boolean=True)

    constraints = [cp.sum(x) == N//2]

    # set diagonal to zero:
    for i in range(N):
        constraints.append(W[i,i] == 0)

    # linearization constraints:
    # W <= X column
    constraints.append(W <= cp.reshape(x, (N,1)))
    constraints.append(W <= cp.reshape(1 - x, (1,N)))
    constraints.append(W >= cp.reshape(x, (N,1)) - cp.reshape(1 - x, (1,N)))

    # B variable
    B = cp.Variable(integer=True)
    constraints.append(B >= 0)

    # for each lag
    for d in range(-N+1, N):
        indices = [(i, i+d) for i in range(N) if 0 <= i+d < N and i != i+d] # exclude diagonal
        if not indices:
            continue
        expr = cp.sum(cp.hstack([W[i,j] for (i,j) in indices]))
        constraints.append(expr <= B)

    # objective
    objective = cp.Minimize(B)

    prob = cp.Problem(objective, constraints)
    # If time_limit passed:
    solve_kwargs = {}
    if time_limit is not None:
        solve_kwargs['solver'] = cp.CBC
        # solver options: need to pass via 'maxSeconds' and maybe 'msg'?
        solve_kwargs['verbose'] = False
        solve_kwargs['solver_options'] = {'maxSeconds': time_limit}
        prob.solve(**solve_kwargs)
    else:
        prob.solve(solver=cp.CBC, verbose=False)

    if prob.status not in ["optimal", "optimal_inaccurate"]:
        raise RuntimeError("MILP not solved status: " + prob.status)
    x_val = np.rint(x.value).astype(int)
    B_val = int(round(B.value))
    return x_val, B_val

import time, random, math
import numpy as np
import cvxpy as cp

def _c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)

def _initial_binary(n):
    # use the global initial_h_values if present, otherwise random
    if "initial_h_values" in globals() and initial_h_values is not None:
        iv = np.asarray(initial_h_values, dtype=float)
        old_n = len(iv)
        x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n,   endpoint=False)
        interp = np.interp(x_new, x_old, iv)
        idx = np.argsort(interp)[-n // 2 :]
        h = np.zeros(n, dtype=float)
        h[idx] = 1.0
        return h
    # pure random construction
    h = np.zeros(n, dtype=float)
    h[np.random.choice(n, n // 2, replace=False)] = 1.0
    return h

def _solve_milp(n, time_limit):
    """Minimise the maximal overlap for a binary vector of length n.
    Returns a binary array and the optimal maximal overlap (as integer count)."""
    x = cp.Variable(n, boolean=True)               # indicator variables
    w = cp.Variable((n, n), boolean=True)          # w[i,j] = x[i] * (1 - x[j])
    B = cp.Variable(integer=True)                  # upper bound on all overlaps

    constraints = [cp.sum(x) == n // 2,
                   B >= 0]

    # linearisation constraints for w[i,j] = x[i]*(1-x[j])  (i!=j)
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            constraints.append(w[i, j] <= x[i])
            constraints.append(w[i, j] <= 1 - x[j])
            constraints.append(w[i, j] >= x[i] - x[j])

    # bound each lag
    for d in range(-n + 1, n):
        if d == 0:
            continue
        if d > 0:
            expr = cp.sum(cp.hstack([w[i, i + d] for i in range(n - d)]))
        else:
            dd = -d
            expr = cp.sum(cp.hstack([w[i, i - dd] for i in range(dd, n)]))
        constraints.append(expr <= B)

    prob = cp.Problem(cp.Minimize(B), constraints)
    # solve with a time cap
    prob.solve(solver=cp.CBC, max_seconds=time_limit, verbose=False)
    if prob.status not in ("optimal", "optimal_inaccurate"):
        raise RuntimeError(f"MILP not solved (status {prob.status})")
    xv = np.rint(x.value).astype(int)
    Bv = int(round(B.value))
    return xv, Bv

def _heuristic_search(n, start, budget):
    """Simple hill‑climbing / simulated annealing for binary vectors."""
    h = _initial_binary(n)
    best_h = h.copy()
    best_c5 = _c5_from_h(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    dx = 2.0 / n

    while time.time() - start < budget * 0.9:
        corr = np.correlate(cur_h, 1 - cur_h, mode="full")
        B = int(np.max(corr))
        lag = int(np.argmax(corr) - (n - 1))

        # indices that contribute to the worst lag
        if lag >= 0:
            bad_one = np.where((cur_h[:n - lag] == 1) & (cur_h[lag:] == 0))[0]
            zeros = np.where(cur_h == 0)[0]
            good_zero = zeros[(zeros + lag >= n) | (cur_h[zeros + lag] == 1)]
        else:
            k = -lag
            bad_one = np.where((cur_h[k:] == 1) & (cur_h[:n - k] == 0))[0] + k
            zeros = np.where(cur_h == 0)[0]
            good_zero = zeros[(zeros - k < 0) | (cur_h[zeros - k] == 1)]

        made = False
        if bad_one.size and good_zero.size:
            i = int(np.random.choice(bad_one))
            j = int(np.random.choice(good_zero))
            cur_h[i] = 0.0
            cur_h[j] = 1.0
            newB = int(np.max(np.correlate(cur_h, 1 - cur_h, mode="full")))
            if newB <= B:
                cur_c5 = newB * dx
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur_h.copy()
                made = True
            if not made:
                cur_h[i] = 1.0
                cur_h[j] = 0.0

        if not made:
            # generic random swap
            one_idx = int(np.random.choice(np.where(cur_h == 1)[0]))
            zero_idx = int(np.random.choice(np.where(cur_h == 0)[0]))
            cur_h[one_idx] = 0.0
            cur_h[zero_idx] = 1.0
            newc5 = _c5_from_h(cur_h)
            if newc5 < best_c5:
                best_c5 = newc5
                best_h = cur_h.copy()
        if best_c5 <= 0.3808:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function (as a numpy array), its C5 bound and point count."""
