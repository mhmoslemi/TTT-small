def _directed_move(h, shift, max_move):
    """Return a new h after moving mass along the given shift,
    or None if no beneficial move found."""
    n = h.size
    if shift == 0:
        # variance‑increasing move: from low to high
        i_low = np.argmin(h)
        i_high = np.argmax(h)
        if h[i_low] <= 0.0 or h[i_high] >= 1.0:
            return None
        delta = min(max_move, h[i_low], 1.0 - h[i_high])
        if delta <= 0.0:
            return None
        new_h = h.copy()
        new_h[i_low] -= delta
        new_h[i_high] += delta
        return new_h

    # shift non‑zero: move mass from larger to smaller across the shift
    if shift > 0:
        i_start, i_end = 0, n - shift
    else:
        i_start, i_end = -shift, n
    idx = np.arange(i_start, i_end)
    jdx = idx + shift
    diff = h[idx] - h[jdx]                     # positive => move from idx→jdx
    mask = diff > 0.0
    if mask.any():
        best = idx[mask][np.argmax(diff[mask])]
        src, dst = best, best + shift
    else:
        # try opposite direction
        diff2 = h[jdx] - h[idx]
        mask2 = diff2 > 0.0
        if not mask2.any():
            return None
        best = idx[mask2][np.argmax(diff2[mask2])]
        src, dst = best + shift, best
    delta = min(max_move, h[src], 1.0 - h[dst])
    if delta <= 0.0:
        return None
    new_h = h.copy()
    new_h[src] -= delta
    new_h[dst] += delta
    return new_h


def _random_pair_move(h, max_move):
    """Simple random transfer between two positions."""
    n = h.size
    i, j = np.random.choice(n, size=2, replace=False)
    # move from i to j
    delta = min(max_move, h[i], 1.0 - h[j])
    if delta <= 0.0:
        return None
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Setup problem size and initial vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        # fallback: uniform half‑height (average 0.5)
        n = kwargs.get("n_points", 95)
        h0 = np.full(n, 0.5, dtype=np.float64)

    n = h0.size
    dx = 2.0 / n
    target_sum = n / 2.0                     # because ∫h = 1 ⇒ sum(h)*dx = 1

    # ensure feasibility (tiny numeric fixes)
    h0 = _make_feasible(h0, target_sum)

    # ------------------------------------------------------------------
    # Initial evaluation
    # ------------------------------------------------------------------
    best_h = h0.copy()
    best_score = evaluate_erdos_solution(best_h)
    cur_h = best_h.copy()
    cur_score = best_score

    # schedule parameters
    max_move_start = 0.12    # larger moves early
    max_move_end   = 0.001   # fine adjustments later
    T_start        = 0.015   # temperature for occasional uphill steps
    T_end          = 0.0001

    start_time = time.time()
    stagnant = 0               # counts iterations without improvement

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.05:
            break
        progress = elapsed / budget_s

        # adaptive parameters
        max_move = max_move_start * (1.0 - progress) + max_move_end * progress
        T = T_start * (1.0 - progress) + T_end * progress

        # --------------------------------------------------------------
        # Identify worst shift and propose a directed move
        # --------------------------------------------------------------
        corr, _ = _correlation_score(cur_h, dx)
        shift = int(corr.argmax() - (n - 1))

        new_h = _directed_move(cur_h, shift, max_move)

        # If directed move not possible, fall back to a random pair move
        if new_h is None:
            new_h = _random_pair_move(cur_h, max_move)

        # If still None (unlikely), skip this iteration
        if new_h is None:
            continue

        # --------------------------------------------------------------
        # Evaluate and decide on acceptance
        # --------------------------------------------------------------
        new_score = evaluate_erdos_solution(new_h)

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_h = new_h.copy()
            stagnant = 0
        else:
            stagnant += 1

        # accept if better, otherwise maybe accept uphill (simulated annealing)
        if new_score <= cur_score:
            cur_h, cur_score = new_h, new_score
        else:
            if np.random.rand() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
                cur_h, cur_score = new_h, new_score

        # --------------------------------------------------------------
        # Periodic diversification if stuck for many iterations
        # --------------------------------------------------------------
        if stagnant >= 2000:
            # perturb the current best a bit
            pert = np.random.uniform(-max_move, max_move, size=n)
            cand = best_h + pert
            cand = np.clip(cand, 0.0, 1.0)
            # enforce exact sum (simple projection)
            cand = _make_feasible(cand, target_sum)
            cand_score = evaluate_erdos_solution(cand)
            cur_h, cur_score = cand, cand_score
            stagnant = 0

    # ------------------------------------------------------------------
    # Final feasibility fix and return
    # ------------------------------------------------------------------
    best_h = _make_feasible(best_h, target_sum)
    final_score = evaluate_erdos_solution(best_h)
    return best_h.astype(np.float32), float(final_score), n
