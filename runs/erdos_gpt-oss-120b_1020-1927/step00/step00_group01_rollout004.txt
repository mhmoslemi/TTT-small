def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def proj_box_sum(v, target):
    # binary search for lambda
    lo, hi = -1.0, 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # small correction for numeric
    diff = target - w.sum()
    if abs(diff) > 1e-10:
        # adjust elements not at bounds
        mask = (w > 0) & (w < 1)
        if mask.any():
            w[mask] += diff / mask.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

def upsample_and_project(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    target_sum = n_new / 2.0

    return h_new

def random_pair_move(h, max_step=0.02):

import time, math, random
import numpy as np

# ---------- helper functions ----------

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx for discrete h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def proj_box_sum(v, target):
    """
    Euclidean projection of vector v onto the box [0,1]^n with sum(v)=target.
    Uses a simple bisection on the Lagrange multiplier λ.
    """
    lo, hi = -1.0, 2.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        mask = (w > 0.0) & (w < 1.0)
        if mask.any():
            w[mask] += diff / mask.sum()
            w = np.clip(w, 0.0, 1.0)
    return w


def upsample_and_project(h_old, n_new):
    """Linear interpolation of h_old to n_new points and projection onto constraints."""
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    target = n_new / 2.0
    h_new = proj_box_sum(h_new, target)
    return h_new


def random_pair_move(h, max_step):
    """
    Propose a move that preserves the total sum:
    pick i!=j and add delta to h[i] while subtracting from h[j].
    Returns a new array or (None, None) if no feasible delta.
    """
    n = len(h)
    i, j = random.sample(range(n), 2)
    # feasible delta range from the bounds of both entries
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if lo >= hi:
        return None, None
    delta = (random.random() * 2.0 - 1.0) * max_step
    delta = max(min(delta, hi), lo)
    if abs(delta) < 1e-14:
        return None, None
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new, delta


def hill_climb(start_h, time_limit):
    """
    Simple simulated-annealing hill climber.
    Keeps the current point and the best seen point.
    """
    start = time.time()
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    n = len(cur_h)
    max_step_initial = 0.2   # initial maximal delta for a pair
    temp_initial = 0.02      # initial temperature for SA

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        frac = elapsed / time_limit

        # schedule step size and temperature
        max_step = max_step_initial * (1.0 - frac) ** 2 + 1e-4
        temp = max(temp_initial * (1.0 - frac), 1e-5)

        cand_h, _ = random_pair_move(cur_h, max_step)
        if cand_h is None:
            continue
        cand_val = compute_c5(cand_h)

        # Metropolis acceptance
        if cand_val < cur_val or random.random() < math.exp((cur_val - cand_val) / temp):
            cur_h, cur_val = cand_h, cand_val
            if cand_val < best_val:
                best_val, best_h = cand_val, cand_h.copy()
                # optional early exit if unbeatable bound reached
                if best_val < 0.3808:
                    break
    return best_h, best_val
