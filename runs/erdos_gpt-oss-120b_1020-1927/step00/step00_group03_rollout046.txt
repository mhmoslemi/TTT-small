def project_box_sum(y, S):
    low = -1.0; high = 2.0
    for _ in range(50):
        lam = (low+high)/2
        x = np.clip(y - lam, 0, 1)
        total = x.sum()
        if total > S:
            low = lam
        else:
            high = lam
    return np.clip(y - (low+high)/2, 0, 1)

def projected_step(h, shift, step):
    N = len(h)
    d = np.zeros(N)
    if shift >=0:
        # h[i+s] contributions
        d[:N-shift] += h[shift:] # h[i+s] = h[i+shift]; using d[i] plus to i range
        d[shift:] += h[:N-shift] # h[i-s] for i >= shift
    else:
        # shift negative: similar
        s = -shift
        d[s:] += h[:N-s] # index i (>=s) gets h[i-s]
        d[:N-s] += h[s:] # index i (< N-s) gets h[i+s]
    # step parameter
    h_new = h + step * d
    # clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # project onto sum
    h_new = project_box_sum(h_new, N/2)
    return h_new

import numpy as np, time, math

def _eval_c5(h: np.ndarray) -> float:
    """Return the Erdős overlap value for a discretised step function."""
    n = h.shape[0]
    # full correlation of h with (1-h)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n

def _project_box_sum(y: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project y onto the set {x | 0 <= x_i <= 1, sum x_i = target_sum}
    using a simple bisection on the Lagrange multiplier.
    """
    low, high = -3.0, 3.0               # wide enough brackets
    for _ in range(60):                # enough for double precision
        lam = (low + high) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            low = lam
        else:
            high = lam
    lam = (low + high) * 0.5
    return np.clip(y - lam, 0.0, 1.0)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step function h:[0,2]->[0,1] with ∫h = 1
    to minimise C5 = max_k ∫ h(x)(1-h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise h
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # fallback: 120 equally spaced points
        n = 120
        h = rng.random(n)
    n = h.shape[0]

    # enforce feasibility
    h = np.clip(h, 0.0, 1.0)
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        h = _project_box_sum(h, target_sum)

    best_h = h.copy()
    best_val = _eval_c5(best_h)
    cur_h = h.copy()
    cur_val = best_val

    # ------------------------------------------------------------------
    # optimisation parameters
    start_time = time.time()
    deadline = start_time + budget_s * 0.95   # keep a safety margin

    # temperature schedule (for simulated‑annealing acceptance)
    T0, Tf = 3e-2, 1e-4
    # step‑size schedule for random mass transfers
    step0, stepf = 0.5, 0.01

    it = 0
    while time.time() < deadline:
        # progress ratio 0..1
        prog = (time.time() - start_time) / (deadline - start_time)
        T = T0 * (1.0 - prog) + Tf * prog
        step_factor = step0 * (1.0 - prog) + stepf * prog

        # --------------------------------------------------------------
        # 1) simple random mass transfer (preserves total mass)
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if rng.random() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        max_transfer = min(cur_h[donor], 1.0 - cur_h[receiver])
        if max_transfer > 0.0:
            delta = rng.random() * max_transfer * step_factor
            cand = cur_h.copy()
            cand[donor] -= delta
            cand[receiver] += delta
            cand_val = _eval_c5(cand)
            dval = cand_val - cur_val
            if dval < 0.0 or rng.random() < math.exp(-dval / (T + 1e-12)):
                cur_h, cur_val = cand, cand_val
                if cand_val < best_val:
                    best_h, best_val = cand.copy(), cand_val

        # --------------------------------------------------------------
        # 2) occasional gradient‑step on the worst shift
        if it % 100 == 0:
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            idx_max = int(np.argmax(corr))
            shift = idx_max - (n - 1)          # signed shift producing max overlap

            # direction that would increase the problematic autocorrelation
            d = np.zeros(n, dtype=float)
            s = shift
            if s >= 0:
                if s < n:
                    d[:n - s] += cur_h[s:]
                    d[s:]     += cur_h[:n - s]
            else:
                s_abs = -s
                if s_abs < n:
                    d[s_abs:] += cur_h[:n - s_abs]
                    d[:n - s_abs] += cur_h[s_abs:]

            if d.max() > 0.0:
                # scale so typical entry change is comparable to step_factor
                scale = step_factor / (d.max() + 1e-12)
                cand = cur_h + scale * d
                cand = np.clip(cand, 0.0, 1.0)
                cand = _project_box_sum(cand, target_sum)
                cand_val = _eval_c5(cand)
                dval = cand_val - cur_val
                if dval < 0.0 or rng.random() < math.exp(-dval / (T + 1e-12)):
                    cur_h, cur_val = cand, cand_val
                    if cand_val < best_val:
                        best_h, best_val = cand.copy(), cand_val

        it += 1

    # final safety projection (numerical drift)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_box_sum(best_h, target_sum)
    final_val = _eval_c5(best_h)

    return best_h, float(final_val), int(n)
