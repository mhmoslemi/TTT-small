def propose_move(h, max_step):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    # current values
    hi, hj = h[i], h[j]
    delta = (np.random.rand()*2 - 1) * max_step

    delta = np.clip(delta, -hj, 1-hi) # increase hi, decrease hj

    delta = np.random.uniform(lo, hi_bound)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def gradient_max_shift(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)  # index in array of length 2n-1
    s = max_idx - (n - 1)  # shift offset
    # Compute gradient
    g = np.ones_like(h)
    # subtract h_{i+s}
    if s >= 0:
        g[:n - s] -= h[s:]  # h_{i+s}
    else:
        # s negative -> i+s refers to earlier indices
        g[-s:] -= h[:n + s]
    # subtract h_{i-s}
    s_neg = -s
    if s_neg >= 0:
        g[:n - s_neg] -= h[s_neg:]
    else:
        g[-s_neg:] -= h[:n + s_neg]
    return g, max_idx, s

import numpy as np
import time
import math

def shifted_vector(h, offset):
    n = h.shape[0]
    res = np.zeros_like(h)
    if offset >= 0:
        if offset < n:
            res[:n - offset] = h[offset:]
    else:
        off = -offset
        if off < n:
            res[off:] = h[:n - off]
    return res

def gradient_max_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    shift = max_idx - (n - 1)
    h_plus = shifted_vector(h, shift)
    h_minus = shifted_vector(h, -shift)
    g = 1.0 - h_plus - h_minus
    return g, max_idx, shift

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_constraints(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) / 2, 0.0, 1.0)
    return proj

def local_optimize(h0, time_budget, rng):

                h = h_new

            new_score = compute_c5(h_new)
            if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(temp, 1e-12)):
                h = h_new
                cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        iteration += 1

            cand = best_h + pert
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_onto_constraints(cand, target)

            if cand_score < best_score:
                best_h = cand.copy()
                best_score = cand_score
                h = cand.copy()
                cur_score = cand_score
    return best_h, best_score

    # Perform several restarts within given budget
    best_h, best_score = local_optimize(h_init, time_budget, rng)
    return best_h, best_score

import numpy as np, time, math

def _shifted(v, off):
    n = v.shape[0]
    r = np.zeros_like(v)
    if off >= 0:
        if off < n:
            r[: n - off] = v[off:]
    else:
        o = -off
        if o < n:
            r[o:] = v[: n - o]
    return r


def _gradient_max(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    s = idx - (n - 1)                     # integer shift giving the max overlap
    g = 1.0 - _shifted(h, s) - _shifted(h, -s)
    return g, idx, s


def _c5(h):
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _project(v, target):
    lo, hi = -2.0, 2.0                     # brackets for binary search on λ
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(v - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _random_feasible(n, rng):
    return _project(rng.random(n), n / 2.0)


def _upsample(h, factor):
    n = len(h)
    new_n = n * factor
    old_x = np.arange(n) * (2.0 / n)
    new_x = np.arange(new_n) * (2.0 / new_n)
    h2 = np.interp(new_x, old_x, h)
    return _project(h2, new_n / 2.0)


def _local_optimize(h0, budget, rng):

    target = n / 2.0

    step, temp = 0.5, 0.01

            h_new = _project(h - step * g, target)
            sc = _c5(h_new)
