def gradient_step(h, step):
    # compute autocorr = correlate h with h
    auto_corr = np.correlate(h, h, mode='full')
    min_val = np.min(auto_corr)
    k_star = np.argmin(auto_corr) - (len(h)-1) # shift index
    # compute gradient: -dx*(shifted vectors)
    dx = 2.0 / len(h)

def compute_c5_and_worst_shift(h):
    auto_corr = np.correlate(h, h, mode='full')
    min_idx = np.argmin(auto_corr)
    min_val = auto_corr[min_idx]
    dx = 2.0 / len(h)
    c5 = dx * (len(h)/2 - min_val) # n/2 = len/2
    shift = min_idx - (len(h)-1) # shift index (may be negative)
    return c5, shift, min_val, auto_corr

def gradient_for_shift(h, shift):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    if shift >= 0:
        grad[:n - shift] -= dx * h[shift:]   # -dx*h_{j+shift}
        grad[shift:] -= dx * h[:n - shift]   # -dx*h_{j-shift}
    else:
        s = -shift
        grad[s:] -= dx * h[:n - s]          # -dx*h_{j - s}
        grad[:n - s] -= dx * h[s:]          # -dx*h_{j + s}
    return grad

def smooth_obj_and_grad(h, target, dx, eps):
    auto = np.correlate(h, h, mode='full')
    min_auto = np.min(auto)
    delta = auto - min_auto
    # shift indices correspond to -(n-1)..(n-1)
    w = np.exp(-delta / eps)
    w_sum = np.sum(w)
    # soft min
    soft_min = min_auto - eps * np.log(w_sum)
    c5 = dx * (target - soft_min)
    # weighted gradient numerator
    n = h.shape[0]
    weighted = np.zeros_like(h)
    # iterate over shifts
    for idx, wk in enumerate(w):
        if wk == 0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            s = shift
            if s < n:
                weighted[:n - s] += wk * h[s:]
                weighted[s:] += wk * h[:n - s]
        else:
            s = -shift
            if s < n:
                weighted[s:] += wk * h[:n - s]
                weighted[:n - s] += wk * h[s:]
    grad = -dx * weighted / w_sum
    return float(c5), grad

import time
import random
import numpy as np
from scipy.optimize import minimize

# global placeholders for the optimizer
_GLOBAL_TARGET = 0.0
_GLOBAL_DX = 0.0
_GLOBAL_EPS = 0.01


def project_to_capped_simplex(v, target):
    """Projection of v onto the hypercube [0,1] with sum == target."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5_and_shift(h):
    """Exact C5 and the shift that attains the maximum (worst) overlap."""
    n = h.shape[0]
    dx = 2.0 / n
    target = n / 2.0
    auto = np.correlate(h, h, mode="full")
    idx = int(np.argmin(auto))
    min_auto = auto[idx]
    shift = idx - (n - 1)          # integer shift (negative allowed)
    c5 = dx * (target - min_auto)  # = max_k ∫ h(1‑h(·+k))
    return float(c5), shift, float(min_auto), auto


def gradient_worst_shift(h, shift):
    """Gradient of C5 w.r.t. h when a particular shift is worst."""
    n = h.shape[0]
    dx = 2.0 / n
    grad = np.zeros_like(h)
    if shift >= 0:
        s = shift
        if s < n:
            grad[: n - s] -= dx * h[s:]
            grad[s:] -= dx * h[: n - s]
    else:
        s = -shift
        if s < n:
            grad[s:] -= dx * h[: n - s]
            grad[: n - s] -= dx * h[s:]
    return grad


def smooth_obj_and_grad(h, target, dx, eps):
    """Smoothed objective (soft‑max of the overlap) and its gradient."""
    n = h.shape[0]
    auto = np.correlate(h, h, mode="full")
    min_auto = np.min(auto)
    delta = auto - min_auto
    w = np.exp(-delta / eps)                     # weights for each shift
    w_sum = np.sum(w)
    soft_min = min_auto - eps * np.log(w_sum)    # smooth approximation of min auto‑correlation
    c5 = dx * (target - soft_min)

    # weighted gradient numerator
    weighted = np.zeros_like(h)
    for idx, wk in enumerate(w):
        if wk == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            s = shift
            if s < n:
                weighted[: n - s] += wk * h[s:]
                weighted[s:] += wk * h[: n - s]
        else:
            s = -shift
            if s < n:
                weighted[s:] += wk * h[: n - s]
                weighted[: n - s] += wk * h[s:]
    grad = -dx * weighted / w_sum
    return float(c5), grad


def obj_fun(h):
    """Objective for scipy.minimize – uses global parameters."""
    return smooth_obj_and_grad(h, _GLOBAL_TARGET, _GLOBAL_DX, _GLOBAL_EPS)[0]


def grad_fun(h):
    """Gradient for scipy.minimize – uses global parameters."""
    return smooth_obj_and_grad(h, _GLOBAL_TARGET, _GLOBAL_DX, _GLOBAL_EPS)[1]


def eq_constraint(h):
    """Equality constraint: sum(h) == target."""
    return np.sum(h) - _GLOBAL_TARGET


def eq_jacobian(h):
    """Jacobian of the equality constraint."""
    return np.ones_like(h)


def random_feasible(n, target):
    """Generate a random vector in [0,1]^n with given sum."""
    v = np.random.rand(n)
    return project_to_capped_simplex(v, target)


def subgradient_descent(h, max_iter, time_limit, start_time):
    """Simple projected sub‑gradient descent on the exact objective."""
    n = h.shape[0]
    target = n / 2.0
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        _, shift, _, _ = compute_c5_and_shift(h)
        grad = gradient_worst_shift(h, shift)
        step = 0.6 / (1.0 + 0.01 * it)      # diminishing stepsize
        h = project_to_capped_simplex(h - step * grad, target)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():

        n = h0.shape[0]
    else:
        n = 80

    target = n / 2.0
    dx = 2.0 / n
    h0 = project_to_capped_simplex(h0, target)

    best_h = h0.copy()
    best_c5, _, _, _ = compute_c5_and_shift(best_h)

    # ------------------------------------------------------------------
    # Phase 1 – deterministic sub‑gradient descent from the given start
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    h1 = subgradient_descent(best_h, max_iter=2000,
                             time_limit=remaining * 0.3,
                             start_time=start)
    c5_1, _, _, _ = compute_c5_and_shift(h1)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # ------------------------------------------------------------------
    # Phase 2 – smooth optimisation with SLSQP (uses global vars)
    # ------------------------------------------------------------------
    global _GLOBAL_TARGET, _GLOBAL_DX, _GLOBAL_EPS
    _GLOBAL_TARGET = target
    _GLOBAL_DX = dx
    _GLOBAL_EPS = 0.05                      # smoothing parameter

    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": eq_constraint, "jac": eq_jacobian}]

    remaining = budget_s - (time.time() - start)

        opt_res = minimize(obj_fun,

        c5_tmp, _, _, _ = compute_c5_and_shift(h_tmp)

        # optional refined optimisation if time permits
        time_rem = budget_s - (time.time() - start)

                               bounds=bounds,

            h_ref = opt_res.x
            h_ref = np.clip(h_ref, 0.0, 1.0)
            h_ref = project_to_capped_simplex(h_ref, target)
            c5_ref, _, _, _ = compute_c5_and_shift(h_ref)
            if c5_ref < best_c5:
                best_c5, best_h = c5_ref, h_ref.copy()
            # restore smoothing for next round
            _GLOBAL_EPS = 0.05

    # ------------------------------------------------------------------
    # Final projection and evaluation
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_capped_simplex(best_h, target)
    final_c5 = compute_c5_and_shift(best_h)[0]

    return best_h, float(final_c5), n
