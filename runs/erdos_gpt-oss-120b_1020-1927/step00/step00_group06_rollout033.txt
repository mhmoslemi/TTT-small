def project_box_simplex(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    # We can also set lo = -np.max(np.abs(v))-1 for safety.
    for _ in range(100):
        lam = (lo+hi)/2.0
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if abs(s - target) < 1e-9:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0, 1)

def neighbor(h):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # Choose delta uniform between -d and d but limited by bounds
    max_inc_i = 1 - h[i]
    max_dec_i = h[i] # can decrease
    max_inc_j = 1 - h[j]
    max_dec_j = h[j]
    # Transfer from i to j: delta from i to j (increase j, decrease i) or opposite.
    # Random delta limited such that both remain in [0,1].
    # Choose delta = random uniform(max(-max_dec_i, -max_inc_j), min(max_inc_i, max_dec_j))
    low = max(-max_dec_i, -max_inc_j)
    high = min(max_inc_i, max_dec_j)
    if low >= high:
        # can't move mass, do nothing
        return h
    delta = np.random.uniform(low, high)
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new
```

def project_box_simplex(v, target):
    # solve min ||x - v||^2 s.t. sum(x) = target, 0<=x<=1
    # Use binary search for lambda (Lagrange multiplier).
    # x_i = min(1, max(0, v_i - lam))
    lo = np.min(v) - 1.0
    hi = np.max(v)
    # adjust bounds to ensure sum above or below target.
    for _ in range(80):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) < 1e-10:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

import time
import numpy as np

def project_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {x: sum(x)=target, 0<=x<=1}."""
    # Bracket lambda: low makes all clipped to 1, high makes all 0.
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def evaluate(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx for the discrete vector."""
    n = h.size
    # autocorrelation of h with itself
    auto = np.correlate(h, h, mode="full")
    min_dot = auto.min()
    return 1.0 - 2.0 * min_dot / n   # equivalent to max overlap integral


def neighbor_swap(h: np.ndarray) -> np.ndarray:
    """Transfer a random amount of mass between two coordinates, preserving sum."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]

    max_inc_i = 1.0 - hi
    max_dec_i = hi
    max_inc_j = 1.0 - hj
    max_dec_j = hj

    low = max(-max_dec_i, -max_inc_j)
    high = min(max_inc_i, max_dec_j)
    if low >= high:
        return h  # cannot move any mass
    delta = np.random.uniform(low, high)  # amount moved from i to j (negative = opposite)
    h2 = h.copy()
    h2[i] = hi - delta
    h2[j] = hj + delta
    return h2


def neighbor_noise(h: np.ndarray, target: float, temp: float) -> np.ndarray:
    """Add Gaussian noise to all coordinates and re‑project."""
    sigma = max(1e-4, 0.15 * temp)
    cand = h + sigma * np.random.randn(h.size)
    return project_box_simplex(cand, target)


def neighbor_gradient(h: np.ndarray, target: float, temp: float) -> np.ndarray:
    """Take a small step in the direction that increases the minimal autocorrelation."""
    n = h.size
    auto = np.correlate(h, h, mode="full")
    min_dot = auto.min()
    # indices achieving the minimum
    idxs = np.where(auto == min_dot)[0]
    grad = np.zeros_like(h)

    for idx in idxs:
        shift = idx - (n - 1)
        if shift >= 0:
            # i with i+shift
            i_end = n - shift
            grad[:i_end] += h[shift:]
            grad[shift:] += h[:i_end]
        else:
            s = -shift
            i_end = n - s
            grad[s:] += h[:i_end]
            grad[:i_end] += h[s:]

    # small step: scale by temperature and a factor
    step = max(1e-4, 0.05 * temp)
    cand = h + step * grad
    # projection back to feasible region
    return project_box_simplex(cand, target)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ----- initialise problem size and starting vector -----------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 120                      # fallback size
        h0 = np.random.rand(n)

    target_sum = n / 2.0
    h0 = project_box_simplex(h0, target_sum)

    # ----- bookkeeping -------------------------------------------------
    best_h = h0.copy()
    best_score = evaluate(best_h)

    cur_h = h0.copy()
    cur_score = best_score

    # ----- simulated‑annealing parameters ------------------------------
    temp = 0.2                           # initial temperature
    final_temp = 1e-6
    decay = 0.9995
    no_improve = 0
    max_no_improve = 5000               # restart after this many non‑improving steps

    # ----- main optimisation loop ------------------------------------
    while time.time() - start < budget_s * 0.95:
        # Choose a mutation type
        r = np.random.rand()
        if r < 0.25:                     # global Gaussian perturbation
            cand = neighbor_noise(cur_h, target_sum, temp)
        elif r < 0.75:                   # simple mass swap
            cand = neighbor_swap(cur_h)
        else:                            # gradient‑guided move
            cand = neighbor_gradient(cur_h, target_sum, temp)

        cand_score = evaluate(cand)

        # Metropolis acceptance
        if cand_score < cur_score or np.exp((cur_score - cand_score) / max(temp, 1e-12)) > np.random.rand():
            cur_h = cand
            cur_score = cand_score

        # Keep the best ever seen
        if cand_score < best_score:
            best_h = cand.copy()
            best_score = cand_score
            no_improve = 0
        else:
            no_improve += 1

        # Temperature schedule
        temp = max(final_temp, temp * decay)

        # Restart if stuck
        if no_improve >= max_no_improve:
            # re‑initialise from the current best with a small jitter
            cur_h = best_h + 0.05 * np.random.randn(n)
            cur_h = project_box_simplex(cur_h, target_sum)
            cur_score = evaluate(cur_h)
            no_improve = 0

    # Return the best-found construction
    return best_h, best_score, n
