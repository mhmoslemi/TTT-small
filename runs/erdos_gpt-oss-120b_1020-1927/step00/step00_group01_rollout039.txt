    n = len(v)
    # initial clamp
    # We'll find lambda such that h_i = min(max(v_i - lambda, 0), 1) and sum = target.

def compute_gradient_and_max_idx(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)  # index
    shift = idx - (n - 1)  # could be negative
    # Build shifted arrays:
    # For i + shift within bounds
    t1 = np.zeros(n)
    if shift >= 0:
        t1[:n-shift] = h[shift:]
    else:
        t1[-shift:] = h[:n+shift]
    # t2 for i - shift
    t2 = np.zeros(n)
    if shift > 0:
        t2[shift:] = h[:n-shift]
    elif shift < 0:
        t2[:n+shift] = h[-shift:]
    # gradient = dx * (1 - t1 - t2)
    grad = dx * (1.0 - t1 - t2)
    return grad, idx

def project_onto_capped_simplex(v, target):
    # binary search for lambda
    # lambda in [low, high] where low gives sum>=target, high gives sum<=target
    low = -1e5
    high = 1e5
    for _ in range(60):
        lam = (low + high) / 2.0
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s > target:
            low = lam
        else:
            high = lam
    # final lambda average
    lam = (low + high) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    # Due to floating error, adjust:
    # If sum not exactly target, adjust by adding/subtract small epsilon to some middle elements
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute diff to entries not at bounds, proportionally.
        # Find indices where 0 < h < 1
        free = np.where((h > 1e-12) & (h < 1-1e-12))[0]
        if free.size > 0:
            # Add diff equally
            inc = diff / free.size
            h[free] += inc
            # Clip again to be safe
            h = np.clip(h, 0.0, 1.0)
        else:
            # Adjust by adding to all and renormalize
            h += diff / len(v)
            h = np.clip(h, 0.0, 1.0)
    # Final enforce sum with small adjustment
    # compute final diff
    final_diff = target - h.sum()
    if abs(final_diff) > 1e-8:
        # adjust first element within bounds
        i = np.argmin(np.abs(h - 0.5))  # arbitrary
        h[i] = np.clip(h[i] + final_diff, 0.0, 1.0)
    return h

def project_onto_capped_simplex(v, target):
    # Use binary search for lambda:
    low = -1e6
    high = 1e6
    for _ in range(60):
        lam = (low + high) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(v - ((low+high)/2.0), 0.0, 1.0)

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper: project a vector onto the intersection of the hyperplane
#   sum(v) = target  and the box 0 <= v_i <= 1.
# It uses a simple binary search on the Lagrange multiplier λ.
def _project_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    w = np.clip(v - lam, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = np.where((w > 1e-12) & (w < 1 - 1e-12))[0]
        if free.size:
            incr = diff / free.size
            w[free] += incr
            w = np.clip(w, 0.0, 1.0)
        else:
            w += diff / v.size
            w = np.clip(w, 0.0, 1.0)
    return w


# ----------------------------------------------------------------------
# Compute the Erdős C5 value for a given step vector.
def _c5_val(h: np.ndarray) -> float:
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


# ----------------------------------------------------------------------
# Gradient of the active (max) shift term.
def _grad_active_shift(h: np.ndarray):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # signed shift that attains the max
    # t_plus[i] = h[i+shift] (0 where out of range)
    t_plus = np.zeros_like(h)
    if shift >= 0:
        t_plus[: n - shift] = h[shift:]
    else:
        t_plus[-shift:] = h[: n + shift]
    # t_minus[i] = h[i-shift] (0 where out of range)
    t_minus = np.zeros_like(h)
    if shift <= 0:
        m = -shift
        t_minus[: n - m] = h[m:]
    else:
        t_minus[shift:] = h[: n - shift]
    # gradient of the active term (to be subtracted)
    grad = dx * (1.0 - t_plus - t_minus)
    return grad, idx

    """
    Returns a step function h (as a NumPy array), its C5 bound, and the number
    of discretisation points used.
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Choose a (even) number of points – larger values give more flexibility.
    N = 200                     # can be changed, 200 is fast enough
    if N % 2:
        N += 1
    target_sum = N / 2.0        # integral = 1  =>  sum(h) = N/2

    # ------------------------------------------------------------------
    # Construct an initial vector.
    # If a parent solution exists we up‑sample it, otherwise start random.
    if "initial_h_values" in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        # Upsample / interpolate to the desired size
        x_old = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, N, endpoint=False)
        h = np.interp(x_new, x_old, parent)
    else:
        h = np.random.rand(N)

    # Force feasibility (bounds and integral)
    h = _project_capped_simplex(h, target_sum)

    best_h = h.copy()
    best_val = _c5_val(best_h)

    # ------------------------------------------------------------------
    # Parameters for the hybrid optimisation
    max_gd_iters = 4000                 # gradient–descent phase
    max_sa_iters = 40000                # simulated‑annealing (pair swaps)
    lr_initial = 0.12
    temp_initial = 0.15

    # ------------------------------------------------------------------
    # 1️⃣ Gradient‑descent phase (sub‑gradient on the active shift)
    for it in range(max_gd_iters):
        if time.time() - start_time > budget_s - 5:

        grad, _ = _grad_active_shift(h)

        # Simple schedule: linearly decreasing step size
        lr = lr_initial * (1.0 - it / max_gd_iters)
        h_candidate = _project_capped_simplex(h - lr * grad, target_sum)

        val = _c5_val(h_candidate)
        if val < best_val:

            best_h = h_candidate.copy()
        # accept if improvement, otherwise keep old point
        if val <= _c5_val(h):
            h = h_candidate

    # ------------------------------------------------------------------
    # 2️⃣ Simulated‑annealing / pair‑swap phase.
    # Randomly move mass between two positions while preserving the sum.
    cur_val = _c5_val(h)
    it = 0
    while time.time() - start_time < budget_s - 5 and it < max_sa_iters:
        # temperature decreases linearly to a small floor
        elapsed = time.time() - start_time
        temp = max(1e-4, temp_initial * (1.0 - elapsed / budget_s))

        i, j = np.random.choice(N, 2, replace=False)

        # Decide direction of transfer
        if np.random.rand() < 0.5:

            max_delta = min(1.0 - h[i], h[j])

            if max_delta <= 0.0:
                it += 1

        h_new = h.copy()

        if val_new < cur_val or np.random.rand() < math.exp(-(val_new - cur_val) / temp):
            h = h_new
            cur_val = val_new
            if val_new < best_val:

                best_h = h_new.copy()

        it += 1

    # Final projection to guarantee feasibility (tiny numerical drift)
    best_h = _project_capped_simplex(best_h, target_sum)

    # Use the provided evaluation routine for the final reported value
    final_c5 = evaluate_erdos_solution(best_h)
