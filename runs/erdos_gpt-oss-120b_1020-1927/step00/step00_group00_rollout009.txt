import time, random, math
import numpy as np

def _project_to_sum_and_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {x : 0<=x<=1, sum(x)=target}."""
    lo, hi = -1e6, 1e6
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Exact C5 value for the discrete step function h."""
    n = h.size
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * (2.0 / n))


def _subgradient_step(h: np.ndarray, step: float) -> np.ndarray:
    """One projected sub‑gradient step for minimizing max_{k} ∫h(1‑h_{·+k})."""
    n = h.size
    conv = np.correlate(h, 1.0 - h, mode="full")
    max_val = conv.max()
    # pick the first lag achieving the maximum
    idx = int(np.where(np.abs(conv - max_val) < 1e-12)[0][0])
    L = idx - (n - 1)

    if L >= 0:
        forward = np.concatenate([h[L:], np.zeros(L)])
        backward = np.concatenate([np.zeros(L), h[: n - L]])
    else:
        k = -L
        forward = np.concatenate([np.zeros(k), h[: n - k]])
        backward = np.concatenate([h[k:], np.zeros(k)])
    grad = 1.0 - forward - backward        # ∇ conv(L)
    new_h = h - step * grad                # move opposite to gradient
    target = n * 0.5
    return _project_to_sum_and_bounds(new_h, target)
