import time
import numpy as np

def _project_to_bounds_and_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {0<=x<=1, sum(x)=target_sum}."""
    lo = np.min(v) - 1.0        # ensures clip(v-lo) = 1 for all entries
    hi = np.max(v)              # ensures clip(v-hi) = 0 for all entries
    for _ in range(60):        # binary search for λ
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        interior = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        cnt = interior.sum()
        if cnt:
            proj[interior] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of  O_s(h)=∑ h_i (1‑h_{i+shift})  w.r.t. h  (including constant part)."""
    n = h.size
    g = np.ones(n)
    if shift >= 0:
        # forward term h_{i+shift}
        g[: n - shift] -= h[shift:]
        # backward term h_{i-shift}
        g[shift:] -= h[: n - shift]
    else:
        s = -shift
        # forward term h_{i-s}
        g[s:] -= h[: n - s]
        # backward term h_{i+s}
        g[: n - s] -= h[s:]
    return g


def _evaluate(h: np.ndarray) -> float:
    """Return C5 for the discrete vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # 1.  Choose resolution and build an initial feasible vector
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values", None)
    target_n = 180                         # a reasonable compromise
    target_sum = target_n / 2.0

    if init is not None:
        init = np.asarray(init, dtype=float)
        n0 = init.size
        # interpolate on a finer uniform grid
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False) + 1.0 / n0
        x_new = np.linspace(0.0, 2.0, target_n, endpoint=False) + 1.0 / target_n
        h = np.interp(x_new, x0, init)
    else:
        # uniform start if no parent is supplied
        h = np.full(target_n, 0.5, dtype=float)

    h = np.clip(h, 0.0, 1.0)
    h = _project_to_bounds_and_sum(h, target_sum)

    best_h = h.copy()
    best_val = _evaluate(best_h)

    # ------------------------------------------------------------------
    # 2.  Projected (sub)‑gradient descent with occasional random restarts
    # ------------------------------------------------------------------
    step = 0.5
    min_step = 1e-5
    no_improve = 0

    while time.time() - start < budget_s * 0.95:
        # correlation and worst shift(s)
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_max = corr.max()
        idx_max = np.where(corr >= cur_max - 1e-12)[0]   # indices achieving maximal value
        shifts = idx_max - (h.size - 1)

        # sub‑gradient: average over all maximal shifts
        grad = np.zeros_like(h)
        for s in shifts:
            grad += _gradient_for_shift(h, int(s))
        grad /= max(1, shifts.size)

        # descent step
        trial = h - step * grad
        trial = _project_to_bounds_and_sum(trial, target_sum)
        val = _evaluate(trial)

        if val < best_val - 1e-12:
            best_val = val
            best_h = trial.copy()
            h = trial
            step = min(step * 1.05, 1.0)
            no_improve = 0
            if best_val < 0.3800:            # good enough – can stop early
                break
        else:
            step = max(step * 0.5, min_step)
            no_improve += 1

        # occasional stochastic kick to escape stagnation
        if no_improve >= 2000:
            noise = rng.normal(scale=0.02, size=h.size)
            h_noise = best_h + noise
            h_noise = np.clip(h_noise, 0.0, 1.0)
            h_noise = _project_to_bounds_and_sum(h_noise, target_sum)
            val_noise = _evaluate(h_noise)
            if val_noise < best_val:
                best_val, best_h = val_noise, h_noise.copy()
                h = h_noise
                step = max(step, 0.05)
            no_improve = 0

    # ------------------------------------------------------------------
    # 3.  Return result
    # ------------------------------------------------------------------
    final_val = _evaluate(best_h)
    return best_h.tolist(), float(final_val), len(best_h)
