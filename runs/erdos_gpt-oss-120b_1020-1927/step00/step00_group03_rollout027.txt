def guided_move(h, step):
    N = len(h)
    # compute correlation to find Lmax
    corr = np.correlate(h, 1 - h, mode='full')
    L_idx = np.argmax(corr)
    L = L_idx - (N - 1)
    # compute gradient g
    g = np.zeros(N)
    for i in range(N):
        # term1: 1 - h[i+L] if i+L in [0,N-1] else 1
        if 0 <= i + L < N:
            term1 = 1.0 - h[i + L]
        else:
            term1 = 1.0
        # term2: h[i - L] if i-L in range else 0
        if 0 <= i - L < N:
            term2 = h[i - L]
        else:
            term2 = 0.0
        g[i] = term1 - term2
    # indices where g > 0 and g < 0
    pos_idx = np.where(g > 1e-9)[0]
    neg_idx = np.where(g < -1e-9)[0]
    if len(pos_idx) == 0 or len(neg_idx) == 0:
        # fallback to random transfer
        return False, None
    i = rng.choice(pos_idx)
    j = rng.choice(neg_idx)

    return True, new_h

import numpy as np, time, math

def _compute_c5(h, dx):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx  (discrete version)."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def _grad_overlap(h, lag):
    """Gradient of overlap O(lag) = Σ h[i]*(1‑h[i+lag]) w.r.t. h."""
    n = h.size
    if lag >= 0:
        # 1‑h[i+lag] part
        term1 = np.empty(n)
        term1[: n - lag] = 1.0 - h[lag:]
        term1[n - lag :] = 1.0
        # h[i‑lag] part (subtracted)
        term2 = np.empty(n)
        term2[lag:] = h[: n - lag]
        term2[:lag] = 0.0
    else:
        lag = -lag
        term1 = np.empty(n)
        term1[lag:] = 1.0 - h[: n - lag]
        term1[:lag] = 1.0
        term2 = np.empty(n)
        term2[: n - lag] = h[lag:]
        term2[n - lag :] = 0.0
    return term1 - term2  # derivative of O(lag)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # ---- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
        factor = 2                         # up‑sample a bit for flexibility
        h = np.repeat(base, factor)
    else:
        # fall back: binary half‑ones, half‑zeros
        n = 140
        h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)], dtype=float)
        rng.shuffle(h)
    n = h.size
    dx = 2.0 / n
    # guarantee exact integral (sum = n/2)
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        # adjust the first non‑boundary element
        for idx in range(n):
            if 0.0 <= h[idx] + diff <= 1.0:
                h[idx] += diff
                break

    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)

    # ---- simulated annealing -----------------------------------------
    start = time.time()
    # parameter schedules (linear annealing)
    T_start, T_end = 0.025, 1e-5          # temperature
    step_start, step_end = 0.30, 0.001    # maximal transfer amount
    max_iters = 10_000_000                # safety upper bound
    # probabilities
    p_guided = 0.55
    p_swap = 0.20
    # main loop
    cur_h = best_h.copy()
    cur_c5 = best_c5
    it = 0
    while it < max_iters:
        now = time.time()
        if now - start > budget_s * 0.95:
            break
        frac = (now - start) / (budget_s * 0.95)   # 0 → 1

        if r < p_guided:                     # guided transfer (based on worst lag)

            corr = np.correlate(cur_h, 1 - cur_h, mode="full")
            lag_idx = int(np.argmax(corr))
            lag = lag_idx - (n - 1)

            grad = _grad_overlap(cur_h, lag)   # derivative of overlap

                new_h[j] -= delta

                j = rng.choice(neg)
                max_dec = min(step, cur_h[i])          # we will decrease i
                max_inc = min(step, 1.0 - cur_h[j])    # increase j
                delta = min(max_dec, max_inc) * rng.random()
                if delta <= 0:

                new_h[i] -= delta
                new_h[j] += delta
        elif r < p_guided + p_swap:          # swap two entries

            new_h[i], new_h[j] = cur_h[j], cur_h[i]
        else:                                 # plain random transfer
            i, j = rng.choice(n, 2, replace=False)

            excess = cand.sum() - target_sum

                # final correction on a single index
                diff = target_sum - cand.sum()
                idx = rng.choice(n)
                cand[idx] = np.clip(cand[idx] + diff, 0.0, 1.0)
            cur_h = cand
            cur_c5 = _compute_c5(cur_h, dx)
            if cur_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5

        it += 1

    # final tiny refinement (deterministic gradient steps)
    refin_iters = 2000
    for _ in range(refin_iters):
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        lag_idx = int(np.argmax(corr))
        lag = lag_idx - (n - 1)
        grad = _grad_overlap(best_h, lag)
        pos = np.where(grad > 1e-12)[0]
        neg = np.where(grad < -1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = pos[np.argmax(grad[pos])]
        j = neg[np.argmin(grad[neg])]
        max_dec = min(0.01, best_h[i])
        max_inc = min(0.01, 1.0 - best_h[j])
        delta = min(max_dec, max_inc)
        if delta <= 0:
            break
        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c5 = _compute_c5(new_h, dx)
        if new_c5 < best_c5 - 1e-12:
            best_h = new_h
            best_c5 = new_c5
        else:
            break

    # final sanity‑check of the integral
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        # correct the smallest‑impact entry
        idx = np.argmin(np.abs(best_h - 0.5))
        best_h[idx] = np.clip(best_h[idx] + diff, 0.0, 1.0)

    final_c5 = _compute_c5(best_h, dx)
    return (best_h.tolist(), float(final_c5), n)
