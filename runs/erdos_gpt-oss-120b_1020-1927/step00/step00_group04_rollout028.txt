def gradient_descent(start_h, max_iters=2000):
    h = start_h.copy()
    N = len(h)
    target_sum = N/2
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    alpha = 0.2  # initial step
    for it in range(max_iters):
        # compute max shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        s = idx_max - (N - 1)
        # compute gradient
        grad = np.zeros(N)
        if s >= 0:
            grad[:N-s] = (1.0 - h[s:])
            grad[s:] -= h[:N-s]
        else:
            p = -s
            grad[p:] = (1.0 - h[:N-p])
            grad[:N-p] -= h[p:]
        # update
        step = alpha / np.sqrt(it+1)  # decaying
        new_h = h - step * grad
        # projection onto [0,1] and sum constraint
        new_h = project_onto_box_and_sum(new_h, target_sum)
        # compute new C
        new_c = compute_c5(new_h)
        # Accept if improvement else maybe accept if not too worse?
        if new_c < c or new_c <= c + 1e-6:
            h = new_h
            c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        # else reject: not updated
        # early stopping if c is very low?
    return best_h, best_c

def project_onto_sum_and_bounds(y, target):
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(y + (lo+hi)/2.0, 0.0, 1.0)

def descend(h, max_iters, alpha0):
    N = len(h)
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    h_curr = h.copy()
    c_curr = best_c
    for it in range(max_iters):
        # compute correlation
        corr = np.correlate(h_curr, 1 - h_curr, mode='full')
        max_idx = np.argmax(corr)
        s = max_idx - (N - 1)
        # compute gradient
        grad = np.zeros(N)
        if s >= 0:
            grad[:N - s] = 1.0 - h_curr[s:]
            grad[s:] -= h_curr[:N - s]
        else:
            p = -s
            grad[p:] = 1.0 - h_curr[:N - p]
            grad[:N - p] -= h_curr[p:]
        # step size schedule
        step = alpha0 / (1.0 + it * 0.01)  # some decreasing
        # attempt step with line search
        new_h = h_curr - step * grad
        # projection
        new_h = project_onto_sum_and_bounds(new_h, target)
        new_c = compute_c5(new_h)
        # accept if improvement or equal (maybe some tolerance)
        if new_c < c_curr - 1e-9:
            h_curr = new_h
            c_curr = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
        else:
            # could reduce step and try again a few times
            # attempt smaller step
            for factor in (0.5, 0.25, 0.125):
                step2 = step * factor
                new_h2 = h_curr - step2 * grad
                new_h2 = project_onto_sum_and_bounds(new_h2, target)
                new_c2 = compute_c5(new_h2)
                if new_c2 < c_curr - 1e-9:
                    h_curr = new_h2
                    c_curr = new_c2
                    if new_c2 < best_c:
                        best_c = new_c2
                        best_h = new_h2.copy()
                    break
        # termination early if c_curr is already below target bound
        if c_curr < 0.3808:
            break
    return best_h, best_c

    def project_onto_sum_and_bounds(y, target):
        # binary search for lambda shift
        lo, hi = -2.0, 2.0
        for _ in range(60):
            mid = (lo + hi) / 2.0
            x = np.clip(y + mid, 0.0, 1.0)
            s = x.sum()
            if s > target:
                hi = mid
            else:
                lo = mid
        x = np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)
        # final tweak to exactly match target (tiny residual)
        diff = target - x.sum()
        if abs(diff) > 1e-12:
            # adjust a non‑saturated element arbitrarily
            free = (x > 0.0) & (x < 1.0)
            if free.any():
                idx = np.where(free)[0][0]
                x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
            else:
                # all at bounds, distribute diff uniformly (should not happen)
                x += diff / len(x)
                x = np.clip(x, 0.0, 1.0)
        return x

                best_c, best_h = c_opt, h_opt
                sigma = max(0.03, sigma * 0.9)
            else:
                sigma = min(0.12, sigma * 1.02)
            sigma = max(0.02, sigma * 0.995)

        # ----- prepare for the next finer stage -----
        if N != stage_sizes[-1]:
            next_N = stage_sizes[stage_sizes.index(N) + 1]
            best_h = upsample_h(best_h, next_N)
            best_h = project_onto_sum_and_bounds(best_h, next_N * 0.5)
            best_c = compute_c5(best_h)

    # final sanity‑check
    N_final = best_h.shape[0]
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_sum_and_bounds(best_h, N_final * 0.5)
    final_c = compute_c5(best_h)

    return best_h, float(final_c), int(N_final)
