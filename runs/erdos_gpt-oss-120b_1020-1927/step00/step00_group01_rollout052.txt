def compute_gradients(h):
    n = len(h)
    # compute correlation only for needed s's.
    # Use np.correlate to get fvec = np.correlate(h, 1-h, mode='full')
    fvec = np.correlate(h, 1.0 - h, mode='full')
    max_val = fvec.max()
    idxs = np.flatnonzero(fvec == max_val)
    # convert idx to shift s: s = idx - (n - 1)
    shifts = idxs - (n - 1)
    grad = np.zeros(n)
    for s in shifts:
        # for i from 0..n-1
        # term1: if i+s in [0,n-1] then (1 - h[i+s])
        # term2: if i-s in [0,n-1] then -h[i-s]
        # So grad_i = (1 - h[i+s] if in range else 0) - (h[i-s] if in range else 0)
        # We can vectorize.
        # term1 array:
        term1 = np.zeros(n)
        if s >= 0:
            term1[:n-s] = 1.0 - h[s:]
        else:
            term1[-s:] = 1.0 - h[:n+s]
        term2 = np.zeros(n)
        if s >= 0:
            term2[s:] = h[:n-s]
        else:
            term2[:n+s] = h[-s:]
        grad += term1 - term2
    grad /= len(shifts)  # average
    return grad
```

import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ---------- projection onto [0,1] with sum = n/2 ----------
    def proj_to_simplex(v):
        n = v.shape[0]
        target = n / 2.0
        # binary search for lambda so that sum(clamp(v+lambda,0,1)) = target
        lo = -np.max(v)
        hi = 1.0 - np.min(v)
        for _ in range(60):
            mid = (lo + hi) / 2.0
            w = np.clip(v + mid, 0.0, 1.0)
            s = w.sum()
            if s < target:
                lo = mid
            else:
                hi = mid
        return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

    # ---------- objective and subgradient ----------
    def obj_and_grad(h):
        n = h.shape[0]
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        idxs = np.flatnonzero(corr == max_val)
        shifts = idxs - (n - 1)
        grad = np.zeros(n)
        for s in shifts:
            if s >= 0:
                grad[: n - s] += 1.0 - h[s:]
                grad[s:] -= h[: n - s]
            else:
                spos = -s
                grad[spos:] += 1.0 - h[: n - spos]
                grad[: n - spos] -= h[spos:]
        grad /= len(shifts)
        # scale by dx = 2/n because the true objective is dx * max_corr
        dx = 2.0 / n
        grad *= dx
        obj = max_val * dx
        return obj, grad

    # ---------- compute C5 directly ----------
    def compute_c5(h):
        n = h.shape[0]
        return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

    # ---------- initial feasible h ----------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # fallback: uniform 0.5 on 47 points (sum = n/2)
        h0 = np.full(47, 0.5, dtype=float)
    h0 = proj_to_simplex(h0)

    # ---------- keep the best solution ----------
    best_h = h0.copy()
    best_val, _ = obj_and_grad(best_h)

    # ---------- subgradient descent on a given start ----------
    def optimize(start_h):
        nonlocal best_h, best_val
        h = start_h.copy()
        n = h.shape[0]
        base_lr = 1.0          # initial learning rate
        min_lr = 0.001         # floor learning rate
        iter_cnt = 0
        max_iters = 10_000_000  # huge, we exit on time

        while time.time() < deadline and iter_cnt < max_iters:
            val, grad = obj_and_grad(h)
            # record global best
            if val < best_val:
                best_val = val
                best_h = h.copy()

            # learning rate schedule (linear decay)
            lr = max(min_lr, base_lr * (1.0 - iter_cnt / 300_000))

            # gradient step (move opposite to subgradient)
            h = h - lr * grad
            h = proj_to_simplex(h)

            iter_cnt += 1

            # occasional stochastic shake to escape plateaus
            if iter_cnt % 5_000 == 0:
                noise = rng.normal(scale=0.02, size=n)
                h = proj_to_simplex(h + noise)

        # final check for the last point
        val, _ = obj_and_grad(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()

    # ---------- optimise at original resolution ----------
    optimize(h0)

    # ---------- if time permits, try a finer resolution ----------
    if time.time() < deadline - 5:
        # double resolution by simple repetition (preserves sum)
        h_fine = np.repeat(best_h, 2)
        h_fine = proj_to_simplex(h_fine)
        optimize(h_fine)

    # final evaluation
    final_c5 = compute_c5(best_h)

    # safety: if the original construction was better, fall back
    orig_val = compute_c5(h0)
    if orig_val < final_c5:
        best_h = h0.copy()
        final_c5 = orig_val

    return best_h, float(final_c5), len(best_h)
