def _project_onto_bounds_and_sum(v, target):
    """Project vector v onto the box [0,1] with sum = target."""
    lo, hi = -1.0, np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny fix for rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        idx = np.where((x > 0) & (x < 1))[0]
        if idx.size:
            x[idx[0]] += diff
            x = np.clip(x, 0.0, 1.0)
    return x
