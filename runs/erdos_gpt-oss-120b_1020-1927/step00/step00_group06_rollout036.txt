    eps = 1e-12

            add_cap = np.minimum(add, 1.0 - x[free])

            added = add_cap.sum()
            x[free] += add_cap
            deficit -= added

            x[free] -= sub_cap
            excess -= removed

                break

import time, math
import numpy as np

def _project_to_constraints(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to exact sum `target` using a water‑filling scheme."""
    x = np.clip(v, 0.0, 1.0)
    cur = x.sum()
    eps = 1e-12

    if abs(cur - target) <= eps:
        return x

    if cur < target:                     # need to add mass
        deficit = target - cur
        while deficit > eps:
            free = np.where(x < 1.0 - eps)[0]
            if free.size == 0:
                break
            add = deficit / free.size
            add_cap = np.minimum(add, 1.0 - x[free])
            added = add_cap.sum()
            x[free] += add_cap
            deficit -= added
            if added < eps:
                break
    else:                                 # need to remove mass
        excess = cur - target
        while excess > eps:
            free = np.where(x > eps)[0]
            if free.size == 0:
                break
            sub = excess / free.size
            sub_cap = np.minimum(sub, x[free])
            removed = sub_cap.sum()
            x[free] -= sub_cap
            excess -= removed
            if removed < eps:
                break

    # final tiny adjustment to hit the target exactly
    diff = target - x.sum()
    if abs(diff) > eps:
        for idx in range(x.size):
            if 0.0 <= x[idx] + diff <= 1.0:
                x[idx] += diff
                break
    return x

    except Exception:                     # no supplied initial_h_values

        use_init = False
        n = int(kwargs.get("n_points", 200))

    target_sum = n * 0.5                     # because ∫h = 1  ⇒  sum(h)*dx =1, dx=2/n
    dx = 2.0 / n

    if use_init and init.size == n:
        h = init.copy()
        # enforce constraints precisely
        h = _project_to_constraints(h, target_sum)
