def project_simplex_box(v, target):
    w = np.clip(v, 0, 1)
    # We need sum w = target.
    # We'll adjust by moving mass between free entries.
    # Use simple iterative scaling:
    for _ in range(10):
        s = w.sum()
        if abs(s - target) < 1e-12:
            break
        diff = s - target
        # Determine indices that are not at bounds in direction of adjustment
        if diff > 0:
            # Need to reduce sum; find indices where w > 0
            free = w > 0
            if free.sum() == 0:
                break
            # Subtract equally from free entries
            step = diff / free.sum()
            w[free] = np.maximum(w[free] - step, 0)
        else:  # diff < 0 need increase sum
            free = w < 1
            if free.sum() == 0:
                break
            step = -diff / free.sum()
            w[free] = np.minimum(w[free] + step, 1)
    # Final clip just in case
    return np.clip(w, 0, 1)

            lo = mid

    n = h.shape[0]

import numpy as np
import time

def _project_to_constraints(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {0<=x<=1, sum(x)=target} (Euclidean projection)."""
    lo = np.min(v) - 1.0          # low enough so that v - lo >= 1 for all entries
    hi = np.max(v)                # high enough so that v - hi <= 0 for all entries
    for _ in range(60):           # binary search for the correct λ
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    return w


def _worst_shift_and_gradient(h: np.ndarray, dx: float):
    """
    Return the shift that attains the maximal overlap,
    its (scaled) gradient and the maximal overlap value.
    """
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)

    grad = np.zeros_like(h)

    if shift >= 0:
        # i from 0 .. n-1-shift  -> term (1 - h[i+shift])
        end = n - shift
        if end > 0:
            grad[:end] += (1.0 - h[shift:])
        # i from shift .. n-1 -> term -h[i-shift]
        if shift < n:
            grad[shift:] -= h[:end]
    else:
        s = -shift
        end = n - s
        if end > 0:
            grad[s:] += (1.0 - h[:end])   # (1 - h[i-s]) for i>=s
            grad[:end] -= h[s:]          # -h[i+s]   for i< n-s

    grad *= dx                      # scale because the objective contains dx
    max_overlap = corr[idx] * dx
    return shift, grad, max_overlap


def _random_feasible(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a random point in the feasible region."""
    v = rng.random(n)
    return _project_to_constraints(v, target)

    Optimize the step function for the Erdős‑minimum‑overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    if "initial_h_values" in globals():

    else:

    h0 = _project_to_constraints(h0, target_sum)

    best_h = h0.copy()

    cur_h = h0.copy()

    no_improve = 0

    # ---- main optimisation loop ----------------------------------
    while time.time() - start_time < budget_s * 0.95:

        shift, grad, worst = _worst_shift_and_gradient(cur_h, dx)
