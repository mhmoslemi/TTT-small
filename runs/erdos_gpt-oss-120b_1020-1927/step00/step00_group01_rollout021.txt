def gradient_for_shift(h, s):
    n = len(h)
    g = np.ones_like(h)
    # right term: h_{i+s}, if i+s < n
    if s >= 0:
        # for i from 0 to n-1-s:
        g[:n-s] -= h[s:]  # subtract h[i+s]
        # left term: h_{i-s}, for i>= s
        g[s:] -= h[:n-s]  # subtract h[i-s]
    else:
        # s negative => -s positive
        d = -s
        # right term: i+s >=0 => i >= d
        g[d:] -= h[:n-d]
        # left term: i-s < n => i < n-d
        g[:n-d] -= h[d:]
    return g

Check for edge cases: When s=0, gradient = 1 - 2*h_i (since right term and left term both equal h_i). So gradient = 1 - 2h_i, which matches derivative of C(0) = sum_i h_i (1 - h_i) = sum_i (h_i - h_i^2). Derivative = 1 - 2h_i. Good.

import time
import random
import numpy as np

def _project_onto_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto the set {0<=x<=1, sum(x)=target}."""
    lo = -np.max(v) - 1.0          # makes all v+lo <= 0
    hi = 1.0 - np.min(v) + 1.0     # makes all v+hi >= 1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _gradient_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Subgradient of C5(t) = max_k sum_i h_i(1-h_{i+k}) at the active shift."""
    n = h.shape[0]
    g = np.ones_like(h)
    if shift >= 0:
        # term h_{i+shift}
        g[: n - shift] -= h[shift:]
        # term h_{i-shift}
        g[shift:] -= h[: n - shift]
    else:
        d = -shift
        g[d:] -= h[: n - d]
        g[: n - d] -= h[d:]
    return g


def _c5_value(h: np.ndarray, dx: float) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx


def _max_shift(conv: np.ndarray, n: int) -> int:
    """Return the integer shift (positive/negative) where conv attains its maximum."""
    idx = int(np.argmax(conv))
    return idx - (n - 1)


def _run_one(start_h: np.ndarray, dx: float, target_sum: float,
             time_deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent from start_h under the time budget."""
    n = start_h.shape[0]
    h = start_h.copy()
    best_h = h.copy()
    best_val = _c5_value(h, dx)

    step = 0.3
    decay_no_improve = 0

    # limit to a reasonable number of iterations; break early if out of time
    for _ in range(12000):
        if time.time() > time_deadline:
            break

        conv = np.correlate(h, 1.0 - h, mode='full')
        shift = _max_shift(conv, n)
        g = _gradient_shift(h, shift)

        # take a step and re‑project
        v = h - step * g
        h_new = _project_onto_box_sum(v, target_sum)

        val_new = _c5_value(h_new, dx)
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()

        if val_new < _c5_value(h, dx):
            h = h_new
            decay_no_improve = 0
            step = min(step * 1.04, 1.0)      # accelerate a bit when improving
        else:
            decay_no_improve += 1
            step *= 0.85                     # shrink step when no progress
            if decay_no_improve > 250:
                # random jitter to escape a plateau
                noise = (np.random.rand(n) - 0.5) * step
                h = _project_onto_box_sum(h + noise, target_sum)
                decay_no_improve = 0

        if step < 1e-6:
            break

    return best_h, best_val


def _binary_hill_climb(h_bin: np.ndarray, dx: float,
                       time_deadline: float) -> tuple[np.ndarray, float]:
    """Simple swap‑based hill climbing for a binary (0/1) vector."""
    n = h_bin.shape[0]
    ones = int(h_bin.sum())
    best = h_bin.copy()
    best_val = _c5_value(best, dx)

    # pre‑compute indices of ones/zeros for faster sampling
    idx_ones = np.flatnonzero(best)
    idx_zeros = np.flatnonzero(1 - best)

    while time.time() < time_deadline:
        if len(idx_ones) == 0 or len(idx_zeros) == 0:
            break
        i = random.choice(idx_ones)
        j = random.choice(idx_zeros)

        cand = best.copy()
        cand[i] = 0
        cand[j] = 1

        val = _c5_value(cand, dx)
        if val < best_val:
            best = cand
            best_val = val
            idx_ones = np.flatnonzero(best)
            idx_zeros = np.flatnonzero(1 - best)

        # occasional random restart to explore new basins
        if random.random() < 0.001:
            cand = np.zeros(n)
            cand[:ones] = 1
            np.random.shuffle(cand)
            val = _c5_value(cand, dx)
            if val < best_val:
                best = cand
                best_val = val
                idx_ones = np.flatnonzero(best)
                idx_zeros = np.flatnonzero(1 - best)

    return best, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of points."""
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()

    # grid size – 200 points give a good trade‑off between quality and speed
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # ------------------------------------------------------------------
    # Build a list of diverse initial vectors
    # ------------------------------------------------------------------
    candidates = []

    # 1) use the provided parent solution if it exists
    if 'initial_h_values' in globals():

                x_tgt = np.linspace(0.0, 2.0, n_points, endpoint=False)
                interp = np.interp(x_tgt, x_src, init)
                interp = np.clip(interp, 0.0, 1.0)
                interp = _project_onto_box_sum(interp, target_sum)
                candidates.append(interp)
        except Exception:
            pass

    # 2) a few random binary patterns (exactly n/2 ones)
    for _ in range(5):
        arr = np.zeros(n_points, dtype=float)
        arr[: int(target_sum)] = 1.0
        np.random.shuffle(arr)
        candidates.append(arr)

    # 3) a few random fractional patterns
    for _ in range(3):
        arr = np.random.rand(n_points)
        arr = _project_onto_box_sum(arr, target_sum)
        candidates.append(arr)

    # ------------------------------------------------------------------
    # Main optimisation loop over candidates
    # ------------------------------------------------------------------
    best_global_h = None
    best_global_val = float('inf')

    for start_h in candidates:
        if time.time() > end_time:
            break

        # 1) projected sub‑gradient descent
        deadline = min(end_time, time.time() + 7.0)  # give each start ~7 s
        h_opt, val_opt = _run_one(start_h, dx, target_sum, deadline)

        if val_opt < best_global_val:
            best_global_val = val_opt
            best_global_h = h_opt.copy()

        # 2) try a binary refinement from the best fractional result
        #    (convert to binary by keeping the largest entries)
        binary = np.zeros_like(h_opt, dtype=float)
        idx = np.argpartition(-h_opt, int(target_sum))[: int(target_sum)]
        binary[idx] = 1.0

        deadline_bin = min(end_time, time.time() + 3.0)
        bin_h, bin_val = _binary_hill_climb(binary, dx, deadline_bin)

    # final safety check – if nothing succeeded, fall back to a simple pattern
    if best_global_h is None:
        best_global_h = np.full(n_points, 0.5)
        best_global_val = _c5_value(best_global_h, dx)

    # ------------------------------------------------------------------
    # Return as required: list of values, bound, and n_points
    # ------------------------------------------------------------------
    return (best_global_h.tolist(), float(best_global_val), n_points)
