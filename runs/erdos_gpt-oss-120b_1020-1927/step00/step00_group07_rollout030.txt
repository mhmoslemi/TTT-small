    import numpy as np, time, math, random, itertools, cvxpy as cp
    random.seed(seed)
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = len(h)
    else:
        n = 10
        # ensure n even
        if n % 2 != 0:
            n+=1
        # simple random start meeting sum constraint
        h = np.random.rand(n)
        # project onto sum constraint using water-filling
        target = n/2
        # scale to have sum target then clip to [0,1]
        h = np.clip(h, 0, 1)
        # use scaling method: adjust by constant then clip iteratively
        # but easier: use optimization:
        # solve linear program to find feasible h close to random
        # We'll just set half entries to 1 and others to value to satisfy sum.
        ones = np.ones(n//2)
        zeros = np.zeros(n - n//2)
        h = np.concatenate([ones, zeros])
        np.random.shuffle(h)

    # Ensure constraints
    N = len(h)
    dx = 2.0 / N
    # Project onto constraints via water-filling
    def project(v):
        target = N/2
        lo = v.min() - 1.0
        hi = v.max()
        for _ in range(50):
            lam = (lo + hi)/2
            proj = np.clip(v - lam, 0.0, 1.0)
            s = proj.sum()
            if s > target:
                lo = lam
            else:
                hi = lam
        return np.clip(v - (lo+hi)/2, 0.0, 1.0)

    h = project(h)

    # compute c5 bound
    corr = np.correlate(h, 1 - h, mode='full') * dx
    c5 = float(corr.max())
    # optional further improvement: simple local search
    start = time.time()
    best_h = h.copy()
    best_c5 = c5
    # hill climbing for limited time
    while time.time() - start < budget_s - 1.0:
        # propose random swap of mass between two indices
        i, j = random.sample(range(N), 2)
        # move delta from i to j
        max_transfer = min(best_h[i], 1 - best_h[j])
        if max_transfer <= 1e-8:
            continue
        delta = random.random() * max_transfer
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # compute correlation
        corr_new = np.correlate(h_new, 1 - h_new, mode='full') * dx
        c5_new = float(corr_new.max())
        if c5_new < best_c5 - 1e-6:
            best_c5 = c5_new
            best_h = h_new
            # optional: continue

import time
import random
import itertools
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the feasible set {0<=h_i<=1, sum(h)=N/2}
def _project_to_simplex(v, target):
    """Project vector v onto the box [0,1] and the hyperplane sum = target."""
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(50):          # binary search for the right offset
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


# ----------------------------------------------------------------------
def _c5_value(h, dx):
    """Return the C5 value for a step vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Find a step function h on [0,2] that (approximately) minimises
    C5 = max_k ∫ h(x)(1‑h(x+k)) dx.

    The function returns (h_values, c5_bound, n_points) where
    h_values is a list of the discretised values, c5_bound the achieved
    bound, and n_points the number of samples used.
    """
    random.seed(seed)
    np.random.seed(seed)

    # --------------------------------------------------------------
    # 1. Load the parent construction if it exists.
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float)
        N = len(h)
    else:
        # fallback: a small even N with a simple feasible start
        N = 12                               # modest size, easy to optimise
        # initialise with half ones, half zeros and shuffle
        h = np.concatenate([np.ones(N // 2), np.zeros(N - N // 2)])
        np.random.shuffle(h)

    # --------------------------------------------------------------
    # 2. Enforce the constraints exactly.
    dx = 2.0 / N
    target_sum = N / 2.0
    h = _project_to_simplex(h, target_sum)

    # --------------------------------------------------------------
    # 3. Basic local optimisation (hill‑climbing with mass swaps).
    best_h = h.copy()
    best_c5 = _c5_value(best_h, dx)

    start_time = time.time()
    # simple simulated‑annealing style schedule
    while time.time() - start_time < budget_s - 1.0:
        # pick two distinct indices
        i, j = random.sample(range(N), 2)

        # amount we can move from i to j without leaving the box
        max_move = min(best_h[i], 1.0 - best_h[j])
        if max_move <= 1e-9:
            continue
        # random step size (geometric cooling could be added)
        delta = random.random() * max_move

        candidate = best_h.copy()
        candidate[i] -= delta
        candidate[j] += delta

        # project back onto the feasible set (cheap because only two entries changed)
        # we only need to restore the total sum; a simple offset works.
        deficit = candidate.sum() - target_sum
        if abs(deficit) > 1e-12:
            # distribute the deficit uniformly over free coordinates
            free = np.where((candidate > 1e-12) & (candidate < 1.0 - 1e-12))[0]
            if free.size:
                candidate[free] -= deficit / free.size
                candidate = np.clip(candidate, 0.0, 1.0)
                # final fix of the sum
                candidate = _project_to_simplex(candidate, target_sum)

        cand_c5 = _c5_value(candidate, dx)

        # accept improvements
        if cand_c5 < best_c5 - 1e-8:
            best_h, best_c5 = candidate, cand_c5

    # --------------------------------------------------------------
    # 4. Return the best solution found.
    return (best_h.tolist(), best_c5, N)
