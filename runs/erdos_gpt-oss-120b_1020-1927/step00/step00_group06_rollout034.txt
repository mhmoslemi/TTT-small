def project_onto_capped_simplex(v, target):
    # v: ndarray
    # returns w <=1, >=0, sum = target
    w = np.clip(v, 0.0, 1.0)
    # compute diff
    max_iter = 100
    for _ in range(max_iter):
        s = w.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase some below 1
            mask = w < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            inc = diff / cnt
            w[mask] += inc
            # Clip again to respect upper bound
            w = np.clip(w, 0.0, 1.0)
        else:
            mask = w > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            dec = -diff / cnt
            w[mask] -= dec
            w = np.clip(w, 0.0, 1.0)
    # final adjustment
    # adjust rounding errors: shift small residual
    s = w.sum()
    if abs(s - target) > 1e-10:
        # adjust uniformly on non-saturated entries
        diff = target - s
        mask = (w > 0.0) & (w < 1.0)
        cnt = mask.sum()
        if cnt > 0:
            w[mask] += diff / cnt
            w = np.clip(w,0.0,1.0)
    return w

import time
import math
import numpy as np

def _project_capped_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project onto the set {0<=x<=1, sum(x)=target_sum}."""
    w = np.clip(v, 0.0, 1.0)
    for _ in range(100):
        s = w.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = w < 1.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            inc = diff / cnt
            w[mask] += inc
            w = np.clip(w, 0.0, 1.0)
        else:
            mask = w > 0.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            dec = -diff / cnt
            w[mask] -= dec
            w = np.clip(w, 0.0, 1.0)
    # final tiny adjustment if needed
    s = w.sum()
    if abs(s - target_sum) > 1e-9:
        diff = target_sum - s
        mask = (w > 0.0) & (w < 1.0)
        cnt = int(mask.sum())
        if cnt:
            w[mask] += diff / cnt
            w = np.clip(w, 0.0, 1.0)
    return w


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Find a step function h on [0,2] with ∫h=1 that minimizes
        C5 = max_k ∫ h(x)(1-h(x+k)) dx .
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    max_time = budget_s * 0.95          # leave a safety margin

    # ------------------------------------------------------------------
    # initialise problem size and starting vector
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=np.float64)
        n = init.size
    except NameError:
        n = 200
        init = None

    target_sum = n / 2.0
    if init is not None:
        h = _project_capped_simplex(np.clip(init, 0.0, 1.0), target_sum)
    else:
        h = _project_capped_simplex(rng.random(n), target_sum)

    # ------------------------------------------------------------------
    # helpers
    # ------------------------------------------------------------------
    def _c5(val):
        # evaluate_erdos_solution is provided by the harness
        return evaluate_erdos_solution(val)

    # initial score
    cur_score = _c5(h)
    best_h = h.copy()
    best_score = cur_score

    # target bound – stop early if reached
    TARGET = 0.3808

    # parameters for the simulated annealing / local search
    init_temp = 0.04
    step_scale = 0.5
    iter_count = 0

    # current worst shift (lag) – recomputed after each accepted move
    cur_corr = np.correlate(h, 1.0 - h, mode="full")
    cur_shift = int(cur_corr.argmax()) - (n - 1)

    while True:
        # time check
        if time.time() - start_time >= max_time:
            break

        # decide move type (targeted vs random)
        use_targeted = rng.random() < 0.65 and n > 2
        if use_targeted:
            shift = cur_shift
            # Build candidate indices that can decrease the current worst count
            if shift >= 0:
                lim = n - shift
                idx = np.arange(lim)
                bad = idx[(h[idx] > 1e-12) & (h[idx + shift] < 1.0 - 1e-12)]
                good = idx[(h[idx] < 1.0 - 1e-12) & (h[idx + shift] > 1e-12)]
            else:
                s = -shift
                idx = np.arange(s, n)
                bad = idx[(h[idx] > 1e-12) & (h[idx - s] < 1.0 - 1e-12)]
                good = idx[(h[idx] < 1.0 - 1e-12) & (h[idx - s] > 1e-12)]
            if bad.size and good.size:
                i = int(rng.choice(bad))
                j = int(rng.choice(good))
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta > 1e-12:
                    delta = max_delta * rng.random() * step_scale
                    if delta > 0.0:
                        # propose move i -> j
                        h[i] -= delta
                        h[j] += delta
                        new_score = _c5(h)
                        accept = False
                        if new_score < cur_score:
                            accept = True
                        else:
                            # simulated annealing acceptance
                            temp = init_temp * max(0.0, 1.0 - (time.time() - start_time) / max_time)
                            if temp > 0 and rng.random() < math.exp(-(new_score - cur_score) / temp):
                                accept = True
                        if accept:
                            cur_score = new_score
                            if new_score < best_score:
                                best_score = new_score
                                best_h = h.copy()
                                if best_score <= TARGET:
                                    break
                            # recompute worst shift for the new vector
                            cur_corr = np.correlate(h, 1.0 - h, mode="full")
                            cur_shift = int(cur_corr.argmax()) - (n - 1)
                        else:
                            # revert
                            h[i] += delta
                            h[j] -= delta
                        iter_count += 1
                        continue  # go to next iteration
            # fallback to random if targeted failed
        # -------------------- random move --------------------
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # decide direction of transfer
        if rng.random() < 0.5:
            # from j to i
            max_delta = min(1.0 - h[i], h[j])
            if max_delta <= 1e-12:
                continue
            delta = max_delta * rng.random() * step_scale
            if delta <= 0.0:
                continue
            h[i] += delta
            h[j] -= delta
        else:
            # from i to j
            max_delta = min(1.0 - h[j], h[i])
            if max_delta <= 1e-12:
                continue
            delta = max_delta * rng.random() * step_scale
            if delta <= 0.0:
                continue
            h[i] -= delta
            h[j] += delta

        new_score = _c5(h)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            temp = init_temp * max(0.0, 1.0 - (time.time() - start_time) / max_time)

        if accept:
            cur_score = new_score
            if new_score < best_score:

            # update worst shift for next iteration
            cur_corr = np.correlate(h, 1.0 - h, mode="full")
            cur_shift = int(cur_corr.argmax()) - (n - 1)
        else:
            # revert move
            if rng.random() < 0.5:
                # we had i+=delta, j-=delta, revert
                h[i] -= delta
                h[j] += delta
            else:
                h[i] += delta
                h[j] -= delta
        iter_count += 1

    # final projection to clean up tiny violations
    best_h = _project_capped_simplex(best_h, target_sum)
    final_score = _c5(best_h)

    return best_h.astype(np.float32), float(final_score), int(n)
