def anneal(h, max_time, start_temp=0.1, end_temp=1e-5):
    start = time.time()
    N = len(h)
    best_h = h.copy()
    best_val = compute_c5(h)
    current_h = h.copy()
    cur_val = best_val
    # schedule based on elapsed fraction
    while time.time() - start < max_time:
        # compute temperature
        fraction = (time.time() - start) / max_time
        T = start_temp * (1 - fraction) + end_temp * fraction  # linear cooling
        # propose swap
        i = np.random.randint(N)
        # find j of opposite
        # ensure hi != hj
        while True:
            j = np.random.randint(N)
            if current_h[j] != current_h[i]:
                break
        # swap
        current_h[i], current_h[j] = current_h[j], current_h[i]
        new_val = compute_c5(current_h)
        delta = new_val - cur_val
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            # accept
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h[:] = current_h
        else:
            # reject: swap back
            current_h[i], current_h[j] = current_h[j], current_h[i]
    return best_h, best_val
```

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a step function h on [0,2]."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _find_swap_for_lag(h: np.ndarray, lag: int):
    """
    For a given lag (shift index) find a pair (i,j) with h[i]=1, h[j]=0 such that
    swapping them reduces the contribution to the mismatch at this lag.
    Returns None if no such pair exists.
    """
    N = h.shape[0]
    if lag >= 0:
        i_start = 0
        i_end = N - lag
        i = np.arange(i_start, i_end, dtype=np.int64)
        t = i + lag
    else:  # lag < 0
        i_start = -lag
        i_end = N
        i = np.arange(i_start, i_end, dtype=np.int64)
        t = i + lag

    hi = h[i]
    ht = h[t]

    # indices where source is 1 and target is 0 (bad contribution)
    i_bad = i[(hi == 1) & (ht == 0)]
    # indices where source is 0 and target is 1 (potential improvement)
    j_bad = i[(hi == 0) & (ht == 1)]

    if i_bad.size == 0 or j_bad.size == 0:
        return None
    i_swap = int(np.random.choice(i_bad))
    j_swap = int(np.random.choice(j_bad))
    return i_swap, j_swap


def _greedy_improve(h0: np.ndarray, max_seconds: float):
    """
    Greedy hill‑climbing with targeted swaps.
    Returns the best vector found and its C5 value.
    """
    N = h0.shape[0]
    best_h = h0.copy()
    best_val = _compute_c5(best_h)

    cur_h = h0.copy()
    cur_val = best_val

    start = time.time()
    it_without_improve = 0
    while time.time() - start < max_seconds:
        # full correlation to locate the worst lag
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(corr))
        lag = max_idx - (N - 1)          # signed shift
        # try a directed swap that helps this lag
        swap = _find_swap_for_lag(cur_h, lag)
        if swap is None:
            # fallback: random opposite‑value pair
            i = np.random.randint(N)
            hi = cur_h[i]
            # find a j of opposite value (few attempts)
            for _ in range(10):
                j = np.random.randint(N)
                if cur_h[j] != hi:
                    break
            swap = (i, j)

        i, j = swap
        # apply swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_val = _compute_c5(cur_h)

        # accept only non‑worsening moves
        if new_val <= cur_val + 1e-12:
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = cur_h.copy()
                it_without_improve = 0
                # early exit if target already reached
                if best_val < 0.3808:
                    break
            else:
                it_without_improve += 1
        else:
            # revert the move
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            it_without_improve += 1

        # occasional random restart to escape dead ends
        if it_without_improve > 5000:
            cur_h = best_h.copy()
            cur_val = best_val
            it_without_improve = 0

    return best_h, best_val
