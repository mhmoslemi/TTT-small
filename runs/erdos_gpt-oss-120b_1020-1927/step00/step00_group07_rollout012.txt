import time
import numpy as np

def _binary_from_values(vals):
    """Return a binary vector with exactly half ones, taking the largest entries."""
    n = len(vals)
    half = n // 2
    idx = np.argsort(-vals)[:half]
    h = np.zeros(n, dtype=np.int8)
    h[idx] = 1
    return h

def _raw_max_corr(h):
    """Maximum of the directional overlap (integer count)."""
    return np.max(np.correlate(h, 1 - h, mode="full"))

def _best_swap_indices(h, shift):
    """Find a pair of indices (i,j) that fixes the given shift.
    Returns (i,j) to be swapped, or (None,None) if no improving pair exists."""
    n = len(h)
    if shift >= 0:
        left_len = n - shift
        left = h[:left_len]
        right = h[shift:]
        A = np.where((left == 1) & (right == 0))[0]          # left 1, right 0
        B = np.where((left == 0) & (right == 1))[0]          # left 0, right 1
        if A.size and B.size:
            i = np.random.choice(A)
            j = np.random.choice(B)
            return i, j
    else:
        k = -shift
        left_len = n - k
        left = h[k:]               # positions i = k..n-1
        right = h[:left_len]       # positions i-k = 0..n-k-1
        A = np.where((left == 1) & (right == 0))[0] + k    # actual indices in h
        B = np.where((left == 0) & (right == 1))[0] + k
        if A.size and B.size:
            i = np.random.choice(A)
            j = np.random.choice(B)
            return i, j
    return None, None

def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function for the Erdős C₅ constant.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : np.ndarray of shape (n_points,) with entries in [0,1]
        c5_bound : float – the achieved bound C₅ = max_k ∫ h(x)(1‑h(x+k))dx
        n_points : int – number of discretisation points used
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.05          # tiny safety margin

    # ----------------------------------------------------------------------
    # initialise problem size and starting vector
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init_vals = np.asarray(initial_h_values, dtype=float)
        n = init_vals.shape[0]
        if n % 2:                               # enforce even length
            n -= 1
            init_vals = init_vals[:n]
        h_cur = _binary_from_values(init_vals)
    else:
        n = 200                                   # default size (even)
        half = n // 2
        h_cur = np.zeros(n, dtype=np.int8)
        h_cur[np.random.choice(n, half, replace=False)] = 1

    # best solution seen so far
    best_h = h_cur.copy()
    best_raw = _raw_max_corr(best_h)
    best_c5 = best_raw * (2.0 / n)

    # target bound (0.3808 as required)
    TARGET_C5 = 0.3808
    target_raw = int(np.floor(TARGET_C5 * n / 2))

    # parameters for the local search
    no_improve = 0
    MAX_NO_IMPROVE = 5000

    # ----------------------------------------------------------------------
    # Main optimisation loop
    # ----------------------------------------------------------------------
    while time.time() < deadline:
        # current score
        cur_raw = _raw_max_corr(h_cur)

        # record improvement
        if cur_raw < best_raw:
            best_raw = cur_raw
            best_h = h_cur.copy()
            best_c5 = cur_raw * (2.0 / n)
            no_improve = 0
            if cur_raw <= target_raw:
                break
        else:
            no_improve += 1

        # locate a shift attaining the current maximum
        conv = np.correlate(h_cur, 1 - h_cur, mode="full")
        max_idxs = np.flatnonzero(conv == conv.max())
        shift_idx = np.random.choice(max_idxs)
        shift = int(shift_idx - (n - 1))

        # try a swap that directly reduces this shift
        i, j = _best_swap_indices(h_cur, shift)
        if i is None:
            # fall back to a random feasible swap
            ones = np.flatnonzero(h_cur == 1)
            zeros = np.flatnonzero(h_cur == 0)
            if ones.size == 0 or zeros.size == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)

        # perform swap
        h_cur[i], h_cur[j] = h_cur[j], h_cur[i]

        # accept only if we did not worsen the objective
        new_raw = _raw_max_corr(h_cur)
        if new_raw > cur_raw:
            # revert
            h_cur[i], h_cur[j] = h_cur[j], h_cur[i]

        # occasional restart if we have been stagnant
        if no_improve > MAX_NO_IMPROVE:
            # start from the best known solution with a few random perturbations
            h_cur = best_h.copy()
            for _ in range(10):
                ones = np.flatnonzero(h_cur == 1)
                zeros = np.flatnonzero(h_cur == 0)
                if ones.size and zeros.size:
                    a = np.random.choice(ones)
                    b = np.random.choice(zeros)
                    h_cur[a], h_cur[b] = h_cur[b], h_cur[a]
            no_improve = 0

    # ----------------------------------------------------------------------
    # Return the best binary step function found
    # ----------------------------------------------------------------------
    h_float = best_h.astype(float)      # convert to float as required
    return h_float, best_c5, n
