def project_onto_box_sum(y, target, lower=0.0, upper=1.0):
    # clip and shift to meet sum using water filling
    lo = -1e5
    hi = 1e5
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(y + mid, lower, upper)
        cur = x.sum()
        if cur < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo+hi)/2, lower, upper)

def project_box_sum(v, target, lo=0.0, hi=1.0):
    # v: numpy array
    # want x = clip(v + λ, lo, hi) such that sum(x) = target
    # binary search λ
    # Lo and hi can be -np.inf/np.inf as start; but we need bound.
    # compute min possible sum after clipping with λ = -inf: sum(lo) = n*lo = n*0 = 0
    # compute max possible sum with λ = +inf: sum(hi) = n*hi = n (since hi=1).
    # target within [0,n]
    # So λ must be within a range that yields that sum. We'll use bracket.
    
    left = -10.0
    right = 10.0
    # ensure bracket includes solution
    # adjust left until sum(clip(v+left)) <= target, right until >=.
    for _ in range(30):
        x_left = np.clip(v + left, lo, hi)
        if x_left.sum() <= target:
            break
        left *= 2  # make more negative
    for _ in range(30):
        x_right = np.clip(v + right, lo, hi)
        if x_right.sum() >= target:
            break
        right *= 2
    # binary search
    for _ in range(50):
        mid = (left + right) / 2
        x_mid = np.clip(v + mid, lo, hi)
        s = x_mid.sum()
        if s < target:
            left = mid
        else:
            right = mid
    return np.clip(v + (left + right) / 2, lo, hi)

import time
import numpy as np

# ------------------------ helper functions ------------------------

def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a step function sampled in h."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / n)


def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the intersection of the box [0,1]^n
    and the affine hyperplane sum(x)=target.
    Uses a simple water‑filling binary search for the Lagrange multiplier.
    """
    lo, hi = -10.0, 10.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _gradient_step(h: np.ndarray, lr: float) -> np.ndarray:
    """
    Perform a single (projected) sub‑gradient step that reduces the
    maximal overlap.  The step size is `lr`.
    """
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = conv.argmax()
    shift = idx - (n - 1)          # signed shift giving the worst overlap
    s = abs(shift)

    if s == 0:
        dr = lr * (2.0 * h - 1.0)                 # sub‑gradient for shift 0
    else:
        # h_{i+s} (zero outside) and h_{i-s} (zero outside)
        h_plus = np.concatenate((h[s:], np.zeros(s)))
        h_minus = np.concatenate((np.zeros(s), h[: n - s]))
        dr = lr * ((h_plus + h_minus) - 1.0)      # opposite sub‑gradient
    return dr


def _random_feasible(n: int) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum = n/2."""
    v = np.random.rand(n)
    return _project_box_sum(v, n / 2.0)


# --------------------------- main entry ---------------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a tuple (h_values, c5_bound, n_points) where
        h_values : numpy array with values in [0,1]
        c5_bound : the achieved C5 value
        n_points : length of the array (sampling points on [0,2])
    The algorithm starts from the provided `initial_h_values` (if any),
    up‑samples it, and then runs a projected sub‑gradient descent with
    occasional random perturbations.
    """
    np.random.seed(seed)
    start = time.time()
    TIME_LIMIT = budget_s * 0.95          # keep a safety margin

    # ---- choose number of sample points ----
    TARGET_N = 200                        # default size (fast & accurate enough)
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float).ravel()
        n0 = init.size
        TARGET_N = max(TARGET_N, n0 * 2)   # give a finer grid than the parent
        TARGET_N = min(TARGET_N, 500)     # stay well under 1000 points
        # linear up‑sampling to the new grid
        src_x = np.linspace(0.0, 2.0, n0, endpoint=False)
        dst_x = np.linspace(0.0, 2.0, TARGET_N, endpoint=False)
        h = np.interp(dst_x, src_x, init)
        h = np.clip(h, 0.0, 1.0)
        h = _project_box_sum(h, TARGET_N / 2.0)
        best_score = _c5_value(h)
    else:
        h = _random_feasible(TARGET_N)
        best_score = _c5_value(h)

    best_h = h.copy()
    n = TARGET_N
    target_sum = n / 2.0

    # ---- optimisation parameters ----
    lr = 0.12                      # initial learning rate
    decay = 0.9997                 # exponential decay per iteration
    min_lr = 1e-5
    iter_no_improve = 0
    max_no_improve = 3000
    max_iters = 50000

    it = 0
    while it < max_iters and (time.time() - start) < TIME_LIMIT:
        # sub‑gradient step
        dr = _gradient_step(h, lr)
        h_new = _project_box_sum(h + dr, target_sum)

        # evaluate

            best_h = h_new.copy()
            iter_no_improve = 0
        else:

        h = h_new

            h_noise = _project_box_sum(h + noise, target_sum)
            noisy_score = _c5_value(h_noise)
            if noisy_score < best_score:
                best_score = noisy_score
                best_h = h_noise.copy()
                h = h_noise
                lr = 0.12                     # reset a little bit
            else:
                # try the noisy point anyway (helps diversification)
                h = h_noise

        # restart if stagnation is detected
        if iter_no_improve > max_no_improve:
            h = _project_box_sum(best_h + 0.04 * np.random.randn(n), target_sum)
            iter_no_improve = 0
            lr = 0.12

        it += 1

    # final evaluation (use the harness helper if available)
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except Exception:
        final_c5 = _c5_value(best_h)

    return (best_h, float(final_c5), int(n))
