def project_to_bounds_and_sum(v, target):
    v = np.clip(v, 0, 1)
    sum_v = v.sum()
    if np.isclose(sum_v, target):
        return v
    if sum_v > target:
        # need to reduce
        while True:
            excess = sum_v - target
            # indices that are >0
            idx = np.where(v > 0)[0]
            if len(idx) == 0: break
            decrement = excess / len(idx)
            # cannot reduce below 0
            reduction = np.minimum(v[idx], decrement)
            v[idx] -= reduction
            sum_v = v.sum()
            if np.all(v[idx] > 0) and np.isclose(sum_v, target):
                break
            # if some become zero, loop again
    else: # sum_v < target
        while True:
            deficit = target - sum_v
            idx = np.where(v < 1)[0]
            if len(idx) == 0: break
            increment = deficit / len(idx)
            inc = np.minimum(1 - v[idx], increment)
            v[idx] += inc
            sum_v = v.sum()
            if np.all(v[idx] < 1) and np.isclose(sum_v, target):
                break
    return v

def repair_sum(v, target):
    # shift to meet target sum while preserving bounds
    # Use iterative water-filling
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if np.isclose(cur, target):
        return v
    if cur > target:
        # need reduce
        excess = cur - target
        # indices not at lower bound
        idx = np.where(v > 0)[0]
        while excess > 1e-12 and len(idx) > 0:
            dec = excess / len(idx)
            dec = np.minimum(v[idx], dec)
            v[idx] -= dec
            excess = v.sum() - target
            idx = np.where(v > 0)[0]
    else:
        deficit = target - cur
        idx = np.where(v < 1)[0]
        while deficit > 1e-12 and len(idx) > 0:
            inc = deficit / len(idx)
            inc = np.minimum(1 - v[idx], inc)
            v[idx] += inc
            deficit = target - v.sum()
            idx = np.where(v < 1)[0]
    return v

# helper utilities

def _repair_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to meet the exact sum `target` using a
    simple water‑filling procedure."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if np.isclose(cur, target, atol=1e-12):
        return v
    if cur < target:
        deficit = target - cur
        while deficit > 1e-12:
            idx = np.where(v < 1.0)[0]
            if idx.size == 0:
                break
            inc = deficit / idx.size
            add = np.minimum(inc, 1.0 - v[idx])
            v[idx] += add
            deficit = target - v.sum()
    else:
        excess = cur - target
        while excess > 1e-12:
            idx = np.where(v > 0.0)[0]
            if idx.size == 0:
                break
            dec = excess / idx.size
            sub = np.minimum(dec, v[idx])
            v[idx] -= sub
            excess = v.sum() - target
    return v


def _max_corr_and_shift(h: np.ndarray, dx: float):
    """Return (max_raw, shift, full_corr) where raw is the correlation
    before the dx scaling, and shift is the integer lag (negative for left
    shift)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_raw = np.max(corr)
    max_idx = np.argmax(corr)
    shift = max_idx - (len(h) - 1)
    return max_raw, shift, corr


def _triangle_initial(N: int) -> np.ndarray:
    """Triangular tent function on [0,2] discretised on N points."""
    dx = 2.0 / N
    x = (np.arange(N) + 0.5) * dx
    h = 1.0 - np.abs(x - 1.0)        # height 1 at centre, 0 at borders
    h = np.clip(h, 0.0, 1.0)
    # enforce exact integral constraint
    return _repair_sum(h, N / 2.0)


def _local_search(start_h: np.ndarray,
                  time_limit: float,
                  target_sum: float) -> tuple:
    """
    Directed hill‑climbing aimed at reducing the maximal overlap.
    Moves transfer a little mass from a “source” point (high h, low neighbour)
    to a “destination” point (low h, high neighbour) for the currently worst
    shift.

    N = len(h_cur)
    dx = 2.0 / N
    # initial evaluation
    raw_cur, shift_cur, _ = _max_corr_and_shift(h_cur, dx)
    best_raw = raw_cur
    best_h = h_cur.copy()
    max_step = 0.2   # start with relatively large transfers
    min_step = 0.001  # final precision

    start = time.time()
    # main loop
    while time.time() - start < time_limit:
        # recompute correlation for the current vector
        raw_cur, shift_cur, corr = _max_corr_and_shift(h_cur, dx)

        if raw_cur < best_raw:
            best_raw = raw_cur
            best_h = h_cur.copy()

        # --------------------------------------------------------------
        #  directed move using the shift that presently gives the max value
        # --------------------------------------------------------------
        d = shift_cur
        if d >= 0:
            a = h_cur[: N - d]          # h[i]
            b = h_cur[d:]               # h[i+d]
            contrib = a * (1.0 - b)      # contribution of each pair
            i_rel = int(np.argmax(contrib))
            src = i_rel
            dst = i_rel + d
        else:
            s = -d
            a = h_cur[s:]                # h[i] where i >= s
            b = h_cur[: N - s]           # h[i-s]
            contrib = a * (1.0 - b)
            i_rel = int(np.argmax(contrib))
            src = s + i_rel
            dst = i_rel

        # schedule step size based on elapsed proportion
        elapsed = time.time() - start
        progress = elapsed / time_limit
        step = max_step * (1.0 - progress) + min_step * progress

        # feasible transfer amount
        max_transfer = min(step, h_cur[src], 1.0 - h_cur[dst])
        if max_transfer > 1e-12:
            delta = np.random.uniform(0.0, max_transfer)
            h_new = h_cur.copy()
            h_new[src] -= delta
            h_new[dst] += delta
            raw_new, _, _ = _max_corr_and_shift(h_new, dx)
            if raw_new <= raw_cur or np.random.rand() < 0.001:
                h_cur = h_new
                raw_cur = raw_new

        # --------------------------------------------------------------
        # occasional random perturbation to escape plateaus
        # --------------------------------------------------------------
        if np.random.rand() < 0.005:

            max_transfer = min(step, h_cur[i], 1.0 - h_cur[j])
            if max_transfer > 1e-12:
                delta = np.random.uniform(0.0, max_transfer)
                h_new = h_cur.copy()
                h_new[i] -= delta
                h_new[j] += delta
                raw_new, _, _ = _max_corr_and_shift(h_new, dx)
                if raw_new <= raw_cur or np.random.rand() < 0.001:
                    h_cur = h_new
                    raw_cur = raw_new

    # final tidy‑up – clip and force exact sum (might be off by 1e‑12)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _repair_sum(best_h, target_sum)
    best_score = best_raw * dx
    return best_h, best_score
