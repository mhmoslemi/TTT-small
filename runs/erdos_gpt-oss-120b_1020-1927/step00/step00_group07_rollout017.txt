import numpy as np
import time

def _compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx for the discretization."""
    n = h.shape[0]
    # full cross‑correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)   # dx = 2/n


def _gradient_of_c_lag(h, lag):
    """Gradient of the overlap integral for a fixed lag."""
    n = h.shape[0]
    g = np.ones(n)
    if lag >= 0:
        # i + lag term
        g[: n - lag] -= h[lag:]
        # i - lag term
        g[lag:] -= h[: n - lag]
    else:               # lag < 0
        lag = -lag
        g[lag:] -= h[: n - lag]
        g[: n - lag] -= h[lag:]
    return g


def _improve_h(initial_h, budget, seed):
    """Local optimisation on a fixed grid size."""
    np.random.seed(seed)
    n = initial_h.shape[0]
    best_h = initial_h.copy()
    best_c5 = _compute_c5(best_h)
    start = time.time()
    # parameters for the stochastic search
    max_no_improve = 2000
    no_improve = 0

    while time.time() - start < budget:
        # correlation and worst lag
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        cur_c5 = corr.max() * (2.0 / n)
        # safety – keep the best value we have seen
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            # best_h already holds this vector

        # indices where the maximum is attained (might be several)
        max_idxs = np.where(corr == corr.max())[0]
        lag_idx = np.random.choice(max_idxs)
        lag = lag_idx - (n - 1)                     # integer shift

        # gradient of the active lag
        g = _gradient_of_c_lag(best_h, lag)

        # pick a decreasing and an increasing coordinate
        pos = np.where(g > 1e-8)[0]   # decreasing h_i reduces the worst overlap
        neg = np.where(g < -1e-8)[0]  # increasing h_j reduces the worst overlap

        if pos.size and neg.size:
            i = np.random.choice(pos)
            j = np.random.choice(neg)
        else:
            # fallback: random pair
            i, j = np.random.choice(n, 2, replace=False)

        # maximum admissible transfer respecting the box constraints
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 1e-12:
            continue

        # step size – shrink as time runs out
        elapsed = time.time() - start
        frac = 1.0 - elapsed / budget
        delta = max_delta * np.random.rand() * (0.5 * frac + 0.02)

        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        new_c5 = _compute_c5(h_new)

        # accept only if improvement (simple descent)
        if new_c5 < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = new_c5
            no_improve = 0
            continue

        # occasional acceptance of a worse move (simulated‑annealing style)
        temp = 0.001 * frac + 1e-6
        if np.random.rand() < np.exp(-(new_c5 - best_c5) / temp):
            best_h = h_new
            best_c5 = new_c5
            no_improve = 0
            continue

        # count stagnation
        no_improve += 1
        if no_improve >= max_no_improve:
            # perform a small random shake to escape a plateau
            idx = np.random.choice(n, 4, replace=False)
            a, b = idx[0], idx[1]
            max_delta = min(best_h[a], 1.0 - best_h[b])
            if max_delta > 1e-12:
                delta = max_delta * np.random.rand() * 0.1
                best_h[a] -= delta
                best_h[b] += delta
            no_improve = 0

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of points.

    The function respects the time budget and uses the supplied
    `initial_h_values` when present.
    """
    # ------------------------------------------------------------------
    # 1.  Starting vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
        # enforce exact feasibility (sum must be n/2)
        n = h0.shape[0]
        target = n / 2.0
        cur = h0.sum()
        if abs(cur - target) > 1e-12:
            # simple proportional shift while keeping box constraints
            diff = target - cur
            # distribute the correction uniformly, then clip
            h0 += diff / n
            np.clip(h0, 0.0, 1.0, out=h0)
            # final fix for any remaining error
            delta = target - h0.sum()
            if abs(delta) > 1e-12:
                # add/subtract uniformly from interior points
                mask = (h0 > 0.0) & (h0 < 1.0)
                idx = np.where(mask)[0]
                if idx.size:
                    adj = delta / idx.size
                    h0[idx] += adj
                    np.clip(h0, 0.0, 1.0, out=h0)
    else:
        # no parent supplied – use a modest grid and a flat start
        n = 200
        h0 = np.full(n, 0.5, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2.  Local optimisation within the allotted time
    # ------------------------------------------------------------------
    start = time.time()
    # reserve a small margin for the final return
    remaining = max(0.0, budget_s - (time.time() - start) - 0.2)

    best_h, best_c5 = _improve_h(h0, remaining, seed)

    # final sanity check (enforce constraints)
    best_h = np.clip(best_h, 0.0, 1.0)
    n = best_h.shape[0]
    target = n / 2.0
    cur = best_h.sum()
    if abs(cur - target) > 1e-9:
        # one‑shot correction preserving bounds
        diff = target - cur
        # add diff to the element farthest from its bounds
        distances = np.where(diff > 0,
                             1.0 - best_h,
                             best_h)
        idx = np.argmax(distances)
        best_h[idx] = np.clip(best_h[idx] + diff, 0.0, 1.0)

    final_c5 = _compute_c5(best_h)

    return best_h, float(final_c5), n
