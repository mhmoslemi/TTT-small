def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time, random, math

    # ------------------------------------------------------------
    # Helper: projection onto box [0,1] with exact sum constraint
    # ------------------------------------------------------------
    def project_box_sum(v, target_sum):
        # Find λ such that sum(clip(v+λ,0,1)) = target_sum (monotone in λ)
        lo = -np.max(v) - 1.0           # makes all entries ≤0 after shift
        hi = 1.0 - np.min(v) + 1.0      # makes all entries ≥1 after shift
        for _ in range(60):            # enough for double precision
            mid = (lo + hi) * 0.5
            proj = np.clip(v + mid, 0.0, 1.0)
            s = proj.sum()
            if s < target_sum:
                lo = mid
            else:
                hi = mid
        lam = (lo + hi) * 0.5
        return np.clip(v + lam, 0.0, 1.0)

    # ------------------------------------------------------------
    # Helper: compute C5 for a given vector h (full correlation version)
    # ------------------------------------------------------------
    def compute_c5(h):
        n = h.size
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * dx

    # ------------------------------------------------------------
    # Determine problem size and an initial feasible vector
    # ------------------------------------------------------------
    try:
        init_arr = np.array(initial_h_values, dtype=np.float64)  # provided by harness
        n = init_arr.shape[0]
        has_init = True
    except Exception:               # no initial_h_values in the environment
        init_arr = None
        n = 70
        has_init = False

    S = n / 2.0                     # required sum of h (integral = 1)
    best_h = None
    best_val = np.inf

    # ------------------------------------------------------------
    # If an initial construction exists, bring it to feasibility
    # ------------------------------------------------------------
    if has_init:
        h0 = project_box_sum(init_arr, S)
        val0 = compute_c5(h0)
        best_h = h0.copy()
        best_val = val0

    # ------------------------------------------------------------
    # Parameters for the search
    # ------------------------------------------------------------
    start_time = time.time()
    end_time = start_time + budget_s
    # allocate portions of the budget
    sa_budget = budget_s * 0.35      # simulated annealing (binary)
    gd_budget = budget_s * 0.60      # gradient descent (continuous)
    # remaining time for final polishing
    final_budget = budget_s * 0.05

    # ------------------------------------------------------------
    # 1. Simulated annealing on binary vectors (preserves exact half‐ones)
    # ------------------------------------------------------------
    def binary_sa(time_limit):
        nonlocal best_h, best_val
        ones_needed = int(round(S))
        # simple temperature schedule
        while time.time() < time_limit:
            # start from a random feasible binary vector
            h_bin = np.zeros(n, dtype=np.float64)
            h_bin[:ones_needed] = 1.0
            np.random.shuffle(h_bin)
            cur_h = h_bin
            cur_val = compute_c5(cur_h)
            if cur_val < best_val:
                best_val, best_h = cur_val, cur_h.copy()
            # inner SA loop
            it = 0
            while time.time() < time_limit and it < 30000:
                it += 1
                # pick a random 1 and a random 0 and swap them
                ones_idx = np.where(cur_h == 1.0)[0]
                zeros_idx = np.where(cur_h == 0.0)[0]
                if ones_idx.size == 0 or zeros_idx.size == 0:
                    break
                i1 = np.random.choice(ones_idx)
                i0 = np.random.choice(zeros_idx)
                new_h = cur_h.copy()
                new_h[i1] = 0.0
                new_h[i0] = 1.0
                new_val = compute_c5(new_h)

                # linear temperature decay (fast cooling)
                elapsed = time.time() - start_time
                T = max(1e-5, 0.08 * (budget_s - elapsed) / budget_s)

                if new_val < cur_val or random.random() < math.exp(-(new_val - cur_val) / T):
                    cur_h, cur_val = new_h, new_val
                    if new_val < best_val:
                        best_val, best_h = new_val, new_h.copy()
            # occasional premature exit to respect the budget
            if time.time() >= time_limit:
                break

    # ------------------------------------------------------------
    # 2. Gradient descent on continuous vectors (starting from best binary)
    # ------------------------------------------------------------
    def gradient_descent(time_limit):
        nonlocal best_h, best_val
        # start from the best binary found (or random feasible if none)
        if best_h is None:
            cur_h = project_box_sum(np.random.rand(n), S)
            cur_val = compute_c5(cur_h)
            best_h, best_val = cur_h.copy(), cur_val
        else:
            cur_h = best_h.copy()
            cur_val = best_val

        eps = 0.3                     # initial step size
        min_eps = 1e-6
        decay = 0.995
        it = 0
        while time.time() < time_limit:
            it += 1
            # correlation and worst shift
            corr_full = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            idx_max = np.argmax(corr_full)
            shift = idx_max - (n - 1)

            # gradient of the worst shift (sub‑gradient of max)
            grad = np.zeros_like(cur_h)
            s = shift
            if s >= 0:
                grad[: n - s] += (1.0 - cur_h[s:])
                grad[s:] -= cur_h[: n - s]
            else:
                t = -s
                grad[t:] += (1.0 - cur_h[: n - t])
                grad[: n - t] -= cur_h[t:]

            # take a step and project back to feasible set
            new_h = project_box_sum(cur_h - eps * grad, S)
            new_val = compute_c5(new_h)

            # keep the best ever seen
            if new_val < best_val - 1e-9:
                best_val, best_h = new_val, new_h.copy()

            # acceptance of the trial point
            if new_val <= cur_val:
                cur_h, cur_val = new_h, new_val
                eps = min(eps * 1.02, 1.0)          # gently increase step when successful
            else:
                eps = max(eps * 0.7, min_eps)      # shrink step after failure

            # occasional tiny random jitter to escape flat regions
            if it % 500 == 0:
                jitter = (np.random.rand(n) - 0.5) * eps * 0.2
                cur_h = project_box_sum(cur_h + jitter, S)
                cur_val = compute_c5(cur_h)

            # slowly decay the step size overall
            eps *= decay
            if eps < min_eps:
                eps = min_eps

    # ------------------------------------------------------------
    # 3. Final polishing (very small steps)
    # ------------------------------------------------------------
    def final_polish(time_limit):
        nonlocal best_h, best_val
        if best_h is None:
            return
        cur_h = best_h.copy()
        cur_val = best_val
        eps = 1e-3
        while time.time() < time_limit:
            corr_full = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            idx_max = np.argmax(corr_full)
            s = idx_max - (n - 1)

            grad = np.zeros_like(cur_h)
            if s >= 0:
                grad[: n - s] += (1.0 - cur_h[s:])
                grad[s:] -= cur_h[: n - s]
            else:
                t = -s
                grad[t:] += (1.0 - cur_h[: n - t])
                grad[: n - t] -= cur_h[t:]

            new_h = project_box_sum(cur_h - eps * grad, S)
            new_val = compute_c5(new_h)

            if new_val < best_val - 1e-12:
                best_val, best_h = new_val, new_h.copy()
                cur_h, cur_val = new_h, new_val
            else:
                eps *= 0.9
                if eps < 1e-7:
                    break

    # ------------------------------------------------------------
    # Run the three phases within the overall time budget
    # ------------------------------------------------------------
    binary_sa(start_time + sa_budget)
    gradient_descent(start_time + sa_budget + gd_budget)
    final_polish(end_time)

    # ------------------------------------------------------------
    # Return the best construction found
    # ------------------------------------------------------------
    return (best_h, float(best_val), n)
