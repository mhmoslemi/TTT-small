def random_move(h, step):
    n = len(h)
    # choose i with h_i < 1, j with h_j > 0
    # to avoid zero transfer we ensure we have space
    i = np.random.randint(n)
    # Determine potential transfer from j to i (increase i)
    # Randomly pick j distinct from i
    j = np.random.randint(n-1)
    if j >= i:
        j += 1
    # Compute max increase constrained by 1 - h_i and h_j
    max_inc = min(step, 1 - h[i], h[j])
    # Compute max decrease (transfer from i to j)
    max_dec = min(step, h[i], 1 - h[j])
    # Choose direction randomly
    if np.random.rand() < 0.5:
        delta = max_inc * np.random.rand()
        if delta < 1e-12:
            return None
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
    else:
        delta = max_dec * np.random.rand()
        if delta < 1e-12:
            return None
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    return new_h

def project_box_sum(v, target):
    # Bisection on lambda
    lo = -1.0  # lower bound for lambda
    hi = 1.0   # upper bound
    # Expand bounds until feasible:
    while np.sum(np.clip(v - lo, 0, 1)) < target:
        lo *= 2
    while np.sum(np.clip(v - hi, 0, 1)) > target:
        hi *= 2
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0, 1)

import time
import numpy as np

def compute_c5(h):
    """Return C5 for the given step values."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with its complement
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_box_sum(v, target):
    """Euclidean projection of v onto the box [0,1] intersect {sum == target}."""
    # bisection on lambda: sum_i clip(v_i - lam,0,1) = target
    lo = -1.0
    hi = 1.0
    while np.sum(np.clip(v - lo, 0.0, 1.0)) < target:
        lo *= 2.0
    while np.sum(np.clip(v - hi, 0.0, 1.0)) > target:
        hi *= 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def upsample_to_n(v_small, n):
    """Upsample a coarse vector to length n by linear interpolation."""
    m = v_small.shape[0]
    xs = np.linspace(0.0, 2.0, m, endpoint=False)
    xt = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(xt, xs, v_small)


def random_init(n):
    """Generate a random feasible vector."""
    v = np.random.rand(n)
    return project_box_sum(v, n / 2.0)


def gradient_step(h, lr, alpha):

    n = h.shape[0]

            grad += 2.0 * w_l * h

            j_slice = slice(l, n)

            grad[j_slice] += w_l * h[i_slice]

    h_new = h + lr * grad
    # enforce bounds and sum
    return project_box_sum(h_new, n / 2.0)


def optimize_candidate(h0, start_time, budget, n):
    """Improve a starting vector until time runs out, returning the best found."""
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    lr = 0.12          # initial learning rate
    alpha = 5.0        # smoothing strength
    it = 0
    while True:
        now = time.time()
        if now - start_time > budget - 0.5:
            break
        h = gradient_step(h, lr, alpha)
        val = compute_c5(h)
        if val + 1e-12 < best_val:
            best_val = val
            best_h = h.copy()
        it += 1
        # schedule tweaks
        if it % 250 == 0:
            lr = max(lr * 0.85, 1e-4)
            alpha = min(alpha * 1.07, 120.0)
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # resolution – a modest size that still gives a good bound
    n = 180
    if n % 2:
        n += 1
    best_h = None
    best_c5 = float("inf")

    # 1) start from the provided construction if present
    if "initial_h_values" in globals():
        try:
            init = np.asarray(initial_h_values, dtype=float)
            h0 = upsample_to_n(init, n)
            h0 = project_box_sum(h0, n / 2.0)
            h_opt, c5_opt = optimize_candidate(h0, start, budget_s, n)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt
        except Exception:
            pass

    # 2) uniform start (simple baseline)
    h_uniform = np.full(n, 0.5, dtype=float)
    h_uniform = project_box_sum(h_uniform, n / 2.0)
    h_opt, c5_opt = optimize_candidate(h_uniform, start, budget_s, n)
    if c5_opt < best_c5:
        best_c5, best_h = c5_opt, h_opt

    # 3) a few random restarts
    for _ in range(6):
        if time.time() - start > budget_s - 2.0:
            break
        h_rand = random_init(n)
        h_opt, c5_opt = optimize_candidate(h_rand, start, budget_s, n)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt

    # final safety projection
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, n / 2.0)
    # recompute final bound (in case of tiny drift)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
