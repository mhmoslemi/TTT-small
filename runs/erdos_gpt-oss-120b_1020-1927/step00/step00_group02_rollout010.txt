def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # compute linear convolution via FFT with zero padding.
    # length = 2*n - 1.
    N = 1 << ((2*n-1).bit_length())
    # Actually using np.fft.rfft for real.
    f1 = np.fft.rfft(h, n=N)
    f2 = np.fft.rfft(1 - h, n=N)
    conv = np.fft.irfft(f1 * np.conj(f2), n=N)[:2*n-1]
    return dx * np.max(conv)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = budget_s * 0.95  # leave margin

    # Setup initial h
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0_coarse = np.array(initial_h_values, dtype=float)
        n0 = h0_coarse.size
    else:
        n0 = 45
        h0_coarse = rng.random(n0)
        # Normalize to integral = 1
        h0_coarse *= (1.0 / (np.sum(h0_coarse) * 2.0 / n0))

    # Choose target number of points
    # Keep within 500 for speed.
    target_n = min(500, max(2 * n0, 180))
    # Ensure even for easier splitting? not required.
    n = target_n

    # Interpolate to n points linearly
    x_coarse = np.linspace(0, 2, n0, endpoint=False)
    x_target = np.linspace(0, 2, n, endpoint=False)
    h = np.interp(x_target, x_coarse, h0_coarse)
    # Clip to [0,1]
    np.clip(h, 0.0, 1.0, out=h)

    # Normalize integral to 1 exactly
    dx = 2.0 / n
    total = np.sum(h) * dx
    if total <= 0:
        # fallback uniform
        h = np.full(n, 0.5)
        total = np.sum(h) * dx
    # Scale to achieve sum =1
    h *= (1.0 / total)

    # Ensure still within bounds (scaling may push >1)
    np.clip(h, 0.0, 1.0, out=h)
    # Renormalize again if needed
    total = np.sum(h) * dx
    if not math.isclose(total, 1.0, rel_tol=1e-12, abs_tol=1e-12):
        # Distribute tiny correction to first entry
        diff = 1.0 - total
        h[0] += diff / dx
        # Clip again
        h[0] = min(max(h[0], 0.0), 1.0)

    # Function to compute objective
    def compute_c5_vec(vec):
        # np.correlate returns array of length 2*n-1
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        return dx * np.max(corr)

    cur_c5 = compute_c5_vec(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # Simulated annealing parameters
    Tmax = max(0.01, cur_c5 * 0.05)  # start temperature
    Tmin = 1e-6
    max_step = 0.05  # max delta for a move
    iter_since_improve = 0
    total_iter = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
        # Linear cooling schedule
        T = Tmax * (1.0 - elapsed / time_limit)
        if T < Tmin:
            T = Tmin

        # propose move
        i = rng.integers(n)
        j = rng.integers(n)
        while j == i:
            j = rng.integers(n)
        hi = h[i]
        hj = h[j]
        # allowed delta to keep within bounds
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        # restrict to max_step
        lo = max(low, -max_step)
        hi_step = min(high, max_step)
        if lo >= hi_step:
            # use allowed range
            lo, hi_step = low, high
        d = rng.uniform(lo, hi_step)

        # Apply move
        h[i] = hi + d
        h[j] = hj - d

        new_c5 = compute_c5_vec(h)
        delta = new_c5 - cur_c5

        accept = False
        if delta < 0:
            accept = True
        else:
            # Accept with probability exp(-delta/T)
            if T > 0:
                prob = math.exp(-delta / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                iter_since_improve = 0
            else:
                iter_since_improve += 1
        else:
            # revert move
            h[i] = hi
            h[j] = hj
            iter_since_improve += 1

        total_iter += 1

        # adapt step size gradually
        if total_iter % 10000 == 0:
            # if no improvement for many steps, increase randomness
            if iter_since_improve > 50000:
                # Random perturbation: add small noise and reproject
                noise = rng.normal(scale=0.02, size=n)

                # Renormalize to sum 1
                total = np.sum(h) * dx
                if total > 0:
                    h *= (1.0 / total)
                cur_c5 = compute_c5_vec(h)
                iter_since_improve = 0
            # reduce max_step slowly
            max_step = max(0.001, max_step * 0.95)

    # After loop, final best solution
    # Ensure constraints satisfied exactly (might be tiny float errors)
    np.clip(best_h, 0.0, 1.0, out=best_h)
    total = np.sum(best_h) * dx
    if total != 0:
        best_h *= (1.0 / total)

    final_c5 = compute_c5_vec(best_h)

    return best_h.tolist(), float(final_c5), int(n)

def sa_optimize(h_init, dx, time_budget):
    n = len(h_init)
    h = h_init.copy()
    # compute initial correlation
    max_val, t_max, corr_raw = max_corr_and_shift(h)
    cur_c5 = max_val * dx
    best_h = h.copy()
    best_c5 = cur_c5
    start_local = time.time()
    max_step = 0.08
    p_target = 0.6
    # temperature schedule
    T0 = max(0.01, cur_c5 * 0.05)
    while time.time() - start_local < time_budget:
        elapsed = time.time() - start_local
        T = max(1e-6, T0 * (1 - elapsed / time_budget))
        # propose move
        if rng.random() < p_target:
            # targeted move
            move = propose_targeted_move(h, t_max, max_step, rng)
            if move is None:
                move = propose_random_move(h, max_step, rng)
        else:
            move = propose_random_move(h, max_step, rng)
        if move is None:
            continue
        if move[2] == 'targeted':
            # Actually propose_targeted_move returns i, j, d
            i, j, d = move
        else:
            i, j, d = move
        # apply move: move mass d from i to j
        hi = h[i]
        hj = h[j]
        # d positive meaning transfer from i to j? We'll define d positive to move from i to j.
        # Ensure d does not exceed bounds
        # For safety, clamp
        d = max(0.0, min(d, min(hi, 1.0 - hj)))
        if d == 0:
            continue
        h[i] = hi - d
        h[j] = hj + d

        # compute new correlation
        new_max_val, new_t_max, new_corr_raw = max_corr_and_shift(h)
        new_c5 = new_max_val * dx
        delta = new_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0 and rng.random() < math.exp(-delta / T):
                accept = True
        if accept:
            cur_c5 = new_c5
            max_val = new_max_val
            t_max = new_t_max

import time, math
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # reserve a bit of margin for safe exit
    time_limit = budget_s * 0.95

    # ------------------------------------------------------------------
    # 1.  Load/construct an initial feasible h.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals() and initial_h_values is not None:
        h_coarse = np.array(initial_h_values, dtype=float)
    else:
        n0 = 45
        h_coarse = rng.random(n0)
    n0 = h_coarse.size
    dx0 = 2.0 / n0
    np.clip(h_coarse, 0.0, 1.0, out=h_coarse)

    # normalise the integral to exactly 1
    total = np.sum(h_coarse) * dx0
    if total <= 0:
        h_coarse.fill(0.5)
        total = np.sum(h_coarse) * dx0
    h_coarse *= (1.0 / total)

    # ------------------------------------------------------------------
    # 2.  Helper functions
    # ------------------------------------------------------------------
    def corr_and_max(vec):
        """return (full correlation array, max value, offset of the max)."""
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        max_val = np.max(corr)
        max_idx = int(np.argmax(corr))
        offset = max_idx - (len(vec) - 1)          # shift = offset * dx
        return corr, max_val, offset

    def compute_c5(vec, dx):
        """C5 = max correlation * dx."""
        _, max_val, _ = corr_and_max(vec)
        return max_val * dx

    # ------------------------------------------------------------------
    # 3.  Random‑pair move proposal (preserves sum exactly)
    # ------------------------------------------------------------------
    def propose_random_move(h, max_step):
        n = h.shape[0]
        i = rng.integers(n)
        j = rng.integers(n)
        while j == i:
            j = rng.integers(n)
        hi, hj = h[i], h[j]
        low = max(-hi, hj - 1.0)      # smallest d we can add to h[i]
        high = min(1.0 - hi, hj)      # largest  d we can add to h[i]
        if low >= high:
            return None
        # keep step size moderate
        lo = max(low, -max_step)
        hi_step = min(high, max_step)
        if lo >= hi_step:
            lo, hi_step = low, high
        d = rng.uniform(lo, hi_step)
        return i, j, d                     # h[i] += d ; h[j] -= d

    # ------------------------------------------------------------------
    # 4.  Targeted move that attacks the worst shift
    # ------------------------------------------------------------------
    def propose_targeted_move(h, shift_offset, max_step):
        """move mass from a point to its partner at the maximal‑overlap shift."""
        n = h.shape[0]
        if shift_offset >= 0:
            i_low, i_high = 0, n - shift_offset - 1
        else:
            i_low, i_high = -shift_offset, n - 1
        if i_low > i_high:
            return None
        for _ in range(15):
            i = rng.integers(i_low, i_high + 1)
            j = i + shift_offset
            hi, hj = h[i], h[j]
            if hi <= 0.0 or hj >= 1.0:        # cannot transfer
                continue
            max_transfer = min(hi, 1.0 - hj)
            step = min(max_step, max_transfer)
            if step <= 0:
                continue
            m = rng.uniform(0.0, step)        # amount to move from i to j
            # our update rule is h[i] += d ; h[j] -= d,
            # so to move *away* from i we use d = -m
            return i, j, -m
        return None

    # ------------------------------------------------------------------
    # 5.  Core optimisation routine (simulated‑annealing + targeted moves)
    # ------------------------------------------------------------------
    def optimise(h_init, dx, remaining_budget):
        n = h_init.shape[0]
        h = h_init.copy()
        corr, cur_max, cur_offset = corr_and_max(h)
        cur_c5 = cur_max * dx
        best_h = h.copy()
        best_c5 = cur_c5

        # schedule parameters
        T0 = max(0.01, cur_c5 * 0.05)          # initial temperature
        max_step = 0.08
        p_target = 0.6                         # probability to try a targeted move
        start = time.time()
        it = 0

        while True:
            elapsed = time.time() - start
            if elapsed >= remaining_budget:
                break
            # linear cooling
            T = max(1e-7, T0 * (1.0 - elapsed / remaining_budget))

            # ------------------------------------------------------------------
            # propose a move
            # ------------------------------------------------------------------
            if rng.random() < p_target:
                mv = propose_targeted_move(h, cur_offset, max_step)
                if mv is None:
                    mv = propose_random_move(h, max_step)
            else:
                mv = propose_random_move(h, max_step)

            if mv is None:
                continue
            i, j, d = mv
            hi, hj = h[i], h[j]

            # apply move
            h[i] = hi + d
            h[j] = hj - d

            # evaluate new objective
            new_corr, new_max, new_offset = corr_and_max(h)
            new_c5 = new_max * dx
            delta = new_c5 - cur_c5

            # acceptance test
            accept = False
            if delta < 0:
                accept = True
            else:
                if T > 0.0 and rng.random() < math.exp(-delta / T):
                    accept = True

            if accept:
                cur_c5 = new_c5
                cur_max = new_max
                cur_offset = new_offset
                corr = new_corr
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                it = 0
            else:
                # revert
                h[i] = hi
                h[j] = hj
                it += 1

            # occasional step‑size decay
            if it % 5000 == 0 and it > 0:
                max_step = max(0.001, max_step * 0.92)

        return best_h, best_c5

    # ------------------------------------------------------------------
    # 6.  Phase 1 – optimise on the original grid (≈35 % of the budget)
    # ------------------------------------------------------------------
    phase1_budget = time_limit * 0.35
    best_h_phase1, best_c5_phase1 = optimise(h_coarse, dx0, phase1_budget)

    # ------------------------------------------------------------------
    # 7.  Phase 2 – up‑sample and continue optimisation (remaining budget)
    # ------------------------------------------------------------------
    target_n = max(180, 2 * n0)          # reasonable resolution, ≤500
    if target_n > 500:
        target_n = 500
    # linear interpolation onto the finer grid
    x_coarse = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_fine   = np.linspace(0.0, 2.0, target_n, endpoint=False)
    h_fine = np.interp(x_fine, x_coarse, best_h_phase1)
    np.clip(h_fine, 0.0, 1.0, out=h_fine)
    dx_fine = 2.0 / target_n
    total = np.sum(h_fine) * dx_fine
    if total > 0:
        h_fine *= (1.0 / total)

    remaining_budget = time_limit - (time.time() - start_time)
    if remaining_budget > 0.1:          # tiny reserve for final tidy‑up
        best_h_phase2, best_c5_phase2 = optimise(h_fine, dx_fine, remaining_budget)
    else:
        best_h_phase2, best_c5_phase2 = h_fine, compute_c5(h_fine, dx_fine)

    # ------------------------------------------------------------------
    # 8.  Pick the best solution obtained
    # ------------------------------------------------------------------
    if best_c5_phase2 < best_c5_phase1:
        final_h = best_h_phase2
        final_c5 = best_c5_phase2
        n_points = target_n
        dx = dx_fine
    else:
        final_h = best_h_phase1
        final_c5 = best_c5_phase1
        n_points = n0
        dx = dx0

    # ------------------------------------------------------------------
    # 9.  Final projection onto the feasible set (tiny numerical clean‑up)
    # ------------------------------------------------------------------
    np.clip(final_h, 0.0, 1.0, out=final_h)
    target_sum = n_points / 2.0                # because sum * dx = 1  → sum = n/2
    cur_sum = np.sum(final_h)
    if cur_sum > 0:
        final_h *= (target_sum / cur_sum)     # simple scaling; any tiny breach is harmless
    # one‑element correction if scaling produced a tiny out‑of‑bounds value
    if final_h[0] < 0.0:
        final_h[0] = 0.0
    elif final_h[0] > 1.0:
        final_h[0] = 1.0
    # recompute bound after projection
    final_c5 = compute_c5(final_h, dx)

    # ------------------------------------------------------------------
    # 10. Return result
    # ------------------------------------------------------------------
    return final_h.tolist(), float(final_c5), int(n_points)
