function propose_move(vec):
    n = len(vec)
    new = vec.copy()
    attempt = 0
    while attempt < 10:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # random direction
        if np.random.rand() < 0.5:
            transfer_max = min(new[i], 1 - new[j])
            if transfer_max > 0:
                delta = transfer_max * np.random.rand()  # uniform up to max
                new[i] -= delta
                new[j] += delta
                break
        else:
            transfer_max = min(new[j], 1 - new[i])
            if transfer_max > 0:
                delta = transfer_max * np.random.rand()
                new[j] -= delta
                new[i] += delta
                break
        attempt += 1
    return new

import numpy as np, time, math, random

def eval_c5(h):
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * (2.0 / len(h))

def random_binary_feasible(n):
    w = n//2
    h = np.zeros(n, dtype=np.float64)
    ones = np.random.choice(n, w, replace=False)
    h[ones] = 1.0
    if n % 2 == 1:
        # set one extra entry to 0.5 at random position not already one
        remaining = np.where(h == 0)[0]
        if len(remaining) > 0:
            idx = np.random.choice(remaining)
            h[idx] = 0.5
    return h

def propose_move(h):
    n = len(h)
    new_h = h.copy()
    # randomly choose direction and pair
    for _ in range(10):
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        if np.random.rand() < 0.5:
            max_transfer = min(new_h[i], 1 - new_h[j])
            if max_transfer > 0:
                delta = max_transfer * np.random.rand()
                new_h[i] -= delta
                new_h[j] += delta
                break
        else:
            max_transfer = min(new_h[j], 1 - new_h[i])
            if max_transfer > 0:
                delta = max_transfer * np.random.rand()
                new_h[j] -= delta
                new_h[i] += delta
                break
    return new_h

def project_sum(h, target):
    # adjust sum via simple elastic method
    # compute diff
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # identify indices can be increased (if diff>0) or decreased (if diff<0)
    if diff > 0:
        inc_indices = np.where(h < 1 - 1e-12)[0]
        if len(inc_indices) == 0:
            return h
        # distribute equally
        add = diff / len(inc_indices)
        # ensure not exceed 1
        for idx in inc_indices:
            can = 1 - h[idx]
            add_i = min(add, can)
            h[idx] += add_i
            diff -= add_i
            # if diff becomes zero break
        # leftover diff can be ignored small
    else:
        diff = -diff
        dec_indices = np.where(h > 1e-12)[0]
        if len(dec_indices) == 0:
            return h
        sub = diff / len(dec_indices)
        for idx in dec_indices:
            can = h[idx]
            sub_i = min(sub, can)
            h[idx] -= sub_i
            diff -= sub_i
    return h

def _eval_c5(vec):
    """C5 value for a given vector."""
    n = vec.shape[0]
    return np.max(np.correlate(vec, 1.0 - vec, mode="full")) * (2.0 / n)

def _random_feasible(n):
    """Return a feasible vector of length n (sum = n/2, values in [0,1])."""
    target = n / 2.0

    else:
        w = n // 2
        h = np.zeros(n, dtype=np.float64)
        ones = np.random.choice(n, w, replace=False)
        h[ones] = 1.0
        # one fractional entry to reach exact sum
        rest = np.setdiff1d(np.arange(n), ones)
        h[np.random.choice(rest)] = 0.5
    return h

def _valid_indices(n, shift):
    """Indices i such that i and i+shift are both in [0,n)."""
    if shift >= 0:
        return np.arange(0, n - shift)
    else:
        return np.arange(-shift, n)

def _propose_random_transfer(v):
    """Random sum‑preserving transfer between two coordinates."""
    n = v.shape[0]
    a = v.copy()
    for _ in range(10):
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        if np.random.rand() < 0.5:
            max_tr = min(a[i], 1.0 - a[j])
            if max_tr > 0.0:
                delta = max_tr * np.random.rand()
                a[i] -= delta
                a[j] += delta
                break
        else:
            max_tr = min(a[j], 1.0 - a[i])
            if max_tr > 0.0:
                delta = max_tr * np.random.rand()
                a[j] -= delta
                a[i] += delta
                break
    return a

def _targeted_move(v):
    """Attempt a move that directly reduces the current maximum overlap."""
    n = v.shape[0]
    corr = np.correlate(v, 1.0 - v, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    inds = _valid_indices(n, shift)
    if inds.size == 0:
        return None
    # masks for uncovered (i) and covered (j) positions w.r.t. this shift
    eps = 1e-12
    vi = v[inds]
    vj = v[inds + shift]
    i_mask = (vi > eps) & (vj < 1.0 - eps)
    j_mask = (vi < 1.0 - eps) & (vj > eps)
    if not (np.any(i_mask) and np.any(j_mask)):
        return None
    i = np.random.choice(inds[i_mask])
    j = np.random.choice(inds[j_mask])
    max_tr = min(v[i], 1.0 - v[j])
    if max_tr <= eps:
        return None
    new = v.copy()
    new[i] -= max_tr
    new[j] += max_tr
    return new

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a small C5 value."""
    np.random.seed(seed)
    start = time.time()

    # ------- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n = h.size
        h = np.clip(h, 0.0, 1.0)
        target_sum = n / 2.0
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            # simple projection onto the sum‑constraint (few adjustments)
            free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
            if free.size:
                add = diff / free.size
                for idx in free:
                    h[idx] = np.clip(h[idx] + add, 0.0, 1.0)
    else:
        n = 100                     # even length makes the binary construction easy
        h = _random_feasible(n)

    target_sum = n / 2.0
    cur = h.copy()
    cur_val = _eval_c5(cur)
    best = cur.copy()
    best_val = cur_val

    # ------- optimisation loop -----------------------------------------
    no_improve = 0
    while True:
        now = time.time()
        if now - start >= budget_s - 0.15:
            break

        # try a targeted move that attacks the current worst shift
        cand = _targeted_move(cur)
        if cand is None:
            # fall back to a random transfer move
            cand = _propose_random_transfer(cur)

        cand_val = _eval_c5(cand)

        # accept if better, otherwise keep the old state (pure hill‑climb)
        if cand_val < cur_val:
            cur, cur_val = cand, cand_val
            no_improve = 0
            if cand_val < best_val:
                best, best_val = cand.copy(), cand_val
        else:
            # occasional stochastic acceptance to escape flat regions
            T = 0.02 * (1.0 - (now - start) / budget_s) + 1e-6
            if np.random.rand() < math.exp(-(cand_val - cur_val) / T):
                cur, cur_val = cand, cand_val
                no_improve = 0
            else:
                no_improve += 1

        # if stuck for a while, inject a random perturbation
        if no_improve > 3000:
            cur = _random_feasible(n)
            cur_val = _eval_c5(cur)
            no_improve = 0

    # final safety projection (tiny rounding errors)
    best = np.clip(best, 0.0, 1.0)
    diff = target_sum - best.sum()
    if abs(diff) > 1e-12:
        free = np.where(best < 1.0 - 1e-12)[0] if diff > 0 else np.where(best > 1e-12)[0]
        if free.size:
            adj = diff / free.size
            for idx in free:
                best[idx] = np.clip(best[idx] + adj, 0.0, 1.0)

    final_val = _eval_c5(best)
    return (best.tolist(), float(final_val), n)
