import numpy as np, time

def project_onto_bounds_and_sum(v, target_sum):
    """Project vector onto box [0,1] and sum constraint using binary search on shift lambda."""
    # Ensure v as float
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(v):
    n = v.shape[0]
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode='full')
    return dx * np.max(corr)

def interpolate_h(v, n_new):
    n_old = len(v)
    old_x = np.linspace(0.0, 2.0, n_old, endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    v_new = np.interp(new_x, old_x, v)
    return v_new

def random_feasible(rng, n, target_sum):

def gradient_descent_one_step(v, target_sum, eta):

import time
import numpy as np

def _project(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the box [0,1] and the hyperplane sum = target_sum."""
    lo = -np.max(v) - 1.0          # makes v+lo <= 0  -> sum = 0
    hi = 1.0 - np.min(v) + 1.0     # makes v+hi >= 1  -> sum = len(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _c5(v: np.ndarray) -> float:
    """Return the overlap constant for step function v on [0,2]."""
    n = v.shape[0]
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full")
    return dx * np.max(corr)


def _interp(v: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of v to a new grid of size n_new."""
    n_old = len(v)
    old_x = np.linspace(0.0, 2.0, n_old, endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, v)


def _random_feasible(rng: np.random.Generator, n: int, target: float) -> np.ndarray:
    """Generate a random feasible vector (values in [0,1], correct sum)."""
    v = rng.random(n)
    return _project(v, target)


def _gd_step(v: np.ndarray, target_sum: float, eta: float) -> np.ndarray:
    """One gradient–descent step on the active lag."""
    n = len(v)
    corr = np.correlate(v, 1 - v, mode="full")
    idx = np.argmax(corr)
    lag = idx - (n - 1)

    g = np.zeros_like(v)
    if lag >= 0:
        i0, i1 = 0, n - lag
        g[i0:i1] = 1 - v[i0 + lag:i1 + lag]
        g[i0 + lag:i1 + lag] -= v[i0:i1]
    else:
        L = -lag
        i0, i1 = 0, n - L
        g[i0 + L:i1 + L] = 1 - v[i0:i1]
        g[i0:i1] -= v[i0 + L:i1 + L]

    v_new = v - eta * g
    return _project(v_new, target_sum)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # a safety margin

    # ----- initial vector -------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = None
    if h0 is None or h0.ndim != 1:
        n0 = 100
        h0 = np.full(n0, 0.5, dtype=float)
    else:
        n0 = len(h0)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_score = _c5(best_h)

    # ----- grid schedule --------------------------------------------------
    max_n = 336
    sizes = [n0]
    while sizes[-1] < max_n:
        nxt = min(max_n, sizes[-1] * 2)
        if nxt == sizes[-1]:
            break
        sizes.append(nxt)

    # ----- optimisation loop ---------------------------------------------
    for n in sizes:
        if time.time() >= deadline:
            break

        if len(best_h) != n:
            best_h = _interp(best_h, n)
            best_h = _project(best_h, n / 2.0)

        target = n / 2.0
        best_score = _c5(best_h)

        # --- gradient descent from current best (few passes) -------------
        for _ in range(2):
            v = best_h.copy()
            for it in range(300):
                if time.time() >= deadline:
                    break
                eta = 0.5 / np.sqrt(it + 1.0)
                v = _gd_step(v, target, eta)
                score = _c5(v)
                if score + 1e-12 < best_score:
                    best_score = score
                    best_h = v.copy()

        # --- random restarts with short gradient descent -----------------
        for _ in range(3):
            if time.time() >= deadline:
                break
            v = _random_feasible(rng, n, target)
            for it in range(100):
                if time.time() >= deadline:
                    break
                eta = 0.4 / np.sqrt(it + 1.0)
                v = _gd_step(v, target, eta)
                score = _c5(v)
                if score + 1e-12 < best_score:
                    best_score = score
                    best_h = v.copy()

        # --- local random perturbations -----------------------------------
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        rand_iters = int(min(3000, max(100, remaining * 2000)))
        step_scale = 0.2
        for _ in range(rand_iters):
            if time.time() >= deadline:
                break
            cand = best_h + rng.normal(scale=step_scale, size=n)
            cand = _project(cand, target)
            cand_score = _c5(cand)
            if cand_score + 1e-12 < best_score:
                best_score = cand_score
                best_h = cand
                step_scale *= 0.99
            else:
                step_scale *= 0.9999

    # ----- final tidy -----------------------------------------------------
    best_h = _project(best_h, len(best_h) / 2.0)
    final_score = _c5(best_h)
    return best_h, final_score, len(best_h)
