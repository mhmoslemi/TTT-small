    best_h = init_h if init_h is not None else random_binary(rng, n, target_sum)
    best_score = evaluate(best_h)
    h = best_h.copy()
    current_score = best_score
    # iterative loop with time check
    start = time.time()
    iter = 0
    while time.time() - start < time_limit:
        # choose random indices
        # Get list of ones positions, zeros positions; we can maintain arrays for speed.
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i] = 0
        h[j] = 1
        new_score = evaluate(h)
        if new_score < current_score:
            current_score = new_score
            # update best if improved
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
        iter += 1
    return best_h, best_score

def gradient_descent_step(h, lr):
    n = len(h)
    dx = 2.0 / n
    O = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(O)
    shift = max_idx - (n - 1)  # integer shift
    grad = np.zeros_like(h)
    if shift >= 0:
        # part1: indices 0..n-1-shift
        grad[:n - shift] += 1 - h[shift:]   # y = 1 - h, but h[shift:] is the shifted h value, for y[i+shift] = 1 - h[i+shift]
        grad[shift:] -= h[:n - shift]
    else:
        t = -shift
        grad[t:] += 1 - h[:n - t]  # because y[i - t] = 1 - h[i - t]
        grad[:n - t] -= h[t:]
    # update
    y = h - lr * grad
    # project onto [0,1] and sum target
    target = n / 2.0
    h_new = proj_sum_box(y, target, 0.0, 1.0)
    return h_new, np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx

def gradient_descent_optimize(initial_h, target_sum, max_iters, lr0, time_deadline):
    h = initial_h.copy()
    best_h = h.copy()
    best_score = evaluate(h)
    n = len(h)
    dx = 2.0 / n
    lr = lr0
    start = time.time()
    for it in range(max_iters):
        if time.time() > time_deadline:
            break
        # Compute overlap
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        # gradient
        grad = np.zeros_like(h)
        if shift >= 0:
            grad[:n - shift] += 1 - h[shift:]
            grad[shift:] -= h[:n - shift]
        else:
            t = -shift
            grad[t:] += 1 - h[:n - t]
            grad[:n - t] -= h[t:]
        # update
        y = h - lr * grad
        h = proj_sum_box(y, target_sum, 0.0, 1.0)
        # eval
        score = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if score < best_score:
            best_score = score
            best_h = h.copy()
        # adaptive lr: optionally reduce if no improvement for many steps
        # we could reduce lr gradually

    return best_h, best_score

import time
import numpy as np

def proj_sum_box(v, target, lower=0.0, upper=1.0):
    """Project vector v onto the set {x: lower≤x≤upper, sum(x)=target}."""
    left = np.min(v) - upper
    right = np.max(v) - lower
    for _ in range(60):
        mid = (left + right) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = np.sum(x)
        if s > target:
            left = mid
        else:
            right = mid
    return np.clip(v - (left + right) / 2.0, lower, upper)


def bound_score(h):
    """Return the Erdős overlap bound for step function h."""
    if 'evaluate_erdos_solution' in globals():
        return float(evaluate_erdos_solution(h))
    n = len(h)
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)


def gradient_step(h, lr, target):
    """One projected sub‑gradient step w.r.t. the worst shift."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)               # integer shift producing the maximum overlap
    grad = np.zeros_like(h)
    if shift >= 0:
        grad[: n - shift] += 1 - h[shift:]
        grad[shift:]      -= h[: n - shift]
    else:
        t = -shift
        grad[t:]          += 1 - h[: n - t]
        grad[: n - t]     -= h[t:]
    y = h - lr * grad
    return proj_sum_box(y, target, 0.0, 1.0)


def random_binary(n, target, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=int(target), replace=False)] = 1.0
    return h


def thue_morse_sequence(n):
    """Binary Thue–Morse sequence of length n (n a power of two)."""
    seq = np.fromiter(((bin(i).count('1') & 1) for i in range(n)), dtype=float, count=n)
    # Adjust to exact half ones if necessary
    target = n // 2
    cur = int(seq.sum())
    if cur != target:
        diff = target - cur
        if diff > 0:
            zeros = np.where(seq == 0)[0][:diff]
            seq[zeros] = 1.0
        else:
            ones = np.where(seq == 1)[0][: -diff]
            seq[ones] = 0.0
    return seq


def hill_climb_binary(start_h, target, rng, deadline, best_h, best_score):
    """Swap‑based hill climbing on binary vectors."""
    h = start_h.copy()
    cur_score = bound_score(h)
    n = len(h)
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_score = bound_score(h)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_score

    start = time.time()
    budget = float(budget_s)

    # ----- discretisation -------------------------------------------------

    target = n / 2.0

    # ----- initial candidate pool -----------------------------------------

        h0 = np.clip(h0, 0.0, 1.0)
        h0 = proj_sum_box(h0, target, 0.0, 1.0)
        candidates.append(h0)

    candidates.append(thue_morse_sequence(n))
