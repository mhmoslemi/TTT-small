def gradient_ascent_step(h, step):
    # compute autocorr
    ac = np.correlate(h, h, mode='full')
    min_ac = np.min(ac)
    # compute lag where min occurs (maybe multiple)
    min_indices = np.where(ac == min_ac)[0]
    # convert index to lag: lag = idx - (len(h)-1)
    lags = min_indices - (len(h)-1)
    # compute gradient as average over these lags
    grad = np.zeros_like(h)
    for lag in lags:
        if lag == 0:
            grad += 2 * h

def solve_milp(n, budget):
    m = n // 2  # number of ones (since n is even)
    a = cp.Variable(n, boolean=True)
    # Create a dict for w variables for each i<j
    # We'll need a list of pairs
    pairs = [(i,j) for i in range(n) for j in range(i+1,n)]
    w_vars = cp.Variable(len(pairs), boolean=True)
    # constraints
    constraints = []
    constraints.append(cp.sum(a) == m)
    # now map pair index
    for idx, (i,j) in enumerate(pairs):
        constraints.append(w_vars[idx] <= a[i])
        constraints.append(w_vars[idx] <= a[j])
        constraints.append(w_vars[idx] >= a[i] + a[j] - 1)
    # variable t
    t = cp.Variable()
    constraints.append(t >= 0)
    # For each lag L from 1..n-1 (including zero maybe)
    for L in range(1, n):
        # list indexes of pairs for this lag: those with j - i == L
        idxs = [idx for idx, (i,j) in enumerate(pairs) if j - i == L]
        # sum of a_i over i from 0..n-1-L
        a_sum_pos = cp.sum(a[:n-L])  # a[0]..a[n-L-1]

        constraints.append(a_sum_pos - w_sum <= t)
        constraints.append(a_sum_neg - w_sum <= t)
    # also L=0 constraints: O(0)=0 satisfied automatically
    # objective
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver='CBC', warm_start=True)
    return a.value, t.value

def _project_box_simplex(v, target):
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        y = np.clip(v + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-8:
        free = (y > 1e-12) & (y < 1 - 1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def _solve_binary_milp(n, time_limit):
