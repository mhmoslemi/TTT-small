def shift_with_zeros(arr, shift):
    n = len(arr)
    if shift > 0:
        out = np.empty(n, dtype=arr.dtype)
        out[:n-shift] = arr[shift:]
        out[n-shift:] = 0.0
    elif shift < 0:
        out = np.empty(n, dtype=arr.dtype)
        out[-shift:] = arr[:n+shift]
        out[: -shift] = 0.0
    else:
        out = arr.copy()
    return out

def shift_with_zeros(arr, shift):
    n = arr.shape[0]
    if shift > 0:
        out = np.empty_like(arr)
        out[:n-shift] = arr[shift:]
        out[n-shift:] = 0.0
    elif shift < 0:
        s = -shift
        out = np.empty_like(arr)
        out[s:] = arr[:n-s]
        out[:s] = 0.0
    else:
        out = arr.copy()
    return out

def project_onto_cap_simplex(y, target):
    # binary search for lambda
    lo = y.min() - 1.0
    hi = y.max()
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(y - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(y - (lo+hi)/2.0, 0.0, 1.0)
    # may have small sum error; adjust by adding small epsilon to non-bound entries.
    err = proj.sum() - target
    if abs(err) > 1e-8:
        # distribute error to entries not at bounds
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            # adjust equally
            adjust = err / free.sum()
            proj[free] -= adjust
            # clip again
            proj = np.clip(proj, 0.0, 1.0)
        else:
            # all at bounds; cannot adjust perfectly; but this should not happen.
            pass
    return proj

def max_shift_and_gradient(h):
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(conv)
    k = idx - (n - 1)  # shift
    # compute shifted versions
    shifted_plus = shift_with_zeros(h, k)
    shifted_minus = shift_with_zeros(h, -k)
    grad = 1.0 - (shifted_plus + shifted_minus)
    return k, grad, conv[idx]

import numpy as np, time

def shift_with_zeros(arr, shift):
    n = arr.shape[0]
    out = np.zeros_like(arr)
    if shift > 0:
        if shift < n:
            out[:n - shift] = arr[shift:]
    elif shift < 0:
        s = -shift
        if s < n:
            out[s:] = arr[:n - s]
    else:
        out = arr.copy()
    return out

def project_onto_cap_simplex(y, target):
    # binary search for lambda
    lo = y.min() - 1.0
    hi = y.max()
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(y - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    # small correction for sum error
    err = proj.sum() - target
    if abs(err) > 1e-9:
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            proj[free] -= err / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def evaluate_h(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def max_shift_and_gradient(h):
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')

    k = idx - (n - 1)

    shifted_minus = shift_with_zeros(h, -k)
    grad = 1.0 - (shifted_plus + shifted_minus)
    return k, grad, conv[idx]

import time
import numpy as np

# ---------- Helper functions ----------
def _shift_with_zeros(arr: np.ndarray, shift: int) -> np.ndarray:
    """Shift arr by `shift` positions, padding with zeros (no wrap)."""
    n = arr.shape[0]
    out = np.zeros_like(arr)
    if shift > 0:
        if shift < n:
            out[: n - shift] = arr[shift:]
    elif shift < 0:
        s = -shift
        if s < n:
            out[s:] = arr[: n - s]
    else:
        out = arr.copy()
    return out


def _project_onto_cap_simplex(y: np.ndarray, target: float) -> np.ndarray:
    """
    Project y onto the set {x | 0 <= x_i <= 1, sum x_i = target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = y.min() - 1.0  # low enough -> all entries become 1 after clipping
    hi = y.max()        # high enough -> all entries become 0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(y - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction if numerical error remained
    err = proj.sum() - target
    if abs(err) > 1e-9:
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            proj[free] -= err / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _evaluate_h(h: np.ndarray) -> float:
    """C5 value for a given sampled function h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)


def _max_shift_and_gradient(h: np.ndarray):
    """Return shift attaining the maximum overlap, its gradient, and the raw
    (unnormalised) maximal convolution value."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(conv))
    k = idx - (n - 1)                     # integer shift
    shifted_plus = _shift_with_zeros(h, k)
    shifted_minus = _shift_with_zeros(h, -k)
    grad = 1.0 - (shifted_plus + shifted_minus)
    return k, grad, float(conv[idx])


# ---------- Optimiser (simulated annealing + gradient) ----------
def _optimise(start_h: np.ndarray, deadline: float, target_bound: float):
    """
    Stochastic optimisation returning a feasible h and its C5 value.
    Uses a mix of gradient moves and random perturbations within a
    simulated‑annealing framework.
    """
    n = start_h.shape[0]
    target_sum = n / 2.0
    h = _project_onto_cap_simplex(start_h, target_sum)
    cur_score = _evaluate_h(h)
    best_h = h.copy()
    best_score = cur_score

    it = 0
    while time.time() < deadline:
        it += 1
        # temperature decays exponentially
        T = max(1e-4, 0.12 * np.exp(-it / 2000.0))

        if np.random.rand() < 0.6:                # gradient style move
            _, grad, _ = _max_shift_and_gradient(h)
            eta = 0.5 / np.sqrt(it)
            y = h - eta * grad
            new_h = _project_onto_cap_simplex(y, target_sum)
        else:                                      # random perturbation
            r = np.random.normal(scale=0.07, size=n)
            r -= r.mean()                         # keep zero sum
            new_h = _project_onto_cap_simplex(h + r, target_sum)

        new_score = _evaluate_h(new_h)
        delta = new_score - cur_score

        if delta < -1e-12 or np.random.rand() < np.exp(-delta / T):
            h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
                if best_score <= target_bound:
                    break
    return best_h, best_score


# ---------- Main entry point ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points).
    h_values is a list of samples in [0,1] satisfying the integral constraint.
    """
    np.random.seed(seed)
    start = time.time()
    target_bound = 0.3808            # we stop as soon as we beat this

    # choose a modest discretisation (still well below 1 k points)
    n = 200

    # build an initial feasible vector
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        m = init.shape[0]
        if m != n:
            xs_old = np.linspace(0.0, 2.0, m, endpoint=False)
            xs_new = np.linspace(0.0, 2.0, n, endpoint=False)
            init = np.interp(xs_new, xs_old, init)
        h0 = np.clip(init, 0.0, 1.0)
    else:
        h0 = np.random.rand(n)

    # enforce feasibility (box + sum)
    target_sum = n / 2.0
    h0 = _project_onto_cap_simplex(h0, target_sum)

    best_h = h0
    best_score = _evaluate_h(best_h)

    # first optimisation run starting from the (possibly) good initial guess
    deadline = start + budget_s - 0.5
    if time.time() < deadline:
        cand_h, cand_score = _optimise(best_h, deadline, target_bound)
        if cand_score < best_score:
            best_h, best_score = cand_h, cand_score

    # further random restarts until time runs out
    while time.time() < start + budget_s - 0.4:
        # a fresh random feasible start
        rand_vec = np.random.rand(n)
        start_vec = _project_onto_cap_simplex(rand_vec, target_sum)
        deadline = start + budget_s - 0.2
        cand_h, cand_score = _optimise(start_vec, deadline, target_bound)
        if cand_score < best_score:
            best_h, best_score = cand_h, cand_score
        if best_score <= target_bound:
            break

    final_score = _evaluate_h(best_h)
    return (best_h.tolist(), float(final_score), int(n))
