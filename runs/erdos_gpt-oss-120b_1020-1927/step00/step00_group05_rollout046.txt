def gradient_for_shift(h, d):
    n = len(h)
    if d >= 0:
        term1 = np.empty(n)
        term1[:d] = 0.0
        term1[d:] = h[:n-d]
        term2 = np.empty(n)
        term2[:n-d] = h[d:]
        term2[n-d:] = 0.0
    else:
        dd = -d
        term1 = np.empty(n)
        term1[:n-dd] = h[dd:]
        term1[n-dd:] = 0.0
        term2 = np.empty(n)
        term2[:dd] = 0.0
        term2[dd:] = h[:n-dd]
    grad = 1.0 - (term1 + term2)
    return grad
```

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1 - h, mode='full') * dx
    return r.max()

def gradient_for_shift(h, d):
    n = len(h)
    if d >= 0:
        term1 = np.empty(n, dtype=h.dtype)
        if d == 0:
            term1 = h
        else:
            term1[:d] = 0.0
            term1[d:] = h[:n-d]
        term2 = np.empty(n, dtype=h.dtype)
        term2[:n-d] = h[d:]
        term2[n-d:] = 0.0
    else:
        dd = -d
        term1 = np.empty(n, dtype=h.dtype)
        term1[:n-dd] = h[dd:]
        term1[n-dd:] = 0.0
        term2 = np.empty(n, dtype=h.dtype)
        term2[:dd] = 0.0
        term2[dd:] = h[:n-dd]
    return 1.0 - (term1 + term2)

def project_onto_bounds_sum(v, target, lower=0.0, upper=1.0, max_iter=50):
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        tau = (lo + hi) / 2.0
        w = np.clip(v + tau, lower, upper)
        s = w.sum()
        if s > target:
            hi = tau
        else:
            lo = tau
    w = np.clip(v + lo, lower, upper)
    # adjust minor rounding error
    # ensure sum close to target, but if not, scale residual across non-boundary entries
    diff = w.sum() - target
    if abs(diff) > 1e-9:
        # simple correction by adding diff uniformly to free variables
        free = (w > lower + 1e-12) & (w < upper - 1e-12)
        if free.any():
            w[free] -= diff / free.sum()
            w = np.clip(w, lower, upper)
        else:
            # fallback: adjust by small constant to avoid violation
            w -= diff / len(w)
            w = np.clip(w, lower, upper)
    return w

    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        tau = (lo + hi) / 2.0
        w = np.clip(v + tau, lower, upper)
        s = w.sum()
        if s > target:
            hi = tau
        else:
            lo = tau

            w = np.clip(w, lower, upper)
        else:
            # adjust uniformly
            w -= diff / len(w)
            w = np.clip(w, lower, upper)

import time

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def gradient_for_shift(h: np.ndarray, d: int) -> np.ndarray:
    """Gradient of the worst–overlap term with respect to h."""
    n = h.shape[0]
    if d >= 0:
        term1 = np.empty(n, dtype=h.dtype)
        term1[:d] = 0.0
        term1[d:] = h[: n - d]
        term2 = np.empty(n, dtype=h.dtype)
        term2[: n - d] = h[d:]
        term2[n - d :] = 0.0
    else:
        dd = -d
        term1 = np.empty(n, dtype=h.dtype)
        term1[: n - dd] = h[dd:]
        term1[n - dd :] = 0.0
        term2 = np.empty(n, dtype=h.dtype)
        term2[:dd] = 0.0
        term2[dd:] = h[: n - dd]
    return 1.0 - (term1 + term2)


def project_onto_bounds_sum(v: np.ndarray, target: float,
                            lower: float = 0.0, upper: float = 1.0,
                            max_iter: int = 60) -> np.ndarray:
    """Euclidean projection onto { lower ≤ x ≤ upper, sum x = target }."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        tau = (lo + hi) * 0.5
        w = np.clip(v + tau, lower, upper)
        s = w.sum()
        if s > target:
            hi = tau
        else:
            lo = tau
    w = np.clip(v + lo, lower, upper)
    diff = w.sum() - target
    if abs(diff) > 1e-10:

            w[free] -= diff / free.sum()
            w = np.clip(w, lower, upper)
        else:
            w -= diff / w.size
            w = np.clip(w, lower, upper)
    return w

                h_new[i] -= delta
                h_new[j] += delta
                h_new = project_onto_bounds_sum(h_new, target_sum)
                new_val = compute_c5(h_new)
                if new_val < cur_val:
                    h = h_new
                    cur_val = new_val
                    improved = True

        if cur_val < best_val - 1e-9:
            best_val = cur_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # occasional restart if stuck
        if no_improve > 3000:
            noise = rng.normal(scale=0.05, size=n)
            h = best_h + noise
            h = project_onto_bounds_sum(h, target_sum)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        # modest decay of learning rate
        if t % 1000 == 0:
            lr = max(lr * 0.8, 1e-4)

    # --- random exploration in remaining time --------------------------------
    while time.time() - start < max_time:
        # random Dirichlet vector (continuous) projected onto the feasible set
        rnd = rng.random(n)
        rnd = project_onto_bounds_sum(rnd, target_sum)
        val = compute_c5(rnd)
        if val < best_val:
            best_val = val
            best_h = rnd.copy()
        # also try pure binary samples occasionally
        if rng.random() < 0.05:
            ones = rng.choice(n, size=n // 2, replace=False)
            bin_vec = np.zeros(n, dtype=float)
            bin_vec[ones] = 1.0
            valb = compute_c5(bin_vec)
            if valb < best_val:
                best_val = valb
                best_h = bin_vec.copy()

    return (best_h.tolist(), float(best_val), n)
