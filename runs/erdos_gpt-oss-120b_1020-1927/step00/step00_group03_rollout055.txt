def shift_contrib(h, s):
    N = len(h)
    w = np.zeros(N)
    if s >= 0:
        # h[i+s] term for i up to N-1-s
        w[:N-s] += h[s:]   # h[i+s] where i from 0..N-1-s
        # h[i-s] term for i>=s
        w[s:] += h[:N-s]   # h[i-s]
    else:
        # s negative => s = -t
        t = -s
        w[t:] += h[:N-t]   # h[i+s] = h[i-t], for i>=t
        w[:N-t] += h[t:]   # h[i-s] = h[i+t], for i<= N-1-t
    return w

        else:

    h = h0.copy()
    cur_val = evaluate(h)
    best_h = h.copy()

    N = len(h)
    T_start = 0.02
    T_end = 1e-5
    it = 0
    while time.time() - start < time_limit:
        it += 1
        elapsed = time.time() - start
        # temperature schedule linear
        T = T_start * (1 - elapsed / time_limit) + T_end * (elapsed / time_limit)
        max_delta = min(0.2, T*10 + 0.001) # small floor

    # binary search lambda
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        lam = (lo+hi)/2
        x = np.clip(u + lam, 0, 1)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    # Final

    diff = target - x.sum()
    if abs(diff) > 1e-9:

        idx = np.where((x > 1e-12) & (x < 1-1e-12))[0]
        if idx.size > 0:
            # add evenly
            add = diff / idx.size
            x[idx] += add
            x = np.clip(x, 0, 1)  # ensure
    return x

import numpy as np
import time
import random

def evaluate(h: np.ndarray) -> float:
    """Return C5 value for a given discretised step function."""
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / N))


def shift_contrib(h: np.ndarray, s: int) -> np.ndarray:
    """
    For a given shift s, return the vector w_i = h[i+s] + h[i-s]
    (out‑of‑range indices contribute 0). Used for the gradient of the
    worst‑overlap shift.
    """
    N = h.shape[0]
    w = np.zeros(N)
    if s >= 0:
        w[: N - s] += h[s:]
        w[s:] += h[: N - s]
    else:
        t = -s
        w[t:] += h[: N - t]
        w[: N - t] += h[t:]
    return w


def project_box_simplex(u: np.ndarray, target: float) -> np.ndarray:
    """
    Project a vector onto the intersection of the hyperplane sum(x)=target
    and the box 0<=x<=1. Uses a binary search on the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(u + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    x = np.clip(u + lo, 0.0, 1.0)
    # tiny correction for possible round‑off
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size:
            add = diff / free.size
            x[free] += add
            x = np.clip(x, 0.0, 1.0)
    return x


def simulated_annealing(h0: np.ndarray, timelimit: float) -> tuple:
    """
    Simple simulated‑annealing optimiser that respects the sum constraint.
    Returns the best vector and its C5 value found within timelimit seconds.
    """
    N = h0.shape[0]
    target = N / 2.0
    h = h0.copy()
    cur_val = evaluate(h)
    best_h, best_val = h.copy(), cur_val

    T_start, T_end = 0.02, 1e-5
    it = 0
    start = time.time()

    while time.time() - start < timelimit:
        it += 1
        elapsed = time.time() - start
        frac = elapsed / timelimit
        T = (1.0 - frac) * T_start + frac * T_end
        max_delta = max(0.001, min(0.2, T * 10.0))

        i, j = np.random.randint(N, size=2)
        if i == j:
            continue

        # feasible transfer from j -> i (positive delta) or i -> j (negative)
        inc = min(max_delta, 1.0 - h[i], h[j])
        dec = min(max_delta, h[i], 1.0 - h[j])

        if inc <= 0.0 and dec <= 0.0:
            continue

        if inc > 0.0 and (dec == 0.0 or np.random.rand() < 0.5):
            delta = np.random.uniform(0.0, inc)
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
        else:
            delta = np.random.uniform(0.0, dec)
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta

        new_val = evaluate(h_new)
        dval = new_val - cur_val

        if dval <= 0.0 or np.random.rand() < np.exp(-dval / T):
            h, cur_val = h_new, new_val
            if cur_val < best_val:
                best_h, best_val = h.copy(), cur_val

        # occasional deterministic improvement on the current worst shift
        if it % 100 == 0:
            corr = np.correlate(h, 1.0 - h, mode="full")
            s = int(np.argmax(corr) - (N - 1))
            w = shift_contrib(h, s)
            mask_up = h < 1.0 - 1e-12
            mask_dn = h > 1e-12
            if mask_up.any() and mask_dn.any():
                w_up = w.copy()
                w_up[~mask_up] = -np.inf
                i_up = int(np.argmax(w_up))
                w_dn = w.copy()
                w_dn[~mask_dn] = np.inf
                i_dn = int(np.argmin(w_dn))
                delta_max = min(0.1, 1.0 - h[i_up], h[i_dn])
                if delta_max > 1e-12:
                    h2 = h.copy()
                    h2[i_up] += delta_max
                    h2[i_dn] -= delta_max
                    val2 = evaluate(h2)
                    if val2 < cur_val:
                        h, cur_val = h2, val2
                        if cur_val < best_val:
                            best_h, best_val = h.copy(), cur_val

    return best_h, best_val

    np.random.seed(seed)

    start_global = time.time()

        h0 = np.interp(x_new, x_old, old_h)

    while True:

        remaining = budget_s - elapsed

        chunk = min(0.5, remaining * 0.5)
