import time, math, random
import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """Correlation based evaluation of the overlap constant."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _worst_gradient(h: np.ndarray):
    """
    Returns the gradient of the worst (maximal) overlap lag.
    Gradient component i = -(h[i+lag] + h[i-lag]) with zero padding.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)
    grad = np.zeros_like(h)

    if lag >= 0:
        L = lag
        if L < n:
            grad[: n - L] -= h[L:]          # term from h[i+L]
            grad[L:] -= h[: n - L]          # term from h[i-L]
    else:
        L = -lag
        if L < n:
            grad[L:] -= h[: n - L]          # term from h[i+lag] (i-L)
            grad[: n - L] -= h[L:]          # term from h[i-lag] (i+L)
    return grad, lag, corr[idx] * (2.0 / n)

            else:
                # occasional uphill acceptance
                temperature = 0.003 * (1.0 - it / max_iter_fine)
                if temperature > 0 and random.random() < math.exp(-(new_c5 - cur_c5) / temperature):
                    h = h_new
                    cur_c5 = new_c5

        # keep the finer result if it beats the coarse one
        if fine_best_c5 < best_c5:
            best_c5 = fine_best_c5
            best_h = fine_best_h
            n = new_n

    # ------------------------------------------------------------------ #
    # Return result
    # ------------------------------------------------------------------ #
    return best_h.tolist(), float(best_c5), n
```
