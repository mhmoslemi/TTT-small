def compute_gradient(h, weights):
    n = len(h)
    g = np.zeros_like(h)
    # w indexed from -(n-1) .. (n-1)
    for idx, w in enumerate(weights):
        l = idx - (n-1)
        if w == 0:
            continue
        # shift
        # compute t1: (1 - h) shifted left by l: i+ l in range
        if l >= 0:
            # i + l in range => i up to n-1-l
            g[:n-l] += w * (1 - h[l:])
            # t2: h shifted right by l: i - l >=0 -> i from l..n-1 -> index i-l
            g[l:] -= w * h[:n-l]
        else:
            # l negative: define lneg = -l >0
            lneg = -l
            # t1: (1 - h) shifted left by l (negative) i + l <=> i - lneg
            # i + l = i - lneg => index shift right
            g[lneg:] += w * (1 - h[:n-lneg])  # i starts at lneg, i-lneg is index
            # t2: h shifted right by l (negative) => h shifted left by lneg
            g[:n-lneg] -= w * h[lneg:]
    return g

def gradient_for_shift(h, shift):
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        # term1 for i in [0, n-shift-1]
        i1 = slice(0, n-shift)
        g[i1] = 1 - h[shift:]
        # term2 for i in [shift, n-1]
        i2 = slice(shift, n)
        g[i2] -= h[:n-shift]
    else:
        # negative shift: shift = -d
        d = -shift
        # term1 for i in [d, n-1] (i + shift = i - d is within)
        i1 = slice(d, n)
        g[i1] = 1 - h[:n-d]
        # term2 for i in [0, n-d-1] (i - shift = i + d)
        i2 = slice(0, n-d)
        g[i2] -= h[d:]
    return g

    for _ in range(max_iter):
        lam = (lo_lam + hi_lam) / 2.0
        w = np.clip(v - lam, lo, hi)
        total = w.sum()
        if abs(total - s) <= tol:
            return w
        if total > s:
            lo_lam = lam
        else:
            hi_lam = lam
    # final return
    return np.clip(v - lam, lo, hi)

def gradient_for_shift(h, shift):
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            g[:n-shift] = 1.0 - h[shift:]
            g[shift:] -= h[:n-shift]
    else:
        d = -shift
        if d < n:
            g[d:] = 1.0 - h[:n-d]
            g[:n-d] -= h[d:]
    return g

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(corr.max() * dx)

def generate_trapezoid_h(n, a):
    # a ∈ [0,1]
    if a <= 0:
        # constant shape with H = 0.5
        return np.full(n, 0.5)
    H = 1.0 / (2.0 - a)
    dx = 2.0 / n
    # sample points at midpoints: x_i = (i+0.5) * dx
    x = (np.arange(n) + 0.5) * dx
    h = np.empty(n, dtype=float)
    left = x <= a
    right = x >= 2.0 - a
    middle = (~left) & (~right)
    # left linear
    h[left] = H * x[left] / a
    # right linear
    h[right] = H * (2.0 - x[right]) / a
    # middle plateau
    h[middle] = H
    return h

import time, math
import numpy as np

# ----------------------------------------------------------------------
# Helper: project a vector onto the box [lo,hi] and enforce sum == s
def _project_box_simplex(v, s, lo=0.0, hi=1.0, tol=1e-10, max_iter=80):
    """Projection onto {w | lo <= w_i <= hi, sum w_i = s}."""
    lo_lam, hi_lam = -1e6, 1e6
    for _ in range(max_iter):
        lam = (lo_lam + hi_lam) * 0.5
        w = np.clip(v - lam, lo, hi)
        total = w.sum()
        if abs(total - s) <= tol:
            return w
        if total > s:
            lo_lam = lam
        else:
            hi_lam = lam
    return np.clip(v - lam, lo, hi)

# Gradient of the correlation for a single shift (sub‑gradient of the max)
def _gradient_shift(h, shift):
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            g[: n - shift] = 1.0 - h[shift:]
            g[shift:] -= h[: n - shift]
    else:
        d = -shift
        if d < n:
            g[d:] = 1.0 - h[: n - d]
            g[: n - d] -= h[d:]
    return g

def _c5_of(h):

    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(corr.max() * dx)

def _trapezoid_h(n, a):
    """
    Trapezoid shape on [0,2] with parameter a in [0,1].
    a = half‑length of the sloping parts.
    Height H = 1/(2-a) guarantees area = 1.
    """
    if a <= 1e-12:
        return np.full(n, 0.5)                     # constant 0.5 (a=0)

    h = np.empty(n, dtype=float)
    left = x <= a
    right = x >= 2.0 - a
    middle = (~left) & (~right)
    h[left] = H * x[left] / a
    h[right] = H * (2.0 - x[right]) / a
    h[middle] = H
    return h
