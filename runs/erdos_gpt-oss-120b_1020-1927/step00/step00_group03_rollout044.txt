def full_correlation(h):
    N = len(h)
    nfft = 1 << ((2*N-1).bit_length())
    fh = np.fft.rfft(h, nfft)
    fb = np.fft.rfft((1-h)[::-1], nfft)
    conv = np.fft.irfft(fh * fb, nfft)[:2*N-1]
    return conv

import time, math
import numpy as np

def _evaluate_c5(h: np.ndarray) -> float:
    """Return max overlap C5 for the discrete step function h."""
    N = h.shape[0]
    dx = 2.0 / N
    # np.correlate implements linear correlation (full mode)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def _project_onto_feasible(y: np.ndarray, target_sum: float,
                           tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Euclidean projection of y onto {0<=x<=1, sum(x)=target_sum}."""
    lo = -np.max(y)          # makes y - lo >= 1  => all clipped to 1
    hi = np.max(y)           # makes y - hi <= 0  => all clipped to 0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    # final clamp
    return np.clip(y - lam, 0.0, 1.0)

            s = -shift
            if s < N:
                g[s:] = 1.0 - h[:N - s]

        # donor (large g, h>0) and receiver (small g, h<1)
        donors = np.where(h > eps)[0]
        receivers = np.where(h < 1.0 - eps)[0]
        if donors.size == 0 or receivers.size == 0:
            break

        donor = donors[np.argmax(g[donors])]
        receiver = receivers[np.argmin(g[receivers])]
        if donor == receiver:
            # fallback random pair
            donor = rng.choice(donors)
            receiver = rng.choice(receivers)
            if donor == receiver:
                it += 1
                continue

        delta = min(step, h[donor], 1.0 - h[receiver])
        if delta <= eps:
            it += 1
            continue

        h_new = h.copy()
        h_new[donor] -= delta
        h_new[receiver] += delta

        new_c5 = _evaluate_c5(h_new)

        # acceptance test
        accept = False
        if new_c5 <= cur_c5 + eps:
            accept = True
        else:
            elapsed = time.time() - start
            T = T0 * max(0.0, (budget_s - elapsed) / budget_s)
            prob = math.exp(-(new_c5 - cur_c5) / (T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()

        # adaptive step size
        if it % 2000 == 0 and it > 0:
            if best_c5 < cur_c5 - 1e-4:
                step = min(step * 1.1, 0.2)
            else:
                step = max(step * 0.9, 0.001)

        # occasional random perturbation to escape plateaus
        if rng.random() < 0.005:
            i, j = rng.choice(N, size=2, replace=False)
            if h[i] > eps and h[j] < 1.0 - eps:
                d = min(step, h[i], 1.0 - h[j])
                h[i] -= d
                h[j] += d

        it += 1

    # final projection (numerical drift)
    best_h = _project_onto_feasible(best_h, target_sum)
    best_c5 = _evaluate_c5(best_h)

    return best_h, best_c5, N
