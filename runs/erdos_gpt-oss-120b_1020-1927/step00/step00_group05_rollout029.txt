def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def improve_binary(h, rng, max_iter, time_limit):
    N = len(h)
    dx = 2.0 / N
    best = h.copy()
    best_score = compute_c5(best)
    start = time.time()
    no_improve = 0
    while time.time() - start < time_limit:
        # compute worst lag
        corr = np.correlate(best, 1 - best, mode='full') * dx
        max_idx = np.argmax(corr)
        shift = max_idx - (N - 1)
        # find indices that cause mismatches
        bad_ones = []
        for i in range(N):
            j = i + shift
            if 0 <= j < N:
                if best[i] == 1 and best[j] == 0:
                    bad_ones.append(i)
        if not bad_ones:
            # no mismatches -> optimum
            break
        zeros = np.where(best == 0)[0]
        # try a few random swaps involving a bad one
        improved = False
        for _ in range(30):
            i = rng.choice(bad_ones)
            j = rng.choice(zeros)
            # swap
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
            score = compute_c5(cand)
            if score < best_score - 1e-9:
                best, best_score = cand, score
                improved = True
                break
        if not improved:
            # random swap
            i = rng.integers(N)
            j = rng.integers(N)
            if best[i] == best[j]:
                continue
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
            score = compute_c5(cand)
            if score < best_score:
                best, best_score = cand, score
                improved = True
        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0
    return best, best_score

def improve_binary(h, rng, time_limit):
    N = len(h)
    dx = 2.0 / N
    best = h.copy()
    best_score = compute_c5(best)
    start = time.time()
    # loop until time limit
    while time.time() - start < time_limit:
        corr = np.correlate(best, 1 - best, mode='full') * dx
        max_idx = np.argmax(corr)
        shift = max_idx - (N - 1)
        # compute mismatches arrays
        if shift >= 0:
            first = best[:N-shift]
            second = best[shift:]
            offset = 0
        else:
            s = -shift
            first = best[s:]
            second = best[:N-s]
            offset = s
        # indices in original best where first corresponds
        bad = np.where((first == 1) & (second == 0))[0] + offset
        good = np.where((first == 0) & (second == 1))[0] + offset
        # if we have both, try swap
        if bad.size and good.size:
            i = rng.choice(bad)
            j = rng.choice(good)
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
            cand_score = compute_c5(cand)
            if cand_score + 1e-12 < best_score:
                best, best_score = cand, cand_score
                continue
        # else random swap
        # pick random i with 1 and j with 0
        ones_idx = np.where(best == 1)[0]
        zeros_idx = np.where(best == 0)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        cand = best.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_score = compute_c5(cand)
        if cand_score + 1e-12 < best_score:
            best, best_score = cand, cand_score
    return best, best_score
```

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

import time
import numpy as np

def _c5(h):
    """C5 for a vector h (dx = 2/n)."""
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full') * dx))

def _adjust_sum(h, target):
    """Project h onto [0,1] with given total (water‑filling)."""
    # simple water‑filling by binary search on lambda
    lo, hi = -1.0, 1.0
    for _ in range(30):
        lam = (lo + hi) / 2.0
        w = np.clip(h - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _binary_improve(h, rng, deadline):
    """Local search on a binary vector (swap 0/1)."""
    n = h.shape[0]
    dx = 2.0 / n
    best = h.copy()
    best_sc = _c5(best)
    while time.time() < deadline:
        corr = np.correlate(best, 1 - best, mode='full') * dx
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)

        if shift >= 0:
            f, s = best[:n - shift], best[shift:]
            off = 0
        else:
            s = -shift
            f, s = best[s:], best[:n - s]
            off = s

        bad = np.where((f == 1) & (s == 0))[0] + off
        good = np.where((f == 0) & (s == 1))[0] + off
        if bad.size and good.size:
            i, j = rng.choice(bad), rng.choice(good)
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
            sc = _c5(cand)
            if sc < best_sc - 1e-12:
                best, best_sc = cand, sc
                continue

        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if ones.size and zeros.size:
            i, j = rng.choice(ones), rng.choice(zeros)
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
            sc = _c5(cand)
            if sc < best_sc - 1e-12:
                best, best_sc = cand, sc
    return best, best_sc

def _continuous_improve(h, rng, deadline):
    """Hill‑climbing with transfers of mass (keeps total fixed)."""
    n = h.shape[0]
    dx = 2.0 / n
    best = h.copy()
    best_sc = _c5(best)
    step = 0.15
    while time.time() < deadline:
        corr = np.correlate(best, 1 - best, mode='full') * dx
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)

        g = np.ones(n, dtype=float)
        if shift >= 0:
            g[:n - shift] -= best[shift:]
            g[shift:] -= best[:n - shift]
        else:
            s = -shift
            g[s:] -= best[:n - s]
            g[:n - s] -= best[s:]

        i_cand = np.where((g == g.max()) & (best > 1e-9))[0]
        j_cand = np.where((g == g.min()) & (best < 1 - 1e-9))[0]

        if i_cand.size and j_cand.size:
            i, j = rng.choice(i_cand), rng.choice(j_cand)
        else:
            i, j = rng.integers(n), rng.integers(n)
            if i == j or best[i] < 1e-9 or best[j] > 1 - 1e-9:
                continue

        delta = min(step, best[i], 1.0 - best[j])
        if delta <= 1e-12:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = _c5(cand)
        if sc < best_sc - 1e-12:
            best, best_sc = cand, sc
    return best, best_sc

def _triangular(N):
    """Triangular “tent” function discretised on N points."""
    x = np.linspace(0, 2, N, endpoint=False) + 1.0 / N
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    h = _adjust_sum(h, N / 2.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h, best_sc, best_n = None, float('inf'), None

    # if a parent solution exists, evaluate it first
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.ndim == 1:
            N0 = len(h0)
            h0 = _adjust_sum(h0, N0 / 2.0)
            sc0 = _c5(h0)
            best_h, best_sc, best_n = h0, sc0, N0

    # candidate lengths (even, not too large)
    lengths = [70, 80, 90, 100, 120, 140]
    while time.time() - start < budget_s * 0.96:
        N = int(rng.choice(lengths))
        if N % 2:
            N += 1
        half = N // 2

        # -------- start from a triangular shape ------------
        h_tri = _triangular(N)
        deadline = time.time() + min(0.3, (budget_s - (time.time() - start)) * 0.2)
        h_tri, sc_tri = _continuous_improve(h_tri, rng, deadline)
        if sc_tri < best_sc:
            best_h, best_sc, best_n = h_tri.copy(), sc_tri, N
            if best_sc <= 0.3808:
                break

        # -------- many random binary starts -----------------
        for _ in range(30):
            if time.time() - start > budget_s * 0.96:
                break
            hb = np.zeros(N, dtype=float)
            idx = rng.choice(N, size=half, replace=False)
            hb[idx] = 1.0

            # binary local improvement
            t_bin = time.time() + min(0.25, (budget_s - (time.time() - start)) * 0.2)
            hb, scb = _binary_improve(hb, rng, t_bin)
            if scb < best_sc:
                best_h, best_sc, best_n = hb.copy(), scb, N
                if best_sc <= 0.3808:
                    break

            # continuous refinement from the binary result
            t_cont = time.time() + min(0.25, (budget_s - (time.time() - start)) * 0.2)
            hb, scb = _continuous_improve(hb, rng, t_cont)
            if scb < best_sc:
                best_h, best_sc, best_n = hb.copy(), scb, N
                if best_sc <= 0.3808:
                    break

    # fallback – simple half‑ones block if nothing was found
    if best_h is None:
        N = 70
        hb = np.zeros(N, dtype=float)
        hb[:N // 2] = 1.0
        best_h, best_sc, best_n = hb, _c5(hb), N

    # final projection to satisfy constraints exactly
    N = best_n
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _adjust_sum(best_h, N / 2.0)

    final_sc = _c5(best_h)
    return (best_h, float(final_sc), N)
