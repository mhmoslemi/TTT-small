    start_time = time.time()

        tol = 0.01

            candidate_h[i] = new_val
            
            # Adjust the total sum if necessary
            delta_sum = new_val - h0[i]
            candidate_h = candidate_h.copy()
            if abs(delta_sum) > 1e-10:
                # This candidate might violate the sum constraint
                pass
            
            # Evaluate this candidate if we haven't found too many bad points
            c5_new = compute_c5(candidate_h)
            evaluations += n  # each evaluation does a full correlation
        
        else:
            # Large jump style
            jump = np.random.randint(1, 10)
            i_rand = np.random.randint(0, n)
            j_rand = (i_rand + np.random.randint(1, jump)) % n
            # But ensure not to change the same value if jump=0
            if i_rand == j_rand and jump == 0:
                pass
            else:
                # Swap two random values
                # Ensure this maintains the sum constraint automatically
                candidate_h = h0.copy()
                candidate_h[i_rand], candidate_h[j_rand] = candidate_h[j_rand], candidate_h[i_rand]
                # No need for delta_sum adjustment
                c5_new = compute_c5(candidate_h)
                evaluations += n
        
        # Track the best solution
        if c5_new < best_c5 or np.isnan(c5_new) or np.isinf(c5_new):
            continue
        
        h_new = best_h
        best_h = candidate_h
        best_c5 = c5_new
