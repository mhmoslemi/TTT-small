    n = len(h)

    i = np.arange(n)
    i_lag = i + lag

    grad[mask] += 1
    grad[mask] -= h[i_lag[mask]]
    i_minus = i - lag

    grad[mask2] -= h[i_minus[mask2]]

def optimize_h(h, max_time):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_h = h.copy()
    step = 0.5  # initial move magnitude
    iter_since_improve = 0
    start = time.time()
    while time.time() - start < max_time:
        conv = np.correlate(cur_h, 1 - cur_h, mode='full')
        cur_max = conv.max()
        cur_val = cur_max * dx
        if cur_val < best_val - 1e-9:
            best_h = cur_h.copy()
            best_val = cur_val
            iter_since_improve = 0
        # choose worst lag
        idx = conv.argmax()
        lag = idx - (n - 1)
        grad = lag_gradient(cur_h, lag)
        # find positive and negative gradient indices
        pos_idx = np.argmax(grad)  # max positive
        neg_idx = np.argmin(grad)  # most negative
        if grad[pos_idx] <= 1e-12 or grad[neg_idx] >= -1e-12:
            # No improving direction, maybe reduce step
            step *= 0.5
            if step < 1e-6:
                break
            continue
        eps = min(step, cur_h[pos_idx], 1 - cur_h[neg_idx])
        if eps <= 0:
            step *= 0.5
            continue
        cur_h[pos_idx] -= eps
        cur_h[neg_idx] += eps
        iter_since_improve += 1
        # adjust step decay
        if iter_since_improve % 500 == 0:
            step *= 0.9
    return best_h, best_val

import numpy as np, time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx, corr

def lag_gradient(h, lag):
    n = len(h)
    grad = np.zeros(n)
    i = np.arange(n)
    i_lag = i + lag
    mask = (i_lag >= 0) & (i_lag < n)
    grad[mask] += 1
    grad[mask] -= h[i_lag[mask]]
    i_minus = i - lag
    mask2 = (i_minus >= 0) & (i_minus < n)
    grad[mask2] -= h[i_minus[mask2]]
    return grad

def improve_h(h, max_time, start_time):
    # gradient descent with loop limited by time
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_val, _ = compute_c5(h)
    cur_h = h.copy()
    step = 0.5   # initial transfer
    # track iteration count for step decay
    iter_no = 0
    while time.time() - start_time < max_time:
        # compute correlation
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        cur_max = corr.max()
        cur_val = cur_max * dx
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = cur_h.copy()
        # find lag with max
        idx = corr.argmax()
        lag = idx - (n - 1)
        # compute gradient
        grad = lag_gradient(cur_h, lag)
        pos = np.argmax(grad)
        neg = np.argmin(grad)
        if grad[pos] <= 1e-12 or grad[neg] >= -1e-12:
            # no direction or maybe random perturb
            step *= 0.5
            if step < 1e-6:
                break
            continue
        eps = min(step, cur_h[pos], 1 - cur_h[neg])
        if eps < 1e-12:
            step *= 0.5
            continue
        cur_h[pos] -= eps
        cur_h[neg] += eps
        iter_no += 1
        if iter_no % 500 == 0:
            step *= 0.9
    return best_h, best_val

import numpy as np
import time

def _corr_max(h):
    """Return max overlap integral and full correlation array."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx, corr

def _lag_grad(h, lag):
    """Gradient of I(lag) = sum h_i (1-h_{i+lag}) w.r.t. h."""
    n = h.size
    i = np.arange(n)
    grad = np.zeros(n, float)

    ip = i + lag
    mask = (ip >= 0) & (ip < n)
    grad[mask] += 1.0
    grad[mask] -= h[ip[mask]]

    im = i - lag
    mask2 = (im >= 0) & (im < n)
    grad[mask2] -= h[im[mask2]]
    return grad

def _project_sum_one(h, target):
    """Adjust h in-place so that sum(h) == target while keeping 0≤h≤1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return
    # try uniform shift first
    shift = diff / h.size
    h += shift
    # clip and fix remaining error
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    # fix residual by greedy fill
    i = 0
    while abs(diff) > 1e-12 and i < h.size:
        if diff > 0 and h[i] < 1.0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0.0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        i += 1

def _subgradient_opt(h, budget):
    """Sub‑gradient descent on the worst lag, preserving the sum."""
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_val, _ = _corr_max(h)

    cur = h.copy()
    step = 0.5
    it = 0
    start = time.time()
    while time.time() - start < budget:
        cur_val, corr = _corr_max(cur)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = cur.copy()
        lag = corr.argmax() - (n - 1)
        grad = _lag_grad(cur, lag)

        i_up = np.argmax(grad)          # largest positive gradient → decrease
        i_down = np.argmin(grad)        # most negative gradient → increase
        if grad[i_up] <= 1e-12 or grad[i_down] >= -1e-12:
            step *= 0.5
            if step < 1e-6:
                break
            continue

        eps = min(step, cur[i_up], 1.0 - cur[i_down])
        if eps <= 1e-12:
            step *= 0.5
            continue

        cur[i_up] -= eps
        cur[i_down] += eps

        it += 1
        if it % 500 == 0:
            step *= 0.9
    return best_h, best_val

def _initial_shape(p, n):
    """Symmetric shape h(x)=c*(1-|x-1|^p)_+ with integral 1."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    t = np.abs(x - 1.0)
    base = np.maximum(0.0, 1.0 - t ** p)
    c = (p + 1.0) / (2.0 * p)
    return c * base

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # choose a modest size (still well below 1000)
    n = 240
    target_sum = n / 2.0

    # start from parent if provided, otherwise from a parametric shape
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        if h0.size != n:
            # linear interpolation to required size
            old_x = (np.arange(h0.size) + 0.5) * (2.0 / h0.size)
            new_x = (np.arange(n) + 0.5) * (2.0 / n)
            h0 = np.interp(new_x, old_x, h0, left=0.0, right=0.0)
    else:
        h0 = _initial_shape(2.0, n)

    _project_sum_one(h0, target_sum)

    best_h, best_val = h0.copy(), _corr_max(h0)[0]

    # small sweep over the shape parameter p
    p_vals = np.linspace(1.0, 4.0, 13)   # 13 candidates
    for p in p_vals:
        if time.time() - start_total > budget_s * 0.85:
            break
        h = _initial_shape(p, n)
        _project_sum_one(h, target_sum)
        # give each candidate a short optimisation window
        remain = budget_s - (time.time() - start_total)
        t_alloc = min(0.07 * remain, 5.0)
        h_opt, val_opt = _subgradient_opt(h, t_alloc)
        if val_opt < best_val:
            best_val, best_h = val_opt, h_opt.copy()

    # final stochastic refinements using random transfers
    while time.time() - start_total < budget_s * 0.95:
        h = best_h.copy()
        # random pair and random transfer respecting bounds
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_tr = min(h[i], 1.0 - h[j])
        if max_tr <= 0:
            continue
        eps = max_tr * rng.random()
        h[i] -= eps
        h[j] += eps
        # short local optimisation
        h_opt, val_opt = _subgradient_opt(h, 1.0)
        if val_opt < best_val:
            best_val, best_h = val_opt, h_opt.copy()

    _project_sum_one(best_h, target_sum)
    final_c5 = _corr_max(best_h)[0]
    return best_h, final_c5, n
