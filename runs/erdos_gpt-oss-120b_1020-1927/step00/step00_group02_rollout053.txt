def random_move(h, max_step):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    # Determine allowable delta range:
    low_i = max(-max_step, -h[i]) # cannot go below 0
    high_i = min(max_step, 1 - h[i])
    # delta applied to i is +delta, to j is -delta
    # So delta must also satisfy j constraints: h[j] - delta in [0, 1]
    # So delta <= h[j] (to ensure h[j] - delta >=0)
    # and delta >= h[j] - 1 (to ensure <=1)
    low_j = max(-max_step, h[j] - 1)  # delta must be >= h[j] - 1 (since -delta <= 1 - h[j] => delta >= h[j] - 1)
    high_j = min(max_step, h[j])     # delta <= h[j] (since -delta >= -h[j] => delta <= h[j])
    # delta allowable range:
    low = max(low_i, low_j)
    high = min(high_i, high_j)
    if low >= high:
        return None, None
    delta = np.random.uniform(low, high)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new, delta

        hi_plus = np.concatenate((h[shift:], np.zeros(shift, dtype=h.dtype)))

def random_move(h, max_step):
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    # compute feasible delta range
    low_i = max(-max_step, -h[i])  # can't go below 0
    high_i = min(max_step, 1.0 - h[i])  # can't exceed 1
    low_j = max(-max_step, h[j] - 1.0)  # delta such that h[j] - delta <=1 -> delta >= h[j] - 1
    high_j = min(max_step, h[j])  # delta such that h[j] - delta >=0 -> delta <= h[j]
    low = max(low_i, low_j)
    high = min(high_i, high_j)
    if low >= high:
        return None
    delta = np.random.uniform(low, high)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new
```

import time

import numpy as np

def _evaluate_c5(h):
    """C₅ = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    # correlation with zero padding, then scale by dx = 2/n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def _gradient_shift(h, shift):
    """
    Gradient of I(shift)=∫h(x)(1‑h(x+shift))dx with respect to h,
    for the currently worst shift.
    Returns an array of size n:
        dI/dh_i = 1 - h_{i+shift} - h_{i-shift}
    (out‑of‑range indices are treated as 0).
    """
    n = h.size
    if shift >= 0:
        hi_plus = np.concatenate((h[shift:], np.zeros(shift, dtype=h.dtype)))
        hi_minus = np.concatenate((np.zeros(shift, dtype=h.dtype), h[:n - shift]))
    else:
        s = -shift
        hi_plus = np.concatenate((np.zeros(s, dtype=h.dtype), h[:n - s]))
        hi_minus = np.concatenate((h[s:], np.zeros(s, dtype=h.dtype)))
    return 1.0 - hi_plus - hi_minus


def _random_pair_move(h, max_step):
    """Return a new feasible vector after moving mass between two coordinates."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)

    # feasible delta for i (increase) and j (decrease) while keeping box constraints
    low_i = max(-max_step, -h[i])          # cannot go below 0
    high_i = min(max_step, 1.0 - h[i])     # cannot exceed 1
    low_j = max(-max_step, h[j] - 1.0)     # delta must keep h[j]-delta ≤ 1
    high_j = min(max_step, h[j])           # delta must keep h[j]-delta ≥ 0

    low = max(low_i, low_j)
    high = min(high_i, high_j)
    if low >= high:
        return None
    delta = np.random.uniform(low, high)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new
