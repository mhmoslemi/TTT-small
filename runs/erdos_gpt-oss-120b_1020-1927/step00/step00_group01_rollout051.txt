def local_search(h, time_limit):
    start_time = time.time()
    best_h = h.copy()
    best_val = evaluate(best_h)
    current_h = h.copy()
    current_val = best_val
    n = len(h)
    iter = 0
    while time.time() - start_time < time_limit:
        # propose move
        i, j = random.sample(range(n), 2)
        # compute max move
        max_delta = min(current_h[i], 1 - current_h[j])
        if max_delta == 0:
            continue
        # propose delta random proportion of max_delta
        delta = random.random() * max_delta
        current_h[i] -= delta
        current_h[j] += delta
        new_val = evaluate(current_h)
        # acceptance
        T = max(0.001, 0.1 * (1 - iter / max_iter))  # schedule
        if new_val < current_val or random.random() < np.exp((current_val - new_val) / (T+1e-12)):
            current_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = current_h.copy()
        else:
            # revert move
            current_h[i] += delta
            current_h[j] -= delta
        iter += 1
    return best_h, best_val

def _evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def _gradient_for_active_shifts(h):
    """combined gradient of the active (max‑value) shifts"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.flatnonzero(corr == max_val) - (n - 1)
    grad = np.zeros(n)
    for s in active:
        if s >= 0:
            # (1‑h[i+s]) part
            i1 = np.arange(n - s)
            grad[i1] += (1.0 - h[i1 + s])
            # -h[i‑s] part
            i2 = np.arange(s, n)
            grad[i2] -= h[i2 - s]

            grad[i1] += (1.0 - h[i1 - s2])
            i2 = np.arange(n - s2)
            grad[i2] -= h[i2 + s2]
    return grad
