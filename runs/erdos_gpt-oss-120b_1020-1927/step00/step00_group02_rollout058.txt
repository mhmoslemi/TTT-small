def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time, math

    # ---------- helper: projection onto box [0,1] with prescribed sum ----------
    def project_sum_box(v, target):
        """Project vector v onto {0<=x_i<=1, sum x_i = target}."""
        # Simple bisection on the shift lambda.
        lo = -np.max(v)                # makes v+lo <= 0
        hi = 1.0 - np.min(v)           # makes v+hi >= 1
        for _ in range(60):
            mid = (lo + hi) / 2.0
            x = np.clip(v + mid, 0.0, 1.0)
            s = x.sum()
            if s < target:
                lo = mid
            else:
                hi = mid
        x = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

        # tiny correction for numerical drift
        diff = target - x.sum()
        if abs(diff) > 1e-10:
            free = (x > 0.0) & (x < 1.0)
            if free.any():
                x[free] += diff / free.sum()
                x = np.clip(x, 0.0, 1.0)
            else:
                # all at bounds – fallback scaling
                scale = target / x.sum()
                x = np.clip(x * scale, 0.0, 1.0)
        return x

    # ---------- parameters ----------
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # number of discretisation points – a compromise between speed and quality
    n = 180                     # < 1000, fits the budget nicely
    dx = 2.0 / n
    target_sum = n / 2.0        # because ∫h = 1  ⇒  sum(h)*dx = 1

    # ---------- initial h ----------
    if "initial_h_values" in globals() and initial_h_values is not None:
        init = np.asarray(initial_h_values, dtype=float)
        n0 = init.size
        # linear interpolation to the new grid
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, init)
        h = np.clip(h, 0.0, 1.0)
        h = project_sum_box(h, target_sum)
    else:
        # random feasible start
        h = rng.random(n)
        h = project_sum_box(h, target_sum)

    # ---------- evaluation utilities ----------
    def corr_vals(vec):
        """Full linear correlation of h with (1‑h), multiplied by dx."""
        return np.correlate(vec, 1.0 - vec, mode="full") * dx

    def c5_of(vec):
        return np.max(corr_vals(vec))

    best_h = h.copy()
    best_c5 = c5_of(best_h)

    # ---------- optimisation parameters ----------
    max_iter = 200000                     # absolute cap; loop will stop earlier by time
    alpha0 = 2.0
    alpha_max = 200.0
    alpha_inc = (alpha_max - alpha0) / max_iter
    eta0 = 0.5

    # ---------- main optimisation loop ----------
    it = 0
    while it < max_iter:
        # time guard
        if time.time() - start_time > budget_s * 0.97:
            break

        # current correlation and objective
        corr = corr_vals(h)
        cur_c5 = np.max(corr)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            # early exit if we already beat the target record
            if best_c5 < 0.3808:
                # keep searching a bit for possible further improvement
                pass

        # smoothing parameter for soft‑max approximation of the max
        alpha = min(alpha_max, alpha0 + it * alpha_inc)

        # soft‑max weights (stable formulation)
        max_corr = np.max(corr)
        exp_vals = np.exp(alpha * (corr - max_corr))
        w = exp_vals / exp_vals.sum()       # shape (2n‑1,)

        # gradient of the smoothed objective
        grad = np.zeros_like(h)
        L = 2 * n - 1
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            s = idx - (n - 1)               # shift corresponding to this entry
            if s >= 0:
                lim = n - 1 - s
                if lim >= 0:
                    # term from Σ h_i
                    grad[: lim + 1] += weight
                    # term from –h_i h_{i+s}
                    grad[: lim + 1] -= weight * h[s:]
                # term from –h_{i-s}
                if s <= n - 1:
                    grad[s:] -= weight * h[: n - s]
            else:  # s < 0
                t = -s
                # term from Σ h_i (i ≥ t)
                grad[t:] += weight
                lim2 = n - 1 - t
                if lim2 >= 0:
                    grad[: lim2 + 1] -= weight * h[t:]
                grad[t:] -= weight * h[: n - t]

        grad *= dx                         # chain rule factor (dx appears in each f_s)

        # projected gradient step
        eta = eta0 / math.sqrt(it + 1)
        h_new = h - eta * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h = project_sum_box(h_new, target_sum)

        # occasional tiny random jump to escape flat regions
        if it % 2000 == 0 and it > 0:
            noise = rng.normal(scale=0.015, size=n)
            h_tmp = np.clip(h + noise, 0.0, 1.0)
            h_tmp = project_sum_box(h_tmp, target_sum)
            if c5_of(h_tmp) < c5_of(h):
                h = h_tmp

        it += 1

    # ---------- final polishing ----------
    # make sure the best solution respects constraints exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_box(best_h, target_sum)
    best_c5 = c5_of(best_h)

    # try a simple binary rounding (top‑half = 1, rest = 0) – keep it if it improves
    k = int(target_sum)                     # number of ones required for binary version
    idx_desc = np.argsort(best_h)[::-1]
    binary_h = np.zeros_like(best_h)
    binary_h[idx_desc[:k]] = 1.0
    binary_c5 = c5_of(binary_h)
    if binary_c5 < best_c5:
        best_h = binary_h
        best_c5 = binary_c5

    return (best_h.tolist(), float(best_c5), int(n))
