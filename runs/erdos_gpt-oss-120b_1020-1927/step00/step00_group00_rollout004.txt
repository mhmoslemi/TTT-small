def project_to_capped_simplex(v, s):
    # v: array-like
    v = np.asarray(v, dtype=float)
    # binary search for lambda such that sum(max(min(v - lam, 1), 0)) = s
    lo = -np.max(v) - 1.0   # ensure v - lo > 1 for all
    hi = np.max(v) + 1.0    # ensure v - hi < 0 for all
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0, 1)
        total = x.sum()
        if total > s:
            lo = lam
        else:
            hi = lam
    # compute final
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0, 1)
    # adjust for small numerical error
    diff = s - x.sum()
    if abs(diff) > 1e-10:
        # distribute diff over non-saturated entries
        free = (x > 0) & (x < 1)
        if free.any():
            # add diff uniformly
            x[free] += diff / free.sum()
            # clip again
            x = np.clip(x, 0, 1)
    # final clip ensure all constraints
    return x
```

import numpy as np
import time
import math

# ------------------------------------------------------------
# Helper functions (must be top‑level, no closures)
# ------------------------------------------------------------

def _project_to_capped_simplex(v, s):
    """Project vector v onto {0≤x≤1, sum(x)=s} minimizing ||x‑v||_2."""
    v = np.asarray(v, dtype=float)
    # binary search for lambda
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        if x.sum() > s:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    # tiny adjustment for numerical error
    diff = s - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            add = diff / free.sum()
            x[free] += add
            x = np.clip(x, 0.0, 1.0)
    return x

def _random_h(n, rng):
    """Generate a random feasible h (0≤h≤1, sum = n/2)."""
    v = rng.random(n)
    return _project_to_capped_simplex(v, n / 2.0)

def _upsample_h(old_h, new_n):
    """Linear interpolation of old_h onto a grid of size new_n."""
    old_n = len(old_h)
    if old_n == new_n:
        return old_h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    up = np.interp(x_new, x_old, old_h)
    up = np.clip(up, 0.0, 1.0)
    return _project_to_capped_simplex(up, new_n / 2.0)

def _anneal(start_h, time_limit, rng):
    """Simple simulated‑annealing on the mass‑transfer neighbourhood."""
    h = start_h.copy()
    cur_score = _evaluate(h)
    best_h = h.copy()
    best_score = cur_score
    n = len(h)
    start = time.time()
    while time.time() - start < time_limit:
        # propose a transfer i -> j
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 1e-12:
            # try opposite direction
            max_transfer = min(h[j], 1.0 - h[i])
            if max_transfer <= 1e-12:
                continue
            i, j = j, i
        # random fraction of the feasible amount
        delta = max_transfer * rng.random() * 0.5
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = _evaluate(h_new)

        if new_score < cur_score:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
        else:
            # annealing acceptance
            elapsed = time.time() - start
            remaining = time_limit - elapsed
            if remaining <= 0:
                break
            T = (remaining / time_limit) * 0.02  # temperature schedule
            if T > 0.0 and rng.random() < math.exp(-(new_score - cur_score) / T):
                h = h_new
                cur_score = new_score
    return best_h, best_score
