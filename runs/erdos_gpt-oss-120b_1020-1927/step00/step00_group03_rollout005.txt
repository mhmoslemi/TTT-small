def project_onto_simplex_bounds(v, target):
    # we want h = clip(v - lam, 0, 1) with sum = target.
    # binary search lam.
    lo = -np.max(v) - 1  # ensures v - lo > 1
    hi = 1 - np.min(v)   # ensures v - hi < 0
    # For safety, we can expand bounds.
    for _ in range(60):
        lam = (lo + hi) / 2
        h = np.clip(v - lam, 0, 1)
        s = h.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    # final projection
    h = np.clip(v - (lo+hi)/2, 0, 1)

    # adjust by distributing residual to interior entries not at bound.

    return h

def project_onto_simplex_bounds(v, target):
    # project v onto {0 <= x_i <= 1, sum x_i = target}
    # using lambda shift and clipping
    lo = -np.max(v) - 1.0  # will give all ones after clipping
    hi = 1.0 - np.min(v)   # will give all zeros after clipping
    for _ in range(60):
        lam = (lo + hi) * 0.5
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    h = np.clip(v - lam, 0.0, 1.0)
    # fix tiny numerical drift
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        # distribute diff over entries not at bounds
        free = (h > 1e-12) & (h < 1 - 1e-12)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    # fallback using np.correlate if evaluate_erdos_solution not present
    n = h.shape[0]
    dx = 2.0 / n
    return (np.correlate(h, 1 - h, mode='full') * dx).max()

import numpy as np
import time

# --------------------------------------------------------------
# Helper: projection onto the capped simplex {0<=x_i<=1, sum x_i = target}
def project_onto_simplex_bounds(v: np.ndarray, target: float) -> np.ndarray:
    # find lambda such that clip(v - lambda,0,1) has the required sum
    lo = -np.max(v) - 1.0          # forces all entries to 1 after clipping
    hi = 1.0 - np.min(v)           # forces all entries to 0 after clipping
    for _ in range(60):
        lam = (lo + hi) * 0.5
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    h = np.clip(v - lam, 0.0, 1.0)

    # tiny correction for numerical drift
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


# Fallback evaluator (used only if the provided one is missing)
def compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode="full") * dx).max()


# --------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ----- initialise h -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float)
        n = h.shape[0]
    else:
        n = 200                      # fall‑back resolution
        h = np.full(n, 0.5, dtype=float)

    target = n / 2.0                 # integral constraint (∫h = 1)
    h = project_onto_simplex_bounds(h, target)

    # ----- Adam state ---------------------------------------------------
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # ----- keep the best solution ----------------------------------------
    try:
        best_score = float(evaluate_erdos_solution(h))
    except Exception:
        best_score = float(compute_c5(h))
    best_h = h.copy()

    # ----- optimisation parameters ----------------------------------------
    gamma = 1.0          # smoothness of the max
    gamma_max = 40.0
    gamma_mul = 1.08
    base_lr = 0.05
    it = 0
    no_improve = 0

    # ----- main loop ------------------------------------------------------
    while True:
        if time.time() - start > budget_s * 0.96:
            break

        # autocorrelation and overlap vector
        sum_h = h.sum()
        autocorr = np.correlate(h, h, mode="full")          # length 2n‑1
        O = sum_h - autocorr                                 # overlap for each shift

        # smooth approximation of max(O)
        maxO = O.max()
        expO = np.exp(gamma * (O - maxO))
        w = expO / expO.sum()                               # weights, sum to 1
        w_rev = w[::-1]

        # gradient = 1 - (conv(w, h) + conv(w_rev, h))
        start_idx = n - 1
        conv1 = np.convolve(w, h, mode="full")[start_idx:start_idx + n]
        conv2 = np.convolve(w_rev, h, mode="full")[start_idx:start_idx + n]
        grad = 1.0 - (conv1 + conv2)

        # Adam update
        it += 1
        lr = base_lr / (1.0 + it * 1e-4)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)

        # enforce bounds and integral constraint
        h = project_onto_simplex_bounds(h, target)

        # periodic evaluation
        if it % 20 == 0:
            try:
                cur = float(evaluate_erdos_solution(h))
            except Exception:
                cur = float(compute_c5(h))
            if cur < best_score - 1e-7:
                best_score = cur
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1

        # slowly sharpen the max approximation
        if it % 100 == 0 and gamma < gamma_max:
            gamma = min(gamma_max, gamma * gamma_mul)

        # occasional random jitter when stuck
        if no_improve >= 500:
            h = best_h + np.random.normal(scale=0.02, size=n)
            h = np.clip(h, 0.0, 1.0)
            h = project_onto_simplex_bounds(h, target)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            gamma = max(1.0, gamma * 0.5)

    n_points = best_h.shape[0]
    return best_h, float(best_score), n_points
