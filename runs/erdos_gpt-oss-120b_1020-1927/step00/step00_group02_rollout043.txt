import time
import numpy as np

def _compute_c5(h):
    """return max_{k} ∫ h(x)(1‑h(x+k))dx  (discretised)"""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def _shape_a(a, n):
    """symmetric shape with linear tails of length a (0≤a≤1) and a flat top.
    a is the geometric length of each tail (in the interval [0,2]).
    """
    dx = 2.0 / n
    x = np.arange(n) * dx + dx * 0.5          # sample at cell centres
    if a <= 0.0:
        # pure rectangle, height 0.5
        h = np.full(n, 0.5, dtype=float)
        return h
    # height required so that area = 1
    H = 1.0 / (2.0 - a)                       # because area = (2‑a)*H =1
    left = x < a
    right = x > (2.0 - a)
    h = np.empty(n, dtype=float)
    # centre part
    centre = ~left & ~right
    h[centre] = H
    # left ramp
    h[left] = H * (x[left] / a)
    # right ramp
    h[right] = H * ((2.0 - x[right]) / a)

    # enforce exact integral (sum = n/2)
    target = n * 0.5
    cur = h.sum()
    if cur != 0.0:
        scale = target / cur
        h *= scale
        np.clip(h, 0.0, 1.0, out=h)
        # tiny remaining deviation – distribute uniformly
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            h += diff / n
            np.clip(h, 0.0, 1.0, out=h)
    return h


def _shape_p(p, n):
    """shape h(x)=sin(πx/2)^p, scaled to have total mass 1."""
    dx = 2.0 / n
    x = np.arange(n) * dx + dx * 0.5
    raw = np.sin(np.pi * x / 2.0) ** p
    # raw.max() == 1, so scaling factor ≤1 ensures bound ≤1
    target = n * 0.5
    cur = raw.sum()
    if cur == 0.0:
        return np.full(n, 0.5, dtype=float)
    scale = target / cur
    if scale > 1.0:
        scale = 1.0
    h = raw * scale
    np.clip(h, 0.0, 1.0, out=h)
    # final tiny correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------

        best_h = init.copy()

    else:

        n = 200                     # moderate size, fast to evaluate

        best_val = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # explore the family with linear tails (parameter a)
    # ------------------------------------------------------------------
