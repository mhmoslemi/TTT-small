    lo = -np.max(v) - 1.0  # perhaps lower bound -max(v) - 1, but safe

import time
import random
import math
import numpy as np

def _project_to_feasible(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {0 <= x_i <= 1, sum x_i = target_sum}."""
    lo = -np.max(v)               # enough to push everything to 0
    hi = 1.0 - np.min(v)          # enough to push everything to 1
    for _ in range(60):           # binary search for the Lagrange multiplier
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = float(x.sum())
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Compute the C5 bound for a given step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation with zero padding, full mode
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step vector h, its C5 bound and the number of points.
    The implementation starts from the provided `initial_h_values`
    (if present) and performs a simple stochastic hill‑climb
    while respecting the constraints.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Load initial construction (if supplied by the harness)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        # fallback: uniform density 0.5 (satisfies the integral automatically)
        n = 200
        h = np.full(n, 0.5, dtype=np.float64)

    n = h.shape[0]
    target = n / 2.0                     # because ∫_0^2 h = 1  →  sum(h)*dx = 1
    h = np.clip(h, 0.0, 1.0)
    h = _project_to_feasible(h, target)

    best_h = h.copy()
    best_val = _c5_value(best_h)

    start_time = time.time()
    time_limit = budget_s * 0.93          # keep a safety margin

    # ------------------------------------------------------------------
    # 2. Simple stochastic local search
    # ------------------------------------------------------------------
    while time.time() - start_time < time_limit:
        # Occasionally restart from a fresh random point
        if random.random() < 0.001:
            cand = _project_to_feasible(np.random.rand(n), target)
        else:
            # Move a random amount of mass from i -> j
            i, j = random.sample(range(n), 2)
            if h[i] <= 1e-12 or h[j] >= 1.0 - 1e-12:
                continue
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 0.0:
                continue
            delta = max_transfer * random.random() * 0.5
            cand = h.copy()
            cand[i] -= delta
            cand[j] += delta

        val = _c5_value(cand)

        # Accept improvement
        if val < best_val:
            best_val = val
            best_h = cand.copy()
            h = cand
        else:
            # occasional probabilistic acceptance (simulated annealing)
            # temperature decreasing linearly with elapsed time
            elapsed = time.time() - start_time
            frac = elapsed / time_limit
            T = 0.05 * (1.0 - frac) + 1e-5
            if random.random() < math.exp(-(val - best_val) / T):
                h = cand

    # ------------------------------------------------------------------
    # 3. Return the best found construction
    # ------------------------------------------------------------------
    final_val = _c5_value(best_h)
    return best_h.tolist(), float(final_val), int(n)
