import time, math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx for a step function sampled on a uniform grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation h with (1-h); mode='full' gives lags -(n-1)..+(n-1)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def project_onto_bounds_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Clamp v to [0,1] and adjust by adding a constant λ so that sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        adj = np.clip(v + lam, 0.0, 1.0)
        s = adj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    if init is not None:
        # Interpolate the parent to the chosen resolution
        xs_old = np.linspace(0.0, 2.0, base_n, endpoint=False)
        xs_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h_parent = np.interp(xs_new, xs_old, init)
        # Project to the feasible box and correct the total mass
        h_parent = project_onto_bounds_and_sum(h_parent, target_sum)
        val_parent = compute_c5(h_parent)
        best_h, best_val = h_parent.copy(), val_parent

        # Also keep a binary version of the parent (useful as a starting point)
        idx = np.argsort(-h_parent)[:target_sum]
        h_bin = np.zeros(n, dtype=float)
        h_bin[idx] = 1.0
        val_bin = compute_c5(h_bin)
        if val_bin < best_val:
            best_h, best_val = h_bin.copy(), val_bin

    # ------------------------------------------------------------------
    # 3. Random‑restart simulated annealing on binary patterns
    # ------------------------------------------------------------------
    # Parameters controlling the annealing schedule
    init_temp = 0.5
    decay = 0.9995
    max_swaps_per_restart = 5000
    target_c5 = 0.3808          # stop early if we reach this bound

    while True:
        now = time.time()
        if now - start_time > budget_s * 0.95:
            break

        # ----- start a fresh random binary configuration -----
        h = np.zeros(n, dtype=float)
        ones_idx = rng.choice(n, size=target_sum, replace=False)
        h[ones_idx] = 1.0
        cur_val = compute_c5(h)
        cur_temp = init_temp

        # ----- annealing loop -----
        for _ in range(max_swaps_per_restart):
            if time.time() - start_time > budget_s * 0.95:
                break

            # pick a random 1 and a random 0 to exchange
            i = rng.choice(np.flatnonzero(h == 1.0))
            j = rng.choice(np.flatnonzero(h == 0.0))
            # propose the swap
            h[i] = 0.0
            h[j] = 1.0
            new_val = compute_c5(h)
            delta = new_val - cur_val

            # accept according to Metropolis rule
            if delta < 0.0 or rng.random() < math.exp(-delta / (cur_temp + 1e-12)):
                cur_val = new_val
            else:
                # revert
                h[i] = 1.0
                h[j] = 0.0

            # keep the best ever seen
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                if best_val <= target_c5:
                    break

            cur_temp *= decay

        if best_val <= target_c5:
            break

    # ------------------------------------------------------------------
    # 4. Final safety check and return
    # ------------------------------------------------------------------
    if best_h is None:
        # fallback: a simple half‑ones pattern
        best_h = np.zeros(n, dtype=float)
        best_h[:target_sum] = 1.0

    final_val = compute_c5(best_h)
    # Ensure feasibility (tiny numerical drift can happen)
    # Clamp and re‑project just in case
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_and_sum(best_h, target_sum)

    return (best_h.tolist(), float(final_val), n)
