def removal_indices(h, lag):
    n = len(h)

def addition_indices(h, lag):
    n = len(h)
    if lag >= 0:
        # condition: h[j]==0 and (j+lag out or h[j+lag]==1)
        out_range = np.arange(n) >= n - lag   # j+lag >= n
        # For j where not out_range, need h[j+lag]==1
        # compute mask2: (h[lag:] == 1) for j from 0 to n-lag-1
        mask2 = np.zeros(n, dtype=bool)
        mask2[:n-lag] = (h[lag:] == 1)
        mask = (h == 0) & (out_range | mask2)
        return np.where(mask)[0]
    else:
        L = -lag
        out_range = np.arange(n) < L   # j+lag <0
        mask2 = np.zeros(n, dtype=bool)
        # For j >= L, need h[j-L]==1
        mask2[L:] = (h[:n - L] == 1)
        mask = (h == 0) & (out_range | mask2)
        return np.where(mask)[0]

def hill_climb_binary(h, max_iters, time_limit):
    n = len(h)
    c = np.correlate(h, 1 - h, mode='full')
    cur_max = c.max()
    best_h = h.copy()
    best_val = cur_max
    start = time.time()
    it = 0
    while it < max_iters and (time.time() - start) < time_limit:
        # compute worst lag
        max_idx = int(np.argmax(c))
        lag = max_idx - (n - 1)
        # compute candidate indices
        rem = removal_indices(h, lag)
        add = addition_indices(h, lag)
        # if empty: fallback random
        if rem.size == 0 or add.size == 0:
            # choose random i where h[i]==1 and random j where h[j]==0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
        else:
            i = np.random.choice(rem)
            j = np.random.choice(add)
        # swap
        h[i] = 0
        h[j] = 1
        # compute new correlation
        c_new = np.correlate(h, 1 - h, mode='full')
        new_max = c_new.max()
        if new_max <= cur_max:
            # accept
            c = c_new
            cur_max = new_max
            if new_max < best_val - 1e-12:  # improvement
                best_val = new_max
                best_h = h.copy()
        else:
            # revert
            h[i] = 1
            h[j] = 0
        it += 1
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random
    np.random.seed(seed)
    random.seed(seed)
    # Acquire initial pattern if provided
    init = kwargs.get('initial_h_values')
    if init is None:
        # fallback to random pattern
        base_n = 42
        base = np.zeros(base_n, dtype=int)
        base[np.random.choice(base_n, base_n//2, replace=False)] = 1
    else:
        base = np.array(init, copy=False)
        base = (base > 0.5).astype(int)
        base_n = base.shape[0]
        # adjust to exactly half ones
        target = base_n // 2
        diff = int(base.sum() - target)
        if diff > 0:
            idx = np.where(base == 1)[0]
            to_zero = np.random.choice(idx, diff, replace=False)
            base[to_zero] = 0
        elif diff < 0:
            idx = np.where(base == 0)[0]
            to_one = np.random.choice(idx, -diff, replace=False)
            base[to_one] = 1
    # Upsample factor
    factor = 4
    n = base_n * factor
    h0 = np.repeat(base, factor).astype(int)
    # Ensure half ones after upsampling (should hold)
    # Hill climbing
    total_time = budget_s
    start_all = time.time()
    # define helper functions
    def removal_indices(arr, lag):
        nloc = len(arr)
        if lag >= 0:
            idx = np.arange(0, nloc - lag)
            mask = (arr[idx] == 1) & (arr[idx + lag] == 0)
            return idx[mask]
        else:
            L = -lag
            idx = np.arange(L, nloc)
            mask = (arr[idx] == 1) & (arr[idx - L] == 0)
            return idx[mask]
    def addition_indices(arr, lag):
        nloc = len(arr)
        if lag >= 0:
            out = np.arange(nloc) >= nloc - lag
            mask = np.zeros(nloc, dtype=bool)
            mask[out] = (arr[out] == 0)
            mask[:nloc - lag] = (arr[:nloc - lag] == 0) & (arr[lag:] == 1)
            return np.where(mask)[0]
        else:
            L = -lag
            out = np.arange(nloc) < L
            mask = np.zeros(nloc, dtype=bool)
            mask[out] = (arr[out] == 0)
            mask[L:] = (arr[L:] == 0) & (arr[:nloc - L] == 1)
            return np.where(mask)[0]
    def hill_climb(start_h, max_iter, time_allowed):

        nloc = len(h)
        c = np.correlate(h, 1 - h, mode='full')
        cur_max = c.max()

        best_val_local = cur_max

import numpy as np, time

def run(seed=42, budget_s=1100, **kwargs):
    # -------------------------------------------------------------
    # initialize random seed
    np.random.seed(seed)

    # -----------------------------------------------------------------
    # obtain initial pattern if it exists (global variable or kwarg)
    try:
        init = initial_h_values
    except NameError:
        init = kwargs.get("initial_h_values", None)

    if init is None:
        # fallback: random half‑ones vector of length 42
        base_n = 42
        base = np.zeros(base_n, dtype=int)
        base[np.random.choice(base_n, base_n // 2, replace=False)] = 1
    else:
        base = np.array(init, copy=False)
        base = (base > 0.5).astype(int)                # binarise
        base_n = base.shape[0]
        target = base_n // 2
        diff = int(base.sum() - target)               # fix exact half‑ones
        if diff > 0:
            idx = np.where(base == 1)[0]
            base[np.random.choice(idx, diff, replace=False)] = 0
        elif diff < 0:
            idx = np.where(base == 0)[0]
            base[np.random.choice(idx, -diff, replace=False)] = 1

    # -----------------------------------------------------------------
    # up‑sample (keep binary, preserve half‑ones)
    factor = 4                     # 42 → 168 points (still fast)
    n = base_n * factor
    h0 = np.repeat(base, factor).astype(int)

    # -----------------------------------------------------------------
    # helper utilities ------------------------------------------------
    def removal_indices(arr, lag):
        """indices i with arr[i]==1 and arr[i+lag]==0 (if in range)"""
        N = len(arr)
        if lag >= 0:
            rng = np.arange(0, N - lag)
            mask = (arr[rng] == 1) & (arr[rng + lag] == 0)
            return rng[mask]
        else:
            L = -lag
            rng = np.arange(L, N)
            mask = (arr[rng] == 1) & (arr[rng - L] == 0)
            return rng[mask]

    def addition_indices(arr, lag):
        """indices j with arr[j]==0 and (j+lag out‑of‑range or arr[j+lag]==1)"""
        N = len(arr)
        if lag >= 0:
            out_of = np.arange(N) >= N - lag
            mask = np.zeros(N, dtype=bool)
            mask[out_of] = (arr[out_of] == 0)
            mask[:N - lag] = (arr[:N - lag] == 0) & (arr[lag:] == 1)
            return np.where(mask)[0]
        else:
            L = -lag
            out_of = np.arange(N) < L
            mask = np.zeros(N, dtype=bool)
            mask[out_of] = (arr[out_of] == 0)
            mask[L:] = (arr[L:] == 0) & (arr[:N - L] == 1)
            return np.where(mask)[0]

    def hill_climb(start_h, max_iter, time_allowed):
        """greedy descent: only keep swaps that do not increase C5"""
        h = start_h.copy()
        N = len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        cur_max = corr.max()
        best_h = h.copy()
        best_val = cur_max
        t0 = time.time()
        it = 0
        while it < max_iter and (time.time() - t0) < time_allowed:
            # worst lag
            idx_max = int(np.argmax(corr))
            lag = idx_max - (N - 1)

            rem = removal_indices(h, lag)
            add = addition_indices(h, lag)
            if rem.size == 0 or add.size == 0:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
            else:
                i = np.random.choice(rem)
                j = np.random.choice(add)

            # try swap
            h[i] = 0
            h[j] = 1
            new_corr = np.correlate(h, 1 - h, mode='full')
            new_max = new_corr.max()
            if new_max <= cur_max + 1e-12:
                corr = new_corr
                cur_max = new_max
                if new_max < best_val - 1e-12:
                    best_val = new_max
                    best_h = h.copy()
            else:
                # revert swap
                h[i] = 1
                h[j] = 0
            it += 1
        return best_h, best_val

    # -----------------------------------------------------------------
    # main search loop ------------------------------------------------
    total_start = time.time()
    best_h = h0.copy()
    best_val = np.correlate(best_h, 1 - best_h, mode='full').max()
    target_bound = 0.3808

    while True:
        elapsed = time.time() - total_start
        if elapsed >= budget_s - 5:          # keep small safety margin
            break
        remaining = budget_s - elapsed
        # propose a starting point: slight perturbation of best or fresh random
        if np.random.rand() < 0.7:
            h_start = best_h.copy()
            for _ in range(3):
                ones = np.where(h_start == 1)[0]
                zeros = np.where(h_start == 0)[0]
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                h_start[i] = 0
                h_start[j] = 1
        else:
            h_start = np.zeros(n, dtype=int)
            h_start[np.random.choice(n, n // 2, replace=False)] = 1

        # allocate a slice of remaining time (max 30 s) for this climb
        slice_time = min(remaining * 0.4, 30.0)
        opt_h, opt_val = hill_climb(h_start, max_iter=50000,
                                    time_allowed=slice_time)
        if opt_val < best_val:
            best_h, best_val = opt_h, opt_val
        if best_val * (2.0 / n) < target_bound:
            break

    c5_bound = best_val * (2.0 / n)
    return best_h.astype(float), float(c5_bound), int(n)
