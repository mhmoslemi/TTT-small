    def compute_c5(h):
        cor = np.correlate(h, 1.0 - h, mode='full')
        return cor.max() * dx

    # ensure constraints
    # Clip and adjust sum exactly using simple water-filling
    def project(h):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target_sum - s
        # apply diff iteratively
        # To keep things simple, distribute diff equally among free entries
        for _ in range(10):

def _compute_c5(h, dx):

    # correlation of h with (1-h)
