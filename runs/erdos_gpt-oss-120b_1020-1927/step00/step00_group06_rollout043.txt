    cur = h0.copy()

    best = cur.copy()

        # Temperature schedule
        T = max(0.001, (time_limit - (time.time() - start)) / time_limit) # linear cooling
        # propose swap
        idx1 = np.random.choice(np.where(cur == 1)[0])
        idx0 = np.random.choice(np.where(cur == 0)[0])

        new_score = compute_maxc(cur)

                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[idx1], cur[idx0] = cur[idx0], cur[idx1]
    return best, best_score

import numpy as np, time, math

def compute_raw_max_missing(h):
    # Returns raw maximum of correlation h * (1 - h_shift)
    return np.max(np.correlate(h, 1 - h, mode='full'))

def project_to_constraints(v, target_sum):
    # Clip to [0,1] then adjust sum by adding diff to a feasible entry
    v = np.clip(v, 0.0, 1.0)
    cur_sum = v.sum()
    diff = target_sum - cur_sum
    if abs(diff) < 1e-12:
        return v
    # Find an index we can adjust without leaving bounds
    for i in range(len(v)):
        if 0 <= v[i] + diff <= 1:
            v[i] += diff
            return v
    # If not found (should not happen), distribute uniformly
    v += diff / len(v)
    v = np.clip(v, 0.0, 1.0)
    # final adjustment if sum still off (tiny)
    return v

import time
import math
import numpy as np

def _raw_max_missing(h: np.ndarray) -> float:
    """Maximum of the raw missing‐overlap sum  Σ h[i]*(1‑h[i+k])."""
    return float(np.max(np.correlate(h, 1 - h, mode="full")))


def _project(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust a single entry to hit the exact sum."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return v
    # try to fix the sum by tweaking one feasible coordinate
    for i in range(v.shape[0]):
        if 0.0 <= v[i] + diff <= 1.0:
            v[i] += diff
            return v
    # fallback – distribute uniformly (tiny correction)
    v += diff / v.shape[0]
    return np.clip(v, 0.0, 1.0)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Find a step function h on [0,2] with integral 1 that lowers the
    Erdős C₅ bound.  Returns (h_values, c5_bound, n_points)."""

    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------
    # 1. initialise problem size and starting vector
    # ------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
        n = h0.size
        target = n / 2.0
        cur = _project(h0, target)
    else:
        # fall back to an even size that is easy to handle as binary
        n = 100
        target = n / 2.0
        cur = np.zeros(n, dtype=np.float64)
        cur[: n // 2] = 1.0
        np.random.shuffle(cur)

    # ------------------------------------------------------------
    # 2. basic bookkeeping
    # ------------------------------------------------------------
    cur_raw = _raw_max_missing(cur)
    best_raw = cur_raw
    best = cur.copy()

    # Simulated‑annealing parameters
    T = 1.0               # initial temperature
    T_min = 1e-6
    decay = 0.99995
    no_improve = 0
    target_bound = 0.3808   # stop early if we beat the record

    # ------------------------------------------------------------
    # 3. main optimisation loop
    # ------------------------------------------------------------
    while time.time() - start < budget_s * 0.95:
        elapsed = time.time() - start
        frac = elapsed / budget_s

        # pick two distinct positions
        i, j = np.random.choice(n, 2, replace=False)
        direction = np.random.rand() < 0.5   # True: move mass from j → i

        if direction:
            max_delta = min(1.0 - cur[i], cur[j])
        else:
            max_delta = min(1.0 - cur[j], cur[i])

        if max_delta <= 0.0:
            continue

        # step size shrinks from ~0.5 to ~0.01 of the allowed amount
        step_factor = 0.5 * (1.0 - frac) + 0.01
        delta = max_delta * step_factor * np.random.rand()

        # apply the transfer
        if direction:
            cur[i] += delta
            cur[j] -= delta
        else:
            cur[j] += delta
            cur[i] -= delta

        new_raw = _raw_max_missing(cur)

        # acceptance test
        if new_raw <= cur_raw:
            accept = True
        else:
            prob = math.exp(-(new_raw - cur_raw) / (T + 1e-12))
            accept = np.random.rand() < prob

        if accept:
            cur_raw = new_raw
            if new_raw < best_raw - 1e-12:

                best = cur.copy()
