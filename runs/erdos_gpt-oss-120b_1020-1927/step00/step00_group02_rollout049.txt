import time
import math
import numpy as np

def _c5_value(h):
    """Compute C5 for a binary step function h on [0,2]."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def _make_initial(n_target=180, rng=None):
    """Return a feasible binary vector of length n_target."""
    # use the provided initial_h_values if present
    if 'initial_h_values' in globals() and initial_h_values is not None:
        base = np.asarray(initial_h_values, dtype=np.float64)
        n0 = base.shape[0]
        factor = max(1, n_target // n0)
        h = np.repeat(base, factor)
        # ensure strict binary values
        h = (h >= 0.5).astype(np.float64)
        n = h.shape[0]
    else:
        n = n_target
        h = np.zeros(n, dtype=np.float64)
        ones = rng.choice(n, size=n // 2, replace=False)
        h[ones] = 1.0
    # adjust length if not exactly n_target
    if h.shape[0] != n_target:
        # simple truncation / padding to attain the right size
        if h.shape[0] > n_target:
            h = h[:n_target]
        else:
            pad = np.zeros(n_target - h.shape[0], dtype=np.float64)
            h = np.concatenate([h, pad])
    # final feasibility check
    cur = int(h.sum())
    target = n_target // 2
    if cur != target:
        # fix by flipping the needed number of bits
        diff = target - cur
        if diff > 0:
            zeros = np.where(h == 0)[0]
            flip = rng.choice(zeros, size=diff, replace=False)
            h[flip] = 1.0
        else:
            ones = np.where(h == 1)[0]
            flip = rng.choice(ones, size=-diff, replace=False)
            h[flip] = 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with a low C5 value."""

    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- parameters --------------------------------------------------------
    N = 180                     # number of discretisation points (must be even)
    MAX_NO_IMPROVE = 30000      # restart after this many futile swaps
    T0 = 0.5                    # initial temperature
    T_end = 1e-4                # final temperature

    # ----- initialise ---------------------------------------------------------
    cur_h = _make_initial(N, rng)
    cur_val = _c5_value(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    no_improve = 0

    # ----- main simulated‑annealing loop ------------------------------------
    while True:
        # time guard
        if time.time() - start_time > budget_s * 0.99:
            break

        # linear cooling schedule
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s * 0.99)
        T = T0 * (1.0 - frac) + T_end * frac

        # propose a swap (keep the half‑ones constraint)
        i = rng.integers(0, N)
        while cur_h[i] != 1.0:
            i = rng.integers(0, N)
        j = rng.integers(0, N)
        while cur_h[j] != 0.0:
            j = rng.integers(0, N)

        # apply swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_val = _c5_value(cur_h)

        delta = new_val - cur_val
        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis criterion
            if T > 0.0:
                prob = math.exp(-delta / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert the swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            no_improve += 1

        # occasional restart to escape deep local minima
        if no_improve >= MAX_NO_IMPROVE:
            # start from a fresh random balanced vector
            cur_h = np.zeros(N, dtype=np.float64)
            ones = rng.choice(N, size=N // 2, replace=False)
            cur_h[ones] = 1.0
            cur_val = _c5_value(cur_h)
            no_improve = 0

    # ----- post‑process -------------------------------------------------------
    # ensure binary step function and exact half‑ones
    best_h = np.where(best_h >= 0.5, 1.0, 0.0)
    cur_sum = int(best_h.sum())
    target = N // 2
    if cur_sum != target:
        diff = target - cur_sum
        if diff > 0:
            zeros = np.where(best_h == 0)[0]
            flip = rng.choice(zeros, size=diff, replace=False)
            best_h[flip] = 1.0
        else:
            ones = np.where(best_h == 1)[0]
            flip = rng.choice(ones, size=-diff, replace=False)
            best_h[flip] = 0.0

    final_val = _c5_value(best_h)

    # return as list for compatibility
    return best_h.tolist(), float(final_val), N
