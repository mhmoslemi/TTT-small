def project_feasible(v, S):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if total == S:
        return v
    if total < S:
        # need to add
        deficit = S - total
        # indices where v < 1
        idx = np.where(v < 1)[0]
        while deficit > 1e-12 and idx.size > 0:
            # equally distribute
            add = deficit / idx.size
            # cannot exceed 1
            inc = np.minimum(1 - v[idx], add)
            v[idx] += inc
            deficit -= inc.sum()
            idx = np.where(v < 1)[0]
    else:
        # need subtract
        excess = total - S
        idx = np.where(v > 0)[0]
        while excess > 1e-12 and idx.size > 0:
            sub = excess / idx.size
            dec = np.minimum(v[idx], sub)
            v[idx] -= dec
            excess -= dec.sum()
            idx = np.where(v > 0)[0]
    return v
```

import numpy as np, time, math

def project_feasible(v, target):
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) <= 1e-12:
        return v
    if total < target:
        deficit = target - total
        # indices where v < 1
        idx = np.where(v < 1)[0]
        while deficit > 1e-12 and idx.size > 0:
            add = deficit / idx.size
            inc = np.minimum(1 - v[idx], add)
            v[idx] += inc
            deficit -= inc.sum()
            idx = np.where(v < 1)[0]
    else:
        excess = total - target
        idx = np.where(v > 0)[0]
        while excess > 1e-12 and idx.size > 0:
            sub = excess / idx.size
            dec = np.minimum(v[idx], sub)
            v[idx] -= dec
            excess -= dec.sum()
            idx = np.where(v > 0)[0]
    return v
```

import time, math
import numpy as np

def _project_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the box [0,1]^n with sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) <= 1e-12:
        return v
    # need to add mass
    if total < target:
        deficit = target - total
        while deficit > 1e-12:
            idx = np.where(v < 1)[0]
            if idx.size == 0:
                break
            add = deficit / idx.size
            inc = np.minimum(1.0 - v[idx], add)
            v[idx] += inc
            deficit -= inc.sum()
    else:  # need to subtract mass
        excess = total - target
        while excess > 1e-12:
            idx = np.where(v > 0)[0]
            if idx.size == 0:
                break
            sub = excess / idx.size
            dec = np.minimum(v[idx], sub)
            v[idx] -= dec
            excess -= dec.sum()
    return v


def _max_corr(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1-h(x+k)) dx for the discretised h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _refine(h: np.ndarray) -> np.ndarray:
    """Local sub‑gradient refinement on the worst shift."""
    n = h.size
    target = n / 2.0
    for _ in range(500):
        raw = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(raw)
        k = idx - (n - 1)  # shift producing the maximum overlap

        # forward[i] = h[i+k] (zero when out of range)
        forward = np.zeros_like(h)
        # backward[i] = h[i-k] (zero when out of range)
        backward = np.zeros_like(h)

        if k >= 0:
            forward[: n - k] = h[k:]
            backward[k:] = h[: n - k]
        else:
            kk = -k
            forward[kk:] = h[: n - kk]
            backward[: n - kk] = h[kk:]

        grad = 1.0 - forward - backward          # ∇O_k
        step = 0.05 * grad                       # small step toward reduction
        h_new = _project_feasible(h - step, target)

        if np.max(np.correlate(h_new, 1.0 - h_new, mode="full")) < np.max(raw):
            h = h_new
        else:
            break
    return h

    best_h = h0.copy()

    # evolutionary search
    pop_size = 30

        order = np.argsort(fitness)
