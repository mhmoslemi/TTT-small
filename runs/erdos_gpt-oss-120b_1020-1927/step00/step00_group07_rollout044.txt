import time
import numpy as np

def _project_sum_and_bounds(x, target, lo=-2.0, hi=2.0, iters=30):
    """Project x onto the box [0,1] while fixing the sum to target.
    Finds lambda such that sum(clip(x+lambda,0,1)) = target."""
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def _gradient_shift(h, k):
    """Gradient of the maximal overlap term for shift k."""
    n = h.shape[0]
    g = np.ones(n, dtype=h.dtype)
    if k >= 0:
        if n - k > 0:
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
    else:
        kk = -k
        if n - kk > 0:
            g[kk:] -= h[: n - kk]
            g[: n - kk] -= h[kk:]
    return g


def _corr_fft(h, L):
    """Linear cross‑correlation of h with (1‑h) using FFT (full length)."""
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(H * G, n=L)
    return conv[: 2 * h.size - 1]


def _initial_feasible(N, target):
    """Create a deterministic feasible start (cosine shape)."""
    x = np.linspace(0.0, 2.0, N, endpoint=False)
    h = 0.5 + 0.5 * np.cos(np.pi * x)   # values already in [0,1]
    # numeric drift may break exact sum; project.
    return _project_sum_and_bounds(h, target)

    np.random.seed(seed)

    # Target number of samples – moderate size for speed / quality trade‑off
    N = 200
    target_sum = N / 2.0
    dx = 2.0 / N
    factor = dx  # max_corr * factor = C5

    # Length for FFT (next power of two >= 2N)
    L = 1 << ((2 * N - 1).bit_length())

    # Build initial vector
    try:
        init = initial_h_values  # provided by the harness, if any
        if len(init) != N:

            new_x = np.linspace(0.0, 2.0, N, endpoint=False)
            h = np.interp(new_x, old_x, init)
        else:
            h = init.astype(float).copy()
        h = _project_sum_and_bounds(h, target_sum)
    except NameError:
        # No parent solution supplied – use a smooth cosine start
        h = _initial_feasible(N, target_sum)

    best_h = h.copy()
    # initial score
    conv = _corr_fft(h, L)
    best_score = conv.max() * factor

    # optimisation parameters
    lr = 0.15            # learning rate
    no_improve = 0
    max_no_improve = 4000
    restart_noise = 0.03

    # main loop
    while time.time() - start_time < budget_s * 0.95:
        # correlation and worst shift
        conv = _corr_fft(h, L)
        max_idx = int(conv.argmax())
        cur_score = conv[max_idx] * factor
        k = max_idx - (N - 1)

        # gradient for that shift and step
        g = _gradient_shift(h, k)
        h_candidate = h - lr * g
        h_candidate = _project_sum_and_bounds(h_candidate, target_sum)

        # evaluate new candidate

        new_score = conv_new.max() * factor

            h = h_candidate
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                no_improve = 0

                no_improve += 1

            lr = 0.12

    best_h = _project_sum_and_bounds(best_h, target_sum)
    final_conv = _corr_fft(best_h, L)
    c5_bound = final_conv.max() * factor
