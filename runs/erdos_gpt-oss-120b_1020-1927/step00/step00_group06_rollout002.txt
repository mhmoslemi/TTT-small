def project_onto_capped_simplex(v, target):
    # ensure sum(v) = target, 0 <= v_i <= 1
    # algorithm: first project onto simplex (nonneg sum = target) then enforce caps by iteratively adjusting.
    # But simpler: use iterative water-filling.
    v = np.clip(v, 0, 1)
    total = v.sum()
    if abs(total - target) < 1e-9:
        return v
    if total < target:
        # need to add (target - total) to entries not at 1
        deficit = target - total
        free = np.where(v < 1)[0]
        while deficit > 1e-9 and free.size > 0:
            add = deficit / free.size
            v[free] += add
            # clip again
            overflow = v[free] > 1
            if overflow.any():
                # some exceed 1 -> set to 1 and recalc deficit
                excess = v[free][overflow] - 1
                v[free][overflow] = 1
                deficit = target - v.sum()
                free = np.where(v < 1)[0]
            else:
                break
    else:
        # total > target, need to subtract
        excess = total - target
        free = np.where(v > 0)[0]
        while excess > 1e-9 and free.size > 0:
            sub = excess / free.size
            v[free] -= sub
            # clip again lower bounds
            under = v[free] < 0
            if under.any():
                # set to 0 and recompute excess
                deficit = -v[free][under]  # negative amount
                v[free][under] = 0
                excess = v.sum() - target
                free = np.where(v > 0)[0]
            else:
                break
    return v

def project_onto_capped_simplex(v, target):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    # water-filling approach
    if total < target:
        # need to add weight
        while True:
            free = v < 1.0
            if not np.any(free):
                break
            deficit = target - v.sum()
            inc = deficit / np.count_nonzero(free)
            v[free] += inc
            # clip overflow
            v = np.clip(v, 0.0, 1.0)
            # break if close
            if abs(v.sum() - target) < 1e-12:
                break
    else:
        # need to reduce
        while True:
            free = v > 0.0
            if not np.any(free):
                break
            excess = v.sum() - target
            dec = excess / np.count_nonzero(free)
            v[free] -= dec

import numpy as np, time, math

def project_onto_capped_simplex(v, target):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    # tolerance
    eps = 1e-12
    if abs(total - target) <= eps:
        return v
    # iterative water-filling
    if total < target:
        # need to increase
        while True:
            deficit = target - v.sum()
            if deficit <= eps:
                break
            free = v < 1.0 - eps
            if not np.any(free):
                # cannot increase further, break
                break
            inc = deficit / np.count_nonzero(free)
            v[free] += inc
            v = np.clip(v, 0.0, 1.0)
    else:
        # need to decrease
        while True:
            excess = v.sum() - target
            if excess <= eps:
                break
            free = v > eps
            if not np.any(free):
                break
            dec = excess / np.count_nonzero(free)
            v[free] -= dec
            v = np.clip(v, 0.0, 1.0)
    return v

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

import time, math
import numpy as np

def _project_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1]^n with sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    eps = 1e-12
    total = v.sum()
    if abs(total - target) <= eps:
        return v
    # increase if below target
    if total < target:
        while True:
            deficit = target - v.sum()
            if deficit <= eps:
                break
            free = v < 1.0 - eps
            if not np.any(free):
                break
            inc = deficit / np.count_nonzero(free)
            v[free] += inc
            v = np.clip(v, 0.0, 1.0)
    # decrease if above target
    else:
        while True:
            excess = v.sum() - target
            if excess <= eps:
                break
            free = v > eps
            if not np.any(free):
                break
            dec = excess / np.count_nonzero(free)
            v[free] -= dec
            v = np.clip(v, 0.0, 1.0)
    return v


def _c5_score(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx for the discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a step function h (as a numpy vector), its C5 bound, and n_points.
    The function tries to minimise C5 using a simple simulated‑annealing search.
    """
    np.random.seed(seed)
    start = time.time()

    # ----- problem size -------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=np.float64)
        n = init.shape[0]
    else:
        # fallback size – small enough to stay fast yet gives some freedom
        n = 120
        init = None

    target_sum = n / 2.0                     # ensures integral 1
    dx = 2.0 / n

    # ----- helpers -------------------------------------------------------
    def random_feasible():
        """Create a random vector satisfying the constraints."""
        v = np.random.rand(n)
        return _project_capped_simplex(v, target_sum)

    # ----- initial solution -----------------------------------------------
    if init is not None:
        cur = _project_capped_simplex(init.copy(), target_sum)
    else:
        cur = random_feasible()
    cur_score = _c5_score(cur)

    best = cur.copy()
    best_score = cur_score

    # ----- simulated annealing parameters ---------------------------------
    base_sigma = 0.01          # minimal perturbation size
    max_sigma = 0.30           # maximal perturbation size at start
    restart_prob = 0.005       # chance to jump to a fresh random point
    min_temp = 1e-3            # temperature never goes to zero

    # ----- main optimisation loop -----------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.97:          # leave a tiny safety margin
            break

        # linearly decreasing temperature (0 → 1)
        T = max(0.0, 1.0 - elapsed / budget_s)
        sigma = base_sigma + max_sigma * T

        # occasional full restart
        if np.random.rand() < restart_prob:
            cand = random_feasible()
            cand_score = _c5_score(cand)
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
            continue

        # propose a neighbour
        cand = cur + sigma * np.random.randn(n)
        cand = _project_capped_simplex(cand, target_sum)
        cand_score = _c5_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
        else:
            temp = max(min_temp, T)
            if np.random.rand() < math.exp(-(cand_score - cur_score) / temp):
                cur, cur_score = cand, cand_score

        # keep the best ever seen
        if cand_score < best_score:
            best, best_score = cand.copy(), cand_score

    # ----- final gentle refinement ----------------------------------------
    # a short deterministic hill‑climb with a tiny sigma
    for _ in range(2000):
        cand = best + 0.005 * np.random.randn(n)
        cand = _project_capped_simplex(cand, target_sum)
        cand_score = _c5_score(cand)
        if cand_score < best_score:
            best, best_score = cand.copy(), cand_score

    best = _project_capped_simplex(best, target_sum)   # final feasibility
    final_score = _c5_score(best)

    return best, final_score, n
