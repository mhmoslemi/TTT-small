import numpy as np
import math
import time

def compute_c5(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1‑h(x+k))dx for the discretised step function."""
    dx = 2.0 / h.shape[0]
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a feasible vector: entries in [0,1] and sum = n/2."""
    h = np.zeros(n, dtype=float)
    half = n // 2
    order = rng.permutation(n)
    h[order[:half]] = 1.0
    if n % 2:
        h[order[half]] = 0.5
    return h


def mutate(h: np.ndarray, rng: np.random.Generator, max_moves: int = 5) -> np.ndarray:
    """Transfer a random amount of mass between indices while preserving the total."""
    n = h.shape[0]
    new_h = h.copy()
    moves = rng.integers(1, max_moves + 1)
    for _ in range(moves):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # choose direction of transfer
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_transfer = min(new_h[src], 1.0 - new_h[dst])
        if max_transfer <= 0.0:
            continue
        delta = max_transfer * rng.random()
        new_h[src] -= delta
        new_h[dst] += delta
    return new_h
