    # clip to [0,1] after shift.

    lo = -1e5; hi = 1e5

        if s > target:
            lo = λ
        else:

    return w

def project_to_sum_and_bounds(v, target):
    # Project onto [0,1] box and sum constraint.
    # Use binary search for lambda.
    lo = -1.0  # low bound for lambda
    hi = 1.0   # high bound for lambda
    # adjust bounds
    # ensure lo and hi are enough: For lambda very negative, v - λ huge >1 => clip yields 1 for all => sum = len(v)
    # For lambda very positive, v - λ <0 => clip yields 0 => sum =0.
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    # binary search
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return w

def local_refine(h0, budget_s):
    import time, numpy as np, random
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    N = len(h)
    # number of 1s and 0s pre-calc
    start_time = time.time()
    # Determine step sizes? Use decreasing epsilon schedule.
    while time.time() - start_time < budget_s:
        # pick random indices i,j with i has value > 0, j has value <1
        i = np.random.randint(N)
        if h[i] <= 0:
            continue
        j = np.random.randint(N)
        if h[j] >= 1:
            continue
        # choose epsilon
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        # epsilon random fraction of max_eps (like 0.1 to 1)
        eps = max_eps * np.random.uniform(0.01, 0.5)  # moderate
        # apply
        h_candidate = h.copy()
        h_candidate[i] -= eps

def solve_milp(N):
    import cvxpy as cp
    import numpy as np
    h = cp.Variable(N, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == N // 2]
    for s in range(1, N):
        m = N - s
        w = cp.Variable(m, boolean=True)
        constraints.append(w <= h[s:])
        constraints.append(w <= h[:N - s])
        constraints.append(w >= h[s:] + h[:N - s] - 1)
        O_s = cp.sum(h[s:]) - cp.sum(w)
        constraints.append(O_s <= t)
    constraints.append(t >= 0)
    prob = cp.Problem(cp.Minimize(t), constraints)
    # use ECOS_BB (default branch-and-bound) or GLPK_MI if available
    try:
        prob.solve(solver=cp.ECOS_BB, verbose=False, max_iters=500000)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False)
        except Exception:
            prob.solve(solver=cp.CBC, verbose=False)
    if h.value is None:
        raise RuntimeError('MILP failed')
    h_sol = np.round(np.array(h.value)).astype(float)
    t_val = float(t.value) if t.value is not None else None
    return h_sol, t_val

import time
import random
import numpy as np
import cvxpy as cp

def _compute_c5(h):
    """Maximum overlap integral for a step function given by samples h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)

def _project_to_sum_and_bounds(v, target_sum):
    """
    Projection of a real vector onto the intersection of the box [0,1]^n
    and the hyperplane sum(v)=target_sum (capped simplex).
    Solved by a simple binary search on the Lagrange multiplier.
    """
    lo = -np.max(v) - 2.0          # low enough: all entries become 1 after clipping
    hi = np.max(v) + 2.0           # high enough: all entries become 0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _random_binary_solution(n):
    """Even n, exactly n/2 ones."""
    h = np.zeros(n)
    idx = np.random.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def _solve_milp(N):
    """
    Solve the binary version of the problem for N points
    (exactly N/2 ones) using a mixed‑integer formulation.
    Returns (binary_h, t) where t = max O_s (integer).
    """
    # binary variables for the main function
    h = cp.Variable(N, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == N // 2]

    # auxiliary binary variables w_{i,s} for each shift s and overlapping index i
    for s in range(1, N):
        m = N - s
        w = cp.Variable(m, boolean=True)                # w[i] corresponds to h[i+s] * h[i]
        # linking constraints for product
        constraints.append(w <= h[s:])                   # w_i <= h_{i+s}
        constraints.append(w <= h[:N - s])               # w_i <= h_i
        constraints.append(w >= h[s:] + h[:N - s] - 1)   # w_i >= h_{i+s}+h_i-1
        # O_s = sum_{i=s}^{N-1} h_i - sum w_i
        O_s = cp.sum(h[s:]) - cp.sum(w)
        constraints.append(O_s <= t)

    constraints.append(t >= 0)

    prob = cp.Problem(cp.Minimize(t), constraints)

    # try a few solvers – ECOS_BB is always available
    try:
        prob.solve(solver=cp.ECOS_BB, verbose=False, max_iters=500000)
    except Exception:  # pragma: no cover
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False)
        except Exception:  # pragma: no cover
            prob.solve(solver=cp.CBC, verbose=False)

    if h.value is None:
        raise RuntimeError("MILP failed to produce a solution")
    binary_h = np.round(np.array(h.value)).astype(float)
    t_val = float(t.value) if t.value is not None else None
    return binary_h, t_val

def _local_refine(h0, budget):
    """
    Simple stochastic hill‑climbing on the continuous feasible set.
    Moves a small amount of mass from one coordinate to another.
    """
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_val = _compute_c5(h)
    start = time.time()
    while time.time() - start < budget:
        # pick source (must have >0 mass) and destination (<1)
        i = np.random.randint(n)
        if h[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if h[j] >= 1.0:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        # 20 % of the time do a full swap, otherwise a small step
        if random.random() < 0.2:
            eps = max_eps
        else:
            eps = max_eps * random.random()
        h_candidate = h.copy()
        h_candidate[i] -= eps
        h_candidate[j] += eps
        val = _compute_c5(h_candidate)
        if val < best_val - 1e-9:
            best_val = val
            best_h = h_candidate.copy()
            h = h_candidate
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """
    Finds a step function h on [0,2] with ∫h=1 that minimizes
    max_k ∫ h(x)(1−h(x+k))dx. Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # keep the best solution seen
    best_h = None
    best_c5 = float("inf")
    best_n = None

    # if an initial construction is supplied, record it
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        init_c5 = _compute_c5(init)
        best_h, best_c5, best_n = init, init_c5, len(init)

    # candidate problem sizes (must be even)
    candidate_N = [60, 80, 100, 120]

    for N in candidate_N:
        if time.time() - start_time > 0.9 * budget_s:
            break
        if N % 2:
            N += 1

        # ---- 1) MILP binary solution ---------------------------------
        try:
            h_bin, _ = _solve_milp(N)
        except Exception:  # pragma: no cover
            # fall back to a random feasible binary vector
            h_bin = _random_binary_solution(N)

        c5_bin = _compute_c5(h_bin)
        if c5_bin < best_c5:
            best_h, best_c5, best_n = h_bin.copy(), c5_bin, N

        # ---- 2) Continuous refinement (if time permits) --------------
        remaining = budget_s - (time.time() - start_time)
        if remaining > 4.0:
            refined_h, refined_c5 = _local_refine(best_h.copy(), remaining * 0.6)
            if refined_c5 < best_c5:
                best_h, best_c5, best_n = refined_h.copy(), refined_c5, len(refined_h)

        # early exit if we already beat a reasonable target
        if best_c5 < 0.3808:
            break

    # -----------------------------------------------------------------
    # final safety projection – guarantees feasibility
    if best_h is None:                                    # very unlikely
        N = 60
        best_h = np.zeros(N)
        best_h[: N // 2] = 1.0
        best_c5 = _compute_c5(best_h)
        best_n = N

    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = best_n / 2.0
    if not np.isclose(best_h.sum(), target_sum, atol=1e-6):
        best_h = _project_to_sum_and_bounds(best_h, target_sum)

    final_c5 = _compute_c5(best_h)

    return best_h, final_c5, best_n
