def adjust_sum(h, target):
    diff = target - h.sum()
    eps = 1e-12
    # iterative distribution
    while abs(diff) > 1e-10:
        if diff > 0:
            free = np.where(h < 1 - eps)[0]
            if len(free) == 0: break
            inc = diff / len(free)
            inc = np.minimum(inc, 1 - h[free])
            h[free] += inc
            diff = target - h.sum()
        else:
            free = np.where(h > eps)[0]
            if len(free) == 0: break
            dec = diff / len(free) # diff negative
            dec = np.maximum(dec, -h[free])
            h[free] += dec  # diff negative reduces
            diff = target - h.sum()

def adjust_sum(arr, target):
    arr = np.clip(arr, 0.0, 1.0)
    diff = target - arr.sum()
    # Tolerance
    eps = 1e-12
    max_iter = 10 * len(arr)
    iter_ = 0
    while abs(diff) > 1e-10 and iter_ < max_iter:
        if diff > 0:
            # need to increase sum
            free = np.where(arr < 1 - eps)[0]
            if len(free) == 0:
                break
            inc = diff / len(free)
            inc = np.minimum(inc, 1 - arr[free])
            arr[free] += inc
        else:
            free = np.where(arr > eps)[0]
            if len(free) == 0:
                break
            dec = diff / len(free)  # diff negative
            dec = np.maximum(dec, -arr[free])
            arr[free] += dec
        diff = target - arr.sum()
        iter_ += 1
    # final clip
    arr = np.clip(arr, 0.0, 1.0)
    return arr

            cur = nxt
            cur_val = nxt_val
            if nxt_val < best_val:
                best_val = nxt_val
                best = nxt.copy()
    best = adjust_sum(best, target)
    best_val = evaluate(best)
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # decide grid size
    try:
        init_h = initial_h_values
        n0 = len(init_h)
        n = max(120, n0 * 4)
    except NameError:
        init_h = None
        n = 180
    target = n / 2.0
    if init_h is not None:
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, init_h)
    else:
        h = random_binary(n, target, rng)
    h = adjust_sum(h, target)

    # main optimisation phase
    remaining = budget_s - (time.time() - start)
    main_time = max(0.5, remaining * 0.9)
    best_h, best_val = local_optimize(h, target, main_time, rng)

    # quick perturb‑restart cycles with any spare time
    while time.time() - start < budget_s * 0.95:
        h2 = perturb(best_h, rng, max_delta=0.2)
        left = budget_s - (time.time() - start)
        if left < 5.0:
            break
        slice_time = min(30.0, left * 0.5)
        cand_h, cand_val = local_optimize(h2, target, slice_time, rng)
        if cand_val < best_val:
            best_h, best_val = cand_h, cand_val

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum(best_h, target)
    best_val = evaluate(best_h)
    return best_h.tolist(), float(best_val), int(n)
