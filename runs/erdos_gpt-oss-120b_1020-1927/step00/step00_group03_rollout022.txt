def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    start = time.time()
    np.random.seed(seed)

    # Determine n, initial h.
    try:
        init = np.array(initial_h_values, dtype=float)
        n0 = init.shape[0]
        rep = max(1, int(200 // n0))  # choose replication factor to get n ~200
        # replicate to get larger n
        h0 = np.repeat(init, rep)
        n = h0.shape[0]
    except Exception:
        n = 200
        h0 = np.full(n, 0.5)

    # Ensure sum correct: compute sum and adjust
    total = n / 2.0
    # scaling maybe needed, but with repeat sum same proportion.
    # But if replication not integer, maybe mismatch. Ensure projection.
    def project(v):
        # project onto [0,1] box and sum total
        lo = -np.max(v) - 2.0
        hi = np.max(v) + 2.0
        for _ in range(60):
            mid = (lo + hi) / 2.0
            x = np.clip(v - mid, 0.0, 1.0)
            s = x.sum()
            if s > total:
                lo = mid
            else:
                hi = mid
        return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    h0 = project(h0)  # just to be safe

    best_h = h0.copy()
    # safe evaluate using provided helper
    try:
        best_val = evaluate_erdos_solution(best_h)
    except Exception:

        dot = np.correlate(best_h, 1 - best_h, mode='full')
        dx = 2.0 / n
        best_val = np.max(dot) * dx

    while time.time() - start < budget_s * 0.94:

import time
import numpy as np

def _project_to_capped_simplex(v, total):
    """
    Project vector v onto the set {x | 0 <= x_i <= 1, sum x_i = total }.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > total:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _compute_overlap_linear(h):
    """Linear (non‑periodic) auto‑correlation O[L] = sum_i h_i h_{i+L}."""
    n = h.size
    corr = np.correlate(h, h, mode='full')
    O = corr[n - 1:]               # O[0] ... O[n-1]
    return O


def _gradient_linear(h, w):
    """
    Gradient of the smoothed max I(L) where
    I(L) = sum(h) - O(L) and O is the linear correlation.
    w are the softmax weights (sum to 1).
    """
    n = h.size
    G = np.ones_like(h)            # derivative of sum(h)

    # L = 0 contributes twice the same shift
    w0 = w[0]
    if w0:
        G -= 2.0 * w0 * h

    for L in range(1, n):
        wL = w[L]
        if wL == 0.0:
            continue
        # forward shift (i+L inside range)
        G[: n - L] -= wL * h[L:]
        # backward shift (i-L inside range)
        G[L:] -= wL * h[: n - L]
    return G


def _softmax(x, beta):
    """Stable softmax with temperature beta."""
    mx = x.max()
    y = np.exp(beta * (x - mx))
    s = y.sum()
    if s == 0.0:
        return np.full_like(x, 1.0 / x.size)
    return y / s
