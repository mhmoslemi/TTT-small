```
def project_onto_constraints(h, target_sum):
    # clip
    h = np.clip(h, 0, 1)
    # adjust to target sum using water filling
    # iterate:
    for _ in range(10):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        # free indices: where not at bound
        if diff > 0:
            free = h < 1 - 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            inc = diff / nfree
            h[free] += inc
            # clip again
            h = np.clip(h, 0, 1)
        else:
            free = h > 1e-12
            nfree = free.sum()
            if nfree == 0:
                break
            dec = diff / nfree
            h[free] += dec  # diff negative so subtract
            h = np.clip(h, 0, 1)
    return h
```

def compute_c5_and_shift(h):
    conv = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(conv)
    max_val = conv[idx]
    shift = idx - (len(h)-1)
    return max_val, shift, conv
```

import numpy as np
import time

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------

def compute_c5(h):
    """Return the C5 value for a discretized step function h."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    max_val = conv.max()
    return max_val * (2.0 / n)


def project_to_constraints(h, target):
    """
    Project vector h onto the set:
        0 <= x_i <= 1   and   sum(x) = target
    The projection can be written as x_i = clip(h_i - lambda, 0, 1)
    with a scalar lambda chosen so that the sum constraint holds.
    """
    lo, hi = -10.0, 10.0
    # binary search for lambda
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(h - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def gradient_max_shift(h, shift):
    """
    Compute gradient of the overlap for the shift that attains the maximum.
    The gradient of   C_s(h) = sum_{i where i+shift in range}
                         h_i * (1 - h_{i+shift})
    w.r.t. each component.
    """
    n = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        m = n - shift
        # part where i contributes as the first factor
        grad[:m] = 1.0 - h[shift:]
        # part where i contributes as the second factor
        grad[shift:] -= h[:m]
    else:
        t = -shift
        m = n - t
        grad[t:] = 1.0 - h[:m]
        grad[:m] -= h[t:]
    return grad


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points)
    where h_values satisfies the Erdős constraints and c5_bound is
    the corresponding overlap constant.
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine the base grid size
    init = globals().get("initial_h_values")
    if init is not None:
        base_n = int(len(init))
    else:
        base_n = 0

    # Use a moderately fine grid (>=200 points) but stay <1000
    n = max(200, base_n)
    if n > 1000:
        n = 1000

    dx = 2.0 / n
    target_sum = n / 2.0

    # ----------------------------------------------------------
    # Construct a smooth candidate: h(x) = sin^2(pi x / 2)
    # This function has integral 1 and, analytically, C5 = 3/8 = 0.375.
    # ----------------------------------------------------------
    xs = (np.arange(n) + 0.5) * dx               # mid‑point sample points
    h = np.sin(np.pi * xs / 2.0) ** 2            # values in [0,1]

    # Project onto the exact linear constraints
    h = project_to_constraints(h, target_sum)

    # Initial evaluation
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # ----------------------------------------------------------
    # Optional local refinement (projected sub‑gradient descent)
    # ----------------------------------------------------------
    eta = 0.1                     # initial step size
    iteration = 0
    max_iters = 20000

    while iteration < max_iters and (time.time() - start_time) < (budget_s - 0.5):
        # Find the shift that gives the current maximum overlap
        conv = np.correlate(best_h, 1.0 - best_h, mode="full")
        idx = int(conv.argmax())
        shift = idx - (n - 1)                     # integer shift index

        # Gradient for this shift
        grad = gradient_max_shift(best_h, shift)

        # Candidate step
        cand = best_h - eta * grad
        cand = project_to_constraints(cand, target_sum)

        cand_c5 = compute_c5(cand)

        # Accept if we improve; otherwise shrink the step size
        if cand_c5 < best_c5:
            best_h = cand
            best_c5 = cand_c5
            eta *= 1.02          # slightly accelerate
        else:
            eta *= 0.5           # be more cautious

        iteration += 1

    # ----------------------------------------------------------
    # Return the final construction
    # ----------------------------------------------------------
    return best_h, best_c5, n
