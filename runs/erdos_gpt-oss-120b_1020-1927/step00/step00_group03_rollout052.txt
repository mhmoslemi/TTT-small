import numpy as np, time

def max_corr_fft(h):
    N = len(h)
    size = 2 * N
    fft_h = np.fft.rfft(h, n=size)
    fft_comp = np.fft.rfft(1.0 - h, n=size)
    corr = np.fft.irfft(fft_h * np.conj(fft_comp), n=size)
    return corr.max()

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    start = time.time()
    N = 200  # can be tuned; maybe go up to 256 for power of 2 for FFT speed
    # Use N=256 (power of 2) to speed up rfft. Ensure N even for half ones.
    N = 256
    # half ones = N//2 = 128
    # Initialize binary vector
    h = np.zeros(N, dtype=np.float64)
    perm = np.random.permutation(N)
    h[perm[:N//2]] = 1.0
    # Evaluate current best
    size = 2*N
    cur_val = max_corr_fft(h) * (2.0/N)
    best_h = h.copy()
    best_val = cur_val

    # Simulated annealing parameters
    max_iters = int(5e6)  # large upper bound
    T0 = 0.5
    T_end = 1e-5
    # Compute exponent factor for exponential schedule
    decay = (T_end / T0) ** (1.0 / max_iters)
    # We'll compute T iteratively: T *= decay each iteration
    T = T0

    iteration = 0
    # number of tries per iteration ~ 1; but we may want to attempt many moves per T step; but schedule fine.

    while time.time() - start < budget_s - 0.5 and iteration < max_iters:
        # Choose two distinct indices i,j with h[i] != h[j]
        # We'll randomly pick i, then find j with opposite value
        i = np.random.randint(N)
        # Determine if h[i] is 1 or 0 (or fractional). We'll pick j with different value direction:
        # Choose j random; if same value, continue.
        j = np.random.randint(N)
        if h[i] == h[j]:
            continue
        # Swap (or exchange fractional).
        # For binary case easier: swap values
        h[i], h[j] = h[j], h[i]
        # Compute new value
        new_val = max_corr_fft(h) * (2.0/N)
        # Check acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
        # Update temperature
        T *= decay
        iteration += 1
        # Possibly restart if stuck? we can also vary step size.

    return best_h, best_val, N

import time
import numpy as np

def _max_corr(h: np.ndarray) -> float:
    """Linear cross‑correlation max_{k} Σ h_i (1‑h_{i+k}) using FFT."""
    n = h.shape[0]
    size = 2 * n
    fft_h = np.fft.rfft(h, n=size)
    fft_c = np.fft.rfft(1.0 - h, n=size)
    corr = np.fft.irfft(fft_h * np.conj(fft_c), n=size)
    return float(corr.max())

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Simulated‑annealing search for a step function h on [0,2] that minimises

        C5 = max_k ∫ h(x) (1‑h(x+k)) dx .

    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # 1.  Starting vector ------------------------------------------------
    # ------------------------------------------------------------------
    try:
        # use the provided construction if it exists
        init = np.asarray(initial_h_values, dtype=np.float64)
        N = init.size
        h = init.copy()
        # enforce the exact integral constraint Σh = N/2
        target = N / 2.0
        cur = h.sum()
        if abs(cur - target) > 1e-9:
            # simple proportional correction followed by a tiny fix on the last entry
            h += (target - cur) / N
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                h[-1] = np.clip(h[-1] + diff, 0.0, 1.0)
    except NameError:
        # no parent construction – fall back to a fresh binary pattern
        N = 256                     # power of two → fast FFT
        h = np.zeros(N, dtype=np.float64)
        perm = np.random.permutation(N)
        h[perm[:N // 2]] = 1.0      # exactly half of the points are 1

    # keep a copy of the best point seen so far
    best_h = h.copy()
    best_val = _max_corr(best_h) * (2.0 / N)

    cur_h = h.copy()
    cur_val = best_val

    # ------------------------------------------------------------------
    # 2.  Simulated‑annealing parameters ---------------------------------
    # ------------------------------------------------------------------
    max_iters = 5_000_000                # generous upper bound
    T_start = 0.5
    T_end = 1e-6
    decay = (T_end / T_start) ** (1.0 / max_iters)
    T = T_start
    it = 0

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop -----------------------------------------
    # ------------------------------------------------------------------
    while time.time() - start < budget_s - 0.5 and it < max_iters:
        # pick two distinct indices with different values
        i = np.random.randint(N)
        j = np.random.randint(N - 1)
        if j >= i:
            j += 1
        if cur_h[i] == cur_h[j]:
            # values identical → pick another pair next iteration
            continue

        # propose a swap (preserves Σh automatically)
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        new_val = _max_corr(cur_h) * (2.0 / N)

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):

            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

    # 4.  Return result ---------------------------------------------------
    # ------------------------------------------------------------------
