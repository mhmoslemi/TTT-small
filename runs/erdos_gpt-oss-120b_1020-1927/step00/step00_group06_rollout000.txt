def _project_to_constraints(v, target):
    """
    Project vector v onto the set {x in [0,1]^n : sum(x) = target}.
    Since the projection onto a hyperplane with box constraints can be
    done by adding a constant tau and then clipping, we binary‑search tau.
    """
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    sigma_start = 0.25
    sigma_end = 1e-4

    # Counters for stagnation detection and occasional restarts
    stagnation = 0
    max_stagnation = 8000

    # Main loop
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.96:
            break

        # Linear interpolation of temperature and sigma
        progress = elapsed / (budget_s * 0.96)
        T = T_start * (1.0 - progress) + T_end * progress
        sigma = sigma_start * (1.0 - progress) + sigma_end * progress

        # ------------------------------------------------------------------
        # Propose a new vector preserving sum and bounds
        # ------------------------------------------------------------------
        # Pick two distinct indices and transfer a small amount delta
        i, j = np.random.randint(n, size=2)
        while i == j:
            j = np.random.randint(n)

        delta = np.random.normal(scale=sigma)

        # Determine admissible interval for delta so that both coordinates stay in [0,1]
        lo = max(-h_cur[i], h_cur[j] - 1.0)
        hi = min(1.0 - h_cur[i], h_cur[j])
        if lo >= hi:
            # Infeasible move; skip this iteration
            continue
        delta = np.clip(delta, lo, hi)

        h_new = h_cur.copy()
        h_new[i] += delta
        h_new[j] -= delta

        # ------------------------------------------------------------------
        # Evaluate and possibly accept
        # ------------------------------------------------------------------
        new_score = score(h_new)

        delta_score = new_score - best_score
        if new_score < best_score:
            best_h = h_new.copy()
            best_score = new_score
            stagnation = 0
        else:
            stagnation += 1

        # Metropolis acceptance for the current state (helps escape local minima)
        accept = False
        if new_score < score(h_cur):
            accept = True
        else:
            # Probability exp(-Δ/T)
            if T > 0:
                prob = math.exp(-(new_score - score(h_cur)) / T)
                if random.random() < prob:
                    accept = True
        if accept:
            h_cur = h_new

        # ------------------------------------------------------------------
        # Restart if stuck for a while
        # ------------------------------------------------------------------
        if stagnation >= max_stagnation:
            # generate a fresh feasible point
            h_cur = _project_to_constraints(np.random.rand(n), target_sum)
            stagnation = 0

    # ------------------------------------------------------------------
    # 4. Final projection and scoring
    # ------------------------------------------------------------------
    best_h = _project_to_constraints(best_h, target_sum)
    final_score = score(best_h)

    return best_h.astype(np.float32), final_score, n
