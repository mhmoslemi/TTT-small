import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def evaluate_h(h):
    """Return (C5 bound, raw max correlation) for a given step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    bound = (2.0 / h.size) * max_corr
    return bound, max_corr


def project_box_simplex(v, total, tol=1e-12, max_iter=60):
    """
    Euclidean projection of v onto the set
        { x : 0 <= x_i <= 1 , sum x_i = total }.
    Uses bisection on the Lagrange multiplier.
    """
    lo = np.min(v - 1.0)   # λ lower bound (makes all entries 1)
    hi = np.max(v)         # λ upper bound (makes all entries 0)
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > total:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)

    # tiny correction to hit the exact sum (numeric safety)
    diff = total - x.sum()
    if abs(diff) > tol:
        free = (x > tol) & (x < 1.0 - tol)
        if free.any():
            add = diff / free.sum()
            x[free] += add
            np.clip(x, 0.0, 1.0, out=x)
    return x


def random_binary_vector(n):
    """Return a binary vector of length n with exactly n/2 ones."""
    v = np.zeros(n, dtype=float)
    v[: n // 2] = 1.0
    np.random.shuffle(v)
    return v


def softmax_weights(corr, beta):
    """
    Compute normalized softmax weights for the smoothed max.
    corr : 1‑D array of correlation values.
    Returns the weight vector (sums to 1).
    """
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    w = exps / exps.sum()
    return w


def gradient_step(h, w, total):
    """
    One projected gradient step.
    h : current vector
    w : softmax weights (len = 2*len(h)-1)
    total : required sum of entries ( = n/2 )
    Returns the new feasible vector.
    """
    n = h.size
    # sum2_i = Σ_ℓ w_ℓ h_{i-ℓ}
    conv2 = np.convolve(w, h, mode="full")
    sum2 = conv2[n - 1 : n - 1 + n]

    # sum1_i = Σ_ℓ w_ℓ h_{i+ℓ}  (use reversed w)
    conv1 = np.convolve(w[::-1], h, mode="full")
    sum1 = conv1[n - 1 : n - 1 + n]

    grad = (1.0 - sum1) - sum2
    # modest step length
    step = 0.2
    h_new = h - step * grad
    return project_box_simplex(h_new, total)


def projected_gradient_descent(
    h0,
    total,
    max_iters=2000,
    beta_init=0.5,
    beta_max=12.0,
    beta_factor=1.015,
    time_deadline=None,
):
    """
    Minimise the smoothed max overlap using projected gradient descent.
    Returns best vector seen and its (C5 bound, raw max corr).
    """
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_bound, _ = evaluate_h(h)

    beta = beta_init
    cur_bound, cur_raw = evaluate_h(h)

    for _ in range(max_iters):
        if time_deadline is not None and time.time() > time_deadline:
            break

        # correlation and softmax weights
        corr = np.correlate(h, 1.0 - h, mode="full")
        w = softmax_weights(corr, beta)

        # gradient step
        h_new = gradient_step(h, w, total)

        # evaluate new point
        new_bound, new_raw = evaluate_h(h_new)

        # keep the best ever seen
        if new_bound < best_bound:
            best_bound, best_h = new_bound, h_new.copy()

        # accept only if non‑worsening (simple descent)
        if new_bound <= cur_bound:
            h, cur_bound, cur_raw = h_new, new_bound, new_raw

        # increase temperature (β) gradually to focus on the true maximum
        if beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

    return best_h, (best_bound, best_raw)


def simple_hill_climb(h, total, max_iters=5000, time_deadline=None):
    """
    Very lightweight 1‑swap local search on a binary representation.
    The vector is forced to stay binary (0/1) and to have exactly total ones.
    """
    n = h.size
    # enforce true binaryness
    h_bin = (h >= 0.5).astype(float)
    # correct number of ones if needed
    cur_ones = int(h_bin.sum())
    target = int(total)
    if cur_ones != target:
        # adjust by flipping random bits
        idx = np.arange(n)
        np.random.shuffle(idx)
        diff = target - cur_ones
        if diff > 0:
            zero_idx = idx[h_bin[idx] == 0][:diff]
            h_bin[zero_idx] = 1.0
        else:
            one_idx = idx[h_bin[idx] == 1][: -diff]
            h_bin[one_idx] = 0.0

    best_h = h_bin.copy()
    best_bound, _ = evaluate_h(best_h)

    while max_iters > 0:
        if time_deadline is not None and time.time() > time_deadline:
            break
        max_iters -= 1

        # pick a random 1 and a random 0 to swap
        one_pos = np.random.choice(np.where(best_h == 1)[0])
        zero_pos = np.random.choice(np.where(best_h == 0)[0])

        cand = best_h.copy()
        cand[one_pos], cand[zero_pos] = 0.0, 1.0
        cand_bound, _ = evaluate_h(cand)

        if cand_bound < best_bound:
            best_h, best_bound = cand, cand_bound
        # keep current best (no simulated‑annealing acceptance)
    return best_h, (best_bound, None)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the step function and return (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine problem size and initial feasible vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        # enforce feasibility (clip + exact sum)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_box_simplex(h0, n / 2.0)
    else:
        n = 80
        h0 = random_binary_vector(n)

    total = n / 2.0
    best_h, (best_bound, _) = projected_gradient_descent(
        h0,
        total,
        max_iters=3000,
        beta_init=0.5,
        beta_max=12.0,
        beta_factor=1.02,
        time_deadline=start_time + budget_s * 0.85,
    )

    # Occasionally polish the result with a binary hill‑climb
    # (useful when the optimum is essentially binary)
    if time.time() - start_time < budget_s * 0.95:
        best_h, (best_bound, _) = simple_hill_climb(
            best_h,
            total,
            max_iters=8000,
            time_deadline=start_time + budget_s * 0.98,
        )

    # Final evaluation (ensuring exact feasibility)
    best_h = project_box_simplex(best_h, total)
    final_bound, _ = evaluate_h(best_h)

    return best_h, final_bound, n
