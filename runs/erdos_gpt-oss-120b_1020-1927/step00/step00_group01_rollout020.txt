def proj_simplex_box(y, s):
    # ensure s between 0 and n (as we have sum target)
    n = len(y)
    # Bisection
    lo = -1e5
    hi = 1e5
    for _ in range(60):
        lam = (lo + hi)/2.0
        x = np.clip(y - lam, 0, 1)
        sum_x = x.sum()
        if sum_x > s:
            lo = lam
        else:
            hi = lam
    # final projection
    x = np.clip(y - (lo+hi)/2.0, 0, 1)
    # Might be small discrepancy; adjust minor rounding
    # ensure sum is close: maybe adjust small fraction to match exactly
    diff = s - x.sum()
    if abs(diff) > 1e-12:
        # adjust by distributing diff across non-bound entries
        # find indices where 0 < x < 1
        free = (x > 1e-12) & (x < 1-1e-12)
        if free.any():
            # adjust uniformly
            x[free] += diff / free.sum()
            # clip again
            x = np.clip(x, 0, 1)
        else:
            # fallback: add diff to first element within [0,1] range
            idx = np.argmax(x)  # pick max
            x[idx] = np.clip(x[idx] + diff, 0, 1)
    return x

    # compute weights: w = exp(alpha * Cvec)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def grad_smooth(h, alpha):
    n = h.shape[0]
    Cvec = np.correlate(h, 1 - h, mode='full')
    # compute stable weights
    maxC = np.max(Cvec)
    w = np.exp(alpha * (Cvec - maxC))
    w_sum = np.sum(w)
    w = w / w_sum  # normalized weights
    grad = np.zeros_like(h)

    # compute shift d index mapping: for idx from 0 to 2n-2, shift d = idx - (n - 1)
    for idx in range(2*n-1):
        d = idx - (n - 1)
        weight = w[idx]
        if weight == 0:
            continue
        # i range where i+d valid:
        i_start = max(0, -d)
        i_end = min(n, n - d)  # exclusive
        if i_start < i_end:
            grad[i_start:i_end] += weight * (1 - h[i_start + d:i_end + d])
        # i range where i-d valid:
        i_start2 = max(0, d)
        i_end2 = min(n, n + d)
        if i_start2 < i_end2:
            grad[i_start2:i_end2] += weight * (-h[i_start2 - d:i_end2 - d])
    return grad

def proj_box_simplex(y, s):
    # find lambda such that sum(clip(y - lam, 0, 1)) = s
    lo = -1.0
    hi = 1.0
    # extend range to ensure sum at lo > s and hi < s
    # Actually for lo negative large, see.
    lo = -np.max(y) - 2.0  # ensures y - lo >> 1
    hi = np.max(y) + 2.0   # ensures y - hi << 0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0, 1)
        total = x.sum()
        if total > s:
            lo = lam
        else:
            hi = lam
    x = np.clip(y - (lo + hi) / 2.0, 0, 1)
    # adjust for small diff:
    diff = s - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff across indices not at bound
        free = (x > 1e-12) & (x < 1 - 1e-12)
        free_idx = np.where(free)[0]
        if free_idx.size > 0:
            # add diff equally
            x[free_idx] += diff / free_idx.size
            x = np.clip(x, 0, 1)
        else:
            # adjust the largest entry within bounds
            # find idx with max capacity to add diff
            idx = np.argmin(x) if diff > 0 else np.argmax(x)
            x[idx] = np.clip(x[idx] + diff, 0, 1)
    return x

import time
import numpy as np

# ---------- helper utilities ----------
def _project_onto_box_simplex(y: np.ndarray, target_sum: float) -> np.ndarray:
    """Project y onto the set {x | 0<=x_i<=1, sum(x)=target_sum}."""
    n = y.shape[0]
    lo = -np.max(y) - 2.0          # low enough → all ones after clipping
    hi = np.max(y) + 2.0           # high enough → all zeros after clipping
    for _ in range(60):           # binary search for the Lagrange multiplier
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:         # tiny correction if rounding broke the sum
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            x[idx] += diff / idx.size
            x = np.clip(x, 0.0, 1.0)
        else:
            # all entries are at a bound – adjust the first one that can move
            i = 0 if diff > 0 else -1
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x


def _c5_value(h: np.ndarray) -> float:
    """Return the C5 constant for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _smooth_gradient(h: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of the smooth approximation of max C(L)."""
    n = h.shape[0]
    # correlation values for every lag
    C = np.correlate(h, 1.0 - h, mode='full')
    # smooth weights (log‑sum‑exp trick)
    m = C.max()
    w = np.exp(alpha * (C - m))
    w /= w.sum()                     # normalised so they sum to 1

    grad = np.zeros_like(h)
    # iterate over all possible lags
    for idx, d in enumerate(range(-(n - 1), n)):
        weight = w[idx]
        if weight == 0.0:
            continue
        # contribution where i+d is inside the array
        i_start = max(0, -d)
        i_end = min(n, n - d)
        if i_start < i_end:
            grad[i_start:i_end] += weight * (1.0 - h[i_start + d:i_end + d])
        # contribution where i-d is inside the array
        i_start2 = max(0, d)
        i_end2 = min(n, n + d)
        if i_start2 < i_end2:
            grad[i_start2:i_end2] += weight * (-h[i_start2 - d:i_end2 - d])
    return grad


def _interpolate(src: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a vector to a new length."""
    old_n = src.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, src)


# ---------- main optimisation ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where
        h_values : np.ndarray of shape (n_points,)
        c5_bound : float  (max_{k} ∫ h(x)(1‑h(x+k))dx )
        n_points : int   (the length of h_values)
    """
    np.random.seed(seed)
    start_time = time.time()
    remaining = lambda: budget_s - (time.time() - start_time)

    # ---- obtain a starting vector ----
    try:
        h0 = np.asarray(initial_h_values, dtype=float)  # provided by the harness
    except Exception:                                   # no starter – use random

    n0 = h0.shape[0]

    # target resolution – a modest upscale of the supplied data
    target_n = max(94, 2 * n0)           # at least 94 points, also even if possible
    if target_n > 400:                  # keep the vector reasonably small
        target_n = 400

    h = _interpolate(h0, target_n)
    h = _project_onto_box_simplex(h, target_n / 2.0)

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # ---- optimisation parameters ----
    max_iters = 12000
    alpha = 5.0          # smoothing factor
    lr = 0.5             # initial step size

    for it in range(max_iters):
        if remaining() < 0.1:      # keep a tiny safety margin
            break

        # smooth gradient of the max‑overlap functional
        g = _smooth_gradient(best_h, alpha)

        # gradient step + projection
        candidate = best_h - lr * g
        candidate = _project_onto_box_simplex(candidate, target_n / 2.0)

        cand_c5 = _c5_value(candidate)
