import time, math, numpy as np

def _evaluate(h):
    """C5 value for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1‑h)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _sa_optimize(h, time_limit, rng):
    """
    Simulated‑annealing style optimisation for fixed‑sum vectors.
    - h : initial vector, already satisfies sum(h)=len(h)/2
    - time_limit : seconds allocated to this stage
    Returns the best vector found and its C5 value.
    """
    start = time.time()
    deadline = start + time_limit
    n = h.shape[0]

    cur_val = _evaluate(h)
    best_h = h.copy()
    best_val = cur_val

    # annealing schedule parameters
    T0 = 0.20
    Tmin = 1e-5

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        # move mass between i and j while keeping each entry in [0,1]
        total = h[i] + h[j]
        lo = max(0.0, total - 1.0)
        hi = min(1.0, total)
        a = rng.random() * (hi - lo) + lo
        b = total - a

        old_i, old_j = h[i], h[j]
        h[i], h[j] = a, b

        new_val = _evaluate(h)
        delta = new_val - cur_val

        # temperature based on progress within this stage
        progress = (now - start) / time_limit
        T = T0 * (1.0 - progress) + Tmin * progress

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = old_i, old_j

        it += 1
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
