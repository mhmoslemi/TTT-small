def smooth_max_hessian(alpha, h):
    # compute f_k
    n = len(h)
    dx = 2.0 / n
    f = np.correlate(h, 1 - h, mode='full') * dx  # length 2n-1
    # compute weights
    max_f = f.max()
    # might overflow; compute stable log-sum-exp
    exp_term = np.exp(alpha * (f - max_f))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp
    # compute gradient via conv
    w_arr = w  # length 2n-1
    n = len(h)
    # compute conv_center
    conv_full = np.convolve(w_arr, h, mode='full')
    # conv_center: length n, each i = conv_full[i + (n-1)]
    conv_center = conv_full[(n-1):(n-1)+n]
    # conv_rev using reversed w
    conv_rev_full = np.convolve(w_arr[::-1], h, mode='full')
    conv_rev_center = conv_rev_full[(n-1):(n-1)+n]
    grad = dx * (1 - conv_center - conv_rev_center)
    # return O = max_f approx via log-sum-exp + ... Actually compute O
    O = max_f + np.log(sum_exp) / alpha
    # O = (1/alpha)*logsumexp(alpha*f) = max_f + (1/alpha)*log(sum_exp)
    O = max_f + (1/alpha) * np.log(sum_exp)
    return O, grad

import time
import numpy as np

def _compute_c5(h):
    """Maximum overlap integral for step function h."""
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_to_feasible(v, target):
    """Project vector v to [0,1]^n with sum(v)=target (euclidean projection)."""
    v = np.clip(v, 0.0, 1.0)
    # binary search for lambda so that sum(clip(v-lambda,0,1))==target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size:
            x[free] += diff / free.size
            x = np.clip(x, 0.0, 1.0)
        else:
            total = x.sum()
            if total > 0:
                x = x * (target / total)
                x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_max(alpha, h):
    """Smooth approximation of max_k ∫h·(1‑h(·+k)) and its gradient."""
    n = h.size
    dx = 2.0 / n
    f = np.correlate(h, 1.0 - h, mode="full") * dx          # length 2n‑1
    m = f.max()
    # stable soft‑max weights
    w = np.exp(alpha * (f - m))
    w_sum = w.sum()
    w /= w_sum
    # convolutions to obtain Σ w_k h_{i±k}
    conv = np.convolve(w, h, mode="full")
    conv_c = conv[(n - 1):(n - 1) + n]
    conv_rev = np.convolve(w[::-1], h, mode="full")
    conv_rc = conv_rev[(n - 1):(n - 1) + n]
    grad = dx * (1.0 - conv_c - conv_rc)
    smooth = m + (1.0 / alpha) * np.log(w_sum)
    return smooth, grad

def _binary_hill_climb(n, ones, rng, time_limit, start):
    """Random binary start + 1‑swap hill climbing."""
    best_h = None
    best_val = np.inf
    while time.time() - start < time_limit:

        idx = rng.choice(n, size=ones, replace=False)
        h[idx] = 1.0

        no_imp = 0
        max_no_imp = 500

            if time.time() - start >= time_limit:
                break
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            h[i] = 0.0
            h[j] = 1.0
            new = _compute_c5(h)
            if new < cur - 1e-12:
                cur = new
                no_imp = 0
            else:

                h[j] = 0.0
                no_imp += 1

        if cur < best_val:
            best_val = cur
            best_h = h.copy()
            if best_val <= 0.381:            # early stop threshold
                break
    return best_h, best_val

def _gradient_descent(n, rng, time_limit, start):
    """Projected gradient descent on a smooth max approximation."""
    best_h = None
    best_val = np.inf
    target = n / 2.0
    while time.time() - start < time_limit:
