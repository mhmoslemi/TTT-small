def project_onto_bounds_and_sum(v, target):
    # water-filling; find lambda such that sum_i clip(v_i - lambda, 0, 1) = target
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(60):
        lam = (low + high) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = lam
        else:
            high = lam
    w = np.clip(v - (low+high)/2.0, 0.0, 1.0)
    # enforce sum exactly (tiny adjustments)
    diff = target - w.sum()
    if abs(diff) > 1e-8:
        # distribute diff among non-boundary indices
        # find mask of values not at bounds
        mask = (w > 1e-12) & (w < 1 - 1e-12)
        if mask.any():
            w[mask] += diff / mask.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            # if all at bounds, adjust uniformly (unlikely)
            w += diff / len(w)
            w = np.clip(w, 0.0, 1.0)
    return w

def solve(initial_h, n, time_budget):
    target = n/2
    dx = 2.0/n
    h = initial_h.copy()
    best_h = h.copy()
    conv = np.correlate(h, h, mode='full')
    O = target - conv
    best_c5 = O.max() * dx
    start = time.time()
    iteration = 0
    step0 = 1.0
    while time.time() - start < time_budget * 0.95:   # leave some time
        # recompute overlap array
        conv = np.correlate(h, h, mode='full')
        O = target - conv
        idx = O.argmax()
        shift = idx - (n-1)
        ns = neighbor_sum_shift(h, shift)
        grad = 1.0 - ns
        step = step0 / (1.0 + iteration * 0.001)   # decays slowly
        h_new = h - step * grad
        h_new = project_onto_bounds_and_sum(h_new, target)
        # compute new C5
        conv_new = np.correlate(h_new, h_new, mode='full')
        O_new = target - conv_new
        c5_new = O_new.max() * dx
        # accept if better else maybe accept with prob
        if c5_new < best_c5 - 1e-8:
            best_c5 = c5_new
            best_h = h_new.copy()
        # acceptance
        if c5_new <= O.max() * dx:
            h = h_new
        else:
            # optional simulated annealing
            # probability decreasing
            T = 0.01 / (1 + iteration*0.01)
            if np.random.rand() < np.exp(-(c5_new - (O.max()*dx)) / T):
                h = h_new
        iteration += 1
        # maybe random restart if stuck
        if iteration % 5000 == 0:
            # if no improvement in last 5000 iter, random restart
            if best_c5 < O.max()*dx - 1e-4:
                continue
            # random restart
            # generate random feasible vector (continuous)
            rnd = np.random.rand(n)
            # project to sum constraint
            h = project_onto_bounds_and_sum(rnd, target)
    return best_h, best_c5
```

def gradient_descent_optimize(h0, budget_s):
    import numpy as np, time, random
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    # compute initial best
    conv = np.correlate(h, h, mode='full')
    O = target - conv
    best_c5 = O.max() * dx
    best_h = h.copy()
    start = time.time()
    iteration = 0
    no_improve = 0
    step0 = 1.0
    while time.time() - start < budget_s * 0.95:
        conv = np.correlate(h, h, mode='full')
        O = target - conv
        idx = O.argmax()
        shift = idx - (n - 1)
        ns = neighbor_sum_shift(h, shift)
        grad = 1.0 - ns

        step = step0 / (1.0 + iteration * 0.001)

def binary_local_search(h_bin, max_iters, time_limit):
    n = len(h_bin)
    target = n//2
    total = target
    conv = np.correlate(h_bin.astype(float), h_bin.astype(float), mode='full')
    O = total - conv
    best_Omax = O.max()
    best_h = h_bin.copy()
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        idx = O.argmax()
        shift = idx - (n-1)
        ns = neighbor_sum_shift(h_bin.astype(float), shift) # returns float, but values 0,1,2
        # indices with ones and zeros
        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]
        # compute ns for those
        ns_ones = ns[ones_idx]
        ns_zeros = ns[zeros_idx]
        i_one = ones_idx[np.argmin(ns_ones)]
        j_zero = zeros_idx[np.argmax(ns_zeros)]
        # compute delta O after swapping i_one (1) and j_zero (0)
        delta = compute_delta_swap(h_bin, i_one, j_zero)  # O_new = O - delta
        O_new = O - delta

            h_bin[j_zero] = 1
            # update O = O_new
            O = O_new
            best_Omax = new_Omax
            best_h = h_bin.copy()
        else:
            # cannot improve, break
            break

def compute_delta_swap(h, i, j):
    n = len(h)
    offset = n-1
    delta = np.zeros(2*n-1, dtype=int)
    # i contribution
    # forward
    t_fwd = np.arange(-i, n-i)
    delta[t_fwd + offset] -= h[i + t_fwd]
    # backward
    t_back = np.arange(i-(n-1), i+1)
    delta[t_back + offset] -= h[i - t_back]
    # j contribution
    t_fwd = np.arange(-j, n-j)
    delta[t_fwd + offset] += h[j + t_fwd]
    t_back = np.arange(j-(n-1), j+1)
    delta[t_back + offset] += h[j - t_back]
    return delta
```

def project_onto_bounds_and_sum(v, target):

    high = np.max(v)

        lam = (low + high) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = lam
        else:
            high = lam
    w = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-8:
        mask = (w > 1e-12) & (w < 1 - 1e-12)
        if mask.any():
            w[mask] += diff / mask.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            w += diff / len(w)
            w = np.clip(w, 0.0, 1.0)

import time, random, math
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the box [0,1] and the affine constraint sum(h)=target
# ----------------------------------------------------------------------
def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {0≤x≤1, sum(x)=target}."""
    low = np.min(v) - 1.0          # guarantee all entries become 1
    high = np.max(v)               # guarantee all entries become 0
    for _ in range(60):            # enough for double precision
        lam = (low + high) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = lam
        else:
            high = lam
    w = np.clip(v - (low + high) * 0.5, 0.0, 1.0)
    # tiny correction in case of round‑off
    diff = target - w.sum()
    if abs(diff) > 1e-8:
        mask = (w > 1e-12) & (w < 1.0 - 1e-12)
        if mask.any():
            w[mask] += diff / mask.sum()
            w = np.clip(w, 0.0, 1.0)
        else:                       # fallback (should rarely happen)
            w += diff / w.size
            w = np.clip(w, 0.0, 1.0)
    return w


# ----------------------------------------------------------------------
# Helper: neighbour sum for a given shift (zero padding)
# ----------------------------------------------------------------------
def _neighbour_sum(h: np.ndarray, shift: int) -> np.ndarray:
    """Return h[i+shift]+h[i-shift] with zero outside the interval."""
    n = h.shape[0]
    ns = np.zeros_like(h)
    if shift >= 0:
        s = shift
        if s < n:
            ns[: n - s] += h[s:]
            ns[s:] += h[: n - s]
    else:
        s = -shift
        if s < n:
            ns[s:] += h[: n - s]
            ns[: n - s] += h[s:]
    return ns


# ----------------------------------------------------------------------
# Compute the objective C5 for a given step function
# ----------------------------------------------------------------------
def _c5_value(h: np.ndarray) -> float:
    n = h.shape[0]
    total = h.sum()
    conv = np.correlate(h, h, mode='full')
    O = total - conv                     # O_k = ∫ h(x)(1−h(x+k))dx (discrete)
    return O.max() * (2.0 / n)


# ----------------------------------------------------------------------
# Gradient descent on the worst shift (sub‑gradient method)
# ----------------------------------------------------------------------
def _gd_optimize(h0: np.ndarray, time_budget: float):
    n = h0.shape[0]
    target = float(n) / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_value(h)

    start = time.time()
    it = 0
    no_imp = 0
    step0 = 1.0

    while time.time() - start < time_budget:
        # current worst shift
        conv = np.correlate(h, h, mode='full')
        O = target - conv
        idx = O.argmax()
        shift = idx - (n - 1)

        # sub‑gradient for this shift
        grad = 1.0 - _neighbour_sum(h, shift)

        # simple diminishing step size
        step = step0 / (1.0 + it * 0.001)
        h_new = h - step * grad
        h_new = _project_box_sum(h_new, target)

        val_new = _c5_value(h_new)

        # keep global best
        if val_new < best_val - 1e-10:
            best_val = val_new
            best_h = h_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        # accept only improvements; occasional random jump when stuck
        if val_new < _c5_value(h) - 1e-12:
            h = h_new
        elif no_imp > 3000 and random.random() < 0.01:
            h = h_new

        # restart around current best if stagnating
        if no_imp > 8000:
            h = best_h + 0.02 * (np.random.rand(n) - 0.5)
            h = _project_box_sum(h, target)
            no_imp = 0

        it += 1

    return best_h, best_val


# ----------------------------------------------------------------------
# Binary swap local search (max‑overlap reduction)
# ----------------------------------------------------------------------
def _swap_delta(h: np.ndarray, i: int, j: int) -> np.ndarray:
    """Change of O_k when swapping a 1 at i with a 0 at j."""
    n = h.shape[0]
    off = n - 1
    delta = np.zeros(2 * n - 1, dtype=int)

    # remove contributions of the former 1 at i
    t_fwd = np.arange(-i, n - i)               # i+ t in [0,n)
    delta[t_fwd + off] -= h[i + t_fwd]
    t_back = np.arange(i - (n - 1), i + 1)      # i - t in [0,n)
    delta[t_back + off] -= h[i - t_back]

    # add contributions of the new 1 at j
    t_fwd = np.arange(-j, n - j)
    delta[t_fwd + off] += h[j + t_fwd]
    t_back = np.arange(j - (n - 1), j + 1)
    delta[t_back + off] += h[j - t_back]

    return delta


def _binary_local_search(h0: np.ndarray, time_budget: float):
    n = h0.shape[0]
    target = n // 2
    h = h0.copy()
    total = float(target)

    conv = np.correlate(h.astype(float), h.astype(float), mode='full')
    O = total - conv
    best_O = O.max()
    best_h = h.copy()

    start = time.time()
    while time.time() - start < time_budget:
        idx = O.argmax()
        shift = idx - (n - 1)

        ns = _neighbour_sum(h.astype(float), shift)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        i_one = ones[np.argmin(ns[ones])]
        j_zero = zeros[np.argmax(ns[zeros])]

        delta = _swap_delta(h, int(i_one), int(j_zero))
        O_new = O - delta
        new_O = O_new.max()

        if new_O < best_O - 1e-9:
            # accept the improving swap
            h[i_one] = 0
            h[j_zero] = 1
            O = O_new
            best_O = new_O
            best_h = h.copy()
            continue

        # try a random pair as a last resort
        i_one = random.choice(ones)
        j_zero = random.choice(zeros)
        delta = _swap_delta(h, int(i_one), int(j_zero))
        O_new = O - delta
        new_O = O_new.max()
        if new_O < best_O - 1e-9:
            h[i_one] = 0
            h[j_zero] = 1
            O = O_new
            best_O = new_O
            best_h = h.copy()
        else:
            break

    return best_h.astype(float), best_O * (2.0 / n)
