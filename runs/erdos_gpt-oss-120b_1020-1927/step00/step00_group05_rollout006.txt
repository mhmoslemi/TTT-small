def project_box_sum(x, target, eps=1e-12):

    lo = -np.max(x)   # enough to make all zero
    hi = 1 - np.min(x) # enough to make all one

        if s > target:
            hi = mid
        else:
            lo = mid

import numpy as np, math, time

def project_box_sum(x, target, eps=1e-12):
    lo = -np.max(x)
    hi = 1 - np.min(x)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0, 1)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) / 2.0, 0, 1)

def compute_c5(h):
    N = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / N)

def compute_c5_shift(h):
    N = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx] * (2.0 / N)
    shift = max_idx - (N - 1)
    return max_val, shift, corr

def gradient_for_shift(h, shift):
    N = len(h)
    g = np.ones(N)
    s = abs(shift)
    if s > 0:
        g[:N-s] -= h[s:]
        g[s:]   -= h[:N-s]
    else:
        # shift ==0, subtract twice same term
        g -= 2 * h
    return g

def optimize_continuous(h0, time_budget, rng):
    start = time.time()
    N = len(h0)
    target = N / 2.0
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    cur_h = h0.copy()
    cur_val = best_val
    max_iters = 200000
    init_temp = 0.02
    final_temp = 1e-6
    for it in range(max_iters):
        if time.time() - start > time_budget:
            break
        # correlation and shift
        cur_val, shift, _ = compute_c5_shift(cur_h)
        # gradient
        grad = gradient_for_shift(cur_h, shift)
        # step size schedule
        step = 0.15 * (1 - it / max_iters) + 0.01
        # propose
        nh = cur_h - step * grad
        nh = np.clip(nh, 0.0, 1.0)
        nh = project_box_sum(nh, target)
        # evaluate
        nval = compute_c5(nh)
        # temperature
        temp = init_temp * (1 - it / max_iters) + final_temp
        if nval < cur_val or rng.random() < math.exp((cur_val - nval) / (temp + 1e-12)):
            cur_h = nh
            cur_val = nval
            if nval < best_val:
                best_val = nval
                best_h = nh.copy()
    return best_h, best_val

def to_binary(h):
    N = len(h)
    k = N // 2
    idx = np.argpartition(h, -k)[-k:]
    bh = np.zeros(N, dtype=float)
    bh[idx] = 1.0
    return bh

    idx_zero = list(np.where(cur_h < 0.5)[0])
    init_temp = 0.02

import numpy as np, math, time

def _proj_box_sum(x, target):
    lo = -np.max(x)
    hi = 1 - np.min(x)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0, 1)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) * 0.5, 0, 1)

def _c5(h):
    N = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / N)

def _c5_full(h):
    N = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    return corr[idx] * (2.0 / N), idx - (N - 1), corr

def _grad_for_shift(h, shift):
    N = h.shape[0]
    g = np.ones(N)
    s = abs(shift)
    if s:
        g[:N - s] -= h[s:]
        g[s:]   -= h[:N - s]
    else:
        g -= 2 * h
    return g

def _continuous_opt(h0, timelimit, rng):
    start = time.time()
    N = h0.shape[0]
    target = N / 2.0
    cur = h0.copy()
    cur_val = _c5(cur)
    best = cur.copy()
    best_val = cur_val
    max_it = 200000
    T0, Tf = 0.02, 1e-6
    for it in range(max_it):
        if time.time() - start > timelimit:
            break
        cur_val, sh, _ = _c5_full(cur)
        grad = _grad_for_shift(cur, sh)
        step = 0.15 * (1 - it / max_it) + 0.01
        cand = cur - step * grad
        cand = np.clip(cand, 0.0, 1.0)
        cand = _proj_box_sum(cand, target)
        cand_val = _c5(cand)
        T = T0 * (1 - it / max_it) + Tf
        if cand_val < cur_val or rng.random() < math.exp((cur_val - cand_val) / (T + 1e-12)):
            cur = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
    return best, best_val

def _to_binary(h):
    N = h.shape[0]
    k = N // 2
    idx = np.argpartition(h, -k)[-k:]
    bh = np.zeros(N, dtype=float)
    bh[idx] = 1.0
    return bh

def _binary_sa(h0, timelimit, rng):
    start = time.time()
    N = h0.shape[0]
    cur = h0.copy()
    cur_val = _c5(cur)
    best = cur.copy()
    best_val = cur_val
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur < 0.5)[0])
    T0 = 0.02
    Tf = 1e-6
    it = 0
    while time.time() - start < timelimit:
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur[i] = 0.0
        cur[j] = 1.0
        new_val = _c5(cur)
        T = T0 * (0.9999 ** it) + Tf
        accept = new_val < cur_val or rng.random() < math.exp((cur_val - new_val) / (T + 1e-12))
        if accept:
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            cur[i] = 1.0
            cur[j] = 0.0
        it += 1
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
