def project_box_sum(y, s, tol=1e-12, max_iter=100):

    low = np.min(y) - 1.0

    high = np.max(y)

    for _ in range(max_iter):

        mid = (low + high) / 2

        x = np.clip(y - mid, 0, 1)

        cur_sum = np.sum(x)

        if np.abs(cur_sum - s) <= tol:

            return x

        if cur_sum > s:

            # λ too low (shifts up less) => need larger λ

            low = mid

        else:

            high = mid

    # Return final x

    return np.clip(y - (low + high) / 2, 0, 1)

def compute_grad(h, k, dx):
    n = len(h)
    grad = np.full(n, dx)
    for i in range(n):
        # contributions when both indices valid:
        if 0 <= i + k < n:
            grad[i] -= dx * h[i + k]
        if 0 <= i - k < n:
            grad[i] -= dx * h[i - k]
    return grad

def gradient_of_shift(h, shift, dx):
    n = h.shape[0]
    grad = np.full(n, dx)  # base from derivative of sum h term
    if shift >= 0:
        # subtract h[i+shift] where i+shift< n

        grad[shift:] -= dx * h[:n - shift]
    else:
        k = -shift
        grad[k:] -= dx * h[:n - k]
        grad[:n - k] -= dx * h[k:]
    return grad

def smooth_gradient(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # compute weights
    w = np.exp(corr / temp)
    w /= w.sum()
    # shift index mapping: idx 0 corresponds to shift = -(n-1)
    shifts = np.arange(-(n-1), n)
    grad = np.full(n, dx)  # base term dx*1
    for idx, k in enumerate(shifts):
        wk = w[idx]
        if wk == 0: continue
        # compute contributions -dx* wk * (h_{i+k} + h_{i-k})
        if k >= 0:
            grad[:n - k] -= wk * dx * h[k:]
            grad[k:] -= wk * dx * h[:n - k]
        else:
            kk = -k
            grad[kk:] -= wk * dx * h[:n - kk]
            grad[:n - kk] -= wk * dx * h[kk:]
    return grad

def local_optimize(h0, max_iters, budget, start_time, n=None):
    # optionally limit iterations by time left
    n = len(h0) if n is None else n
    total = n/2.0
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    # hyperparameters
    alpha0 = 0.5  # step size
    temp = 0.1   # temperature for smoothing
    for it in range(max_iters):
        # time check
        if time.time() - start_time > budget:
            break
        # compute correlation and max shift
        corr = np.correlate(h, 1 - h, mode='full') * dx
        cmax = corr.max()
        if cmax < best_c:
            best_c = float(cmax)
            best_h = h.copy()
        # gradient using smoothing optional
        # Use smoothed gradient only for early iterations; after certain # maybe use subgradient.
        if it < max_iters // 2:
            # smoothed gradient
            grad = smooth_gradient(h, temp)
        else:
            # subgradient of max
            idx = np.argmax(corr)
            shift = idx - (n - 1)
            grad = gradient_of_shift(h, shift, dx)
        # step size schedule
        alpha = alpha0 / np.sqrt(it+1)
        y = h - alpha * grad
        h = project_to_feasible(y, total)

        temp *= 0.99
        # occasional random perturbation
        if it % 200 == 0 and it > 0:
            noise = np.random.randn(n) * 0.01
            h = project_to_feasible(h + noise, total)
    return best_h, best_c

def smooth_gradient(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    weights = np.exp(corr / temp)
    weights /= weights.sum()
    shifts = np.arange(-(n-1), n)
    grad = np.full(n, dx)  # base term
    for wk, k in zip(weights, shifts):
        if wk == 0.0:
            continue
        if k >= 0:
            # for indices where i + k in range
            grad[:n - k] -= wk * dx * h[k:]
            grad[k:] -= wk * dx * h[:n - k]
        else:
            kk = -k
            grad[kk:] -= wk * dx * h[:n - kk]
            grad[:n - kk] -= wk * dx * h[kk:]
    return grad

def local_optimize(h0, max_iters, start_time, time_limit):
    n = len(h0)
    total = n/2.0
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    # initial temperature
    temp = 0.2
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        # correlation
        corr = np.correlate(h, 1 - h, mode='full') * dx
        cmax = float(np.max(corr))
        if cmax < best_c:
            best_c = cmax
            best_h = h.copy()
        # use smoothed gradient for first half iterations
        if it < max_iters // 2:
            grad = smooth_gradient(h, temp)
            temp *= 0.995  # decrease slowly
        else:
            idx = np.argmax(corr)
            shift = idx - (n - 1)
            grad = gradient_of_shift(h, shift, dx)
        # step size schedule
        alpha = 0.5 / np.sqrt(it + 1)
        y = h - alpha * grad
        h = project_to_feasible(y, total)
        # occasional random jitter
        if it % 200 == 0 and it > 0:
            y = h + np.random.randn(n) * 0.01
            h = project_to_feasible(y, total)
    return best_h, best_c

def gradient_of_shift(h, shift, dx):
    n = h.shape[0]
    grad = np.full(n, dx)
    if shift >= 0:
        k = shift
        if k < n:
            grad[:n - k] -= dx * h[k:]
            grad[k:] -= dx * h[:n - k]
    else:
        k = -shift
        grad[k:] -= dx * h[:n - k]
        grad[:n - k] -= dx * h[k:]
    return grad

def project_to_feasible(y, total):
    # y: numpy array
    # total: required sum
    low = np.min(y) - 1.0
    high = np.max(y)
    for _ in range(60):
        mid = (low + high) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > total:
            low = mid
        else:
            high = mid
    return np.clip(y - (low + high) / 2.0, 0.0, 1.0)

import time
import numpy as np

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(corr.max())

def _gradient_shift(h, shift, dx):
    n = h.shape[0]
    g = np.full(n, dx)
    if shift >= 0:
        k = shift
        if k < n:
            g[: n - k] -= dx * h[k:]
            g[k:] -= dx * h[: n - k]
    else:
        k = -shift
        g[k:] -= dx * h[: n - k]
        g[: n - k] -= dx * h[k:]
    return g

def _smooth_gradient(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    w = np.exp(corr / temp)
    w /= w.sum()
    shifts = np.arange(-(n - 1), n)
    g = np.full(n, dx)                 # d/dh of Σ h_i term
    for wk, sh in zip(w, shifts):
        if wk == 0.0:
            continue
        if sh >= 0:
            k = sh
            if k < n:
                g[: n - k] -= wk * dx * h[k:]
                g[k:] -= wk * dx * h[: n - k]
        else:
            k = -sh
            g[k:] -= wk * dx * h[: n - k]
            g[: n - k] -= wk * dx * h[k:]
    return g

def _project_box_sum(y, total):
    low = np.min(y) - 1.0
    high = np.max(y)
    for _ in range(60):
        mid = (low + high) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > total:
            low = mid
        else:
            high = mid
    return np.clip(y - (low + high) / 2.0, 0.0, 1.0)

def _local_optimize(h0, max_iters, start, limit):
    n = len(h0)
    total = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c = _compute_c5(h)
    temp = 0.2
    for it in range(max_iters):
        if time.time() - start > limit:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        cur_c = float(corr.max())
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if it < max_iters // 2:
            grad = _smooth_gradient(h, temp)
            temp *= 0.995
        else:
            idx = np.argmax(corr)
            shift = idx - (n - 1)
            grad = _gradient_shift(h, shift, dx)
        alpha = 0.5 / np.sqrt(it + 1.0)
        y = h - alpha * grad
        h = _project_box_sum(y, total)
        if it % 200 == 0 and it > 0:
            y = h + np.random.randn(n) * 0.01
            h = _project_box_sum(y, total)
    return best_h, best_c

def _local_search(h, steps, start, limit):
    n = len(h)
    total = n / 2.0
    best_h = h.copy()
    best_c = _compute_c5(h)
    for _ in range(steps):
        if time.time() - start > limit:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = np.random.uniform(-0.05, 0.05)
        hi = best_h[i] + delta
        hj = best_h[j] - delta
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        cand = best_h.copy()
        cand[i] = hi
        cand[j] = hj
        cnew = _compute_c5(cand)
        if cnew < best_c:
            best_c = cnew
            best_h = cand
    # final projection to be safe
    best_h = _project_box_sum(best_h, total)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    limit = budget_s - 0.2   # small safety margin

    best_h = None
    best_c = np.inf
    best_n = None

    def consider(start_vec):
        nonlocal best_h, best_c, best_n, start
        if time.time() - start > limit:
            return
        h_opt, c_opt = _local_optimize(start_vec, max_iters=2500,
                                       start=start, limit=limit)
        if c_opt < best_c:
            best_c, best_h, best_n = c_opt, h_opt.copy(), len(h_opt)
        # polish with local swap search
        h_pol, c_pol = _local_search(best_h, steps=2000,
                                     start=start, limit=limit)
        if c_pol < best_c:
            best_c, best_h, best_n = c_pol, h_pol.copy(), len(h_pol)

    # use parent construction if supplied
    init_vals = globals().get('initial_h_values')
    if init_vals is not None:
        arr = np.array(init_vals, dtype=float)
        arr = _project_box_sum(arr, len(arr) / 2.0)
        consider(arr)

    # systematic exploration of a few sizes
    for n in [60, 80, 100, 120, 150, 180, 200]:
        if time.time() - start > limit:
            break
        total = n / 2.0
        x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n   # points in (0,2)
        # linear ramps
        ramp = x / 2.0
        ramp = _project_box_sum(ramp, total)
        consider(ramp)
        # decreasing ramp
        dec = 1.0 - x / 2.0
        dec = _project_box_sum(dec, total)
        consider(dec)
        # triangular peak at centre
        tri = 1.0 - np.abs(x - 1.0)
        tri = np.clip(tri, 0.0, 1.0)
        tri = _project_box_sum(tri, total)
        consider(tri)
        # a few random starts
        for _ in range(3):
            if time.time() - start > limit:
                break
            rnd = np.random.rand(n)
            rnd = _project_box_sum(rnd, total)
            consider(rnd)

    # fallback
    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5)
        best_c = _compute_c5(best_h)
        best_n = n

    # final safety projection
    best_h = _project_box_sum(np.clip(best_h, 0.0, 1.0), best_n / 2.0)
    return best_h.tolist(), float(best_c), int(best_n)
