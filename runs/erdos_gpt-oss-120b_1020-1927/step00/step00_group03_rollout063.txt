def compute_c5(arr):
    N = arr.shape[0]
    dx = 2.0 / N
    corr = np.correlate(arr, 1 - arr, mode='full') * dx
    return corr.max()  # as float

def solve_exact_mip(N, time_limit):
    import cvxpy as cp, numpy as np, time
    x = cp.Variable(N, boolean=True)
    w = cp.Variable((N,N), boolean=True)
    t = cp.Variable()
    constraints = []
    constraints.append(cp.sum(x) == N // 2)
    # w[i,i] = 0
    constraints.append(cp.diag(w) == 0)
    # w constraints:
    # w <= x[:, None]
    constraints.append(w <= cp.reshape(x, (N,1)))
    constraints.append(w <= 1 - cp.reshape(x, (1,N)))
    constraints.append(w >= cp.reshape(x, (N,1)) - cp.reshape(x, (1,N)))
    # and w >= 0 is redundant for boolean

    # shift constraints
    Nhalf = N / 2.0
    for k in range(-N+1, N):
        if k == 0:
            continue
        if k > 0:
            I = np.arange(N - k)
            J = I + k
        else:
            I = np.arange(-k, N)
            J = I + k
        # Sum w[I,J]
        constraints.append(cp.sum(w[I, J]) <= t * Nhalf)
    constraints.append(t >= 0)

    prob = cp.Problem(cp.Minimize(t), constraints)
    # Solve with CBC, set max seconds
    solve_kwargs = {}
    # Provide time limit via "max_seconds"
    solve_kwargs['max_seconds'] = time_limit
    # We may also pass 'msg': False to suppress log.
    prob.solve(solver=cp.CBC, **solve_kwargs)
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None
    x_val = x.value
    # Ensure binary
    x_binary = np.rint(x_val).astype(int)
    return x_binary

def try_milp(N, time_limit, initial=None):
    import cvxpy as cp, numpy as np, time
    # variables
    x = cp.Variable(N, boolean=True)
    w = cp.Variable((N, N), boolean=True)
    t = cp.Variable()
    constraints = []
    constraints.append(cp.sum(x) == N // 2)
    constraints.append(t >= 0)
    # diagonal zero
    constraints.append(cp.diag(w) == 0)
    # w constraints for all i != j
    for i in range(N):
        for j in range(N):
            if i == j:
                continue
            constraints.append(w[i, j] <= x[i])
            constraints.append(w[i, j] <= 1 - x[j])
            constraints.append(w[i, j] >= x[i] - x[j])
    # shift constraints
    Nhalf = N / 2.0
    for k in range(-N+1, N):
        if k == 0:
            continue
        if k > 0:
            I = np.arange(N - k)
            J = I + k
        else:
            I = np.arange(-k, N)
            J = I + k
        constraints.append(cp.sum(w[I, J]) <= t * Nhalf)

    prob = cp.Problem(cp.Minimize(t), constraints)
    # solve with CBC, time limit
    solve_kwargs = {}
    # Put time limit if supported
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_limit, verbose=False)
    except TypeError:
        # older cvxpy may use "time_limit"
        prob.solve(solver=cp.CBC, time_limit=time_limit, verbose=False)

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        xv = x.value
        # round
        hv = np.rint(xv).astype(float)
        # Ensure sum constraint satisfied; maybe adjust
        diff = int(hv.sum() - N // 2)
        if diff != 0:
            # adjust naive
            if diff > 0:
                idx = np.where(hv == 1)[0][:diff]
                hv[idx] = 0
            else:
                idx = np.where(hv == 0)[0][: -diff]
                hv[idx] = 1
        return hv
    else:
        return None

import time
import numpy as np
import cvxpy as cp

# ------------------------------------------------------------
def _c5_score(h):
    """Return C5 for a vector h (np.ndarray, values in [0,1])."""
    n = h.shape[0]
    dx = 2.0 / n
    # full linear correlation (zero padding outside the interval)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max()), int(corr.argmax() - (n - 1)), corr


# ------------------------------------------------------------
def _milp_solve(N, time_limit, init=None):
    """
    Solve the exact integer programme for length N.
    Variables:
        x_i ∈ {0,1}, sum x_i = N/2
        w_{i,j} = x_i * (1 - x_j)  (continuous, enforced by linear inequalities)
        t  = max_{k}  Σ_{i} w_{i,i+k}
    Minimise t.
    Returns a binary numpy array of length N or None if not solved.
    """
    # binary variables for the set
    x = cp.Variable(N, boolean=True)
    # continuous auxiliary variables for the products
    w = cp.Variable((N, N))
    t = cp.Variable()

    constraints = []

    # weight condition
    constraints.append(cp.sum(x) == N // 2)
    constraints.append(t >= 0)

    # diagonal of w must be zero (product of a variable with its own complement)
    constraints.append(cp.diag(w) == 0)

    # bounds that enforce w[i,j] == x[i] * (1 - x[j])
    for i in range(N):
        # w[i, :] ≤ x[i]  (row bound)
        constraints.append(w[i, :] <= x[i])
        # w[:, i] ≤ 1 - x[i]  (column bound)
        constraints.append(w[:, i] <= 1 - x[i])
    # lower bounds w[i,j] ≥ x[i] - x[j]
    for i in range(N):
        for j in range(N):
            if i == j:
                continue
            constraints.append(w[i, j] >= x[i] - x[j])

    # constraints for every shift k (non‑zero)
    half = N / 2.0
    for k in range(-N + 1, N):
        if k == 0:
            continue
        if k > 0:
            I = np.arange(N - k)
            J = I + k
        else:
            I = np.arange(-k, N)
            J = I + k
        constraints.append(cp.sum(w[I, J]) <= t * half)

    prob = cp.Problem(cp.Minimize(t), constraints)

    # CBC accepts maxSeconds as a time limit
    try:
        prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
    except TypeError:
        # fallback for older cvxpy versions
        prob.solve(solver=cp.CBC, time_limit=time_limit, verbose=False)

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None

    xv = np.rint(x.value).astype(float)  # rounding removes tiny numerical noise
    # Safety: enforce exact weight if rounding slipped
    target = N // 2
    cur = int(xv.sum())
    if cur != target:
        diff = cur - target
        if diff > 0:
            idx = np.where(xv == 1)[0][:diff]
            xv[idx] = 0
        else:
            idx = np.where(xv == 0)[0][: -diff]
            xv[idx] = 1
    return xv

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # 1) initialise a feasible vector h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        N = len(h0)
        h0 = np.clip(h0, 0.0, 1.0)
        # force binary – easier for the later search
        h0 = (h0 >= 0.5).astype(float)
        target = N // 2
        cur = int(h0.sum())
        if cur != target:
            diff = cur - target
            if diff > 0:          # too many ones
                idx = np.where(h0 == 1)[0][:diff]
                h0[idx] = 0
            else:                 # too few ones
                idx = np.where(h0 == 0)[0][: -diff]
                h0[idx] = 1
    else:
        N = 200                           # a reasonable size when no parent is given
        h0 = np.zeros(N, dtype=float)
        h0[: N // 2] = 1
        np.random.shuffle(h0)

    # enforce even length (required for exact half‑weight)
    if N % 2 == 1:
        h0 = h0[:-1]
        N -= 1

    # ------------------------------------------------------------------
    # 2) try an exact MILP solution for modest N (≤80)
    remaining = budget_s - (time.time() - start)
    best_h = h0.copy()
    best_score, _, _ = _c5_score(best_h)

    if N <= 80 and remaining > 30:
        milp_h = _milp_solve(N, min(remaining, 120), init=best_h)
        if milp_h is not None:
            milp_score, _, _ = _c5_score(milp_h)
            if milp_score < best_score:
                best_h, best_score = milp_h, milp_score

    # ------------------------------------------------------------------
    # 3) stochastic hill climbing with simulated‑annealing acceptance
    cur_h = best_h.copy()
    cur_score = best_score

    T0 = 0.008                # initial temperature
    decay = 4e-6              # exponential decay factor
    it = 0
    check_interval = 200000  # restart interval

    while time.time() - start < budget_s * 0.94:
        it += 1
        T = T0 * np.exp(-decay * it)

        # identify the shift that currently dominates the max
        _, shift, _ = _c5_score(cur_h)

        # indices that contribute to that shift (1 followed by 0)
        if shift > 0:
            mask = (cur_h[: N - shift] == 1) & (cur_h[shift:] == 0)
            bad = np.where(mask)[0]
        elif shift < 0:
            s = -shift
            mask = (cur_h[s:] == 1) & (cur_h[: N - s] == 0)
            bad = np.where(mask)[0] + s
        else:
            bad = np.array([], dtype=int)

        # pick a 1 from a bad position if possible, otherwise any 1
        if bad.size:
            i = int(np.random.choice(bad))
        else:
            i = int(np.random.choice(np.where(cur_h == 1)[0]))
        # pick a 0 somewhere else
        zero_candidates = np.where(cur_h == 0)[0]
        if zero_candidates.size == 0:
            break
        j = int(np.random.choice(zero_candidates))

        # perform the swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score, _, _ = _c5_score(cur_h)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
                if best_score <= 0.3808:      # target reached – stop early
                    break
        else:
            # revert the move
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        # periodic restart from the best found so far
        if it % check_interval == 0:
            cur_h = best_h.copy()
            cur_score = best_score

    # ------------------------------------------------------------------
    # 4) final evaluation and return
    try:
        final_c5 = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_c5, _, _ = _c5_score(best_h)

    return best_h.astype(float), final_c5, N
