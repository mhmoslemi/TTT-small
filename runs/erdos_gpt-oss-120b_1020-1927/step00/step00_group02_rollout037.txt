project_to_bounds_sum(v, target_sum):

def project_to_bounds_sum(v, target_sum, lower=0., upper=1., max_iter=100, tol=1e-12):
    # Use binary search for lambda
    lo = -np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, lower, upper)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2.0, lower, upper)

Now gradient descent:

def gradient_descent(h, target_sum, time_limit, start_time):
    N = h.shape[0]
    dx = 2.0 / N
    best_h = h.copy()
    best_val = compute_bound(h)
    cur_h = h.copy()
    cur_val = best_val
    init_alpha = 0.5  # may adjust
    while True:
        now = time.time()
        if now - start_time > time_limit:
            break
        # compute correlation to find worst lag
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        idx = np.argmax(corr)
        lag = idx - (N - 1)
        L = abs(lag)
        # compute gradient
        g = np.ones(N)
        if L == 0:
            g -= 2 * cur_h
        else:
            # Overlap indices
            if lag >= 0:
                g[:N-L] -= cur_h[L:]
                g[L:] -= cur_h[:N-L]
            else:
                g[:N-L] -= cur_h[L:]
                g[L:] -= cur_h[:N-L]
        # step size schedule
        elapsed = now - start_time
        # linearly decay alpha
        alpha = init_alpha * (1.0 - elapsed / time_limit)
        if alpha < 1e-6:
            alpha = 1e-6
        # update
        new_h = cur_h - alpha * g
        # project to constraints
        new_h = project_to_bounds_sum(new_h, target_sum)
        new_val = compute_bound(new_h)
        # accept if improves (allow tiny tolerance)
        if new_val < cur_val - 1e-12:
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_h = new_h.copy()
                best_val = new_val
        else:
            # maybe try small backtrack? skip for simplicity
            pass
    return best_h, best_val

Now random local search:

def random_local_search(h, target_sum, time_limit, start_time, best_h, best_val):
    # use repeated random swaps to try to improve.
    N = h.shape[0]
    cur_h = h.copy()
    cur_val = compute_bound(cur_h)
    while True:
        now = time.time()
        if now - start_time > time_limit:
            break
        # choose random i != j
        i, j = np.random.randint(0,N, size=2)
        if i == j:
            continue
        hi, hj = cur_h[i], cur_h[j]
        dmin = max(-hi, hj - 1.0)
        dmax = min(1.0 - hi, hj)
        if dmax <= dmin:
            continue
        delta = np.random.uniform(dmin, dmax)
        cur_h[i] = hi + delta
        cur_h[j] = hj - delta
        # ensure within [0,1] automatically satisfied
        new_val = compute_bound(cur_h)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = hi
            cur_h[j] = hj
    return best_h, best_val

import time
import numpy as np

def _project_to_bounds_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=200):
    """Project v onto the box [lower,upper] while enforcing sum(v)=target."""
    lo = -np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, lower, upper)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            hi = mid
        else:
            lo = mid
    # fallback
    return np.clip(v + (lo + hi) * 0.5, lower, upper)


def _bound(h):
    """Return C5 bound for the step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def _subgradient(h, lag):
    """Sub‑gradient of the active correlation (worst lag) w.r.t. h."""
    n = h.shape[0]
    g = np.ones(n)
    L = abs(lag)
    if L == 0:
        g -= 2.0 * h
    else:
        # overlapping pairs (i, i+L)
        g[: n - L] -= h[L:]
        g[L:] -= h[: n - L]
    return g


def _descent_step(h, lag, cur_val):
    """Perform a single descent step moving mass from positive to negative gradient entries."""
    n = h.shape[0]
    g = _subgradient(h, lag)
    pos = np.where(g > 0)[0]
    neg = np.where(g < 0)[0]
    if pos.size == 0 or neg.size == 0:
        return h, cur_val  # nothing to do
    i = np.random.choice(pos)
    j = np.random.choice(neg)
    # amount we can transfer while staying inside [0,1]
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 1e-12:
        return h, cur_val
    delta = np.random.uniform(0.0, max_delta)
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    new_val = _bound(h_new)
    if new_val < cur_val - 1e-12:
        return h_new, new_val
    # reject the move
    return h, cur_val


def _random_swap(h, cur_val):
    """Randomly exchange a small amount between two coordinates."""
    n = h.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return h, cur_val
    hi, hj = h[i], h[j]
    dmin = max(-hi, hj - 1.0)
    dmax = min(1.0 - hi, hj)
    if dmax <= dmin:
        return h, cur_val
    delta = np.random.uniform(dmin, dmax)
    h[i] = hi + delta
    h[j] = hj - delta
    new_val = _bound(h)
    if new_val < cur_val - 1e-12:
        return h, new_val
    # revert
    h[i] = hi
    h[j] = hj
    return h, cur_val


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # ------------------------------------------------------------------
    # initialise h
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        n = 120
        h = np.full(n, 0.5, dtype=float)

    n = h.shape[0]
    target_sum = n / 2.0
    h = _project_to_bounds_sum(h, target_sum)
    best_h = h.copy()
    best_val = _bound(best_h)

    # time budgeting
    total_end = start + budget_s * 0.97
    # ------------------------------------------------------------------
    # Phase 1 : focused gradient descent on the active lag
    while time.time() < total_end * 0.6:
        # current worst lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        lag = idx - (n - 1)

        # one descent step
        h, cur_val = _descent_step(h, lag, best_val if best_val < _bound(h) else _bound(h))
        # keep global best
        cur_val = _bound(h)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()

    # ------------------------------------------------------------------
    # Phase 2 : random local swaps / perturbations
    while time.time() < total_end:
        if np.random.rand() < 0.7:
            h, cur_val = _random_swap(h, _bound(h))
        else:
            # occasional stronger move: pick worst lag again and do a directed transfer
            corr = np.correlate(h, 1.0 - h, mode="full")
            lag = int(np.argmax(corr)) - (n - 1)
            h, cur_val = _descent_step(h, lag, _bound(h))
        cur_val = _bound(h)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()

    # ------------------------------------------------------------------
    # final projection to guarantee feasibility (tiny numeric drift)
    best_h = _project_to_bounds_sum(best_h, target_sum)

    # ------------------------------------------------------------------
    return best_h, float(best_val), n
