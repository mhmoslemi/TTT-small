def compute_c5(h):
    n = len(h)
    m = 2*n
    # zero pad to length m
    H = np.fft.fft(h, m)
    Hc = np.fft.fft((1 - h)[::-1], m)
    conv = np.fft.ifft(H * Hc).real
    # conv length m, but true convolution length is 2n-1. Use conv[:2*n-1]
    conv = conv[:2*n-1]
    c5 = conv.max() * (2.0 / n)
    return c5

def compute_c5_fft(h):
    n = len(h)
    m = 2*n  # zero padding
    H = np.fft.fft(h, m)
    G = np.fft.fft((1 - h)[::-1], m)
    conv = np.fft.ifft(H * G).real
    # conv length m; use first 2n-1 entries
    conv = conv[:2*n-1]
    c5 = conv.max() * (2.0 / n)
    idx = conv.argmax()
    shift = idx - (n - 1)
    return c5, shift

def compute_c5_fft(h):
    n = h.shape[0]
    m = 2 * n
    H = np.fft.fft(h, m)
    G = np.fft.fft((1 - h)[::-1], m)
    conv = np.fft.ifft(H * G).real[:2 * n - 1]
    max_val = conv.max()
    shift = conv.argmax() - (n - 1)
    c5 = max_val * (2.0 / n)
    return c5, shift

def project_onto_box_sum(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - lo, 0.0, 1.0)
    # adjust residual
    resid = target - x.sum()
    if abs(resid) > 1e-10:
        mask = (x > 1e-9) & (x < 1 - 1e-9)
        count = mask.sum()
        if count > 0:
            x[mask] += resid / count
            # clamp again
            x = np.clip(x, 0.0, 1.0)
    return x

def binary_step_vector(n, a, b, c):
    # widths: w1=a, w2=b, w3=c, w4=1-b, w5=1-a-c
    # ensure non-negative (should be)
    # breakpoints
    x = np.arange(n) * (2.0 / n) + (1.0 / n)  # maybe use midpoints for better representation
    # but we can use x just as positions start at 0; we use <= for breakpoints
    # compute breakpoints as cumulated.
    # We'll compute as float positions.
    b1 = a
    b2 = a + b
    b3 = a + b + c
    b4 = a + 1.0 + c  # a + 1 + c
    # Determine mask:
    h = np.empty(n)
    # Using vectorized boolean operations
    # define condition order from 0 to 2.
    # We'll compute arrays of bool:
    # conditions: x < b1 (seg1), x >= b1 & x < b2 (seg2), x >= b2 & x < b3 (seg3), x >= b3 & x < b4 (seg4), else seg5
    seg1 = x < b1
    seg2 = (x >= b1) & (x < b2)
    seg3 = (x >= b2) & (x < b3)
    seg4 = (x >= b3) & (x < b4)
    # seg5 else
    h[seg1] = 1.0
    h[seg2] = 0.0
    h[seg3] = 1.0
    h[seg4] = 0.0
    # remaining  5
    mask5 = ~(seg1|seg2|seg3|seg4)
    h[mask5] = 1.0
    return h

def _compute_c5_fft(h):
    """Return (C5 value, shift index where max occurs) for vector h."""
    n = h.shape[0]
    m = 2 * n
    H = np.fft.fft(h, m)
    G = np.fft.fft((1.0 - h)[::-1], m)
    conv = np.fft.ifft(H * G).real[: 2 * n - 1]
    idx = conv.argmax()
    c5 = conv[idx] * (2.0 / n)
    shift = idx - (n - 1)          # integer shift producing the max
    return float(c5), int(shift)

    if L >= 0:
        hi = np.empty_like(h)
        hi[: n - L] = h[L:]
        hi[n - L :] = 0.0
        hm = np.empty_like(h)
        hm[L:] = h[: n - L]
        hm[:L] = 0.0
    else:
        k = -L
        hi = np.empty_like(h)
        hi[k:] = h[: n - k]
        hi[:k] = 0.0
        hm = np.empty_like(h)
        hm[: n - k] = h[k:]
        hm[n - k :] = 0.0
    return 1.0 - hi - hm   # sum_i grad_i = 0


def _project_box_sum(v, target):
    """Euclidean projection of v onto {0<=x<=1, sum(x)=target}."""
    lo = v.min() - 1.0          # enough to make all entries >=1 after shift
    hi = v.max()                # enough to make all entries <=0 after shift
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - lo, 0.0, 1.0)
    resid = target - x.sum()
    if abs(resid) > 1e-12:
        free = (x > 1e-9) & (x < 1.0 - 1e-9)
        cnt = free.sum()
        if cnt:
            x[free] += resid / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _binary_step_vector(n, a, b, c):
    """3‑interval‑of‑ones, 2 intervals of zeros (5‑step) binary pattern."""
    # widths: w1=a, w2=b, w3=c, w4=1-b, w5=1-a-c  (all >=0, sum=2)
    x = np.arange(n) * (2.0 / n) + (1.0 / n)   # mid‑point sample positions
    b1 = a
    b2 = a + b
    b3 = a + b + c
    b4 = a + 1.0 + c          # = a + (1-b) + b + c = a + 1 + c
    h = np.empty(n, dtype=float)
    seg1 = x < b1
    seg2 = (x >= b1) & (x < b2)
    seg3 = (x >= b2) & (x < b3)
    seg4 = (x >= b3) & (x < b4)
    h[seg1] = 1.0
    h[seg2] = 0.0
    h[seg3] = 1.0
    h[seg4] = 0.0
    h[~(seg1 | seg2 | seg3 | seg4)] = 1.0   # last segment
    return h


def _random_step_search(n, time_budget, rng):
    """Search over the 5‑step binary family."""
    target = n / 2.0
    best_h = None
    best_val = np.inf
    t0 = time.time()
    while time.time() - t0 < time_budget:
        a = rng.random()
        c = rng.random()
        if a + c > 1.0:
            continue
        b = rng.random()
        h = _binary_step_vector(n, a, b, c)
        h = _project_box_sum(h, target)
        val, _ = _compute_c5_fft(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val


def _gradient_descent(h0, time_budget):
    """Simple projected gradient descent on the worst shift."""
    n = h0.shape[0]
    target = n / 2.0
    cur = _project_box_sum(h0, target)
    cur_val, cur_shift = _compute_c5_fft(cur)
    best = cur.copy()
    best_val = cur_val
    step = 0.3
    step_max = 0.5
    step_min = 1e-4
    it = 0
    t0 = time.time()
    while time.time() - t0 < time_budget:
        # gradient step for the currently worst shift
        grad = _gradient_shift(cur, cur_shift)
        trial = _project_box_sum(cur - step * grad, target)
        trial_val, trial_shift = _compute_c5_fft(trial)

        if trial_val < cur_val - 1e-12:
            cur, cur_val, cur_shift = trial, trial_val, trial_shift
            if trial_val < best_val:
                best, best_val = trial.copy(), trial_val
            step = min(step_max, step * 1.02)
        else:
            step = max(step_min, step * 0.5)

        # occasional random mass move for escaping plateaus
        if it % 120 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            max_move = min(0.11, cur[i], 1.0 - cur[j])
            if max_move > 0:
                amt = np.random.rand() * max_move
                h2 = cur.copy()
                h2[i] -= amt
                h2[j] += amt
                val2, _ = _compute_c5_fft(h2)
                if val2 < cur_val:
                    cur, cur_val = h2, val2
                    if val2 < best_val:
                        best, best_val = h2.copy(), val2
        it += 1
    return best, best_val


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    n = 256                         # number of samples (kept modest)
    target = n / 2.0
    # ---- initial construction -------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, init)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = _project_box_sum(h0, target)
    else:
        # start from a random half‑ones binary vector
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    # ---- quick random search in the 5‑step family ----------------------------
    elapsed = time.time()
    remaining = budget_s - (time.time() - elapsed)
    # allocate at most 20 % of the remaining budget (or 30 s) to this phase
    step_time = min(30.0, remaining * 0.2)
    step_h, step_val = _random_step_search(n, step_time, rng)

    # keep the better of the two starters
    cur_h, cur_val = h0, _compute_c5_fft(h0)[0]
    if step_h is not None and step_val < cur_val:
        cur_h, cur_val = step_h, step_val

    # ---- projected gradient descent -----------------------------------------
    elapsed = time.time()
    remaining = budget_s - (time.time() - elapsed) - 5.0  # 5 s safety margin
    if remaining > 0:
        refined_h, refined_val = _gradient_descent(cur_h, remaining)
        if refined_val < cur_val:
            cur_h, cur_val = refined_h, refined_val

    # final sanity projection and exact evaluation
    cur_h = _project_box_sum(cur_h, target)
    final_val, _ = _compute_c5_fft(cur_h)

    return cur_h, float(final_val), n
