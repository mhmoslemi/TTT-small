import time
import numpy as np

def _project_capped_simplex(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=200):
    """Project v onto {x: lower <= x_i <= upper, sum x = target}."""
    n = v.shape[0]
    target = max(lower * n, min(target, upper * n))
    # lambda low gives all values at lower bound, lambda high at upper bound
    lam_low = -np.max(v) - 1.0      # ensures v+lam_low <= 0
    lam_high = upper - np.min(v) + 1.0  # ensures v+lam_high >= upper
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        x = np.clip(v + lam, lower, upper)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lam_high = lam
        else:
            lam_low = lam
    # final projection
    lam = (lam_low + lam_high) * 0.5
    return np.clip(v + lam, lower, upper)


def _corr_max(h, dx):
    """Return max_{k} ∫ h(x)(1-h(x+k))dx computed by discrete correlation."""
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)


def _smooth_objective_and_grad(h, beta, dx):
    """Smooth approximation of the max overlap and its gradient."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full') * dx          # length 2n-1
    # log‑sum‑exp for stability
    t = beta * corr
    t_max = t.max()
    exp_t = np.exp(t - t_max)
    sum_exp = exp_t.sum()
    weights = exp_t / sum_exp                                 # sum to 1
    obj = (np.log(sum_exp) + t_max) / beta                    # smooth max

    grad = np.zeros_like(h)
    # each shift contributes weight * gradient of that shift
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        shift = idx - (n - 1)    # 0‑shift at index n-1
        if shift >= 0:
            m = shift
            if m < n:
                # (1 - h[i+m]) part
                grad[: n - m] += w * dx * (1.0 - h[m:])
                # -h[i-m] part
                grad[m:] += w * dx * (-h[: n - m])
        else:
            p = -shift
            if p < n:
                grad[p:] += w * dx * (1.0 - h[: n - p])
                grad[: n - p] += w * dx * (-h[p:])
    return obj, grad
