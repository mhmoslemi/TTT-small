def project_vec(vec, target):
    lo = -1.0
    hi = 1.0
    for _ in range(40):
        mid = (lo+hi)/2
        proj = np.clip(vec - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(vec - (lo+hi)/2, 0.0, 1.0)

def compute_c5_full(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx, corr

import time
import math
import numpy as np

def _project_to_sum(vec, target):
    """Project vec onto the box [0,1]^n with given sum = target."""
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        proj = np.clip(vec - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(vec - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h):
    """Return (C5, full correlation array) for a given h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx, corr


def _gradient_for_shift(h, shift):
    """Gradient of the maximal overlap term for a given integer shift."""
    n = h.shape[0]
    if shift == 0:
        # D(0) = sum h_i (1 - h_i) -> grad = 1 - 2*h
        return 1.0 - 2.0 * h
    # forward part: h_i * (1 - h_{i+shift})
    fwd = np.concatenate((h[shift:], np.zeros(shift)))
    # backward part: h_{i-shift} appears in the product of the opposite direction
    bwd = np.concatenate((np.zeros(shift), h[: n - shift]))
    return np.ones(n) - fwd - bwd


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # initialise h and problem size
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.shape[0]
    else:
        n = 200                     # a modest size, fast to evaluate
        h = rng.random(n)
        target_sum = n / 2.0
        h = h / h.sum() * target_sum
        h = _project_to_sum(h, target_sum)

    target_sum = n / 2.0
    # ensure feasibility
    h = _project_to_sum(h, target_sum)

    best_h = h.copy()
    best_c5, _ = _c5_value(best_h)

    # ------------------------------------------------------------------
    # optimisation loop (sub‑gradient descent + occasional noise)
    # ------------------------------------------------------------------
    it = 0
    eta0 = 0.7                     # initial step size
    max_iters = 10_000_000

    while time.time() - start_time < budget_s * 0.95 and it < max_iters:
        it += 1

        # current value and the shift achieving it
        cur_c5, corr = _c5_value(h)
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
            # early exit if we already beat the target bound
            if best_c5 < 0.38:
                break

        # shift corresponding to the maximum entry
        idx_max = int(np.argmax(corr))
        lag = idx_max - (n - 1)                # lag ∈ [-(n-1), n-1]
        s = abs(lag)

        # sub‑gradient for this lag
        grad = _gradient_for_shift(h, s)

        # simple diminishing step size
        eta = eta0 / math.sqrt(it + 1.0)

        # gradient step + box projection
        h_new = h - eta * grad
        h_new = np.clip(h_new, 0.0, 1.0)

        # enforce the integral constraint exactly
        h_new = _project_to_sum(h_new, target_sum)

        h = h_new

        # occasional random jitter to avoid stagnation
        if it % 20000 == 0:
            noise = (rng.random(n) - 0.5) * 0.02
            h = np.clip(h + noise, 0.0, 1.0)
            h = _project_to_sum(h, target_sum)

    # one final evaluation of the best candidate
    best_c5, _ = _c5_value(best_h)

    # return as required: list of values, bound, and number of points
    return (best_h.tolist(), float(best_c5), n)
