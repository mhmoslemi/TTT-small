def shift_with_zero(arr, L):
    n = arr.shape[0]
    if L >= 0:
        out = np.empty_like(arr)
        out[:n-L] = arr[L:]
        out[n-L:] = 0.0
    else:
        L2 = -L
        out = np.empty_like(arr)
        out[L2:] = arr[:n-L2]
        out[:L2] = 0.0
    return out
```

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time, random, math

    # ------------------------------------------------------------------
    # Helpers
    # ------------------------------------------------------------------
    def compute_c5(h):
        """Return max_k ∫ h(x)(1-h(x+k))dx for the discretised h."""
        n = h.shape[0]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * (2.0 / n)

    def shift_zero(arr, lag):
        """Shift arr by lag positions, filling out‑of‑range with zeros."""
        n = arr.shape[0]
        if lag >= 0:
            out = np.empty_like(arr)
            out[: n - lag] = arr[lag:]
            out[n - lag :] = 0.0
        else:
            lag = -lag
            out = np.empty_like(arr)
            out[lag:] = arr[: n - lag]
            out[:lag] = 0.0
        return out

    def random_feasible(n):
        """Uniform 0‑½ vector that satisfies the integral constraint."""
        # start from 0.5 everywhere (guaranteed feasible)
        return np.full(n, 0.5, dtype=float)

    # ------------------------------------------------------------------
    # Initialise
    # ------------------------------------------------------------------
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    if "initial_h_values" in globals():
        h_start = np.array(initial_h_values, dtype=float)  # use provided step function
    else:
        # fallback: 47 points uniform 0.5
        h_start = random_feasible(47)

    n = h_start.shape[0]
    dx = 2.0 / n

    cur_h = h_start.copy()
    cur_val = compute_c5(cur_h)

    # If the supplied construction already beats the target, return it immediately.
    target = 0.3808
    if cur_val <= target:
        return cur_h, cur_val, n

    best_h = cur_h.copy()
    best_val = cur_val

    # Parameters for the local search
    max_step = 0.02          # maximum transferred mass in one move
    no_improve = 0
    restart_interval = 20000
    tolerance = 1e-12

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.97:
        # --- compute the lag that currently realizes the maximum ---
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx_max = int(np.argmax(corr))
        lag = idx_max - (n - 1)                     # signed lag
        # --- compute the derivative‑like weights for this lag ---
        shifted = shift_zero(cur_h, lag)
        d = 1.0 - shifted                         # larger -> bigger contribution

        # --- choose source (high d, positive entry) and target (low d, room) ---
        src_mask = cur_h > tolerance
        tgt_mask = cur_h < 1.0 - tolerance

        if not src_mask.any() or not tgt_mask.any():
            break

        # source index: maximal d among admissible sources
        d_src = d.copy()
        d_src[~src_mask] = -1.0
        src_i = int(np.argmax(d_src))

        # target index: minimal d among admissible targets
        d_tgt = d.copy()
        d_tgt[~tgt_mask] = np.inf
        tgt_i = int(np.argmin(d_tgt))

        # avoid degenerate choice
        if src_i == tgt_i:
            # fallback to secondary candidates
            src_candidates = np.argsort(-d)  # descending d
            tgt_candidates = np.argsort(d)   # ascending d
            for cand in src_candidates:
                if cur_h[cand] > tolerance and cand != tgt_i:
                    src_i = cand
                    break
            for cand in tgt_candidates:
                if cur_h[cand] < 1.0 - tolerance and cand != src_i:
                    tgt_i = cand
                    break

        # --- propose a deterministic move (transfer from src to tgt) ---
        delta = min(max_step, cur_h[src_i], 1.0 - cur_h[tgt_i])
        if delta > tolerance:
            h_new = cur_h.copy()
            h_new[src_i] -= delta
            h_new[tgt_i] += delta
            val_new = compute_c5(h_new)

            if val_new + tolerance < cur_val:          # improvement on current state
                cur_h = h_new
                cur_val = val_new
                no_improve = 0
                if val_new + tolerance < best_val:
                    best_h = h_new.copy()
                    best_val = val_new
                if best_val <= target:

        i, j = np.random.randint(0, n, size=2)

    return best_h, final_val, n
