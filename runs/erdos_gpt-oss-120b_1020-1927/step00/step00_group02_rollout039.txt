def project_to_feasible(v, target):
    lo = -1e6
    hi = 1e6
    for _ in range(60):
        lam = (lo + hi) / 2.0
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def max_overlap(h, dx):
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def gradient_for_shift(h, s):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if s >= 0:
        ns = n - s
        grad[:ns] += 1.0
        grad[:ns] -= h[s:]
        if s < ns:
            grad[s:n - s] -= h[: n - 2 * s]
    else:
        t = -s
        nt = n - t
        grad[t:] += 1.0
        grad[t:] -= h[: nt]
        if t < nt:
            grad[: nt] -= h[t:]
    return grad

def descent(h_start, dx, target_sum, max_iters, lr0):
    h = h_start.copy()
    best_h = h.copy()
    best_val = max_overlap(h, dx)
    lr = lr0
    n = h.shape[0]
    for it in range(max_iters):
        # compute correlation raw (no dx) for shift detection
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr_raw))
        cur_val = corr_raw[idx] * dx
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        s = idx - (n - 1)
        grad = gradient_for_shift(h, s)
        h = project_to_feasible(h - lr * grad, target_sum)
        lr *= 0.9995
    return best_h, best_val

import time
import random
import numpy as np

def _project_to_feasible(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {0<=h_i<=1, sum(h)=target_sum}."""
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = (lo + hi) * 0.5
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray, dx: float) -> float:
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k)) dx using discrete correlation."""
    return (np.correlate(h, 1.0 - h, mode="full") * dx).max()


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Gradient of the correlation value at the given integer shift.
    shift = lag index - (n-1) where lag runs 0..2n-2.
    """
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        ns = n - shift
        # first part: +1 - h[i+shift]
        g[:ns] += 1.0
        g[:ns] -= h[shift:]
        # second part: -h[i-shift] for i in [shift, n-shift)
        if shift < ns:  # i.e. shift < n/2
            g[shift:n - shift] -= h[: n - 2 * shift]
    else:
        t = -shift
        nt = n - t
        # first part: +1 - h[i-t]
        g[t:] += 1.0
        g[t:] -= h[:nt]
        # second part: -h[i+t] for i in [0, nt)
        if t < nt:  # t < n/2
            g[:nt] -= h[t:]
    return g


def _descent(
    h0: np.ndarray,
    dx: float,
    target_sum: float,
    max_iters: int,
    lr0: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent that reduces max correlation."""
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = _c5_value(h, dx)

    lr = lr0
    for _ in range(max_iters):
        # correlation (raw, without dx) – only needed for index of max
        corr_raw = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr_raw))
        cur_val = corr_raw[idx] * dx
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        shift = idx - (n - 1)
        grad = _gradient_for_shift(h, shift)
        h = _project_to_feasible(h - lr * grad, target_sum)
        lr *= 0.9995  # gentle decay
        if time.time() >= deadline:
            break
    return best_h, best_val

    best_h = None
