def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, target_sum, tol=1e-9, max_iter=100):
    # Clip and adjust sum while respecting bounds.
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) <= tol:
            break
        # find indices not at bounds
        free = np.where((h > 0 + 1e-12) & (h < 1 - 1e-12))[0]
        if len(free) == 0:
            # distribute equally among all?
            # adjust all with small delta and clip again
            delta = diff / len(h)
            h += delta
            h = np.clip(h, 0.0, 1.0)
        else:
            delta = diff / len(free)
            h[free] += delta
            h = np.clip(h, 0.0, 1.0)
    # final clip

def project_simple(h, target):
    # Clip and adjust sum
    h = np.clip(h, 0.0, 1.0)
    # Compute difference
    diff = target - np.sum(h)
    # Identify adjustable indices
    # We'll perform simple water-filling:
    # Use while diff not zero and there are free indices
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size == 0:
            # No free indices, break to avoid infinite loop
            break
        delta = diff / free.size
        # apply delta
        h[free] += delta
        # Clip again
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - np.sum(h)
    # Final small fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Add diff to a non-saturated element if possible
        idx = np.where((h > 0) & (h < 1))[0]
        if idx.size > 0:
            i = idx[0]
            h[i] = min(max(h[i] + diff, 0.0), 1.0)
    return h

import time, math
import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """max overlap integral for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate with mode='full' already implements the required sum
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _project(h: np.ndarray, target: float) -> np.ndarray:
    """Project h onto [0,1]^n with sum == target (Euclidean projection)."""
    h = np.clip(h, 0.0, 1.0)
    # water‑filling style iteration – fast for the tiny n we use
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size == 0:
            # nothing free – give up early
            break
        delta = diff / free.size
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
    # final tiny correction if needed
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.where((h > 0) & (h < 1))[0]
        if idx.size:
            i = idx[0]
            h[i] = min(max(h[i] + diff, 0.0), 1.0)
    return h


def _sinusoid_candidate(n: int, m: int, phi: float = 0.0) -> np.ndarray:
    """Build h(x)=½(1‑cos(mπ(x‑φ))) on the uniform grid, then project."""
    xs = np.arange(n) * (2.0 / n)
    h = 0.5 * (1.0 - np.cos(m * np.pi * (xs - phi)))
    return _project(h, n / 2.0)


def _explore_resolution(best_h, best_c, n, rnd):
    """Search sinusoidal families on a grid of size n."""
    target = n / 2.0
    max_m = max(1, n // 4)          # at least four points per half‑period
    # deterministic sweep over m
    for m in range(1, max_m + 1):
        h = _sinusoid_candidate(n, m, 0.0)
        c = _c5_of(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    # few random phase attempts
    for _ in range(30):
        m = rnd.randint(1, max_m + 1)
        phi = rnd.rand() * (2.0 / m)   # shift inside one period
        h = _sinusoid_candidate(n, m, phi)
        c = _c5_of(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    return best_h, best_c


def _local_search(h: np.ndarray, best_c: float, budget_s: float, start: float):
    """Simple simulated‑annealing on pairwise adjustments, respecting constraints."""
    n = h.shape[0]
    target = n / 2.0
    temp = 0.1
    best_h = h.copy()
    it = 0
    while time.time() - start < budget_s * 0.95:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 0.2  # move up to ±0.1
        hi, hj = h[i], h[j]
        new_i, new_j = hi + delta, hj - delta
        if not (0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0):
            continue
        h[i], h[j] = new_i, new_j
        c = _c5_of(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
            temp = max(temp * 0.9995, 1e-6)
        else:
            if np.random.rand() < math.exp(-(c - best_c) / temp):
                # accept worse move
                temp = max(temp * 0.9999, 1e-6)
            else:
                h[i], h[j] = hi, hj   # revert
        # occasional re‑projection to keep the sum exact
        if it % 5000 == 0:
            h = _project(h, target)
        it += 1
    best_h = _project(best_h, target)
    best_c = _c5_of(best_h)
    return best_h, best_c
