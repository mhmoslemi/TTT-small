```
def optimize_h(initial_h, max_iter=2000, time_limit=budget_s*0.9):
    h = initial_h.copy()
    n = len(h)
    dx = 2.0/n
    step = 0.05  # initial step size
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        # compute autocorrelation
        corr_auto = np.correlate(h, h, mode='full')
        # compute O_k = corr_auto * dx
        O_vals = corr_auto * dx
        # find min O
        Omin = O_vals.min()
        # find active shift distances
        idxs = np.where(O_vals <= Omin + 1e-12)[0]  # indices in full array
        # map to shift distances:
        shifts = idxs - (n - 1)
        abs_shifts = np.unique(np.abs(shifts))
        # compute gradient
        g = np.zeros_like(h)
        for d in abs_shifts:
            if d == 0:
                g += 2 * h
            else:
                # slice contributions
                g[:n-d] += h[d:]
                g[d:]   += h[:n-d]

            best_c5 = c5_new
            # keep step size
        else:

            step *= 0.7
            # optionally continue without update
        h = h_new
    return best_h, best_c5
```

```
def project_box_sum(v, target):
    # find lambda s.t. sum(clip(v - lam, 0, 1)) = target
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # adjust slight rounding errors
    diff = proj.sum() - target
    if abs(diff) > 1e-9:
        # distribute diff across interior points
        interior = (proj > 1e-12) & (proj < 1-1e-12)
        if interior.any():
            proj[interior] -= diff / interior.sum()
            proj = np.clip(proj, 0.0, 1.0)
        else:
            # fall back scaling
            proj *= target / proj.sum()
    return proj
```

def project_box_sum(v, target):
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # tiny adjustment
    diff = proj.sum() - target
    if abs(diff) > 1e-9:
        # adjust interior points if any
        interior = (proj > 1e-12) & (proj < 1-1e-12)
        if interior.any():
            proj[interior] -= diff / interior.sum()
            proj = np.clip(proj, 0.0, 1.0)
        else:
            proj *= target / proj.sum()
    return proj

import time
import numpy as np

# ---------- helper functions -------------------------------------------------

def compute_c5(h):
    """C5 value for a step function given by samples h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def autocorr(h):
    """Autocorrelation ∫ h(x)h(x+k) dx for all integer shifts k."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.correlate(h, h, mode="full") * dx


def project_box_sum(v, target):
    """
    Euclidean projection of vector v onto the set
        { x : 0 <= x_i <= 1 , sum x_i = target }.
    The solution is x_i = clip(v_i - λ, 0, 1) for a suitable λ.
    """
    lo, hi = -2.0, 2.0          # λ bounds certainly bracket the solution
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction for numerical drift
    diff = proj.sum() - target
    if abs(diff) > 1e-10:
        interior = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if interior.any():
            proj[interior] -= diff / interior.sum()
            proj = np.clip(proj, 0.0, 1.0)
        else:
            proj *= target / proj.sum()
    return proj


def improve_h(initial_h, time_limit, rng):
    """
    Heuristic ascent on the minimal autocorrelation.
    Returns the best vector seen and its C5.
    """
    h = initial_h.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # parameters
    step = 0.02
    sigma = 0.005
    last_improve = time.time()
    start = time.time()

    while time.time() - start < time_limit:
        # autocorrelation and its minimum
        auto = autocorr(h)
        O_min = auto.min()
        # indices where the minimum is attained (within tiny tolerance)
        idxs = np.where(auto <= O_min + 1e-12)[0]
        shifts = idxs - (n - 1)                # signed shifts
        abs_shifts = np.unique(np.abs(shifts))

        # gradient that increases the minimal autocorrelation
        g = np.zeros_like(h)
        for d in abs_shifts:
            if d == 0:
                g += 2.0 * h
            else:
                g[: n - d] += h[d:]
                g[d:]     += h[: n - d]

        # ascent step + small random jitter
        h_candidate = h + step * g + sigma * rng.standard_normal(n)
        h_candidate = project_box_sum(h_candidate, target)

        c5_candidate = compute_c5(h_candidate)

        if c5_candidate < best_c5 - 1e-9:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
            h = h_candidate
            # gently enlarge step after success
            step *= 1.02
            sigma *= 1.01
            last_improve = time.time()
        else:
            # no improvement: shrink step and jitter
            step *= 0.70
            sigma *= 0.90
            # occasional random restart if we are stuck for a while
            if time.time() - last_improve > 5.0 and step < 1e-5:
                # random binary start respecting the sum
                h = np.zeros(n)
                ones = rng.choice(n, size=n // 2, replace=False)
                h[ones] = 1.0
                h = project_box_sum(h, target)
                step = 0.02
                sigma = 0.005
                last_improve = time.time()
            else:
                # keep the current h (do not replace)
                pass

    return best_h, best_c5


# ---------- main entry point ------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points)
    h_values – list of samples defining the step function,
    c5_bound – the achieved C5 value,
    n_points – number of samples used.
    """
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # ----- obtain an initial construction -----------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        init = np.asarray(init, dtype=float)
        n0 = init.shape[0]
        # upscale if possible (factor 4 gives finer resolution, still cheap)
        factor = 4 if n0 * 4 <= 600 else 2
        h0 = np.repeat(init, factor)
    else:
        # fall back to a simple random half‑ones pattern
        n0 = 120
        h0 = np.zeros(n0)
        ones = rng.choice(n0, size=n0 // 2, replace=False)
        h0[ones] = 1.0

    n = h0.shape[0]
    target_sum = n / 2.0
    h0 = project_box_sum(h0, target_sum)

    # ----- optimisation ------------------------------------------------------
    remaining = budget_s * 0.92 - (time.time() - start_total)
    if remaining < 0.0:
        remaining = 0.0
    best_h, best_c5 = improve_h(h0, remaining, rng)

    # final safety projection (to guarantee feasibility)
    best_h = project_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h.tolist(), float(best_c5), n
