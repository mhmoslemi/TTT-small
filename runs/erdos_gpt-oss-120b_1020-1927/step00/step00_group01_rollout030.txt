import time
import numpy as np
from scipy.optimize import minimize, Bounds

def _compute_c5(h):
    """Return C5 for a given h (numpy array)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _grad_c5(h):
    """Sub‑gradient of C5 w.r.t. h (for the current maximal lag)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    L = idx - (n - 1)                # lag corresponding to the maximum
    # overlapping indices i such that 0<=i<n and 0<=i+L<n
    start = max(0, -L)
    end = min(n - 1, n - 1 - L)
    inds = np.arange(start, end + 1, dtype=int)
    j = inds + L                     # partner indices
    grad = np.zeros(n, dtype=float)
    grad[inds] += 1.0 - h[j]        # derivative from the linear term
    grad[j]    += -h[inds]          # derivative from the quadratic term
    return dx * grad

def _project_to_bounds_and_sum(x, target):
    """Project x onto the box [0,1]^n while keeping sum(x)=target."""
    # Simple water‑filling projection
    n = len(x)
    # Clip to the box first
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    if np.isclose(s, target, atol=1e-12):
        return x
    # If we need to add mass, distribute to non‑full coordinates
    # If we need to remove mass, take from non‑empty coordinates
    # Iterative approach (fast for our modest n)
    deficit = target - s
    while abs(deficit) > 1e-12:
        free = np.where((x > 0.0) & (x < 1.0))[0]
        if free.size == 0:
            # all variables are at a bound; distribute tiny amounts to keep feasibility
            free = np.arange(n)
        delta = deficit / free.size
        # clamp delta to respect bounds
        delta = np.clip(delta, -np.min(x[free]), 1.0 - np.max(x[free]))
        x[free] += delta
        s = np.sum(x)
        deficit = target - s
    return x

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points).
    h_values is a numpy array satisfying the constraints.
    """
    start_time = time.time()
    deadline = start_time + budget_s * 0.99  # a small safety margin

    rng = np.random.default_rng(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        init_parent = np.asarray(initial_h_values, dtype=float)
        n = init_parent.shape[0]
        # make sure parent respects the sum exactly
        target_sum = n / 2.0
        init_parent = _project_to_bounds_and_sum(init_parent, target_sum)
        initial_candidates = [init_parent]
    else:
        # default size – even for a perfect midpoint rule
        n = 100
        initial_candidates = []

    # Additional generic starts
    target_sum = n / 2.0
    dx = 2.0 / n

    # uniform 0.5
    uniform = np.full(n, 0.5, dtype=float)
    initial_candidates.append(uniform)

    # triangular (mid‑point rule makes the integral exact)
    x = (np.arange(n) + 0.5) * dx
    triangular = np.minimum(x, 2.0 - x)
    # projection just in case of odd n
    triangular = _project_to_bounds_and_sum(triangular, target_sum)
    initial_candidates.append(triangular)

    # a few random feasible vectors
    for _ in range(3):
        rnd = rng.random(n)
        rnd = _project_to_bounds_and_sum(rnd, target_sum)
        initial_candidates.append(rnd)

    best_h = None
    best_c5 = np.inf

    # bounds for SLSQP
    bounds = Bounds(0.0, 1.0)

    # equality constraint: sum(h) = n/2
    eq_cons = {
        "type": "eq",
        "fun": lambda h: np.sum(h) - target_sum,
        "jac": lambda h: np.ones_like(h),
    }

    # ---------- local optimisation with SLSQP ----------
    for h0 in initial_candidates:
        if time.time() >= deadline:
            break
        # SLSQP optimisation
        res = minimize(
            _compute_c5,
            h0,
            method="SLSQP",
            jac=_grad_c5,
            bounds=bounds,
            constraints=[eq_cons],
            options={"maxiter": 500, "ftol": 1e-12, "disp": False},
        )
        if not res.success:
            # fall back to the raw candidate
            cand_h = h0.copy()
        else:
            cand_h = res.x
        # enforce feasibility (tiny numeric drift)
        cand_h = _project_to_bounds_and_sum(cand_h, target_sum)
        cur_c5 = _compute_c5(cand_h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = cand_h.copy()

        # ---------- simple pairwise hill‑climbing ----------
        # try to improve the solution by random mass exchanges
        h = cand_h.copy()
        cur_score = cur_c5
        step_factor = 0.2  # typical magnitude for delta
        while time.time() < deadline:
            # propose a pairwise exchange
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # maximal admissible delta preserving bounds
            max_delta = min(h[i], 1.0 - h[j])
            min_delta = -min(h[j], 1.0 - h[i])
            if max_delta <= 0.0 and min_delta >= 0.0:
                continue
            delta = rng.uniform(min_delta, max_delta) * step_factor
            # apply
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            # evaluate
            new_score = _compute_c5(h_new)
            if new_score < cur_score:
                h = h_new
                cur_score = new_score
                if cur_score < best_c5:
                    best_c5 = cur_score
                    best_h = h.copy()
            # occasionally shrink step_factor to refine search
            if rng.random() < 0.001:
                step_factor *= 0.9
                step_factor = max(step_factor, 1e-4)

    # If nothing succeeded (should not happen), fall back to uniform
    if best_h is None:
        best_h = uniform.copy()
        best_c5 = _compute_c5(best_h)

    # Final safety projection
    best_h = _project_to_bounds_and_sum(best_h, target_sum)
    best_c5 = _compute_c5(best_h)

    return best_h, float(best_c5), int(n)
