def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    np.random.seed(seed)

    # ------------------------------------------------------------
    # Load or create an initial feasible h
    # ------------------------------------------------------------
    try:
        init = initial_h_values  # pre‑imported if available
        n = int(init.shape[0])
        h = init.astype(float).copy()
    except Exception:
        n = 64                       # fallback size
        h = np.full(n, 0.5, dtype=float)

    dx = 2.0 / n
    target_sum = n / 2.0

    # project onto the hyperplane sum(h)=target_sum while staying in [0,1]
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        # distribute the difference uniformly over free coordinates
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # clamp one entry
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
        else:
            inc = diff / free.size
            for idx in free:
                h[idx] = np.clip(h[idx] + inc, 0.0, 1.0)
        # final tiny correction
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            # adjust the last entry (small enough to stay in bounds)
            h[-1] = np.clip(h[-1] + diff, 0.0, 1.0)

    # ------------------------------------------------------------
    # Helper: evaluate C5 for a given h
    # ------------------------------------------------------------
    def evaluate(arr):
        # correlation of h with (1-h); np.correlate implements full cross‑correlation
        corr = np.correlate(arr, 1.0 - arr, mode="full")
        return np.max(corr) * dx

    cur_val = evaluate(h)
    best_h = h.copy()
    best_val = cur_val

    # ------------------------------------------------------------
    # Simulated annealing / hill‑climbing
    # ------------------------------------------------------------
    total_time = max(0.0, budget_s - 0.2)   # small safety margin
    start = time.time()
    T0 = 5e-2          # initial temperature
    T_end = 1e-5

    # pre‑allocate a small array for temporary corrections
    while True:
        elapsed = time.time() - start
        if elapsed >= total_time:
            break

        # linear temperature schedule
        progress = elapsed / total_time
        T = T0 * (1.0 - progress) + T_end * progress

        # propose a move that preserves the sum:
        i, j = np.random.randint(0, n, size=2)
        while j == i:
            j = np.random.randint(0, n)
        # direction: +1 moves mass from j to i, -1 the opposite
        direction = 1 if np.random.rand() < 0.5 else -1
        if direction == 1:
            max_delta = min(1.0 - h[i], h[j])
        else:
            max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 1e-14:
            continue

        # step size scales with temperature (larger steps early)
        frac = 0.5 + 0.5 * (T / T0)   # between 0.5 and 1.0
        delta = direction * max_delta * np.random.rand() * frac

        # apply
        h[i] += delta
        h[j] -= delta
        # (tiny numerical drift possible)
        h[i] = min(max(h[i], 0.0), 1.0)
        h[j] = min(max(h[j], 0.0), 1.0)

        new_val = evaluate(h)

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(new_val - cur_val) / T) if T > 0 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta

        # occasional small jitter to help escape tiny plateaus
        if int(elapsed * 10) % 2000 == 0:
            # add a tiny random perturbation while keeping the sum fixed
            idxs = np.random.choice(n, 2, replace=False)
            a, b = idxs[0], idxs[1]
            eps = 1e-4 * np.random.rand()
            if h[a] + eps <= 1.0 and h[b] - eps >= 0.0:
                h[a] += eps
                h[b] -= eps
            elif h[a] - eps >= 0.0 and h[b] + eps <= 1.0:
                h[a] -= eps
                h[b] += eps

    # ------------------------------------------------------------
    # Final feasibility clean‑up (enforce exact sum)
    # ------------------------------------------------------------
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny residual to a non‑saturated coordinate
        for idx in range(n):
            if diff > 0 and best_h[idx] < 1.0:
                add = min(diff, 1.0 - best_h[idx])
                best_h[idx] += add
                diff -= add
            elif diff < 0 and best_h[idx] > 0.0:
                sub = min(-diff, best_h[idx])
                best_h[idx] -= sub

        # as a last resort, tweak the last element
        if abs(diff) > 1e-14:
            best_h[-1] = np.clip(best_h[-1] + diff, 0.0, 1.0)

    final_val = evaluate(best_h)

    return best_h, final_val, n
