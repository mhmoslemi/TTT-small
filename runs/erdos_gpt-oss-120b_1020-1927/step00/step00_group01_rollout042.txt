def project_onto_bounds_and_sum(x, target):
   # Clip to [0,1]
   x = np.clip(x, 0, 1)
   current_sum = x.sum()
   diff = target - current_sum
   if abs(diff) < 1e-12: return x
   # Determine free indices where 0 < x_i < 1
   free = np.where((x > 0) & (x < 1))[0]
   if free.size == 0:
       # All at bounds; cannot adjust; fallback to scaling
       # if diff>0, set some to 1; else to 0 (should not happen)
       # We'll modify simply by redistributing uniformly and clipping again
       x = np.clip(x + diff / len(x), 0, 1)
       return x

       delta = diff / free.size
       x[free] += delta
       # Clip again and update diff
       # Any that exceed bounds become fixed and removed from free
       # Recompute diff and free

   return x

def project_onto_simplex_box(v, target):
    # This projects v onto the intersection of [0,1]^n and sum = target.
    # Implementation via iterative water-filling:
    # Clip to [0,1]
    v = np.clip(v, 0, 1)
    # Compute current sum & deficit
    s = v.sum()
    # If s == target done.
    if abs(s - target) < 1e-12:
        return v
    # Determine direction to adjust: need to add delta if s < target, else remove.
    # We'll use while loop.
    # At each iteration, compute free indices where v is not at bound (0< v <1).
    # Compute delta = (target - s) / len(free)
    # Add delta to v[free]; then clip again to [0,1]; recompute s, diff. Continue.
    max_iter = 1000
    for _ in range(max_iter):
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((v > 0) & (v < 1))[0]
        if free.size == 0:
            # All at bounds, can't adjust; break anyway
            break
        delta = diff / free.size
        # Apply delta
        v[free] += delta
        # Clip
        v = np.clip(v, 0, 1)
        # Continue loop until diff small or free zero
    # After loop, if still not exact target (due to floating diff ~ small),
    # adjust a single component within bounds.
    diff = target - v.sum()
    if abs(diff) > 1e-10:
        # Find any index that can be adjusted within bounds.
        # For diff>0, find idx where v < 1; for diff<0, where v>0.
        if diff > 0:
            candidates = np.where(v < 1)[0]
            if candidates.size > 0:
                idx = candidates[0]
                v[idx] = min(1, v[idx] + diff)
        else:
            candidates = np.where(v > 0)[0]
            if candidates.size > 0:
                idx = candidates[0]
                v[idx] = max(0, v[idx] + diff)  # diff negative
        # After adjusting one component, diff may not be entirely fixed if capacity limited.
        # Loop again maybe.
        # Ensure final sum within tolerance
    v = np.clip(v, 0, 1)
    return v

Simplified algorithm:

General algorithm:

def project_onto_simplex_box(v, target):
    v = np.clip(v, 0, 1)
    # Using iterative approach
    eps = 1e-12
    while True:
        s = v.sum()
        diff = target - s
        if abs(diff) <= eps:
            break
        # Determine indices that are not at bounds (i.e., can be adjusted)
        if diff > 0:
            free = np.where(v < 1 - eps)[0]
            if free.size == 0:
                break
        else:
            free = np.where(v > eps)[0]
            if free.size == 0:
                break
        delta = diff / free.size
        # Apply delta but cap at bounds
        # For each idx in free:
        # In vectorized form:
        adj = np.full_like(v, 0.0)
        adj[free] = delta
        v = v + adj
        # Clip again
        v = np.clip(v, 0, 1)
        # Continue
    # Ensure final sum
    # If still mismatch due to rounding, adjust by small epsilon on some free entries
    if abs(v.sum() - target) > 1e-9:
        # Simple correction
        diff = target - v.sum()
        # modify some index with capacity
        if diff > 0:
            idx = np.argmax(v < 1 - eps)
            if idx >= 0:
                v[idx] = min(1.0, v[idx] + diff)
        else:
            idx = np.argmax(v > eps)
            if idx >= 0:
                v[idx] = max(0.0, v[idx] + diff)
        # final clip
        v = np.clip(v, 0, 1)
    return v

import time
import numpy as np

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def _gradient_and_lag(h):
    """gradient of the worst (max) term and its lag."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)          # lag = index - centre
    g = np.zeros_like(h)

    if lag >= 0:
        L = lag
        if L < n:
            g[: n - L] += 1.0 - h[L:]
            g[L:] += -h[: n - L]
    else:
        L = -lag
        if L < n:
            g[L:] += 1.0 - h[: n - L]
            g[: n - L] += -h[L:]

    g *= dx
    return g, lag


def _project_to_bounds_and_sum(v, target):
    """Euclidean projection onto {0≤x≤1, sum x = target}."""
    v = np.clip(v, 0.0, 1.0)
    eps = 1e-12
    for _ in range(500):
        s = v.sum()
        diff = target - s
        if abs(diff) <= eps:
            break
        # free variables (not stuck at a bound)
        if diff > 0:
            free = np.where(v < 1.0 - eps)[0]
        else:
            free = np.where(v > eps)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # tiny residual fix
    diff = target - v.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            cand = np.where(v < 1.0 - eps)[0]
        else:
            cand = np.where(v > eps)[0]
        if cand.size:
            i = cand[0]
            v[i] = np.clip(v[i] + diff, 0.0, 1.0)
    return np.clip(v, 0.0, 1.0)


def _random_feasible(n, target, rng):
    v = rng.random(n)
    return _project_to_bounds_and_sum(v, target)


def _gradient_descent(start_h, target, max_iters, time_limit):
    """Simple projected sub‑gradient descent on the max‑overlap objective."""
    h = start_h.copy()
    cur = _compute_c5(h)
    best_h, best_val = h.copy(), cur
    step = 1.0
    start = time.time()
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        g, _ = _gradient_and_lag(h)
        improved = False
        for fac in (1.0, 0.5, 0.25):
            h_new = h - step * fac * g
            h_new = _project_to_bounds_and_sum(h_new, target)
            val = _compute_c5(h_new)
            if val < cur - 1e-12:
                h, cur = h_new, val
                improved = True
                if val < best_val:
                    best_h, best_val = h_new.copy(), val
                step *= 1.05
                break
        if not improved:
            step *= 0.5
        if step < 1e-8:
            break
    return best_h, best_val


def _hill_climb(start_h, max_iters, time_limit):
    """Greedy transfer of mass between two coordinates."""
    n = start_h.shape[0]
    target = n / 2.0
    h = start_h.copy()
    cur = _compute_c5(h)
    best_h, best_val = h.copy(), cur
    rng = np.random.default_rng()
    start = time.time()
    no_improve = 0
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        # move mass from i to j (or opposite) while respecting bounds
        if h[i] > 1e-12 and h[j] < 1.0 - 1e-12:
            max_delta = min(h[i], 1.0 - h[j])
            src, dst = i, j
        elif h[j] > 1e-12 and h[i] < 1.0 - 1e-12:
            max_delta = min(h[j], 1.0 - h[i])
            src, dst = j, i
        else:
            continue
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * 0.5
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        val = _compute_c5(h_new)
        if val < cur - 1e-12:
            h, cur = h_new, val
            no_improve = 0
            if val < best_val:
                best_h, best_val = h_new.copy(), val
        else:
            no_improve += 1
    return best_h, best_val

    start_all = time.time()

    # --- decide the discretisation size ---------------------------------
    if "initial_h_values" in globals():
        init_vec = np.asarray(initial_h_values, dtype=float)
        n = init_vec.shape[0]
    else:
        # a modest size that still gives flexibility
        n = 200
        init_vec = None

    target_sum = n / 2.0

    # make sure the (possibly) provided vector obeys the constraints
    if init_vec is not None and init_vec.shape[0] == n:
        init_vec = _project_to_bounds_and_sum(init_vec, target_sum)
        best_h = init_vec.copy()
        best_val = _compute_c5(best_h)
    else:
        best_h = None
        best_val = np.inf

    # --- main optimisation loop -----------------------------------------
    restart = 0
    while time.time() - start_all < budget_s - 5:
        # time left for this restart (leave a small safety margin)
        time_left = budget_s - (time.time() - start_all) - 2
        if time_left <= 0:

        # initialise a start point
        if restart == 0 and init_vec is not None:
            h0 = init_vec.copy()
        else:
            h0 = _random_feasible(n, target_sum, rng)

        # 1) greedy hill‑climb
        hc_h, hc_val = _hill_climb(h0, max_iters=20000, time_limit=time_left * 0.6)

        if hc_val < best_val:
            best_h, best_val = hc_h, hc_val

        # 2) short gradient‑descent refinement

        if gd_val < best_val:
            best_h, best_val = gd_h, gd_val

        restart += 1
        if restart >= 20:       # safety cap on restarts
            break

    # final safety projection
    best_h = _project_to_bounds_and_sum(best_h, target_sum)
    best_val = _compute_c5(best_h)

    return best_h, float(best_val), n
