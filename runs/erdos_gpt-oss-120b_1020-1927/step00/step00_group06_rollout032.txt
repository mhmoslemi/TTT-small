def project(v):

import time
import numpy as np

def project_to_capped_simplex(v, target):
    n = v.size
    lo = -np.max(v) - 1.0
    hi = 1 - np.min(v) + 1.0
    for _ in range(60):
        theta = (lo + hi) / 2.0
        x = np.clip(v + theta, 0, 1)
        s = x.sum()
        if s < target:
            lo = theta
        else:
            hi = theta
    x = np.clip(v + (lo+hi)/2.0, 0, 1)
    return x

def evaluate_c5(h):

import time
import numpy as np

def _project_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {x∈[0,1]^n : sum(x)=target}."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        theta = (lo + hi) / 2.0
        x = np.clip(v + theta, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = theta
        else:
            hi = theta
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Return the overlap constant C₅ for a vector h."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / h.size)   # dx = 2/n

    if tri_c5 < best_c5:
