import time

# ------------------------------------------------------------
# Helper utilities

def _shift_array(arr: np.ndarray, shift: int) -> np.ndarray:
    """Return arr shifted by +shift (zero padding)."""
    n = arr.shape[0]
    out = np.empty_like(arr)
    if shift >= 0:
        out[: n - shift] = arr[shift:]
        out[n - shift :] = 0.0
    else:
        s = -shift
        out[s:] = arr[: n - s]
        out[:s] = 0.0
    return out


def _project_to_sum_and_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector `v` onto the set {x | 0<=x_i<=1, sum x_i = target}.
    Uses a simple bisection on a common additive shift lambda.
    """
    lo = -np.max(v) - 2.0          # guarantee all zeros
    hi = 1.0 - np.min(v) + 2.0     # guarantee all ones
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Integral value defined in the statement."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))
