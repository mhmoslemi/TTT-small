import numpy as np
import time

def _adjust_sum(v, target):
    """Adjust one entry of v so that sum(v) == target while staying in [0,1]."""
    diff = target - v.sum()
    if abs(diff) < 1e-14:
        return v
    # Find an index where we can add diff without violating bounds.
    feasible = np.where((v + diff >= 0.0) & (v + diff <= 1.0))[0]
    if feasible.size == 0:
        # Fallback: pick any index that can be moved partially.
        if diff > 0:
            idx = np.argmax(v)   # the largest value has room upwards if not at 1
        else:
            idx = np.argmin(v)   # the smallest value has room downwards
        # Clip after adjustment.
        v[idx] = np.clip(v[idx] + diff, 0.0, 1.0)
    else:
        idx = feasible[0]
        v[idx] += diff
    return v

    Erdős overlap constant C5.

    Returns:
        h_values : np.ndarray of shape (n_points,)
        c5_bound : float, the computed C5 value
        n_points : int, length of the discretisation
    """
    # Determine number of points (use the length of the supplied parent if any)
    if 'initial_h_values' in globals() and isinstance(initial_h_values, np.ndarray):
        n = int(initial_h_values.shape[0])
    else:

    i = np.arange(n)

    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))

    # Enforce the exact integral constraint: sum(h) == n/2
    target_sum = n / 2.0
