def propose(h, eps=0.1):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    if hi > hj:
        # propose moving from i (higher) to j (lower)
        max_move = min(eps, hi - 0, 1 - hj)
        if max_move <= 0:
            return None
        delta = np.random.uniform(0, max_move)
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    else:
        # hi <= hj, move from j to i
        max_move = min(eps, hj - 0, 1 - hi)
        if max_move <= 0:
            return None
        delta = np.random.uniform(0, max_move)
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
    return new_h

def _c5_value(h):
    """Return the C5 value for a discretised step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")

def _initial_vector():
    """Return a feasible starting vector."""
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        # enforce exact sum (might be off by 1e-12)
        target = h0.size / 2.0
        cur = h0.sum()
        if abs(cur - target) > 1e-12:
            diff = target - cur
            # distribute the tiny diff on a non‑boundary entry
            idxs = np.where((h0 > 0) & (h0 < 1))[0]
            if idxs.size == 0:
                idxs = np.arange(h0.size)
            i = np.random.choice(idxs)
            h0[i] = np.clip(h0[i] + diff, 0.0, 1.0)
        return h0
    # fallback: binary random start
    n = 80
    return _random_binary(n)

def _propose_move(h, eps):
    """Create a neighbour of h by moving a small amount between two coordinates."""
    n = h.size
    i, j = np.random.randint(0, n, size=2, replace=False)
    hi, hj = h[i], h[j]
    if hi > hj:
        max_move = min(eps, hi, 1.0 - hj)
        if max_move <= 0.0:
            return None
        delta = np.random.rand() * max_move
        h_new = h.copy()
        h_new[i] = hi - delta
        h_new[j] = hj + delta
    elif hj > hi:
        max_move = min(eps, hj, 1.0 - hi)
        if max_move <= 0.0:
            return None
        delta = np.random.rand() * max_move
        h_new = h.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta
    else:
        return None
    return h_new

def run(seed=42, budget_s=1100, **_):
    """Return a step‑function h, its C5 bound and the number of discretisation points."""
    np.random.seed(seed)
    random.seed(seed)

    h_best = _initial_vector()
    n = h_best.size
    score_best = _c5_value(h_best)

    # If a pre‑existing solution is present, keep it as a reference
    try:
        score_initial = _c5_value(h_best)
    except Exception:
        score_initial = score_best

    # Current working point
    h_cur = h_best.copy()
    score_cur = score_best

    start_time = time.time()
    time_limit = budget_s * 0.96  # a safety margin

    eps = 0.5          # maximal mass transferred per move
    T0 = 0.02          # initial "temperature"
    it = 0

    while time.time() - start_time < time_limit:
        # propose a neighbour
        h_new = _propose_move(h_cur, eps)
        if h_new is None:
            continue
        score_new = _c5_value(h_new)

        # annealed acceptance
        elapsed = time.time() - start_time
        T = T0 * (1.0 - elapsed / time_limit)
        accept = False
        if score_new < score_cur:
            accept = True
        else:
            prob = np.exp(-(score_new - score_cur) / (T + 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            h_cur = h_new
            score_cur = score_new
            if score_new < score_best:
                h_best = h_new.copy()
                score_best = score_new
                # early stop if we already beat the target
                if score_best <= 0.3808:
                    break

        # schedule adjustments
        it += 1
        if it % 20000 == 0:
            eps = max(0.01, eps * 0.8)   # gradually shrink step size
        if it % 50000 == 0:
            # occasional random restart around the best point
            if np.random.rand() < 0.3:
                # add tiny noise to the best solution
                noise = (np.random.rand(n) - 0.5) * eps * 0.5
                h_cur = np.clip(h_best + noise, 0.0, 1.0)
                # re‑project to the exact mass constraint
                diff = (n / 2.0) - h_cur.sum()
                # distribute the diff uniformly on non‑boundary entries
                idxs = np.where((h_cur > 0.0) & (h_cur < 1.0))[0]
                if idxs.size:
                    adj = diff / idxs.size
                    h_cur[idxs] += adj
                    h_cur = np.clip(h_cur, 0.0, 1.0)
                score_cur = _c5_value(h_cur)

    # Final projection to satisfy the integral exactly
    total = h_best.sum()
    target = n / 2.0
    diff = target - total
    if abs(diff) > 1e-12:
        # modify a single interior coordinate if possible
        idxs = np.where((h_best > 0.0) & (h_best < 1.0))[0]
        if idxs.size == 0:
            idxs = np.arange(n)
        i = np.random.choice(idxs)
        h_best[i] = np.clip(h_best[i] + diff, 0.0, 1.0)

    h_best = np.clip(h_best, 0.0, 1.0)

    # final evaluation (use the provided helper if it exists)
    try:
        final_score = evaluate_erdos_solution(h_best)
    except Exception:
        final_score = _c5_value(h_best)

    return h_best.tolist(), float(final_score), n
