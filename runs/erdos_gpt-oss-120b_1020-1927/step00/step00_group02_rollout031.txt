import math
import time
import random
import numpy as np

def _stats(h):
    """Return (C5, correlation vector s_k = sum_i h_i h_{i+k})."""
    n = h.shape[0]
    # autocorrelation (linear, zero‑padded)
    full = np.correlate(h, h, mode="full")
    s = full[n - 1 :]                 # k = 0 .. n-1
    min_s = s.min()
    c5 = 1.0 - (2.0 / n) * min_s       # 1 - dx * min_s,  dx = 2/n
    return c5, s


def _grad_shift(h, k):
    """Gradient of s_k = sum_i h_i h_{i+k} w.r.t. h (k >= 0)."""
    n = h.shape[0]
    if k == 0:
        return 2.0 * h
    grad = np.zeros_like(h)
    grad[: n - k] += h[k:]
    grad[k:] += h[: n - k]
    return grad


def _random_move(h, step):
    """Return a new h after moving a small amount from a positive to a free entry."""
    pos = np.where(h > 1e-12)[0]
    free = np.where(h < 1.0 - 1e-12)[0]
    if pos.size == 0 or free.size == 0:
        return None
    i = np.random.choice(pos)
    j = np.random.choice(free)
    delta = min(step, h[i], 1.0 - h[j])
    if delta <= 0:
        return None
    h2 = h.copy()
    h2[i] -= delta
    h2[j] += delta
    return h2


def _project_sum_one(h, target):
    """Adjust h to satisfy sum(h)=target while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # simple scan‑and‑fill
    for i in range(len(h)):
        if diff > 0 and h[i] < 1.0:
            inc = min(1.0 - h[i], diff)
            h[i] += inc
            diff -= inc
        elif diff < 0 and h[i] > 0.0:
            dec = min(h[i], -diff)
            h[i] -= dec
            diff += dec
        if abs(diff) < 1e-12:
            break
    return h

        k_star = int(np.argmin(cur_s))          # could be 0, but that's fine
        grad = _grad_shift(h, k_star)

        # candidate indices for increase / decrease
        inc_mask = (grad == grad.max()) & (h < 1.0 - 1e-12)
        dec_mask = (grad == grad.min()) & (h > 1e-12)
        inc_idx = np.where(inc_mask)[0]
        dec_idx = np.where(dec_mask)[0]

        if inc_idx.size and dec_idx.size:
            i_up = np.random.choice(inc_idx)
            i_down = np.random.choice(dec_idx)
            delta = min(step, 1.0 - h[i_up], h[i_down])
            if delta > 1e-12:
                h_new = h.copy()
                h_new[i_up] += delta
                h_new[i_down] -= delta
            else:
                h_new = _random_move(h, step)
        else:
            h_new = _random_move(h, step)

        if h_new is None:
            continue

        new_c5, new_s = _stats(h_new)
        d = new_c5 - cur_c5

        if d < -1e-12 or random.random() < math.exp(-d / max(T, 1e-12)):
            # accept
            h = h_new
            cur_c5 = new_c5
            cur_s = new_s
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h.copy()
                if best_c5 < 0.3808:               # early stop if target met
                    break

    # final tiny re‑projection to guarantee exact mass
    best_h = _project_sum_one(best_h, target_sum)

    final_c5, _ = _stats(best_h)
    return best_h.tolist(), float(final_c5), int(n)
