def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # Use FFT for correlation
    size = 1 << ((2*n - 1).bit_length())
    # FFT
    H = np.fft.rfft(h, n=size)
    G = np.fft.rfft(1.0 - h, n=size)
    corr = np.fft.ifft(H * np.conj(G)).real[:2*n-1]
    return float(np.max(corr) * dx)

def project_pair_move(h, i, j, delta):

    h_new[i] += delta

def local_search(h, time_budget, start_step=0.2, step_factor=0.995, min_step=1e-4):
    start = time.time()
    n = h.shape[0]
    dx = 2.0 / n
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    step = start_step
    while time.time() - start < time_budget:

        i, j = random.sample(range(n), 2)
        hi = current_h[i]
        hj = current_h[j]
        # compute allowable delta range
        delta_min = max(-hi, hj - 1.0)
        delta_max = min(1.0 - hi, hj)
        if delta_max <= delta_min:
            continue
        delta = random.uniform(delta_min, delta_max)
        # Maybe scale delta by step factor to limit magnitude
        # Actually we can incorporate step by limiting amplitude:
        max_abs = step
        if delta > max_abs:
            delta = max_abs
        elif delta < -max_abs:
            delta = -max_abs
        # apply move
        new_h = current_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # compute new c5
        new_c5 = compute_c5(new_h)
        # acceptance: greedy
        if new_c5 <= current_c5:

            current_c5 = new_c5

        # step decay
        step = max(step * step_factor, min_step)
    return best_h, best_c5

def adjust_sum_to_target(h, target):
    cur = h.sum()
    diff = target - cur
    n = len(h)
    if abs(diff) < 1e-12:
        return h
    # We adjust using random indices in a loop
    # While diff significant
    max_iter = 10000
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        idx = random.randrange(n)
        # Determine maximum we can add/subtract at this index
        if diff > 0:
            # we need to increase
            can = 1.0 - h[idx]
            if can <= 0:
                continue
            inc = min(can, diff)
            h[idx] += inc
            diff -= inc
        else:
            # need to decrease
            can = h[idx]  # amount we can subtract without going below 0
            if can <= 0:
                continue
            dec = min(can, -diff)
            h[idx] -= dec
            diff += dec
    # after loop, due to rounding may still be off small; adjust all equally
    if abs(diff) > 1e-9:
        # distribute evenly
        h += diff / n
        h = np.clip(h, 0, 1)
    return h

    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    step = start_step
    T = start_temp
    iters = 0
    start = time.time()
    while time.time() - start < time_budget:
        iters += 1
        # pick indices
        i, j = random.sample(range(n), 2)
        hi = cur_h[i]; hj = cur_h[j]
        delta_min = max(-hi, hj - 1.0)
        delta_max = min(1.0 - hi, hj)
        if delta_max <= delta_min:
            continue
        delta = random.uniform(delta_min, delta_max)
        # bound by step
        if delta > step:
            delta = step
        elif delta < -step:
            delta = -step
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_c5 = compute_c5(new_h)
        delta_c5 = new_c5 - cur_c5
        accepted = False
        if delta_c5 <= 0:
            accepted = True
        else:
            if T > 0:
                prob = math.exp(-delta_c5 / T)
                if random.random() < prob:
                    accepted = True
        if accepted:
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        # update step size and temp
        step = max(step * step_factor, min_step)
        T = max(T * temp_decay, min_temp)
    return best_h, best_c5

    n = len(h)
    g = np.zeros(n)
    # contributions
    if k >= 0:
        # for i where i + k < n: add h[i+k]
        g[:n-k] += h[k:]  # h[i+k] for i=0..n-k-1
        # for i where i - k >= 0: add h[i - k]
        if k > 0:
            g[k:] += h[:n-k]  # h[i-k] for i=k..n-1
    else:
        # k negative: handle similarly with reversed sign
        kk = -k
        g[kk:] += h[:n-kk]
        if kk > 0:
            g[:n-kk] += h[kk:]
    return g

def evaluate_all(h):
    n = h.shape[0]
    dx = 2.0 / n
    size = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=size)
    # conv between h and (1-h) reversed
    R1 = np.fft.rfft((1.0 - h)[::-1], n=size)
    conv1 = np.fft.irfft(H * R1, n=size)[:2 * n - 1]
    max_corr = np.max(conv1)
    # auto correlation
    auto = np.fft.irfft(H * np.conj(H), n=size)[:2 * n - 1]
    # Determine min auto correlation and index
    min_idx = np.argmin(auto)
    min_corr = auto[min_idx]
    C5 = max_corr * dx
    return C5, auto, min_idx

def compute_gradient(h, k):
    n = h.shape[0]
    g = np.zeros(n)
    if k >= 0:
        # i + k < n
        if k < n:
            g[: n - k] += h[k:]
            if k > 0:
                g[k:] += h[: n - k]
    else:
        kk = -k
        if kk < n:
            g[kk:] += h[: n - kk]
            if kk > 0:
                g[: n - kk] += h[kk:]
    return g

def guided_move(h, min_idx, step):
    n = h.shape[0]
    k = min_idx - (n - 1)
    g = compute_gradient(h, k)
    # masks
    eps = 1e-12
    src_mask = (h > eps)
    tgt_mask = (h < 1.0 - eps)
    if not np.any(src_mask) or not np.any(tgt_mask):
        return None
    # find indices with extreme gradient
    src_vals = g[src_mask]
    min_g = src_vals.min()
    src_candidates = np.where(src_mask & (np.abs(g - min_g) < 1e-12))[0]
    if src_candidates.size == 0:
        src_candidates = np.where(src_mask)[0]
    source = int(np.random.choice(src_candidates))

    tgt_vals = g[tgt_mask]
    max_g = tgt_vals.max()
    tgt_candidates = np.where(tgt_mask & (np.abs(g - max_g) < 1e-12))[0]
    if tgt_candidates.size == 0:
        tgt_candidates = np.where(tgt_mask)[0]
    target = int(np.random.choice(tgt_candidates))

    # amount to transfer
    delta_max = min(step, h[source], 1.0 - h[target])
    if delta_max <= 0:
        return None
    delta = np.random.uniform(0, delta_max)
    h_new = h.copy()
    h_new[source] -= delta
    h_new[target] += delta

def adjust_sum_to_target(h, target):
    cur = h.sum()
    diff = target - cur
    n = h.shape[0]
    if abs(diff) < 1e-12:
        return h
    # simple distribution
    # Add/subtract diff/n to all
    h = h + diff / n
    # Clip and fix residual
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    # If still not zero within tolerance, fix iteratively
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # distribute diff uniformly again
        add = diff / n
        h = h + add
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        diff = target - cur
    # If still leftover (rare), adjust random indices
    if abs(diff) > 1e-12:
        indices = np.arange(n)
        np.random.shuffle(indices)
        for idx in indices:
            if diff > 0:
                possible = 1.0 - h[idx]
                inc = min(possible, diff)
                h[idx] += inc
                diff -= inc
            else:
                possible = h[idx]
                dec = min(possible, -diff)
                h[idx] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
    return h

import numpy as np
import random
import time
import math

# ---------- helper functions ----------

def evaluate_all(h):
    """Return (C5, full auto‑correlation array, index of minimum auto‑correlation)."""
    n = h.shape[0]
    dx = 2.0 / n
    size = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n-1
    H = np.fft.rfft(h, n=size)
    # correlation of h with (1‑h)
    R1 = np.fft.rfft((1.0 - h)[::-1], n=size)
    conv1 = np.fft.irfft(H * R1, n=size)[:2 * n - 1]
    max_corr = np.max(conv1)
    # auto‑correlation of h
    auto = np.fft.irfft(H * np.conj(H), n=size)[:2 * n - 1]
    min_idx = int(np.argmin(auto))
    C5 = max_corr * dx
    return C5, auto, min_idx


def compute_c5(h):
    """Convenient wrapper returning only C5."""
    C5, _, _ = evaluate_all(h)
    return C5


def compute_gradient(h, k):
    """Gradient of the autocorrelation at shift k (g_i = h_{i+k}+h_{i‑k})."""
    n = h.shape[0]
    g = np.zeros(n, dtype=float)
    if k >= 0:
        if k < n:
            g[:n - k] += h[k:]               # h_{i+k}
            if k > 0:
                g[k:] += h[:n - k]           # h_{i‑k}
    else:
        kk = -k
        if kk < n:
            g[kk:] += h[:n - kk]             # h_{i‑kk} = h_{i+k}
            if kk > 0:
                g[:n - kk] += h[kk:]         # h_{i+kk} = h_{i‑k}
    return g


def guided_move(h, min_idx, step):
    """Perform a move that tries to raise the worst autocorrelation."""
    n = h.shape[0]
    k = min_idx - (n - 1)          # actual shift
    g = compute_gradient(h, k)

    eps = 1e-12
    src_mask = h > eps
    tgt_mask = h < 1.0 - eps
    if not np.any(src_mask) or not np.any(tgt_mask):
        return None

    # source: minimal gradient among feasible decreases
    src_vals = g[src_mask]
    min_g = src_vals.min()
    src_candidates = np.where(src_mask & (np.abs(g - min_g) < 1e-12))[0]
    if src_candidates.size == 0:
        src_candidates = np.where(src_mask)[0]
    source = int(np.random.choice(src_candidates))

    # target: maximal gradient among feasible increases
    tgt_vals = g[tgt_mask]
    max_g = tgt_vals.max()
    tgt_candidates = np.where(tgt_mask & (np.abs(g - max_g) < 1e-12))[0]
    if tgt_candidates.size == 0:
        tgt_candidates = np.where(tgt_mask)[0]
    target = int(np.random.choice(tgt_candidates))

    delta_max = min(step, h[source], 1.0 - h[target])
    if delta_max <= 0:
        return None
    delta = np.random.uniform(0.0, delta_max)

    h_new = h.copy()
    h_new[source] -= delta
    h_new[target] += delta
    return h_new


def local_search(h_init, time_budget,
                 start_step=0.3, step_factor=0.9995,
                 min_step=1e-6, guided_prob=0.7):
    """Projected gradient‑like search with occasional random pair moves."""
    start = time.time()
    h = h_init.copy()
    C5_cur, auto_cur, min_idx_cur = evaluate_all(h)
    best_h = h.copy()
    best_c5 = C5_cur
    step = start_step

    while time.time() - start < time_budget:
        if random.random() < guided_prob:
            h_new = guided_move(h, min_idx_cur, step)
            if h_new is not None:
                C5_new, auto_new, min_idx_new = evaluate_all(h_new)
                if C5_new <= C5_cur + 1e-12:
                    h, C5_cur, auto_cur, min_idx_cur = h_new, C5_new, auto_new, min_idx_new
                    if C5_new < best_c5:
                        best_c5, best_h = C5_new, h_new.copy()
                step = max(step * step_factor, min_step)
                continue

        # ----- random pair move -----
        n = h.shape[0]
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]
        delta_min = max(-hi, hj - 1.0)
        delta_max = min(1.0 - hi, hj)
        if delta_max <= delta_min:
            step = max(step * step_factor, min_step)
            continue
        delta = random.uniform(delta_min, delta_max)
        # bound by current step size
        if delta > step:
            delta = step
        elif delta < -step:
            delta = -step
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        C5_new, auto_new, min_idx_new = evaluate_all(h_new)
        if C5_new <= C5_cur + 1e-12:
            h, C5_cur, auto_cur, min_idx_cur = h_new, C5_new, auto_new, min_idx_new
            if C5_new < best_c5:
                best_c5, best_h = C5_new, h_new.copy()
        step = max(step * step_factor, min_step)

    return best_h, best_c5


def adjust_sum_to_target(h, target):
    """Clamp into [0,1] and enforce exact sum = target (using simple redistribution)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    n = h.shape[0]
    if abs(diff) < 1e-12:
        return h
    # first uniform tweak
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    # iterate a few times if needed
    for _ in range(5):
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        diff = target - cur
    # final catch‑all using random indices
    if abs(diff) > 1e-12:
        idxs = np.arange(n)
        np.random.shuffle(idxs)
        for idx in idxs:
            if diff > 0:
                inc = min(1.0 - h[idx], diff)
                h[idx] += inc
                diff -= inc
            else:
                dec = min(h[idx], -diff)
                h[idx] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
    return h


def interpolate_h(h_small, n_new):
    """Linear interpolation of a coarser step function onto a finer grid."""
    n_small = h_small.shape[0]
    x_small = np.linspace(0.0, 2.0, n_small, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_small, h_small)
    h_new = np.clip(h_new, 0.0, 1.0)
    target = n_new / 2.0
    h_new = adjust_sum_to_target(h_new, target)
    return h_new
