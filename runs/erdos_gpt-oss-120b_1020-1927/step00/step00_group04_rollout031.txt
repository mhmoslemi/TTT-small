def shift_with_zeros(arr, shift):
    n = arr.shape[0]
    if shift > 0:
        out = np.empty_like(arr)
        out[:shift] = 0.0

        return out
    elif shift < 0:
        shift = -shift
        out = np.empty_like(arr)
        out[-shift:] = 0.0
        out[:n-shift] = arr[shift:]
        return out

        return arr.copy()

def project_box_sum(x, target, lower=0.0, upper=1.0):
    # project onto box [lower, upper] and equality sum constraint
    # Based on bisection approach
    lo = -1e5
    hi = 1e5
    for _ in range(60):
        lam = (lo + hi) / 2.0
        y = np.clip(x - lam, lower, upper)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    y = np.clip(x - (lo + hi) / 2.0, lower, upper)
    return y

import time
import numpy as np
import random

# ---------- helper utilities ----------

def shift_with_zeros(arr: np.ndarray, shift: int) -> np.ndarray:
    """Return arr shifted by `shift` positions; out‑of‑range entries are 0."""
    n = arr.shape[0]
    if shift == 0:
        return arr.copy()
    if shift > 0:
        out = np.empty_like(arr)
        out[:shift] = 0.0
        out[shift:] = arr[: n - shift]
        return out
    # shift < 0
    shift = -shift
    out = np.empty_like(arr)
    out[-shift:] = 0.0
    out[: n - shift] = arr[shift:]
    return out


def compute_c5(h: np.ndarray) -> float:
    """Exact objective used by the evaluator."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def project_box_sum(x: np.ndarray,
                    target: float,
                    lower: float = 0.0,
                    upper: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of `x` onto the intersection
    { lower ≤ z ≤ upper , sum(z)=target }.
    """
    lo, hi = -1e5, 1e5
    for _ in range(60):
        lam = (lo + hi) * 0.5
        z = np.clip(x - lam, lower, upper)
        s = z.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) * 0.5, lower, upper)


def gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the maximal‑shift overlap term."""
    n = h.shape[0]
    dx = 2.0 / n
    plus = shift_with_zeros(h, shift)
    minus = shift_with_zeros(h, -shift)
    return dx * (1.0 - plus - minus)


def projected_gradient_descent(h0: np.ndarray,
                               max_iters: int,
                               target: float) -> tuple:
    """
    Simple projected gradient descent that always moves opposite to the
    gradient of the currently maximal shift.
    Returns (best_h, best_c5).
    """
    n = h0.shape[0]
    lower, upper = 0.0, 1.0
    h = project_box_sum(h0, target, lower, upper)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    lr = 5.0  # initial step size
    factors = [0.0, 0.2, 0.5, 1.0, 2.0, 5.0]

    for _ in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)

        grad = gradient_for_shift(h, shift)

        # try a few step‑size factors and keep the best candidate
        cur_c5 = compute_c5(h)
        best_local_h = h
        best_local_c5 = cur_c5
        for f in factors:
            cand = project_box_sum(h - lr * f * grad, target, lower, upper)
            cand_c5 = compute_c5(cand)
            if cand_c5 < best_local_c5:
                best_local_c5 = cand_c5
                best_local_h = cand

        # record global best
        if best_local_c5 < best_c5 - 1e-12:
            best_c5 = best_local_c5
            best_h = best_local_h.copy()

        # update current point and adapt learning rate
        if best_local_c5 < cur_c5:
            h = best_local_h
            lr = min(lr * 1.1, 20.0)
        else:
            h = best_local_h
            lr = max(lr * 0.5, 1e-6)

    return best_h, best_c5


def random_binary_h(n: int) -> np.ndarray:
    """Uniform random binary vector with exactly n/2 ones (n must be even)."""
    if n % 2 != 0:
        # fallback to a continuous random vector
        return random_initial_h(n)
    ones = n // 2
    h = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, ones, replace=False)
    h[idx] = 1.0
    return h


def random_initial_h(n: int) -> np.ndarray:
    """Random vector in [0,1]^n projected onto the required simplex."""
    x = np.random.rand(n)
    target = n / 2.0
    return project_box_sum(x, target)


def upsample(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from `h` (size n) to a finer grid of size `new_n`."""
    n = h.shape[0]
    if new_n == n:
        return h.copy()
    dx_old = 2.0 / n
    dx_new = 2.0 / new_n
    old_x = (np.arange(n) + 0.5) * dx_old
    new_x = (np.arange(new_n) + 0.5) * dx_new
    h_new = np.interp(new_x, old_x, h, left=0.0, right=0.0)
    target = new_n / 2.0
    return project_box_sum(h_new, target)


def binary_local_search(h_start: np.ndarray,
                        max_swaps: int = 2000) -> tuple:
    """Greedy hill‑climbing on binary vectors (swap a 1 with a 0)."""
    n = h_start.shape[0]
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for _ in range(max_swaps):
        i, j = np.random.randint(0, n, size=2)
        if h[i] == h[j]:
            continue
        # ensure i is the 1‑position, j the 0‑position
        if h[i] < h[j]:
            i, j = j, i
        h[i], h[j] = 0.0, 1.0
        cand_c5 = compute_c5(h)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = h.copy()
        else:
            # revert the swap
            h[i], h[j] = 1.0, 0.0

    return best_h, best_c5


# ---------- main entry point ----------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where h_values is a numpy array
    representing a step function on [0,2] with the required constraints.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()

    # ------------------------------------------------------------------
    # initialise from the provided parent if available
    if "initial_h_values" in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64)
        n0 = h0.shape[0]
        target0 = n0 / 2.0
        h0 = project_box_sum(h0, target0)
        best_h, best_c5 = h0, compute_c5(h0)
        best_n = n0
    else:
        # fallback initial construction: 50 % ones, 50 % zeros
        n0 = 42
        h0 = random_binary_h(n0)
        best_h, best_c5 = h0, compute_c5(h0)
        best_n = n0

    # ------------------------------------------------------------------
    # multi‑resolution optimisation loop
    max_n = 512                     # upper bound on grid size
    max_iters_per_stage = 1500      # gradient iterations per start
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.92:        # leave a small safety margin
            break
        if best_n >= max_n:
            break

        next_n = min(best_n * 2, max_n)
        target_next = next_n / 2.0

        # upsample current best solution to the finer grid
        h_up = upsample(best_h, next_n)

        # collect a few different starting points
        starters = [h_up,
                    random_binary_h(next_n),
                    random_initial_h(next_n)]
        # a few extra random binary attempts
        for _ in range(2):
            starters.append(random_binary_h(next_n))

        stage_best_h = None
        stage_best_c5 = float("inf")

        for h_start in starters:
            # binary hill‑climb if the start happens to be binary
            if np.all(np.logical_or(h_start == 0, h_start == 1)):
                h_bin, c5_bin = binary_local_search(h_start, max_swaps=2500)
                if c5_bin < stage_best_c5:
                    stage_best_c5, stage_best_h = c5_bin, h_bin

            # gradient descent refinement
            h_opt, c5_opt = projected_gradient_descent(
                h_start,
                max_iters=max_iters_per_stage,
                target=target_next
            )
            if c5_opt < stage_best_c5:
                stage_best_c5, stage_best_h = c5_opt, h_opt

        # accept the new grid if it improves the global record
        if stage_best_c5 < best_c5 - 1e-6:
            best_h, best_c5 = stage_best_h, stage_best_c5
            best_n = next_n
        else:
            # even without improvement we keep the finer grid for future
            best_h, best_c5 = stage_best_h, stage_best_c5
            best_n = next_n

    # final sanity projection and evaluation
    best_h = project_box_sum(best_h, best_h.shape[0] / 2.0)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), best_h.shape[0])
