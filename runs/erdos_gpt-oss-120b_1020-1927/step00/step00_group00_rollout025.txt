def compute_gradient(h, shift):
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        # contribution from h[i+shift]
        g[:n-shift] -= h[shift:]
        # contribution from h[i-shift]

        s = -shift
        # shift negative: i+shift = i - s
        g[s:] -= h[:n-s]
        # i-shift = i + s
        g[:n-s] -= h[s:]
    return g

import time
import numpy as np

def _project_sum_box(x: np.ndarray, target: float, tol: float = 1e-12) -> np.ndarray:
    """Project vector x onto the intersection {0≤z≤1, sum(z)=target}."""
    lo, hi = -2.0, 2.0
    # binary search for the correct shift λ
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def _gradient(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the worst correlation w.r.t. h."""
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        s = shift
        if s < n:
            g[:n - s] -= h[s:]
            g[s:]    -= h[:n - s]
    else:
        s = -shift
        if s < n:
            g[s:]    -= h[:n - s]
            g[:n - s] -= h[s:]
    return g


def _c5_score(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def _upscale(h: np.ndarray) -> np.ndarray:
    """Linear interpolation to double the number of points."""
    n = h.size
    new_n = n * 2
    # cell centres for old and new grids
    old_x = (np.arange(n) + 0.5) / n * 2.0
    new_x = (np.arange(new_n) + 0.5) / new_n * 2.0
    h_new = np.interp(new_x, old_x, h, left=0.0, right=0.0)
    # enforce box and sum constraints
    return _project_sum_box(h_new, new_n / 2.0)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float)
    else:
        n0 = 80
        h = rng.random(n0)
        h = _project_sum_box(h, n0 / 2.0)

    n = h.size
    target_sum = n / 2.0

    best_h = h.copy()
    best_score = _c5_score(best_h)

    # ----- main optimisation loop ------------------------------------
    step = 0.5                # initial step size for gradient descent
    decay = 0.9995            # geometric decay of step
    no_improve = 0            # counter for stagnation

    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.88:      # leave margin for final work
            break

        # current worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        cur_score = corr.max() * (2.0 / n)

        # gradient step
        g = _gradient(h, shift)
        step = max(step * decay, 1e-6)
        h_candidate = _project_sum_box(h - step * g, target_sum)

        cand_score = _c5_score(h_candidate)

        # keep the best ever seen
        if cand_score < best_score:
            best_score = cand_score
            best_h = h_candidate.copy()
            no_improve = 0
        else:
            no_improve += 1

        # accept if improvement, otherwise stay
        if cand_score <= cur_score:
            h = h_candidate
        else:
            # occasional random kick to escape flat regions
            if no_improve > 2000:
                eps = rng.normal(scale=0.02, size=n)
                h_kick = _project_sum_box(h + eps, target_sum)
                if _c5_score(h_kick) < best_score:
                    best_h = h_kick.copy()
                    best_score = _c5_score(h_kick)
                h = h_kick
                no_improve = 0

        # ----- occasional resolution refinement -----------------------
        # When we have spent a reasonable amount of time and the grid is
        # still modest, double the resolution to give the optimiser more
        # freedom.
        if (elapsed > budget_s * 0.4) and (n < 240) and (no_improve > 5000):
            # upscale once
            h = _upscale(best_h)
            n = h.size
            target_sum = n / 2.0
            step = 0.5
            no_improve = 0
            # recompute best score for the new resolution
            best_score = _c5_score(h)
            best_h = h.copy()
            continue

    # ----- final polishing with simple hill‑climbing ------------------
    # Use the remaining budget for cheap random exchanges.
    while time.time() - start < budget_s * 0.98:
        i, j = rng.choice(n, size=2, replace=False)
        # max possible transfer without breaking bounds
        eps_max = min(best_h[i], 1.0 - best_h[j])
        if eps_max <= 1e-8:

        eps = rng.random() * eps_max
        h_new = best_h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        sc = _c5_score(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # final check – ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_sum_box(best_h, len(best_h) / 2.0)
    best_score = _c5_score(best_h)

    return (best_h.tolist(), float(best_score), len(best_h))
