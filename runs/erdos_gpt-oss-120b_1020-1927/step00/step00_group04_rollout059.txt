def project(h):
    # Project onto [0,1] and sum target = n/2
    n = len(h)
    target = n/2.0
    # Use water-filling algorithm (bisection)
    lower = np.min(h - target) # not exactly; we just choose large range.
    # Bisection extremes:
    lo = -1.0
    hi = 1.0
    def sum_proj(l):
        return np.clip(h - l, 0.0, 1.0).sum()
    # Expand lo/hi
    while sum_proj(lo) < target:
        lo -= 1.0
    while sum_proj(hi) > target:
        hi += 1.0
    for _ in range(50):
        mid = (lo+hi)/2.0
        s = sum_proj(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    return proj

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step function."""
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(conv.max() * dx)


def compute_worst_shift(h: np.ndarray):
    """Return shift (k) attaining the maximal C5 and the corresponding raw value."""
    n = h.size
    conv = np.correlate(h, 1.0 - h, mode='full')
    idx = int(conv.argmax())
    shift = idx - (n - 1)
    return shift, float(conv[idx])


def project_onto_box_sum(v: np.ndarray, target_sum: float,
                         lower: float = 0.0, upper: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of vector v onto the set:
        { x | lower <= x_i <= upper,  sum(x) = target_sum }.
    Implemented by a simple bisection on the Lagrange multiplier.
    """
    if target_sum < lower * v.size or target_sum > upper * v.size:
        raise ValueError("Target sum infeasible for given box.")
    lo = -2.0
    hi = 2.0

    def s(lam):
        return np.clip(v - lam, lower, upper).sum()

    while s(lo) < target_sum:
        lo -= 1.0
    while s(hi) > target_sum:
        hi += 1.0

    for _ in range(60):
        mid = (lo + hi) * 0.5
        cur = s(mid)
        if cur > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, lower, upper)


def random_move(h: np.ndarray, magnitude: float = 0.1) -> np.ndarray:
    """Transfer a small amount of mass between two random indices, preserving the sum."""
    n = h.size
    i, j = np.random.choice(n, size=2, replace=False)
    max_pos = min(h[i], 1.0 - h[j])
    max_neg = -min(h[j], 1.0 - h[i])
    delta = np.random.uniform(-magnitude, magnitude)
    delta = np.clip(delta, max_neg, max_pos)
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new


# ---------- main routine ----------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where:
        h_values  – list of step heights on a uniform grid of [0,2]
        c5_bound  – the computed max overlap integral for this h
        n_points  – number of discretisation points (len(h_values))
    """
    np.random.seed(seed)
    start = time.time()

    # --- initial vector -------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        # fallback: uniform 0.5 on a modest grid
        h0 = np.full(64, 0.5, dtype=np.float64)

    n0 = h0.size
    h0 = project_onto_box_sum(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # --- resolution schedule -------------------------------------------------
    max_points = 800               # keep below 1k, still enough freedom
    stages = [n0]
    cur = n0
    while cur * 2 <= max_points:
        cur *= 2
        stages.append(cur)

    # --- optimisation loop over stages ------------------------------------
    for stage_n in stages:
        # upscale if needed
        if best_h.size != stage_n:
            x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, stage_n, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = project_onto_box_sum(h_up, stage_n / 2.0)
            best_h = h_up
            best_c5 = compute_c5(best_h)

        # time bookkeeping for this stage
        time_left = budget_s - (time.time() - start)
        if time_left < 5.0:
            break

        # a few random restarts before the main descent
        for _ in range(4):
            cand = np.random.rand(stage_n)
            cand = project_onto_box_sum(cand, stage_n / 2.0)
            cand_c5 = compute_c5(cand)
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()

        # main projected sub‑gradient descent
        h_cur = best_h.copy()
        cur_c5 = best_c5
        it = 0
        # rough iteration cap; actual stop governed by budget
        max_iter = 50000 + 20000 * (stage_n // 100)

        while True:
            # time stop
            if time.time() - start > budget_s - 1.0:
                break
            if it >= max_iter:
                break

            shift, _ = compute_worst_shift(h_cur)

            # direction = h_{i+shift} + h_{i-shift}
            direction = np.zeros_like(h_cur)
            if shift >= 0:
                s = shift
                direction[:stage_n - s] += h_cur[s:]
                direction[s:] += h_cur[:stage_n - s]
            else:
                s = -shift
                direction[s:] += h_cur[:stage_n - s]
                direction[:stage_n - s] += h_cur[s:]

            # normalized step
            norm = np.linalg.norm(direction)
            if norm > 0:
                base_step = 0.02
                # halve the step every ~1500 iterations
                step = base_step * (0.5 ** (it // 1500))
                h_new = h_cur + step * direction / norm
            else:
                h_new = h_cur

            h_new = project_onto_box_sum(h_new, stage_n / 2.0)
            new_c5 = compute_c5(h_new)

            # accept if improvement
            if new_c5 < cur_c5 - 1e-12:
                h_cur = h_new
                cur_c5 = new_c5
                if new_c5 < best_c5 - 1e-12:
                    best_c5 = new_c5
                    best_h = h_new.copy()
            else:
                # occasional exploratory move
                if it % 200 == 0:
                    trial = random_move(h_cur, magnitude=0.07)
                    trial_c5 = compute_c5(trial)
                    if trial_c5 < cur_c5:
                        h_cur = trial
                        cur_c5 = trial_c5
                        if trial_c5 < best_c5:
                            best_c5 = trial_c5
                            best_h = trial.copy()

            # occasional jitter around the best known point
            if it % 5000 == 0 and it > 0:
                jitter = np.random.normal(scale=0.02, size=stage_n)
                h_j = best_h + jitter
                h_j = np.clip(h_j, 0.0, 1.0)
                h_j = project_onto_box_sum(h_j, stage_n / 2.0)
                c5_j = compute_c5(h_j)
                if c5_j < best_c5:
                    best_c5 = c5_j
                    best_h = h_j.copy()
                    h_cur = h_j.copy()
                    cur_c5 = c5_j

            it += 1

        # ensure best_h respects the constraints after this stage
        best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
        best_c5 = compute_c5(best_h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, best_h.size)
