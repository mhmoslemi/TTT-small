import time, numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # correlation using np.correlate, returns length 2n-1
    conv = np.correlate(h, 1 - h, mode="full")
    return np.max(conv) * dx

def hill_climb(h, max_no_improve, rng):
    n = len(h)
    dx = 2.0 / n
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    # list of ones/zeros indices
    ones = np.flatnonzero(cur == 1)
    zeros = np.flatnonzero(cur == 0)
    no_improve_counter = 0
    while no_improve_counter < max_no_improve:
        # pick random swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur[i], cur[j] = 0, 1
        new_c5 = compute_c5(cur)
        if new_c5 < cur_c5 - 1e-12:
            cur_c5 = new_c5
            # update indices
            # after swap, i becomes zero, j becomes one
            # update arrays efficiently:
            # Remove i from ones and add i to zeros, remove j from zeros and add j to ones
            # we can just recompute each time for simplicity
            ones = np.flatnonzero(cur == 1)
            zeros = np.flatnonzero(cur == 0)
            no_improve_counter = 0
        else:
            # revert
            cur[i], cur[j] = 1, 0
            no_improve_counter += 1
    return cur, cur_c5

def simulated_annealing(h, rng, max_iters, init_temp, final_temp):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    n = len(h)
    ones = np.flatnonzero(cur == 1)

    for it in range(max_iters):
        # schedule temperature
        t = init_temp + (final_temp - init_temp) * it / max_iters  # linear cooling

import numpy as np, time, math

def compute_c5(vec):
    n = len(vec)
    dx = 2.0 / n
    return np.max(np.correlate(vec, 1 - vec, mode='full')) * dx

def simulated_annealing(n, rng, time_limit, start_vec=None):
    if start_vec is None:
        vec = np.zeros(n, dtype=np.float64)
        vec[:n//2] = 1.0
        rng.shuffle(vec)
    else:
        vec = start_vec.astype(np.float64)  # ensure copy
    cur_c5 = compute_c5(vec)
    best_vec = vec.copy()
    best_c5 = cur_c5

    # parameters
    max_iters = 2000000  # large upper bound, will break on time
    T0 = 0.05
    Tf = 0.0005

    start = time.time()
    for it in range(max_iters):
        # time guard
        if time.time() - start > time_limit:
            break
        # temperature schedule linear
        tau = it / max_iters
        T = T0 * (1 - tau) + Tf * tau

        # propose swap
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j = j + 1
        # ensure opposite bits to preserve sum
        if vec[i] == vec[j]:
            continue
        # swap
        vec[i], vec[j] = vec[j], vec[i]
        new_c5 = compute_c5(vec)
        delta = new_c5 - cur_c5
        if delta <= 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            # accept
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_vec = vec.copy()
        else:
            # revert
            vec[i], vec[j] = vec[j], vec[i]
    return best_vec, best_c5

import time, math
import numpy as np

def _compute_c5(vec: np.ndarray) -> float:
    """Return the C5 value for a step function defined by `vec`."""
    n = vec.shape[0]
    dx = 2.0 / n
    # full cross‑correlation of h with (1‑h)
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return float(np.max(corr) * dx)


def _random_binary(rng: np.random.Generator, n: int) -> np.ndarray:
    """Generate a binary vector of length n with exactly n/2 ones."""
    v = np.zeros(n, dtype=np.float64)
    v[: n // 2] = 1.0
    rng.shuffle(v)
    return v


def _simulated_annealing(rng: np.random.Generator, n: int,
                         time_limit: float, start_vec: np.ndarray = None):
    """Simple SA that preserves the number of ones (swap moves)."""
    if start_vec is None:
        cur = _random_binary(rng, n)
    else:
        cur = start_vec.astype(np.float64).copy()
    cur_c5 = _compute_c5(cur)

    best_vec = cur.copy()
    best_c5 = cur_c5

    # annealing schedule parameters
    max_iters = 400_000          # generous upper bound; will stop on time
    T_start = 0.08
    T_end = 0.0005

    start_time = time.time()
    it = 0
    while it < max_iters:
        if time.time() - start_time > time_limit:
            break

        # linear temperature decay
        tau = it / max_iters
        T = T_start * (1.0 - tau) + T_end * tau

        # propose a swap between a 1 and a 0
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if cur[i] == cur[j]:          # not a genuine swap, skip
            it += 1
            continue

        # perform swap
        cur[i], cur[j] = cur[j], cur[i]
        new_c5 = _compute_c5(cur)
        delta = new_c5 - cur_c5

        # Metropolis acceptance
        if delta <= 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_vec = cur.copy()
        else:
            # revert
            cur[i], cur[j] = cur[j], cur[i]

        it += 1

    return best_vec, best_c5


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Find a step function h on [0,2] with ∫h = 1 that minimizes
    max_k ∫ h(x)(1‑h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    target = 0.3808          # we stop early if we hit this bound

    # Choose the discretisation size – a moderate even number works well.
    n = 200                  # can be changed; must be even
    best_h = None
    best_score = math.inf

    # Helper to see how much time is left (with a small safety margin)
    def time_left():
        return budget_s - (time.time() - start_time)

    # Main loop: many short SA runs until budget expires or target reached.
    while time_left() > 1.0:
        # allocate a slice of the remaining budget for this SA run
        slice_time = min(5.0, time_left() * 0.5)

        # random starting point (binary, half ones)
        start_vec = _random_binary(rng, n)

        # run SA
        vec, score = _simulated_annealing(rng, n, slice_time, start_vec)

        if score < best_score:
            best_score = score
            best_h = vec.copy()
            if best_score <= target:
                break   # good enough – exit early

    # Fallback: if nothing was found (should never happen)
    if best_h is None:
        best_h = _random_binary(rng, n)
        best_score = _compute_c5(best_h)

    # Ensure exact integral constraint (numerical safety)
    # binary vectors already satisfy sum = n/2; enforce via clipping just in case
    best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, float(best_score), n
