def project_capped_simplex(y, S, lower=0.0, upper=1.0, tol=1e-9, max_iter=50):
    lo = -np.max(y) - 1.0  # ensure lower bound
    hi = upper - np.min(y) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2
        x = np.clip(y + lam, lower, upper)
        s = x.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    return x

def _project_capped_simplex(y, target_sum, low=0.0, high=1.0, max_iter=60, tol=1e-12):
    """Project y onto the set {x | low ≤ x ≤ high, sum(x)=target_sum}."""
    # initial bounds for the Lagrange multiplier λ
    lo = low - np.max(y) - 1.0
    hi = high - np.min(y) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(y + lam, low, high)
        s = x.sum()
        if np.abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            hi = lam
        else:
            lo = lam
    return np.clip(y + (lo + hi) * 0.5, low, high)

def _random_jitter(h, rng, target_sum, magnitude=0.02):
    """Add small random perturbation and re‑project onto the feasible set."""
    delta = (rng.random(h.shape) - 0.5) * magnitude
    cand = np.clip(h + delta, 0.0, 1.0)
    return _project_capped_simplex(cand, target_sum)

    dx = 2.0 / h.shape[0]

                best_c5, best_h = cand_c5, cand_h.copy()
            step = min(step * 1.07, 0.5)     # become more daring
        else:
            step = max(step * 0.65, 1e-5)    # become more cautious
            # If we become too timid, shake the current point
            if step < 1e-5:
                cur_h = _random_jitter(cur_h, rng, target_sum, magnitude=0.07)
                cur_c5 = _c5_value(cur_h)
                step = 0.05

        # Periodic mild perturbation around the best known solution
        if it % 4800 == 0 and it > 0:
            cur_h = _random_jitter(best_h, rng, target_sum, magnitude=0.02)
            cur_c5 = _c5_value(cur_h)
            step = 0.05

        it += 1

    # -----------------------------------------------------------------
    # 3. Final polishing: try a pure 0/1 version of the best point.
    # -----------------------------------------------------------------
    S_int = int(round(target_sum))
    top = np.argpartition(best_h, -S_int)[-S_int:]
    binary_h = np.zeros_like(best_h)
    binary_h[top] = 1.0
    binary_c5 = _c5_value(binary_h)
    if binary_c5 < best_c5:
        best_h, best_c5 = binary_h, binary_c5

    # -----------------------------------------------------------------
    # 4. Return the best construction discovered.
    # -----------------------------------------------------------------
    return (best_h, float(best_c5), n)
