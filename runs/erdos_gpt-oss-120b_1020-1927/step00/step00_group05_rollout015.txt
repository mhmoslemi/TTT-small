import numpy as np
import time
import random

def _project_to_sum(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of vec onto the set {x in [0,1]^n : sum(x)=target_sum}."""
    low, high = -2.0, 2.0
    for _ in range(60):
        lam = (low + high) * 0.5
        proj = np.clip(vec - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            low = lam
        else:
            high = lam
    lam = (low + high) * 0.5
    return np.clip(vec - lam, 0.0, 1.0)


def _compute_c5(vec: np.ndarray) -> float:
    """C5 for a discrete step function given by vec."""
    n = vec.shape[0]
    dx = 2.0 / n
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return float((corr * dx).max())


def _grad_overlap(vec: np.ndarray, shift: int) -> np.ndarray:
    """
    Gradient of the overlap for a given shift.
    Overlap(shift) = sum_i h_i * (1 - h_{i+shift})   (zero outside array)
    d/ dh_j = (1 - h_{j+shift})  (if j+shift in range)  - h_{j-shift} (if j-shift in range)
    """
    n = vec.shape[0]
    grad = np.zeros_like(vec)
    if shift >= 0:
        # term (1 - h_{i+shift}) for i = 0 .. n-1-shift
        grad[: n - shift] += 1.0 - vec[shift:]
        # term -h_{i-shift} for i = shift .. n-1
        grad[shift:] -= vec[: n - shift]
    else:
        s = -shift
        # term (1 - h_{i-s}) for i = s .. n-1
        grad[s:] += 1.0 - vec[: n - s]
        # term -h_{i+s} for i = 0 .. n-s-1
        grad[: n - s] -= vec[s:]
    return grad
