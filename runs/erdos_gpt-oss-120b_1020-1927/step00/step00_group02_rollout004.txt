def gradient_max_lag(h, lag, dx):
    n = len(h)
    grad = np.zeros_like(h)
    # term (1 - h[i+lag]) for valid i+lag
    if lag >= 0:
        grad[:n-lag] += (1 - h[lag:])
        # -h[i-lag] for i>=lag, i-lag valid range
        grad[lag:] -= h[:n-lag]
    else:
        # lag negative, i+lag = i - |lag|
        lag_abs = -lag
        grad[lag_abs:] += (1 - h[:n-lag_abs])
        grad[:n-lag_abs] -= h[lag_abs:]
    return grad * dx

import math, time
import numpy as np

# ------------------------------------------------------------
# Helper functions

def project_onto_box_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {0<=x_i<=1, sum(x)=target_sum}."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny fix for rounding errors
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size:
            i = free[0]
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x


def compute_c5(h: np.ndarray) -> float:
    """Exact C5 value for a discrete step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def gradient_max_lag(h: np.ndarray, lag: int, dx: float) -> np.ndarray:
    """Gradient of overlap for a single lag (the one attaining the maximum)."""
    n = h.size
    grad = np.zeros_like(h)
    if lag >= 0:
        grad[: n - lag] += (1.0 - h[lag:])
        grad[lag:] -= h[: n - lag]
    else:
        a = -lag
        grad[a:] += (1.0 - h[: n - a])
        grad[: n - a] -= h[a:]
    return grad * dx


def subgradient_and_value(h: np.ndarray):
    """Subgradient (average over all maximal lags) and the C5 value."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    # indices where the maximum is reached (tolerance for floating errors)
    idxs = np.where(np.abs(corr - max_val) < 1e-12)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        lag = idx - (n - 1)
        grad += gradient_max_lag(h, lag, dx)
    grad /= max(idxs.size, 1)
    return grad, max_val * dx


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


# ------------------------------------------------------------
# Core optimizer for a fixed resolution
# ------------------------------------------------------------
def optimize_at_resolution(h0: np.ndarray, time_budget: float, rng: np.random.Generator):
    n = h0.size
    target_sum = n / 2.0
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)

    # ----- phase 1 : projected gradient descent (≈50% of budget) -----
    phase1_end = start + 0.5 * time_budget
    while time.time() < phase1_end:
        grad, cur_val = subgradient_and_value(h)
        if cur_val < best_val:
            best_val, best_h = cur_val, h.copy()
        step = 0.2
        improved = False
        while step > 1e-5:
            h_new = project_onto_box_sum(h - step * grad, target_sum)
            new_val = compute_c5(h_new)
            if new_val < cur_val - 1e-14:
                h, improved = h_new, True
                break
            step *= 0.5
        if not improved:
            # small random shake to escape flat regions
            delta = (rng.random(n) - 0.5) * 0.02
            h = project_onto_box_sum(h + delta, target_sum)

    # ----- phase 2 : pair‑wise hill climbing / simulated annealing -----
    while time.time() - start < time_budget:
        # direction from current subgradient
        grad, cur_val = subgradient_and_value(h)

        pos = np.where(grad > 1e-9)[0]  # decreasing these helps
        neg = np.where(grad < -1e-9)[0]  # increasing these helps
        if pos.size and neg.size:
            i = rng.choice(pos)
            j = rng.choice(neg)
            max_step = min(h[i], 1.0 - h[j], 0.05)
            if max_step > 1e-12:
                delta = rng.random() * max_step
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
            else:
                delta = (rng.random(n) - 0.5) * 0.02
                h_new = project_onto_box_sum(h + delta, target_sum)
        else:
            delta = (rng.random(n) - 0.5) * 0.02
            h_new = project_onto_box_sum(h + delta, target_sum)

        new_val = compute_c5(h_new)
        if new_val < best_val - 1e-14:
            best_val, best_h, h = new_val, h_new.copy(), h_new
            continue

        # occasional Metropolis acceptance
        remaining = max(time_budget - (time.time() - start), 1e-12)
        T = max(1e-4, 0.02 * remaining / time_budget)
        if rng.random() < math.exp(-(new_val - best_val) / T):
            h = h_new

    # final clean‑up
    best_h = project_onto_box_sum(best_h, target_sum)
    best_val = compute_c5(best_h)
    return best_h, best_val


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_global = time.time()

    # ----------------------------------------------------------------
    # Prepare an initial feasible vector
    # ----------------------------------------------------------------
    try:
        h_initial = np.array(initial_h_values, dtype=float)  # type: ignore
        n_initial = h_initial.size
        target0 = n_initial / 2.0
        if not np.isclose(h_initial.sum(), target0, atol=1e-9):
            h_initial = project_onto_box_sum(h_initial, target0)
    except Exception:
        # fallback when no global variable is provided
        n_initial = 45
        h_initial = random_feasible_vector(n_initial, rng)

    best_h, best_val, best_n = None, float("inf"), None

    # resolutions we will try (keep total points ≤ 1000)
    resolutions = [45, 90, 180, 360]
    for idx, n_target in enumerate(resolutions):
        if time.time() - start_global > budget_s * 0.97:
            break

        # up‑sample / interpolate to the new resolution
        if n_target != h_initial.size:
            x_old = np.linspace(0.0, 2.0, h_initial.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
            h_up = np.interp(x_new, x_old, h_initial)
            h_up = project_onto_box_sum(h_up, n_target / 2.0)
        else:
            h_up = h_initial.copy()

        # allocate time for this level (remaining evenly among the rest)
        remaining_levels = len(resolutions) - idx
        time_left = max(0.0, budget_s - (time.time() - start_global))
        level_budget = time_left / remaining_levels

        # optimise at this resolution
        h_opt, val_opt = optimize_at_resolution(h_up, level_budget, rng)

        if val_opt < best_val:
            best_h, best_val, best_n = h_opt.copy(), val_opt, n_target

        # use this solution as seed for the next finer grid
        h_initial = h_opt.copy()

    # final sanity projection
    if best_h is None:
        # should never happen, but fallback
        best_n = n_initial
        best_h = project_onto_box_sum(h_initial, best_n / 2.0)
        best_val = compute_c5(best_h)

    best_h = project_onto_box_sum(best_h, best_n / 2.0)
    best_val = compute_c5(best_h)

    # return list for compatibility with the harness
    return best_h.tolist(), float(best_val), int(best_n)
