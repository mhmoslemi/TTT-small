def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    from scipy.optimize import minimize, Bounds
    rng = np.random.default_rng(seed)
    start = time.time()
    # get initial h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=np.float64)
    else:
        n = 47
        target = n/2
        h_init = np.zeros(n, dtype=np.float64)
        num_one = int(np.floor(target))
        h_init[:num_one] = 1.0
        if target - num_one > 1e-12:
            h_init[num_one] = target - num_one
        rng.shuffle(h_init)
    n = len(h_init)
    target = n/2.0
    dx = 2.0 / n

    # function to compute C5
    def compute_c5(h):
        # guarantee 1-h
        # Use np.correlate for speed
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    # evaluate initial
    best_h = np.copy(h_init)
    best_val = compute_c5(best_h)

    # option: ensure equality constraint exactly (maybe adjust)
    # but compute_c5 may not need exact sum but we keep it.

    # define constraints for SLSQP
    # equality constraint: sum(h) - target = 0
    eq_cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target}
    # bounds
    bounds = Bounds(0.0, 1.0)

    # try local optimizer on best_h
    def attempt_optimize(h_start):
        nonlocal best_h, best_val, start
        if time.time() - start > budget_s * 0.95:
            return
        # call minimize
        try:
            res = minimize(compute_c5, h_start, method='SLSQP',
                           bounds=bounds, constraints=[eq_cons],
                           options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
        except Exception as e:
            return
        if not res.success:
            # maybe ignore
            h_opt = res.x
        else:
            h_opt = res.x
        # clip due to numeric issues
        h_opt = np.clip(h_opt, 0.0, 1.0)
        # enforce sum exactly by simple adjustment
        cur_sum = np.sum(h_opt)
        diff = target - cur_sum
        if abs(diff) > 1e-9:
            # find index not at bounds
            # attempt to distribute diff to first index not at bound
            # if diff >0 need to increase some entry <1, else reduce some >0.
            idxs = np.where((diff > 0) & (h_opt < 1-1e-12) |
                            (diff < 0) & (h_opt > 1e-12))[0]
            if len(idxs) == 0:
                # fallback: distribute to all
                h_opt += diff / n
                h_opt = np.clip(h_opt, 0.0, 1.0)
            else:
                # distribute diff equally onto those entries
                per = diff / len(idxs)
                for i in idxs:
                    h_opt[i] += per
                # reclip
                h_opt = np.clip(h_opt, 0.0, 1.0)
                # recompute sum diff again possibly leftover; can iterate once
                cur_sum = np.sum(h_opt)
                diff = target - cur_sum
                if abs(diff) > 1e-9:
                    # distribute residual to first non-bound entry
                    i = idxs[0]
                    h_opt[i] += diff
                    h_opt = np.clip(h_opt, 0.0, 1.0)
        val = compute_c5(h_opt)
        if val < best_val - 1e-12:
            best_h = h_opt
            best_val = val

    # first attempt with initial
    attempt_optimize(best_h)

    # number of random restarts based on time budget
    # For small n, we can afford many restarts. Let's approximate: each attempt ~0.01-0.02 seconds? Not sure.
    # We'll loop while time left > some small threshold.
    # We'll generate random feasible vectors and call attempt_optimize.
    # We'll limit maximum restarts maybe 200.
    max_restarts = 200
    for _ in range(max_restarts):
        elapsed = time.time() - start
        if elapsed > budget_s * 0.9:
            break
        # generate random binary pattern with exactly target sum.
        # We'll create vector random permutations of ones and zeros.
        # The number of 1's needed = floor(target). leftover fractional part = target - floor(target).
        num_one = int(np.floor(target))
        h_rand = np.zeros(n)
        h_rand[:num_one] = 1.0
        frac = target - num_one
        if frac > 1e-12:
            h_rand[num_one] = frac
        rng.shuffle(h_rand)
        # add small random noise to avoid discrete stuck
        # noise amplitude = 0.05
        noise = (rng.random(n) - 0.5) * 0.1  # +/-0.05
        h_rand = np.clip(h_rand + noise, 0.0, 1.0)
        # Rebalance sum to target again: compute diff and adjust first element not at bound
        cur_sum = np.sum(h_rand)
        diff = target - cur_sum
        # adjust first element (or find one)
        if abs(diff) > 1e-12:
            # try to add diff to a random entry that can accept it
            idxs = np.where(((diff > 0) & (h_rand < 1-1e-12)) |
                            ((diff < 0) & (h_rand > 1e-12)))[0]
            if len(idxs) > 0:
                i = rng.choice(idxs)
                h_rand[i] = np.clip(h_rand[i] + diff, 0.0, 1.0)
            else:
                # distribute evenly
                h_rand += diff / n
                h_rand = np.clip(h_rand, 0.0, 1.0)
        attempt_optimize(h_rand)

    # optionally run hill climbing on best so far for a while
    # simple mass-transfer random moves
    h_cur = best_h.copy()
    cur_val = best_val
    # temperature schedule
    init_T = 0.005
    while time.time() - start < budget_s * 0.95:
        elapsed = time.time() - start
        T = init_T * (1 - elapsed / budget_s)
        # pick two indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # max transfer from i to j
        max_transfer = min(h_cur[i], 1.0 - h_cur[j])
        if max_transfer <= 0:
            continue
        # choose delta (maybe random proportion)
        delta = max_transfer * rng.random()
        # maybe small step factor: multiply by 0.5
        # delta = max_transfer * rng.random() * 0.5
        h_new = h_cur.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_val = compute_c5(h_new)
        if new_val < cur_val:
            h_cur = h_new
            cur_val = new_val
            if new_val < best_val:
                best_h = h_new.copy()
                best_val = new_val
        else:
            # accept with probability exp(-(new - cur)/T)
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if rng.random() < prob:
                    h_cur = h_new
                    cur_val = new_val

    # final best
    # ensure final h satisfies constraints exactly
    # clip
    best_h = np.clip(best_h, 0.0, 1.0)
    # we can scale to target sum iteratively
    cur_sum = np.sum(best_h)
    diff = target - cur_sum
    if abs(diff) > 1e-9:
        # find index where we can add/subtract
        if diff > 0:
            idxs = np.where(best_h < 1-1e-12)[0]
        else:
            idxs = np.where(best_h > 1e-12)[0]
        if len(idxs) > 0:
            # distribute equally across them
            per = diff / len(idxs)
            for i in idxs:
                best_h[i] = np.clip(best_h[i] + per, 0.0, 1.0)
        else:
            # fallback uniform scaling
            best_h += diff / n
        # clip again
        best_h = np.clip(best_h, 0.0, 1.0)

    # final val
    final_val = compute_c5(best_h)
    # decide final return (h, c5, n)
    return best_h, final_val, n

import time, math
import numpy as np
from scipy.optimize import minimize, Bounds

def _c5(h: np.ndarray) -> float:
    """C5 for a discrete step function."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_to_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to hit the exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # indices that can be increased / decreased
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if idx.size == 0:
        # fallback uniform adjustment
        h += diff / h.size
        return np.clip(h, 0.0, 1.0)
    per = diff / idx.size
    h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    # Second‑pass for remaining rounding error
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        for _ in range(10):
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                cand = np.where(h < 1.0 - 1e-12)[0]
                inc = min(diff, 1.0 - h[cand[0]]) if cand.size else 0.0
                h[cand[0]] += inc
                diff -= inc
            else:
                cand = np.where(h > 1e-12)[0]
                dec = max(diff, -h[cand[0]]) if cand.size else 0.0
                h[cand[0]] += dec
                diff -= dec
    return np.clip(h, 0.0, 1.0)
