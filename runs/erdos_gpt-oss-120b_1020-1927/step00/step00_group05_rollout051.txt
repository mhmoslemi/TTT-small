import time, random, math
import numpy as np

def _compute_c5(vec, dx):
    """Maximum overlap C5 for a given step vector."""
    # correlation of h with (1-h)
    return np.max(np.correlate(vec, 1.0 - vec, mode='full')) * dx

def _initial_binary(vals, n):
    """Convert a real vector to a feasible (binary/half) vector with sum=n/2."""
    idx = np.argsort(-vals)               # descending order
    h = np.zeros(n, dtype=float)
    half_int = n // 2
    h[idx[:half_int]] = 1.0
    if n % 2:
        h[idx[half_int]] = 0.5
    return h

def _project_capped_simplex(v, target):
    """Project onto {0<=x<=1, sum(x)=target} using a simple bisection on λ."""
    lo, hi = -2.0, 2.0                     # wide enough bounds
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # ----------------------------------------------------------------------
    # 1. Problem size and initial vector
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 70
        h0 = np.random.rand(n)

    dx = 2.0 / n
    target_sum = n / 2.0

    # initial feasible vector (binary, possibly one 0.5 when n is odd)
    cur_h = _initial_binary(h0, n)
    cur_c5 = _compute_c5(cur_h, dx)

    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ----------------------------------------------------------------------
    # 2. Helper for C5
    # ----------------------------------------------------------------------
    def c5_of(v):
        return _compute_c5(v, dx)

    # ----------------------------------------------------------------------
    # 3. Simulated‑annealing hill climbing (many restarts)
    # ----------------------------------------------------------------------
    max_iters_per_restart = 200_000
    init_temp = 0.02

    while True:
        # time guard
        if time.time() - start > budget_s * 0.95:
            break
        # start this restart from the current global best (helps convergence)
        cur_h = best_h.copy()
        cur_c5 = best_c5

        for it in range(max_iters_per_restart):
            # occasional time check (every 5000 iters)
            if (it & 0xFFF) == 0 and (time.time() - start > budget_s * 0.95):
                break

            # pick two different indices with different values
            i = random.randrange(n)
            j = random.randrange(n - 1)
            if j >= i:
                j += 1
            if cur_h[i] == cur_h[j]:
                continue

            # propose swap
            new_h = cur_h.copy()
            new_h[i], new_h[j] = cur_h[j], cur_h[i]
            new_c5 = c5_of(new_h)

            delta = new_c5 - cur_c5
            temp = init_temp * (1.0 - it / max_iters_per_restart)

            # Metropolis acceptance
            if delta < 0.0 or (temp > 0.0 and random.random() < math.exp(-delta / temp)):
                cur_h, cur_c5 = new_h, new_c5
                if new_c5 < best_c5:
                    best_c5, best_h = new_c5, new_h.copy()

        # early stop if we already beat the target
        if best_c5 < 0.3808:
            break
