def project_onto_box_sum(x, target, lo=0.0, hi=1.0):

    # Binary search λ to satisfy sum clip(x - λ, lo, hi) = target.

    # λ range can be from min(x) - hi to max(x) - lo.

    lo_lambda = min(x) - hi
    hi_lambda = max(x) - lo

    for _ in range(40):
        lam = (lo_lambda + hi_lambda) /2
        y = np.clip(x - lam, lo, hi)
        s = y.sum()
        if s > target:
            lo_lambda = lam
        else:
            hi_lambda = lam
    return y

    for _ in range(max_iter):
        lam = (low + high) / 2.0
        y = np.clip(x - lam, lo, hi)
        s = y.sum()
        if s > target:
            low = lam
        else:
            high = lam
    return y

def gradient_of_shift(h, shift):
    n = len(h)
    grad = np.zeros_like(h)
    if shift >= 0:
        i = np.arange(n - shift)
        grad[i] = 1 - h[i + shift]
        grad[i + shift] -= h[i]
    else:
        s = -shift
        i = np.arange(s, n)
        grad[i] = 1 - h[i - s]
        grad[i - s] -= h[i]
    return grad

def subgradient_descent(initial_h, max_iters, target, time_limit):
    h = initial_h.copy()
    best_h = h.copy()
    best_val = evaluate(h)
    n = len(h)
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        # compute correlation vector
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        grad = gradient_of_shift(h, shift)
        # step size schedule
        lr = 0.5 / (1 + it * 0.001)  # decreasing
        new_raw = h - lr * grad
        # project
        h = project_onto_box_sum(new_raw, target)
        val = np.max(np.correlate(h, 1 - h, mode='full') * dx)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        # optionally occasional random restart if stagnation
    return best_h, best_val

import numpy as np, time

def _eval_c5(h):
    n = h.size
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c.max()


def _proj_box_sum(x, target, lo=0.0, hi=1.0, it=50):
    """Project x onto { lo<=y<=hi , sum(y)=target } via bisection on a shift."""
    low = np.min(x) - hi
    high = np.max(x) - lo
    for _ in range(it):
        lam = (low + high) * 0.5
        y = np.clip(x - lam, lo, hi)
        s = y.sum()
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(x - (low + high) * 0.5, lo, hi)


def _grad_shift(h, shift):
    """Gradient of C(shift)=∑ h[i]*(1-h[i+shift]) wrt h."""
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        i = np.arange(n - shift)
        g[i] = 1.0 - h[i + shift]
        g[i + shift] -= h[i]
    else:
        s = -shift
        i = np.arange(s, n)
        g[i] = 1.0 - h[i - s]
        g[i - s] -= h[i]
    return g


def _subgrad_opt(h0, max_it, target, budget):
    """Projected sub‑gradient descent on max C(·)."""
    h = h0.copy()
    n = h.size
    best_h, best_val = h.copy(), _eval_c5(h)
    start = time.time()
    for it in range(max_it):
        if time.time() - start > budget:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        grad = _grad_shift(h, shift)
        lr = 0.8 / (1 + it * 0.001)          # slowly decreasing step
        h_candidate = _proj_box_sum(h - lr * grad, target)
        val = _eval_c5(h_candidate)
        if val < best_val - 1e-9:
            best_val, best_h = val, h_candidate.copy()
            # occasional aggressive step when improvement is big
            if val < 0.39:
                # tighten further with smaller learning rate
                for _ in range(3):
                    grad = _grad_shift(best_h, shift)
                    best_h = _proj_box_sum(best_h - lr * 0.5 * grad, target)
                    best_val = _eval_c5(best_h)
        h = h_candidate
    return best_h, best_val


def _make_linear(n, a):
    """Symmetric linear shape with endpoint a and centre b=1−a."""
    dx = 2.0 / n
    x = np.arange(n) * dx
    b = 1.0 - a
    h = np.where(
        x <= 1.0,
        a + (b - a) * x,
        b + (a - b) * (x - 1.0),
    )
    # enforce sum exactly
    target = n / 2.0
    scale = target / h.sum()
    h = np.clip(h * scale, 0.0, 1.0)
    # final projection (tiny adjustments)
    h = _proj_box_sum(h, target)
    return h


def _upscale(h, new_n):
    old_n = h.size
    dx_old = 2.0 / old_n
    dx_new = 2.0 / new_n
    old_x = (np.arange(old_n) + 0.5) * dx_old
    new_x = (np.arange(new_n) + 0.5) * dx_new
    h_new = np.interp(new_x, old_x, h)
    target = new_n / 2.0
    h_new = _proj_box_sum(h_new, target)
    return h_new


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_total = time.time()
    # main parameters
    n_small = 200                # quick optimisation size
    target_small = n_small / 2.0
    # start from a linear shape with a≈0.9 (empirically good)
    a0 = 0.9
    h0 = _make_linear(n_small, a0)

    # sub‑gradient optimisation on the small grid
    time_rem = budget_s - (time.time() - start_total) - 200
    if time_rem < 0:
        time_rem = 0
    best_h, best_val = _subgrad_opt(h0, max_it=30000, target=target_small,
                                    budget=time_rem * 0.6)

    # if we already beat the target, return immediately
    if best_val <= 0.3808:
        return best_h, float(best_val), n_small

    # upscale to a finer grid and polish a little more
    n_big = 400
    target_big = n_big / 2.0
    h_big = _upscale(best_h, n_big)

    # a few more refinement steps on the finer grid
    time_left = budget_s - (time.time() - start_total)
    if time_left > 0:
        h_big, best_val = _subgrad_opt(h_big, max_it=15000,
                                       target=target_big,
                                       budget=time_left * 0.9)

    # final safety projection (enforce sum exactly)
    h_big = _proj_box_sum(h_big, target_big)

    # evaluate final bound
    final_c5 = _eval_c5(h_big)

    return h_big, float(final_c5), n_big
