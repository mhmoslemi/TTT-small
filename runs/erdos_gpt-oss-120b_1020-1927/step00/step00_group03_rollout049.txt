def project_to_sum_box(x, target):
    # ensure we get y = clip(x - lam, 0, 1) s.t. sum(y) = target
    # Use bisection
    lo = -1e6
    hi = 1e6
    for _ in range(60):
        lam = (lo + hi) / 2.0
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    y = np.clip(x - (lo+hi)/2.0, 0.0, 1.0)
    # If sum not exactly target due to rounding, adjust small residual
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        # adjust a non-boundary element by diff if possible, else ignore
        # find indices where 0<y<1
        idx = np.where((y > 1e-12) & (y < 1-1e-12))[0]
        if idx.size > 0:
            i = idx[0]
            y[i] = np.clip(y[i] + diff, 0.0, 1.0)
    return y

import numpy as np, random, time, math

def compute_c5(h):
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(conv.max() * (2.0 / n))

def project_to_sum_box(x, target):
    # Bisection
    lo = -1e6
    hi = 1e6
    for _ in range(60):
        lam = 0.5 * (lo + hi)
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    # adjust residual
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        idx = np.where((y > 0.0) & (y < 1.0))[0]
        if idx.size > 0:
            i = idx[0]
            y[i] = np.clip(y[i] + diff, 0.0, 1.0)
    return y

def construct_shape(a, n):
    dx = 2.0 / n
    x = np.arange(n) * dx + dx / 2.0
    ramp_len = max(0.0, 2.0 - 2.0 * a)  # avoid negative due to a>1
    if ramp_len < 1e-12:
        # step function
        h = np.where(x <= a, 1.0, 0.0)
    else:
        # compute ramp region
        # h=1 for x <= a; h=0 for x >= 2-a; else linear
        h = np.where(x <= a, 1.0,
                     np.where(x >= 2 - a, 0.0,
                              1.0 - (x - a) / ramp_len))
    return h

    m = old_h.shape[0]

import numpy as np
import random
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * (2.0 / n))


def project_to_sum_box(x: np.ndarray, target: float) -> np.ndarray:
    """
    Project x onto the intersection of the box [0,1]^n and the hyperplane sum(x)=target.
    Uses bisection on the Lagrange multiplier for the simple clipping formulation.
    """
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = 0.5 * (lo + hi)
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

    # Tiny residual correction (optional, keeps sum exactly target)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        interior = np.where((y > 0.0) & (y < 1.0))[0]
        if interior.size:
            i = interior[0]
            y[i] = np.clip(y[i] + diff, 0.0, 1.0)
    return y


def construct_shape(a: float, n: int) -> np.ndarray:
    """
    Monotone shape: plateau of height 1 on [0,a],
    linear descent to 0 on [a,2-a],
    zero thereafter.  The integral is exactly 1 for any a∈[0,1].
    """
    dx = 2.0 / n
    x = np.arange(n) * dx + dx / 2.0
    ramp_len = max(0.0, 2.0 - 2.0 * a)
    if ramp_len < 1e-12:               # step function (a≈1)
        h = np.where(x <= a, 1.0, 0.0)
    else:
        h = np.where(
            x <= a,
            1.0,
            np.where(
                x >= 2.0 - a,
                0.0,
                1.0 - (x - a) / ramp_len,
            ),
        )
    return h.astype(float)


def resample_h(old_h: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a pre‑existing vector onto a new grid."""
    m = old_h.shape[0]
    dx_old = 2.0 / m
    dx_new = 2.0 / n
    old_x = np.arange(m) * dx_old + dx_old / 2.0
    new_x = np.arange(n) * dx_new + dx_new / 2.0
    new_h = np.interp(new_x, old_x, old_h)
    return project_to_sum_box(new_h, n / 2.0)


def hill_climb(start_h: np.ndarray, end_time: float) -> (np.ndarray, float):
    """
    Simulated‑annealing style local search.
    Moves preserve the total sum by transferring a small amount δ from one index to another.
    """
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    T = 0.02               # initial temperature
    cooling = 0.99995
    target = n / 2.0

    while time.time() < end_time:
        i = random.randrange(n)
        j = random.randrange(n - 1)
        if j >= i:
            j += 1

        max_inc = 1.0 - cur_h[i]
        max_dec = cur_h[j]
        max_delta = min(max_inc, max_dec)
        if max_delta <= 1e-12:
            continue

        # temperature‑scaled step size
        scale = max(T * 30.0, 0.001)
        if scale > 1.0:
            scale = 1.0
        delta = random.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # apply tentative change
        cur_h[i] += delta
        cur_h[j] -= delta
        new_c5 = compute_c5(cur_h)

        accept = False
        if new_c5 < cur_c5 - 1e-12:
            accept = True
        else:
            prob = math.exp(-(new_c5 - cur_c5) / max(T, 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] -= delta
            cur_h[j] += delta

        T = max(T * cooling, 1e-7)

    # final projection to guarantee feasibility
    best_h = project_to_sum_box(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

    for a in np.linspace(0.0, 1.0, 151):

        best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # 4.  Local optimisation (simulated annealing / hill climbing)
    # ------------------------------------------------------------------
    start = time.time()
    end = start + budget_s - 0.2   # tiny safety margin
    best_h, best_c5 = hill_climb(best_h, end)

    # Ensure constraints are satisfied one final time
