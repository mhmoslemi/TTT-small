def project_box_sum(v, S):
    low = -max(v) - 1
    high = max(v) + 1
    for _ in range(50):
        mid = (low + high) / 2
        proj = np.clip(v - mid, 0, 1)
        s = proj.sum()
        if s > S:
            low = mid  # need increase lambda to reduce sum
        else:
            high = mid
    return np.clip(v - (low+high)/2, 0, 1)

def project_box_sum(v, S, max_iter=100, tol=1e-9):
    # Find λ such that sum(clipped(v - λ)) = S
    # because sum is monotonic decreasing in λ.
    lo = -np.max(v) - 1.0  # ensures v - lo > 1 -> all clipped to 1
    hi = np.max(v) + 1.0   # ensures v - hi < 0 -> all clipped to 0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - S) <= tol:
            return proj
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - ((lo+hi)/2.0), 0.0, 1.0)

def shift(arr, s):
    n = arr.shape[0]
    if s == 0:
        return arr.copy()
    if s > 0:
        out = np.empty_like(arr)
        out[:s] = 0.0
        out[s:] = arr[:n-s]
        return out
    else:
        s = -s
        out = np.empty_like(arr)
        out[-s:] = 0.0
        out[:-s] = arr[s:]
        return out

def project_box_sum(v, S, max_iter=50):
    # v: array float, project onto [0,1] and sum S
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if np.abs(s - S) <= 1e-12:
            return proj
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - ((lo+hi)/2.0), 0.0, 1.0)

def c5_of(h):
    n = h.shape[0]

import time
import numpy as np

def _shift(arr: np.ndarray, s: int) -> np.ndarray:
    """Shift arr by s positions with zero padding (right shift for s>0)."""
    n = arr.shape[0]
    if s == 0:
        return arr.copy()
    if s > 0:
        out = np.empty_like(arr)
        out[:s] = 0.0
        out[s:] = arr[: n - s]
        return out
    # s < 0  (left shift)
    s = -s
    out = np.empty_like(arr)
    out[-s:] = 0.0
    out[: n - s] = arr[s:]
    return out


def _project_box_sum(v: np.ndarray, target_sum: float, iters: int = 50) -> np.ndarray:
    """Euclidean projection of v onto {0≤x≤1, sum x = target_sum}."""
    lo = -np.max(v) - 1.0          # forces all entries to 1 after clipping
    hi = np.max(v) + 1.0           # forces all entries to 0 after clipping
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target_sum) <= 1e-12:
            return proj
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - ((lo + hi) * 0.5), 0.0, 1.0)


def _c5(h: np.ndarray) -> float:
    """C5 value for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function giving a low C₅ bound."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # grid size –  even for simplicity
    n = 200
    dx = 2.0 / n
    target_sum = n / 2.0

    best_h = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1) Gradient ascent on the minimal auto‑correlation (continuous h)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        try:
            init = np.asarray(initial_h_values, dtype=float)
            if init.size != n:
                x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                init = np.interp(x_new, x_old, init)
            cur = _project_box_sum(init, target_sum)
            cur_val = _c5(cur)
            if cur_val < best_val:
                best_val, best_h = cur_val, cur.copy()

            for it in range(2000):
                # worst shift = one giving maximal overlap (i.e. minimal correlation)
                corr = np.correlate(cur, cur, mode="full")
                idx = np.argmin(corr)
                shift_k = idx - (n - 1)

                # direction that would increase correlation for this shift
                direction = _shift(cur, shift_k) + _shift(cur, -shift_k)
                step = 0.5 / np.sqrt(it + 1.0)
                cand = _project_box_sum(cur + step * direction * dx, target_sum)

                cand_val = _c5(cand)
                if cand_val < cur_val:
                    cur, cur_val = cand, cand_val
                    if cur_val < best_val:
                        best_val, best_h = cur_val, cur.copy()

                if time.time() - start > budget_s * 0.65:
                    break
        except Exception:
            pass

    # ---------------------------------------------------------------
    # 2) Binary hill‑climbing (targeted swaps) for the remaining time
    # ---------------------------------------------------------------
    while time.time() - start < budget_s * 0.95:
        # random feasible binary start
        cur = np.zeros(n, dtype=float)
        ones = rng.choice(n, size=int(target_sum), replace=False)
        cur[ones] = 1.0
        cur_val = _c5(cur)

        no_improve = 0
        # local improvement loop
        while True:
            # find the shift that currently gives the largest overlap
            corr_full = np.correlate(cur, 1.0 - cur, mode="full")
            idx = np.argmax(corr_full)
            max_overlap = corr_full[idx] * dx
            shift_k = idx - (n - 1)

            # positions that actually contribute to this overlap
            shifted = _shift(cur, shift_k)
            bad = np.where((cur == 1) & (shifted == 0))[0]   # mismatch ones
            good = np.where((cur == 0) & (shifted == 1))[0]  # zeroes that could be turned on

            improved = False
            if bad.size and good.size:
                i_bad = rng.choice(bad)
                i_good = rng.choice(good)
                cur[i_bad] = 0.0
                cur[i_good] = 1.0
                new_val = _c5(cur)
                if new_val < cur_val - 1e-12:
                    cur_val = new_val
                    improved = True
                else:
                    # revert
                    cur[i_bad] = 1.0
                    cur[i_good] = 0.0

            if not improved:
                # fall back to a random swap
                i_one = rng.choice(np.flatnonzero(cur == 1))
                i_zero = rng.choice(np.flatnonzero(cur == 0))
                cur[i_one] = 0.0
                cur[i_zero] = 1.0
                new_val = _c5(cur)
                if new_val < cur_val - 1e-12:
                    cur_val = new_val
                    improved = True
                else:
                    cur[i_one] = 1.0
                    cur[i_zero] = 0.0

            if improved:
                no_improve = 0
                if cur_val < best_val:
                    best_val, best_h = cur_val, cur.copy()
            else:
                no_improve += 1

            # termination of this local search
            if no_improve >= 5 * n:
                break
            if time.time() - start > budget_s * 0.95:
                break

    # fallback if nothing was found (should never happen)
    if best_h is None:
        best_h = np.full(n, 0.5, dtype=float)
        best_val = _c5(best_h)

    return best_h, float(best_val), n
