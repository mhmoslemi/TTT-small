def hill_climb(h0, max_iters, step_initial=0.2, step_final=0.001, time_limit=None):
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    n = len(h0)
    S = n/2
    t0 = time.time()
    for i in range(max_iters):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        step = step_initial + (step_final - step_initial) * (i / max_iters)
        # propose move
        i1, i2 = np.random.randint(0, n, size=2)
        while i1 == i2:
            i2 = np.random.randint(0, n)
        # compute allowed delta range
        lower = max(-cur_h[i1], -(1 - cur_h[i2]), -step)
        upper = min(1 - cur_h[i1], cur_h[i2], step)
        if lower >= upper:
            # try another pair or skip
            continue
        delta = np.random.uniform(lower, upper)
        new_h = cur_h.copy()
        new_h[i1] += delta
        new_h[i2] -= delta
        # sum remains unchanged, bounds satisfied.
        new_val = compute_c5(new_h)
        if new_val < cur_val - 1e-12:
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_h = new_h.copy()
                best_val = new_val
    return best_h, best_val

def simulated_annealing(initial_h, time_limit):
    best_h = initial_h.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val

    cooling = 0.9995  # each iteration multiply
    iterations = 0
    while time.time() - t0 < time_limit:
        # propose move
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # step bound
        step = 0.5  # initial step size
        # Choose delta as random uniform in allowed range [-step, step] w.r.t bounds
        lower = max(-cur_h[i], -(1 - cur_h[j]), -step)
        upper = min(1 - cur_h[i], cur_h[j], step)
        if lower >= upper:
            continue
        delta = np.random.uniform(lower, upper)
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_val = compute_c5(new_h)
        dE = new_val - cur_val
        if dE < 0 or np.random.rand() < np.exp(-dE / T):

            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
        # update temperature
        T *= cooling
        iterations += 1

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def project_to_capped_simplex(v, S):
    # Clip values maybe not needed, just solve with bisection on lambda
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + hi, 0.0, 1.0)
    return x

import numpy as np, time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def project_to_capped_simplex(v, S):
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v)
    for _ in range(60):  # sufficient
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < S:
            lo = mid
        else:
            hi = mid
    return np.clip(v + hi, 0.0, 1.0)

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return C₅ for a given step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _project_capped_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set
        {x | 0 ≤ x_i ≤ 1, sum(x) = target_sum }.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0          # forces all entries to 0
    hi = 1.0 - np.min(v)           # forces all entries to 1
    for _ in range(60):           # enough for double precision
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + hi, 0.0, 1.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises a step function h on [0,2] with ∫h = 1
    to obtain a low value of C₅ = maxₖ∫h(x)(1‑h(x+k))dx.
    Returns (h_array, C₅_value, number_of_grid_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # --------------------------------------------------------------
    # 1. Choose a working resolution
    # --------------------------------------------------------------
    n = 256                     # number of samples – 256 gives a good trade‑off
    target_sum = n / 2.0        # because Σh * (2/n) = 1  ⇒  Σh = n/2

    # --------------------------------------------------------------
    # 2. Build an initial feasible vector
    # --------------------------------------------------------------
    if "initial_h_values" in globals() and isinstance(initial_h_values, (list, np.ndarray)):
        init = np.asarray(initial_h_values, dtype=np.float64)

        if n0 != n:

            old_x = np.linspace(0.0, 2.0, n0, endpoint=False) + 0.5 * (2.0 / n0)
            new_x = np.linspace(0.0, 2.0, n, endpoint=False) + 0.5 * (2.0 / n)
            h0 = np.interp(new_x, old_x, init)
        else:
            h0 = init.copy()
    else:
        # a simple feasible start – the constant ½ satisfies the integral
        h0 = np.full(n, 0.5, dtype=np.float64)

    h0 = _project_capped_simplex(h0, target_sum)

    # --------------------------------------------------------------
    # 3. Simulated‑annealing style search
    # --------------------------------------------------------------
    best_h = h0.copy()

    cur_val = best_val

    # temperature schedule (purely for occasional uphill moves)
