def project_onto_bounds_and_sum(v, target):
    # Clip v to [0,1] first? Actually projection will handle bounds via clamp after subtract λ.
    # We'll perform water-filling via bisection over λ.

    # Determine λ range: low = min(v)-1 maybe; high = max(v) maybe.
    # Actually we need to find λ such that sum_i min(1, max(0, v_i - λ)) = target.
    # For λ very negative -> all v_i - λ >1, clamp to 1 => sum = n (max). For λ very large -> sum=0.
    # So we can binary search λ in [low, high].
    # low bound: λ_low = min(v) - 1 ; high bound: λ_high = max(v)

    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(x)
        if s < target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v - lo, 0.0, 1.0)
    # Ensure sum after final slightly adjust due to numeric.
    diff = target - np.sum(x)
    # Adjust small by adding diff uniformly to non-boundary entries
    if abs(diff) > 1e-8:
        # Find indices not at bounds
        free = (x > 0) & (x < 1)
        if np.any(free):
            x[free] += diff / np.sum(free)
            x = np.clip(x, 0.0, 1.0)
        # else nothing.
    return x

weights = exp_terms / np.sum(exp_terms)

def compute_gradient(h, weights):
    n = h.shape[0]
    grad = np.zeros_like(h)
    for shift_idx, w in enumerate(weights):
        if w == 0:
            continue
        shift = shift_idx - (n - 1)
        if shift >= 0:
            i = np.arange(0, n - shift)
            j = i + shift
        else:
            i = np.arange(-shift, n)
            j = i + shift
        grad[i] += w * (1 - h[j])
        grad[j] -= w * h[i]
    return grad

import time, math
import numpy as np

def _project_onto_bounds_and_sum(v, target):
    """Project v onto {0<=x<=1, sum(x)=target} via bisection on a shift."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v - lo, 0.0, 1.0)
    # fix tiny residual sum error
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 0) & (x < 1)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_and_weights(h, alpha):
    """Return raw correlation (no dx) and smooth-max weights."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    # log‑sum‑exp trick
    shifted = corr - mx
    expv = np.exp(alpha * shifted)
    w = expv / expv.sum()
    return corr, w

def _grad_from_weights(h, w):
    """Gradient of the smooth‑max objective w.r.t. h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    # shift index runs from -(n-1) .. n-1
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            i = slice(0, n - shift)
            j = slice(shift, n)
        else:
            i = slice(-shift, n)
            j = slice(0, n + shift)
        hi = h[i]
        hj = h[j]
        # ∂/∂h_i term
        grad[i] += weight * (1.0 - hj)
        # ∂/∂h_j term (negative)
        grad[j] -= weight * hi
    return grad

def _c5_from_h(h):
    """C5 value using the prescribed evaluation."""
    dx = 2.0 / h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def _initial_from_parent(parent, n):
    """Upsample/downsample the provided parent to length n."""
    parent = np.asarray(parent, dtype=float)
    m = parent.size
    xs = np.linspace(0.0, 2.0, m, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(xs_new, xs, parent)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    h = _project_onto_bounds_and_sum(h, target)
    return h

def _triangular(n):
    """Symmetric triangle of height 1, area exactly 1."""
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))
    # exact integral is 1 for continuous; adjust discretisation sum.
    target = n / 2.0
    return _project_onto_bounds_and_sum(h, target)

    start_time = time.time()
    np.random.seed(seed)

    # --- configuration -------------------------------------------------
    candidates = [45, 80]                 # try a small and a larger mesh
    best_h = None
    best_val = float('inf')
    best_n = None

    # the smoothing parameter for the soft‑max; it will be lifted gradually
    base_alpha = 8.0

    # iterate over mesh sizes
    for n in candidates:
        dx = 2.0 / n
        target_sum = n / 2.0

        # initial guess -------------------------------------------------
        if 'initial_h_values' in globals() and initial_h_values is not None:

            h0 = _triangular(n)

            h0 = np.clip(h0 + noise, 0.0, 1.0)
            h0 = _project_onto_bounds_and_sum(h0, target_sum)

        # multiple random restarts for this mesh
        restarts = 12

            # a bigger random perturbation for later restarts
