def optimize_h(h0, n, dx, budget, rng):
    start = time.time()
    a = h0.copy()
    target_sum = n/2
    best_a = a.copy()
    best_val = compute_c5(a, dx)
    max_iter = 100000  # large but we will break early
    beta = 5.0
    beta_max = 30.0
    step0 = 0.5   # learning rate

    for it in range(max_iter):
        if time.time() - start > budget:
            break
        # compute r
        r_full = np.correlate(a, 1 - a, mode='full') * dx
        cur_val = r_full.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # compute softmax weights
        # Use scaling to avoid overflow
        max_r = r_full.max()
        exps = np.exp(beta * (r_full - max_r))
        w = exps / exps.sum()
        # compute gradient
        grad = np.ones_like(a)
        offset = n - 1
        # We can vectorize loops using slices but will stay in loop
        for s in range(-n+1, n):
            ws = w[offset + s]
            if ws == 0:
                continue
            if s >= 0:
                # a_{j+s}
                grad[:n-s] -= ws * a[s:]
                # a_{j-s}
                grad[s:] -= ws * a[:n-s]
            else:
                t = -s
                grad[t:] -= ws * a[:n-t]
                grad[:n-t] -= ws * a[t:]

        step = step0 / math.sqrt(it + 1)
        # update
        a_new = a - step * grad

        a = a_new

            beta = min(beta_max, beta * 1.2)
    # final refine maybe random small perturbation
    # Additional improvement: try occasional random variation
    # We'll skip due to time
    return best_a, best_val

def project_to_capped_simplex(v, S, lo=0.0, hi=1.0, max_iter=100, tol=1e-9):
    # Using binary search for theta
    # Note: function f(theta) = sum(clipped(v - theta, lo, hi)) - S is monotonic decreasing in theta.
    low = v.min() - hi
    high = v.max() - lo
    # Ensure total at low >= S and at high <= S
    # Might need to adjust extremes:
    for _ in range(max_iter):
        theta = (low + high) / 2.0
        w = np.clip(v - theta, lo, hi)
        total = w.sum()
        if abs(total - S) <= tol:
            return w
        if total > S:
            low = theta
        else:
            high = theta
    # Return result after max_iter
    return np.clip(v - (low + high) / 2.0, lo, hi)

def capped_simplex_projection(v, S, lo=0.0, hi=1.0, max_iter=50, tol=1e-7):
    lo_theta = v.min() - hi
    hi_theta = v.max() - lo
    for it in range(max_iter):
        theta = (lo_theta + hi_theta) / 2.0
        w = np.clip(v - theta, lo, hi)
        total = w.sum()
        if abs(total - S) <= tol:
            return w
        if total > S:
            lo_theta = theta
        else:
            hi_theta = theta
    # final return
    return np.clip(v - (lo_theta + hi_theta) / 2.0, lo, hi)

def optimize_one(a0, n, dx, budget, rng):
    a = a0.copy()
    target_sum = n / 2.0
    best_a = a.copy()
    best_val = compute_c5(a, dx)
    start = time.time()
    beta = 5.0
    beta_max = 50.0
    step0 = 1.0  # initial step size
    iteration = 0
    while time.time() - start < budget:
        # correlation
        r = np.correlate(a, 1 - a, mode='full') * dx
        cur_val = r.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # softmax weights
        max_r = r.max()
        exps = np.exp(beta * (r - max_r))
        w = exps / exps.sum()
        # gradient
        grad = np.ones_like(a)
        offset = n - 1
        # iterate shifts
        for s in range(-n+1, n):
            ws = w[offset + s]
            if ws == 0:
                continue
            if s >= 0:
                # term a_{j+s}
                grad[:n - s] -= ws * a[s:]
                # term a_{j-s}
                grad[s:] -= ws * a[:n - s]
            else:
                t = -s
                grad[t:] -= ws * a[:n - t]
                grad[:n - t] -= ws * a[t:]
        grad = grad * dx
        # step size schedule
        step = step0 / np.sqrt(iteration + 1)
        a_new = a - step * grad
        # keep values within bounds via projection
        a_new = capped_simplex_projection(a_new, target_sum, lo=0.0, hi=1.0)
        a = a_new
        # adjust beta gradually
        if iteration % 200 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.2)
        iteration += 1
    # final evaluate best after loop
    return best_a, best_val

def compute_c5(a, dx):
    r = np.correlate(a, 1 - a, mode='full') * dx
    return r.max()

import numpy as np
import time
import math

def _capped_simplex_projection(v, S, lo=0.0, hi=1.0, max_iter=60, tol=1e-9):
    """Project v onto {x : lo<=x_i<=hi, sum x_i = S}."""
    lo_theta = v.min() - hi
    hi_theta = v.max() - lo
    for _ in range(max_iter):
        theta = (lo_theta + hi_theta) * 0.5
        w = np.clip(v - theta, lo, hi)
        total = w.sum()
        if abs(total - S) <= tol:
            return w
        if total > S:
            lo_theta = theta
        else:
            hi_theta = theta
    # final approximation
    theta = (lo_theta + hi_theta) * 0.5
    return np.clip(v - theta, lo, hi)


def _c5_value(a, dx):
    """Maximum overlap for a step function a (array)."""
    return np.max(np.correlate(a, 1 - a, mode="full") * dx)


def _optimize_one(a0, n, dx, time_budget, rng):
    """Projected Adam optimization of max‑overlap."""
    a = a0.copy()
    S = n / 2.0
    best_a = a.copy()
    best_val = _c5_value(a, dx)

    start = time.time()
    beta = 5.0           # soft‑max temperature
    beta_max = 50.0
    lr = 1.5             # base learning rate for Adam
    eps = 1e-8
    b1, b2 = 0.9, 0.999  # Adam coefficients
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() - start < time_budget:
        # correlation and current bound
        r = np.correlate(a, 1 - a, mode="full") * dx
        cur = r.max()
        if cur < best_val:
            best_val = cur
            best_a = a.copy()

        # soft‑max weights over all shifts
        max_r = r.max()
        expw = np.exp(beta * (r - max_r))
        w = expw / expw.sum()

        # gradient of the smooth max (dx factor included later)
        grad = np.ones_like(a)
        off = n - 1
        for s in range(-n + 1, n):
            ws = w[off + s]
            if ws == 0.0:
                continue
            if s >= 0:
                grad[: n - s] -= ws * a[s:]
                grad[s:]      -= ws * a[: n - s]
            else:
                t = -s
                grad[t:]      -= ws * a[: n - t]
                grad[: n - t] -= ws * a[t:]
        grad *= dx

        # Adam update
        it += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1 - b1 ** it)
        v_hat = v / (1 - b2 ** it)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # feasibility projection
        a = _capped_simplex_projection(a_new, S, lo=0.0, hi=1.0)

        # occasional random perturbation (exploration)
        if it % 400 == 0:
            noise = rng.normal(scale=0.01, size=n)
            a_try = _capped_simplex_projection(a + noise, S, lo=0.0, hi=1.0)
            if _c5_value(a_try, dx) < cur:
                a = a_try

        # raise temperature gradually
        if it % 200 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.2)

    return best_a, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    rng = np.random.default_rng(seed)

    # number of discretisation points (trade‑off between quality and speed)
    n = 200
    dx = 2.0 / n
    target_sum = n / 2.0

    start_all = time.time()
    best_h = None
    best_val = float("inf")
    max_restarts = 5

    for restart in range(max_restarts):
        elapsed = time.time() - start_all
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break
        # allocate roughly equal share of the remaining time
        time_here = remaining / (max_restarts - restart)

        # initialise a feasible start vector
        if restart == 0 and "initial_h_values" in globals():
            init = np.asarray(initial_h_values, dtype=float)
            n_init = init.shape[0]
            xs = np.linspace(0.0, 2.0, n_init, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(xt, xs, init)
        else:
            u = rng.uniform(size=n)
            h0 = _capped_simplex_projection(u, target_sum, lo=0.0, hi=1.0)

        h_opt, val_opt = _optimize_one(h0, n, dx, time_here, rng)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()

    # safety fallback
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_h = _capped_simplex_projection(best_h, target_sum, lo=0.0, hi=1.0)
        best_val = _c5_value(best_h, dx)

    # final sanity projection (ensures exact feasibility)
    best_h = _capped_simplex_projection(best_h, target_sum, lo=0.0, hi=1.0)
    best_val = float(_c5_value(best_h, dx))

    return best_h, best_val, n
