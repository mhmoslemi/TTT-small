def build_h(l, g, n):
    # l length r array, g length r+1 array
    # return binary vector of length n
    # use np.concatenate of zeros/ones.
    parts = []
    for i in range(len(l)):
        parts.append(np.zeros(g[i], dtype=np.float64))
        parts.append(np.ones(l[i], dtype=np.float64))
    parts.append(np.zeros(g[-1], dtype=np.float64))
    h = np.concatenate(parts)
    # Sometimes due to rounding errors the concatenated length may not be exactly n if arrays lengths not sum properly.

    if h.shape[0] != n:

        if h.shape[0] < n:
            h = np.concatenate([h, np.zeros(n - h.shape[0])])
        else:
            h = h[:n]
    return h

import numpy as np
import math
import time

def _max_overlap(h: np.ndarray) -> int:
    """Return the integer C5 count: max_k ∫ h·(1‑h) dx approximated on the grid."""
    return int(np.max(np.correlate(h, 1.0 - h, mode="full")))


def _build_h(lens: np.ndarray, gaps: np.ndarray, n: int) -> np.ndarray:
    """Construct the 0‑1 step function from interval lengths and gaps."""
    parts = []
    r = len(lens)
    for i in range(r):
        parts.append(np.zeros(gaps[i], dtype=np.float64))
        parts.append(np.ones(lens[i], dtype=np.float64))
    parts.append(np.zeros(gaps[-1], dtype=np.float64))
    h = np.concatenate(parts)
    if h.shape[0] != n:
        if h.shape[0] < n:
            h = np.concatenate([h, np.zeros(n - h.shape[0])])
        else:
            h = h[:n]
    return h


def _local_search_intervals(n: int, r: int, rng: np.random.Generator,
                           step_limit: int = 3000) -> tuple[np.ndarray, int]:
    """Simulated‑annealing on the space of r interval (block) placements."""
    total_ones = n // 2
    total_zeros = n - total_ones

    # random composition of ones into r blocks and zeros into r+1 gaps
    lens = rng.multinomial(total_ones, np.full(r, 1.0 / r))
    gaps = rng.multinomial(total_zeros, np.full(r + 1, 1.0 / (r + 1)))

    h = _build_h(lens, gaps, n)
    cur_score = _max_overlap(h)
    best_h, best_score = h.copy(), cur_score

    T = 2.0
    cooling = 0.9995
    for _ in range(step_limit):
        if rng.random() < 0.5:                     # move a unit between intervals
            if r < 2:
                continue
            src = rng.integers(r)
            if lens[src] == 0:
                for __ in range(3):
                    src = rng.integers(r)
                    if lens[src] > 0:
                        break
                else:
                    continue
            dst = rng.integers(r)
            while dst == src:
                dst = rng.integers(r)
            new_lens = lens.copy()
            new_lens[src] -= 1
            new_lens[dst] += 1
            new_h = _build_h(new_lens, gaps, n)
            move = "interval"
        else:                                      # move a unit between gaps
            src = rng.integers(r + 1)
            if gaps[src] == 0:
                for __ in range(3):
                    src = rng.integers(r + 1)
                    if gaps[src] > 0:
                        break
                else:
                    continue
            dst = rng.integers(r + 1)
            while dst == src:
                dst = rng.integers(r + 1)
            new_gaps = gaps.copy()
            new_gaps[src] -= 1
            new_gaps[dst] += 1
            new_h = _build_h(lens, new_gaps, n)
            move = "gap"

        new_score = _max_overlap(new_h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            if move == "interval":
                lens = new_lens
            else:
                gaps = new_gaps
            h, cur_score = new_h, new_score
            if cur_score < best_score:
                best_score, best_h = cur_score, h.copy()
        T *= cooling
        if T < 1e-4:
            T = 1e-4
    return best_h, best_score


def _local_search_swaps(n: int, rng: np.random.Generator,
                        step_limit: int = 3000) -> tuple[np.ndarray, int]:
    """Simulated‑annealing on the full binary space by swapping a 1↔0."""
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0

    cur_score = _max_overlap(h)
    best_h, best_score = h.copy(), cur_score

    T = 2.0
    cooling = 0.9995
    for _ in range(step_limit):
        # pick a random 1 and a random 0
        idx_ones = np.where(h > 0.5)[0]
        idx_zeros = np.where(h < 0.5)[0]
        if idx_ones.size == 0 or idx_zeros.size == 0:
            break
        i = rng.choice(idx_ones)
        j = rng.choice(idx_zeros)

        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0

        new_score = _max_overlap(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            h, cur_score = h_new, new_score
            if cur_score < best_score:
                best_score, best_h = cur_score, h.copy()
        T *= cooling
        if T < 1e-4:
            T = 1e-4
    return best_h, best_score


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with a small Erdős overlap bound."""
    start = time.time()
    rng = np.random.default_rng(seed)

    target_bound = 0.3808                      # desired upper bound
    n_candidates = [140, 160, 180, 200]        # even numbers < 1000
    best_h = None
    best_n = None
    best_bound = float("inf")

    # --- incorporate a provided initial construction if present -----------------
    init = globals().get("initial_h_values")
    if init is not None:
        init_arr = np.asarray(init, dtype=np.float64)
        n0 = init_arr.shape[0]
        init_arr = np.clip(init_arr, 0.0, 1.0)
        cur = init_arr.sum()
        need = n0 / 2
        if not np.isclose(cur, need, atol=1e-6) and cur > 0:
            init_arr = np.clip(init_arr * (need / cur), 0.0, 1.0)
        b0 = _max_overlap(init_arr) * (2.0 / n0)
        best_h, best_n, best_bound = init_arr.copy(), n0, b0

    # ---------------------------- main optimisation loop -----------------------
    while time.time() - start < budget_s * 0.95:
        n = int(rng.choice(n_candidates))
        if n % 2:
            n += 1

        # Choose a search strategy: interval‑based or full‑swap
        if rng.random() < 0.6:
            r = int(rng.integers(3, 8))               # 3..7 intervals
            h_loc, score_loc = _local_search_intervals(n, r, rng,
                                                       step_limit=3000)
        else:
            h_loc, score_loc = _local_search_swaps(n, rng,
                                                   step_limit=3000)

        bound_loc = score_loc * (2.0 / n)
        if bound_loc < best_bound:
            best_h, best_n, best_bound = h_loc.copy(), n, bound_loc
            if best_bound <= target_bound:
                break

        # occasional pure random binary trial (cheap diversification)
        if rng.random() < 0.05:
            h_rand = np.zeros(n, dtype=np.float64)
            ones = rng.choice(n, size=n // 2, replace=False)
            h_rand[ones] = 1.0
            b_rand = _max_overlap(h_rand) * (2.0 / n)
            if b_rand < best_bound:
                best_h, best_n, best_bound = h_rand, n, b_rand
                if best_bound <= target_bound:
                    break

    # -------------------------------------------------------------------------
    # Ensure the returned vector satisfies the exact integral constraint.
    if best_h is None:
        # fall back to the trivial half‑interval construction
        best_n = 200
        best_h = np.zeros(best_n, dtype=np.float64)
        best_h[:best_n // 2] = 1.0
        best_bound = _max_overlap(best_h) * (2.0 / best_n)

    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    needed = best_n / 2
    if not np.isclose(cur_sum, needed, atol=1e-6):
        diff = int(round(needed - cur_sum))
        if diff > 0:
            zeros = np.where(best_h < 0.5)[0]
            best_h[zeros[:diff]] = 1.0
        elif diff < 0:
            ones = np.where(best_h > 0.5)[0]
            best_h[ones[: -diff]] = 0.0

    final_bound = _max_overlap(best_h) * (2.0 / best_n)
    return best_h, float(final_bound), int(best_n)
