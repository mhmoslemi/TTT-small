def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # make sure h within [0,1], sum close to n/2
    # compute convolution with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def project_bounds_sum(v, target):
    # projection of v onto the box [0,1] with sum constraint = target
    lo = -2.0
    hi = 2.0
    # expand bounds if necessary
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.8          # initial step size
    decay = 0.998       # geometric decay
    min_step = 1e-5

        # worst shift (largest c5 contribution)
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = int(np.argmax(corr))

        d = np.zeros(n, dtype=np.float64)

import numpy as np, time, math, random

def _c5_of(h: np.ndarray) -> float:
    """Compute the C₅ value for a step function given by the samples h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) gives the integrand for every discrete shift
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_onto_box_and_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the hypercube [0,1]^n
    intersected with the affine hyperplane sum(v)=target_sum.
    Solved by a simple bisection on the Lagrange multiplier λ for:
        proj_i = clip(v_i - λ, 0, 1)
    such that Σ proj_i = target_sum.
    """
    lo, hi = -2.0, 2.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _improve(h: np.ndarray, target_sum: float, time_limit: float):
    """
    Simple sub‑gradient descent for the minimax problem.
    At each iteration we locate the shift that currently attains
    C₅ (the worst shift) and move the values of h towards the
    direction that most improves the overlap for that shift,
    while keeping the integral constraint satisfied.
    """
    n = h.shape[0]
    start = time.time()
    best_h = h.copy()
    best_val = _c5_of(h)

    step = 0.8          # initial step magnitude
    decay = 0.998       # geometric decay of the step
    min_step = 1e-5

    while time.time() - start < time_limit:
        # correlation to find the worst shift (largest contribution)
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)

        # neighbor‑sum d_i = h_{i+shift} + h_{i-shift}
        d = np.zeros_like(h)
        if shift >= 0:
            s = shift
            if s < n:
                d[: n - s] += h[s:]
                d[s:] += h[: n - s]
        else:
            s = -shift
            if s < n:
                d[s:] += h[: n - s]
                d[: n - s] += h[s:]

        # direction = increase where d is large, keep zero‑mean to roughly preserve sum
        direction = d - d.mean()
        h = h + step * direction
        h = _project_onto_box_and_sum(h, target_sum)

        cur = _c5_of(h)
        if cur < best_val:
            best_val = cur
            best_h = h.copy()

        step *= decay
        if step < min_step:
            step = 0.8

    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function that (hopefully) improves the known C₅ bound."""
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # 1.  Determine the working resolution.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n0 = init.size
    else:
        init = None
        n0 = 0

    # aim for at least ~200 points but stay modest to keep evaluation cheap
    target_n = max(200, (n0 // 1) * 4)
    if target_n % 2:
        target_n += 1                     # need an even number for the ½‑integral
    target_sum = target_n / 2.0           # because ∫h = 1  ⇔  Σh·dx = 1  with dx=2/n

    # ------------------------------------------------------------------
    # 2.  Build an initial feasible vector.
    # ------------------------------------------------------------------
    if init is not None:
        # up‑sample the provided construction by simple repetition
        factor = int(math.ceil(target_n / n0))
        h = np.repeat(init, factor)[:target_n].astype(float)
    else:
        # random binary start (exactly half ones)
        h = np.zeros(target_n, dtype=float)
        ones = np.random.choice(target_n, target_n // 2, replace=False)
        h[ones] = 1.0

    h = _project_onto_box_and_sum(h, target_sum)

    # ------------------------------------------------------------------
    # 3.  Quick check – maybe the supplied construction already meets the goal.
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_val = _c5_of(h)
    if best_val <= 0.38085:
        return best_h, best_val, target_n

    # ------------------------------------------------------------------
    # 4.  Allocate time and run the optimisation loop.
    # ------------------------------------------------------------------
    elapsed = time.time() - 0.0
    remaining = budget_s - elapsed - 0.5   # leave a small safety margin
    if remaining > 0.0:

    return best_h, best_val, target_n
