import numpy as np, time, math

def project_onto_box_sum(v, target):
    # water-filling algorithm: find lambda such that sum_i min(max(v_i - lam, 0),1) = target.
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def random_delta_move(h, rng):
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return None, None, None
    hi = h[i]
    hj = h[j]
    # step direction +/- ensures feasible
    lower = max(-hi, hj - 1.0)   # min delta
    upper = min(1.0 - hi, hj)    # max delta
    if lower >= upper:
        return None, None, None
    delta = rng.uniform(lower, upper)
    # Return indices and delta
    return i, j, delta

def simulated_annealing(h0, budget, rng):
    n = len(h0)
    target_sum = n / 2.0
    cur_h = h0.copy()
    cur_cost = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_cost = cur_cost
    # Parameters
    start_time = time.time()
    T = 0.02  # initial temperature
    decay = 0.9995
    iter = 0
    while True:
        if time.time() - start_time > budget - 0.1:
            break
        # propose move
        move = random_delta_move(cur_h, rng)
        if move[0] is None:
            continue
        i, j, delta = move
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_cost = compute_c5(new_h)
        # acceptance
        if new_cost < cur_cost or rng.random() < math.exp(-(new_cost - cur_cost) / max(T, 1e-12)):
            cur_h = new_h
            cur_cost = new_cost
            if new_cost < best_cost:
                best_h = new_h.copy()
                best_cost = new_cost
        # schedule
        T *= decay
        iter += 1
        # occasional random reset if stuck
        if iter % 50000 == 0:
            # after many iterations, maybe restart with best + noise
            if rng.random() < 0.2:
                cur_h = best_h + rng.normal(scale=0.05, size=n)
                cur_h = np.clip(cur_h, 0.0, 1.0)
                cur_h = project_onto_box_sum(cur_h, target_sum)
                cur_cost = compute_c5(cur_h)
    return best_h, best_cost

def hill_climb(h0, budget, rng):
    n = len(h0)
    target_sum = n/2.0
    cur_h = h0.copy()
    cur_cost = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_cost = cur_cost
    start = time.time()
    # small steps
    step = 0.05
    while time.time() - start < budget - 0.1:
        i, j, delta = random_delta_move(cur_h, rng)
        if i is None:
            continue
        # optionally limit delta magnitude
        max_delta = step
        # restrict delta within max_delta
        delta = max(min(delta, max_delta), -max_delta)
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_cost = compute_c5(new_h)
        if new_cost < cur_cost:
            cur_h = new_h
            cur_cost = new_cost
            if new_cost < best_cost:
                best_h = new_h.copy()
                best_cost = new_cost
    return best_h, best_cost

import time, math
import numpy as np

def _project_onto_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {0<=x<=1, sum(x)=target} using a bisection on λ."""
    lo, hi = -1.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Evaluate C5 = max_k ∫ h(x)(1‑h(x+k))dx using the prescribed correlate."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _propose_move(h: np.ndarray, max_step: float, rng: np.random.Generator):
    """Return (i, j, delta) that respects 0≤h≤1 and keeps the sum unchanged."""
    n = h.shape[0]
    i = rng.integers(n)
    # pick a different index j
    j = rng.integers(n - 1)
    if j >= i:
        j += 1
    hi, hj = h[i], h[j]
    # feasible interval for delta so that both entries stay in [0,1]
    lo = max(-hi, hj - 1.0)
    hi_ = min(1.0 - hi, hj)
    if lo >= hi_:
        return None
    # restrict magnitude to max_step
    lo = max(lo, -max_step)
    hi_ = min(hi_, max_step)
    if lo >= hi_:
        return None
    delta = rng.uniform(lo, hi_)
    return i, j, delta


def _simulated_annealing(start_h: np.ndarray, time_budget: float, rng: np.random.Generator):
    """SA that tries to minimise C5 while honouring the constraints."""
    n = start_h.shape[0]
    target_sum = n / 2.0
    cur_h = start_h.copy()
    cur_val = _c5_value(cur_h)
    best_h, best_val = cur_h.copy(), cur_val

    start = time.time()
    T = 0.05               # initial temperature
    decay = 0.9998         # geometric decay
    step = 0.1             # initial maximal delta
    it = 0

    while time.time() - start < time_budget - 0.05:
        mv = _propose_move(cur_h, step, rng)
        if mv is None:
            continue
        i, j, delta = mv
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_val = _c5_value(new_h)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val

        # temperature & step schedule
        T *= decay
        step = max(0.005, step * decay)
        it += 1

        # occasional re‑shake from the current best
        if it % 50000 == 0 and rng.random() < 0.2:
            cur_h = best_h + rng.normal(scale=0.03, size=n)
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = _project_onto_box_sum(cur_h, target_sum)
            cur_val = _c5_value(cur_h)

    return best_h, best_val


def _hill_climb(start_h: np.ndarray, time_budget: float, rng: np.random.Generator):
    """Simple greedy local search (only accepts improvements)."""
    n = start_h.shape[0]
    target_sum = n / 2.0
    cur_h = start_h.copy()
    cur_val = _c5_value(cur_h)
    best_h, best_val = cur_h.copy(), cur_val

    start = time.time()
    step = 0.02
    while time.time() - start < time_budget - 0.05:
        mv = _propose_move(cur_h, step, rng)
        if mv is None:
            continue
        i, j, delta = mv
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_val = _c5_value(new_h)
        if new_val < cur_val:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
    return best_h, best_val
