def propose_swap(h, rng, max_delta):
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    delta = rng.uniform(-max_delta, max_delta)
    # Determine max allowed delta to keep within [0,1]:
    # delta must satisfy -h[i] <= delta <= 1 - h[i]
    # and - ( -h[j])? Wait we adjust j opposite sign: h[j] - delta must be within [0,1]
    lo_i = -h[i]
    hi_i = 1 - h[i]
    lo_j = h[j] - 1  # because h[j] - delta >= 0 => delta <= h[j]
    hi_j = h[j]      # because h[j] - delta <= 1 => delta >= h[j] - 1
    # Actually delta lower bound: need h[j] - delta >=0 => delta <= h[j]
    # and <=? Let's derive:
    # h[j] - delta ≤ 1 => - delta ≤ 1 - h[j] => delta ≥ h[j] - 1
    delta_low = max(lo_i, h[j] - 1)
    delta_high = min(hi_i, h[j])
    if delta_low > delta_high:
        return None # not feasible; try new pair
    # Clip delta to feasible range.
    delta = np.clip(delta, delta_low, delta_high)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h
```

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / h.shape[0])

def transfer_swap(h, k, rng):
    n = h.shape[0]
    if k >= 0:
        i_range = np.arange(n - k)
    else:
        i_range = np.arange(-k, n)
    j_range = i_range + k
    # mismatched indices where h[i]>0 and h[j]<1

    j = i + k

        return None
    delta = max_delta * rng.uniform(0.1, 0.5)  # random fraction
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h, k

import time
import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """Return C5 for a step function represented by array h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _transfer_on_worst_shift(h: np.ndarray, shift: int, rng: np.random.Generator):
    """Move a little mass from a mismatching 1‑position to its 0‑partner
    for the given shift.  Returns a new vector or None if no move possible."""
    n = h.shape[0]
    if shift >= 0:
        i_idx = np.arange(n - shift)
        j_idx = i_idx + shift
    else:
        i_idx = np.arange(-shift, n)
        j_idx = i_idx + shift
    mask = (h[i_idx] > 0.0) & (h[j_idx] < 1.0)
    if not np.any(mask):
        return None
    cand_i = i_idx[mask]
    i = rng.choice(cand_i)
    j = i + shift
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0.0:
        return None
    delta = max_delta * rng.uniform(0.1, 0.5)   # moderate step
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def _random_pair_move(h: np.ndarray, rng: np.random.Generator, max_delta: float):
    """Pick two indices and exchange a feasible amount of mass."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    lo = max(-h[i], h[j] - 1.0)          # lower bound for delta (h[i] + delta, h[j] - delta)
    hi = min(1.0 - h[i], h[j])
    if lo >= hi:
        return None
    delta = rng.uniform(lo, hi)
    if abs(delta) < 1e-12:
        return None
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h
