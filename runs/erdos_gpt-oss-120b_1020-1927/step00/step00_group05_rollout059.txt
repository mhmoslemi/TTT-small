    else:

    def random_binary():
        # Choose n/2 positions of ones
        ones = rng.choice(n, n//2, replace=False)
        h = np.zeros(n, dtype=np.int8)
        h[ones] = 1
        return h

    # Compute correlation max for a vector
    def max_correlation(h):
        # Compute correlation vector (use float for speed)
        c = np.correlate(h, 1 - h, mode='full')
        return c.max()

    # Initialize best
    best_h = None
    best_max = None

    # Dim count of loops
    # We'll run while time < budget_s - some margin
    max_iterations = 1000000  # high

    # We'll linearly anneal T over total iterations or time.
    # We'll compute fraction based on elapsed ratio.

    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, copy=False)
        if init.shape[0] != n:
            # if shape diff, ignore
            pass
        else:

            h0 = (init > 0.5).astype(np.int8)

                diff = cur_sum - n//2
                if diff > 0:
                    # too many ones: set diff random ones to zero
                    idx = rng.choice(np.where(h0==1)[0], diff, replace=False)
                    h0[idx] = 0
                elif diff < 0:
                    idx = rng.choice(np.where(h0==0)[0], -diff, replace=False)
                    h0[idx] = 1
            cur_max = max_correlation(h0)
            best_h = h0
            best_max = cur_max

    # Number of restarts
    restarts = 10

    while time.time() - start_time < budget_s * 0.95:  # keep margin
        # generate random restart
        h = random_binary()
        cur_max = max_correlation(h)
        # local search loops for this restart
        itr = 0
        while time.time() - start_time < budget_s * 0.95:
            # compute temperature based on elapsed fraction (linear schedule)
            elapsed = time.time() - start_time
            fraction = elapsed / budget_s
            T = T_start * (1 - fraction) + T_end * fraction

            # pick i from ones, j from zeros
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]

            h[j] = 1

                    best_h = h.copy()

                        break

                h[i] = 1
                h[j] = 0
            itr += 1
            # optional early stop for local loop after some iterations
            if itr % 1000 == 0:
                # if improvement hasn't been seen for while maybe break
                pass
        # after local search, maybe break if bound satisfied
        if best_max is not None and best_max * (2.0 / n) <= 0.3808:

    if best_h is None:
        # fallback: use uniform half vector

        best_max = max_correlation(best_h.astype(int))  # not correct but anyway
    c5 = best_max * (2.0 / n)

    # Convert best_h to float values [0,1] as required
    h_vals = best_h.astype(float)
    # Ensure sum constraint within tolerance
    # If continuous, not needed
    return (h_vals, float(c5), n)

import time, math
import numpy as np


def _max_corr(arr: np.ndarray) -> int:
    """Maximum of the correlation of h with (1‑h).  Returns integer count."""
    return int(np.correlate(arr, 1 - arr, mode="full").max())


def _project_onto_constraints(v: np.ndarray) -> np.ndarray:
    """
    Project vector onto the intersection of the box [0,1]^n and the hyperplane
    sum(v) = n/2 (n must be even).  Uses a simple bisection on the Lagrange
    multiplier for the equality constraint.
    """
    n = v.shape[0]
    target = n / 2.0
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):

    C₅ = max_k ∫ h(x)(1−h(x+k))dx.

        n = int(np.asarray(initial_h_values).shape[0])
    else:
        n = 70
    if n % 2:
        n += 1                     # must be even for sum = n/2
    target_sum = n // 2

    # helper to evaluate C5 for a candidate vector
    def _eval(h):
        maxc = _max_corr(h)
        return float(maxc * (2.0 / n)), maxc

        if m % 2 == 0:
            return m == 2

        for d in range(3, r + 1, 2):
            if m % d == 0:
                return False
        return True

    # ------------------------------------------------------------------
    start_time = time.time()
    T_start, T_end = 5.0, 1e-4

    while time.time() - start_time < 0.95 * budget_s:
        # start from a fresh random binary vector
        cur = np.zeros(n, dtype=np.int8)
        cur[np.random.choice(n, target_sum, replace=False)] = 1
        cur_max = _max_corr(cur)

        # local search loop
        it = 0
        while time.time() - start_time < 0.95 * budget_s:
            # temperature schedule (linear in elapsed fraction)
            elapsed = time.time() - start_time
            frac = elapsed / budget_s
            T = T_start * (1.0 - frac) + T_end * frac

            # pick a random 1 and a random 0 and swap them
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cur[i] = 0
            cur[j] = 1

            new_max = _max_corr(cur)
            delta = new_max - cur_max
            accept = False
            if delta <= 0:
                accept = True
            else:
                if rng.random() < math.exp(-delta / T):
                    accept = True

            if accept:
                cur_max = new_max
                cur_val = cur_max * (2.0 / n)
                if best_val is None or cur_val < best_val:
                    best_val = cur_val
                    best_h = cur.astype(float).copy()
                    best_max = int(cur_max)
                    if best_val <= 0.3808:
                        break
            else:
                # revert the swap
                cur[i] = 1
                cur[j] = 0

            it += 1
            if it % 2000 == 0 and best_val is not None and best_val <= 0.3808:
                break
        if best_val is not None and best_val <= 0.3808:
            break

    # ------------------------------------------------------------------
    # 4) final safety checks and return
    # ------------------------------------------------------------------
    if best_h is None:
        # fallback: the constant 0.5 function (already feasible)
        best_h = np.full(n, 0.5, dtype=float)
        best_val, _ = _eval(best_h)

    # enforce constraints exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(best_h.sum() - target_sum) > 1e-9:
        best_h = _project_onto_constraints(best_h)

    return (best_h, float(best_val), n)
