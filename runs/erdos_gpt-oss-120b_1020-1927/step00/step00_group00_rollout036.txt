def compute_delta_c(old_h, i, j):
    n = len(old_h)
    L = 2*n - 1
    d = np.empty(L, dtype=int)
    for idx in range(L):
        s = idx - (n-1)
        term = 0
        # term1:
        ii = i - s

def weighted_shift_sum(h, w):
    n = len(h)
    L = len(w)
    shift_offset = n - 1
    # compute sum_{s} w_s * h[i+s] with zero outside
    result = np.zeros_like(h, dtype=float)
    for s in range(-n+1, n):
        idx = s + shift_offset
        w_s = w[idx]
        # shift h by s
        if s >= 0:
            result[:n-s] += w_s * h[s:]
        else:
            result[-s:] += w_s * h[:n+s]
    return result

def weighted_shift_sum_vec(h, w):
    n = len(h)
    L = len(w)
    offset = n-1
    result = np.zeros_like(h, dtype=float)
    for s in range(-n+1, n):
        idx = s + offset
        ws = w[idx]
        if ws == 0:
            continue
        if s >= 0:
            result[:n-s] += ws * h[s:]
        else:
            result[-s:] += ws * h[:n+s]
    return result

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine n from initial_h_values if exists:
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = len(init)
        # may be odd; if odd make even by truncating or adding zero.
        if n % 2 != 0:
            # make even by truncating last element
            init = init[:-1]
            n -= 1
        h0 = np.clip(init,0.0,1.0)
    else:
        n = 120
        h0 = np.full(n, 0.5)
    # ensure even:
    if n % 2 != 0:
        n = n+1
        h0 = np.append(h0, 0.5)
    # set target sum and project
    target_sum = n/2
    # projection to fulfill sum exactly (as floats)
    def project(x):
        x = np.clip(x,0.0,1.0)
        lo = -1.0
        hi = 1.0
        for _ in range(30):
            lam = (lo+hi)/2.0
            y = np.clip(x - lam, 0.0, 1.0)
            s = y.sum()
            if s > target_sum:
                lo = lam
            else:
                hi = lam
        return np.clip(x - (lo+hi)/2.0, 0.0, 1.0)
    h = project(h0)

    best_h = h.copy()
    # compute initial score using helper
    def compute_score(vec):
        # Assume evaluate_erdos_solution exists
        if 'evaluate_erdos_solution' in globals():
            return float(evaluate_erdos_solution(vec))
        else:
            nlen = len(vec)
            dx = 2.0 / nlen
            corr = np.correlate(vec, 1-vec, mode='full')
            return float(np.max(corr) * dx)

    best_score = compute_score(best_h)

    # parameters
    alpha = 30.0
    eta = 0.5
    decay = 0.9995
    max_iter = 200000
    iter_count = 0

    n = len(h)
    dx = 2.0 / n
    offset = n - 1
    L = 2*n - 1

    # Main loop
    while time.time() - start < budget_s * 0.95 and iter_count < max_iter:
        iter_count += 1
        # correlation raw (no dx)
        R = np.correlate(h, 1 - h, mode='full')
        maxR = R.max()
        # softmax weights
        exp_term = np.exp(alpha * (R - maxR))
        w = exp_term / exp_term.sum()
        # compute A and B
        A = np.zeros(n, dtype=float)
        B = np.zeros(n, dtype=float)
        for s in range(-n+1, n):
            idx = s + offset
            ws = w[idx]
            if ws == 0.0:
                continue
            if s >= 0:
                A[:n - s] += ws * h[s:]
                B[s:] += ws * h[:n - s]
            else:
                t = -s
                A[t:] += ws * h[:n - t]
                B[:n - t] += ws * h[t:]
        grad = dx * (1.0 - A - B)
        # step
        h_new = h - eta * grad
        # projection
        h = project(h_new)

        # decay step size
        eta *= decay

        # occasionally evaluate score
        if iter_count % 100 == 0:
            cur_score = compute_score(h)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
    # final projection to ensure sum exact
    best_h = project(best_h)
    final_score = compute_score(best_h)
    return (best_h.astype(float), float(final_score), len(best_h))

import time
import numpy as np

def _project_to_box_and_sum(x, target):
    """Project x onto [0,1]^n with sum equal to target (capped simplex)."""
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(30):
        lam = (lo + hi) / 2.0
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)


def _compute_score(vec):
    """Exact C5 value as used by the harness."""
    if 'evaluate_erdos_solution' in globals():
        return float(evaluate_erdos_solution(vec))
    n = len(vec)
    dx = 2.0 / n
    return float(np.max(np.correlate(vec, 1 - vec, mode='full')) * dx)


def run(seed=42, budget_s=1100, **_):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and initial vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = h0.shape[0]
        if n % 2:
            # make size even by discarding the last element
            h0 = h0[:-1]
            n -= 1
    else:
        n = 120                     # a moderate size that fits the budget
        h0 = np.full(n, 0.5)

    target_sum = n / 2.0
    # ensure feasibility
    h = _project_to_box_and_sum(h0, target_sum)

    best_h = h.copy()
    best_score = _compute_score(best_h)

    # ------------------------------------------------------------------
    # Optimization parameters
    # ------------------------------------------------------------------
    alpha = 30.0          # smooth‑max temperature
    eta0 = 0.8            # initial step size
    decay = 0.9995        # step‑size decay
    max_iters = 500000    # absolute limit, actual stop is time‑based

    dx = 2.0 / n
    offset = n - 1
    L = 2 * n - 1

    it = 0
    # ------------------------------------------------------------------
    # Main optimisation loop (gradient descent on a smooth approximation)
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.94 and it < max_iters:
        it += 1
        # ---------- correlation (raw, without dx) ----------
        R = np.correlate(h, 1.0 - h, mode='full')      # length 2n‑1
        maxR = R.max()
        # ---------- smooth max (soft‑max) ----------
        exp_shift = np.exp(alpha * (R - maxR))
        w = exp_shift / exp_shift.sum()               # weights, sum to 1

        # ---------- weighted shift sums A and B ----------
        # A_i = Σ_s w_s h_{i+s}
        A_full = np.correlate(w, h, mode='full')      # length 3n‑2
        A = A_full[:n]
        # B_i = Σ_s w_s h_{i‑s} = Σ_s w_{-s} h_{i+s}
        w_rev = w[::-1]
        B_full = np.correlate(w_rev, h, mode='full')
        B = B_full[:n]

        # ---------- gradient of smooth‑max ----------
        grad = dx * (1.0 - A - B)                     # shape (n,)

        # ---------- step & projection ----------
        eta = eta0 * (decay ** it)
        h_new = h - eta * grad
        h = _project_to_box_and_sum(h_new, target_sum)

        # ---------- occasional true score evaluation ----------
        if it % 200 == 0:
            cur = _compute_score(h)
            if cur < best_score:
                best_score = cur
                best_h = h.copy()
            # early exit if already very good
            if best_score <= 0.3808:
                break

    # final polishing (a few extra iterations with a smaller step)
    eta = 0.05
    for _ in range(2000):
        R = np.correlate(best_h, 1.0 - best_h, mode='full')
        maxR = R.max()
        exp_shift = np.exp(alpha * (R - maxR))
        w = exp_shift / exp_shift.sum()
        A = np.correlate(w, best_h, mode='full')[:n]
        B = np.correlate(w[::-1], best_h, mode='full')[:n]
        grad = dx * (1.0 - A - B)
        cand = best_h - eta * grad
        best_h = _project_to_box_and_sum(cand, target_sum)

    best_score = _compute_score(best_h)

    return (best_h.astype(float), float(best_score), n)
