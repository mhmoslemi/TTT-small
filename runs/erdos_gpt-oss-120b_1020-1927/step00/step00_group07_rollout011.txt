def project_capped_simplex(v, target):
    low = v.min() - 1.0
    high = v.max()
    for _ in range(50):
        mid = (low + high) / 2.0
        proj = np.clip(v - mid, 0, 1)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    return proj

def compute_gradient(h, shift):
    n = h.size
    grad = np.zeros_like(h)
    if shift >= 0:
        # first term: indices i from 0 .. n-1-shift
        i1 = np.arange(0, n - shift)
        grad[i1] += (1 - h[i1 + shift])
        # second term: indices i from shift .. n-1
        i2 = np.arange(shift, n)
        grad[i2] -= h[i2 - shift]
    else:
        s = -shift
        # first term: i from s .. n-1
        i1 = np.arange(s, n)

def project_to_capped_simplex(v, target):
    # v: np.ndarray
    low = v.min() - 1.0
    high = v.max()
    # Binary search for lambda
    for _ in range(60):
        mid = (low + high) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    # Use high for final lambda (ensures sum <= target)
    proj = np.clip(v - high, 0.0, 1.0)
    # Fix any tiny residual
    residual = target - proj.sum()
    if abs(residual) > 1e-12:
        # add residual to a non-saturated element if possible
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if free.any():
            idx = np.argmax(free)  # pick first
            proj[idx] = np.clip(proj[idx] + residual, 0.0, 1.0)

            idx = np.argmin(proj) if residual < 0 else np.argmax(proj)
            proj[idx] = np.clip(proj[idx] + residual, 0.0, 1.0)
    return proj

import numpy as np
import time

# ---------- helper functions ----------

def _project_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {x | 0<=x_i<=1, sum(x)=target}."""
    # binary search for lambda such that sum(clip(v-lambda,0,1)) = target
    low = v.min() - 1.0           # forces all entries to 1 -> sum=n > target
    high = v.max()                # forces all entries to 0 -> sum=0 < target
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    proj = np.clip(v - high, 0.0, 1.0)  # sum <= target
    # fix tiny residuals
    residual = target - proj.sum()
    if abs(residual) > 1e-12:
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if free.any():
            idx = np.flatnonzero(free)[0]
            proj[idx] = np.clip(proj[idx] + residual, 0.0, 1.0)
        else:
            # all at bounds, adjust first element safely
            idx = 0
            proj[idx] = np.clip(proj[idx] + residual, 0.0, 1.0)
    return proj


def _compute_score(h: np.ndarray) -> float:
    """C5 value for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx


def _score_and_shift(h: np.ndarray):
    """Return (C5, shift index, full correlation array)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    dx = 2.0 / n
    return corr[idx] * dx, idx - (n - 1), corr


def _gradient(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the maximal overlap term w.r.t. h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        k = shift
        i1 = np.arange(0, n - k)
        g[i1] += (1.0 - h[i1 + k])
        i2 = np.arange(k, n)
        g[i2] -= h[i2 - k]
    else:
        s = -shift
        i1 = np.arange(s, n)
        g[i1] += (1.0 - h[i1 - s])
        i2 = np.arange(0, n - s)
        g[i2] -= h[i2 + s]
    return g


# ---------- main routine ----------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) trying to improve the C5 bound."""
    start = time.time()
    np.random.seed(seed)

    # ---- initialise h -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # fallback: random binary vector with exactly n/2 ones
        n = 80
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0
        np.random.shuffle(h0)

    n = h0.size
    target = n / 2.0                     # sum(h) must equal target
    h0 = _project_capped_simplex(h0, target)

    best_h = h0.copy()
    best_score = _compute_score(best_h)

    # ---- time budget handling -----------------------------------------
    time_limit = budget_s * 0.95          # keep a safety margin

    # ---- simulated annealing / hill climbing --------------------------
    max_iters = 200_000
    T0 = 0.01
    cur_h = best_h.copy()
    cur_score = best_score

    it = 0
    while it < max_iters and (time.time() - start) < time_limit:
        # pick a random 1 and a random 0 to swap
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # propose swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score, _, _ = _score_and_shift(cur_h)

        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            T = T0 * (1.0 - it / max_iters)
            if T < 1e-8:
                T = 1e-8
            if np.exp(-delta / T) > np.random.rand():
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        it += 1

    # ---- continuous refinement (projected sub‑gradient) ---------------
    gd_h = best_h.copy()
    gd_iter = 0
    max_gd_iter = 2000
    while gd_iter < max_gd_iter and (time.time() - start) < time_limit:

        grad = _gradient(gd_h, shift)
        step = 0.5 / np.sqrt(gd_iter + 1.0)

        gd_h = _project_capped_simplex(gd_h, target)

        new_score = _compute_score(gd_h)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_h = gd_h.copy()
        gd_iter += 1

    # final projection to guarantee feasibility
    best_h = _project_capped_simplex(best_h, target)
    best_score = _compute_score(best_h)

    return best_h, best_score, n
