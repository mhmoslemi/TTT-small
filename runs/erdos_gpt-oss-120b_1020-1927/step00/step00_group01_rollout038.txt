def hill_climb(h, target_sum, budget, start_time):
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(best_h)
    cur_h = h.copy()
    cur_val = best_val
    # parameters
    sigma = 0.2
    T_initial = 0.01
    T_final = 1e-6
    iter_count = 0
    while time.time() - start_time < budget:
        elapsed = time.time() - start_time
        frac = elapsed / budget
        # decaying sigma
        sigma = 0.2 * (1 - frac) + 0.001
        T = T_initial * (1 - frac) + T_final
        # propose
        v = cur_h + sigma * np.random.randn(n)
        # project
        proposal = project_onto_box_and_sum(v, target_sum)
        # evaluate
        val = compute_c5(proposal)
        # acceptance
        delta = val - cur_val
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            cur_h = proposal
            cur_val = val
            if val < best_val:
                best_val = val
                best_h = proposal.copy()
        iter_count += 1
        # occasional restart: every 5000 iters, random perturb to escape
        if iter_count % 5000 == 0:
            # small jitter from best
            cur_h = best_h + 0.05 * np.random.randn(n)
            cur_h = project_onto_box_and_sum(cur_h, target_sum)
            cur_val = compute_c5(cur_h)
    return best_h, best_val

def project_onto_box_and_sum(v, target, lower=0.0, upper=1.0, max_iter=60):
    # lambda search
    lo = -1.0
    hi = 1.0
    # Expand bounds until sum condition met.
    while True:
        cur = np.clip(v - lo, lower, upper)
        s = cur.sum()
        if s > target:
            lo -= 1.0
        else:
            break
    while True:
        cur = np.clip(v - hi, lower, upper)
        s = cur.sum()
        if s < target:
            hi += 1.0
        else:
            break
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        cur = np.clip(v - lam, lower, upper)
        s = cur.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, lower, upper)

def gradient_worst_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (n - 1)
    s_abs = abs(s)
    grad = np.zeros_like(h)
    if s_abs > 0:
        grad[:n - s_abs] += 1.0 - h[s_abs:]
        grad[s_abs:] -= h[:n - s_abs]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad, s_abs

import time
import math
import numpy as np

def compute_c5(h):
    """return max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def project_onto_box_and_sum(v, target, lower=0.0, upper=1.0, iters=60):
    """Euclidean projection of v onto [lower,upper]^n with ∑x = target."""
    lo = -1e5
    hi = 1e5
    for _ in range(iters):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, lower, upper)


def gradient_worst_shift(h):
    """gradient of the worst (largest) overlap integral with respect to h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    s = idx - (n - 1)               # signed shift
    s_abs = abs(s)

    grad = np.zeros_like(h)
    if s_abs == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        # contributions for i in [0, n-s_abs)
        grad[: n - s_abs] += 1.0 - h[s_abs:]
        # contributions for i in [s_abs, n)
        grad[s_abs:] -= h[: n - s_abs]
    return grad, s_abs


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # ---------------------------------------------------------
    #   initialise h
    # ---------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n0 = init.size
        n = max(200, n0)               # a modest size that still runs fast
        if n != n0:
            x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
            x = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x, x0, init)
        else:
            h = init.copy()
    else:
        n = 200
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))   # triangular start

    target_sum = n / 2.0
    h = project_onto_box_and_sum(h, target_sum)

    best_h = h.copy()
    best_val = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_val = best_val

    # ---------------------------------------------------------
    #   optimisation parameters
    # ---------------------------------------------------------
    step0 = 0.15          # initial step size for gradient moves
    temp0 = 0.025         # initial temperature for simulated annealing
    max_time = budget_s * 0.95   # keep a small margin for return handling
    it = 0

    while time.time() - start < max_time:
        elapsed = time.time() - start
        frac = elapsed / max_time
        # slowly shrink step and temperature
        step = step0 * (1.0 - frac) + 1e-5
        temp = temp0 * (1.0 - frac) + 1e-7

        # gradient step for the currently worst shift
        grad, s_abs = gradient_worst_shift(cur_h)

        # propose a gradient move
        v = cur_h - step * grad
        prop = project_onto_box_and_sum(v, target_sum)
        val = compute_c5(prop)

        delta = val - cur_val
        if delta < 0 or math.exp(-delta / temp) > np.random.rand():
            cur_h = prop
            cur_val = val
            if val < best_val:
                best_val = val
                best_h = prop.copy()

        # occasionally try a random jump to avoid stagnation
        if np.random.rand() < 0.001:
            cur_h = best_h + 0.05 * np.random.randn(n)
            cur_h = project_onto_box_and_sum(cur_h, target_sum)
            cur_val = compute_c5(cur_h)

        # a very cheap local improvement: move mass along the worst shift
        if s_abs > 0:
            diff = cur_h[: n - s_abs] - cur_h[s_abs:]
            i = int(np.argmax(diff))
            if diff[i] > 0:
                delta_move = min(step, diff[i] / 2.0,
                                 cur_h[i],
                                 1.0 - cur_h[i + s_abs])
                if delta_move > 0:
                    trial = cur_h.copy()
                    trial[i] -= delta_move
                    trial[i + s_abs] += delta_move
                    val2 = compute_c5(trial)
                    if val2 < cur_val:
                        cur_h = trial
                        cur_val = val2
                        if val2 < best_val:
                            best_val = val2
                            best_h = trial.copy()
        it += 1

    # final projection to guarantee feasibility
    best_h = project_onto_box_and_sum(best_h, target_sum)
    final_val = compute_c5(best_h)

    return best_h.tolist(), float(final_val), int(n)
