def compute_gradient(h, shift):
    n = len(h)
    if shift >= 0:
        left = np.empty_like(h)
        left[:n-shift] = h[shift:]
        left[n-shift:] = 0.0
        right = np.empty_like(h)
        right[shift:] = h[:n-shift]
        right[:shift] = 0.0
    else:
        s = -shift
        left = np.empty_like(h)
        left[s:] = h[:n-s]
        left[:s] = 0.0
        right = np.empty_like(h)
        right[:n-s] = h[s:]
        right[n-s:] = 0.0
    grad = 1.0 - (left + right)
    return grad

def project_onto_capped_simplex(v, target):
    # Implementation described above
    x = np.clip(v, 0.0, 1.0)
    cur = x.sum()
    if abs(cur - target) < 1e-12:
        return x
    # adjust
    # We'll use water-filling algorithm with loops up to few iterations
    max_iter = 1000
    for _ in range(max_iter):
        if cur > target:
            excess = cur - target
            mask = x > 0
            count = np.count_nonzero(mask)
            if count == 0:
                break
            delta = excess / count
            x[mask] -= delta
            x = np.clip(x, 0.0, 1.0)
        elif cur < target:
            deficit = target - cur
            mask = x < 1
            count = np.count_nonzero(mask)
            if count == 0:
                break
            delta = deficit / count
            x[mask] += delta
            x = np.clip(x, 0.0, 1.0)
        else:
            break
        new_cur = x.sum()
        if abs(new_cur - target) < 1e-12:
            break
        if abs(new_cur - cur) < 1e-12:
            # cannot adjust further due to bounds; break
            break
        cur = new_cur
    return x

import time
import numpy as np

def _project_onto_capped_simplex(v, target):
    """Project vector v onto {0<=x_i<=1, sum x_i = target}."""
    x = np.clip(v, 0.0, 1.0)
    s = x.sum()
    if abs(s - target) < 1e-12:
        return x
    # simple water‑filling loop – converges fast for modest sizes
    for _ in range(1000):
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        delta = diff / cnt
        # apply and re‑clip
        x[mask] += delta
        x = np.clip(x, 0.0, 1.0)
        s = x.sum()
    return x


def _c5_value(h, dx):
    """Return max overlap C5 for a given h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _gradient_worst_shift(h, shift):
    """Gradient of g_shift(h)=∑ h_i (1-h_{i+shift}) with zero padding."""
    n = h.shape[0]
    if shift >= 0:
        left = np.empty_like(h)
        left[: n - shift] = h[shift:]
        left[n - shift:] = 0.0
        right = np.empty_like(h)
        right[shift:] = h[: n - shift]
        right[:shift] = 0.0
    else:
        s = -shift
        left = np.empty_like(h)
        left[s:] = h[: n - s]
        left[:s] = 0.0
        right = np.empty_like(h)
        right[: n - s] = h[s:]
        right[n - s:] = 0.0
    return 1.0 - (left + right)
