import time, math
import numpy as np

def _project_to_sum_and_bounds(x, target):
    """Project x onto the hyperplane sum(x)=target with 0<=x<=1 using water‑filling."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


def _compute_c(h, dx):
    """Return max_{k} ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx


def _baseline_two_level(n, dx):
    """Simple two‑level step family: 1 on both edges, constant a in the middle."""
    w = n * 0.5
    best_c = np.inf
    best_h = None
    max_p = n // 4
    for p in range(max_p + 1):
        q = n - 2 * p
        if q <= 0:
            continue
        a = (w - 2 * p) / q
        if not (0.0 <= a <= 1.0):
            continue
        h = np.empty(n, dtype=np.float64)
        if p:
            h[:p] = 1.0
            h[-p:] = 1.0
        h[p:n - p] = a
        c = _compute_c(h, dx)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


def _optimize(n, init_h, time_budget, rng):
    """Simulated‑annealing style optimisation for a given length n."""
    dx = 2.0 / n
    target_sum = n * 0.5

    # initialise
    if init_h is not None:
        h = np.array(init_h, dtype=np.float64)
    else:
        # start from the simple two‑level family
        h, _ = _baseline_two_level(n, dx)

    # small random jitter and projection onto constraints
    h += rng.normal(scale=0.01, size=n)
    np.clip(h, 0.0, 1.0, out=h)
    h = _project_to_sum_and_bounds(h, target_sum)

    best_h = h.copy()
    best_c = _compute_c(best_h, dx)
    cur_c = best_c

    start = time.time()
    while time.time() - start < time_budget:
        # correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)

        # indices participating in that shift
        if shift >= 0:
            src = np.arange(0, n - shift)
            tgt = src + shift
        else:
            src = np.arange(-shift, n)
            tgt = src + shift          # shift is negative here

        contrib = h[src] * (1.0 - h[tgt])
        local = int(np.argmax(contrib))
        i = int(src[local])
        j = int(tgt[local])

        # move mass from i → j (reduces the offending term)
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max > 1e-12:
            delta = delta_max * rng.random() * 0.5 + 1e-12
            h[i] -= delta
            h[j] += delta
            new_c = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
            if new_c <= cur_c + 1e-13:
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                # possible uphill acceptance (simulated annealing)
                prog = (time.time() - start) / time_budget
                T = 0.02 * (1.0 - prog)          # linear cooling
                if T > 0 and rng.random() < math.exp(-(new_c - cur_c) / T):
                    cur_c = new_c
                    if new_c < best_c:
                        best_c = new_c
                        best_h = h.copy()
                else:
                    h[i] += delta
                    h[j] -= delta
        else:
            # fall back to a random feasible transfer
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            delta_max = min(1.0 - h[i], h[j])
            if delta_max > 1e-12:
                delta = delta_max * rng.random() * 0.5
                h[i] += delta
                h[j] -= delta
                new_c = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
                if new_c <= cur_c + 1e-13:
                    cur_c = new_c
                    if new_c < best_c:
                        best_c = new_c
                        best_h = h.copy()
                else:
                    # uphill with tiny probability
                    prog = (time.time() - start) / time_budget
                    T = 0.02 * (1.0 - prog)
                    if T > 0 and rng.random() < math.exp(-(new_c - cur_c) / T):
                        cur_c = new_c
                        if new_c < best_c:
                            best_c = new_c
                            best_h = h.copy()
                    else:
                        h[i] -= delta
                        h[j] += delta
        # end of iteration
    # final correction of the sum (tiny numerical drift)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        # try to fix by adjusting a single component inside bounds
        for idx in range(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break
        else:
            best_h = _project_to_sum_and_bounds(best_h, target_sum)

    return best_h, best_c

    start_all = time.time()
    total_budget = budget_s * 0.96   # reserve a tiny margin

    # candidate 1 : use the provided initial vector (if any)
