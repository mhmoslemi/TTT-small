def project_feasible(x, target, max_iter=1000):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        s = np.sum(x)
        if np.isclose(s, target, atol=1e-9):
            break
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to reduce sum
            mask = x > 0
        else:
            mask = x < 1
        m = np.count_nonzero(mask)
        if m == 0:
            # cannot adjust further
            break
        adjust = diff / m
        # subtract adjust (if diff>0 need reduce)
        if diff > 0:
            x[mask] -= adjust
        else:
            x[mask] -= adjust  # diff negative, adjust negative, adds
        # ensure bounds
        x = np.clip(x, 0.0, 1.0)
    # final correction: ensure sum is target exactly? small error may remain
    # distribute residual uniformly among non-bound indices.
    residual = target - np.sum(x)
    if abs(residual) > 1e-9:
        # find indices not at bound
        mask = (x > 0) & (x < 1)
        m = np.count_nonzero(mask)
        if m > 0:
            x[mask] += residual / m
            x = np.clip(x, 0.0, 1.0)
    return x

def grad_smooth(h):

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions -------------------------------------------------


def project_feasible(x, target):
    """Project x onto the box [0,1] and the hyperplane sum(x)=target."""
    x = np.clip(x, 0.0, 1.0)
    # iterative redistribution of the excess/deficit
    for _ in range(1000):
        s = np.sum(x)
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = x > 0.0
        else:
            mask = x < 1.0
        m = np.count_nonzero(mask)
        if m == 0:
            break
        adjust = diff / m
        x[mask] -= adjust
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction (if any)
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        m = np.count_nonzero(free)
        if m:
            x[free] += residual / m
            x = np.clip(x, 0.0, 1.0)
    return x


def compute_c5(h, dx):
    """Exact C5 for a given h."""
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def smooth_obj_grad(h, beta, target_sum, dx, lam):
    """
    Smooth approximation of max_{k} integral + quadratic penalty.
    Returns f, grad where grad is the gradient of f w.r.t. h.
    """
    n = h.shape[0]
    # correlation (already scaled by dx)
    c = np.correlate(h, 1 - h, mode="full") * dx
    maxc = np.max(c)

    # softmax weights
    shifted = c - maxc
    expv = np.exp(beta * shifted)
    sumexp = np.sum(expv)
    smooth = maxc + np.log(sumexp) / beta

    sumh = np.sum(h)
    penalty = lam * (sumh - target_sum) ** 2
    f = smooth + penalty

    w = expv / sumexp                # shape (2n-1)
    grad = np.zeros_like(h)

    # gradient of the smooth max (weighted sum of ∂c/∂h)
    for idx, ws in enumerate(w):
        if ws == 0.0:
            continue
        s = idx - (n - 1)            # shift value
        if s >= 0:
            i_end = n - s
            # i: 0 .. i_end-1
            grad[:i_end] += ws * (1 - h[s:s + i_end])
            grad[s:s + i_end] += ws * (-h[:i_end])
        else:
            shift = -s
            i_end = n - shift
            grad[:i_end] += ws * (-h[shift:shift + i_end])
            grad[shift:shift + i_end] += ws * (1 - h[:i_end])

    # penalty gradient
    grad += 2.0 * lam * (sumh - target_sum) * np.ones_like(h)
    return f, grad


def local_search(h, dx, time_budget):
    """
    Simple pairwise mass‑exchange hill‑climb that keeps the sum exactly constant.
    Stops after ``time_budget`` seconds.
    """
    n = h.size
    best_h = h.copy()
    best_val = compute_c5(best_h, dx)
    start = time.time()
    while time.time() - start < time_budget:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        # allowable positive/negative delta keeping both in [0,1]
        max_pos = min(1.0 - hi,   hj)   # increase i, decrease j
        max_neg = min(hi,        1.0 - hj)   # decrease i, increase j
        if max_pos <= 0.0 and max_neg <= 0.0:
            continue
        if np.random.rand() < 0.5:
            if max_pos <= 0.0:
                continue
            delta = np.random.rand() * max_pos
        else:
            if max_neg <= 0.0:
                continue
            delta = -np.random.rand() * max_neg
        new_h = best_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        val = compute_c5(new_h, dx)
        if val < best_val:
            best_h = new_h
            best_val = val
    return best_h

def run(seed=42, budget_s=1100, **_):
    np.random.seed(seed)
    start_time = time.time()

    # ---- obtain initial vector ------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=np.float64)   # provided by harness
    except NameError:
        init = None

    if init is None:
        n = 80
        h0 = np.full(n, 0.5, dtype=np.float64)   # already feasible
    else:
        n = init.size
        h0 = init.astype(np.float64)

    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    # ---- bookkeeping ---------------------------------------------------------
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    # ---- optimisation parameters ---------------------------------------------
    lam = 500.0                     # penalty weight
    betas = [30.0, 80.0, 200.0]     # increasing smoothness
    maxfun_phase = 3000
    bounds = [(0.0, 1.0)] * n

    def optimise_from(start_vec):
        nonlocal best_h, best_c5, start_time
        x = start_vec.copy()
        for beta in betas:
            # abort if time is running out
            if time.time() - start_time > budget_s * 0.94:
                break
            args = (beta, target_sum, dx, lam)
            res = minimize(
                fun=smooth_obj_grad,
                x0=x,
                args=args,
                method="L-BFGS-B",
                jac=True,
                bounds=bounds,
                options={"maxfun": maxfun_phase, "maxiter": maxfun_phase,
                         "ftol": 1e-12, "gtol": 1e-12, "disp": False}
            )
            x = project_feasible(res.x, target_sum)
            cur_c5 = compute_c5(x, dx)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = x.copy()
        return x

    # ---- first optimisation from the supplied (or default) start -------------
    optimise_from(best_h)

    # ---- random restarts ------------------------------------------------------
    while time.time() - start_time < budget_s * 0.85:
        # generate a random feasible vector (uniform then project)
        rnd = np.random.rand(n)
        rnd = project_feasible(rnd, target_sum)
        optimise_from(rnd)
        # stop early if we already beat the target record
        if best_c5 <= 0.3808:
            break

    # ---- local refinement -----------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 2.0:
        best_h = local_search(best_h, dx, time_budget=remaining * 0.8)

    final_c5 = compute_c5(best_h, dx)
    return best_h.tolist(), float(final_c5), n
