    h = np.clip(h, 0, 1)
    # water-filling
    lower = np.zeros_like(h)
    upper = np.ones_like(h)

        free = (h > 0) & (h < 1)
        free_count = free.sum()
        if free_count == 0:
            # cannot adjust further
            break
        diff = sum_target - h.sum()
        if abs(diff) < 1e-12:

def project_sum_bounds(x, target_sum):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # water-filling
    while True:
        # compute current sum
        cur_sum = x.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        # free indices: those not at bound 0 or 1
        free = (x > 0) & (x < 1)
        free_count = free.sum()
        if free_count == 0:
            # cannot adjust further
            break
        delta = diff / free_count
        # apply delta, but keep within bounds
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    return x
```

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    n = 200
    # create initial guess
    if 'initial_h_values' in globals():
        init = upsample(np.asarray(initial_h_values, dtype=float), n)
    else:
        xs = np.arange(n) * (2.0 / n)
        init = 0.5 + 0.5 * np.cos(np.pi * xs / 2)  # shape
    init = project_sum_bounds(init, n/2.0)
    best_h = init.copy()
    best_val = compute_c5(best_h)
    # main loop
    while time.time() - start < budget_s:
        # random restart: generate random feasible h (maybe perturb best)
        if np.random.rand() < 0.2:
            # fresh random
            h0 = np.random.rand(n)
            h0 = project_sum_bounds(h0, n/2.0)
        else:
            # perturb best
            h0 = best_h + np.random.normal(scale=0.05, size=n)
            h0 = np.clip(h0, 0, 1)
            h0 = project_sum_bounds(h0, n/2.0)
        # Run local hill climbing for a short time (e.g., 5 seconds)
        cur_h, cur_val = local_search(h0, min(10.0, budget_s - (time.time() - start)))
        if cur_val < best_val:
            best_h = cur_h
            best_val = cur_val
        # optionally break early if we reached target bound
        if best_val <= 0.3808:
            break
    best_h = project_sum_bounds(best_h, n/2.0)
    best_val = compute_c5(best_h)
    return (best_h, best_val, n)

    mask3 = xs > r + p
    if r > 0:
        h[mask3] = a * (2 - xs[mask3]) / r
    else:
        h[mask3] = 0
    return h

def local_search(h0, time_budget):
    n = len(h0)
    target_sum = n/2.0
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    start = time.time()
    checks = 0
    fail_counter = 0
    while time.time() - start < time_budget:
        # propose move
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        # compute allowed eps range
        max_up = min(cur_h[i], 1 - cur_h[j])
        max_down = min(cur_h[j], 1 - cur_h[i])
        # choose eps in [-max_down, max_up]
        low = -max_down
        high = max_up
        if low >= high:
            continue
        eps = np.random.uniform(low, high)
        if abs(eps) < 1e-12:
            continue
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        # evaluate
        new_val = compute_c5(new_h)
        if new_val < cur_val:
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
                fail_counter = 0
        else:
            # acceptance with temperature
            elapsed = time.time() - start
            temp = max(0.01 * (1 - elapsed / time_budget), 1e-5)
            if np.random.rand() < np.exp(-(new_val - cur_val) / temp):
                cur_h = new_h
                cur_val = new_val
        # periodic random restart if stagnated
        fail_counter += 1
        if fail_counter > 20000:
            # random perturbation of best_h
            new_h = best_h.copy()
            # add Gaussian noise
            noise = np.random.normal(scale=0.02, size=n)
            new_h += noise
            new_h = np.clip(new_h, 0, 1)
            new_h = project_sum_bounds(new_h, target_sum)
            cur_h = new_h
            cur_val = compute_c5(cur_h)
            fail_counter = 0
        # optional break after many iterations if best_val <= target (maybe 0.38)
        if best_val <= 0.3808:
            break
    best_h = project_sum_bounds(best_h, target_sum)
    best_val = compute_c5(best_h)
    return best_h, best_val

import time
import numpy as np

def compute_c5(h):
    """Return C5 bound for a discrete step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / h.size)

def project_sum_bounds(x, target):
    """Clip to [0,1] and adjust to hit the exact sum = target."""
    x = np.clip(x, 0.0, 1.0)
    # water‑filling style projection onto the simplex with box bounds
    while True:
        cur = x.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    return x

def upsample(old, n_new):
    """Linear interpolation of a step vector to a new length."""
    n_old = old.shape[0]
    xs_old = np.arange(n_old) * (2.0 / n_old)
    xs_new = np.arange(n_new) * (2.0 / n_new)
    return np.interp(xs_new, xs_old, old)

def piecewise_h(a, n):
    """Build the symmetric piecewise‑linear shape parameterised by a."""
    r = 2.0 - 1.0 / a          # ramp length
    p = 2.0 / a - 2.0          # plateau length
    xs = np.arange(n) * (2.0 / n)
    if r <= 1e-12:
        h = np.full(n, a)
    else:
        h = np.empty_like(xs)
        mask1 = xs < r
        h[mask1] = a * xs[mask1] / r
        mask2 = (xs >= r) & (xs <= r + p)
        h[mask2] = a
        mask3 = xs > r + p
        h[mask3] = a * (2.0 - xs[mask3]) / r
    return h

def local_search(start_h, time_budget):
    """Hill‑climbing / simulated‑annealing on the max‑overlap objective."""
    n = start_h.shape[0]
    target = n / 2.0
    best_h = start_h.copy()
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    start = time.time()
    stagn = 0
    max_stagn = 20000
    while time.time() - start < time_budget:
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # allowed epsilon so that both stay in [0,1]
        up = min(cur_h[i], 1.0 - cur_h[j])     # move from i to j
        down = min(cur_h[j], 1.0 - cur_h[i])   # move from j to i (negative eps)
        lo, hi = -down, up
        if lo >= hi:
            continue
        eps = np.random.uniform(lo, hi)
        if abs(eps) < 1e-14:
            continue
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_val = compute_c5(new_h)
        if new_val < cur_val:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
                stagn = 0
        else:
            # occasional uphill acceptance
            elapsed = time.time() - start
            temp = max(0.02 * (1.0 - elapsed / time_budget), 1e-5)
            if np.random.rand() < np.exp(-(new_val - cur_val) / temp):
                cur_h, cur_val = new_h, new_val
            stagn += 1
        if stagn >= max_stagn:
            # random perturbation around the current best
            new_h = best_h + np.random.normal(scale=0.02, size=n)
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = project_sum_bounds(new_h, target)
            cur_h, cur_val = new_h, compute_c5(new_h)
            stagn = 0
        if best_val <= 0.3808:   # early stop if target achieved
            break
    best_h = project_sum_bounds(best_h, target)
    best_val = compute_c5(best_h)
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_all = time.time()

    # ---- 1. scan the one‑parameter family (piecewise linear) ----
    n_scan = 400
    best_a = 0.5
    best_val = np.inf
    for a in np.linspace(0.5, 1.0, 501):
        h = piecewise_h(a, n_scan)
        h = project_sum_bounds(h, n_scan / 2.0)
        val = compute_c5(h)
        if val < best_val:
            best_val, best_a = val, a
    # finer local scan around the best a
    for a in np.linspace(max(0.5, best_a - 0.01), min(1.0, best_a + 0.01), 101):
        h = piecewise_h(a, n_scan)
        h = project_sum_bounds(h, n_scan / 2.0)
        val = compute_c5(h)
        if val < best_val:
            best_val, best_a = val, a

    # ---- 2. build a higher‑resolution starting vector ----
    n_opt = 800
    h_start = piecewise_h(best_a, n_opt)
    h_start = project_sum_bounds(h_start, n_opt / 2.0)

    # mix with the provided initial construction if it exists
    if 'initial_h_values' in globals():

            init = np.asarray(initial_h_values, dtype=float)
            h_init = upsample(init, n_opt)
            h_init = project_sum_bounds(h_init, n_opt / 2.0)
            if compute_c5(h_init) < compute_c5(h_start):
                h_start = h_init
        except Exception:
            pass  # ignore any unexpected shape issues

    # ---- 3. local optimisation within the remaining time ----
    elapsed = time.time() - start_all
    remaining = max(0.0, budget_s - elapsed)

        best_val = compute_c5(h_best)

    h_best = project_sum_bounds(h_best, h_best.shape[0] / 2.0)

    return (h_best, final_val, int(h_best.shape[0]))
