def project_to_hyperplane_box(y, target):
    n = len(y)
    x = y.copy()
    # initial active set is all indices
    active = np.arange(n)
    fixed = np.zeros(n, dtype=bool)
    while True:
        if active.size == 0:
            break
        # compute lambda needed to satisfy sum constraint
        sum_fixed = x[fixed].sum()
        lambda_val = (target - sum_fixed) / active.size
        # apply to active entries
        x[active] = y[active] + lambda_val
        # clip to bounds
        over_low = x < 0
        over_high = x > 1
        # If nothing violates, break
        if not np.any(over_low) and not np.any(over_high):
            break
        # For any violations, fix them to bound and mark fixed; adjust y for active set
        to_fix_low = active[over_low[active]]
        to_fix_high = active[over_high[active]]
        # set them to 0 or 1
        x[to_fix_low] = 0
        x[to_fix_high] = 1
        # mark as fixed
        fixed[to_fix_low] = True
        fixed[to_fix_high] = True
        # remove from active
        mask = np.ones_like(active, dtype=bool)
        mask[np.isin(active, np.concatenate([to_fix_low, to_fix_high]))] = False
        active = active[mask]
        # update y to original y of remaining active (since you might have changed x)
        # Actually we want to recompute lambda based on y for remaining.
        # Keep original y unchanged; but for active entries we still want to shift from original y.
    # ensure sum meets target (maybe due to numeric)
    # adjust slight remainder by adding to any free entry
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff to non-fixed?
        # pick index with room
        for i in range(n):
            if (diff > 0 and x[i] < 1) or (diff < 0 and x[i] > 0):
                step = min(diff, 1 - x[i]) if diff > 0 else max(diff, -x[i])
                x[i] += step
                diff -= step

    return x

def subgradient(h):
    n = len(h)
    dx = 2.0/n
    conv = np.correlate(h, 1 - h, mode='full') * dx
    max_val = conv.max()
    # tolerance
    tol = 1e-12
    idxs = np.where(conv >= max_val - 1e-12)[0]
    lags = idxs - (n-1)
    grad = np.zeros_like(h)
    for s in lags:
        if s >= 0:
            # term h[j + s] for j from 0..n-1-s
            grad[:n-s] -= dx * h[s:]  # for j in 0..n-1-s, add -dx*h[j+s]

            grad[t:] -= dx * h[:n-t]
            # j - s = j + t: valid for j <= n-1 - t

Goal: Solve for variables x_i in [0,1] with sum = target that are closest to y (the unprojected). This is a standard problem. The solution is like truncated shift: x_i = min(1, max(0, y_i + λ)) where λ is chosen s.t. sum x_i = target. That's projection onto a box and hyperplane using "clipping with a shift". Indeed the Euclidean projection onto a hyperplane intersecting a box is given by shifting all elements by same λ and then clipping to bounds. That's because the problem is minimizing ||x - y||^2 s.t. sum x = target & bounds. The solution indeed is x_i = Π_{[0,1]} (y_i + λ) where λ is found such that sum_i x_i = target. This is a monotonic function of λ, can be found via binary search because sum after clipping is monotonic non-decreasing in λ. Indeed if we increase λ, each term y_i+λ increases until hitting 1; otherwise saturates at 0 when low. So we can find λ using binary search over range [-max(y_i), 1 - min(y_i)] but more precisely [-max(y_i), 1 - min(y_i)] possibly.

So we can implement binary search for λ:

- set lo = -max(y) - 1 (ensures all shifted values are <=0, sum=0). Actually shift large negative gives all <0 => after clipping to 0, sum=0.

- set hi = 1 - min(y) + 1 (ensures all shifted >1 => after clipping to 1, sum=n). Actually we need hi large enough to produce sum = n (max). So we can set hi = 1 - min(y) + 1.

- Then binary search λ until sum_i clip(y_i + λ) approx target.

def project_onto_simplex_box(y, target):
    n = len(y)
    # If sum y already equals target and all in [0,1], maybe just clip
    # but we still need to enforce constraints.
    # Use binary search for lambda.
    lo = -np.max(y) - 1.0  # ensure y+lo <= -something
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(60):
        lam = (lo + hi)/2.0
        x = np.clip(y + lam, 0, 1)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    x = np.clip(y + lo, 0, 1)
    # due to rounding, may not equal target exactly; adjust diff
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff proportionally to entries not at bounds
        # find indices where 0 < x < 1
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size > 0:
            # add diff equally
            add = diff / free.size
            # may exceed bounds; clip
            x[free] += add
            # clip again
            x = np.clip(x, 0, 1)
            # if still diff not fixed, adjust iteratively (repeat)
        else:
            # if no free variables (rare), just adjust arbitrary min bound variable
            idx = np.argmin(x) if diff > 0 else np.argmax(x)
            x[idx] = np.clip(x[idx] + diff, 0, 1)
    # final adjust for rounding
    # Ensure sum close within tolerance
    # Possibly re-balance tiny difference by distributing across free
    return x

def project_box_sum(y, target):
    # y is array
    # Find λ such that sum(clip(y + λ, 0, 1)) = target
    # Use binary search between low and high bounds
    low = -np.max(y) - 1.0
    high = 1.0 - np.min(y) + 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        x = np.clip(y + mid, 0, 1)
        s = x.sum()
        if s > target:
            high = mid
        else:
            low = mid
    # final x
    x = np.clip(y + low, 0, 1)
    # due to numeric, correct small diff
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # adjust small # while preserving bounds
        # try to distribute to non-fixed entries
        # find indices where x not at bounds
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size > 0:
            # distribute equally
            add = diff / free.size
            x[free] += add
            x = np.clip(x, 0, 1)
        else:
            # adjust any index that can absorb
            idx = np.argmin(x) if diff > 0 else np.argmax(x)
            x[idx] = np.clip(x[idx] + diff, 0, 1)
    return x

import time, math, random
import numpy as np

def compute_c5(h):
    """Exact evaluation used by the harness."""
    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full").max() * dx

def project_box_sum(y, target):
    """Euclidean projection of y onto {0≤x≤1, sum x = target}."""
    lo = -np.max(y) - 1.0       # guarantees all entries ≤0 after shift
    hi = 1.0 - np.min(y) + 1.0  # guarantees all entries ≥1 after shift
    for _ in range(55):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + lo, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size:
            add = diff / free.size
            x[free] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            idx = int(np.argmin(x) if diff > 0 else np.argmax(x))
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def smooth_gradient(h, dx, alpha):
    """Gradient of (1/α)·log Σ exp(α·O_k) where O_k = ∫h(1‑h(·+k))."""
    n = h.size
    conv = np.correlate(h, h, mode="full")          # Σ h_i·h_{i+k}
    O = 1.0 - dx * conv                              # overlap values for each lag
    # soft‑max weights
    shift = O.max()
    exp_a = np.exp(alpha * (O - shift))
    w = exp_a / exp_a.sum()
    grad = np.zeros_like(h)
    off = n - 1
    for idx, wk in enumerate(w):
        s = idx - off
        if s == 0:
            continue
        if s > 0:
            grad[: n - s] += wk * (-dx) * h[s:]
            grad[s:]      += wk * (-dx) * h[: n - s]
        else:
            t = -s
            grad[t:] += wk * (-dx) * h[: n - t]
            grad[: n - t] += wk * (-dx) * h[t:]
    return grad

def optimize_h(h0, budget, start_time):
    """Gradient‑descent on a smoothed max‑overlap objective."""
    n = h0.size
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    max_iter = 15000
    alpha = 12.0          # initial smoothing
    lr0 = 0.45
    for it in range(1, max_iter + 1):
        if time.time() - start_time > budget * 0.94:
            break
        # occasionally sharpen the approximation
        if it % 3000 == 0 and alpha < 200:
            alpha *= 1.7
        g = smooth_gradient(h, dx, alpha)
        step = lr0 / (it ** 0.58)
        h_new = h - step * g
        h_new = project_box_sum(h_new, target)
        cur = compute_c5(h_new)
        if cur < best_score:
            best_score = cur
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

    n = h.size

    # ---- initial construction ------------------------------------------------
    try:                         # provided by the harness
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:            # fallback – half ones, half zeros
        n0 = 45
        h0 = np.concatenate([np.ones(n0 // 2), np.zeros(n0 - n0 // 2)])
    n0 = h0.size
    target0 = n0 / 2.0
    h0 = project_box_sum(h0, target0)

    # ---- first optimisation on the original grid ---------------------------
    best_h, best_score = optimize_h(h0, budget_s, start)

    # ---- up‑sample and refine (twice, if time permits) ----------------------
    for factor in (2, 2):                     # 45→90→180 points
        if time.time() - start > budget_s * 0.94:
            break
        n_old = best_h.size
        n_new = n_old * factor
        x_old = np.linspace(0, 2, n_old, endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_box_sum(h_up, n_new / 2.0)
        h_up, sc_up = optimize_h(h_up, budget_s, start)
        if sc_up < best_score:
            best_h, best_score = h_up, sc_up

    # ---- final fine‑tuning with random moves -------------------------------
    if time.time() - start < budget_s * 0.98:
        best_h, best_score = hill_climb(best_h, budget_s, start)

    # ---- final sanity projection --------------------------------------------
    n_final = best_h.size
    best_h = project_box_sum(best_h, n_final / 2.0)
    best_score = compute_c5(best_h)

    return (best_h, float(best_score), n_final)
