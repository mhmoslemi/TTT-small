import time
import random
import numpy as np

def _c5_score(h):
    """Return C5 for a given h (array of length n)."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate implements cross‑correlation = convolution with reversed second argument
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _random_h(n):
    """Generate a feasible h with sum = n/2 (respecting 0/1 bounds)."""
    h = np.zeros(n, dtype=float)
    ones = n // 2
    idx = np.random.choice(n, ones, replace=False)
    h[idx] = 1.0
    if n % 2:                         # odd n → need a 0.5 entry to reach n/2
        remaining = np.setdiff1d(np.arange(n), idx)
        h[np.random.choice(remaining)] = 0.5
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Optimise the Erdős C5 bound."""
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    total_time = budget_s * 0.95          # keep a small safety margin

    # ------------------------------------------------------------------
    # initialise problem size and starting vector
    # ------------------------------------------------------------------
    try:
        h = np.array(initial_h_values, copy=True, dtype=float)   # provided by harness
        n = h.shape[0]
    except NameError:
        n = 48                              # fallback size (even → binary feasible)
        h = _random_h(n)

    # make sure the sum constraint holds exactly (possible tiny drift)
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # simple fix: scale the first non‑boundary entry
        for i in range(n):
            if 0.0 < h[i] < 1.0:
                h[i] += target_sum - cur_sum
                break
        # final safeguard – clip and re‑normalise on the simplex with bounds
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - target_sum
        if abs(excess) > 1e-12:
            # distribute excess uniformly over interior entries
            interior = np.where((h > 0.0) & (h < 1.0))[0]
            if interior.size:
                h[interior] -= excess / interior.size
                h = np.clip(h, 0.0, 1.0)

    # ------------------------------------------------------------------
    # optimisation parameters

    cur_score = _c5_score(h)

    while True:

        j = np.random.randint(n - 1)
        if j >= i:

        direction = np.random.randint(2)
