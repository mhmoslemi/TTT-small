def propose_move(h, step=0.1):
    h_new = h.copy()
    # random number of indices to change, e.g., up to 5
    m = np.random.randint(1, 5)
    idx = np.random.choice(len(h), size=m, replace=False)
    # generate random deltas in [-step, step]
    deltas = np.random.uniform(-step, step, size=m)
    h_new[idx] += deltas
    # clip to [0,1]
    np.clip(h_new, 0, 1, out=h_new)
    # adjust sum to target
    target = len(h) / 2
    current = h_new.sum()
    diff = current - target
    if abs(diff) > 1e-12:
        # distribute diff across indices with slack
        if diff > 0:
            # need to reduce sum
            # find indices with >0
            idx_pos = np.where(h_new > 1e-12)[0]
            # randomly order
            np.random.shuffle(idx_pos)
            for j in idx_pos:
                dec = min(diff, h_new[j])
                h_new[j] -= dec
                diff -= dec
                if diff <= 1e-12:
                    break
        else:
            # need to increase sum
            need = -diff
            idx_neg = np.where(h_new < 1 - 1e-12)[0]
            np.random.shuffle(idx_neg)
            for j in idx_neg:
                inc = min(need, 1 - h_new[j])
                h_new[j] += inc
                need -= inc
                if need <= 1e-12:
                    break
    # final adjustment: maybe slight numeric diff
    # correct any leftover diff using tiny epsilon distribution
    # fudge: compute final diff, subtract through first index
    final_diff = h_new.sum() - target
    if abs(final_diff) > 1e-12:
        # adjust first entry within bounds
        j = 0
        h_new[j] -= final_diff
        # clip again if needed
        h_new[j] = min(max(h_new[j], 0.0), 1.0)
    return h_new
```

    idx_max = int(np.argmax(corr))

    delta = rng.uniform(0, delta_max)
    h_new = h.copy()
    h_new[i_sel] -= delta
    h_new[j_sel] += delta
    return h_new

import time
import numpy as np

def compute_c5(h: np.ndarray, dx: float) -> float:
    """Maximum overlap integral for a step function given by h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def propose_pair_move(h: np.ndarray, rng: np.random.Generator, max_delta: float) -> np.ndarray:
    """Transfer a random amount between two entries while preserving the sum."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # random direction
    if rng.random() < 0.5:
        # i -> j
        delta_max = min(max_delta, h[i], 1.0 - h[j])
        if delta_max <= 0.0:
            return h.copy()
        delta = rng.uniform(0.0, delta_max)
        out = h.copy()
        out[i] -= delta
        out[j] += delta
    else:
        # j -> i
        delta_max = min(max_delta, h[j], 1.0 - h[i])
        if delta_max <= 0.0:
            return h.copy()
        delta = rng.uniform(0.0, delta_max)
        out = h.copy()
        out[i] += delta
        out[j] -= delta
    return out


def propose_targeted_move(h: np.ndarray, rng: np.random.Generator, max_delta: float) -> np.ndarray:
    """Move mass from a high‑value entry to its worst‑overlap partner."""
    n = h.shape[0]
    # identify the shift with the largest overlap
    corr = np.correlate(h, 1 - h, mode="full")
    idx_max = int(np.argmax(corr))
    shift = idx_max - (n - 1)
    if shift == 0:
        # no useful direction, fall back
        return propose_pair_move(h, rng, max_delta)

    i_all = np.arange(n)
    j_all = i_all + shift
    valid = (j_all >= 0) & (j_all < n)
    if not np.any(valid):
        return propose_pair_move(h, rng, max_delta)

    i_valid = i_all[valid]
    j_valid = j_all[valid]
    contrib = h[i_valid] * (1.0 - h[j_valid])
    # choose the pair contributing most to the current maximum
    best_idx = int(np.argmax(contrib))
    i_sel = i_valid[best_idx]
    j_sel = j_valid[best_idx]

    delta_max = min(max_delta, h[i_sel], 1.0 - h[j_sel])
    if delta_max <= 0.0:
        return propose_pair_move(h, rng, max_delta)

    delta = rng.uniform(0.0, delta_max)
    out = h.copy()
    out[i_sel] -= delta
    out[j_sel] += delta
    return out


def propose_candidate(h: np.ndarray, rng: np.random.Generator, max_delta: float) -> np.ndarray:
    """Randomly choose a proposal type."""
    if rng.random() < 0.55:
        return propose_targeted_move(h, rng, max_delta)
    else:
        return propose_pair_move(h, rng, max_delta)

    return (best_h.tolist(), float(final_score), n)
