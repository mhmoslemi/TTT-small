def propose_swap(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_pos = np.argmax(corr)
    shift = max_pos - (n-1)
    bad_ones = []
    good_zeros = []
    if shift >= 0:
        for i in np.where(h == 1)[0]:
            j = i + shift
            if j < n and h[j] == 0:
                bad_ones.append(i)

def simulated_annealing(h_init, max_time, target_c5=0.3808):
    n = len(h_init)
    curr_h = h_init.copy()
    curr_c5 = evaluate_c5(curr_h)
    best_h = curr_h.copy()
    best_c5 = curr_c5
    start = time.time()
    T0 = 0.5
    Tmin = 1e-4
    while True:
        elapsed = time.time() - start
        if elapsed > max_time:
            break
        # thermal schedule
        T = max(Tmin, T0 * (1 - elapsed / max_time))
        # propose move
        i, j = propose_swap_targeted(curr_h)
        # produce new
        h_new = curr_h.copy()
        h_new[i] = 0
        h_new[j] = 1
        new_c5 = evaluate_c5(h_new)
        delta = new_c5 - curr_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            curr_h = h_new
            curr_c5 = new_c5
            if curr_c5 < best_c5:
                best_h = curr_h.copy()
                best_c5 = curr_c5
                if best_c5 <= target_c5:
                    # maybe break early
                    # but we continue to try to improve further; optionally break

    return pairs

import time
import numpy as np

# ---------- evaluation ----------
def evaluate_c5(h: np.ndarray) -> float:
    """Return the C5 value for a step function h on [0,2] discretised."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


# ---------- targeted swap ----------
def propose_swap_targeted(h: np.ndarray) -> tuple:
    """Pick a swap (i_one, j_zero) that tries to reduce the worst shift."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = int(np.argmax(corr))
    shift = max_idx - (n - 1)          # shift = k  (positive = right)

    if shift >= 0:
        rng = np.arange(0, n - shift)
        bad = rng[(h[rng] == 1) & (h[rng + shift] == 0)]
        good = rng[(h[rng] == 0) & (h[rng + shift] == 1)]
    else:
        s = -shift
        rng = np.arange(s, n)
        bad = rng[(h[rng] == 1) & (h[rng - s] == 0)]
        good = rng[(h[rng] == 0) & (h[rng - s] == 1)]

    if bad.size and good.size:
        i = np.random.choice(bad)
        j = np.random.choice(good)
        return int(i), int(j)

    # fallback random swap
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    i = int(np.random.choice(ones))
    j = int(np.random.choice(zeros))
    return i, j


# ---------- simulated annealing ----------
def simulated_annealing(h_start: np.ndarray, time_budget: float, target: float = 0.3808):
    """Simple SA on binary vectors keeping exactly sum(h)=n/2."""
    n = h_start.shape[0]
    half = n // 2
    curr = h_start.copy()
    curr_val = evaluate_c5(curr)
    best = curr.copy()
    best_val = curr_val

    start = time.time()
    T0 = 0.5
    Tmin = 1e-4

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_budget:
            break
        T = max(Tmin, T0 * (1 - elapsed / time_budget))

        i, j = propose_swap_targeted(curr)
        cand = curr.copy()
        cand[i] = 0
        cand[j] = 1
        cand_val = evaluate_c5(cand)

        delta = cand_val - curr_val
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            curr, curr_val = cand, cand_val
            if curr_val < best_val:
                best, best_val = curr.copy(), curr_val
                if best_val <= target:
                    # good enough – keep going a little longer for possible improvement
                    pass
    return best, best_val


# ---------- greedy hill climbing ----------
def hill_climb(h_start: np.ndarray, max_iter: int = 2000):
    """Deterministic improvement by swapping any pair that lowers C5."""
    best = h_start.copy()
    best_val = evaluate_c5(best)
    n = best.shape[0]

    for _ in range(max_iter):
        improved = False
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        for i in ones:
            for j in zeros:
                cand = best.copy()
                cand[i] = 0
                cand[j] = 1
                cand_val = evaluate_c5(cand)
                if cand_val < best_val - 1e-9:
                    best, best_val = cand, cand_val
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best, best_val

    """Try integer programming (h binary, maximize minimal overlap)."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None

    half = n // 2
    # build pair lists for each shift
    shift_pairs = {}
    for k in range(-n + 1, n):
        pairs = []
        if k >= 0:
            for i in range(n - k):
                pairs.append((i, i + k))
        else:
            s = -k
            for i in range(s, n):
                pairs.append((i, i - s))
        shift_pairs[k] = pairs

    h = cp.Variable(n, boolean=True)
    s = cp.Variable()               # minimum overlap
    constraints = [cp.sum(h) == half, s >= 0, s <= half]

    # auxiliary y variables for products h_i * h_{i+k}

        if not pairs:
            continue
        idx_i = np.array([i for i, _ in pairs], dtype=int)
        idx_j = np.array([j for _, j in pairs], dtype=int)
        y = cp.Variable(len(pairs), boolean=True)
        constraints += [
            y <= h[idx_i],
            y <= h[idx_j],
            y >= h[idx_i] + h[idx_j] - 1,
            cp.sum(y) >= s
        ]
    prob = cp.Problem(cp.Maximize(s), constraints)

    # try several solvers; most environments have CBC or GLPK_MI
    for solver in (cp.CBC, cp.GLPK_MI, cp.ECOS_BB):
        try:
            prob.solve(solver=solver, verbose=False, max_seconds=max_seconds)
            if prob.status in ("optimal", "optimal_inaccurate"):
                break
        except Exception:
            continue

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None, None

    h_val = np.rint(h.value).astype(int)
    # ensure exact half ones (repair tiny rounding errors)
    cur = int(h_val.sum())
    if cur != half:
        diff = cur - half
        if diff > 0:
            idx = np.where(h_val == 1)[0]
            np.random.shuffle(idx)
            h_val[idx[:diff]] = 0
        else:
            idx = np.where(h_val == 0)[0]
            np.random.shuffle(idx)
            h_val[idx[: -diff]] = 1

    s_val = float(s.value)
    c5 = (half - s_val) * (2.0 / n)
    return h_val.astype(float), c5


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # ----- choose a working resolution -----
    try:
        base_n = len(initial_h_values)          # provided by the harness
    except Exception:
        base_n = 40

    # we will try a modest MILP size first, then fall back to SA on a larger grid
    milp_n = 60
    if milp_n % 2 != 0:
        milp_n += 1
    # make sure we don't exceed the global budget
    remaining = budget_s - (time.time() - start_time)
    milp_time = min(300.0, remaining * 0.5)       # give up to 5 min for MILP

    best_h, best_c5 = milp_optimize(milp_n, max_seconds=milp_time)
    if best_h is not None:
        # MILP succeeded – keep this result
        elapsed = time.time() - start_time
        # attempt a quick refinement with hill climb if time permits
        if budget_s - elapsed > 5.0:
            refined_h, refined_c5 = hill_climb(best_h, max_iter=1000)
            if refined_c5 < best_c5:
                best_h, best_c5 = refined_h, refined_c5
        return best_h, best_c5, milp_n

    # ----- if MILP failed, use simulated annealing on a larger grid -----
    # decide final number of points (even, not too large)
    factor = 1
    while base_n * factor < 80 and base_n * factor <= 200:
        factor *= 2
    n_points = base_n * factor
    if n_points % 2:
        n_points += 1
    half = n_points // 2

    # create an initial binary vector
    if 'initial_h_values' in globals():
        # repeat the provided pattern to match the new size
        repeat = n_points // base_n
        h0 = np.repeat(initial_h_values, repeat)
        h_bin = (h0 >= 0.5).astype(float)
        cur = int(h_bin.sum())
        if cur > half:
            idx = np.where(h_bin == 1)[0]
            np.random.shuffle(idx)
            h_bin[idx[:cur - half]] = 0
        elif cur < half:
            idx = np.where(h_bin == 0)[0]
            np.random.shuffle(idx)
            h_bin[idx[:half - cur]] = 1
    else:
        h_bin = np.zeros(n_points)
        h_bin[:half] = 1
        np.random.shuffle(h_bin)

    # run SA within the remaining budget
    sa_budget = max(5.0, budget_s - (time.time() - start_time) - 5.0)
    best_h, best_c5 = simulated_annealing(h_bin, sa_budget)

    # final local hill climbing if time left
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        hc_h, hc_c5 = hill_climb(best_h, max_iter=2000)
        if hc_c5 < best_c5:
            best_h, best_c5 = hc_h, hc_c5

    # assure integral constraint exactly (tiny numerical drift possible)
    dx = 2.0 / n_points
    total = best_h.sum() * dx
    if abs(total - 1.0) > 1e-8:
        # adjust by flipping a tiny number of entries
        diff = int(round((total - 1.0) / dx))
        if diff > 0:
            ones = np.where(best_h == 1)[0]
            np.random.shuffle(ones)
            best_h[ones[:diff]] = 0
        elif diff < 0:
            zeros = np.where(best_h == 0)[0]
            np.random.shuffle(zeros)
            best_h[zeros[: -diff]] = 1

    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n_points
