def project_onto_bounds_and_sum(v, s):
    # clip to [0,1]
    v = np.clip(v, 0, 1)
    # successive adjustment
    # solve: find lambda such that sum(clip(v + lambda, 0, 1)) = s
    # This is typical.
    # We can use bisection on lambda, as the sum of clipped function is monotonic in lambda.
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        lam = (lo+hi)/2
        w = v + lam
        w = np.clip(w, 0, 1)
        cur = w.sum()
        if cur < s:
            lo = lam
        else:
            hi = lam
    return w

def shift_array(arr, shift):
    # shift>0: shift left by shift (i.e., arr shifted to earlier indices)
    # returns array of same length
    n = len(arr)
    if shift >= 0:
        out = np.empty(n, dtype=arr.dtype)
        if shift < n:

            out[n-shift:] = 0.0
        else:
            out[:] = 0.0

        out = np.empty(n, dtype=arr.dtype)
        if t < n:
            out[t:] = arr[:n-t]
            out[:t] = 0.0
        else:
            out[:] = 0.0
    return out

def weighted_shifts(p, h):
    n = h.shape[0]
    center = n - 1
    result = np.zeros_like(h)
    for idx, weight in enumerate(p):
        if weight == 0.0:
            continue
        m = idx - center

        out_pos = shift_array(h, m)
        out_neg = shift_array(h, -m)
        result += weight * (out_pos + out_neg)
    return result

def project_onto_bounds_and_sum(v, target_sum, tol=1e-12, max_iter=100):

        if s < target_sum:
            lo = lam

            hi = lam

    return w

def _shift_array(arr: np.ndarray, shift: int) -> np.ndarray:
    """Shift `arr` by `shift` places. Positive shift moves elements left,
    negative shift moves them right. Vacated entries are filled with 0."""
    n = arr.shape[0]
    out = np.zeros_like(arr)
    if shift > 0:
        out[:n - shift] = arr[shift:]
    elif shift < 0:
        t = -shift
        out[t:] = arr[:n - t]
    else:
        out[:] = arr
    return out


def _weighted_shifts(p: np.ndarray, h: np.ndarray) -> np.ndarray:
    """Return Σ_m p[m]·(h shifted by +m + h shifted by –m)."""
    n = h.shape[0]
    L = 2 * n - 1                     # length of p
    # convolution gives Σ_m p[m]·h_{j‑m}
    conv1 = np.convolve(p, h, mode="full")
    conv2 = np.convolve(p[::-1], h, mode="full")
    start = L - 1                     # index that aligns with j=0
    end = start + n
    return conv1[start:end] + conv2[start:end]


def _project_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto the box [0,1] and the affine subspace Σv = target."""
    # Clamp to [0,1] first (does not affect the monotone search later)
    v = np.clip(v, 0.0, 1.0)
    # Find λ such that Σ clip(v+λ,0,1) = target.
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = v + lam
        np.clip(w, 0.0, 1.0, out=w)
        s = w.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    w = v + lam
    np.clip(w, 0.0, 1.0, out=w)
    return w


def _cmax(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step function `h`."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _softmax_weights_and_grad(h: np.ndarray, alpha: float):
    """Soft‑max weights for the correlation vector and its gradient."""
    raw = np.correlate(h, 1.0 - h, mode="full")
    m = raw.max()
    # Stabilised exponentials
    ex = np.exp(alpha * (raw - m))
    p = ex / ex.sum()
    grad = 1.0 - _weighted_shifts(p, h)
    # smooth approximation of the max (not needed for the update itself)
    smooth = m + np.log(ex.sum()) / alpha
    return p, grad, smooth


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function `h` on [0,2] that yields a low C₅ bound.

    Returns
    -------
    (h_values, c5_bound, n_points)
    """
    random.seed(seed)
    np.random.seed(seed)

    # ---- initial data -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: half‑ones, half‑zeros (satisfies the integral)
        n = 80
        h = np.concatenate([np.ones(n // 2), np.zeros(n // 2)])
    n = h.size
    target_sum = n / 2.0
    h = _project_bounds_sum(h, target_sum)

    best_h = h.copy()
    best_val = _cmax(best_h)

    start_time = time.time()
    time_limit = budget_s * 0.94  # keep a margin for final polishing

    # ---- parameters ---------------------------------------------------
    alpha = 5.0          # soft‑max sharpness
    lr = 0.25            # learning rate for projected gradient steps
    no_improve = 0
    improve_tol = 1e-6

    # ---- projected gradient descent ------------------------------------
    while time.time() - start_time < time_limit:
        # current max value
        cur_val = _cmax(h)

        # compute soft‑max weights, gradient and a smooth objective
        _, grad, _ = _softmax_weights_and_grad(h, alpha)

        # gradient step with projection
        h_new = h - lr * grad
        h_new = _project_bounds_sum(h_new, target_sum)

        new_val = _cmax(h_new)

        # keep the best ever seen
        if new_val < best_val - improve_tol:
            best_val = new_val
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # adapt learning rate if progress stalls
        if no_improve > 500:
            lr = max(lr * 0.5, 1e-4)
            no_improve = 0

        # increase the soft‑max sharpness slowly
        if alpha < 80.0:
            alpha = min(alpha * 1.03, 80.0)

        h = h_new

    # ---- greedy local refinement ---------------------------------------
    # Small random moves that preserve the sum.
    h = best_h.copy()
    for _ in range(15000):
        if time.time() - start_time > budget_s * 0.98:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = (np.random.rand() - 0.5) * 0.08  # step size ~[-0.04,0.04]
        hi = h[i] + delta
        hj = h[j] - delta
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        h_candidate = h.copy()
        h_candidate[i] = hi
        h_candidate[j] = hj
        cand_val = _cmax(h_candidate)
        if cand_val < best_val - 1e-8:
            best_val = cand_val
            best_h = h_candidate.copy()
            h = h_candidate
        # occasionally accept neutral moves to escape plateaus
        elif cand_val < best_val + 1e-5 and random.random() < 0.01:
            h = h_candidate

    # final projection to guarantee exact feasibility
    best_h = _project_bounds_sum(best_h, target_sum)
    best_val = _cmax(best_h)

    return best_h.tolist(), float(best_val), n
