import numpy as np, time, math, random

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def optimize(h, max_iter, time_limit):
    start = time.time()
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(best_h)
    # initial values for step size
    step = 0.1
    iteration = 0
    while iteration < max_iter and time.time() - start < time_limit:
        # random pick i<j
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # compute epsilon
        eps = (np.random.rand() - 0.5) * step * 2  # uniform [-step, step]
        # propose new values
        hi = best_h[i] + eps
        hj = best_h[j] - eps
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        # create new candidate
        cand = best_h.copy()
        cand[i] = hi
        cand[j] = hj
        # compute score
        score = compute_c5(cand)
        # accept if improvement
        if score < best_score:
            best_h = cand
            best_score = score
            # optionally reduce step
        # else maybe accept with probability
        # ...
        iteration += 1
        # decay step slowly
        step = max(step * 0.99999, 1e-5)
    return best_h, best_score

import numpy as np
import time
import random

def _c5_value(h: np.ndarray) -> float:
    """Compute the C5 value for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # Full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_sum_one(h: np.ndarray, target: float) -> np.ndarray:
    """
    Adjust a vector h (already respecting 0<=h<=1) so that its sum equals *target*.
    The routine moves mass between two coordinates, preserving the box constraints.
    """
    # Simple two‑point correction – find any pair that can absorb the difference.
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Locate a coordinate that can increase and one that can decrease.
    if diff > 0:
        inc_idx = np.where(h < 1.0)[0]
        dec_idx = np.where(h > 0.0)[0]
    else:
        inc_idx = np.where(h > 0.0)[0]
        dec_idx = np.where(h < 1.0)[0]
    if inc_idx.size == 0 or dec_idx.size == 0:
        return h
    i = inc_idx[0]
    j = dec_idx[0]
    # Apply the smallest feasible adjustment.
    eps = min(abs(diff), 1.0 - h[i] if diff > 0 else h[i])
    eps = min(eps, h[j] if diff > 0 else 1.0 - h[j])
    eps = eps if diff > 0 else -eps
    h[i] += eps
    h[j] -= eps
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):

    Optimise a step function h on [0,2] (discretised) to minimise the
    Erdős C5 constant.  The routine works with any supplied initial
    configuration (named ``initial_h_values``) but will fall back to a
    uniform 0.5 start if none is present.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    try:
        # ``initial_h_values`` is injected by the harness if available.
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.shape[0]
    except NameError:
        # Default to a modest size – 81 (odd) helps avoid the exact half‑shift.
        n = 81
        h = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0
    # project onto the feasible set (box + exact sum)
    h = np.clip(h, 0.0, 1.0)
    # simple scaling – if sum deviates, correct by pairwise moves.
    if not np.isclose(h.sum(), target_sum, atol=1e-12):
        h = _project_sum_one(h, target_sum)

    best_h = h.copy()
    best_score = _c5_value(best_h)

    # ------------------------------------------------------------------
    # local stochastic optimisation (two‑point swaps)
    # ------------------------------------------------------------------
    start_time = time.time()
    timeout = budget_s * 0.93            # keep a small safety margin
    max_iters = 2_000_000                # hard cap – actual run will stop earlier
    step_size = 0.2                      # initial amplitude of perturbations

    it = 0
    while it < max_iters and (time.time() - start_time) < timeout:
        # pick two distinct positions
        i, j = random.sample(range(n), 2)
        # propose a symmetric move (preserves the total sum)
        eps = (random.random() - 0.5) * step_size * 2.0  # in [-step, step]
        hi = best_h[i] + eps
        hj = best_h[j] - eps
        # enforce the box constraints

            it += 1
            continue

        cand = best_h.copy()
        cand[i] = hi
        cand[j] = hj

                break

        # gradually shrink the proposal size
        if it % 5000 == 0 and step_size > 1e-4:
            step_size *= 0.98
        it += 1

    # final safety projection (might have drifted by 1e-12)
    best_h = np.clip(best_h, 0.0, 1.0)
    if not np.isclose(best_h.sum(), target_sum, atol=1e-9):
        best_h = _project_sum_one(best_h, target_sum)

    final_score = _c5_value(best_h)

    # return as requested: values, bound, number of points
    return best_h.tolist(), float(final_score), n
