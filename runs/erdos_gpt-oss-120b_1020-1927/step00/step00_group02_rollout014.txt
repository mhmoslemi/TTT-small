import time, math
import numpy as np

def _project_sum_box(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {x: 0<=x<=1, sum(x)=target_sum}."""
    lo = - (v.max() + 1.0)
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _gradient_max_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the maximal overlap term at the given shift."""
    N = h.shape[0]
    g = np.zeros(N, dtype=float)

    if shift >= 0:
        ov = N - shift                 # length of the overlapping region
        g[:ov] += 1.0                  # Σ h_i term
        g[:ov] -= h[shift:]            # -h_i·h_{i+shift} contribution (i part)
        if shift == 0:
            g[:ov] -= h[:ov]           # second identical contribution when shift==0
        else:
            g[shift:] -= h[:ov]        # -h_{i}·h_{i-shift} contribution (i‑shift part)
    else:
        s = -shift
        ov = N - s
        g[s:] += 1.0                   # Σ h_i term (i from s to N‑1)
        g[s:] -= h[:ov]                # -h_i·h_{i-s} contribution (i part)
        g[:ov] -= h[s:]                # -h_{i}·h_{i+s} contribution (i+s part)
    return g


def _correlation_max(h: np.ndarray):
    """Return (full correlation vector, max value scaled by dx)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr, corr.max() * (2.0 / h.shape[0])


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- obtain a base pattern (if supplied) -----
    try:
        base = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
    except Exception:
        base = np.full(45, 0.5, dtype=float)

    base_len = base.shape[0]

    # target number of samples (keep it modest and < 1000)
    repeat = max(4, int(360 // base_len))
    n = base_len * repeat
    if n > 1000:
        repeat = max(1, 1000 // base_len)
        n = base_len * repeat

    # start from the repeated parent pattern
    h = np.repeat(base, repeat)
    target_sum = n / 2.0
    h = _project_sum_box(h, target_sum)

    # initialise bookkeeping
    best_h = h.copy()
    _, best_c5 = _correlation_max(h)
    stagnation = 0
    iter_cnt = 0
    step0 = 0.6

    while True:
