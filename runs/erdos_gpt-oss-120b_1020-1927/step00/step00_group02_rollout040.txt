def run(seed=42, budget_s=1100, **kwargs):
    # Choose a size that makes the zero/one blocks exactly 1/3 of the interval.
    # n must be a multiple of 6 so that a = (n/6)*(2/n) = 1/3.
    n = 60                # < 1 000 samples, fast to evaluate
    p = n // 6            # length of the leading zeros and trailing ones

    # Build the step function:
    #   [0, a)        -> 0
    #   [a, 2‑a)      -> 0.5
    #   [2‑a, 2]      -> 1
    h = np.empty(n, dtype=float)
    h[:p] = 0.0
    h[p:n-p] = 0.5
    h[n-p:] = 1.0

    # Normalisation check (optional, always satisfied by construction)
    # dx = 2 / n ; sum(h) * dx == 1
    dx = 2.0 / n
    # Evaluate the maximal overlap integral C₅
    corr = np.correlate(h, 1 - h, mode="full") * dx
    c5 = float(corr.max())   # best achievable bound for this construction

    return h, c5, n
