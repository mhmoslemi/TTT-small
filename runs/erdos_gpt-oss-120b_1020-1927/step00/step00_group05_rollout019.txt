import time
import numpy as np

def _project_to_feasible(v, target_sum):
    """Project vector v onto the box [0,1] intersect {sum = target_sum}."""
    lo = -np.max(v) - 5.0          # ensure all become 0
    hi = 1.0 - np.min(v) + 5.0     # ensure all become 1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _compute_gradient(h, probs):
    """Gradient of the soft‑max of f(k)=⟨h,1−shift_k h⟩."""
    n = h.shape[0]
    offset = n - 1
    g = np.zeros_like(h)
    for idx, w in enumerate(probs):
        if w == 0.0:
            continue
        s = idx - offset          # shift amount
        if s >= 0:
            end = n - s
            # part contributing +1‑h[i+s]
            g[:end] += w * (1.0 - h[s:])
            # part contributing −h[i]
            g[s:]   += w * (-h[:end])
        else:
            s = -s
            end = n - s
            # i runs from s .. n‑1
            g[s:]   += w * (1.0 - h[:end])
            g[:end] += w * (-h[s:])
    return g


def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points)."""
    start_time = time.time()
