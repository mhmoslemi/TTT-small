    """C₅ value for a step vector v on [0,2]."""

    """Uniform random binary vector with exactly `half` ones."""

            # blind random swap of any 1 with any 0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)

        # Execute tentative swap
        h[i], h[j] = h[j], h[i]

        # Evaluate new state
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_score = float(np.max(new_corr) * (2.0 / n))

        # ----- acceptance test ------------------------------------------------
        delta = new_score - cur_score
        elapsed = time.time() - start
        t_frac = max(0.0, (deadline - time.time()) / total_time)
        temperature = max_temp * t_frac + min_temp

        if delta <= 0 or rng.random() < math.exp(-delta / temperature):
            # accept move
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_h = h.copy()
                last_improve_iter = iter_count
        else:
            # reject: undo swap
            h[i], h[j] = h[j], h[i]

        iter_count += 1

        # ----- stagnation handling -------------------------------------------
        if iter_count - last_improve_iter > 30000:
            h = _random_binary(rng, n, half)
            last_improve_iter = iter_count

        # ----- early exit if target reached ----------------------------------
        if best_score <= 0.3808:
            break

    # Guarantee the integral constraint (average = 0.5) at the very end.
    target_sum = n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        diff = target_sum - cur_sum
        if diff > 0:
            zeros = np.where(best_h < 1.0)[0]
            inc = diff / zeros.size if zeros.size else 0
            best_h[zeros] = np.clip(best_h[zeros] + inc, 0.0, 1.0)
        else:
            ones = np.where(best_h > 0.0)[0]
            dec = -diff / ones.size if ones.size else 0
            best_h[ones] = np.clip(best_h[ones] - dec, 0.0, 1.0)

    final_score = _c5_of(best_h)

    return (best_h.tolist(), float(final_score), n)
