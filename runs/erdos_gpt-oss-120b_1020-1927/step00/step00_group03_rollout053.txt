import time
import numpy as np

def _init_h(seed, n):
    """Create an initial feasible h of length n (half ones, half zeros)."""
    rng = np.random.default_rng(seed)
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def _eval_c5_from_R(R, target_sum, n):
    """C5 from current auto‑correlation R."""
    dx = 2.0 / n
    return ((target_sum - R).max()) * dx

def _gradient_for_shift(h, k):
    """Gradient ∂R[k]/∂h_i = h_{i+k}+h_{i‑k} (zero outside)."""
    n = len(h)
    g = np.zeros(n, dtype=float)
    if k >= 0:
        # i+k term
        g[: n - k] += h[k:]
        # i‑k term
        g[k:] += h[: n - k]
    else:
        kk = -k
        g[kk:] += h[: n - kk]
        g[: n - kk] += h[kk:]
    return g

def _delta_R(h, i, j, di, dj):
    """
    Compute change in auto‑correlation R when h[i]←h[i]+di, h[j]←h[j]+dj.
    Uses padded view for O(n) vectorised work.
    """
    n = len(h)
    L = 2 * n - 1
    offset = n - 1
    pad = offset
    h_pad = np.concatenate((np.zeros(pad), h, np.zeros(pad)))

    # contributions from index i
    slice_i = h_pad[i : i + L]
    contrib_i = di * (slice_i + slice_i[::-1])

    # contributions from index j
    slice_j = h_pad[j : j + L]
    contrib_j = dj * (slice_j + slice_j[::-1])

    dR = contrib_i + contrib_j

    # cross terms (d_i * d_j) at k = j‑i and k = i‑j
    shift1 = j - i
    idx1 = shift1 + offset
    dR[idx1] += di * dj
    shift2 = -shift1
    idx2 = shift2 + offset
    dR[idx2] += di * dj
    return dR

    # -----------------------------------------------------------------
