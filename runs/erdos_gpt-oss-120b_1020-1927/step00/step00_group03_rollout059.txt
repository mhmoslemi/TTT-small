def project_onto_simplex_box(v, target_sum):
    # v: array, target_sum = total sum required.
    # returns x in [0,1] with sum = target_sum, minimizing ||x - v||^2.
    # This is projection onto bounded simplex.
    #
    # Use algorithm: start with active set = all indices.
    # Loop:
    #   Compute λ = (target_sum - sum(v_free) ) / len(active)
    #   x = v
    #   For i in active: x_i = v_i + λ
    #   Clip x_i to [0,1] -> if x_i <0 -> set to 0, if x_i >1 -> set to 1
    #   If any clipping occurred, fix those indices as saturated, remove from active, update target_sum accordingly (subtract already fixed values), and repeat.
    #   else break.
    #
    # Complexity O(N^2) worst-case but fine.
    #
    # Implementation reminiscent of projecting onto simplex with bounds.
    N = len(v)
    x = np.copy(v)
    # initial active indices all
    active = np.arange(N)
    # keep track of fixed values
    # We'll treat iterative approach.
    while True:
        if len(active) == 0:
            # all fixed; just adjust if rounding error
            # Already sum check:
            break
        # Compute λ for active entries
        sum_fixed = np.sum(x) - np.sum(x[active])  # sum of already fixed entries (i.e., those not in active)
        # Want sum_fixed + sum_{i in active} (v_i + λ) = target_sum => λ = (target_sum - sum_fixed - sum(v_active)) / len(active)
        v_active = v[active]  # original v values (not current x) for active entries
        λ = (target_sum - sum_fixed - np.sum(v_active)) / len(active)
        # Update tentative x for active
        x_new = v_active + λ
        # Clip
        below = x_new < 0
        above = x_new > 1
        # If any clipping, fix them
        if np.any(below) or np.any(above):
            # For below, set to 0; for above set to 1.
            # Update x for those indices:
            # Determine indices in active where clip occurs
            fix_low = active[below]
            fix_high = active[above]
            # Set them
            x[fix_low] = 0.0
            x[fix_high] = 1.0
            # Remove them from active
            mask = np.ones(len(active), dtype=bool)
            mask[below] = False
            mask[above] = False
            active = active[mask]
            # Continue loop
            continue
        else:
            # No clipping, set all active to x_new and break
            x[active] = x_new
            break
    # Ensure tiny numerical errors corrected
    # Adjust sum if necessary by distributing small diff among non-bound entries
    total = np.sum(x)
    diff = target_sum - total
    if abs(diff) > 1e-12:
        # distribute diff proportionally to interior entries (0 < x < 1)
        interior = np.where((x > 0) & (x < 1))[0]
        if interior.size > 0:
            # distribute equally
            x[interior] += diff / interior.size
            # Clip again ensure within bounds
            x = np.clip(x, 0., 1.)
            # After that sum could be off by a bit due to clipping; final correction:
            total = np.sum(x)
            diff = target_sum - total
            if abs(diff) > 1e-9:
                # adjust one interior entry
                idx = interior[0]
                x[idx] = np.clip(x[idx] + diff, 0., 1.)
        else:
            # adjust any entry
            idx = np.argmax(np.abs(x - 0.5))  # irrelevant
            x[idx] = np.clip(x[idx] + diff, 0., 1.)
    # final ensure sum exact within tolerance
    return x

import time
import math
import numpy as np

def _compute_c5(h):
    """Return the C5 value for a given step function h."""
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / N)

def _project_onto_bounds_sum(v, target):
    """Euclidean projection of v onto the box [0,1]^n with sum(v)=target."""
    # Find λ such that sum(clip(v+λ,0,1)) = target.
    lo = -np.max(v) - 1.0       # makes v+lo <= 0
    hi = 1.0 - np.min(v) + 1.0  # makes v+hi >= 1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny fix for rounding errors
    diff = target - np.sum(x)
    if abs(diff) > 1e-10:
        free = (x > 0.0) & (x < 1.0)
        cnt = np.count_nonzero(free)
        if cnt:
            x[free] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
        else:
            # all coordinates are at bounds – adjust one arbitrarily
            i = np.argmin(np.abs(x - 0.5))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def _gradient_for_shift(h, shift):
    """Gradient of O_k = Σ h_i (1‑h_{i+shift}) wrt h at a given shift."""
    N = h.shape[0]
    g = np.zeros(N)
    if shift >= 0:
        L = N - shift
        g[:L] += 1.0 - h[shift:]
        g[shift:] += -h[:L]
    else:
        k = -shift
        L = N - k
        g[k:] += 1.0 - h[:L]
        g[:L] += -h[k:]
    return g

def _random_two_swap(h, rng):
    """Generate a feasible neighbour by moving mass between two coordinates."""
    N = h.shape[0]
    i, j = rng.integers(N, size=2)
    if i == j:
        return None
    hi, hj = h[i], h[j]
    lower = max(-hi, hj - 1.0)
    upper = min(1.0 - hi, hj)
    if lower >= upper:
        return None
    delta = rng.random() * (upper - lower) + lower
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a small C₅ value."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------------------
    # 1. Initialise h
    # --------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)   # provided by the harness
    except NameError:
        init = None

    if init is not None:
        h = init.copy()
        N = h.shape[0]
    else:
        N = 70                       # a modest size if no parent is supplied
        h = np.full(N, 0.5, dtype=float)

    target_sum = N / 2.0
    # Ensure feasibility
    h = np.clip(h, 0.0, 1.0)
    h = _project_onto_bounds_sum(h, target_sum)

    cur_h = h.copy()
    cur_score = _compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    # --------------------------------------------------------------------
    # 2. optimisation loop
    # --------------------------------------------------------------------
    max_iter = 10**7               # a very large bound – we stop by time
    step_size = 0.5                # initial gradient step length
    step_decay = 0.9995            # geometric decay per iteration
    prob_random = 0.25             # chance to try a random two‑swap move
    iter_no_improve = 0
    last_improve_iter = 0

    for it in range(max_iter):
        # time check
        if time.time() - start_time > budget_s * 0.99:
            break

        # compute correlation and locate all maximal shifts
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_val = corr.max()
        max_idx = np.where(corr == max_val)[0]
        # convert to shifts relative to centre
        shifts = max_idx - (N - 1)

        # ----------------------------------------------------------------
        # 2a. decide move type
        # ----------------------------------------------------------------
        if rng.random() < prob_random:
            cand = _random_two_swap(cur_h, rng)
            if cand is None:
                continue
        else:
            # average gradient over all worst shifts
            g = np.zeros(N)
            for sh in shifts:
                g += _gradient_for_shift(cur_h, sh)
            g /= len(shifts) if shifts.size else 1
            # descent step + projection
            cand = _project_onto_bounds_sum(cur_h - step_size * g, target_sum)

        cand_score = _compute_c5(cand)

        # ----------------------------------------------------------------
        # 2b. accept if improvement (greedy) – occasionally allow uphill
        # ----------------------------------------------------------------
        accept = False
        if cand_score < cur_score - 1e-12:
            accept = True
        else:
            # simple simulated‑annealing acceptance
            temperature = max(1e-6, (budget_s - (time.time() - start_time)) / budget_s)
            prob = math.exp(-(cand_score - cur_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_h = cand
            cur_score = cand_score
            step_size = max(step_size * step_decay, 1e-4)
            iter_no_improve = 0
            if cand_score < best_score - 1e-12:
                best_h = cand.copy()
                best_score = cand_score
                last_improve_iter = it
        else:
            iter_no_improve += 1

        # ----------------------------------------------------------------
        # 2c. occasional restart if stuck for a while
        # ----------------------------------------------------------------
        if iter_no_improve > 5000:
            # perturb a random subset while preserving the total mass
            mask = rng.random(N) < 0.1
            if np.any(mask):
                pert = cur_h[mask] + rng.normal(scale=0.2, size=mask.sum())
                cur_h[mask] = pert
                cur_h = _project_onto_bounds_sum(cur_h, target_sum)
                cur_score = _compute_c5(cur_h)
            iter_no_improve = 0

    # --------------------------------------------------------------------
    # 3. Final tidy‑up (enforce constraints a final time)
    # --------------------------------------------------------------------
    best_h = _project_onto_bounds_sum(best_h, target_sum)
    final_score = _compute_c5(best_h)

    return (best_h, float(final_score), N)
