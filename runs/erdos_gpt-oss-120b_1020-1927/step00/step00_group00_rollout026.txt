import numpy as np, time, math

def compute_c5(h: np.ndarray) -> float:
    """return max overlap integral for a given step vector h."""
    n = h.size
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def max_c5_and_lag(h: np.ndarray):
    """return (max value, lag producing it). lag = index-(n-1)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    max_val = float(corr[max_idx] * (2.0 / n))
    lag = max_idx - (n - 1)
    return max_val, lag


def grad_h_for_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """gradient of A(lag)=∫h(1‑h(·+lag)) w.r.t. h (sub‑gradient of the active term)."""
    n = h.size
    forward = np.zeros(n, dtype=np.float64)
    backward = np.zeros(n, dtype=np.float64)
    if lag >= 0:
        forward[: n - lag] = h[lag:]
        backward[lag:] = h[: n - lag]
    else:
        k = -lag
        forward[k:] = h[: n - k]
        backward[: n - k] = h[k:]
    return 1.0 - forward - backward


def project_sum_and_bounds(h: np.ndarray, target: float) -> np.ndarray:
    """clip to [0,1] and enforce sum == target (tiny adjustments)."""
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    n = h.size
    # try to fix the deficit/excess with a single entry
    if diff > 0:
        # need to increase
        idx = np.argmax(1.0 - h)          # farthest from 1
        add = min(diff, 1.0 - h[idx])
        h[idx] += add
    else:
        idx = np.argmin(h)                # farthest from 0
        sub = min(-diff, h[idx])
        h[idx] -= sub
    # any tiny remainder is spread uniformly (will stay inside bounds)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        h += diff2 / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def random_binary_h(N: int, rng: np.random.Generator) -> np.ndarray:
    """binary (or half‑binary) vector with exact integral 1."""
    h = np.zeros(N, dtype=np.float64)
    half = N // 2
    ones = rng.choice(N, size=half, replace=False)
    h[ones] = 1.0
    if N % 2:                     # odd length: add a 0.5 entry
        remaining = np.setdiff1d(np.arange(N), ones)
        h[rng.choice(remaining)] = 0.5
    return h


def gradient_descent_once(h: np.ndarray, target: float, max_iters: int,
                         rng: np.random.Generator, deadline: float):
    """simple sub‑gradient descent on the active lag, keep the best seen."""
    best_h = h.copy()
    best_score = compute_c5(h)
    n = h.size
    for it in range(max_iters):
        if time.time() > deadline:
            break
        cur_score, lag = max_c5_and_lag(h)
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
        g = grad_h_for_lag(h, lag)
        # decreasing step size
        alpha = 0.5 * (1.0 - it / max_iters) + 0.001
        h_new = h - alpha * g
        np.clip(h_new, 0.0, 1.0, out=h_new)
        h_new = project_sum_and_bounds(h_new, target)
        new_score = compute_c5(h_new)
        # accept only if we do not worsen the active lag
        if new_score <= cur_score:
            h = h_new
    # final check
    final_score = compute_c5(best_h)
    if final_score > compute_c5(h):
        best_h = h
        best_score = compute_c5(h)
    return best_h, best_score


def annealing_refinement(h: np.ndarray, target: float, rng: np.random.Generator,
                         deadline: float, start_time: float, total_time: float):
    """simple simulated‑annealing style pair updates."""
    cur_h = h.copy()
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    n = cur_h.size
    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_time
        progress = min(1.0, elapsed / total_time)
        temp = (1.0 - progress) * 0.01 + 1e-5
        eps = (1.0 - progress) * 0.05 + 0.001

        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        max_step = min(cur_h[i], 1.0 - cur_h[j])
        if max_step <= 0.0:
            continue
        step = min(eps, max_step)
        new_h = cur_h.copy()
        new_h[i] -= step
        new_h[j] += step
        new_score = compute_c5(new_h)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / (temp + 1e-12)):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()
        # occasional tiny sum correction
        if int(time.time() * 10) % 97 == 0:
            diff = target - cur_h.sum()
            if abs(diff) > 1e-12:
                cur_h[0] = np.clip(cur_h[0] + diff, 0.0, 1.0)
    return best_h, best_score

    rng = np.random.default_rng(seed)

    init = globals().get("initial_h_values")

    h0 = project_sum_and_bounds(h0, target_sum)

    start = time.time()
    end = start + budget_s * 0.95   # reserve a small margin

    # 1) deterministic descent from the (possibly) good initial vector
    best_h, best_score = gradient_descent_once(h0, target_sum,
                                               max_iters=800000,
                                               rng=rng, deadline=end)

    # 2) a few random restarts with shorter descents
    while time.time() + 0.5 < end:

    best_h = project_sum_and_bounds(best_h, target_sum)
