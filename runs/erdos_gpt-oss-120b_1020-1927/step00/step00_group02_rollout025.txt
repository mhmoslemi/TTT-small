def project_onto_box_simplex(v, total):
    # Clip to [0,1] after projection via lambda.
    # We'll find lambda such that sum(np.clip(v - lam, 0, 1)) = total.
    lo = -np.max(v) - 1.0  # lower bound for λ
    hi = np.max(v) + 1.0   # upper bound
    for _ in range(50):
        lam = (lo + hi) / 2.0
        y = np.clip(v - lam, 0, 1)
        s = y.sum()
        if s > total:
            lo = lam
        else:
            hi = lam
    y = np.clip(v - (lo+hi)/2.0, 0, 1)
    # At this point s may not equal total due to bounds; adjust small remainder if any via adding to non-locked values.
    # compute diff = total - sum(y)
    diff = total - y.sum()
    if abs(diff) > 1e-10:
        # distribute diff uniformly to entries not at bounds (0<y<1)
        mask = (y > 0) & (y < 1)
        if mask.any():
            y[mask] += diff / mask.sum()
            # Clip again
            y = np.clip(y, 0, 1)
        else:
            # No interior, fallback: distribute to all
            y += diff / len(y)
            y = np.clip(y, 0, 1)
        # finally adjust any remaining diff iteratively; but simple approach.
    return y

import time
import numpy as np

# ---------- helper utilities ----------
def compute_c5(h):
    """Return the max overlap integral for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # full cross‑correlation of h and (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def project_onto_box_simplex(v, total):
    """
    Euclidean projection of vector v onto the set
    { y | 0 ≤ y_i ≤ 1, sum(y)=total }.
    """
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        y = np.clip(v - lam, 0.0, 1.0)
        if y.sum() > total:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    y = np.clip(v - lam, 0.0, 1.0)

    # tiny fix for numerical drift
    diff = total - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > 0.0) & (y < 1.0)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, 0.0, 1.0)
        else:                         # all at the bounds – distribute uniformly
            y += diff / len(y)
            y = np.clip(y, 0.0, 1.0)
    return y


def make_three_region_h(L_mid, a_val, n):
    """
    Symmetric three‑region step function.
    L_mid : length of the middle region (0<L_mid<2)
    a_val : height of the middle region (0≤a_val≤1)
    n     : number of discretisation points
    Returns h, the array of length n.
    """
    total = n / 2.0                     # integral must be 1 ⇒ sum = n/2
    if L_mid >= 2.0:
        L_mid = 1.9999
    w = (2.0 - L_mid) / 2.0             # each outer region length
    # compute outer height from the integral constraint
    denom = 2.0 - L_mid
    if denom == 0:
        return None
    b_val = (1.0 - a_val * L_mid) / denom
    if not (0.0 <= b_val <= 1.0):
        return None

    dx = 2.0 / n
    x = np.arange(n) * dx + dx / 2.0    # mid‑point locations
    left_edge = x < w
    middle = (x >= w) & (x < w + L_mid)
    h = np.where(left_edge, b_val,
                 np.where(middle, a_val, b_val))
    # enforce the exact integral by projection (tiny correction may be needed)
    h = project_onto_box_simplex(h, total)
    return h


def subgradient_step(h, step):
    """Perform one sub‑gradient descent step for the max‑overlap objective."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    shift = max_idx - (n - 1)               # integer lag
    g = np.zeros_like(h)

    for i in range(n):
        j = i + shift
        if 0 <= j < n:
            g[i] += 1.0 - h[j]
            g[j] -= h[i]
        else:
            g[i] += 1.0                     # (1‑h) = 1 outside the interval
    # gradient step and projection
    v = h - step * g
    total = n / 2.0
    h_new = project_onto_box_simplex(v, total)
    return h_new


def random_pair_move(h, eps):
    """Swap a small amount of mass between two coordinates, preserving the total sum."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # direction: try to move eps from i to j (or opposite if not feasible)
    delta = np.random.uniform(-eps, eps)
    if delta >= 0:
        inc = min(delta, 1.0 - h[i])
        dec = min(delta, h[j])
        actual = min(inc, dec)
        if actual > 0:
            h[i] += actual
            h[j] -= actual
    else:
        delta = -delta
        dec = min(delta, h[i])
        inc = min(delta, 1.0 - h[j])
        actual = min(dec, inc)
        if actual > 0:
            h[i] -= actual
            h[j] += actual
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # ---- choose discretisation size ----
    if "initial_h_values" in globals():
        base_n = len(initial_h_values)
        n_points = max(200, base_n * 2)   # a modest resolution
    else:
        n_points = 200

    total_needed = n_points / 2.0          # required sum of h

    # ---- initialise from the provided construction if present ----
    if "initial_h_values" in globals():
        # linear interpolation onto the current grid
        base_n = len(initial_h_values)
        x_old = np.linspace(0.0, 2.0, base_n, endpoint=False) + 1.0 / base_n
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False) + 1.0 / n_points
        h_curr = np.interp(x_new, x_old, np.asarray(initial_h_values, dtype=float))
        h_curr = np.clip(h_curr, 0.0, 1.0)
        h_curr = project_onto_box_simplex(h_curr, total_needed)
        best_h = h_curr.copy()
        best_c5 = compute_c5(best_h)
    else:
        # uniform 0.5 is a feasible start
        best_h = np.full(n_points, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)

    # ---- Phase 1: systematic search in the three‑region family ----
    L_vals = np.linspace(0.2, 1.0, 81)      # step ≈0.01
    a_vals = np.linspace(0.5, 1.0, 51)      # step ≈0.01
    for L_mid in L_vals:
        if time.time() - start_time > 0.35 * budget_s:
            break
        for a_val in a_vals:
            h_candidate = make_three_region_h(L_mid, a_val, n_points)
            if h_candidate is None:
                continue
            c5 = compute_c5(h_candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate.copy()

    # ---- Phase 2: sub‑gradient descent from the current best ----
    # step size schedule: start relatively large and decay
    max_iter = 8000
    for it in range(max_iter):
        if time.time() - start_time > 0.75 * budget_s:
            break
        step_size = 0.3 / np.sqrt(it + 1)
        h_new = subgradient_step(best_h, step_size)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-9:
            best_c5 = c5_new
            best_h = h_new.copy()

    # ---- Phase 3: simple stochastic hill‑climbing for fine tuning ----
    eps = 0.02
    while time.time() - start_time < budget_s * 0.95:
        h_trial = best_h.copy()
        h_trial = random_pair_move(h_trial, eps)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial
            eps = max(eps * 0.9, 0.005)   # gradually reduce step size when improving
        else:
            eps = min(eps * 1.05, 0.05)   # occasionally increase to escape plateaus

    # final safety projection (ensures feasibility)
    best_h = project_onto_box_simplex(best_h, total_needed)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_points
