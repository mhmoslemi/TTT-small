def project_onto_bounds_and_sum(y, S, tol=1e-9, max_iter=100):
    # compute lower and upper lambda bounds such that sum at lambda lower <= S <= sum at lambda upper.
    # Since sum increases with lambda, we can start low = -max(y) - 1, high = 1 - min(y) + 1 maybe.
    lo = -np.max(y) - 1  # ensures all y+lo <=0 -> sum=0.
    hi = 1 - np.min(y) + 1  # ensures all y+hi >=1 -> sum=n.
    # Actually we need to guarantee extreme by: at lo sum = sum of zeros = number_of_elements where y_i+lo >0? Wait if lo is far negative, all clip results will be 0. So sum=0.
    # At hi far positive, all clip results will be 1 => sum=n.
    # So we can set lo = -1 - np.max(y) and hi = 1 - np.min(y) + 1.
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    # ensure bounds: at lo sum <= S <= sum at hi.
    # Use binary search.
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        total = x.sum()
        if abs(total - S) <= tol:
            return x
        if total < S:
            lo = mid
        else:
            hi = mid
    # after loop, return x with closest
    return np.clip(y + (lo + hi)/2.0, 0.0, 1.0)

def compute_gradient(h):
    n = len(h)
    # correlation array (raw)
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (n - 1)
    # Compute gradient w.r.t raw correlation (not scaled)
    grad = np.ones_like(h)
    if s >= 0:
        # term1: subtract h[i + s] for i from 0..n-s-1
        grad[:n - s] -= h[s:]
        # term2: subtract h[i - s] for i from s..n-1
        if s != 0:
            grad[s:] -= h[:n - s]
        else:
            grad -= h  # both terms are h; double subtraction equivalently subtract 2*h
    else:
        t = -s
        # i + s = i - t ; valid for i >= t
        grad[t:] -= h[:n - t]
        # i - s = i + t ; valid for i <= n - t -1
        grad[:n - t] -= h[t:]
    # That's gradient of raw correlation; scaling factor dx irrelevant.
    return grad, s, corr.max()

def compute_gradient(h):
    n = len(h)
    # full correlation
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    s = idx - (n - 1)
    # gradient
    grad = np.ones_like(h)
    if s == 0:
        grad = 1 - 2 * h
    elif s > 0:
        # i + s term
        grad[:n - s] -= h[s:]
        # i - s term
        grad[s:] -= h[:n - s]
    else:
        t = -s
        grad[t:] -= h[:n - t]
        grad[:n - t] -= h[t:]
    return grad, s, corr[idx]

def project_onto_bounds_and_sum(v, target_sum, tol=1e-9, max_iter=100):
    # find lambda such that sum(np.clip(v + lam, 0, 1)) = target_sum
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

import time, numpy as np

def evaluate_c5(h):
    """C5 value using the evaluation rule."""
    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full").max() * dx

def project_onto_bounds_and_sum(v, target, tol=1e-12, max_iter=100):
    """Euclidean projection of v onto {0≤x≤1, sum(x)=target}."""
    lo = -np.max(v) - 1.0      # makes all entries ≤0 → sum=0
    hi = 1.0 - np.min(v) + 1.0  # makes all entries ≥1 → sum=n
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def compute_gradient(h):
    """Sub‑gradient of max_k ∫h(x)(1‑h(x+k))dx at the current worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    s = idx - (n - 1)                 # shift attaining the maximum
    if s == 0:
        grad = 1.0 - 2.0 * h
    elif s > 0:
        grad = np.ones_like(h)
        grad[:n - s] -= h[s:]
        grad[s:]    -= h[:n - s]
    else:  # s < 0
        t = -s
        grad = np.ones_like(h)
        grad[t:]    -= h[:n - t]
        grad[:n - t] -= h[t:]
    return grad, s, corr[idx]

def local_optimize(h0, target_sum, max_iter=4000, time_limit=None, start_time=None):
    """Simple projected sub‑gradient descent."""
    if start_time is None:
        start_time = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate_c5(h)
    cur_val = best_val
    lr = 0.5
    for it in range(max_iter):
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        grad, _, _ = compute_gradient(h)
        h_new = project_onto_bounds_and_sum(h - lr * grad, target_sum)
        new_val = evaluate_c5(h_new)
        if new_val < cur_val:
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
        else:
            # no improvement – shrink step and perturb a little
            lr *= 0.7
            if np.random.rand() < 0.5:
                pert = (np.random.rand(h.size) - 0.5) * 0.05
                h_new = project_onto_bounds_and_sum(h + pert, target_sum)
                new_val = evaluate_c5(h_new)
                if new_val < cur_val:
                    h = h_new
                    cur_val = new_val
        lr *= 0.9995
        if (it + 1) % 300 == 0:
            # occasional jitter to escape plateaus
            jitter = (np.random.rand(h.size) - 0.5) * 0.02
            h = project_onto_bounds_and_sum(h + jitter, target_sum)
            cur_val = evaluate_c5(h)
    return best_h, best_val

def generate_shape(n, a):
    """One‑parameter family: linear ramps of width a and a flat top of height H."""
    if a <= 1e-12:                     # degenerate case → constant 0.5
        return np.full(n, 0.5)
    H = 1.0 / (2.0 - a)                # Height needed to make area = 1
    L = 2.0 - 2.0 * a                  # Length of the flat top (may be zero)
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx      # mid‑point sampling
    h = np.empty(n, dtype=float)
    left = x < a
    h[left] = H * x[left] / a
    middle = (x >= a) & (x <= a + L)
    h[middle] = H
    right = x > a + L
    h[right] = H * (2.0 - x[right]) / a
    return h
