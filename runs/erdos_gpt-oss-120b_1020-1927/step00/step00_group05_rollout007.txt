def project_onto_capped_simplex(y, S, cap=1.0):
    # return projection onto sum=S and 0 <= x <= cap
    # Use bsearch on lambda
    lo = y.min() - cap - 1 # lower bound, not important
    hi = y.max() # upper bound
    # function f(lambda) = sum_i min(max(y_i - λ, 0), cap)
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, cap)
        s = x.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo+hi)/2, 0, cap)
    # due to numeric may deviate S; adjust small difference:
    diff = S - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff among entries not at bounds.
        # pick a mask of free entries where 0 < x < cap
        mask = (x > 0) & (x < cap)
        if mask.sum() > 0:
            # add diff equally
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0, cap)
        else:
            # cannot adjust, fallback to scaling
            x = np.clip(y, 0, cap)
            # adjust by scaling of all entries
            scale = S / x.sum()
            x = np.clip(x * scale, 0, cap)
    return x

import numpy as np, time, math

def evaluate(h):
    C = np.max(np.correlate(h, 1-h, mode='full')) * (2.0 / len(h))
    return C

def project_capped_simplex(y, S, cap=1.0):
    # binary search for lambda
    # Since sum_i min(max(y_i - lam, 0), cap) is decreasing in lam.
    # Determine feasible lam bounds:
    # lam_low ensures all values at cap -> lam < y_i - cap
    # lam_high ensures all values at 0 -> lam > y_i
    # So we can set low = y.min() - cap - 1
    # high = y.max() + 1
    lo = y.min() - cap - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, cap)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    x = np.clip(y - lam, 0.0, cap)
    # after binary search sum may be slightly off due to tolerance
    # adjust small difference
    diff = S - x.sum()
    if abs(diff) > 1e-8:
        mask = (x > 0) & (x < cap)
        if mask.any():
            x[mask] += diff / mask.sum()
            # clip again
            x = np.clip(x, 0.0, cap)
        else:
            # fallback: scale
            # ensure not division by zero
            if x.sum() > 0:
                x = np.clip(x * (S / x.sum()), 0.0, cap)
    return x

import numpy as np
import time
import math
import random

def evaluate_h(h: np.ndarray) -> float:
    """Return C5 for the discretized step function h."""
    n = h.shape[0]
    # correlation of h with (1-h) over all integer shifts
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)  # dx = 2/n


def project_capped_simplex(y: np.ndarray, S: float, cap: float = 1.0) -> np.ndarray:
    """
    Projection of y onto the set {x | sum(x)=S, 0<=x<=cap}.
    Solved by finding λ such that x_i = clip(y_i-λ,0,cap) and sum(x)=S.
    """
    lo = y.min() - cap - 1.0
    hi = y.max() + 1.0
    for _ in range(80):                       # enough for double precision
        lam = (lo + hi) * 0.5
        x = np.clip(y - lam, 0.0, cap)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(y - lam, 0.0, cap)
    # tiny correction if numerical error remains
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 0.0) & (x < cap)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, cap)
        else:
            # fallback to scaling (should rarely happen)
            total = x.sum()
            if total > 0:
                x = np.clip(x * (S / total), 0.0, cap)
    return x


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step function h on [0,2] with integral 1 to minimise
    C5 = max_k ∫ h(x) (1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()

    # ------------------------------------------------------------------
    # initialise h
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.shape[0]
    else:
        n = 70
        h = np.random.rand(n)
        S = n / 2.0
        h = project_capped_simplex(h, S, cap=1.0)

    S = n / 2.0                     # required sum of h[i]
    dx = 2.0 / n

    best_h = h.copy()
    best_val = evaluate_h(h)

    # ------------------------------------------------------------------
    # parameters for the iterative scheme
    max_iter = 20000                # generous upper bound, will be cut by time
    step_start = 0.6
    step_end = 5e-5
    alpha_start = 1.0
    alpha_end = 80.0
    temp_start = 0.12
    target = 0.3808                 # we try to beat this

    # ------------------------------------------------------------------
    # main optimisation loop (projected gradient with soft‑max weighting)
    for it in range(max_iter):
        # time guard
        if time.time() - start_time > budget_s * 0.94:
            break

        # current correlation and C5
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_val = np.max(corr) * dx

        # update best if improved
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            if best_val <= target:
                break

        # soft‑max weights on the correlation values (focus on large ones)
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iter)
        maxc = np.max(corr)
        exp_vals = np.exp(alpha * (corr - maxc))
        w = exp_vals / exp_vals.sum()        # sum_j w_j = 1

        # gradient of the smooth approximation of the max
        g = np.zeros_like(h)
        # each shift contributes +wj to every component (the "1" part)
        # which after the whole loop equals a vector of ones (since sum w = 1)
        # we add it inside the loop for simplicity
        for shift_idx, wj in enumerate(w):
            if wj == 0.0:

            j = shift_idx - (n - 1)          # actual shift (negative … positive)

            if j >= 0:

        new_val = evaluate_h(h_new)

        if new_val < cur_val:

        else:

            prob = math.exp(-(new_val - cur_val) / (temp + 1e-12))

    # ------------------------------------------------------------------

    h = best_h.copy()

        h_candidate = h.copy()
        h_candidate[i] -= eps
        h_candidate[j] += eps
        cand_val = evaluate_h(h_candidate)
        if cand_val < best_val:
            best_val = cand_val
            h = h_candidate
            best_h = h_candidate.copy()
            if best_val <= target:
                break

    # final check: ensure feasibility (tiny numerical adjustments)
    best_h = project_capped_simplex(best_h, S, cap=1.0)
    best_val = evaluate_h(best_h)

    return (best_h.tolist(), best_val, n)
