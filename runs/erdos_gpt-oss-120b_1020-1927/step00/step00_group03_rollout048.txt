import time
from scipy.optimize import differential_evolution

def max_overlap(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def evaluate_coeffs(a, xs):
    h = 0.5 + sum_{m} a[m-1] * np.cos(m * np.pi * xs)
    # check bounds
    if np.any(h < -1e-12) or np.any(h > 1 + 1e-12):
        return 1e3  # penalty
    return max_overlap(h)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, sys
    from scipy.optimize import differential_evolution
    np.random.seed(seed)
    start = time.time()
    N = 256  # number of points
    xs = np.arange(N) * (2.0 / N)
    M = 5
    bounds = [(-0.5, 0.5) for _ in range(M)]

    def objective(a):
        # compute h from coefficients
        # a is array of shape (M,)
        h = np.full(N, 0.5)
        for m, coeff in enumerate(a, start=1):
            h += coeff * np.cos(m * np.pi * xs)
        # enforce feasible quickly
        if np.min(h) < -1e-12 or np.max(h) > 1 + 1e-12:
            # penalty: large value proportional to violation
            return 1e6 + 1e6 * (np.abs(np.min(h)) + max(0, np.max(h)-1))
        # compute max overlap
        c5 = max_overlap(h)
        return c5

    # callback to enforce time limit
    def callback(xk, convergence):
        if time.time() - start > budget_s - 5.0:
            return True
        return False

    # run differential evolution
    result = differential_evolution(objective,
                                    bounds,
                                    strategy='best1bin',
                                    maxiter=500,
                                    popsize=15,
                                    tol=1e-7,
                                    mutation=(0.5, 1),
                                    recombination=0.7,
                                    seed=seed,
                                    polish=False,
                                    callback=callback,
                                    init='latinhypercube')
    best_a = result.x
    # compute best h
    best_h = np.full(N, 0.5)
    for m, coeff in enumerate(best_a, start=1):
        best_h += coeff * np.cos(m * np.pi * xs)

    # clip to [0,1] due to possible numeric tiny breach
    best_h = np.clip(best_h, 0.0, 1.0)

    # adjust sum to exact N/2
    total = best_h.sum()
    target = N / 2.0
    diff = target - total
    # add diff to a point that doesn't break bounds
    if abs(diff) > 1e-12:
        idx = np.argmin(best_h) if diff > 0 else np.argmax(best_h)
        # ensure staying in bounds
        new_val = best_h[idx] + diff
        if 0 <= new_val <= 1:
            best_h[idx] = new_val
        else:
            # distribute small adjustment across all points
            best_h += diff / N
            best_h = np.clip(best_h, 0.0, 1.0)

    c5_val = max_overlap(best_h)

    # If we still have time, try simple local search improvement
    elapsed = time.time() - start
    remaining = budget_s - elapsed - 1.0
    if remaining > 0.5:
        # hill climbing
        cur_a = best_a.copy()
        cur_c5 = c5_val
        sigma = 0.05
        while time.time() - start < budget_s - 0.5:

            trial_a = cur_a + sigma * np.random.randn(M)
            # enforce bounds manually
            trial_a = np.clip(trial_a, -0.5, 0.5)
            # compute h

            if np.min(h) < -1e-9 or np.max(h) > 1 + 1e-9:

            # compute
            val = max_overlap(h)
            if val < cur_c5:
                cur_a = trial_a
                cur_c5 = val
                best_h = np.clip(h, 0.0, 1.0)

    best_h = np.clip(best_h, 0.0, 1.0)
    # adjust sum exactly once more

    diff = target - total
    if abs(diff) > 1e-12:

        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = max_overlap(best_h)

    return best_h, float(final_c5), N

import time
import numpy as np
from scipy.optimize import differential_evolution

def _max_overlap(h: np.ndarray) -> float:
    """Return C5 for a sampled step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear (zero‑padded) correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a step function h, its C5 bound, and the number of sample points.
    The function tries to minimise C5 using a cosine series and
    differential evolution followed by a short hill‑climb.
    """
    np.random.seed(seed)
    start = time.time()

    # --- discretisation ----------------------------------------------------
    N = 256                     # number of sample points (even)
    xs = np.arange(N) * (2.0 / N)   # sample positions on [0,2)

    # --- cosine basis ------------------------------------------------------
    M = 5                       # number of cosine terms
    # matrix rows: m = 1..M, columns: sample points
    cos_mat = np.cos(np.outer(np.arange(1, M + 1), np.pi * xs))

    # --- objective for the optimiser ---------------------------------------
    def objective(a):
        """C5 for the function defined by coefficients a."""
        h = 0.5 + a @ cos_mat          # shape (N,)
        # feasibility test (hard bounds)
        if h.min() < -1e-12 or h.max() > 1.0 + 1e-12:
            # large penalty, proportional to violation magnitude
            viol = max(0.0, -h.min()) + max(0.0, h.max() - 1.0)
            return 1e6 + 1e6 * viol
        return _max_overlap(h)

    # --- time‑limited callback ---------------------------------------------
    def callback(_xk, _convergence):
        return (time.time() - start) > (budget_s - 5.0)

    # --- differential evolution --------------------------------------------
    bounds = [(-0.5, 0.5) for _ in range(M)]
    de_result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=800,          # generous, will be cut by the callback
        popsize=15,
        tol=1e-7,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        polish=False,
        init='latinhypercube',
        callback=callback,
    )
    best_a = de_result.x
    # build the corresponding step function
    best_h = 0.5 + best_a @ cos_mat
    best_h = np.clip(best_h, 0.0, 1.0)

    # Exact sum correction (should already be N/2 up to rounding)
    target_sum = N / 2.0
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        # adjust the first element – the correction is tiny (<dx)
        best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)

    best_c5 = _max_overlap(best_h)

    # --- short hill‑climbing refinement ------------------------------------
    elapsed = time.time() - start
    remaining = budget_s - elapsed - 0.5
    if remaining > 0.0:
        cur_a = best_a.copy()
        cur_c5 = best_c5
        sigma = 0.07
        # simple random walk with decreasing step size
        while time.time() - start < budget_s - 0.3:
            trial_a = cur_a + sigma * np.random.randn(M)
            trial_a = np.clip(trial_a, -0.5, 0.5)
            h = 0.5 + trial_a @ cos_mat
            if h.min() < -1e-12 or h.max() > 1.0 + 1e-12:
                continue
            val = _max_overlap(h)
            if val < cur_c5:
                cur_a, cur_c5, best_h = trial_a, val, np.clip(h, 0.0, 1.0)
                best_c5 = cur_c5
            sigma *= 0.99995    # slowly anneal step size

        # final tiny sum fix
        cur_sum = best_h.sum()
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)
        best_c5 = _max_overlap(best_h)

    # -----------------------------------------------------------------------
    # Return the step function, its C5 bound, and the number of points.
    return best_h, float(best_c5), N
