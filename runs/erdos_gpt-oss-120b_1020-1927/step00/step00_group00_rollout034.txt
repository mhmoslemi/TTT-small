    rng = np.random.RandomState(seed)

    best_h = None
    best_val = np.inf

    n = 50

    # initial h = 0.5 everywhere
    h = np.full(n, 0.5)
    # We'll also create some random perturbations as starting points.

    def max_f(h):
        corr = np.fft.ifft(np.fft.fft(h, 2*n) * np.conj(np.fft.fft(1-h, 2*n))).real
        # take first 2n-1 entries
        c = corr[:2*n-1]
        return np.max(c) * dx

    current_val = max_f(h)
    best_h = h.copy()
    best_val = current_val

    # Simulated annealing schedule:
    T0 = 0.1
    T_end = 1e-4
    # We'll compute fraction of time
    iter = 0
    while time.time() - start_time < budget_s * 0.9:
        # produce neighbor
        i, j = rng.choice(n, size=2, replace=False)
        # compute delta range
        delta_low = max(-h[i], h[j] - 1.0)
        delta_high = min(1.0 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        delta = rng.uniform(delta_low, delta_high)
        # apply
        h[i] += delta
        h[j] -= delta
        new_val = max_f(h)
        # acceptance probability
        # temperature schedule linear in time
        t_frac = (time.time() - start_time) / (budget_s * 0.9)
        T = T0 * (1 - t_frac) + T_end * t_frac
        if new_val < current_val or rng.rand() < np.exp((current_val - new_val) / T):

            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                # optionally print progress for debugging? Not needed.
        else:
            # revert
            h[i] -= delta
            h[j] += delta

        iter += 1

    # After SA, we might try local refinement: small random perturbations around best.

def max_overlap(h):
    n = h.shape[0]
    dx = 2.0 / n
    # compute correlation by FFT:
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1 - h, N)
    c = np.fft.irfft(H * np.conj(G), N)
    # c contains length N real values; we need first 2*n-1 entries:
    corr = c[:2*n-1]
    # multiply by dx
    return np.max(corr) * dx

def neighbor(h, rng):
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # compute delta limit
    low = max(-h[i], h[j] - 1.0)
    high = min(1.0 - h[i], h[j])
    if low >= high:
        return None, None, None
    delta = rng.uniform(low, high)
    # apply
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new, i, delta

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    n = 200  # number of points
    h = (np.arange(n, dtype=np.float64) + 0.5) / n
    dx = 2.0 / n
    # compute overlap
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * dx)
    return h, c5, int(n)
