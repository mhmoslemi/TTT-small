def project_onto_capped_simplex(z, target):
    n = z.size
    # define lower and upper bound for lambda
    lo = -np.max(z) - 1.0
    hi = 1.0 - np.min(z) + 1.0

        s = np.sum(x)

            lo = mid
        else:
            hi = mid

def optimize(start_h, target_sum, time_limit, start_time):
    h = start_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    iter = 0
    while time.time() - start_time < time_limit:
        # compute correlation
        O = np.correlate(h, 1 - h, mode='full')
        O = O * (2.0 / n)
        max_val = np.max(O)
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
        # compute shift
        idx_max = np.argmax(O)
        s = idx_max - (n - 1)
        # gradient
        grad = compute_gradient(h, s)

        lr = 0.5 / math.sqrt(iter+1)

def project_onto_capped_simplex(z, target):
    n = len(z)
    # lower bound on lambda
    lam_low = -np.max(z) - 1.0
    lam_high = 1.0 - np.min(z) + 1.0
    # compute sum at low and high to ensure target within
    # Might have issues if target is out of range but not.
    for _ in range(60):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(z + lam, 0.0, 1.0)
        s = np.sum(x)
        if s < target:
            lam_low = lam
        else:
            lam_high = lam
    x = np.clip(z + (lam_low + lam_high) / 2.0, 0.0, 1.0)
    # To fix tiny numerical error, adjust by distributing diff over non-bound entries
    diff = target - np.sum(x)
    if abs(diff) > 1e-8:
        # distribute diff to entries not at bounds (0< x <1)
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if np.any(mask):
            x[mask] += diff / np.sum(mask)
            # clip again
            x = np.clip(x, 0, 1)
        else:
            # if all at bounds, adjust uniformly tiny
            x += diff / n
            x = np.clip(x, 0, 1)
    return x

import time
import math
import random
import numpy as np

def _project_capped_simplex(z: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of z onto {x | 0<=x_i<=1, sum x_i = target}."""
    n = z.size
    # bounds for lambda so that after shift everything is clipped to 0 or 1
    lo = -np.max(z) - 1.0
    hi = 1.0 - np.min(z) + 1.0
    # binary search for λ
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(z + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    x = np.clip(z + (lo + hi) * 0.5, 0.0, 1.0)
    # fix tiny residual error
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
    return x


def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k)) dx for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float((corr * dx).max())


def _worst_shift(h: np.ndarray):
    """Return (max value, shift index) where shift = idx-(n-1)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (n - 1)


def _gradient(h: np.ndarray, s: int) -> np.ndarray:
    """Sub‑gradient of the worst overlap term with respect to h."""
    n = h.shape[0]
    grad = np.ones(n)
    i_start = max(0, -s)
    i_end = min(n, n - s)
    i = np.arange(i_start, i_end, dtype=int)
    j = i + s
    # contributions -h_j to grad[i] and -h_i to grad[j]
    grad[i] -= h[j]
    grad[j] -= h[i]
    return grad


def _optimize(start_h: np.ndarray, target_sum: float,
              time_limit: float, start_time: float):
    """Projected sub‑gradient descent with occasional random swaps."""
    n = start_h.shape[0]
    h = start_h.copy()
    best_h = h.copy()
    best_val = _compute_c5(h)
    it = 0
    while time.time() - start_time < time_limit:
        # current worst shift
        cur_val, s = _worst_shift(h)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
        # sub‑gradient step
        g = _gradient(h, s)
        step = 0.5 / math.sqrt(it + 1)
        h_candidate = _project_capped_simplex(h - step * g, target_sum)
        # accept candidate (even if worse, projection keeps feasibility)
        h = h_candidate
        it += 1

        # occasional random perturbations to escape plateaus
        if it % 3000 == 0:
            for _ in range(6):
                i, j = np.random.choice(n, 2, replace=False)
                if h[i] > 0.0 and h[j] < 1.0:
                    delta = min(h[i], 1.0 - h[j]) * np.random.rand() * 0.2
                    h_tmp = h.copy()
                    h_tmp[i] -= delta
                    h_tmp[j] += delta
                    val = _compute_c5(h_tmp)
                    if val < best_val:
                        best_val = val
                        best_h = h_tmp.copy()
                        h = h_tmp
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """
    Find a step function h on [0,2] with ∫h=1 that minimizes
    C5 = max_k ∫ h(x)(1‑h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # obtain initial vector if supplied by the harness
    try:
        h0 = np.asarray(initial_h_values, dtype=float).reshape(-1)
    except NameError:
        h0 = np.random.rand(70)

    n = h0.size
    target = n / 2.0
    h0 = _project_capped_simplex(h0, target)

    # allocate most of the time to a first optimisation run
    time_allow = budget_s * 0.88
    best_h, best_val = _optimize(h0, target, time_allow, start)

    # try a fresh random start if time remains
    if time.time() - start < budget_s * 0.97:
        rand_start = np.random.rand(n)
        rand_start = _project_capped_simplex(rand_start, target)
        h1, val1 = _optimize(rand_start, target,
                             budget_s * 0.08, start)
        if val1 < best_val:
            best_h, best_val = h1, val1

    # final sanity check (clip any tiny numerical drift)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_capped_simplex(best_h, target)

    return (best_h, best_val, n)
