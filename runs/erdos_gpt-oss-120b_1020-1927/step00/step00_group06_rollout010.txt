import numpy as np, time, math

def _project_to_sum(v, target):
    # adjust small drift
    cur = np.sum(v)
    diff = target - cur
    if abs(diff) < 1e-12:
        return v
    # try to adjust a single element
    for i in range(v.size):
        lo, hi = 0.0, 1.0
        if lo <= v[i] + diff <= hi:
            v[i] += diff
            return v
    # fallback: distribute uniformly
    v += diff / v.size
    v = np.clip(v, 0.0, 1.0)
    # final correction
    cur = np.sum(v)
    diff = target - cur
    if abs(diff) > 1e-12:
        # try again
        for i in range(v.size):
            lo, hi = 0.0, 1.0
            if lo <= v[i] + diff <= hi:
                v[i] += diff
                break
    return v

import time, math
import numpy as np

def _project_to_sum(v, target):
    """tiny correction to keep sum(v)==target while respecting 0‑1 bounds"""
    cur = float(v.sum())
    diff = target - cur
    if abs(diff) < 1e-12:
        return v
    # try a single element
    for i in range(v.size):
        lo, hi = 0.0, 1.0
        if lo <= v[i] + diff <= hi:
            v[i] += diff
            return v
    # fallback: spread uniformly and clip
    v += diff / v.size
    np.clip(v, 0.0, 1.0, out=v)
    cur = float(v.sum())
    diff = target - cur
    if abs(diff) > 1e-12:
        for i in range(v.size):
            lo, hi = 0.0, 1.0
            if lo <= v[i] + diff <= hi:
                v[i] += diff
                break
    return v
