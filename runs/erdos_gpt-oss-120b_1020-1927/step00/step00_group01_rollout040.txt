import time
import numpy as np

def project_onto_box_sum(v, target, lower=0.0, upper=1.0, iters=60):
    """Euclidean projection of v onto the box [lower,upper] with sum constraint."""
    lo, hi = -2.0, 2.0  # wide enough for the needed shift
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, lower, upper)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, lower, upper)


def compute_c5(h):
    """C5 value for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def build_symmetric_5step(a, d2, n):
    """Construct a symmetric 5‑step function (0,a,1,a,0) on n points."""
    # lengths according to the derivation in the analysis
    d1 = 0.5 + d2 * (a - 1.0)          # left/right zero segment length
    d3 = 2.0 - 2.0 * (d1 + d2)        # central 1‑segment length
    if d1 < 0.0 or d3 < 0.0:
        return None
    dx = 2.0 / n
    # break points
    b0 = 0.0
    b1 = d1
    b2 = d1 + d2
    b3 = d1 + d2 + d3
    b4 = d1 + 2.0 * d2 + d3
    x = (np.arange(n) + 0.5) * dx
    h = np.empty(n, dtype=np.float64)
    h[x < b1] = 0.0
    h[(x >= b1) & (x < b2)] = a
    h[(x >= b2) & (x < b3)] = 1.0
    h[(x >= b3) & (x < b4)] = a
    h[x >= b4] = 0.0
    return h


def random_candidate_params(rng):
    """Draw a random (a,d2) pair respecting feasibility."""
    a = rng.random()
    if a <= 0.0:
        d2_max = 0.5
    elif a >= 1.0:
        d2_max = 0.5
    else:
        d2_max = min(0.5 / (1.0 - a), 0.5 / a)
    d2 = rng.random() * d2_max
    return a, d2


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # --------------------------------------------------------------
    # Choose a working grid size (even for symmetry)
    n = 200
    target_sum = n / 2.0

    # --------------------------------------------------------------
    # Initialise best solution with the provided parent if any
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        init_vec = np.asarray(init_vec, dtype=np.float64)
        n0 = init_vec.size
        if n0 != n:
            # linear interpolation onto the new grid
            x_old = (np.arange(n0) + 0.5) * (2.0 / n0)
            x_new = (np.arange(n) + 0.5) * (2.0 / n)
            h_best = np.interp(x_new, x_old, init_vec, left=0.0, right=0.0)
        else:
            h_best = init_vec.copy()
        h_best = project_onto_box_sum(h_best, target_sum, 0.0, 1.0)
        best_c5 = compute_c5(h_best)
    else:
        # fallback: uniform 0.5 function (already satisfies the integral)
        h_best = np.full(n, 0.5, dtype=np.float64)
        best_c5 = compute_c5(h_best)

    # --------------------------------------------------------------
    # Random search over the symmetric 5‑step family
    max_iter = 200000
    while True:
        if time.time() - start > budget_s * 0.70:
            break
        a, d2 = random_candidate_params(rng)
        h = build_symmetric_5step(a, d2, n)
        if h is None:

        h = project_onto_box_sum(h, target_sum, 0.0, 1.0)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h.copy()
        # a simple early‑stop check for a very good value
        if best_c5 < 0.381:

    # Local refinement: small pairwise transfers preserving the sum
    remaining = budget_s - (time.time() - start)

            if (step + 1) % 5000 == 0:
