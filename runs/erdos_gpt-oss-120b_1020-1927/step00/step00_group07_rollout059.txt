def max_overlap(h):
    n = h.shape[0]
    nfft = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft((1 - h)[::-1], n=nfft)
    conv = np.fft.irfft(H * G, n=nfft)
    max_val = conv.max()
    dx = 2.0 / n

import time
import numpy as np

def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the set {x | 0 <= x_i <= 1, sum(x) = target}
    by finding a scalar shift alpha and clipping.
    """
    low = -np.max(v)            # all entries become <= 0 after shift
    high = 1.0 - np.min(v)      # all entries become >= 1 after shift
    for _ in range(50):
        mid = (low + high) * 0.5
        clipped = np.clip(v + mid, 0.0, 1.0)
        s = clipped.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, 0.0, 1.0)


def _max_overlap(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1−h(x+k))dx for discrete step function h."""
    n = h.shape[0]
    conv = np.convolve(h, (1.0 - h)[::-1])      # full correlation
    return (2.0 / n) * conv.max()


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] (sampled uniformly) to minimize
    C5 = max_k ∫ h(x)(1−h(x+k))dx, under 0≤h≤1 and ∫h=1.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise ---------------------------------------------------------
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=np.float64)
        n = init_h.shape[0]
        if n % 2 == 1:                     # enforce even length for exact half‑mass
            # pad with zero to make length even
            init_h = np.r_[init_h, 0.0]
            n += 1
        h = np.clip(init_h, 0.0, 1.0)
    else:
        n = 200
        if n % 2 == 1:
            n += 1
        h = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0

    target_sum = n / 2.0
    h = _project_box_sum(h, target_sum)

    best_h = h.copy()
    best_score = _max_overlap(best_h)

    # ----- optimisation parameters -------------------------------------------
    eta = 0.5                     # initial step size
    decay = 0.9995                # geometric decay per iteration
    max_iter = 200000             # hard cap; actual loops stop by time budget
    last_improve = 0
    restart_interval = 5000       # steps without improvement -> perturb

    iteration = 0
    while True:
        # time check
        if time.time() - start_time > budget_s * 0.95:
            break
        if iteration >= max_iter:
            break

        # current correlation and score
        conv = np.convolve(h, (1.0 - h)[::-1])
        max_val = conv.max()
        cur_score = (2.0 / n) * max_val

        # keep best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_h = h.copy()
            last_improve = iteration

        # gradient of the active shift(s)
        max_idxs = np.where(conv == max_val)[0]
        grad = np.zeros_like(h)
        for idx in max_idxs:
            k = n - 1 - idx          # shift (can be negative)
            grad += 1.0
            if k >= 0:
                if k < n:
                    grad[:n - k] -= h[k:]
                    grad[k:] -= h[:n - k]
            else:
                pos = -k
                grad[pos:] -= h[:n - pos]
                grad[:n - pos] -= h[pos:]

        grad /= len(max_idxs)

        # gradient step with projection onto box & sum constraint
        h_new = h - eta * grad
        h_new = _project_box_sum(h_new, target_sum)

        # update
        h = h_new
        eta = max(eta * decay, 1e-6)
        iteration += 1

        # occasional random perturbation if stagnating
        if iteration - last_improve > restart_interval:
            # perturb around the best found so far
            noise = rng.normal(scale=0.02, size=n)
            h = _project_box_sum(best_h + noise, target_sum)
            eta = 0.5
            last_improve = iteration

    # final evaluation on best solution
    final_score = _max_overlap(best_h)

    return best_h, final_score, n
