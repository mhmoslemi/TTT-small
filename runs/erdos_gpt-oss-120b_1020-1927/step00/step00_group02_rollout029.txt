import time
import numpy as np

# --------------------------------------------------------------
# Projection of a vector onto the box [0,1] together with a sum
# constraint sum(x) = target (in L2 sense)
# --------------------------------------------------------------
def project_onto_box_and_sum(x, target):
    """Euclidean projection of x onto {0<=z<=1, sum(z)=target}."""
    lo, hi = -1.0, 1.0                     # bounds for the dual variable λ
    for _ in range(60):                    # binary search for λ
        mid = (lo + hi) * 0.5
        z = np.clip(x - mid, 0.0, 1.0)
        s = z.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction for numerical drift
    diff = target - z.sum()
    if abs(diff) > 1e-12:
        idx = np.where((z > 0.0) & (z < 1.0))[0]
        if idx.size:
            i = idx[0]
            z[i] = min(1.0, max(0.0, z[i] + diff))
        else:                               # all at bounds – rescale proportionally
            s = z.sum()
            if s > 0:
                z *= target / s
    return z


# --------------------------------------------------------------
# Objective functions for the Erdős C5 problem
# --------------------------------------------------------------
def compute_c5(h):
    """Exact C5 value using the problem's correlation definition."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def compute_O(h):
    """Vector O_ℓ = Σ_{i< n-ℓ} h_i (1 - h_{i+ℓ}) for all lags ℓ."""
    n = h.shape[0]
    prefix = np.concatenate(([0.0], np.cumsum(h)))          # length n+1
    corr = np.correlate(h, h, mode="full")
    corr_part = corr[n - 1:]                               # ordered products for ℓ=0..n-1
    ℓ = np.arange(n)
    O = prefix[n - ℓ] - corr_part
    return O


def gradient_of_O(h, ell):
    """Gradient of O_ell (the current worst lag) with respect to h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if ell < n:
        m = n - ell
        g[:m] += 1.0
        g[:m] -= h[ell:]          # -h_{i+ell}
        g[ell:] -= h[:m]          # -h_{i-ell}
    return g


# --------------------------------------------------------------
# Simple projected gradient descent on the worst lag
# --------------------------------------------------------------
def local_optimize(h_init, target_sum, time_budget):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    no_improve = 0
    alphas = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]

    while time.time() - start < time_budget:
        O = compute_O(h)
        ell = int(np.argmax(O))            # the lag that currently gives max overlap
        g = gradient_of_O(h, ell)

        improved = False
        for a in alphas:
            y = h - a * g
            y = project_onto_box_and_sum(y, target_sum)
            val = compute_c5(y)
            if val < best_val - 1e-12:
                best_val = val
                best_h = y.copy()
                h = y
                improved = True
                no_improve = 0
                break

        if not improved:
            # small random jitter to escape plateaux
            y = h + 0.01 * (np.random.rand(len(h)) - 0.5)
            y = project_onto_box_and_sum(y, target_sum)
            val = compute_c5(y)
            if val < best_val:
                best_val = val
                best_h = y.copy()
                h = y
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 200:
                    break
    return best_h, best_val
