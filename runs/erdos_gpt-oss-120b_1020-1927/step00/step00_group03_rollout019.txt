    def project_sum(vec):

        np.clip(vec, 0.0, 1.0, out=vec)

            np.clip(vec, 0.0, 1.0, out=vec)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Starting vector (use provided one if present)
    # ------------------------------------------------------------------
    init_glob = globals().get('initial_h_values')
    if init_glob is not None:
        base = np.asarray(init_glob, dtype=np.float64)
        n0 = base.shape[0]
        # modest up‑sampling (keep total points ≤ 200)
        max_n = 200
        factor = max_n // n0
        factor = max(1, min(factor, 3))
        h = np.repeat(base, factor) if factor > 1 else base.copy()
    else:
        # fallback: random feasible vector
        n = 140
        h = rng.uniform(size=n)
        target_sum = n / 2.0
        h = h * (target_sum / h.sum())
    n = h.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # 2.  Projection onto the box [0,1] and the affine sum constraint
    # ------------------------------------------------------------------
    def project_sum(v):
        np.clip(v, 0.0, 1.0, out=v)
        diff = target_sum - v.sum()
        for _ in range(10):
            if abs(diff) < 1e-12:
                break
            free = (v > 1e-12) & (v < 1.0 - 1e-12)
            cnt = int(free.sum())
            if cnt == 0:
                break
            delta = diff / cnt
            v[free] += delta
            np.clip(v, 0.0, 1.0, out=v)
            diff = target_sum - v.sum()
        if abs(diff) > 1e-9:
            free = (v > 1e-12) & (v < 1.0 - 1e-12)
            idx = np.flatnonzero(free)
            if idx.size:
                i = idx[0]
                v[i] = np.clip(v[i] + diff, 0.0, 1.0)
        return v

    h = project_sum(h)

    # ------------------------------------------------------------------
    # 3.  Objective (max overlap)
    # ------------------------------------------------------------------
    def max_overlap(arr):
        return np.max(np.correlate(arr, 1.0 - arr, mode='full')) * dx

    cur_val = max_overlap(h)
    best_h = h.copy()
    best_val = cur_val

    # ------------------------------------------------------------------
    # 4.  Simulated‑annealing loop
    # ------------------------------------------------------------------
    start = time.time()
    T0 = 0.05                       # initial temperature
    total = max(budget_s - 0.5, 0.1) # leave a tiny margin
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= total:
            break
        T = T0 * max(0.0, 1.0 - elapsed / total)

        # ---- simple two‑index delta move (preserves sum) ----
        i, j = rng.integers(0, n, size=2)
        while j == i:
            j = rng.integers(0, n)
        hi, hj = h[i], h[j]
        low = max(-hi, -(1.0 - hj))
        high = min(1.0 - hi, hj)
        if low >= high:
            continue

        max_step = min(0.2, (high - low) * 0.5, T * 5.0 if T > 0 else 0.2)

        hi_n = hi + delta
        hj_n = hj - delta
        if not (0.0 <= hi_n <= 1.0 and 0.0 <= hj_n <= 1.0):
            continue
        cand = h.copy()
        cand[i], cand[j] = hi_n, hj_n
        val = max_overlap(cand)
        if val < cur_val or rng.random() < math.exp(-(val - cur_val) / (T + 1e-12)):
            h = cand
            cur_val = val
            if val < best_val:
                best_val, best_h = val, cand.copy()

        # ---- occasional gradient descent on the current worst shift ----
        if (it & 1023) == 0:              # every 1024 iterations
