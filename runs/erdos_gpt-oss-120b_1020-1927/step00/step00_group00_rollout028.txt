def opt_h(initial_h, max_iter=500, beta=50, tol=1e-9):

    bounds = [(0,1)]*n

    def obj(h):
        conv = np.correlate(h, 1 - h, mode='full')
        # smooth max

        # compute smoothed max using log-sum-exp stable

import time
import numpy as np
from scipy.optimize import minimize

# global smoothing parameter for the objective
OBJECTIVE_BETA = 1.0


def project_onto_sum_and_box(v: np.ndarray, target: float, tol: float = 1e-9) -> np.ndarray:
    """Project v onto the hyperplane sum(v)=target with box 0≤v≤1."""
    lo, hi = -1.0, 1.0  # lambda bounds guarantee sums n and 0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > tol:
        free = (w > 0.0) & (w < 1.0)
        cnt = int(free.sum())
        if cnt:
            w[free] += diff / cnt
            w = np.clip(w, 0.0, 1.0)
    return w


def constraint_sum(h: np.ndarray) -> float:
    """Equality constraint: integral of h must be 1 (sum = n/2)."""
    return h.sum() - h.size / 2.0


def compute_exact_c5(h: np.ndarray) -> float:
    """Exact C5 value from the definition."""
    n = h.size
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / n)


def smooth_obj_and_grad(h: np.ndarray):
    """Smooth approximation of max_{k} ∫h(1‑hₖ) and its gradient."""
    n = h.size
    dx = 2.0 / n
    beta = OBJECTIVE_BETA
    q = 1.0 - h
    conv = np.correlate(h, q, mode="full")
    m = conv.max()
    # stable log‑sum‑exp
    exp_shift = np.exp(beta * (conv - m))
    Z = exp_shift.sum()
    w = exp_shift / Z                # softmax weights
    sval = m + np.log(Z) / beta      # smooth max
    obj = sval * dx

    # gradient
    rev_q = q[::-1]
    # term1 = Σ w_i * rev_q[i‑k]
    cor1 = np.correlate(w, rev_q, mode="full")
    term1 = cor1[n - 1 :: -1][:n]    # pick n entries decreasing from centre
    # term2 = Σ w_i * h[i+k-(n‑1)]
    cor2 = np.correlate(w, h, mode="full")
    term2 = cor2[:n]
    grad = (term1 - term2) * dx
    return obj, grad


def objective_smooth(h: np.ndarray) -> float:
    return smooth_obj_and_grad(h)[0]


def grad_smooth(h: np.ndarray) -> np.ndarray:
    return smooth_obj_and_grad(h)[1]


def hill_climb(h: np.ndarray, iters: int, rng: np.random.Generator, step: float = 0.02):
    """Simple random 2‑point exchange that never breaks the constraints."""
    best = h.copy()
    best_val = compute_exact_c5(best)
    n = best.size
    for _ in range(iters):
        i, j = rng.choice(n, 2, replace=False)
        # delta added to i, subtracted from j
        dmin = max(-best[i], best[j] - 1.0)
        dmax = min(1.0 - best[i], best[j])
        if dmax <= dmin:
            continue
        delta = rng.uniform(dmin, dmax)
        if abs(delta) > step:
            delta = step if delta > 0 else -step
        cand = best.copy()
        cand[i] += delta
        cand[j] -= delta
        val = compute_exact_c5(cand)
        if val + 1e-12 < best_val:
            best, best_val = cand, val
    return best, best_val

    best_h = project_onto_sum_and_box(best_h, best_n / 2.0)
    best_c5 = compute_exact_c5(best_h)

    return best_h, float(best_c5), int(best_n)
