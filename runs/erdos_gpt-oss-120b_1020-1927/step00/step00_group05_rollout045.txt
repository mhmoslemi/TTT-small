def compute_crosscorr(h):
    return np.correlate(h, h, mode='full') # size 2n-1.

def gradient(h, alpha):
    A = compute_crosscorr(h)  # shape (2n-1)
    B = -A  # negative
    # compute softmax weights w
    max_b = np.max(alpha * B) # may be large; better stable: compute softmax stability
    # w = exp(alpha*B)/sum(exp(alpha*B))
    # Compute exp(alpha*(B - maxB/alpha))? Actually compute w = np.exp(alpha*B - logsumexp(alpha*B)), using scipy.special.logsumexp or implement stable.
    from scipy.special import logsumexp
    logZ = logsumexp(alpha * B)
    w = np.exp(alpha * B - logZ) # shape (2n-1)
    # compute convolutions
    conv1 = np.convolve(w, h, mode='full')
    conv2 = np.convolve(w[::-1], h, mode='full')
    start = len(h)-1
    grad = - (conv1[start:start+len(h)] + conv2[start:start+len(h)])
    return grad

def project_to_box_sum(v, target):
    # v may have values outside [0,1] as we call with triangle shape (within [0,1] already).
    # We'll solve min ||x - v||^2 s.t. sum x = target, 0 ≤ x ≤ 1.
    # We'll use water-filling: find λ such that sum clamp(v - λ, 0,1) = target.
    # Use binary search between low and high λ.
    lo = -1.0
    hi = 1.0
    # Expand bounds: while sum(clamp(v - lo)) < target: lo -= 1
    # Actually for λ decreasing (more negative), v-λ increases → clamp to 1 saturates -> sum increase.

    # We'll start lo = -2.0, hi = 2.0 and expand if needed.

        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, 0, 1)

import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the box [0,1]^n together with a sum constraint
# ------------------------------------------------------------
def _proj_box_sum(v: np.ndarray, target: float, it: int = 60) -> np.ndarray:
    """
    Find the Euclidean projection of v onto the set
        {x | 0 <= x_i <= 1, sum(x) = target }.
    This is solved by a simple binary search on the Lagrange multiplier λ
    such that sum(clip(v - λ, 0, 1)) = target.
    """
    lo, hi = -5.0, 5.0

    # enlarge the interval until the monotone condition is satisfied
    while np.clip(v - lo, 0.0, 1.0).sum() <= target:
        lo -= 5.0
    while np.clip(v - hi, 0.0, 1.0).sum() >= target:
        hi += 5.0

    for _ in range(it):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


# ------------------------------------------------------------
# Helper: compute the C5 value for a discrete step function
# ------------------------------------------------------------
def _c5_value(h: np.ndarray) -> float:
    """max_k ∫ h(x)(1‑h(x+k))dx approximated by discrete correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


# ------------------------------------------------------------
# Main optimisation routine (simple gradient descent on a smooth
# approximation of the max‑overlap objective)
# ------------------------------------------------------------
def _smooth_max_gradient(h: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of   (1/α) log Σ exp(α·B_s)   with B_s = -⟨h,shift_s h⟩."""
    n = h.size
    # auto‑correlation A_s = Σ_i h_i h_{i+s}
    A = np.correlate(h, h, mode="full")          # length 2n‑1, shift s = idx-(n‑1)
    B = -A                                       # B_s = –A_s
    # soft‑max weights w_s = exp(α·B_s) / Σ exp(α·B_s)
    scaled = alpha * B
    max_scaled = scaled.max()
    exp_shifted = np.exp(scaled - max_scaled)    # numerical stability
    w = exp_shifted / exp_shifted.sum()

    # gradient = - Σ_s w_s (h_{j+s}+h_{j‑s})
    conv_fwd = np.convolve(w, h, mode="full")
    conv_rev = np.convolve(w[::-1], h, mode="full")
    start = n - 1
    grad = -(conv_fwd[start:start + n] + conv_rev[start:start + n])
    return grad


# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Constructs a step function that beats the reference bound 0.38085.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # --------------------------------------------------------
    # 1.  Build a triangular shape (the known extremal continuous solution)
    # --------------------------------------------------------
    n = 400                               # number of discretisation points (fast enough)
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx        # mid‑points of the sub‑intervals
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))

    # enforce the exact integral constraint Σ h_i = n/2
    target = n * 0.5
    h = _proj_box_sum(h, target)

    # --------------------------------------------------------
    # 2.  Optional short refinement – a few gradient‑descent steps
    # --------------------------------------------------------
    # parameters for the smoothing and the steps
    alpha = 30.0                # larger → closer to the true max
    lr = 0.8                    # learning rate
    max_iter = 80

    # current best value
    best_c5 = _c5_value(h)

    for it in range(max_iter):
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break

        grad = _smooth_max_gradient(h, alpha)

        # simple backtracking line‑search
        for _ in range(12):
            h_candidate = _proj_box_sum(h - lr * grad, target)
            cand_c5 = _c5_value(h_candidate)
            if cand_c5 < best_c5:
                h = h_candidate
                best_c5 = cand_c5
                lr *= 1.05          # a tiny boost after success
                break
            lr *= 0.5               # shrink step size if no improvement
        else:
            # no improvement – stop early
            break

    # --------------------------------------------------------
    # 3.  Final evaluation and return
    # --------------------------------------------------------
    final_c5 = _c5_value(h)

    return (h, float(final_c5), n)
