def evaluate(h):
    conv = np.correlate(h, h, mode='full')
    # product integral values
    # min product:
    min_prod = conv.min() * (2 / len(h))
    # Compute C5 bound
    c5 = 1 - min_prod
    return c5, conv

def project(v):
    target = len(v) / 2  # sum = n/2
    # find lambda such that sum(clip(v + lam, 0, 1)) = target
    lo = -5.0
    hi = 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid

def shift_fill(arr, shift):
    n = len(arr)
    res = np.zeros_like(arr)
    if shift > 0:
        res[shift:] = arr[:n-shift]
    elif shift < 0:
        res[:n+shift] = arr[-shift:]
    else:
        res = arr.copy()

def grad_min_corr(h):
    conv = np.correlate(h, h, mode='full')
    s_idx = conv.argmin()
    shift = s_idx - (len(h)-1)
    # gradient: shift_fill(h, shift) + shift_fill(h, -shift)
    g = shift_fill(h, shift) + shift_fill(h, -shift)

def anneal_binary(h, time_limit, start_time):
    n = len(h)
    k = n // 2
    cur = h.copy()
    cur_val = compute_c5(cur)
    best = cur.copy()
    best_val = cur_val
    T0 = 0.05
    # Determine approximate number of steps: we will use while time < limit
    it = 0
    while time.time() - start_time < time_limit:
        # schedule temperature:
        elapsed = time.time() - start_time
        frac = elapsed / time_limit
        T = T0 * (1 - frac)  # linear cooling
        # propose swap
        one_idx = np.where(cur > 0.5)[0]
        zero_idx = np.where(cur < 0.5)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:

        cand = cur.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_val = compute_c5(cand)
        delta = cand_val - cur_val
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-9)):
            cur = cand
            cur_val = cand_val
            if cur_val < best_val:
                best = cur.copy()
                best_val = cur_val

import numpy as np
import time

def _project_box_sum(v, target):
    """Project v onto [0,1] box with sum(v)=target."""
    lo = -np.max(v) - 10.0
    hi = 1.0 - np.min(v) + 10.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)


def _shift_fill(arr, s):
    """Shift arr by s (positive => right) with zero fill."""
    n = arr.shape[0]
    out = np.zeros_like(arr)
    if s > 0:
        out[s:] = arr[:n - s]
    elif s < 0:
        out[:n + s] = arr[-s:]
    else:
        out[:] = arr
    return out


def _gradient_min_corr(h):
    """Gradient of the minimal autocorrelation term."""
    conv = np.correlate(h, h, mode='full')
    idx = conv.argmin()
    shift = idx - (len(h) - 1)
    grad = _shift_fill(h, shift) + _shift_fill(h, -shift)
    return grad, shift, conv[idx]


def _c5_val(h):
    """Compute the Erdős C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _random_binary(n):
    """Uniform random binary vector with exactly n/2 ones (n assumed even)."""
    k = n // 2
    h = np.zeros(n, dtype=float)
    h[:k] = 1.0
    np.random.shuffle(h)
    return h


def _anneal_binary(start_h, max_seconds):
    """Simple simulated‑annealing on binary vectors."""
    n = start_h.shape[0]
    cur = start_h.copy()
    cur_val = _c5_val(cur)
    best = cur.copy()
    best_val = cur_val
    T0 = 0.02
    start = time.time()
    while time.time() - start < max_seconds:
        frac = (time.time() - start) / max_seconds
        T = T0 * (1.0 - frac)
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = cur.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_val = _c5_val(cand)
        delta = cand_val - cur_val
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            cur = cand
            cur_val = cand_val
            if cur_val < best_val:
                best = cur.copy()
                best_val = cur_val
    return best, best_val


def _refine_fractional(start_h, max_seconds):
    """Gradient ascent on the minimal autocorrelation (fractional h)."""
    n = start_h.shape[0]
    target = n / 2.0
    cur = start_h.copy()
    cur_val = _c5_val(cur)
    best = cur.copy()
    best_val = cur_val
    step = 0.2
    start = time.time()
    while time.time() - start < max_seconds:
        grad, _, _ = _gradient_min_corr(cur)
        cand = _project_box_sum(cur + step * grad, target)
        cand_val = _c5_val(cand)
        if cand_val < best_val:
            best, best_val = cand, cand_val
            cur, cur_val = cand, cand_val
            step = min(step * 1.1, 1.5)
        else:
            cur, cur_val = cand, cand_val
            step = max(step * 0.7, 1e-4)
        if np.random.rand() < 0.02:
            pert = np.random.randn(n) * 0.05
            cur = _project_box_sum(cur + pert, target)
            cur_val = _c5_val(cur)
    return best, best_val


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_total = time.time()

    # number of discretisation points
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
    else:
        n = 70
    target_sum = n / 2.0

    # start from a binary version of the parent if available
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        idx = np.argsort(-init_arr)[:int(target_sum)]
        best_h = np.zeros(n, dtype=float)
        best_h[idx] = 1.0
    else:
        best_h = _random_binary(n)

    best_val = _c5_val(best_h)

    # ----- simulated annealing on many short runs -----
    elapsed = time.time() - start_total
    ann_time = budget_s * 0.6 - elapsed
    if ann_time > 0:
        ann_start = time.time()
        while time.time() - ann_start < ann_time:
            # each short run gets about 1 s (or less at the end)
            slice_time = min(1.0, ann_time - (time.time() - ann_start))
            if slice_time <= 0:
                break
            start_vec = _random_binary(n)
            cand_h, cand_val = _anneal_binary(start_vec, slice_time)
            if cand_val < best_val:
                best_h, best_val = cand_h, cand_val

    # ----- gradient refinement on the best solution -----
    elapsed = time.time() - start_total
    remain = budget_s - elapsed - 0.1
    if remain > 0.1:
        refined_h, refined_val = _refine_fractional(best_h, remain)
        if refined_val < best_val:
            best_h, best_val = refined_h, refined_val

    # final projection to guarantee feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_box_sum(best_h, target_sum)
    final_val = _c5_val(best_h)

    return best_h, float(final_val), n
