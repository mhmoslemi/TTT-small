def compute_overlap_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_val = corr.max()
    # Identify all lags with max
    max_idxs = np.where(corr == max_val)[0]
    # compute gradient by averaging over all max lags
    grad = np.zeros_like(h)
    for idx in max_idxs:
        lag = idx - (n - 1)
        if lag >= 0:
            s = lag
            # forward shift
            # for i 0..n-s-1: + (1 - h[i+s])*dx
            grad[:n - s] += (1 - h[s:]) * dx
            # for i s..n-1: -h[i - s] * dx
            grad[s:] += -h[:n - s] * dx
        else:
            s = -lag
            # backward shift
            # for i s..n-1: + (1 - h[i - s])*dx
            grad[s:] += (1 - h[:n - s]) * dx
            # for i 0..n - s -1: - h[i + s] * dx
            grad[:n - s] += -h[s:] * dx
    grad /= len(max_idxs)
    return max_val, grad

            grad[:n-s] += -h[s:] * dx

    # scale max_val
    max_val = max_val_unscaled * dx
    return max_val, grad

import time
import random
import numpy as np

# --------------------------------------------------------------
# Helper functions (top level, no closures)
# --------------------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Max overlap integral for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_sum_box(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Projection of v onto the hypercube [0,1]^n under the additional affine
    constraint sum(v)=target_sum.  Uses a simple bisection on the Lagrange
    multiplier lambda for the projection:  clip(v-lambda,0,1).
    """
    lo, hi = -1e6, 1e6
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def compute_overlap_and_grad(h: np.ndarray):
    """
    Returns (max_overlap, grad) where max_overlap is the same quantity as
    compute_c5 and grad is a sub‑gradient of this max w.r.t. h.
    """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_unscaled = float(np.max(corr))
    idxs = np.where(corr == max_unscaled)[0]

    grad = np.zeros_like(h)
    for idx in idxs:
        lag = idx - (n - 1)            # lag ∈ [-(n-1), n-1]
        if lag >= 0:                    # forward shift
            s = lag
            if s < n:
                grad[: n - s] += (1.0 - h[s:]) * dx
                grad[s:] += -h[: n - s] * dx
        else:                           # backward shift
            s = -lag
            if s < n:
                grad[s:] += (1.0 - h[: n - s]) * dx
                grad[: n - s] += -h[s:] * dx
    grad /= float(len(idxs))
    max_scaled = max_unscaled * dx
    return max_scaled, grad


def binary_hill_climb(start_h: np.ndarray, time_limit: float):
    """
    Simple hill‑climbing on binary vectors (0/1) that preserves exactly
    n/2 ones.  Swaps a random 1 with a random 0 and accepts the move
    if the objective does not increase.
    """
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)

    best_h = cur_h.copy()
    best_c = cur_c

    ones = list(np.where(cur_h == 1)[0])
    zeros = list(np.where(cur_h == 0)[0])

    t0 = time.time()
    while time.time() - t0 < time_limit:
        i = random.choice(ones)   # index with a 1
        j = random.choice(zeros)  # index with a 0

        # perform tentative swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c = compute_c5(cur_h)

        if new_c <= cur_c + 1e-12:                # accept non‑worsening move
            cur_c = new_c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)

            if new_c < best_c - 1e-12:
                best_c = new_c
                best_h = cur_h.copy()
        else:                                      # reject
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    return best_h, best_c


def gradient_descent(start_h: np.ndarray, max_iters: int, time_limit: float):
    """
    Projected sub‑gradient descent on the continuous problem.
    """
    n = start_h.shape[0]
    target = n / 2.0
    h = project_onto_sum_box(start_h.astype(float), target)

    best_h = h.copy()
    best_c = compute_c5(h)

    step = 0.5
    t0 = time.time()
    for _ in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        _, grad = compute_overlap_and_grad(h)
        new_h = h - step * grad
        new_h = project_onto_sum_box(new_h, target)
        new_c = compute_c5(new_h)

        if new_c < best_c - 1e-12:
            best_c = new_c
            best_h = new_h.copy()
            h = new_h
            step = min(step * 1.05, 5.0)        # slightly enlarge step after improvement
        else:
            step *= 0.7                         # shrink step on failure
            if step < 1e-6:                     # occasional random shake‑up
                step = 0.5
                eps = np.random.uniform(-0.1, 0.1, size=n)
                h = np.clip(h + eps, 0.0, 1.0)
                h = project_onto_sum_box(h, target)
    return best_h, best_c


def random_binary_vector(n: int) -> np.ndarray:
    """Binary vector of length n with exactly n/2 ones."""
    arr = np.zeros(n, dtype=float)
    arr[: n // 2] = 1.0
    np.random.shuffle(arr)
    return arr
