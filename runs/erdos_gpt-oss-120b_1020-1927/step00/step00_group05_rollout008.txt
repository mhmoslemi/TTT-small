def phi_and_grad(h, alpha, dx):
    n = len(h)
    b = 1 - h
    corr = np.correlate(h, b, mode='full')
    F = corr * dx
    # compute softmax weights w over full length 2n-1
    maxF = F.max()
    exp_shift = np.exp(alpha * (F - maxF))
    w = exp_shift / exp_shift.sum()
    # compute term1 using sliding dot product
    # vectorization: we can compute T1 via convolution? loops.
    T1 = np.empty(n)
    # compute slices using precomputed w slices for each j
    # loop
    for j in range(n):
        start = n - 1 - j
        T1[j] = np.dot(w[start:start+n], b)
    # term2
    rev_h = h[::-1]
    T2 = np.empty(n)
    for j in range(n):
        T2[j] = np.dot(w[j:j+n], rev_h)
    grad = dx * (T1 - T2)
    # phi value
    phi = (maxF + np.log(np.exp(alpha * (F - maxF)).sum())) / alpha
    return phi, grad

import time
import numpy as np

# ---------- helper functions ----------

def project_to_capped_simplex(z, target):
    """Project z onto {x: 0<=x_i<=1, sum x_i = target}."""
    # We solve x_i = clip(z_i - lam, 0, 1) for a scalar lam.
    low, high = -1.0, 2.0  # initial bounds for lambda
    # ensure brackets contain a feasible solution
    for _ in range(60):
        s_low = np.clip(z - low, 0.0, 1.0).sum()
        if s_low < target:
            low -= (high - low)  # expand lower bound
        else:
            break
    for _ in range(60):
        s_high = np.clip(z - high, 0.0, 1.0).sum()
        if s_high > target:
            high += (high - low)  # expand upper bound
        else:
            break
    # bisection
    for _ in range(60):
        mid = (low + high) * 0.5
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(z - lam, 0.0, 1.0)


def compute_C5(h, dx):
    """Exact (discretized) overlap constant for a given h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def phi_and_grad(h, alpha, dx):
    """
    Smoothed max objective phi = (1/alpha) logsumexp(alpha*F)
    where F = corr * dx.
    Returns phi and gradient wrt h.
    """
    n = h.size
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')          # length 2n-1
    F = corr * dx                                    # scaled overlaps

    # softmax weights
    maxF = F.max()
    exp_shift = np.exp(alpha * (F - maxF))
    w = exp_shift / exp_shift.sum()                # shape (2n-1,)

    # term1: sum_ℓ w_ℓ * b[j+ℓ]   (indices outside range are treated as 0)
    T1 = np.empty(n)
    for j in range(n):
        start = n - 1 - j
        T1[j] = np.dot(w[start:start + n], b)

    # term2: sum_ℓ w_ℓ * h[j-ℓ]  (using reversed h)
    rev_h = h[::-1]
    T2 = np.empty(n)
    for j in range(n):
        T2[j] = np.dot(w[j:j + n], rev_h)

    grad = dx * (T1 - T2)                           # gradient of phi

    # actual phi value
    phi = (maxF + np.log(exp_shift.sum())) / alpha
    return phi, grad


def projected_gradient_descent(h, target, dx, max_iter, alphas,
                              time_limit, start_time):
    """Simple PGD using the smoothed max as surrogate."""
    n = h.size
    for alpha in alphas:
        step = 0.5
        for _ in range(max_iter):
            if time.time() - start_time > time_limit:
                return h
            phi, grad = phi_and_grad(h, alpha, dx)

            # backtracking line search
            ls_step = step
            for _ in range(10):
                cand = project_to_capped_simplex(h - ls_step * grad, target)
                cand_phi, _ = phi_and_grad(cand, alpha, dx)
                if cand_phi <= phi:
                    h = cand
                    step = ls_step * 1.05
                    break
                ls_step *= 0.5
            else:
                # no improvement; shrink step and exit inner loop
                step *= 0.5
                if step < 1e-6:
                    break
    return h


def local_search(h, target, dx, time_limit, start_time, rng):
    """Simple pairwise mass transfers to greedily lower C5."""
    n = h.size
    best = h.copy()
    best_c5 = compute_C5(best, dx)

    # pre‑compute indices that can give positive deltas
    while time.time() - start_time < time_limit:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount that can be moved from i to j while staying in [0,1]
        max_delta = min(best[i], 1.0 - best[j])
        if max_delta <= 1e-12:
            continue
        delta = max_delta * (0.5 + 0.5 * rng.random())   # between 0.5*max and max
        new_h = best.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c5 = compute_C5(new_h, dx)
        if new_c5 < best_c5 - 1e-9:
            best = new_h
            best_c5 = new_c5
    return best


# ---------- main entry point ----------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = budget_s

    # ----- initialise h -----
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=np.float64)
        # up‑sample for a finer grid (factor 2 works well and stays fast)
        factor = 2
        h0 = np.repeat(init, factor)
    else:
        # fallback: uniform 0.5 function
        n = 120
        h0 = np.full(n, 0.5, dtype=np.float64)

    n = h0.size
    dx = 2.0 / n
    target = n * 0.5                     # sum must equal n/2

    h0 = project_to_capped_simplex(h0, target)

    best_h = h0.copy()
    best_c5 = compute_C5(best_h, dx)

    # ----- main optimisation loop -----
    # allocate most of the budget to iterative improvement
    while time.time() - start_time < time_limit - 5.0:
        # generate a starting point
        if rng.random() < 0.3:
            # small perturbation of the current best
            start_vec = best_h + rng.normal(scale=0.04, size=n)
        else:
            # fresh random point in the box
            start_vec = rng.random(n)
        start_vec = project_to_capped_simplex(start_vec, target)

        # gradient descent with a schedule of smoothing parameters
        start_vec = projected_gradient_descent(
            start_vec, target, dx,
            max_iter=150,
            alphas=[5.0, 15.0, 30.0, 60.0],
            time_limit=time_limit,
            start_time=start_time
        )

        # optional local refinement
        start_vec = local_search(
            start_vec, target, dx,
            time_limit=time_limit,
            start_time=start_time,
            rng=rng
        )

        cur_c5 = compute_C5(start_vec, dx)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = start_vec.copy()

    return best_h.tolist(), float(best_c5), n
