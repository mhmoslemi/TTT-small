def compute_c5(h, dx):
    return np.max(np.convolve(h, (1 - h)[::-1])) * dx

def random_binary(n, rng):
    h = np.zeros(n, dtype=float)
    idx = np.arange(n)
    rng.shuffle(idx)
    h[idx[:n//2]] = 1.0
    return h

import time
import numpy as np

def _random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = np.arange(n)
    rng.shuffle(idx)
    h[idx[: n // 2]] = 1.0
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function h (as a numpy array), its C5 bound, and the
    number of discretisation points.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Initialise ----------------------------------------------------
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        h0 = np.asarray(init, dtype=float).copy()
        n = h0.shape[0]
        # ensure binary – if not, fall back to a random binary start
        if not np.all(np.logical_or(np.isclose(h0, 0), np.isclose(h0, 1))):
            h0 = _random_binary(n, rng)
    else:
        n = 200  # a reasonable size when no starting vector is given
        h0 = _random_binary(n, rng)

    # guarantee exact half‑mass (required for binary vectors)
    target_ones = n // 2
    if int(h0.sum()) != target_ones:
        h0[:] = 0.0
        h0[:target_ones] = 1.0
        rng.shuffle(h0)

    dx = 2.0 / n

    def _c5(h_vec: np.ndarray) -> float:
        """C5 evaluation for a binary vector h_vec."""
        # correlation = convolution of h with reversed (1‑h)
        conv = np.convolve(h_vec, (1.0 - h_vec)[::-1])
        return np.max(conv) * dx

    best_h = h0.copy()
    best_val = _c5(best_h)

    cur_h = best_h.copy()
    cur_val = best_val

    # current positions of 1s and 0s (use numpy arrays for fast indexing)
    ones = np.where(cur_h == 1)[0]
    zeros = np.where(cur_h == 0)[0]

    # ------------------------------------------------------------------
    # 2.  Simulated‑annealing search ------------------------------------
    # ------------------------------------------------------------------
    start = time.time()
    deadline = start + budget_s - 0.1  # tiny safety margin

    max_iters = 10_000_000
    it = 0
    no_imp = 0
    # after this many useless swaps we restart from a fresh random state
    restart_after = 30_000

    while time.time() < deadline and it < max_iters:
        # temperature shrinks linearly from 0.01 to ~0
        elapsed = time.time() - start
        frac = min(elapsed / (budget_s * 0.9), 1.0)
        temp = (1.0 - frac) * 0.01 + 1e-9

        # propose a swap of a 1 and a 0
        i = rng.choice(ones)
        j = rng.choice(zeros)

        cand = cur_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_val = _c5(cand)

        # Metropolis acceptance
        accept = cand_val < cur_val or rng.random() < np.exp(-(cand_val - cur_val) / temp)
        if accept:
            cur_h = cand
            cur_val = cand_val
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]

            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        # restart if stagnation observed
        if no_imp >= restart_after:
            cur_h = _random_binary(n, rng)
            cur_val = _c5(cur_h)
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            no_imp = 0

        it += 1

    # ------------------------------------------------------------------
    # 3.  Return --------------------------------------------------------
    # ------------------------------------------------------------------
    return best_h, float(best_val), n
