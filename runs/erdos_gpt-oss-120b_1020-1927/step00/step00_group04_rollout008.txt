import time, math
import numpy as np

def _compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx for the discrete vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)

def _random_binary(n, rng):
    """Binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = rng.permutation(n)[: n // 2]
    h[idx] = 1.0
    return h

def _sturmian_binary(n, theta, phi, rng):
    """Balanced Sturmian‑type sequence, then forced to have exactly n/2 ones."""
    phases = (np.arange(n, dtype=float) * theta + phi) % 1.0
    h = (phases < 0.5).astype(float)
    # Fix the count if it drifts by at most a few positions
    diff = int(h.sum() - n // 2)
    if diff > 0:          # too many ones, turn some to zero
        ones = np.flatnonzero(h)
        to_flip = rng.choice(ones, size=diff, replace=False)
        h[to_flip] = 0.0
    elif diff < 0:        # too few ones, turn some zeros to one
        zeros = np.flatnonzero(1.0 - h)
        to_flip = rng.choice(zeros, size=-diff, replace=False)
        h[to_flip] = 1.0
    return h

def _improve_once(h, rng):
    """
    Perform a single targeted swap aimed at reducing the current worst shift.
    Returns True if a swap was performed (whether it helped or not).
    """
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(conv.argmax())
    d = idx - (n - 1)

    # overlapping region for shift d
    if d >= 0:
        i0, i1 = 0, n - d
    else:
        i0, i1 = -d, n

    mism_one = []
    mism_zero = []
    for i in range(i0, i1):
        if h[i] > 0.5 and h[i + d] < 0.5:
            mism_one.append(i)
        elif h[i] < 0.5 and h[i + d] > 0.5:
            mism_zero.append(i)

    if mism_one and mism_zero:
        i = rng.choice(mism_one)
        j = rng.choice(mism_zero)
        h[i], h[j] = h[j], h[i]
        return True
    return False

    """Search for a step function with a small Erdős overlap constant C₅.

    Returns
    -------
    h_values : list of float
        The discrete function values (length n_points).
    c5_bound : float
        The achieved max overlap integral.
    n_points : int
        Number of discretisation points.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + float(budget_s) - 0.2   # tiny safety margin

    # ------------------------------------------------------------
    # grid size – use the supplied initial solution if present,
    # otherwise fall back to a modest size (84 points).
    # ------------------------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
        # sanity check – keep it only if it already satisfies the mass constraint
        if not np.isclose(init.sum(), n / 2):
            raise ValueError
        have_initial = True
    except Exception:
        init = None
        n = 84
        have_initial = False

    best_h = None
    best_val = math.inf

    if have_initial:
        val = _compute_c5(init)
        best_h = init.copy()
        best_val = val

    # a few fixed irrational slopes for Sturmian candidates
    thetas = [
        (2 ** 0.5 - 1.0),
        (3 ** 0.5 - 1.0) % 1.0,
        (5 ** 0.5 - 1.0) / 2.0,
        0.6180339887498949,
        0.4142135623730950,

    target = 0.3808  # we stop as soon as we beat this

    # ------------------------------------------------------------
    # Main search loop – generate candidates, evaluate, and try a
    # tiny local improvement step.
    # ------------------------------------------------------------
    while time.time() < deadline:

        # 0 – pure random binary
        # 1 – random binary derived from best (few swaps)
        # 2 – Sturmian deterministic sequence with random phase
        # ------------------------------------------------------------------

        if method == 0:
            h = _random_binary(n, rng)
        elif method == 1 and best_h is not None:

            # a few random swaps to diversify
            for _ in range(rng.integers(1, 6)):
                ones = np.flatnonzero(h > 0.5)
                zeros = np.flatnonzero(h < 0.5)
                if ones.size == 0 or zeros.size == 0:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i], h[j] = h[j], h[i]

        best_val = _compute_c5(best_h)
