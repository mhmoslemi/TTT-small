import time
import numpy as np

# evaluate_erdos_solution is supplied by the harness
# it returns the C5 value for a given step function vector h

def _c5_of_h(h):
    """Return C5 for vector h using the provided evaluator."""
    return evaluate_erdos_solution(h)


def _hill_climb_until(h_start, stop_time, rng, start_time, end_time):
    """
    Simple gradient‑based hill climbing that preserves the sum constraint.
    Returns the best vector found and its C5 value.
    """
    h = h_start.copy()
    best_h = h.copy()
    best_val = _c5_of_h(h)

    while time.time() < stop_time:
        n = h.size
        # full correlation of h with (1‑h)
        corr = np.correlate(h, 1 - h, mode="full")
        idx = int(np.argmax(corr))
        cur_val = corr[idx] * (2.0 / n)          # integral value
        shift = idx - (n - 1)                    # integer shift giving the worst overlap

        # keep global best
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        # gradient of the worst‑shift term (the one we try to improve)
        s = abs(shift)
        grad = np.zeros_like(h)
        if s == 0:
            grad = 2.0 * h
        else:
            # contribution from i+s and i‑s where defined
            grad[: n - s] += h[s:]
            grad[s:] += h[: n - s]

        # choose a position to increase (high gradient, room to grow)
        inc_mask = h < 1.0 - 1e-12
        dec_mask = h > 1e-12
        if inc_mask.any() and dec_mask.any():
            inc_idx = np.where(inc_mask)[0]
            dec_idx = np.where(dec_mask)[0]
            inc_scores = grad[inc_idx] * (1.0 - h[inc_idx])
            dec_scores = grad[dec_idx] * h[dec_idx]
            i = inc_idx[int(np.argmax(inc_scores))]
            j = dec_idx[int(np.argmin(dec_scores))]
        else:
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)

        # how much we can move from j to i while staying in [0,1]
        max_inc = min(1.0 - h[i], h[j])
        if max_inc <= 1e-15:
            continue

        # step size shrinks as we consume the budget
        fraction = max(0.01, (stop_time - time.time()) / (end_time - start_time))
        delta = max_inc * rng.random() * fraction

        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        new_val = _c5_of_h(h_new)

        # accept only if it improves the worst overlap (or tiny random acceptance)
        if new_val < cur_val - 1e-12:
            h = h_new
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
        elif rng.random() < 0.001:
            h = h_new

    return best_h, best_val

    start_time = time.time()
