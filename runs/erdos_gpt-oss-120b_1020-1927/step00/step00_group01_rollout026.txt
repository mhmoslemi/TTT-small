def project_to_feasible(v, target):
    # binary search
    lo = -2.0  # lower bound for lambda
    hi = 2.0   # upper bound
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid

import math
import random
import time
import numpy as np

# ----- helper functions (top level) ---------------------------------

def _compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_to_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Projection of v onto the intersection of the box [0,1]^n and the hyperplane
    sum(x) = target.  Uses a simple bisection on the Lagrange multiplier λ:
        x_i = clip(v_i - λ, 0, 1)
    The function sum_i x_i is monotone decreasing in λ, so bisection works.
    """
    lo, hi = -2.0, 2.0                      # safe bounds for λ
    for _ in range(60):                    # enough for double precision
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction for possible rounding error
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 0.0) & (proj < 1.0)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _subgradient_step(h: np.ndarray, eta: float) -> (np.ndarray, float):
    """
    One sub‑gradient step for ``max_s  h·(1‑h_shift)``.
    Returns the new h and the C5 value after the step.
    """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)

    # indices where the correlation attains the maximum (within tight tolerance)
    eps = max_val * 1e-12 + 1e-14
    max_inds = np.flatnonzero(np.abs(corr - max_val) <= eps)

    grad = np.zeros_like(h)

    for idx in max_inds:
        s = idx - (n - 1)                     # actual shift
        if s >= 0:
            i_end = n - s
            grad[:i_end] += (1.0 - h[s:])
            grad[s:]   -= h[:i_end]
        else:
            s_abs = -s
            i_end = n - s_abs
            grad[s_abs:] += (1.0 - h[:i_end])
            grad[:i_end] -= h[s_abs:]

    grad *= dx                                # account for the outer dx factor
    h_new = h - eta * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    target = n / 2.0
    h_new = _project_to_box_sum(h_new, target)
    new_c5 = max_val * dx
    return h_new, new_c5


def _random_swap_move(h: np.ndarray) -> (np.ndarray, float):
    """
    Propose a move that transfers a small amount of mass from one coordinate
    to another while preserving the total sum and the box constraints.
    """
    n = h.shape[0]
    i, j = random.sample(range(n), 2)

    # maximal admissible transfer from j to i (positive delta)
    max_inc_i = 1.0 - h[i]
    max_dec_j = h[j]
    delta_pos = min(max_inc_i, max_dec_j)

    # maximal admissible transfer from i to j (negative delta)
    max_inc_j = 1.0 - h[j]
    max_dec_i = h[i]
    delta_neg = -min(max_inc_j, max_dec_i)

    if delta_pos + delta_neg == 0.0:        # both coordinates are at a bound
        return h, _compute_c5(h)

    delta = random.uniform(delta_neg, delta_pos)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new, _compute_c5(h_new)


# ----- main entry point ---------------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Optimised discretised step function (values in [0,1]).
    c5_bound : float
        The achieved C5 value (max overlap integral).
    n_points : int
        Number of discretisation points (len(h_values)).
    """
    random.seed(seed)
    np.random.seed(seed)

    # ---- initialise ------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=float).copy()
        n = h.shape[0]
    else:
        # fallback: moderate even size, triangular start shape
        n = 200
        dx = 2.0 / n
        x = (np.arange(n) + 0.5) * dx                # mid‑point of each cell
        h = 1.0 - np.abs(x - 1.0)                    # triangle of total area 1
        # enforce exact sum constraint
        h = _project_to_box_sum(h, n / 2.0)

    # guarantee feasibility (clip + project)
    h = np.clip(h, 0.0, 1.0)
    h = _project_to_box_sum(h, n / 2.0)

    best_h = h.copy()
    best_val = _compute_c5(best_h)

    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    # ---- phase 1: sub‑gradient descent -----------------------------
    max_iter_sub = 6000
    eta0 = 1.0                                        # initial stepsize

    for it in range(max_iter_sub):
        if elapsed() > budget_s * 0.55:              # keep some budget for SA
            break
        eta = eta0 / math.sqrt(it + 1.0)
        h, cur_val = _subgradient_step(h, eta)

        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

    # ---- phase 2: simple simulated annealing -----------------------
    # keep the current point as the start of the SA phase
    cur_h = h
    cur_val = _compute_c5(cur_h)

    # temperature schedule (linear decay)
    temp_start = max(0.1, best_val)
    temp_end = 0.001
    it_sa = 0

    while elapsed() < budget_s * 0.99:
        # linear temperature in terms of iteration count (arbitrarily scaled)
        frac = min(1.0, it_sa / 200000.0)
        temp = (1.0 - frac) * temp_start + frac * temp_end

        new_h, new_val = _random_swap_move(cur_h)

        # Metropolis acceptance
        if new_val < cur_val or random.random() < math.exp(-(new_val - cur_val) / (temp + 1e-12)):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
        it_sa += 1

    # final projection to be safe
    best_h = _project_to_box_sum(np.clip(best_h, 0.0, 1.0), n / 2.0)
    final_val = _compute_c5(best_h)

    return best_h, final_val, n
