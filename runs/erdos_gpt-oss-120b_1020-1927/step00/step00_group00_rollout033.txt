def binary_simulated_annealing(h, iterations, rng, time_limit):
    n = len(h)
    target_sum = n/2
    best_h = h.copy()
    best_score = evaluate(h)
    current_h = h.copy()
    current_score = best_score
    # prepare sets
    ones = set(np.where(current_h > 0.5)[0])
    zeros = set(np.where(current_h < 0.5)[0])
    initial_T = 0.02
    for it in range(iterations):
        if time.time() > time_limit:
            break
        T = max(1e-6, initial_T * (1 - it / iterations))
        # pick random i from ones, j from zeros
        if not ones or not zeros:
            break
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        # swap
        new_h = current_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_score = evaluate(new_h)
        delta = new_score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / T):
                accept = True
        if accept:
            current_h = new_h
            current_score = new_score
            # update sets
            ones.remove(i); zeros.add(i)
            zeros.remove(j); ones.add(j)
            # maybe also accept if improvement:
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()
    return best_h, best_score

import numpy as np, time, math

def evaluate_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def project_sum_bounds(v, target):
    # enforce 0<=v<=1 and sum(v)=target
    v = np.clip(v, 0.0, 1.0)
    # adjust sum using simple water-filling
    for _ in range(10):
        cur = np.sum(v)
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (v < 1.0)
            nfree = np.count_nonzero(free)
            if nfree == 0:
                break
            inc = diff / nfree
            v[free] = np.minimum(v[free] + inc, 1.0)
        else:
            free = (v > 0.0)
            nfree = np.count_nonzero(free)
            if nfree == 0:
                break
            dec = diff / nfree
            v[free] = np.maximum(v[free] + dec, 0.0)  # dec is negative
    # final small fix
    cur = np.sum(v)
    diff = target - cur
    if abs(diff) > 1e-12:
        # adjust a single free element
        idx = np.where((v > 0.0) & (v < 1.0))[0]
        if idx.size == 0:
            idx = np.arange(len(v))
        i = idx[0]
        v[i] = np.clip(v[i] + diff, 0.0, 1.0)
    return v

def gradient_descent(start_h, max_iter, rng, deadline):
    h = start_h.copy()
    n = len(h)
    target = n / 2.0
    cur_score = evaluate_h(h)
    best_h = h.copy()
    best_score = cur_score
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # compute correlation to find worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        lag_idx = np.argmax(corr)
        L = lag_idx - (n - 1)
        # compute shifted arrays a = h shifted by L (zero padded), b = h shifted by -L
        if L >= 0:
            a = np.concatenate((h[L:], np.zeros(L)))
            b = np.concatenate((np.zeros(L), h[:n - L]))
        else:
            shift = -L
            a = np.concatenate((np.zeros(shift), h[:n - shift]))
            b = np.concatenate((h[shift:], np.zeros(shift)))
        grad = 1.0 - (a + b)
        # step size schedule
        eta = 0.2 / (1.0 + it * 0.01)
        h_new = h - eta * grad
        h_new = project_sum_bounds(h_new, target)
        new_score = evaluate_h(h_new)
        # accept if improvement, else accept with small prob
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(eta, 1e-9)):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
    return best_h, best_score

def binary_sa(start_h, max_iter, rng, deadline):

    target = n / 2.0

            if new_score < best_score:

                best_h = cur_h.copy()
    return best_h, best_score

def thue_morse_pattern(n):
    # generate binary Thue-Morse sequence of length n
    # Use parity of popcount
    arr = np.fromiter((bin(i).count('1') % 2 for i in range(n)), dtype=float)
    # ensure sum equals n/2 (true for power of 2). For other n, adjust by flipping some bits.
    target = n / 2.0
    cur = np.sum(arr)
    diff = int(target - cur)
    if diff > 0:
        zeros = np.where(arr == 0)[0]
        if zeros.size >= diff:
            idx = zeros[:diff]
            arr[idx] = 1.0
    elif diff < 0:
        ones = np.where(arr == 1)[0]
        if ones.size >= -diff:
            idx = ones[: -diff]
            arr[idx] = 0.0
    return arr

def upsample(h0, n_new):
    n0 = len(h0)
    x0 = (np.arange(n0) + 0.5) * 2.0 / n0
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x0, h0, left=h0[0], right=h0[-1])
    # enforce bounds
    h_new = np.clip(h_new, 0.0, 1.0)
    target = n_new / 2.0
    h_new = project_sum_bounds(h_new, target)
    return h_new

import numpy as np
import time
import math

def _evaluate_h(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _project_sum_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to make Σv == target (water‑filling)."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        cur = v.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = v < 1.0
            nfree = int(free.sum())
            if nfree == 0:
                break
            inc = diff / nfree
            v[free] = np.minimum(v[free] + inc, 1.0)
        else:
            free = v > 0.0
            nfree = int(free.sum())
            if nfree == 0:
                break
            dec = diff / nfree  # negative
            v[free] = np.maximum(v[free] + dec, 0.0)
    # final tiny correction on a non‑blocked entry
    cur = v.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        idx = np.where((v > 0.0) & (v < 1.0))[0]
        if idx.size == 0:
            idx = np.arange(v.size)
        i = idx[0]
        v[i] = np.clip(v[i] + diff, 0.0, 1.0)
    return v


def _gradient_descent(start_h: np.ndarray, max_iter: int, rng, deadline: float):
    """Sub‑gradient descent on the max‑overlap objective."""
    h = start_h.copy()
    n = h.size
    target = n / 2.0
    cur_score = _evaluate_h(h)
    best_h, best_score = h.copy(), cur_score

    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        lag_idx = int(np.argmax(corr))
        L = lag_idx - (n - 1)

        if L >= 0:
            a = np.concatenate((h[L:], np.zeros(L)))
            b = np.concatenate((np.zeros(L), h[: n - L]))
        else:
            s = -L
            a = np.concatenate((np.zeros(s), h[: n - s]))
            b = np.concatenate((h[s:], np.zeros(s)))

        grad = 1.0 - (a + b)               # ∂f/∂h for the worst lag
        eta = 0.2 / (1.0 + it * 0.01)      # decreasing step size
        h_new = _project_sum_bounds(h - eta * grad, target)
        new_score = _evaluate_h(h_new)

        # accept if better or with a small Metropolis probability
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(eta, 1e-9)):
            h, cur_score = h_new, new_score
            if new_score < best_score:
                best_h, best_score = h_new.copy(), new_score
    return best_h, best_score


def _binary_sa(start_h: np.ndarray, max_iter: int, rng, deadline: float):
    """Simulated annealing on binary vectors (preserving Σh = n/2)."""
    n = start_h.size
    cur_h = start_h.copy()
    cur_score = _evaluate_h(cur_h)
    best_h, best_score = cur_h.copy(), cur_score

    ones = set(np.where(cur_h > 0.5)[0])
    zeros = set(np.where(cur_h < 0.5)[0])
    init_T = 0.02

    for it in range(max_iter):
        if time.time() > deadline:
            break
        T = max(1e-7, init_T * (1.0 - it / max_iter))
        if not ones or not zeros:
            break
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))

        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_score = _evaluate_h(new_h)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_h, cur_score = new_h, new_score
            ones.remove(i); zeros.add(i)
            zeros.remove(j); ones.add(j)
            if new_score < best_score:
                best_h, best_score = new_h.copy(), new_score
    return best_h, best_score


def _thue_morse(n: int) -> np.ndarray:
    """Binary Thue–Morse sequence of length n (adjusted to exact half‑ones)."""
    seq = np.fromiter((bin(i).count("1") % 2 for i in range(n)), dtype=float)
    target = n / 2.0
    cur = seq.sum()
    diff = int(round(target - cur))
    if diff > 0:
        zeros = np.where(seq == 0)[0]
        seq[zeros[:diff]] = 1.0
    elif diff < 0:
        ones = np.where(seq == 1)[0]
        seq[ones[: -diff]] = 0.0
    return seq


def _upsample(vec: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation from vec (length m) to length n_new with constraints."""
    m = vec.size
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = (np.arange(n_new) + 0.5) * 2.0 / n_new
    up = np.interp(xt, xs, vec, left=vec[0], right=vec[-1])
    up = np.clip(up, 0.0, 1.0)
    return _project_sum_bounds(up, n_new / 2.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the time budget."""
    rng = np.random.default_rng(seed)
    start = time.time()

    n_target = 256

    candidates = []

    # 3) up‑sample any provided parent construction

        diff = int(target_sum - cur)
