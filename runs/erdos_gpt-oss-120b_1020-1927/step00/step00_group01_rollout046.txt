import time
import numpy as np

def _project_onto_bounds_sum(y, target, lo=0.0, hi=1.0):
    """
    Project y onto the set {x | lo <= x_i <= hi, sum x_i = target}
    using a simple bisection on the Lagrange multiplier.
    """
    # bounds for the multiplier; wide enough to cover all cases
    lam_low, lam_high = -1e6, 1e6
    for _ in range(60):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(y - lam, lo, hi)
        s = x.sum()
        if s > target:          # need larger lambda to decrease the sum
            lam_low = lam
        else:                   # need smaller lambda
            lam_high = lam
    return np.clip(y - (lam_low + lam_high) / 2.0, lo, hi)


def _score(h, dx):
    """exact objective value: max_k ∫ h(x)(1‑h(x+k))dx"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)


def _softmax_grad(h, dx, alpha):
    """
    Gradient of the smoothed objective
    f_alpha(h) = dx * Σ p_s * corr_s,
    where p_s = exp(alpha*corr_s) / Σ exp(alpha*corr_t).
    Returns (grad, current_exact_score).
    """
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)

    # softmax weights, numerically stable
    exp_shift = np.exp(alpha * (corr - max_corr))
    probs = exp_shift / exp_shift.sum()

    grad = np.zeros_like(h)
    # shifts run from -(n-1) … (n-1)
    for idx, s in enumerate(range(-(n - 1), n)):
        p = probs[idx]
        if p == 0.0:
            continue
        if s >= 0:
            # j + s inside range
            if n - s > 0:
                grad[: n - s] += p * (1.0 - h[s:])
                grad[s:]      += p * (-h[: n - s])
        else:
            t = -s
            if n - t > 0:
                grad[t:]      += p * (1.0 - h[: n - t])
                grad[: n - t] += p * (-h[t:])
    grad *= dx
    exact = dx * max_corr
    return grad, exact


def _optimise_one_start(h_start, dx, target_sum, budget, rng):
    """
    Gradient‑descent with soft‑max smoothing from a single start point.
    Stops when the local time budget expires.
    Returns the best (h, score) seen during the run.
    """
    # parameters
    step = 0.015                # base step size
    alpha = 2.0                 # initial soft‑max sharpness
    best_h = h_start.copy()
    best_score = _score(best_h, dx)

    h = h_start.copy()
    start_time = time.time()
    while time.time() - start_time < budget:
        grad, cur_score = _softmax_grad(h, dx, alpha)

        # gradient step
        h_new = h - step * grad
        # enforce constraints
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _project_onto_bounds_sum(h_new, target_sum)

        new_score = _score(h_new, dx)

        # accept if improvement, otherwise keep old (no Metropolis move)
        if new_score < cur_score:
            h = h_new
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()

        # slowly tighten the soft‑max and reduce the step size
        elapsed = time.time() - start_time
        if elapsed > 0.3 * budget:
            alpha = min(alpha * 1.02, 200.0)
        if elapsed > 0.6 * budget:
            step *= 0.9995

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points)
    where h_values is a numpy array satisfying the constraints,
    c5_bound is the achieved C5 value, and n_points is the length of h.
    """
    rng = np.random.default_rng(seed)
    # Choose a modest grid size; 100 points give enough flexibility.
    n = 100
    dx = 2.0 / n
    target = n / 2.0

    # Build an initial guess.
    if 'initial_h_values' in globals():
        # Upsample the provided vector to our grid size.
        h0 = np.asarray(initial_h_values, dtype=float)
        m = h0.size
        old_x = (np.arange(m) + 0.5) * (2.0 / m)
        new_x = (np.arange(n) + 0.5) * dx
        h_start = np.interp(new_x, old_x, h0)
    else:
        h_start = rng.random(n)

    # Project onto the feasible set.
    h_start = np.clip(h_start, 0.0, 1.0)
    h_start = _project_onto_bounds_sum(h_start, target)

    best_h = h_start.copy()
    best_score = _score(best_h, dx)

    # Time budgeting
    start_total = time.time()
    remaining = budget_s
    # Allocate a fraction of the total budget to each restart.
    # We will perform several restarts (including the one based on the parent).
    while time.time() - start_total < 0.95 * budget_s:
        # allocate up to 8 seconds per restart (adjustable)

        # Randomize start point (jitter around current best)

        h0 = _project_onto_bounds_sum(h0, target)

        h_opt, score_opt = _optimise_one_start(
            h0, dx, target, time_for_restart, rng)

        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt.copy()

        # stop early if we already beat the target record
        if best_score <= 0.3808:
