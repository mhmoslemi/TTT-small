import time
import numpy as np

def _generate_h(p, n, dx):
    """Create a symmetric step‑function with slope length p (0≤p≤1)."""
    xs = (np.arange(n) + 0.5) * dx          # mid‑points of the sub‑intervals
    if p < 1e-12:                           # uniform limit
        return np.full(n, 0.5)

    H = 1.0 / (2.0 - p)                     # height chosen so that the integral equals 1
    h = np.empty(n)

    left = xs < p
    right = xs > (2.0 - p)
    middle = ~left & ~right

    # rising edge
    h[left] = H * xs[left] / p
    # flat top
    h[middle] = H
    # falling edge (symmetric)
    h[right] = H * (2.0 - xs[right]) / p
    return h


def _project_onto_feasible(v, target):
    """Project v onto the box [0,1]^n while fixing the linear sum = target."""
    lo, hi = -2.0, 2.0                      # wide enough bounds for the Lagrange multiplier
    for _ in range(60):                    # binary search for the correct shift
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_of_h(h, dx):
    """Return the objective value C₅ for a given discretised h."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

    rng = np.random.RandomState(seed)
    start_time = time.time()

    # 1.  Choose a moderate grid size (large enough for accuracy,
    #     small enough for speed). 200 points give a good trade‑off.
    # -------------------------------------------------------------
    n = 200
    dx = 2.0 / n
    target_sum = n / 2.0                     # because ∫h = 1  ⇔  Σh * dx = 1

    best_h = None
    best_c5 = np.inf
    best_p = None

    # -------------------------------------------------------------
    # 2.  Coarse grid search over the shape parameter p ∈ [0,1].
    # -------------------------------------------------------------
    for p in np.linspace(0.0, 1.0, 101):
        # respect time budget
        if time.time() - start_time > budget_s * 0.9:
            break
        h_raw = _generate_h(p, n, dx)
        h = _project_onto_feasible(h_raw, target_sum)
        c5 = _c5_of_h(h, dx)
        if c5 < best_c5:

    # -------------------------------------------------------------

                c5 = _c5_of_h(h, dx)
                if c5 < best_c5:

        for _ in range(max_iter):
            i, j = rng.choice(n, 2, replace=False)
            # propose moving a tiny amount from i to j
            max_delta = min(h_cur[i], 1.0 - h_cur[j])
            min_delta = -min(h_cur[j], 1.0 - h_cur[i])
            if max_delta <= 0 and min_delta >= 0:
                continue
            delta = (rng.rand() * (max_delta - min_delta) + min_delta) * 0.2
            h_new = h_cur.copy()
            h_new[i] -= delta
            h_new[j] += delta
            c5_new = _c5_of_h(h_new, dx)
            if c5_new < c5_cur:
                h_cur, c5_cur = h_new, c5_new
                if c5_new < best_c5:
                    best_c5, best_h = c5_new, h_new.copy()
        # final projection to guarantee feasibility (should already hold)
        best_h = _project_onto_feasible(best_h, target_sum)

    # -------------------------------------------------------------
    # 5.  Fallback – if everything failed, return a uniform function.
    # -------------------------------------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_c5 = _c5_of_h(best_h, dx)

    return best_h, best_c5, n
