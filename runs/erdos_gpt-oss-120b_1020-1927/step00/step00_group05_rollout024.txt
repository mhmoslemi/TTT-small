def project_onto_simplex_bounds(y, K, lower=0.0, upper=1.0, max_iter=50):
    # We want to find lambda such that sum(clip(y + lambda, lower, upper)) = K.
    # lambda range: [lb, ub] where lb = lower - max(y) (lowest possible lambda to bring all values to lower bound) and ub = upper - min(y).
    lo = lower - np.max(y)
    hi = upper - np.min(y)
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(y + lam, lower, upper)
        s = x.sum()
        if s > K:
            hi = lam
        else:
            lo = lam
    return np.clip(y + (lo+hi)/2.0, lower, upper)

import time, math, random
import numpy as np

def compute_c5(h):
    """return C5 = max_k ∫ h(x)(1‑h(x+k)) dx  (discretised)"""
    dx = 2.0 / h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_sum(y, target, lo=0.0, hi=1.0, iters=50, eps=1e-12):
    """
    Euclidean projection of y onto the hyperplane sum(x)=target
    intersected with box [lo,hi]^n.
    """
    # find λ so that sum(clamp(y+λ, lo, hi)) = target
    lam_low = lo - np.max(y)
    lam_high = hi - np.min(y)
    for _ in range(iters):
        lam = (lam_low + lam_high) * 0.5
        x = np.clip(y + lam, lo, hi)
        s = x.sum()
        if s > target:
            lam_high = lam
        else:
            lam_low = lam
        if lam_high - lam_low < eps:
            break
    return np.clip(y + (lam_low + lam_high) * 0.5, lo, hi)


def thue_morse_binary(N):
    """balanced 0/1 Thue‑Morse sequence (exactly N/2 ones for powers of two)"""
    arr = np.empty(N, dtype=float)
    for i in range(N):
        arr[i] = (i.bit_count() & 1)  # parity of bits
    return arr


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    # ---- choose a working resolution ---------------------------------
    N = 256                     # power of two makes the TM sequence balanced
    target_sum = N / 2.0

    # ---- initial construction ----------------------------------------
    h = thue_morse_binary(N)
    # maybe its complement is better
    if compute_c5(1.0 - h) < compute_c5(h):
        h = 1.0 - h
    h = project_to_sum(h, target_sum)      # enforce exact sum

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start = time.time()
    # parameters for the smooth‑max optimisation
    alpha = 5.0          # starts smooth, grows later
    eta = 0.6            # basic step magnitude
    it = 0
    stagnant = 0

    while time.time() - start < budget_s * 0.95:
        it += 1

        # correlation of h with itself (linear, zero‑padded)
        corr = np.correlate(h, h, mode="full")
        K = np.sum(h)               # should stay = target_sum
        g = K - corr                # g_s = Σ h_i (1‑h_{i+s})

        # smooth approximation of max(g)
        g_max = np.max(g)
        w_raw = np.exp(alpha * (g - g_max))
        w = w_raw / w_raw.sum()     # weights sum to 1, length 2N‑1

        # gradient of the smooth‑max (see analysis)
        conv1 = np.convolve(w, h, mode="full")
        conv2 = np.convolve(w[::-1], h, mode="full")
        off = N - 1
        term1 = conv1[off:off + N]
        term2 = conv2[off:off + N]
        grad = -(term1 + term2)

        # step with diminishing size
        step = eta / math.sqrt(it)
        h_new = h - step * grad
        h_new = project_to_sum(h_new, target_sum)

        cur_c5 = compute_c5(h_new)

        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h_new.copy()
            stagnant = 0
        else:
            stagnant += 1

                stagnant = 0
