```
def project_box_sum(v, target, lo, hi):
    # v initial guess; we need to find x minimizing ||x - v|| subject to sum x = target, lo <= x <= hi.
    # Use binary search for lambda (Lagrangian multiplier).
    lo_lambda = -1e6
    hi_lambda = 1e6
    for _ in range(60):
        lam = (lo_lambda + hi_lambda) / 2
        x = np.clip(v + lam, lo, hi)
        s = x.sum()
        if s > target:
            hi_lambda = lam
        else:
            lo_lambda = lam
    # final projection
    x = np.clip(v + (lo_lambda+hi_lambda)/2, lo, hi)
    # adjust for possible off by small epsilon due to numeric
    # optional small correction to meet exact target:
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff proportionally among non-boundary entries
        free = (x > lo+1e-12) & (x < hi-1e-12)
        if free.any():
            x[free] += diff / free.sum()
            # clip again
            x = np.clip(x, lo, hi)
        else:
            # if all at bounds, adjust by min modifications (unlikely)
            pass
    return x
```

```
def propose_move(h, max_step=None):
    n=len(h)
    i, j = np.random.randint(0,n, size=2)
    while i==j:
        i, j = np.random.randint(0,n,size=2)
    # compute max feasible delta for increase i (positive)
    pos_max = min(1.0 - h[i], h[j])
    # max feasible negative delta (decrease i) magnitude
    neg_max = min(h[i], 1.0 - h[j])
    if pos_max == 0 and neg_max == 0:
        return None, None
    # Determine allowed delta range
    low = -neg_max
    high = pos_max
    # optionally limit step size
    if max_step is not None:
        # restrict magnitude
        if high > max_step:
            high = max_step
        if low < -max_step:
            low = -max_step
        if low > high: # not feasible
            return None, None
    delta = low + np.random.rand() * (high - low)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h, (i, j, delta)
```

```
def project_onto_box_and_sum(x, target, lo=0.0, hi=1.0):
    # Return y closest to x (in L2) satisfying sum(y)=target, lo <= y <= hi.
    # Use water-filling via binary search on lambda.
    lo_lam = -np.max([abs(x[i] - lo) for i in range(len(x))]) - 10
    hi_lam = np.max([abs(hi - x[i]) for i in range(len(x))]) + 10
    for _ in range(60):
        lam = (lo_lam + hi_lam) / 2.0
        y = np.clip(x + lam, lo, hi)
        s = y.sum()
        if s > target:
            hi_lam = lam
        else:
            lo_lam = lam
    y = np.clip(x + (lo_lam + hi_lam) / 2.0, lo, hi)
    # Slight adjustment to meet exact target due to rounding
    diff = target - y.sum()
    if abs(diff) > 1e-9:
        # distribute diff among non-boundary entries (if any)
        free = (y > lo + 1e-8) & (y < hi - 1e-8)
        if free.any():
            # Distribute equally
            y[free] += diff / free.sum()
            # Clip again
            y = np.clip(y, lo, hi)
        # else, adjust by small modifications to boundaries in some direction - ignore small diff.
    return y
```

```
def simulated_annealing(h, time_limit):
    n = len(h)
    cur = h.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        remaining = time_limit - elapsed
        # temperature decays linearly from T0 to Tf
        T0 = 0.02
        Tf = 0.0001
        T = max(Tf, T0 * (remaining / time_limit))
        # propose move
        new_h, _ = propose_move(cur, max_step=0.2)  # bigger steps
        if new_h is None:
            continue
        new_score = compute_c5(new_h)
        d = new_score - cur_score
        if d < 0 or random.random() < math.exp(-d / T):
            cur = new_h
            cur_score = new_score
            if new_score < best_score:
                best = new_h.copy()
                best_score = new_score
    return best, best_score
```

```
def subgrad_max_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    # find indices with max
    idxs = np.where(np.abs(corr - max_val) <= 1e-12)[0]
    mid = n - 1
    g = np.zeros_like(h)
    for idx in idxs:
        lag = idx - mid
        # compute gradient contributions for this lag
        if lag >= 0:
            L = lag
            # term from i: gradient with respect to h_i: (1 - h_{i+L}) if i+L < n else 1
            # term2: -h_{i-L} if i-L >= 0
            # Using vectorized updates:
            g[:n-L] += (1 - h[L:])
            g[n-L:] += 1.0
            g[L:] -= h[:n-L]
        else:
            L = -lag
            # term1:
            g[:L] += 1.0
            g[L:] += (1 - h[:n-L])
            # term2:
            g[:n-L] -= h[L:]
        # If multiple max lags, sum contributions; will average later
    g /= len(idxs)
    return g
```

import numpy as np, random, math, time

def _project_onto_box_sum(v, target, lo=0.0, hi=1.0):
    # Simple water-filling via bisection on lambda
    lo_lam = -1e6
    hi_lam = 1e6
    for _ in range(70):
        lam = (lo_lam + hi_lam) * 0.5
        y = np.clip(v + lam, lo, hi)
        s = y.sum()
        if s > target:
            hi_lam = lam
        else:
            lo_lam = lam
    y = np.clip(v + (lo_lam + hi_lam) * 0.5, lo, hi)
    # tiny correction for numerical error
    diff = target - y.sum()
    if abs(diff) > 1e-10:
        free = (y > lo + 1e-12) & (y < hi - 1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, lo, hi)
    return y

def _c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def _subgrad_max_corr(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    mids = np.where(np.abs(corr - max_val) <= 1e-12)[0]
    mid = n - 1
    g = np.zeros_like(h)
    for idx in mids:
        lag = idx - mid
        if lag >= 0:
            L = lag
            # j+L in range
            g[:n - L] += 1.0 - h[L:]
            # j+L out of range -> 1
            g[n - L:] += 1.0
            # term -h_{j-L}
            g[L:] -= h[:n - L]
        else:
            L = -lag
            # j-L out of range -> 1
            g[:L] += 1.0
            # j-L in range: (1 - h_{j-L})
            g[L:] += 1.0 - h[:n - L]
            # term -h_{j+L}
            g[:n - L] -= h[L:]
    g /= float(len(mids))
    return g

def _subgradient_descent(h0, time_limit):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_score = _c5_value(h)
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        g = _subgrad_max_corr(h) * dx
        eta = 0.5 / (math.sqrt(it) + 1.0)
        h_new = h - eta * g
        h_new = _project_onto_box_sum(h_new, target, 0.0, 1.0)
        cur = _c5_value(h)
        new = _c5_value(h_new)
        if new < best_score:
            best_score = new
            best_h = h_new.copy()
        if new <= cur:
            h = h_new
        # optional occasional random perturbation to escape plateaus
        if random.random() < 0.001:
            # random swap move
            i, j = random.sample(range(n), 2)
            delta = (random.random() - 0.5) * 0.1
            # enforce bounds
            lo = max(0.0, h[i] - delta,  -1e9)
            hi = min(1.0, h[i] + delta,  1e9)
            # for simplicity just reproject after change
            h[i] = np.clip(h[i] + delta, 0.0, 1.0)
            h[j] = np.clip(h[j] - delta, 0.0, 1.0)
            h = _project_onto_box_sum(h, target, 0.0, 1.0)
    return best_h, best_score

def _propose_move(h, max_step):
    n = len(h)
    i, j = random.sample(range(n), 2)
    # feasible delta range
    pos_max = min(1.0 - h[i], h[j])
    neg_max = min(h[i], 1.0 - h[j])
    if pos_max == 0 and neg_max == 0:
        return None, None
    low = -neg_max
    high = pos_max
    if max_step is not None:
        if high > max_step:
            high = max_step
        if low < -max_step:
            low = -max_step
        if low > high:
            return None, None
    delta = low + random.random() * (high - low)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h, (i, j, delta)

def _simulated_annealing(h0, time_limit):
    n = len(h0)
    target = n / 2.0
    cur = h0.copy()
    cur_score = _c5_value(cur)
    best = cur.copy()
    best_score = cur_score
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = 1.0 - elapsed / time_limit
        T = 0.02 * frac + 1e-6
        max_step = 0.2 * frac + 1e-6
        new_h, _ = _propose_move(cur, max_step)
        if new_h is None:
            continue
        new_score = _c5_value(new_h)
        d = new_score - cur_score
        if d < 0 or random.random() < math.exp(-d / T):
            cur = new_h
            cur_score = new_score
            if new_score < best_score:
                best = new_h.copy()
                best_score = new_score
    # final projection
    best = _project_onto_box_sum(best, target, 0.0, 1.0)
    return best, best_score

def _upscale_h(initial_h, n):
    m = len(initial_h)
    # endpoints at centers of bins
    xs = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    xs0 = np.linspace(0.0, 2.0, m, endpoint=False) + 1.0 / m
    h = np.interp(xs, xs0, initial_h, left=initial_h[0], right=initial_h[-1])
    h = np.clip(h, 0.0, 1.0)
    h = _project_onto_box_sum(h, n / 2.0, 0.0, 1.0)
    return h

        if best_score < 0.3808:

        elapsed = time.time() - start
        frac = 1.0 - elapsed / time_limit
        T = 0.02 * frac + 1e-7
        step = 0.2 * frac + 1e-7
        new_h, _ = _propose_move(cur, step)
        if new_h is None:
            continue
        new_score = _c5_value(new_h)
        d = new_score - cur_score
        if d < 0 or random.random() < math.exp(-d / T):
            cur = new_h
            cur_score = new_score
            if new_score < best_score:
                best = new_h.copy()
                best_score = new_score
    best = _project_onto_box_sum(best, target, 0.0, 1.0)
    return best, best_score

# ---------- upscale a parent pattern ----------
def _upscale_h(parent, n):
    m = len(parent)
    xs = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    xs0 = np.linspace(0.0, 2.0, m, endpoint=False) + 1.0 / m
    h = np.interp(xs, xs0, parent, left=parent[0], right=parent[-1])
    h = np.clip(h, 0.0, 1.0)
    h = _project_onto_box_sum(h, n / 2.0, 0.0, 1.0)
    return h

# ---------- optimisation for a fixed grid size ----------
def _optimize_n(n, budget):
    best_h, best_score = None, float('inf')
    start = time.time()
    while time.time() - start < budget:
        remaining = budget - (time.time() - start)
        if remaining < 0.3:
            break
        # start point
        if 'initial_h_values' in globals() and random.random() < 0.5:
            h0 = _upscale_h(np.asarray(initial_h_values, dtype=float), n)
        else:
            h0 = np.random.rand(n)
            h0 = _project_onto_box_sum(h0, n / 2.0, 0.0, 1.0)
        # short subgradient run
        per_run = min(remaining * 0.6, 4.0)
        h_opt, sc_opt = _subgradient_descent(h0, per_run)
        # optional annealing refinement
        if remaining - per_run > 0.2:
            h_opt, sc_opt = _simulated_annealing(h_opt, remaining - per_run - 0.1)
        if sc_opt < best_score:
            best_score, best_h = sc_opt, h_opt.copy()
        if best_score < 0.3808:       # early stop if record already broken
            break
    return best_h, best_score

# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    total = budget_s
    start = time.time()
    # try a few grid sizes
    n_candidates = [100, 150, 200]
    best_h, best_score = None, float('inf')
    for idx, n in enumerate(n_candidates):
        if time.time() - start > total * 0.95:
            break
        remaining = total - (time.time() - start)
        # split remaining time evenly among the still‑to‑try sizes
        time_for_n = remaining * 0.9 / (len(n_candidates) - idx)
        if time_for_n < 0.5:
            continue
        h_n, sc_n = _optimize_n(n, time_for_n)
        if h_n is not None and sc_n < best_score:
            best_h, best_score = h_n, sc_n
    # fallback if nothing succeeded
    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5)
        best_h = _project_onto_box_sum(best_h, n / 2.0, 0.0, 1.0)
        best_score = _c5_value(best_h)
    # final projection (ensures exact constraints)
    n = len(best_h)
    best_h = _project_onto_box_sum(best_h, n / 2.0, 0.0, 1.0)
    best_score = _c5_value(best_h)
    return best_h, float(best_score), n
