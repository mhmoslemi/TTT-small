def weighted_grad(h, alpha):
    n = len(h)
    conv = np.correlate(h, 1 - h, mode='full')
    # compute weights using softmax over conv*alpha
    # conv can be negative? Actually (1-h) non-negative and h non-negative; product is >=0, so conv >=0. Good.
    # So weights
    max_conv = conv.max()
    # subtract max for numerical stability
    exp_vals = np.exp(alpha * (conv - max_conv))
    w = exp_vals / exp_vals.sum()
    # accumulate gradient
    g = np.zeros_like(h)
    # iterate over each lag index and weight
    for idx, weight in enumerate(w):
        if weight == 0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            # forward part
            g[:n-shift] += weight * (1 - h[shift:])
            # backward part
            g[shift:] -= weight * h[:n-shift]
        else:
            s = -shift
            g[s:] += weight * (1 - h[:n-s])
            g[:n-s] -= weight * h[s:]
    return g, conv.max() * (2.0/n)

def _project_box_and_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project vector v onto the box [0,1]^n with exact sum = target_sum."""
    lo, hi = -2.0, 2.0  # guarantees that sum(clamp(v-lam,0,1)) spans [0,n]
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)

        if s > target_sum:

    n = h.shape[0]

        alpha = base_alpha + 0.05 * iteration
        if alpha > max_alpha:
            alpha = max_alpha
        step = step0 / (1.0 + 0.001 * iteration)

        # ----- gradient and candidate ------------------------------------------------
        grad, cur_c5 = _weighted_gradient(h, alpha)
        max_abs = np.max(np.abs(grad))
        if max_abs > 0:
            step_scaled = step / max_abs
        else:
            step_scaled = 0.0
        cand = h - step_scaled * grad
        cand = _project_box_and_sum(cand, target_sum)

        cand_c5 = _c5_of_h(cand)

        # ----- acceptance ---------------------------------------------------------
        if cand_c5 < best_c5:
            best_h = cand.copy()
            best_c5 = cand_c5
            h = cand
        else:
            # simulated‑annealing style fallback
            T = 0.01 / (1.0 + 0.0005 * iteration)
            if np.random.rand() < math.exp(-(cand_c5 - best_c5) / max(T, 1e-12)):
                h = cand

        # ----- occasional random perturbation ------------------------------------
        if iteration % 200 == 0 and iteration > 0:
            pert = best_h + np.random.normal(scale=0.02, size=n_points)
            pert = _project_box_and_sum(pert, target_sum)
            pert_c5 = _c5_of_h(pert)
            if pert_c5 < best_c5:
                best_h = pert.copy()
                best_c5 = pert_c5
                h = pert

        iteration += 1

    # final clean‑up
    best_h = _project_box_and_sum(best_h, target_sum)
    final_c5 = _c5_of_h(best_h)

    return best_h, final_c5, n_points
