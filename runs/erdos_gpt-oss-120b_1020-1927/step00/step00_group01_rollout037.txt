import numpy as np, time, math

def project_box_sum(v, S, max_iter=60):
    low = -np.max(v) - 2.0    # ensure f(low) >= S
    high = np.max(v) + 2.0    # ensure f(high) <= S
    for _ in range(max_iter):
        tau = (low + high) / 2.0
        x = np.clip(v - tau, 0.0, 1.0)
        s = x.sum()
        if s > S:
            low = tau
        else:
            high = tau
    return np.clip(v - (low+high)/2.0, 0.0, 1.0)
```

def compute_c5(h):

    g = h.sum() - conv
    return (2.0 / n) * np.max(g)

import time, math, numpy as np

def _project_box_sum(v: np.ndarray, target_sum: float, iters: int = 60) -> np.ndarray:
    """Projection of v onto {x : 0<=x<=1, sum(x)=target_sum}."""
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(iters):
        tau = (lo + hi) * 0.5
        x = np.clip(v - tau, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_of(h: np.ndarray) -> float:
    """C5 value for a step vector h (dx handled inside)."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with its complement = sum_h - correlation(h,h)
    conv = np.correlate(h, h, mode="full")
    g = h.sum() - conv                         # g_k = sum_h - conv_k
    return dx * np.max(g)                      # max over all shifts


def _initial_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a feasible starter: upsample parent if available, otherwise random."""
    parent = globals().get("initial_h_values", None)
    if parent is not None:
        parent = np.asarray(parent, dtype=float)
        # linear interpolation to size n
        old_n = parent.shape[0]
        old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
        new_x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        h = np.interp(new_x, old_x, parent)
    else:
        # random binary start
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
    # ensure feasibility
    return _project_box_sum(h, n / 2)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] minimizing
        C5 = max_k ∫ h(x) (1-h(x+k)) dx .
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # resolution – a few hundred points gives a good trade‑off
    n = 300
    h = _initial_vector(n, rng)
    cur_val = _c5_of(h)
    best_h = h.copy()
    best_val = cur_val

    # Simulated‑annealing on the simplex {0≤h_i≤1, sum= n/2}
    #   – each move transfers a small amount δ from one coordinate to another
    #   – temperature decays exponentially
    max_steps = 200_000
    T0 = 0.02
    T_end = 1e-5

    for step in range(max_steps):
        # time check
        if time.time() - start_time > budget_s * 0.94:
            break

        # temperature schedule
        T = T0 * ((T_end / T0) ** (step / max_steps))

        # propose a transfer of mass δ from i (source) to j (dest)
        i, j = rng.choice(n, 2, replace=False)
        # amount we may move without breaking bounds
        max_down = h[i]               # how much we can take from i
        max_up = 1.0 - h[j]           # how much we can add to j
        if max_down == 0 or max_up == 0:
            continue
        # random δ, limited to a small fraction of the feasible range
        delta = rng.random() * min(max_down, max_up, 0.1)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        # projection is not needed (we kept within bounds and sum preserved)

        new_val = _c5_of(h_new)
        d = new_val - cur_val
        if d <= 0.0 or rng.random() < math.exp(-d / T):
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()

        # occasional gentle random perturbation to escape plateaus
        if step % 5000 == 0 and rng.random() < 0.12:
            noise = rng.normal(scale=0.02, size=n)
            h_noisy = np.clip(best_h + noise, 0.0, 1.0)
            h_noisy = _project_box_sum(h_noisy, n / 2)
            noisy_val = _c5_of(h_noisy)
            if noisy_val < best_val:
                best_h = h_noisy
                best_val = noisy_val
                h = h_noisy
                cur_val = noisy_val

    # final projection to satisfy constraints exactly
    best_h = _project_box_sum(best_h, n / 2)
    best_val = _c5_of(best_h)

    # ensure result is a plain python list for the harness
    return (best_h.tolist(), float(best_val), n)
