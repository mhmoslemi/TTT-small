    # Use bisection on lambda in [low, high]

import numpy as np
import time
from scipy.optimize import minimize

def _compute_c5(h):
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear correlation, zero‑padding outside the interval
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * dx

def _project_to_simplex_box(v, target_sum):
    """
    Project vector v onto the intersection of the box [0,1]^n
    and the hyperplane sum(v)=target_sum.
    Uses a simple water‑filling (binary search on the offset λ).
    """
    n = v.shape[0]
    lo = np.min(v) - 1.0      # enough to make all entries 1 after clipping
    hi = np.max(v) + 1.0      # enough to make all entries 0 after clipping
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction if we're off by a rounding error
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = free.sum()
        if cnt > 0:
            x[free] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step‑function h on [0,2] (represented by n points)
    to minimise C₅ = max_k ∫ h(x)(1‑h(x+k))dx.

    Returns (list_of_h, c5_value, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    if "initial_h_values" in globals() and initial_h_values is not None:
        h = np.asarray(initial_h_values, dtype=float).copy()
        n = h.size
    else:
        n = 42                     # a modest size (fast to evaluate)
        h = np.full(n, 0.5)       # feasible (sum = n/2)
    target_sum = n / 2.0           # because ∫h = 1  →  Σh·dx = 1  → Σh = n/2
    # ensure feasibility
    h = np.clip(h, 0.0, 1.0)
    h = _project_to_simplex_box(h, target_sum)

    best_h = h.copy()
    best_c5 = _compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    # ------------------------------------------------------------------
    # hill‑climbing with mass‑exchange moves
    # ------------------------------------------------------------------
    max_step = 0.3          # initial maximum amount transferred in a move
    min_step = 0.005        # final minimum step size
    it = 0
    while True:
        # time termination
        elapsed = time.time() - start_time
        if elapsed > budget_s - 0.5:
            break
        # linear schedule for step magnitude
        frac = elapsed / budget_s
        step_limit = max_step * (1.0 - frac) + min_step * frac

        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # feasible delta range keeping both entries in [0,1]
        lo = max(-current_h[i], current_h[j] - 1.0)
        hi = min(1.0 - current_h[i], current_h[j])
        # additionally respect the global step limit
        lo = max(lo, -step_limit)
        hi = min(hi, step_limit)
        if lo >= hi:
            continue

        delta = np.random.uniform(lo, hi)
        h_new = current_h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        # evaluate
        c5_new = _compute_c5(h_new)

        # accept strictly improving moves
        if c5_new < current_c5:
            current_h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                # early exit if we already beat the target record
                if best_c5 <= 0.3808:
                    break

        # occasional random restart to escape possible plateaus
        it += 1
        if it % 50000 == 0:
            # generate a fresh feasible point (uniform Dirichlet then clip)
            rnd = np.random.rand(n)
            rnd = _project_to_simplex_box(rnd, target_sum)
            current_h = rnd
            current_c5 = _compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

    # ------------------------------------------------------------------
    # final refinement with a few gradient‑based steps (optional)
    # ------------------------------------------------------------------
    # compute the lag that currently realises the maximum
    corr = np.correlate(best_h, 1.0 - best_h, mode="full")
    lag = np.argmax(corr) - (n - 1)

    # a single projected gradient step can tidy up small irregularities
    grad = np.ones_like(best_h)
    if lag >= 0:
        grad[: n - lag] -= best_h[lag:]
        grad[lag:]   -= best_h[: n - lag]
    else:
        p = -lag
        grad[p:]    -= best_h[: n - p]
        grad[: n - p] -= best_h[p:]

    step = 0.1
    h_tmp = best_h - step * grad
    h_tmp = _project_to_simplex_box(h_tmp, target_sum)
    c5_tmp = _compute_c5(h_tmp)
    if c5_tmp < best_c5:
        best_h = h_tmp
        best_c5 = c5_tmp

    # ensure final feasibility (tiny numerical drift may appear)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_simplex_box(best_h, target_sum)

    # ------------------------------------------------------------------
    # return the solution
    # ------------------------------------------------------------------
    final_c5 = _compute_c5(best_h)
    return (best_h.tolist(), float(final_c5), int(n))
