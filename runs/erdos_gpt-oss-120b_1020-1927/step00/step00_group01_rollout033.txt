import time, math, random
import numpy as np

def _c5_value(h):
    """Return the C5 value for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] that yields a small
    Erdős overlap constant C5.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : numpy.ndarray of shape (n_points,)
        c5_bound : float, the achieved C5 value
        n_points : int, number of sampled points (len(h_values))
    """
    # ------------------------------------------------------------
    # basic settings
    # ------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # target resolution – even number makes binary constructions easy
    n = 200                     # feel free to change; 200 gives a good trade‑off
    dx = 2.0 / n
    half = n // 2               # exact number of 1's required

    # ------------------------------------------------------------
    # helper to compute the worst shift and the list of “bad’’ positions
    # ------------------------------------------------------------
    def _worst_shift_and_bad_positions(arr):
        """Return (shift, bad_first_indices) for the current max correlation."""
        nloc = arr.shape[0]
        corr = np.correlate(arr, 1.0 - arr, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (nloc - 1)          # integer lag (may be negative)

        if shift >= 0:
            i = np.arange(0, nloc - shift)
            bad = i[(arr[i] == 1) & (arr[i + shift] == 0)]
        else:
            s = -shift
            i = np.arange(s, nloc)
            bad = (i - s)[(arr[i - s] == 1) & (arr[i] == 0)]
        return shift, bad

    # ------------------------------------------------------------
    # keep the best solution seen so far
    # ------------------------------------------------------------
    best_h = None
    best_val = float("inf")

    # ------------------------------------------------------------
    # if a parent construction is provided, evaluate it
    # ------------------------------------------------------------
    if "initial_h_values" in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        # parent may have odd length; its integral should already be 1.
        # evaluate using its own grid
        parent_val = _c5_value(parent)
        best_h = parent.copy()
        best_val = parent_val

    # ------------------------------------------------------------
    # main optimisation loop – multiple random restarts
    # ------------------------------------------------------------
    while time.time() - start_time < budget_s - 5.0:
        # ---- start a fresh random binary candidate ----------------
        h = np.zeros(n, dtype=float)
        h[np.random.choice(n, half, replace=False)] = 1.0
        cur_val = _c5_value(h)

        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        # ---- simulated‑annealing style walk ----------------------
        # iteration budget for this restart adapts to remaining time
        time_left = budget_s - (time.time() - start_time)
        max_iter = int(min(50000, time_left * 15))   # a rough budget
        temperature = 0.12

        for _ in range(max_iter):
            # abort early if time is almost exhausted
            if time.time() - start_time >= budget_s - 2.0:
                break

            # locate the worst shift and a “bad’’ index at that shift
            shift, bad = _worst_shift_and_bad_positions(h)
            if bad.size:
                i = int(np.random.choice(bad))
            else:
                # fallback – any 1‑position
                i = int(np.random.choice(np.where(h == 1)[0]))

            # pick a random 0‑position to exchange mass with
            zero_idx = np.where(h == 0)[0]
            if zero_idx.size == 0:
                break
            j = int(np.random.choice(zero_idx))

            if i == j:
                continue

            # perform the swap
            h[i], h[j] = 0.0, 1.0
            new_val = _c5_value(h)

            # Metropolis acceptance
            accept = False
            if new_val < cur_val:
                accept = True
            else:
                # uphill moves are occasionally tolerated
                delta = new_val - cur_val
                if temperature > 1e-9 and random.random() < math.exp(-delta / temperature):
                    accept = True

            if accept:
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_h = h.copy()
                temperature *= 0.9997
            else:
                # revert the swap
                h[i], h[j] = 1.0, 0.0
                temperature *= 0.9997

            # early exit if we already beat the target bound
            if best_val <= 0.3808:
                break

        # optional cheap local polishing of the current best
        if best_val <= 0.3808:
            break

        # --------------------------------------------------------
        # a quick hill‑climb around the best known configuration
        # --------------------------------------------------------
        if best_h is not None and best_h.shape[0] == n:
            local_h = best_h.copy()
            local_val = best_val
            improved = True
            while improved and (time.time() - start_time < budget_s - 2.0):
                improved = False
                # attempt a few random swaps that reduce the max
                for _ in range(200):
                    i = int(np.random.choice(np.where(local_h == 1)[0]))
                    j = int(np.random.choice(np.where(local_h == 0)[0]))
                    local_h[i], local_h[j] = 0.0, 1.0
                    cand = _c5_value(local_h)
                    if cand < local_val:
                        local_val = cand
                        improved = True
                        if cand < best_val:
                            best_val = cand
                            best_h = local_h.copy()
                    else:
                        local_h[i], local_h[j] = 1.0, 0.0
                # stop if no improvement after a whole sweep
                if not improved:
                    break

    # ------------------------------------------------------------
    # final sanity check and return
    # ------------------------------------------------------------
    if best_h is None:                     # should never happen
        best_h = np.zeros(n, dtype=float)
        best_h[:half] = 1.0
    final_n = best_h.shape[0]
    final_c5 = _c5_value(best_h)

    return best_h, final_c5, final_n
