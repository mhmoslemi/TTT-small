def project_to_feasible(h, target_sum):
    low, high = -1.0, 1.0
    for _ in range(30):
        mid = (low + high) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(h - (low+high)/2, 0, 1)

def gradient_of_corr(h, shift):
    N = len(h)
    d = np.zeros(N)
    if shift >=0:
        # i from 0..N-1-shift have i+shift in range
        d[:N-shift] += 1 - h[shift:]
        # i from shift..N-1 have i-shift in range
        d[shift:] -= h[:N-shift]
    else:
        s = -shift
        # i from s..N-1 have i+shift = i - s in range
        d[s:] += 1 - h[:N-s]
        # i from 0..N-1-s have i-shift = i + s in range
        d[:N-s] -= h[s:]
    return d

import time
import math
import numpy as np

def evaluate_c5(h: np.ndarray) -> float:
    """Return the overlap bound C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) using zero padding
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _project_to_feasible(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of x onto the intersection
    { 0 <= y_i <= 1 } and { sum y_i = target_sum }.
    Uses a simple bisection on the Lagrange multiplier.
    """
    low, high = -1.0, 1.0          # guarantees sum in [0, n]
    for _ in range(50):
        mid = (low + high) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(x - (low + high) * 0.5, 0.0, 1.0)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ------------------------------------------------------------------
    # initialise h
    try:
        init = initial_h_values          # provided by the harness
    except NameError:
        init = None
    if init is not None:
        h = np.array(init, dtype=np.float64).copy()
        n = h.shape[0]
    else:
        n = 70
        h = rng.random(n)
        h = _project_to_feasible(h, n / 2.0)

    # best solution seen so far
    best_h = h.copy()
    best_val = evaluate_c5(best_h)

    # parameters for the simulated‑annealing style search
    target_sum = n / 2.0
    T0 = 2.0e-2            # initial temperature
    T_end = 1.0e-7         # final temperature
    min_step = 0.02        # minimal fraction of the available transfer

    # ------------------------------------------------------------------
    # main optimisation loop
    while time.time() - start < budget_s * 0.95:
        # current correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        dx = 2.0 / n
        corr_scaled = corr * dx
        cur_val = float(np.max(corr_scaled))
        # all shifts achieving the current maximum
        max_inds = np.flatnonzero(corr_scaled == cur_val)
        shift_idx = rng.choice(max_inds)
        shift = int(shift_idx - (n - 1))        # actual integer shift

        # propose a move
        h_new = h.copy()
        # linear schedule for the step size fraction (shrinks with time)
        elapsed = time.time() - start
        frac = max(min_step, 0.5 * (1.0 - elapsed / budget_s))

        if shift != 0:
            if shift > 0:
                # contributions for i where i+shift is inside range
                contrib = h[: n - shift] * (1.0 - h[shift:])
                i_local = int(np.argmax(contrib))
                i = i_local
                j = i + shift
            else:
                s = -shift
                contrib = h[s:] * (1.0 - h[: n - s])
                i_local = int(np.argmax(contrib))
                i = i_local + s
                j = i - s
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max <= 1e-12:          # fallback to a random pair
                i = rng.integers(n)
                j = rng.integers(n)
                while j == i:
                    j = rng.integers(n)
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max <= 0.0:
                    continue
            delta = delta_max * frac
            if delta <= 0.0:
                continue
            h_new[i] -= delta
            h_new[j] += delta
        else:
            # shift 0 gives no useful directed move – perform a random transfer
            i = rng.integers(n)
            j = rng.integers(n)
            while j == i:
                j = rng.integers(n)
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max <= 0.0:
                continue
            delta = delta_max * frac
            h_new[i] -= delta
            h_new[j] += delta

        # enforce bounds and exact sum (tiny drift may appear)
        np.clip(h_new, 0.0, 1.0, out=h_new)
        if abs(h_new.sum() - target_sum) > 1e-9:
            h_new = _project_to_feasible(h_new, target_sum)

        new_val = evaluate_c5(h_new)

        # simulated‑annealing acceptance
        T = T0 + (T_end - T0) * (elapsed / budget_s)
        if new_val <= cur_val or rng.random() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
            h = h_new
            cur_val = new_val

        # keep global best
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()

    # final evaluation (ensuring feasibility one last time)
    best_h = _project_to_feasible(best_h, target_sum)
    final_val = evaluate_c5(best_h)
    return best_h, final_val, best_h.shape[0]
