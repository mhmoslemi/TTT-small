import numpy as np, time, math

def overlap_conv(h):
    N = h.shape[0]
    L = 2 * N
    H = np.fft.rfft(h, n=L)
    H_rev = np.fft.rfft(h[::-1], n=L)
    conv = np.fft.irfft(H * H_rev, n=L)
    return conv[:2*N-1]

def compute_c5(h):
    N = len(h)
    total = h.sum()
    conv = overlap_conv(h)
    min_ov = conv.min()
    if min_ov < 0:
        min_ov = 0.0
    dx = 2.0 / N
    return (total - min_ov) * dx

def project_box_hyperplane(v, target):
    # Find lambda such that sum(clip(v+lam,0,1)) = target.
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + hi, 0.0, 1.0)

        l = -lag

import numpy as np, time, math

def _overlap_conv(h: np.ndarray) -> np.ndarray:
    """Linear autocorrelation of h with its reverse (i.e. Σ h_i h_{i+lag})."""
    N = h.shape[0]
    L = 2 * N
    H = np.fft.rfft(h, n=L)
    H_rev = np.fft.rfft(h[::-1], n=L)
    conv = np.fft.irfft(H * H_rev, n=L)
    return conv[: 2 * N - 1]   # length 2N‑1, index k ↔ lag = N‑1‑k


def _c5_of(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    N = h.shape[0]
    total = h.sum()               # = N/2 (exact after projection)
    conv = _overlap_conv(h)       # overlap Σ h_i h_{i+lag}
    min_ov = conv.min()
    if min_ov < 0:
        min_ov = 0.0               # numerical safety
    dx = 2.0 / N
    return (total - min_ov) * dx   # = max correlation * dx


def _project_box_hyperplane(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project v onto the set {x | 0≤x_i≤1, Σx_i = target}.
    Uses a simple bisection on the common offset λ such that
    x_i = clip(v_i+λ,0,1) and Σx_i = target.
    """
    lo, hi = -2.0, 2.0           # bounds wide enough for all feasible λ
    for _ in range(60):          # enough for double precision
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + hi, 0.0, 1.0)


def _grad_for_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Sub‑gradient of the maximal correlation w.r.t. the lag giving the max."""
    N = h.shape[0]
    g = np.ones(N)
    if lag >= 0:
        if lag == 0:
            g -= 2.0 * h
        else:
            g[: N - lag] -= h[lag:]
            g[lag:] -= h[: N - lag]
    else:
        l = -lag
        g[l:] -= h[: N - l]
        g[: N - l] -= h[l:]
    return g
