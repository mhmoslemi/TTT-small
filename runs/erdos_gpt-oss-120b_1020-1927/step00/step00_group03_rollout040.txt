def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, random
    start = time.time()
    rng = np.random.default_rng(seed)
    # Determine n_points and initial h
    try:
        init = initial_h_values
        n = init.shape[0]
        h = init.copy()
    except NameError:
        n = 70  # default
        # start with half ones randomly
        h = np.zeros(n)
        indices = rng.choice(n, size=n//2, replace=False)
        h[indices] = 1.0
    dx = 2.0 / n
    # function to compute c5
    def compute_c5(vec):
        conv = np.correlate(vec, 1.0 - vec, mode='full')
        return np.max(conv) * dx
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    # Keep best and iterate
    # We'll use simple swap moves: choose i where h[i]=1 and j where h[j]=0, swap
    # We'll precompute indices of ones and zeros.
    # For each move evaluate new candidate and accept if better, else maybe reject.
    # Use simulated annealing:
    temperature = 1.0
    while time.time() - start < budget_s * 0.95:  # leave margin
        # randomly pick positions to swap
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # Create new candidate by swapping
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        c5 = compute_c5(cand)
        # Accept if improved or with probability exp(-(c5 - best_c5)/temperature)
        if c5 < best_c5 or rng.random() < np.exp(-(c5 - best_c5) / temperature):
            best_h = cand
            best_c5 = c5
        # Decrease temperature gradually
        temperature *= 0.9999
    return best_h, best_c5, n

def run(seed=42, budget_s=1100, **kwargs):

import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # ----- problem size and initial vector -----
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
    except Exception:
        n = 70
        init = None
    dx = 2.0 / n
    target = n / 2.0                     # sum of h must equal this
    # ----- projection onto {0≤h≤1, sum=h_target} -----
    def proj(v):
        lo = np.min(v) - 1.0
        hi = np.max(v) + 1.0
        for _ in range(50):
            mid = (lo + hi) / 2.0
            p = np.clip(v - mid, 0.0, 1.0)
            s = p.sum()
            if s > target:
                lo = mid
            else:
                hi = mid
        return np.clip(v - lo, 0.0, 1.0)
    # ----- C5 evaluation -----
    def c5_val(vec):
        conv = np.correlate(vec, 1.0 - vec, mode="full")
        return np.max(conv) * dx
    # ----- starting point -----
    if init is not None:
        h = proj(init.copy())
    else:
        x = np.linspace(dx / 2.0, 2.0 - dx / 2.0, n)
        h = np.clip(1.0 - np.abs(x - 1.0), 0.0, 1.0)
        h = proj(h)
    best_h = h.copy()
    best_c5 = c5_val(best_h)
    # ----- gradient descent with soft‑max smoothing -----
    inner_iters = 250
    beta0, beta1 = 1.0, 30.0          # smoothing schedule
    lr0, lr1 = 0.30, 0.03             # learning‑rate schedule
    while time.time() - start < budget_s * 0.92:
        # random restart
        v0 = rng.random(n)
        h = proj(v0)
        for it in range(inner_iters):
            if time.time() - start >= budget_s * 0.92:
                break
            conv = np.correlate(h, 1.0 - h, mode="full")
            beta = beta0 + (beta1 - beta0) * it / inner_iters
            logits = beta * conv
            m = np.max(logits)
            w = np.exp(logits - m)
            w /= w.sum()                     # soft‑max weights over all shifts
            grad = np.zeros_like(h)
            # accumulate weighted derivatives of each shift
            for s in range(-n + 1, n):
                ws = w[s + n - 1]
                if ws == 0.0:
                    continue
                if s >= 0:
                    # forward part
                    grad[: n - s] += ws * (1.0 - h[s:])
                    # backward part
                    grad[s:] -= ws * h[: n - s]
                else:
                    k = -s
                    grad[k:] += ws * (1.0 - h[: n - k])
                    grad[: n - k] -= ws * h[k:]
            grad *= beta
            lr = lr0 + (lr1 - lr0) * it / inner_iters
            h = proj(h - lr * grad)
        cur = c5_val(h)
        if cur < best_c5:
            best_c5, best_h = cur, h.copy()
    # ----- binary refinement (swap hill‑climbing) -----
    ones_target = int(round(target))
    idx = np.argsort(-best_h)               # take the largest entries as ones
    bin_h = np.zeros_like(best_h)
    bin_h[idx[:ones_target]] = 1.0
    cur_h = bin_h.copy()
    cur_c5 = c5_val(cur_h)
    best_bin_c5, best_bin_h = cur_c5, cur_h.copy()
    temp = 0.5
    while time.time() - start < budget_s * 0.98:
        conv = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(conv))
        shift = max_idx - (n - 1)          # integer shift giving current max
        if shift >= 0:
            mism = np.where((cur_h[: n - shift] == 1) & (cur_h[shift:] == 0))[0]
        else:
            k = -shift
            mism = np.where((cur_h[k:] == 1) & (cur_h[: n - k] == 0))[0] + k
        if mism.size == 0:
            i = rng.choice(np.flatnonzero(cur_h == 1))
        else:
            i = rng.choice(mism)
        j = rng.choice(np.flatnonzero(cur_h == 0))
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_c5 = c5_val(cur_h)
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(temp, 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_bin_c5:
                best_bin_c5, best_bin_h = new_c5, cur_h.copy()
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0
        temp *= 0.99995
    # ----- return the best found solution -----
    if best_bin_c5 < best_c5:
        return best_bin_h, best_bin_c5, n
    else:
        return best_h, best_c5, n
