    # binary search λ in some range: min possible λ such that sum <= target? We'll search λ in [min(v)-upper, max(v)-lower] maybe.

        u = np.clip(v - lam, lower, upper)
        s = u.sum()
        if s > target:
            # need bigger λ to reduce sum (since subtract λ bigger reduces v-lam)
            lo = lam
        else:
            hi = lam
    u = np.clip(v - (lo+hi)/2, lower, upper)
    # adjust small numerical error: rescale difference first? Might not exactly equal target; we can distribute residual onto free variables (those not at bounds).
    # If sum deviates slightly, we can adjust by adding epsilon to some variables not at bounds.
    diff = target - u.sum()
    # add diff equally to variables not at bound and within small epsilon
    # careful to stay within bounds.
    if abs(diff) > 1e-8:
        free = (u > lower + 1e-12) & (u < upper - 1e-12)
        count = free.sum()
        if count > 0:
            u[free] += diff / count
            u = np.clip(u, lower, upper)
    # final adjust if sum still off due to rounding: small tolerance ok.

def project_onto_simplex_bounded(v, target_sum, lower=0.0, upper=1.0):
    # Using binary search for lambda.
    # v is np.ndarray float.
    # We search lam such that sum_i clip(v_i - lam, lower, upper) == target_sum.
    lo = -1e5
    hi = 1e5
    # adjust lower/upper to ensure there is feasible lam (finite)
    # Since clipping saturates at bounds, there will be lam extreme where sum >= target, etc.
    # We'll run for fixed iterations.
    for _ in range(60):
        lam = (lo + hi) / 2.0
        u = np.clip(v - lam, lower, upper)
        s = u.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    u = np.clip(v - (lo + hi) / 2.0, lower, upper)
    # Due to numerical errors, adjust small diff
    diff = target_sum - u.sum()
    if abs(diff) > 1e-8:
        # Adjust only among free variables (not at bounds)
        free = (u > lower + 1e-10) & (u < upper - 1e-10)
        cnt = free.sum()
        if cnt > 0:
            # distribute diff equally
            u[free] += diff / cnt
            # clamp again to bounds
            u = np.clip(u, lower, upper)
        # if still diff, we could do another small adjust but ignore.
    return u

import time
import numpy as np

def _project_simplex_bounded(v: np.ndarray, target: float,
                             lo: float = 0.0, hi: float = 1.0) -> np.ndarray:
    """Project v onto the set {x | sum(x)=target, lo <= x_i <= hi}."""
    # binary search for lambda such that sum(clamp(v - λ, lo, hi)) = target
    lam_low, lam_high = -1e5, 1e5
    for _ in range(60):
        lam = (lam_low + lam_high) * 0.5
        u = np.clip(v - lam, lo, hi)
        s = u.sum()
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
    u = np.clip(v - (lam_low + lam_high) * 0.5, lo, hi)
    # fix tiny residuals caused by rounding
    diff = target - u.sum()
    if abs(diff) > 1e-9:
        free = (u > lo + 1e-12) & (u < hi - 1e-12)
        cnt = int(free.sum())
        if cnt:
            u[free] += diff / cnt
            u = np.clip(u, lo, hi)
    return u


def _c5_value(h: np.ndarray, dx: float) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def _upsample(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a step function to a finer grid."""
    old_n = old.shape[0]
    if old_n == new_n:
        return old.astype(float, copy=True)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, old).astype(float)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Find a step function h on [0,2] with ∫h=1 that minimises C₅."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # target grid size – a balance between flexibility and speed
    n = 200
    dx = 2.0 / n
    target_sum = n / 2.0

    # --- initial candidate -------------------------------------------------
    if "initial_h_values" in globals():
        h = _upsample(np.asarray(initial_h_values, dtype=float), n)
    else:
        h = rng.random(n)
    h = _project_simplex_bounded(h, target_sum, 0.0, 1.0)

    best_h = h.copy()
    best_val = _c5_value(best_h, dx)

    step = 0.2               # initial step length
    no_improve = 0           # counter for stagnation

    # main optimisation loop – runs until time expires
    while time.time() - start < budget_s * 0.95:
        # correlation and current worst shift
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        cur_val = float(corr.max())

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # indices attaining the maximum (there can be several)
        max_idxs = np.where(np.abs(corr - cur_val) < 1e-12)[0]
        idx = rng.choice(max_idxs)
        shift = idx - (n - 1)          # integer shift in grid units

        # sub‑gradient of the active overlap functional
        g = np.ones_like(h)
        if shift >= 0:
            g[: n - shift] -= h[shift:]
            g[shift:]    -= h[: n - shift]
        else:
            t = -shift
            g[t:]  -= h[: n - t]
            g[: n - t] -= h[t:]

        # propose a new point and project back onto feasibility region
        h_new = h - step * g
        h_new = _project_simplex_bounded(h_new, target_sum, 0.0, 1.0)

        new_val = _c5_value(h_new, dx)

        # accept if we do not worsen the objective
        if new_val <= cur_val + 1e-12:
            h = h_new
            step = min(step * 1.02, 2.0)   # gently accelerate
        else:
            step = max(step * 0.5, 1e-5)   # backtrack
            if no_improve > 2000:          # escape if stuck
                h = best_h + rng.normal(scale=0.05, size=n)
                h = np.clip(h, 0.0, 1.0)
                h = _project_simplex_bounded(h, target_sum, 0.0, 1.0)
                step = 0.2
                no_improve = 0

    # final safeguard – ensure feasibility one last time
    best_h = _project_simplex_bounded(np.asarray(best_h, dtype=float),
                                      target_sum, 0.0, 1.0)

    return best_h.tolist(), float(best_val), int(n)
