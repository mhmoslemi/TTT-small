import cvxpy as cp
import numpy as np

def lp_relaxation(h0=None):
    n = len(h0) if h0 is not None else 70
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2]
    # For each shift k from 0 to n-1
    for k in range(n):
        length = n - k
        if length == 0:
            continue
        p = cp.Variable(length)
        constraints += [p <= h[:length]]
        constraints += [p <= h[k:]]
        constraints += [p >= h[:length] + h[k:] - 1]
        constraints += [p >= 0]  # optional
        constraints += [cp.sum(p) >= t]
    prob = cp.Problem(cp.Maximize(t), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    return h.value, t.value
```

def improve_h(h0, total_time):
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    n = len(h0)
    target = n / 2.0
    start = time.time()
    iter = 0
    while time.time() - start < total_time:
        # subgradient step
        corr = np.correlate(h0, h0, mode='full')
        min_idx = corr.argmin()
        k = min_idx - (n - 1)
        # compute gradient
        g = np.zeros(n)
        if k >= 0:
            g[:n - k] += h0[k:]
            g[k:] += h0[:n - k]
        else:
            kk = -k
            g[kk:] += h0[:n - kk]
            g[:n - kk] += h0[kk:]
        # direction with zero sum
        d = g - g.mean()
        # step size schedule: a decreasing factor
        elapsed = time.time() - start
        remaining = total_time - elapsed
        # adaptive step: start large, shrink
        step = 0.5 * (remaining / total_time)  # start at 0.5 then shrink
        # propose new
        y = h0 + step * d
        h_new = project_onto_box_and_sum(y, target)
        new_c5 = evaluate_erdos_solution(h_new)
        # accept if better
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        # for hill climbing decide acceptance: if new better or with prob
        if new_c5 <= evaluate_erdos_solution(h0) or np.random.rand() < 0.01:
            h0 = h_new
        # occasional random perturb
        if iter % 500 == 0:
            # small random jitter
            y = best_h + 0.02 * (np.random.rand(n) - 0.5)
            y = np.clip(y, 0.0, 1.0)
            # renormalize sum
            y = project_onto_box_and_sum(y, target)
            h0 = y
        iter += 1
    return best_h, best_c5

import time
import numpy as np

# evaluate_erdos_solution and possibly initial_h_values are provided externally.
# They are expected to be available in the global namespace when run() is called.

def _project_onto_box_and_sum(y, target):
    """Euclidean projection of y onto the box [0,1]^n with sum(y)=target."""
    # We solve for λ in sum(clip(y-λ,0,1)) = target by bisection.
    lo = -np.max(np.abs(y)) - 2.0
    hi = np.max(np.abs(y)) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)


def _max_overlap(h):
    """Return the C5 value for a given step function h."""
    # evaluate_erdos_solution is supplied by the harness.
    return evaluate_erdos_solution(h)


def _subgradient_step(h, target_sum, step):
    """Perform one projected sub‑gradient step that increases the minimal autocorrelation."""
    n = h.size
    # full autocorrelation (non‑circular, zero‑padded)
    corr = np.correlate(h, h, mode='full')
    min_idx = int(np.argmin(corr))
    lag = min_idx - (n - 1)          # lag = shift (can be negative)

    # gradient of the correlation at this lag
    g = np.zeros_like(h)
    if lag >= 0:
        # i contributes h[i+lag] and i-lag contributes h[i-lag]
        g[: n - lag] += h[lag:]
        g[lag:]      += h[: n - lag]
    else:
        k = -lag
        g[k:]        += h[: n - k]
        g[: n - k]   += h[k:]

    # enforce zero‑sum direction (stay on the hyperplane sum = target)
    d = g - g.mean()
    y = h + step * d
    # project back onto feasible set
    return _project_onto_box_and_sum(y, target_sum)


def _local_search(start_h, time_allow):
    """Hill‑climbing / simulated‑annealing style improvement."""
    n = start_h.size
    target = n / 2.0
    best_h = start_h.copy()
    best_val = _max_overlap(best_h)

    cur_h = best_h.copy()
    cur_val = best_val

    start = time.time()
    it = 0
    while time.time() - start < time_allow:
        elapsed = time.time() - start
        remaining = time_allow - elapsed

        # sub‑gradient ascent on the minimal autocorrelation
        # step size shrinks linearly to zero
        step = 0.5 * (remaining / time_allow)
        cand_h = _subgradient_step(cur_h, target, step)
        cand_val = _max_overlap(cand_h)

        # accept if it improves, otherwise accept with small prob.
        if cand_val < cur_val or np.random.rand() < 0.02:
            cur_h, cur_val = cand_h, cand_val
            if cand_val < best_val:
                best_h, best_val = cand_h.copy(), cand_val

        # occasional random perturbation to escape plateaus
        if it % 400 == 0 and np.random.rand() < 0.1:
            pert = best_h + 0.05 * (np.random.rand(n) - 0.5)
            pert = np.clip(pert, 0.0, 1.0)
            pert = _project_onto_box_and_sum(pert, target)
            cur_h, cur_val = pert, _max_overlap(pert)

        it += 1

    return best_h, best_val


def _random_feasible(n):
    """Generate a random feasible h (sum = n/2, entries in [0,1])."""
    # start from a uniform Dirichlet then clip and re‑project
    v = np.random.rand(n)
    v = v / v.sum() * (n / 2.0)
    v = np.clip(v, 0.0, 1.0)
    v = _project_onto_box_and_sum(v, n / 2.0)
    return v


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Determine number of points
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
        # ensure feasibility (tiny numerical drift possible)
        init = _project_onto_box_and_sum(init, n / 2.0)
        candidates = [(init, _max_overlap(init))]
    else:
        # fallback: use a smooth cosine start
        n = 120  # a modest size well below the 1 k limit
        x = np.arange(n)
        init = 0.5 + 0.5 * np.cos(2.0 * np.pi * x / n)  # average ≈0.5
        init = _project_onto_box_and_sum(init, n / 2.0)
        candidates = [(init, _max_overlap(init))]

    # Allocate a short budget for a few random restarts
    remaining = budget_s - (time.time() - start_time)
    for _ in range(3):
        if remaining <= 0:
            break
        rnd = _random_feasible(n)
        val = _max_overlap(rnd)
        candidates.append((rnd, val))
        remaining = budget_s - (time.time() - start_time)

    # Pick the best starter
    cur_h, cur_val = min(candidates, key=lambda p: p[1])

    # Main optimisation phase
    stay_time = budget_s - (time.time() - start_time) - 0.5  # keep a tiny buffer
    if stay_time > 0:
        cur_h, cur_val = _local_search(cur_h, stay_time)

    # Final safety check – a quick random perturbation might still improve it
    final_check_time = budget_s - (time.time() - start_time)
    if final_check_time > 0.1:
        for _ in range(5):
            pert = cur_h + 0.03 * (np.random.rand(n) - 0.5)
            pert = np.clip(pert, 0.0, 1.0)
            pert = _project_onto_box_and_sum(pert, n / 2.0)
            val = _max_overlap(pert)
            if val < cur_val:
                cur_h, cur_val = pert, val

    # Return as plain Python list for compatibility
    return (cur_h.tolist(), float(cur_val), int(n))
