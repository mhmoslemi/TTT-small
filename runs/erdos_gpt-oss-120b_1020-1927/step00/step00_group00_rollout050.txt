def project(v, target):
    # Ensure v is numpy array float
    # binary search for lambda
    lo = -1.0
    hi = np.max(v)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            hi = mid
        else:
            lo = mid
    # final projection
    proj = np.clip(v - lo, 0.0, 1.0)
    # In rare case due to numeric, adjust small diff
    diff = proj.sum() - target
    if abs(diff) > 1e-8:
        # adjust via scaling of interior points (non-bound)
        # approximate: shift values slightly by diff/(count of interior)
        interior = (proj > 0) & (proj < 1)

            # clip again
            proj = np.clip(proj, 0.0, 1.0)
        # if still off, final normalization
        proj = project(v, target)  # recursion (but unlikely)
    return proj

def project(v, target):
    # binary search lambda.
    lo = -1.0
    hi = np.max(v)
    for _ in range(40):
        mid = (lo + hi)/2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    # small adjust for sum equality
    diff = proj.sum() - target
    if abs(diff) > 1e-8:
        interior = (proj > 0) & (proj < 1)
        cnt = interior.sum()
        if cnt > 0:
            proj[interior] -= diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

import time, math
import numpy as np

def _project_to_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto {0≤x≤1, sum(x)=target_sum} using a binary search on the
    Lagrange multiplier for the sum constraint."""
    lo, hi = -1.0, np.max(v)
    for _ in range(40):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s < target_sum:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    # tiny correction to enforce the sum exactly (within numerical tolerance)
    diff = proj.sum() - target_sum
    if abs(diff) > 1e-9:
        interior = (proj > 0) & (proj < 1)
        cnt = interior.sum()
        if cnt > 0:
            proj[interior] -= diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _c5_score(h: np.ndarray) -> float:
    """C₅ value for a discrete step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def _subgradient(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the active overlap functional."""
    n = h.size
    g = np.ones(n)
    if shift >= 0:
        if shift < n:
            g[: n - shift] -= h[shift:]
            g[shift:] -= h[: n - shift]
    else:
        s = -shift
        if s < n:
            g[s:] -= h[: n - s]
            g[: n - s] -= h[s:]
    return g

        n2 = n * 2
        x_old = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_fine = np.interp(x_new, x_old, best)
        target2 = n2 / 2.0
        cur = _project_to_constraints(h_fine, target2)

        # sub‑gradient on the finer grid
        max_iter_sub2 = 4000
        step0 = 0.12
        for it in range(max_iter_sub2):
            if time.time() - start > budget_s * 0.95:
                break
            corr = np.correlate(cur, 1.0 - cur, mode="full")
            idx = int(corr.argmax())
            shift = idx - (n2 - 1)
            g = _subgradient(cur, shift)
            eta = step0 / math.sqrt(it + 1.0)
            cand = cur - eta * g
            cand = _project_to_constraints(cand, target2)
            sc = _c5_score(cand)
            if sc < best_score:
                best_score, best = sc, cand.copy()
            cur = cand

        # short random search on the finer grid
        cur = best.copy()
        sigma = 0.09
        max_iter_rand2 = 20000
        for it in range(max_iter_rand2):
            if time.time() - start > budget_s * 0.95:
                break
            sigma = max(0.001, sigma * 0.9995)
            v = rng.normal(size=n2)
            v -= v.mean()
            cand = cur + sigma * v
            cand = _project_to_constraints(cand, target2)
            sc = _c5_score(cand)
            if sc < best_score:
                best_score, best = sc, cand.copy()
            cur = cand

        # prepare for possible next refinement
        n, target = n2, target2
        best = _project_to_constraints(best, target)

    best = _project_to_constraints(best, len(best) / 2.0)
    best_score = _c5_score(best)
    return best.tolist(), float(best_score), len(best)
