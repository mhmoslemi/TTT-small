    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    # find indices where corr equals max (allow small tolerance)
    idxs = np.where(np.abs(corr - max_val) < 1e-12)[0]  # at most few
    # Now compute gradient sum of contributions for each k (lag) where k = idx - (n-1)
    grad = np.zeros(n)
    for idx in idxs:
        s = idx - (n - 1)  # shift; can be negative.
        # for each i meeting 0<=i, i+s < n
        # contribution from derivative of term h_i * (1 - h_{i+s}):
        # derivative w.r.t h_i: (1 - h_{i+s}) if i+s in [0,n-1]; else 0
        # derivative w.r.t h_{i+s}: -h_i
        # So for each i with i+s valid, grad[i] += (1 - h[i+s]); grad[i+s] += -h[i]

def project_box_sum(x):
    n = len(x)
    target = n/2.0
    lo = -np.max(x) - 1.0
    hi = np.max(x) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo+hi)/2.0, 0.0, 1.0)

import time
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant C5 for the step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(corr))


def subgrad_c5(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient of C5 w.r.t. h (using the worst shift only)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # can be negative
    grad = np.zeros_like(h)

    if shift >= 0:
        i = np.arange(0, n - shift)
        grad[i] = 1.0 - h[i + shift]
        grad[i + shift] = -h[i]
    else:
        p = -shift
        i = np.arange(p, n)
        grad[i] = 1.0 - h[i - p]
        grad[i - p] = -h[i]

    return grad * dx


def project_box_sum(x: np.ndarray) -> np.ndarray:
    """
    Project x onto the set {0 <= h_i <= 1 , sum h_i = n/2}.
    Uses a simple bisection on the shift lambda.
    """
    n = x.shape[0]
    target = n / 2.0
    lo = -np.max(x) - 1.0          # sum will be n (all 1) -> > target
    hi = np.max(x) + 1.0           # sum will be 0 -> < target

    for _ in range(50):
        mid = (lo + hi) * 0.5
        clipped = np.clip(x - mid, 0.0, 1.0)
        s = clipped.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def gradient_descent_projected(h0: np.ndarray,
                               time_budget: float,
                               lr_initial: float = 5.0) -> tuple:
    """Simple projected sub‑gradient descent with exponential decay."""
    start = time.time()
    h = h0.copy()
    cur_val = compute_c5(h)
    best_h, best_val = h.copy(), cur_val
    lr = lr_initial

    while time.time() - start < time_budget:
        grad = subgrad_c5(h)
        h_new = project_box_sum(h - lr * grad)
        val_new = compute_c5(h_new)

        if val_new < cur_val:
            h, cur_val = h_new, val_new
            if val_new < best_val:
                best_h, best_val = h_new.copy(), val_new

        lr *= 0.9995
        if lr < 1e-7:
            break

    return best_h, best_val


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1] with sum = n/2."""
    x = rng.random(n)
    return project_box_sum(x)


def local_search(h_start: np.ndarray, time_budget: float) -> tuple:
    """Stochastic pairwise exchange search with simulated‐annealing acceptance."""
    n = h_start.shape[0]
    rng = np.random.default_rng()
    cur_h = h_start.copy()
    cur_val = compute_c5(cur_h)
    best_h, best_val = cur_h.copy(), cur_val

    start = time.time()
    it = 0
    max_delta_factor = 0.5
    T0 = 0.01  # initial temperature for annealing

    while time.time() - start < time_budget:
        it += 1

        # choose a donor and a receiver
        i = rng.integers(n)
        if cur_h[i] <= 1e-12:
            continue
        j = rng.integers(n)
        if cur_h[j] >= 1.0 - 1e-12:
            continue

        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0.0:
            continue

        delta = max_transfer * rng.random() * max_delta_factor
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_val = compute_c5(new_h)

        if new_val < cur_val:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
        else:
            elapsed = time.time() - start
            frac = elapsed / time_budget
            T = (1.0 - frac) * T0
            if T > 0.0 and math.exp(-(new_val - cur_val) / T) > rng.random():
                cur_h, cur_val = new_h, new_val

        # occasional reset to a fresh random feasible point
        if it % 20000 == 0 and rng.random() < 0.5:
            cur_h = random_feasible(n, rng)
            cur_val = compute_c5(cur_h)

        # slowly shrink the exploratory step size
        if it % 5000 == 0:
            max_delta_factor = max(0.01, max_delta_factor * 0.95)

    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Starting vector
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        # fallback: uniform 0.5 on 47 points
        n0 = 47
        h0 = np.full(n0, 0.5, dtype=np.float64)

    # Ensure feasibility (box + sum)
    h0 = project_box_sum(h0)

    # ------------------------------------------------------------------
    # 2. Time allocation
    total_budget = float(budget_s)
    start_time = time.time()

    # 2a. Gradient descent phase (≈40% of the budget)
    gd_budget = min(0.4 * total_budget, 300.0)
    # protect against overshoot if start already took time
    elapsed = time.time() - start_time
    gd_budget = max(0.0, min(gd_budget, total_budget - elapsed))

    best_h, best_val = gradient_descent_projected(h0, gd_budget, lr_initial=5.0)

    # 2b. Local‑search phase (use the remaining time)
    elapsed = time.time() - start_time
    ls_budget = max(0.0, total_budget - elapsed)
    best_h2, best_val2 = local_search(best_h, ls_budget)

    if best_val2 < best_val:
        best_h, best_val = best_h2, best_val2

    # Final sanity projection (ensures exact constraints)
    best_h = project_box_sum(best_h)
    best_val = compute_c5(best_h)

    # Return as plain Python list for the harness
    return best_h.tolist(), float(best_val), best_h.shape[0]
