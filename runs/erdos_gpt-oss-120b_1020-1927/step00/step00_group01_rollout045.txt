def subgradient(h):
    n = len(h)
    # compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()
    # find active shift indices
    active = np.where(corr == max_val)[0]
    grad = np.zeros(n)
    # For each active shift
    for idx in active:
        s = idx - (n - 1)  # shift amount, can be negative or positive.
        # compute contributions

        if s >= 0:

def optimize_n(n, init_h, rng, max_iter, step0):
    h = init_h.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    for it in range(max_iter):
        grad = subgradient(h)
        # step size schedule
        eta = step0 / np.sqrt(it + 1)
        v = h - eta * grad
        h = project_to_feasible(v)
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val

import time, math
import numpy as np

def _c5_value(h):
    """Return C5 for a discrete step function h on [0,2]."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * corr.max()


def _subgradient(h):
    """Sub‑gradient of the max‑overlap functional at h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    # shifts that achieve the maximum (index = shift + n - 1)
    shifts = np.where(corr == max_val)[0] - (n - 1)

    grad = np.zeros(n)
    for s in shifts:
        if s >= 0:
            # first term: (1 - h[i+s]) for i = 0 .. n-s-1
            grad[: n - s] += 1.0 - h[s:]
            # second term: -h[i-s] for i = s .. n-1
            grad[s:] -= h[: n - s]
        else:
            t = -s
            # first term for negative shift
            grad[t:] += 1.0 - h[: n - t]
            # second term
            grad[: n - t] -= h[t:]
    return grad


def _project_onto_capped_simplex(v, S):
    """
    Euclidean projection of v onto {x | 0≤x_i≤1, sum x_i = S}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    n = len(v)
    # bounds for lambda: sufficiently low -> all clipped to 1, high -> all 0
    lo = np.min(v) - 1.0 - 1.0
    hi = np.max(v) + 1.0 + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _upscale(initial_h, n):
    """Interpolate a parent vector to a new length n and project to feasibility."""
    n0 = len(initial_h)
    # sample points at interval mid‑points
    x0 = np.linspace(0, 2, n0, endpoint=False) + 1.0 / n0
    x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    h = np.interp(x, x0, initial_h)
    h = np.clip(h, 0.0, 1.0)
    h = _project_onto_capped_simplex(h, n / 2.0)
    return h


def _optimize_one_start(h0, max_iter, step0, rng):
    """Projected sub‑gradient descent from a feasible start."""
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_value(h)
    n = len(h)

    for it in range(max_iter):
        grad = _subgradient(h)
        eta = step0 / math.sqrt(it + 1.0)
        v = h - eta * grad
        h = _project_onto_capped_simplex(v, n / 2.0)

        val = _c5_value(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val
