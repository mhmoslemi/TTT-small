    """Project v onto the box [0,1]^n while forcing sum(v)=target."""
    lo, hi = -2.0, 2.0

            # step size linearly decays, never reaches zero
            elapsed = (now - start) / budget_s
            step_frac = 0.5 * (1.0 - elapsed) + 0.001
            delta = rng.random() * max_delta * step_frac
            if delta <= 0:
                continue

            new_h = cur_h.copy()
            new_h[i_inc] += delta
            new_h[i_dec] -= delta

        else:                                        # random pair move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # move mass from the larger entry to the smaller one
            if cur_h[i] > cur_h[j]:
                i_inc, i_dec = j, i
            else:
                i_inc, i_dec = i, j
            max_inc = 1.0 - cur_h[i_inc]
            max_dec = cur_h[i_dec]
            max_delta = min(max_inc, max_dec, 0.02)
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            new_h = cur_h.copy()
            new_h[i_inc] += delta
            new_h[i_dec] -= delta

        # ----- evaluate ------------------------------------------------
        new_c5 = compute_c5(new_h)

        # Accept improvements outright
        if new_c5 < cur_c5:
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        else:
            # occasional uphill acceptance (simulated annealing)
            elapsed = (now - start) / budget_s
            temp = 0.001 * (1.0 - elapsed)
            if temp > 0 and rng.random() < np.exp((cur_c5 - new_c5) / temp):
                cur_h = new_h
                cur_c5 = new_c5

        # Occasionally perturb the best solution to escape plateaus
        if rng.random() < 0.001:
            pert = best_h.copy()
            for _ in range(5):
                a, b = rng.integers(0, n, size=2)
                max_inc = 1.0 - pert[a]
                max_dec = pert[b]
                max_delta = min(max_inc, max_dec, 0.01)
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta
                pert[a] += delta
                pert[b] -= delta
            cur_h = pert
            cur_c5 = compute_c5(cur_h)

    # ----- final reporting --------------------------------------------
    final_c5 = compute_c5(best_h)
    return best_h.tolist(), float(final_c5), n
