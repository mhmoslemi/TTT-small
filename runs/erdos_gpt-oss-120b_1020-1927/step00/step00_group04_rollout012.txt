from scipy.signal import fftconvolve

def _max_corr(h):
    # compute max overlap count
    # correlation of h with (1-h)
    return np.max(fftconvolve(h, (1-h)[::-1], mode='full'))

def _prepare_initial(h0, N, rng):
    # upscale or random
    if h0 is None:
        # random
        h = np.zeros(N, dtype=float)
        h[rng.choice(N, N//2, replace=False)] = 1.0
        return h
    # upscale: repeat each element
    orig = np.asarray(h0, dtype=float)
    L = orig.size
    # binarize
    orig = (orig > 0.5).astype(float)
    repeat = N // L
    remainder = N - repeat * L
    h = np.repeat(orig, repeat)
    if remainder > 0:
        extra_idx = rng.choice(L, remainder, replace=False)
        h = np.concatenate([h, orig[extra_idx]])
    # adjust sum
    target = N // 2
    cur = int(h.sum())
    if cur > target:
        idx = np.where(h == 1)[0]
        to_zero = rng.choice(idx, cur - target, replace=False)
        h[to_zero] = 0
    elif cur < target:
        idx = np.where(h == 0)[0]
        to_one = rng.choice(idx, target - cur, replace=False)
        h[to_one] = 1
    return h

    If init_arr is given, it is up‑scaled by repetition before adjusting.
    """

        h = np.zeros(N, dtype=float)
        h[rng.choice(N, N // 2, replace=False)] = 1.0
        return h

    v = np.asarray(init_arr, dtype=float)
