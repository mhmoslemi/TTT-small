import time
import numpy as np

def _evaluate(h):
    """Return the C5 value for a vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def _project_to_bounds_and_sum(v, target):
    """
    Project vector v onto the box [0,1] while fixing the sum to `target`.
    Solves min ||x-(v)|| s.t. 0<=x<=1 and sum(x)=target.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -10.0
    hi = 10.0
    # expand bounds until the bracketing condition holds
    while np.clip(v - lo, 0.0, 1.0).sum() < target:
        lo -= 5.0
    while np.clip(v - hi, 0.0, 1.0).sum() > target:
        hi += 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _gradient(h, weights):
    """Gradient of the soft‑max‑approximated max‑overlap w.r.t. h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    n_full = 2 * n - 1
    for idx in range(n_full):
        s = idx - (n - 1)
        if s == 0:
            continue
        w = weights[idx]
        if w == 0.0:
            continue
        if s > 0:
            grad[s:] += w * (1.0 - h[: n - s])
            grad[: n - s] -= w * h[s:]
        else:
            t = -s
            grad[: n - t] += w * (1.0 - h[t:])
            grad[t:] -= w * h[: n - t]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()
    # ------------------------------------------------------------------ #
    # 1. Determine a working discretisation size n and a feasible start.
    # ------------------------------------------------------------------ #
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.ndim != 1:
            h0 = h0.ravel()
        n0 = h0.shape[0]
        # make n even (binary feasible) by repeating if necessary
        if n0 % 2 == 1:
            h0 = np.repeat(h0, 2)
            n0 = h0.shape[0]
        n = n0
        # enforce constraints
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        h0 = _project_to_bounds_and_sum(h0, target)
    else:
        # fallback: use 100 points, uniform 0.5 (already feasible)
        n = 100
        h0 = np.full(n, 0.5, dtype=float)
        target = n / 2.0

    # ------------------------------------------------------------------ #
    # 2. Optimisation parameters
    # ------------------------------------------------------------------ #
    max_restarts = 3
    iters_per_restart = 4000
    best_h = None
    best_score = np.inf
    target = n / 2.0

    # ------------------------------------------------------------------ #
    # 3. Main optimisation loop (multiple random restarts)
    # ------------------------------------------------------------------ #
    for restart in range(max_restarts):
        if restart == 0 and "h0" in locals():
            h = h0.copy()
        else:
            # random feasible start
            h = rng.rand(n)
            h = _project_to_bounds_and_sum(h, target)

        step = 0.5
        beta = 4.0

        for it in range(iters_per_restart):
            # time guard
            if time.time() - start > budget_s * 0.94:
                break

            # correlation (unnormalised, without dx)
            corr = np.correlate(h, 1 - h, mode="full")
            cur_max = corr.max()
            # keep best ever
            if cur_max < best_score:
                best_score = cur_max
                best_h = h.copy()

            # soft‑max weights approximating the max
            w = np.exp(beta * (corr - cur_max))
            w /= w.sum()

            # gradient step
            g = _gradient(h, w)
            h_new = h - step * g

            # project back to feasible region
            h = _project_to_bounds_and_sum(h_new, target)

            # annealing of step and beta
            if (it + 1) % 200 == 0:
                step *= 0.85
                beta = min(beta * 1.08, 60.0)

        # after a restart we may already have exhausted budget
        if time.time() - start > budget_s * 0.94:
            break

    # ------------------------------------------------------------------ #
    # 4. Final evaluation and return
    # ------------------------------------------------------------------ #
    if best_h is None:
        best_h = h0 if "h0" in locals() else np.full(n, 0.5)
        best_score = _evaluate(best_h)

    c5_bound = _evaluate(best_h)
    # Return as a plain Python list for safety
    return (best_h.tolist(), float(c5_bound), int(n))
