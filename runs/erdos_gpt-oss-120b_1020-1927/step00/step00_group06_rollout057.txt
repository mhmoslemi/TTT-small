def project_onto_sum_box(v, target_sum):
    # v is 1D numpy array
    # Project onto [0,1] box and sum constraint.
    # Use binary search for lambda.
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

import time
import numpy as np

def _project_to_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {x : 0<=x<=1, sum(x)=target}."""
    lo = -v.max() - 10.0
    hi = v.max() + 10.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _max_overlap(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k))dx approximated by discrete correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    return float(corr.max())


def _max_shift_and_score(h: np.ndarray):
    """Return (shift, score) where score is the current max overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    idx = int(np.argmax(corr))
    return idx - (n - 1), float(corr[idx])


def _gradient_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the overlap score w.r.t. h for a fixed shift k (the worst shift)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if k >= 0:
        if k < n:
            g[: n - k] += h[k:]
            g[k:] += h[: n - k]
    else:
        ka = -k
        if ka < n:
            g[ka:] += h[: n - ka]
            g[: n - ka] += h[ka:]
    return g


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --------------------------------------------------------------------
    # initialise h
    # --------------------------------------------------------------------
    try:
        # use the provided construction if available
        h0 = np.array(initial_h_values, dtype=float, copy=True)
        n = h0.size
    except Exception:
        # fall back to a modest size if no parent is present
        n = 120
        h0 = rng.random(n).astype(float)

    # enforce feasibility (bounds + integral)
    h0 = _project_to_sum_box(h0, n / 2.0)

    cur_h = h0.copy()
    cur_score = _max_overlap(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    # --------------------------------------------------------------------
    # simulated‑annealing / subgradient loop
    # --------------------------------------------------------------------
    T = 0.05                # initial temperature
    T_min = 1e-6
    decay = 0.9999          # geometric cooling
    step_pair = 0.12        # magnitude for random pairwise moves
    step_grad = 0.03        # magnitude for gradient moves
    prob_grad = 0.3         # chance to attempt a gradient step
    stagnation_limit = 40000
    last_improve = 0
    it = 0

    while True:
        now = time.time()
        if now - start > budget_s - 0.05:
            break

        it += 1
        # decide move type
        if rng.random() < prob_grad:
            # ------------------------------------------------------------
            # gradient move aimed at reducing the worst shift
            # ------------------------------------------------------------
            k_star, _ = _max_shift_and_score(cur_h)
            grad = _gradient_for_shift(cur_h, k_star)
            new_h = cur_h + step_grad * grad
            # project back to feasible region
            new_h = _project_to_sum_box(new_h, n / 2.0)
        else:
            # ------------------------------------------------------------
            # random pairwise redistribution preserving the sum
            # ------------------------------------------------------------
            i, j = rng.integers(0, n, size=2, endpoint=False)

            delta_max = min(cur_h[i], 1.0 - cur_h[j])
            if delta_max <= 0.0:

            new_h[i] -= delta
            new_h[j] += delta
            # bounds already respected by construction

        delta_score = new_score - cur_score

        if delta_score <= 0.0 or rng.random() < np.exp(-delta_score / max(T, 1e-12)):

            cur_score = new_score
            if new_score < best_score - 1e-12:

                best_score = new_score

        step_grad = max(step_grad * 0.9999, 1e-4)

    # final tidy‑up
    best_h = _project_to_sum_box(best_h, n / 2.0)
    best_score = _max_overlap(best_h)
