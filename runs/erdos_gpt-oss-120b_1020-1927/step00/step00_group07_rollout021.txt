def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    # Determine N: if initial_h_values in globals, use its length else use 80
    if 'initial_h_values' in globals():
        initial = np.array(initial_h_values, copy=False)
        N = initial.shape[0]
        # create binary initial by thresholding at median (ensuring exactly N/2 ones)
        # using argsort
        idx = np.argsort(-initial)
        h = np.zeros(N, dtype=np.int8)
        h[idx[:N//2]] = 1
    else:
        N = 80
        h = np.zeros(N, dtype=np.int8)
        # simple pattern alternating
        h[::2][:N//2] = 1
    dx = 2.0 / N

    def score(arr):
        return np.max(np.correlate(arr, 1 - arr, mode='full')) * dx

    current = h.copy()
    cur_score = score(current)
    best = current.copy()
    best_score = cur_score

    # Precompute indices lists for fast random selection
    ones = np.flatnonzero(current == 1)
    zeros = np.flatnonzero(current == 0)

    start = time.time()
    T = 0.5
    Tmin = 1e-4
    decay = 0.9995
    iter = 0
    no_improve = 0
    # We'll run until budget - small margin
    while time.time() - start < budget_s - 0.5:
        # Pick random 1 and 0
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        # swap
        current[i1] = 0
        current[i0] = 1
        # compute new score
        new_score = score(current)
        delta = new_score - cur_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            # Metropolis
            prob = math.exp(-delta / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:
            cur_score = new_score
            # update ones/zeros arrays efficiently
            # swap indices in lists
            # remove i1 from ones, add to zeros; remove i0 from zeros, add to ones
            # simplest: recompute after each acceptance maybe okay O(N)
            # But we can do incremental updates.
            # We'll recompute every 1000 steps.
            no_improve = 0
        else:
            # revert swap
            current[i1] = 1
            current[i0] = 0
            # no update
            no_improve += 1

        # Keep best
        if cur_score < best_score - 1e-12:
            best = current.copy()
            best_score = cur_score
            no_improve = 0

        # temperature decay
        T = max(T * decay, Tmin)

        iter += 1

        # periodic recompute ones/zeros maybe to keep consistent
        if iter % 2000 == 0:
            ones = np.flatnonzero(current == 1)
            zeros = np.flatnonzero(current == 0)

        # random restart if stuck
        if no_improve > 5000:
            # reinitialize randomly but keep best
            # random binary with exactly N/2 ones
            perm = rng.permutation(N)
            current = np.zeros(N, dtype=np.int8)
            current[perm[:N//2]] = 1
            cur_score = score(current)
            ones = np.flatnonzero(current == 1)
            zeros = np.flatnonzero(current == 0)
            no_improve = 0

    # after loop
    final_h = best.astype(float)
    final_score = best_score
    return (final_h, final_score, N)

        N = init.shape[0]
    else:

    def score(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Construct initial binary candidate
    if 'initial_h_values' in globals():

        idx = np.argsort(-init)
        binary = np.zeros(N, dtype=np.int8)

        binary[:N//2] = 1
        rng.shuffle(binary)
    # Evaluate
    cur_bin = binary.copy()
    cur_score = score(cur_bin.astype(float))
    best_bin = cur_bin.copy()
    best_score = cur_score

    start = time.time()
    # Some time budget split phases
    # Phase 1: binary SA for t1 seconds or up to certain iterations
    phase1_time = budget_s * 0.5
    T = 0.5
    Tmin = 1e-4
    decay = 0.9995
    iter = 0
    no_improve = 0
    while time.time() - start < phase1_time:

        ones_idx = np.flatnonzero(cur_bin == 1)
        zeros_idx = np.flatnonzero(cur_bin == 0)
        if len(ones_idx) == 0 or len(zeros_idx) == 0:

        cur_bin[i1] = 0
        cur_bin[i0] = 1
        new_score = score(cur_bin.astype(float))

import numpy as np, time, math, cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        init_array = np.array(initial_h_values, copy=False)
        N = init_array.shape[0]
    else:
        N = 80
    # Ensure N even
    if N % 2 != 0:
        N += 1

    dx = 2.0 / N

    # Helper for computing score
    def compute_score(arr):
        return np.max(np.correlate(arr, 1 - arr, mode='full')) * dx

    # Try exact binary MILP
    try:
        # Build MILP
        h = cp.Variable(N, boolean=True)
        # Build pair list
        pair_index = {}
        indices_by_L = [[] for _ in range(N)]
        idx = 0
        for i in range(N):
            for j in range(i, N):
                pair_index[(i, j)] = idx
                L = j - i
                indices_by_L[L].append(idx)
                idx += 1
        M = idx
        y = cp.Variable(M, boolean=True)
        C = cp.Variable()
        constraints = []

        # sum h = N/2
        constraints.append(cp.sum(h) == N // 2)

        # product constraints
        # iterate over i,j pairs
        for (i, j), k in pair_index.items():
            constraints.append(y[k] <= h[i])
            constraints.append(y[k] <= h[j])
            constraints.append(y[k] >= h[i] + h[j] - 1)

        # overlap constraints for each lag L
        for L in range(N):
            # sum of first N-L elements of h
            if N - L > 0:
                overlap_expr = cp.sum(h[:N - L]) - cp.sum(cp.vstack([y[k] for k in indices_by_L[L]]))
            else:
                overlap_expr = 0
            constraints.append(overlap_expr <= C)

        # C non-negative
        constraints.append(C >= 0)

        prob = cp.Problem(cp.Minimize(C), constraints)

        # Solve within time budget
        start_solve = time.time()
        # solver may accept 'max_iters' but not typical; we just call solve
        prob.solve(solver=cp.ECOS_BB, verbose=False)
        solve_time = time.time() - start_solve
        # If solution found
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = np.rint(h.value).astype(int)  # round to nearest integer
            # Ensure sum constraint
            if np.sum(h_opt) != N // 2:
                # adjust by flipping few bits
                diff = int(N // 2 - np.sum(h_opt))
                # if diff>0, set diff zeros to ones
                zeros = np.where(h_opt == 0)[0]
                ones = np.where(h_opt == 1)[0]
                if diff > 0:
                    choose = rng.choice(zeros, size=diff, replace=False)
                    h_opt[choose] = 1
                elif diff < 0:
                    choose = rng.choice(ones, size=-diff, replace=False)
                    h_opt[choose] = 0
            c5_val = compute_score(h_opt.astype(float))
            return (h_opt.astype(float), c5_val, N)
    except Exception:
