def project_box_sum(x, target, tol=1e-9, max_iter=100):
    lo = -np.max(x) - 1  # low enough to clamp all to 0
    hi = 1 - np.min(x) + 1  # high enough to clamp all to 1
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        v = np.clip(x + mid, 0, 1)
        s = v.sum()
        if abs(s - target) < tol:
            return v
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo+hi)/2, 0, 1)

def project_box_sum(x, target, tol=1e-9, max_iter=100):
    lo = -np.max(x) - 2.0  # ensure all x+lo <= 0
    hi = 1.0 - np.min(x) + 2.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        v = np.clip(x + mid, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) <= tol:
            return v
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

def project_box_sum(x, target, tol=1e-8, max_iter=100):

    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0

    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        v = np.clip(x + mid, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) <= tol:
            return v
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo+hi)/2.0, 0.0, 1.0)

import time, math, random
import numpy as np

def _project_box_sum(x: np.ndarray, target: float, tol: float = 1e-8, max_iter: int = 80) -> np.ndarray:
    """Project x onto {0≤v_i≤1, sum(v)=target} by finding a common shift."""
    lo = -np.max(x) - 1.0          # makes x+lo ≤0 → all clipped to 0
    hi = 1.0 - np.min(x) + 1.0     # makes x+hi ≥1 → all clipped to 1
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        v = np.clip(x + mid, 0.0, 1.0)
        s = v.sum()
        if abs(s - target) <= tol:
            return v
        if s < target:
            lo = mid
        else:
            hi = mid
    # final compromise
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Return C5 for a discretised step function."""
    n = h.size
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / n)


def _subgrad_max_shift(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_{k} Σ h_i(1‑h_{i+k}) w.r.t. h (only for the worst shifts)."""
    n = h.size
    raw = np.correlate(h, 1.0 - h, mode="full")
    max_raw = raw.max()
    # offsets where the maximum is attained
    offs = np.where(np.isclose(raw, max_raw, atol=1e-12))[0] - (n - 1)
    w = 1.0 - h
    g = np.zeros_like(h)
    for r in offs:
        if r >= 0:
            # w term: i ∈ [0, n‑r‑1]
            g[: n - r] += w[r:]
            # -h term: i ∈ [r, n‑1]
            g[r:] -= h[: n - r]
        else:
            k = -r
            # w term: i ∈ [k, n‑1]
            g[k:] += w[: n - k]
            # -h term: i ∈ [0, n‑k‑1]
            g[: n - k] -= h[k:]
    return g


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    # ------------------------------------------------------------
    # setup
    random.seed(seed)
    np.random.seed(seed)

    # choose a modest grid size (larger than the provided 42 points)
    N = 200                     # number of sample points on [0,2]
    target_sum = N // 2         # integral constraint  ∑h = N/2
    # ------------------------------------------------------------
    # initial h: up‑sample the given construction if present,
    # otherwise start from the uniform ½‑function with a small
    # random perturbation.
    if "initial_h_values" in globals():
        init = initial_h_values.astype(float)
        n0 = init.shape[0]
        if n0 == N:
            h = init.copy()
        else:
            # linear interpolation onto the larger grid
            x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
            x = np.linspace(0.0, 2.0, N, endpoint=False)
            h = np.interp(x, x0, init)
            h = np.clip(h, 0.0, 1.0)
            h = _project_box_sum(h, target_sum)
    else:
        h = np.full(N, 0.5)
        # add a tiny random bump to leave the flat plateau
        h += 0.12 * (np.random.rand(N) - 0.5)
        h = np.clip(h, 0.0, 1.0)
        h = _project_box_sum(h, target_sum)

    best_h = h.copy()
    best_val = _c5_value(best_h)

    start = time.time()
    deadline = start + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------
    # main optimisation loop: many short gradient descents interleaved
    # with a brief simulated‑annealing polish.
    while time.time() < deadline:
        # ----- gradient descent from the current start -----------------
        cur = h.copy()
        cur_val = _c5_value(cur)
        eta = 0.5                       # initial step size
        # limit the inner descent to a few thousand iterations
        for _ in range(2500):
            if time.time() >= deadline:
                break
            grad = _subgrad_max_shift(cur)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-9:
                break
            # try a step; shrink if it does not improve
            for _ in range(4):
                cand = cur - eta * grad
                cand = _project_box_sum(cand, target_sum)
                cand_val = _c5_value(cand)
                if cand_val < cur_val - 1e-12:
                    cur, cur_val = cand, cand_val
                    break
                eta *= 0.5
            eta *= 0.99
        # keep the best seen so far
        if cur_val < best_val:
            best_val, best_h = cur_val, cur.copy()
            if best_val <= 0.38081:          # early stop
                break

        # ----- short simulated‑annealing refinement -------------------
        # use the gradient‑descent output as a warm start
        SA_h = cur.copy()
        SA_val = cur_val
        T = 0.018
        decay = 0.99997
        for step in range(40000):
            if time.time() >= deadline:
                break
            i, j = np.random.randint(0, N, size=2)
            while j == i:
                j = np.random.randint(0, N)
            hi, hj = SA_h[i], SA_h[j]
            lo = max(-hi, hj - 1.0)
            hi_bound = min(1.0 - hi, hj)
            if hi_bound <= lo:
                continue
            delta = lo + (hi_bound - lo) * np.random.rand()
            cand = SA_h.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta
            cand_val = _c5_value(cand)
            d = cand_val - SA_val
            if d < -1e-12 or (T > 1e-12 and np.random.rand() < math.exp(-d / T)):
                SA_h, SA_val = cand, cand_val
                if SA_val < best_val:
                    best_val, best_h = SA_val, SA_h.copy()
                    if best_val <= 0.38081:
                        break
            T *= decay
        # ----- random restart for the next outer iteration ------------
        # (keeps the global best untouched)
        h = np.full(N, 0.5)
        h += 0.18 * (np.random.rand(N) - 0.5)   # fresh perturbation
        h = np.clip(h, 0.0, 1.0)
        h = _project_box_sum(h, target_sum)

    # final safety projection and evaluation
    final_h = _project_box_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    final_val = _c5_value(final_h)

    return final_h, float(final_val), N
