def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx
```

def anneal_h(initial_h, max_iter, time_limit):
    h = initial_h.copy()
    N = len(h)
    dx = 2.0 / N
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val

    T0 = 0.01
    T = T0
    decay = 0.9999

    start_time = time.time()
    i = 0
    while True:
        if time.time() - start_time > time_limit:
            break
        # propose move
        i_idx, j_idx = np.random.randint(N, size=2)
        if i_idx == j_idx:
            continue
        # direction: choose increase i
        if np.random.rand() < 0.5:
            inc_idx, dec_idx = i_idx, j_idx
        else:
            inc_idx, dec_idx = j_idx, i_idx
        # allowed delta
        max_inc = min(1.0 - h[inc_idx], h[dec_idx])
        if max_inc <= 0:
            continue
        # choose delta: random fraction of max_inc
        delta = np.random.rand() * max_inc
        # maybe also allow full swap delta = max_inc if we want binary flips
        # apply
        new_h = h.copy()
        new_h[inc_idx] += delta
        new_h[dec_idx] -= delta

        new_val = compute_c5(new_h)

        diff = new_val - cur_val
        if diff < 0 or np.random.rand() < np.exp(-diff / T):
            h = new_h
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_h = h.copy()
        # schedule temperature
        T *= decay
        if T < 1e-6:
            T = 1e-6

        i += 1
    return best_h, best_val
```

import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic search for a step function h on [0,2] that minimizes
        C5 = max_k ∫ h(x) (1-h(x+k)) dx
    under the constraints 0≤h≤1 and ∫ h = 1.
    The function returns (h_array, c5_value, n_points).
    """
    np.random.seed(seed)
    start_total = time.time()
    # keep a small safety margin
    total_budget = budget_s * 0.96

    # ----------------------------------------------------------------------
    # Helper utilities
    # ----------------------------------------------------------------------
    def raw_max_corr(arr):
        """raw maximum of the discrete correlation ∑ h[i]*(1-h[i+s])"""
        return np.correlate(arr, 1.0 - arr, mode='full').max()

    def c5_from_raw(arr):
        """scaled C5 value from the raw correlation max."""
        n = arr.shape[0]
        return raw_max_corr(arr) * (2.0 / n)

    def random_binary(n):
        """binary vector with exactly n//2 ones."""
        a = np.zeros(n, dtype=np.float64)
        idx = np.random.choice(n, n // 2, replace=False)
        a[idx] = 1.0
        return a

    def random_feasible(n):
        """continuous feasible vector: uniform in [0,1] then project to sum=n/2."""
        a = np.random.rand(n)
        target = n / 2.0
        cur = a.sum()
        # simple affine adjustment (may slightly violate bounds, then clip)
        a = a * (target / cur)
        # clip and fix any residual
        a = np.clip(a, 0.0, 1.0)
        diff = target - a.sum()
        if abs(diff) > 1e-12:
            # adjust a single non‑boundary entry
            for i in range(n):
                if 0.0 < a[i] < 1.0:
                    a[i] += diff
                    break
        return a

    # ----------------------------------------------------------------------
    # Determine initial solution (if provided) and its size
    # ----------------------------------------------------------------------
    try:
        init_h = initial_h_values.copy()
        n_init = init_h.shape[0]
    except Exception:
        init_h = None
        n_init = None

    # ----------------------------------------------------------------------
    # Core simulated‑annealing routine (works for any length)
    # ----------------------------------------------------------------------
    def anneal(initial_h, time_allow):
        """returns (best_h, best_raw_value) found within time_allow seconds."""
        h = initial_h.copy()
        n = h.shape[0]
        cur_raw = raw_max_corr(h)
        best_h = h.copy()
        best_raw = cur_raw

        # annealing schedule
        T = 0.02           # initial temperature
        decay = 0.99997    # per‑iteration decay
        min_T = 1e-7

        stagnate = 0       # steps without improvement
        it = 0
        start = time.time()

        while time.time() - start < time_allow:
            # pick two distinct indices
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue

            # decide which one will increase, which one will decrease
            if np.random.rand() < 0.5:
                inc, dec = i, j
            else:
                inc, dec = j, i

            # maximum admissible transfer while staying in [0,1]
            max_inc = min(1.0 - h[inc], h[dec])
            if max_inc <= 0.0:
                continue

            # choose delta: often full swap, sometimes a fraction
            if np.random.rand() < 0.3:
                delta = max_inc               # full possible swap
            else:
                delta = np.random.rand() * max_inc

            # propose new vector
            h_new = h.copy()
            h_new[inc] += delta
            h_new[dec] -= delta

            new_raw = raw_max_corr(h_new)
            diff = new_raw - cur_raw

            if diff < 0 or np.random.rand() < np.exp(-diff / max(T, 1e-12)):
                h = h_new
                cur_raw = new_raw
                if cur_raw < best_raw - 1e-12:
                    best_raw = cur_raw
                    best_h = h.copy()
                    stagnate = 0
                else:
                    stagnate += 1
            else:
                stagnate += 1

            # temperature decay
            T = max(min_T, T * decay)

            # occasional restart if stuck too long
            if stagnate > 25000:
                # random binary restart (keeps the sum exact)
                h = random_binary(n)
                cur_raw = raw_max_corr(h)
                stagnate = 0

            it += 1
        return best_h, best_raw

    # ----------------------------------------------------------------------
    # Main search: first use the provided construction (if any), then
    # explore a larger resolution if time permits.
    # ----------------------------------------------------------------------
    best_overall_h = None
    best_overall_raw = np.inf
    best_overall_n = None

    # allocate half of the budget to the initial size (if we have one)
    if init_h is not None:
        half_budget = total_budget * 0.55
        h_opt, raw_opt = anneal(init_h,
                                 time_allow=min(half_budget,
                                                total_budget - (time.time() - start_total)))
        if raw_opt < best_overall_raw:
            best_overall_raw = raw_opt
            best_overall_h = h_opt.copy()
            best_overall_n = h_opt.shape[0]

    # allocate remaining time to a larger discretisation (e.g. 100 points)
    remaining = total_budget - (time.time() - start_total)
    if remaining > 5.0:               # need a non‑trivial slice
        n_large = 100
        # start from a random feasible vector (continuous)
        h_start = random_feasible(n_large)
        h_opt, raw_opt = anneal(h_start,
                                 time_allow=remaining)
        if raw_opt < best_overall_raw:
            best_overall_raw = raw_opt
            best_overall_h = h_opt.copy()
            best_overall_n = n_large

    # ----------------------------------------------------------------------
    # Final sanity – enforce exact sum constraint (tiny adjustment if needed)
    # ----------------------------------------------------------------------
    if best_overall_h is not None:
        target_sum = best_overall_n / 2.0
        cur_sum = best_overall_h.sum()
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            # adjust a non‑boundary entry if possible
            for idx in range(best_overall_n):
                if 0.0 <= best_overall_h[idx] + diff <= 1.0:
                    best_overall_h[idx] += diff
                    break
        # recompute the bound with the final (exact) vector
        final_c5 = c5_from_raw(best_overall_h)
    else:
        # fallback – should never happen, but keep a trivial construction
        best_overall_n = 70
        best_overall_h = np.zeros(best_overall_n, dtype=np.float64)
        best_overall_h[:best_overall_n // 2] = 1.0
        final_c5 = c5_from_raw(best_overall_h)

    return best_overall_h, float(final_c5), int(best_overall_n)
