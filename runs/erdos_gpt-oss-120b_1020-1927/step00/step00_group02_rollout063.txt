def project_box_sum(v, target):
    # target sum across elements.
    # Use binary search on lambda.
    lo = -1.0
    hi = 1.0
    # adjust lo, hi large enough
    for _ in range(60):
        lam = (lo+hi)/2
        h = np.clip(v + lam, 0.0, 1.0)
        s = h.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    h = np.clip(v + (lo+hi)/2, 0.0, 1.0)
    # In case of rounding error adjust by small epsilon (should be very close)
    return h

def run(seed=42, budget_s=1100, **kwargs):

    # seed
    np.random.seed(seed)
    random.seed(seed)

    # get global initial_h_values if exists:
    try:
        init = np.array(initial_h_values, dtype=float)
    except NameError:
        init = None

    n = init.shape[0] if init is not None else 45
    target_sum = n/2
    dx = 2.0 / n

    if init is None:
        # random feasible
        h = np.random.rand(n)
        # project onto bounds and sum:
        h = project_box_sum(h, target_sum)
    else:
        h = init.copy()
        # ensure feasible via projection
        h = np.clip(h, 0.0, 1.0)
        # adjust to target sum via projection
        h = project_box_sum(h, target_sum)

    def compute_c5(vec):
        corr = np.correlate(vec, 1 - vec, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)

    start = time.time()
    max_iter = 200000
    # parameters
    step0 = 0.1
    temp0 = 0.05

    for it in range(max_iter):
        elapsed = time.time() - start
        if elapsed > budget_s * 0.95:
            break

        # compute C5 and shift with max
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_c5 = corr[max_idx] * dx
        # best update
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()

        # shift s index:
        s = max_idx - (n - 1)
        # compute gradient for shift s
        # vectorized gradient
        grad = np.ones(n) * 1.0  # base 1 for i - s out-of-range

        # subtract term for i + s

import numpy as np
import time
import math
import random

# ------------------------------------------------------------
# helper: project a vector onto the box [0,1] and the affine
# constraint sum(x) = target (here target = n/2)
def _project_box_sum(v, target):
    """Projection of v onto {0<=x<=1, sum(x)=target}."""
    # binary search on lambda such that sum(clip(v+lam,0,1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


# ------------------------------------------------------------
def _c5_value(h, dx):
    """Compute C5 for a given vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


# ------------------------------------------------------------
def _gradient(h, shift):
    """Sub‑gradient of the max‑shift term (shift may be negative)."""
    n = h.shape[0]
    g = np.ones(n, dtype=np.float64)

    if shift >= 0:
        # part (1 - h[i-shift])
        if shift > 0:
            g[shift:] = 1.0 - h[: n - shift]
        # subtract h[i+shift]
        if shift > 0:
            g[: n - shift] -= h[shift:]
    else:
        sp = -shift
        # (1 - h[i+sp]) where defined
        if sp > 0:
            g[: n - sp] = 1.0 - h[sp:]
        # subtract h[i-sp] where defined
        if sp > 0:
            g[sp:] -= h[: n - sp]

    # for shift == 0 this reduces to 1 - 2h
    return g


# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h (as a list), its C5 bound and n_points."""
    np.random.seed(seed)
    random.seed(seed)

    # ----------------------------------------------------------------
    # obtain initial vector if the harness supplied one
    try:
        init = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        init = None

    # length of the discretisation
    if init is not None:
        n = int(init.shape[0])
        h = init.copy()
    else:
        n = 45                     # fallback size
        h = np.random.rand(n)

    target_sum = n / 2.0          # because sum(h)*dx == 1, dx = 2/n
    dx = 2.0 / n

    # project the starting point onto the feasible set
    h = np.clip(h, 0.0, 1.0)
    h = _project_box_sum(h, target_sum)

    # current C5 and best seen
    cur_c5 = _c5_value(h, dx)
    best_h = h.copy()
    best_c5 = cur_c5

    # optimisation parameters
    max_iter = 500_000
    step0 = 0.6                  # initial step size for gradient descent
    temp0 = 0.08                 # initial temperature for simulated‑annealing moves
    last_improve = 0
    stall_limit = 30_000

    start = time.time()
    for it in range(1, max_iter + 1):
        # time check
        if time.time() - start > budget_s * 0.95:
            break

        # ----------------------------------------------------
        # 1) locate a worst shift (one attaining the max)
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = corr.max()
        idxs = np.nonzero(corr == max_val)[0]
        shift_idx = np.random.choice(idxs)
        shift = shift_idx - (n - 1)

        # 2) gradient step for that shift
        g = _gradient(h, shift)
        step = step0 * (1.0 - it / max_iter)
        h_candidate = h - step * g
        h_candidate = _project_box_sum(h_candidate, target_sum)

        cand_c5 = _c5_value(h_candidate, dx)

        # 3) simulated‑annealing acceptance
        if cand_c5 < cur_c5:
            accept = True
        else:
            T = temp0 * (1.0 - it / max_iter) + 1e-9
            accept = (np.random.rand() < math.exp(-(cand_c5 - cur_c5) / T))

        if accept:
            h = h_candidate
            cur_c5 = cand_c5
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = h.copy()
                last_improve = it

        # ----------------------------------------------------
        # occasional random transfer move to increase exploration
        if it % 250 == 0:
            i, j = random.sample(range(n), 2)
            # move a small amount from i to j (preserving sum)
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta > 1e-12:
                delta = max_delta * np.random.rand()
                h_tmp = h.copy()
                h_tmp[i] -= delta
                h_tmp[j] += delta
                tmp_c5 = _c5_value(h_tmp, dx)
                if tmp_c5 < cur_c5:
                    h = h_tmp
                    cur_c5 = tmp_c5
                    if cur_c5 < best_c5 - 1e-12:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        last_improve = it

        # ----------------------------------------------------
        # restart if stuck for a while
        if it - last_improve > stall_limit:
            # perturb the best solution slightly and continue
            noise = np.random.normal(scale=0.05, size=n)
            h = best_h + noise
            h = np.clip(h, 0.0, 1.0)
            h = _project_box_sum(h, target_sum)
            cur_c5 = _c5_value(h, dx)
            last_improve = it

    # final safety projection (to eliminate tiny drift)
    best_h = _project_box_sum(best_h, target_sum)
    final_c5 = _c5_value(best_h, dx)

    return best_h.tolist(), float(final_c5), int(n)
