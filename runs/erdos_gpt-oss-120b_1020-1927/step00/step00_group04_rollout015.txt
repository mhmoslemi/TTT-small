def compute_max_overlap(h):
    n = len(h)
    dx = 2.0/n
    a = h.astype(np.float64)
    b = 1 - a
    # reverse b
    b_rev = b[::-1]
    L = 1 << ((2*n - 1).bit_length())
    Fa = np.fft.rfft(a, n=L)
    Fb = np.fft.rfft(b_rev, n=L)
    conv = np.fft.irfft(Fa * Fb, n=L)[:2*n-1]
    return np.max(conv) * dx

```
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    a = h.astype(np.float64)
    b = 1 - a
    # Convolution a * b_rev:
    b_rev = b[::-1]
    L = 1 << ((2*n - 1).bit_length())
    Fa = np.fft.rfft(a, n=L)
    Fb_rev = np.fft.rfft(b_rev, n=L)
    conv = np.fft.irfft(Fa * Fb_rev, n=L)[:2*n-1]
    return conv.max() * dx
```

    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    Fa = np.fft.rfft(a, n=L)
    Fb = np.fft.rfft(b_rev, n=L)
    conv = np.fft.irfft(Fa * Fb, n=L)[: 2 * n - 1]
    return float(conv.max() * dx)


def hill_climb_binary(n, init_h, max_swaps, deadline):
    """Swap‑based simulated annealing for a binary vector of length n
    with exactly n/2 ones."""
    cur = init_h.copy()
    cur_score = compute_c5(cur)
    best_h = cur.copy()
    best_score = cur_score
    ones = np.flatnonzero(cur)
    zeros = np.flatnonzero(1 - cur)
    T0 = cur_score * 0.03 + 1e-6

    for it in range(max_swaps):
        if time.time() > deadline:
            break
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)

        # propose swap
        cur[i1] = 0
        cur[i0] = 1
        new_score = compute_c5(cur)
        delta = new_score - cur_score
        T = T0 * (1 - it / max_swaps)

        if delta < 0 or np.random.rand() < math.exp(-delta / max(T, 1e-12)):
            cur_score = new_score
            ones = np.flatnonzero(cur)
            zeros = np.flatnonzero(1 - cur)
            if new_score < best_score:
                best_score = new_score
                best_h = cur.copy()
        else:
            # revert
            cur[i1] = 1
            cur[i0] = 0

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95

    # obtain any provided construction
    try:
        base = initial_h_values
        base_n = len(base)
    except NameError:
        base = None
        base_n = None

    # candidate sample sizes
    sizes = []
    if base_n:
        sizes.append(base_n)
        mult = 2
        while base_n * mult <= 400:
            sizes.append(base_n * mult)
            mult <<= 1
    else:
        sizes = [100, 200, 300]

    best_h = None
    best_score = np.inf

    for n in sizes:
        if time.time() > deadline:
            break
        half = n // 2

        # initialise a binary vector with the correct weight
        if base is not None and n % base_n == 0:
            rep = n // base_n
            h0 = np.repeat(base, rep)
            h0 = (h0 > 0.5).astype(np.int8)
            cur = int(h0.sum())
            diff = half - cur
            if diff > 0:
                z = np.where(h0 == 0)[0]
                np.random.shuffle(z)
                h0[z[:diff]] = 1
            elif diff < 0:
                o = np.where(h0 == 1)[0]
                np.random.shuffle(o)
                h0[o[: -diff]] = 0
        else:
            h0 = np.zeros(n, dtype=np.int8)
            h0[:half] = 1
            np.random.shuffle(h0)

        # amount of effort scales with n
        max_swaps = 40000 if n <= 200 else 60000
        h_opt, sc = hill_climb_binary(n, h0, max_swaps, deadline)

        if sc < best_score:
            best_score = sc
            best_h = h_opt.copy()

    if best_h is None:
        # fallback: constant 0.5 (always feasible)
        n = 100
        best_h = np.full(n, 0.5, dtype=float)
        best_score = compute_c5(best_h)

    return best_h.astype(float), float(best_score), int(best_h.shape[0])
