import time
import math
import random
import numpy as np

def compute_c5(arr: np.ndarray, dx: float) -> float:
    """Return max_{k} ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    # correlation of h with its complement; arr is 0/1 (or any [0,1] values)
    corr = np.correlate(arr.astype(float), 1.0 - arr.astype(float), mode="full")
    return float(np.max(corr) * dx)


def make_binary(arr: np.ndarray) -> np.ndarray:
    """Round to 0/1 and enforce the exact half‑mass constraint."""
    n = arr.size
    target = n // 2
    bin_arr = np.rint(arr).astype(int)
    bin_arr = np.clip(bin_arr, 0, 1)

    cur = int(bin_arr.sum())
    if cur > target:                     # too many ones
        excess = cur - target
        ones = np.flatnonzero(bin_arr)
        if excess > 0:
            flip = np.random.choice(ones, size=excess, replace=False)
            bin_arr[flip] = 0
    elif cur < target:                   # too few ones
        deficit = target - cur
        zeros = np.flatnonzero(1 - bin_arr)
        if deficit > 0:
            flip = np.random.choice(zeros, size=deficit, replace=False)
            bin_arr[flip] = 1
    return bin_arr


def random_binary(n: int) -> np.ndarray:
    """Uniform random 0/1 vector with exactly n/2 ones."""
    ones = n // 2
    arr = np.zeros(n, dtype=int)
    arr[np.random.choice(n, size=ones, replace=False)] = 1
    return arr


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a low‑overlap step function.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : np.ndarray of shape (n_points,) with entries in [0,1]
        c5_bound : float, the computed max overlap integral
        n_points : int, length of the discretisation
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()

    # ------------------------------------------------------------------
    # initialise problem size and starting vector
    # ------------------------------------------------------------------
    try:
        # existing parent construction
        init = np.array(initial_h_values, copy=False)
        n = int(init.size)
        if n % 2 != 0:                       # enforce even length
            # drop the last sample to keep things simple
            init = init[:-1]
            n -= 1
        cur = make_binary(init)
    except NameError:
        # no parent supplied – pick a modest size that still gives a fine grid
        n = 200
        cur = random_binary(n)

    dx = 2.0 / n
    cur_score = compute_c5(cur, dx)

    best_arr = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # simulated‑annealing style search
    # ------------------------------------------------------------------
    temperature = 0.02            # initial “accept‑worse” scale
    decay = 0.9999                # exponential cooling per iteration
    min_temp = 1e-6
    max_swap = n // 2             # we may swap up to half the points at once
    it = 0
    last_improve = 0

    while True:
        # time check
        if time.time() - start_time > budget_s - 0.2:
            break

        # propose a neighbour by swapping t ones with t zeros
        t = np.random.randint(1, max_swap + 1)
        ones_idx = np.flatnonzero(cur)
        zeros_idx = np.flatnonzero(1 - cur)

        # ensure we can actually swap t elements
        max_possible = min(ones_idx.size, zeros_idx.size)
        if t > max_possible:
            t = max_possible
        if t == 0:
            break

        sel_ones = np.random.choice(ones_idx, size=t, replace=False)
        sel_zeros = np.random.choice(zeros_idx, size=t, replace=False)

        new = cur.copy()
        new[sel_ones] = 0
        new[sel_zeros] = 1

        new_score = compute_c5(new, dx)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0 or np.random.rand() < math.exp(-delta / max(temperature, 1e-12)):
            cur = new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_arr = cur.copy()
                last_improve = it

        # temperature schedule
        temperature = max(min_temp, temperature * decay)

        # restart if stagnation is long
        if it - last_improve > 200_000:
            cur = random_binary(n)
            cur_score = compute_c5(cur, dx)
            temperature = 0.02
            last_improve = it

        it += 1

    # ------------------------------------------------------------------
    # return the best discovered configuration
    # ------------------------------------------------------------------
    h_vals = best_arr.astype(float)
    c5_bound = float(best_score)
    return (h_vals, c5_bound, n)
