def random_move(h, max_transfer=0.1):
    # choose i with h[i] > 0, j with h[j] < 1
    idx_pos = np.where(h > 0)[0]
    idx_neg = np.where(h < 1)[0]
    if len(idx_pos)==0 or len(idx_neg)==0: return h.copy()
    i = np.random.choice(idx_pos)
    j = np.random.choice(idx_neg)
    # ensure i != j
    while j == i:
        j = np.random.choice(idx_neg)
    max_delta = min(h[i], 1 - h[j])
    delta = max_delta * (np.random.rand() - 0.5) * 2 * max_transfer
    # ensure delta within bounds
    delta = np.clip(delta, -max_delta, max_delta)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    # due to rounding error ensure sum within tolerance
    return new_h
```

import numpy as np
import time
import random

def _c5_of_h(h):
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # full cross‑correlation of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _worst_shift_move(h):
    """
    Produce a neighbour of h by moving a small amount of mass
    from an index that contributes heavily to the current worst shift
    to its partner at that shift.
    """
    n = h.shape[0]
    # correlation to locate the worst shift
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)                # signed shift giving the max overlap

    if shift >= 0:
        i_range = np.arange(n - shift)          # i such that i+shift is in range
        j_range = i_range + shift
    else:
        i_range = np.arange(-shift, n)           # i such that i+shift stays ≥0
        j_range = i_range + shift

    # contribution of each pair (i, j) to the worst‑shift overlap
    prod = h[i_range] * (1.0 - h[j_range])
    total = prod.sum()
    if total <= 0.0:
        # degenerate case – just pick a random feasible pair
        i = np.random.choice(i_range)
    else:
        probs = prod / total
        i = np.random.choice(i_range, p=probs)

    j = i + shift
    # amount we may transfer without breaking bounds
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0.0:
        return h  # nothing feasible to move
    # move a random fraction (up to half) of that amount
    delta = max_delta * np.random.rand() * 0.5
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new


def _random_transfer_move(h, scale=0.4):
    """Transfer a small random amount between two unrelated feasible positions."""
    n = h.shape[0]
    pos = np.where(h > 1e-12)[0]
    neg = np.where(h < 1.0 - 1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return h
    i = np.random.choice(pos)
    j = np.random.choice(neg)
    while j == i:
        j = np.random.choice(neg)
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0.0:
        return h
    delta = max_delta * np.random.rand() * scale
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

    Optimize the step function for the Erdős minimum‑overlap problem.
    Returns (h_values, c5_bound, n_points).
    """

    random.seed(seed)
    start = time.time()
