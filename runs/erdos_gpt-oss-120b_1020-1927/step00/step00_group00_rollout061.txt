import numpy as np, time, math, random

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def simulated_annealing(h_init, budget, seed):
    rng = np.random.default_rng(seed)
    h = h_init.copy()
    n = len(h)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    start = time.time()
    # schedule: total_time proportion assigned for this size
    total = budget
    it = 0
    while time.time() - start < total:
        # choose two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = h[i], h[j]
        # compute max delta for increasing hi and decreasing hj
        max_inc = min(1 - hi, hj)
        max_dec = min(hi, 1 - hj)
        # decide direction
        if max_inc > 0 and max_dec > 0:
            if rng.random() < 0.5:
                max_delta = max_inc
                sign = 1
            else:
                max_delta = max_dec
                sign = -1
        elif max_inc > 0:
            max_delta = max_inc
            sign = 1
        elif max_dec > 0:
            max_delta = max_dec
            sign = -1
        else:
            continue  # both at bounds
        # schedule step factor decaying with iteration
        # compute temperature
        elapsed = time.time() - start
        t_ratio = elapsed / total
        T = (best_c5) * (0.02 + (0.5 - 0.02)*(1 - t_ratio))  # initial high then low
        step_factor = max(0.001, 1 - t_ratio)  # gradually shrink
        delta = rng.random() * max_delta * step_factor
        if delta == 0:
            continue
        # apply move
        if sign == 1:
            h[i] += delta
            h[j] -= delta
        else:
            h[i] -= delta
            h[j] += delta
        # evaluate
        new_c5 = compute_c5(h)
        # acceptance
        if new_c5 < current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Metropolis acceptance
            # Probability exp(-(new-current)/T)
            if rng.random() < math.exp(-(new_c5 - current_c5) / T):
                current_c5 = new_c5
            else:
                # revert
                if sign == 1:
                    h[i] -= delta
                    h[j] += delta
                else:
                    h[i] += delta
                    h[j] -= delta
        it += 1
    return best_h, best_c5
```

def optimize_h(h_initial, time_budget, seed):
    rng = np.random.default_rng(seed)
    h = h_initial.copy()
    n = len(h)
    cur = compute_c5(h)
    best = cur
    best_h = h.copy()
    start = time.time()
    # Initialize temperature based on initial cur
    T0 = cur * 0.5  # or something
    # Set some schedule
    while time.time() - start < time_budget:
        # Choose indices
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = h[i], h[j]
        # Determine allowed max delta if we move from hi to hi+delta, hj-delta (keeping sum)
        max_up = min(1.0 - hi, hj)  # hi up, j down
        max_down = min(hi, 1.0 - hj)  # hi down, j up
        # choose direction
        if max_up > 0 and max_down > 0:
            if rng.random() < 0.5:
                max_delta = max_up
                direction = 1
            else:
                max_delta = max_down
                direction = -1
        elif max_up > 0:
            max_delta = max_up
            direction = 1
        elif max_down > 0:
            max_delta = max_down
            direction = -1
        else:
            continue
        # step size decays
        elapsed = time.time() - start
        t_ratio = elapsed / time_budget
        # compute delta based on max_delta * (0.1 + 0.9 * (1 - t_ratio))
        step_scale = max(0.001, 1.0 - t_ratio)  # decays from 1 to 0.001
        delta = rng.random() * max_delta * step_scale
        if delta <= 0:
            continue
        # apply move
        if direction == 1:
            h[i] = hi + delta
            h[j] = hj - delta
        else:
            h[i] = hi - delta
            h[j] = hj + delta
        new = compute_c5(h)
        # acceptance
        accept = False
        if new < cur:
            accept = True
        else:
            # compute temperature schedule
            T = T0 * (1.0 - t_ratio) + 1e-6
            if rng.random() < math.exp(-(new - cur) / T):
                accept = True
        if accept:
            cur = new
            if new < best:
                best = new
                best_h = h.copy()
        else:
            # revert
            if direction == 1:
                h[i] = hi
                h[j] = hj
            else:
                h[i] = hi
                h[j] = hj
        # early stop if best below target
        if best < 0.3808:
            break
    return best_h, best
```

    # sample at centers
    x = (np.arange(n) + 0.5) * dx
    h = 0.5 + a * np.cos(np.pi * x / 2)

    target = n / 2.0

        else:

def anneal_optimize(h0, budget, seed):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    # compute initial
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur = np.max(corr) * dx
    best = cur
    start = time.time()
    # Determine initial temperature heuristically: may use cur
    T0 = cur * 0.5 + 1e-3
    while time.time() - start < budget:
        # compute current correlation and lag
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        ell = max_idx - (n - 1)
        # Determine overlap region
        if ell >= 0:
            i_start = 0
            i_end = n - ell
        else:
            i_start = -ell
            i_end = n
        # Overlap region length
        overlap_len = i_end - i_start
        # Choose i weighted by contribution if possible
        if overlap_len > 0:
            cont = h[i_start:i_end] * (1.0 - h[i_start+ell:i_end+ell])
            # pick i biased to high cont
            probs = cont / (cont.sum() + 1e-12)
            i_rel = rng.choice(overlap_len, p=probs)
            i = i_start + i_rel
        else:
            i = rng.integers(n)
        # Choose j distinct
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # decide direction: aim to decrease contribution
        # compute contribution for i
        if 0 <= i+ell < n:
            ci = h[i] * (1.0 - h[i+ell])
        else:
            ci = 0.0
        # try to move mass from i to j (decrease hi)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            # try opposite direction
            max_delta = min(1.0 - h[i], h[j])
            if max_delta <= 0:
                continue
            direction = 1  # increase i, decrease j
        else:
            direction = -1  # decrease i, increase j
        # schedule step_scale
        elapsed = time.time() - start
        t_ratio = elapsed / budget
        step_scale = max(0.001, 1.0 - t_ratio)
        delta = rng.random() * max_delta * step_scale
        if delta <= 0:
            continue
        # apply move
        if direction == -1:
            h[i] -= delta
            h[j] += delta
        else:
            h[i] += delta
            h[j] -= delta
        # compute new score
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new = np.max(new_corr) * dx
        # acceptance
        accept = False
        if new < cur:
            accept = True
        else:
            T = T0 * (1.0 - t_ratio) + 1e-12
            if rng.random() < math.exp(-(new - cur) / T):
                accept = True
        if accept:
            cur = new
            if new < best:
                best = new
                best_h = h.copy()
        else:
            # revert
            if direction == -1:
                h[i] += delta
                h[j] -= delta
            else:
                h[i] -= delta
                h[j] += delta
        # early exit if reached target low
        if best < 0.3808:
            break
    return best_h, best

def upsample(h, factor, rng):
    n = len(h)
    n2 = n * factor
    dx = 2.0 / n
    dx2 = 2.0 / n2
    x_old = (np.arange(n) + 0.5) * dx
    x_new = (np.arange(n2) + 0.5) * dx2
    h2 = np.interp(x_new, x_old, h)
    h2 = np.clip(h2, 0.0, 1.0)
    target = n2 / 2.0
    diff = target - h2.sum()
    if abs(diff) > 1e-10:
        idx = rng.integers(n2)
        # try few attempts to adjust within bounds
        for _ in range(5):
            if 0 <= h2[idx] + diff <= 1:
                h2[idx] += diff
                break
            idx = rng.integers(n2)
        else:
            # fallback linear correction
            h2 += diff / n2
            h2 = np.clip(h2, 0.0, 1.0)
    return h2

def enforce_sum(h):

    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # try a single element adjustment
    mask = (h + diff >= 0) & (h + diff <= 1)
    idx = np.nonzero(mask)[0]

        return h

            if rng.random() < math.exp(-(new - cur) / T):
                accept = True
        if accept:
            cur = new
            if new < best:
                best, best_h = new, h.copy()
        else:
            if direction == -1:
                h[i] += delta
                h[j] -= delta
            else:
                h[i] -= delta
                h[j] += delta
        if best < 0.3808:
            break
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    base_n = 120                     # moderate resolution start
    candidates = []

    # use parent if supplied
    try:
        init = np.array(initial_h_values, dtype=float)
        if init.size != base_n:
            init = resample_to_n(init, base_n)
        candidates.append((init, compute_c5(init)))
    except NameError:
        pass

    # scan cosine family
    elapsed = time.time() - start
    if elapsed < budget_s * 0.05:
        cos_h, cos_c = best_cosine_shape(base_n)
        candidates.append((cos_h, cos_c))

    # pick best starter
    best_h, best_val = None, np.inf
    for h, v in candidates:
        if v < best_val:
            best_h, best_val = h.copy(), v
    if best_h is None:                # fallback
        best_h = np.full(base_n, 0.5)
        best_val = compute_c5(best_h)

    # main annealing phase
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 5:
        h_opt, v_opt = anneal_optimize(best_h, remaining * 0.7, seed + 1)
        if v_opt < best_val:
            best_h, best_val = h_opt, v_opt

    # optional refinement on a finer grid
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 5:
        h_fine = upsample(best_h, 2, rng)            # double points
        h_opt2, v_opt2 = anneal_optimize(h_fine, remaining * 0.7, seed + 2)
        if v_opt2 < best_val:
            best_h, best_val = h_opt2, v_opt2

    # final clean‑up
    np.clip(best_h, 0.0, 1.0, out=best_h)
    best_h = enforce_sum(best_h)
    final_c5 = compute_c5(best_h)
    return best_h, float(final_c5), best_h.size
