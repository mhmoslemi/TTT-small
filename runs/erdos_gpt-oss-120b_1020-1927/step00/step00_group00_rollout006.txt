def adjust_sum(h, target):
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # Try to add diff to elements that can accommodate
    if diff > 0:
        # find indices where h < 1
        idx = np.where(h < 1 - 1e-12)[0]
        if idx.size == 0:
            raise ValueError('cannot increase sum')
        # find idx where capacities >= diff
        capacities = 1 - h[idx]
        # find index where capacity >= diff
        big = idx[capacities >= diff]
        if big.size > 0:
            i = big[0]
            h[i] += diff
            return h
        # else, distribute across many entries
        # allocate diff equally over all these indices but not exceed each.
        # We'll just add capacity of each until diff done.
        for i in idx:
            add = min(diff, 1 - h[i])
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
        if abs(diff) > 1e-9:
            # something wrong
            pass
    else:
        diff = -diff
        idx = np.where(h > 1e-12)[0]

        if big.size > 0:
            i = big[0]
            h[i] -= diff
            return h

            sub = min(diff, h[i])
            h[i] -= sub
            diff -= sub
            if diff <= 1e-12:
                break
    return h

    cur_h = h0.copy()

    itr = 0
    max_step = 0.2  # initial max step
    # we can schedule decreasing max_step each time
    # use temperature for acceptance of worse moves.
    T0 = 0.01
    while time.time() - start_time < time_limit:
        itr += 1
        # optionally decay step
        max_step = max(0.005, max_step * 0.9999)  # gradually shrink
        # compute correlation to get worst shift
        corr = np.correlate(cur_h, 1 - cur_h, mode='full') * dx
        idx_max = np.argmax(corr)

        # try targeted move

        # try up to 5 attempts to find viable move
        for _ in range(5):
            new_h, delta = attempt_targeted_move(cur_h, shift, max_step)
            if new_h is not None:
                break
        if new_h is None:
            # fallback random move
            new_h = random_move(cur_h, max_step)
        # evaluate new
        new_score = evaluate(new_h)
        # acceptance
        diff = new_score - cur_score
        if diff < 0 or np.random.rand() < np.exp(-diff / (T0 * (1 + itr * 1e-5))):

            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_h = new_h.copy()
                best_score = new_score
        # optionally restart if stagnation
        if itr % 5000 == 0 and time.time() - start_time > 0.1 * time_limit:
            # restart from best with random perturb
            cur_h = best_h.copy()
            # add small random noise while preserving sum
            cur_h = perturb(cur_h, magnitude=0.05)
            cur_h = project_sum(cur_h)
            cur_score = evaluate(cur_h)
    return best_h, best_score

import time
import math
import numpy as np

# ------------------ helper functions ---------------------------------

def evaluate(h: np.ndarray) -> float:
    """Return C5 for a given discretized step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # cross‑correlation h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def adjust_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Change h in‑place so that sum(h) == target while keeping 0≤h≤1."""
    total = float(np.sum(h))
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    n = h.shape[0]
    if diff > 0.0:               # need to increase some entries
        caps = 1.0 - h
        idx = np.where(caps >= diff)[0]
        if idx.size:
            h[idx[0]] += diff
            return h
        order = np.argsort(-caps)  # largest capacity first
        for i in order:
            add = min(diff, caps[i])
            if add <= 0.0:
                continue
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:                         # need to decrease some entries
        diff = -diff
        caps = h
        idx = np.where(caps >= diff)[0]
        if idx.size:
            h[idx[0]] -= diff
            return h
        order = np.argsort(-caps)  # largest value first
        for i in order:
            sub = min(diff, caps[i])
            if sub <= 0.0:
                continue
            h[i] -= sub
            diff -= sub
            if diff <= 1e-12:
                break
    # tiny residual fix
    residual = target - float(np.sum(h))
    if abs(residual) > 1e-12:
        for i in range(n):
            if 0.0 <= h[i] + residual <= 1.0:
                h[i] += residual
                break
    return h


def upsample(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid, then re‑project to constraints."""
    old_n = h.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, h)
    new_h = np.clip(new_h, 0.0, 1.0)
    new_h = adjust_sum(new_h, new_n / 2.0)
    return new_h


def attempt_targeted_move(h: np.ndarray, shift: int, max_step: float):
    """Move mass from a source to its shift‑partner to reduce the worst lag."""
    n = h.shape[0]
    if shift >= 0:
        start, end = 0, n - shift
    else:
        start, end = -shift, n
    if end <= start:
        return None, 0.0
    idx_range = np.arange(start, end)
    cond = (h[idx_range] > 1e-12) & (h[idx_range + shift] < 1.0 - 1e-12)
    valid = idx_range[cond]
    if valid.size == 0:
        return None, 0.0
    i = np.random.choice(valid)
    allowed = min(h[i], 1.0 - h[i + shift], max_step)
    if allowed <= 1e-12:
        return None, 0.0
    delta = np.random.uniform(0.0, allowed)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[i + shift] += delta
    return new_h, delta


def random_move(h: np.ndarray, max_step: float) -> np.ndarray:
    """Swap a small amount of mass between two random positions."""
    n = h.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    lo = max(-h[i], h[j] - 1.0, -max_step)
    hi = min(1.0 - h[i], h[j], max_step)
    if lo >= hi:
        return h.copy()
    delta = np.random.uniform(lo, hi)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    # Clip tiny numerical overshoots and re‑balance sum if needed
    new_h = np.clip(new_h, 0.0, 1.0)
    target = n / 2.0
    if abs(new_h.sum() - target) > 1e-12:
        new_h = adjust_sum(new_h, target)
    return new_h


def perturb(h: np.ndarray, magnitude: float = 0.05) -> np.ndarray:
    """Add small uniform noise while preserving the constraints."""
    n = h.shape[0]
    noise = np.random.uniform(-magnitude, magnitude, size=n)
    new_h = np.clip(h + noise, 0.0, 1.0)
    new_h = adjust_sum(new_h, n / 2.0)
    return new_h


def optimize_h(initial: np.ndarray, time_budget: float):
    """Local stochastic optimisation for C5."""
    n = initial.shape[0]
    target_sum = n / 2.0
    cur = initial.copy()
    cur = np.clip(cur, 0.0, 1.0)
    cur = adjust_sum(cur, target_sum)
    cur_score = evaluate(cur)

    best = cur.copy()
    best_score = cur_score

    start = time.time()
    itr = 0
    max_step = 0.2
    T0 = 0.01  # temperature for occasional uphill moves

    while time.time() - start < time_budget:
        itr += 1
        # shrink step size slowly
        max_step = max(0.005, max_step * 0.99997)

        # correlation to locate the worst lag
        dx = 2.0 / n
        corr = np.correlate(cur, 1.0 - cur, mode='full') * dx
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)

        # try a directed move
        new_h, _ = attempt_targeted_move(cur, shift, max_step)
        if new_h is None:
            new_h = random_move(cur, max_step)

        new_score = evaluate(new_h)

        # Metropolis acceptance
        diff = new_score - cur_score
        if diff < 0.0 or np.random.rand() < math.exp(-diff / (T0 * (1.0 + itr * 1e-5))):
            cur = new_h
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best = new_h.copy()
                best_score = new_score
                # early exit if we beat the known record comfortably
                if best_score < 0.3808:
                    break

        # occasional restart from best with a jitter
        if itr % 5000 == 0:
            cur = perturb(best, magnitude=0.02)
            cur_score = evaluate(cur)

    return best, best_score


# ------------------ main driver -------------------------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points).
    h_values is a NumPy array sampled uniformly on [0,2] with length n_points.
    """
    np.random.seed(seed)
    start = time.time()

    # ---- obtain an initial construction --------------------------------
    try:
        init_h = initial_h_values  # provided by the harness
    except NameError:
        # fallback: uniform 0.5 on 45 points
        init_h = np.full(45, 0.5, dtype=float)

    # work at 90 points (even, convenient for binary‑type patterns)
    target_n = 90
    if init_h.shape[0] != target_n:
        h0 = upsample(init_h, target_n)
    else:
        h0 = init_h.astype(float).copy()
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = adjust_sum(h0, target_n / 2.0)

    # ---- allocate time --------------------------------------------------
    elapsed = time.time() - start
    remaining = max(0.0, budget_s - elapsed)
    # spend most of the remaining budget on optimisation
    opt_time = remaining * 0.95

    # ---- optimise -------------------------------------------------------
    best_h, best_score = optimize_h(h0, opt_time)

    # final sanity check (tiny numerical errors are tolerated)
    n = best_h.shape[0]
    dx = 2.0 / n
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12), "bounds violation"
    assert abs(best_h.sum() * dx - 1.0) < 1e-9, "integral constraint violated"

    return best_h, best_score, n
