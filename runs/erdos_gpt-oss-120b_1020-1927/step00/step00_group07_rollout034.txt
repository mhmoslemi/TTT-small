def evaluate(h):
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def project(v, target):
    # binary search for lambda
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2, 0.0, 1.0)

def gradient_shift(h, shift):
    N = len(h)
    g = np.zeros(N)
    if shift >= 0:
        g[:N - shift] += h[shift:]
        g[shift:] += h[:N - shift]
    else:
        s = -shift
        # shift negative => same as shift positive due to symmetry
        g[:N - s] += h[s:]
        g[s:] += h[:N - s]
    return g

def gradient_shift(h, shift):
    n = len(h)
    g = np.zeros(n, dtype=np.float64)
    if shift >= 0:
        # term h_i * h_{i+shift}
        g[:n-shift] += h[shift:]   # h_i+shift for i in [0,n-shift-1]
        g[shift:] += h[:n-shift]   # h_i-shift for i in [shift, n-1]
    else:
        s = -shift
        g[:n-s] += h[s:]
        g[s:] += h[:n-s]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time

    np.random.seed(seed)
    start = time.time()
    # Determine n_points
    try:
        h0 = initial_h_values.astype(np.float64)
        n = h0.size
        # ensure sum correct
        target_sum = n / 2.0
        # Due to tolerance, project
        def proj(v):
            # projection onto box [0,1] and sum=target_sum
            lo = -np.max(v) - 1.0
            hi = 1.0 - np.min(v) + 1.0
            for _ in range(50):
                mid = (lo + hi) / 2.0
                w = np.clip(v + mid, 0.0, 1.0)
                s = w.sum()
                if s < target_sum:
                    lo = mid
                else:
                    hi = mid
            return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
        h = proj(h0)
    except NameError:
        n = 200
        target_sum = n / 2.0
        h = np.zeros(n)
        idx = np.random.choice(n, int(target_sum), replace=False)
        h[idx] = 1.0
    dx = 2.0 / n
    target_sum = n / 2.0

    # evaluate function
    def evaluate(h):
        # max correlation with complement
        return np.max(np.correlate(h, 1 - h, mode='full') * dx)

    # initial evaluation
    cur_c = evaluate(h)
    best_h = h.copy()
    best_c = cur_c
    # parameters
    step = 0.5   # large as starting maybe
    iter_count = 0
    # track improvement
    last_improve_iter = 0

    # define projection function for later
    def project(v):
        lo = -np.max(v) - 1.0
        hi = 1.0 - np.min(v) + 1.0
        for _ in range(60):
            mid = (lo + hi) / 2.0
            w = np.clip(v + mid, 0.0, 1.0)
            s = w.sum()
            if s < target_sum:
                lo = mid
            else:
                hi = mid
        return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

    while time.time() - start < budget_s * 0.93:
        # compute a correlation (auto)
        a = np.correlate(h, h, mode='full') * dx
        a_min = a.min()
        # shift index of min
        idx_min = a.argmin()
        shift = idx_min - (n - 1)
        s = abs(shift)
        # compute gradient for this shift
        g = np.zeros(n, dtype=np.float64)
        if s > 0:
            g[:n - s] += h[s:]
            g[s:] += h[:n - s]
        else:
            g[:] = 2 * h  # shift zero
        # step update (increase product)
        new_h = h + step * g
        new_h = project(new_h)
        # evaluate
        new_c = evaluate(new_h)
        if new_c < cur_c:
            h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()
                last_improve_iter = iter_count
        else:
            # occasional random perturb to escape
            if np.random.rand() < 0.02:
                pert = np.random.uniform(-0.1, 0.1, size=n)
                new_h = project(h + pert)
                new_c = evaluate(new_h)
                if new_c < cur_c:
                    h = new_h
                    cur_c = new_c
                    if new_c < best_c:
                        best_c = new_c
                        best_h = new_h.copy()
                        last_improve_iter = iter_count
        # adapt step
        if iter_count % 500 == 0 and iter_count > 0:
            # if no improvement in last 2000 iters, reduce step
            if iter_count - last_improve_iter > 2000:
                step *= 0.5
                if step < 1e-4:
                    step = 1e-4
            else:
                # could increase step a bit
                step = min(step * 1.1, 2.0)
        iter_count += 1

    # final projection for best_h
    best_h = project(best_h)
    final_c = evaluate(best_h)
    return best_h, float(final_c), n

import numpy as np
import time

def _project_to_bounds_and_sum(v, target_sum):
    """
    Euclidean projection of v onto the set {x: 0<=x_i<=1 , sum(x)=target_sum}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _evaluate(h, dx):
    """return max_k ∫ h(x)(1-h(x+k))dx using linear correlation."""
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)


def _gradient_of_worst_shift(h, shift):
    """
    Gradient of the correlation at the worst shift.
    shift is an integer (may be negative).  Returns a vector of same size as h.
    The gradient of c_k = Σ h_i (1 - h_{i+shift}) (with zero padding)
    with respect to h is:
        g_i = (1 - h_{i+shift})  (if i+shift in range)
              - h_{i-shift}      (if i-shift in range)
    """
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        # forward part: i+shift in range
        g[: n - shift] += 1.0 - h[shift:]
        # backward part: i-shift in range
        g[shift:] -= h[: n - shift]
    else:
        s = -shift
        # forward part: i+shift = i-s in range  → i >= s
        g[s:] += 1.0 - h[: n - s]
        # backward part: i-shift = i+s in range → i <= n-s-1
        g[: n - s] -= h[s:]
    return g


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimise
    C5 = max_k ∫ h(x)(1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # initialise h
    # ----------------------------------------------------------------------
    try:
        # provided by the harness (if any)
        h = np.array(initial_h_values, dtype=np.float64)
        n = h.size
    except NameError:
        # fall back to a modest default size
        n = 200
        h = np.zeros(n, dtype=np.float64)

    target_sum = n / 2.0               # because ∫_0^2 h = 1  and dx = 2/n
    dx = 2.0 / n

    # ensure feasibility
    h = _project_to_bounds_and_sum(h, target_sum)

    # ----------------------------------------------------------------------
    # optimisation loop
    # ----------------------------------------------------------------------
    best_h = h.copy()
    best_c = _evaluate(best_h, dx)

    # parameters for the simple projected‑gradient descent
    step = 0.2
    it = 0
    last_improve = 0

    while time.time() - start_time < budget_s * 0.93:
        # compute correlation vector (full mode)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        cur_c = corr.max()
        # indices where the maximum is attained (might be a plateau)
        max_idxs = np.flatnonzero(corr == cur_c)
        # pick a random one among the maximal shifts to avoid stagnation
        idx = np.random.choice(max_idxs)
        shift = idx - (n - 1)

        # gradient of the worst shift
        g = _gradient_of_worst_shift(h, shift)

        # proposed update (descent direction)
        h_new = h - step * g
        h_new = _project_to_bounds_and_sum(h_new, target_sum)

        new_c = _evaluate(h_new, dx)

        # accept if we improve (or with a tiny probability to escape)
        if new_c < cur_c - 1e-9 or (np.random.rand() < 0.001):
            h = h_new
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h_new.copy()
                last_improve = it
        else:
            # occasional random jitter to break possible plateaus
            if np.random.rand() < 0.01:

                h_j = _project_to_bounds_and_sum(h + jitter, target_sum)
                c_j = _evaluate(h_j, dx)
                if c_j < cur_c:
                    h = h_j
                    cur_c = c_j
                    if c_j < best_c:
                        best_c = c_j
                        best_h = h_j.copy()
                        last_improve = it

    # final projection (just to be safe) and evaluation

    final_c = _evaluate(best_h, dx)

    return best_h, float(final_c), n
