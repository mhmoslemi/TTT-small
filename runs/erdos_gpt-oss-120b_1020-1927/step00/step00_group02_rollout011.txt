def grad_of_shift(h, L):
    n = len(h)
    g = np.zeros_like(h)
    # for indices where i+L in range
    if L >= 0:
        # i from 0 to n-1-L
        i = np.arange(0, n - L)
        g[i] += 1 - h[i + L]
        # second term: i from L to n-1
        i2 = np.arange(L, n)
        g[i2] += -h[i2 - L]

        i = np.arange(r, n)

        # second term: i from 0 to n-1-r
        i2 = np.arange(0, n - r)
        g[i2] += -h[i2 + r]

def grad_of_shift(h, shift):
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        # term1: i from 0 to n-1-shift
        i = np.arange(0, n - shift)
        g[i] += 1 - h[i + shift]
        # term2: i from shift to n-1
        i2 = np.arange(shift, n)
        g[i2] += -h[i2 - shift]
    else:
        r = -shift
        # term1: i from r to n-1
        i = np.arange(r, n)
        g[i] += 1 - h[i - r]
        # term2: i from 0 to n - r -1
        i2 = np.arange(0, n - r)
        g[i2] += -h[i2 + r]
    return g
```

import time, random, math
import numpy as np

# ---------- Helper Functions ----------

def evaluate_overlap(h):
    """Maximum overlap C5 for a step function h (dx = 2/len(h))."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def grad_of_shift(h, shift):
    """Gradient of the overlap for a single shift (lag)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        # term 1: i from 0 .. n-1-shift
        i1 = np.arange(0, n - shift)
        g[i1] += 1.0 - h[i1 + shift]
        # term 2: i from shift .. n-1
        i2 = np.arange(shift, n)
        g[i2] += -h[i2 - shift]
    else:
        r = -shift
        # term 1: i from r .. n-1
        i1 = np.arange(r, n)
        g[i1] += 1.0 - h[i1 - r]
        # term 2: i from 0 .. n-1-r
        i2 = np.arange(0, n - r)
        g[i2] += -h[i2 + r]
    return g


def project_to_feasible(y, target_sum):
    """
    Euclidean projection of y onto {x : 0<=x_i<=1, sum x_i = target_sum}.
    Uses binary search on the Lagrange multiplier λ in x_i = clip(y_i+λ,0,1).
    """
    lo = -np.max(y) - 1.0    # guarantees all clipped to 0 -> sum=0
    hi = 1.0 - np.min(y) + 1.0  # guarantees all clipped to 1 -> sum=n
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = np.sum(x)
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    λ = (lo + hi) * 0.5
    x = np.clip(y + λ, 0.0, 1.0)
    # tiny numerical correction (should be unnecessary in practice)
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        if np.any(free):
            x[free] += diff / np.sum(free)
            x = np.clip(x, 0.0, 1.0)
    return x


def random_feasible(n):
    """Uniform random point in the feasible set."""
    y = np.random.rand(n)
    return project_to_feasible(y, n / 2.0)


def upsample_to_target(arr, target_n):
    """Simple up‑sampling by integer repeat (truncates/extends as needed)."""
    n0 = arr.shape[0]
    if target_n == n0:
        return arr.copy()
    factor = max(1, target_n // n0)
    rep = np.repeat(arr, factor)
    if rep.shape[0] > target_n:
        rep = rep[:target_n]
    while rep.shape[0] < target_n:
        need = target_n - rep.shape[0]
        rep = np.append(rep, arr[:need])
    return rep


# ---------- Core Optimisation ----------

def subgradient_step_multi(h, step, top_m=3):
    """
    One sub‑gradient descent step using a sum of gradients from the top‑m
    overlapping shifts.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_m >= corr.size:
        idxs = np.arange(corr.size)
    else:
        # indices of the largest overlaps (unsorted)
        idxs = np.argpartition(-corr, top_m - 1)[:top_m]
    g = np.zeros_like(h)
    for idx in idxs:
        shift = idx - (n - 1)
        g += grad_of_shift(h, shift)
    # simple normalisation to keep step magnitude sensible
    norm = np.linalg.norm(g, ord=2)
    if norm > 0:
        g = g / norm
    h_new = h - step * g
    return project_to_feasible(h_new, n / 2.0)


def minimise_max_overlap(h0, time_limit, max_iters=10000, top_m=3):
    """Projected sub‑gradient descent on max overlap."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate_overlap(h)
    n = h.shape[0]
    it = 0
    while it < max_iters and (time.time() - start) < time_limit:
        it += 1
        # diminishing step size
        step = 0.5 / math.sqrt(it)
        h = subgradient_step_multi(h, step, top_m=top_m)
        val = evaluate_overlap(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        # early exit if already excellent
        if best_val <= 0.3808:
            break
    return best_h, best_val


def multistart_optimize(n, total_time, base_h=None):
    """Run several local searches (random restarts) within the time budget."""
    start = time.time()
    best_h = None
    best_val = np.inf

    # If a base solution is supplied, give it a short dedicated run
    if base_h is not None:
        portion = min(0.2 * total_time, 30.0)
        h_opt, val = minimise_max_overlap(base_h, portion)
        best_h, best_val = h_opt, val

    # Repeated random restarts
    while True:
        elapsed = time.time() - start
        if elapsed >= total_time * 0.95:
            break
        remaining = total_time - elapsed
        if remaining < 5.0:
            break
        # allocate a modest chunk per restart
        chunk = min(25.0, remaining * 0.5)
        h0 = random_feasible(n)
        h_opt, val = minimise_max_overlap(h0, chunk)
        if val < best_val:
            best_h, best_val = h_opt, val
        if best_val <= 0.3808:
            break
    return best_h, best_val
