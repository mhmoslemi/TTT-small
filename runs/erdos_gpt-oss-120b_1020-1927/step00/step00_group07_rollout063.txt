def project_box_simplex(v, target):
    # Find lambda such that x = np.clip(v - lam, 0, 1) satisfies sum(x) = target
    lo = -1e6
    hi = 1e6
    # adjust bounds: we can use sum of clipped values at lam=lo and lam=hi.
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

def optimize_h(h0, dx, max_iter=2000, alpha0=0.5):
    n = len(h0)
    target = n / 2
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    alpha = alpha0
    for it in range(max_iter):
        # compute correlation array
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        # compute gradient for shift
        shift = max_idx - (n - 1)
        grad = np.zeros_like(h)
        if shift >= 0:
            k = shift
            grad[:n - k] += 1 - h[k:]
            grad[k:] -= h[:n - k]
        else:
            k = -shift
            grad[k:] += 1 - h[:n - k]
            grad[:n - k] -= h[k:]
        # take step
        v = h - alpha * grad  # move opposite gradient to reduce max_corr (since gradient points up of f)

        # evaluate new objective
        cur_val = compute_c5(h, dx)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        # adapt step size
        if (it+1) % 200 == 0:
            alpha *= 0.5
    return best_h, best_val

import time
import numpy as np

def _c5_value(h, dx):
    """Scaled max correlation C5 for a given step array."""
    return dx * np.max(np.correlate(h, 1.0 - h, mode="full"))

def _project_box_simplex(v, target):
    """
    Project vector v onto the set {0 <= x_i <= 1, sum x_i = target}
    using a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _max_shift_and_corr(h):
    """Return (shift, maximum raw correlation) for the current h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (len(h) - 1)
    return shift, corr[idx]

def _gradient_for_shift(h, shift):
    """Sub‑gradient of the correlation term that attains the maximum."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        k = shift
        # forward contribution: (1 - h_{i+k})
        g[: n - k] += 1.0 - h[k:]
        # backward contribution: -h_{i-k}
        g[k:] -= h[: n - k]
    else:
        k = -shift
        # forward: (1 - h_{i-k})
        g[k:] += 1.0 - h[: n - k]
        # backward: -h_{i+k}
        g[: n - k] -= h[k:]
    return g

def _optimize_one_start(h0, dx, iters=3000, alpha0=1.0):
    """Projected sub‑gradient descent starting from h0."""
    n = h0.shape[0]
    target = n * 0.5
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_value(h, dx)

    for it in range(iters):
        shift, cur_corr = _max_shift_and_corr(h)
        grad = _gradient_for_shift(h, shift)
        # diminishing step size
        step = alpha0 / np.sqrt(it + 1.0)
        v = h - step * grad
        h = _project_box_simplex(v, target)

        val = _c5_value(h, dx)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        # occasional tiny random perturbation to escape plateaus
        if (it & 127) == 0:
            noise = np.random.normal(scale=0.005, size=n)
            h = _project_box_simplex(h + noise, target)

    return best_h, best_val

def run(seed=42, budget_s=1100, **_kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # determine problem size
    if 'initial_h_values' in globals():
        h_init = np.asarray(initial_h_values, dtype=float)
        n = h_init.shape[0]
        candidates = [h_init.copy()]
    else:
        n = 80
        candidates = []

    dx = 2.0 / n
    target_sum = n * 0.5

    # deterministic seeds: uniform, triangular, flipped triangular
    candidates.append(np.full(n, 0.5))
    pos = np.linspace(0.0, 1.0, n)
    tri = np.maximum(0.0, 1.0 - np.abs(2.0 * pos - 1.0))
    candidates.append(tri)
    candidates.append(tri[::-1])

    # random starts
    for _ in range(5):
        r = np.random.rand(n)
        candidates.append(_project_box_simplex(r, target_sum))

    best_h = None
    best_val = np.inf

    # main optimisation loop over candidates
    for h0 in candidates:
        if time.time() - start_time > budget_s * 0.92:
            break
        h_opt, val = _optimize_one_start(h0, dx, iters=2500, alpha0=1.0)
        if val < best_val:
            best_val, best_h = val, h_opt.copy()

    # a final refinement on the current best
    if best_h is not None and time.time() - start_time < budget_s * 0.85:
        h_opt, val = _optimize_one_start(best_h, dx, iters=5000, alpha0=0.5)
        if val < best_val:
            best_val, best_h = val, h_opt.copy()

    # ensure feasibility (tiny numerical drift may occur)
    best_h = _project_box_simplex(best_h, target_sum)

    c5_bound = _c5_value(best_h, dx)
    return best_h, float(c5_bound), n
