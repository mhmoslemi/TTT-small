def project_to_bounds_sum(x, target, lo=0.0, hi=1.0):
    # first clip to bounds? Actually we need to adjust by λ, then clip.
    # Use binary search.
    low = np.min(x) - 1.0
    high = np.max(x)  # such that all clipped to 0.
    for _ in range(60):
        lam = (low + high) / 2.0
        y = np.clip(x - lam, lo, hi)
        s = y.sum()
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(x - (low+high)/2.0, lo, hi)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random, math
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine N
    N = 200  # Might adjust if initial_h_values not None maybe length?
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        # Upsample to N
        x0 = np.linspace(0, 2, n0, endpoint=False) + (2/n0)/2  # maybe midpoints?
        x = np.linspace(0, 2, N, endpoint=False) + (2/N)/2
        h = np.interp(x, x0, h0)
    else:
        # simple initial guess: half ones near start
        h = np.concatenate([np.ones(N//2), np.zeros(N - N//2)])
    # Ensure feasible
    target_sum = N/2  # sum of h must equal N/2
    h = project_to_bounds_sum(h, target_sum)
    # evaluation function
    dx = 2.0 / N
    def compute_c5(h):
        return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    best_h = h.copy()
    best_score = compute_c5(best_h)
    # Simulated annealing
    # parameters
    max_iters = 2000000  # maybe limit by time
    T0 = 0.01
    T_end = 1e-5
    # schedule as exponential: T = T0 * (T_end/T0)^(t/max_iters)
    for t in range(max_iters):
        if t % 10000 == 0:
            # time check
            if time.time() - start_time > budget_s * 0.95:
                break
            # maybe adapt temperature
        # temperature
        T = T0 * ((T_end / T0) ** (t / max_iters))
        # randomly pick two indices
        i, j = rng.integers(N, size=2)
        while i == j:
            j = rng.integers(N)
        # compute maximum epsilon
        # delta positive means h[i] increase, h[j] decrease (preserve sum)
        max_inc = 1.0 - h[i]
        max_dec = h[j] - 0.0
        eps_max = min(max_inc, max_dec)
        max_inc2 = h[i] - 0.0
        max_dec2 = 1.0 - h[j]
        eps_min = -min(max_inc2, max_dec2)  # negative epsilon allowed (i decrease, j increase)
        if eps_max <= 0 and eps_min >= 0:
            # can't move, continue
            continue
        eps = rng.uniform(eps_min, eps_max)
        if eps == 0:
            continue
        new_h = h.copy()
        new_h[i] += eps
        new_h[j] -= eps
        # evaluate
        new_score = compute_c5(new_h)
        dE = new_score - best_score
        if dE < 0 or rng.random() < math.exp(-dE / (T + 1e-12)):
            # accept move
            h = new_h
            if new_score < best_score:

                best_h = new_h.copy()
    # At end, final best_h ensure feasible (project just in case)
    best_h = project_to_bounds_sum(best_h, target_sum)
    best_score = compute_c5(best_h)
    return best_h, best_score, N

import numpy as np, time, math

def _project_bounds_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of x onto the set {y : 0<=y_i<=1, sum(y)=target}.
    Uses a simple bisection on the Lagrange multiplier λ.
    """
    lo = np.min(x) - 1.0          # ensures all clip(x-λ)=1
    hi = np.max(x)                # ensures all clip(x-λ)=0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_of(h: np.ndarray) -> float:
    """max_k ∫ h(x)*(1-h(x+k)) dx  (discrete approximation)"""
    dx = 2.0 / h.size
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initial construction -------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        n0 = base.size
        # target number of points – a power of two for cheap FFT if ever needed
        N = 256 if n0 < 200 else 2 * n0
        # linear interpolation to N points
        xs0 = np.linspace(0.0, 2.0, n0, endpoint=False) + (2.0 / n0) * 0.5
        xs = np.linspace(0.0, 2.0, N, endpoint=False) + (2.0 / N) * 0.5
        h = np.interp(xs, xs0, base)
    else:
        # simple feasible start: half ones, half zeros
        N = 200
        h = np.concatenate([np.ones(N // 2), np.zeros(N - N // 2)])
    # enforce feasibility exactly
    target_sum = N / 2.0
    h = _project_bounds_sum(h, target_sum)

    # ---- optimisation parameters ---------------------------------------------
    dx = 2.0 / N
    best_h = h.copy()
    best_score = _c5_of(best_h)

    curr_h = best_h.copy()
    curr_score = best_score

    # simulated‑annealing schedule
    T0 = 0.02
    T_end = 1e-6
    max_iters = 2_000_000                     # will be cut by time limit

    # pre‑compute index masks for fast random moves
    zero_mask = curr_h == 0.0
    one_mask  = curr_h == 1.0

    for it in range(max_iters):
        # time‑check
        if (it & 1023) == 0:   # every 1024 iterations
            if time.time() - start > budget_s * 0.95:
                break

        # temperature (linear cooling)
        T = T0 + (T_end - T0) * (it / max_iters)

        # -------- propose a pairwise move preserving the sum -----------------
        inc_candidates = np.where(curr_h < 1.0)[0]
        dec_candidates = np.where(curr_h > 0.0)[0]
        if inc_candidates.size == 0 or dec_candidates.size == 0:
            continue
        i = rng.choice(inc_candidates)
        j = rng.choice(dec_candidates)
        # maximal allowed transfer from j to i
        max_transfer = min(1.0 - curr_h[i], curr_h[j])
        if max_transfer <= 0.0:
            continue
        # random step size (up to the full allowed amount)
        eps = rng.random() * max_transfer
        if eps == 0.0:
            continue
        new_h = curr_h.copy()
        new_h[i] += eps
        new_h[j] -= eps

        # -------- evaluate ---------------------------------------------------
        new_score = _c5_of(new_h)

        # -------- acceptance -------------------------------------------------
        dE = new_score - curr_score
        if dE < 0.0 or rng.random() < math.exp(-dE / (T + 1e-12)):
            curr_h = new_h
            curr_score = new_score
            # keep best overall

                best_h = new_h.copy()
