    # compute a, c
    a = (1 - b) / 2.0
    c = 1.0 - b

    # flat region
    mask_flat = (x >= a + b) & (x < a + b + c)

    mask_down = (x >= a + b + c) & (x < a + 2*b + c)

def normalize_h(h, target):
    cur = h.sum()
    if cur == 0:
        return h
    h = h * (target / cur)
    # clip
    h = np.clip(h, 0.0, 1.0)
    # adjust sum after clipping:
    diff = target - h.sum()
    # fix by adding diff to random feasible entries
    # iterate small steps:
    while abs(diff) > 1e-12:
        if diff > 0:
            # need increase
            idx = np.where(h < 1 - 1e-12)[0]
            if len(idx) == 0:

            i = np.random.choice(idx)

def local_search(h0, init_val, time_limit):

    dx = 2.0 / N
    h = h0.copy()
    cur_val = init_val
    best_h = h.copy()
    best_val = cur_val
    target = N/2

    # parameters
    max_iter = 10_000_000

            break
        # propose move
        # pick i with h[i] > eps, j with h[j] < 1-eps
        i_candidates = np.where(h > 1e-12)[0]
        j_candidates = np.where(h < 1 - 1e-12)[0]
        if len(i_candidates) == 0 or len(j_candidates) == 0:
            break
        i = np.random.choice(i_candidates)
        j = np.random.choice(j_candidates)
        if i == j:
            continue
        epsilon_max = min(h[i], 1.0 - h[j])
        if epsilon_max <= 0:
            continue
        # step size factor gradually decreasing
        factor = max(0.02, 1.0 - it / 500_000)  # linear schedule down to 0.02
        epsilon = epsilon_max * factor * np.random.rand()
        if epsilon <= 0:
            continue
        # perform move
        h[i] -= epsilon
        h[j] += epsilon
        new_val = max_overlap(h) * dx
        # accept if better, else maybe revert
        if new_val < cur_val:
            cur_val = new_val
            # keep move
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] += epsilon
            h[j] -= epsilon
        it += 1
    return best_h, best_val

import numpy as np, time, math

def _max_overlap(h: np.ndarray) -> float:
    """Return max_{lag} Σ h[i]·(1‑h[i+lag]) (no dx factor)."""
    # np.correlate implements linear cross‑correlation with zero padding.
    return np.max(np.correlate(h, 1.0 - h, mode='full'))

def _normalize(h: np.ndarray, target: float) -> np.ndarray:
    """Scale h to have sum == target and clip to [0,1]."""
    cur = h.sum()
    if cur == 0:
        return h
    h = h * (target / cur)
    h = np.clip(h, 0.0, 1.0)
    # Simple redistribution of the tiny remaining error.
    diff = target - h.sum()
    eps = 1e-12
    rnd = np.random.RandomState(0)
    while abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - eps)[0]
            if idx.size == 0:
                break
            i = rnd.choice(idx)
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            idx = np.where(h > eps)[0]
            if idx.size == 0:
                break
            i = rnd.choice(idx)
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    return h

def _shape_h(b: float, N: int) -> np.ndarray:
    """Symmetric step‑ramp shape with total area 1.
    b = width of each ramp (0 < b <= 1)."""
    a = (1.0 - b) / 2.0          # zero margin on each side
    c = 1.0 - b                  # length of the flat top (height 1)
    dx = 2.0 / N
    xs = (np.arange(N) + 0.5) * dx   # sample at cell centres
    h = np.zeros(N, dtype=np.float64)

    if b > 0:
        mask_up = (xs >= a) & (xs < a + b)
        h[mask_up] = (xs[mask_up] - a) / b
        mask_down = (xs >= a + b + c) & (xs < a + 2 * b + c)
        h[mask_down] = (a + 2 * b + c - xs[mask_down]) / b
    mask_flat = (xs >= a + b) & (xs < a + b + c)
    h[mask_flat] = 1.0
    return h

def _initial_from_parent(N: int) -> np.ndarray | None:
    """Upsample the global `initial_h_values` if it exists."""
    try:

    target = N / 2.0

def _local_search(h0: np.ndarray, init_val: float, time_budget: float):
    """Simple simulated‑annealing‑style coordinate moves."""
    N = h0.shape[0]
    dx = 2.0 / N
    target = N / 2.0
    # internal RNG – deterministic for reproducibility
    rng = np.random.RandomState(12345)

    h = h0.copy()
    cur_val = init_val
    best_h = h.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    eps = 1e-12
    # schedule: linear reduction of the typical step size
    while True:
        if time.time() - start > time_budget:
            break
        # pick i with h[i] > 0 and j with h[j] < 1
        i = rng.randint(N)
        while h[i] <= eps:
            i = rng.randint(N)
        j = rng.randint(N)
        while h[j] >= 1.0 - eps:
            j = rng.randint(N)
        if i == j:
            continue
        # maximal transferable amount respecting bounds
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= eps:
            continue
        # step size shrinks as iterations advance
        factor = max(0.02, 1.0 - it / 500_000)
        delta = delta_max * factor * rng.rand()
        if delta <= eps:
            continue

        # perform tentative move
        h[i] -= delta
        h[j] += delta

        new_val = _max_overlap(h) * dx
        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta

        it += 1

    # final safeguard – ensure the sum constraint is still satisfied
    best_h = _normalize(best_h, target)
    best_val = _max_overlap(best_h) * dx
    return best_h, best_val

        else:
            h_start = shape_h
            cur_val = shape_val
    else:
        h_start = shape_h
        cur_val = shape_val

    # 3) remaining time for a local stochastic refinement
    remaining = budget_s - (time.time() - start_total)
    if remaining > 1.0:
        best_h, best_val = _local_search(h_start, cur_val, remaining * 0.90)
    else:
        best_h = h_start
        best_val = cur_val

    # final evaluation (ensuring exact arithmetic)
    final_val = _max_overlap(best_h) * dx
    return best_h, final_val, N
