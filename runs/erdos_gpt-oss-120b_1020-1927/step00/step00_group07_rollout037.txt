import numpy as np
import time

def compute_max_corr(h):
    return np.max(np.correlate(h, 1 - h, mode='full'))

def project_to_sum_and_bounds(v, target, lo=0.0, hi=1.0, tol=1e-10, max_iter=100):
    # binary search for lambda
    low = np.min(v - hi) - 1.0  # ensure low enough
    high = np.max(v - lo) + 1.0  # ensure high enough
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        proj = np.clip(v - lam, lo, hi)
        s = np.sum(proj)
        if s > target:
            low = lam
        else:
            high = lam
    proj = np.clip(v - (low + high) / 2.0, lo, hi)
    # due to numeric errors may be off by small epsilon
    # adjust to match target if needed by adding epsilon to some entries not at bounds
    diff = target - np.sum(proj)
    if abs(diff) > 1e-8:
        # distribute diff over non-bound entries
        mask = (proj > lo + 1e-12) & (proj < hi - 1e-12)
        idx = np.where(mask)[0]
        if len(idx) > 0:
            # distribute equally
            proj[idx] += diff / len(idx)
            proj = np.clip(proj, lo, hi)
    return proj

import numpy as np
import time

def compute_max_corr(h):
    return np.max(np.correlate(h, 1 - h, mode='full'))

def project_to_sum_and_bounds(v, target, lo=0.0, hi=1.0, tol=1e-9, max_iter=100):
    # binary search for lambda
    lam_low = -np.max(v) - 1.0
    lam_high = np.max(v) - lo + 1.0
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(v - lam, lo, hi)
        s = np.sum(x)
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
    x = np.clip(v - (lam_low + lam_high) / 2.0, lo, hi)
    diff = target - np.sum(x)
    if abs(diff) > tol:
        mask = (x > lo + 1e-12) & (x < hi - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, lo, hi)
    return x

def gradient_descent(h, target, rng, max_iter=2000, alpha=30.0, lr=0.5):
    n = h.shape[0]
    for it in range(max_iter):
        # smooth max
        s = np.correlate(h, 1 - h, mode='full')
        max_s = np.max(s)
        exp_s = np.exp(alpha * (s - max_s))
        w = exp_s / np.sum(exp_s)  # weights for each shift
        # compute gradient
        grad = np.empty_like(h)
        # precompute for speed: get indices for each shift
        # We'll use Python loops
        for j in range(n):
            total = 0.0
            # iterate over shifts
            for idx, wk in enumerate(w):
                k = idx - (n - 1)
                i1 = j + k
                if 0 <= i1 < n:
                    total += wk * h[i1]
                i2 = j - k
                if 0 <= i2 < n:
                    total += wk * h[i2]
            grad[j] = 1.0 - total
        # gradient step
        h = h - lr * grad
        # project
        h = project_to_sum_and_bounds(h, target)
        # decay learning rate
        lr *= 0.9995
    return h

def local_refine(h, target, rng, steps=2000, init_step=0.2):
    n = len(h)
    cur_h = h.copy()
    cur_val = compute_max_corr(cur_h)
    step = init_step
    T = 0.2
    decay_T = 0.9995
    decay_step = 0.9995
    for i in range(steps):
        # pick indices
        i_up = rng.integers(n)
        i_dn = rng.integers(n)
        while i_dn == i_up:
            i_dn = rng.integers(n)
        # decide direction (increase i_up, decrease i_dn)
        max_up = min(step, 1.0 - cur_h[i_up])
        max_dn = min(step, cur_h[i_dn])
        max_transfer = min(max_up, max_dn)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        new_h = cur_h.copy()
        new_h[i_up] += delta
        new_h[i_dn] -= delta
        new_val = compute_max_corr(new_h)
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / (T + 1e-12))
            accept = rng.random() < prob
        if accept:
            cur_h = new_h
            cur_val = new_val
        if i % 100 == 0:
            T *= decay_T
            step *= decay_step
            if step < 1e-5:
                step = 1e-5
            if T < 1e-5:
                T = 1e-5
    return cur_h

        h0 = project_to_sum_and_bounds(v, target)
    else:

        h0 = project_to_sum_and_bounds(h0, target)
    best_h = h0.copy()
    best_val = compute_max_corr(best_h)
    # store best bound

    while time.time() - start < budget_s * 0.95:

        else:
            v = rng.random(n)
            h_start = project_to_sum_and_bounds(v, target)
        # gradient descent

        # local refinement
        h_opt = local_refine(h_opt, target, rng, steps=1500, init_step=0.2)
        val = compute_max_corr(h_opt)
        if val < best_val:

            best_h = h_opt.copy()
        # early stop if good enough
        if best_val * dx <= 0.3808:
            break
    # final projection to guarantee feasibility
    best_h = project_to_sum_and_bounds(best_h, target)

    return (best_h, float(final_val), n)

import numpy as np
import time

def compute_max_corr(h):
    """Maximum value of the correlation h * (1-h) (unscaled)."""
    return np.max(np.correlate(h, 1 - h, mode="full"))


def project_to_sum_and_bounds(v, target, lo=0.0, hi=1.0, tol=1e-9, max_iter=100):
    """
    Euclidean projection of v onto the hyperplane sum(x)=target with box bounds [lo,hi].
    """
    lam_low = -np.max(v) - 1.0
    lam_high = np.max(v) - lo + 1.0
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(v - lam, lo, hi)
        s = np.sum(x)
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
    x = np.clip(v - (lam_low + lam_high) / 2.0, lo, hi)
    diff = target - np.sum(x)
    if abs(diff) > tol:
        mask = (x > lo + 1e-12) & (x < hi - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt:
            x[mask] += diff / cnt
            x = np.clip(x, lo, hi)
    return x


def gradient_descent(h, target, rng, max_iter=2000, alpha=30.0, lr=0.5):
    """
    Smooth‑max gradient descent.
    The smoothed objective is
        F_alpha(h) = (1/alpha) log sum_k exp(alpha * s_k(h)),
    where s_k = correlation for shift k.
    """
    n = h.shape[0]
    for _ in range(max_iter):
        s = np.correlate(h, 1 - h, mode="full")
        mx = np.max(s)
        w = np.exp(alpha * (s - mx))
        w /= np.sum(w)                     # weights for each shift
        grad = np.empty_like(h)
        for j in range(n):
            tot = 0.0
            for idx, wk in enumerate(w):
                k = idx - (n - 1)
                i1 = j + k
                if 0 <= i1 < n:
                    tot += wk * h[i1]
                i2 = j - k
                if 0 <= i2 < n:
                    tot += wk * h[i2]
            grad[j] = 1.0 - tot
        h = h - lr * grad
        h = project_to_sum_and_bounds(h, target)
        lr *= 0.9995
    return h


def local_refine(h, target, rng, steps=2000, init_step=0.2):
    """
    Small‑move simulated‑annealing hill climbing to polish the solution.
    """
    n = len(h)
    cur = h.copy()
    cur_val = compute_max_corr(cur)
    step = init_step
    T = 0.2
    decay_T = 0.9995
    decay_step = 0.9995
    for i in range(steps):
        inc = rng.integers(n)
        dec = rng.integers(n)
        while dec == inc:
            dec = rng.integers(n)
        max_inc = min(step, 1.0 - cur[inc])
        max_dec = min(step, cur[dec])
        mx = min(max_inc, max_dec)
        if mx <= 0:
            continue
        delta = rng.random() * mx
        cand = cur.copy()
        cand[inc] += delta
        cand[dec] -= delta
        cand_val = compute_max_corr(cand)
        if cand_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(cand_val - cur_val) / (T + 1e-12))
            accept = rng.random() < prob
        if accept:
            cur = cand
            cur_val = cand_val
        if i % 100 == 0:
            T *= decay_T
            step *= decay_step
            if step < 1e-5:
                step = 1e-5
            if T < 1e-5:
                T = 1e-5
    return cur


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --- problem size -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        h0 = project_to_sum_and_bounds(h0, n / 2.0)
    else:
        n = 80
        h0 = project_to_sum_and_bounds(rng.random(n), n / 2.0)

    target = n / 2.0
    best_h = h0.copy()
    best_val = compute_max_corr(best_h)
    dx = 2.0 / n

    # --- main optimisation loop ---------------------------------------
    while time.time() - start < budget_s * 0.95:
        # start point: either current best or a fresh random vector
        if rng.random() < 0.3:
            start_vec = best_h.copy()
        else:
            start_vec = project_to_sum_and_bounds(rng.random(n), target)

        # smooth‑max gradient descent
        vec = gradient_descent(start_vec, target, rng,
                               max_iter=1500, alpha=30.0, lr=0.3)

        # local polishing
        vec = local_refine(vec, target, rng, steps=1500, init_step=0.2)

        val = compute_max_corr(vec)
        if val < best_val:
            best_val = val
            best_h = vec.copy()

        if best_val * dx <= 0.3808:          # reached target bound
            break

    # final projection and bound computation
    best_h = project_to_sum_and_bounds(best_h, target)
    final_bound = compute_max_corr(best_h) * dx
    return (best_h, float(final_bound), n)
