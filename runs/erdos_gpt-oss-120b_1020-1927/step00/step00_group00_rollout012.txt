            break

import time

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a given discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _adjust_sum(h: np.ndarray) -> np.ndarray:
    """Project onto the hyperplane sum(h)=n/2 while keeping 0≤h≤1."""
    n = h.shape[0]
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # try to fix the whole difference with a single entry that has room
    if diff > 0:  # need to increase
        idx = np.argmax(1.0 - h)  # most headroom
        add = min(diff, 1.0 - h[idx])
        h[idx] += add
        diff -= add
    else:  # need to decrease
        idx = np.argmax(h)       # largest value
        sub = min(-diff, h[idx])
        h[idx] -= sub
        diff += sub
    # if a tiny remainder remains, distribute it evenly (rare)
    if abs(diff) > 1e-12:
        # distribute over all entries that are not at a bound
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # fallback: spread over whole vector, then clip
            h += diff / n
            np.clip(h, 0.0, 1.0, out=h)
        else:
            chunk = diff / free.size
            for i in free:
                h[i] += chunk
            np.clip(h, 0.0, 1.0, out=h)
    # final tiny correction
    rem = target - h.sum()
    if abs(rem) > 1e-12:
        # push the correction onto a random free position

        if free.size:

            h[i] = np.clip(h[i] + rem, 0.0, 1.0)
    return h


def _simulated_annealing(start_h: np.ndarray, time_limit: float) -> tuple:
    """Simple SA that preserves the sum constraint by exchanging mass between two indices."""
    n = start_h.shape[0]
    best_h = start_h.copy()

    cur_h = best_h.copy()
    cur_score = best_score
    t0 = time.time()
    max_T = 0.05
    min_T = 1e-7

    while True:
        elapsed = time.time() - t0
        if elapsed >= time_limit:
            break
        # linear cooling
        T = max_T * max(0.0, 1.0 - elapsed / time_limit) + min_T

        i, j = np.random.randint(0, n, size=2)
        if i == j:

        # random direction of transfer
        if np.random.rand() < 0.5:
            i, j = j, i

        max_up = 1.0 - cur_h[i]
        max_down = cur_h[j]
        max_delta = max_up if max_up < max_down else max_down
        if max_delta <= 0.0:

        if np.random.rand() < 0.02:
            delta = max_delta
        else:
            delta = np.random.rand() * max_delta

        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_score = _compute_c5(new_h)

        if new_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0

        if accept:
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:

                best_score = new_score

    best_h = _adjust_sum(best_h)
    best_score = _compute_c5(best_h)
