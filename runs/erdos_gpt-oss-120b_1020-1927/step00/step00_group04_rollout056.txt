    v = np.clip(v, 0, 1)
    s = v.sum()
    # If sum matches target within tolerance, done
    if abs(s - target) < 1e-12:
        return v
    # Determine whether need to add or subtract
    if s < target:
        # Need to add delta = target - s across indices where v < 1
        # total capacity = sum(1 - v_i) over i where v_i < 1
        caps = 1 - v
        total_cap = caps.sum()
        if total_cap == 0:
            return v
        # distribute proportionally to caps (or equally)
        # simplest: add proportionally to caps
        v += caps / total_cap * (target - s)
    else:
        # Need to subtract from indices where v > 0
        caps = v
        total_cap = caps.sum()
        if total_cap == 0:
            return v
        v -= caps / total_cap * (s - target)
    # Clip again to ensure within bounds
    v = np.clip(v, 0, 1)
    return v

def compute_c5(h):
    # h is np.ndarray
    n = len(h)
    prod = np.convolve(h, h[::-1])
    # length 2n-1, index shift k corresponds to n-1 + k
    S = prod  # product sums per lag
    # compute corr = total_mass - product sum
    total = h.sum()
    # convert to C5 = max (corr) * dx
    corr = total - S
    # max over all k
    max_corr = np.max(corr)
    dx = 2.0/n
    return max_corr * dx
```

def compute_c5_fft(h):
    n = len(h)
    prod = np.fft.ifft(np.fft.fft(h) * np.fft.fft(h[::-1])).real

    total = h.sum()

import numpy as np
import time

def _project_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1]^n while enforcing sum(v)=target.
    The solution has the form clamp(v+λ,0,1) for some λ, found by bisection."""
    lo = -np.max(v)          # makes v+lo <= 0  → all zeros after clamp
    hi = 1.0 - np.min(v)     # makes v+hi >= 1  → all ones after clamp
    for _ in range(60):      # enough for double precision
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k))dx using FFT‑based linear convolution."""
    n = h.shape[0]
    # length for linear convolution (next power of two for speed)
    L = 1 << ((2 * n - 1).bit_length())

    H_rev = np.fft.rfft(h[::-1], L)
    prod = np.fft.irfft(H * H_rev, L)[: 2 * n - 1]   # linear convolution h * rev(h)
    min_prod = prod.min()
    total = h.sum()               # should be n/2
    dx = 2.0 / n
    return (total - min_prod) * dx

        h = 0.5 + 0.5 * np.cos(np.pi * x)      # values in [0,1]
    n = h.shape[0]
    target_sum = n / 2.0

    # enforce the linear constraints exactly
    h = np.clip(h, 0.0, 1.0)
    if not np.isclose(h.sum(), target_sum, atol=1e-12):
        h = _project_sum(h, target_sum)

    # ---- optimisation ------------------------------------------------------
    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    cur_h = h.copy()
    cur_c5 = best_c5

    start_time = time.time()
    max_time = budget_s * 0.95  # keep a small safety margin

    # temperature schedule (simulated‑annealing style)
    T_start, T_end = 0.04, 0.0005
    # step‑size schedule (max transferred mass per elementary move)
    step_start, step_end = 0.20, 0.01

    # counters for stagnation handling
    stagnation = 0
    STAGNATION_LIMIT = 20000

    while time.time() - start_time < max_time:
        elapsed = time.time() - start_time
        fraction = elapsed / max_time
        # linear schedules
        T = T_start * (1.0 - fraction) + T_end * fraction
        step = step_start * (1.0 - fraction) + step_end * fraction

        # ---- propose an elementary transfer move (preserves sum) ------------
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        if np.random.rand() < 0.5:
            # move from i → j
            max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0.0:
                continue
            delta = np.random.rand() * max_delta
            cur_h[i] -= delta
            cur_h[j] += delta
            direction = 1        # remember for possible rollback
        else:
            # move from j → i
            max_delta = min(step, cur_h[j], 1.0 - cur_h[i])
            if max_delta <= 0.0:
                continue
            delta = np.random.rand() * max_delta
            cur_h[j] -= delta
            cur_h[i] += delta
            direction = -1       # opposite direction

        new_c5 = _c5_value(cur_h)

        # ---- Metropolis acceptance -----------------------------------------
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_c5 - cur_c5) / T):
                accept = True

        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = cur_h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            # rollback the move
            if direction == 1:
                cur_h[i] += delta
                cur_h[j] -= delta
            else:
                cur_h[j] += delta
                cur_h[i] -= delta
            stagnation += 1

        # ---- escape deep stagnation ----------------------------------------
        if stagnation >= STAGNATION_LIMIT:
            # perturb the best solution a bit and re‑project
            noise = np.random.randn(n) * step * 0.5
            cur_h = best_h + noise
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = _project_sum(cur_h, target_sum)
            cur_c5 = _c5_value(cur_h)
            stagnation = 0

    # -----------------------------------------------------------------------
    # Return the best discovered construction
    return best_h, best_c5, best_h.shape[0]
