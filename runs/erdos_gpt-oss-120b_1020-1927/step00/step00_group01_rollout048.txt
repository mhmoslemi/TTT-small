def gradient_of_max(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    max_val = corr.max()
    idx = np.flatnonzero(corr == max_val)[0]  # first

    grad = np.zeros(n)
    lo = max(0, -k)
    hi = min(n-1, n-1-k)
    # Iterate i in [lo,hi]
    for i in range(lo, hi+1):
        grad[i] += 1.0
        j = i + k
        if 0 <= j < n:
            grad[i] -= h[j]

    # we can use algorithm: treat as solving w_i = min(upper, max(lower, v_i + lambda)), find lambda s.t sum w_i = target.

def project_to_sum_bounds(v, lower, upper, target):
    # Binary search for lambda
    lo = -1e4
    hi = 1e4
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v + lam, lower, upper)
        s = w.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    w = np.clip(v + (lo+hi)/2, lower, upper)
    return w

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    from scipy.optimize import minimize

    start = time.time()
    np.random.seed(seed)

    # Determine candidate n values
    n_candidates = [80, 100, 120, 160, 200]  # sorted maybe.
    # We can also include len(initial_h_values) if given.
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        if n0 not in n_candidates:
            n_candidates.insert(0, n0)

    best_h = None
    best_val = float('inf')
    best_n = None

    for n in n_candidates:
        # If time exceeded, break
        if time.time() - start > budget_s * 0.95:
            break
        dx = 2.0 / n

        # prepare list of initial guesses
        init_guesses = []

        # if provided initial_h_values
        if 'initial_h_values' in globals():
            # interpolate to length n
            hv = initial_h_values
            # Interpolate
            x_old = np.linspace(0, 2, len(hv), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, hv)
            # Project onto constraints
            h_proj = project_to_sum_bounds(h_interp, 0.0, 1.0, n/2)
            init_guesses.append(h_proj)

        # add random restarts
        num_random = 5 if n <= 120 else 3  # reduce for larger n
        for _ in range(num_random):
            v = np.random.rand(n)
            h_rand = project_to_sum_bounds(v, 0.0, 1.0, n/2)
            init_guesses.append(h_rand)

        for h0 in init_guesses:
            # Check budget
            if time.time() - start > budget_s * 0.95:
                break

            # define objective and gradient closures capturing n
            def obj(h):
                # return value (scalar)
                corr = np.correlate(h, 1-h, mode='full')
                return corr.max() * (2.0 / n)

            def jac(h):
                corr = np.correlate(h, 1-h, mode='full')
                maxval = corr.max()
                idx = np.flatnonzero(corr == maxval)[0]
                k = idx - (n - 1)
                grad = np.zeros_like(h)
                lo = max(0, -k)
                hi = min(n-1, n-1-k)
                for i in range(lo, hi+1):
                    grad[i] += 1.0
                    j = i + k
                    if 0 <= j < n:
                        grad[i] -= h[j]
                        grad[j] -= h[i]
                return grad * (2.0 / n)

            # equality constraint for sum
            cons = ({'type': 'eq', 'fun': lambda h, target=n/2: np.sum(h) - target,
                     'jac': lambda h, target=n/2: np.ones_like(h)},
                    )
            bounds = [(0.0, 1.0) for _ in range(n)]

            # Use SLSQP
            res = minimize(obj, h0, method='SLSQP', jac=jac,
                           bounds=bounds, constraints=cons,
                           options={'ftol': 1e-9, 'maxiter': 200, 'disp': False})

            h_opt = res.x
            # After optimization, enforce constraints again by projection
            h_opt = project_to_sum_bounds(h_opt, 0.0, 1.0, n/2)

            val = obj(h_opt)

            if val < best_val - 1e-9:
                best_val = val
                best_h = h_opt
                best_n = n
                # maybe print progress
                # print(f"New best: n={n}, val={val:.6f}")

    # If no solution found, fallback to initial_h_values or uniform.
    if best_h is None:
        if 'initial_h_values' in globals():
            h = initial_h_values
            n = len(h)
        else:
            n = 100
            h = np.full(n, 0.5)  # not feasible = sum = n/2? Actually sum=0.5*n so satisfies.
        best_h = h
        best_n = len(h)
        best_val = np.max(np.correlate(best_h, 1-best_h, mode='full')) * (2.0 / best_n)

    return best_h, best_val, best_n

import time
import numpy as np
from scipy.optimize import minimize

# ----- globals used by the optimizer -----
opt_n = None          # current problem size
opt_target_sum = None # n/2

# ----- helper functions -----
def _project_to_sum(v, lower, upper, target):
    """Project v onto [lower,upper] with sum constraint = target."""
    # binary search for lambda so that sum(clamp(v+lam)) = target
    lo, hi = -1e5, 1e5
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, lower, upper)
        s = w.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, lower, upper)


def _objective(h):
    """C5 value for the current global n."""
    n = opt_n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _gradient(h):
    """Sub‑gradient of the max‑correlation objective (uses first argmax)."""
    n = opt_n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    idx = np.flatnonzero(corr == max_val)[0]      # first occurrence
    k = idx - (n - 1)                            # shift amount
    grad = np.zeros_like(h)

    lo = max(0, -k)
    hi = min(n - 1, n - 1 - k)                    # inclusive
    for i in range(lo, hi + 1):
        grad[i] += 1.0
        j = i + k
        # both i and j are guaranteed inside [0,n)
        grad[i] -= h[j]          # derivative w.r.t. h[i] from term -h[i]*h[j]
        grad[j] -= h[i]          # derivative w.r.t. h[j] from the same term
    return grad * (2.0 / n)


def _eq_constraint(h):
    return np.sum(h) - opt_target_sum


def _eq_jac(_):
    return np.ones_like(_)


def _initial_guesses(n, seed):
    """Return a list of feasible starting vectors for size n."""
    rng = np.random.default_rng(seed)
    guesses = []

    # use the provided construction if available (interpolated if needed)
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.size != n:
            x_old = np.linspace(0.0, 2.0, base.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(x_new, x_old, base)
        else:
            interp = base.copy()
        guesses.append(_project_to_sum(interp, 0.0, 1.0, n / 2.0))

    # a handful of random feasible vectors
    rnd = 8 if n <= 120 else 4
    for _ in range(rnd):
        v = rng.random(n)
        guesses.append(_project_to_sum(v, 0.0, 1.0, n / 2.0))
    return guesses


def _refine(h, current_val, deadline):
    """Simple hill‑climbing refinement with random swaps."""
    n = h.size
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a small amount from i to j (preserving bounds)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        if delta < 1e-8:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        val = _objective(h_new)
        if val + 1e-12 < current_val:
            h, current_val = h_new, val
    return h, current_val

    global opt_n, opt_target_sum
