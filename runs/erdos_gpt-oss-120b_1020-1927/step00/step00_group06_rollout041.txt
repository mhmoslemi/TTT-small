def project_box_simplex(y, S):
    # y: np.array shape (n,)
    # returns x satisfying 0 <= x <= 1, sum = S
    # use binary search on lambda
    lo = -np.max(y) - 1  # ensure low enough that all clipped to 1
    hi = np.max(y)       # high enough that all clipped to 0

    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0, 1)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    # final x at hi
    x = np.clip(y - hi, 0, 1)
    # may not sum exactly due to numeric; adjust tiny diff
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        # distribute diff uniformly among indices not at bounds?
        # simple fix: add diff / n to all and clip again, repeat a few times
        # but diff will be very small; just adjust one element within bounds
        idx = np.where((x > 0) & (x < 1))[0]
        if len(idx) > 0:
            x[idx[0]] += diff
            x = np.clip(x, 0, 1)
        else:
            # all at bounds, cannot adjust; but diff should be zero
            pass
    return x

import time
import math
import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a binary (or [0,1]) vector."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) over all integer shifts
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max() * dx)


def _binary_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.float64)
    h[: n // 2] = 1.0
    rng.shuffle(h)
    return h

    """
    Find a pair (i,j) that improves the given shift (if any) and return a new vector.
    Returns (new_h, True) if a swap was performed, otherwise (h, False).
    """
    n = h.shape[0]
    if shift == 0:
        return h, False

    if shift > 0:
        i_range = np.arange(n - shift)
        hi = h[i_range]
        h_shift = h[i_range + shift]
    else:  # shift < 0
        s = -shift
        i_range = np.arange(s, n)
        hi = h[i_range]
        h_shift = h[i_range - s]

    bad1 = i_range[(hi == 1) & (h_shift == 0)]   # 1 followed by 0
    bad2 = i_range[(hi == 0) & (h_shift == 1)]   # 0 followed by 1
    if bad1.size == 0 or bad2.size == 0:
        return h, False

    i = int(rng.choice(bad1))
    j = int(rng.choice(bad2))
    new_h = h.copy()
    new_h[i] = 0.0
    new_h[j] = 1.0
    return new_h, True


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h defined on [0,2] (discretised) that minimises
    the Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Length‑n array with entries in [0,1] and sum equal to n/2.
    c5_bound : float
        The achieved upper bound on C5.
    n_points : int
        Number of discretisation points used.
    """
    rng = np.random.default_rng(seed)

    # ---- Problem size -------------------------------------------------
    n = 200                     # even to allow a binary representation
    target_ones = n // 2        # exactly half of the points are 1

    # ---- Initial solution ---------------------------------------------
    # If a global `initial_h_values` is provided we try to reuse it,
    # otherwise we start from a random binary vector.
    try:
        # the harness may provide an initial array; we convert it to binary
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.shape[0] != n:
            # If the supplied size differs we ignore it – a fresh start is fine.
            raise ValueError
        # Threshold at 0.5, then adjust the count to match target_ones.
        h = (init >= 0.5).astype(np.float64)
        cur = int(h.sum())
        if cur > target_ones:
            # turn some 1's into 0's
            idx = np.where(h == 1)[0]
            to_zero = rng.choice(idx, cur - target_ones, replace=False)
            h[to_zero] = 0.0
        elif cur < target_ones:
            idx = np.where(h == 0)[0]
            to_one = rng.choice(idx, target_ones - cur, replace=False)
            h[to_one] = 1.0
    except Exception:
        # No suitable initial vector – generate a random binary one.
        h = _binary_initial(n, rng)

    best_h = h.copy()
    best_c5 = _c5_of(best_h)

    start_time = time.time()
    deadline = start_time + budget_s * 0.95   # leave a safety margin

    # Simulated‑annealing parameters
    temp_initial = 0.02
    temp_min = 5e-4

    # Main optimisation loop
    while time.time() < deadline:
        # Current worst shift
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)

        # Try a targeted swap that specifically helps the worst shift
        new_h, performed = _targeted_swap(best_h, shift, rng)

        # If no targeted move was possible, fall back to a random 1↔0 swap
        if not performed:
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            new_h = best_h.copy()
            new_h[i] = 0.0
            new_h[j] = 1.0

        new_c5 = _c5_of(new_h)
        delta = new_c5 - best_c5

        # Acceptance rule (simulated annealing)
        elapsed = time.time() - start_time
        progress = elapsed / (deadline - start_time)
        temperature = max(temp_min, temp_initial * (1.0 - progress))
        if delta <= 0.0 or rng.random() < math.exp(-delta / temperature):
            best_h = new_h
            best_c5 = new_c5

    # Ensure the returned vector satisfies the constraints exactly (tiny rounding)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Enforce exact sum = n/2 by a tiny redistribution if needed
    current_sum = best_h.sum()
    diff = target_ones - current_sum
    if abs(diff) > 1e-12:
        # adjust a single element that is not at a bound
        free = np.where((best_h > 0) & (best_h < 1))[0]
        if free.size:
            best_h[free[0]] += diff
            best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = _c5_of(best_h)
    return best_h, final_c5, n
