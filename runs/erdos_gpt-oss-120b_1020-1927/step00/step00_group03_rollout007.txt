def hill_climb(h, max_iters, time_limit):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    N = len(h)
    start = time.time()
    it = 0
    while it < max_iters and time.time() - start < time_limit:
        # random swap of a 1 and 0
        ones = np.where(best_h > 0.5)[0] # or exactly 1
        zeros = np.where(best_h < 0.5)[0]
        i = random.choice(ones)
        j = random.choice(zeros)
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        score = compute_c5(new_h)
        if score < best_score - 1e-8: # improvement
            best_h = new_h
            best_score = score
        it += 1
    return best_h, best_score
```

import time, math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    rng.shuffle(h)
    return h


def enforce_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry so that the sum equals target (keeps bounds)."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # try to fix the first element, otherwise the first non‑boundary element
    for idx in range(h.shape[0]):
        new_val = h[idx] + diff
        if 0.0 <= new_val <= 1.0:
            h[idx] = new_val
            return h
    # fallback: clip and redistribute the tiny residue
    h = np.clip(h, 0.0, 1.0)
    residue = target - h.sum()
    if abs(residue) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))  # a middle point
        h[idx] = min(max(h[idx] + residue, 0.0), 1.0)
    return h


def hill_climb(start_h: np.ndarray, time_limit: float,
               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy local search: move mass between two positions if it improves C5."""
    h = start_h.copy()
    n = h.shape[0]
    cur_score = compute_c5(h)
    start = time.time()
    # number of random trials per improvement step
    trials_per_step = max(30, n)

    while time.time() - start < time_limit:
        improved = False
        for _ in range(trials_per_step):
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            # feasible delta so that 0 ≤ h ≤ 1 after the move
            lo = -min(h[i], 1.0 - h[j])
            hi = min(1.0 - h[i], h[j])
            if hi <= lo:
                continue
            # pick a random delta inside the feasible interval
            delta = rng.uniform(lo, hi)
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            new_score = compute_c5(new_h)
            if new_score < cur_score - 1e-12:
                h = new_h
                cur_score = new_score
                improved = True
                break
        if not improved:
            break
    return h, cur_score


def optimize(initial_h: np.ndarray, total_budget: float,
             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Repeated random restarts + hill climbing within the time budget."""
    n = initial_h.shape[0]
    target = n / 2.0
    best_h = enforce_sum(initial_h.copy(), target)
    best_score = compute_c5(best_h)

    start = time.time()
    # allocate at most 0.5 s per restart (adjusted later)
    min_slice = 0.3
    while time.time() - start < total_budget:
        remaining = total_budget - (time.time() - start)
        slice_time = min(max(min_slice, remaining * 0.1), remaining)
        # random starting point: either a perturbed best or fresh binary
        if rng.random() < 0.5:
            start_h = best_h.copy()
            # apply a few random feasible moves to diversify
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                lo = -min(start_h[i], 1.0 - start_h[j])
                hi = min(1.0 - start_h[i], start_h[j])
                if hi <= lo:
                    continue
                delta = rng.uniform(lo, hi)
                start_h[i] += delta
                start_h[j] -= delta
        else:
            start_h = random_binary(n, rng)

        cand_h, cand_score = hill_climb(start_h, slice_time, rng)
        if cand_score < best_score:
            best_h, best_score = cand_h, cand_score
            if best_score <= 0.3808:
                break
    return best_h, best_score
