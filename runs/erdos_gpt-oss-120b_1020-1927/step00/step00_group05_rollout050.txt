def project_capped_simplex(v, S):
    # binary search lambda
    lower = v.min() - 1.0  # maybe -2.0
    upper = v.max()
    for _ in range(50):
        lam = (lower + upper) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > S:
            lower = lam
        else:
            upper = lam
    return np.clip(v - lam, 0, 1)

import numpy as np
import time
from scipy.optimize import minimize

def project_onto_capped_simplex(v, total):
    """Project onto {x: 0<=x<=1, sum(x)=total}."""
    lo = v.min() - 1.0   # ensures all clips to 1 → sum=n
    hi = v.max()        # ensures all clips to 0 → sum=0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > total:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = total - x.sum()
    if abs(diff) > 1e-10:
        mask = (x > 0) & (x < 1)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
    return x

def max_overlap(h, dx):
    """C₅ for a discrete step function."""
    return float(np.max(np.correlate(h, 1 - h, mode='full') * dx))

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.size
    else:
        n = 70
        h0 = np.zeros(n, dtype=np.float64)
        h0[rng.choice(n, size=n // 2, replace=False)] = 1.0
    total = n / 2.0                     # integral constraint (∑h = n/2)
    dx = 2.0 / n
    h = project_onto_capped_simplex(h0, total)

    best_h = h.copy()
    best_c5 = max_overlap(best_h, dx)

    start = time.time()
    # ------------------------------------------------------------------
    # Phase 1 : smooth‑max gradient descent with projection
    T0 = 0.02
    T_min = 1e-5
    T_decay = 0.9995
    lr = 0.2
    lr_min = 1e-4
    lr_decay = 0.9995
    it = 0
    phase1_deadline = start + budget_s * 0.65

    while time.time() < phase1_deadline:
        # raw correlation (dx factor omitted – does not affect minimiser)
        v_raw = np.correlate(h, 1 - h, mode='full')
        temp = max(T_min, T0 * (T_decay ** it))
        vmax = v_raw.max()
        exp_vals = np.exp((v_raw - vmax) / temp)
        w = exp_vals / exp_vals.sum()          # soft‑max weights, sum=1

        # gradient of the smooth max (base term = 1)
        grad = np.ones_like(h)
        N = n
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            shift = idx - (N - 1)
            if shift == 0:
                grad -= 2.0 * weight * h
            elif shift > 0:

                grad[shift:]    -= weight * h[:N - shift]
            else:  # shift < 0

                grad[s:]       -= weight * h[:N - s]
                grad[:N - s]   -= weight * h[s:]

        # projected step
        h = project_onto_capped_simplex(h - lr * grad, total)

        cur_c5 = max_overlap(h, dx)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        # schedules
        lr = max(lr_min, lr * lr_decay)
        it += 1

        # occasional jitter / reset to escape plateaus
        if it % 2000 == 0:

        if it % 5000 == 0:

                best_c5 = cand_c5

    if remaining > 5.0:

                    grad[:N - s]   -= weight * x[s:]
            grad += 2.0 * mu * diff
            return smooth, grad

        res = minimize(fun=lambda y: obj_and_grad(y)[0],
                       x0=best_h,
                       jac=lambda y: obj_and_grad(y)[1],
                       method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxfun': 2000, 'maxiter': 2000, 'ftol': 1e-12})

            refined_c5 = max_overlap(refined, dx)
