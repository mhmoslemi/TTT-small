import numpy as np, time, math

def project_onto_box_simplex(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    h_fft = np.fft.fft(h)
    dot = np.real(np.fft.ifft(h_fft * np.conj(h_fft)))
    f = 1.0 - dx * dot
    return float(np.max(f)), f

def optimize_h(h0, iters, lr, alpha, target_sum, time_limit):
    import time, numpy as np
    start = time.time()
    h = np.clip(h0, 0.0, 1.0)
    h = project_onto_box_simplex(h, target_sum)
    best_h = h.copy()
    best_c5, _ = compute_c5(h)
    n = len(h)
    dx = 2.0 / n
    # Parameter schedule
    cur_alpha = alpha
    for i in range(iters):
        if time.time() - start > time_limit:
            break
        # Compute correlation and f
        h_fft = np.fft.fft(h)
        dot = np.real(np.fft.ifft(h_fft * np.conj(h_fft)))
        f = 1.0 - dx * dot
        # compute softmax weights
        max_f = np.max(f)
        exp_arg = cur_alpha * (f - max_f)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / n
        else:
            w /= w_sum
        # compute gradient using FFT
        w_rev = np.empty_like(w)
        w_rev[0] = w[0]
        if n > 1:
            w_rev[1:] = w[:0:-1]
        # Convolutions
        A = np.real(np.fft.ifft(np.fft.fft(w_rev) * h_fft))
        B = np.real(np.fft.ifft(np.fft.fft(w) * h_fft))

import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the box [0,1] with a fixed sum.
# ------------------------------------------------------------
def _proj_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {x | 0<=x<=1, sum x = target} w.r.t. Euclidean norm."""
    lo = np.min(v) - 1.0      # ensures all entries clip to 1 -> sum=n
    hi = np.max(v)            # ensures all entries clip to 0 -> sum=0
    for _ in range(60):       # enough for double precision
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


# ------------------------------------------------------------
# Compute the Erdős overlap constant C5 for a given step function.
# ------------------------------------------------------------
def _c5_value(h: np.ndarray) -> (float, np.ndarray):
    n = h.shape[0]
    dx = 2.0 / n
    hf = np.fft.fft(h)
    dot = np.real(np.fft.ifft(hf * np.conj(hf)))          # h_i·h_{i+k}
    f = 1.0 - dx * dot                                   # overlap for each shift
    return float(np.max(f)), f


# ------------------------------------------------------------
# Core optimizer: projected (soft‑max) gradient descent.
# ------------------------------------------------------------
def _optimize(h0: np.ndarray,
              iters: int,
              lr: float,
              alpha_start: float,
              target_sum: float,
              time_limit: float) -> (np.ndarray, float):
    start = time.time()
    h = np.clip(h0, 0.0, 1.0)
    h = _proj_box_sum(h, target_sum)

    best_h = h.copy()
    best_c5, _ = _c5_value(h)

    n = h.shape[0]
    dx = 2.0 / n
    alpha = alpha_start

    for it in range(iters):
        if time.time() - start > time_limit:
            break

        # ----------- evaluate current f ----------
        hf = np.fft.fft(h)
        dot = np.real(np.fft.ifft(hf * np.conj(hf)))
        f = 1.0 - dx * dot

        # ----------- soft‑max weights ------------
        m = np.max(f)
        w = np.exp(alpha * (f - m))
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / n)

        # ----------- gradient (FFT based) -------
        # w_rev corresponds to the kernel for the (+k) shift
        w_rev = np.empty_like(w)
        w_rev[0] = w[0]
        if n > 1:
            w_rev[1:] = w[:0:-1]                     # [w0, w_{n-1},...,w1]

        A = np.real(np.fft.ifft(np.fft.fft(w_rev) * hf))   # Σ w_k h_{i+k}
        B = np.real(np.fft.ifft(np.fft.fft(w) * hf))       # Σ w_k h_{i‑k}
        grad = 1.0 - (A + B)

        # ----------- step & projection -----------
        h_new = h - lr * grad
        h = _proj_box_sum(h_new, target_sum)

        # ----------- keep the best ---------------
        c5_val, _ = _c5_value(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()

        # ----------- schedule --------------------
        if (it + 1) % 100 == 0:
            alpha = min(alpha * 1.2, 80.0)          # sharpen the max
            # optional: slowly decay lr (not mandatory)

    return best_h, best_c5

# Main entry point expected by the evaluation harness.

    start_total = time.time()

    target_sum = n / 2.0

    # (b) smooth sinusoidal start (already satisfies the sum)

        best_h = _proj_box_sum(best_h, target_sum)
        best_c5, _ = _c5_value(best_h)

    # ensure constraints are exactly satisfied (tiny numeric drift)
    best_h = _proj_box_sum(best_h, target_sum)

    return (best_h.tolist(), float(best_c5), n)
