    n = len(h)
    # compute O via correlation
    O = np.correlate(h, 1 - h, mode='full')
    # compute softmax weights
    expa = np.exp(alpha * O - np.max(alpha * O))  # stability
    sum_expa = expa.sum()
    w = expa / sum_expa
    # iterate over shifts
    grad = np.zeros_like(h)
    center = n - 1
    for d in range(-(n-1), n):
        w_d = w[center + d]
        if w_d == 0:
            continue
        # term (1 - h[i+d]) for i where i+d in range
        if d >= 0:

            grad[:n-d] += w_d * (1 - h[d:])

            grad[:n-d_abs] -= w_d * h[d_abs:]

import math
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper: apply a swap (or value exchange) of two positions in h
# and update the full correlation array C = corr(h, 1-h, mode='full')
# ----------------------------------------------------------------------
def apply_swap(h: np.ndarray, C: np.ndarray, a: int, b: int) -> None:
    """Swap values h[a] and h[b] and update correlation C in‑place."""
    n = h.shape[0]
    centre = n - 1
    va_old = h[a]
    vb_old = h[b]
    if va_old == vb_old:                     # nothing changes
        return
    # new values after swap
    va_new = vb_old
    vb_new = va_old
    # update contributions for all i ≠ a,b
    for i in range(n):
        if i == a or i == b:
            continue
        hi = h[i]

        # (i , a)   and   (i , b)
        d = a - i
        C[centre + d] += hi * (1 - va_new) - hi * (1 - va_old)
        d = b - i
        C[centre + d] += hi * (1 - vb_new) - hi * (1 - vb_old)

        # (a , i)   and   (b , i)
        d = i - a
        C[centre + d] += va_new * (1 - hi) - va_old * (1 - hi)
        d = i - b
        C[centre + d] += vb_new * (1 - hi) - vb_old * (1 - hi)

    # pair (a , b)   and   (b , a)
    d = b - a
    C[centre + d] += va_new * (1 - vb_new) - va_old * (1 - vb_old)
    d = a - b
    C[centre + d] += vb_new * (1 - va_new) - vb_old * (1 - va_old)

    # finally swap the values
    h[a] = va_new
    h[b] = vb_new


# ----------------------------------------------------------------------
# Objective for continuous optimisation (soft‑max of the overlaps)
# ----------------------------------------------------------------------
class ErdosObjective:
    def __init__(self, n: int, target_sum: float,
                 alpha: float = 20.0, lam: float = 1000.0):
        self.n = n
        self.target_sum = target_sum
        self.alpha = alpha
        self.lam = lam
        self.centre = n - 1

    def __call__(self, x: np.ndarray):
        # overlap for every shift
        O = np.correlate(x, 1.0 - x, mode='full')
        a = self.alpha
        # stable soft‑max
        aO = a * O
        m = aO.max()
        expa = np.exp(aO - m)
        sum_expa = expa.sum()
        w = expa / sum_expa               # soft‑max weights
        # smooth approximation of the maximum
        smooth_max = (m + np.log(sum_expa)) / a
        # quadratic penalty for the integral constraint
        diff = x.sum() - self.target_sum
        penalty = self.lam * diff * diff
        f = smooth_max + penalty

        # gradient of the soft‑max part
        grad = np.zeros_like(x)
        n = self.n
        c = self.centre
        for d in range(-(n - 1), n):
            wd = w[c + d]
            if wd == 0.0:
                continue
            if d >= 0:
                # (1‑h[i+d]) term
                grad[:n - d] += wd * (1.0 - x[d:])
                # –h[i‑d] term
                grad[d:] -= wd * x[:n - d]
            else:
                dabs = -d
                grad[dabs:] += wd * (1.0 - x[:n - dabs])
                grad[:n - dabs] -= wd * x[dabs:]

        # gradient of the penalty
        grad += 2.0 * self.lam * diff
        return f, grad


def erdos_fun(x, obj):
    return obj(x)[0]


def erdos_grad(x, obj):
    return obj(x)[1]

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------------
    # Choose a working resolution (even, power‑of‑two, not too large)
    # --------------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n0 = init.size
        n = 1
        while n < max(n0, 256):
            n <<= 1
        if n == n0:
            h = init.copy()
        else:
            factor = n // n0
            h = np.repeat(init, factor)
    else:
        n = 512                      # default size
        h = np.zeros(n, dtype=float)

    # force binary with exactly n/2 ones (feasible for the original problem)
    half = n // 2
    if not np.all((h == 0) | (h == 1)):
        h[:] = 0.0
        idx = np.arange(n)
        rng.shuffle(idx)
        h[idx[:half]] = 1.0
    else:
        # ensure exact count if something went wrong
        cur = int(h.sum())
        if cur != half:
            h[:] = 0.0
            idx = np.arange(n)
            rng.shuffle(idx)
            h[idx[:half]] = 1.0

    # --------------------------------------------------------------
    # Continuous optimisation (soft‑max L‑BFGS‑B)
    # --------------------------------------------------------------
    target_sum = half
    best_h = h.copy()
    best_c5 = (2.0 / n) * np.max(np.correlate(best_h, 1 - best_h, mode='full'))

    # parameters for the optimisation
    alpha = 25.0                     # sharper soft‑max
    lam = 2000.0                     # penalty weight
    max_restarts = 8
    max_iter_per = 180

    for _ in range(max_restarts):
        # random feasible start (Dirichlet → half‑sum, clipped)
        raw = rng.gamma(1.0, 1.0, size=n)
        x0 = raw / raw.sum() * target_sum
        # Clip into the box; the penalty will pull the sum back
        np.clip(x0, 0.0, 1.0, out=x0)

        obj = ErdosObjective(n, target_sum, alpha=alpha, lam=lam)

        res = minimize(
            fun=erdos_fun,
            x0=x0,
            args=(obj,),
            jac=erdos_grad,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': max_iter_per, 'ftol': 1e-14, 'disp': False}
        )
        cand = np.clip(res.x, 0.0, 1.0)

        # tiny projection onto the exact integral plane (iterative)
        diff = cand.sum() - target_sum
        if abs(diff) > 1e-8:
            # distribute the excess uniformly among interior points
            free = (cand > 0.0) & (cand < 1.0)
            if free.any():
                adjustment = diff / free.sum()
                cand[free] -= adjustment
                np.clip(cand, 0.0, 1.0, out=cand)

        c5 = (2.0 / n) * np.max(np.correlate(cand, 1 - cand, mode='full'))
        if c5 < best_c5:
            best_c5 = c5
            best_h = cand.copy()
            if best_c5 <= 0.3808:
                break

        # time guard
        if time.time() - start_time > budget_s * 0.90:
            break

    # --------------------------------------------------------------
    # Optional short greedy hill‑climbing on the best binary‑ish vector
    # --------------------------------------------------------------
    # Turn the current best into a strict binary vector (helps the swap loop)
    bin_h = (best_h > 0.5).astype(float)
    bin_h[:] = 0.0
    idx = np.arange(n)
    rng.shuffle(idx)
    bin_h[idx[:half]] = 1.0
    C = np.correlate(bin_h, 1 - bin_h, mode='full')
    cur_c5 = (2.0 / n) * C.max()
    best_c5 = cur_c5
    best_h = bin_h.copy()

    # simple simulated‑annealing style swap search
    T0 = 0.018
    it = 0
    while True:
        if time.time() - start_time > budget_s * 0.96:
            break
        a = rng.integers(n)
        b = rng.integers(n - 1)
        if b >= a:
            b += 1
        if bin_h[a] == bin_h[b]:
            continue

        apply_swap(bin_h, C, a, b)                # perform swap
        new_c5 = (2.0 / n) * C.max()
        delta = new_c5 - cur_c5
        # linear cooling
        frac = (time.time() - start_time) / (budget_s * 0.96)
        T = T0 * (1.0 - frac)
        if delta <= 0.0 or (T > 1e-12 and rng.random() < math.exp(-delta / T)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = bin_h.copy()
                if best_c5 <= 0.3808:
                    break
        else:
            # revert
            apply_swap(bin_h, C, a, b)
        it += 1
        if it % 50000 == 0 and best_c5 <= 0.3808:
            break

    # --------------------------------------------------------------
    # Final clean‑up, guarantee constraints within tolerance
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    total = best_h.sum()
    if abs(total - target_sum) > 1e-6:
        # simple scaling (rare)
        best_h *= target_sum / total
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = (2.0 / n) * np.max(np.correlate(best_h, 1 - best_h, mode='full'))

    return best_h.tolist(), float(final_c5), n
