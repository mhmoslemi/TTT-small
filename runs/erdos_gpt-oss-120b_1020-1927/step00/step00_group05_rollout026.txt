def gradient_max_shift(h, shift):
    n = len(h)
    grad = np.zeros(n)
    if shift >=0:
        # forward contribution for indices where j+shift < n:
        grad[:n-shift] += (1 - h[shift:])
        # backward contribution for indices where j-shift >=0:
        grad[shift:] -= h[:n-shift]
    else:
        s = -shift

import time, math
import numpy as np

def _project_box_sum(y: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of y onto the set
        {x : 0 <= x_i <= 1,  sum(x) == target }.
    Implemented by binary search for the Lagrange multiplier λ.
    """
    lo = -np.max(y)          # makes y+λ <= 0 for all i
    hi = 1.0 - np.min(y)     # makes y+λ >= 1 for all i
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(y + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(y + lo, 0.0, 1.0)


def _grad_max_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Gradient of f_s(h)=∑_i h_i (1-h_{i+shift}) w.r.t. h,
    where out‑of‑range indices are ignored.
    """
    n = h.shape[0]
    g = np.zeros_like(h)

    if shift >= 0:
        # forward part: (1 - h_{i+shift}) for i where i+shift < n
        g[: n - shift] += 1.0 - h[shift:]
        # backward part: -h_{i-shift} for i where i-shift >= 0
        g[shift:] -= h[: n - shift]
    else:
        s = -shift
        # forward part for negative shift
        g[s:] += 1.0 - h[: n - s]
        # backward part for negative shift
        g[: n - s] -= h[s:]

    return g


def _random_feasible(rng: np.random.Generator, n: int, target: float) -> np.ndarray:
    """Generate a random vector in [0,1]^n with prescribed sum."""
    y = rng.random(n)
    return _project_box_sum(y, target)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns a step function h (as a NumPy array), its C5 bound,
    and the number of sample points.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise h
    # ------------------------------------------------------------------
    try:

        init = initial_h_values       # type: ignore[name-defined]

    except Exception:  # NameError or any issue – fall back to a fresh start
        n = 200                     # a modest size (under 1000 as requested)
        target_sum = n / 2.0
        h = _random_feasible(rng, n, target_sum)

    dx = 2.0 / n

    # ------------------------------------------------------------------
    # 2. Keep the best solution we have seen
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_val = float(evaluate_erdos_solution(best_h))

    start_time = time.time()
    iteration = 0

    # ------------------------------------------------------------------
    # 3. Main optimisation loop (gradient‑based subgradient descent)
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s - 0.3:
        # a short descent run from the current point
        for _ in range(2000):
            # correlation and worst shift
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_corr = np.max(corr)
            max_idx = np.argmax(corr)
            shift = max_idx - (n - 1)

            # current bound
            cur_val = max_corr * dx
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_h = h.copy()
                # stop early if we already beat the target threshold
                if best_val < 0.3808:
                    return (best_h, float(best_val), n)

            # gradients of all shifts attaining the maximum (subgradient)
            tol = 1e-9
            mask = np.abs(corr - max_corr) <= tol
            shs = np.where(mask)[0] - (n - 1)
            g = np.zeros_like(h)
            for sh in shs:
                g += _grad_max_shift(h, int(sh))
            g /= len(shs)

            # step size (diminishing)
            alpha = 0.8 / math.sqrt(iteration + 1.0)

            # tentative step and projection
            y = h - alpha * g
            y = _project_box_sum(y, target_sum)

            # accept if it does not increase the worst overlap
            new_corr = np.max(np.correlate(y, 1.0 - y, mode='full'))
            if new_corr <= max_corr + 1e-12:
                h = y
                max_corr = new_corr
            else:

                improved = False
                for factor in (0.5, 0.25, 0.125):

                    y2 = _project_box_sum(y2, target_sum)
                    c2 = np.max(np.correlate(y2, 1.0 - y2, mode='full'))
                    if c2 <= max_corr + 1e-12:
                        h = y2
                        max_corr = c2
                        improved = True
                        break
                if not improved:

                    pass

            iteration += 1

        # ------------------------------------------------------------------
        h = best_h.copy()
        h = np.clip(h + rng.normal(scale=0.05, size=n), 0.0, 1.0)
        h = _project_box_sum(h, target_sum)

    # time budget exhausted – return the best construction found
    best_val = float(evaluate_erdos_solution(best_h))
    return (best_h, best_val, n)
