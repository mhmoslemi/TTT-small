def compute_subgradient(h):
    n = len(h)
    # correlation without dx
    corr = np.correlate(h, 1-h, mode='full')
    # find max value(s)
    max_val = corr.max()
    # Tolerance for equality:
    tol = 1e-12
    # get list of shifts where close to max
    indices = np.where(np.abs(corr - max_val) <= tol)[0]
    g = np.zeros(n)
    for idx in indices:
        shift = idx - (n - 1)  # positive shift means h shifted right?
        if shift >= 0:
            # term1 for i: (1 - h[i+shift]) for i< n-shift else 0
            term1 = np.concatenate([1 - h[shift:], np.zeros(shift)])
            term2 = np.concatenate([np.zeros(shift), h[:n-shift]])
        else:
            s = -shift
            term1 = np.concatenate([np.zeros(s), 1 - h[:n-s]])
            term2 = np.concatenate([h[s:], np.zeros(s)])
        g += term1 - term2
    # If multiple shifts, average (subgradient). We'll divide by count
    g = g / len(indices)
    return g

def project_to_box_and_sum(v, target_sum):
    # find λ in appropriate range, we can set low = min(v)-1, high = max(v)+1.
    # We'll binary search for λ such that sum(min(1, max(0, v - λ))) = target_sum
    lo = -1.0  # can be less
    hi = 1.0   # maybe 2
    for _ in range(50):
        mid = (lo + hi) / 2
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - hi, 0.0, 1.0)
    return proj

def smooth_max_gradient(h, beta):
    n = len(h)
    O = np.correlate(h, 1 - h, mode='full')
    maxO = O.max()
    # stability:
    exp_weights = np.exp(beta * (O - maxO))
    sum_w = exp_weights.sum()
    weights = exp_weights / sum_w
    grad = np.zeros_like(h)
    # shift index offset
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            term1 = np.concatenate([1 - h[shift:], np.zeros(shift)])
            term2 = np.concatenate([np.zeros(shift), h[:n - shift]])
        else:
            s = -shift
            term1 = np.concatenate([np.zeros(s), 1 - h[:n - s]])
            term2 = np.concatenate([h[s:], np.zeros(s)])
        grad += w * (term1 - term2)
    return grad

def smooth_max_gradient(h, beta):
    n = len(h)
    O = np.correlate(h, 1 - h, mode='full')
    maxO = O.max()
    exp_weights = np.exp(beta * (O - maxO))
    sum_w = exp_weights.sum()
    if sum_w == 0:
        return np.zeros_like(h)
    weights = exp_weights / sum_w
    grad = np.zeros_like(h)
    # iterate over shift indexes
    for idx, w in enumerate(weights):
        if w == 0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            # term1: 1 - h[shift:] on indices 0:n-shift
            # term2: h[:n-shift] on indices shift:n
            grad[:n-shift] += w * (1 - h[shift:])
            grad[shift:] -= w * h[:n-shift]
        else:
            s = -shift
            # term1 on indices s:n: 1 - h[:n-s]
            # term2 on indices 0:n-s: h[s:]
            grad[s:] += w * (1 - h[:n-s])
            grad[:n-s] -= w * h[s:]
    return grad

def project_to_sum_box(v, target):
    # Use water-filling via binary search for lambda
    # Solve min ||x - v||^2 s.t. sum x = target, 0 <= x <= 1.
    # Equivalent to x_i = min(1, max(0, v_i - λ))
    # Find λ such that sum_i x_i = target.
    # λ is monotonic decreasing; use binary search.
    lo = -1.0  # lower bound for λ may be less than min(v)-1
    hi = 1.0   # upper bound for λ may be greater than max(v)
    # expand range if needed
    # ensure that sum at lo > target and sum at hi < target for monotonic decreasing
    # compute sums at lo and hi
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - hi, 0.0, 1.0)
    # due to floating error, adjust small residual
    residual = target - x.sum()
    if abs(residual) > 1e-8:
        # distribute residual uniformly among non-bound entries
        free = (x > 0.0) & (x < 1.0)
        count = free.sum()
        if count > 0:
            x[free] += residual / count
            # again clip if any exceed bounds (rare)
            x = np.clip(x, 0.0, 1.0)
    return x

import numpy as np
import time

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def _project_to_sum_box(v, target):
    # water‑filling via binary search for λ in x_i = clip(v_i - λ,0,1)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - hi, 0.0, 1.0)
    # tiny residual correction
    res = target - x.sum()
    if abs(res) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt:
            x[free] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_max_gradient(h, beta):
    n = h.shape[0]
    O = np.correlate(h, 1.0 - h, mode="full")
    m = O.max()
    w = np.exp(beta * (O - m))
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.zeros_like(h)
    w = w / w_sum
    grad = np.zeros_like(h)
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            # (1 - h[i+shift]) contributes on i=0..n-shift-1
            grad[: n - shift] += weight * (1.0 - h[shift:])
            # -h[i-shift] contributes on i=shift..n-1
            grad[shift:] -= weight * h[: n - shift]
        else:
            s = -shift
            grad[s:] += weight * (1.0 - h[: n - s])
            grad[: n - s] -= weight * h[s:]
    return grad

def _gradient_descent(start_h, iters=1500, lr0=0.25, beta0=2.0, beta_inc=0.008):
    h = start_h.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_val = _compute_c5(h)
    cur_val = best_val
    for it in range(iters):
        beta = beta0 + beta_inc * it
        g = _smooth_max_gradient(h, beta)
        lr = lr0 * (1.0 - it / iters)
        h_new = _project_to_sum_box(h - lr * g, target)
        new_val = _compute_c5(h_new)
        if new_val < best_val:
            best_val, best_h = new_val, h_new.copy()
        # simple acceptance rule
        if new_val <= cur_val or np.random.rand() < 0.01:
            h, cur_val = h_new, new_val
    return best_h, best_val

def _local_refine(h, steps=200):
    n = h.shape[0]
    target = n / 2.0
    cur_h = h.copy()
    cur_c = _compute_c5(cur_h)
    for _ in range(steps):
        # choose i weighted by value, j weighted by complement
        probs_i = cur_h / cur_h.sum()
        probs_j = (1.0 - cur_h) / (1.0 - cur_h).sum()
        i = np.random.choice(n, p=probs_i)
        j = np.random.choice(n, p=probs_j)
        eps = min(cur_h[i], 1.0 - cur_h[j]) * 0.1
        if eps <= 0.0:
            continue
        cand = cur_h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = _project_to_sum_box(cand, target)
        cand_c = _compute_c5(cand)
        if cand_c < cur_c:
            cur_h, cur_c = cand, cand_c
    return cur_h, cur_c
