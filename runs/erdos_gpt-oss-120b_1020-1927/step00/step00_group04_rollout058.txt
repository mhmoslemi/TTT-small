import time, numpy as np

def compute_mismatch_counts(h):
    conv = np.convolve(h, 1-h, mode='full')
    return conv    # length 2n-1

def max_mismatch(conv):
    return conv.max()

def find_worst_shift(conv, n):
    idx = np.argmax(conv)
    L = idx - (n-1)
    return L, idx

def improve_one_step(h, L):
    n = len(h)
    if L > 0:
        # indices where i+L < n
        i_range = np.arange(n - L)
        # mismatches f where h[i]==1 and h[i+L]==0
        bad_i = i_range[(h[i_range] == 1) & (h[i_range + L] == 0)]
        # mismatches where h[i]==0 and h[i+L]==1
        bad_j = i_range[(h[i_range] == 0) & (h[i_range + L] == 1)]
    elif L < 0:
        # similar using negative L
        Lpos = -L
        i_range = np.arange(Lpos, n)
        bad_i = i_range[(h[i_range] == 1) & (h[i_range - Lpos] == 0)]   # h[i]=1, h[i-L]=0
        bad_j = i_range[(h[i_range] == 0) & (h[i_range - Lpos] == 1)]
    else:
        return False   # L=0 no mismatches

    if bad_i.size and bad_j.size:
        # pick random index from each
        i = int(bad_i[np.random.randint(bad_i.size)])
        j = int(bad_j[np.random.randint(bad_j.size)])
        # swap
        h[i], h[j] = h[j], h[i]
        return True
    return False

    np.random.seed(seed)
    total_time = budget_s
    start = time.time()
    # choose n: we can adapt to time; maybe try 400 first.
    n = 400
    half = n // 2
    # initial vector:
    h = np.zeros(n, dtype=np.int8)
    # set half ones randomly
    h[:half] = 1
    np.random.shuffle(h)
    # compute initial max mismatches
    conv = np.convolve(h, 1 - h, mode='full')
    max_mismatch = int(conv.max())
    best_h = h.copy()
    best_mismatch = max_mismatch
    # Precompute indices groups
    while time.time() - start < total_time * 0.95:
        # compute worst lag
        conv = np.convolve(h, 1 - h, mode='full')
        max_mismatch = int(conv.max())
        if max_mismatch < best_mismatch:
            best_mismatch = max_mismatch
            best_h = h.copy()
            # optional early exit if already below target
            if best_mismatch * 2.0 / n <= 0.3808:
                break
        # find worst lag index
        idx = np.argmax(conv)
        lag = idx - (n - 1)
        # attempt constructive swap
        improved = False
        if lag != 0:
            if lag > 0:
                i_range = np.arange(n - lag)
                bad_i = i_range[(h[i_range] == 1) & (h[i_range + lag] == 0)]
                bad_j = i_range[(h[i_range] == 0) & (h[i_range + lag] == 1)]
            else:
                L = -lag
                i_range = np.arange(L, n)
                bad_i = i_range[(h[i_range] == 1) & (h[i_range - L] == 0)]
                bad_j = i_range[(h[i_range] == 0) & (h[i_range - L] == 1)]

                j = int(bad_j[np.random.randint(bad_j.size)])

import time, math
import numpy as np

def _c5_from_binary(h):
    """Return max overlap count (not scaled) for binary vector h."""
    # correlation of h with its complement, as used by the evaluator
    corr = np.correlate(h, 1 - h, mode="full")
    return int(corr.max())


def _target_lag_indices(n, lag):
    """Return index arrays for a given lag (positive or negative)."""
    if lag > 0:
        i = np.arange(n - lag)
        i2 = i + lag
    elif lag < 0:
        L = -lag
        i = np.arange(L, n)
        i2 = i - L
    else:
        return np.array([], dtype=int), np.array([], dtype=int)
    return i, i2


def _try_targeted_swap(h, lag):
    """Find a beneficial 0/1 swap that reduces mismatches for this lag."""
    n = h.shape[0]
    i, i2 = _target_lag_indices(n, lag)
    if i.size == 0:
        return False, h
    # positions where (1,0) occurs for this lag
    bad_i = i[(h[i] == 1) & (h[i2] == 0)]
    # positions where (0,1) occurs for this lag
    bad_j = i[(h[i] == 0) & (h[i2] == 1)]
    if bad_i.size == 0 or bad_j.size == 0:
        return False, h
    # pick a random pair and swap
    ii = int(bad_i[np.random.randint(bad_i.size)])
    jj = int(bad_j[np.random.randint(bad_j.size)])
    new_h = h.copy()
    new_h[ii], new_h[jj] = new_h[jj], new_h[ii]
    return True, new_h


def _random_swap(h):
    """Perform a random 0/1 swap preserving the number of ones."""
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)
    i = int(ones[np.random.randint(ones.size)])
    j = int(zeros[np.random.randint(zeros.size)])
    new_h = h.copy()
    new_h[i], new_h[j] = new_h[j], new_h[i]
    return new_h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function (binary vector) that gives a low
    C5 bound.  Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95   # a safety margin

    # ---- parameters -------------------------------------------------
    n = 400                     # number of discretisation points (even)
    half = n // 2               # must have exactly half ones
    target_c5 = 0.3808          # we stop early if we beat this
    # ---------------------------------------------------------------

    best_h = None
    best_mis = n                # maximal mismatch count (the larger the worse)

    # we may run several independent restarts while time permits
    while time.time() < time_limit:
        # ----- initialise a balanced binary vector -----------------
        h = np.zeros(n, dtype=np.int8)
        h[:half] = 1
        np.random.shuffle(h)

        # quick evaluation
        cur_mis = _c5_from_binary(h)
        if cur_mis < best_mis:
            best_mis, best_h = cur_mis, h.copy()
            if best_mis * 2.0 / n <= target_c5:
                break

        # ----- simulated‑annealing style improvement ---------------
        T = 1.0                 # temperature
        # each inner loop consumes a small chunk of the budget
        inner_end = min(time_limit, time.time() + 6.0)
        while time.time() < inner_end:
            # recompute the worst lag
            conv = np.correlate(h, 1 - h, mode="full")
            cur_mis = int(conv.max())
            if cur_mis < best_mis:
                best_mis, best_h = cur_mis, h.copy()
                if best_mis * 2.0 / n <= target_c5:
                    break

            worst_idx = int(conv.argmax())
            lag = worst_idx - (n - 1)   # signed shift producing the max

            # try a targeted swap that directly attacks this lag
            made, trial_h = _try_targeted_swap(h, lag)
            if not made:
                # fallback to a completely random swap
                trial_h = _random_swap(h)

            new_mis = _c5_from_binary(trial_h)
            delta = new_mis - cur_mis
            # accept if better, otherwise with Metropolis probability
            if delta <= 0 or math.exp(-delta / max(T, 1e-9)) > np.random.rand():
                h = trial_h
                cur_mis = new_mis

            # cool down
            T *= 0.9995
            # exit early if we already meet the target
            if cur_mis * 2.0 / n <= target_c5:
                break

        # safety check – break outer loop if target achieved
        if best_h is not None and best_mis * 2.0 / n <= target_c5:
            break

    # -----------------------------------------------------------------
    if best_h is None:          # should never happen, but guard against it
        best_h = np.zeros(n, dtype=np.int8)

    # convert to float as the harness expects a step function in [0,1]
    best_h_float = best_h.astype(float)

    # final bound (scaled by dx = 2/n)
    c5_bound = best_mis * (2.0 / n)

    # If the provided helper evaluate_erdos_solution exists, prefer its value
    try:
        c5_bound = evaluate_erdos_solution(best_h_float)
    except Exception:
        pass

    return best_h_float, float(c5_bound), n
