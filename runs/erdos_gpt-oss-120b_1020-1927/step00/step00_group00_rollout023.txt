def propose_move(h, step_scale):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    # allowable delta such that 0 <= h_i - delta <= 1 and 0 <= h_j + delta <= 1
    delta_max = min(step_scale, h[i], 1 - h[j])
    delta_min = -min(step_scale, 1 - h[i], h[j])
    delta = np.random.uniform(delta_min, delta_max)
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new
```

We preserve sum because subtraction and addition cancel.

    h = initial_h.copy()
    best_h = h.copy()
    best_score = objective(h)
    T0 = best_score

def binary_anneal(h, target_sum, n_iters, rng, start_temp):
    # h is binary (0/1)
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h < 0.5)
    best_h = h.copy()
    best_score = objective(h)
    cur_h = h.copy()
    cur_score = best_score
    for it in range(n_iters):
        # choose random i in ones, j in zeros
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = objective(cur_h)
        delta = new_score - cur_score
        # temperature schedule
        T = start_temp * (1 - it / n_iters)
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_score = new_score
            # update ones/zeros arrays
            # swap lists
            # We'll update by swapping indices in ones and zeros arrays
            # one approach: just recompute ones/zeros each iteration - cheap.
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            # nothing else
        if cur_score < best_score:
            best_score = cur_score
            best_h = cur_h.copy()
    return best_h, best_score
```

import time, math, numpy as np
from scipy.optimize import minimize

def _c5_of_h(h):
    """Maximum overlap integral for a step function sampled by vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def _subgrad(h):
    """Sub‑gradient of the max correlation (multiplied by dx)."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    lag = idx - (n - 1)          # lag ∈ [-(n-1), n-1]
    g = np.zeros_like(h)
    if lag >= 0:
        g[:n - lag] += 1 - h[lag:]      # ∂/∂h_i of  h_i term
        g[lag:]      -= h[:n - lag]     # ∂/∂h_i of  -h_i*h_{i+lag}
    else:
        l = -lag
        g[l:]  += 1 - h[:n - l]
        g[:n - l] -= h[l:]
    return g * (2.0 / n)               # include dx factor

def _random_binary(n, rng):
    """Binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    #   initialise problem size and a feasible start vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        base_h = np.asarray(initial_h_values, dtype=float)
        base_n = base_h.size
    else:
        base_n = 45
        base_h = _random_binary(base_n, rng)

    # upscale resolution (larger n gives a tighter bound)
    factor = 6                     # 45*6 = 270 points, even
    n = base_n * factor
    if n % 2:
        factor += 1
        n = base_n * factor

    # repeat the parent pattern, then force binary and correct the sum
    h = np.repeat(base_h, factor)
    h = (h >= 0.5).astype(float)
    target_sum = n // 2
    cur_sum = int(h.sum())
    while cur_sum < target_sum:
        z = np.where(h < 0.5)[0]
        i = rng.choice(z)
        h[i] = 1.0
        cur_sum += 1
    while cur_sum > target_sum:
        o = np.where(h > 0.5)[0]
        i = rng.choice(o)
        h[i] = 0.0
        cur_sum -= 1

    best_h = h.copy()
    best_score = _c5_of_h(best_h)

    # ------------------------------------------------------------------
    #   simulated annealing on the binary space
    # ------------------------------------------------------------------
    cur_h = best_h.copy()
    cur_score = best_score
    max_iters = 120000                     # generous but will stop earlier
    start_T = best_score * 0.15 + 1e-9

    for it in range(max_iters):
        if time.time() - start > budget_s * 0.65:
            break
        # linear temperature decay
        T = start_T * (1.0 - it / max_iters)
        # pick a 1 and a 0 to swap
        ones = np.flatnonzero(cur_h > 0.5)
        zeros = np.flatnonzero(cur_h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = _c5_of_h(cur_h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0

        # occasional restart from the current best to escape stagnation
        if (it + 1) % 20000 == 0:
            cur_h = best_h.copy()
            cur_score = best_score

    # ------------------------------------------------------------------
    #   continuous refinement with SLSQP (uses gradient of the max lag)
    # ------------------------------------------------------------------
    def eq_sum(x):
        return np.sum(x) - target_sum

    cons = ({'type': 'eq', 'fun': eq_sum},)
    bounds = [(0.0, 1.0)] * n

    starts = [best_h]
    for _ in range(3):
        starts.append(_random_binary(n, rng))

    for start_vec in starts:
        if time.time() - start > budget_s * 0.95:
            break
        res = minimize(_c5_of_h, start_vec, method='SLSQP',
                       jac=_subgrad, bounds=bounds,
                       constraints=cons,
                       options={'maxiter': 250, 'ftol': 1e-9, 'disp': False})
        if not res.success:
            continue
        cand = np.clip(res.x, 0.0, 1.0)
        # tiny correction to enforce the sum exactly
        diff = np.sum(cand) - target_sum
        if abs(diff) > 1e-8:
            free = np.where((cand > 1e-9) & (cand < 1 - 1e-9))[0]
            if free.size:
                cand[free] -= diff / free.size
                cand = np.clip(cand, 0.0, 1.0)
        sc = _c5_of_h(cand)
        if sc < best_score:
            best_score = sc
            best_h = cand.copy()

    # final sum correction (in case of tiny drift)
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = np.sum(best_h) - target_sum
    if abs(diff) > 1e-8:
        # adjust a non‑boundary entry
        for idx in range(n):
            if 0.0 < best_h[idx] < 1.0:
                best_h[idx] -= diff
                break
        best_h = np.clip(best_h, 0.0, 1.0)

    final_score = _c5_of_h(best_h)
    return (best_h, final_score, n)
