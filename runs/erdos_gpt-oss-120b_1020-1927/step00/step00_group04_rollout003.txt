def correlation_fft(a,b):
    n=len(a)
    L = 1 << ((2*n-1).bit_length())  # next power of two
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    c = np.fft.irfft(fa * np.conj(fb), L)
    return c[:2*n-1]

def correlation_max(h):
    n = h.shape[0]
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, L)
    Hc = np.fft.rfft(1 - h, L)
    conv = np.fft.irfft(H * np.conj(Hc), L)[:2*n-1]
    max_val = conv.max()
    idx = conv.argmax()
    k = idx - (n-1)
    return max_val, k, conv

def shift_vec(a, k):
    if k > 0:
        out = np.empty_like(a)
        out[:k] = 0.0
        out[k:] = a[:-k]
        return out
    elif k < 0:
        kk = -k
        out = np.empty_like(a)
        out[-kk:] = 0.0
        out[:-kk] = a[kk:]
        return out
    else:
        return a.copy()

def project_box_sum(v, target):
    # v may be outside [0,1]; we first clip to bounds
    v = np.clip(v, 0.0, 1.0)

        if s > target:

        else:

import numpy as np
import time

def _correlation_max(h):
    """Return max correlation value, the shift that attains it, and the full array."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    H = np.fft.rfft(h, L)
    Hc = np.fft.rfft(1.0 - h, L)
    conv = np.fft.irfft(H * np.conj(Hc), L)[: 2 * n - 1]
    idx = int(conv.argmax())
    max_val = float(conv[idx])
    k = idx - (n - 1)                            # shift corresponding to idx
    return max_val, k, conv


def _shift_vec(v, k):
    """Shift vector by k places, filling empty entries with 0 (no wrap)."""
    if k > 0:
        out = np.empty_like(v)
        out[:k] = 0.0
        out[k:] = v[:-k]
        return out
    if k < 0:
        kk = -k
        out = np.empty_like(v)
        out[-kk:] = 0.0
        out[:-kk] = v[kk:]
        return out
    return v.copy()


def _project_box_sum(v, target):
    """Project v onto the set {0≤x≤1, sum(x)=target}."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -2.0, 2.0                           # guaranteed to bracket the solution
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _symmetrize_and_project(v, target):
    """Force mirror symmetry (v[i]=v[n-1-i]) and project onto the feasible set."""
    v = (v + v[::-1]) * 0.5
    return _project_box_sum(v, target)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95          # safety margin

    # ------------------------------------------------------------------
    # Determine resolution and initialise h
    # ------------------------------------------------------------------
    init = globals().get('initial_h_values', None)
    if init is not None:
        init = np.asarray(init, dtype=float)
        n0 = init.size
    else:
        n0 = None

    # target size (even, not too large)
    n = 256
    if n0 is not None:
        n = min(800, max(256, n0 * 4))
        if n % 2:
            n += 1

    # up‑sample initial construction if available, else start from random binary
    if init is not None:
        x_old = np.linspace(0.0, 2.0, init.size)
        x_new = np.linspace(0.0, 2.0, n)
        h = np.interp(x_new, x_old, init)
    else:
        h = np.zeros(n)
        h[: n // 2] = 1.0
        np.random.shuffle(h)

    target_sum = n / 2.0                               # because ∑h = n/2 ⇒ ∫h =1
    h = _symmetrize_and_project(h, target_sum)

    # ------------------------------------------------------------------
    # Main optimisation loop (projected sub‑gradient descent with occasional restarts)
    # ------------------------------------------------------------------
    best_h = h.copy()
    max_val, _, _ = _correlation_max(h)
    dx = 2.0 / n
    best_c5 = max_val * dx

    iter_no_improve = 0
    iteration = 0
    base_lr = 0.6                                      # initial step size

    while time.time() < deadline:
        iteration += 1

        # current objective
        cur_max, cur_k, _ = _correlation_max(h)
        cur_c5 = cur_max * dx

        # keep the best ever seen
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
            iter_no_improve = 0
        else:
            iter_no_improve += 1

        # gradient of the worst‑case overlap (sub‑gradient of max)
        grad = 1.0 - _shift_vec(h, cur_k) - _shift_vec(h, -cur_k)

        # learning‑rate schedule (slow decay)
        lr = base_lr / (1.0 + iteration * 1e-3)

        # tentative update
        h_candidate = h - lr * grad
        h_candidate = _symmetrize_and_project(h_candidate, target_sum)

        # evaluate candidate
        cand_max, _, _ = _correlation_max(h_candidate)
        cand_c5 = cand_max * dx

        # accept if improvement; otherwise keep old point (very small probability of uphill move)
        if cand_c5 < cur_c5 or np.random.rand() < 0.003:
            h = h_candidate
            cur_c5 = cand_c5
            # reset iteration counter for learning‑rate dynamics
        else:
            # occasional aggressive perturbation to escape plateaus
            if iter_no_improve >= 3000:
                # random restart around current best
                noise = (np.random.rand(n) - 0.5) * 0.2
                h = best_h + noise
                h = _symmetrize_and_project(h, target_sum)
                iter_no_improve = 0

    # final projection to make sure constraints hold exactly
    best_h = _symmetrize_and_project(best_h, target_sum)

    # recompute final bound (just in case)
    final_max, _, _ = _correlation_max(best_h)
    final_c5 = final_max * (2.0 / best_h.shape[0])

    return best_h, final_c5, best_h.shape[0]
