import numpy as np, time, math

def project_onto_box_simplex(v, target):
    """Project vector v onto [0,1] box and sum=target."""
    # Use bisection on lambda for clipping.
    lo = -1.0
    hi = 2.0
    # In principle lambda range such that after shifting and clipping sum = target.
    for _ in range(60):
        lam = (lo + hi) / 2
        w = np.clip(v - lam, 0, 1)
        s = np.sum(w)
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2, 0, 1)

```python
def compute_c5(h, dx):
    n = len(h)
    N = 1 << (2*n - 1).bit_length()  # next power of two >= 2n
    # Actually pad to N.
    fft_h = np.fft.rfft(h, n=N)
    fft_one_minus_h = np.fft.rfft(1 - h, n=N)
    corr = np.fft.irfft(fft_h * np.conj(fft_one_minus_h), n=N)
    # The linear correlation for lags -(n-1) to (n-1) is corr[0:2*n-1]
    # Note: irfft returns real values.
    r = corr[:2*n-1] * dx
    return np.max(r)
```

    """Project v onto the box [0,1]^n while fixing sum(v)=target."""
    # Find λ such that sum(clip(v-λ,0,1)) = target by bisection.
    lo = np.min(v) - 2.0   # enough to make all entries clip to 1
    hi = np.max(v) + 2.0   # enough to make all entries clip to 0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    return w

def _compute_c5(h, dx):
    """Return max_k ∫ h(x)(1‑h(x+k))dx for the discretised h."""
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1
    H = np.fft.rfft(h, n=N)
    O = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * np.conj(O), n=N)
    vals = corr[:2 * n - 1] * dx
    return np.max(vals)

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)

    # ---- initialise ----
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.shape[0]
        target = n / 2.0
        if abs(h.sum() - target) > 1e-12:
            h = _project_to_bounds_and_sum(h, target)
    else:
        n = 150                     # any even number < 1000 works well
        h = np.random.rand(n)
        target = n / 2.0
        h = _project_to_bounds_and_sum(h, target)

    dx = 2.0 / n
    cur_c = _compute_c5(h, dx)
    best_h = h.copy()
    best_c = cur_c

    # ---- simulated annealing ----
    T0 = 0.01
    T_end = 1e-6
    max_time = budget_s * 0.95
    iter_since_best = 0
    max_no_improve = 50000

    while time.time() - start < max_time:
        elapsed = time.time() - start
        frac = elapsed / max_time
        T = T0 * (T_end / T0) ** frac          # exponential cooling
        scale = T / T0                         # ≤1, shrinks step size

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]

        low  = max(-hi,      hj - 1.0)   # minimal delta (negative)
        high = min(1.0 - hi,  hj)        # maximal delta (positive)
        if low >= high:
            continue

        delta = np.random.uniform(low, high) * scale
        h[i] = hi + delta
        h[j] = hj - delta

        new_c = _compute_c5(h, dx)

        accept = False
        if new_c < cur_c:
            accept = True
        elif T > 1e-9:
            prob = math.exp(-(new_c - cur_c) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_c = new_c
            if new_c + 1e-12 < best_c:
                best_c = new_c
                best_h = h.copy()
                iter_since_best = 0
            else:
                iter_since_best += 1
        else:
            # revert
            h[i] = hi
            h[j] = hj
            iter_since_best += 1

        if iter_since_best > max_no_improve:
            # random restart, keep the best found so far
            h = np.random.rand(n)
            h = _project_to_bounds_and_sum(h, target)
            cur_c = _compute_c5(h, dx)
            iter_since_best = 0

    # ---- final greedy refinement on the best found solution ----
    h = best_h.copy()
    cur_c = best_c
    for _ in range(20000):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        low  = max(-hi,      hj - 1.0)
        high = min(1.0 - hi,  hj)
        if low >= high:
            continue
        # tiny step
        max_delta = min(high, -low) * 0.2
        if max_delta <= 0:
            continue
        delta = np.random.uniform(-max_delta, max_delta)
        h[i] = hi + delta
        h[j] = hj - delta
        new_c = _compute_c5(h, dx)
        if new_c + 1e-12 < cur_c:
            cur_c = new_c
            best_c = new_c
            best_h = h.copy()
        else:
            h[i] = hi
            h[j] = hj

    final_c = _compute_c5(best_h, dx)
    return (best_h.tolist(), float(final_c), n)
