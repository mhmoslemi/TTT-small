def project_onto_box_sum(v, lower, upper, target_sum, max_iter=100, tol=1e-9):
    # solve for λ s.t. sum_i clip(v_i - λ, lower, upper) = target_sum
    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):
        mid = (lo+hi)/2
        proj = np.clip(v - mid, lower, upper)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, lower, upper)

import time
import numpy as np

def _project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0, iters=60):
    """Project v onto the set {x: lo <= x <= hi, sum(x)=target_sum}."""
    low = -1e6
    high = 1e6
    for _ in range(iters):
        mid = (low + high) * 0.5
        proj = np.clip(v - mid, lo, hi)
        s = proj.sum()
        if s > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) * 0.5, lo, hi)


def _evaluate(h):
    """Return the C5 bound for a given step vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _subgradient_step(h, target_sum, step):
    """One sub‑gradient step minimizing max_{k} ∫h(1‑h(·+k))."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    idxs = np.where(corr >= m - 1e-12)[0]  # indices achieving the max
    g = np.zeros_like(h)
    for idx in idxs:
        shift = idx - (n - 1)
        if shift > 0:
            g[: n - shift] -= h[shift:]
            g[shift:] -= h[: n - shift]
        elif shift < 0:
            d = -shift
            g[d:] -= h[: n - d]
            g[: n - d] -= h[d:]
        else:  # shift == 0
            g -= 2.0 * h
        g += 1.0               # the constant “+1” part of the gradient
    g /= len(idxs)
    v = h - step * g
    return _project_onto_box_sum(v, target_sum)


def _local_refine(h, target_sum, moves=1500):
    """Simple hill‑climbing: transfer mass between two coordinates."""
    n = h.shape[0]
    best = h.copy()
    best_val = _evaluate(best)
    for _ in range(moves):
        i, j = np.random.choice(n, 2, replace=False)
        # propose a small transfer delta keeping sum constant
        delta = np.random.uniform(-0.08, 0.08)
        # clip to stay inside [0,1]
        if delta > 0:
            delta = min(delta, 1.0 - best[i], best[j])
        else:
            delta = -min(-delta, best[i], 1.0 - best[j])
        if abs(delta) < 1e-12:
            continue
        cand = best.copy()
        cand[i] += delta
        cand[j] -= delta
        val = _evaluate(cand)
        if val < best_val:
            best = cand
            best_val = val
    # final projection (may be unnecessary but safe)
    return _project_onto_box_sum(best, target_sum), best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function giving a low C5 bound."""
    np.random.seed(seed)
    start = time.time()

    # Determine dimension
    init = globals().get("initial_h_values", None)
    if init is not None:
        init = np.asarray(init, dtype=float)
        n0 = init.shape[0]
        n = max(140, n0 * 2)               # a modest refinement of the given grid
        # linear up‑sampling of the supplied vector
        xs_old = np.linspace(0, 2, n0, endpoint=False)
        xs_new = np.linspace(0, 2, n, endpoint=False)
        h0 = np.interp(xs_new, xs_old, init)
    else:
        n = 140
        h0 = np.random.rand(n)

    target_sum = n * 0.5                     # because ∫h = 1 and dx = 2/n
    h0 = _project_onto_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_val = _evaluate(best_h)

    # Parameters for the optimisation loops
    max_iters_per_restart = 2500
    step_initial = 1.0

    restart = 0
    while time.time() - start < budget_s * 0.95:
        # fresh start: either the supplied construction (first restart) or random
        if restart == 0 and init is not None:
            h = h0.copy()
        else:
            h = _project_onto_box_sum(np.random.rand(n), target_sum)

        # sub‑gradient descent
        for t in range(1, max_iters_per_restart + 1):
            alpha = step_initial / np.sqrt(t)
            h = _subgradient_step(h, target_sum, alpha)

            # occasional evaluation
            if t % 300 == 0 or t == max_iters_per_restart:
                val = _evaluate(h)
                if val < best_val:
                    best_val = val
                    best_h = h.copy()
                # early exit if we already beat the target record
                if best_val < 0.3808:
                    break

        # local refinement
        h_refined, val_refined = _local_refine(h, target_sum, moves=2000)
        if val_refined < best_val:
            best_val = val_refined
            best_h = h_refined.copy()
            if best_val < 0.3808:
                break

        restart += 1

    # final projection to guarantee feasibility
    best_h = _project_onto_box_sum(best_h, target_sum)

    return best_h.tolist(), float(best_val), int(best_h.shape[0])
