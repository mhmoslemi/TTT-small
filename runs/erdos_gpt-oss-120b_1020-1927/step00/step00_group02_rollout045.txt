def hill_climb_binary(n, time_limit, start_h=None):
    dx = 2.0 / n
    # initial h: if start_h provided and len matches n and sum constraint, else random binary.
    if start_h is not None and len(start_h) == n and abs(start_h.sum() - n/2) < 1e-6:
        h = start_h.copy()
    else:
        h = np.zeros(n, dtype=np.int8)
        h[:n//2] = 1
        np.random.shuffle(h)
    # Compute best
    best_h = h.copy()
    best_val = evaluate(h)
    # maintain list of positions of ones and zeros for fast swap
    ones = np.where(h==1)[0]
    zeros = np.where(h==0)[0]
    start = time.time()
    iter = 0
    while time.time() - start < time_limit:
        # pick random swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Perform swap in copy
        h[i], h[j] = 0, 1
        val = evaluate(h)
        if val < best_val - 1e-12: # improvement
            best_val = val
            best_h = h.copy()
            # update ones, zeros by swapping
            # Since we swapped i->0, j->1
            # update arrays
            # i becomes zero, j becomes one
            # Use python list for fast update?
            # For simplicity just recompute ones, zeros occasionally.
        else:
            # revert swap
            h[i], h[j] = 1, 0
        iter += 1
        # maybe update ones zeros after each iteration to keep accurate.
        # For speed we can maintain arrays using set; but okay for n=100.
        # We'll recompute after rejection/pick? slower but fine.
        if iter % 1000 == 0:
            ones = np.where(h==1)[0]
            zeros = np.where(h==0)[0]
    return best_h, best_val

            # temperature linearly decays from 5e‑3 to 0
            T = 0.005 * (remaining / time_budget)
            if T > 0.0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
                cur_h, cur_val = new_h, new_val
            non_improve += 1

        if non_improve >= max_non_improve:
            # restart from the best solution found so far
            cur_h, cur_val = best_h.copy(), best_val
            non_improve = 0

    return best_h, best_val


def _project_to_feasible(h: np.ndarray) -> np.ndarray:
    """
    Project an arbitrary vector onto the feasible set:
    0 ≤ h ≤ 1 and sum(h) = n/2.
    Uses a simple water‑filling (box‑constrained) projection.
    """
    n = h.size
    target = n / 2.0
    # First clip to the box
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if np.isclose(s, target):
        return h

    # Adjust by a constant λ, then re‑clip, repeat (few iterations are enough)
    for _ in range(10):
        delta = (target - s) / n
        h += delta
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if np.isclose(s, target, atol=1e-12):
            break
    # Small residual correction (should be tiny)
    diff = target - s
    if abs(diff) > 1e-12:
        # distribute the remaining amount to non‑saturated entries
        mask = (h > 0.0) & (h < 1.0)
        if np.any(mask):
            h[mask] += diff / np.count_nonzero(mask)
            h = np.clip(h, 0.0, 1.0)
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Entry point required by the evaluation harness.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    total_start = time.time()
    # ------------------------------------------------------------------
    # Obtain initial vector (if supplied) or build a simple binary start.
    # ------------------------------------------------------------------
    try:
        init_vec = np.asarray(initial_h_values, dtype=np.float64)
        n_pts = init_vec.size
        h0 = _project_to_feasible(init_vec)
    except NameError:
        # No supplied construction – use a random binary pattern.
        n_pts = 90                     # a modest even size for speed
        h0 = np.zeros(n_pts, dtype=np.float64)
        h0[: n_pts // 2] = 1.0
        np.random.shuffle(h0)

    # ------------------------------------------------------------------
    # Run the stochastic improvement within the allocated time.
    # ------------------------------------------------------------------
    # Spend most of the budget on optimisation; keep a small safety margin.
    time_for_opt = max(budget_s * 0.93, 0.1)
    best_h, best_val = improve_h(h0, time_for_opt)

    # Final safety re‑evaluation (in case of rounding drift)
    best_h = _project_to_feasible(best_h)
    final_val = evaluate_c5(best_h)

    # ------------------------------------------------------------------
    # Return result as plain Python objects.
    # ------------------------------------------------------------------
    elapsed = time.time() - total_start
    # (the harness may ignore elapsed, but we keep it for debugging)
    return best_h.tolist(), float(final_val), n_pts
