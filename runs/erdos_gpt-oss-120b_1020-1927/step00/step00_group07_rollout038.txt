import time
import numpy as np

def _c5_score(h):
    """Return the C5 value for a step function given by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); numpy correlates a with b shifted.
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, int(corr.max())   # scaled value and integer count


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that minimises C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Problem size and basic parameters
    # ------------------------------------------------------------------
    # If the parent supplied an initial vector, use its length; otherwise 80.
    init_vec = globals().get('initial_h_values')
    if init_vec is not None:
        h0 = np.asarray(init_vec, dtype=np.float64)
        n = h0.shape[0]
    else:
        n = 80
        h0 = None

    dx = 2.0 / n
    target_sum = n / 2.0          # because ∫h = 1  -> sum(h) = n/2

    # ------------------------------------------------------------------
    # 2. Starting points
    # ------------------------------------------------------------------
    # a) Constant 0.5 function (always feasible)
    best_h = np.full(n, 0.5, dtype=np.float64)
    best_score, best_int = _c5_score(best_h)

    # b) If a parent vector is supplied, use it (after a tiny projection)
    if h0 is not None:
        cand = np.clip(h0, 0.0, 1.0)
        # project onto the affine hyperplane sum = target_sum
        cur_sum = cand.sum()
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            # distribute the needed correction uniformly; clip again if needed
            cand += diff / n
            cand = np.clip(cand, 0.0, 1.0)
            # a second tiny correction to restore the exact sum
            cur_sum = cand.sum()
            diff = target_sum - cur_sum
            if abs(diff) > 1e-12:
                # fix by adding the remainder to a random component that can still move
                idx = np.random.choice(np.where((cand > 0.0) & (cand < 1.0))[0])
                cand[idx] = np.clip(cand[idx] + diff, 0.0, 1.0)
        s, si = _c5_score(cand)
        if s < best_score:
            best_h = cand.copy()
            best_score, best_int = s, si

    # ------------------------------------------------------------------
    # 3. Local optimisation loop (simulated annealing / hill‑climbing)
    # ------------------------------------------------------------------
    cur_h = best_h.copy()
    cur_score, cur_int = best_score, best_int

    # pre‑compute indexes of entries that are not at a bound (to allow a swap)
    eps = 1e-9
    max_no_improve = 3000  # after this many tries without improvement we randomise a bit
    no_improve_cnt = 0

    while True:
        # time check
        if time.time() - start_time > budget_s - 0.5:
            break

        # compute the correlation and the worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = int(corr.argmax())
        max_int = int(corr.max())
        shift = max_idx - (n - 1)                # can be negative
        abs_shift = abs(shift)

        # ------------------------------------------------------------------
        # 3a. Try a targeted move that directly attacks the worst shift
        # ------------------------------------------------------------------
        improved = False
        if abs_shift > 0:
            # indices where h[i]=1 and h[i+abs_shift]=0   (contribute to the max)
            left = cur_h[: n - abs_shift]
            right = cur_h[abs_shift:]
            mask_one = (left > 0.5) & (right < 0.5)   # 1‑0 pattern
            mask_zero = (left < 0.5) & (right > 0.5)  # 0‑1 pattern
            idx_one = np.flatnonzero(mask_one)
            idx_zero = np.flatnonzero(mask_zero)

            if idx_one.size and idx_zero.size:
                i_one = np.random.choice(idx_one)            # position of a 1
                i_zero = np.random.choice(idx_zero)          # position of a 0
                # perform a swap (full amounts, keeps the sum exact)
                new_h = cur_h.copy()
                new_h[i_one] = 0.0
                new_h[i_zero] = 1.0
                new_score, new_int = _c5_score(new_h)
                if new_int <= cur_int:
                    cur_h, cur_score, cur_int = new_h, new_score, new_int
                    improved = True
                    no_improve_cnt = 0
                    if new_int < best_int:
                        best_h, best_score, best_int = new_h.copy(), new_score, new_int
        # ------------------------------------------------------------------
        # 3b. If the targeted move did not help, perform a small random transfer
        # ------------------------------------------------------------------
        if not improved:
            # pick a source (h > 0) and a destination (h < 1)
            src_candidates = np.flatnonzero(cur_h > eps)
            dst_candidates = np.flatnonzero(cur_h < 1.0 - eps)
            if src_candidates.size and dst_candidates.size:
                i_src = np.random.choice(src_candidates)
                i_dst = np.random.choice(dst_candidates)
                if i_src != i_dst:
                    # move a random amount not exceeding the bounds
                    max_delta = min(cur_h[i_src], 1.0 - cur_h[i_dst])
                    if max_delta > 1e-12:
                        delta = np.random.rand() * max_delta
                        new_h = cur_h.copy()
                        new_h[i_src] -= delta
                        new_h[i_dst] += delta
                        new_score, new_int = _c5_score(new_h)

                        # simulated‑annealing acceptance
                        elapsed = time.time() - start_time
                        temp = max(0.001, (budget_s - elapsed) / budget_s) * 0.5
                        accept = False
                        if new_int <= cur_int:
                            accept = True
                        else:
                            prob = np.exp(-(new_int - cur_int) / temp)
                            if np.random.rand() < prob:
                                accept = True
                        if accept:
                            cur_h, cur_score, cur_int = new_h, new_score, new_int
                            no_improve_cnt = 0 if new_int < cur_int else no_improve_cnt + 1
                            if new_int < best_int:
                                best_h, best_score, best_int = new_h.copy(), new_score, new_int
                        else:
                            no_improve_cnt += 1
        # ------------------------------------------------------------------
        # 3c. Periodically inject a random perturbation to escape plateaus
        # ------------------------------------------------------------------
        if no_improve_cnt >= max_no_improve:
            # perform a few random swaps (binary style) to diversify
            for _ in range(5):
                ones = np.flatnonzero(cur_h > 0.5)
                zeros = np.flatnonzero(cur_h < 0.5)
                if ones.size and zeros.size:
                    a = np.random.choice(ones)
                    b = np.random.choice(zeros)
                    cur_h[a], cur_h[b] = 0.0, 1.0
            cur_score, cur_int = _c5_score(cur_h)
            if cur_int < best_int:
                best_h, best_score, best_int = cur_h.copy(), cur_score, cur_int
            no_improve_cnt = 0

    # ------------------------------------------------------------------
    # 4. Return the best found solution
    # ------------------------------------------------------------------
    # Ensure the returned vector respects the exact integral constraint
    # (tiny numerical drift may appear); project back if necessary.
    total = best_h.sum()
    diff = target_sum - total
    if abs(diff) > 1e-9:
        # adjust the first component that is not saturated
        adjustable = np.where((best_h > eps) & (best_h < 1.0 - eps))[0]
        if adjustable.size:
            best_h[adjustable[0]] += diff
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            # fallback: uniform correction
            best_h += diff / n
            best_h = np.clip(best_h, 0.0, 1.0)

    final_score, _ = _c5_score(best_h)
    return (best_h.tolist(), float(final_score), n)
