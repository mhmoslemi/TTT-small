import numpy as np
import time

def _project_to_bounds_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1]^n while fixing sum(v)=target."""
    lo, hi = -1.0, 2.0          # wide enough to contain the feasible λ
    for _ in range(60):        # enough for double precision
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _overlap_vals(h: np.ndarray, dx: float) -> np.ndarray:
    """C5 values for all integer shifts (full correlation)."""
    return np.correlate(h, 1.0 - h, mode="full") * dx


def _gradient_weighted(h: np.ndarray, weight: np.ndarray, dx: float) -> np.ndarray:
    """Gradient of the smooth‑max objective Σ weight_k * O(k)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    off = n - 1                     # index where shift=0 sits
    for idx, w in enumerate(weight):
        if w == 0.0:
            continue
        s = idx - off               # integer shift
        if s >= 0:
            # (1 – h[i+s]) for i = 0 … n‑1‑s
            grad[: n - s] += w * (1.0 - h[s:])
            # –h[i‑s] for i = s … n‑1
            grad[s:] -= w * h[: n - s]
        else:
            t = -s
            # (1 – h[i‑t]) for i = t … n‑1
            grad[t:] += w * (1.0 - h[: n - t])
            # (1 – 0) = 1 for i = 0 … t‑1
            grad[:t] += w
            # –h[i+t] for i = 0 … n‑t‑1
            grad[: n - t] -= w * h[t:]
    grad *= dx
    return grad


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step‑function h, its C₅ bound, and the number of points."""
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise h : use the provided construction if present,
    # otherwise fall back to a simple uniform start.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n = h.shape[0]
    else:
        n = 200
        h = np.full(n, 0.5, dtype=np.float64)

    dx = 2.0 / n
    target_sum = n / 2.0            # because ∫h = 1  ⇔  Σh·dx = 1

    h = _project_to_bounds_and_sum(h, target_sum)

    # best solution seen so far
    best_h = h.copy()
    best_val = np.max(_overlap_vals(best_h, dx))

    # optimisation parameters
    step = 0.2
    alpha = 15.0                     # smoothing for the max
    stagn = 0
    it = 0
    max_iters = 10 ** 7

    # ------------------------------------------------------------------
    # main optimisation loop (projected gradient descent with simple
    # step‑size adaptation and occasional random perturbations).
    # ------------------------------------------------------------------
    while (time.time() - start) < budget_s * 0.96 and it < max_iters:
        O = _overlap_vals(h, dx)               # O(k) for all shifts
        cur_val = O.max()

        # record best
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
            stagn = 0
        else:
            stagn += 1

        # smooth‑max weights (softmax)
        maxO = O.max()
        exp_term = np.exp(alpha * (O - maxO))
        weight = exp_term / exp_term.sum()

        # gradient of the smooth objective
        grad = _gradient_weighted(h, weight, dx)

        # tentative step and projection
        h_new = _project_to_bounds_and_sum(h - step * grad, target_sum)

        # evaluate the true objective after the step
        new_val = np.max(_overlap_vals(h_new, dx))

        # accept / reject and adapt step size
        if new_val < cur_val:
            h = h_new
            step = min(step * 1.05, 1.0)       # slowly enlarge if successful
        else:
            step *= 0.5                         # shrink if not improving
            # occasional random kick to escape plateaus
            if stagn % 500 == 0:
                h = np.clip(h + np.random.normal(scale=0.01, size=n), 0.0, 1.0)
                h = _project_to_bounds_and_sum(h, target_sum)

        # slowly increase the smoothing parameter – it sharpens the
        # approximation to the true max as optimisation proceeds.
        if it % 2000 == 0 and alpha < 30.0:
            alpha *= 1.02

        it += 1

    # final evaluation of the best vector discovered
    final_c5 = float(np.max(_overlap_vals(best_h, dx)))
    return best_h, final_c5, n
