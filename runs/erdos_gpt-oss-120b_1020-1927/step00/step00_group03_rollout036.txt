        if n - dpos > 0:
            g[dpos:] += 1 - h[:n-dpos]
            g[:n-dpos] -= h[dpos:]

import time
import math
import numpy as np

def _evaluate_max_O(h: np.ndarray) -> float:
    """Return max O = max_x ∫ h(x)(1‑h(x+k))dx approximated by correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear correlation, missing entries are treated as zero
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _gradient_for_shift(h: np.ndarray, d: int) -> np.ndarray:
    """Sub‑gradient of O(d) w.r.t. h for a given integer shift d."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if d >= 0:
        nd = n - d
        if nd > 0:
            g[:nd] += 1.0 - h[d:]
            g[d:] -= h[:nd]
    else:
        dpos = -d
        nd = n - dpos
        if nd > 0:
            g[dpos:] += 1.0 - h[:nd]
            g[:nd] -= h[dpos:]
    return g

def _project_to_bounds_and_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the box [0,1]^n while keeping the sum exactly target_sum."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    eps = 1e-12
    if abs(cur - target_sum) < eps:
        return v
    # water‑filling style iterative projection
    for _ in range(100):
        diff = target_sum - cur
        free = (v > 0.0) & (v < 1.0)
        nfree = int(free.sum())
        if nfree == 0:
            break
        delta = diff / nfree
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
        cur = v.sum()
        if abs(cur - target_sum) < eps:
            break
    return v

def _gradient_descent(start_h: np.ndarray, max_iters: int, deadline: float):
    """Simple sub‑gradient descent for the max‑O problem."""
    h = start_h.astype(float).copy()
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_val = _evaluate_max_O(h)

    for it in range(max_iters):
        if time.time() > deadline:
            break
        # raw correlation (no dx) – enough to locate the worst shift
        raw = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(raw))
        d = idx - (n - 1)

        cur_val = float(raw[idx] * (2.0 / n))
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        # sub‑gradient step
        g = _gradient_for_shift(h, d)
        # decreasing step size
        alpha = 0.2 * (1.0 - it / max_iters)
        h = _project_to_bounds_and_sum(h - alpha * g, target_sum)

    final_val = _evaluate_max_O(best_h)
    return best_h, final_val
