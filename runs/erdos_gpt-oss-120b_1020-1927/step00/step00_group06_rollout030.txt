def project_onto_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {0<=x<=1, sum(x)=target}."""
    lo, hi = -1.0, 1.0
    for _ in range(60):          # binary search for the Lagrange multiplier
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

            continue
        delta = (np.random.rand() * 2.0 - 1.0) * step_range
        delta = np.clip(delta, delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_c5 = compute_c5(new_h, dx)

        if (new_c5 < cur_c5 or
            np.random.rand() <
                math.exp(-(new_c5 - cur_c5) / (temperature + 1e-12))):
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5

        temperature = max(temperature * cooling, min_temp)

        # ----- occasional random restart -------------------------------- #
        if it % restart_interval == 0:
            cur_h = random_feasible_vector(n, target_sum)
            cur_c5 = compute_c5(cur_h, dx)
            if cur_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
            temperature = 0.5

    # ------------------------------------------------------------------ #
    # 3.  Greedy polishing of the best vector --------------------------- #
    # ------------------------------------------------------------------ #
    cur_h = best_h.copy()
    cur_c5 = best_c5
    for _ in range(20000):
        i, j = np.random.randint(n, size=2)
        while i == j:
            j = np.random.randint(n)
        hi, hj = cur_h[i], cur_h[j]

        delta_max = min(hi, 1.0 - hj)
        delta_min = -min(hj, 1.0 - hi)
        if delta_max <= delta_min:
            continue

        for delta in (delta_max, delta_min):
            if abs(delta) < 1e-12:
                continue
            cand = cur_h.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_c5 = compute_c5(cand, dx)
            if cand_c5 + 1e-12 < cur_c5:
                cur_h = cand
                cur_c5 = cand_c5
                if cand_c5 < best_c5:
                    best_h = cand.copy()
                    best_c5 = cand_c5
                break

    # ------------------------------------------------------------------ #
    # 4.  Final sanity check and return -------------------------------- #
    # ------------------------------------------------------------------ #
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)

    return best_h.astype(np.float32), float(final_c5), n
