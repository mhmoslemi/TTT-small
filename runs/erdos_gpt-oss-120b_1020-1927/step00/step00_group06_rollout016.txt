def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random
    random.seed(seed)
    np.random.seed(seed)
    start=time.time()
    # Determine n_points
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        N = len(h0)
    else:
        N=150
        h0 = None
    target = N/2.0
    dx = 2.0/N
    # define evaluation
    def c5_of(h):
        H = np.fft.fft(h, n=2*N)
        G = np.fft.fft(1.0 - h, n=2*N)
        corr = np.real(np.fft.ifft(H * np.conj(G)))[:2*N-1]
        return np.max(corr) * dx
    # projection function
    def project(v):
        # solve sum clamp(v+lam) = target using bisection
        lo = -np.max(v) - 1.0
        hi = 1.0 - np.min(v) + 1.0
        for _ in range(50):
            mid = (lo+hi)/2.0
            x = np.clip(v + mid, 0.0, 1.0)
            s = x.sum()
            if s < target:
                lo = mid
            else:
                hi = mid
        x = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
        # small correction for numerical error
        err = target - x.sum()
        if abs(err) > 1e-9:
            # distribute error uniformly on free variables
            free = (x > 1e-12) & (x < 1-1e-12)
            if free.any():
                idx = np.where(free)[0][0]
                x[idx] = min(1.0, max(0.0, x[idx] + err))
            else:
                # fallback: adjust all equally
                x += err / len(x)
                x = np.clip(x, 0.0, 1.0)
        return x
    # random feasible generator
    def random_feasible():
        v = np.random.rand(N)
        return project(v)
    # Initialize population
    popsize=20
    population=[]
    if h0 is not None:
        # ensure feasible
        h0_proj = project(h0)
        best_val = c5_of(h0_proj)
        population.append((best_val, h0_proj))
    while len(population) < popsize:
        cand = random_feasible()
        population.append((c5_of(cand), cand))
    # sort
    population.sort(key=lambda x:x[0])
    best_score, best_h = population[0]
    # main loop
    iter=0
    sigma0=0.2
    sigma_min=0.001
    decay=0.9995

    # maintain best recorded
    while time.time() - start < budget_s*0.95:

import time
import random
import numpy as np

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Evolutionary+gradient search for a step function h on [0,2] that
    minimizes   max_k ∫ h(x)*(1-h(x+k)) dx .
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    time_limit = budget_s * 0.95  # safety margin

    # ------------------------------------------------------------
    # problem size and basic parameters
    # ------------------------------------------------------------
    try:
        init_vec = np.array(initial_h_values, dtype=np.float64)  # provided by harness
    except Exception:
        init_vec = None

    if init_vec is not None:
        N = init_vec.size
    else:
        N = 150                     # fallback size if no parent is supplied
    target_sum = N / 2.0            # ∫h = 1  =>  sum(h) = N/2
    dx = 2.0 / N

    # ------------------------------------------------------------
    # helper: projection onto intersection of [0,1]^N and sum = target_sum
    # ------------------------------------------------------------
    def project(v: np.ndarray) -> np.ndarray:
        """Return projection of v onto {0<=x_i<=1, sum x_i = target_sum}."""
        lo = -np.max(v) - 1.0      # guarantee all clamp(v+lo) = 0
        hi = 1.0 - np.min(v) + 1.0  # guarantee all clamp(v+hi) = 1
        for _ in range(50):
            mid = (lo + hi) * 0.5
            x = np.clip(v + mid, 0.0, 1.0)
            s = x.sum()
            if s < target_sum:
                lo = mid
            else:
                hi = mid
        x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
        # tiny numerical fix
        err = target_sum - x.sum()
        if abs(err) > 1e-9:
            free = (x > 1e-12) & (x < 1.0 - 1e-12)
            if free.any():
                i = np.flatnonzero(free)[0]
                x[i] = min(1.0, max(0.0, x[i] + err))
            else:
                x += err / N
                x = np.clip(x, 0.0, 1.0)
        return x

    # ------------------------------------------------------------
    # helper: generate a random feasible vector
    # ------------------------------------------------------------
    def random_feasible() -> np.ndarray:
        v = np.random.rand(N)
        return project(v)

    # ------------------------------------------------------------
    # helper: evaluate C5 for a given h
    # ------------------------------------------------------------
    def c5_of(h: np.ndarray) -> float:
        # linear cross‑correlation via FFT (zero‑padded)
        H = np.fft.fft(h, n=2 * N)
        G = np.fft.fft(1.0 - h, n=2 * N)
        corr_full = np.real(np.fft.ifft(H * np.conj(G)))
        corr = corr_full[: 2 * N - 1]          # linear (full) correlation
        return np.max(corr) * dx

    # ------------------------------------------------------------
    # helper: one gradient‑descent step on the worst shift
    # ------------------------------------------------------------
    def gradient_step(h: np.ndarray, eta: float) -> np.ndarray:
        H = np.fft.fft(h, n=2 * N)
        G = np.fft.fft(1.0 - h, n=2 * N)
        corr_full = np.real(np.fft.ifft(H * np.conj(G)))
        corr = corr_full[: 2 * N - 1]
        idx = int(np.argmax(corr))
        shift = idx - (N - 1)            # shift ∈ [-(N-1), N-1]

        grad = np.ones(N, dtype=np.float64)   # derivative of Σ h_i part

        if shift >= 0:
            # contribution from h_{j+shift}
            if shift < N:
                grad[: N - shift] -= h[shift:]
                grad[shift:] -= h[: N - shift]
        else:
            s = -shift
            if s < N:
                grad[s:] -= h[: N - s]
                grad[: N - s] -= h[s:]

        # descent step
        nh = h - eta * grad
        return project(nh)

    # ------------------------------------------------------------
    # initialise best candidate
    # ------------------------------------------------------------
    if init_vec is not None:
        best_h = project(init_vec.copy())
    else:
        best_h = random_feasible()
    best_score = c5_of(best_h)

    # ------------------------------------------------------------
    # search parameters
    # ------------------------------------------------------------
    sigma = 0.2          # mutation amplitude
    eta = 0.05           # gradient step size
    sigma_min = 0.001
    eta_min = 0.0005
    sigma_decay = 0.9995
    eta_decay = 0.9995

    it = 0
    last_improve_it = 0

    # ------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------
    while time.time() - start_time < time_limit:
        it += 1

        # ------- Gaussian + sinusoidal mutation -------
        child = best_h + sigma * np.random.randn(N)

        if np.random.rand() < 0.3:                      # occasional wave perturbation
            freq = np.random.randint(1, max(2, N // 2))
            amp = sigma * np.random.uniform(-1.0, 1.0)
            phase = np.random.rand() * 2.0 * np.pi
            i = np.arange(N)
            child += amp * np.sin(2.0 * np.pi * freq * i / N + phase)

        child = project(child)
        child_score = c5_of(child)
        if child_score < best_score:
            best_h, best_score = child, child_score
            last_improve_it = it

        # ------- Gradient descent step -------
        g_child = gradient_step(best_h, eta)
        g_score = c5_of(g_child)
        if g_score < best_score:
            best_h, best_score = g_child, g_score
            last_improve_it = it

        # ------- occasional random restart if stagnating -------
        if it - last_improve_it > 5000:
            rnd = random_feasible()
            rnd_score = c5_of(rnd)
            if rnd_score < best_score:
                best_h, best_score = rnd, rnd_score
                last_improve_it = it

        # decay parameters
        sigma = max(sigma_min, sigma * sigma_decay)
        eta = max(eta_min, eta * eta_decay)

    # ------------------------------------------------------------
    # final return
    # ------------------------------------------------------------
    return (best_h.tolist(), float(best_score), N)
