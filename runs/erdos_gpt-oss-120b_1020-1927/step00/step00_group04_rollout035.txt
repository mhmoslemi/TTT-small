import time
import math
import numpy as np

def _compute_c5(vec):
    """Return C5 for a binary (or [0,1]) vector."""
    n = vec.shape[0]
    # correlation of vec with its complement (full mode)
    corr = np.correlate(vec, 1 - vec, mode="full")
    # dx = length / n = 2 / n
    return float(np.max(corr) * (2.0 / n))

def _initial_binary(seed, target_n, init_vals=None, rng=None):
    """Create a binary start vector of length target_n with exactly target_n/2 ones."""
    if rng is None:
        rng = np.random.default_rng(seed)
    # If we have an initial construction, up‑sample it and binarise.
    if init_vals is not None:
        init_vals = np.asarray(init_vals, dtype=float)
        n0 = init_vals.shape[0]
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x1 = np.linspace(0.0, 2.0, target_n, endpoint=False)
        interp = np.interp(x1, x0, init_vals)
        start = (interp > 0.5).astype(int)
    else:
        start = np.zeros(target_n, dtype=int)
        ones = rng.choice(target_n, size=target_n // 2, replace=False)
        start[ones] = 1

    # Fix exact weight if rounding introduced an error.
    needed = target_n // 2 - start.sum()
    if needed > 0:
        zero_idx = np.where(start == 0)[0]
        flip = rng.choice(zero_idx, size=needed, replace=False)
        start[flip] = 1
    elif needed < 0:
        one_idx = np.where(start == 1)[0]
        flip = rng.choice(one_idx, size=-needed, replace=False)
        start[flip] = 0
    return start

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Desired number of points (keep it modest and even)
    n = 200
    # Use the provided initial_h_values if present
    init_vals = globals().get("initial_h_values", None)
    h = _initial_binary(seed, n, init_vals, rng)

    cur_c5 = _compute_c5(h.astype(float))
    best_h = h.copy()
    best_c5 = cur_c5

    # Simulated‑annealing parameters
    init_temp = 0.02   # temperature in the C5‑scale
    min_temp = 1e-6

    # Main optimisation loop
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break

        # Linear cooling
        temp = max(min_temp, init_temp * (1.0 - elapsed / budget_s))

        # Find the worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)          # positive => right shift, negative => left shift

        # Determine overlapping indices for this shift
        if shift >= 0:
            i_range = np.arange(0, n - shift, dtype=int)
            j_range = i_range + shift
        else:
            p = -shift
            i_range = np.arange(p, n, dtype=int)
            j_range = i_range - p

        # Mismatched pairs (1 with 0) and opposite (0 with 1)
        mask12 = (h[i_range] == 1) & (h[j_range] == 0)
        mask21 = (h[i_range] == 0) & (h[j_range] == 1)

        idx12 = i_range[mask12]   # positions where a 1 is paired with a 0
        idx21 = i_range[mask21]   # positions where a 0 is paired with a 1

        made_swap = False
        # Try to fix the worst shift by swapping a complementary mismatched pair
        if idx12.size and idx21.size:
            i = rng.choice(idx12)
            j = rng.choice(idx21)
            # perform swap
            h[i], h[j] = h[j], h[i]
            new_c5 = _compute_c5(h.astype(float))
            if new_c5 <= cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / temp):
                cur_c5 = new_c5
                made_swap = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                # revert
                h[i], h[j] = h[j], h[i]

        # Fallback: random swap of any 1 and 0
        if not made_swap:
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            if ones_idx.size and zeros_idx.size:
                i = rng.choice(ones_idx)
                j = rng.choice(zeros_idx)
                h[i], h[j] = 0, 1
                new_c5 = _compute_c5(h.astype(float))
                if new_c5 <= cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / temp):
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                else:
                    h[i], h[j] = 1, 0

        # Early exit if we already beat the target
        if best_c5 <= 0.3808:
            break

    # Return the best solution discovered
    return (best_h.astype(float), float(best_c5), n)
