def compute_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()
    # tolerance for maxima
    tol = 1e-12
    idx_max = np.where(np.abs(corr - max_val) <= tol)[0]  # indices in corr array
    # note idx corresponds to shift s = idx - (n-1)
    s_vals = idx_max - (n - 1)
    # compute gradient contributions
    grads = np.zeros(n)
    rng = np.arange(n)
    for s in s_vals:
        # mask for i such that i-s in range:
        mask1 = ((rng - s) >= 0) & ((rng - s) < n)
        # plus term:
        grads[mask1] += 1.0
        # minus term 1:
        grads[mask1] -= h[rng[mask1] - s]
        # mask2 for i+s in range:

        grads[mask2] -= h[rng[mask2] + s]
    grads *= dx
    return grads

import time
import numpy as np

# ---------------- helper functions ---------------- #

def project_onto_bounds_and_sum(v, target):
    """Euclidean projection of v onto {0<=x<=1, sum(x)=target}."""
    lo, hi = -10.0, 10.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def compute_corr_and_grad(h):
    """Return (C5 value, subgradient) for current h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = corr.max()
    # active shifts (may be several with same max)
    idxs = np.where(np.abs(corr - max_val) <= 1e-12)[0]
    grad = np.zeros(n)
    n_minus_1 = n - 1
    for idx in idxs:
        k = idx - n_minus_1          # shift used by np.correlate
        if k >= 0:
            i = np.arange(0, n - k)
            j = i + k
        else:
            i = np.arange(-k, n)
            j = i + k
        grad[i] += 1.0 - h[j]
        grad[j] -= h[i]
    grad *= dx
    return max_val * dx, grad


def compute_obj(h):
    """C5 value for a given h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def subgrad_optimize(h0, max_iter, time_limit, start_time):
    """Projected sub‑gradient descent."""
    h = h0.copy()
    n = h.shape[0]
    target = n * 0.5
    best_h = h.copy()
    best_val = compute_obj(h)
    step0 = 1.0
    for t in range(1, max_iter + 1):
        if time.time() - start_time > time_limit:
            break
        val, grad = compute_corr_and_grad(h)
        step = step0 / np.sqrt(t)
        h_new = h - step * grad
        h_new = project_onto_bounds_and_sum(h_new, target)
        val_new = compute_obj(h_new)
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_val


def local_refine(h0, cur_val, time_limit, start_time, rng):
    """Simple pairwise random adjustments (hill‑climbing)."""
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = cur_val
    while time.time() - start_time < time_limit:
        i, j = rng.randint(0, n, size=2)
        if i == j:
            continue
        # permissible delta keeping both entries in [0,1]
        delta_max = min(1.0 - h[i], h[j])
        delta_min = -min(h[i], 1.0 - h[j])
        if delta_max <= delta_min:
            continue
        delta = rng.uniform(delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        val_new = compute_obj(h_new)
        if val_new < best_val - 1e-12:
            best_val = val_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_val


def optimize_n(n, init_h, time_budget, rng):
    """Optimize for a fixed grid size n."""
    target = n * 0.5
    best_h = None
    best_val = np.inf
    start_total = time.time()
    # keep performing restarts while time remains
    while time.time() - start_total < time_budget:
        remaining = time_budget - (time.time() - start_total)
        if remaining < 5.0:
            break
        # each restart gets a slice of the remaining budget
        slice_budget = min(30.0, remaining / 3.0)
        # initialise
        if init_h is not None and init_h.shape[0] == n and best_h is None:
            h0 = project_onto_bounds_and_sum(init_h.copy(), target)
        else:
            rnd = rng.rand(n)
            h0 = project_onto_bounds_and_sum(rnd, target)
        # sub‑gradient phase
        sub_start = time.time()
        h_opt, val_opt = subgrad_optimize(h0, max_iter=4000,
                                          time_limit=slice_budget,
                                          start_time=sub_start)
        # local refinement
        elapsed = time.time() - sub_start
        left = slice_budget - elapsed
        if left > 0.5:
            h_opt, val_opt = local_refine(h_opt, val_opt, left,
                                          sub_start, rng)
        # keep best over all restarts
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
        # break if we already beat the target bound
        if best_val < 0.3808:
            break
    if best_h is None:
        # fallback uniform solution
        best_h = np.full(n, 0.5)
        best_h = project_onto_bounds_and_sum(best_h, target)
        best_val = compute_obj(best_h)
    else:
        best_h = project_onto_bounds_and_sum(best_h, target)
        best_val = compute_obj(best_h)
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) fitting the budget."""
    rng = np.random.RandomState(seed)
    start = time.time()

    # make use of a provided parent solution if present
    init_present = 'initial_h_values' in globals()
    if init_present:

        n0 = init_h_full.size
    else:
        init_h_full = None
        n0 = None

    # candidate grid sizes (small, fast, and a doubled version)
    candidates = []
    if n0 is not None:
        candidates.append(n0)
        if n0 * 2 <= 200:
            candidates.append(n0 * 2)
        if n0 * 3 <= 300:
            candidates.append(n0 * 3)
    else:
        candidates = [80, 120, 200]

    # allocate time roughly equally among candidate sizes
    total_budget = max(0.0, budget_s - 0.2)      # tiny safety margin
    per_n = total_budget / len(candidates)

    best_h = None
    best_val = np.inf
    best_n = None

    for n in candidates:
        elapsed = time.time() - start
        remaining = budget_s - elapsed - 0.05

            break
        time_for_n = min(per_n, remaining)

        init_here = init_h_full if (init_present and n == n0) else None
