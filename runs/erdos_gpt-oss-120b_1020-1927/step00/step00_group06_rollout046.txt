import time
import numpy as np

def _project_onto_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the feasible set {x | 0<=x<=1, sum(x)=target}."""
    # Binary search for a shift lambda such that sum(clip(v+lambda,0,1)) = target
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_of(h: np.ndarray) -> float:
    """Return the C5 value of h using the definition from the statement."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

    Perform a simple local‑search/pair‑transfer optimisation for the
    Erdős minimum‑overlap problem.
