import cvxpy as cp, numpy as np, time

def solve_milp(N, time_limit):
    M = N // 2
    h = cp.Variable(N, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == M]
    # optional symmetry break: h[0] = 1 maybe
    # constraints.append(h[0] == 1)
    for k in range(1, N):
        n_k = N - k
        if n_k <= 0:
            continue
        # w variables for this shift
        w = cp.Variable(n_k, boolean=True)
        # linking constraints
        constraints += [w <= h[:n_k],
                        w <= h[k:],
                        w >= h[:n_k] + h[k:] - 1]
        # t bound
        constraints.append(M - cp.sum(w) <= t)
    constraints.append(t >= 0)
    # objective minimize t
    prob = cp.Problem(cp.Minimize(t), constraints)
    # solve with time limit param
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_iters=5e6, eps_abs=1e-4, # maybe other options
                   # possible options for CBC:
                   # use solver_opts param
                   )
    except Exception as e:
        # fallback
        prob.solve(solver=cp.CBC, verbose=False)
    return h.value, t.value
