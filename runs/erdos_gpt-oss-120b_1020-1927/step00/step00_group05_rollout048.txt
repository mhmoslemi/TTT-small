import time
import math
import numpy as np

def compute_c5(h, dx):
    """Maximum overlap integral for a discrete step function."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx

def project_sum_and_bounds(x, target):
    """Project onto the box [0,1]^n and the hyperplane sum(x)=target."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) < 1e-12:
        return x
    # water‑filling style iteration
    for _ in range(100):
        free = (x > 0.0) & (x < 1.0)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            break
        delta = diff / n_free
        x[free] += delta
        over = x > 1.0
        under = x < 0.0
        if over.any():
            x[over] = 1.0
        if under.any():
            x[under] = 0.0
        diff = target - np.sum(x)
        if abs(diff) < 1e-12:
            break
    # tiny correction if needed
    if abs(np.sum(x) - target) > 1e-9:
        idx = np.where((x > 0) & (x < 1))[0]
        if idx.size == 0:
            idx = np.arange(len(x))
        i = idx[0]
        x[i] = np.clip(x[i] + (target - np.sum(x)), 0.0, 1.0)
    return np.clip(x, 0.0, 1.0)

def subgradient_Q_shift(h, shift):
    """Sub‑gradient of Q_l(h)=∑ h_i(1‑h_{i+shift}) for the given shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    l = shift
    if l >= 0:
        end = n - l
        if end > 0:
            g[0:end] += 1 - h[l:n]
            g[l:n]   -= h[0:end]
    else:
        d = -l
        start = d
        if start < n:
            g[start:n]   += 1 - h[0:n-d]
            g[0:n-d]     -= h[start:n]
    return g

def binary_from_continuous(h, target):
    """Make a binary vector (0/1) keeping the largest `target` entries."""
    idx = np.argsort(-h)
    bin_h = np.zeros_like(h)
    bin_h[idx[:target]] = 1.0
    return bin_h

def hill_climb_binary(h_bin, dx, rng, max_trials=2000):
    """Simple 1‑swap hill climbing on a binary vector."""
    best = h_bin.copy()
    best_val = compute_c5(best, dx)
    for _ in range(max_trials):
        ones = np.flatnonzero(best == 1)
        zeros = np.flatnonzero(best == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        best[i] = 0.0
        best[j] = 1.0
        new_val = compute_c5(best, dx)
        if new_val < best_val:
            best_val = new_val
        else:
            best[i] = 1.0
            best[j] = 0.0
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------ discretisation choice ------------
    n = 200                     # even number, reasonable resolution
    target_sum = n // 2         # integral condition: ∫h = 1  ⇒  Σh = n/2
    dx = 2.0 / n

    # ------------ initial candidate ------------
    if "initial_h_values" in globals():
        init = initial_h_values
        init_n = init.shape[0]
        xs_init = np.linspace(0.0, 2.0, init_n, endpoint=False)
        xs_target = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(xs_target, xs_init, init)
        order = np.argsort(-interp)
        h0 = np.zeros(n)
        h0[order[:target_sum]] = 1.0
    else:
        h0 = np.zeros(n)
        h0[:target_sum] = 1.0
        rng.shuffle(h0)

    best_h = h0.copy()
    best_val = compute_c5(best_h, dx)

    # Try a quick binary hill‑climb from the initial point
    bin_h, bin_val = hill_climb_binary(best_h.copy(), dx, rng, max_trials=3000)
    if bin_val < best_val:
        best_h, best_val = bin_h, bin_val

    # ------------ main optimisation loop ------------
    restart = 0
    while time.time() - start < budget_s * 0.93:
        restart += 1

        # random binary start
        h = np.zeros(n)
        h[:target_sum] = 1.0
        rng.shuffle(h)

        # sub‑gradient descent on the worst shift
        max_iters = 2500
        improved_this_restart = False
        for it in range(max_iters):
            # current objective
            Q = np.correlate(h, 1 - h, mode="full") * dx
            cur_obj = np.max(Q)

            # time guard
            if time.time() - start > budget_s * 0.93:
                break

            # worst shift
            idx_max = int(np.argmax(Q))
            shift = idx_max - (n - 1)

            # sub‑gradient and step
            g = subgradient_Q_shift(h, shift)
            step = 0.5 / math.sqrt(it + 1)
            h_new = project_sum_and_bounds(h - step * g, target_sum)

            # evaluate candidate
            Q_new = np.correlate(h_new, 1 - h_new, mode="full") * dx
            new_obj = np.max(Q_new)

            if new_obj <= cur_obj:
                h = h_new
                if new_obj < best_val - 1e-9:
                    best_val = new_obj
                    best_h = h_new.copy()
                    improved_this_restart = True
            # else keep the old h (no acceptance of uphill moves)

        # binary refinement of the final continuous point
        bin_candidate = binary_from_continuous(h, target_sum)
        bin_candidate, bin_val = hill_climb_binary(bin_candidate, dx, rng, max_trials=3000)
        if bin_val < best_val:
            best_val = bin_val
            best_h = bin_candidate.copy()

            if time.time() - start > budget_s * 0.80:
                break

    # final feasibility projection
    best_h = project_sum_and_bounds(best_h, target_sum)
    final_val = compute_c5(best_h, dx)

    return (best_h, float(final_val), n)
