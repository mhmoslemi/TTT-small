def _project_to_box_and_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto [0,1]^n intersect {sum = target} by a simple binary search."""
    lo, hi = -1.0, 1.0  # enough to move any entry to 0 or 1

        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)

        if s > target:
            hi = mid
        else:

def _smooth_max_and_grad(h: np.ndarray, dx: float, alpha: float,
                         target_sum: float, penalty: float):
    """
    Returns (smooth_max, gradient) of
    smooth_max_alpha( corr(h,1-h) ) + penalty*(sum(h)-target_sum)^2 .
    """
    N = h.size
    # correlation (full, zero padding)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx

    # smooth max via log-sum-exp
    a_corr = alpha * corr
    a_max = a_corr.max()
    exp_a = np.exp(a_corr - a_max)
    sum_exp = exp_a.sum()
    smooth = (a_max + np.log(sum_exp)) / alpha

    # softmax weights (sum to 1)
    w = exp_a / sum_exp

    # gradient of smooth part
    grad = np.full(N, dx)                     # contribution of the constant '1' term
    # shifts from -(N-1) .. N-1
    shift_vals = np.arange(-N + 1, N, dtype=int)
    for idx, d in enumerate(shift_vals):
        wd = w[idx]
        if wd == 0.0:
            continue
        if d >= 0:
            # h[j+d] term
            grad[: N - d] -= wd * dx * h[d:]
            # h[j-d] term
            grad[d:] -= wd * dx * h[: N - d]
        else:
            m = -d
            grad[m:] -= wd * dx * h[: N - m]
            grad[: N - m] -= wd * dx * h[m:]

    # penalty part
    sum_err = h.sum() - target_sum
    smooth += penalty * sum_err * sum_err
    grad += 2.0 * penalty * sum_err * np.ones_like(h)

    return smooth, grad


def _compute_c5(h: np.ndarray) -> float:
    """Exact C5 value as used by the evaluator."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise C5.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Choose a reasonable number of points (balance speed vs flexibility)
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        base_n = base.size
        n = max(300, base_n * 4)          # upscale modestly if a tiny starter is given
        # upscale by linear interpolation
        x_small = np.linspace(0.0, 2.0, base_n, endpoint=False)
        x_big = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_big, x_small, base)
    else:
        n = 400
        h0 = np.full(n, 0.5)              # uniform feasible start

    dx = 2.0 / n
    target_sum = n / 2.0                  # because ∫h =1 ↔ sum(h)*dx =1

    # ensure feasibility of the initial point
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _project_to_box_and_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # Parameters for the optimisation
    # ------------------------------------------------------------------
    penalty = 100.0                       # quadratic penalty for the integral constraint
    alphas = [2.0, 6.0, 12.0, 30.0]       # annealing schedule for the smooth max
    max_iters_per_stage = 200

    # ------------------------------------------------------------------
    # Main optimisation loop (several restarts, respecting the time budget)
    # ------------------------------------------------------------------
    restart = 0
    while time.time() - start < 0.9 * budget_s:
        restart += 1
        # start from the best known plus a little noise to escape possible plateaux
        if restart == 1:
            cur_h = best_h.copy()
        else:
            cur_h = best_h.copy()
            cur_h += 0.02 * np.random.randn(n)          # gentle perturbation
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = _project_to_box_and_sum(cur_h, target_sum)

        for alpha in alphas:
            # objective/gradient closure for current alpha
            def fun(x):
                val, _ = _smooth_max_and_grad(x, dx, alpha, target_sum, penalty)
                return val

            def jac(x):
                _, g = _smooth_max_and_grad(x, dx, alpha, target_sum, penalty)
                return g

            bounds = [(0.0, 1.0)] * n
            res = optimize.minimize(fun, cur_h, method='L-BFGS-B',
                                    jac=jac, bounds=bounds,
                                    options={'maxiter': max_iters_per_stage,
                                             'disp': False})
            cur_h = res.x
            # project to satisfy the integral exactly (still within [0,1])
            cur_h = _project_to_box_and_sum(cur_h, target_sum)

        cur_c5 = _compute_c5(cur_h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = cur_h.copy()
        # a cheap local random search to polish the result
        for _ in range(200):
            if time.time() - start > 0.95 * budget_s:
                break
            # transfer a tiny amount from a random index i to j
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            delta = 0.01 * (np.random.rand() - 0.5)  # signed small step
            if delta > 0:
                allowed = min(delta, best_h[i], 1.0 - best_h[j])
            else:
                allowed = max(delta, -best_h[j], best_h[i] - 1.0)
            if allowed == 0:
                continue
            trial = best_h.copy()
            trial[i] -= allowed
            trial[j] += allowed
            # project only to keep within bounds (sum stays unchanged)
            trial = np.clip(trial, 0.0, 1.0)
            trial_c5 = _compute_c5(trial)
            if trial_c5 < best_c5 - 1e-8:
                best_c5 = trial_c5
                best_h = trial
        # early exit if we beat the target record
        if best_c5 <= 0.3808:
            break

    # final sanity check – ensure constraints are satisfied
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_box_and_sum(best_h, target_sum)
    final_c5 = _compute_c5(best_h)

    return best_h.tolist(), float(final_c5), n
