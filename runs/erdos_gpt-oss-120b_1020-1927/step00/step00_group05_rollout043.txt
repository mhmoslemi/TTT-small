def project_onto_box_and_sum(v, target_sum, lower=0.0, upper=1.0):
    # Clip to [lower, upper]
    v_clipped = np.clip(v, lower, upper)
    # Compute difference from target due to clipping
    diff = target_sum - v_clipped.sum()
    # If diff is zero, done.
    # Else we need to add diff distributed over non-clipped entries.
    # We'll use waterfilling: find set of indices where v not at bound; add equal portion.
    # Could solve via simple iterative approach.
    for _ in range(10):
        free = (v_clipped > lower + 1e-12) & (v_clipped < upper - 1e-12)
        n_free = free.sum()
        if n_free == 0:
            break
        delta = diff / n_free
        # Ensure we do not cross bounds
        v_new = v_clipped.copy()
        v_new[free] += delta
        # Clip again
        v_new = np.clip(v_new, lower, upper)
        # Update diff
        diff = target_sum - v_new.sum()
        v_clipped = v_new
    return v_clipped

import time, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0/n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_bounds_sum(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Adjust to meet sum
    diff = target_sum - h.sum()
    # Use water-filling
    for _ in range(10):
        free = ( (h > 0) & (h < 1) )
        n_free = np.count_nonzero(free)
        if n_free == 0:
            break
        delta = diff / n_free
        # adjust but respect bounds
        add = np.maximum(np.minimum(delta, 1 - h[free]), -h[free])
        h[free] += add
        diff = target_sum - h.sum()
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = len(h0)

        h0 = np.clip(h0, 0.0, 1.0)
        target_sum = n/2.0
        h0 = project_bounds_sum(h0, target_sum)
    else:
        # Choose an even n <= 800 for speed/accuracy
        n = 300
        # start with binary random
        h0 = np.zeros(n, dtype=np.float64)
        h0[:n//2] = 1.0

    target_sum = n/2.0
    cur_h = h0.copy()
    cur_c5 = compute_c5(cur_h)

    best_c5 = cur_c5

    max_step = 0.5
    min_step = 1e-4
    # temperature schedule for acceptance
    temp_start = 0.02
    temp_end = 0.0001
    
    iter = 0

    while time.time() - start_time < budget_s * 0.95 and iter < max_iter:
        # linearly schedule step and temperature
        frac = iter / max_iter
        step = max_step * (1 - frac) + min_step * frac

import time
import numpy as np

def _project_sum_to_target(x, target):
    """Clip to [0,1] and then adjust to hit the exact target sum."""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    # water‑filling on the free variables
    for _ in range(15):
        free = (x > 0.0) & (x < 1.0)
        nfree = int(free.sum())
        if nfree == 0:
            break
        delta = diff / nfree
        # limit the step so we stay in bounds
        add = np.where(delta > 0,
                       np.minimum(delta, 1.0 - x[free]),
                       np.maximum(delta, -x[free]))
        x[free] += add
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
    # final tiny clip (numeric safety)
    return np.clip(x, 0.0, 1.0)


def _c5_value(h):
    """Compute the Erdős C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _swap_move(h, idx_one, idx_zero):
    """Swap values at the two indices (keeps the vector binary)."""
    h[idx_one], h[idx_zero] = h[idx_zero], h[idx_one]


def _initial_binary_vector(n, seed):
    """Generate a binary vector of length n with exactly n/2 ones."""
    rng = np.random.default_rng(seed)
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and starting vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h_start = np.asarray(initial_h_values, dtype=np.float64)
        n = h_start.shape[0]
        target_sum = n / 2.0
        # force feasibility (clip + exact sum)
        h_start = _project_sum_to_target(h_start, target_sum)
        # keep it binary for the combinatorial search – rounding makes the
        # total exactly n/2 while staying within the box constraints.
        h_start = (h_start > 0.5).astype(np.float64)
    else:
        n = 300                     # even, modest size for fast evaluation
        target_sum = n / 2.0
        h_start = _initial_binary_vector(n, seed)

    # ------------------------------------------------------------------
    # Helper variables for the stochastic search
    # ------------------------------------------------------------------
    cur_h = h_start.copy()
    cur_c5 = _c5_value(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # pre‑compute index masks for fast random swaps
    def _indices_of(val):
        return np.flatnonzero(cur_h == val)

    # annealing schedule
    max_iter = 5_000_000          # generous upper bound; loop ends by time
    temp_start = 0.02
    temp_end = 0.0001
    accept_no_improve = 0

    it = 0
    while time.time() - start < budget_s * 0.95 and it < max_iter:
        # linear schedules for temperature and (optional) step size
        frac = it / max_iter
        temperature = temp_start * (1.0 - frac) + temp_end * frac

        # pick a random 1 and a random 0 and swap them
        ones = np.flatnonzero(cur_h == 1.0)
        zeros = np.flatnonzero(cur_h == 0.0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # try the swap
        _swap_move(cur_h, i, j)
        new_c5 = _c5_value(cur_h)

        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / (temperature + 1e-12)):
            # accept move
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = cur_h.copy()
                accept_no_improve = 0
            else:
                accept_no_improve += 1
        else:
            # reject – undo the swap
            _swap_move(cur_h, i, j)
            accept_no_improve += 1

        # occasional random perturbation if we have been stuck for a while
        if accept_no_improve > 20_000:
            # start again from the best known configuration and jiggle it a bit
            cur_h = best_h.copy()
            # perform a short burst of random swaps (no acceptance test)
            for _ in range(200):
                ones = np.flatnonzero(cur_h == 1.0)
                zeros = np.flatnonzero(cur_h == 0.0)
                if ones.size == 0 or zeros.size == 0:

                i = rng.choice(ones)
                j = rng.choice(zeros)
                _swap_move(cur_h, i, j)
            cur_c5 = _c5_value(cur_h)
            accept_no_improve = 0

        it += 1

    # final projection onto the feasible set (ensures exact sum)
    best_h = _project_sum_to_target(best_h, n / 2.0)
    final_c5 = _c5_value(best_h)

    return best_h.tolist(), float(final_c5), n
