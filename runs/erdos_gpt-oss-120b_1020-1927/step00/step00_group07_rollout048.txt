        if s > target:
            lo = mid

            proj = np.clip(proj, 0, 1)
        # after adjusting still may be off due to rounding -> ignore small.
    return proj

import time
import numpy as np
import scipy.optimize as _opt

# ---------- helper functions ----------
def _sigmoid(x: np.ndarray) -> np.ndarray:
    """Sigmoid mapping: ℝ → (0,1)."""
    return 1.0 / (1.0 + np.exp(-x))


def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project onto the intersection of the box [0,1]^n and the affine hyperplane sum = target.
    Uses bisection on the Lagrange multiplier λ for the problem:
        min‖clip(v‑λ,0,1) – v‖²  s.t.  sum = target
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny residual correction (adds to a free coordinate if any)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = np.where((proj > 0.0) & (proj < 1.0))[0]
        if free.size:
            proj[free[0]] += diff
            proj = np.clip(proj, 0.0, 1.0)
    return proj


class _SmoothObj:
    """
    Smooth approximation of max_k ∫h(1‑h∘shift) with a quadratic penalty for the integral constraint.
    """
    def __init__(self, alpha: float, n: int, dx: float, target_sum: float, lam: float):
        self.alpha = alpha
        self.n = n
        self.dx = dx
        self.target = target_sum
        self.lam = lam

    def __call__(self, y: np.ndarray):
        h = _sigmoid(y)
        B = 1.0 - h
        conv = np.correlate(h, B, mode='full')
        m = conv.max()
        # log‑sum‑exp trick
        exp_shifted = np.exp(self.alpha * (conv - m))
        sum_exp = exp_shifted.sum()
        p = exp_shifted / sum_exp
        smooth_max = m + np.log(sum_exp) / self.alpha

        sum_h = h.sum()
        penalty = self.lam * (sum_h - self.target) ** 2
        obj = smooth_max * self.dx + penalty

        # gradient w.r.t. h (double loop, n≤≈200 → cheap)
        n = self.n
        grad_h = np.zeros(n, dtype=np.float64)
        for i in range(n):
            hi = h[i]
            for j in range(n):
                idx = i - j + n - 1          # position in p
                w = p[idx]
                grad_h[i] += w * (1.0 - h[j])
                grad_h[j] += w * (-hi)

        grad_h = grad_h * self.dx + 2.0 * self.lam * (sum_h - self.target) * np.ones_like(h)

        # chain rule dy = dh/dy * grad_h, where dh/dy = h·(1‑h)
        dy = grad_h * h * (1.0 - h)
        return obj, dy

    try:
        return evaluate_erdos_solution(h)          # pre‑imported when available

        return np.correlate(h, 1.0 - h, mode='full').max() * dx
