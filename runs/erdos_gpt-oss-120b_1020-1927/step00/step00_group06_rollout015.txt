def propose(h):
    new_h = h.copy()
    n = h.size
    # choose random index i
    i = np.random.randint(n)
    # choose random j != i
    j = np.random.randint(n-1)
    if j >= i:
        j += 1
    # compute max transferable mass from i to j
    max_pos = min(h[i], 1 - h[j])
    max_neg = min(h[j], 1 - h[i])  # transferring opposite direction
    # total possible delta: between -max_neg and max_pos.
    if max_pos + max_neg == 0:
        return new_h
    # choose delta uniformly
    delta = np.random.uniform(-max_neg, max_pos)
    new_h[i] -= delta
    new_h[j] += delta
    return new_h
```

import numpy as np, time

def _repair(v):
    """Project v onto the box [0,1]^n with sum(v)=n/2."""
    n = v.shape[0]
    target = n * 0.5
    # binary search for shift lambda so that sum(clip(v+lam,0,1))=target
    lo, hi = -1.0, 1.0
    for _ in range(30):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    v = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # fix tiny residual by a uniform tweak on free entries
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        free = (v > 0.0) & (v < 1.0)
        if free.any():
            v[free] += diff / free.sum()
            v = np.clip(v, 0.0, 1.0)
        else:
            # fallback: adjust a random coordinate
            i = np.random.randint(n)
            v[i] = np.clip(v[i] + diff, 0.0, 1.0)
    return v

def _propose(v, rng):
    """Return a neighbour of v that respects the sum constraint."""
    n = v.shape[0]
    # choose move type
    if rng.random() < 0.7:                     # mass transfer between two indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_pos = min(v[i], 1.0 - v[j])       # i → j
        max_neg = min(v[j], 1.0 - v[i])       # j → i
        if max_pos + max_neg == 0.0:
            return v
        delta = rng.uniform(-max_neg, max_pos)
        w = v.copy()
        w[i] -= delta
        w[j] += delta
        return w
    else:                                      # small Gaussian noise on a few points
        w = v.copy()
        k = rng.integers(1, 5)                # up to 4 perturbed entries
        idx = rng.choice(n, size=k, replace=False)
        w[idx] += rng.normal(scale=0.04, size=k)
        w = np.clip(w, 0.0, 1.0)
        w = _repair(w)
        return w

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # -------------------------------------------------
    # initialise a feasible vector
    # -------------------------------------------------
    if 'initial_h_values' in globals():
        h_small = np.asarray(initial_h_values, dtype=np.float64)
        n_small = h_small.size
        n = max(200, n_small * 3)                     # upscale for finer search
        # linear interpolation to desired size
        xs = np.linspace(0.0, 2.0, n_small, endpoint=False)
        xl = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(xl, xs, h_small)
        h = np.clip(h, 0.0, 1.0)
        h = _repair(h)
    else:
        n = 300
        h = rng.random(n)
        h = _repair(h)

    # -------------------------------------------------
    # optimisation loop (simulated annealing)
    # -------------------------------------------------
    best_h = h.copy()
    best_score = evaluate_erdos_solution(best_h)
    cur_h = h.copy()
    cur_score = best_score

    start = time.time()
    max_iter = 500_000
    T0, T1 = 0.015, 1e-6

    for it in range(max_iter):
        if time.time() - start > budget_s * 0.98:
            break
        # temperature decay (exponential)
        T = T0 * ((T1 / T0) ** (it / max_iter))
        cand = _propose(cur_h, rng)
        cand_score = evaluate_erdos_solution(cand)

        # accept if better, otherwise with Boltzmann probability
        if cand_score < cur_score - 1e-12:

            if cand_score < best_score - 1e-12:

        else:

        # occasional diversification

    final_score = evaluate_erdos_solution(best_h)
    return best_h.astype(np.float32), float(final_score), n
