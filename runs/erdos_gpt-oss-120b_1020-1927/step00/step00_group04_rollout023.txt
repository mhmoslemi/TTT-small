def gradient_of_R(h, shift):
    n = len(h)
    g = np.ones(n)
    if shift >= 0:
        # i from 0 to n-1-shift: subtract h[i+shift]
        g[:n-shift] -= h[shift:]
        # i from shift to n-1: subtract h[i-shift]
        g[shift:] -= h[:n-shift]
    else:
        s = -shift
        # For negative shift, same as shift = -s
        # Equivalent to g[s:] -= h[:n-s]  (since i + shift = i - s)
        # Actually for negative shift, term h[i+shift] refers to h[i - s], which only exists if i-s >=0 => i>=s.
        # So subtract h[i-s] for i>=s.
        # Similarly subtract h[i - shift] = h[i + s] for i <= n-1-s.
        g[s:] -= h[:n-s]
        g[:n-s] -= h[s:]
    return g

    n = len(h0)
    target = n/2

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def gradient_R(h, shift):
    """gradient of R_shift = sum_i h_i (1 - h_{i+shift}) w.r.t. h."""
    n = h.shape[0]
    g = np.ones(n)
    if shift >= 0:
        s = shift
        if s < n:
            g[:n - s] -= h[s:]          # -h_{i+shift}
            g[s:] -= h[:n - s]          # -h_{i-shift}
    else:
        s = -shift
        if s < n:
            g[s:] -= h[:n - s]
            g[:n - s] -= h[s:]
    return g

def project_to_bounds_and_sum(x, target):
    """Euclidean projection of x onto [0,1]^n with sum(x)=target."""
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        free = np.where((y > 0.0) & (y < 1.0))[0]
        if free.size:
            y[free[0]] = np.clip(y[free[0]] + diff, 0.0, 1.0)
        else:
            y[0] = np.clip(y[0] + diff, 0.0, 1.0)
    return y

        j = int(np.argmax(corr))
