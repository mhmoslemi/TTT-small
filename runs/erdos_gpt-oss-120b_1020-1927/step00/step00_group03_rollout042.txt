def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------
    # helpers
    # ------------------------------------------------------------
    def compute_c5(arr):
        """Maximum overlap C5 for a step function given by arr."""
        corr = np.correlate(arr, 1.0 - arr, mode="full")
        return np.max(corr) * (2.0 / arr.size)

    # ------------------------------------------------------------
    # baseline from possible parent construction
    # ------------------------------------------------------------
    best_h = None

    best_n = None

        init_h = np.asarray(initial_h_values, dtype=float)
        n0 = init_h.size

        if cur_sum0 != 0:
            init_h = init_h * (target_sum0 / cur_sum0)
        init_h = np.clip(init_h, 0.0, 1.0)
        # fix tiny mass imbalance
        diff0 = target_sum0 - init_h.sum()
        if abs(diff0) > 1e-12:
            free0 = np.where((init_h > 1e-12) & (init_h < 1 - 1e-12))[0]
            if free0.size:
                init_h[free0[0]] = np.clip(init_h[free0[0]] + diff0, 0.0, 1.0)
            else:
                init_h[0] = np.clip(init_h[0] + diff0, 0.0, 1.0)

        init_c = compute_c5(init_h)

        best_n = n0

    dx = 2.0 / n

        # avoid division by zero

        c = 1.0 - 2.0 * b * p

        h[mask1] = p
