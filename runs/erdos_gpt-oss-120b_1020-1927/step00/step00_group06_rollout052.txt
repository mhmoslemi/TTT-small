def _correlation_array(h: np.ndarray) -> np.ndarray:
    """Full correlation of h with (1-h) multiplied by dx."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate implements linear correlation (no wrap‑around)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def _max_overlap(h: np.ndarray) -> float:
    """Maximum overlap integral C₅ for the given step function."""
    return float(np.max(_correlation_array(h)))


def _max_shift_info(h: np.ndarray):
    """
    Return (max_val, offset) where ``offset`` is the shift index counted
    from the centre (i.e. 0 corresponds to zero shift).
    """
    corr = _correlation_array(h)
    max_idx = int(np.argmax(corr))
    n = h.shape[0]
    offset = max_idx - (n - 1)          # centre of full mode is n-1
    return float(corr[max_idx]), int(offset)

    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # simple proportional scaling followed by a final correction on a single entry
        if cur_sum > 0:
            h *= target_sum / cur_sum
        else:  # degenerate case – fill first entry
            h[0] = target_sum
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        # distribute the tiny residual on a random coordinate that can absorb it
        idx = rng.integers(n)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

    best_h = h.copy()
    best_val = _max_overlap(best_h)

    # ------------------------------------------------------------------
    # Stochastic local search
    # ------------------------------------------------------------------
    TARGET = 0.38085               # known record – we stop if we beat it
    # control parameters (they decay with time)
    MAX_ITER = 10_000_000          # safety upper bound – loop also checks wall‑time
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.99:
            break
        if best_val <= TARGET:
            break

        # progress fraction determines mutation amplitude and temperature
        prog = elapsed / budget_s
        delta_scale = max(0.01, (1.0 - prog) * 0.15)   # typical step size
        temperature = max(1e-7, (1.0 - prog) * 0.02)  # for occasional uphill moves

        # --------------------------------------------------------------
        # Pick a move: 70 % targeted at the worst shift, 30 % random pair
        # --------------------------------------------------------------
        if rng.random() < 0.70:
            # Targeted move based on the shift that currently gives max overlap
            _, offset = _max_shift_info(h)
            if offset != 0:
                if offset > 0:
                    idx_range = np.arange(n - offset)
                    terms = h[idx_range] * (1.0 - h[idx_range + offset])
                    total = terms.sum()
                    if total > 0:
                        probs = terms / total
                        i = rng.choice(idx_range, p=probs)
                        j = i + offset
                    else:
                        i, j = rng.integers(n, size=2)
                else:  # offset < 0
                    off = -offset
                    idx_range = np.arange(off, n)
                    terms = h[idx_range] * (1.0 - h[idx_range - off])
                    total = terms.sum()
                    if total > 0:
                        probs = terms / total
                        i = rng.choice(idx_range, p=probs)
                        j = i - off
                    else:
                        i, j = rng.integers(n, size=2)
            else:
                i, j = rng.integers(n, size=2)
        else:
            i, j = rng.integers(n, size=2)

        if i == j:
            continue

        # --------------------------------------------------------------
        # Propose a mass transfer between i and j (preserve sum)
        # --------------------------------------------------------------
        hi, hj = h[i], h[j]

        # Randomly decide direction of transfer
        if rng.random() < 0.5:
            delta_max = min(hi, 1.0 - hj)
            if delta_max <= 0:
                continue
            delta = delta_max * rng.random() * delta_scale
            if delta <= 0:
                continue
            h_candidate = h.copy()
            h_candidate[i] -= delta
            h_candidate[j] += delta
        else:
            delta_max = min(hj, 1.0 - hi)
            if delta_max <= 0:
                continue
            delta = delta_max * rng.random() * delta_scale
            if delta <= 0:
                continue
            h_candidate = h.copy()
            h_candidate[i] += delta
            h_candidate[j] -= delta

        # --------------------------------------------------------------
        # Evaluate candidate
        # --------------------------------------------------------------
        cand_val = _max_overlap(h_candidate)

        # --------------------------------------------------------------
        # Acceptance rule (simulated annealing style)
        # --------------------------------------------------------------
        # Always accept strict improvements; otherwise accept with probability.
        if cand_val < best_val - 1e-12:
            best_val = cand_val
            best_h = h_candidate.copy()
        # use current state's max overlap as reference for uphill acceptance
        cur_val = _max_overlap(h)   # inexpensive relative to the whole loop
        if cand_val < cur_val - 1e-12:
            h = h_candidate
        else:
            # uphill acceptance probability
            prob = math.exp(-(cand_val - cur_val) / temperature)
            if rng.random() < prob:
                h = h_candidate

        iteration += 1
        if iteration >= MAX_ITER:
            break

    # ------------------------------------------------------------------
    # Final evaluation (use the provided helper if available)
    # ------------------------------------------------------------------
    final_bound = None
    try:
        final_bound = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_bound = _max_overlap(best_h)

    return best_h.tolist(), final_bound, n
