import time
import random
import math
import numpy as np

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic search for a low‑overlap step function.

    Returns
    -------
    h_values : list of float
        Discretised step function on [0,2].
    c5_bound : float
        Corresponding value of C5.
    n_points : int
        Number of samples (length of h_values).
    """
    # ------------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------------
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    time_limit = lambda: time.time() - start

    # --------------------------------------------------------------
    # Load / initialise h
    # --------------------------------------------------------------
    if 'initial_h_values' in globals() and initial_h_values is not None:
        # start from the given construction, double its resolution
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        # duplicate each entry to get a finer grid (45 -> 90)
        h = np.repeat(h0, 2)
    else:
        # fall‑back: random feasible vector with n=45
        n0 = 45
        h = np.random.rand(n0)
        h = np.clip(h, 0., 1.)
        target_sum = n0 / 2.0
        h *= target_sum / h.sum()
        # ensure exact sum after clipping
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            # distribute the tiny surplus uniformly
            h += diff / n0

    n = len(h)                     # length of the optimisation vector
    dx = 2.0 / n
    target_sum = n / 2.0           # required sum of h entries

    # ------------------------------------------------------------------
    # Helper: evaluate C5 for a candidate h
    # ------------------------------------------------------------------
    def evaluate(vec):
        corr = np.correlate(vec, 1.0 - vec, mode="full")
        return np.max(corr) * dx

    # ------------------------------------------------------------------
    # Project to feasibility (box + integral) – used rarely
    # ------------------------------------------------------------------
    def project(vec):
        vec = np.clip(vec, 0.0, 1.0)
        s = vec.sum()
        if s == 0:
            return vec
        # Scale to hit the exact integral, then clip again
        vec = vec * (target_sum / s)
        vec = np.clip(vec, 0.0, 1.0)
        # Small residual correction
        diff = target_sum - vec.sum()
        if abs(diff) > 1e-12:
            # distribute over non‑saturated entries
            free = np.where((vec > 0.0) & (vec < 1.0))[0]
            if free.size > 0:
                vec[free] += diff / free.size
                vec = np.clip(vec, 0.0, 1.0)
            else:
                vec *= target_sum / vec.sum()
        return vec

    # ------------------------------------------------------------------
    # Ensure start is feasible
    # ------------------------------------------------------------------
    h = project(h)
    best_h = h.copy()
    best_val = evaluate(best_h)
    cur_h = best_h.copy()
    cur_val = best_val

    # ------------------------------------------------------------------
    # Simulated annealing parameters
    # ------------------------------------------------------------------
    TARGET = 0.3808                     # desired bound
    T0 = 5e-3                           # initial temperature
    max_step = 0.5                      # maximal transfer amount (relative)
    step_decay = 5e4                    # exponential decay of step size
    total_budget = budget_s * 0.95      # keep a safety margin

    it = 0
    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time_limit() < total_budget:
        # Temperature linearly decays to zero
        frac = time_limit() / total_budget
        T = T0 * (1.0 - frac)

        # Step size decays exponentially
        step = max_step * math.exp(-it / step_decay)

        # Choose two distinct indices
        i, j = random.sample(range(n), 2)
        # Random direction of transfer
        if random.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        # Maximum admissible transfer respecting the box constraints
        max_delta = min(cur_h[src], 1.0 - cur_h[dst])
        if max_delta <= 0.0:
            it += 1
            continue

        # Random amount (scaled by current step)
        delta = random.random() * step * max_delta
        if delta <= 0.0:
            it += 1
            continue

        # Propose new vector
        new_h = cur_h.copy()
        new_h[src] -= delta
        new_h[dst] += delta

        # Evaluate
        new_val = evaluate(new_h)
        dobj = new_val - cur_val

        # Metropolis acceptance
        accept = False
        if dobj < 0.0:
            accept = True
        elif T > 0.0:
            accept = random.random() < math.exp(-dobj / T)

        if accept:
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
                if best_val <= TARGET:
                    break

        it += 1

    # ------------------------------------------------------------------
    # Final projection to ensure exact feasibility
    # ------------------------------------------------------------------
    best_h = project(best_h)
    final_val = evaluate(best_h)

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    return (best_h.tolist(), float(final_val), n)
