import numpy as np, time

def project_box_sum(v, target):
    # binary search
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_and_shift(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(conv)
    shift = idx - (n - 1)
    return conv[idx] * dx, shift

def compute_gradient(h, shift):
    n = len(h)
    g = np.zeros_like(h)
    if shift == 0:
        g[:] = 1.0 - 2.0 * h
        return g
    if shift > 0:
        # +1 term
        g[:n - shift] += 1.0
        # -h_{i+shift}
        g[:n - shift] -= h[shift:]
        # -h_{i-shift}
        g[shift:] -= h[:n - shift]
    else:
        t = -shift
        g[t:] += 1.0
        g[t:] -= h[:n - t]
        g[:n - t] -= h[t:]
    return g

import numpy as np
import time

def _project_box_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto {0≤x≤1, sum x = target_sum} using a simple bisection."""
    lo, hi = -1.0, 2.0  # guarantees that clipping will give sum=n and sum=0 respectively
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5(h: np.ndarray) -> float:
    """Return the overlap constant C5 for a given step vector h."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * (2.0 / n)


def _c5_and_shift(h: np.ndarray):
    """Return (C5 value, shift achieving the maximum) for h."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(conv))
    shift = idx - (n - 1)                 # shift ∈ [-(n-1), n-1]
    return conv[idx] * (2.0 / n), shift


def _gradient(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the active linear functional (the maximal shift)."""
    n = h.size
    g = np.zeros_like(h)
    if shift == 0:
        g[:] = 1.0 - 2.0 * h
        return g
    if shift > 0:
        # +1 for indices where the left term appears
        g[:n - shift] += 1.0
        # -h_{i+shift}
        g[:n - shift] -= h[shift:]
        # -h_{i-shift}
        g[shift:] -= h[:n - shift]
    else:
        t = -shift
        g[t:] += 1.0
        g[t:] -= h[:n - t]          # -h_{i+shift}
        g[:n - t] -= h[t:]          # -h_{i-shift}
    return g


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Optimise a step function for the Erdős C₅ problem."""
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.size
    else:
        n = 120                                 # a modest size when no starter is given
        h0 = np.random.rand(n)

    target = n / 2.0                           # ∫h = 1  ⇒  sum h = n/2
    h = _project_box_sum(h0, target)

    best_h = h.copy()
    best_val = _c5(best_h)

    start = time.time()

    # ------------------------------------------------------------------
    # Phase 1 – sub‑gradient descent on the active shift
    lr = 0.5                                    # initial step size
    phase1_deadline = start + budget_s * 0.6
    while time.time() < phase1_deadline:
        cur_val, shift = _c5_and_shift(h)

        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            if best_val <= 0.3808:
                break

        g = _gradient(h, shift)

        # backtracking line search
        step = lr
        for _ in range(10):
            candidate = _project_box_sum(h - step * g, target)
            cand_val = _c5(candidate)
            if cand_val <= cur_val:
                h = candidate
                lr = step * 1.02          # modest increase after success
                break
            step *= 0.5
        else:
            lr *= 0.5                      # all trials failed – shrink learning rate

        lr *= 0.9995                      # slow decay
        if lr < 1e-7:
            break

    # ------------------------------------------------------------------
    # Phase 2 – stochastic refinement (small mass transfers)
    cur_val = _c5(h)
    temp0 = 0.02                                 # initial temperature for SA

    while time.time() - start < budget_s * 0.98:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # propose a transfer of mass between i and j
        if np.random.rand() < 0.5:
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            max_delta = min(h[j], 1.0 - h[i])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta

        new_val = _c5(h_new)

        elapsed = time.time() - start
        T = temp0 * max(0.0, 1.0 - elapsed / budget_s)   # linear cooling

        accept = False
        if new_val < cur_val:
            accept = True
        elif T > 0:
            prob = np.exp(-(new_val - cur_val) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
                if best_val <= 0.3808:
                    break

    # ------------------------------------------------------------------
    # final safety projection and evaluation
    best_h = _project_box_sum(best_h, target)
    final_val = _c5(best_h)
    return best_h, final_val, n
