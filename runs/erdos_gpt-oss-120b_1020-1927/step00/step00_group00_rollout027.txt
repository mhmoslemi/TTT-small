import numpy as np

def _c5_of_h(h):
    """Compute the C₅ value for a discretised step function h."""
    dx = 2.0 / h.size
    # np.correlate gives the linear correlation of h with (1‑h)
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def run(seed=42, budget_s=1100, **kwargs):
    """
    Produce a step‑function h on [0,2] with ∫h = 1 that gives a
    provably small C₅ value.  The construction is the simple linear

    n_points = 200

    total = h.sum() * dx
