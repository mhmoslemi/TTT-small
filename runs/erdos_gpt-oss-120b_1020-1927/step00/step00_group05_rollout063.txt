def project_box_simplex(v, target):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        s = v.sum()
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        # identify indices that are not at bounds and can be moved in direction of diff
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        if not np.any(mask):
            # can't adjust further; break
            break
        # amount to add per item
        add = diff / np.sum(mask)
        # adjust
        v[mask] += add
        # clip again
        v = np.clip(v, 0.0, 1.0)
    return v

def compute_c5(h):
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / len(h))

def shift_array(v, k):
    n = v.shape[0]
    if k >= 0:
        out = np.empty_like(v)
        out[:n - k] = v[k:]
        out[n - k:] = 0.0
        return out
    else:
        k = -k
        out = np.empty_like(v)
        out[k:] = v[:n - k]
        out[:k] = 0.0
        return out

def project_to_sum(v, target):
    # Clip to [0,1] first
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Determine which entries can be adjusted
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        if not np.any(mask):
            break
        # Amount to add per allowed entry
        add = diff / np.count_nonzero(mask)
        v[mask] += add
        # Clip again
        v = np.clip(v, 0.0, 1.0)
    return v

def simulated_anneal(n, total_time, rng):
    # Build initial h as triangular shape
    dx = 2.0 / n
    x = np.arange(n) * dx + dx/2.0
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    # Project to sum constraint
    target_sum = n/2.0
    h = project_sum(h, target_sum)
    cur_h = h
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    start = time.time()
    # Parameters
    T0 = 0.2
    # move fraction start
    move_frac_start = 0.5
    while time.time() - start < total_time:
        elapsed = time.time() - start
        frac = elapsed / total_time
        T = T0 * (0.01 ** frac)  # decays to near zero
        move_frac = max(0.01, move_frac_start * (1 - frac))
        # propose move
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        if rng.random() < 0.5:
            max_move = min(cur_h[i], 1 - cur_h[j])
            if max_move <= 0:
                continue
            delta = rng.random() * max_move * move_frac
            cand = cur_h.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            max_move = min(cur_h[j], 1 - cur_h[i])
            if max_move <= 0:
                continue
            delta = rng.random() * max_move * move_frac
            cand = cur_h.copy()
            cand[i] += delta
            cand[j] -= delta
        # No need to enforce sum (still sum unchanged)
        # Clip for safety
        # (should already stay within bounds)
        # Evaluate
        cand_c5 = compute_c5(cand)
        delta_c = cand_c5 - cur_c5

            cur_h = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5

import time
import numpy as np

def _compute_c5_fft(h: np.ndarray) -> float:
    """Return C5 for a discrete step function h using FFT correlation."""
    n = h.shape[0]
    size = 2 * n - 1
    H = np.fft.fft(h, n=size)
    B = np.fft.fft(1.0 - h, n=size)
    corr = np.fft.ifft(H * np.conj(B)).real
    # max over full correlation, scaled by dx = 2/n
    return float(corr.max() * (2.0 / n))

def _shift_array(v: np.ndarray, k: int) -> np.ndarray:
    """Shift v by k with zero padding (v[i+k] at position i)."""
    n = v.shape[0]
    if k >= 0:
        out = np.empty_like(v)
        out[: n - k] = v[k:]
        out[n - k :] = 0.0
    else:
        k = -k
        out = np.empty_like(v)
        out[k:] = v[: n - k]
        out[:k] = 0.0
    return out

def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the hyperplane sum(v)=target while keeping 0≤v≤1."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(15):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        add = diff / cnt
        v[mask] += add
        v = np.clip(v, 0.0, 1.0)
    return v

def _simulated_annealing(n: int, time_budget: float, rng: np.random.Generator):
    """Search for a good h on a grid of n points within time_budget seconds."""
    # initialise with a symmetric triangular shape (area = 1)
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))
    h = _project_to_sum(h, n / 2.0)

    cur_h = h.copy()
    cur_c5 = _compute_c5_fft(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    start = time.time()
    # annealing schedule parameters
    T0 = 0.20          # initial temperature
    move_frac_start = 0.5
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = T0 * (0.01 ** frac)          # exponential cooling
        move_frac = max(0.01, move_frac_start * (1 - frac))

        i, j = rng.integers(n, size=2)
        if i == j:
            continue

        if rng.random() < 0.5:
            max_move = min(cur_h[i], 1.0 - cur_h[j])
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move * move_frac
            candidate = cur_h.copy()
            candidate[i] -= delta
            candidate[j] += delta
        else:
            max_move = min(cur_h[j], 1.0 - cur_h[i])
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move * move_frac
            candidate = cur_h.copy()
            candidate[i] += delta
            candidate[j] -= delta

        cand_c5 = _compute_c5_fft(candidate)
        d = cand_c5 - cur_c5
        if d <= 0.0 or rng.random() < np.exp(-d / (T + 1e-12)):
            cur_h = candidate
            cur_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
    return best_h, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a (h_values, c5_bound, n_points) tuple improving the C5 bound."""
    rng = np.random.default_rng(seed)
    overall_start = time.time()
    best_h = None
    best_c5 = float("inf")
    best_n = None

    # --- start from the provided initial construction if it exists ---

        best_h = init_h.copy()
        best_c5 = float(init_c5)
        best_n = init_h.shape[0]
    except Exception:
        pass

    # grid sizes to explore (keep them ≤ 1000)
    n_grid = [120, 200, 300, 400, 500]
    total_time = budget_s * 0.95
    per_n = total_time / len(n_grid)

    for n in n_grid:
        if time.time() - overall_start > total_time:
            break
        h_opt, c5_opt = _simulated_annealing(n, per_n, np.random.default_rng(seed + n))
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = float(c5_opt)
            best_n = n

    # final sanity projection (ensures constraints exactly)
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        target = best_n / 2.0
        diff = target - best_h.sum()
        if abs(diff) > 1e-8:

            for idx in range(best_n):
                if diff == 0.0:

                if diff > 0 and best_h[idx] < 1.0:
                    add = min(diff, 1.0 - best_h[idx])
                    best_h[idx] += add
                    diff -= add
                elif diff < 0 and best_h[idx] > 0.0:
                    sub = min(-diff, best_h[idx])
                    best_h[idx] -= sub
                    diff += sub

        final_c5 = _compute_c5_fft(best_h)
        best_c5 = float(final_c5)
    else:
        # fallback: uniform half‑density function
        best_n = 70
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c5 = _compute_c5_fft(best_h)

    return (best_h.tolist(), best_c5, int(best_n))
