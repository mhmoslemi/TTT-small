def project_onto_box_and_sum(v, target):
    # find offset s such that sum(clip(v - s,0,1)) = target
    low = -np.max(v)  # ensures v - low >= max(v) + something large > 1 => clipped 1
    high = np.max(v)   # ensures v - high <=0 => clipped 0
    for _ in range(50):  # enough iterations
        mid = (low + high) / 2
        x = np.clip(v - mid, 0, 1)
        s = np.sum(x)
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low+high)/2, 0, 1)

import time
import numpy as np

def _shift(v: np.ndarray, off: int) -> np.ndarray:
    """Shift vector v by integer offset off.
    Positive off → v[i+off] (elements shifted left, zeros on right).
    Negative off → v[i+off] (elements shifted right, zeros on left)."""
    n = v.shape[0]
    if off == 0:
        return v
    if off > 0:
        out = np.empty_like(v)
        out[: n - off] = v[off:]
        out[n - off :] = 0.0
        return out
    # off < 0
    k = -off
    out = np.empty_like(v)
    out[k:] = v[: n - k]
    out[:k] = 0.0
    return out


def _project_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the box [0,1] while enforcing sum(x)=target.
    Returns a vector y = clip(x - s,0,1) with sum(y)=target."""
    # binary search for offset s
    lo = -np.max(x)          # yields all ones after clipping
    hi = np.max(x)           # yields all zeros after clipping
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray, dx: float) -> float:
    """Compute the C5 overlap bound for vector h."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * dx
