import numpy as np
import time

# ---------- projection onto [0,1] with a fixed sum ----------
def _project(v: np.ndarray, target_sum: float) -> np.ndarray:
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


# ---------- correlation handled by FFT ----------
def _full_corr(h: np.ndarray, L: int) -> np.ndarray:
    a = np.fft.fft(h, L)
    b = np.fft.fft((1.0 - h)[::-1], L)
    conv = np.fft.ifft(a * b).real
    return conv[: 2 * len(h) - 1]


def _corr_max(h: np.ndarray, L: int):
    conv = _full_corr(h, L)
    idx = int(np.argmax(conv))
    return conv, conv[idx], idx


# ---------- gradient of a single shift ----------
def _grad_shift(h: np.ndarray, s: int) -> np.ndarray:
    n = len(h)
    i = np.arange(n)
    plus = i + s
    term1 = np.full(n, 1.0)
    mask = (plus >= 0) & (plus < n)
    term1[mask] = 1.0 - h[plus[mask]]

    minus = i - s
    term2 = np.zeros(n)
    mask2 = (minus >= 0) & (minus < n)
    term2[mask2] = h[minus[mask2]]
    return term1 - term2


# ---------- local optimisation from a start vector ----------
def _optimize_from(h0: np.ndarray, max_iter: int, L: int,
                   target_sum: float, dx: float,
                   start: float, budget: float):
    h = h0.copy()
    best_h = h.copy()
    conv, max_raw, _ = _corr_max(h, L)
    best_val = max_raw * dx
    step = 0.8
    n = len(h)

    for it in range(max_iter):
        if time.time() - start > budget * 0.95:
            break

        conv, max_raw, max_idx = _corr_max(h, L)
        cur_val = max_raw * dx

        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        # all shifts attaining the maximum (within numerical tolerance)
        tol = 1e-12
        max_idxs = np.where(np.abs(conv - max_raw) < tol)[0]

        grad = np.zeros(n)
        for idx in max_idxs:
            s = idx - (n - 1)
            grad += _grad_shift(h, s)
        grad /= max_idxs.size

        # gradient step + projection
        y = h - step * grad
        h_new = _project(y, target_sum)

        conv_new, max_raw_new, _ = _corr_max(h_new, L)
        new_val = max_raw_new * dx

        if new_val <= cur_val:
            h = h_new
        else:
            step *= 0.9
            if step < 1e-5:
                step = 0.8

    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)

    # use the provided construction if it exists
    init_parent = None
    if 'initial_h_values' in globals():
        init_parent = np.array(initial_h_values, dtype=float)
        n0 = init_parent.shape[0]
    else:
        n0 = None

    # target resolution (must be even, < 1000)
    n = 200
    if n0 is not None and n0 > n:

    if n % 2:
        n += 1

    dx = 2.0 / n
    target_sum = n / 2.0
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    start_time = time.time()

    best_h = None
    best_val = np.inf

    # ---------- build a few diverse starting vectors ----------
    starts = []

    # 1. up‑sample the given construction if possible
    if init_parent is not None:
        if init_parent.shape[0] == n:
            starts.append(init_parent.copy())
        else:
            # linear interpolation to the new grid
            x_old = np.linspace(0.0, 2.0, init_parent.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            up = np.interp(x_new, x_old, init_parent)

    xs = (np.arange(n) + 0.5) * dx

    bin_pat[: n // 2] = 1.0
    np.random.shuffle(bin_pat)

    rnd = np.random.rand(n)
    starts.append(_project(rnd, target_sum))

    for h0 in starts:

        max_iters = int(40000 * remaining / budget_s)
        if max_iters < 2000:

            h0, max_iters, L, target_sum, dx, start_time, budget_s
        )
        if val_opt < best_val:

            best_h = h_opt.copy()
        if best_val <= 0.3808:          # target already reached
            break

    # final projection (numerical safety) and exact evaluation
    best_h = _project(best_h, target_sum)
    conv, max_raw, _ = _corr_max(best_h, L)
    best_val = max_raw * dx

    return (best_h, best_val, n)
