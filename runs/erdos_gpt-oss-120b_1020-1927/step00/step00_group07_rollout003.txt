def project_onto_box_simplex(y, w):
    # find lambda such that sum clip(y_i+lambda,0,1) = w
    # Use binary search.
    lo = -1.0 - np.max(y)  # enough low to make all <=0
    hi = 1.0 - np.min(y)   # enough high to make all >=1
    for _ in range(50):
        mid = (lo+hi)/2
        x = np.clip(y+mid, 0,1)
        s = x.sum()
        if s < w:
            lo = mid
        else:
            hi = mid
    x = np.clip(y+ (lo+hi)/2, 0,1)
    return x

import numpy as np

def _c5_of_h(h, dx):
    """Return C5 = max correlation * dx for a given h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_box_sum(y, target_sum, eps=1e-12, max_iter=60):
    """
    Project y onto the feasible set:
        0 <= x_i <= 1  and  sum(x) = target_sum
    """
    lo = -np.max(y) - 5.0
    hi = 1.0 - np.min(y) + 5.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numeric drift
    diff = target_sum - x.sum()
    if abs(diff) > eps:
        # distribute the remaining mass to any coordinate that can accommodate it
        for i in range(len(x)):
            if diff > 0 and x[i] < 1.0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > 0.0:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) < eps:
                break
    return x

def _gradient_of_max_corr(h):
    """Gradient of the maximal correlation term (before multiplying by dx)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)          # shift value that attains the maximum
    g = np.zeros_like(h)
    if lag >= 0:
        i = np.arange(n - lag)
        g[i] += (1.0 - h[i + lag])
        g[i + lag] += -h[i]
    else:
        t = -lag
        i = np.arange(n - t)
        g[i + t] += (1.0 - h[i])
        g[i] += -h[i + t]
    return g, lag

    cur_c5 = best_c5
    no_improve = 0
    max_no_improve = 5000           # restart after this many non‑improving tries

    while time.time() - start < budget_s * 0.88:
        # pick a random 1‑index and a random 0‑index and swap them
        ones_idx = np.where(cur_h == 1.0)[0]
        zeros_idx = np.where(cur_h == 0.0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i_one = np.random.choice(ones_idx)
        i_zero = np.random.choice(zeros_idx)

        # perform the tentative swap
        cur_h[i_one] = 0.0
        cur_h[i_zero] = 1.0
        new_c5 = _c5_of_h(cur_h, dx)

        if new_c5 < cur_c5 - 1e-12:                 # improvement
            cur_c5 = new_c5
            no_improve = 0
            if cur_c5 < global_best_c5 - 1e-12:
                global_best_c5 = cur_c5
                global_best_h = cur_h.copy()
        else:                                       # reject
            cur_h[i_one] = 1.0
            cur_h[i_zero] = 0.0
            no_improve += 1

        # restart when stuck
        if no_improve >= max_no_improve:
            # start near the current global best with a few random swaps
            cur_h = global_best_h.copy()
            for _ in range(15):
                o = np.where(cur_h == 1.0)[0]
                z = np.where(cur_h == 0.0)[0]
                if len(o) == 0 or len(z) == 0:
                    break
                a = np.random.choice(o)
                b = np.random.choice(z)
                cur_h[a] = 0.0
                cur_h[b] = 1.0
            cur_c5 = _c5_of_h(cur_h, dx)
            no_improve = 0

        # safety break
        if time.time() - start > budget_s * 0.88:
            break

    # ------------------------------------------------------------------
    # Phase B – continuous refinement of the best binary pattern.
    # ------------------------------------------------------------------
    cur_h = global_best_h.astype(float)
    cur_c5 = global_best_c5
    max_iter_g = 3000
    alpha0 = 5.0                     # initial step magnitude

    for it in range(max_iter_g):
        if time.time() - start > budget_s * 0.97:
            break
        # gradient of the currently maximal correlation term
        grad, _ = _gradient_of_max_corr(cur_h)

        # decreasing step size
        alpha = alpha0 * (0.99 ** it)

        # gradient step then projection onto the feasible polytope
        y = cur_h - alpha * grad * dx
        cur_h = _project_box_sum(y, weight)

        new_c5 = _c5_of_h(cur_h, dx)
        if new_c5 < cur_c5 - 1e-12:
            cur_c5 = new_c5
            if new_c5 < global_best_c5 - 1e-12:
                global_best_c5 = new_c5
                global_best_h = cur_h.copy()
        # optional early stop when improvement stalls
        if it % 200 == 0 and alpha * np.linalg.norm(grad) * dx < 1e-6:
            # step becoming negligible – stop early
            break

    # ------------------------------------------------------------------
    # Return the best construction found.
    # ------------------------------------------------------------------
    h_values = global_best_h.tolist()
    c5_bound = float(global_best_c5)
    n_points = n
    return (h_values, c5_bound, n_points)
