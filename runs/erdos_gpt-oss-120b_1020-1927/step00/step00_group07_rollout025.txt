def project_onto_box_and_sum(v, target):
    # binary search lambda so sum(clip(v+lambda,0,1)) = target
    lo, hi = -1.0, 1.0

def gradient(h, w):
    n = len(h)
    grad = np.ones_like(h)
    for idx, weight in enumerate(w):
        if weight == 0:
            continue
        s = idx - (n - 1)
        if s >= 0:
            if n - s > 0:
                grad[:n-s] -= weight * h[s:]
                grad[s:] -= weight * h[:n-s]
        else:
            s_abs = -s
            if n - s_abs > 0:
                grad[s_abs:] -= weight * h[:n - s_abs]
                grad[:n - s_abs] -= weight * h[s_abs:]
    return grad

import time
import math
import numpy as np

def _max_corr(h):
    """Maximum of the overlap integral (without dx factor)."""
    return np.max(np.correlate(h, 1 - h, mode="full"))

def _project_box_sum(v, target):
    """
    Project v onto the set {x | 0 <= x_i <= 1, sum x_i = target}
    using a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0
    # 40 iterations give ~1e-12 precision, more than enough.
    for _ in range(40):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _gradient(h, weights):
    """
    Gradient of the smooth max objective.
    For each shift s we have contribution 1 - (h_{j+s}+h_{j-s}),
    weighted by the softmax weight.
    """
    n = h.shape[0]
    grad = np.ones_like(h)                     # the "1" part
    # iterate over all shifts s = idx-(n-1)
    for idx, w in enumerate(weights):
        if w == 0.0:
            continue
        s = idx - (n - 1)
        if s >= 0:
            if n - s > 0:
                grad[:n - s] -= w * h[s:]       # forward term
                grad[s:]     -= w * h[:n - s]   # backward term
        else:
            s = -s
            if n - s > 0:
                grad[s:]     -= w * h[:n - s]   # forward term (negative shift)
                grad[:n - s] -= w * h[s:]       # backward term
    return grad

    else:
