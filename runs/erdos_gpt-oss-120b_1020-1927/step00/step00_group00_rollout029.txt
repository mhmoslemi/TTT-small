def anneal(h, dx, rng, max_iter, init_temp):
    current_score = compute_c5(h,dx)
    best_local_score = current_score
    best_local_h = h.copy()
    for it in range(max_iter):
        # temperature schedule
        T = init_temp * (1 - it / max_iter) # linear cooling
        # propose swap
        i = rng.choice(np.where(h==1)[0])
        j = rng.choice(np.where(h==0)[0])
        # swap
        h[i], h[j] = 0, 1
        new_score = compute_c5(h,dx)
        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_score = new_score
            # accept move
            if new_score < best_local_score:
                best_local_score = new_score
                best_local_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1, 0
    return best_local_h, best_local_score

import numpy as np, time, math

def _c5_of(h):
    """C5 = max correlation * dx for binary/continuous h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _rand_binary(n, rng):
    """Even n, exactly n/2 ones."""
    h = np.zeros(n, dtype=np.int8)
    ones = n // 2
    idx = rng.choice(n, size=ones, replace=False)
    h[idx] = 1
    return h.astype(float)

def _thue_morse(n):
    """Generate first n bits of the Thue‑Morse sequence (0/1)."""
    s = np.array([0], dtype=np.int8)
    while s.size < n:
        s = np.concatenate([s, 1 - s])
    return s[:n].astype(float)

def _adjust_to_target(h, target, rng):
    """Flip random entries until sum(h) == target (target integer)."""
    cur = int(round(np.sum(h)))
    diff = target - cur
    if diff == 0:
        return h
    idx_one = np.where(h == 1)[0]
    idx_zero = np.where(h == 0)[0]
    for _ in range(abs(diff)):
        if diff > 0:                # need more ones
            i = rng.choice(idx_zero)
            h[i] = 1
            idx_zero = np.delete(idx_zero, np.where(idx_zero == i))
            idx_one = np.append(idx_one, i)
        else:                       # need fewer ones
            i = rng.choice(idx_one)
            h[i] = 0
            idx_one = np.delete(idx_one, np.where(idx_one == i))
            idx_zero = np.append(idx_zero, i)
    return h

def _anneal(h0, dx, rng, max_iter, init_temp):
    """Simple simulated annealing on binary vectors preserving sum."""
    h = h0.copy()
    best_h = h.copy()
    cur_score = _c5_of(h)
    best_score = cur_score
    n = h.shape[0]
    for it in range(max_iter):
        t = init_temp * (1.0 - it / max_iter)
        # pick a random 1 and a random 0 to swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_score = _c5_of(h)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / max(t, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0
    return best_h, best_score
