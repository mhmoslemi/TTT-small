def propose_move(h, rng, max_step):
    N = len(h)
    i, j = rng.choice(N, size=2, replace=False)
    hi, hj = h[i], h[j]
    max_inc_i = 1 - hi   # can increase at most this
    max_dec_i = hi       # can decrease at most this
    max_inc_j = 1 - hj
    max_dec_j = hj
    # we will move delta from j to i (i gains, j loses)
    # So delta ∈ [ -max_dec_i, max_inc_i ] (increase i) and delta ∈ [ -max_inc_j, max_dec_j ] (increase j is negative case)
    # Equivalent feasible interval:
    low = max(-hi, hj - 1)   # if delta negative, i decreases hi by |delta|, j increases.
    high = min(1 - hi, hj)   # delta positive
    if low >= high:
        return None
    # Clip magnitude:
    dmax = max_step
    # feasible interval length = high-low. Set d = clamp(rand within, but limited by dmax)
    if high - low > 2 * dmax:
        # sample a delta within +/- dmax around a random point in interval
        center = rng.uniform(low + dmax, high - dmax)
        d = rng.uniform(-dmax, dmax) + center
        # ensure within bounds
        d = max(min(d, high), low)
    else:
        d = rng.uniform(low, high)
    new_h = h.copy()
    new_h[i] += d
    new_h[j] -= d
    return new_h

    # To increase overlap (decrease Corr), we need to increase hi for indices where partner[i] is large, and decrease hi where partner[i] is small.

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _max_corr_and_idx(h: np.ndarray):
    """Return (max_correlation, index_of_max) for the vector h."""
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx


def _project_to_sum(v: np.ndarray, target: float, max_iter: int = 10):
    """
    Project v onto the affine set sum(v) = target while keeping 0<=v<=1.
    Simple iterative redistribution; fast for our small problems.
    """
    for _ in range(max_iter):
        cur = v.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = np.where((v > 0) & (v < 1))[0]
        if free.size == 0:                     # all at a bound – distribute equally
            v += diff / v.size
            v = np.clip(v, 0.0, 1.0)
        else:
            add = diff / free.size
            v[free] += add
            v = np.clip(v, 0.0, 1.0)
    return v


def _target_shift(idx: int, n: int) -> int:
    """Convert correlation index to shift (can be negative)."""
    return idx - (n - 1)


def _partner_sums(h: np.ndarray, shift: int):
    """
    For a given shift, return for each position i the sum of the
    partner values that contribute to h[i] * h[i+shift] in the overlap.
    """
    n = h.size
    partner = np.zeros(n, dtype=float)
    if shift >= 0:
        # overlapping positions: i in [0, n-shift)
        partner[: n - shift] += h[shift:]
        partner[shift:] += h[: n - shift]
    else:
        s = -shift
        partner[s:] += h[: n - s]
        partner[: n - s] += h[s:]
    return partner


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) – a step function improving the C5 bound."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size and initial vectors
    # --------------------------------------------------------------
    try:
        init_arr = np.asarray(initial_h_values, dtype=float)   # provided by harness
        n = init_arr.size
    except Exception:        # pragma: no cover
        # fallback size if no parent construction is available
        n = 45
        init_arr = None

    dx = 2.0 / n
    target_sum = n / 2.0                      # because ∫h = 1 ⇒ sum(h) = n/2

    candidates = []

    # 1) use the supplied initial construction (if any)
    if init_arr is not None:
        a = np.clip(init_arr, 0.0, 1.0)
        a = _project_to_sum(a, target_sum)
        candidates.append(a)

    # 2) a simple “V‑shaped” function (edges high, centre low)
    xs = (np.arange(n) + 0.5) * dx
    v = np.abs(xs - 1.0)                     # values in [0,1], edges = 1, centre = 0
    v = _project_to_sum(v, target_sum)
    candidates.append(v)

    # 3) a random feasible vector (useful for diversification)
    rand = rng.random(n)
    rand = rand / rand.sum() * target_sum
    rand = np.clip(rand, 0.0, 1.0)
    rand = _project_to_sum(rand, target_sum)
    candidates.append(rand)

    # --------------------------------------------------------------
    # Pick the best start among the candidates
    # --------------------------------------------------------------
    best_h = None
    best_c5 = float("inf")
    best_corr = None
    best_idx = None

    for h in candidates:
        corr_val, idx = _max_corr_and_idx(h)
        c5_val = corr_val * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            best_corr = corr_val
            best_idx = idx

    # --------------------------------------------------------------
    # Local search / simulated annealing
    # --------------------------------------------------------------
    # Parameters (tuned to work under the time budget)
    step_max = 0.08                 # maximal mass transferred per move
    temp = 0.005                    # initial temperature for occasional uphill moves
    decay = 0.99994                 # temperature decay factor per iteration
    min_temp = 1e-6
    no_improve = 0
    improve_limit = 2000
    target_c5 = 0.3808               # stop if we manage to go below this

    cur_h = best_h.copy()
    cur_corr = best_corr
    cur_idx = best_idx

    while time.time() - start_time < budget_s * 0.97:
        # --------------------------------------------------------------
        # Evaluate the worst lag (the one giving the current max correlation)
        # --------------------------------------------------------------
        shift = _target_shift(cur_idx, n)

        # --------------------------------------------------------------
        # Compute partner sums for that lag – guides a directed move
        # --------------------------------------------------------------
        partners = _partner_sums(cur_h, shift)

        # Choose an index to increase (high partner) and one to decrease (low partner)
        inc_mask = (cur_h < 1.0 - 1e-12) & (partners > np.median(partners))
        dec_mask = (cur_h > 1e-12) & (partners < np.median(partners))

        inc_candidates = np.where(inc_mask)[0]
        dec_candidates = np.where(dec_mask)[0]

        if inc_candidates.size and dec_candidates.size:
            i = int(rng.choice(inc_candidates))
            j = int(rng.choice(dec_candidates))
        else:
            # fallback to a random pair
            i, j = rng.choice(n, size=2, replace=False)

        hi, hj = cur_h[i], cur_h[j]
        low = max(-hi, hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:                     # infeasible – try a small random perturbation instead
            delta = 0.0
        else:
            # limit the step size to stay within a modest neighbourhood
            feasible = high - low
            max_step = min(step_max, feasible)
            delta = rng.uniform(-max_step / 2, max_step / 2)
            # shift delta into the feasible interval
            delta = max(low, min(high, delta))

        if abs(delta) > 1e-12:
            cand = cur_h.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta

            cand_corr, cand_idx = _max_corr_and_idx(cand)
            cand_c5 = cand_corr * dx

            # Accept improvements outright; occasionally accept worse moves
            accept = False
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()
                best_corr = cand_corr
                best_idx = cand_idx
                accept = True
                no_improve = 0
            else:
                prob = np.exp(-(cand_c5 - best_c5) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_h = cand
                cur_corr = cand_corr
                cur_idx = cand_idx
            else:
                no_improve += 1
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # Temperature schedule and occasional random perturbation
        # --------------------------------------------------------------
        temp = max(min_temp, temp * decay)

        if no_improve >= improve_limit:
            # inject a small random jitter while preserving the sum constraint
            noise = rng.normal(scale=0.02, size=n)
            jitter = cur_h + noise
            jitter = np.clip(jitter, 0.0, 1.0)
            jitter = _project_to_sum(jitter, target_sum)
            cur_h = jitter
            cur_corr, cur_idx = _max_corr_and_idx(cur_h)
            no_improve = 0

        # --------------------------------------------------------------
        # Early exit if we already beat the target bound
        # --------------------------------------------------------------
        if best_c5 < target_c5:
            break

    # final sanity – enforce constraints exactly (tiny numerical drift may appear)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_sum(best_h, target_sum)

    final_c5 = _max_corr_and_idx(best_h)[0] * dx
    return best_h.tolist(), float(final_c5), n
