import time
import numpy as np

def compute_c5(h):
    """Return max_{k} ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    # correlation h  * (1-h)  (zero‑padding outside the interval)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def project_to_sum_and_bounds(v, target, tol=1e-12, max_iter=60):
    """
    Project vector v onto the feasible set:
        0 <= w_i <= 1  and  sum(w) = target
    using a simple bisection for the Lagrange multiplier.
    """
    lo = -np.max(v) - 2.0
    hi = 1.0 - np.min(v) + 2.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - w.sum()
    if abs(diff) > tol:
        free = (w > 0.0) & (w < 1.0)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            w += diff / w.size
            w = np.clip(w, 0.0, 1.0)
    return w


def random_feasible(n, target):
    """Return a random vector in [0,1]^n with prescribed sum."""
    v = np.random.rand(n)
    return project_to_sum_and_bounds(v, target)


def golden_seed(n, phase):
    """Binary Sturmian word with slope (sqrt(5)-1)/2 and a phase shift."""
    alpha = (np.sqrt(5.0) - 1.0) / 2.0
    i = np.arange(n, dtype=np.float64)
    s = np.sin(2.0 * np.pi * alpha * i + phase) > 0
    return s.astype(float)
