def hill_climb(N, time_limit):
    # random initial with exactly N/2 ones
    h = np.zeros(N, dtype=np.int8)
    ones = np.random.choice(N, N//2, replace=False)
    h[ones]=1

                best_h = h.copy()
        else:
            # revert swap
            h[i]=1; h[j]=0
        # Optional: early stop if Omax <= target Omax
        # maybe break if Omax <= target

        frac = (time.time() - start) / time_limit
        T = T0 * (1 - frac) + T1 * frac

        i = random.choice(ones)
        j = random.choice(zeros)

        # propose swap
        h[i] = 0
        h[j] = 1
        o_new = _o_max(h)

        delta = o_new - o_curr
        accept = False
        if delta <= 0:
            accept = True
        else:
            if random.random() < math.exp(-delta / T):
                accept = True

        if accept:
            o_curr = o_new
            # update index lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)

            if o_new < best_o:
                best_o = o_new
                best_h = h.copy()
                if best_o <= _target_o(n):
                    break
        else:
            # revert swap
            h[i] = 1
            h[j] = 0

    return best_h, best_o

def _search_for_n(n: int, time_budget: float) -> tuple:
    """Run several hill‑climb attempts within the allotted time."""
    best_h, best_o = None, n
    start = time.time()
    # each attempt gets a slice of the remaining budget (min 4 s)
    while time.time() - start < time_budget:
        slice_time = min(6.0, time_budget - (time.time() - start))
        if slice_time <= 0:
            break
        h, o = _hill_climb_one_start(n, slice_time)
        if o < best_o:
            best_o, best_h = o, h
            if best_o <= _target_o(n):
                break
    return best_h, best_o

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function giving a low C₅ bound."""
    np.random.seed(seed)
    random.seed(seed)

    overall_start = time.time()
    best_h, best_o, best_n = None, None, None

    # evaluate the supplied construction, if any
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.ndim == 1:
            n0 = init.shape[0]
            # enforce the integral constraint by simple scaling (tiny adjustments)
            total = init.sum()
            if total != 0:
                init = init * (n0 / 2) / total
            # clip to [0,1] and recompute O_max (works for floats as well)
            init = np.clip(init, 0.0, 1.0)
            o0 = int(np.max(np.correlate(init, 1 - init, mode="full")))
            best_h, best_o, best_n = init, o0, n0

    # list of sample sizes to try (moderate lengths give enough freedom)
    candidate_ns = [200, 250, 300, 160, 180, 120, 140, 100]

    for n in candidate_ns:
        now = time.time()
        remaining = budget_s - (now - overall_start) - 0.1
        if remaining <= 0:
            break

        # allocate a share of the remaining budget; keep a minimum per size
        alloc = min(remaining / (len(candidate_ns) - candidate_ns.index(n)), remaining)
        h, o = _search_for_n(n, alloc)

        if h is not None:
            if best_o is None or o < best_o:
                best_h, best_o, best_n = h.astype(float), o, n
                # quick exit if we already beat the target constant
                if _c5_from_o(best_o, best_n) <= 0.3808:
                    break

    # fallback: simple half‑ones pattern if nothing was found
    if best_h is None:
        n = 100
        h = np.zeros(n, dtype=np.int8)
        h[: n // 2] = 1
        np.random.shuffle(h)
        best_h, best_o, best_n = h.astype(float), _o_max(h), n

    final_c5 = _c5_from_o(best_o, best_n)
    # ensure exact integral constraint (sum = n/2) – rounding issues are tiny for binary vectors
    return (best_h.tolist(), float(final_c5), int(best_n))
