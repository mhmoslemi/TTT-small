import time
import math
import numpy as np

def _evaluate(h):
    """
    Evaluate C5 for a given step function array `h`.
    Uses the pre‑imported evaluator if available, otherwise falls back
    to the specification (full correlation multiplied by dx).
    """
    try:
        # Provided by the harness, may be missing in isolated testing.
        return evaluate_erdos_solution(h)
    except Exception:
        # fallback implementation (linear convolution with zero padding)
        n = h.shape[0]
        dx = 2.0 / n
        return np.max(np.correlate(h, 1 - h, mode="full") * dx)


def _initial_h(n):
    """
    Construct a deterministic feasible start when the harness does not
    supply `initial_h_values`.  A simple triangular shape satisfies the
    integral constraint exactly.
    """
    # triangle: 0→1 at centre→0, area = 1 (integral requirement)
    x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n  # mid‑point sample
    h = np.where(x <= 1.0, x, 2.0 - x)          # linear rise/fall
    # ensure exact integral
    target = n / 2.0
    cur = h.sum()
    if cur != 0:
        h *= target / cur
    # clip to [0,1] (the triangle already respects this)
    np.clip(h, 0.0, 1.0, out=h)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        # size 80 is a good default; adjust if needed
        n = 80
        h = _initial_h(n)

    n = h.shape[0]
    target_sum = n / 2.0                     # because ∫h = 1  ⇒  Σh·dx = 1
    # enforce feasibility (tiny numerical drift may appear)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        # distribute the difference on a random feasible index
        idx = rng.choice(np.where((h > 0) & (h < 1))[0]) if diff < 0 else \
              rng.choice(np.where(h < 1)[0])
        h[idx] += diff
        h[idx] = np.clip(h[idx], 0.0, 1.0)

    # ------------------------------------------------------------------
    # initial evaluation
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_score = _evaluate(best_h)
    cur_h = h.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------------
    start_time = time.time()
    time_limit = budget_s * 0.94          # keep a little margin
    max_iter = 10_000_000                 # safety cap
    # simulated‑annealing style temperature schedule
    T0 = 0.02
    T_end = 5e-5

    # ------------------------------------------------------------------
    # helper for a feasible random delta move
    # ------------------------------------------------------------------
    def propose_move(arr):
        """Return (i_inc, i_dec, delta) respecting box constraints."""
        # indices that can be increased / decreased
        inc_candidates = np.where(arr < 1.0 - 1e-12)[0]
        dec_candidates = np.where(arr > 1e-12)[0]
        if inc_candidates.size == 0 or dec_candidates.size == 0:
            return None
        i_inc = rng.choice(inc_candidates)
        i_dec = rng.choice(dec_candidates)
        # maximum feasible transfer
        max_inc = 1.0 - arr[i_inc]
        max_dec = arr[i_dec]
        max_delta = min(max_inc, max_dec)
        if max_delta <= 0:
            return None
        delta = rng.random() * max_delta
        return i_inc, i_dec, delta

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    it = 0
    no_improve = 0
    while time.time() - start_time < time_limit and it < max_iter:
        it += 1
        # linear temperature decay
        progress = it / max_iter
        T = T0 * (1 - progress) + T_end * progress

        move = propose_move(cur_h)
        if move is None:
            break
        i_inc, i_dec, delta = move

        # apply move
        cur_h[i_inc] += delta
        cur_h[i_dec] -= delta

        new_score = _evaluate(cur_h)
        delta_score = new_score - cur_score

        if delta_score < 0 or rng.random() < math.exp(-delta_score / T):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # reject – revert
            cur_h[i_inc] -= delta
            cur_h[i_dec] += delta
            no_improve += 1

        # occasional small random perturbation to escape plateaus
        if no_improve > 20000:
            # re‑seed from the best solution with a tiny shake
            cur_h = best_h.copy()
            for _ in range(5):
                shake = propose_move(cur_h)
                if shake:
                    ii, jj, d = shake
                    cur_h[ii] += d
                    cur_h[jj] -= d
            cur_score = _evaluate(cur_h)
            no_improve = 0

    # ------------------------------------------------------------------
    # final sanity check – enforce exact integral
    # ------------------------------------------------------------------
    final_sum = best_h.sum()
    diff = target_sum - final_sum
    if abs(diff) > 1e-9:
        # correct the tiny deviation on a random feasible index
        feasible = np.where((best_h > 0) & (best_h < 1))[0]
        if feasible.size == 0:
            feasible = np.arange(n)
        idx = rng.choice(feasible)
        best_h[idx] = np.clip(best_h[idx] + diff, 0.0, 1.0)

    # final evaluation (using the same routine as the loop)
    final_score = _evaluate(best_h)

    return best_h, float(final_score), int(n)
