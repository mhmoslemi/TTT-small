def project_to_box_and_sum(v, target_sum):

    lo = -1.0
    hi = 1.0

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_box_and_sum(v, target_sum, tol=1e-12, max_iter=60):
    # projection onto {0 <= x_i <= 1, sum x_i = target_sum}
    lo = -1.0
    hi = 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny adjustment for numerical error
    diff = target_sum - proj.sum()
    if abs(diff) > tol:
        free = np.where((proj > 1e-12) & (proj < 1 - 1e-12))[0]
        if free.size > 0:
            proj[free] += diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj

import numpy as np
import time

def compute_c5(h):
    """C5 for a discrete step function h on [0,2]."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def project_to_box_and_sum(v, target_sum, tol=1e-12, max_iter=60):
    """
    Euclidean projection of a vector onto the set
        {x : 0 <= x_i <= 1, sum x_i = target_sum}.
    The solution has the form x_i = clip(v_i + λ, 0, 1) for a suitable λ.
    """
    lo, hi = -1.0, 1.0  # λ limits are safe for 0/1 box
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target_sum - proj.sum()
    if abs(diff) > tol:
        free = np.where((proj > 1e-12) & (proj < 1 - 1e-12))[0]
        if free.size:
            proj[free] += diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def random_feasible(N, target_sum=None):

    v = np.random.rand(N)

def _gradient_worst_shift(h, shift):
    """
    Gradient of   f_s(h) = Σ_i h_i (1 - h_{i+shift})
    with respect to h, for a given non‑negative integer shift.
    The gradient points to *increase* f_s; we will move opposite.
    """
    N = h.shape[0]
    grad = np.zeros(N)
    if shift == 0:
        # f_0 = Σ h_i - Σ h_i^2
        grad[:] = 1.0 - 2.0 * h
        return grad

    overlap = N - shift
    if overlap <= 0:
        return grad  # no overlap – gradient is zero

    # linear part Σ h_i contributes +1 on the overlapping segment
    grad[:overlap] += 1.0

    # product part Σ h_i h_{i+shift}
    grad[:overlap] -= h[shift:]          # derivative from h_i term
    grad[shift:] -= h[:N - shift]       # derivative from h_{i+shift} term
    return grad
