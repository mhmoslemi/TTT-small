def project_onto_sum_and_box(v, target):
    low = -np.max(v)  # to ensure v+lambda >=0 for all
    high = 1 - np.min(v)  # to ensure <=1
    # Binary search for lambda such that sum(clip(v+λ,0,1)) = target

        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            high = mid
        else:
            low = mid
    return np.clip(v + mid, 0.0, 1.0)

import numpy as np, time, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_box_sum(v, target):
    low = -np.max(v)   # enough to bring all >=0
    high = 1 - np.min(v)  # enough to bring all <=1
    for _ in range(50):
        lam = (low + high) / 2.0
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            high = lam
        else:
            low = lam
    return np.clip(v + lam, 0.0, 1.0)

def random_feasible(n, rng):
    v = rng.random(n)
    target = n / 2.0
    v = project_box_sum(v, target)
    return v

def subgradient_descent(h0, max_iters, step0, step_decay, time_limit):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    step = step0
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr) - (n - 1)
        # gradient
        g = np.ones(n)
        if k >= 0:
            if k < n:
                g[:n - k] -= h[k:]
                g[k:] -= h[:n - k]
        else:
            k = -k
            if k < n:
                g[k:] -= h[:n - k]
                g[:n - k] -= h[k:]
        h_new = h - step * g
        h_new = project_box_sum(h_new, target)
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
        step *= step_decay
        if step < 1e-8:
            break
    return best_h, best_val

import time
import math
import numpy as np

# ---------- helper functions ----------

def compute_c5(h):
    """Return C5 for a step function given by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)


def project_box_sum(v, target):
    """
    Project vector v onto the intersection of the box [0,1]^n
    and the hyperplane sum(v) = target.
    Uses a simple bisection on the Lagrange multiplier.
    """
    low = -np.max(v)          # ensure v+lam >= 0
    high = 1.0 - np.min(v)    # ensure v+lam <= 1
    for _ in range(60):
        lam = (low + high) * 0.5
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            high = lam
        else:
            low = lam
    return np.clip(v + lam, 0.0, 1.0)


def random_feasible(n, rng):
    """Generate a random feasible vector of length n."""
    v = rng.random(n)               # uniform in [0,1)
    target = n * 0.5
    return project_box_sum(v, target)


def subgradient_descent(h0, max_iters, step0, decay, time_limit):
    """
    Projected sub‑gradient descent on the max–overlap objective.
    Stops early if the allotted time expires.
    """
    n = h0.shape[0]
    target = n * 0.5
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    step = step0
    start = time.time()

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break

        # locate the shift attaining the maximal overlap
        corr = np.correlate(h, 1.0 - h, mode='full')
        k = np.argmax(corr) - (n - 1)          # integer shift

        # sub‑gradient of the max term
        g = np.ones(n)
        if k >= 0:
            if k < n:
                g[:n - k] -= h[k:]
                g[k:]   -= h[:n - k]
        else:
            kk = -k
            if kk < n:
                g[kk:] -= h[:n - kk]
                g[:n - kk] -= h[kk:]

        # gradient step and projection
        h_new = h - step * g
        h_new = project_box_sum(h_new, target)

        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()

        h = h_new
        step *= decay
        if step < 1e-9:
            break

    return best_h, best_val


def upgrade_resolution(h, new_n):
    """Linear interpolation to a higher resolution followed by projection."""
    old_n = h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_interp = np.interp(x_new, x_old, h)
    return project_box_sum(h_interp, new_n * 0.5)


# ---------- main routine ----------

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s

    # initialise from the provided solution if it exists
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float)
    else:
        h_start = random_feasible(45, rng)

    n = h_start.shape[0]
    h_start = project_box_sum(h_start, n * 0.5)
    best_h = h_start.copy()
    best_val = compute_c5(best_h)

    # ---- deterministic descent from the initial point ----
    remaining = end_time - time.time()
    if remaining > 5.0:
        # generous number of iterations; step decays slowly
        best_h, best_val = subgradient_descent(
            best_h, max_iters=30000, step0=0.20, decay=0.999,
            time_limit=remaining
        )

    # ---- random‑restart hill climbing (subgradient) ----
    while time.time() < end_time - 1.0:
        remaining = end_time - time.time()
        if remaining < 2.0:
            break

        # create a fresh feasible start (occasionally blend with current best)
        if rng.random() < 0.25:
            mix = rng.random()
            h0 = mix * best_h + (1.0 - mix) * random_feasible(n, rng)
            h0 = project_box_sum(h0, n * 0.5)
        else:
            h0 = random_feasible(n, rng)

        # spend a few seconds on each restart
        budget = min(5.0, remaining)
        h_opt, val_opt = subgradient_descent(
            h0, max_iters=5000, step0=0.20, decay=0.995, time_limit=budget
        )
        if val_opt < best_val:
            best_val, best_h = val_opt, h_opt.copy()
            if best_val <= 0.3808:
                break

    # ---- optional refinement at higher resolution ----
    if time.time() < end_time - 5.0:
        # double the number of points
        h_hi = upgrade_resolution(best_h, n * 2)
        hi_n = h_hi.shape[0]
        remaining = end_time - time.time()
        # a short descent on the finer grid
        h_hi, val_hi = subgradient_descent(
            h_hi, max_iters=20000, step0=0.15, decay=0.999,
            time_limit=remaining
        )
        if val_hi < best_val:
            best_h, best_val, n = h_hi, val_hi, hi_n

    # final safety projection and evaluation
    best_h = project_box_sum(best_h, best_h.shape[0] * 0.5)
    best_val = compute_c5(best_h)

    # return as required: (h_values, c5_bound, n_points)
    return (best_h.tolist(), float(best_val), best_h.shape[0])
