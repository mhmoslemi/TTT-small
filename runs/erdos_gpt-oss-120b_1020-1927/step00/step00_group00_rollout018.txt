def gradient_at_shift(h, k):
    n = len(h)
    g = np.ones_like(h)
    idx = np.arange(n)
    plus = idx + k
    minus = idx - k
    valid_plus = (plus >= 0) & (plus < n)
    valid_minus = (minus >= 0) & (minus < n)
    g[valid_plus] -= h[plus[valid_plus]]
    g[valid_minus] -= h[minus[valid_minus]]
    return g
```

def project_onto_bounds_and_sum(v, S, low=0.0, high=1.0, max_iter=60):
    # find λ such that sum clip(v - λ) == S
    lam_low = np.min(v) - high
    lam_high = np.max(v) - low
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        w = np.clip(v - lam, low, high)
        s = w.sum()
        if s > S:
            lam_low = lam
        else:
            lam_high = lam
    return w
```

def random_local_search(h, best_val, S, time_budget):

    while time.time() - start < time_budget:

        if i == j:
            continue
        lower = max(-cur[i], cur[j] - 1.0)
        upper = min(1.0 - cur[i], cur[j])

        new[j] -= delta
        new_val = compute_c5(new)
        if new_val < cur_val - 1e-12:
            cur = new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:

                best_val = new_val

def compute_c5(h):
    """Return the Erdős C5 value of a step function h."""
    # evaluate_erdos_solution is pre‑imported in the execution environment

def max_corr_shift(h):
    """Return the lag k (int) giving the maximal overlap and the raw max value."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    i = int(np.argmax(corr))
    k = i - (n - 1)
    return k, float(corr[i])


def gradient_at_shift(h, k):
    """Gradient of f_k(h)=∫h(x)(1‑h(x+k))dx w.r.t. h (unscaled by dx)."""
    n = len(h)
    g = np.ones(n, dtype=np.float64)
    idx = np.arange(n)
    plus = idx + k
    minus = idx - k
    mask_plus = (plus >= 0) & (plus < n)
    mask_minus = (minus >= 0) & (minus < n)
    g[mask_plus] -= h[plus[mask_plus]]
    g[mask_minus] -= h[minus[mask_minus]]
    return g


def project_onto_bounds_and_sum(v, S, low=0.0, high=1.0, max_iter=60):
    """
    Euclidean projection of v onto the box [low,high] together with the affine
    constraint sum(x)=S.
    """
    lam_low = np.min(v) - high      # sum will be n*high > S
    lam_high = np.max(v) - low      # sum will be 0 < S
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        w = np.clip(v - lam, low, high)
        s = w.sum()
        if s > S:
            lam_low = lam
        else:
            lam_high = lam
    return w


def projected_gradient_step(h, k, alpha, S):
    """One projected gradient descent step on the active lag k."""
    g = gradient_at_shift(h, k)
    v = h - alpha * g
    return project_onto_bounds_and_sum(v, S)


def random_feasible(n, S):
    """Draw a random point in [0,1]^n with sum exactly S."""
    v = np.random.rand(n)
    return project_onto_bounds_and_sum(v, S)


def gradient_descent(start_h, S, time_budget):
    """Simple projected gradient descent on the active lag."""
    cur = start_h.copy()
    cur_val = compute_c5(cur)
    best = cur.copy()
    best_val = cur_val
    alpha = 0.2                     # initial step size
    start = time.time()
    while time.time() - start < time_budget:
        k, _ = max_corr_shift(cur)
        new = projected_gradient_step(cur, k, alpha, S)
        new_val = compute_c5(new)
        if new_val < cur_val - 1e-12:
            cur = new
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best = new.copy()
                best_val = new_val
        else:
            # shrink step size if we cannot improve
            alpha *= 0.9
        # slow exponential decay to keep a foothold
        alpha = max(alpha * 0.9999, 1e-4)
    return best, best_val


def random_local_search(start_h, start_val, S, time_budget):
    """Simulated‑annealing style pairwise move search."""
    cur = start_h.copy()
    cur_val = start_val
    best = cur.copy()
    best_val = start_val
    n = len(cur)
    start = time.time()
    while time.time() - start < time_budget:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # keep sum constant: h[i] += δ, h[j] -= δ
        lower = max(-cur[i], cur[j] - 1.0)
        upper = min(1.0 - cur[i], cur[j])
        if lower >= upper:
            continue
        delta = np.random.uniform(lower, upper)
        new = cur.copy()
        new[i] += delta
        new[j] -= delta
        new_val = compute_c5(new)
        if new_val < cur_val:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new.copy()
                best_val = new_val
        else:
            # annealing acceptance
            elapsed = time.time() - start
            frac = 1.0 - elapsed / time_budget
            T = 0.02 * max(frac, 0.0)    # temperature decays to 0
            if T > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
                cur = new
                cur_val = new_val
    return best, best_val


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) satisfying the problem constraints."""
    np.random.seed(seed)
    random.seed(seed)

    # -----------------------------------------------------------------
    # Load initial construction if provided, otherwise start from 0.5
    # -----------------------------------------------------------------
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64)
    except NameError:
        # fallback: uniform 0.5 (feasible after projection)
        h0 = np.full(45, 0.5, dtype=np.float64)

    n = len(h0)
    S = n / 2.0                      # required sum of h_i (since dx = 2/n)

    # make sure the starting point satisfies all constraints
    h0 = project_onto_bounds_and_sum(h0, S)

    # -----------------------------------------------------------------
    # Phase 1 – gradient descent on the current best point
    # -----------------------------------------------------------------
    start_time = time.time()
    gd_time = min(budget_s * 0.4, max(5.0, budget_s * 0.4))
    best_h, best_val = gradient_descent(h0, S, gd_time)

    # -----------------------------------------------------------------
    # Phase 2 – extensive simulated‑annealing style search
    # -----------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = max(0.0, budget_s - elapsed)
    if remaining > 0.5:               # leave a tiny slack for finalisation
        best_h, best_val = random_local_search(best_h, best_val, S, remaining)

    # -----------------------------------------------------------------
    # Final projection (just in case) and final evaluation
    # -----------------------------------------------------------------
    best_h = project_onto_bounds_and_sum(best_h, S)
    final_c5 = compute_c5(best_h)

    # Return a plain Python list, the bound as a float, and the number of points
    return best_h.tolist(), float(final_c5), int(n)
