def gradient_smooth_max(h, alpha=20.0):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    # compute softmax probabilities
    # subtract max for numeric stability:
    max_corr = corr.max()
    # exponent:
    exp = np.exp(alpha * (corr - max_corr))
    w = exp / exp.sum()
    # compute t_i = correlation of w with (1 - h)
    t_full = np.correlate(w, 1-h, mode='full')
    t = t_full[n-1 : n-1 + n]
    # compute s_i = convolution of w with h, get segment
    conv_full = np.convolve(w, h, mode='full')
    # start index: len(w)-1 = 2n-2
    start = len(w) - 1
    s = conv_full[start : start + n]
    grad = t - s
    return grad, corr

def compute_c5(h):
    """Exact overlap bound for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def project_onto_constraints(y, target):
    """
    Projection of y onto the set {x | 0<=x_i<=1, sum x_i = target}.
    Uses a simple bisection on the Lagrange multiplier λ.
    """
    lo = -np.max(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)


def grad_smooth_max(h, alpha):
    """
    Gradient of a smooth approximation of max_lag sum_i h_i (1-h_{i+lag}),
    using softmax weights with parameter alpha.
    """
    n = h.shape[0]
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")          # length 2n-1
    max_corr = corr.max()
    w = np.exp(alpha * (corr - max_corr))
    w /= w.sum()                                          # softmax probabilities

    # t_i = sum_{lag} w_lag * (1 - h_{i+lag})
    t_full = np.correlate(w, 1.0 - h, mode="full")       # length 3n-2
    t = t_full[n - 1 : n - 1 + n]

    # s_i = sum_{lag} w_lag * h_{i-lag}
    conv_full = np.convolve(w, h, mode="full")          # length 3n-2
    start = len(w) - 1                                   # 2n-2
    s = conv_full[start : start + n]

    grad = t - s
    return grad, corr


def upsample_h(h, new_n):
    """Linear interpolation of h onto a finer uniform grid."""
    old_n = h.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)


def optimize_h(h0, target_sum, max_iter, time_limit, rng):
    """
    Projected gradient descent on the smoothed max objective.
    Returns the best (h, c5) seen within the allotted time.
    """
    n = h0.shape[0]
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        # schedule for smoothing and step size
        alpha = 5.0 + 75.0 * (it / max_iter)          # 5 → 80
        lr = 0.5 / math.sqrt(it + 1.0)               # decaying learning rate

        # occasional random nudges to avoid stagnation
        if it % 500 == 0 and it > 0:
            noise = rng.normal(scale=0.02, size=n)
            h = project_onto_constraints(h + noise, target_sum)

        grad, corr = grad_smooth_max(h, alpha)

        # gradient step + projection
        h_new = h - lr * grad
        h = project_onto_constraints(h_new, target_sum)

        c5 = corr.max() * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            if best_c5 <= 0.3808:        # target reached – stop early
                break

    return best_h, best_c5


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # ------- obtain an initial feasible construction -------------------
    init_h = None
    if "initial_h_values" in globals():
        init_h = np.array(initial_h_values, dtype=float)

    if init_h is None:
        # fall back to a random start with 200 points
        n0 = 200
        init_h = rng.random(n0)
    # project to satisfy sum = n/2
    n0 = init_h.shape[0]
    target0 = n0 / 2.0
    init_h = project_onto_constraints(init_h, target0)

    # ------- first optimisation pass on the original resolution -------
    remaining = budget_s - (time.time() - start_total)
    h_best, c5_best = optimize_h(init_h, target0, max_iter=20000,
                                 time_limit=remaining * 0.4, rng=rng)
    n_best = n0

    # early exit if already good enough
    if c5_best <= 0.3808:
        return (h_best.tolist(), c5_best, n_best)

    # ------- iterative refinement on finer grids ----------------------
    # we keep doubling the number of points while time permits.
    while True:
        elapsed = time.time() - start_total
        if elapsed >= budget_s:
            break
        if n_best >= 376:                 # cap to keep work reasonable
            break

        new_n = n_best * 2
        up_h = upsample_h(h_best, new_n)
        up_h = project_onto_constraints(up_h, new_n / 2.0)

        remaining = budget_s - (time.time() - start_total)
        if remaining <= 0:
            break

        # allocate a fraction of the remaining budget for this stage
        stage_time = min(remaining * 0.4, 250.0)
        # iteration count scaled with size (more points → fewer iterations)
        max_iter = int(25000 * (200.0 / new_n))

        h_opt, c5_opt = optimize_h(up_h, new_n / 2.0, max_iter,
                                   time_limit=stage_time, rng=rng)

        if c5_opt < c5_best:
            h_best, c5_best, n_best = h_opt, c5_opt, new_n

        if c5_best <= 0.3808:
            break

    # final bound recomputed (ensures consistency)
    final_c5 = compute_c5(h_best)

    return (h_best.tolist(), final_c5, n_best)
