import time
import numpy as np

def _c5_score(h: np.ndarray) -> float:
    """Return the C5 value for a step function sampled in h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _project_bounds_sum(x: np.ndarray,
                        lower: float = 0.0,
                        upper: float = 1.0,
                        target: float = 0.0) -> np.ndarray:
    """Project x onto the box [lower,upper] while enforcing sum(x)=target."""
    n = x.size
    # quick feasibility handling
    if target <= n * lower:
        return np.full_like(x, lower)
    if target >= n * upper:
        return np.full_like(x, upper)

    lo, hi = -1e6, 1e6
    for _ in range(60):                # binary search for the right shift
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, lower, upper)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) * 0.5, lower, upper)

    # small correction to hit the target sum (numerical)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        free = (y > lower + 1e-12) & (y < upper - 1e-12)
        if free.any():
            y[free] += diff / free.sum()
        else:
            y += diff / n
        y = np.clip(y, lower, upper)
    return y

def _upscale(h: np.ndarray, factor: int) -> np.ndarray:
    """Linearly up‑sample a vector by an integer factor."""
    n = h.size
    new_n = n * factor
    dx_small = 2.0 / n
    dx_large = 2.0 / new_n
    xs = np.arange(n) * dx_small
    xl = np.arange(new_n) * dx_large
    return np.interp(xl, xs, h, left=h[0], right=h[-1])
