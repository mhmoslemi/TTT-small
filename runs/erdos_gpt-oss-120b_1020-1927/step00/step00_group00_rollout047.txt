def propose_move(h, max_delta):
    i, j = np.random.choice(N, size=2, replace=False)
    # Determine allowable range for delta: delta in [ -h[i], 1-h[j] ]
    lo = -h[i]
    hi = 1 - h[j]

        return None
    delta = np.random.uniform(lo, hi)
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta

import time, math, random
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def max_overlap(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)(1-h(x+k))dx via discrete correlation."""
    dx = 2.0 / h.shape[0]
    # np.correlate already does Σ h_i * (1-h)_{i+k}
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def project_to_simplex_bounds(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust to meet exact sum using a simple water‑filling."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(15):
        s = x.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        n_free = int(free.sum())
        if n_free == 0:
            # nothing can move – break to avoid infinite loop
            break
        delta = diff / n_free
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # Final tiny correction (distribute residual uniformly)
    residual = target_sum - x.sum()
    if abs(residual) > 1e-12:
        idx = np.argmin(np.abs(x - 0.5))  # pick a central entry
        x[idx] = np.clip(x[idx] + residual, 0.0, 1.0)
    return x


def upsample(h: np.ndarray, factor: int) -> np.ndarray:
    """Linear interpolation to increase resolution by an integer factor."""
    n = h.shape[0]
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n * factor, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    # enforce constraints
    target = h_new.shape[0] / 2.0
    return project_to_simplex_bounds(h_new, target)


def perturb(h: np.ndarray, scale: float) -> np.ndarray:
    """Add a zero‑mean random perturbation of given amplitude."""
    n = h.shape[0]
    delta = np.random.randn(n)
    delta -= delta.mean()
    max_abs = np.abs(delta).max()
    if max_abs > 0:
        delta = delta / max_abs * scale
    h_new = h + delta
    h_new = np.clip(h_new, 0.0, 1.0)
    target = n / 2.0
    return project_to_simplex_bounds(h_new, target)


def propose_move(cur: np.ndarray, max_delta: float):
    """Generate a feasible move that preserves the total sum."""
    n = cur.shape[0]
    i, j = np.random.randint(n, size=2)
    if i == j:
        return None
    hi, hj = cur[i], cur[j]
    # allowable delta: subtract from i, add to j
    lo = -hi               # cannot go below 0 at i
    hi_bound = 1.0 - hj    # cannot exceed 1 at j
    lo = max(lo, -max_delta)
    hi_bound = min(hi_bound, max_delta)
    if lo >= hi_bound:
        return None
    delta = np.random.uniform(lo, hi_bound)
    new = cur.copy()
    new[i] = hi - delta
    new[j] = hj + delta
    return new


def optimize_h(start_h: np.ndarray, time_limit: float):
    """Heuristic (simulated‑annealing) search for a lower C5."""
    n = start_h.shape[0]
    target_sum = n / 2.0
    cur = start_h.copy()
    cur_val = max_overlap(cur)
    best = cur.copy()
    best_val = cur_val

    start = time.time()
    iter_cnt = 0
    max_delta_initial = 0.2

    while time.time() - start < time_limit:
        # linearly decay max_delta
        frac = min(iter_cnt / 200000.0, 1.0)
        max_delta = max_delta_initial * (1.0 - frac) + 0.001 * frac

        cand = propose_move(cur, max_delta)
        if cand is None:
            continue

        cand_val = max_overlap(cand)

        # simple SA acceptance
        if cand_val < cur_val:
            accept = True
        else:
            # temperature schedule (very low to keep occasional uphill moves)
            T = 0.001 * (1.0 - frac) + 1e-6
            accept = random.random() < math.exp((cur_val - cand_val) / T)

        if accept:
            cur = cand
            cur_val = cand_val
            if cand_val < best_val:
                best = cand.copy()
                best_val = cand_val

        iter_cnt += 1

        # occasional restart from the current best with a modest perturbation
        if iter_cnt % 20000 == 0:
            cur = perturb(best, max_delta_initial * 0.5)
            cur_val = max_overlap(cur)

    return best, best_val

    random.seed(seed)

    # ------------------------------------------------------------------
    # 2. Allocate time to stages
    # ------------------------------------------------------------------
    start_all = time.time()

    # Subsequent up‑sampling stages (double twice → up to ≈180 points)
    up_factors = [2, 2]          # each entry multiplies current length by 2
    for idx, fac in enumerate(up_factors):
        elapsed = time.time() - start_all
