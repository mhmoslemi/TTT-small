    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # identify free indices
    free = np.where((h > 0) & (h < 1))[0]
    while free.size > 0 and abs(diff) > 1e-12:
        delta = diff / free.size
        # update free indices but ensure stay in bounds
        add = np.minimum(np.maximum(delta, -h[free]), 1 - h[free])  # limit
        h[free] += add
        diff = target - h.sum()
        # update free set
        free = np.where((h > 0) & (h < 1))[0]
    # In rare case diff remains and all variables at bounds, we might need to adjust with small epsilon
    return h

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (must be top‑level according to the rules)
# ------------------------------------------------------------

def _c5_score(h):
    """Return the C5 value for a given step function h (numpy array)."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' yields 2n-1 entries
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)


def _gradient(h, shift):
    """Gradient of the overlap term that attains the current maximum.

    For a given integer shift (can be negative) the derivative of
    O_s = Σ h[i]*(1-h[i+shift]) w.r.t. h is:
        g[i] = (1 - h[i+shift]) - h[i-shift]   (out‑of‑range terms are 0)
    """
    n = h.shape[0]
    g = np.zeros_like(h)
    idx = np.arange(n)

    plus = idx + shift
    mask = (plus >= 0) & (plus < n)
    if np.any(mask):
        g[mask] += 1.0 - h[plus[mask]]

    minus = idx - shift
    mask2 = (minus >= 0) & (minus < n)
    if np.any(mask2):
        g[mask2] -= h[minus[mask2]]

    return g


def _project_to_sum_and_bounds(x, target):
    """Projection of vector x onto the intersection
       {0 ≤ v ≤ 1, Σ v = target}.
    Uses a simple bisection on a scalar λ such that
    v = clip(x - λ, 0, 1) satisfies Σv = target.
    """
    lo, hi = -3.0, 3.0  # wide enough interval for λ
    for _ in range(60):
        lam = (lo + hi) * 0.5
        v = np.clip(x - lam, 0.0, 1.0)
        s = v.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


def _initial_vector(n):
    """
    Build a starting vector of length n.
    If the global `initial_h_values` exists and its length divides n,
    repeat it to reach length n.  Otherwise use a random binary vector
    with exactly n/2 ones.
    """
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        m = base.size
        if n % m == 0:
            repeat = n // m
            h = np.repeat(base, repeat)
            # enforce exact sum = n/2 through projection
            target = n / 2.0
            h = _project_to_sum_and_bounds(h, target)
            return h
    # fallback: random binary with exact half ones
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    np.random.shuffle(h)
    return h


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a tuple (h_values, c5_bound, n_points) where
      h_values – list of step heights on a uniform grid of n_points,
      c5_bound – the achieved C5 value (max overlap integral),
      n_points – the grid size used.
    The routine performs a simple projected sub‑gradient descent
    (with occasional random perturbations) to lower the maximal overlap.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.995   # tiny safety margin

    # --------------------------------------------------------
    # Choose a grid size.
    # The provided construction has length 45; we upscale it by 4
    # (giving 180 points) which is a good compromise between speed
    # and expressive power.
    # --------------------------------------------------------
    base_len = 45
    if 'initial_h_values' in globals():
        base_len = len(initial_h_values)
    # use a factor that yields a moderate size (≤ 250)
    factor = max(1, 180 // base_len)
    n = base_len * factor
    target_sum = n / 2.0

    # --------------------------------------------------------
    # Initialise the vector
    # --------------------------------------------------------
    h = _initial_vector(n)

    # Keep the best solution ever seen
    best_h = h.copy()
    best_score = _c5_score(best_h)

    # Sub‑gradient descent parameters
    step = 0.3                # initial step size
    step_decay = 0.9995       # geometric decay per iteration
    min_step = 1e-6

    # restart handling
    no_improve = 0
    max_no_improve = 20000    # after this many iterations restart

    iteration = 0
    while time.time() < deadline:
        # ----------------------------------------------------
        # evaluate current C5 and record the best one
        # ----------------------------------------------------
        score = _c5_score(h)
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----------------------------------------------------
        # compute the shift corresponding to the current maximum
        # ----------------------------------------------------
        conv = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(conv.argmax())
        shift = max_idx - (n - 1)          # shift range: -(n-1)…+(n-1)

        # ----------------------------------------------------
        # sub‑gradient step
        # ----------------------------------------------------
        g = _gradient(h, shift)

        # take a step and project back onto feasible set
        h_new = h - step * g
        h = _project_to_sum_and_bounds(h_new, target_sum)

        # ----------------------------------------------------
        # schedule step size
        # ----------------------------------------------------
        step = max(step * step_decay, min_step)

        # ----------------------------------------------------
        # occasional random perturbation to escape flat regions
        # ----------------------------------------------------
        if iteration % 1000 == 0 and np.random.rand() < 0.07:
            noise = np.random.normal(scale=0.02, size=n)
            h = _project_to_sum_and_bounds(h + noise, target_sum)

        # ----------------------------------------------------
        # restart if stuck
        # ----------------------------------------------------
        if no_improve > max_no_improve:
            # a fresh random binary start
            h = np.zeros(n, dtype=float)
            h[: n // 2] = 1.0
            np.random.shuffle(h)
            no_improve = 0
            step = 0.3

        iteration += 1

    # Final evaluation using the official helper (ensures exact format)
    final_score = float(evaluate_erdos_solution(best_h.tolist()))

    return best_h.tolist(), final_score, n
