import numpy as np, time, math, random
def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx

def project_to_box_sum(y, target):
    # binary search for lambda
    lo = -2.0
    hi = 2.0
    # Expand bounds enough
    lo = np.min(y) - 2.0
    hi = np.max(y) + 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient_and_shift(h):
    N = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    shift = max_idx - (N - 1)

    if shift == 0:
        g = 1.0 - 2.0 * h
    elif shift > 0:
        g = np.ones(N)
        g[:N-shift] -= h[shift:]
        g[shift:] -= h[:N-shift]
    else: # shift < 0
        s = -shift
        g = np.ones(N)
        g[s:] -= h[:N-s]
        g[:N-s] -= h[s:]
    return g, shift, corr[max_idx]

def subgradient_descent(h0, max_iters, step0, decay):
    h = h0.copy()
    N = len(h)
    target = N / 2.0
    for t in range(max_iters):
        grad, shift, maxval = compute_gradient_and_shift(h)
        alpha = step0 * (decay ** t)
        y = h - alpha * grad
        h = project_to_box_sum(y, target)
    return h

def binary_sa(N, weight, time_limit, rng):
    # initial h random
    indices = np.arange(N)
    ones = rng.choice(indices, size=weight, replace=False)
    h = np.zeros(N, dtype=int)
    h[ones] = 1

    # precompute mism and max
    weight_float = float(weight)
    corr = np.correlate(h, h, mode='full')
    mism = weight_float - corr  # mism array length 2N-1
    cur_max = mism.max()
    best_h = h.copy()
    best_max = cur_max
    best_corr = corr

    # Precompute list of ones and zeros indices each iteration for quick selection
    # We'll update lists after swaps.

    start = time.time()
    T = 2.0  # initial temperature
    iteration = 0
    while time.time() - start < time_limit:
        # select random i in ones and j in zeros
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        # choose random i,j
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # compute new mism efficiently: recompute full correlation (slower but okay)
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        corr_new = np.correlate(h_new, h_new, mode='full')
        mism_new = weight_float - corr_new
        max_new = mism_new.max()
        delta = max_new - cur_max
        accept = False
        if delta <= 0:
            accept = True
        else:
            prob = math.exp(-delta / T) if T > 1e-12 else 0.0
            if rng.random() < prob:
                accept = True
        if accept:
            h = h_new
            cur_max = max_new
            # update best if improved
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        # cooling schedule
        iteration += 1
        # Update temperature
        # Maybe decrease T gradually each iteration
        T = max(0.001, T * 0.99995)  # e.g., exponential cooling
    # return best found
    c5 = best_max * (2.0 / N)
    return best_h.astype(float), c5

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine N
    if 'initial_h_values' in globals():
        N = len(initial_h_values)
    else:
        N = 70  # default
    weight = N // 2

    # SA parameters
    time_limit = budget_s * 0.9
    # Initialize binary h
    # Could start from randomly chosen ones: choose weight indices.
    indices = np.arange(N)
    initial_one_idx = rng.choice(indices, size=weight, replace=False)
    h = np.zeros(N, dtype=int)
    h[initial_one_idx] = 1

    # Compute mismatch max
    weight_float = float(weight)
    corr = np.correlate(h, h, mode='full')
    mism = weight_float - corr
    cur_max = mism.max()
    best_h = h.copy()
    best_max = cur_max

    # maintain lists of ones and zeros
    ones = set(initial_one_idx.tolist())
    zeros = set(indices.tolist()) - ones

    # SA schedule
    T0 = 2.0
    T = T0
    iteration = 0
    # We'll run while time remains
    while time.time() - start < time_limit:
        # pick random i (from ones) and j (from zeros)
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        # propose swap
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1

        corr_new = np.correlate(h_new, h_new, mode='full')
        mism_new = weight_float - corr_new
        max_new = mism_new.max()
        delta = max_new - cur_max
        if delta <= 0:
            accept = True
        else:
            prob = math.exp(-delta / T) if T > 1e-12 else 0.0
            accept = rng.random() < prob
        if accept:
            h = h_new
            cur_max = max_new
            # update sets
            ones.remove(i)
            zeros.add(i)
            zeros.remove(j)
            ones.add(j)
            # check best
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        # cooling
        iteration += 1
        # Update temperature gradually; e.g., exponential
        # For simplicity, after each iteration multiply by factor
        T = max(0.001, T * 0.99997)   # tune factor to reduce gradually

    # After SA, do local refinement: simple hill climbing (accept only improvements)
    # Continue for some steps
    refine_time = budget_s * 0.05
    while time.time() - start < budget_s - 0.1:
        # pick random i in ones, j in zeros
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        corr_new = np.correlate(h_new, h_new, mode='full')
        mism_new = weight_float - corr_new
        max_new = mism_new.max()
        if max_new < cur_max:
            h = h_new
            cur_max = max_new
            # update sets
            ones.remove(i)
            zeros.add(i)
            zeros.remove(j)
            ones.add(j)
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        # break if max_new equals best max maybe
        if time.time() - start > budget_s - 0.1:
            break

def h_from_r(r, N):
    if r <= 0:
        return np.full(N, 0.5)
    # compute H and lengths in actual domain:
    H = 1.0 / (2.0 - r)
    p = 2.0 - 2.0 * r
    # positions x from 0 to 2
    dx = 2.0 / N
    x = (np.arange(N) + 0.5) * dx
    # compute h:
    h = np.empty(N)
    # left ramp: x < r
    mask1 = x < r
    h[mask1] = H * x[mask1] / r
    # plateau: r <= x < r + p
    mask2 = (x >= r) & (x < r + p)
    h[mask2] = H
    # right ramp: x >= r + p
    mask3 = x >= r + p
    h[mask3] = H * (2.0 - x[mask3]) / r
    return h

import time, math
import numpy as np

def compute_c5(h):
    """Return the overlap constant C5 for a step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def project_to_sum_box(y, target):
    """Project a vector y onto {0<=x<=1, sum(x)=target}."""
    lo = np.min(y) - 2.0
    hi = np.max(y) + 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def gradient_and_shift(h):
    """Sub‑gradient of max_{k} ∫h(x)(1−h(x+k))dx at the worst shift."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)                # integer shift that attains the max

    if shift == 0:
        g = 1.0 - 2.0 * h
    elif shift > 0:
        g = np.ones(n)
        g[: n - shift] -= h[shift:]
        g[shift:]      -= h[: n - shift]
    else:                               # shift < 0
        s = -shift
        g = np.ones(n)
        g[s:]        -= h[: n - s]
        g[: n - s]   -= h[s:]
    return g, shift, corr[idx]

def generate_h(r, n):
    """One‑parameter family: ramps of length r, a flat top, symmetric on [0,2]."""
    if r <= 0.0:
        return np.full(n, 0.5)
    if r >= 1.0:
        # triangular shape (r=1)
        dx = 2.0 / n
        x = (np.arange(n) + 0.5) * dx
        return np.where(x <= 1.0, x, 2.0 - x)

    H = 1.0 / (2.0 - r)          # height ensuring total area = 1
    p = 2.0 - 2.0 * r            # width of the flat top
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx

    h = np.empty(n)
    left  = x < r
    middle = (x >= r) & (x < r + p)
    right = x >= r + p

    h[left]   = H * x[left] / r
    h[middle] = H
    h[right]  = H * (2.0 - x[right]) / r
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # use a modest but well‑resolved discretisation
    N = 200                     # number of sample points (must be even)
    target_sum = N / 2.0

    # --------------------------------------------------------------
    # 1.  Scan the one‑parameter family for a good initial guess.
    # --------------------------------------------------------------
    best_h = None
    best_c5 = float('inf')
    best_r = 0.0

    # coarse grid
    for r in np.linspace(0.0, 1.0, 101):
        h = generate_h(r, N)
        h = project_to_sum_box(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_r = r

    # fine grid around the best coarse point
    lo = max(0.0, best_r - 0.1)
    hi = min(1.0, best_r + 0.1)
    for r in np.linspace(lo, hi, 201):
        h = generate_h(r, N)
        h = project_to_sum_box(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_r = r

    # --------------------------------------------------------------
    # 2.  Refine with a simple sub‑gradient descent.
    # --------------------------------------------------------------
    h = best_h.copy()
    iters = 8000
    for t in range(iters):
        if time.time() - start > budget_s * 0.93:
            break
        grad, _, _ = gradient_and_shift(h)
        step = 0.02 / (1.0 + t * 0.001)   # decreasing step size
        h = project_to_sum_box(h - step * grad, target_sum)
        # occasional re‑evaluation of the bound
        if (t & 31) == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # final bound for the best candidate found
    final_c5 = compute_c5(best_h)

    return (best_h.tolist(), float(final_c5), N)
