import time, math
import numpy as np

def _evaluate_cost(h, dx):
    # correlation (full) of h and (1-h), then max scaled by dx
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_to_feasible(h, target_sum):
    """Project vector h onto box [0,1] while keeping the sum = target_sum.
    Simple water-filling: clamp, then redistribute the excess."""
    # clamp
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) < 1e-12:
        return h
    # redistribute linearly (will stay in bounds because steps are tiny)
    diff = target_sum - cur
    # distribute diff uniformly, then re‑clip (repeat a few times)
    for _ in range(5):
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ---- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
        target_sum = n / 2.0
        # enforce the exact sum constraint (tiny tolerances allowed)
        init = _project_to_feasible(init, target_sum)
        cur_h = init.copy()
    else:
        # choose a modest size if no starter is provided
        n = 200
        target_sum = n / 2.0
        cur_h = np.zeros(n, dtype=float)
        ones = rng.choice(n, size=n // 2, replace=False)
        cur_h[ones] = 1.0

    dx = 2.0 / n
    cur_cost = _evaluate_cost(cur_h, dx)

    best_h = cur_h.copy()
    best_cost = cur_cost

    start_time = time.time()
    # annealing schedule parameters
    max_temp = 5e-2
    min_temp = 1e-5
    max_step = 0.20
    min_step = 0.001

    n_iter = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break

        # linear schedules
        t = elapsed / budget_s
        temp = max_temp * (1.0 - t) + min_temp * t
        step = max_step * (1.0 - t) + min_step * t

        # propose a transfer of mass d from j -> i
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        if cur_h[i] >= 1.0 or cur_h[j] <= 0.0:
            continue
        max_transfer = min(step, 1.0 - cur_h[i], cur_h[j])
        if max_transfer <= 0.0:
            continue
        d = rng.random() * max_transfer

        new_h = cur_h.copy()
        new_h[i] += d
        new_h[j] -= d

        # the move preserves the sum; only box constraints may be violated slightly
        # (they cannot because we limited d), so no projection needed
        new_cost = _evaluate_cost(new_h, dx)
        delta = new_cost - cur_cost
        if delta < 0.0 or rng.random() < math.exp(-delta / max(temp, 1e-12)):
            cur_h = new_h
            cur_cost = new_cost
            if new_cost < best_cost:
                best_cost = new_cost
                best_h = new_h.copy()

        n_iter += 1

        # occasional random restart if stagnation is detected
        if n_iter % 20000 == 0:
            # if we haven't improved for a while, restart from a fresh binary guess
            if best_cost - cur_cost > 1e-6:

            cur_h = np.zeros(n, dtype=float)
            cur_h[rng.choice(n, size=n // 2, replace=False)] = 1.0
            cur_cost = _evaluate_cost(cur_h, dx)

    # final tiny polishing: small deterministic sweep improving the worst lag

        lag = worst_idx - (n - 1)                # signed lag

        low = idx_first[best_h[idx_first] < 0.5]
