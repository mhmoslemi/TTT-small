def project_onto_bounds_and_sum(x, target_sum, lower=0.0, upper=1.0, tol=1e-9):
    # Find lambda such that sum_i clip(x_i + lambda, lower, upper) = target_sum.
    # Binary search lambda.
    lo = -np.max(np.abs(x)) - 10
    hi = np.max(np.abs(x)) + 10
    for _ in range(100):
        mid = (lo+hi)/2
        proj = np.clip(x + mid, lower, upper)
        s = proj.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol: break
    return np.clip(x + (lo+hi)/2, lower, upper)

import numpy as np
import random
import time

def _project_box_sum(x, target, lo=0.0, hi=1.0, tol=1e-12, max_iter=100):
    # projection onto [lo,hi] with sum == target (Euclidean)
    # find lambda such that sum(clip(x+λ, lo, hi)) == target
    left = -np.max(x) - 1.0      # makes x+left <= 0 -> all clipped to lo
    right = (hi - np.min(x)) + 1.0  # makes x+right >= hi -> all clipped to hi
    for _ in range(max_iter):
        mid = (left + right) * 0.5
        proj = np.clip(x + mid, lo, hi)
        s = proj.sum()
        if s < target:
            left = mid
        else:
            right = mid
        if right - left < tol:
            break
    return np.clip(x + (left + right) * 0.5, lo, hi)


def _c5(h, dx):
    """max overlap integral for a discretised step function."""
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def _binary_optimize(h_cont, dx):
    """Turn a continuous vector into a 0/1 vector of the same mass
    and perform a simple swap‑based local search."""
    n = len(h_cont)
    target_int = int(round(n / 2.0))
    # initial binary by taking the largest entries
    idx = np.argpartition(-h_cont, target_int - 1)[:target_int]
    bin_vec = np.zeros(n, dtype=float)
    bin_vec[idx] = 1.0
    best_val = _c5(bin_vec, dx)
    improved = True
    while improved:
        improved = False
        ones = np.flatnonzero(bin_vec)
        zeros = np.flatnonzero(1.0 - bin_vec)
        for i in ones:
            for j in zeros:
                bin_vec[i] = 0.0
                bin_vec[j] = 1.0
                val = _c5(bin_vec, dx)
                if val + 1e-12 < best_val:
                    best_val = val
                    improved = True
                    break
                # revert
                bin_vec[i] = 1.0
                bin_vec[j] = 0.0
            if improved:
                break
    return bin_vec, best_val


def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()

    # ------------------------------------------------------------------
    # initialise problem size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 200                     # even for an integer mass
        h0 = np.random.rand(n)

    dx = 2.0 / n
    target_sum = n / 2.0

    # make the initial point feasible
    h0 = _project_box_sum(h0, target_sum)
    best_h = h0.copy()
    best_val = _c5(best_h, dx)

    # a quick binary improvement of the start point
    bin_h, bin_val = _binary_optimize(best_h, dx)
    if bin_val < best_val:
        best_h, best_val = bin_h, bin_val

    # ------------------------------------------------------------------
    # main stochastic optimisation loop
    max_time = budget_s - 0.2  # a tiny safety margin

    while time.time() - start < max_time:
        # random start – either fresh or noisy best
        if random.random() < 0.3:
            h = np.random.rand(n)
        else:
            h = best_h + 0.1 * (np.random.rand(n) - 0.5)
        h = _project_box_sum(h, target_sum)

        beta = 0.5          # smoothing for the max
        eta = 0.5           # step size
        # number of gradient steps (adaptive to remaining time)
        max_iters = 2000
        for it in range(max_iters):
            if time.time() - start >= max_time:
                break

            # current correlation (full range)
            corr = np.correlate(h, 1.0 - h, mode="full") * dx

            # smooth‑max weights
            m = np.max(corr)
            w_raw = np.exp(beta * (corr - m))
            w = w_raw / w_raw.sum()

            # gradient of the smooth max
            conv1 = np.convolve(w, h, mode="full")
            conv2 = np.convolve(w[::-1], h, mode="full")
            offset = n - 1
            u = conv1[offset:offset + n]
            v = conv2[offset:offset + n]
            grad = dx * (1.0 - (u + v))

            # gradient step + projection
            h = h - eta * grad
            h = _project_box_sum(h, target_sum)

            # schedule updates
            beta = min(beta * 1.02, 30.0)
            eta = max(eta * 0.998, 1e-4)

            # occasional check for improvement
            if (it & 31) == 0:
                cur = np.max(corr)
                if cur < best_val:
                    best_val = float(cur)
                    best_h = h.copy()

        # occasional binary refinement of the best found so far
        if random.random() < 0.2:
            bin_h, bin_val = _binary_optimize(best_h, dx)
            if bin_val < best_val:
                best_h, best_val = bin_h, bin_val

    # final binary polishing
    bin_h, bin_val = _binary_optimize(best_h, dx)
    if bin_val < best_val:
        best_h, best_val = bin_h, bin_val

    # final enforce of constraints (tiny numeric drift)
    best_h = _project_box_sum(best_h, target_sum)
    final_val = _c5(best_h, dx)

    return (best_h, float(final_val), n)
