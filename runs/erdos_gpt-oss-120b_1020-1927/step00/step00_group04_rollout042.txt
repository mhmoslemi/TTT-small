def perturb(h, max_step):

    # find source and target indices
    # pick randomly source with h>0 and target with h<1
    # use vectorized approach of random index selection repeated until succeed
    i = np.random.randint(n)

    # compute allowable delta
    delta_max = min(max_step, h[i], 1 - h[j])

def local_search(initial_h, max_iters, max_step_initial):
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = max_step_initial

        if cand_c5 < compute_c5(h) or np.random.rand() < math.exp(-(cand_c5 - compute_c5(h)) / step):

        # Reduce step gradually

def normalize_h(h):
    # clip to [0,1], then adjust to meet sum = n/2 while staying within bounds.
    h = np.clip(h, 0, 1)
    target = len(h)/2
    diff = target - h.sum()

    # simple method: while |diff| > 1e-10:
    # find indices with room to increase (if diff>0) or decrease (if diff<0)
    # allocate equally.
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase sum
            room = 1 - h
            total_room = room.sum()
            if total_room < 1e-12: break
            inc = diff * room / total_room
            h += inc
        else:
            # need to decrease sum
            room = h
            total_room = room.sum()
            if total_room < 1e-12: break
            dec = diff * room / total_room  # diff negative
            h += dec  # subtract (since diff negative)
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
    return h

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Return the C5 value for a step function h given as a 1‑D array."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def normalize_sum_one(h):
    """Project h onto [0,1] with the exact sum = n/2."""
    n = h.size
    target = n / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    cur = h.sum()
    if abs(cur - target) < 1e-12:
        return h
    # simple proportional scaling, then a few correction passes
    if cur > 0:
        h *= target / cur
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(5):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = free.sum()
        if cnt == 0:
            h += diff / n
        else:
            h[free] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h


def random_binary(n):
    """A binary (0/1) vector with exactly n/2 ones (plus a 0.5 if n odd)."""
    h = np.zeros(n)
    half = n // 2
    idx = np.random.choice(n, half, replace=False)
    h[idx] = 1.0
    if n % 2:
        # add a 0.5 to meet the exact sum constraint
        remaining = np.setdiff1d(np.arange(n), idx)
        h[remaining[0]] = 0.5
    return h


def propose(h, max_transfer):
    """Transfer a small amount of mass from a positive entry to a lower one."""
    n = h.size
    pos = np.where(h > 1e-12)[0]
    neg = np.where(h < 1.0 - 1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return h.copy()
    i = np.random.choice(pos)
    j = np.random.choice(neg)
    delta = min(max_transfer, h[i], 1.0 - h[j])
    if delta <= 0:
        return h.copy()
    delta *= np.random.rand()
    h2 = h.copy()
    h2[i] -= delta
    h2[j] += delta
    return h2


def targeted_propose(h, max_transfer, corr):
    """
    Move mass from the overlapping region of the worst lag to the outside region
    (or vice‑versa) to directly reduce the dominant O(L).
    """
    n = h.size
    idx = np.argmax(corr)          # index of maximal correlation entry
    lag = idx - (n - 1)            # signed lag
    if lag >= 0:
        src_region = np.arange(0, n - lag)
        tgt_region = np.arange(n - lag, n)
    else:
        lag = -lag
        src_region = np.arange(lag, n)
        tgt_region = np.arange(0, lag)

    src = src_region[h[src_region] > 1e-12]
    tgt = tgt_region[h[tgt_region] < 1.0 - 1e-12]
    if src.size == 0 or tgt.size == 0:
        return h.copy()
    i = np.random.choice(src)
    j = np.random.choice(tgt)
    delta = min(max_transfer, h[i], 1.0 - h[j])
    if delta <= 0:
        return h.copy()
    delta *= np.random.rand()
    h2 = h.copy()
    h2[i] -= delta
    h2[j] += delta
    return h2


def sa_optimize(start_h, time_budget):
    """Simulated‑annealing style optimizer for a fixed n."""
    n = start_h.size
    dx = 2.0 / n
    best_h = start_h.copy()
    best_c5 = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
    cur_c5 = cur_corr.max() * dx

    step = 0.2                     # maximal transfer amount
    start = time.time()
    it = 0
    max_iter = 200000              # safety upper bound
    while time.time() - start < time_budget and it < max_iter:
        # choose proposal type
        if np.random.rand() < 0.35:
            cand_h = targeted_propose(cur_h, step, cur_corr)
        else:
            cand_h = propose(cur_h, step)

        cand_corr = np.correlate(cand_h, 1.0 - cand_h, mode="full")
        cand_c5 = cand_corr.max() * dx

        if cand_c5 < best_c5:
            best_h = cand_h.copy()
            best_c5 = cand_c5

        # acceptance rule (temperature decays linearly)
        if cand_c5 < cur_c5:
            cur_h, cur_corr, cur_c5 = cand_h, cand_corr, cand_c5
        else:
            temp = 0.06 * (1.0 - it / 10000.0)   # decays to ~0 after 10k iters
            if temp > 1e-9 and np.random.rand() < math.exp(-(cand_c5 - cur_c5) / temp):
                cur_h, cur_corr, cur_c5 = cand_h, cand_corr, cand_c5

        it += 1
        if it % 400 == 0:
            step = max(step * 0.9, 1e-4)

    return best_h, best_c5

    """

    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------

    if 'initial_h_values' in globals():

    while True:

        trial_h, trial_c5 = sa_optimize(trial_h, slice_time)

        if trial_c5 < best_c5:

                pass

    # 3. final sanity check & output
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = normalize_sum_one(best_h)
    final_c5 = compute_c5(best_h)
