def upscale(h, factor):
    n0 = len(h)
    n = n0 * factor
    x_old = np.arange(n0) / n0 * 2.0
    x_new = np.arange(n) / n * 2.0
    h_new = np.interp(x_new, x_old, h)
    # adjust sum
    target_sum = n/2.0
    cur_sum = np.sum(h_new)
    scale = target_sum / cur_sum
    h_new *= scale
    # clip to [0,1] and renorm if needed
    h_new = np.clip(h_new, 0.0, 1.0)
    # recompute sum and adjust
    cur_sum = np.sum(h_new)
    if abs(cur_sum - target_sum) > 1e-6:
        # adjust by adding diff to a random index with capacity
        diff = target_sum - cur_sum
        if diff > 0:
            # find indices with <1
            idx = np.where(h_new < 1)[0]
        else:
            idx = np.where(h_new > 0)[0]
        if idx.size > 0:
            i = np.random.choice(idx)
            h_new[i] = np.clip(h_new[i] + diff, 0, 1)
    return h_new

def _c5_of(h: np.ndarray) -> float:
    """C5 value for a discrete step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def _normalize(h: np.ndarray) -> np.ndarray:
    """Force 0≤h≤1 and Σh = n/2 (up to tiny numerical error)."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute the missing mass over free coordinates
        if diff > 0:
            free = np.where(h < 1.0)[0]
        else:
            free = np.where(h > 0.0)[0]
        if free.size:
            inc = diff / free.size
            h[free] = np.clip(h[free] + inc, 0.0, 1.0)
    # final tiny correction on a random free index
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) > 1e-12:
        if diff > 0:
            cand = np.where(h < 1.0)[0]
        else:
            cand = np.where(h > 0.0)[0]
        if cand.size:
            i = np.random.choice(cand)
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def _upscale(h: np.ndarray, factor: int) -> np.ndarray:
    """Linear interpolation to a finer grid (factor must be integer)."""
    if factor == 1:
        return h.copy()
    n0 = h.shape[0]
    n = n0 * factor
    x_old = np.arange(n0, dtype=float) / n0 * 2.0
    x_new = np.arange(n, dtype=float) / n * 2.0
    h_new = np.interp(x_new, x_old, h)
    # enforce the integral constraint
    target = n / 2.0
    cur = np.sum(h_new)
    if cur > 0:
        h_new *= (target / cur)
    h_new = np.clip(h_new, 0.0, 1.0)
    return _normalize(h_new)

    best_h = cur_h.copy()
    best_c5 = cur_c5

    t0 = time.time()
    it = 0
    while time.time() - t0 < secs:
        it += 1
        # temperature linearly decreasing to 1e‑6
        remaining = secs - (time.time() - t0)
        T = max(1e-6, 0.05 * remaining / secs)

        # pick donor (value>0) and receiver (value<1)
        donors = np.where(cur_h > 1e-12)[0]
        receivers = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or receivers.size == 0:
            break

        j = np.random.choice(receivers)
        if i == j:
            continue

        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0:
            continue
        delta = max_transfer * np.random.rand()  # any amount up to the feasible bound
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_c5 = _c5_of(new_h)
        diff = new_c5 - cur_c5
        if diff < 0 or np.random.rand() < np.exp(-diff / T):

    best_h = _normalize(best_h)
    best_c5 = _c5_of(best_h)
