def adjust_sum(h, target):
    # assume h already in [0,1]
    cur_sum = h.sum()
    diff = target - cur_sum
    if abs(diff) < 1e-12:
        return h
    # if diff >0, need to increase some entries where h<1
    # if diff<0, need to decrease some entries where h>0
    direction = 1 if diff > 0 else -1
    # compute indices eligible
    if direction == 1:
        eligible = np.where(h < 1 - 1e-12)[0]
    else:
        eligible = np.where(h > 1e-12)[0]
    # if none eligible; just return
    if eligible.size == 0:
        return h
    # Get random permutation
    rng = np.random.default_rng()
    # We'll loop until diff approx 0
    while abs(diff) > 1e-12 and eligible.size >0:
        i = rng.choice(eligible)
        if direction == 1:
            max_inc = 1 - h[i]
            inc = min(max_inc, diff)
            h[i] += inc
            diff -= inc
            if h[i] >= 1 - 1e-12:
                eligible = eligible[eligible != i]  # remove this index
        else:
            max_dec = h[i]
            dec = min(max_dec, -diff)
            h[i] -= dec
            diff += dec
            if h[i] <= 1e-12:
                eligible = eligible[eligible != i]
    return h

import numpy as np
import time
import math

def run(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimizer for the Erdős minimum overlap constant C5.
    Returns a tuple (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    budget = float(budget_s)

    # ------------------------------------------------------------------
    # Helpers
    # ------------------------------------------------------------------
    def eval_h(h):
        """Return the C5 value for a candidate vector h."""
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * dx

    def project_sum_bounds(h, target):
        """Clip to [0,1] and force sum(h) == target (simple iterative fix)."""
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            return h
        # indices that can move in the needed direction
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if idx.size == 0:
            return h
        # distribute the remainder greedily
        while abs(diff) > 1e-12 and idx.size:
            i = rng.choice(idx)
            if diff > 0:
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
                diff -= inc
                if h[i] >= 1.0 - 1e-12:
                    idx = idx[idx != i]
            else:
                dec = min(-diff, h[i])
                h[i] -= dec
                diff += dec
                if h[i] <= 1e-12:
                    idx = idx[idx != i]
        return h

    def propose_move(h, max_step):
        """Propose a new h by moving a small amount from one index to another."""
        n = len(h)
        i, j = rng.choice(n, size=2, replace=False)
        # feasible delta keeping both entries inside [0,1]
        low = -min(h[i], 1.0 - h[j])
        high = min(1.0 - h[i], h[j])
        if low >= high:
            return None
        low = max(low, -max_step)
        high = min(high, max_step)
        if low >= high:
            return None
        delta = rng.uniform(low, high)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        return h_new

    # Starting point

    try:

    best_overall_n = None

    # ------------------------------------------------------------------
    # Main optimisation loop over chosen grid sizes
    # ------------------------------------------------------------------
    for n in grid_sizes:
        # initialise h for this grid
        if n == n_orig:
            h = h0.copy()
        else:
            # linear interpolation to increase resolution
            old_x = np.linspace(0.0, 2.0, n_orig, endpoint=False)
