import time, math, random
import numpy as np

def _compute_c5(vec, dx):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx using the prescribed discrete evaluation."""
    # np.correlate follows the problem’s definition (zero padding outside).
    return np.max(np.correlate(vec, 1.0 - vec, mode="full")) * dx

def _random_feasible(n, rng):
    """Generate a random h with entries in [0,1] and sum = n/2."""
    # Dirichlet gives a point on the simplex Σh = 1; scale to Σh = n/2.
    v = rng.dirichlet(np.ones(n))
    h = v * (n / 2.0)
    # Clip to bounds; fix the remaining slack on the first non‑saturated entry.
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-12:
        for i in range(n):
            room = 1.0 - h[i] if diff > 0 else h[i]
            change = min(abs(diff), room)
            change = change if diff > 0 else -change
            h[i] += change
            diff = n / 2.0 - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def _anneal_one_start(h0, dx, rng, time_deadline):
    """
    Simulated annealing starting from h0.
    Returns the best vector and its C5 found during this run.
    """
    n = h0.size
    h = h0.copy()
    cur = _compute_c5(h, dx)
    best_h = h.copy()
    best_val = cur

    # annealing schedule
    max_iters = 500_000
    T_start = 0.03
    T_end = 1e-5
    log_factor = math.log(T_end / T_start)

    it = 0
    while it < max_iters and time.time() < time_deadline:
        # temperature decreases smoothly
        T = T_start * math.exp(log_factor * (it / max_iters))

        # propose a feasible transfer between two indices
        i, j = rng.integers(0, n, size=2)
        while j == i:
            j = rng.integers(0, n)
        hi, hj = h[i], h[j]

        d_min = max(-hi, hj - 1.0)
        d_max = min(1.0 - hi, hj)
        if d_max <= d_min:
            it += 1
            continue

        delta = rng.random() * (d_max - d_min) + d_min
        h[i] = hi + delta
        h[j] = hj - delta

        new_val = _compute_c5(h, dx)
        diff = new_val - cur
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            cur = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj

        it += 1

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """
    Find a step function h on [0,2] with ∫h=1 that minimizes
    C5 = max_k ∫ h(x)(1‑h(x+k))dx.

    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95   # safe margin

    # ------------------------------------------------------------------
    # Initialise a feasible vector.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        # Project onto the feasible set if needed (rare).
        cur_sum = h0.sum()
        if abs(cur_sum - n / 2.0) > 1e-9:
            # Simple scaling + clipping, then fix the remaining slack.
            h0 = np.clip(h0 * (n / 2.0) / cur_sum, 0.0, 1.0)
            diff = n / 2.0 - h0.sum()
            for i in range(n):
                room = 1.0 - h0[i] if diff > 0 else h0[i]
                change = min(abs(diff), room)
                change = change if diff > 0 else -change
                h0[i] += change
                diff = n / 2.0 - h0.sum()
                if abs(diff) < 1e-12:
                    break
    else:
        n = 200                # a modest size when no parent is supplied
        h0 = _random_feasible(n, rng)

    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Run a first annealing from the (possibly) good parent solution.
    # ------------------------------------------------------------------
    best_vec, best_val = _anneal_one_start(h0, dx, rng, time_limit)

    # ------------------------------------------------------------------
    # If time permits, try a few fresh random starts.
    # ------------------------------------------------------------------
    attempts = 0
    while time.time() < time_limit - 0.5 and attempts < 5:
        attempts += 1
        h_rand = _random_feasible(n, rng)
        vec, val = _anneal_one_start(h_rand, dx, rng, time_limit)
        if val < best_val:
            best_val, best_vec = val, vec

    # ------------------------------------------------------------------
    # Final projection: ensure the sum constraint holds exactly.
    # ------------------------------------------------------------------
    diff = n / 2.0 - best_vec.sum()
    if abs(diff) > 1e-12:
        # Adjust the first entry that can absorb the difference without leaving [0,1].
        for i in range(n):
            low, high = 0.0, 1.0
            if low <= best_vec[i] + diff <= high:
                best_vec[i] += diff
                break

    # ------------------------------------------------------------------
    # Compute the final bound using the official evaluator if available.
    # ------------------------------------------------------------------
    if "evaluate_erdos_solution" in globals():
        final_c5 = float(evaluate_erdos_solution(best_vec.tolist()))
    else:
        final_c5 = float(_compute_c5(best_vec, dx))

    return best_vec.tolist(), final_c5, n
