import cvxpy as cp
def project_onto_constraints(x):
    n = len(x)
    target = n/2
    var = cp.Variable(n)
    obj = cp.Minimize(cp.sum_squares(var - x))
    cons = [var >= 0, var <= 1, cp.sum(var) == target]
    prob = cp.Problem(obj, cons)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=10000)
    return var.value

import time
import numpy as np

def _compute_c5(h):
    """C5 = max_{k} ∫ h(x)*(1-h(x+k)) dx  (discretized)"""
    # correlation h with (1-h) using full mode
    conv = np.convolve(h, (1 - h)[::-1], mode="full")
    return conv.max() * (2.0 / h.size)


def _max_lag(conv, n):
    """Return lag (relative shift) that gives the maximal correlation."""
    idx = int(conv.argmax())
    return idx - (n - 1)


def _gradient(h, lag):
    """Sub‑gradient of the maximal correlation w.r.t. h."""
    n = h.size
    g = np.zeros_like(h)
    if lag >= 0:
        l = lag
        # term (1 - h[i+l]) for i where i+l < n
        i1 = np.arange(n - l)
        g[i1] += 1.0 - h[i1 + l]
        # term -h[i-l] for i where i-l >= 0
        i2 = np.arange(l, n)
        g[i2] -= h[i2 - l]
    else:
        l = -lag
        # term (1 - h[i-l]) for i where i-l >= 0  <=> i >= l
        i1 = np.arange(l, n)
        g[i1] += 1.0 - h[i1 - l]
        # term -h[i+l] for i where i+l < n  <=> i < n-l
        i2 = np.arange(0, n - l)
        g[i2] -= h[i2 + l]
    return g


def _project_sum_one(h):
    """Project to the simplex {0≤h_i≤1, sum h = n/2} preserving bounds."""
    n = h.size
    target = n / 2.0
    # simple iterative redistribution
    while True:
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            # forced correction on any coordinate (will stay in bounds)
            free = np.arange(n)
        inc = diff / free.size
        # clip the increment so we stay in [0,1]
        inc = np.clip(inc, -h[free], 1 - h[free])
        h[free] += inc
    return h
