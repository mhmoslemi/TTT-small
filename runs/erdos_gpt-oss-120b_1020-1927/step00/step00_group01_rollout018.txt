```
def compute_c5(a):
    n = a.size
    dx = 2.0 / n
    corr = np.correlate(a.astype(np.float64), 1.0 - a.astype(np.float64), mode='full')
    max_corr = np.max(corr)
    return max_corr * dx, max_corr
```

```
def search_binary(n, target_max_corr=None, time_limit=..., seed=42):
    rng = np.random.default_rng(seed)
    # Initialize
    a = np.zeros(n, dtype=np.int8)
    # random shuffle indices
    perm = rng.permutation(n)
    a[perm[:n//2]] = 1
    max_bound, max_corr = compute_c5(a)
    best_a = a.copy()
    best_corr = max_corr
    best_bound = max_bound

    start = time.time()
    iteration = 0
    T0 = 5.0
    # schedule: T = T0 * (1 - elapsed/ time_limit)
    while time.time() - start < time_limit:
        # propose swap
        # choose i where a[i]==1
        i = rng.choice(np.flatnonzero(a))
        j = rng.choice(np.flatnonzero(1-a))
        # swap
        a[i], a[j] = 0, 1
        _, new_corr = compute_c5(a)
        # acceptance
        delta = new_corr - max_corr
        if delta <= 0 or rng.random() < np.exp(-delta / (T0 * (1 - (time.time() - start) / time_limit) + 1e-12)):
            # accept
            max_corr = new_corr
            max_bound = max_corr * (2.0 / n)
            if max_corr < best_corr:
                best_corr = max_corr
                best_bound = max_corr * (2.0 / n)
                best_a = a.copy()
                if target_max_corr and max_corr <= target_max_corr:
                    break
        else:
            # reject: undo swap
            a[i], a[j] = 1, 0
        iteration += 1
```

import numpy as np
import time
import math

def project_onto_box_sum(v, lo, hi, target):
    left = -1e6
    right = 1e6
    for _ in range(60):
        mid = (left + right) * 0.5
        y = np.clip(v - mid, lo, hi)
        s = y.sum()
        if s > target:
            left = mid
        else:
            right = mid
    return np.clip(v - (left + right) * 0.5, lo, hi)

def compute_gradient(h, lags):
    n = h.shape[0]
    g = np.zeros_like(h)
    for lag in lags:
        if lag >= 0:
            g[:n - lag] -= h[lag:]
            g[lag:]    -= h[:n - lag]
        else:
            L = -lag
            g[L:]    -= h[:n - L]
            g[:n - L] -= h[L:]
    return g

def optimize_h(h0, max_iters, start_time, time_limit):
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_val = np.max(np.correlate(h, 1 - h, mode='full'))
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        cur_max = corr.max()
        if cur_max < best_val:
            best_val = cur_max
            best_h = h.copy()
        # lags with max correlation
        lag_idxs = np.where(corr == cur_max)[0] - (n - 1)
        grad = compute_gradient(h, lag_idxs)
        eta = 0.5 / math.sqrt(it + 1)
        v = h - eta * grad
        h = project_onto_box_sum(v, 0.0, 1.0, target)
    final_val = np.max(np.correlate(best_h, 1 - best_h, mode='full'))
    return best_h, final_val

def upscale_h(h_small, n_target):
    x_small = np.linspace(0.0, 2.0, len(h_small))
    x_target = np.linspace(0.0, 2.0, n_target)

    dx = 2.0 / n
    max_corr = np.max(np.correlate(h, 1 - h, mode='full'))

import numpy as np
import time
import math

def _project_onto_box_sum(v, lo, hi, target):
    """Project vector v onto the box [lo,hi] while forcing the sum to `target`."""
    left, right = -1e6, 1e6
    for _ in range(60):
        mid = (left + right) * 0.5
        y = np.clip(v - mid, lo, hi)
        s = y.sum()
        if s > target:
            left = mid
        else:
            right = mid
    return np.clip(v - (left + right) * 0.5, lo, hi)


def _gradient_for_lags(h, lags):
    """Sub‑gradient of the maximal overlap for the given lags."""
    n = h.shape[0]
    g = np.zeros_like(h)
    for lag in lags:
        if lag >= 0:
            g[: n - lag] -= h[lag:]
            g[lag:]      -= h[: n - lag]
        else:
            L = -lag
            g[L:]       -= h[: n - L]
            g[: n - L]  -= h[L:]
    return g


def _optimize_continuous(h0, max_iters, start, deadline):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_val = np.max(np.correlate(h, 1 - h, mode="full"))
    for it in range(max_iters):
        if time.time() - start > deadline:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        cur_max = corr.max()
        if cur_max < best_val:
            best_val = cur_max
            best_h = h.copy()
        # all lags attaining the current maximum
        lag_idxs = np.where(corr == cur_max)[0] - (n - 1)
        grad = _gradient_for_lags(h, lag_idxs)
        eta = 0.5 / math.sqrt(it + 1)
        h = _project_onto_box_sum(h - eta * grad, 0.0, 1.0, target)
    final_val = np.max(np.correlate(best_h, 1 - best_h, mode="full"))
    return best_h, final_val


def _upscale(h_small, n_target):

    xs = np.linspace(0.0, 2.0, len(h_small))
    xt = np.linspace(0.0, 2.0, n_target)
    h = np.interp(xt, xs, h_small)
    h = np.clip(h, 0.0, 1.0)
    return _project_onto_box_sum(h, 0.0, 1.0, n_target / 2.0)

    n = h.shape[0]
    dx = 2.0 / n
    raw = np.max(np.correlate(h, 1 - h, mode="full"))
    return float(raw * dx), float(raw)
