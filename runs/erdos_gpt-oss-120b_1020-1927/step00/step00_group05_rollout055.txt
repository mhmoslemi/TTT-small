def project_box_sum(v, target, max_iter=60):
    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):
        mid = (lo+hi)/2
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, 0, 1)

        w = np.exp(-alpha * (corr - cmin))
        w_sum = w.sum()
        w = w / w_sum

        # compute gradient
        g = np.zeros_like(h)
        offset = n - 1
        # lag 0
        w0 = w[offset]
        g += 2.0 * w0 * h
        # positive lags
        for l in range(1, n):
            wpos = w[offset + l] + w[offset - l]  # both signs
            if wpos == 0:  # skip
                continue
            # contributions
            g[:n - l] += wpos * h[l:]
            g[l:] += wpos * h[:n - l]

        # gradient ascent step
        h_new = h + eta * g

        # projection onto feasible set
        h = project(h_new)

        # evaluate C5
        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        if c5 < best_c5 - 1e-6:
            best_c5 = c5
            best_h = h.copy()

        # update parameters
        alpha = min(alpha * alpha_inc, 10.0)  # limit alpha
        eta = max(eta * eta_decay, 1e-4)

        # Occasionally maybe random perturbation to avoid local minima
        if np.random.rand() < 0.001:
            # random jitter
            jitter = (np.random.rand(n) - 0.5) * 0.1
            h = project(h + jitter)

    # Return best
    return best_h, best_c5, n

import time
import numpy as np

def _project_onto_box_sum(v, target, max_iter=60):
    """
    Euclidean projection of v onto the set
        { x : 0 <= x_i <= 1 , sum(x) = target }.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1e6, 1e6
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # -----------------------------------------------------------------
    # initialise h
    # -----------------------------------------------------------------
    try:
        init = initial_h_values          # provided by the harness
        n = int(init.shape[0])

    best_h = h.copy()
    dx = 2.0 / n
    best_c5 = np.max(np.correlate(best_h, 1.0 - best_h, mode='full')) * dx

    # parameters for the heuristic
    eta = 0.06                # base step size
    eta_min = 1e-5
    decay = 0.9997            # exponential decay of step size
    no_improve = 0
    stagnation_limit = 4000   # restart after this many non‑improvements
    worst_pool = 5            # how many of the current worst lags to use
    offset = n - 1            # index of lag 0 in full correlation array

    while time.time() - start_time < budget_s * 0.96:
        # current performance
        cur_c5 = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

        # autocorrelation of h with itself
        corr = np.correlate(h, h, mode='full')

        # indices of a few worst lags (smallest autocorrelation)
        worst_idx = np.argpartition(corr, worst_pool)[:worst_pool]

        # gradient that tries to increase those worst autocorrelations
        g = np.zeros_like(h)
        for idx in worst_idx:

            l = abs(L)
            if l == 0:

            else:

                g[: n - l] += h[l:]
                g[l:]       += h[: n - l]

        # average the contribution of the selected worst lags
        g /= float(len(worst_idx))

        # take a step and project back onto the feasible set
        h_new = _project_onto_box_sum(h + eta * g, target_sum)

        # evaluate the new candidate
        cand_c5 = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * dx

        else:

            eta = 0.06
            no_improve = 0

    # final sanity check (projection may drift by <1e-12)
    best_h = _project_onto_box_sum(best_h, target_sum)
    best_c5 = np.max(np.correlate(best_h, 1.0 - best_h, mode='full')) * dx

    return best_h, float(best_c5), n
