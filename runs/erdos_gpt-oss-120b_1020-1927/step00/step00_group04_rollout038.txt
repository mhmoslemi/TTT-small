    # Find lambda such that sum_i clip(v_i - lambda, 0, 1) = S
    # Equivalent to water-filling.
    # Solve by sorting v.
    n = len(v)
    # sort v ascending
    u = np.sort(v)
    # compute prefix sums
    # We'll find lambda such that after adjusting downwards (i.e., subtract lambda from each, then clip to [0,1]) sum = S.
    # clip(v_i - lambda, 0, 1) = min(max(v_i - lambda, 0), 1).
    # This is like shifting values downward (lambda positive). Many resources present algorithm for projection onto simplex with box constraints.
    # Implementation: binary search on lambda on interval [min(v)-1, max(v)].
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2, 0, 1)

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def binary_milp(n, M_target):
    import cvxpy as cp
    S = n // 2
    # variables
    h = cp.Variable(n, boolean=True)
    # define pair index dictionary and variable y
    pair_idx = {}
    pair_vars = []
    # we will create y for each pair (i,j) where i<j
    for i in range(n):
        for j in range(i+1, n):
            pair_idx[(i,j)] = len(pair_vars)
            pair_vars.append(cp.Variable(boolean=True))
    y = cp.Variable(len(pair_vars), boolean=True)
    constraints = []
    constraints.append(cp.sum(h) == S)
    # linearization constraints
    for (i,j), idx in pair_idx.items():
        constraints.append(y[idx] <= h[i])
        constraints.append(y[idx] <= h[j])
        constraints.append(y[idx] >= h[i] + h[j] - 1)
    # shift constraints
    for k in range(1, n):
        idxs_i = np.arange(0, n - k)
        # compute sum of h_i for i in idxs_i
        sum_h = cp.sum(cp.hstack([h[i] for i in idxs_i]))
        # sum of y_{i,i+k}
        y_idxs = [pair_idx[(i,i+k)] for i in idxs_i]
        sum_y = cp.sum(cp.hstack([y[j] for j in y_idxs]))
        constraints.append(sum_h - sum_y <= M_target)
    # objective dummy (zero)
    prob = cp.Problem(cp.Minimize(0), constraints)
    try:
        prob.solve(solver=cp.GLPK_MI, warm_start=True, verbose=False, glpk={'msg_lev': 'GLP_MSG_OFF'})
    except Exception:
        try:
            prob.solve(solver=cp.CBC, warm_start=True, verbose=False)
        except Exception:
            return None
    if prob.status in [cp.OPTIMAL, cp.FEASIBLE]:
        h_val = np.round(h.value).astype(int)
        # may need to clip to 0/1
        h_val = np.clip(h_val,0,1)
        return h_val
    else:
        return None

def project_to_box_and_sum(v, target_sum):
    import numpy as np
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(40):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v - (lo + hi) / 2.0, 0, 1)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-8:
        mask = (proj > 0) & (proj < 1)
        cnt = mask.sum()
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0, 1)
    return proj

import time
import numpy as np

def compute_c5(h):
    """return C5 for a vector h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def project_to_box_and_sum(v, target):
    """Euclidean projection of v onto {0≤x≤1, sum x = target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(40):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-9:
        mask = (proj > 0.0) & (proj < 1.0)
        cnt = mask.sum()
        if cnt:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

            h = np.interp(x_new, x_old, h)
    else:
        h = np.zeros(n, dtype=float)
        h[:S] = 1.0
        np.random.shuffle(h)
    h = project_to_box_and_sum(np.clip(h, 0.0, 1.0), S)

    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c

    start = time.time()
    T0, Tmin = 0.04, 0.0005
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        # temperature and step size schedules
        T = max(Tmin, T0 * (1.0 - it / max_iters))
        sigma = 0.5 * (1.0 - it / max_iters) + 0.02

        # random perturbation of a few coordinates
        idx = np.random.randint(0, n, size=5)
        dv = np.random.normal(scale=sigma, size=5)
        new_h = cur_h.copy()
        new_h[idx] += dv
        new_h = project_to_box_and_sum(np.clip(new_h, 0.0, 1.0), S)

        new_c = compute_c5(new_h)

        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()

        # Metropolis acceptance
        if new_c < cur_c or np.random.rand() < np.exp((cur_c - new_c) / max(T, 1e-12)):
            cur_h, cur_c = new_h, new_c

        # early exit when already very good
        if best_c <= 0.3808:
            break
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)

    best_h, best_c, best_n = None, float("inf"), None

    # try the supplied construction first
    try:
        init_vec = np.array(initial_h_values, dtype=float)
        c0 = compute_c5(init_vec)
        if c0 < best_c:
            best_h, best_c, best_n = init_vec, c0, len(init_vec)
    except Exception:
        init_vec = None

    # grid sizes we will explore (all even)
    n_candidates = [42, 84, 126, 168, 210]

    for n in n_candidates:
        remaining = budget_s - (time.time() - start)
        if remaining < 5.0:
            break

        # allocate iterations roughly proportional to remaining time
        max_iters = int(min(30000, remaining * 30))

        # three independent starts per grid size
        for _ in range(3):
            # use the current best as warm start when sizes match or interpolate otherwise
            start_vec = None
            if best_h is not None and best_n == n:
                start_vec = best_h
            elif init_vec is not None:
                start_vec = init_vec
            h_opt, c_opt = anneal_optimize(
                n, max_iters, budget_s - (time.time() - start), start_vec
            )
            if c_opt < best_c:
                best_h, best_c, best_n = h_opt, c_opt, n
            if best_c <= 0.3808:
                break
        if best_c <= 0.3808:
            break

    # final safety projection and recomputation
    if best_h is None:
        n = 42
        best_h = np.full(n, 0.5)
        best_c = compute_c5(best_h)
        best_n = n
    else:
        best_h = np.array(best_h, dtype=float)
        best_h = project_to_box_and_sum(best_h, len(best_h) // 2)
        best_c = compute_c5(best_h)
        best_n = len(best_h)

    return best_h.tolist(), float(best_c), int(best_n)
