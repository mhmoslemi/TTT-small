def smoothed_grad(h, α):
    # Compute S = np.correlate(h, h, mode='full') * dx  (length 2n-1)
    S = np.correlate(h, h, mode='full') * dx
    # compute weights: w = np.exp(-α * S)
    w = np.exp(-α * S)
    w /= w.sum()
    # compute gradient of f = - (1/α) log sum exp(-α S) = weighted sum ∇S.
    # compute ∇S as described.
    n = len(h)
    grad = np.zeros_like(h)
    # note w_k corresponds to shift k from -(n-1) to n-1
    for k in range(0, n):
        wk = w[n-1 + k]  # shift k
        if wk == 0:
            continue
        if k == 0:
            grad += 2 * wk * dx * h
        else:
            # overlapping parts
            grad[:n-k] += wk * dx * h[k:]
            grad[k:]   += wk * dx * h[:n-k]
    return grad

import numpy as np
import time

def _project_onto_box_and_sum(v, target_sum):
    """
    Euclidean projection of v onto the set {h | 0 <= h_i <= 1, sum(h)=target_sum}.
    Uses a simple bisection on the Lagrange multiplier λ for the constraint sum(h)=target_sum.
    """
    low, high = -1e6, 1e6
    for _ in range(60):
        lam = (low + high) * 0.5
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            low = lam
        else:
            high = lam
    lam = (low + high) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _evaluate_c5(h, dx):
    """
    Compute the Erdős C5 value for a discretised function h.
    """
    # correlation of h with (1-h), mode='full' gives all integer shifts
    overlap = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(overlap.max())


def _gradient_soft_min(h, dx, alpha):
    """
    Gradient of the smooth approximation of min_{k} S_k,
    where S_k = sum_i h_i h_{i+k} * dx.
    The smooth min is   - (1/α) log Σ_k exp(-α S_k).
    """
    n = h.shape[0]
    # autocorrelation S_k (full length 2n-1)
    corr = np.correlate(h, h, mode='full') * dx
    # soft‑min weights
    w = np.exp(-alpha * corr)
    w /= w.sum()

    # compute gradient ∇f = Σ_k w_k ∇S_k
    grad = np.zeros_like(h)
    off = n - 1                     # index offset for k = 0
    for k in range(0, n):
        wk = w[off + k]
        if wk == 0.0:
            continue
        if k == 0:
            # derivative of S_0 = Σ_i h_i² * dx   => 2*dx*h_i
            grad += 2.0 * wk * dx * h
        else:
            # contributions from h_i * h_{i+k} and h_{i} * h_{i-k}
            grad[: n - k] += wk * dx * h[k:]          # term h_{i+k}
            grad[k:]       += wk * dx * h[: n - k]    # term h_{i-k}
    return grad


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to obtain a low C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine number of discretisation points. Use the provided initial vector length if available.
    if 'initial_h_values' in globals():
        h_initial = np.asarray(initial_h_values, dtype=float)
        n = h_initial.shape[0]
    else:
        n = 200                     # a moderate size (fast evaluation, enough flexibility)
        h_initial = None

    dx = 2.0 / n
    target_sum = n / 2.0            # because ∫ h = 1 => sum h * dx = 1

    # ----------------------------------------------------------------------
    # Helper: evaluate current best and keep global best pair
    # ----------------------------------------------------------------------
    def _maybe_update_best(candidate_h, candidate_c5):
        nonlocal best_h, best_c5
        if candidate_c5 < best_c5:
            best_c5 = candidate_c5
            best_h = candidate_h.copy()

    # ----------------------------------------------------------------------
    # Start from a feasible point
    # ----------------------------------------------------------------------
    if h_initial is not None:
        h0 = _project_onto_box_and_sum(h_initial, target_sum)
    else:
        # simple feasible start: uniform 0.5 plus a tiny random perturbation
        h0 = np.full(n, 0.5) + (np.random.rand(n) - 0.5) * 0.01
        h0 = _project_onto_box_and_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = _evaluate_c5(best_h, dx)

    # ----------------------------------------------------------------------
    # Main optimisation loop – gradient ascent on a smooth approximation of
    # the min‑autocorrelation (equivalently minimising max overlap).
    # ----------------------------------------------------------------------
    # Parameters controlling the optimisation schedule
    max_total_iters = 15000           # generous upper bound, actual count limited by time
    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 3:       # keep a safety margin
            break
        # Adaptive parameters
        alpha = 5.0 + 45.0 * (it / max_total_iters)   # from 5 to 50
        step = 0.5 * (1.0 - it / max_total_iters) + 0.02   # from ~0.52 down to ~0.02

        # Gradient step
        grad = _gradient_soft_min(best_h, dx, alpha)
        h_new = _project_onto_box_and_sum(best_h + step * grad, target_sum)

        # Evaluate
        c5_new = _evaluate_c5(h_new, dx)

        # Keep the best seen so far
        _maybe_update_best(h_new, c5_new)

        # Simple acceptance rule: move if the new point is not worse,
        # otherwise occasionally accept a worsening move to escape shallow minima.
        if c5_new <= _evaluate_c5(best_h, dx) or np.random.rand() < 0.02:
            best_h = h_new

        it += 1
        if it >= max_total_iters:
            break

    # ----------------------------------------------------------------------
    # A quick local polishing pass – random pairwise mass transfers
    # ----------------------------------------------------------------------
    h = best_h.copy()
    c5_curr = _evaluate_c5(h, dx)
    for _ in range(3000):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 2:
            break
        # pick two distinct indices i (donor) and j (receiver)
        i, j = np.random.choice(n, 2, replace=False)
        # amount to transfer (scaled by how far we are from bounds)
        eps = 0.02 * np.random.rand()
        if h[i] <= eps or h[j] >= 1.0 - eps:
            continue
        h_trial = h.copy()
        h_trial[i] -= eps
        h_trial[j] += eps
        # maintain box constraints explicitly
        h_trial[i] = max(h_trial[i], 0.0)
        h_trial[j] = min(h_trial[j], 1.0)
        # sum is already preserved by construction
        c5_trial = _evaluate_c5(h_trial, dx)
        if c5_trial < c5_curr:
            h, c5_curr = h_trial, c5_trial
            if c5_curr < best_c5:
                best_h, best_c5 = h.copy(), c5_curr

    # Final sanity check – ensure constraints are satisfied within tolerance
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_onto_box_and_sum(best_h, target_sum)

    final_c5 = _evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n)
