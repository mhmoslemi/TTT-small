def get_bad_good_indices(h, lag):
    N = len(h)
    bad = []   # i where h[i]==1 and i+lag in range and h[i+lag]==0
    good = []  # j where h[j]==0 and j+lag in range and h[j+lag]==1
    for i in range(N):
        j = i + lag
        if 0 <= j < N:
            if h[i] == 1 and h[j] == 0:
                bad.append(i)
            elif h[i] == 0 and h[j] == 1:
                good.append(i)
    return np.array(bad), np.array(good)

def project_to_sum(v, target, lo=0., hi=1., tol=1e-12, max_iter=100):
    w = np.clip(v, lo, hi)
    for _ in range(max_iter):
        s = w.sum()
        if abs(s - target) <= tol:
            break
        diff = target - s
        free_idx = np.where((w > lo + tol) & (w < hi - tol))[0]
        if len(free_idx) == 0:
            # cannot adjust further, break
            break
        delta = diff / len(free_idx)
        # update w at free indices
        w[free_idx] += delta
        # clip again
        w = np.clip(w, lo, hi)
    return w
```

import time
import numpy as np

def _project_to_sum(v, target, lo=0.0, hi=1.0, tol=1e-12, max_iter=100):
    """
    Euclidean projection of v onto the box [lo, hi] with the additional linear
    constraint sum(v) = target.
    """
    w = np.clip(v, lo, hi)
    for _ in range(max_iter):
        s = w.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        free = (w > lo + tol) & (w < hi - tol)
        n_free = free.sum()
        if n_free == 0:
            break
        delta = diff / n_free
        w[free] += delta
        w = np.clip(w, lo, hi)
    return w


def _evaluate_c5(h):
    """Return the C5 value for the step function h."""
    dx = 2.0 / h.shape[0]
    # correlation h * (1 - h) with zero padding (full mode)
    val = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
    return float(val)


def _worst_lag(h):
    """Return (lag, correlation array) where the overlap is maximal."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    lag = idx - (h.shape[0] - 1)
    return lag, corr


def _gradient_wrt_h(h, lag):
    """Gradient of the worst‑lag overlap term w.r.t. h."""
    n = h.shape[0]
    g = np.zeros(n, dtype=np.float64)
    if lag >= 0:
        L = lag
        # (1 - h[i+L]) part
        g[: n - L] += 1.0 - h[L:]
        # -h[i-L] part
        g[L:] -= h[: n - L]
    else:
        D = -lag
        # (1 - h[i-D]) for i >= D
        g[D:] += 1.0 - h[: n - D]
        # -h[i+D] for i < n-D
        g[: n - D] -= h[D:]
    return g


def _propose_move_gradient(h, lag, step, rng):
    """Propose a mass‑transfer move guided by the gradient at a given lag."""
    grad = _gradient_wrt_h(h, lag)
    # indices where increasing h would decrease the objective (grad < 0)
    inc_idx = np.where((grad < 0) & (h < 1.0 - 1e-12))[0]
    # indices where decreasing h would decrease the objective (grad > 0)
    dec_idx = np.where((grad > 0) & (h > 1e-12))[0]
    if inc_idx.size == 0 or dec_idx.size == 0:
        # fall back to a random safe move
        return _propose_move_random(h, step, rng)

    i = rng.choice(dec_idx)  # we will decrease h[i]
    j = rng.choice(inc_idx)  # we will increase h[j]
    max_delta = min(step, h[i], 1.0 - h[j])
    if max_delta <= 0:
        return h.copy()
    delta = max_delta * rng.random()
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def _propose_move_random(h, step, rng):
    """Random transfer of mass from a positive entry to a non‑full entry."""
    pos = np.where(h > 1e-12)[0]
    neg = np.where(h < 1.0 - 1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return h.copy()
    i = rng.choice(pos)
    j = rng.choice(neg)
    if i == j:
        return h.copy()
    max_delta = min(step, h[i], 1.0 - h[j])
    if max_delta <= 0:
        return h.copy()
    delta = max_delta * rng.random()
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Initialise the step function
    # ------------------------------------------------------------------
    try:
        # use the provided construction if present
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = int(h0.shape[0])
        # ensure exact sum constraint
        h0 = _project_to_sum(h0, target=n / 2.0)
    except Exception:  # NameError or other issues – fall back to a fresh start
        n = 100  # a modest size that stays well below the 1 k limit
        h0 = rng.random(n)
        h0 = _project_to_sum(h0, target=n / 2.0)

    # current state and the best seen so far
    cur_h = h0.copy()
    cur_score = _evaluate_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Simulated‑annealing loop
    # ------------------------------------------------------------------
    max_temp = 0.012
    min_temp = 1e-6
    max_step = 0.45          # max transferable mass at start
    min_step = 0.001         # finer adjustments later
    prob_gradient = 0.70     # probability to use a gradient‑guided move

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed > budget_s * 0.97:      # leave a tiny safety margin
            break

        frac = elapsed / budget_s
        temp = max_temp * (1.0 - frac) + min_temp * frac
        step = max_step * (1.0 - frac) + min_step * frac

        # worst lag for the current configuration
        lag, _ = _worst_lag(cur_h)

        # propose a new configuration
        if rng.random() < prob_gradient:
            cand_h = _propose_move_gradient(cur_h, lag, step, rng)
        else:
            cand_h = _propose_move_random(cur_h, step, rng)

        # all moves keep the total mass unchanged; we only need to re‑clip
        cand_h = np.clip(cand_h, 0.0, 1.0)

        cand_score = _evaluate_c5(cand_h)
        delta = cand_score - cur_score

        # Acceptance test (Metropolis)
        if delta < -1e-13 or rng.random() < np.exp(-delta / (temp + 1e-12)):
            cur_h = cand_h
            cur_score = cand_score
            if cand_score < best_score - 1e-13:
                best_h = cand_h.copy()
                best_score = cand_score

    # ------------------------------------------------------------------
    # Final clean‑up: enforce the exact linear constraint one last time
    # ------------------------------------------------------------------
    best_h = _project_to_sum(best_h, target=n / 2.0)
    best_score = _evaluate_c5(best_h)

    return best_h, best_score, n
