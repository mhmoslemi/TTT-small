def grad_smoothmax(h, dx, alpha):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # smoothed weights
    maxc = np.max(corr)
    exp_alpha = np.exp(alpha * (corr - maxc))
    w = exp_alpha / np.sum(exp_alpha)   # length 2N-1
    N = len(h)
    g = np.zeros(N)

    for k in range(N):
        wk = w[N - 1 + k]
        if wk == 0:

def compute_grad(h, dx, alpha):
    N = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full') * dx
    maxc = np.max(corr)
    # compute normalized weights
    exp_shift = np.exp(alpha * (corr - maxc))
    w = exp_shift / np.sum(exp_shift)
    g = np.zeros_like(h)
    # positive lags and zero
    # we iterate over k, but to avoid separate handling for k=0 and others, unify:
    for k in range(N):
        wk = w[N - 1 + k]  # lag k
        if wk == 0.0:
            continue
        if k == 0:
            # g += wk*dx*(1 - 2*h)  # as derived
            g += wk * dx * (1 - 2 * h)
        else:
            # forward term
            g[:N - k] += wk * dx * (1 - h[k:])
            # backward term
            g[k:] -= wk * dx * h[:N - k]
    # negative lags
    for k in range(1, N):

        if wk == 0.0:

        g[k:] += wk * dx * (1 - h[:N - k])
        # backward term (i <= N-1 - k)
        g[:N - k] -= wk * dx * h[k:]
    return g, np.max(corr)

import numpy as np, time, math
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # get initial solution if given
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    def project_sum_box(v, target):
        # binary search for lambda
        lo = -np.max(v) - 1.0
        hi = 1.0 - np.min(v) + 1.0
        for _ in range(60):
            mid = (lo + hi) / 2.0
            x = np.clip(v + mid, 0.0, 1.0)
            s = x.sum()
            if s < target:
                lo = mid
            else:
                hi = mid
        return np.clip(v + hi, 0.0, 1.0)

    def cmax(h):
        N = h.shape[0]
        dx = 2.0 / N
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return corr.max()

    def smooth_grad(h, dx, alpha):
        N = h.shape[0]
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        m = np.max(corr)
        w_raw = np.exp(alpha * (corr - m))
        w = w_raw / np.sum(w_raw)   # length 2N-1, sum 1
        g = np.zeros_like(h)
        # positive lags (including zero)
        for k in range(N):
            wk = w[N - 1 + k]
            if wk == 0.0:
                continue
            if k == 0:
                g += wk * dx * (1.0 - 2.0 * h)
            else:
                g[:N - k] += wk * dx * (1.0 - h[k:])
                g[k:]     -= wk * dx * h[:N - k]
        # negative lags
        for k in range(1, N):
            wk = w[N - 1 - k]
            if wk == 0.0:
                continue
            g[k:]   += wk * dx * (1.0 - h[:N - k])
            g[:N - k] -= wk * dx * h[k:]
        return g, corr.max()

    # define hierarchical stages
    base_n = 84
    if init_h is not None:
        base_n = max(base_n, init_h.shape[0] * 2)
    stages = [base_n]
    while stages[-1] < 500:
        stages.append(min(stages[-1] * 2, 500))
    # limit to distinct values
    stages = sorted(set(stages))

    best_h = None
    best_val = float('inf')
    for N in stages:
        # time guard
        if time.time() - start > budget_s - 5:
            break
        dx = 2.0 / N
        target_sum = N / 2.0
        # initialize h at this resolution
        if init_h is not None:
            # upsample existing best if available, else init_h
            if best_h is not None:
                src = best_h
            else:
                src = init_h
            x_src = np.linspace(0.0, 2.0, src.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 2.0, N, endpoint=False)
            h = np.interp(x_new, x_src, src)
        else:
            h = np.full(N, 0.5)
        # enforce constraints
        h = project_sum_box(h, target_sum)

        # optimizer parameters
        alpha = 30.0
        lr = 0.4
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        m = np.zeros(N)
        v = np.zeros(N)
        iter = 0
        max_iter = 5000  # generous; will stop early on time
        while iter < max_iter:
            if time.time() - start > budget_s - 5:
                break
            grad, cur_max = smooth_grad(h, dx, alpha)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** (iter + 1))
            v_hat = v / (1 - beta2 ** (iter + 1))
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = h - step
            h_new = project_sum_box(h_new, target_sum)
            # evaluate real max
            cur = cmax(h_new)
            if cur < best_val:
                best_val = cur
                best_h = h_new.copy()
                # also keep best at current resolution for upsampling later
            h = h_new
            iter += 1

    # If time left, try a few random restarts at finest resolution
    if best_h is not None:
        N = best_h.shape[0]
    else:
        N = stages[-1] if stages else 200
    dx = 2.0 / N
    target_sum = N / 2.0
    remaining = budget_s - (time.time() - start)
    if remaining > 20.0:
        restarts = 5
        for _ in range(restarts):
            if time.time() - start > budget_s - 5:
                break
            h = np.random.rand(N)
            h = project_sum_box(h, target_sum)
            m = np.zeros(N)
            v = np.zeros(N)
            lr = 0.4
            for it in range(1000):
                if time.time() - start > budget_s - 5:
                    break
                grad, _ = smooth_grad(h, dx, alpha)
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * (grad * grad)
                m_hat = m / (1 - beta1 ** (it + 1))
                v_hat = v / (1 - beta2 ** (it + 1))
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                h = project_sum_box(h - step, target_sum)
            cur = cmax(h)
            if cur < best_val:
                best_val = cur
                best_h = h.copy()

    # final sanity clip
    if best_h is None:
        # fallback: constant
        N = base_n
        best_h = np.full(N, 0.5)
        best_val = cmax(best_h)

    # ensure constraints are met within tolerance
    best_h = np.clip(best_h, 0.0, 1.0)
    # adjust integral exactly by projection
    best_h = project_sum_box(best_h, best_h.shape[0] / 2.0)

    return (best_h, float(best_val), int(best_h.shape[0]))

import numpy as np
import time

def _project_sum_box(v, target):
    """Project v onto the hyperplane sum(v)=target with box [0,1]."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        if x.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + hi, 0.0, 1.0)


def _cmax(h):
    """Exact objective for a discretised h."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()


def _smooth_grad(h, dx, alpha):
    """Gradient of the smooth max (log‑sum‑exp) of the overlap values."""
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    w_raw = np.exp(alpha * (corr - m))
    w = w_raw / w_raw.sum()                     # length 2N‑1, sum 1
    g = np.zeros_like(h)

    # k >= 0 (including 0)
    for k in range(N):
        wk = w[N - 1 + k]
        if wk == 0.0:
            continue
        if k == 0:
            g += wk * dx * (1.0 - 2.0 * h)
        else:
            g[: N - k] += wk * dx * (1.0 - h[k:])
            g[k:]      -= wk * dx * h[: N - k]

    # k < 0
    for k in range(1, N):
        wk = w[N - 1 - k]
        if wk == 0.0:
            continue
        g[k:]        += wk * dx * (1.0 - h[: N - k])
        g[: N - k]   -= wk * dx * h[k:]

    return g, corr.max()


def _shape_h(p, N):
    """Symmetric piecewise‑linear shape with parameter p∈[0,1]."""
    if p <= 1e-12:
        return np.full(N, 0.5)
    if p >= 1.0:
        x = np.linspace(0.0, 2.0, N, endpoint=False)
        return 1.0 - np.abs(x - 1.0)
    h0 = 1.0 / (2.0 - p)                        # plateau height
    x = np.linspace(0.0, 2.0, N, endpoint=False)
    h = np.empty_like(x)

    left = x < p
    right = x > 2.0 - p
    mid = ~left & ~right

    h[left] = h0 * x[left] / p                  # rising ramp
    h[right] = h0 * (2.0 - x[right]) / p         # falling ramp
    h[mid] = h0
    return h


def _local_optimize(h, budget, start):
    """Few Adam steps starting from h (keeps constraints)."""
    N = h.shape[0]
    dx = 2.0 / N
    target = N / 2.0
    alpha = 30.0
    lr = 0.45
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    while True:
        if time.time() - start > budget - 5:
            break
        grad, _ = _smooth_grad(h, dx, alpha)
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = _project_sum_box(h - step, target)
        h = h_new
        if it >= 2000:
            break
    return h
