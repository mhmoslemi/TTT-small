def gradient_for_shift(h, shift):
    N = len(h)
    g = np.zeros(N)
    if shift >= 0:
        # i from 0 to N-1-shift
        i = np.arange(N - shift)
        # first part: + (1 - h[i+shift])
        g[i] += 1 - h[i + shift]
        # second part: - h[i]
        g[i + shift] -= h[i]
    else:
        # shift negative: shift = -s where s>0
        s = -shift
        # i from s to N-1: mapping i to i-s  (since i+l where l negative)
        i = np.arange(s, N)
        # first part + (1 - h[i - s])
        g[i] += 1 - h[i - s]
        # second part - h[i]
        g[i - s] -= h[i]
    return g

def proj_box_simplex(v, total):
    # v is original vector.
    # We need to find lambda such that x_i = min(max(v_i + lambda, 0), 1) and sum(x_i) = total.
    # We can solve via bisection on lambda.
    lo = -np.max(v) - 1  # low bound (so all values + lo <= 0)
    hi = 1 - np.min(v) + 1  # high bound (so all values + hi >=1)
    for _ in range(50):
        mid = (lo + hi) / 2
        x = np.clip(v + mid, 0, 1)
        s = x.sum()
        if s < total:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo+hi)/2, 0, 1)   # final
    # May be small adjustment to ensure sum close to total due to clipping.
    # We'll also adjust by adding uniform delta to free variables.
    # But given tolerance fine.
    return x

def compute_c5(h):

    corr = np.correlate(h, 1 - h, mode='full') * dx

def proj_box_simplex(v, total):
    # Use binary search on lambda.
    lower = -np.max(v) - 1.0  # ensure all v+lower <=0
    upper = 1 - np.min(v) + 1.0  # ensure all v+upper >= 1
    for _ in range(60):
        mid = (lower + upper) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < total:
            lower = mid
        else:
            upper = mid
    x = np.clip(v + (lower + upper) / 2.0, 0.0, 1.0)
    # Simple fix to meet total exactly: adjust remaining slack equally among non-bound entries.
    diff = total - x.sum()
    if abs(diff) > 1e-8:
        # Distribute diff over free variables (those not at bounds)
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.sum() > 0:
            x[free] += diff / free.sum()
            # Clip again
            x = np.clip(x, 0.0, 1.0)
    return x

def gradient_for_shift(h, shift):
    N = len(h)
    g = np.zeros(N, dtype=np.float64)
    if shift >= 0:
        i = np.arange(N - shift)
        g[i] += 1 - h[i + shift]
        g[i + shift] -= h[i]
    else:
        s = -shift
        i = np.arange(s, N)
        g[i] += 1 - h[i - s]
        g[i - s] -= h[i]
    return g

import time
import random
import numpy as np

# ------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1‑h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def proj_box_simplex(v: np.ndarray, total: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x | 0≤x_i≤1, sum(x)=total}.
    Uses a binary search for the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0          # makes v+lo ≤ 0  → all clipped to 0
    hi = 1.0 - np.min(v) + 1.0      # makes v+hi ≥ 1  → all clipped to 1
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < total:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

    # tiny adjustment to hit the sum exactly (if a few free vars exist)
    diff = total - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            add = diff / free.sum()
            x[free] += add
            x = np.clip(x, 0.0, 1.0)
    return x


def gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Sub‑gradient of the overlap functional for the shift that currently
    attains the maximum.  shift ∈ [-(n‑1), …, n‑1].
    """
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        i = np.arange(n - shift)
        g[i] += 1.0 - h[i + shift]
        g[i + shift] -= h[i]
    else:
        s = -shift
        i = np.arange(s, n)
        g[i] += 1.0 - h[i - s]
        g[i - s] -= h[i]
    return g


def upsample_h(orig: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation of a step function to a different grid size."""
    n0 = orig.shape[0]
    if n0 == target_n:
        return orig.copy()
    x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    return np.interp(x_new, x_old, orig)

    """

    np.random.seed(seed)
    random.seed(seed)

    # -----------------------------------------------------------------
    # 1. Initialise a feasible h (use the supplied parent if available)
    # -----------------------------------------------------------------
    if "initial_h_values" in globals():

    # Gradient‑descent parameters

    max_iters = 15000                     # safety cap – will break on time

    cur_c5 = best_c5

    # -----------------------------------------------------------------
    while time.time() - start < budget_s - 1.0:

        max_dec = best_h[i]
        max_inc = 1.0 - best_h[j]
        if max_dec == 0 or max_inc == 0:

        delta = min(max_dec, max_inc) * random.random() * 0.3
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new

    # Final projection (ensures exact feasibility) and re‑evaluation
    best_h = proj_box_simplex(best_h, total_mass)
    best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), int(best_h.shape[0])
