def proj_box_simplex(v, target):
    # v: array
    # target: scalar sum
    # Returns projection onto [0,1] box with sum constraint.
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        y = np.clip(v - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return y
