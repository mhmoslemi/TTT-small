def propose_move(h, rng):
    N = len(h)
    i, j = rng.integers(0, N, size=2, endpoint=False)
    while i == j:
        j = rng.integers(0, N)
    # compute eps range
    low = max(-h[i], -(1 - h[j]))
    high = min(1 - h[i], h[j])
    if high <= low:
        return None # no feasible move
    eps = rng.random() * (high - low) + low
    new_h = h.copy()
    new_h[i] += eps
    new_h[j] -= eps
    return new_h

import time, math
import numpy as np

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a step function h (as a numpy array), its C5 value, and the number of points.
    The algorithm performs a simple simulated‑annealing search while respecting
    the constraints 0 ≤ h ≤ 1 and sum(h) = n/2.
    """
    rng = np.random.default_rng(seed)

    # ---------- fast C5 evaluator ----------
    def fast_c5(h: np.ndarray) -> float:
        """Integral C5 = max_k ∫ h(x) (1-h(x+k)) dx, discretized."""
        n = h.size
        dx = 2.0 / n
        # full correlation, zero padding outside the interval
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * dx

    # ---------- obtain initial vector ----------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.size
        # enforce sum constraint if necessary (very small adjustments)
        target_sum = n / 2.0
        cur_sum = h0.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            # simple projection onto the hyperplane while staying in [0,1]
            diff = target_sum - cur_sum
            free = np.where((h0 > 1e-12) & (h0 < 1 - 1e-12))[0]
            if free.size == 0:
                # fallback: distribute uniformly (should never happen)
                h0 += diff / n
                h0 = np.clip(h0, 0.0, 1.0)
            else:
                add = diff / free.size
                h0[free] += add
                h0 = np.clip(h0, 0.0, 1.0)
    else:
        # construct a simple feasible start (half 1, half 0, one 0.5 if needed)
        n = 100
        h0 = np.zeros(n, dtype=np.float64)
        half = n // 2
        h0[:half] = 1.0
        if n % 2:
            h0[half] = 0.5
        rng.shuffle(h0)

    # ---------- initial state ----------
    cur_h = h0.copy()
    curC = fast_c5(cur_h)
    best_h = cur_h.copy()
    bestC = curC

    target_bound = 0.38085  # threshold we aim to beat
    if bestC <= target_bound:

    # ---------- annealing schedule ----------
    T0 = max(0.5, curC)      # start temperature (generous)
    T_end = 1e-7             # final temperature
    start_time = time.time()

    # ---------- main search loop ----------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= allowed:

        # linear cooling based on elapsed fraction
        progress = elapsed / allowed
        T = T0 * (1.0 - progress) + T_end * progress

        # propose a move: transfer a feasible amount epsilon between two indices
        i = rng.integers(n)
        j = rng.integers(n)
        while j == i:
            j = rng.integers(n)

        # feasible interval for epsilon so that both entries stay in [0,1]
        lo = max(-cur_h[i], -(1.0 - cur_h[j]))
        hi = min(1.0 - cur_h[i], cur_h[j])
        if hi <= lo:

        new_h = cur_h.copy()
        new_h[i] += eps
        new_h[j] -= eps

        newC = fast_c5(new_h)

        # Metropolis acceptance
        if newC < curC or rng.random() < math.exp(-(newC - curC) / T):
            cur_h = new_h
            curC = newC
            if newC < bestC:
                bestC = newC
                best_h = new_h.copy()
                if bestC <= target_bound:
