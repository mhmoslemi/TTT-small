### Functions

1. `project_to_feasible(v, total_sum)`:
    - Input vector v (numpy array). Output vector p satisfying 0 ≤ p_i ≤ 1 and sum(p) = total_sum.
    Steps: Clip p = np.clip(v, 0, 1). Compute diff = total_sum - p.sum().
    While abs(diff) > 1e-12:
        - find indices where p_i not at bound considering diff sign: if diff>0 then indices where p_i < 1 else indices where p_i > 0.
        - if none, break (cannot adjust).
        - add inc = diff / len(free) to those entries.
        - update p[free] = np.clip(p[free] + inc, 0, 1)
        - recompute diff.
    This iterative approach should converge in few iterations because diff decreases at each step and bound indices may become saturated.

            bwd = shift_array(h, -k)
            grad += (1.0 - fwd - bwd)

        conv_new = correlation_full(h_new)
        new_val = conv_new.max() * dx

        # compute temperature decaying
        temp = 0.001 * step0 / np.sqrt(it + 1)

    best_val = compute_C5(best_h, dx)

def project_to_feasible(v, total_sum):
    # Clip after shifting by lambda (water-filling)
    # We'll use bisection
    v = np.asarray(v, dtype=np.float64)
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        lam = (lo + hi) / 2.0
        p = np.clip(v - lam, 0.0, 1.0)
        s = p.sum()
        if s > total_sum:
            lo = lam
        else:
            hi = lam
    p = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust possible minor sum error by scaling residual on free variables
    diff = total_sum - p.sum()
    if abs(diff) > 1e-10:
        # distribute diff uniformly among not-at-bound indices
        free = np.where((p > 1e-12) & (p < 1 - 1e-12))[0]
        if free.size > 0:
            inc = diff / free.size
            p[free] += inc
            # clip again just in case
            p = np.clip(p, 0.0, 1.0)
    return p

import time
import math
import numpy as np

# ----- helper functions -----------------------------------------------------


def shift_array(arr: np.ndarray, k: int) -> np.ndarray:
    """Shift arr by k positions with zero padding (positive = right shift)."""
    n = arr.shape[0]
    out = np.empty_like(arr)
    if k >= 0:
        out[: n - k] = arr[k:]
        out[n - k :] = 0.0
    else:
        k = -k
        out[k:] = arr[: n - k]
        out[:k] = 0.0
    return out


def correlation_full(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h) using FFT (same as np.correlate)."""
    n = h.shape[0]
    L = 2 * n
    H = np.fft.fft(h, L)
    G = np.fft.fft((1.0 - h)[::-1], L)
    conv = np.fft.ifft(H * G).real[: 2 * n - 1]
    return conv


def compute_c5(h: np.ndarray):
    """Return (C5 value, full correlation array) for step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = correlation_full(h)
    return conv.max() * dx, conv


def project_to_feasible(v: np.ndarray, total_sum: float) -> np.ndarray:
    """
    Project vector v onto the set {0<=x_i<=1, sum x_i = total_sum}
    using a water‑filling (bisection) scheme.
    """
    v = np.asarray(v, dtype=np.float64)
    lo = np.min(v) - 1.0  # ensures v - lo >= 1 for all i
    hi = np.max(v)        # ensures v - hi <= 0 for all i
    for _ in range(60):
        lam = (lo + hi) * 0.5
        p = np.clip(v - lam, 0.0, 1.0)
        s = p.sum()
        if s > total_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    p = np.clip(v - lam, 0.0, 1.0)

    # tiny residual correction (very rare)
    diff = total_sum - p.sum()
    if abs(diff) > 1e-12:
        free = np.where((p > 1e-12) & (p < 1.0 - 1e-12))[0]
        if free.size:
            p[free] += diff / free.size
            p = np.clip(p, 0.0, 1.0)
    return p


def subgradient(h: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """Average subgradient of C5 over the active shifts."""
    if shifts.size == 0:
        return np.zeros_like(h)
    grad = np.zeros_like(h)
    for k in shifts:
        grad += 1.0 - shift_array(h, k) - shift_array(h, -k)
    grad /= float(shifts.size)
    return grad
