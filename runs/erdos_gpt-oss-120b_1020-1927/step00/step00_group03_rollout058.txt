import time, numpy as np

def _project_box_sum(v, target):
    """Project vector v onto {0<=x<=1, sum(x)=target} via bisection on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _eval_c5(vec):
    """Exact evaluation used by the harness."""
    dx = 2.0 / vec.size
    corr = np.correlate(vec, 1 - vec, mode="full")
    return float(corr.max() * dx)


def _softmax_gradient(h, beta):
    """Gradient of the smooth max L = (1/beta) log sum exp(beta*f) where
    f = correlation(h, 1-h). Returns (grad, f)."""
    n = h.size
    f = np.correlate(h, 1 - h, mode="full")
    f_max = f.max()
    # stable softmax weights
    w = np.exp(beta * (f - f_max))
    w /= w.sum()

    ks = np.arange(-(n - 1), n)          # shift values corresponding to w indices
    grad = np.zeros_like(h)

    for weight, k in zip(w, ks):
        if weight == 0.0:
            continue
        if k >= 0:
            m = n - k                     # length of overlapping region
            # +1 on the first m positions
            grad[:m] += weight
            # -h[i+k] on the first m positions
            grad[:m] -= weight * h[k:]
            # -h[i] on the shifted positions
            grad[k:] -= weight * h[:m]
        else:
            kk = -k
            m = n - kk
            grad[kk:] += weight
            grad[kk:] -= weight * h[:m]
            grad[:m] -= weight * h[kk:]

    return grad, f


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) initialise h
    # ------------------------------------------------------------------
    try:                                   # use the parent construction if present
        h = np.array(initial_h_values, dtype=np.float64)
        n = h.size
    except NameError:                      # fall back to a simple uniform start
        n = 200
        h = np.full(n, 0.5, dtype=np.float64)

    target_sum = n * 0.5
    h = _project_box_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = _eval_c5(best_h)

    # ------------------------------------------------------------------
    # 2) optimisation parameters
    # ------------------------------------------------------------------
    start_time = time.time()
    it = 0
    # schedule limits – they adapt as the loop runs
    beta_start, beta_end = 0.5, 8.0
    lr_start,   lr_end   = 0.6, 0.02
    max_iters = 100000   # generous upper bound; real stop is time budget

    while it < max_iters and (time.time() - start_time) < (budget_s - 0.5):
        # linear schedules based on progress
        frac = it / max_iters
        beta = beta_start + (beta_end - beta_start) * frac
        lr   = lr_start   + (lr_end   - lr_start)   * frac

        grad, _ = _softmax_gradient(h, beta)

        # gradient‐descent step and projection back onto feasibility set
        h = _project_box_sum(h - lr * grad, target_sum)

        # keep the best observed solution
        cur_c5 = _eval_c5(h)

        # occasional random shake to escape possible plateaus
        if it % 500 == 0 and it > 0:
            shake = rng.normal(scale=0.02, size=n)
            h = _project_box_sum(best_h + shake, target_sum)

        it += 1

    # final safety evaluation
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, int(n)
