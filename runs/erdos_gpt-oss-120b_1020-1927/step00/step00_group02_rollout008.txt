def evaluate_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

    return np.correlate(h, h, mode='full')

def shifted_by(h, shift):
    n = h.shape[0]
    if shift >= 0:
        # shift to the left: h[i+shift]
        return np.concatenate((h[shift:], np.zeros(shift, dtype=h.dtype)))
    else:
        # negative shift: pad at front
        s = -shift
        return np.concatenate((np.zeros(s, dtype=h.dtype), h[:n - s]))

import time, math

def evaluate_c5(h: np.ndarray) -> float:
    """Integral max_{k} h(x)(1-h(x+k)) approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def correlation_self(h: np.ndarray) -> np.ndarray:
    """Self‑correlation (overlap) of h."""
    return np.correlate(h, h, mode="full")

def shifted_by(h: np.ndarray, shift: int) -> np.ndarray:
    """Zero‑padded shift of vector h (positive = forward)."""
    n = h.shape[0]
    if shift >= 0:
        return np.concatenate([h[shift:], np.zeros(shift, dtype=h.dtype)])
    else:
        s = -shift
        return np.concatenate([np.zeros(s, dtype=h.dtype), h[: n - s]])
