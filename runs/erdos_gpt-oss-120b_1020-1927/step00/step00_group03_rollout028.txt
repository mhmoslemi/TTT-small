def simulated_annealing(h0, max_time):
    h_cur = h0.copy()
    cur_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = cur_c5
    n = len(h_cur)
    K = n//2
    ones = np.flatnonzero(h_cur == 1)
    zeros = np.flatnonzero(h_cur == 0)
    start = time.time()
    # predetermined iterations steps
    while time.time() - start < max_time:
        # maybe schedule temperature based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / max_time
        T = max(0.5 * (1 - frac), 0.001)
        # propose swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # make swap
        h_cur[i] = 0
        h_cur[j] = 1
        new_c5 = compute_c5(h_cur)
        delta = new_c5 - cur_c5
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            # accept
            cur_c5 = new_c5
            # update indices
            # (swap sets)
            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = h_cur.copy()
                # early exit if below target
                if best_c5 <= 0.3808:
                    break
        else:
            # reject swap: revert
            h_cur[i] = 1
            h_cur[j] = 0
        # optionally also occasional random perturbations
    return best_h, best_c5

import time
import math
import numpy as np

def _c5_of(h):
    """Compute the C5 value for a binary (or 0‑1) vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1‑h); mode='full' gives all integer shifts
    c = np.correlate(h, 1 - h, mode="full")
    return float(c.max() * dx)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discrete vector) with
    integral 1 (i.e. sum(h)=n/2) that minimizes
        C5 = max_k ∫ h(x)(1‑h(x+k))dx .
    The evaluation used by the harness is exactly the same as
    np.correlate(h, 1‑h, mode='full')*dx.

    The algorithm is a simple simulated‑annealing on binary vectors.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and a feasible starting point
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.shape[0]
        # sanity check – fall back to a random binary vector if it is not feasible
        if not (np.all((h0 >= 0) & (h0 <= 1)) and
                abs(h0.sum() * (2.0 / n) - 1.0) < 1e-9):
            h0 = None
    else:
        h0 = None
        n = 70                     # a safe default (even, small enough for speed)

    # ensure even length (required for sum = n/2 integer)
    if n % 2 == 1:
        n -= 1

    half = n // 2                 # number of ones required
    dx = 2.0 / n

    # start from the supplied vector if it is feasible, otherwise from a random binary one
    if h0 is not None:
        cur_h = h0.copy()
    else:
        cur_h = np.zeros(n, dtype=int)
        cur_h[rng.choice(n, size=half, replace=False)] = 1

    cur_c5 = _c5_of(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ------------------------------------------------------------------
    # Simulated annealing
    # ------------------------------------------------------------------
    start_time = time.time()
    # set a small target; exiting early if we beat it
    target_c5 = 0.3808

    # current index lists (updated only when a move is accepted)
    ones_idx = np.flatnonzero(cur_h == 1)
    zeros_idx = np.flatnonzero(cur_h == 0)

    # main loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.2:      # keep a tiny margin for the return
            break

        # linear temperature schedule
        T = max(0.5 * (1.0 - elapsed / budget_s), 0.0005)

        # propose a swap of a 1 and a 0 (keeps the sum constraint exact)
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)

        # perform the swap on the current solution
        cur_h[i] = 0
        cur_h[j] = 1
        new_c5 = _c5_of(cur_h)
        delta = new_c5 - cur_c5

        accept = False
        if delta <= 0.0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_c5 = new_c5
            # update the index lists (cheap for n≈70)
            ones_idx = np.flatnonzero(cur_h == 1)
            zeros_idx = np.flatnonzero(cur_h == 0)

            # record best solution seen so far
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                if best_c5 <= target_c5:
                    break
        else:
            # revert the move
            cur_h[i] = 1
            cur_h[j] = 0

        # occasional random restart to escape possible plateaus
        if rng.random() < 0.001:
            cur_h[:] = 0
            cur_h[rng.choice(n, size=half, replace=False)] = 1
            cur_c5 = _c5_of(cur_h)
            ones_idx = np.flatnonzero(cur_h == 1)
            zeros_idx = np.flatnonzero(cur_h == 0)

    # ------------------------------------------------------------------
    # Final quick local improvement (deterministic hill climbing)
    # ------------------------------------------------------------------
    # try a bounded number of random swaps that improve the best solution
    attempts = 0
    max_attempts = 20000
    cur_h = best_h.copy()
    cur_c5 = best_c5
    ones_idx = np.flatnonzero(cur_h == 1)
