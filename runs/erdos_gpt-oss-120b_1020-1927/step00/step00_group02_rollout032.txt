def project_box_sum(v, S):
    lo = -1e6
    hi = 1e6
    for _ in range(100):
        lam = (lo + hi) / 2
        h = np.clip(v + lam, 0, 1)
        s = h.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    h = np.clip(v + (lo+hi)/2, 0, 1)
    # adjust if due to rounding sum not exactly S, do small correction:
    diff = S - h.sum()
    if abs(diff) > 1e-9:
        # distribute diff among non-bound variables
        # find indices where 0< h_i <1 (free)
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            # equally distribute small correction
            h[free] += diff / free.size
            # clip again and ignore small errors
            h = np.clip(h, 0, 1)
        else:
            # if no free vars, adjust by adding diff to first index (unlikely)
            h[0] = np.clip(h[0] + diff, 0, 1)
    return h

def project_box_sum(v, S):
    # find lambda such that sum(clip(v + lambda, 0, 1)) = S
    lo = -1.0
    hi = 1.0
    # extend range but ensure hi enough; we can set lo=-max(v), hi=1-max(v)

        h = np.clip(v + lam, 0.0, 1.0)
        s = h.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    h = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # Usually sum will be very close; adjust small diff if needed.
    diff = S - h.sum()
    if abs(diff) > 1e-8:

        if free.any():

            h[free] += adj
            h = np.clip(h, 0.0, 1.0)

            # no free variables; adjust by adding diff to first element within bounds
            idx = np.argmax(v)  # pick arbitrary
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def init_h(n):

        # Interpolate to n size
        x0 = (np.arange(n0) + 0.5) * (2.0 / n0)

import numpy as np
import time

def _project_box_sum(v, target_sum):
    """Project vector v onto the box [0,1] with a fixed sum = target_sum."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        h = np.clip(v + lam, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            hi = lam
        else:
            lo = lam
    h = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if round‑off missed the sum
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback: adjust first element (should be rare)
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def _init_binary_h(n):
    """Initialize a binary step function of length n, respecting the sum constraint."""
    target_ones = n // 2
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        n0 = base.size
        # linear interpolation to size n
        x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
        x = (np.arange(n) + 0.5) * (2.0 / n)
        h_cont = np.interp(x, x0, base)
        # pick the top target_ones entries
        idx = np.argsort(-h_cont)[:target_ones]
        h = np.zeros(n, dtype=int)
        h[idx] = 1
        return h
    # otherwise a random binary vector
    perm = np.random.permutation(n)
    h = np.zeros(n, dtype=int)
    h[perm[:target_ones]] = 1
    return h

def _max_mismatch(h):
    """Return max_{k} ∫ h(x)(1‑h(x+k)) dx approximated on the grid (without the dx factor)."""
    # h is a 0/1 ndarray
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max()

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    # ------------------------------------------------------------
    # parameters
    n = 200                     # number of sample points (even)
    dx = 2.0 / n
    target_c5 = 0.3808          # we try to beat this
    # ------------------------------------------------------------
    # initial solution
    h = _init_binary_h(n)
    cur_val = _max_mismatch(h) * dx
    best_h = h.copy()
    best_val = cur_val

    start_time = time.time()
    # Simple simulated‑annealing / greedy loop
    iteration = 0
    while time.time() - start_time < budget_s - 0.3:
        # recompute correlation and the worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        cur_val = max_corr * dx
        # store best so far
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
            if best_val <= target_c5:
                break

        # locate all shifts attaining the maximum (pick one at random)
        max_idxs = np.flatnonzero(corr == max_corr)
        idx = max_idxs[np.random.randint(len(max_idxs))]
        shift = idx - (n - 1)          # shift 0 corresponds to index n‑1
        s = abs(shift)

        # ----- propose a swap ---------------------------------
        # try a targeted swap that directly improves the worst shift
        accepted_swap = False
        if s > 0:
            shifted = np.concatenate([h[s:], np.zeros(s, dtype=int)])
            left_one = np.where((h == 1) & (shifted == 0))[0]
            right_one = np.where((h == 0) & (shifted == 1))[0]
            if left_one.size and right_one.size:
                i = left_one[np.random.randint(left_one.size)]
                j = right_one[np.random.randint(right_one.size)]
                accepted_swap = True
        if not accepted_swap:
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            i = zeros[np.random.randint(zeros.size)]
            j = ones[np.random.randint(ones.size)]

        # apply swap on a copy
        h_new = h.copy()
        h_new[i] = 1
        h_new[j] = 0

        # evaluate new solution
        new_corr = np.correlate(h_new, 1 - h_new, mode='full')
        new_val = new_corr.max() * dx

        # ----- acceptance rule (simulated annealing) ----------
        elapsed = time.time() - start_time
        T = max(0.005, 0.05 * (1.0 - elapsed / budget_s))   # linear cooling
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - cur_val) / T):
                accept = True
        if accept:
            h = h_new
            cur_val = new_val

        iteration += 1

    # final best solution in list form
    return (best_h.astype(float).tolist(), best_val, n)
