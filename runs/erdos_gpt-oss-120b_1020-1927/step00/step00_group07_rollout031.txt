def project_h(h):
    # clip to [0,1]
    np.clip(h, 0.0, 1.0, out=h)
    total = np.sum(h)
    target = len(h) / 2  # n/2
    diff = target - total
    if abs(diff) > 1e-12:
        # adjust the first element that can accommodate diff
        for idx in range(len(h)):
            cap = 1.0 - h[idx] if diff > 0 else h[idx]
            if cap > 1e-12:
                delta = min(abs(diff), cap)
                if diff > 0:
                    h[idx] += delta
                else:
                    h[idx] -= delta
                diff = target - np.sum(h)
                if abs(diff) < 1e-12:
                    break
    return h

import numpy as np, random, time, math

def compute_cost(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h):
    n = len(h)
    target = n / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    total = np.sum(h)
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # distribute diff greedily
    for i in range(n):
        if diff > 0:
            cap = 1.0 - h[i]
            if cap > 0:
                delta = min(cap, diff)
                h[i] += delta
                diff -= delta
        else:
            cap = h[i]
            if cap > 0:
                delta = min(cap, -diff)
                h[i] -= delta
                diff += delta
        if abs(diff) < 1e-12:
            break
    # final clip
    np.clip(h, 0.0, 1.0, out=h)
    return h

def generate_beatty_binary(n):
    phi = (1 + 5 ** 0.5) / 2  # golden ratio
    vals = (np.arange(n) * phi) % 1.0
    h = (vals < 0.5).astype(np.float64)
    target = n / 2.0
    diff = target - h.sum()
    if diff > 0:
        # turn some 0s into 1s
        zero_idx = np.where(h == 0)[0]
        h[zero_idx[:int(diff)]] = 1.0
    elif diff < 0:
        one_idx = np.where(h == 1)[0]
        h[one_idx[:int(-diff)]] = 0.0
    return h

def random_binary(n):
    h = np.zeros(n, dtype=np.float64)
    ones = np.random.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def shaped_candidate(n, a):
    # a in [0,1] fraction of domain occupied by each ramp
    dx = 2.0 / n
    ramp_len = a  # length in continuous domain
    r = max(1, int(round(ramp_len / dx)))
    central_len = n - 2 * r
    if central_len < 0:
        r = n // 4
        central_len = n - 2 * r
    c = 1.0 / (2.0 - a)
    if c > 1.0:
        c = 1.0
    h = np.empty(n, dtype=np.float64)
    # left ramp
    h[:r] = np.linspace(0.0, c, r, endpoint=False)
    # central constant
    if central_len > 0:
        h[r:r+central_len] = c
    # right ramp descending
    h[-r:] = np.linspace(c, 0.0, r, endpoint=False)
    # adjust sum
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if central_len > 0:
            h[r:r+central_len] += diff / central_len
        else:
            h[:r] += diff / r
        np.clip(h, 0.0, 1.0, out=h)
    return h

def local_search(start_h, budget_s, start_time):
    n = len(start_h)
    h = start_h.copy()
    cost = compute_cost(h)
    best_h = h.copy()
    best_cost = cost
    no_improve = 0
    T0 = 0.02
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
        # temperature and step size decay
        T = T0 * (1.0 - elapsed / budget_s)
        step_factor = max(0.01, 0.5 * (1.0 - elapsed / budget_s))
        # pick donor (h>0) and receiver (h<1)
        pos = np.where(h > 1e-12)[0]
        neg = np.where(h < 1.0 - 1e-12)[0]
        if pos.size == 0 or neg.size == 0:
            break
        i = np.random.choice(pos)
        j = np.random.choice(neg)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * step_factor
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_cost = compute_cost(h_new)
        accept = False
        if new_cost < cost:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_cost - cost) / T)
                if np.random.rand() < prob:
                    accept = True
        if accept:
            h = h_new
            cost = new_cost
            if cost < best_cost - 1e-12:
                best_h = h.copy()
                best_cost = cost
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        # occasional restart if stuck
        if no_improve > 20000:
            # perturb current solution a bit
            perturb = np.random.randn(n) * 0.01
            h = np.clip(best_h + perturb, 0.0, 1.0)
            # reproject sum
            h = project_h(h)
            cost = compute_cost(h)
            no_improve = 0
    # final projection of best
    best_h = project_h(best_h)
    best_cost = compute_cost(best_h)
    return best_h, best_cost

    np.random.seed(seed)
    start_time = time.time()
    # Determine problem size
    if 'initial_h_values' in globals():

import numpy as np
import random
import time
import math

def compute_cost(h: np.ndarray) -> float:
    """Return C5 for a given vector h using full correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum_i h_i = n/2 exactly."""
    n = h.shape[0]
    target = n / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    total = float(np.sum(h))
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # Greedy redistribution of the remaining mass
    for i in range(n):
        if diff > 0:
            cap = 1.0 - h[i]
            if cap > 0:
                delta = min(cap, diff)
                h[i] += delta
                diff -= delta
        else:
            cap = h[i]
            if cap > 0:
                delta = min(cap, -diff)
                h[i] -= delta
                diff += delta
        if abs(diff) < 1e-12:
            break
    np.clip(h, 0.0, 1.0, out=h)
    return h


def random_binary(n: int) -> np.ndarray:
    """Random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.float64)
    ones = np.random.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def generate_beatty_binary(n: int) -> np.ndarray:
    """Binary sequence obtained from the Beatty (golden‑ratio) construction."""
    phi = (1.0 + 5.0 ** 0.5) / 2.0
    vals = (np.arange(n, dtype=np.float64) * phi) % 1.0
    h = (vals < 0.5).astype(np.float64)
    target = n / 2.0
    diff = int(round(target - h.sum()))
    if diff > 0:
        zero_idx = np.where(h == 0)[0]
        h[zero_idx[:diff]] = 1.0
    elif diff < 0:
        one_idx = np.where(h == 1)[0]
        h[one_idx[: -diff]] = 0.0
    return h


def shaped_candidate(n: int, a: float) -> np.ndarray:
    """
    Piecewise linear shape:
      - left ramp of length a,
      - middle plateau of height c,
      - right ramp (mirror of left).
    Parameter a ∈ [0,1] (continuous length of each ramp).
    """
    dx = 2.0 / n
    ramp_len = max(1, int(round(a / dx)))          # at least one point
    central_len = n - 2 * ramp_len
    if central_len < 0:                            # fallback for large a
        ramp_len = n // 4
        central_len = n - 2 * ramp_len
    c = 1.0 / (2.0 - a) if a < 2.0 else 1.0
    c = min(1.0, c)

    h = np.empty(n, dtype=np.float64)
    h[:ramp_len] = np.linspace(0.0, c, ramp_len, endpoint=False)
    if central_len > 0:
        h[ramp_len:ramp_len + central_len] = c
    h[-ramp_len:] = np.linspace(c, 0.0, ramp_len, endpoint=False)

    # Fix the total integral to 1 (sum = n/2)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if central_len > 0:
            h[ramp_len:ramp_len + central_len] += diff / central_len
        else:
            h[:ramp_len] += diff / ramp_len
        np.clip(h, 0.0, 1.0, out=h)
    return h


def local_search(start_h: np.ndarray, budget_s: float, start_time: float):
    """Simulated‑annealing‑style hill climbing respecting the constraints."""
    n = start_h.shape[0]
    h = start_h.copy()
    cost = compute_cost(h)
    best_h = h.copy()
    best_cost = cost
    no_improve = 0
    T0 = 0.02                     # initial temperature
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
        # temperature and step size decay linearly
        T = T0 * (1.0 - elapsed / budget_s)
        step_factor = max(0.01, 0.5 * (1.0 - elapsed / budget_s))

        # choose a donor (h>0) and a receiver (h<1)
        donors = np.where(h > 1e-12)[0]
        receivers = np.where(h < 1.0 - 1e-12)[0]
        if donors.size == 0 or receivers.size == 0:
            break
        i = np.random.choice(donors)
        j = np.random.choice(receivers)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * np.random.rand() * step_factor
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_cost = compute_cost(h_new)

        accept = False
        if new_cost < cost:
            accept = True
        else:
            if T > 0.0:
                prob = math.exp(-(new_cost - cost) / T)
                if np.random.rand() < prob:
                    accept = True
        if accept:
            h = h_new
            cost = new_cost
            if cost < best_cost - 1e-12:
                best_h = h.copy()
                best_cost = cost
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional perturbation if stuck
        if no_improve > 20000:
            pert = np.random.randn(n) * 0.01
            h = np.clip(best_h + pert, 0.0, 1.0)
            h = project_h(h)
            cost = compute_cost(h)
            no_improve = 0

    best_h = project_h(best_h)
    best_cost = compute_cost(best_h)
    return best_h, best_cost


def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) meeting the Erdos C5 constraints."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size
    if 'initial_h_values' in globals():
        init_h = np.asarray(initial_h_values, dtype=np.float64)
        n = init_h.shape[0]
    else:
        n = 80
        init_h = None

    # Build a small pool of diverse starting points
    candidates = []
    if init_h is not None:
        candidates.append(init_h.copy())
    candidates.append(random_binary(n))
    candidates.append(generate_beatty_binary(n))
    for a in (0.1, 0.3, 0.5, 0.7, 0.9):
        candidates.append(shaped_candidate(n, a))

    # Keep the best of the raw candidates

    best_cost = float('inf')

        cst = compute_cost(cand)

            best_h = cand.copy()
            best_cost = cst

    # Refine with local search

    # Final safety projection
    best_h = project_h(best_h)
    best_cost = compute_cost(best_h)

    return best_h, best_cost, n
