def project_boxed_simplex(v, target, lower=0., upper=1., max_iter=100):
    # Solve min ||x - v||^2 s.t. lower <= x <= upper, sum x = target
    # using bisection on lambda for x_i = clip(v_i - lam, lower, upper)

    hi = np.max(v - lower) + 1.0
    for _ in range(60):

        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2, lower, upper)

def gradient_step(h, lag, step_max):
    # compute direction vector
    n = len(h)
    left = np.zeros(n)
    right = np.zeros(n)
    if lag > 0:
        left[lag:] = h[:-lag]  # h[i-lag] for i >= lag
        right[:n-lag] = 1 - h[lag:]  # i where i+lag valid
    elif lag < 0:
        L = -lag
        left[:n-L] = h[L:]  # i-L = i+lag valid for i < n-L (since i-L >=0)
        right[L:] = 1 - h[:n-L]  # i+lag valid for i >= L: i+lag = i - L
    else: # lag ==0
        left = h.copy()
        right = 1 - h
    direction = left - right
    # choose i able to increase (direction positive and h_i < 1)
    inc_candidates = np.where((direction > 1e-12) & (h < 1-1e-12))[0]
    dec_candidates = np.where((direction < -1e-12) & (h > 1e-12))[0]
    if len(inc_candidates)==0 or len(dec_candidates)==0:
        return None # no move possible
    i = inc_candidates[np.argmax(direction[inc_candidates])] # largest positive direction
    j = dec_candidates[np.argmin(direction[dec_candidates])] # most negative direction
    # maximum epsilon we can transfer
    max_inc = 1 - h[i]
    max_dec = h[j]
    max_move = min(max_inc, max_dec, step_max)
    if max_move <= 0:
        return None
    eps = step_max * rng.random() # maybe random up to max_move?
    eps = min(eps, max_move)
    new_h = h.copy()
    new_h[i] += eps
    new_h[j] -= eps
    # ensure not stray with eps small.
    return new_h

def random_move(h, rng, max_delta=0.2):
    n = len(h)
    # Choose source i where h[i] > 0
    i = rng.choice(np.where(h > 1e-12)[0])
    # Choose target j where h[j] < 1
    j = rng.choice(np.where(h < 1-1e-12)[0])
    if i == j:
        return None
    max_transfer = min(h[i], 1 - h[j], max_delta)
    if max_transfer <= 0:
        return None
    eps = rng.random() * max_transfer
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

    # compute direction
    left = np.zeros(n)
    right = np.zeros(n)

    dec_mask = (direction < -eps) & (h > eps)

import time
import numpy as np

def _project_boxed_simplex(v, target, lower=0.0, upper=1.0, max_iter=60):
    """Project v onto {x: lower<=x<=upper, sum(x)=target}."""
    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, lower, upper)


def _score(h):
    """C5 value for a given h."""
    n = h.shape[0]
    dx = 2.0 / n
    cur = np.correlate(h, 1.0 - h, mode="full")
    return np.max(cur) * dx


def _gradient_move(h, lag, step, rng):
    """Move mass from a decreasing to an increasing direction for the worst lag."""
    n = h.shape[0]
    left = np.zeros(n)
    right = np.zeros(n)

    if lag > 0:
        left[lag:] = h[:-lag]                     # h[i‑lag]
        right[: n - lag] = 1.0 - h[lag:]          # 1‑h[i+lag]
    elif lag < 0:
        L = -lag
        left[: n - L] = h[L:]                     # h[i‑lag]  (i‑lag >=0)
        right[L:] = 1.0 - h[: n - L]              # 1‑h[i+lag] (i+lag < n)
    else:                                          # lag == 0
        left = h
        right = 1.0 - h

    direction = left - right                       # -grad of the worst term
    eps = 1e-12
    inc_mask = (direction > eps) & (h < 1.0 - eps)
    dec_mask = (direction < -eps) & (h > eps)
    inc_idx = np.where(inc_mask)[0]
    dec_idx = np.where(dec_mask)[0]
    if inc_idx.size == 0 or dec_idx.size == 0:
        return None

    i = inc_idx[np.argmax(direction[inc_idx])]
    j = dec_idx[np.argmin(direction[dec_idx])]

    max_inc = 1.0 - h[i]
    max_dec = h[j]
    max_move = min(max_inc, max_dec, step)
    if max_move <= 0.0:
        return None
    delta = rng.random() * max_move
    if delta < 1e-12:
        delta = max_move * 0.5
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h


def _random_move(h, rng, step):
    """Transfer a random amount of mass between two indices."""
    n = h.shape[0]
    pos = np.where(h > 1e-12)[0]
    neg = np.where(h < 1.0 - 1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return None
    i = rng.choice(pos)
    j = rng.choice(neg)
    if i == j:
        return None
    max_transfer = min(h[i], 1.0 - h[j], step)
    if max_transfer <= 0.0:
        return None
    delta = rng.random() * max_transfer
    if delta < 1e-12:
        delta = max_transfer * 0.5
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def _random_feasible(n, target, rng):
    v = rng.random(n)
    return _project_boxed_simplex(v, target)


def run(seed=42, budget_s=1100):
    rng = np.random.default_rng(seed)

    # --- initialise ---------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        # no supplied start -> build a random feasible vector
        n = 101
        target_sum = n / 2.0
        h0 = _random_feasible(n, target_sum, rng)
    n = h0.shape[0]
    target_sum = n / 2.0
    h = _project_boxed_simplex(h0, target_sum)

    best_h = h.copy()
    best_val = _score(best_h)

    start = time.perf_counter()
    step = 0.2                     # maximal transferred mass per move
    it = 0
    no_improve = 0
    reset_after = 50000           # iterations without improvement before a random restart
    while True:
        if time.perf_counter() - start > budget_s * 0.95:
            break

        # worst lag for the current configuration
        conv = np.correlate(h, 1.0 - h, mode="full")
        worst_idx = int(np.argmax(conv))
        lag = worst_idx - (n - 1)

        # try a directed gradient step
        cand = _gradient_move(h, lag, step, rng)
        improved = False
        if cand is not None:
            val = _score(cand)
            if val < best_val - 1e-12:
                best_val = val
                best_h = cand.copy()
                h = cand
                improved = True
                no_improve = 0
        if not improved:
            # fallback to a random swap move
            cand2 = _random_move(h, rng, step)
            if cand2 is not None:
                val2 = _score(cand2)
                if val2 < best_val - 1e-12:
                    best_val = val2
                    best_h = cand2.copy()
                    h = cand2
                    improved = True
                    no_improve = 0

        if not improved:
            no_improve += 1

        # schedule: slowly shrink step size

        if it % 5000 == 0:

        # occasional random restart if stuck

            no_improve = 0
            step = 0.2

    # final projection to be safe
    best_h = _project_boxed_simplex(best_h, target_sum)
    final_val = _score(best_h)

    return best_h.tolist(), float(final_val), int(n)
