import time
import numpy as np

def _c5_of_h(h):
    """Maximum overlap integral for a given discretised h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def _project_box_hyperplane(v, target):
    """
    Euclidean projection of v onto the set
        {x | 0 <= x_i <= 1 , sum x_i = target }.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -5.0, 5.0          # wide enough to guarantee feasibility
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _gradient_softmin(h, beta):
    """
    Gradient of the smooth approximation of -min_k <h,h_shifted_k>.
    Returns grad = - sum_d w_d * (h_{i+d}+h_{i-d}) with proper bounds.
    """
    n = h.shape[0]
    A = np.correlate(h, h, mode="full")          # autocorrelation
    m = A.min()
    w = np.exp(-beta * (A - m))
    w /= w.sum()                                 # weights emphasise small A_k

    grad = np.zeros_like(h)
    N = n
    off = N - 1                                   # shift of index 0

    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        d = idx - off
        if d >= 0:
            if d < N:
                grad[:N - d] -= weight * h[d:]           # h_{i+d}
                grad[d:]      -= weight * h[:N - d]      # h_{i-d}
        else:
            dpos = -d
            if dpos < N:
                grad[dpos:]   -= weight * h[:N - dpos]   # h_{i-d}
                grad[:N - dpos] -= weight * h[dpos:]    # h_{i+d}
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound and the number of points."""
    rng = np.random.default_rng(seed)
    start = time.time()

    best_c5 = _c5_of_h(best_h)
