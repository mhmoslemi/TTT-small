def smooth_max_and_grad(h, alpha):
    n = len(h)
    sum_h = np.sum(h)
    c = np.correlate(h, h, mode='full')
    f = sum_h - c  # shape (2n-1)
    max_f = np.max(f)
    # Stabilize
    shifted = alpha * (f - max_f)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)  # weights shape (2n-1)
    # Gradient constant term: weighted sum of 1 => 1
    # Compute conv1 = np.convolve(w, h, mode='full')[n-1:n-1+n]
    conv_full = np.convolve(w, h, mode='full')
    start = n-1
    conv1 = conv_full[start:start+n]  # sum_v w_v * h_{i - v}

    conv_full_rev = np.convolve(w[::-1], h, mode='full')
    conv2 = conv_full_rev[start:start+n]  # sum_v w_v * h_{i + v}
    grad = 1.0 - (conv1 + conv2)
    # smooth objective (softmax)
    sm_val = np.sum(w * f)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_onto_box_sum(x, target, lower=0.0, upper=1.0):
    # find lambda s.t sum clip(x+lambda) = target.
    lo = lower - np.max(x)  # ensures all become lower bound after adding lo
    hi = upper - np.min(x)  # ensures all become upper bound after adding hi
    # bisection for up to 60 iter.
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, lower, upper)
        s = np.sum(y)
        if s > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + lo, lower, upper)
    # adjust small residual
    diff = target - np.sum(y)
    if abs(diff) > 1e-12:
        # distribute to free variables
        free = (y > lower + 1e-12) & (y < upper - 1e-12)
        if np.any(free):

            y = np.clip(y, lower, upper)
        else:
            # adjust all within min diff
            y += diff / len(y)
            y = np.clip(y, lower, upper)
    return y

import time
import numpy as np

def _evaluate_c5(h):
    """Exact C5 for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def _project_box_sum(x, target, lower=0.0, upper=1.0, iters=60):
    """
    Euclidean projection of x onto the set
        {y | lower <= y_i <= upper, sum(y)=target}
    The solution has the form y_i = clip(x_i + λ, lower, upper) for a suitable λ.
    """
    lo = -np.max(x) - 1.0          # makes all entries <= lower after shift
    hi = 1.0 - np.min(x) + 1.0     # makes all entries >= upper after shift

    for _ in range(iters):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, lower, upper)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + lo, lower, upper)

    # Tiny residual correction (to keep the sum exactly target)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        free = (y > lower + 1e-12) & (y < upper - 1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, lower, upper)
        else:
            # all variables are at a bound – spread uniformly (unlikely)
            y += diff / len(y)
            y = np.clip(y, lower, upper)
    return y


def _smooth_max_and_grad(h, alpha):
    """
    Smooth approximation of max_k f_k where f_k = sum_i h_i (1 - h_{i+k})
    and its gradient.
    """
    n = h.shape[0]
    sum_h = h.sum()
    # autocorrelation of h with itself
    c = np.correlate(h, h, mode="full")
    f = sum_h - c                     # f_k = Σ h_i - Σ h_i h_{i+k}
    max_f = f.max()
    # softmax weights
    shifted = alpha * (f - max_f)
    w = np.exp(shifted)
    w /= w.sum()

    # conv1 = Σ_v w_v * h_{i-v}
    conv_full = np.convolve(w, h, mode="full")
    start = n - 1
    conv1 = conv_full[start:start + n]

    # conv2 = Σ_v w_v * h_{i+v}
    conv_full_rev = np.convolve(w[::-1], h, mode="full")
    conv2 = conv_full_rev[start:start + n]

    grad = 1.0 - (conv1 + conv2)      # ∂/∂h_i of weighted max f
    smooth_val = np.dot(w, f)        # softmax value (≈ max f)
    return smooth_val, grad

    """
