import time, numpy as np

# ---------- helper functions -----------------------------------------

def project_onto_box_simplex(v, total):
    """Euclidean projection of v onto {x | 0<=x_i<=1, sum x_i = total}."""
    lo = np.min(v) - 1.0      # guarantee all 1 after shift
    hi = np.max(v) + 1.0      # guarantee all 0 after shift
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > total:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def compute_max_overlap(h):
    """C5 value for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_gradient(h, lag):
    """Gradient of the worst‑lag correlation term."""
    n = h.shape[0]
    g = np.ones(n)
    if lag >= 0:
        if lag < n:
            g[:n - lag] -= h[lag:]
            g[lag:]    -= h[:n - lag]
    else:
        l = -lag
        if l < n:
            g[l:]    -= h[:n - l]
            g[:n - l] -= h[l:]
    return g


def upsample(src, n_target):
    """Linear interpolation of src (length m) to length n_target."""
    m = src.shape[0]
    xs = np.linspace(0.0, 2.0, m, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h = np.interp(xt, xs, src)
    total = n_target / 2.0
    return project_onto_box_simplex(h, total)


def random_feasible(n):
    """Random point in the feasible polytope."""
    y = np.random.rand(n)
    return project_onto_box_simplex(y, n / 2.0)


def improve_h(h0, total, start, budget):
    """Projected gradient descent on the max‑overlap objective."""
    h = h0.copy()
    n = h.shape[0]
    cur_val = compute_max_overlap(h)
    best_h, best_val = h.copy(), cur_val
    max_it = 200000
    it = 0
    while it < max_it and (time.time() - start) < budget:
        # identify worst lag
        corr = np.correlate(h, 1.0 - h, mode='full')
        widx = np.argmax(corr)
        lag = widx - (n - 1)

        # gradient of that lag
        grad = compute_gradient(h, lag)

        # simple decreasing step size
        step = 0.5 * (1.0 - it / max_it)

        nh = h - step * grad
        nh = project_onto_box_simplex(nh, total)

        nval = compute_max_overlap(nh)
        if nval < best_val:
            best_val, best_h = nval, nh.copy()
        if nval <= cur_val:
            h, cur_val = nh, nval
        else:
            # tiny chance to accept worsening move (helps escape plateaus)
            if np.random.rand() < 0.001:
                h, cur_val = nh, nval
        it += 1
    return best_h, best_val


# ---------- main entry point -----------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h, best_val = None, 1e9

    # lengths to try (prefer even lengths for exact sum=n/2)
    lengths = [94, 188]
    # include the provided construction if present
    try:
        init0 = initial_h_values  # provided by the harness
        lengths = [len(init0)] + lengths
    except Exception:
        init0 = None

    for n in lengths:
        total = n / 2.0
        # build a starting vector
        if init0 is not None and len(init0) != n:
            h_start = upsample(init0, n)
        elif init0 is not None:
            h_start = np.copy(init0)
            # enforce exact sum
            h_start = project_onto_box_simplex(h_start, total)
        else:
            h_start = random_feasible(n)

        # local optimisation from this start
        remaining = budget_s - (time.time() - start_time)
        h_opt, val_opt = improve_h(h_start, total, start_time, remaining * 0.6)
        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt

        # a few random restarts under the same n
        while (time.time() - start_time) < budget_s * 0.94:
            h_rand = random_feasible(n)
            remaining = budget_s - (time.time() - start_time)
            h_opt2, val_opt2 = improve_h(h_rand, total, start_time, remaining * 0.4)
            if val_opt2 < best_val:
                best_h, best_val = h_opt2, val_opt2
            if best_val <= 0.3808:
                break
        if best_val <= 0.3808:
            break

    # safety fallback if nothing succeeded
    if best_h is None:
        n = 47
        best_h = np.full(n, 0.5)
        best_h = project_onto_box_simplex(best_h, n / 2.0)
        best_val = compute_max_overlap(best_h)

    # final projection to guarantee feasibility and recompute exact bound
    best_h = project_onto_box_simplex(best_h, len(best_h) / 2.0)
    best_val = compute_max_overlap(best_h)

    return best_h, float(best_val), int(len(best_h))
