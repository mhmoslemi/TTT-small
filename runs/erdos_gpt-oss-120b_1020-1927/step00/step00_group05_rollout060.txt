def solve_lp(n):
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2]
    # for each shift
    for s in range(1, n):
        size = n - s
        p = cp.Variable(size)
        hi = h[s:]
        hj = h[:size]
        constraints += [p >= 0,
                        p <= hi,
                        p <= hj,
                        p >= hi + hj - 1,
                        cp.sum(p) >= t]
    # objective minimize -t (maximize t)
    prob = cp.Problem(cp.Maximize(t), constraints)
    prob.solve(solver='GLPK', glpk={'msg_lev': 'GLP_MSG_OFF'})
    return h.value

import numpy as np, time, random, math

def proj_box_sum(v, target):
    lo = np.min(v - 1)
    hi = np.max(v) + 1.0   # ensure hi large enough.
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    max_corr = np.max(np.correlate(h, 1 - h, mode='full'))
    return max_corr * (2.0 / n)

def compute_gradient(h, shift):
    n = len(h)
    if shift == 0:
        return 1 - 2 * h
    s = abs(shift)
    grad = np.ones(n)
    grad[s:] -= h[:n - s]
    grad[:n - s] -= h[s:]
    return grad

def gradient_descent(h0, max_iter=2000, step0=0.3):
    h = h0.copy()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(h)
    for it in range(1, max_iter + 1):
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        grad = compute_gradient(h, shift)
        alpha = step0 / math.sqrt(it)
        h = h - alpha * grad
        # project
        h = proj_box_sum(h, target)
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val

def simulated_annealing(h0, max_iter=20000, init_temp=0.01, final_temp=1e-5, max_step=0.1):
    h = h0.copy()
    n = len(h)
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val
    for it in range(1, max_iter + 1):
        # linear cooling temperature
        t = init_temp + (final_temp - init_temp) * (it / max_iter)
        # pick two indices
        i, j = random.sample(range(n), 2)
        # compute allowable delta range
        # delta added to i, sub from j
        max_pos = min(max_step, 1.0 - h[i], h[j])
        max_neg = min(max_step, h[i], 1.0 - h[j])
        if max_pos == 0 and max_neg == 0:
            continue
        delta = random.uniform(-max_neg, max_pos)
        if delta == 0:
            continue
        h_new = h.copy()
        h_new[i] = h[i] + delta
        h_new[j] = h[j] - delta
        # compute new bound
        new_val = compute_c5(h_new)
        dE = new_val - cur_val
        if dE < 0 or random.random() < math.exp(-dE / (t + 1e-12)):
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
    return best_h, best_val

import time, random, math
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the hyperplane sum=h_target with 0/1 box
def proj_box_sum(v, target, iters=60):
    # Find λ such that sum_i clip(v_i - λ,0,1) = target
    lo = np.min(v - 1.0)           # low enough => all entries become 1 after clipping
    hi = np.max(v) + 1.0           # high enough => all entries become 0
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    λ = (lo + hi) * 0.5
    return np.clip(v - λ, 0.0, 1.0)

# ------------------------------------------------------------
# Compute C5 bound for a discrete step function h
def compute_c5(h):
    n = h.shape[0]
    max_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
    return max_corr * (2.0 / n)

# ------------------------------------------------------------
# Gradient of the active overlap term (the one attaining the maximum)
def compute_gradient(h, shift):
    n = h.shape[0]
    if shift == 0:
        # O_0 = Σ h_i (1-h_i)  →  grad = 1 - 2 h_i
        return 1.0 - 2.0 * h
    s = abs(shift)
    grad = np.ones(n)
    # subtract h[i - shift] where defined
    grad[s:] -= h[:n - s]
    # subtract h[i + shift] where defined
    grad[:n - s] -= h[s:]
    return grad

# ------------------------------------------------------------
# Simple projected sub‑gradient descent
def subgradient_opt(h0, max_iter=2500, step0=0.5):
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h, best_val = h.copy(), compute_c5(h)

    for it in range(1, max_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)                # integer shift corresponding to max
        grad = compute_gradient(h, shift)
        alpha = step0 / math.sqrt(it)
        h -= alpha * grad
        h = proj_box_sum(h, target)
        val = compute_c5(h)
        if val < best_val:
            best_val, best_h = val, h.copy()
    return best_h, best_val

# ------------------------------------------------------------
# Stochastic local search (pairwise swaps) with cooling
def stochastic_opt(h0, time_budget, start_temp=0.02, final_temp=1e-6,
                   start_eps=0.12, final_eps=5e-4, max_iter=200000):
    h = h0.copy()
    n = h.shape[0]
    cur_val = compute_c5(h)
    best_h, best_val = h.copy(), cur_val
    start = time.time()

    for it in range(1, max_iter + 1):
        # time guard
        if time.time() - start > time_budget - 0.2:
            break
        # linear cooling schedules
        frac = it / max_iter
        temp = start_temp + (final_temp - start_temp) * frac
        eps = start_eps + (final_eps - start_eps) * frac

        i, j = random.sample(range(n), 2)
        # feasible delta range while keeping box constraints
        max_pos = min(eps, 1.0 - h[i], h[j])
        max_neg = min(eps, h[i], 1.0 - h[j])
        if max_pos == 0.0 and max_neg == 0.0:
            continue
        delta = random.uniform(-max_neg, max_pos)
        if abs(delta) < 1e-15:
            continue
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_val = compute_c5(h_new)
        dE = new_val - cur_val
        if dE < 0.0 or random.random() < math.exp(-dE / (temp + 1e-12)):
            h, cur_val = h_new, new_val
            if new_val < best_val:
                best_val, best_h = new_val, h_new.copy()
    return best_h, best_val
