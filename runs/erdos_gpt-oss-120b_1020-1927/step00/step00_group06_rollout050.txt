def compute_grad(h, shift_index):
    # shift_index is index in correlation array (len=2*n-1, shift = index - (n-1))
    shift = shift_index - (len(h) - 1)
    n = len(h)
    if shift >= 0:
        # sum over i = 0 .. n-1-shift
        # term h[i] * (1 - h[i+shift])
        # gradient w.r.t. h[i]: (1 - h[i+shift]) for i in 0..n-1-shift
        # gradient w.r.t. h[i+shift]: -h[i] for i in 0..n-1-shift
        grad = np.zeros_like(h)
        # contributions to h[i] for i within range
        grad[:n-shift] += (1 - h[shift:])  # (1 - h[i+shift])
        # contributions to h[i+shift] from -h[i]
        grad[shift:] -= h[:n-shift]
        # No contributions from other indices (zero padded)
    else:
        shift_abs = -shift
        # sum over i = shift_abs .. n-1:
        # term h[i] * (1 - h[i-shift_abs]) where i-shift_abs valid.
        # Equivalent to i' = i, j = i - shift_abs; SHIFT negative means h_i * (1 - h_{i - shift})
        # So original correlation for negative shift is sum_{i=shift_abs}^{n-1} h[i] * (1 - h[i - shift_abs]).
        grad = np.zeros_like(h)
        # gradient w.r.t. h[i] (i shift) for i>=shift_abs: (1 - h[i - shift_abs])
        grad[shift_abs:] += (1 - h[:n-shift_abs])
        # gradient w.r.t. h[i - shift_abs] -h[i] for i>=shift_abs:
        grad[:n-shift_abs] -= h[shift_abs:]
    return grad

def compute_gradient_hmax(h, shift_idx):
    n = len(h)
    shift = shift_idx - (n - 1)
    grad = np.zeros_like(h)
    if shift >= 0:
        # i in [0, n-1-shift]
        i_end = n - shift
        grad[:i_end] += (1 - h[shift:])
        grad[shift:] -= h[:i_end]
    else:
        shift_abs = -shift
        i_start = shift_abs
        i_end = n
        # i from shift_abs to n-1
        # contributions to h[i] and h[i - shift_abs]
        grad[i_start:] += (1 - h[:n-shift_abs])
        grad[:n-shift_abs] -= h[i_start:]
    return grad

def local_refine(h, dx, target, steps=500, eta=0.01):
    best_h = h.copy()
    best_val = evaluate_c5(best_h, dx)
    for t in range(steps):
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k = np.argmax(corr)
        grad = compute_gradient_hmax(best_h, k)
        # step size adapt?
        step = eta / (1 + t*0.01)  # decreasing
        cand = best_h - step * grad
        cand = project_onto_capped_simplex(cand, target)
        cand_val = evaluate_c5(cand, dx)
        if cand_val < best_val:
            best_val = cand_val
            best_h = cand
    return best_h, best_val

import time
import random
import numpy as np

# ---------- projection onto the capped simplex ----------
def _proj_capped_simplex(v, target):
    """Euclidean projection of v onto {x in [0,1]^n, sum x = target}."""
    lo, hi = -1.0, 2.0               # guarantees sum(lo) = n, sum(hi) = 0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)

    # tiny correction to hit the sum exactly (numerical safety)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


# ---------- evaluation ----------
def _c5_value(h, dx):
    """Return the overlap integral C5 for a discretised function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


# ---------- deterministic starter patterns ----------
def _thue_morse(n):
    """Generate the first n bits of the Thue–Morse sequence (0/1)."""
    indices = np.arange(n, dtype=np.uint64)
    # popcount parity: count of ones modulo 2
    bits = np.unpackbits(indices[:, None].view(np.uint8), axis=1).sum(axis=1) & 1
    return bits.astype(float)


def _alternating(n):
    """Simple alternating pattern 0101..., balanced when n even."""
    pat = np.arange(n) % 2
    return pat.astype(float)


def _half_left(n):
    """First half 1's, second half 0's."""
    pat = np.zeros(n, dtype=float)
    pat[: n // 2] = 1.0
    return pat


# ---------- mutation / crossover ----------
def _mutate(h, sigma, target):
    cand = h + np.random.normal(scale=sigma, size=h.shape)
    return _proj_capped_simplex(cand, target)


def _crossover(a, b, target):
    cand = (a + b) * 0.5
    return _proj_capped_simplex(cand, target)


# ---------- gradient of the active (max) shift ----------
def _grad_of_max_shift(h, shift_idx):
    """Sub‑gradient of the max‑shift term for vector h."""
    n = h.size
    shift = shift_idx - (n - 1)          # actual shift value (negative allowed)
    grad = np.zeros_like(h)

    if shift >= 0:
        end = n - shift
        grad[:end] += (1.0 - h[shift:])      # ∂/∂h[i] for i in [0,end)
        grad[shift:] -= h[:end]              # ∂/∂h[i+shift]
    else:
        s = -shift
        start = s
        grad[start:] += (1.0 - h[: n - s])   # ∂/∂h[i] for i >= s
        grad[: n - s] -= h[start:]           # ∂/∂h[i] for i < n-s
    return grad


# ---------- local refinement via sub‑gradient steps ----------
def _local_refine(h, dx, target, steps=300, eta_start=0.05, eta_end=0.005):
    best_h = h.copy()
    best_val = _c5_value(best_h, dx)

    for it in range(steps):
        cor = np.correlate(best_h, 1.0 - best_h, mode='full')
        k = int(np.argmax(cor))
        grad = _grad_of_max_shift(best_h, k)

        # decreasing step size
        eta = eta_start + (eta_end - eta_start) * (it / steps)
        cand = best_h - eta * grad
        cand = _proj_capped_simplex(cand, target)

        val = _c5_value(cand, dx)
        if val < best_val:
            best_val = val
            best_h = cand
    return best_h, best_val


# ---------- main entry point ----------
def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # decide number of points (use 200 for enough flexibility)
    n = 200
    target_sum = n / 2.0
    dx = 2.0 / n

    # ---------- build initial population ----------
    pop_size = 60
    elite_frac = 0.2
    elite_sz = max(2, int(pop_size * elite_frac))

    population = []

    # 1) incorporate the provided initial vector (interpolated)
    if 'initial_h_values' in globals():
        init_old = np.asarray(initial_h_values, dtype=float)
        m = init_old.size
        x_old = np.linspace(0.0, 2.0, m, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h_interp = np.interp(x_new, x_old, init_old)
        h_interp = _proj_capped_simplex(h_interp, target_sum)
        population.append({'h': h_interp,
                           'c5': _c5_value(h_interp, dx)})

    # 2) deterministic seeds
    for maker in (_thue_morse, _alternating, _half_left):
        h = maker(n)
        h = _proj_capped_simplex(h, target_sum)
        population.append({'h': h,
                           'c5': _c5_value(h, dx)})

    # 3) fill the rest with random feasible vectors
    while len(population) < pop_size:
        h = np.random.rand(n)
        h = _proj_capped_simplex(h, target_sum)
        population.append({'h': h,
                           'c5': _c5_value(h, dx)})

    # track global best
    best_entry = min(population, key=lambda x: x['c5'])
    best_h = best_entry['h'].copy()
    best_c5 = best_entry['c5']

    # annealing schedule for mutation strength
    max_sigma = 0.6
    min_sigma = 0.001
    iteration = 0

    # ---------- evolutionary loop ----------
    while time.time() - start_time < budget_s * 0.95:
        # keep elite
        population.sort(key=lambda x: x['c5'])
        elite = population[:elite_sz]

        # progress‑dependent mutation size
        elapsed = time.time() - start_time
        prog = elapsed / (budget_s * 0.95)
        sigma = max_sigma * (1.0 - prog) + min_sigma * prog

        children = []

        # mutations of elites
        for e in elite:
            child_h = _mutate(e['h'], sigma, target_sum)
            children.append({'h': child_h,
                             'c5': _c5_value(child_h, dx)})

        # cross‑overs among elites (pairwise, limited)
        for i in range(elite_sz):
            for j in range(i + 1, elite_sz):
                if len(children) + elite_sz >= pop_size:
                    break
                child_h = _crossover(elite[i]['h'], elite[j]['h'], target_sum)
                child_h = _mutate(child_h, sigma * 0.5, target_sum)
                children.append({'h': child_h,
                                 'c5': _c5_value(child_h, dx)})
            if len(children) + elite_sz >= pop_size:
                break

        # random newcomers if still space
        while len(children) + elite_sz < pop_size:
            h = np.random.rand(n)
            h = _proj_capped_simplex(h, target_sum)
            children.append({'h': h,
                             'c5': _c5_value(h, dx)})

        # new generation
        population = elite + children
        population.sort(key=lambda x: x['c5'])
        population = population[:pop_size]

        # update global best
        if population[0]['c5'] < best_c5:
            best_c5 = population[0]['c5']
            best_h = population[0]['h'].copy()
            # early stop if record already beaten
            if best_c5 <= 0.3808:
                break

        # occasional local refinement
        if iteration % 20 == 0 and prog > 0.4:
            refined_h, refined_val = _local_refine(best_h, dx, target_sum,
                                                   steps=150,
                                                   eta_start=0.03,
                                                   eta_end=0.003)
            if refined_val < best_c5:
                best_c5 = refined_val
                best_h = refined_h.copy()
                population[-1] = {'h': best_h, 'c5': best_c5}
        iteration += 1

    # ---------- final polishing ----------
    best_h, best_c5 = _local_refine(best_h, dx, target_sum,
                                    steps=400,
                                    eta_start=0.02,
                                    eta_end=0.001)

    # try a binary rounding (exactly half ones) – keep if better
    if n % 2 == 0:
        binary = np.zeros_like(best_h)
        idx = np.argpartition(best_h, -int(target_sum))[-int(target_sum):]
        binary[idx] = 1.0
        binary_c5 = _c5_value(binary, dx)
        if binary_c5 < best_c5:
            best_h = binary
            best_c5 = binary_c5

    return best_h, float(best_c5), n
