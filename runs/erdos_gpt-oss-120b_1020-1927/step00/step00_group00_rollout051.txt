import time, math
import numpy as np

# ---------- Helper ----------

def _project_to_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto [0,1]^n with sum(v)=target using bisection on Lagrange multiplier."""
    lo, hi = -np.max(v) - 1.0, np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _score(h: np.ndarray) -> float:
    """C5 value for discretised h."""
    n = h.size
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode="full")
    return float(cor.max() * dx)


def _shift_zero(arr: np.ndarray, k: int) -> np.ndarray:
    """Shift arr by k with zero padding (no wrap). Positive k shifts right."""
    n = arr.size
    out = np.zeros_like(arr)
    if k > 0:
        out[k:] = arr[: n - k]
    elif k < 0:
        kk = -k
        out[: n - kk] = arr[kk:]
    else:
        out[:] = arr
    return out


def _gd_step(x: np.ndarray, step: float) -> np.ndarray:
    """One sub‑gradient step for the worst shift."""
    n = x.size
    # worst shift (max overlap)
    corr = np.correlate(x, 1.0 - x, mode="full")
    w_idx = int(np.argmax(corr))
    shift = w_idx - (n - 1)

    # direction = h_{i+shift} + h_{i-shift}
    dir_vec = _shift_zero(x, shift) + _shift_zero(x, -shift)
    new = x + step * dir_vec
    return _project_to_box_sum(new, n / 2.0)


# ---------- Main routine ----------

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s

    # ----- start from parent if available -----
    try:

    except NameError:

    if init is None:

    cur = _project_to_box_sum(cur, n0 / 2.0)

    best_h = cur.copy()
    best_score = _score(best_h)

    # ----- gradient‑descent phase on original grid -----
    max_gd = 3000
    for it in range(max_gd):
        if time.time() > deadline - 2.0:
            break
        # decreasing step
        frac = it / max_gd
        step = 0.10 * (1.0 - frac) + 0.001 * frac
        cur = _gd_step(cur, step)

        if sc < best_score:
            best_score = sc
            best_h = cur.copy()
        # occasional small random jitter

            sc_tmp = _score(tmp)
