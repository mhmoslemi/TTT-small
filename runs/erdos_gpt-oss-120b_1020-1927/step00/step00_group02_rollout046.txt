def _proj_box_sum(v, target):
    """Project v onto {0<=x<=1, sum x = target} by bisection on a dual shift."""
    lo = -np.max(v) - 2.0
    hi = 1.0 - np.min(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


# ------------------------------------------------------------
# Overlap and gradient for a given integer shift
# ------------------------------------------------------------
def _overlap_full(h):
    """Full correlation array: np.correlate(h,1-h,'full') (no scaling)."""
    return np.correlate(h, 1.0 - h, mode='full')


def _gradient_shift(h, shift):
    """Sub‑gradient of the overlap term that attains the current maximum."""
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        # first part: 1 - h[i+shift] for i = 0..n-1-shift
        g[: n - shift] += 1.0 - h[shift:]
        # second part: -h[i] for i+shift = shift..n-1
        g[shift:] -= h[: n - shift]
    else:
        t = -shift
        # first part: 1 - h[i-t] for i = t..n-1
        g[t:] += 1.0 - h[: n - t]
        # second part: -h[i] for i = 0..n-1-t (paired with i+t)
        g[: n - t] -= h[t:]
    return g


def _c5_value(h, dx):
    """Current bound C5 for a vector h."""
    return np.max(_overlap_full(h)) * dx

# Local projected sub‑gradient descent
# ------------------------------------------------------------
def _local_optimize(start_h, dx, target_sum, time_deadline,
                    max_iters=20000, step0=0.5):
    h = start_h.copy()
    best_h = h.copy()
    best_val = _c5_value(h, dx)

    n = h.size
    it = 0
    while it < max_iters and time.time() < time_deadline:
        # current worst shift
        full = _overlap_full(h)
        idx = int(np.argmax(full))
        shift = idx - (n - 1)

        # gradient of that shift
        grad = _gradient_shift(h, shift)

        # simple diminishing step size
        step = step0 / np.sqrt(it + 1.0)
        h = _proj_box_sum(h - step * grad, target_sum)

        # keep the best seen
        cur = _c5_value(h, dx)
        if cur < best_val:
            best_val = cur
            best_h = h.copy()

        # occasional jitter to escape plateaus
        if (it + 1) % 500 == 0:
            jitter = (np.random.rand(n) - 0.5) * 0.02
            h = _proj_box_sum(best_h + jitter, target_sum)

        it += 1
    return best_h, best_val

# Main entry point required by the harness
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # choose a working resolution (you can change it if time permits)
    n = 180                      # points on [0,2]
    dx = 2.0 / n
    target_sum = n / 2.0

    # --------------------------------------------------------
    # Build a list of starting vectors
    # --------------------------------------------------------
    starts = []

    # 1) interpolated version of the provided construction (if any)
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if base.size != n:
            x_old = np.linspace(0.0, 2.0, base.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            base = np.interp(x_new, x_old, base)
        starts.append(base)

    # 2) a few random starts
    for _ in range(3):
        rand = np.random.rand(n)
        starts.append(rand)

    # Project all starts onto the feasible set
    starts = [_proj_box_sum(v, target_sum) for v in starts]

    # --------------------------------------------------------
    # Run local optimisation from each start (time‑bound)
    # --------------------------------------------------------
    best_h = None
    best_val = np.inf
    # give each start an equal share of the remaining budget
    per_start = max(5, int(time_left() // max(1, len(starts))))

    for h0 in starts:
        if time_left() < 2.0:
            break
        deadline = time.time() + min(per_start, time_left())
        h_opt, val_opt = _local_optimize(
            h0, dx, target_sum, deadline,
            max_iters=30000, step0=0.7
        )
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()

    # Final safety projection
    best_h = _proj_box_sum(best_h, target_sum)

    # Compute the final bound using the harness helper (if present)
    try:
        c5_bound = evaluate_erdos_solution(best_h)
    except Exception:                     # fallback to our own computation
        c5_bound = _c5_value(best_h, dx)

    # Return as requested
    return best_h.tolist(), float(c5_bound), n
