def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # C5 is max of corr (they can be negative? but we care max)
    return np.max(corr)

import time, math
import numpy as np

def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {x in [0,1]^n : sum(x)=target}
    using a simple binary search for the Lagrange multiplier."""
    lo = np.min(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            add = diff / free.sum()
            x[free] += add
            x = np.clip(x, 0.0, 1.0)
    return x


def _c5_score(h: np.ndarray) -> float:
    """Return the C5 value for a given discretised h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.RandomState(seed)

    # ---- decide number of points ------------------------------------
    try:
        # use the provided construction if present (its length is reasonable)
        init = np.asarray(initial_h_values, dtype=float)
        n = init.shape[0]
    except NameError:
        n = 95                     # fallback size, small enough to stay fast

    target_sum = n / 2.0
    dx = 2.0 / n

    # ---- start from an analytical shape (sin^2) --------------------
    x = np.arange(n) * dx + dx * 0.5               # centre of each sub‑interval
    h = np.sin(np.pi * x / 2.0) ** 2                # integral of sin^2 on [0,2] equals 1
    h = _project_to_sum(h, target_sum)              # enforce the exact integral constraint
    h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_score = _c5_score(best_h)

    cur_h = best_h.copy()
    cur_score = best_score

    start = time.time()
    max_iters = 500000
    T_start = 0.02
    T_end = 1e-5

    for it in range(max_iters):
        # time budget check
        if time.time() - start > budget_s * 0.95:
            break

        # cooling schedule (linear)
        t_frac = it / max_iters
        T = T_start * (1.0 - t_frac) + T_end * t_frac

        # ---- mutation ------------------------------------------------
        if rng.rand() < 0.78:                 # pairwise mass exchange (keeps the sum exact)
            i, j = rng.choice(n, size=2, replace=False)
            lo = max(-cur_h[i], cur_h[j] - 1.0)
            hi = min(1.0 - cur_h[i], cur_h[j])
            if lo >= hi:
                continue
            delta = rng.uniform(lo, hi)
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:                                 # global Gaussian perturbation + projection
            sigma = 0.07 * (1.0 - t_frac)      # shrink as we cool
            new_h = cur_h + rng.randn(n) * sigma
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = _project_to_sum(new_h, target_sum)

        new_score = _c5_score(new_h)

        # ---- Metropolis acceptance ------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / (T + 1e-12))
            if rng.rand() < prob:
                accept = True

        if accept:
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score

        # occasional diversification
        if (it + 1) % 60000 == 0:
            # restart near the best with a burst of noise
            cur_h = best_h + rng.randn(n) * 0.015
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = _project_to_sum(cur_h, target_sum)
            cur_score = _c5_score(cur_h)

    # final sanity check (project once more just in case)
    best_h = _project_to_sum(best_h, target_sum)
    best_score = _c5_score(best_h)

    return best_h.tolist(), float(best_score), n
