def compute_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    # Not multiply by dx because gradient will use same factors.
    max_val = corr.max()
    max_shifts = np.where(np.abs(corr - max_val) < 1e-12)[0]  # indices in full array
    # shift offset = idx - (n-1)
    grads = np.zeros(n)
    for idx in max_shifts:
        m = idx - (n - 1)
        # forward: for i where i+m in [0, n-1]
        # i ranges max(0, -m) to min(n-1, n-1-m)
        # we can vectorize:
        if m >= 0:
            i_range = np.arange(0, n - m)
            forward_idx = i_range + m
            # term1 for i_range: (1 - h[forward_idx])
            grads[i_range] += (1 - h[forward_idx])
            # term2 for i_range+m? Actually backward term: -h[i_range]? Not exactly.
            # Let's compute for all i: 
            # term2 for i = i_range+m: -h[i_range] because i_backward = i - m
            # So we set grads[i_range + m] += -h[i_range]
            grads[i_range + m] += -h[i_range]
        else:  # m < 0
            m_neg = -m
            i_range = np.arange(m_neg, n)  # such that i-m >=0 => i >= m_neg

    grads *= dx
    grads /= len(max_shifts)  # average over multiple max shifts

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return (corr.max()) * dx

def compute_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()
    # tolerance relative
    eps = 1e-12
    idxs = np.where(np.abs(corr - max_val) <= eps)[0]
    grads = np.zeros(n)
    for idx in idxs:
        m = idx - (n - 1)
        if m >= 0:
            i = np.arange(0, n - m)
            forward = i + m
            grads[i] += (1 - h[forward])
            grads[forward] += -h[i]
        else:
            p = -m
            i = np.arange(p, n)
            forward = i - p
            grads[i] += (1 - h[forward])
            grads[forward] += -h[i]
    grads *= dx
    grads /= len(idxs)
    return grads

def project_onto_constraints(y, total):
    # binary search tau to satisfy sum constraints after clipping y-tau
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        tau = (lo + hi) / 2.0
        x = np.clip(y - tau, 0.0, 1.0)
        s = x.sum()
        if s > total:
            lo = tau
        else:
            hi = tau
    # final projection
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def pgd(h0, max_iters, time_remaining, best_global_c5):

def upsample(h, factor):
    n = len(h)
    new_n = n * factor
    dx = 2.0 / n
    old_x = np.arange(n) * dx + dx/2.0
    new_dx = 2.0 / new_n

import time, math
import numpy as np

def _evaluate(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def _gradient(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    # indices attaining the maximum (allow tiny tolerance)
    mask = np.isclose(corr, mx, atol=1e-12, rtol=1e-9)
    idxs = np.nonzero(mask)[0]
    g = np.zeros(n)
    for idx in idxs:
        m = idx - (n - 1)          # shift
        if m >= 0:
            i = np.arange(0, n - m)
            j = i + m
            g[i] += (1.0 - h[j])
            g[j] += -h[i]
        else:  # m < 0
            p = -m
            i = np.arange(p, n)
            j = i - p
            g[i] += (1.0 - h[j])
            g[j] += -h[i]
    g *= dx / max(1, len(idxs))
    return g

def _project(y, total):
    # project y onto {0<=x_i<=1, sum x_i = total}
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        tau = (lo + hi) * 0.5
        x = np.clip(y - tau, 0.0, 1.0)
        s = x.sum()
        if s > total:
            lo = tau
        else:
            hi = tau
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def _pgd(h0, max_iters, time_budget, best_global):
    h = h0.copy()
    n = h.size
    total = n * 0.5
    best_h = h.copy()
    best_val = _evaluate(h)
    start = time.time()
    for it in range(1, max_iters + 1):
        if time.time() - start > time_budget:
            break
        grad = _gradient(h)
        step = 0.6 / math.sqrt(it)
        # try a few step reductions if no improvement
        for _ in range(5):
            cand = _project(h - step * grad, total)
            val = _evaluate(cand)
            if val < best_val - 1e-12:
                h = cand
                best_h = cand
                best_val = val
                break
            step *= 0.5
        # occasional jitter to escape plateaus
        if it % 400 == 0:
            jitter = np.random.uniform(-0.04, 0.04, n)
            cand = _project(np.clip(h + jitter, 0.0, 1.0), total)
            val = _evaluate(cand)
            if val < best_val:
                h = cand
                best_h = cand
                best_val = val
        if best_val < 0.38070:        # early exit if already good
            break
    return best_h, best_val

def _upsample(h, factor):
    n = h.size
    new_n = n * factor
    dx = 2.0 / n
    old_x = np.arange(n) * dx + dx * 0.5
    new_dx = 2.0 / new_n
    new_x = np.arange(new_n) * new_dx + new_dx * 0.5
    nh = np.interp(new_x, old_x, h)
    return _project(nh, new_n * 0.5)

def _to_binary(h):
    n = h.size
    half = n // 2
    idx = np.argsort(-h)[:half]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def _binary_hill_climb(b0, max_swaps, time_budget):
    n = b0.size
    ones = np.flatnonzero(b0 > 0.5)
    zeros = np.flatnonzero(b0 < 0.5)
    best = b0.copy()
    best_val = _evaluate(best)
    start = time.time()
    for _ in range(max_swaps):
        if time.time() - start > time_budget:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = best.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        val = _evaluate(cand)
        if val < best_val:
            best = cand
            best_val = val
            # update lists
            ones = ones[ones != i]
            zeros = zeros[zeros != j]
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_all = time.time()

    # ---- initial vector ----------------------------------------------------
    if 'initial_h_values' in globals():

    cur_n = cur_h.size

            remaining = budget_s - elapsed

        # c) random binary restarts (optional, cheap)
        elapsed = time.time() - start_all
        remaining = budget_s - elapsed
        if remaining > 8.0 and cur_h.size % 2 == 0:

    final_c5 = _evaluate(best_h)
    return best_h.tolist(), float(final_c5), best_h.size
