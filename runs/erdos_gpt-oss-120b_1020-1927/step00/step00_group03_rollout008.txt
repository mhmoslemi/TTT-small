import numpy as np
import time

def _project_onto_box_sum(v, target, lo=0.0, hi=1.0, tol=1e-9, max_iter=100):
    """
    Project vector v onto the set {x: lo <= x_i <= hi, sum(x)=target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    low = -1e6
    high = 1e6
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, lo, hi)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) * 0.5, lo, hi)


def _c5_of(h):
    """Exact C5 evaluation used by the harness."""
    dx = 2.0 / h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Choose discretisation size.

    N = 200                      # number of samples (feel free to adjust)

    except Exception:          # fallback if no initial_h_values
        t = np.linspace(0.0, 2.0, N, endpoint=False)
        h = 0.5 + 0.5 * np.cos(np.pi * t)   # a smooth feasible start

    h = _project_onto_box_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = _c5_of(best_h)

    tau = 5.0               # start temperature for the smooth max
