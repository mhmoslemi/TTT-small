def project_to_bounds_and_sum(v, target_sum):
    # Want to find x = clip(v + λ, 0, 1) s.t. sum(x) = target_sum.
    # Since sum is monotonic in λ, binary search λ.
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)

def grad_corr(h, k):
    n = h.shape[0]
    grad = np.zeros(n)
    if k >= 0:
        i_start = 0
        i_end = n - 1 - k   # inclusive
        # term1: +1 for i in that range
        grad[i_start:i_end+1] += 1
        # term2: subtract h[i + k] for i in range
        grad[i_start:i_end+1] -= h[i_start + k : i_end + k + 1]
        # term3: subtract h[i - k] for i >= k
        if k > 0:
            grad[k:] -= h[: n - k]
    else:
        s = -k
        i_start = s
        i_end = n - 1
        # term1: +1 for i in [s, n-1]
        grad[i_start:] += 1
        # term2: subtract h[i - s] for i in that range: corresponds to h[0:n-s]
        grad[i_start:] -= h[: n - s]
        # term3: subtract h[i + s] for i in [0, n-1-s]
        grad[: n - s] -= h[s:]
    return grad

def upsample_to_n(arr, n):
    m = len(arr)
    if m == n:
        return arr.copy()
    x_old = np.linspace(0, 2, m, endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    return np.interp(x_new, x_old, arr)

import time, numpy as np

def _project_to_sum_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1]^n with sum(v)=target using a simple bisection."""
    lo = -np.max(v) - 1.0          # ensures v+lo <= 0  -> all clipped to 0
    hi = 1.0 - np.min(v) + 1.0     # ensures v+hi >= 1  -> all clipped to 1
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        if x.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _grad_corr(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of sum_i h_i (1 - h_{i+k}) w.r.t. h (full range handling)."""
    n = h.size
    g = np.zeros_like(h)
    if k >= 0:
        i_end = n - 1 - k
        if i_end >= 0:
            g[: i_end + 1] += 1                 # term +1
            g[: i_end + 1] -= h[k : k + i_end + 1]   # -h_{i+k}
        if k > 0:
            g[k:] -= h[: n - k]                 # -h_{i-k}
    else:
        s = -k
        i_start = s
        g[i_start:] += 1                         # term +1 for i>=s
        g[i_start:] -= h[: n - s]                # -h_{i-s}
        g[: n - s] -= h[s:]                      # -h_{i+s}
    return g


def _c5_value(h: np.ndarray) -> float:
    """Compute C5 for a step function h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def _initial_vector(target_n: int) -> np.ndarray:
    """Construct a start vector of length target_n."""
    # use a small random perturbation around 0.5, then project to feasibility
    rnd = np.random.rand(target_n) * 0.2 - 0.1   # [-0.1,0.1]
    h = np.full(target_n, 0.5) + rnd
    h = np.clip(h, 0.0, 1.0)
    return _project_to_sum_bounds(h, target_n / 2.0)


def _maybe_upsample(init: np.ndarray, n: int) -> np.ndarray:
    """Upsample a given array to length n by linear interpolation."""
    if init.size == n:
        return init.copy()
    x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, init)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C5 bound."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2

    # ----- choose discretisation size -----
    n = 200                         # even, gives dx=0.01
    target_sum = n / 2.0

    # ----- initialise h -----
    if "initial_h_values" in globals():
        h0 = _maybe_upsample(np.asarray(initial_h_values, dtype=float), n)
    else:
        h0 = _initial_vector(n)

    h = _project_to_sum_bounds(h0, target_sum)
    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # ----- optimisation parameters -----
    lr0 = 0.06                     # initial learning rate
    decay = 1e-4
    top_k = 5                     # number of top shifts to average gradients
    no_imp = 0
    it = 0

    while time.time() < deadline:
        # correlation and the k‑values that are largest
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        max_c = corr[max_idx]
        # indices of a few largest correlations (including the max)
        if top_k >= len(corr):
            idxs = np.arange(len(corr))
        else:
            idxs = np.argpartition(corr, -top_k)[-top_k:]
        # average gradient of the selected shifts
        grad = np.zeros_like(h)
        for idx in idxs:
            k = idx - (n - 1)
            grad += _grad_corr(h, k)
        grad /= float(len(idxs))

        # step with diminishing learning rate
        lr = lr0 / (1.0 + it * decay)
        h_new = h - lr * grad
        h_new = _project_to_sum_bounds(h_new, target_sum)

        cur_c5 = _c5_value(h_new)

        # keep the best ever seen
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        # accept the move if it does not worsen too much, else keep old
        if cur_c5 <= max_c or np.random.rand() < 0.02:
            h = h_new
        else:
            # occasional random shake to escape plateaus
            if no_imp > 300:
                h = _initial_vector(n)
                no_imp = 0
        it += 1

    return best_h, best_c5, n
