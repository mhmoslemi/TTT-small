def propose_move(h, max_move):
    # choose random i, j distinct
    i = rng.integers(n)
    j = rng.integers(n-1)
    if j >= i: j+=1
    # ensure direction of mass movement: ensure h[i] > (some) and h[j] < (some)
    # If not, swap i and j.
    if h[i] == 0 or h[j] == 1:
        # find suitable pair via loop up to attempts.
        for _ in range(10):
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i: j+=1
            if h[i] > 0 and h[j] < 1:
                break
    max_delta = min(h[i], 1 - h[j], max_move)
    if max_delta <= 0:
        return None
    delta = rng.random() * max_delta
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h, (i, j, delta)

def evaluate_h(h):
    n = h.shape[0]

    min_overlap = np.min(np.concatenate([conv[:center], conv[center+1:]]))
