import numpy as np
import time
import math

def max_correlation(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1‑h(x+k)) dx for the discrete step function."""
    N = h.shape[0]
    dx = 2.0 / N
    # linear correlation (no wrap) – same as the reference implementation
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)


def make_initial_binary(N: int, rng: np.random.RandomState) -> np.ndarray:
    """Create a length‑N binary vector with exactly N/2 ones.
    If a global `initial_h_values` is present and long enough we start from it."""
    init = globals().get("initial_h_values")
    target = N // 2
    if init is not None and len(init) >= N:
        base = np.array(init[:N], dtype=float)
        bin_vec = (base > 0.5).astype(float)          # simple threshold
        cur = int(bin_vec.sum())
        diff = target - cur
        if diff > 0:
            zeros = np.where(bin_vec == 0)[0]
            if len(zeros) >= diff:
                pick = rng.choice(zeros, size=diff, replace=False)
                bin_vec[pick] = 1.0
        elif diff < 0:
            ones = np.where(bin_vec == 1)[0]
            if len(ones) >= -diff:
                pick = rng.choice(ones, size=-diff, replace=False)
                bin_vec[pick] = 0.0
        return bin_vec
    # pure random start
    vec = np.zeros(N, dtype=float)
    ones = rng.choice(N, size=target, replace=False)
    vec[ones] = 1.0
    return vec


def hill_climb(start_vec: np.ndarray,
