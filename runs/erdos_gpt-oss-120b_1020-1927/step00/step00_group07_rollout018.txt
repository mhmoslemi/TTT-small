    n = len(v)

import time, math
import numpy as np

def _c5_score(h: np.ndarray) -> float:
    """Return C5 = dx * max_{k} sum h[i]*(1-h[i+k])   (full correlation)."""
    n = h.size
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / n)   # dx = 2/n


def _proj_box_sum(v: np.ndarray, target: float, eps: float = 1e-10) -> np.ndarray:
    """Project v onto {x : 0<=x_i<=1, sum x_i = target}."""
    n = v.size
    if target <= 0:
        return np.zeros_like(v)
    if target >= n:
        return np.ones_like(v)

    # lambda bounds so that clipped(v+lam) is all 0 or all 1
    lam_low = -np.max(v) - 1.0
    lam_high = 1.0 - np.min(v) + 1.0

    for _ in range(60):
        lam = (lam_low + lam_high) * 0.5
        p = np.clip(v + lam, 0.0, 1.0)
        s = p.sum()
        if s < target:
            lam_low = lam
        else:
            lam_high = lam
        if lam_high - lam_low < eps:
            break
    return np.clip(v + lam, 0.0, 1.0)


def _worst_shift_gradient(h: np.ndarray):
    """Gradient of the worst (max) shift term."""
    n = h.size
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(conv.argmax())
    k = idx - (n - 1)                     # shift amount (may be negative)
    grad = np.zeros_like(h)

    i0 = max(0, -k)
    i1 = min(n, n - k)
    if i0 >= i1:
        return grad, k

    i = np.arange(i0, i1, dtype=int)
    ip = i + k
    grad[i] += 1.0 - h[ip]                # derivative w.r.t. h_i
    grad[ip] -= h[i]                      # derivative w.r.t. h_{i+k}
    return grad, k


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) with a good upper bound for C5."""
    np.random.seed(seed)
    t0 = time.time()

    # ---- initialise ---------------------------------------------------------
    try:
        h = np.array(initial_h_values, dtype=float)   # provided by the harness
        n = h.size
    except Exception:
        n = 80
        v = np.random.rand(n)
        h = _proj_box_sum(v, target=n / 2.0)

    # make sure the start satisfies the constraints exactly
    h = _proj_box_sum(h, target=n / 2.0)
    cur_score = _c5_score(h)

    best_h = h.copy()
    best_score = cur_score

    # ---- optimisation parameters -------------------------------------------
    target_sum = n / 2.0
    step = 0.12                # initial step length
    step_decay = 0.9996
    temp = 0.015               # simulated‑annealing temperature
    temp_decay = 0.9995
    min_temp = 1e-7

    it = 0
    last_improve = 0

    # ---- main loop ---------------------------------------------------------
    while time.time() - t0 < budget_s - 2.0:
        # gradient of the worst shift
        grad, _ = _worst_shift_gradient(h)

        # tentative move and projection onto the feasible set
        h_new = h - step * grad
        h_new = _proj_box_sum(h_new, target=target_sum)

        new_score = _c5_score(h_new)

        # acceptance rule (deterministic improvement or SA)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis step
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            h = h_new
            cur_score = new_score

        # keep global best
        if new_score < best_score:
            best_h = h_new.copy()
            best_score = new_score
            last_improve = it

        # parameter schedules
        step *= step_decay
        temp = max(temp * temp_decay, min_temp)
        it += 1

        # occasional diversification if stuck
        if it - last_improve > 25000:
            # jitter the current best and continue
            noise = np.random.normal(scale=0.04, size=n)
            h = best_h + noise
            h = np.clip(h, 0.0, 1.0)
            h = _proj_box_sum(h, target=target_sum)
            cur_score = _c5_score(h)
            last_improve = it
            # reset step/temperature a little
            step = 0.12
            temp = 0.015

    # final projection (ensures exact sum)
    best_h = _proj_box_sum(best_h, target=target_sum)
    best_score = _c5_score(best_h)

    return (best_h, float(best_score), n)
