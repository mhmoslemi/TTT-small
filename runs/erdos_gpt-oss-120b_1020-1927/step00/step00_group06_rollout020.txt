def compute_C5(h):
    N = len(h)
    a = h
    b = 1 - h
    # compute cross-correlation full: same as convolution of a and reverse(b)
    b_rev = b[::-1]
    n = 2 * N - 1
    # perform FFT with zero padding to next power of two size
    size = 1 << (2 * N - 1).bit_length()
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b_rev, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    # Note: correlation index shift: conv[0] corresponds to shift = -(N-1) ...
    # multiplication factor dx
    dx = 2.0 / N
    return np.max(conv) * dx

def project_onto_box_and_sum(w, target, low=0.0, high=1.0):
    # w: array of floats
    # find lambda such that sum(clip(w - lambda, low, high)) = target.
    # lambda can be any real.
    lo = -1e5
    hi = 1e5
    # Actually need to bound lambda such that w - lambda within [0,1]
    # we can search in range [min(w)-1, max(w)] maybe.
    lo = np.min(w) - 1.0
    hi = np.max(w) - 0.0
    # Use binary search for tolerance
    for _ in range(60):
        mid = (lo + hi) / 2.0
        v = np.clip(w - mid, 0.0, 1.0)
        s = v.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(w - (lo+hi)/2.0, 0.0, 1.0)

import time
import math
import random
import numpy as np

def _project_onto_box_and_sum(v: np.ndarray, target: float, low: float = 0.0, high: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x : low <= x_i <= high, sum(x)=target}.
    Uses a water‑filling (binary search for the Lagrange multiplier).
    """
    # Initial search interval for lambda
    lo = np.min(v) - high   # makes v - lo >= high  => sum = n*high
    hi = np.max(v) - low    # makes v - hi <= low   => sum = 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, low, high)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) * 0.5, low, high)

    # Tiny correction if numerical error leaves sum off by >1e-12
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = np.where((proj > low + 1e-12) & (proj < high - 1e-12))[0]
        if free.size:
            proj[free] += diff / free.size
            proj = np.clip(proj, low, high)
    return proj

def _c5_value(h: np.ndarray, dx: float) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k)) dx for the discretised h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that lowers the
    Erdős overlap constant C₅.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # ---- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        N = h.shape[0]
    else:
        N = 120                     # fallback size (still < 1000)
        h = np.full(N, 0.5, dtype=np.float64)

    target_sum = N / 2.0
    # Make sure the starting vector satisfies the constraints
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target_sum) > 1e-12:
        h = _project_onto_box_and_sum(h, target_sum)

    dx = 2.0 / N
    best_h = h.copy()
    best_c5 = _c5_value(best_h, dx)

    # ---- optimisation parameters ------------------------------------
    lr = 0.6                     # step‑size multiplier
    temp = 0.04                  # simulated‑annealing temperature
    iter_cnt = 0
    last_improve_iter = 0
    # after this many iterations without improvement we trigger a small random restart
    stagnation_limit = 3000

    while True:
        now = time.time()
        if now - start_time > budget_s * 0.97:
            break

        # current correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_c5 = np.max(corr) * dx
        idx_max = np.argmax(corr)
        shift = idx_max - (N - 1)                # shift relative to zero

        # direction that tends to lower the worst overlap
        neigh = np.zeros(N, dtype=np.float64)
        if shift >= 0:
            neigh[: N - shift] += h[shift:]
            neigh[shift:]      += h[: N - shift]
        else:
            s = -shift
            neigh[s:]         += h[: N - s]
            neigh[: N - s]    += h[s:]

        # build a proposal
        noise = np.random.randn(N) * (lr * 0.12)
        proposal = h + lr * neigh + noise
        proposal = _project_onto_box_and_sum(proposal, target_sum)

        # evaluate proposal
        prop_c5 = _c5_value(proposal, dx)

        # deterministic improvement check
        improved = prop_c5 < best_c5 - 1e-9
        if improved:
            best_c5 = prop_c5
            best_h = proposal.copy()
            last_improve_iter = iter_cnt

        # SA acceptance for the current state (not the global best)
        accept = False
        if prop_c5 < cur_c5:
            accept = True
        else:
            # probability to accept a worse move
            delta = prop_c5 - cur_c5
            if np.random.rand() < math.exp(-delta / (temp + 1e-12)):
                accept = True
        if accept:
            h = proposal

        # decay learning rate & temperature
        if iter_cnt % 500 == 0 and iter_cnt:
            lr = max(0.02, lr * 0.985)
            temp = max(1e-5, temp * 0.985)

        # occasional random jitter if we stagnate too long
        if iter_cnt - last_improve_iter > stagnation_limit:
            # create a fresh start around the current best
            jitter = np.random.randn(N) * 0.15
            h = best_h + jitter
            h = _project_onto_box_and_sum(h, target_sum)
            last_improve_iter = iter_cnt

        iter_cnt += 1

    # final projection to guarantee feasibility
    best_h = _project_onto_box_and_sum(best_h, target_sum)
    final_c5 = _c5_value(best_h, dx)

    return (best_h, final_c5, N)
