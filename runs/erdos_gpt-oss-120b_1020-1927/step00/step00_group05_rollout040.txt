def project_box_simplex(x, target):
    lo, hi = -1.0, 2.0
    for _ in range(40):
        lam = (lo + hi) / 2.0
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
```

import numpy as np, time, math

def project_box_simplex(v, target):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def evaluate_and_gradient(h):
    n = h.shape[0]
    conv = np.correlate(h, 1 - h, mode='full')
    idx = int(np.argmax(conv))
    max_val = conv[idx] * (2.0 / n)
    L = idx - (n - 1)
    g = np.ones_like(h)
    if L >= 0:
        if L < n:
            g[:n - L] -= h[L:]
            g[L:]   -= h[:n - L]
    else:
        d = -L
        if d < n:
            g[d:]   -= h[:n - d]
            g[:n - d] -= h[d:]
    return max_val, g, L

def compute_c5(h):
    n = h.shape[0]
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # number of discretization points (must be even)
    n = 512
    target_sum = n / 2.0
    best_h = None
    best_val = float('inf')
    # helper to create a feasible vector from raw data
    def make_feasible(v):
        return project_box_simplex(v, target_sum)
    # main loop with restarts until time runs out
    while time.time() - start < budget_s * 0.9:
        # initialise start vector
        if rng.random() < 0.3:
            # triangular shape (continuous)
            dx = 2.0 / n
            xs = (np.arange(n) + 0.5) * dx
            h0 = np.where(xs <= 1.0, xs, 2.0 - xs)
            # adjust sum to exact target
            s = h0.sum()
            diff = target_sum - s
            if abs(diff) > 1e-12:
                if diff > 0:
                    # increase values where possible
                    cap = 1.0 - h0
                    order = np.argsort(-cap)   # biggest capacity first
                    for i in order:
                        add = min(cap[i], diff)
                        h0[i] += add
                        diff -= add
                        if diff <= 1e-12:
                            break
                else:
                    diff = -diff
                    cap = h0
                    order = np.argsort(cap)    # smallest values first
                    for i in order:
                        sub = min(cap[i], diff)
                        h0[i] -= sub
                        diff -= sub
                        if diff <= 1e-12:
                            break
            h = h0.copy()
        else:
            # random feasible point
            raw = rng.random(n)
            h = make_feasible(raw)
        # evaluate initial
        cur_val = compute_c5(h)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        # gradient descent for this start
        max_iters = 60000
        step0 = 0.5
        for it in range(max_iters):
            if time.time() - start > budget_s * 0.9:
                break
            val, grad, L = evaluate_and_gradient(h)
            step = step0 / math.sqrt(it + 1)
            h_temp = h - step * grad
            h = make_feasible(h_temp)
            new_val = compute_c5(h)
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()

                    noise = rng.normal(scale=0.02, size=n)
                    h = make_feasible(h + noise)
        # end of this restart
    # final evaluation using provided helper
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), float(final_c5), n)

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper functions (top level as required)
# --------------------------------------------------------------

def _project_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the intersection of the box [0,1]^n
    and the hyperplane sum(x) = target.
    """
    lo, hi = -1.0, 2.0  # guarantees sum at lo > target, sum at hi < target
    for _ in range(50):
        lam = (lo + hi) / 2.0
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _evaluate_and_gradient(h: np.ndarray):
    """
    Returns (C5 value, subgradient at the worst shift, shift index).
    Subgradient of the max‑overlap functional is the gradient of the
    specific shift attaining the maximum.
    """
    n = h.shape[0]
    conv = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(conv))
    max_val = conv[idx] * (2.0 / n)
    L = idx - (n - 1)                     # integer shift
    g = np.ones_like(h)                  # gradient of f_L = Σ h_i(1‑h_{i+L})

    if L >= 0:
        if L < n:
            g[: n - L] -= h[L:]          # term (1‑h_{i+L})
            g[L :]   -= h[: n - L]       # term (‑h_{i‑L})
    else:
        d = -L
        if d < n:
            g[d :]   -= h[: n - d]       # (1‑h_{i+L}) where i+L in range
            g[: n - d] -= h[d :]         # (‑h_{i‑L})
    return max_val, g, L


def _compute_c5(h: np.ndarray) -> float:
    """Plain evaluation of the objective (max overlap * dx)."""
    n = h.shape[0]
    conv = np.correlate(h, 1 - h, mode="full")
    return np.max(conv) * (2.0 / n)


# --------------------------------------------------------------
# Main optimisation routine
# --------------------------------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # discretisation size (even, ≤1000). 512 gives a fine grid.
    n = 512
    target_sum = n / 2.0          # integral constraint ⇒ Σh = n/2

    best_h = None
    best_val = float("inf")

    # ----------------------------------------------------------
    # Local helper to turn any vector into a feasible one
    # ----------------------------------------------------------
    def _make_feasible(v):
        return _project_box_simplex(v, target_sum)

    # ----------------------------------------------------------
    # Main optimisation loop with many random restarts
    # ----------------------------------------------------------
    while time.time() - start < budget_s * 0.9:
        # ---- initialise a starting point ---------------------------------
        if rng.random() < 0.3:
            # triangular shape (continuous, area 1)
            dx = 2.0 / n
            xs = (np.arange(n) + 0.5) * dx
            h = np.where(xs <= 1.0, xs, 2.0 - xs)

            # adjust to meet the exact integral constraint
            cur = h.sum()
            diff = target_sum - cur
            if abs(diff) > 1e-12:
                if diff > 0:
                    # add mass where we have room (closest to 1)
                    capacity = 1.0 - h
                    order = np.argsort(-capacity)
                    for i in order:
                        add = min(capacity[i], diff)
                        h[i] += add
                        diff -= add
                        if diff <= 1e-12:
                            break
                else:
                    diff = -diff
                    # remove mass from the smallest entries first
                    order = np.argsort(h)
                    for i in order:
                        sub = min(h[i], diff)
                        h[i] -= sub
                        diff -= sub
                        if diff <= 1e-12:
                            break
        else:
            # completely random feasible point
            raw = rng.random(n)
            h = _make_feasible(raw)

        # ---- keep the best seen so far ------------------------------------
        cur_val = _compute_c5(h)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        # ---- gradient‑based refinement ------------------------------------
        max_iters = 60000
        step0 = 0.5
        for it in range(max_iters):
            if time.time() - start > budget_s * 0.9:
                break

            val, grad, _ = _evaluate_and_gradient(h)
            step = step0 / math.sqrt(it + 1.0)
            h = _make_feasible(h - step * grad)

            new_val = _compute_c5(h)
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()

            # occasional random jitter to avoid tiny plateaus
            if (it & 0xFFF) == 0 and rng.random() < 0.3:
                h = _make_feasible(h + rng.normal(scale=0.02, size=n))

        # end of a restart -------------------------------------------------

    # Final evaluation using the supplied harness function
    final_c5 = evaluate_erdos_solution(best_h)

    # Return (h_values, c5_bound, n_points) as required
    return (best_h.tolist(), float(final_c5), n)
