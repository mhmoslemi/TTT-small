def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    # multiply by dx
    return np.max(corr) * (2.0 / len(h))

    N = len(h)

    G = np.fft.rfft(np.concatenate([1 - h, np.zeros(n - N)]))
    conv = np.fft.irfft(H * G, n=n)

    conv = conv[:2*N-1]
    return np.max(conv) * (2.0 / N)

def propose_move(parent, eps):
    i, j = rng.choice(N, size=2, replace=False)
    # Determine max possible transfer
    # Random direction
    direction = rng.choice([1, -1])

def local_optimize(h_start, time_limit, start_time):
    h = h_start.copy()
    N = h.shape[0]
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score
    # schedule parameters
    max_iters = 500
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        # find worst shift k
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        k = idx - (N - 1)
        # gradient
        g = np.zeros(N, dtype=float)
        if k >= 0:
            if N - k > 0:
                g[:N - k] += 1.0 - h[k:]
                g[k:] -= h[:N - k]
        else:
            kk = -k
            if N - kk > 0:
                g[kk:] += 1.0 - h[:N - kk]
                g[:N - kk] -= h[kk:]
        # step size schedule
        step = 0.1 * (1.0 - it / max_iters) + 0.001
        h_new = h - step * g
        # enforce bounds and sum
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_sum(h_new, N / 2)
        new_score = compute_c5(h_new)
        # accept if better else maybe accept with small prob
        if new_score < cur_score:
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            # simple simulated annealing temperature decreasing
            temp = 0.01 * (1.0 - it / max_iters) + 1e-5
            if np.exp((cur_score - new_score) / temp) > np.random.rand():
                h = h_new

                g[:m] -= h[kk:]
        # step size that shrinks linearly
        step = 0.15 * (1.0 - it / max_iter) + 0.001
        h_new = h - step * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _project_sum(h_new, n / 2.0)
        val = _c5(h_new)
        # greedy accept, with occasional uphill moves
        if val < cur:
            h, cur = h_new, val
            if val < best_val:
                best_val, best_h = val, h_new.copy()
        else:
            temp = 0.02 * (1.0 - it / max_iter) + 1e-5
            if math.exp((cur - val) / temp) > np.random.rand():
                h, cur = h_new, val
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise ---------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = _project_sum(h0, n / 2.0)
        best_h = h0.copy()
        best_val = _c5(best_h)
    else:
        n = 150
        h0 = rng.random(n)
        h0 = _project_sum(h0, n / 2.0)
        best_h = h0.copy()
        best_val = _c5(best_h)

    # ---- main optimisation loop --------------------------------------------
    deadline = budget_s - 0.3
    while time.time() - start < deadline:
        # new starting point: perturb best or draw fresh
        if rng.random() < 0.6:
            cand = best_h + rng.normal(scale=0.12, size=n)
        else:
            cand = rng.random(n)
        cand = np.clip(cand, 0.0, 1.0)
        cand = _project_sum(cand, n / 2.0)

        # local optimisation from this start
        opt_h, opt_val = _local_optimize(cand, deadline, start)
        if opt_val < best_val:
            best_val, best_h = opt_val, opt_h.copy()

    return (best_h, float(best_val), int(n))
