    rng = np.random.default_rng(seed)
    start = time.time()

    n = 200
    dx = 2.0 / n

    h[ones] = 1.0

    def compute_max_count(v):

        # v is np.ndarray float
        corr = np.correlate(v, 1 - v, mode='full')
        return int(corr.max())

import time
import math
import numpy as np

def _max_corr_info(h):
    """Return (max_count, lag, full_corr) for binary-like h."""
    corr = np.correlate(h, 1 - h, mode='full')
    idx = int(corr.argmax())
    max_val = int(round(corr[idx]))
    lag = idx - (len(h) - 1)
    return max_val, lag, corr

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with ∫h=1, 0≤h≤1
    that minimizes C5 = max_k ∫ h(x)(1-h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.98   # safety margin

    # ----- parameters -----
    n = 300                     # number of samples (adjustable)
    dx = 2.0 / n
    target_counts = int(math.floor(0.3808 * n / 2))  # required max count to beat record
    max_temp = 20.0            # initial temperature for SA
    prob_target = 0.8          # probability to target current worst lag
    max_no_improve = 150000    # iterations without improvement before restart

    # ----- initial binary vector (random) -----
    h = np.zeros(n, dtype=np.float64)
    ones_idx = rng.choice(n, size=n // 2, replace=False)
    h[ones_idx] = 1.0

    cur_max, cur_lag, _ = _max_corr_info(h)
    best_h = h.copy()
    best_max = cur_max
    best_lag = cur_lag

    last_improve_iter = 0
    iteration = 0

    # main SA loop
    while time.time() < deadline:
        # temperature decreases linearly with time
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s * 0.98)
        T = max_temp * (1.0 - frac) + 0.001

        # ----- choose swap -----
        if rng.random() < prob_target and cur_max > 0:
            lag = cur_lag
            if lag >= 0:
                # find positions where 1 aligns with 0 at this lag
                mask = (h[:n - lag] == 1) & (h[lag:] == 0)
                bad = np.flatnonzero(mask)
                if bad.size == 0:
                    i = rng.choice(np.flatnonzero(h == 1))
                else:
                    i = rng.choice(bad)
                # choose a zero elsewhere, avoiding the partner if possible
                zeros = np.flatnonzero(h == 0)
                partner = i + lag
                if zeros.size > 1 and partner < n and h[partner] == 0:
                    zeros = zeros[zeros != partner]
                j = rng.choice(zeros)
            else:
                L = -lag
                mask = (h[:n - L] == 0) & (h[L:] == 1)
                bad = np.flatnonzero(mask)
                if bad.size == 0:
                    i = rng.choice(np.flatnonzero(h == 0))
                else:
                    i = rng.choice(bad)
                ones = np.flatnonzero(h == 1)
                partner = i + L
                if ones.size > 1 and partner < n and h[partner] == 1:
                    ones = ones[ones != partner]
                j = rng.choice(ones)
        else:
            # pure random swap (one from ones, one from zeros)
            i = rng.choice(np.flatnonzero(h == 1))
            j = rng.choice(np.flatnonzero(h == 0))

        # perform the swap
        h[i], h[j] = h[j], h[i]

        # evaluate new state
        new_max, new_lag, _ = _max_corr_info(h)

        delta = new_max - cur_max
        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_max, cur_lag = new_max, new_lag
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                best_lag = new_lag
                last_improve_iter = iteration
                # early exit if we already beat the target threshold
                if best_max <= target_counts:
                    break
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]

        iteration += 1

        # restart if stuck too long
        if iteration - last_improve_iter > max_no_improve:
            h[:] = 0.0
            ones_idx = rng.choice(n, size=n // 2, replace=False)
            h[ones_idx] = 1.0
            cur_max, cur_lag, _ = _max_corr_info(h)
            last_improve_iter = iteration

    c5_bound = best_max * dx
    return best_h.tolist(), c5_bound, n
