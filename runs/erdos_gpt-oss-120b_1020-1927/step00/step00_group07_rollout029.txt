def compute_score(h):
    N = len(h)
    dx = 2.0 / N
    a = h
    b = 1 - h
    m = 2*N - 1
    # Next pow2 for speed
    size = 1 << (m-1).bit_length()  # next power of two
    Fa = np.fft.rfft(np.pad(a, (0, size - N), 'constant'))
    Fb = np.fft.rfft(np.pad(b, (0, size - N), 'constant'))
    # cross-correlation via conj(Fa) * Fb
    # Actually using conj(Fa) * Fb yields correlation a with b.
    # Compute conv = irfft(Fa.conj() * Fb, n=size)
    corr = np.fft.irfft(Fa.conj() * Fb, n=size)
    # correlation length = size, but first m entries are meaningful
    corr = corr[:m] * dx
    return float(np.max(corr))

    # returns x satisfying 0<=x<=1 and sum equals S, minimizing Euclidean distance.
    # Use water-filling via binary search on lambda.

    high = 1.0 - np.min(y) + 1.0

    for _ in range(50):
        mid = (low + high) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        s = np.sum(x)
        if s > S:
            high = mid
        else:
            low = mid
    # final:
    x = np.clip(y + low, 0.0, 1.0)
    # The sum may be slightly off due to precision; adjust by distributing residual among free variables.
    diff = S - np.sum(x)
    if abs(diff) > 1e-8:
        # adjust small diff via adding equally to variables not at bounds, respecting bounds
        free = (x > 0) & (x < 1)
        if np.any(free):
            # distribute equally
            inc = diff / np.sum(free)
            x[free] += inc
            # clip again
            x = np.clip(x, 0.0, 1.0)
        else:
            # If all variables at bounds (rare), then adjust minimal.
            pass
    return x

def project_box_sum(y, target_sum):
    # Clip not required initially but needed for search.
    # We'll binary search lambda such that sum(clamp(y+lambda,0,1)) = target_sum
    # lower bound can be -max(y) - 1 (ensuring y+lambda <=0).
    # upper bound = 1 - min(y) + 1.
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + lo, 0.0, 1.0)
    # Due to rounding, sum may differ a bit; enforce exactly target by adjusting free variables
    diff = target_sum - x.sum()
    if abs(diff) > 1e-8:
        free = (x > 0) & (x < 1)
        if np.any(free):
            # distribute diff equally to free entries, ensuring bounds
            inc = diff / free.sum()
            x[free] += inc
            # clip again (should be within bounds)
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: adjust one entry within bounds
            # find index that can adjust
            for i in range(len(x)):
                # compute max adjustment possible
                max_adj = 1 - x[i] if diff > 0 else -x[i]
                adj = np.clip(diff, -x[i], 1 - x[i])
                x[i] += adj
                diff -= adj
                if abs(diff) < 1e-12:
                    break
    return x

def compute_score(h):
    N = h.shape[0]
    dx = 2.0 / N
    # next power of 2 length for zero padding
    n = 1 << ((2*N-1)-1).bit_length()
    # FFT of h and 1-h
    H = np.fft.rfft(np.concatenate([h, np.zeros(n - N)]))
    G = np.fft.rfft(np.concatenate([1 - h, np.zeros(n - N)]))
    corr = np.fft.irfft(H.conj() * G, n=n)
    # correlation size = n, linear part first 2N-1
    corr = corr[:2*N-1] * dx
    # return max
    return float(np.max(corr))

def run(seed=42,budget_s=1100,**kwargs):

    rng = np.random.default_rng(seed)
    start = time.time()
    # Choose number of points
    N = 200  # base grid
    dx = 2.0 / N
    target_sum = N / 2

    # Setup compute_score closure with N, dx.
    # Already defined.

    # Phase 1: param search over epsilon, a
    best_h = None
    best_score = math.inf

    # Define epsilon grid
    eps_vals = np.linspace(0.05, 0.45, 41)  # step 0.01
    a_vals = np.linspace(0.6, 1.0, 9)  # try values

    xs = np.linspace(0,2,N,endpoint=False)  # sample points

    for eps in eps_vals:
        for a in a_vals:
            # compute b from area: 2*eps*a + (2-2*eps)*b = 1
            denom = 2 - 2*eps
            if denom <= 0:
                continue
            b = (1 - 2*eps*a) / denom
            if b < 0 or b > a or b > 1:
                continue

            h = project_box_sum(h, target_sum)
            score = compute_score(h)
            if score < best_score:
                best_score = score
                best_h = h.copy()
    # Use also constant 0.5 as baseline
    h_const = np.full(N, 0.5)
    h_const = project_box_sum(h_const, target_sum)
    score_const = compute_score(h_const)
    if score_const < best_score:
        best_score = score_const
        best_h = h_const.copy()

    # If initial_h_values provided and length differs, resample or pad.
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        if init.size != N:
            # Interpolate to N points
            x_old = np.linspace(0,2,init.size,endpoint=False)
            h_resampled = np.interp(xs, x_old, init)
        else:
            h_resampled = init.copy()
        h_resampled = project_box_sum(h_resampled, target_sum)
        score = compute_score(h_resampled)
        if score < best_score:
            best_score = score
            best_h = h_resampled.copy()

    # Phase 2: evolutionary search
    popsize = 80
    # Initialize population
    population = np.empty((popsize, N), dtype=float)
    # First member = best_h found
    population[0] = best_h
    # Fill rest random
    for i in range(1, popsize):
        y = rng.random(N)
        population[i] = project_box_sum(y, target_sum)

    # Evolution parameters
    sigma = 0.2  # initial mutation scale
    min_sigma = 0.001
    sigma_decay = 0.995
    elite_frac = 0.1
    generation = 0

    best_h_global = best_h.copy()

        scores = np.empty(popsize)
        for i in range(popsize):

        # Update global best
        idx_best = np.argmin(scores)
        if scores[idx_best] < best_score_global:
            best_score_global = scores[idx_best]
            best_h_global = population[idx_best].copy()
        # Selection: elite individuals
        elite_count = max(1, int(popsize * elite_frac))
        elite_idx = np.argpartition(scores, elite_count)[:elite_count]
        elites = population[elite_idx]

        # Create new population
        new_pop = np.empty_like(population)
        # Keep elite
        new_pop[:elite_count] = elites
        # Fill rest
        for i in range(elite_count, popsize):

            child = parent + sigma * rng.standard_normal(N)
            child = project_box_sum(child, target_sum)

        # Decay sigma
        sigma = max(min_sigma, sigma * sigma_decay)

        generation += 1
        # Optional early stop if score very low
        if best_score_global < 0.3808:
            break

    # Return final best
    return best_h_global, best_score_global, N

def binary_sa(N, rng, time_limit):
    # start with random binary vector with N/2 ones.
    h = np.zeros(N, dtype=float)
    ones_idx = rng.choice(N, N//2, replace=False)
    h[ones_idx] = 1.0
    # compute score
    best_h = h.copy()
    best_score = compute_score(h)
    cur_h = h.copy()
    cur_score = best_score

    # temperature schedule
    temperature = 0.1
    min_temp = 1e-5
    decay = 0.9995

    while time.time() < time_limit:
        # pick i (1) and j (0)
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if len(ones) ==0 or len(zeros) ==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_score = compute_score(cur_h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta/temperature):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # reject -> swap back
            cur_h[i], cur_h[j] = 1.0, 0.0
        temperature = max(min_temp, temperature * decay)
        if best_score < 0.3808:
            break
    return best_h, best_score

def compute_score(h):
    N = len(h)
    dx = 2.0 / N
    n = 1 << ((2*N-1)-1).bit_length()  # next power of 2
    H = np.fft.rfft(np.concatenate([h, np.zeros(n - N)]))
    G = np.fft.rfft(np.concatenate([1 - h, np.zeros(n - N)]))
    corr = np.fft.irfft(H.conj() * G, n=n)
    # take first 2N-1 values
    max_val = np.max(corr[:2*N-1]) * dx
    return float(max_val)

import time, math
import numpy as np

def _project_box_sum(y: np.ndarray, target_sum: float) -> np.ndarray:
    """Project y onto the set {0≤x≤1, sum(x)=target_sum} (Euclidean projection)."""
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + lo, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0) & (x < 1)
        if free.any():
            inc = diff / free.sum()
            x[free] += inc
            x = np.clip(x, 0.0, 1.0)
    return x


def _compute_score(h: np.ndarray) -> float:
    """C5 value for a discrete step function h on [0,2] with N points."""
    N = h.shape[0]
    dx = 2.0 / N
    # next power of two ≥ 2N‑1 for linear convolution via FFT
    n = 1 << ((2 * N - 1) - 1).bit_length()
    H = np.fft.rfft(np.concatenate([h, np.zeros(n - N)]))
    G = np.fft.rfft(np.concatenate([1 - h, np.zeros(n - N)]))
    corr = np.fft.irfft(H.conj() * G, n=n)
    max_val = np.max(corr[: 2 * N - 1]) * dx
    return float(max_val)
