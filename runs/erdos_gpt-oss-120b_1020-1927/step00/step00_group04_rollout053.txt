```
def project_onto_constraints(x, target):
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # iterate
    for _ in range(10):
        current_sum = x.sum()
        diff = target - current_sum
        if abs(diff) < 1e-10:
            break
        free = (x > 0) & (x < 1)
        free_idx = np.where(free)[0]
        if len(free_idx) == 0:
            # all at bounds, cannot adjust
            break
        delta = diff / len(free_idx)
        x[free_idx] += delta
        # clip again
        x = np.clip(x, 0.0, 1.0)
    return x
```

import numpy as np
import time

def _evaluate_c5(h: np.ndarray) -> float:
    """Return the C5 value for a given step function h."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / h.size)


def _project_onto_constraints(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector x onto the box [0,1]^n while preserving the linear sum = target_sum.
    Simple water-filling iteration – converges in a few passes.
    """
    x = np.clip(x, 0.0, 1.0)
    for _ in range(20):
        cur = x.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        idx = np.where(free)[0]
        if idx.size == 0:
            # all variables at a bound; distribute the remaining diff uniformly and clip again
            x += diff / x.size
            x = np.clip(x, 0.0, 1.0)
            continue
        delta = diff / idx.size
        x[idx] += delta
        x = np.clip(x, 0.0, 1.0)
    return x


def _gradient_of_worst_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Gradient of the overlap functional for the shift that currently attains the maximum.
    shift > 0 corresponds to xi shifted right by `shift` positions.
    """
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        # first term: (1 - h[i+shift]) for i in [0, n-shift)
        g[: n - shift] += 1.0 - h[shift:]
        # second term: -h[i-shift] for i in [shift, n)
        g[shift:] -= h[: n - shift]
    else:
        t = -shift
        # first term for negative shift
        g[t:] += 1.0 - h[: n - t]
        # second term
        g[: n - t] -= h[t:]
    return g
