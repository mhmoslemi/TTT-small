def project_box_simplex(v, target, lower=0.0, upper=1.0):
    # Solve min ||x - v||^2 s.t. lower <= x_i <= upper, sum x_i = target
    # Use binary search for lambda.
    lo = -1e6
    hi = 1e6
    for _ in range(80):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2.0, lower, upper)

    # If sum not exactly target due to hitting bounds, we need to distribute leftover onto free variables.

import time, math
import numpy as np

def _proj_box_simplex(v, target, lower=0.0, upper=1.0):
    """Project v onto {x | lower<=x_i<=upper, sum x_i = target}."""
    lo, hi = -1e6, 1e6
    for _ in range(80):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, lower, upper)

    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------
    n = 200                         # number of sample points on [0,2]
    dx = 2.0 / n
    target_sum = n / 2.0

    # ------------------------------------------------------------

    # ------------------------------------------------------------
    if 'initial_h_values' in globals():
        raw = np.asarray(initial_h_values, dtype=float)

            xs_old = np.linspace(0.0, 2.0, raw.size, endpoint=False)
            xs_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(xs_new, xs_old, raw)
        else:
            h0 = raw.copy()
    else:
        # triangular function: height 1 at centre, linear to 0 at edges
        x = (np.arange(n) + 0.5) * dx
        h0 = np.maximum(0.0, 1.0 - np.abs(x - 1.0))

    h0 = _proj_box_simplex(h0, target_sum)

    # ------------------------------------------------------------
    # Helper to evaluate the max overlap (the objective)
    # ------------------------------------------------------------
    def max_overlap(h):
        return np.correlate(h, 1.0 - h, mode='full').max() * dx

    best_h = h0.copy()
