import numpy as np
import time

def compute_c5(h):
    """Return the overlap constant C5 for a discrete step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # Length of the vector: use the provided construction if present,
    # otherwise fall back to a modest default.
    if "initial_h_values" in globals():
        n = len(initial_h_values)
    else:
        n = 200

    dx = 2.0 / n

    # Mid‑point sampling of the symmetric triangular function:
    #   h(x) = min(x, 2−x)  on [0,2].
    x = (np.arange(n) + 0.5) * dx          # points at the centre of each sub‑interval
    h = np.minimum(x, 2.0 - x)

    # Enforce the exact integral constraint ∑h·dx = 1  (i.e. ∑h = n/2).
    target_sum = n / 2.0
    scale = target_sum / h.sum()
    h = np.clip(h * scale, 0.0, 1.0)

    # Initial C5 value.
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Simple hill‑climbing (optional, runs only if time permits).
    # Swaps a tiny amount δ between two indices while preserving the sum.
    # This can only improve the bound, never worsen it.
    time_left = budget_s - (time.time() - start)
    if time_left > 20.0 and best_c5 > 0.38:
        max_iter = int(1e5 * (time_left / budget_s))
        for _ in range(max_iter):
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # propose moving a small amount from j to i
            delta = np.random.uniform(-0.02, 0.02)
            hi, hj = best_h[i], best_h[j]
            # keep values in [0,1] after the move
            if not (0.0 <= hi + delta <= 1.0 and 0.0 <= hj - delta <= 1.0):
                continue
            cand = best_h.copy()
            cand[i] += delta
            cand[j] -= delta
            c5cand = compute_c5(cand)
            if c5cand < best_c5:
                best_h, best_c5 = cand, c5cand
                # early exit if we already beat the target
                if best_c5 <= 0.3808:
                    break

    # Return as plain Python list for maximum compatibility.
    return (best_h.tolist(), best_c5, n)
