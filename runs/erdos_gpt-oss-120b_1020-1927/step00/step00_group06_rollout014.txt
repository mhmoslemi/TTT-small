def _project_onto_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the set {x : 0<=x_i<=1, sum(x)=target}.
    Uses bisection on the shifting parameter λ such that
    x_i = clip(v_i - λ, 0, 1) and sum(x)=target.
    """
    lo, hi = -1.0, 2.0  # guarantees sum at lo >= target, at hi <= target
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny adjustment to hit target exactly (optional)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        interior = (x > 0.0) & (x < 1.0)
        idx = np.where(interior)[0]
        if idx.size:
            add = diff / idx.size
            x[idx] += add
            np.clip(x, 0.0, 1.0, out=x)
    return x

        h = _project_onto_capped_simplex(h, target_sum)
    else:
        # half ones, half zeros, one 0.5 if n is odd
        h = np.zeros(n, dtype=np.float64)
        half = n // 2
        h[:half] = 1.0
        if n % 2:
            h[half] = 0.5
        rng.shuffle(h)

    # ----- basic evaluation -------------------------------------------
    curr_h = h.copy()
    curr_score = evaluate_erdos_solution(curr_h)
    best_h = curr_h.copy()
    best_score = curr_score

    # ----- simulated‑annealing style search ---------------------------
    start = time.time()
    max_fraction = 0.97  # keep a safety margin for finalisation
    iters = 0
    # parameters for the temperature schedule
    T_start, T_end = 0.008, 1e-6

    while True:
        now = time.time()
        if now - start >= budget_s * max_fraction:
            break

        # propose a two‑index move that preserves the sum exactly
        i, j = rng.integers(n, size=2)
        if i == j:
            continue

        # feasible delta range so that both entries stay in [0,1]
        lo = max(-curr_h[i],            curr_h[j] - 1.0)
        hi = min( 1.0 - curr_h[i],      curr_h[j])
        if lo >= hi:
            continue

        delta = rng.uniform(lo, hi)    # exact amount transferred from j to i
        new_h = curr_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_score = evaluate_erdos_solution(new_h)

        # acceptance rule (simulated annealing)
        if new_score < curr_score:
            accept = True
        else:
            # linear temperature decay based on elapsed fraction
            frac = 1.0 - (now - start) / budget_s
            T = T_start * frac + T_end
            accept = rng.random() < np.exp(-(new_score - curr_score) / T)

        if accept:
            curr_h = new_h
            curr_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score

        iters += 1

        # occasional aggressive perturbation around the best solution
        if iters % 5000 == 0 and rng.random() < 0.15:
            cand = best_h.copy()
            # apply a few random pair transfers with reduced magnitude
            for _ in range(rng.integers(1, 5)):
                a, b = rng.integers(n, size=2)
                if a == b:
                    continue
                lo = max(-cand[a],            cand[b] - 1.0)
                hi = min( 1.0 - cand[a],      cand[b])
                if lo >= hi:
                    continue
                d = rng.uniform(lo, hi) * 0.4
                cand[a] += d
                cand[b] -= d
            cand = np.clip(cand, 0.0, 1.0)
            cand = _project_onto_capped_simplex(cand, target_sum)
            cand_score = evaluate_erdos_solution(cand)
            if cand_score < best_score:
                best_h = cand.copy()
                best_score = cand_score
                curr_h = cand
                curr_score = cand_score

        # periodic re‑projection to avoid drift
        if iters % 10000 == 0:
            curr_h = _project_onto_capped_simplex(curr_h, target_sum)

    # ----- finalise ----------------------------------------------------
    best_h = _project_onto_capped_simplex(best_h, target_sum)
    final_score = evaluate_erdos_solution(best_h)

    return best_h.astype(np.float32), float(final_score), n
