def project_onto_simplex_bounded(v, target):
    # solve min ||x - v||^2 subject to sum x = target, 0 <= x <= 1
    # using water-filling: find lambda such that sum clip(v - lambda, 0, 1) = target
    n = v.shape[0]

    lower = np.min(v) - 1.0
    upper = np.max(v)
    for _ in range(100):
        lam = (lower + upper) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if np.abs(s - target) < 1e-12:
            break
        if s > target:
            # we need to increase lam to reduce sum
            lower = lam
        else:
            upper = lam
    return x
```

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    start = time.time()
    np.random.seed(seed)

    # Determine n_points
    global initial_h_values
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n_points = 70
        h0 = np.random.rand(n_points)
    n = h0.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0

    # Project initial onto feasible set (bounded simplex)
    h = project_onto_simplex_bounded(h0, target_sum)

    best_h = h.copy()
    best_score = compute_c5(h, dx)

    # Subgradient descent phase
    max_iter = 20000
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    eps = 1e-8
    for t in range(1, max_iter+1):
        # time check
        if time.time() - start > budget_s - 0.5:
            break

        # compute Hs
        conv = np.correlate(h, 1 - h, mode='full')
        H = conv * dx
        max_val = H.max()
        # track best
        if max_val < best_score:
            best_score = max_val
            best_h = h.copy()
        # get active shifts
        active_idx = np.where(np.isclose(H, max_val, atol=1e-12))[0]
        # compute subgradient
        grad = np.zeros_like(h)
        for idx in active_idx:
            s = idx - (n - 1)
            # i all indices
            indices = np.arange(n)
            i_plus_mask = (indices + s >= 0) & (indices + s < n)
            i_plus = indices[i_plus_mask]
            j_plus = i_plus + s

            i_minus_mask = (indices - s >= 0) & (indices - s < n)

            k_minus = i_minus - s
            grad[i_minus] -= dx * h[k_minus]

import time
import numpy as np

def _compute_c5(h, dx):
    """Return max overlap C5 for a given h."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _project_onto_bounded_simplex(v, target):
    """
    Project vector v onto {x: sum(x)=target, 0<=x<=1}.
    Uses water-filling (binary search on lambda).
    """
    lo = np.min(v) - 1.0          # ensures all clipped to 1 -> sum = len(v)
    hi = np.max(v) + 1.0          # ensures all clipped to 0 -> sum = 0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return x

def _subgradient_step(h, dx, target, m, v, t, beta1, beta2, eps):
    """
    Perform one Adam‑style subgradient step aimed at decreasing
    the max correlation.
    Returns (new_h, new_m, new_v, current_max).
    """
    n = h.size
    # correlation and max
    conv = np.correlate(h, 1.0 - h, mode="full")
    H = conv * dx
    max_val = H.max()
    # indices attaining the maximum (subgradient set)
    active = np.where(np.abs(H - max_val) < 1e-10)[0]
    # subgradient of the max (average over active shifts)
    grad = np.zeros_like(h)
    idxs = np.arange(n)
    for a in active:
        s = a - (n - 1)                # shift value
        # first term: i s.t. i+s in range
        mask_plus = (idxs + s >= 0) & (idxs + s < n)
        i_plus = idxs[mask_plus]
        j_plus = i_plus + s
        grad[i_plus] += dx * (1.0 - h[j_plus])
        # second term: i s.t. i-s in range
        mask_minus = (idxs - s >= 0) & (idxs - s < n)
        i_minus = idxs[mask_minus]
        k_minus = i_minus - s
        grad[i_minus] -= dx * h[k_minus]
    grad /= max(len(active), 1)

    # Adam update
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    lr = 0.5 / np.sqrt(t)                 # decreasing base step
    step = lr * m_hat / (np.sqrt(v_hat) + eps)

    y = h - step
    h_new = _project_onto_bounded_simplex(y, target)
    return h_new, m, v, max_val

def _simulated_annealing_binary(h_bin, dx, target, time_limit):
    """Improve a binary vector using swap‑based simulated annealing."""
    n = h_bin.size
    curr = h_bin.copy()
    best = curr.copy()
    curr_score = _compute_c5(curr, dx)
    best_score = curr_score

    start = time.time()
    it = 0
    # linear cooling in sync with remaining time
    while time.time() - start < time_limit:
        # pick random 1 and 0 positions
        ones = np.where(curr == 1)[0]
        zeros = np.where(curr == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = curr.copy()
        cand[i] = 0
        cand[j] = 1
        cand_score = _compute_c5(cand, dx)

        # temperature proportional to remaining budget fraction
        elapsed = time.time() - start
        frac = max(0.0, 1.0 - elapsed / time_limit)
        T = 0.02 * frac + 1e-12

        if cand_score < curr_score or np.random.rand() < np.exp(-(cand_score - curr_score) / T):
            curr = cand
            curr_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        it += 1
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the C5 bound."""
    np.random.seed(seed)
    start_time = time.time()

    # ----- initialise ---------------------------------------------------------
    try:

    h = _project_onto_bounded_simplex(h0, target_sum)

    # ----- phase 1: subgradient (Adam) optimisation -------------------------
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)

    # allocate roughly one third of the total budget to this phase
    phase1_time = budget_s * 0.35
    t = 1
    best_h = h.copy()
    best_score = _compute_c5(h, dx)

    while time.time() - start_time < phase1_time:
        h, m, v, cur_max = _subgradient_step(
            h, dx, target_sum, m, v, t, beta1, beta2, eps
        )
        if cur_max < best_score:
            best_score = cur_max
            best_h = h.copy()
        t += 1

    # ----- phase 2: binary simulated annealing -------------------------------
    # turn the best continuous vector into a feasible binary one
    order = np.argsort(best_h)[::-1]
    binary_h = np.zeros_like(best_h)
    binary_h[order[:int(target_sum)]] = 1.0

    remaining = budget_s - (time.time() - start_time) - 0.2
    if remaining > 0.0:
        bin_best, bin_score = _simulated_annealing_binary(
            binary_h, dx, target_sum, remaining
        )
        # pick the better of the two phases
        if bin_score < best_score:
            best_h = bin_best.astype(float)
            best_score = bin_score

    # final sanity – project once more to guarantee feasibility
    best_h = _project_onto_bounded_simplex(best_h, target_sum)
    final_score = _compute_c5(best_h, dx)

    return best_h.tolist(), float(final_score), n
