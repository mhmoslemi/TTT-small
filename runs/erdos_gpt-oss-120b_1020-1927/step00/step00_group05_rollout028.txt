    target_sum = n / 2.0
    # Helper function to evaluate C5 (scaled)
    def get_c5(h):
        # correlation using np.correlate
        c = np.correlate(h, 1 - h, mode='full')
        return np.max(c) * dx
    # Projection onto [0,1] and sum constraint
    def project(v):

        # Use binary search for lambda: x_i = clip(v_i - lam, 0,1)

                hi = lam

       i, j = rng.integers(0, n, size=2)
       if i==j: continue

def run(seed=42,budget_s=1100,**kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    # get n
    try:
        init = np.array(initial_h_values, dtype=np.float64)
        n = init.size
    except NameError:
        n = 100
        init = None

    dx = 2.0 / n
    target_sum = n / 2.0

    def project(v):
        lo = -np.max(v) - 1.0
        hi = np.max(v) + 1.0
        for _ in range(60):
            lam = (lo + hi) * 0.5
            x = np.clip(v - lam, 0.0, 1.0)
            s = x.sum()
            if s > target_sum:
                lo = lam
            else:
                hi = lam
        return np.clip(v - (lo+hi)*0.5, 0.0, 1.0)

    def get_c5(h):
        # correlation then max times dx
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # start with sin^2 shape
    x = (np.arange(n) + 0.5) * dx   # midpoint sampling
    h = (1.0 - np.cos(np.pi * x)) * 0.5   # sin^2(π x /2)
    # ensure within bounds
    h = np.clip(h, 0.0, 1.0)
    h = project(h)

    best_h = h.copy()
    best_c5 = get_c5(best_h)

    # optionally incorporate initial_h if exists and better
    if init is not None:
        init = np.clip(init,0.0,1.0)
        init = project(init)
        c5_init = get_c5(init)
        if c5_init < best_c5:
            best_h = init.copy()
            best_c5 = c5_init

    start = time.time()
    # gradient descent loops
    it = 0
    # schedule parameters
    max_iters = 200000
    while time.time() - start < budget_s * 0.94 and it < max_iters:
        # worst shift and gradient
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        # compute shift
        s = max_idx - (n - 1)
        # shift arrays
        if s >= 0:
            shift_plus = np.concatenate((h[s:], np.zeros(s)))
            shift_minus = np.concatenate((np.zeros(s), h[:n-s]))
        else:
            ss = -s
            shift_plus = np.concatenate((np.zeros(ss), h[:n-ss]))
            shift_minus = np.concatenate((h[ss:], np.zeros(ss)))
        grad = 1.0 - shift_plus - shift_minus
        # step size
        eta = 0.3 / math.sqrt(it + 1.0)
        h_new = h - eta * grad
        h_new = project(h_new)
        c5_new = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
        if c5_new < best_c5 - 1e-8:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        it += 1
        # occasional random perturb to escape plateaus
        if it % 5000 == 0:
            # random shuffle small mass
            i, j = rng.integers(0, n, size=2)
            # small delta
            delta = min(0.05, h[i], 1.0 - h[j])
            if delta > 1e-6:
                h[i] -= delta
                h[j] += delta
                h = project(h)
    # final local refinement (pair swaps)
    for _ in range(5000):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving small mass delta
        # calculate possible delta respecting bounds
        max_delta = min(0.02, h[i], 1.0 - h[j])
        if max_delta <= 1e-8:
            continue
        delta = max_delta * rng.random()

import time, math
import numpy as np

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # --- problem size -------------------------------------------------
    try:
        init_arr = np.array(initial_h_values, dtype=np.float64)  # type: ignore
        n = init_arr.size
    except Exception:
        n = 70            # fallback if the harness does not provide it
        init_arr = None

    dx = 2.0 / n
    target_sum = n / 2.0

    # --- projection onto {0<=h<=1, sum(h)=target_sum} -----------------
    def proj(v: np.ndarray) -> np.ndarray:
        """Euclidean projection onto the box [0,1]^n intersected with a hyperplane."""
        lo = -np.max(v) - 1.0
        hi = np.max(v) + 1.0
        # binary search for lambda such that sum(clip(v-lambda)) == target_sum
        for _ in range(45):
            lam = (lo + hi) * 0.5
            x = np.clip(v - lam, 0.0, 1.0)
            s = x.sum()
            if s > target_sum:
                lo = lam
            else:
                hi = lam
        lam = (lo + hi) * 0.5
        return np.clip(v - lam, 0.0, 1.0)

    # --- evaluation ----------------------------------------------------
    def c5_val(h: np.ndarray) -> float:
        """Maximum overlap integral for a given discretised h."""
        return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

    # --- starting point ------------------------------------------------
    # 1) a smooth sine‑squared profile (integrates to 1 analytically)
    xs = (np.arange(n) + 0.5) * dx                # midpoint sampling
    h = (1.0 - np.cos(np.pi * xs)) * 0.5          # sin^2(π x/2)
    h = np.clip(h, 0.0, 1.0)
    h = proj(h)

    best_h = h.copy()
    best_c5 = c5_val(best_h)

    # 2) possibly improve with the supplied initial construction
    if init_arr is not None:
        cand = proj(np.clip(init_arr, 0.0, 1.0))
        cand_c5 = c5_val(cand)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand, cand_c5

    # ------------------------------------------------------------------
    start = time.time()
    it = 0
    max_iters = 20000                     # enough for convergence, cheap per step
    # main sub‑gradient descent on the worst shift
    while time.time() - start < budget_s * 0.94 and it < max_iters:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        s = max_idx - (n - 1)             # integer shift (in samples)

        # shifted versions of h needed for the gradient
        if s >= 0:
            shift_plus = np.concatenate((h[s:], np.zeros(s)))
            shift_minus = np.concatenate((np.zeros(s), h[:n - s]))
        else:
            ss = -s
            shift_plus = np.concatenate((np.zeros(ss), h[:n - ss]))
            shift_minus = np.concatenate((h[ss:], np.zeros(ss)))

        grad = 1.0 - shift_plus - shift_minus   # ∂/∂h of the worst overlap term
        eta = 0.25 / math.sqrt(it + 1.0)        # diminishing step size
        h_new = proj(h - eta * grad)

        cur_c5 = c5_val(h_new)
        if cur_c5 < best_c5 - 1e-9:
            best_c5, best_h = cur_c5, h_new.copy()
        h = h_new
        it += 1

        # occasional tiny random mass move to escape flat regions
        if it % 4000 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = min(0.04, h[i], 1.0 - h[j])
                if delta > 1e-9:
                    h[i] -= delta
                    h[j] += delta
                    h = proj(h)

    # ------------------------------------------------------------------
    # small local swaps (pairwise mass transfers) for polishing
    for _ in range(6000):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a tiny amount from i to j (preserving sum)
        max_delta = min(0.015, h[i], 1.0 - h[j])
        if max_delta <= 1e-9:
            continue
        delta = max_delta * rng.random()
        h_try = h.copy()
        h_try[i] -= delta
        h_try[j] += delta
        c5_try = c5_val(h_try)
        if c5_try < best_c5 - 1e-8:
            best_c5, best_h = c5_try, h_try.copy()
            h = h_try

    # final feasible projection and evaluation
    best_h = proj(best_h)
    final_c5 = c5_val(best_h)

    return best_h.tolist(), float(final_c5), n
