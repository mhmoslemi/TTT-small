def project_onto_box_and_sum(v, target_sum):
    # Clip to [0,1] first
    v_clipped = np.clip(v, 0, 1)
    # Compute sum, if sum is target, done.
    s = v_clipped.sum()
    if np.isclose(s, target_sum):
        return v_clipped
    # else need to adjust with water-filling
    # We want to find lambda such that sum(clip(v - lambda, 0, 1)) = target.
    # Since clipping after subtracting lambda for each component produce monotonic sum in lambda.
    # We can binary search lambda.
    lo = -1.0   # lower bound for lambda (increase values)
    hi = 1.0    # upper bound (decrease values)
    for _ in range(50):
        mid = (lo+hi)/2
        w = np.clip(v - mid, 0, 1)
        s = w.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return w

def optimize_h(N, h_init, target_sum, start_time, budget_s):
    h = h_init.copy()
    best_h = h.copy()
    best_c5, _ = compute_max_overlap(h)
    # pre compute dx
    dx = 2.0 / N
    max_iter = 20000
    lr0 = 1.0
    for it in range(max_iter):
        # time check
        now = time.time()
        if now - start_time > budget_s - 2:
            break
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        g = corr * dx
        # find index of max
        idx_max = np.argmax(g)
        k_star = idx_max - (N - 1)  # shift integer
        # compute gradient
        shift_plus = np.zeros(N)
        shift_minus = np.zeros(N)
        if k_star >= 0:
            shift_plus[:N - k_star] = h[k_star:]
            shift_minus[k_star:] = h[:N - k_star]
        else:
            s = -k_star
            shift_plus[s:] = h[:N - s]
            shift_minus[:N - s] = h[s:]
        grad = dx * (1 - shift_plus - shift_minus)
        # Update
        lr = lr0 * (1.0 - it / max_iter)
        h = h - lr * grad
        # Projection to [0,1] and sum constraint
        h = project_onto_box_and_sum(h, target_sum)
        # compute new C5
        c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Random restart/perturbations occasionally
        if it % 500 == 0 and it > 0:
            # add small random noise to escape plateau

Implementation details of projection function:

def project_onto_box_and_sum(v, target):
    # we need to find w = clip(v - λ, 0, 1) such that sum w = target.
    # Use binary search for λ.
    lo = -1.0
    hi = 1.0
    # Ensure lo sufficiently low to make sum > target; hi sufficiently high to make sum < target.
    # For λ very negative, v - λ is large, clipping to 1 gives all ones -> sum=N.
    # For λ very positive, v - λ is negative, clipping to 0 gives 0 -> sum=0.
    # So we can set lo=-2, hi=2 to be safe.
    for _ in range(50):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    # final w
    w = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # After binary search we may get slight deviation; correct with small adjustment:
    # adjust to meet exact sum by adding diff to interior entries (not hitting bounds) if possible.
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        # distribute diff uniformly among entries not at bounds
        mask = (w > 0) & (w < 1)
        cnt = mask.sum()
        if cnt > 0:
            w[mask] += diff / cnt
            # clip again
            w = np.clip(w, 0.0, 1.0)
    return w

import numpy as np
import time
import math

def _project_box_sum(v, target):
    """Project v onto {0<=x<=1, sum(x)=target} using a simple binary search on a shift."""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if rounding left a deficit/excess
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        mask = (w > 0.0) & (w < 1.0)
        cnt = mask.sum()
        if cnt:
            w[mask] += diff / cnt
            w = np.clip(w, 0.0, 1.0)
    return w

def _c5_and_idx(h):
    """Return (max overlap, index of that lag) for a given step vector h."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    idx = int(np.argmax(g))
    return float(g[idx]), idx

def _gradient_max_shift(h, idx_max):
    """Gradient (sub‑gradient) of the max‑overlap term w.r.t. h."""
    N = h.shape[0]
    dx = 2.0 / N
    k = idx_max - (N - 1)          # integer shift
    grad = np.zeros(N, dtype=np.float64)

    if k >= 0:
        # i where i+k is inside the vector
        i1 = np.arange(N - k)
        grad[i1] += dx * (1.0 - h[i1 + k])
        # i where i-k is inside the vector
        i2 = np.arange(k, N)
        grad[i2] -= dx * h[i2 - k]
    else:
        k2 = -k
        i1 = np.arange(k2, N)
        grad[i1] += dx * (1.0 - h[i1 - k2])
        i2 = np.arange(N - k2)
        grad[i2] -= dx * h[i2 + k2]

    return grad

def _hill_climb(h, budget, start):
    """Simple simulated‑annealing style pairwise‑swap search."""
    N = h.shape[0]
    target = N / 2
    best_h = h.copy()
    best_val = _c5_and_idx(best_h)[0]
    cur_h = h.copy()
    cur_val = best_val
    it = 0
    max_it = 200_000  # generous upper bound; loop will break on time
    while it < max_it and time.time() - start < budget - 0.1:
        it += 1
        # temperature and move size schedule
        T = 0.02 * (1.0 - it / max_it)
        eps = 0.04 * (1.0 - it / max_it)  # max delta
        i, j = np.random.choice(N, 2, replace=False)
        delta = np.random.uniform(-eps, eps)
        hi, hj = cur_h[i] + delta, cur_h[j] - delta
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            new_h = cur_h.copy()
            new_h[i], new_h[j] = hi, hj
            val = _c5_and_idx(new_h)[0]
            improve = val < cur_val - 1e-12
            if improve or np.random.rand() < math.exp(-(val - cur_val) / (T + 1e-12)):
                cur_h, cur_val = new_h, val
                if val < best_val - 1e-12:
                    best_h, best_val = new_h.copy(), val
    return best_h, best_val

def _optimize(h, budget, start):
    """Gradient‑descent phases followed by a hill‑climb."""
    N = h.shape[0]
    target = N / 2
    best_h = h.copy()
    best_val = _c5_and_idx(best_h)[0]

    # ----- gradient descent on the max‑overlap subgradient -----
    max_gd_iter = 8_000
    for it in range(max_gd_iter):
        if time.time() - start > budget - 5.0:
            break
        val, idx = _c5_and_idx(h)
        if val < best_val - 1e-12:
            best_h, best_val = h.copy(), val
        grad = _gradient_max_shift(h, idx)
        # simple diminishing step size
        lr = 1.5 * (1.0 - it / max_gd_iter)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _project_box_sum(h_new, target)
        h = h_new

    # ----- final random‑swap hill climbing -----
    h, val = _hill_climb(h, budget, start)
    if val < best_val:
        best_h, best_val = h, val
    return best_h, best_val
