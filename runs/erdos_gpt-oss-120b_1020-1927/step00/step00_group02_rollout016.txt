    v = np.zeros(n)

        val = rng.random() * max_val

        remaining -= val

                break

                break

import time
import numpy as np

def _c5_value(h, dx):
    """Compute the C5 value for a given step vector."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_box_sum(v, target):
    """
    Project a vector onto the intersection of the box [0,1]^n
    and the hyperplane sum(v) = target using a simple bisection on the
    Lagrange multiplier.
    """
    lo = -1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        h = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] += diff / np.count_nonzero(free)
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback: adjust a random free entry
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def _random_binary_restart(n, target, rng):
    """Return a random binary (or one fractional) feasible vector."""
    full = int(np.floor(target))
    frac = target - full
    h = np.zeros(n, dtype=float)
    perm = rng.permutation(n)
    h[perm[:full]] = 1.0
    if frac > 1e-12:
        h[perm[full]] = frac
    return h

def _gradient_step(h, cur_val, dx, rng, step_factor):
    """Attempt a sub‑gradient step that reduces the worst shift."""
    n = h.size
    # worst shift
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    lag = max_idx - (n - 1)          # integer lag (may be negative)

    # compute h[i+lag] and h[i-lag] with zero padding
    if lag >= 0:
        h_fwd = np.empty(n)
        h_fwd[: n - lag] = h[lag:]
        h_fwd[n - lag :] = 0.0
        h_bwd = np.empty(n)
        h_bwd[lag:] = h[: n - lag]
        h_bwd[: lag] = 0.0
    else:
        k = -lag
        h_fwd = np.empty(n)
        h_fwd[k:] = h[: n - k]
        h_fwd[: k] = 0.0
        h_bwd = np.empty(n)
        h_bwd[: n - k] = h[k:]
        h_bwd[n - k :] = 0.0

    grad = 1.0 - (h_fwd + h_bwd)           # ∇ g(lag)
    d = -grad + np.mean(grad)              # zero‑sum direction

    # feasible scaling
    pos = d > 1e-12
    neg = d < -1e-12
    max_step = np.inf
    if np.any(pos):
        max_step = min(max_step, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        max_step = min(max_step, np.min((-h[neg]) / d[neg]))
    if not np.isfinite(max_step) or max_step <= 0.0:
        return h, cur_val, False

    # back‑track line search
    for _ in range(12):
        step = step_factor * max_step
        nh = h + step * d
        val = _c5_value(nh, dx)
        if val < cur_val - 1e-12:
            return nh, val, True
        step_factor *= 0.5
    return h, cur_val, False

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals() and initial_h_values is not None:
        base = np.asarray(initial_h_values, dtype=float)
        n0 = base.size
        # use double resolution when possible
        n = n0 * 2
        h = np.repeat(base, 2)
    else:
        n = 45
        # simple uniform start (feasible because sum = n/2)
        h = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0
    # tiny projection just in case
    h = _project_box_sum(h, target_sum)

    dx = 2.0 / n

    cur_h = h.copy()
    cur_val = _c5_value(cur_h, dx)

    best_h = cur_h.copy()
    best_val = cur_val

    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    # ----- optimisation parameters ------------------------------------
    prob_grad = 0.6                # chance of performing a gradient step
    step_factor = 1.0              # start with full feasible step
    max_iter = 10_000_000
    no_improve = 0
    restart_after = 200_000        # iterations without global improvement

    # ----- main loop --------------------------------------------------
    for it in range(max_iter):
        if elapsed() > budget_s * 0.95:
            break

        if rng.random() < prob_grad:
            # gradient descent move
            new_h, new_val, improved = _gradient_step(
                cur_h, cur_val, dx, rng, step_factor
            )
        else:
            # random two‑coordinate mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            if rng.random() < 0.5:
                max_del = min(cur_h[i], 1.0 - cur_h[j])
                if max_del <= 0.0:
                    continue
                delta = rng.random() * max_del
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
            else:
                max_del = min(cur_h[j], 1.0 - cur_h[i])
                if max_del <= 0.0:
                    continue
                delta = rng.random() * max_del
                new_h = cur_h.copy()
                new_h[i] += delta
                new_h[j] -= delta
            new_val = _c5_value(new_h, dx)
            improved = new_val < cur_val - 1e-12

        # accept if improvement (strict)
        if improved:
            cur_h = new_h
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_h = new_h.copy()
                best_val = new_val
        else:
            no_improve += 1

        # occasional restart to escape stagnation
        if no_improve >= restart_after:
            cur_h = _random_binary_restart(n, target_sum, rng)
            cur_val = _c5_value(cur_h, dx)
            no_improve = 0

    # final safety projection (maintains feasibility)
    best_h = _project_box_sum(best_h, target_sum)
    best_val = _c5_value(best_h, dx)

    return best_h.tolist(), float(best_val), int(n)
