def project_to_constraints(h, target):
    # water-filling
    # While not close enough:
    h = np.clip(h, 0., 1.)
    s = np.sum(h)
    # tolerance small.
    for _ in range(10):
        if abs(s - target) < 1e-9:
            break
        if s < target:
            # need to add mass
            free = (h < 1 - 1e-12)
            if not np.any(free):
                break
            increment = (target - s) / np.sum(free)
            h[free] += increment
        else:
            free = (h > 1e-12)
            if not np.any(free):
                break
            decrement = (s - target) / np.sum(free)
            h[free] -= decrement
        h = np.clip(h, 0., 1.)
        s = np.sum(h)
    return h

import time
import numpy as np

def _compute_c5(h, dx):
    """C5 = max_k ∫ h(x)(1‑h(x+k)) dx  (discrete version)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_to_constraints(h, target_sum, max_iter=20):
    """Clamp to [0,1] and enforce ∑h = target_sum (exactly)."""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    for _ in range(max_iter):
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            free = h < 1.0
            if not np.any(free):
                break
            inc = diff / np.sum(free)
            h[free] += inc
        else:
            free = h > 0.0
            if not np.any(free):
                break
            dec = -diff / np.sum(free)
            h[free] -= dec
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
    return h

def _gradient_for_shift(h, shift):
    """Sub‑gradient of C(shift)=∑h[i](1‑h[i+shift]) w.r.t. h."""
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        # indices where i+shift is inside the array
        i = np.arange(n - shift)
        g[i] += (1.0 - h[i + shift])
        # indices where i‑shift is inside the array
        j = np.arange(shift, n)
        g[j] -= h[j - shift]
    else:
        d = -shift
        i = np.arange(d, n)               # i‑d >= 0
        g[i] += (1.0 - h[i - d])
        j = np.arange(0, n - d)           # i+d < n
        g[j] -= h[j + d]
    return g

        # accept if it improves (otherwise keep current)
        # --------------------------------------------------------------
        if new_val < cur_val:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
                # early exit if we already beat the target record
                if best_val <= 0.3808:
                    break
        # occasional uphill step to avoid getting stuck
        elif np.random.rand() < 0.001:
            cur_h, cur_val = new_h, new_val

        # every few thousand iterations, re‑project to wipe tiny numerical drift
        if (it & 0xFFF) == 0:
            cur_h = _project_to_constraints(cur_h, target_sum)

    # ------------------------------------------------------------------
    # 3.  Final sanity check and return.
    # ------------------------------------------------------------------
    best_h = _project_to_constraints(best_h, target_sum)
    final_val = _compute_c5(best_h, dx)

    return best_h.tolist(), float(final_val), int(n)
```
