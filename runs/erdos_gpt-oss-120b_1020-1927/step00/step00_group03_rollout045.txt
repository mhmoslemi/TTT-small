            elapsed = time.time() - start_time
            frac = elapsed / budget
            T = max(1e-6, T0 * (1.0 - frac))

            i, j = np.random.randint(0, n, size=2)
            if i != j:
                if np.random.rand() < 0.5:
                    src, tgt = i, j
                else:
                    src, tgt = j, i
                max_delta = min(max_delta_rand,
                                h[src],
                                1.0 - h[tgt])
                if max_delta > 0:
                    delta = max_delta * np.random.rand()
                    h[src] -= delta
                    h[tgt] += delta
                    new_c5 = compute_c5(h)
                    d_obj = new_c5 - cur_c5
                    if d_obj <= 0 or np.random.rand() < math.exp(-d_obj / T):
                        cur_c5 = new_c5
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                    else:
                        # revert
                        h[src] += delta
                        h[tgt] -= delta

        # occasional tiny re‑balancing to keep the sum exact
        if iteration % 5000 == 0:
            drift = h.sum() - target_sum
            if abs(drift) > 1e-12:
                # distribute the drift onto a non‑boundary entry
                non_bounds = np.where((h > 0) & (h < 1))[0]
                if non_bounds.size == 0:
                    non_bounds = np.arange(n)
                idx = int(non_bounds[0])
                h[idx] -= drift
                h[idx] = min(1.0, max(0.0, h[idx]))

        iteration += 1

    # ---- short deterministic polishing ---------------------------------
    h_polish = best_h.copy()
    cur_c5 = best_c5
    for _ in range(3000):
        corr_full = np.correlate(h_polish, 1.0 - h_polish, mode="full")
        idx_max = int(np.argmax(corr_full))
        L = idx_max - (n - 1)
        if L == 0:
            break
        if L > 0:
            src_start, src_end = 0, n - L
        else:
            src_start, src_end = -L, n
        if src_end <= src_start:
            continue
        src_idx = np.arange(src_start, src_end, dtype=int)
        tgt_idx = src_idx + L
        contrib = h_polish[src_idx] * (1.0 - h_polish[tgt_idx])
        local_max = int(np.argmax(contrib))
        src = int(src_idx[local_max])
        tgt = int(tgt_idx[local_max])
        max_delta = min(0.05, h_polish[src], 1.0 - h_polish[tgt])
        if max_delta <= 0:
            continue
        delta = max_delta
        h_polish[src] -= delta
        h_polish[tgt] += delta
        new_c5 = compute_c5(h_polish)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            best_c5 = new_c5
            best_h = h_polish.copy()
        else:
            # undo
            h_polish[src] += delta
            h_polish[tgt] -= delta

    # final evaluation (use helper if available)
    try:
        final_c5 = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_c5 = float(best_c5)

    return (best_h, final_c5, n)
