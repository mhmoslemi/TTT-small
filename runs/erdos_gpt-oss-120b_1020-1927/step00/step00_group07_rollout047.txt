def project_simplex_bounds(v, target):
    # Computes projection onto set {x: sum(x)=target, 0<=x<=1}
    # Use binary search on λ.
    n = v.shape[0]
    # choose lower bound l: ensure after subtract λ, all v_i - λ >= 1 => clip to 1 -> sum = n.
    # For λ <= min(v_i - 1), v_i - λ >= 1 => clip to 1.
    # So l = min(v) - 1
    lo = np.min(v) - 1.0
    # upper bound: λ >= max(v) => v_i - λ <= 0 => sum = 0
    hi = np.max(v)
    # binary search
    for _ in range(64):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # Due to numerical error ensure sum exactly target by tiny correction
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        # distribute diff among non-boundary variables
        free = (proj > 1e-9) & (proj < 1 - 1e-9)
        if free.any():
            proj[free] += diff / free.sum()
            # clip again
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def random_pair_move(h, target_sum, rng):

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return C5 for a given step function sampled on [0,2]."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv.max() * (2.0 / n)

def _project_onto_sum_and_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x : sum(x)=target, 0<=x<=1}.
    Uses a simple bisection on the Lagrange multiplier λ solving
        sum_i clip(v_i - λ, 0, 1) = target.
    """
    lo = np.min(v) - 1.0          # makes all v_i-λ >= 1  → sum = n
    hi = np.max(v)                # makes all v_i-λ <= 0 → sum = 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # Tiny correction for numerical drift
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 1e-9) & (proj < 1.0 - 1e-9)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _gradient_move(h: np.ndarray, target_sum: float, step_factor: float = 0.2) -> np.ndarray:
    """Move h in the direction that increases the smallest auto‑correlation."""
    n = h.shape[0]
    # shift that gives the current worst overlap (largest conv)
    conv = np.correlate(h, 1.0 - h, mode='full')
    s = conv.argmax() - (n - 1)          # signed shift
    grad = np.zeros_like(h)

    if s >= 0:
        # i where i+s < n
        grad[: n - s] += h[s:]
        grad[s:] += h[: n - s]
    else:
        s2 = -s
        grad[s2:] += h[: n - s2]
        grad[: n - s2] += h[s2:]

    # Zero‑mean to keep the sum unchanged
    grad -= grad.mean()
    max_abs = np.abs(grad).max()
    if max_abs == 0.0:
        return h
    alpha = step_factor / max_abs
    y = h + alpha * grad
    y = np.clip(y, 0.0, 1.0)
    return _project_onto_sum_and_bounds(y, target_sum)

def _pair_move(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap a small amount of mass between two coordinates."""
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    while j == i:
        j = rng.integers(0, n)
    # Random direction
    if rng.random() < 0.5:
        inc, dec = i, j
    else:
        inc, dec = j, i
    max_inc = 1.0 - h[inc]
    max_dec = h[dec]
    delta = min(max_inc, max_dec) * rng.random() * 0.5
    if delta <= 0.0:
        return h
    new_h = h.copy()
    new_h[inc] += delta
    new_h[dec] -= delta
    return new_h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to obtain a low upper bound for C5.
    Returns (h_values, c5_bound, n_points).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise h
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=np.float64)
        n = init_h.shape[0]
    except Exception:
        n = 200                     # fallback size
        init_h = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0
    h = np.clip(init_h, 0.0, 1.0)
    h = _project_onto_sum_and_bounds(h, target_sum)

    best_h = h.copy()
    best_val = _compute_c5(best_h)
    cur_h = h.copy()
    cur_val = best_val

    # ------------------------------------------------------------------
    # 2. Main optimisation loop
    # ------------------------------------------------------------------
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break
        it += 1

        # Choose move type
        if rng.random() < 0.55:
            new_h = _gradient_move(cur_h, target_sum, step_factor=0.25)
        else:
            new_h = _pair_move(cur_h, rng)

        new_val = _compute_c5(new_h)

        # Accept if improves current state
        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val - 1e-12:
                best_h, best_val = new_h.copy(), new_val
        else:
            # Simulated‑annealing style occasional acceptance
            T = max(0.003, 1.0 - elapsed / budget_s)
            prob = math.exp(-(new_val - cur_val) / (T * 0.001))
            if rng.random() < prob:
                cur_h, cur_val = new_h, new_val

        # Periodic re‑projection to repair tiny drift
        if it % 2000 == 0:
            cur_h = _project_onto_sum_and_bounds(cur_h, target_sum)

    # ------------------------------------------------------------------
    # 3. Finalise and return
    # ------------------------------------------------------------------
    best_h = _project_onto_sum_and_bounds(best_h, target_sum)
    final_c5 = _compute_c5(best_h)

    return best_h, final_c5, n
