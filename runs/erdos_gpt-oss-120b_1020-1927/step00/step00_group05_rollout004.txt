import numpy as np
import time
import random
import math

def _compute_c5(h):
    """Return max overlap integral C5 for a step function h sampled on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _worst_shift(h):
    """Return (shift, value) where shift is integer lag (negative allowed) giving max overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return shift, corr[idx]

def _mismatch_pair(h, shift):
    """
    For a given shift, find a pair (i_one, i_zero) such that
    h[i_one]==1, h[i_one+shift]==0 (or the symmetric case for negative shift).
    Returns None if no such pair exists.
    """
    n = h.shape[0]
    if shift > 0:
        pref = h[: n - shift]
        succ = h[shift:]
        mask = (pref == 1) & (succ == 0)
        idxs = np.nonzero(mask)[0]
        if idxs.size == 0:
            return None
        i_one = int(idxs[0])
        i_zero = i_one + shift
        return i_one, i_zero
    elif shift < 0:
        k = -shift
        pref = h[: n - k]
        succ = h[k:]
        mask = (pref == 0) & (succ == 1)
        idxs = np.nonzero(mask)[0]
        if idxs.size == 0:
            return None
        i_zero = int(idxs[0])
        i_one = i_zero + k
        return i_one, i_zero
    else:
        return None

def _random_swap_pair(h):
    """Pick indices (i_one, i_zero) with h[i_one]==1 and h[i_zero]==0."""
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1.0 - h)
    i_one = int(np.random.choice(ones))
    i_zero = int(np.random.choice(zeros))
    return i_one, i_zero

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] (discretised) to minimise the
    Erdős overlap constant C5. Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.shape[0]
    else:
        n = 70                     # fallback size
        init = None

    target_sum = n / 2.0          # because integral must be 1, dx = 2/n

    # Build an initial binary feasible vector
    if init is not None:
        # Choose the positions with the largest initial values to be 1
        order = np.argsort(-init)               # descending
        h_current = np.zeros(n, dtype=float)
        h_current[order[: int(target_sum)]] = 1.0
    else:
        h_current = np.zeros(n, dtype=float)
        h_current[np.random.choice(n, int(target_sum), replace=False)] = 1.0

    # Ensure exact sum (numeric safety)
    cur_sum = h_current.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        # Adjust by flipping a few entries – this should never happen for binary construction
        diff = int(round(target_sum - cur_sum))
        if diff > 0:
            zeros = np.flatnonzero(h_current == 0)
            flip = np.random.choice(zeros, diff, replace=False)
            h_current[flip] = 1.0
        elif diff < 0:
            ones = np.flatnonzero(h_current == 1)
            flip = np.random.choice(ones, -diff, replace=False)
            h_current[flip] = 0.0

    best_h = h_current.copy()
    best_c5 = _compute_c5(best_h)

    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    # Main simulated‑annealing / targeted‑swap loop
    while elapsed() < budget_s:
        # Determine temperature (linear cooling)
        prog = elapsed() / budget_s
        temperature = max(1e-8, 0.02 * (1.0 - prog))

        # Find worst shift for the current configuration
        shift, _ = _worst_shift(h_current)

        # Decide move type
        if random.random() < 0.75:
            pair = _mismatch_pair(h_current, shift)
            if pair is None:
                # fallback to a random swap
                i_one, i_zero = _random_swap_pair(h_current)
            else:
                i_one, i_zero = pair
        else:
            i_one, i_zero = _random_swap_pair(h_current)

        # Propose swap
        h_new = h_current.copy()
        h_new[i_one] = 0.0
        h_new[i_zero] = 1.0

        # Evaluate
        c_new = _compute_c5(h_new)

        # Acceptance step
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

        # Metropolis criterion
        if c_new < _compute_c5(h_current) or random.random() < math.exp(-(c_new - _compute_c5(h_current)) / temperature):
            h_current = h_new

        # Periodically re‑seed from the best solution to reinforce progress
        if random.random() < 0.01:
            h_current = best_h.copy()

    # Final safety projection (in case of tiny numerical drift)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Adjust sum exactly (binary shape makes this trivial)
    excess = best_h.sum() - target_sum
    if abs(excess) > 1e-8:

            to_flip = np.random.choice(zeros, int(round(-excess)), replace=False)
            best_h[to_flip] = 1.0

    final_c5 = _compute_c5(best_h)

    return (best_h, float(final_c5), int(n))
