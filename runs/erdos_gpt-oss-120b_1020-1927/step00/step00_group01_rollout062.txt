def project_onto_box_and_sum(a, target, tol=1e-9):
    lo = -np.max(a) - 1.0  # ensure all a_i+lambda <=0
    hi = 1.0 - np.min(a) + 1.0  # ensure all >=1
    for _ in range(100):
        mid = (lo+hi)/2
        h = np.clip(a+mid, 0, 1)
        s = h.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(a+ (lo+hi)/2, 0, 1)
    # adjust for numeric errors:
    diff = target - h.sum()
    # distribute diff among non-bound indices
    free = np.where((h>0+1e-12) & (h<1-1e-12))[0]
    if len(free) > 0:
        h[free] += diff / len(free)
        h = np.clip(h, 0, 1)
    return h

def hill_climb(h_initial, time_budget):
    n = len(h_initial)
    best_h = h_initial.copy()
    best_score = compute_c5(best_h)
    current_h = best_h.copy()
    current_score = best_score
    start = time.time()
    iter_count = 0
    T0 = 0.01 # initial temperature (maybe bigger)
    # The scale of score: typical C5 maybe 0.5; differences small; using temperature relative.
    while time.time() - start < time_budget:
        # pick random pair
        i, j = np.random.randint(0, n, size=2)
        while i == j:
            j = np.random.randint(0, n)
        # random direction: move from i -> j or j -> i
        if np.random.rand() < 0.5:
            # move from i to j
            max_delta = min(current_h[i], 1 - current_h[j])
            if max_delta <= 1e-12:
                continue
            delta = np.random.rand() * max_delta
            new_h = current_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # move from j to i
            max_delta = min(current_h[j], 1 - current_h[i])
            if max_delta <= 1e-12:
                continue
            delta = np.random.rand() * max_delta
            new_h = current_h.copy()
            new_h[j] -= delta
            new_h[i] += delta
        new_score = compute_c5(new_h)
        delta_score = new_score - current_score
        # temperature schedule
        T = max(1e-6, T0 * (1 - iter_count / 2000000))  # gradually decrease, large iteration count
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / T):
            current_h = new_h
            current_score = new_score
            if new_score < best_score:
                best_h = new_h
                best_score = new_score
        iter_count += 1

def project_to_sum_and_bounds(v, target):
    # ensure target between 0 and len(v)
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo+hi)/2
        h = np.clip(v+mid, 0.0, 1.0)
        s = h.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v + (lo+hi)/2, 0.0, 1.0)
    # adjust for small diff due to rounding
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        # distribute diff proportionally among free
        free = np.where((h > 1e-12) & (h < 1-1e-12))[0]
        if len(free)>0:
            h[free] += diff / len(free)
    return h

def hill_climb(h0, time_budget):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    start = time.time()
    it = 0
    # parameters
    T0 = 0.01
    # We'll adapt T based on iteration count
    while time.time() - start < time_budget:
        # choose two distinct indices
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        # random direction: move from i to j or opposite
        if np.random.rand() < 0.5:
            # move from i to j
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            # move from j to i
            max_delta = min(h[j], 1.0 - h[i])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta
            h_new = h.copy()
            h_new[j] -= delta
            h_new[i] += delta
        new_score = compute_c5(h_new)
        delta_score = new_score - cur_score
        # temperature schedule, using exponential decay
        T = T0 * (0.99 ** (it // 1000))
        # acceptance
        if delta_score < 0 or np.random.rand() < math.exp(-delta_score / max(T, 1e-12)):
            h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        it += 1
        # occasionally random reheat to avoid stuck too early
        if it % 50000 == 0:
            T0 = max(0.001, T0 * 0.9)
    return best_h, best_score

def project_to_sum_and_bounds(a, target, tol=1e-12, max_iter=100):
    lo = -np.max(a) - 1.0
    hi = 1.0 - np.min(a) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        h = np.clip(a + mid, 0.0, 1.0)
        s = h.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a + (lo + hi) / 2.0, 0.0, 1.0)
    # adjust for any tiny diff
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        # distribute equally among unfixed entries
        free = (h > 0 + 1e-12) & (h < 1 - 1e-12)
        n_free = free.sum()
        if n_free > 0:
            h[free] += diff / n_free
            h = np.clip(h, 0.0, 1.0)
    return h
```

def gradient_optim(h0, time_budget):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h = project_to_sum_and_bounds(h0, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    iter_count = 0
    while time.time() - start < time_budget:
        corr = np.correlate(h, h, mode='full')
        half = corr[n-1:]  # s>=0
        min_g = half.min()
        # compute C5
        cur_c5 = dx * (target - min_g)
        # compute gradient
        eps = 1e-12
        min_shifts = np.where(np.abs(half - min_g) < eps)[0]
        grad = np.zeros_like(h)
        for s in min_shifts:
            if s == 0:
                grad += 2 * h
            else:
                # overlap length = n - s
                # contributions
                grad[:n-s] += h[s:]
                grad[s:] += h[:n-s]
        # if gradient zero (all entries zero) break to avoid infinite loop
        if np.allclose(grad, 0):
            break
        # try step sizes
        improved = False
        for alpha in [0.2, 0.1, 0.05, 0.02, 0.01]:
            h_new = h + alpha * grad
            # Clip to [0,1]
            h_new = np.clip(h_new, 0.0, 1.0)
            # Project to sum
            h_new = project_to_sum_and_bounds(h_new, target)
            new_c5 = compute_c5(h_new)
            if new_c5 < best_c5 - 1e-9:
                best_c5 = new_c5
                best_h = h_new.copy()
                improved = True
            if new_c5 < cur_c5 - 1e-9:
                h = h_new
                cur_c5 = new_c5
                improved = True
                break
        if not improved:
            # stochastic perturbation
            i, j = np.random.randint(0, n, size=2)
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta > 1e-8:
                delta = max_delta * np.random.rand() * 0.5
                h2 = h.copy()
                h2[i] -= delta
                h2[j] += delta
                h2 = project_to_sum_and_bounds(h2, target)
                c5_2 = compute_c5(h2)
                if c5_2 < cur_c5:
                    h = h2
                    cur_c5 = c5_2
                    if c5_2 < best_c5:
                        best_c5 = c5_2
                        best_h = h2.copy()
        iter_count += 1
        # optional early stop if best_c5 below threshold like 0.3808
        if best_c5 <= 0.3808:
            break
    return best_h, best_c5

import math
import time
import random
import numpy as np

def _project_to_sum_and_bounds(a, target, tol=1e-12, max_iter=80):
    """Project a vector onto the box [0,1] and the hyperplane sum=target."""
    lo = -np.max(a) - 1.0
    hi = 1.0 - np.min(a) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        h = np.clip(a + mid, 0.0, 1.0)
        s = h.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > tol:
        free = (h > tol) & (h < 1.0 - tol)
        cnt = free.sum()
        if cnt:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def _c5_from_h(h):
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k))dx using the discrete formula."""
    n = h.shape[0]
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, h, mode='full')
    min_g = corr.min()
    return dx * (target - min_g)


def _anneal_opt(h0, time_budget):
    """Simulated‑annealing style optimisation of the C5 value."""
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    cur = _c5_from_h(h)
    best_h = h.copy()
    best = cur
    start = time.time()
    it = 0
    while time.time() - start < time_budget:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # direction: move mass from i -> j or j -> i
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_delta = min(h[src], 1.0 - h[dst])
        if max_delta <= 1e-14:
            continue
        # temperature schedule
        elapsed = time.time() - start
        frac = elapsed / time_budget
        T = max(1e-5, 0.2 * (1.0 - frac))
        # choose a delta scaled by temperature
        delta = max_delta * (0.05 + 0.95 * np.random.rand()) * (0.5 + 0.5 * T / 0.2)
        h_new = h.copy()
        h_new[src] -= delta
        h_new[dst] += delta
        # sum preserved, bounds respected by construction
        new = _c5_from_h(h_new)
        if new < cur or np.random.rand() < math.exp((cur - new) / T):
            h = h_new
            cur = new
            if new < best:
                best = new
                best_h = h_new.copy()
        it += 1
        if best <= 0.3808:
            break
    return best_h, best


def _optimize_for_n(n, time_budget, base_vecs):
    """Run several annealing runs for a given length n and return the best."""
    target = n / 2.0
    best_h = None
    best_val = math.inf
    share = time_budget / max(1, len(base_vecs))
    for vec in base_vecs:
        if time.time() - start_time >= budget_s - 0.5:
            break
        h0 = _project_to_sum_and_bounds(vec, target)
        h_opt, val_opt = _anneal_opt(h0, share)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt
        if best_val <= 0.3808:
            break
    return best_h, best_val

        # build a pool of start vectors for this n

        # simple piecewise linear “V’’ shape (high at edges, low in centre)

    return best_h, best_c5, best_n
