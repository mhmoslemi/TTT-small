def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum_and_bounds(v, target_sum):
    # Find lambda such that sum(clip(v+lambda, 0, 1)) = target_sum
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    w = np.clip(v + lo, 0.0, 1.0)
    diff = target_sum - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0) & (w < 1)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

import numpy as np
import time
import math

# ---------- helper utilities ----------

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_sum_and_bounds(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector `v` onto the hyperplane sum(v)=target_sum while keeping
    all entries in [0,1].  Uses a simple bisection on the offset λ where
    w = clip(v+λ,0,1).
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    w = np.clip(v + lo, 0.0, 1.0)
    diff = target_sum - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0.0) & (w < 1.0)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w


def random_feasible_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1] with the required integral."""
    v = rng.random(n)
    return project_to_sum_and_bounds(v, n / 2.0)


def init_h_for_n(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build an initial feasible step function for a given length.
    If a global `initial_h_values` exists, it is interpolated;
    otherwise a random feasible vector is produced.
    """
    if "initial_h_values" in globals() and initial_h_values is not None:
        init = np.asarray(initial_h_values, dtype=float)
        m = init.shape[0]
        if n == m:
            h = np.clip(init, 0.0, 1.0)
            return project_to_sum_and_bounds(h, n / 2.0)
        # linear interpolation to new grid
        x_old = np.linspace(0.0, 2.0, m, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, init)
        h = np.clip(h, 0.0, 1.0)
        return project_to_sum_and_bounds(h, n / 2.0)
    # no parent available – fall back to a random start
    return random_feasible_vector(n, rng)


# ---------- core optimization primitives ----------

def simulated_annealing_with_time_limit(
    h0: np.ndarray, time_budget: float, rng: np.random.Generator
) -> tuple[np.ndarray, float]:
    """
    Simple SA that preserves the integral exactly.
    Allows two move types:
        * swap two entries
        * transfer a small amount δ from i to j (preserving sum)
    """
    n = len(h0)
    cur_h = h0.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    start = time.time()
    max_iters = int(5e6)          # hard upper bound on iterations
    max_delta = 0.5               # maximal amount moved in one transfer
    T0, T_end = 0.02, 1e-5        # temperature schedule boundaries

    it = 0
    while time.time() - start < time_budget and it < max_iters:
        frac = it / max_iters
        if frac > 1.0:
            frac = 1.0
        T = T0 * ((T_end / T0) ** frac)

        # decide move type
        if rng.random() < 0.2:                     # swap move
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            new_h = cur_h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
        else:                                      # transfer move
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            lo = max(-cur_h[j], cur_h[i] - 1.0, -max_delta)
            hi = min(1.0 - cur_h[j], cur_h[i], max_delta)
            if lo >= hi:
                it += 1
                continue
            delta = rng.random() * (hi - lo) + lo
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        new_val = compute_c5(new_h)
        dE = new_val - cur_val
        if dE < 0.0 or rng.random() < math.exp(-dE / T):
            cur_h, cur_val = new_h, new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
        it += 1

    return best_h, best_val


def greedy_local_improve(h: np.ndarray, max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Deterministic local refinement:
    repeatedly lessen the contribution from the current worst shift
    by moving a fraction of mass from a high‑contribution site to a
    low‑contribution partner.
    """
    n = len(h)
    best_val = compute_c5(h)

    for _ in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)                # signed offset producing the max overlap

        # indices that actually participate in this shift
        i_start = max(0, shift)
        i_end = min(n - 1, n - 1 + shift)
        if i_start > i_end:
            break
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = i_vals - shift
        contrib = h[i_vals] * (1.0 - h[j_vals])

        # pick the pair with the largest contribution
        pos = int(np.argmax(contrib))
        i, j = int(i_vals[pos]), int(j_vals[pos])

        # try to shift a modest fraction of mass from i to j (or opposite)
        if h[i] > 1e-12 and h[j] < 1.0 - 1e-12:
            delta = 0.5 * min(h[i], 1.0 - h[j])
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        elif h[j] > 1e-12 and h[i] < 1.0 - 1e-12:
            delta = 0.5 * min(h[j], 1.0 - h[i])
            new_h = h.copy()
            new_h[j] -= delta
            new_h[i] += delta
        else:
            break

        new_val = compute_c5(new_h)
        if new_val < best_val - 1e-12:
            h, best_val = new_h, new_val
        else:
            break

    return h, best_val


# ---------- main driver ----------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize the step function for the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_all = time.time()
    total_budget = budget_s * 0.95               # keep a safety margin

    # try a couple of resolutions; smaller vectors are cheap to evaluate
    lengths = [45, 90]
    best_h = None
    best_val = float("inf")
    best_n = None

    # allocate roughly equal time per resolution
    time_per_len = total_budget / len(lengths)

    for n in lengths:
        now = time.time()
        time_left = start_all + total_budget - now
        if time_left <= 0.0:
            break
        allocated = min(time_per_len, time_left)

        # build a feasible starting vector
        h0 = init_h_for_n(n, rng)

        # run a fairly long simulated‑annealing phase
        h_opt, val_opt = simulated_annealing_with_time_limit(h0, allocated, rng)

        # polish with a short deterministic greedy sweep
        h_opt, val_opt = greedy_local_improve(h_opt, max_iters=2000)

        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()

    # fallback – should never happen, but guard against pathological cases
    if best_h is None:
        if "initial_h_values" in globals() and initial_h_values is not None:
            best_h = np.asarray(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_val = compute_c5(best_h)
