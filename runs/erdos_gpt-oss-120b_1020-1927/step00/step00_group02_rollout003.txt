    # Compute FFT

import time
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Compute the C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear correlation: sum_i h[i] * (1 - h[i+k]) for each shift k
    # np.correlate with mode='full' gives all k (including negative)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def propose_move(h: np.ndarray):
    """
    Propose a feasible move that keeps the sum of h unchanged.
    Returns (i, j, delta) meaning h[i] += delta, h[j] -= delta,
    with both new values staying in [0,1].
    """
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # delta must satisfy 0 <= h[i] + delta <= 1 and 0 <= h[j] - delta <= 1
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if lo >= hi:
        return None
    delta = np.random.uniform(lo, hi)
    return i, j, delta

def random_binary_feasible(n: int) -> np.ndarray:
    """Generate a binary (or half‑binary) feasible vector with sum = n/2."""
    h = np.zeros(n, dtype=np.float64)
    half = n // 2
    h[:half] = 1.0
    if n % 2:
        h[half] = 0.5
    np.random.shuffle(h)
    return h

def _adjust_sum_to_target(h: np.ndarray, target: float):
    """Adjust a single entry (if possible) to hit the exact target sum."""
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return
    for idx in range(h.size):
        new_val = h[idx] + diff
        if 0.0 <= new_val <= 1.0:
            h[idx] = new_val
            break

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function h, its C5 bound and the number of points.
    The algorithm starts from a provided initial construction (if any)
    and performs simulated‑annealing style local optimisation.
    """
    np.random.seed(seed)
    start = time.time()
    # reserve a tiny margin for post‑processing
    max_time = budget_s - 0.5

    # ------------------------------------------------------------------
    # initialise h
    if 'initial_h_values' in globals() and initial_h_values is not None:
        base = np.array(initial_h_values, dtype=np.float64)
        n0 = base.size
        # aim for at least ~120 points for flexibility
        desired = max(120, n0)
        factor = max(1, int(math.ceil(desired / n0)))
        h = np.repeat(base, factor)
        n = h.size
    else:
        n = 120
        h = random_binary_feasible(n)

    target_sum = n / 2.0
    _adjust_sum_to_target(h, target_sum)

    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5

    # ------------------------------------------------------------------
    # simulated‑annealing loop
    T0 = 0.02                # initial temperature
    eps = 1e-12
    it = 0
    while True:
        if time.time() - start >= max_time:
            break

        mv = propose_move(h)
        if mv is None:
            continue
        i, j, delta = mv

        # apply move
        h[i] += delta
        h[j] -= delta

        new_c5 = compute_c5(h)
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / max_time)   # linear cooling

        accept = False
        if new_c5 <= current_c5:
            accept = True
        else:
            # uphill acceptance with probability exp(-Δ/T)
            if np.random.rand() < math.exp(-(new_c5 - current_c5) / (T + eps)):
                accept = True

        if accept:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta

        it += 1
        # occasional cheap check without recomputing temperature too often
        if it % 200 == 0 and time.time() - start >= max_time:
            break

    # ------------------------------------------------------------------
    # final greedy refinement (temperature ≈ 0)
    while time.time() - start < max_time:
        mv = propose_move(best_h)
        if mv is None:
            continue
        i, j, delta = mv
        best_h[i] += delta
        best_h[j] -= delta
        new_c5 = compute_c5(best_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            best_h[i] -= delta
            best_h[j] += delta
        # occasional time check
        if time.time() - start >= max_time:
            break

    # ------------------------------------------------------------------
    # final feasibility safeguard
    _adjust_sum_to_target(best_h, best_h.size / 2.0)

    final_c5 = compute_c5(best_h)
    return best_h.tolist(), float(final_c5), best_h.size
