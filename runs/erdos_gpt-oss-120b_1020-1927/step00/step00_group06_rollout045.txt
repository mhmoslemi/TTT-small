    # Initialize population

    # include init as first
    h0 = init.copy()
    s0 = compute_c5(h0)
    pop.append(h0)
    scores.append(s0)
    for _ in range(pop_size-1):

        v = project_capped_simplex(v, target)
        pop.append(v)
        scores.append(compute_c5(v))
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx]

    # parameters
    sigma0 = 0.2
    sigma = sigma0
    decay = 0.995  # per iteration
    stagnation = 0
    max_stagnation = 200
    iter_count = 0

    while time.time() - start < budget_s * 0.95:

        order = np.argsort(scores)
        pop = [pop[i] for i in order]

        # elites

        # number to create
        to_create = pop_size - elite_size

            parent = elites[rng.integers(elite_size)]

        for i in range(random_replace):

            v = project_capped_simplex(v, target)
            new_pop[idx] = v
            new_scores[idx] = compute_c5(v)
        pop = new_pop
        scores = new_scores
        # Update best
        cur_best_idx = int(np.argmin(scores))
        cur_best_score = scores[cur_best_idx]

            best_h = pop[cur_best_idx].copy()
            stagnation = 0

        # adapt sigma (e.g., reduce if stagnation)
        if stagnation > max_stagnation:
            sigma *= 0.5
            stagnation = 0
        else:

        iter_count += 1

    # After loop, maybe fine-tune via hill climbing

import time
import numpy as np

def _project_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the set {w | 0 <= w_i <= 1, sum(w) = target}.
    Uses bisection on the shift τ such that w_i = clip(v_i - τ, 0, 1) satisfies the sum.
    """
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix tiny residual sum errors
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = np.where((w > 1e-12) & (w < 1.0 - 1e-12))[0]
        if free.size:
            w[free[0]] += diff
        else:
            w += diff / w.size
        w = np.clip(w, 0.0, 1.0)
    return w


def _c5_score(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k)) dx, discretized."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)
