def propose_move(h):

        j = rng.integers(0,n)

def gradient_for_shift(h, shift):
    n = len(h)
    g = np.empty(n)
    # compute term1: 1 - h[p+shift] for valid positions else 0
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    if shift >= 0:
        # term1 for p in [0, n-1-shift]
        term1[:n-shift] = 1 - h[shift:]
        # term2 for p in [shift, n-1] (i.e., p - shift >=0)
        term2[shift:] = h[:n-shift]
    else:

        term1[s:] = 1 - h[:n-s]
        # term2 for p in [0, n-1-s] (p - shift = p + s)
        term2[:n-s] = h[s:]
    g = term1 - term2
    return g

import time, math

def _fft_max_corr(h):
    """Return (max_corr, shift) where max_corr = max correlation
    sum_i h[i]*(1-h[i+shift]) and shift is the corresponding integer shift."""
    n = h.size
    r = 1.0 - h
    N = 1 << ((2 * n - 1).bit_length())      # next power of two
    H = np.fft.rfft(h, N)
    R = np.fft.rfft(r[::-1], N)
    conv = np.fft.irfft(H * R, N)[: 2 * n - 1]   # correlation (full mode)
    idx = int(conv.argmax())
    max_corr = conv[idx]
    shift = idx - (n - 1)                       # index → shift
    return max_corr, shift, conv

def _c5_from_corr(max_corr, n):
    """Convert raw correlation to the C5 value."""
    return max_corr * (2.0 / n)

def _gradient_worst_shift(h, shift):
    """Gradient of the worst‑shift C5 w.r.t. h (g[p] = 1 - h[p+shift] - h[p-shift])."""
    n = h.size
    g = np.zeros(n, dtype=h.dtype)
    if shift >= 0:
        s = shift
        # term1: 1 - h[p+s] for p+s < n
        g[: n - s] += 1.0 - h[s:]
        # term2: -h[p-s] for p-s >= 0
        g[s:] -= h[: n - s]
    else:
        s = -shift
        # term1 for p >= s : 1 - h[p-s]
        g[s:] += 1.0 - h[: n - s]
        # term2 for p < n-s : -h[p+s]
        g[: n - s] -= h[s:]
    return g

def _feasible_pair(g, h, rng):
    """Find i (decrease) with g[i]>0 and h[i]>0, and j (increase) with g[j]<0 and h[j]<1."""
    pos = np.where((g > 1e-12) & (h > 1e-12))[0]
    neg = np.where((g < -1e-12) & (h < 1.0 - 1e-12))[0]
    if pos.size == 0 or neg.size == 0:
        return None
    i = pos[np.argmax(g[pos])]          # largest positive gradient
    j = neg[np.argmin(g[neg])]          # most negative gradient
    eps = min(h[i], 1.0 - h[j])
    if eps <= 0:
        return None
    return i, j, eps

def _make_random_feasible(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    target = n * 0.5
    h = rng.random(n)
    h = h * (target / h.sum())
    # enforce bounds iteratively
    for _ in range(5):
        over = h > 1.0
        if not over.any():
            break
        excess = h[over] - 1.0
        total_ex = excess.sum()
        h[over] = 1.0
        under = h < 1.0
        if under.any():
            h[under] += total_ex / under.sum()
    # final tiny adjustment to hit the exact sum
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            idx = rng.choice(np.where(mask)[0])
            h[idx] += diff
            h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h

    while time.time() < deadline:
        # initialise a fresh feasible point
        cur_h = _init_from_parent(n, rng)
        cur_c5 = evaluate(cur_h)
        stagnation = 0

        # inner optimisation steps
        while time.time() < deadline:
            max_corr, shift, _ = _fft_max_corr(cur_h)
            cur_c5 = _c5_from_corr(max_corr, n)

            # gradient of the worst shift
            g = _gradient_worst_shift(cur_h, shift)

            pair = _feasible_pair(g, cur_h, rng)

            if pair is None:
                # no obvious descent direction – try a small random move
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                max_eps = min(cur_h[i], 1.0 - cur_h[j])
                if max_eps <= 0:
                    continue
                eps = max_eps * rng.random() * 0.2   # modest step
                new_h = cur_h.copy()
                new_h[i] -= eps
                new_h[j] += eps
            else:
                i, j, max_eps = pair
                # take a sizeable but not full step to keep changes smooth
                eps = max_eps * (0.6 + 0.4 * rng.random())
                new_h = cur_h.copy()
                new_h[i] -= eps
                new_h[j] += eps

            new_c5 = evaluate(new_h)

            # accept improvement or occasional uphill move
            if new_c5 < cur_c5:
                cur_h, cur_c5 = new_h, new_c5
                stagnation = 0
                if new_c5 < best_c5:
                    best_h, best_c5 = new_h.copy(), new_c5
            else:
                # uphill acceptance with a tiny temperature that cools down
                temp = 0.001 * (1.0 - (time.time() - start_time) / (budget_s * 0.96))
                if rng.random() < math.exp(-(new_c5 - cur_c5) / max(temp, 1e-8)):
                    cur_h, cur_c5 = new_h, new_c5
                stagnation += 1

            # occasional re‑normalisation to erase accumulated numerical drift
            if stagnation % 500 == 0:
                diff = (n * 0.5) - cur_h.sum()
                if abs(diff) > 1e-12:
                    mask = (cur_h > 1e-12) & (cur_h < 1.0 - 1e-12)
                    if mask.any():
                        cur_h[mask] += diff / mask.sum()
                        cur_h = np.clip(cur_h, 0.0, 1.0)

            # escape if stuck for a while
            if stagnation > 4000:
                break

    # final safety‑check/fix of the best solution
    if best_h is None:
        best_h = _make_random_feasible(n, rng)
        best_c5 = evaluate(best_h)

    # enforce exact integral constraint
    target = n * 0.5
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        mask = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        if mask.any():
            best_h[mask] += diff / mask.sum()
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            idx = rng.integers(0, n)
            best_h[idx] = np.clip(best_h[idx] + diff, 0.0, 1.0)

    best_c5 = evaluate(best_h)   # recompute after final fixing
    return best_h, best_c5, n
