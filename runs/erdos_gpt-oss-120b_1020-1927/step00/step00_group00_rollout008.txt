def local_search(initial_h, max_iter=500000, time_limit=..., start_step=0.2, end_step=0.001):
    n = len(initial_h)
    best_h = np.copy(initial_h)
    best_score = compute_c5(best_h)
    cur_h = np.copy(best_h)
    cur_score = best_score
    start_time = time.time()
    step = start_step
    # schedule step decays each iteration: step = start_step * (end_step/start_step) ** (iter/ max_iter)
    for iter in range(max_iter):
        now = time.time()
        if now - start_time > time_limit: break
        # produce random move
        i = np.random.randint(n)
        j = np.random.randint(n)
        while i == j:
            j = np.random.randint(n)

        max_transfer = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0:

        eps = np.random.uniform(0, max_transfer)

import time, math, random
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return the overlap integral C5 for a discretised step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlate h with (1-h); mode='full' gives all integer shifts
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(conv.max() * dx)


def _make_feasible(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust to satisfy the integral constraint exactly."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    # simple scaling then projection
    cur = h.sum()
    if cur > 0:
        h *= target / cur
    h = np.clip(h, 0.0, 1.0)

    # fix any residual error by distributing it greedily
    diff = target - h.sum()
    rng = np.random.default_rng()
    while abs(diff) > 1e-12:
        if diff > 0:
            cand = np.where(h < 1.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            add = min(diff, 1.0 - h[i])
            h[i] += add
        else:
            cand = np.where(h > 0.0)[0]
            if cand.size == 0:
                break
            i = rng.choice(cand)
            sub = min(-diff, h[i])
            h[i] -= sub
        diff = target - h.sum()
    return h


def _improve_h(initial_h: np.ndarray,
               time_limit: float,
               start_step: float = 0.2,
               end_step: float = 1e-4,
               max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """Local search that tries to lower C5."""
    n = initial_h.shape[0]
    target_sum = n / 2.0
    rng = np.random.default_rng()
    cur_h = initial_h.copy()
    cur_score = _c5_value(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    start = time.time()
    last_improve = 0

    for it in range(max_iter):
        # time guard
        if time.time() - start > time_limit:
            break

        # linear decay of step size and temperature
        frac = it / max_iter
        step = start_step * (1.0 - frac) + end_step * frac
        temp = step  # use same value as temperature for acceptance

        # compute the shift with maximal overlap
        conv = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        max_idx = int(conv.argmax())
        shift = max_idx - (n - 1)   # shift in index units, can be negative

        moved = False
        # try a few directed moves that attack the worst shift
        if shift != 0:
            for _ in range(5):
                if shift > 0:
                    i = rng.integers(0, n - shift)
                    j = i + shift
                else:
                    i = rng.integers(-shift, n)
                    j = i + shift
                # direction: move mass from a high‑value to a low‑value
                if cur_h[i] > 0 and cur_h[j] < 1:
                    max_transfer = min(step, cur_h[i], 1.0 - cur_h[j])
                else:
                    max_transfer = 0.0
                if max_transfer <= 1e-12:
                    # try opposite direction
                    if cur_h[j] > 0 and cur_h[i] < 1:
                        max_transfer = min(step, cur_h[j], 1.0 - cur_h[i])
                        if max_transfer > 1e-12:
                            eps = rng.random() * max_transfer
                            new_h = cur_h.copy()
                            new_h[j] -= eps
                            new_h[i] += eps
                            moved = True
                            break
                    continue
                eps = rng.random() * max_transfer
                new_h = cur_h.copy()
                new_h[i] -= eps

        if not moved:

            while j == i:

            max_transfer = min(step, cur_h[i], 1.0 - cur_h[j])

        # accept if better or with Metropolis probability
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / (temp + 1e-12)):

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h, its C5 value, and the number of grid points."""
    np.random.seed(seed)
    random.seed(seed)
    start_total = time.time()

    # --------------------------------------------------------------------
    # Obtain an initial feasible vector.
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        h0 = _make_feasible(h0)
    else:
        # fall back to a simple binary‑fraction pattern (half ones, half zeros)
        n0 = 45
        target = n0 / 2.0
        h0 = np.zeros(n0, dtype=np.float64)
        full = int(math.floor(target))
        remainder = target - full
        idx = np.random.choice(n0, full, replace=False)
        h0[idx] = 1.0
        if remainder > 1e-12:
            free = np.where(h0 == 0)[0]
            h0[np.random.choice(free)] = remainder
        h0 = _make_feasible(h0)

    # --------------------------------------------------------------------
    # Candidate grid sizes (include the original size and a few refinements)
    base_n = h0.shape[0]
    candidates = [base_n, 60, 90, 120, 180]

    best_h = h0.copy()
    best_score = _c5_value(best_h)
    best_n = base_n

    # Main optimisation loop over different resolutions
    for n in candidates:
        elapsed = time.time() - start_total
        if elapsed >= budget_s * 0.96:
            break
        remaining = budget_s - elapsed
        # ----------------------------------------------------------------
        # Build a starting vector for the chosen resolution.
        if n == base_n:
            start_h = h0.copy()
        else:
            # Linear interpolation onto a finer grid
            x_old = np.linspace(0.0, 2.0, base_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            start_h = np.interp(x_new, x_old, h0)
            start_h = _make_feasible(start_h)

        # ----------------------------------------------------------------
        # Allocate a slice of the remaining budget for this resolution.
        # Larger n gets fewer iterations (quadratic cost).
        max_iter = int(250000 * (base_n / n))
        max_iter = max(5000, max_iter)
        time_for_this = remaining * 0.5
        time_for_this = max(time_for_this, 5.0)   # guarantee a minimal window

        # ----------------------------------------------------------------
        # Run the local optimiser.
        opt_h, opt_score = _improve_h(start_h,
                                      time_limit=time_for_this,
                                      start_step=0.2,
                                      end_step=1e-4,
                                      max_iter=max_iter)

        if opt_score < best_score:
            best_h, best_score, best_n = opt_h, opt_score, n

        # Early exit if we beat the target threshold.
        if best_score < 0.3808:
            break

    # Final safety projection (should already be feasible)
    best_h = _make_feasible(best_h)
    best_score = _c5_value(best_h)

    # Return the result
    return best_h, best_score, int(best_n)
