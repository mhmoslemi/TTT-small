def grad_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    max_corr = np.max(corr)
    # compute stable softmax
    expo = np.exp(alpha * (corr - max_corr))
    w = expo / np.sum(expo)  # shape (2n-1)
    # compute term1 and term2
    grad = np.zeros(n)
    # vectorized approach: for each shift s, add contributions.
    for s_idx, s in enumerate(range(-(n-1), n)):
        weight = w[s_idx]
        # term1 contributions: to h_j portion (1 - h_{j+s}) if j+s in range
        # This adds weight * (1 - h_{j+s}) to grad_j.
        # Equivalent to adding weight * ( (1 - h) shifted by -s ) to grad.
        # In numpy, we can shift using slicing.
        if s >= 0:
            # j from 0 to n-1-s: j + s in range, so contribution for grad[0:n-s] adds weight * (1 - h[s:])
            grad[0:n-s] += weight * (1 - h[s:])
            # term2: -weight * h[j - s] for j - s in range
            # This is for j >= s: j - s from 0 to n-1-s
            grad[s:n] -= weight * h[0:n-s]
        else:
            # s negative: s = -t, t>0
            t = -s
            # term1 when j + s in range -> index j - t in range => j >= t.
            # So for j from t to n-1: contribution weight * (1 - h[j - t]) where h index j - t
            grad[t:n] += weight * (1 - h[0:n-t])
            # term2: -weight * h[j - s] = -weight * h[j + t] for j + t in range -> j <= n-1 - t.
            grad[0:n-t] -= weight * h[t:n]
    grad = grad * dx
    return grad

def smooth_grad(h, alpha, penalty):
   n = len(h); dx = 2.0 / n; S = n/2
   corr = np.correlate(h, 1 - h, mode='full')
   max_corr = np.max(corr)
   expo = np.exp(alpha * (corr - max_corr))
   w = expo / np.sum(expo)   # shape (2n-1)
   # compute gradient of max_corr approximated by weighted sum of per-shift derivatives.
   # Weighted derivative = sum_s w_s * d corr_s/dh_j.
   # We'll compute using loops as earlier but include dx scaling and weight.
   grad = np.zeros(n)
   # For each shift s and weight w_s, add contributions
   # Link to earlier loop with weight
   for idx in range(2*n-1):
       s = idx - (n-1)
       weight = w[idx]
       if weight == 0:
           continue
       if s >= 0:
           # term1: (1 - h_{j+s}) for j = 0 .. n-s-1
           grad[0:n-s] += weight * (1 - h[s:])
           # term2: -h_{j-s} for j = s .. n-1
           grad[s:n] -= weight * h[0:n-s]
       else:
           t = -s
           # term1 for j = t .. n-1
           grad[t:n] += weight * (1 - h[0:n-t])
           # term2 for j = 0 .. n-t-1
           grad[0:n-t] -= weight * h[t:n]
   # Multiply by dx
   grad = grad * dx
   # penalty gradient
   diff = np.sum(h) - S
   grad += 2.0 * penalty * diff  # since d/dh_i diff = 1
   return grad

def project_to_box_sum(v, target_sum):
    # ensure within bounds after projection.
    # using bisection for lambda.
    n = v.shape[0]
    lower = -10.0
    upper = 10.0
    for _ in range(50):
        lam = (lower + upper) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lower = lam
        else:
            upper = lam
    x = np.clip(v - (lower + upper) / 2.0, 0.0, 1.0)
    # Might still be slightly off; adjust by distributing residual among non-bound values.
    diff = target_sum - x.sum()
    if abs(diff) > 1e-8:
        # distribute to free elements
        free = (x > 0.0) & (x < 1.0)
        nfree = free.sum()
        if nfree > 0:
            x[free] += diff / nfree
            x = np.clip(x, 0.0, 1.0)
        else:
            # if no free, adjust first element (clamp)
            idx = np.argmin(x)  # smallest value
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

import time
import numpy as np

def _evaluate_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def _project_to_box_sum(v, target):
    """Project vector v onto [0,1]^n while forcing sum(v)=target."""
    n = v.shape[0]
    # Bisection on lambda for clipping
    lo, hi = -10.0, 10.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny adjustment if not exact
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 0.0) & (x < 1.0)
        nfree = free.sum()
        if nfree > 0:
            x[free] += diff / nfree
            x = np.clip(x, 0.0, 1.0)
        else:
            # adjust a random element within bounds
            ix = np.argmin(x)  # smallest entry
            x[ix] = np.clip(x[ix] + diff, 0.0, 1.0)
    return x

def _upscale_h(h_small, n_large):
    """Linear interpolation of a periodic step function."""
    n_small = h_small.shape[0]
    xs = np.linspace(0.0, 2.0, n_small, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_large, endpoint=False)
    h_large = np.interp(xt, xs, h_small, period=2.0)
    return h_large

def _gradient_submax(h, weight_mask):

    n = h.shape[0]

        else:

            grad[t:]     += w * (1.0 - h[:n - t])
            grad[:n - t] -= w * h[t:]
    # scale by dx
    return grad * (2.0 / n)

import time

def _evaluate_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def _project_to_box_sum(v, target):
    """Project v onto [0,1]^n while enforcing sum(v)=target."""
    n = v.shape[0]
    lo, hi = -10.0, 10.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 0.0) & (x < 1.0)
        nfree = free.sum()
        if nfree > 0:
            x[free] += diff / nfree
            x = np.clip(x, 0.0, 1.0)
        else:
            ix = np.argmin(x)
            x[ix] = np.clip(x[ix] + diff, 0.0, 1.0)
    return x

def _upscale_h(h_small, n_large):

    xt = np.linspace(0.0, 2.0, n_large, endpoint=False)
    return np.interp(xt, xs, h_small, period=2.0)

def _gradient_submax(h, weight_mask):
