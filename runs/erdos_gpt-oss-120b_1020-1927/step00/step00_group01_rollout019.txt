def project_to_sum(v, S):
    # Lagrangian lambda to shift and clip.
    # We need sum_i clip(v_i + lam, 0,1) = S.
    # binary search lam.
    low = -np.max(v) - 1.0 # ensure low low enough (like -2)
    high = 1 - np.min(v) + 1.0 # ensure high enough (like +2)
    # Actually we can compute more precise: lam in [-1,1].
    low = -1.0
    high = 1.0
    for _ in range(60):
        mid = (low + high) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s > S:
            high = mid
        else:
            low = mid
    return np.clip(v + (low + high) / 2.0, 0.0, 1.0)

def compute_grad_worst(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    k = idx - (N - 1)
    if k >= 0:
        L = 0
        R = N - 1 - k
    else:
        L = -k
        R = N - 1
    grad = np.zeros(N, dtype=float)
    # mask
    grad[L:R+1] = dx
    # first factor subtract
    i = np.arange(L, R+1)
    grad[i] -= dx * h[i + k]
    # second factor subtract
    j_start = L + k
    j_end = R + k
    if j_start < 0:
        j_start = 0
    if j_end > N - 1:
        j_end = N - 1
    if j_start <= j_end:
        j = np.arange(j_start, j_end + 1)

def compute_grad_worst(h, dx):
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    k = idx - (N - 1)
    # overlapping region
    if k >= 0:
        L = 0
        R = N - 1 - k
    else:
        L = -k
        R = N - 1
    grad = np.zeros(N, dtype=float)
    # overlap mask contribution
    grad[L:R+1] = dx
    # subtract contribution from product term
    i = np.arange(L, R+1)
    grad[i] -= dx * h[i + k]
    # second contribution: indices j where j - k in [L,R]
    j_start = L + k
    j_end = R + k
    if j_start < 0:
        j_start = 0
    if j_end > N - 1:
        j_end = N - 1
    if j_start <= j_end:
        j = np.arange(j_start, j_end + 1)
        grad[j] -= dx * h[j - k]
    cur_val = corr[idx] * dx
    return grad, k, cur_val

import numpy as np, time, math

def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the hyperplane sum=v_target with 0<=v_i<=1."""
    lo, hi = -1.0, 1.0               # λ bounds guarantee sum∈[0,len(v)]
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _grad_worst_shift(h: np.ndarray, dx: float):
    """Gradient of the maximal overlap term (the active shift)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    k = idx - (n - 1)

    if k >= 0:
        L, R = 0, n - 1 - k
    else:
        L, R = -k, n - 1

    grad = np.zeros_like(h)
    # contribution of the linear part (the sum of overlapping h_i)
    grad[L:R + 1] = dx

    i = np.arange(L, R + 1)
    grad[i] -= dx * h[i + k]

    j_start = L + k
    j_end = R + k
    if j_start < 0:
        j_start = 0
    if j_end > n - 1:
        j_end = n - 1
    if j_start <= j_end:
        j = np.arange(j_start, j_end + 1)
        grad[j] -= dx * h[j - k]

    cur_val = corr[idx] * dx
    return grad, k, cur_val


def _c5_value(h: np.ndarray, dx: float) -> float:
    """Compute C5 for a given h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
