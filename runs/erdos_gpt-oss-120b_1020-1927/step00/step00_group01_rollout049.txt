def optimize_h(initial_h, time_limit):
    h = initial_h.copy()
    best_h = h.copy()
    best_score = evaluate(h)
    n = len(h)
    cur_score = best_score
    t0 = 0.01
    # Usually we want temperature decreasing from t0 to near 0
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # linear schedule
        T = t0 * (1 - it / 1e7) # but we don't know iteration count
        # For simplicity keep T=0.001 maybe.
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        # Determine max move front
        max_move = min(1.0 - h[i], h[j])
        if max_move <= 0:
            continue
        # step size random fraction up to max_move, but not too small
        delta = max_move * np.random.rand() * 0.5   # scale factor
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # Compute score
        new_score = evaluate(new_h)
        dscore = new_score - cur_score
        accept = False
        if dscore < 0:
            accept = True
        else:
            # accept with probability exp(-dscore / T)
            # avoid zero temp
            if T > 1e-12:
                accept = np.random.rand() < np.exp(-dscore / T)
        if accept:
            h = new_h
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        # optionally occasionally random reset to best plus random noise
        if it % 10000 == 0:
            # small random jitter to try escape
            if np.random.rand() < 0.1:
                # Phase: perturbe best solution
                perturb = best_h.copy()
                # random pair move larger delta
                i, j = np.random.randint(n), np.random.randint(n)
                max_move = min(1.0 - perturb[i], perturb[j])
                if max_move > 0:
                    delta = max_move * np.random.rand()
                    perturb[i] += delta
                    perturb[j] -= delta
                h = perturb

def worst_shift_and_gradient(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    idx_max = np.argmax(corr)
    max_val = corr[idx_max]
    lag = idx_max - (n - 1)
    # compute gradient
    g = np.ones(n)
    if lag >= 0:
        # j+lag within range for j from 0 to n-1-lag
        g[:n-lag] -= h[lag:]   # j from 0 to n-lag-1: subtract h[j+lag]
        # j-lag: indices from lag to n-1 have j-lag >=0
        g[lag:] -= h[:n-lag]   # j>=lag: subtract h[j-lag]
    else:
        # lag negative: shift left
        pos_lag = -lag
        # j+lag = j - pos_lag, valid for j >= pos_lag
        g[pos_lag:] -= h[:n-pos_lag]  # subtract h[j+lag] = h[j - pos_lag]
        # j - lag = j + pos_lag, valid for j <= n-1-pos_lag
        g[:n-pos_lag] -= h[pos_lag:]  # subtract h[j - lag] = h[j + pos_lag]
    return lag, max_val, g

def optimize_h(h0, time_budget):
    import numpy as np, time
    n = len(h0)
    dx = 2.0 / n
    def evaluate(vec):
        corr = np.correlate(vec, 1 - vec, mode='full')
        return dx * np.max(corr)

    def worst_shift_and_grad(vec):
        n = len(vec)
        corr = np.correlate(vec, 1 - vec, mode='full')
        idx_max = int(np.argmax(corr))
        max_val = corr[idx_max]
        lag = idx_max - (n - 1)
        # compute gradient g as described
        g = np.ones(n)
        if lag > 0:
            # subtract h_i+lag for i in [0, n-lag)
            g[:n-lag] -= vec[lag:]
            # subtract h_i-lag for i in [lag, n)
            g[lag:] -= vec[:n-lag]
        elif lag < 0:
            p = -lag
            # subtract h_i+lag = h_i-p for i >= p
            g[p:] -= vec[:n-p]
            # subtract h_i-lag = h_i+p for i < n-p
            g[:n-p] -= vec[p:]
        # else lag ==0 not used, but g stays as ones (but then max correlation is zero)
        return lag, max_val, g

    best_h = h0.copy()
    best_score = evaluate(best_h)
    cur_h = h0.copy()
    cur_score = best_score

    start = time.time()
    # parameters
    step_factor = 0.5
    # for safety: also random move probability
    while time.time() - start < time_budget:

        if max_corr <= 1e-12:

import time
import numpy as np

def _evaluate_h(h):
    """Return C5 for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _worst_shift_and_grad(h):
    """
    Compute the lag (relative shift) that attains the maximal overlap,
    its correlation value, and the subgradient of that correlation
    with respect to h (ignoring the equality constraint on the sum).
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = corr[idx]
    lag = idx - (n - 1)                     # range: -(n-1)..+(n-1)

    # subgradient: g_j = 1 - h_{j+lag} - h_{j-lag}
    g = np.ones(n, dtype=float)
    if lag > 0:
        # j+lag part
        g[: n - lag] -= h[lag:]
        # j-lag part
        g[lag:] -= h[: n - lag]
    elif lag < 0:
        p = -lag
        # j+lag part (j-p)
        g[p:] -= h[: n - p]
        # j-lag part (j+p)
        g[: n - p] -= h[p:]
    # lag == 0 gives correlation zero and is never maximal, gradient stays 1
    return lag, max_val, g


def _project_to_sum_bounds(v, target):
    """Project v onto [0,1]^len(v) while enforcing sum(v)=target (water‑filling)."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny residual correction
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _optimize_h(h_start, time_budget):
    """Improves h_start using a deterministic sub‑gradient descent with occasional random jumps."""
    n = h_start.shape[0]
    best_h = h_start.copy()
    best_score = _evaluate_h(best_h)
    cur_h = h_start.copy()
    cur_score = best_score

    start = time.time()
    stagnation = 0
    while time.time() - start < time_budget:
        lag, _, grad = _worst_shift_and_grad(cur_h)

        # fast exit if the maximal correlation is already zero
        if cur_score <= 1e-14:
            break

        # indices that can receive / give mass
        inc_idx = np.where(cur_h < 1.0 - 1e-12)[0]
        dec_idx = np.where(cur_h > 1e-12)[0]
        if inc_idx.size == 0 or dec_idx.size == 0:
            break

        i = inc_idx[np.argmin(grad[inc_idx])]
        j = dec_idx[np.argmax(grad[dec_idx])]
        if i == j:
            i = np.random.choice(inc_idx)
            j = np.random.choice(dec_idx)

        max_move = min(1.0 - cur_h[i], cur_h[j])
        if max_move <= 0:
            # fallback random move
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            max_move = min(1.0 - cur_h[i], cur_h[j])
            if max_move <= 0:
                continue

        # step size shrinks as time passes
        elapsed = time.time() - start
        prog = elapsed / time_budget
        step_factor = 0.5 * (1.0 - prog) + 0.01
        delta = max_move * step_factor

        cur_h[i] += delta
        cur_h[j] -= delta
        new_score = _evaluate_h(cur_h)

        if new_score < cur_score - 1e-12:
            cur_score = new_score
            stagnation = 0
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert the move
            cur_h[i] -= delta
            cur_h[j] += delta
            stagnation += 1

        # occasional random perturbation to escape plateaus
        if stagnation > 5000:
            cur_h = best_h.copy()
            cur_score = best_score
            for _ in range(8):
                a, b = np.random.randint(n, size=2)
                if a == b:
                    continue
                max_mv = min(1.0 - cur_h[a], cur_h[b])
                if max_mv <= 0:
                    continue
                d = max_mv * np.random.rand() * 0.3
                cur_h[a] += d
                cur_h[b] -= d
            cur_score = _evaluate_h(cur_h)
            stagnation = 0

    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of sampled points."""
    np.random.seed(seed)

    # obtain the supplied construction if it exists
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        # fallback: uniform half‑density on 47 points
        h0 = np.full(47, 0.5, dtype=float)

    n0 = h0.shape[0]
    # we will also try 2× and 4× refinements (still modest size)
    n_candidates = [n0]
    for mult in (2, 4):
        n_new = n0 * mult
        if n_new <= 400:          # keep the problem small enough
            n_candidates.append(n_new)

    total_start = time.time()
    best_overall_h = None
    best_overall_score = float('inf')
    best_n = None

    for idx, n in enumerate(n_candidates):
        elapsed = time.time() - total_start
        time_left = budget_s - elapsed
        if time_left <= 0:
            break
        # split remaining time evenly among the still‑to‑process sizes
        remaining = len(n_candidates) - idx
        time_for_this = time_left / remaining

        if n == n0:
            start_h = h0.copy()
        else:
            factor = n // n0
            start_h = np.repeat(h0, factor)

        # tiny numerical drifts may appear after repetition
        target_sum = n / 2.0
        if abs(start_h.sum() - target_sum) > 1e-9:
            start_h = _project_to_sum_bounds(start_h, target_sum)

        refined_h, refined_score = _optimize_h(start_h, time_for_this)

        if refined_score < best_overall_score:
            best_overall_score = refined_score
            best_overall_h = refined_h
            best_n = n

    # Safety fallback
    if best_overall_h is None:
        best_overall_h = h0.copy()
        best_overall_score = _evaluate_h(best_overall_h)
        best_n = n0

    return best_overall_h, float(best_overall_score), int(best_n)
