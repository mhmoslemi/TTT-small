def project_box_sum(h, target_sum):
    # Project onto [0,1] and sum = target_sum (Euclidean projection)
    # Use simple heuristic: iterative adjust mean offset and clip
    for _ in range(10):

def gradient_step(h, eta):

    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()
    # find all lags achieving max (within tolerance)
    tol = 1e-12

    grad = np.zeros_like(h)

```
def project_box_sum(x, target):
    # Euclidean projection onto intersection of [0,1]^n and sum(x)=target.
    # Simple iterative water-filling.
    n = x.shape[0]
    # initial clip
    np.clip(x, 0, 1, out=x)
    for _ in range(10):
        cur_sum = x.sum()
        diff = target - cur_sum
        if abs(diff) < 1e-12:
            break
        # indices that can move in direction of diff
        if diff > 0:
            free = np.where(x < 1 - 1e-12)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            # Limit inc to max allowed for each free index
            max_inc = 1 - x[free]
            inc = min(inc, max_inc.max())
            # might overshoot for some; safe approach:
            # add inc, then clip again
            x[free] += inc
        else:
            free = np.where(x > 1e-12)[0]
            if free.size == 0:
                break
            dec = diff / free.size  # negative
            max_dec = -x[free]
            dec = max(dec, max_dec.min())
            x[free] += dec
        np.clip(x, 0, 1, out=x)
    # Final small correction via single entry adjustment
    cur_sum = x.sum()
    diff = target - cur_sum
    if abs(diff) > 1e-12:
        # adjust a free index if possible
        idx = np.argmin(np.abs(x - 0.5))
        x[idx] = min(1.0, max(0.0, x[idx] + diff))
    return x
```

def max_corr(h: np.ndarray) -> float:
    """Raw maximum of the correlation ∫h(x)(1−h(x+k))dx (without dx factor)."""
    return np.max(np.correlate(h, 1 - h, mode='full'))

def compute_c5(h: np.ndarray) -> float:
    """C5 with proper dx factor."""
    n = h.shape[0]
    return max_corr(h) * (2.0 / n)

def shift_array(arr: np.ndarray, lag: int) -> np.ndarray:
    """Shift arr by lag positions, filling vacated entries with 0.
    Positive lag shifts to the right."""
    n = arr.shape[0]
    if lag > 0:
        out = np.empty_like(arr)
        out[:lag] = 0.0
        out[lag:] = arr[:n - lag]
    elif lag < 0:
        k = -lag
        out = np.empty_like(arr)
        out[:n - k] = arr[k:]
        out[n - k:] = 0.0
    else:
        out = arr.copy()
    return out

def project_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of x onto [0,1]^n ∩ {sum = target}."""
    n = x.size
    np.clip(x, 0.0, 1.0, out=x)
    # simple iterative water‑filling
    for _ in range(10):
        cur = x.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        x += diff / n
        np.clip(x, 0.0, 1.0, out=x)
    # final tiny correction on a non‑boundary entry
    cur = x.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # pick an element not at a bound if possible
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size == 0:
            free = np.arange(n)
        i = free[0]
        x[i] = min(1.0, max(0.0, x[i] + diff))
    return x

def upsample(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a finer grid, followed by projection."""
    old_n = h.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, h)
    return project_box_sum(new_h, new_n / 2.0)

# ----- core optimizer -----------------------------------------------------

def subgradient_optimize(h0: np.ndarray, time_budget: float):
    """Subgradient descent on max_corr with simple random restarts."""
    start = time.time()
    n = h0.shape[0]
    best_h = h0.copy()
    best_raw = max_corr(best_h)
    cur_h = best_h.copy()
    cur_raw = best_raw
    no_improve = 0
    step0, step_min = 0.5, 0.001

    while True:
        if time.time() - start >= time_budget:
            break
        # decreasing step size
        elapsed = time.time() - start
        eta = step0 * (1.0 - elapsed / time_budget) + step_min * (elapsed / time_budget)

        # current correlation and maximal lag
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_val = corr.max()
        idxs = np.where(np.abs(corr - max_val) < 1e-12)[0]
        idx = np.random.choice(idxs)
        L = idx - (n - 1)

        # subgradient for the active lag
        fwd = shift_array(cur_h, L)
        bwd = shift_array(cur_h, -L)
        grad = 1.0 - fwd - bwd

        # gradient step + projection
        new_h = cur_h - eta * grad
        new_h = project_box_sum(new_h, n / 2.0)
        new_raw = max_corr(new_h)

        # acceptance
        if new_raw <= cur_raw:
            cur_h, cur_raw = new_h, new_raw
            if new_raw < best_raw - 1e-12:
                best_h, best_raw = new_h.copy(), new_raw
                no_improve = 0
            else:
                no_improve += 1
        else:
            # occasional uphill move (simulated‑annealing style)
            prob = math.exp(-(new_raw - cur_raw) / (eta + 1e-12))
            if np.random.rand() < prob:
                cur_h, cur_raw = new_h, new_raw
            no_improve += 1

        # random perturbation if stuck
        if no_improve > 2000:
            jitter = np.random.randn(n) * (eta * 0.2)
            cur_h = best_h + jitter
            cur_h = project_box_sum(cur_h, n / 2.0)
            cur_raw = max_corr(cur_h)
            no_improve = 0

    return best_h, best_raw

# ----- entry point --------------------------------------------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function h with a low Erdős C₅ bound."""
    np.random.seed(seed)
    start_total = time.time()

    # --- initial vector --------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 45
        h = np.random.rand(n0)
        h = h / h.sum() * (n0 / 2.0)
    h = project_box_sum(h, len(h) / 2.0)

    # --- phase 1: optimise at current resolution -------------------------
    elapsed = time.time() - start_total
    time_left = budget_s - elapsed
    phase1_time = min(time_left * 0.4, 300.0)
    best_h, best_raw = subgradient_optimize(h, phase1_time)

    # --- phase 2: upsample (if time permits) ----------------------------
    elapsed = time.time() - start_total
    time_left = budget_s - elapsed
    if time_left > 30.0 and len(best_h) * 2 <= 300:
        h_fine = upsample(best_h, len(best_h) * 2)
        best_h2, best_raw2 = subgradient_optimize(h_fine, time_left)
        if best_raw2 < best_raw:
            best_h, best_raw = best_h2, best_raw2

    # --- final result ----------------------------------------------------
    final_c5 = best_raw * (2.0 / len(best_h))
    return (best_h, float(final_c5), len(best_h))
