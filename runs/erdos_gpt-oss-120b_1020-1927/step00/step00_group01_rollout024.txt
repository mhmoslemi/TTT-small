```
def sturmian(n, alpha, threshold=0.5):
    # generate points x_i = fractional part(i * alpha)
    # assign 1 if x_i < threshold else 0
    fracs = (np.arange(n) * alpha) % 1.0
    h = (fracs < threshold).astype(float)
    # adjust to exact n/2 ones
    current = int(h.sum())
    target = n // 2
    diff = current - target
    if diff > 0:
        # too many ones: turn off some random ones
        idx = np.where(h == 1)[0]
        turn_off = np.random.choice(idx, diff, replace=False)
        h[turn_off] = 0
    elif diff < 0:
        idx = np.where(h == 0)[0]
        turn_on = np.random.choice(idx, -diff, replace=False)
        h[turn_on] = 1
    return h
```

def run(seed=42, budget_s=1100, **kwargs):

    import numpy as np, time, math, random, cvxpy as cp

    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    best_h = None
    best_val = float('inf')
    best_n = None

    # Choose a list of n values to try (even)
    candidate_ns = [80, 100, 120]  # maybe

    for n in candidate_ns:
        # ensure time
        if time.time() - start_time > budget_s * 0.9:
            break

        w = n // 2
        dx = 2.0 / n

        # Build MILP
        h = cp.Variable(n, boolean=True)
        t = cp.Variable()
        constraints = [cp.sum(h) == w, t >= 0]

        # For each lag
        for d in range(1, n):
            nd = n - d
            # product variables
            W = cp.Variable(nd, boolean=True)
            # linking
            constraints += [W <= h[:nd]]
            constraints += [W <= h[d:]]
            constraints += [W >= h[:nd] + h[d:] - 1]

            I = cp.sum(W)
            S_fwd = cp.sum(h[:nd])
            S_rev = cp.sum(h[d:])
            constraints += [dx * (S_fwd - I) <= t]
            constraints += [dx * (S_rev - I) <= t]

        prob = cp.Problem(cp.Minimize(t), constraints)

        # solve with time limit
        remaining = max(budget_s - (time.time() - start_time), 1.0)
        try:
            prob.solve(solver=cp.CBC, max_seconds=remaining, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.GLPK_MI, max_seconds=remaining, verbose=False)
            except Exception:
                # fallback to heuristic
                break

        if prob.status not in ["optimal", "optimal_inaccurate"]:
            # maybe solver timed out or not optimal
            # we can still extract solution if it exists
            # but if not feasible, continue
            continue

        h_val = np.round(np.array(h.value)).astype(int)  # binary
        # sanity: ensure sum = w
        cur_sum = int(h_val.sum())
        if cur_sum != w:
            # adjust: flip some bits
            diff = cur_sum - w
            if diff > 0:
                # turn off diff ones randomly
                ones_idx = np.where(h_val == 1)[0]
                if len(ones_idx) >= diff:
                    turn_off = np.random.choice(ones_idx, diff, replace=False)
                    h_val[turn_off] = 0
            elif diff < 0:
                zeros_idx = np.where(h_val == 0)[0]
                turn_on = np.random.choice(zeros_idx, -diff, replace=False)
                h_val[turn_on] = 1

        # compute C5 using np.correlate
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        C5 = np.max(corr) * dx

        if C5 < best_val:
            best_val = C5
            best_h = h_val.astype(float)  # convert to floats
            best_n = n

    if best_h is None:
        # fallback: simple random binary
        n = 100
        w = n // 2
        h = np.zeros(n)
        ones_idx = np.random.choice(n, w, replace=False)
        h[ones_idx] = 1.0
        dx = 2.0 / n
        C5 = np.max(np.correlate(h, 1-h, mode='full')) * dx
        best_h = h
        best_val = C5
        best_n = n

    return best_h.tolist(), float(best_val), int(best_n)

import time, random, math
import numpy as np
import cvxpy as cp

def _c5_value(h):
    """compute the C5 value for a binary (or [0,1]) vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Find a step function h on [0,2] (discretised) that minimises
    max_k ∫ h(x)(1‑h(x+k))dx  (the Erdős C5 constant).

    Returns
    -------
    h_values : list of float  (length = n_points)
    c5_bound : float           (the achieved C5 value)
    n_points : int
    """
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    best_h = None
    best_val = float("inf")
    best_n = None

    # try a few even grid sizes; larger grids give finer bounds but are slower.
    candidate_ns = [80, 100, 120]

    for n in candidate_ns:
        # stop early if time is almost exhausted
        if time.time() - start_time > budget_s * 0.95:
            break

        # we need exactly n/2 ones (integral of h = 1)
        w = n // 2
        dx = 2.0 / n

        # ----- build the mixed‑integer linear program -----
        h = cp.Variable(n, boolean=True)      # indicator of the set A
        t = cp.Variable()                     # upper bound on the overlap
        constraints = [cp.sum(h) == w, t >= 0]

        # for each possible shift d (positive and negative)
        for d in range(1, n):
            nd = n - d                         # size of the overlapping region
            # auxiliary binary variables for the product h_i * h_{i+d}
            W = cp.Variable(nd, boolean=True)

            # linearisation of the product
            constraints += [W <= h[:nd]]
            constraints += [W <= h[d:]]
            constraints += [W >= h[:nd] + h[d:] - 1]

            I = cp.sum(W)                      # number of overlapping 1‑1 pairs
            S_fwd = cp.sum(h[:nd])            # ones in the left part of the overlap
            S_rev = cp.sum(h[d:])              # ones in the right part of the overlap

            # both directions must respect the bound t
            constraints += [dx * (S_fwd - I) <= t]
            constraints += [dx * (S_rev - I) <= t]

        prob = cp.Problem(cp.Minimize(t), constraints)

        # solve with a mixed‑integer solver; CBC is the default choice
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            # fall back to GLPK if CBC is unavailable
            try:
                prob.solve(solver=cp.GLPK_MI, verbose=False)
            except Exception:
                # if no MIP solver works, give up on this n
                continue

        # if solver did not return a feasible solution, skip this n
        if prob.status not in ("optimal", "optimal_inaccurate"):
            continue

        # retrieve a (near‑)binary solution and force exact integrality
        h_val = np.rint(np.asarray(h.value).flatten()).astype(int)
        cur_sum = int(h_val.sum())
        # correct possible rounding drift
        if cur_sum != w:
            diff = cur_sum - w
            if diff > 0:
                ones = np.where(h_val == 1)[0]
                turn_off = np.random.choice(ones, diff, replace=False)
                h_val[turn_off] = 0
            else:
                zeros = np.where(h_val == 0)[0]
                turn_on = np.random.choice(zeros, -diff, replace=False)
                h_val[turn_on] = 1

        # evaluate the actual C5 value (the same metric the scorer uses)
        cur_val = _c5_value(h_val)

        if cur_val < best_val:
            best_val = cur_val
            best_h = h_val.astype(float)
            best_n = n

        # stop if we already beat the target record
        if best_val <= 0.3808:
            break

    # ----- fallback: a simple random feasible vector if optimisation failed -----
    if best_h is None:
        n = 100
        w = n // 2
        h = np.zeros(n, dtype=float)
        idx = np.random.choice(n, w, replace=False)
        h[idx] = 1.0
        best_h = h
        best_val = _c5_value(best_h)
        best_n = n

    # final sanity check: ensure integral constraint holds exactly
    # (tiny floating‑point errors are harmless for the scorer)
    assert abs(best_h.sum() * (2.0 / best_n) - 1.0) < 1e-12
    assert np.all((best_h >= 0) & (best_h <= 1))

    return best_h.tolist(), float(best_val), int(best_n)
