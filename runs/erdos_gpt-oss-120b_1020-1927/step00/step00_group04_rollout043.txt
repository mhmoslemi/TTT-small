def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    n = 200
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * x / 2.0) ** 2

    # ensure sum close to n/2, small adjust:
    current_sum = h.sum()
    target = n / 2.0
    if abs(current_sum - target) > 1e-12:
        # scale to match
        factor = target / current_sum
        h *= factor
        # clip to [0,1]
        np.clip(h, 0, 1, out=h)

        # After clipping, adjust sum again by distributing diff uniformly to non-saturated values
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # find indices not at bounds
            mask = (h > 1e-12) & (h < 1 - 1e-12)
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += diff / cnt
                np.clip(h, 0, 1, out=h)

    # compute c5 bound
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx

    return (h, float(c5), n)

import time
import numpy as np

def _compute_c5(h, dx):
    """Return the C5 value for a given step function h."""
    # correlation of h with (1-h), full mode, then max and scale by dx
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Build a simple two‑level step function (edges at height 1, interior at a constant
    level) and scan the edge length to minimise the Erdős C5 value.
    """
    np.random.seed(seed)
    start = time.time()

    # number of discretisation points (keep moderate for speed)
    n = 200                     # even, so that the integral constraint is exact
    dx = 2.0 / n
    target_sum = n / 2.0        # ∑h_i must equal this (integral = 1)

    best_c5 = np.inf
    best_h = None
    best_n_edge = None

    # Scan all possible symmetric edge lengths (total number of points with value 1)
    for n_edge in range(0, n // 2 + 1):
        # interior value t is forced by the integral constraint
        if n_edge == n:
            continue
        t = (target_sum - n_edge) / (n - n_edge) if (n - n_edge) != 0 else 0.0
        if t < 0.0 or t > 1.0:
            continue

        # construct h: left edge, interior, right edge
        h = np.full(n, t, dtype=float)
        left = n_edge // 2
        right = n_edge - left
        if left:
            h[:left] = 1.0
        if right:
            h[-right:] = 1.0

        # compute C5
        c5 = _compute_c5(h, dx)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n_edge = n_edge

        # time guard
        if time.time() - start > budget_s * 0.9:
            break

    # If the simple family did not improve over the initial, fall back to a smooth
    # cosine shape (which is guaranteed to satisfy the constraints).
    if best_h is None:
        x = (np.arange(n) + 0.5) * dx
        h = np.sin(np.pi * x / 2.0) ** 2
        best_h = h
        best_c5 = _compute_c5(best_h, dx)

    # final check that the constraints are satisfied (tiny numerical tolerance)
    total = best_h.sum() * dx
    if not np.all((best_h >= -1e-12) & (best_h <= 1.0 + 1e-12)):
        # clip just in case of rounding errors
        np.clip(best_h, 0.0, 1.0, out=best_h)
        total = best_h.sum() * dx
    if abs(total - 1.0) > 1e-9:
        # adjust uniformly to enforce the integral exactly
        diff = 1.0 - total
        mask = (best_h > 0.0) & (best_h < 1.0)
        cnt = mask.sum()
        if cnt > 0:
            best_h[mask] += diff / cnt
            np.clip(best_h, 0.0, 1.0, out=best_h)

    # recompute C5 after any possible adjustment
    best_c5 = _compute_c5(best_h, dx)

    return best_h, best_c5, n
