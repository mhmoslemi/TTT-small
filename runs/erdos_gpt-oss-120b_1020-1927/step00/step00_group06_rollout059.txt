def project_to_sum_and_box(v, target):
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) <= 1e-9:
        return v
    # iterative fix
    # If total > target -> need to subtract
    # else need to add
    # We'll use simple water-filling algorithm
    if total > target:
        # reduce
        excess = total - target
        # we want to subtract from positive entries proportionally until we reach target, but not below 0.
        # Use sorting descending of v to know when some values become zero.
        # Equivalent to projecting onto simplex with upper bound 1.
        # Let's implement using numpy sorting algorithm.
        # This problem is same as solve min ||w-v|| s.t. 0<=w<=1, sum w = target.
        # that's a bounded simplex projection. Known solution: find lambda s.t. w_i = min(max(v_i - lambda, 0), 1) and sum w_i = target.
        # We'll compute lambda using bisection.
        lo = -1.0
        hi = 1.0
        for _ in range(40):
            lam = (lo + hi) / 2.0
            w = np.clip(v - lam, 0.0, 1.0)
            s = w.sum()
            if s > target:
                lo = lam
            else:
                hi = lam
        return w
    else:
        # total < target, need to increase

                hi = lam

def get_candidate_pairs(h, s):
    n = h.shape[0]
    if s >=0:
        i_range = np.arange(n - s)
        i_cond = (h[i_range] == 1) & (h[i_range + s] == 0)
        j_cond = (h[i_range] == 0) & (h[i_range + s] == 1)
        i_candidates = i_range[i_cond]
        j_candidates = i_range[j_cond]
    else:
        s = -s
        i_range = np.arange(s, n)
        i_cond = (h[i_range] == 1) & (h[i_range - s] == 0)
        j_cond = (h[i_range] == 0) & (h[i_range - s] == 1)
        i_candidates = i_range[i_cond]
        j_candidates = i_range[j_cond]
    return i_candidates, j_candidates

import time, math, random
import numpy as np

# ------------------------------------------------------------
# Helper functions (top level, no closures)
# ------------------------------------------------------------
def evaluate(h):
    """Return C5 bound for a step vector h (np.ndarray)."""
    n = h.shape[0]
    # linear (non‑circular) correlation as used by the harness
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)


def project_sum_box(v, target):
    """Project v onto the box [0,1]^n while enforcing sum(v)=target.

    Uses a simple bisection on the parameter λ in
        w_i = clip(v_i - λ, 0, 1)
    which yields a unique solution because the right‑hand side is monotone.
    """
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def random_binary(n):
    """Return a random binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    np.random.shuffle(h)
    return h


def candidate_pairs(h, shift):
    """Indices that can be swapped to improve the correlation at a given shift."""
    n = h.shape[0]
    if shift >= 0:
        rng = np.arange(n - shift)
        i_cond = (h[rng] == 1) & (h[rng + shift] == 0)
        j_cond = (h[rng] == 0) & (h[rng + shift] == 1)
    else:
        s = -shift
        rng = np.arange(s, n)
        i_cond = (h[rng] == 1) & (h[rng - s] == 0)
        j_cond = (h[rng] == 0) & (h[rng - s] == 1)
    return rng[i_cond], rng[j_cond]


def fractional_swap(h, max_delta=0.2):
    """Transfer a small amount of mass from one entry to another while keeping the sum."""
    n = h.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    # feasible positive delta (increase i, decrease j)
    pos = min(1.0 - h[i], h[j])
    # feasible negative delta (decrease i, increase j)
    neg = -min(h[i], 1.0 - h[j])
    delta = np.random.uniform(-max_delta, max_delta)
    if delta > pos:
        delta = pos
    elif delta < neg:
        delta = neg
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a low C5 bound.

    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + 0.96 * budget_s   # leave a safety margin

    # -----------------------------------------------------------------
    # Choose a problem size (even, < 1000) and an initial vector.
    # -----------------------------------------------------------------
    n = 200                     # moderate size, easy to handle
    target_sum = n / 2.0

    # Golden‑ratio (Sturmian) construction – a good deterministic start.
    phi = (math.sqrt(5.0) - 1.0) / 2.0
    pos = (np.arange(n) * phi) % 1.0
    order = np.argsort(pos)
    h = np.zeros(n, dtype=float)
    h[order[: n // 2]] = 1.0
    h = project_sum_box(h, target_sum)

    best_h = h.copy()
    best_score = evaluate(best_h)

    # -----------------------------------------------------------------
    # Simulated‑annealing / hill‑climbing loop.
    # -----------------------------------------------------------------
    cur_h = h.copy()
    cur_score = best_score
    no_improve = 0
    MAX_STUCK = 3000   # restart after this many failed attempts

    it = 0
    while time.time() < deadline:
        # temperature decays linearly to a small floor
        elapsed = time.time() - start
        T = max(0.005, 0.5 * (1.0 - elapsed / budget_s))

        # Decide which kind of move to propose
        if random.random() < 0.6:                     # emphasis on targeted swaps
            # Find the shift that currently dominates the max.
            corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            idx = int(np.argmax(corr))
            shift = idx - (n - 1)
            i_set, j_set = candidate_pairs(cur_h, shift)
            if i_set.size and j_set.size:
                i = int(np.random.choice(i_set))
                j = int(np.random.choice(j_set))
                new_h = cur_h.copy()
                new_h[i], new_h[j] = cur_h[j], cur_h[i]
            else:
                # fallback: ordinary binary swap
                ones = np.flatnonzero(cur_h == 1)
                zeros = np.flatnonzero(cur_h == 0)
                if ones.size == 0 or zeros.size == 0:
                    new_h = fractional_swap(cur_h)
                else:
                    i = int(np.random.choice(ones))
                    j = int(np.random.choice(zeros))
                    new_h = cur_h.copy()
                    new_h[i], new_h[j] = cur_h[j], cur_h[i]
        else:
            # pure fractional mass transfer
            new_h = fractional_swap(cur_h)

        # Ensure feasibility (clip + project – cheap because sum is already kept)
        new_h = np.clip(new_h, 0.0, 1.0)
        # numerical drift may break the sum; re‑project occasionally
        if abs(new_h.sum() - target_sum) > 1e-9:
            new_h = project_sum_box(new_h, target_sum)

        new_score = evaluate(new_h)

        # Acceptance test (simulated annealing)
        if new_score < cur_score or random.random() < math.exp(-(new_score - cur_score) / T):
            cur_h, cur_score = new_h, new_score
            no_improve = 0
            if new_score < best_score:
                best_h, best_score = new_h.copy(), new_score
        else:
            no_improve += 1

        # Restart if we have been stuck for a while
        if no_improve >= MAX_STUCK:
            cur_h = random_binary(n)
            cur_score = evaluate(cur_h)
            no_improve = 0

        it += 1

    # -----------------------------------------------------------------
    # Final tidy‑up: enforce constraints exactly and recompute final bound.
    # -----------------------------------------------------------------
    best_h = project_sum_box(best_h, target_sum)
    final_score = evaluate(best_h)

    return best_h, final_score, n
