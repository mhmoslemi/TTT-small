    # ensuring target between 0 and n (the sum bound for variables in [0,1]).
    low = np.min(z) - 1.0
    high = np.max(z)
    for _ in range(50):

def gradient_for_shift(h, k):
    n = len(h)
    # compute shift arrays
    if k >= 0:
        # forward shift: h[i+k]
        h_fwd = np.concatenate([h[k:], np.zeros(k)])
        # backward shift: h[i - k]; shift backward means shift all elements right by k
        h_bwd = np.concatenate([np.zeros(k), h[:n-k]])

def compute_correlation_fft(h):
    n = len(h)
    m = 2*n-1
    N = 1 << (m-1).bit_length()  # next power of two
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1 - h, N)
    # cross-correlation (full) = ifft(conj(H) * G) but thinking of reversed? Actually correlation using formula conj(FFT(a)) * FFT(b). For correlation of a with b (no reversal) is ifft(conj(FFT(a))*FFT(b)). For cross-correlation we need conj of first.
    # So we can compute:
    corr = np.fft.ifft(np.conj(H) * G).real
    # This yields circular correlation of length N. We need linear full correlation length 2n-1.
    # The linear correlation can be extracted from indices 0..2n-2 after zero-padding to length N.
    corr = corr[:2*n-1]
    return corr

def project_to_capped_simplex(z, target):
    # compute projection onto [0,1] and sum target via water-filling
    n = len(z)
    # shortcuts if already feasible within tolerance
    if abs(z.sum() - target) < 1e-12 and np.all(z >= 0) and np.all(z <= 1):
        return z
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(60):
        lam = (lo + hi) / 2.0
        p = np.clip(z - lam, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the intersection of the box [0,1]^n
# and the affine hyperplane sum(h)=target (capped simplex).
# ------------------------------------------------------------
def _project_capped_simplex(z: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of z onto {0<=x<=1, sum(x)=target}."""
    # If already feasible (within tiny tolerance) just return a copy.
    if (
        abs(z.sum() - target) < 1e-12
        and np.all(z >= -1e-12)
        and np.all(z <= 1 + 1e-12)
    ):
        return np.clip(z, 0.0, 1.0)

    lo = np.min(z) - 1.0            # ensures all entries are at most 1 after shift
    hi = np.max(z)                  # ensures all entries are at least 0 after shift

    for _ in range(60):             # enough for double precision
        lam = (lo + hi) * 0.5
        p = np.clip(z - lam, 0.0, 1.0)
        s = p.sum()
        if s > target:               # we need a larger λ to lower the sum
            lo = lam
        else:                        # sum too small → decrease λ
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(z - lam, 0.0, 1.0)


# ------------------------------------------------------------
# Helper: evaluate the Erdős C5 value for a given step function.
# ------------------------------------------------------------
def _c5_value(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = len(h)
    # np.correlate gives the linear cross‑correlation (length 2n‑1)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)      # dx = 2/n


# ------------------------------------------------------------
# Main entry point.
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --------------------------------------------------------
    # Choose discretisation size (small enough to be fast,
    # large enough to give flexibility). 200 works well.
    # --------------------------------------------------------
    n = 200
    target_sum = n / 2.0

    # --------------------------------------------------------
    # Build an initial shape.
    # Prefer the supplied parent construction if available;
    # otherwise fall back to a triangular profile.
    # --------------------------------------------------------
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
        n0 = base.size
        # Mid‑point positions for the parent and for the new grid
        xs0 = (np.arange(n0) + 0.5) * (2.0 / n0)
        xs = (np.arange(n) + 0.5) * (2.0 / n)
        h0 = np.interp(xs, xs0, base)
    else:
        xs = (np.arange(n) + 0.5) * (2.0 / n)
        h0 = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))   # triangular hat

    h0 = _project_capped_simplex(h0, target_sum)

    best_h = None
    best_val = math.inf

    # --------------------------------------------------------
    # Main optimisation loop: several restarts, sub‑gradient descent,
    # followed by a short stochastic refinement.
    # --------------------------------------------------------
    while time.time() - start < budget_s * 0.9:
        # Randomise the start a little to escape local minima
        h = h0 + rng.normal(scale=0.07, size=n)
        h = np.clip(h, 0.0, 1.0)
        h = _project_capped_simplex(h, target_sum)

        # ---------- sub‑gradient descent on the worst shift ----------
        max_iter = 2500
        for it in range(max_iter):
            # compute full correlation and locate worst shift
            corr = np.correlate(h, 1.0 - h, mode="full")
            idx = int(np.argmax(corr))
            cur_val = corr[idx] * (2.0 / n)

            # stop early if time is almost exhausted
            if time.time() - start > budget_s * 0.95:
                break

            shift = idx - (n - 1)          # lag corresponding to idx

            # gradient of the active correlation term r_k w.r.t. h
            if shift >= 0:
                fwd = np.concatenate([h[shift:], np.zeros(shift, dtype=h.dtype)])
                bwd = np.concatenate([np.zeros(shift, dtype=h.dtype), h[: n - shift]])
            else:
                s = -shift
                fwd = np.concatenate([np.zeros(s, dtype=h.dtype), h[: n - s]])
                bwd = np.concatenate([h[s:], np.zeros(s, dtype=h.dtype)])
            grad = 1.0 - fwd - bwd          # ∂r_k/∂h

            # diminishing step size (classical sub‑gradient schedule)
            eta = 0.45 / math.sqrt(it + 1.0)
            z = h - eta * grad
            h = _project_capped_simplex(z, target_sum)

        # ---------- short stochastic refinement ----------
        cur_val = _c5_value(h)
        for _ in range(1800):
            if time.time() - start > budget_s * 0.95:
                break
            i, j = rng.choice(n, size=2, replace=False)
            hi, hj = h[i], h[j]
            # allowed delta keeping both variables inside [0,1]
            delta_max = min(0.015, 1.0 - hi, hj)
            delta_min = -min(0.015, hi, 1.0 - hj)
            if delta_max <= delta_min:
                continue
            delta = rng.uniform(delta_min, delta_max)
            if abs(delta) < 1e-12:
                continue
            h_trial = h.copy()
            h_trial[i] = hi + delta
            h_trial[j] = hj - delta
            val_trial = _c5_value(h_trial)
            if val_trial < cur_val:
                h = h_trial
                cur_val = val_trial

        # Record the best solution seen so far
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            # Use this improved shape as the new warm‑start
            h0 = best_h.copy()

    # If for some reason no feasible solution was found (should not happen)
    if best_h is None:
        best_h = np.full(n, 0.5, dtype=float)
        best_h = _project_capped_simplex(best_h, target_sum)
        best_val = _c5_value(best_h)

    # Final projection to guarantee feasibility (tiny numerical drift)
    best_h = _project_capped_simplex(best_h, target_sum)

    # Compute the bound using the provided evaluator if present,
    # otherwise fall back to our own implementation.
    try:
        c5_bound = float(evaluate_erdos_solution(best_h))
    except Exception:
        c5_bound = float(_c5_value(best_h))

    return best_h, c5_bound, n
