import time
import numpy as np

def _c5_eval(h):
    """Compute the overlap constant for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full").max() * dx


def _gradient_sets(h, shift):
    """Return indices where increasing (g<0) or decreasing (g>0) h reduces the
    current worst overlap (shift = argmax convolution)."""
    n = h.shape[0]
    g = np.full(n, 1.0, dtype=float)
    if shift >= 0:
        # h[i+shift] exists for i < n-shift
        g[: n - shift] -= h[shift:]
        # h[i-shift] exists for i >= shift
        g[shift:] -= h[: n - shift]
    else:
        s = -shift
        g[s:] -= h[: n - s]
        g[: n - s] -= h[s:]
    inc = np.where(g < 0)[0]   # want to increase these entries
    dec = np.where(g > 0)[0]   # want to decrease these entries
    return inc, dec


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin
    limit = start + budget_s * 0.94

    # ------------------------------------------------------------------
    # initialise the vector h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        n = 200
        h = np.full(n, 0.5, dtype=float)

    n = h.shape[0]
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # simple rescaling + clipping to repair the sum constraint
        h *= target_sum / cur_sum
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            mask = (h > 0) & (h < 1)
            if mask.any():
                h[mask] += diff / mask.sum()
                h = np.clip(h, 0.0, 1.0)

    c = 1.0 - h
    offset = n - 1
    L = 2 * n - 1
    conv = np.correlate(h, c, mode="full")
    cur_best = conv.max()
    best_h = h.copy()
    best_val = cur_best

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    max_step = 0.5
    min_step = 1e-4
    max_temp = 0.02
    min_temp = 1e-5
    no_improve = 0

    while time.time() < limit:
        elapsed = time.time() - start
        frac = elapsed / (limit - start)

        # annealing schedule
        step_scale = max_step * (1.0 - frac) + min_step * frac
        temp = max_temp * (1.0 - frac) + min_temp * frac

        # decide whether to use a directed move (based on the worst shift)
        if rng.random() < 0.7:

        else:
            i, j = rng.integers(0, n, size=2)

        if i == j:

        d = rng.random() * max_inc * step_scale
        if d <= 1e-12:

        # ------------------------------------------------------------------
        delta = np.zeros(L, dtype=float)
