def project_onto_feasible(v, target):
    # Clip to [0,1]
    w = np.clip(v, 0, 1)
    # Compute shortage/excess
    s = np.sum(w) - target
    if abs(s) < 1e-9:
        return w
    # Use Lagrange multiplier lambda to subtract/add equally to free variables
    # Determine free set: indices where 0 < w_i < 1 (not at bounds).
    # While loop:
    while True:
        free = (w > 0) & (w < 1)
        m = np.sum(free)
        if m == 0:
            # All at bounds cannot adjust, need to adjust some that are at bounds
            # In this case we need to distribute the mass by moving some from 1 to lower or 0 to higher etc.
            # For simplicity use simple scaling approach: w = w + (target - np.sum(w)) / n
            w += (target - np.sum(w)) / len(w)
            w = np.clip(w, 0, 1)
            continue
        # compute adjustment needed per free variable: delta = s / m

def gradient_step(h, dx, target_sum, step_size):
    # compute correlation and identify worst lag
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    L = idx - (len(h) - 1)
    # compute gradient g_i = 1 - h[i+L] if in bounds else 1
    n = len(h)
    # compute g
    g = np.ones(n, dtype=np.float64)  # default 1
    if L >= 0:
        # for i s.t. i+L < n
        g[:n-L] = 1.0 - h[L:]
    else:
        shift = -L
        # i+L = i-shift => valid for i>=shift
        g[shift:] = 1.0 - h[:n-shift]
    # direction to decrease objective: -g, but need zero-sum.
    d = -g + np.mean(g)  # ensures sum zero
    # step
    new_h = h + step_size * d
    # project onto feasible set
    new_h = project_onto_feasible(new_h, target_sum)
    return new_h

import time
import math
import numpy as np

def _project_onto_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector onto the set {0<=x_i<=1, sum x_i = target}."""
    v = np.asarray(v, dtype=np.float64)
    lo = v.min() - 1.0          # ensures all clipped to 1 when lambda=lo
    hi = v.max()                # ensures all clipped to 0 when lambda=hi
    for _ in range(60):        # binary search for the Lagrange multiplier
        mid = (lo + hi) * 0.5
        w = v - mid
        np.clip(w, 0.0, 1.0, out=w)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = v - (lo + hi) * 0.5
    np.clip(w, 0.0, 1.0, out=w)
    return w


def _c5_value(h: np.ndarray, dx: float) -> float:
    """Compute the overlap constant C5 for a given discretisation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _pairwise_move(h: np.ndarray) -> np.ndarray:
    """Generate a neighbour preserving the sum and bounds using a pairwise exchange."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]
    eps_max = min(1.0 - hi, hj)
    eps_min = -min(hi, 1.0 - hj)
    if eps_max <= eps_min:
        return h.copy()
    eps = np.random.uniform(eps_min, eps_max)
    nh = h.copy()
    nh[i] = hi + eps
    nh[j] = hj - eps
    return nh


def _global_perturb(h: np.ndarray, sigma: float, target: float) -> np.ndarray:
    """Add Gaussian noise to all entries and re‑project onto the feasible set."""
    v = h + sigma * np.random.randn(h.shape[0])
    return _project_onto_feasible(v, target)


def _gradient_step_worst(h: np.ndarray, target: float, step: float) -> np.ndarray:
    """One projected gradient step aimed at the lag attaining the current maximum."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    L = idx - (n - 1)

    # compute g_i = 1 - h_{i+L} - h_{i-L}
    g = np.ones(n, dtype=np.float64)

    if L >= 0:
        # i+L  in range for i=0..n-L-1
        g[: n - L] -= h[L:]
        # i-L in range for i=L..n-1
        g[L:] -= h[: n - L]
    else:
        off = -L
        # i+L = i-off in range for i=off..n-1
        g[off:] -= h[: n - off]
        # i-L = i+off in range for i=0..n-off-1
        g[: n - off] -= h[off:]

    # direction that decreases the maximal term
    d = -g
    d -= d.mean()                     # keep the sum unchanged
    nh = h + step * d
    return _project_onto_feasible(nh, target)
