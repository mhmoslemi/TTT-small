def project_onto_hypercube_simplex(v, target):
    # v: initial vector
    # returns x with 0<=x<=1 and sum = target
    n = len(v)
    # start with x = np.clip(v, 0, 1)
    x = np.clip(v, 0, 1)
    # Compute residual
    residual = target - x.sum()
    # Keep track of free indices.
    free = np.arange(n)
    # We'll perform water-filling: add same amount to all free variables.
    while abs(residual) > 1e-12:
        free_mask = (x > 0) & (x < 1)
        free_idx = np.where(free_mask)[0]
        if len(free_idx) == 0:
            # cannot adjust further; break
            break
        delta = residual / len(free_idx)
        # Update x on free indices:

def project_to_sum_and_box(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # fix minor rounding error
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff across non-boundary elements
        free = (x > 0) & (x < 1)
        if np.any(free):
            x[free] += diff / free.sum()
            # clip again
            x = np.clip(x, 0.0, 1.0)
    return x

    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction to hit the target exactly
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


def _c5_of(h: np.ndarray) -> float:
    """compute C5 = max_k ∫ h(x)(1‑h(x+k)) dx for a discrete step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _random_binary_half(n: int) -> np.ndarray:
    """binary vector with exactly n/2 ones (n must be even)."""
    m = n // 2
    arr = np.zeros(n, dtype=float)
    arr[:m] = 1.0
    np.random.shuffle(arr)
    return arr


def _smooth_circular(v: np.ndarray, w: int) -> np.ndarray:
    """circular moving‑average of width w (odd). Keeps the total mass unchanged."""
    if w <= 1:
        return v.copy()
    half = w // 2
    out = np.zeros_like(v)
    for shift in range(-half, half + 1):
        out += np.roll(v, shift)
    out /= w
    return out


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise h
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        h_cur = _project_sum_box(h0, n / 2.0)
    else:
        n = 70                     # fallback size (must be even)
        h_cur = np.full(n, 0.5)    # uniform start, will be projected
        h_cur = _project_sum_box(h_cur, n / 2.0)

    best_h = h_cur.copy()
    best_score = _c5_of(best_h)
    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # optimisation parameters
    T0 = 0.02          # initial temperature for SA
    Tmin = 1e-5        # final temperature
    eps_start = 0.25   # max magnitude for two‑index swaps
    eps_end = 0.02

    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.96:
            break

        # linear schedules
        t = max(Tmin, T0 * (1.0 - elapsed / (budget_s * 0.96)))
        eps = eps_end + (eps_start - eps_end) * (1.0 - elapsed / (budget_s * 0.96))

        # --------------------------------------------------------------
        # Phase A : random binary + smoothing (every other iteration)
        if it % 2 == 0:
            # generate a fresh binary half‑vector
            cand = _random_binary_half(n)

            # choose a smoothing width (odd, 3‑15)
            w = np.random.choice([3, 5, 7, 9, 11, 13, 15])
            cand = _smooth_circular(cand, w)

            # already respects sum, bounds guaranteed by averaging
        else:
            # Phase B : two‑index random swap (simulated annealing)
            i, j = np.random.choice(n, size=2, replace=False)
            # feasible delta respecting box constraints
            max_inc = min(eps, 1.0 - h_cur[i], h_cur[j])
            min_inc = -min(eps, h_cur[i], 1.0 - h_cur[j])
            if max_inc <= min_inc:
                it += 1
                continue
            delta = np.random.uniform(min_inc, max_inc)
            cand = h_cur.copy()
            cand[i] += delta
            cand[j] -= delta
            # sum unchanged by construction; bounds honoured by construction

        # --------------------------------------------------------------
        # evaluate candidate
        cand_score = _c5_of(cand)

        # accept according to Metropolis criterion
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            if np.random.rand() < math.exp(-(cand_score - best_score) / t):
                accept = True

        if accept:
            h_cur = cand
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand.copy()

        it += 1

    # final projection to guarantee exact feasibility
    best_h = _project_sum_box(best_h, target_sum)
    final_score = _c5_of(best_h)

    # return as plain Python list to satisfy the interface
    return best_h.tolist(), float(final_score), n
