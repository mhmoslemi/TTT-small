def project_capped_simplex(v, target, tol=1e-9, max_iter=100):
    n = len(v)
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            low = mid
        else:
            high = mid
    # after loop
    return np.clip(v - (low + high) / 2, 0.0, 1.0)
```

import time, math
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for step function h on [0,2]."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_capped_simplex(v, target, tol=1e-10, max_iter=80):
    """Project v onto the set {x in [0,1]^n : sum(x)=target}."""
    low = -np.max(v) - 1.0
    high = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) * 0.5, 0.0, 1.0)


def optimize_h(h, time_budget):
    """Simple simulated‑annealing / gradient descent on a fixed grid."""
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(best_h)
    curr_h = h.copy()
    curr_val = best_val
    start = time.time()
    it = 0
    # schedule constants
    eta0 = 0.08
    eps0 = 0.03
    temp0 = best_val * 0.5 + 1e-9

    while time.time() - start < time_budget:
        # gradient of the maximal correlation term
        corr = np.correlate(curr_h, 1.0 - curr_h, mode="full")
        idx = np.argmax(corr)
        k = idx - (n - 1)

        grad = np.zeros(n)
        if k >= 0:
            grad[: n - k]   += 1.0 - curr_h[k:]
            grad[k:        ] -= curr_h[: n - k]
        else:
            kk = -k
            grad[kk:       ] += 1.0 - curr_h[: n - kk]
            grad[: n - kk ] -= curr_h[kk:]

        # occasional random push to avoid stagnation
        if np.random.rand() < 0.15:
            grad += np.random.randn(n) * 0.1

        # step size decays with iterations
        eta = eta0 / (1.0 + it * 4e-4)
        cand = curr_h - eta * grad

        # tiny random jitter
        if np.random.rand() < 0.1:
            cand += (np.random.rand(n) - 0.5) * eps0

        cand = project_capped_simplex(cand, target)
        cand_val = compute_c5(cand)

        # deterministic improvement, otherwise Metropolis acceptance
        if cand_val < curr_val:
            curr_h, curr_val = cand, cand_val
            if cand_val < best_val:
                best_h, best_val = cand.copy(), cand_val
        else:
            temp = max(1e-9, temp0 * math.exp(-it * 1e-4))
            if np.random.rand() < math.exp(-(cand_val - curr_val) / temp):
                curr_h, curr_val = cand, cand_val

        # periodic restart around the best solution
        if it % 5000 == 0 and it > 0:
            rand = best_h + (np.random.rand(n) - 0.5) * eps0
            curr_h = project_capped_simplex(rand, target)
            curr_val = compute_c5(curr_h)

        it += 1

    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h on [0,2] with a small C5 bound."""
    np.random.seed(seed)
    start_total = time.time()

    # ----- start from the provided construction, if any -----
    init = globals().get("initial_h_values")
    if init is not None:

    else:

    # enforce constraints at the base resolution
    n0 = h0.shape[0]
    h0 = project_capped_simplex(h0, n0 / 2.0)

    best_h = h0
    best_c5 = compute_c5(best_h)

    # ---- hierarchical refinement (42 → 84 → 168 → 336 ...) ----
    sizes = [n0]
    while sizes[-1] < 500:
        sizes.append(sizes[-1] * 2)
    sizes = [s for s in sizes if s <= 800]  # keep work reasonable
    total_stages = len(sizes)

    for stage, n in enumerate(sizes):
        # time left for this stage (90 % of remaining shared equally)
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break
        stages_left = total_stages - stage
        alloc = remaining * 0.9 / stages_left
