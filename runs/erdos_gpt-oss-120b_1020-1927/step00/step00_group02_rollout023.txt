import numpy as np, random, time, math

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # Use provided initial_h_values if present, else construct triangular start.
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 200
        dx = 2.0 / n
        x = (np.arange(n) + 0.5) * dx
        h0 = np.minimum(x, 2.0 - x)          # triangular shape
        # adjust to exact integral
        target_sum = n / 2.0
        cur_sum = h0.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            h0[0] += target_sum - cur_sum
            h0[0] = min(max(h0[0], 0.0), 1.0)

    # Ensure constraints
    n = h0.size
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    # Fix sum if needed
    diff = target_sum - h0.sum()
    if abs(diff) > 1e-12:
        # add diff to a random index that can absorb it
        idxs = [i for i in range(n) if 0 <= h0[i] + diff <= 1]
        if idxs:
            h0[random.choice(idxs)] += diff
        else:
            # fallback: distribute uniformly
            h0 += diff / n
            h0 = np.clip(h0, 0.0, 1.0)

    dx = 2.0 / n

    def corr_vec(v):
        return np.correlate(v, 1.0 - v, mode='full')

    def c5_val(v):
        return np.max(corr_vec(v)) * dx

    cur_h = h0.copy()
    cur_corr = corr_vec(cur_h)
    cur_c5 = np.max(cur_corr) * dx
    best_h = cur_h.copy()
    best_c5 = cur_c5

    start = time.time()
    limit = budget_s - 0.3

    # parameters for annealing
    T0 = 0.02
    step0 = 0.5
    no_improve = 0

    while True:
        now = time.time()
        if now - start >= limit:
            break
        elapsed = now - start
        prog = elapsed / limit
        T = max(1e-6, T0 * (1 - prog))
        step = max(0.005, step0 * (1 - prog))

        # Decide move type
        if random.random() < 0.9 and np.max(cur_corr) > 0:
            # directed move based on current worst shift
            max_idx = int(np.argmax(cur_corr))
            d = max_idx - (n - 1)        # shift value (can be negative)
            if d >= 0:
                i_min, i_max = 0, n - 1 - d
            else:
                i_min, i_max = -d, n - 1
            i = random.randint(i_min, i_max)
            j = i + d
            direction = random.choice([1, -1])
        else:
            # random pair move
            i, j = random.sample(range(n), 2)
            direction = random.choice([1, -1])

        # compute feasible delta
        if direction == 1:
            max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
        else:
            max_delta = min(step, cur_h[j], 1.0 - cur_h[i])
        if max_delta <= 1e-12:
            continue
        delta = random.uniform(0.0, max_delta)

        # apply move
        if direction == 1:
            cur_h[i] -= delta
            cur_h[j] += delta
        else:
            cur_h[j] -= delta
            cur_h[i] += delta

        new_corr = corr_vec(cur_h)
        new_c5 = np.max(new_corr) * dx

        # accept or revert
        accept = False
        if new_c5 < cur_c5 - 1e-12:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp(-(new_c5 - cur_c5) / T) if T > 0 else 0.0
            if random.random() < prob:
                accept = True

        if accept:
            cur_c5 = new_c5
            cur_corr = new_corr
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = cur_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert
            if direction == 1:
                cur_h[i] += delta
                cur_h[j] -= delta
            else:
                cur_h[j] += delta
                cur_h[i] -= delta
            no_improve += 1

        # occasional restart if stagnated
        if no_improve > 30000:
            # random binary restart
            cur_h = np.zeros(n, dtype=float)
            ones = random.sample(range(n), n // 2)
            cur_h[ones] = 1.0
            cur_corr = corr_vec(cur_h)
            cur_c5 = np.max(cur_corr) * dx
            no_improve = 0

    # final projection onto feasible set
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-9:
        # distribute diff uniformly (small) and re‑clip
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = c5_val(best_h)

    return best_h, float(final_c5), int(n)

import numpy as np, random, time, math

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step‑function h on [0,2] (as a discrete vector) with a low
    Erdős‑overlap bound C₅.
    The vector satisfies 0≤h[i]≤1 and sum(h)=n/2, where n is its length.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ----- initial vector -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        # triangular start, n modest for speed
        n = 200
        dx = 2.0 / n
        x = (np.arange(n) + 0.5) * dx               # mid‑points of sub‑intervals
        h0 = np.minimum(x, 2.0 - x)                # 0 at ends, 1 at centre
        # enforce exact integral = 1  (i.e. sum = n/2)
        target = n / 2.0
        diff = target - h0.sum()
        if abs(diff) > 1e-12:
            h0[0] += diff
        h0 = np.clip(h0, 0.0, 1.0)

    n = h0.size
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    # tiny correction to satisfy the linear constraint exactly
    diff = target_sum - h0.sum()
    if abs(diff) > 1e-12:
        # adjust a random entry that can absorb the difference
        idx = random.randint(0, n - 1)
        new_val = h0[idx] + diff
        if 0.0 <= new_val <= 1.0:
            h0[idx] = new_val
        else:
            h0 += diff / n
            h0 = np.clip(h0, 0.0, 1.0)

    dx = 2.0 / n

    # ----- helpers --------------------------------------------------------
    def corr_vec(v):
        """Full discrete correlation of h with 1‑h."""
        return np.correlate(v, 1.0 - v, mode='full')

    def c5_val(v):
        """Current C₅ bound for vector v."""
        return np.max(corr_vec(v)) * dx

    # ----- initialise search ----------------------------------------------
    cur_h = h0.copy()
    cur_corr = corr_vec(cur_h)
    cur_c5 = np.max(cur_corr) * dx
    best_h = cur_h.copy()
    best_c5 = cur_c5

    start = time.time()
    time_limit = budget_s - 0.3      # safety margin

    # annealing schedule parameters
    T0 = 0.02
    step0 = 0.5
    stagnation = 0

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        prog = elapsed / time_limit
        T = max(1e-6, T0 * (1.0 - prog))
        step = max(0.005, step0 * (1.0 - prog))

        # ---- decide which move to try ------------------------------------
        if random.random() < 0.9 and np.max(cur_corr) > 0:
            # targeted move based on the presently worst shift
            max_idx = int(np.argmax(cur_corr))
            d = max_idx - (n - 1)                # signed shift
            if d >= 0:
                i_min, i_max = 0, n - 1 - d
            else:
                i_min, i_max = -d, n - 1
            i = random.randint(i_min, i_max)
            j = i + d
            # move mass from the larger to the smaller entry to increase the product
            if cur_h[i] > cur_h[j]:
                src, dst = i, j
            else:
                src, dst = j, i
        else:
            # random pair, direction free
            src, dst = random.sample(range(n), 2)
            # pick direction that tends to equalise the two values
            if cur_h[src] < cur_h[dst]:
                src, dst = dst, src

        # ---- feasible delta ------------------------------------------------
        max_delta = min(step, cur_h[src], 1.0 - cur_h[dst])
        if max_delta <= 1e-12:
            continue
        delta = random.uniform(0.0, max_delta)

        # ---- apply move ---------------------------------------------------
        cur_h[src] -= delta
        cur_h[dst] += delta

        new_corr = corr_vec(cur_h)
        new_c5 = np.max(new_corr) * dx

        # ---- Metropolis acceptance -----------------------------------------
        accept = False
        if new_c5 < cur_c5 - 1e-12:
            accept = True
        else:
            prob = math.exp(-(new_c5 - cur_c5) / T) if T > 0 else 0.0
            if random.random() < prob:
                accept = True

        if accept:
            cur_c5 = new_c5
            cur_corr = new_corr
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = cur_h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            # revert
            cur_h[src] += delta
            cur_h[dst] -= delta
            stagnation += 1

        # ---- occasional restart if stuck ----------------------------------
        if stagnation > 30000:
            # binary random restart (exactly n/2 ones)
            cur_h[:] = 0.0
            one_pos = random.sample(range(n), n // 2)
            cur_h[one_pos] = 1.0
            cur_corr = corr_vec(cur_h)
            cur_c5 = np.max(cur_corr) * dx
            stagnation = 0

    # ----- final projection onto the feasible set -------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-9:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = c5_val(best_h)

    return best_h.tolist(), float(final_c5), int(n)
