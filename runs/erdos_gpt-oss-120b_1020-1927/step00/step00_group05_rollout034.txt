def grad_worst(h, k, dx):
    n = len(h)
    grad = np.full(n, dx)  # base term 1.
    if k >= 0:
        n_k = n - k
        if n_k > 0:
            grad[:n_k] -= dx * h[k:]
            grad[k:] -= dx * h[:n_k]
    else:
        k_abs = -k
        n_k = n - k_abs
        if n_k > 0:
            grad[k_abs:] -= dx * h[:n_k]
            grad[:n_k] -= dx * h[k_abs:]
    return grad

def project_box_sum(v, target, lower=0., upper=1., max_iter=100):
    # binary search lambda
    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):
        mid = (lo+hi)/2
        h = np.clip(v - mid, lower, upper)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo+hi)/2.0, lower, upper)
    # adjust for small floating error
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute diff among unfixed elements
        free = (h > lower + 1e-12) & (h < upper - 1e-12)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, lower, upper)
        else:
            # all at bounds; adjust one element
            # pick random index
            idx = np.random.randint(len(h))
            h[idx] = np.clip(h[idx] + diff, lower, upper)
    return h

    target = n/2

    return h

def optimize_pairwise(h_start, budget, step_max=0.05):
    n = len(h_start)
    dx = 2.0 / n
    target = n/2
    h = h_start.copy()
    best_h = h.copy()
    best_val = evaluate_h(h, dx)

    start = time.time()
    while time.time() - start < budget:
        # compute O for current h
        O = np.correlate(h, 1 - h, mode='full') * dx
        idx_max = np.argmax(O)
        k = idx_max - (n-1)
        cur_val = O[idx_max]

        # gradient for worst shift
        grad = grad_worst(h, k, dx)

        # choose indices
        i_pos = np.argmax(grad)
        i_neg = np.argmin(grad)

        # compute possible delta
        delta = min(step_max, h[i_pos], 1 - h[i_neg])
        if delta <= 0:
            # cannot improve, break or jitter
            # perform random jitter small
            perturb = (np.random.rand(n) - 0.5) * step_max
            v = h + perturb
            v = np.clip(v, 0, 1)
            v = project_box_sum(v, target)
            h = v
            continue

        # update
        h[i_pos] -= delta
        h[i_neg] += delta

        # evaluate new max
        O_new = np.correlate(h, 1 - h, mode='full') * dx
        val_new = O_new.max()

        # if new val better, update best
        if val_new < best_val - 1e-12:
            best_val = val_new
            best_h = h.copy()
            # increase step size maybe
            step_max = min(step_max * 1.05, 0.2)
        else:
            # maybe revert? we may keep because step may still move.
            # If no improvement, revert and try smaller step
            h[i_pos] += delta
            h[i_neg] -= delta
            step_max = step_max * 0.7
            # random small perturbation
            i1, i2 = np.random.choice(n, 2, replace=False)
            eps = min(step_max, h[i1], 1 - h[i2])
            if eps > 0:
                h[i1] -= eps
                h[i2] += eps

        # optionally early exit if no improvement after many iterations

    return best_h, best_val

def optimize_pairwise(start_h, budget):
    h = start_h.copy()
    n = len(h)
    dx = 2.0 / n
    target_sum = n/2
    best_h = h.copy()
    best_val = evaluate_h(h, dx)
    start = time.time()
    step_max = 0.05  # initial max transfer amount
    no_improve_iters = 0
    while time.time() - start < budget:
        O = np.correlate(h, 1 - h, mode='full') * dx
        idx_max = np.argmax(O)
        k = idx_max - (n - 1)
        cur_val = O[idx_max]

        grad = grad_worst(h, k, dx)

        # find max positive and min negative gradient indices
        i_pos = np.argmax(grad)  # largest positive
        i_neg = np.argmin(grad)  # most negative

        # compute possible transfer delta limited by bounds and step_max
        delta = min(step_max, h[i_pos], 1 - h[i_neg])
        if delta <= 0:
            # no possible transfer in this direction, try opposite direction
            delta = min(step_max, h[i_neg], 1 - h[i_pos])
            if delta <= 0:
                # random jitter
                jitter = (np.random.rand(n) - 0.5) * step_max
                v = h + jitter
                v = np.clip(v, 0, 1)
                # project onto sum constraint
                v = project_box_sum(v, target_sum)
                h = v
                continue
            else:
                # swap roles: move from i_neg to i_pos
                i_from, i_to = i_neg, i_pos
        else:
            i_from, i_to = i_pos, i_neg

        # perform transfer
        h[i_from] -= delta
        h[i_to] += delta

        # evaluate new
        new_val = evaluate_h(h, dx)
        if new_val < best_val - 1e-12:
            best_val = new_val
            best_h = h.copy()
            no_improve_iters = 0
            step_max = min(step_max * 1.05, 0.2)
        else:
            # revert? we might keep it and continue, but if it worsens a lot we may revert
            # Simple rule: if new_val > cur_val: revert; else keep.
            if new_val > cur_val:
                # revert transfer
                h[i_from] += delta
                h[i_to] -= delta
                # decrease step size
                step_max = max(step_max * 0.7, 1e-4)
                no_improve_iters += 1
                # random perturb after many no improvements
                if no_improve_iters > 100:
                    # random reinitialize from best_h with jitter
                    jitter = (np.random.rand(n) - 0.5) * step_max
                    v = best_h + jitter
                    v = np.clip(v, 0, 1)
                    v = project_box_sum(v, target_sum)
                    h = v
                    no_improve_iters = 0
            else:
                # keep; maybe minor improvement? but we necessary.
                # update best if new_val better anyway
                if new_val < best_val - 1e-12:
                    best_val = new_val
                    best_h = h.copy()
                no_improve_iters = 0

        # time check
        if time.time() - start > budget:
            break

    return best_h, best_val

import time
import numpy as np

def evaluate_h(h, dx):
    """Maximum overlap integral for discrete h."""
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)


def grad_worst(h, k, dx):
    """Gradient of the worst overlap term (lag k) w.r.t. h."""
    n = h.size
    grad = np.zeros(n, dtype=np.float64)

    if k >= 0:
        lim = n - k
        # contribution from h_i term (when i+k in range)
        grad[:lim] += dx * (1 - h[k:])
        # contribution from product term (when i = j-k)
        grad[k:] -= dx * h[:lim]
    else:
        ka = -k
        lim = n - ka
        # contribution from h_i term (i+k >=0)
        grad[ka:] += dx * (1 - h[:lim])
        # contribution from product term (i = j+ka)
        grad[:lim] -= dx * h[ka:]

    return grad


def project_box_sum(v, target, lower=0.0, upper=1.0, iters=60):
    """Project onto box [lower,upper] with sum constraint = target."""
    lo, hi = -1e6, 1e6
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, lower, upper)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        free = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        cnt = free.sum()
        if cnt > 0:
            proj[free] += diff / cnt
            proj = np.clip(proj, lower, upper)
        else:
            # all at bounds – adjust a random free coordinate
            idx = np.random.randint(len(proj))
            proj[idx] = np.clip(proj[idx] + diff, lower, upper)
    return proj


def triangle_h(n):
    """Triangular shape h(x)=max(0,1-|x-1|) sampled on n points."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = project_box_sum(h, target)
    return h


def random_h(n, target):
    """Random feasible vector."""
    v = np.random.rand(n)
    return project_box_sum(v, target)


def optimize(h_start, budget, step_init=0.05):
    """Pairwise mass‑transfer local search for given time budget."""
    h = h_start.copy()
    n = h.size
    dx = 2.0 / n
    target = n / 2.0

    best_h = h.copy()
    best_val = evaluate_h(h, dx)

    start = time.time()
    step = step_init
    stagnation = 0

    while time.time() - start < budget:
        # worst shift
        corr = np.correlate(h, 1 - h, mode='full') * dx
        idx = int(np.argmax(corr))
        k = idx - (n - 1)
        cur_val = corr[idx]

        # gradient of this worst term
        g = grad_worst(h, k, dx)

        i_pos = int(np.argmax(g))      # largest positive gradient
        i_neg = int(np.argmin(g))      # most negative gradient

        delta = min(step, h[i_pos], 1.0 - h[i_neg])
        if delta > 0:
            h[i_pos] -= delta
            h[i_neg] += delta
            new_val = evaluate_h(h, dx)

            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h.copy()
                stagnation = 0
                step = min(step * 1.05, 0.2)
            else:
                # revert if it made things worse
                if new_val > cur_val:
                    h[i_pos] += delta
                    h[i_neg] -= delta
                    step = max(step * 0.7, 1e-4)
                    stagnation += 1
                else:
                    stagnation = 0
        else:
            # no feasible transfer – small random jitter
            jitter = (np.random.rand(n) - 0.5) * step
            v = np.clip(h + jitter, 0.0, 1.0)
            v = project_box_sum(v, target)
            h = v
            stagnation += 1

        if stagnation > 2000:
            # restart from best with jitter
            jitter = (np.random.rand(n) - 0.5) * step
            v = np.clip(best_h + jitter, 0.0, 1.0)
            v = project_box_sum(v, target)
            h = v
            stagnation = 0

        # early exit if already very good
        if best_val <= 0.3805:
            break

    return best_h, best_val

    except Exception:                     # no parent construction
        n = 70
        target = n / 2.0
        init = None

    # build a few start points
    starters = []
    if init is not None:
        starters.append(init)
    starters.append(triangle_h(n))
    for _ in range(3):
        starters.append(random_h(n, target))

    best_h = None
    best_val = float('inf')

    # allocate time among starts
    remaining = budget_s
    for s in starters:
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed
        if remaining < 0.5:
            break
        sub_budget = min(remaining * 0.25, 12.0)   # up to ~12 s per start
        h_opt, val_opt = optimize(s, sub_budget)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
        if best_val <= 0.3805:
            break

    # final refinement on the best found solution
    elapsed = time.time() - start_total
    remaining = budget_s - elapsed
    if remaining > 1.0 and best_h is not None:
        h_opt, val_opt = optimize(best_h, remaining)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()

    # safety projection before return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, n / 2.0)

    final_val = evaluate_h(best_h, 2.0 / n)
    return best_h, float(final_val), int(n)
