import time
import numpy as np

def _c5(h: np.ndarray) -> float:
    """return C_5 for a step function given by the samples h."""
    n = h.shape[0]
    # full correlation of h with (1-h); zero padding outside the interval
    return np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n)


def _hill_climb(h: np.ndarray, max_moves_factor: int = 5) -> np.ndarray:
    """
    Simple greedy / simulated‑annealing style improvement for a binary vector.
    Keeps sum(h) = n/2 and h∈{0,1}. Tries to reduce the maximal overlap.
    """
    n = h.shape[0]
    max_moves = n * max_moves_factor
    for _ in range(max_moves):
        # current overlap profile
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        max_val = float(corr[max_idx])
        if max_val == 0.0:                     # perfect, cannot improve
            break

        shift = max_idx - (n - 1)               # integer shift producing the max
        if shift == 0:                          # impossible (overlap is zero), pick another
            shift = np.random.randint(1, n)

        # indices that contribute a “1,0” pattern for this shift
        if shift > 0:
            a_mask = (h[: n - shift] == 1) & (h[shift:] == 0)
            b_mask = (h[: n - shift] == 0) & (h[shift:] == 1)
            A = np.flatnonzero(a_mask)
            B = np.flatnonzero(b_mask)
        else:  # shift < 0
            s = -shift
            a_mask = (h[s:] == 1) & (h[: n - s] == 0)
            b_mask = (h[s:] == 0) & (h[: n - s] == 1)
            A = np.flatnonzero(a_mask) + s
            B = np.flatnonzero(b_mask) + s

        # try a targeted swap that fixes the worst shift
        improved = False
        if A.size and B.size:
            i = np.random.choice(A)
            j = np.random.choice(B)
            h[i], h[j] = 0.0, 1.0                # swap 1↔0
            new_max = np.correlate(h, 1.0 - h, mode="full").max()
            if new_max < max_val:
                improved = True
            else:
                h[i], h[j] = 1.0, 0.0            # revert
        if not improved:
            # fall back to a random 1↔0 swap; accept only if it does not worsen
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
            if ones.size == 0 or zeros.size == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h[i], h[j] = 0.0, 1.0
            new_max = np.correlate(h, 1.0 - h, mode="full").max()
            if new_max > max_val:
                h[i], h[j] = 1.0, 0.0          # revert, no improvement
    return h


def _optimize_binary(n: int, time_budget: float, global_best: float):
    """
    Search for a binary vector of length n (with exactly n/2 ones)
    that minimises C_5, using many random restarts and the hill‑climb above.
    """
    start = time.time()
    best_h = None
    best_val = float("inf")
    while time.time() - start < time_budget:
        h = np.zeros(n, dtype=np.float64)
        h[np.random.choice(n, n // 2, replace=False)] = 1.0
        # quick improvement
        h = _hill_climb(h)
        val = _c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        # early exit if we already beat the current global best
        if best_val < global_best - 1e-12:
            break
    return best_h, best_val
