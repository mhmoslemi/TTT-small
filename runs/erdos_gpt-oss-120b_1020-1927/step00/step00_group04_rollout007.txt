import numpy as np
import time
import math

# ------------------------------------------------------------
# Helper: project a vector onto the box [0,1] with a given sum.
# ------------------------------------------------------------
def _project_sum_box(x: np.ndarray, target: float) -> np.ndarray:
    # Find λ such that sum(clip(x - λ,0,1)) = target.
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        z = np.clip(x - mid, 0.0, 1.0)
        s = z.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


# ------------------------------------------------------------
# Compute the C5 value (max overlap) for a step vector h.
# ------------------------------------------------------------
def _c5_of(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' yields all integer shifts.
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(conv.max() * dx)


# ------------------------------------------------------------
# Return (C5, worst_shift) where worst_shift is integer offset
# (positive means h shifted right).
# ------------------------------------------------------------
def _c5_and_shift(h: np.ndarray):
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    idx = int(conv.argmax())
    shift = idx - (n - 1)
    return float(conv[idx] * (2.0 / n)), shift

# Perform a steepest‑descent binary local search from a given start.
# h must contain exactly n/2 ones and n/2 zeros (as floats 0/1).
# ------------------------------------------------------------
def _binary_local_search(start_h: np.ndarray):
    n = start_h.shape[0]
    h = start_h.copy()
    best_val = _c5_of(h)
    improved = True
    while improved:
        improved = False
        ones = np.flatnonzero(h)               # indices where h==1
        zeros = np.flatnonzero(1.0 - h)         # indices where h==0
        best_swap = None
        best_val_local = best_val
        # explore all possible swaps
        for i in ones:
            for j in zeros:
                # swap i (1) with j (0)
                h[i], h[j] = 0.0, 1.0
                val = _c5_of(h)
                if val < best_val_local - 1e-13:
                    best_val_local = val
                    best_swap = (i, j)
                # revert
                h[i], h[j] = 1.0, 0.0
        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = 0.0, 1.0
            best_val = best_val_local
            improved = True
    return h, best_val

    target = n // 2
    best_h = None

            h_bin[order[:target]] = 1.0
            val = _c5_of(h_bin)
            if val < best_val:
                best_val, best_h = val, h_bin.copy()

        h0 = np.zeros(n, dtype=np.float64)

    return best_h, best_val

def _gradient_refine(h: np.ndarray, time_budget: float):
    n = h.shape[0]
    target = n / 2.0
    dx = 2.0 / n
    start = time.time()
    it = 0
    step0 = 0.5
    while time.time() - start < time_budget:
        # find worst shift (max overlap)
        conv = np.correlate(h, 1.0 - h, mode='full')
        idx = int(conv.argmax())
        shift = idx - (n - 1)
        # gradient of C5 w.r.t. h (sub‑gradient using the worst shift)
        grad = np.ones(n, dtype=np.float64)

            grad[:n - s] -= h[s:]
            grad[s:]    -= h[:n - s]

            grad[ss:]   -= h[:n - ss]
            grad[:n - ss] -= h[ss:]
        it += 1
        step = step0 / math.sqrt(it)
        h = h - step * grad
        h = _project_sum_box(h, target)
    return h
