        h0 = np.zeros(n, dtype=float)
        idx = np.random.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    n = h0.shape[0]
    half = n // 2

    # force binary and correct total count
    idx1 = np.argsort(-h0)[:half]
    h_bin = np.zeros(n, dtype=int)
    h_bin[idx1] = 1

    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()

    # ------------------------------------------------------------------
    # Try a mixed‑integer formulation (if time permits)
    # ------------------------------------------------------------------
    time_left = budget_s - (time.time() - start_time)
    # reserve at most 600 s for the MIP, otherwise skip it
    if time_left > 300:
        try:
            import cvxpy as cp
        except Exception:  # cvxpy not available
            cp = None

        if cp is not None:
            m = n
            x = cp.Variable(m, boolean=True)
            t = cp.Variable(integer=True)

            cons = [cp.sum(x) == half, t >= 0]

            # build constraints for each shift L (both directions)
            for L in range(1, m):
                region = m - L
                x_left = x[:region]
                x_right = x[L:]
                y = cp.Variable(region, boolean=True)

                cons.append(y <= x_left)
                cons.append(y <= x_right)
                cons.append(y >= x_left + x_right - 1)

                # h·(1‑h shifted) = sum(x_left) - sum(y)
                cons.append(cp.sum(x_left) - cp.sum(y) <= t)
                # the opposite direction
                cons.append(cp.sum(x_right) - cp.sum(y) <= t)

            prob = cp.Problem(cp.Minimize(t), cons)

            # try a few common MILP solvers
            solved = False
            for solver_name in ('GLPK_MI', 'CBC', 'ECOS_BB'):
                try:
                    prob.solve(solver=solver_name, verbose=False)
                    if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                        solved = True
                        break
                except Exception:
                    continue

            if solved and x.value is not None:
                xv = np.rint(np.array(x.value)).astype(int)
                # adjust possible rounding errors to keep exact half ones
                diff = half - xv.sum()
                if diff != 0:
                    if diff > 0:
                        zeros = np.where(xv == 0)[0]
                        np.random.shuffle(zeros)
                        xv[zeros[:diff]] = 1
                    else:
                        ones = np.where(xv == 1)[0]
                        np.random.shuffle(ones)
                        xv[ones[: -diff]] = 0
                best_h = xv.astype(float)
                best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Local search (hill climbing with random restarts)
    # ------------------------------------------------------------------
    time_elapsed = time.time() - start_time
    time_remaining = max(0.0, budget_s - time_elapsed - 0.05)  # tiny buffer

    if time_remaining > 0.0:
        dx = 2.0 / n

        def hill_climb(curr_h, curr_c5, time_budget):
            cur = curr_h.copy()
            cur_c5 = curr_c5
            best_loc = cur.copy()
            best_val = cur_c5
            start = time.time()
            stagnant = 0

            while time.time() - start < time_budget:
                ones = np.where(cur == 1)[0]
                zeros = np.where(cur == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = np.random.choice(ones)
                j = np.random.choice(zeros)

                # propose swap
                cur[i] = 0
                cur[j] = 1
                new_c5 = compute_c5(cur)

                if new_c5 + 1e-12 < best_val:
                    best_loc = cur.copy()
                    best_val = new_c5
                    cur_c5 = new_c5
                    stagnant = 0
                else:
                    # reject move
                    cur[i] = 1
                    cur[j] = 0
                    stagnant += 1

                if stagnant > 20000:
                    # random restart
                    cur = np.zeros(n, dtype=int)
                    idx = np.random.choice(n, half, replace=False)
                    cur[idx] = 1
                    cur_c5 = compute_c5(cur)
                    if cur_c5 < best_val:
                        best_loc = cur.copy()
                        best_val = cur_c5
                    stagnant = 0
            return best_loc, best_val

        best_h, best_c5 = hill_climb(best_h, best_c5, time_remaining)

    # final sanity check: enforce exact half ones
    total = int(best_h.sum())
    if total != half:
        diff = half - total
        if diff > 0:
            zeros = np.where(best_h == 0)[0]
            np.random.shuffle(zeros)
            best_h[zeros[:diff]] = 1
        else:
            ones = np.where(best_h == 1)[0]
            np.random.shuffle(ones)
            best_h[ones[: -diff]] = 0

    final_c5 = compute_c5(best_h)

    return (best_h.astype(float), final_c5, n)
