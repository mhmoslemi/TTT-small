def project_to_capped_simplex(v, target):
    # Clip via water-filling to [0,1] and sum target (target within [0,n])
    # find tau via bisection.
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(60):
        mid = (low + high) / 2.0
        w = np.clip(v - mid, 0, 1)
        s = w.sum()
        if s > target:
            low = mid
        else:
            high = mid
    w = np.clip(v - (low + high) / 2.0, 0, 1)
    return w

def gradient_step(h, step_size, target):
    n = h.shape[0]
    # compute correlation of h with itself (full)
    conv = np.correlate(h, h, mode='full')
    # find minimal value
    min_val = conv.min()
    # indices of minimal values (could be multiple)
    min_indices = np.where(conv == min_val)[0]
    # compute gradient
    g = np.zeros_like(h)
    for idx in min_indices:
        shift = idx - (n - 1)  # integer shift
        if shift >= 0:
            # overlapping region size = n - shift
            g[:n-shift] += h[shift:]  # hi + shift
            g[shift:] += h[:n-shift]  # hi - shift
        else:
            s = -shift
            g[s:] += h[:n-s]
            g[:n-s] += h[s:]
    # step
    v = h + step_size * g
    # project
    h_new = project_to_capped_simplex(v, target)
    return h_new

def targeted_move(h, step):
    # compute correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    idx_max = np.argmax(corr)
    offset = idx_max - (len(h) - 1)
    n = len(h)
    # compute contributions array
    contrib = np.empty(n)
    for i in range(n):
        j = i + offset
        if 0 <= j < n:
            contrib[i] = h[i] * (1 - h[j])
        else:
            contrib[i] = h[i]  # since out of bounds => (1 - 0) = 1
    i_high = np.argmax(contrib)
    # candidate j where contribution low (prefer small hi)
    # Might pick j = argmin contrib, but ensure we have capacity to receive mass.
    # Or pick random index with h[j] < 1 and low contribution.
    # We'll sample several candidates and pick best.
    low_candidates = np.where(h < 1)[0]  # can receive
    if len(low_candidates) == 0:

    # pick j with minimal h/j (or minimal contribution)
    # Evaluate contributions at j: similar formula
    j_scores = np.empty_like(low_candidates, dtype=float)

    return h

import time
import numpy as np


def _project_to_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the intersection
    of the box [0,1]^n and the hyperplane sum(v)=target.
    Uses a simple bisection (water‑filling) on the Lagrange multiplier.
    """
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _evaluate_c5(h: np.ndarray) -> float:
    """
    Compute C5 as defined in the task:
        C5 = max_k ∫ h(x)(1‑h(x+k)) dx
    approximated on the discrete grid.
    """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a triple (h_values, c5_bound, n_points) where
        - h_values is a list of values in [0,1] with sum = n_points/2,
        - c5_bound is the achieved C5 value,
        - n_points is the length of the discretisation.
    The routine tries to improve the known bound by a simple
    stochastic hill‑climbing that preserves the constraints.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)  # type: ignore
    except NameError:
        # fallback: linear “tent’’ shape (high at the ends, zero in the middle)
        n = 200
        idx = np.arange(n, dtype=float)
        h0 = np.abs(idx - (n - 1) / 2.0) / ((n - 1) / 2.0)  # average = 0.5
    n = h0.size
    target_sum = n / 2.0

    # Ensure feasibility exactly (box + linear constraint)
    h = _project_to_capped_simplex(h0, target_sum)

    best_h = h.copy()
    best_c5 = _evaluate_c5(best_h)

    start_time = time.time()
    # ------------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------------
    step_max = 0.5                 # maximal amount moved in one transfer
    decay = 0.9995                # exponential decay of step size
    random_move_prob = 0.7        # probability to perform a blind random transfer
    min_accept_gain = 1e-12       # tolerance for accepting improvements

    # ------------------------------------------------------------------
    # main loop (time limited)
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s - 0.3:
        old_h = h.copy()

        if rng.random() < random_move_prob:
            # -------------------------- random transfer -----------------
            i, j = rng.choice(n, size=2, replace=False)
            if h[i] <= 0.0 or h[j] >= 1.0:
                continue
            max_transfer = min(step_max, h[i], 1.0 - h[j])
            if max_transfer <= 0.0:
                continue
            delta = max_transfer * rng.random()
            h[i] -= delta
            h[j] += delta
        else:
            # -------------------------- targeted move ------------------
            # identify the shift that currently gives the worst (largest) C5 term
            full_corr = np.correlate(h, 1.0 - h, mode="full")
            idx_max = int(full_corr.argmax())
            offset = idx_max - (n - 1)   # integer shift (may be negative)

            # contributions of each index to the maximal correlation term
            contrib = np.empty(n, dtype=float)
            if offset >= 0:
                # overlapping part
                contrib[: n - offset] = h[: n - offset] * (1.0 - h[offset:])
                # tail where partner is out of range -> partner = 0
                contrib[n - offset :] = h[n - offset :]
            else:
                off = -offset
                contrib[off:] = h[off:] * (1.0 - h[: n - off])
                contrib[:off] = h[:off]

            i_src = int(contrib.argmax())          # a source with large contribution
            # pick a destination that can receive mass and has a small contribution
            possible = np.where((h < 1.0) & (np.arange(n) != i_src))[0]
            if possible.size == 0:
                h = old_h
                continue
            j_dst = int(rng.choice(possible))
            max_transfer = min(step_max, h[i_src], 1.0 - h[j_dst])
            if max_transfer <= 0.0:
                h = old_h
                continue
            delta = max_transfer * rng.random()
            h[i_src] -= delta
            h[j_dst] += delta

        # projection to keep feasibility (exact sum and box)
        h = _project_to_capped_simplex(h, target_sum)

        # evaluate
        current_c5 = _evaluate_c5(h)
        if current_c5 + min_accept_gain < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        else:
            h = old_h  # reject

        # slowly shrink the step size
        step_max *= decay

    # final projection (just in case of tiny numerical drift)
    best_h = _project_to_capped_simplex(best_h, target_sum)
    final_c5 = _evaluate_c5(best_h)

    # return as plain Python list to avoid hidden NumPy dependencies for the harness
    return best_h.tolist(), float(final_c5), n
