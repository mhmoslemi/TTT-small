    # Projection onto simplex intersected with [0,1]^N
    v_clipped = np.clip(v, 0, 1)
    s = np.sum(v_clipped)
    target = v.size / 2.0
    
    if abs(s - target) < 1e-5:
        return v_clipped
    
    # Scale projection
    if s < target:
        scale = target / s
        v_scaled = v_clipped * scale
        # Fill residual deficit by increasing least saturated points
        remainder = target - np.sum(np.clip(v_scaled, 0, min(1, np.max(v_scaled)))  # Approximation
        if remainder > 1e-5:
            # Distribute remainder
            active = np.flatnonzero(v_scaled < 1)
            if len(active) > 0:
                v_scaled[active[:int(remainder)]] = np.clip(v_scaled[active[:int(remainder)]] + remainder / len(active), 0, 1)
    else:
        # Excess case: scale down and remove any saturation
        scale = target / s
        v_scaled = v_clipped * scale
        # Trim excess by setting superfluous values to median
        sort_idx = np.argsort(v_scaled)
        trim_count = max(0, int(round((sum(v_scaled) - target) * 10)))
        v_scaled[sort_idx[:trim_count]] = 0.5  # Reduction point
    return v_scaled
