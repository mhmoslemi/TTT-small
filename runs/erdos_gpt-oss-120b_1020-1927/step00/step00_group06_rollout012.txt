    # finds lambda via binary search in [lower - max(v), upper]
    # Use lambda_low = -max(v) (so v - lambda_low is huge positive, clip to upper) basically sum = N*upper.
    # lambda_high = max(v) - lower (so v - lambda_high is <= lower) thus x_i = lower -> sum = N*lower.
    # We'll search between these.
    lo = -np.max(v) - 10 # maybe lower bound
    hi = np.max(v) + 10

    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = mid
        else:
            hi = mid
    # Return final

import time
import math
import numpy as np

def _project_onto_capped_simplex(v: np.ndarray, target: float,
                                 lower: float = 0.0, upper: float = 1.0,
                                 max_iter: int = 60, tol: float = 1e-9) -> np.ndarray:
    """
    Projection of v onto the set {x | lower <= x_i <= upper, sum(x) = target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0          # ensures all v - lo >= upper
    hi = np.max(v) + 1.0           # ensures all v - hi <= lower
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, lower, upper)

        N = h.size
    else:
        N = 100                     # fallback size
        h = rng.random(N)
    target_sum = N / 2.0
    h = _project_onto_capped_simplex(h, target_sum)

    # helper to evaluate C5 (same as evaluation harness)
    dx = 2.0 / N

    def _score(vec: np.ndarray) -> float:
        # correlation of h with (1-h), max over all integer shifts
        c = np.correlate(vec, 1.0 - vec, mode='full')
        return float(c.max() * dx)

    best_h = h.copy()
    best_score = _score(best_h)

    # ------------------------------------------------------------------
    # optimisation parameters
    start = time.time()
    limit = budget_s * 0.97           # safety margin
    T0 = best_score * 0.30 + 1e-8     # initial SA temperature
    step_size = 0.5                   # gradient step magnitude
    curr_h = best_h.copy()
    curr_score = best_score
    it = 0
    last_improve_it = 0

    while time.time() - start < limit:
        # decide move type
        if rng.random() < 0.6:
            # ----- gradient move for the current worst shift -----
            corr = np.correlate(curr_h, 1.0 - curr_h, mode='full')
            t_max = int(np.argmax(corr))
            shift = t_max - (N - 1)

            g = np.zeros(N, dtype=np.float64)
            if shift >= 0:
                s = shift
                g[:N - s] += 1.0 - curr_h[s:]
                g[s:] += -curr_h[:N - s]
            else:
                s = -shift
                g[s:] += 1.0 - curr_h[:N - s]
                g[:N - s] += -curr_h[s:]

            norm = np.linalg.norm(g)
            if norm > 0:
                delta = (step_size / norm) * g
                cand = curr_h - delta
            else:
                cand = curr_h.copy()
            cand = _project_onto_capped_simplex(cand, target_sum)
        else:
            # ----- simple pairwise mass transfer -----
            i, j = rng.choice(N, size=2, replace=False)
            if rng.random() < 0.5:
                # move from j to i
                max_delta = min(1.0 - curr_h[i], curr_h[j])
                if max_delta <= 0.0:
                    continue
                delta = rng.random() * max_delta * 0.5
            else:
                # move from i to j
                max_delta = min(curr_h[i], 1.0 - curr_h[j])
                if max_delta <= 0.0:
                    continue
                delta = -rng.random() * max_delta * 0.5
            cand = curr_h.copy()
            cand[i] += delta
            cand[j] -= delta
            # no need to re‑project – sum preserved and bounds respected by construction

        cand_score = _score(cand)

        # ----- simulated‑annealing acceptance -----
        elapsed = time.time() - start
        frac = elapsed / limit
        T = (1.0 - frac) * T0 + 1e-12
        accept = False
        if cand_score < curr_score:
            accept = True
        else:
            if rng.random() < math.exp((curr_score - cand_score) / T):
                accept = True

        if accept:
            curr_h = cand
            curr_score = cand_score
            if cand_score < best_score:
                best_h = cand.copy()
                best_score = cand_score
                step_size = max(step_size * 0.95, 1e-4)
                last_improve_it = it

        # occasional large random perturbation (exploration)
        if it % 1000 == 0 and it > 0:
            sigma = 0.2
            noise = rng.normal(scale=sigma, size=N)
            noise -= noise.mean()
            pert = best_h + noise
            pert = _project_onto_capped_simplex(pert, target_sum)
            pert_score = _score(pert)
            if pert_score < curr_score or rng.random() < math.exp((curr_score - pert_score) / T):
                curr_h = pert
                curr_score = pert_score
                if pert_score < best_score:
                    best_h = pert.copy()
                    best_score = pert_score
                    step_size = max(step_size * 0.95, 1e-4)
                    last_improve_it = it

        # simple restart if stagnant for a long time
        if it - last_improve_it > 5000:
            # re‑initialise around the best point
            sigma = 0.1
            noise = rng.normal(scale=sigma, size=N)
            noise -= noise.mean()
            curr_h = _project_onto_capped_simplex(best_h + noise, target_sum)
            curr_score = _score(curr_h)
            last_improve_it = it

        it += 1
        if best_score < 0.3808:
            break

    # final safety projection
    best_h = _project_onto_capped_simplex(np.clip(best_h, 0.0, 1.0), target_sum)
    final_score = _score(best_h)

    return best_h.tolist(), float(final_score), int(N)
