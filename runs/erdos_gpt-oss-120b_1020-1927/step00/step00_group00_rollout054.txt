def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # Use FFT for correlation.
    size = 2 * n
    # Compute convolution of h and reversed(1-h). Use real FFT.
    a = np.fft.rfft(h, n=size)
    b = np.fft.rfft((1 - h)[::-1], n=size)
    conv = np.fft.irfft(a * b, n=size)
    # conv length = size.
    # correlation length = 2*n - 1 -> indexes 0 to 2n-2; the conv contains indices shifting corresponding to? The convolution result
    # Actually convolution of length n + n - 1 = 2n - 1; using zero padding size=2n gives conv length 2n. The last element corresponds to zero because of padding, we take first (2n-1) entries.
    max_val = conv[:2*n-1].max()
    return max_val * dx

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    size = 2*n
    a_fft = np.fft.rfft(h, n=size)
    b_fft = np.fft.rfft((1 - h)[::-1], n=size)
    conv = np.fft.irfft(a_fft * b_fft, n=size)
    # conv length = size
    max_val = conv[:2*n-1].max()
    return max_val * dx

def project_to_sum_box(v, S):
    # project onto [0,1] box and sum = S.
    # Solve for lambda.
    # Use bisection.
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        total = proj.sum()
        if total > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, 0.0, 1.0)

def propose_move(h, step_scale):
    n = len(h)
    # Choose i, j distinct
    i = np.random.randint(n)
    j = np.random.randint(n-1)
    if j >= i:
        j += 1
    # Determine direction randomly
    if np.random.rand() < 0.5:
        # move from i to j
        max_delta = min(h[i], 1 - h[j])
        if max_delta <= 0:
            return None
        # choose delta
        if np.random.rand() < 0.2:
            delta = max_delta  # full transfer
        else:
            delta = max_delta * step_scale * np.random.rand()
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    else:
        # move from j to i
        max_delta = min(h[j], 1 - h[i])
        if max_delta <= 0:
            return None
        if np.random.rand() < 0.2:
            delta = max_delta
        else:
            delta = max_delta * step_scale * np.random.rand()
        new_h = h.copy()
        new_h[j] -= delta
        new_h[i] += delta

import numpy as np
import time
import math
import random

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    size = 2 * n
    # convolution of h with reversed (1‑h); the max of the linear correlation.
    a = np.fft.rfft(h, n=size)
    b = np.fft.rfft((1.0 - h)[::-1], n=size)
    conv = np.fft.irfft(a * b, n=size)
    max_val = np.max(conv[: 2 * n - 1])
    return max_val * dx


def _project_to_sum_box(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {0 ≤ x ≤ 1, Σx = target_sum}."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    # tiny adjustment if rounding left a small deficit/excess
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        free = np.where((proj > 1e-12) & (proj < 1 - 1e-12))[0]
        if free.size:
            proj[free] += diff / free.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj
