def compute_c5(h, dx):

def random_binary_candidate(n, target_sum, rng):
    h = np.zeros(n, dtype=np.float64)
    int_target = int(np.floor(target_sum))
    if int_target > 0:
        idx = rng.choice(n, int_target, replace=False)
        h[idx] = 1.0
    frac = target_sum - int_target
    if frac > 1e-9:
        # Use half.

        half_idx = rng.choice(remaining, 1)[0]
        h[half_idx] = 0.5

def hillclimb(h, dx, rng, max_trials=2000, max_no_improve=500):
    val = compute_c5(h, dx)
    n = h.size
    no_improve = 0
    for _ in range(max_trials):
        i, j = rng.choice(n, 2, replace=False)
        if h[i] == h[j]:

        else:
