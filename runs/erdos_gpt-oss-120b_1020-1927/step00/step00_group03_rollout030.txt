import time
import numpy as np
import cvxpy as cp
import random

def _correlation_max(h):
    """Maximum of the overlap integral (discrete version)."""
    return int(np.correlate(h, 1 - h, mode="full").max())

def _solve_milp(n, w, time_limit):
    """
    Solve the MILP:
        min M
        s.t. sum_i h_i = w
             for each shift s=1..n-1:
                 C_s = sum_{i=0}^{n-s-1} h_i - sum_{i=0}^{n-s-1} y_{i,s} <= M
                 y_{i,s} = h_i * h_{i+s}  (linearised)
        h_i, y_{i,s} binary.
    Returns (h, M) or (None, None) if unsolved within time_limit.
    """
    # variables
    h = cp.Variable(n, boolean=True)
    M = cp.Variable(integer=True)

    constraints = [cp.sum(h) == w, M >= 0]

    # for each shift create auxiliary y variables and constraints
    for s in range(1, n):
        L = n - s
        if L == 0:
            continue
        y = cp.Variable(L, boolean=True)
        # linearisation of product h[i] * h[i+s]
        constraints += [
            y <= h[:L],
            y <= h[s:],
            y >= h[:L] + h[s:] - 1,
            cp.sum(h[:L]) - cp.sum(y) <= M,
        ]

    prob = cp.Problem(cp.Minimize(M), constraints)

    # try a few solvers in order of speed / availability
    solvers = [
        (cp.CBC, {}),          # COIN-OR CBC
        (cp.GLPK_MI, {}),      # GLPK mixed‑integer
        (cp.ECOS_BB, {}),      # ECOS branch‑and‑bound (fallback)
    ]

    start = time.time()
    for solver, opts in solvers:
        try:
            prob.solve(solver=solver, **opts, warm_start=True)
        except (cp.error.SolverError, Exception):
            continue
        # stop if we have a feasible solution within the time budget
        if prob.status in (cp.OPTIMAL, cp.FEASIBLE):
            break
        if time.time() - start > time_limit:
            break

    # if we didn't finish optimally, return None
    if prob.status not in (cp.OPTIMAL, cp.FEASIBLE):
        return None, None

    h_val = np.rint(np.array(h.value)).astype(int)
    M_val = int(round(M.value))
    return h_val, M_val


def _local_search(h, max_iter, time_budget, best_cmax):
    """Simple hill‑climbing / simulated‑annealing improvement."""
    n = len(h)
    w = h.sum()
    dx = 2.0 / n
    start = time.time()
    cur = h.copy()
    cur_cmax = best_cmax

    # pre‑compute indices of ones and zeros
    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]

    # simple temperature schedule
    T0 = 1.0
    for it in range(max_iter):
        if time.time() - start > time_budget:
            break
        # pick a random shift that currently attains the max
        corr = np.correlate(cur, 1 - cur, mode="full")
        max_val = int(corr.max())
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)   # signed shift value

        # collect mismatched positions for this shift
        if shift >= 0:
            L = n - shift
            left = cur[:L]
            right = cur[shift:]
            mismatched = np.where((left == 1) & (right == 0))[0]
            opposite   = np.where((left == 0) & (right == 1))[0]
        else:
            s = -shift
            L = n - s
            left = cur[s:]
            right = cur[:L]
            mismatched = np.where((left == 1) & (right == 0))[0] + s
            opposite   = np.where((left == 0) & (right == 1))[0] + s

        improved = False
        # try a few targeted swaps
        if mismatched.size and opposite.size:
            for _ in range(3):
                i = int(np.random.choice(mismatched))
                j = int(np.random.choice(opposite))
                if i == j:
                    continue
                cur[i], cur[j] = cur[j], cur[i]
                new_cmax = int(np.correlate(cur, 1 - cur, mode="full").max())
                if new_cmax <= max_val:
                    cur_cmax = new_cmax
                    improved = True
                    break
                # revert
                cur[i], cur[j] = cur[j], cur[i]

        if not improved:
            # random swap (maintains weight)
            i = int(np.random.choice(np.where(cur == 1)[0]))
            j = int(np.random.choice(np.where(cur == 0)[0]))
            cur[i], cur[j] = cur[j], cur[i]
            new_cmax = int(np.correlate(cur, 1 - cur, mode="full").max())
            # accept if not worse, otherwise revert with Metropolis rule
            if new_cmax <= cur_cmax:
                cur_cmax = new_cmax
            else:
                # simulated‑annealing acceptance
                T = T0 * (1 - it / max_iter)
                if np.random.rand() < np.exp(-(new_cmax - cur_cmax) / max(T, 1e-8)):
                    cur_cmax = new_cmax
                else:
                    cur[i], cur[j] = cur[j], cur[i]

        # keep best found so far
        if cur_cmax < best_cmax:
            best_cmax = cur_cmax
            best_h = cur.copy()
        # occasionally reshuffle to escape deep local minima
        if it % 500 == 0 and random.random() < 0.01:
            np.random.shuffle(cur)
            cur_cmax = int(np.correlate(cur, 1 - cur, mode="full").max())

    return best_h, best_cmax


def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where
    - h_values  : numpy array of step heights in [0,1]
    - c5_bound  : the achieved upper bound on C₅
    - n_points  : number of discretisation points (length of h_values)
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine problem size
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
        # enforce exact weight (might be already correct)
        w_target = n // 2
        # if the provided vector does not meet the weight, adjust by simple balancing
        cur_weight = int(round(h0.sum()))
        if cur_weight != w_target:
            # force a binary vector with correct weight as a fallback
            h = np.zeros(n, dtype=int)
            h[:w_target] = 1
            np.random.shuffle(h)
        else:
            # keep the supplied vector (it may be fractional; we will binarise)
            h = (h0 > 0.5).astype(int)
    else:
        # default size (small enough for fast optimisation)
        n = 70
        w_target = n // 2
        h = np.zeros(n, dtype=int)
        h[:w_target] = 1
        np.random.shuffle(h)

    w_target = n // 2
    dx = 2.0 / n

    start_time = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start_time))

    # ------------------------------------------------------------------
    # Try to solve the exact MILP first (fast for n~70)
    # ------------------------------------------------------------------
    milp_time = min(30.0, time_left() * 0.5)   # give at most 30 s or half the budget
    h_opt, M_opt = _solve_milp(n, w_target, milp_time)

    if h_opt is not None:
        # exact solution found
        c5_bound = M_opt * dx
        h_opt = h_opt.astype(float)
        return (h_opt, c5_bound, n)

    # ------------------------------------------------------------------
    # Fallback: heuristic local search starting from the current vector
    # ------------------------------------------------------------------
    best_h = h.astype(int)
    best_cmax = _correlation_max(best_h)

    # allocate remaining time between several restarts
    while time_left() > 5.0:
        # perform a local search run
        max_iter = 2000
        time_budget = min(time_left() - 0.5, 5.0)
        candidate_h, candidate_cmax = _local_search(
            best_h.copy(),
            max_iter,
            time_budget,
            best_cmax,
        )
        if candidate_cmax < best_cmax:
            best_cmax = candidate_cmax
            best_h = candidate_h.copy()
        # random restart occasionally
        if random.random() < 0.2:
            rnd = np.zeros(n, dtype=int)
            rnd[:w_target] = 1
            np.random.shuffle(rnd)
            best_h = rnd
            best_cmax = _correlation_max(best_h)

    c5_bound = best_cmax * dx
    return (best_h.astype(float), c5_bound, n)
