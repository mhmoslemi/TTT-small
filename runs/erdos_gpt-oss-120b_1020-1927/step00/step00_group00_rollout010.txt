### Move selection
- Choose two distinct indices i,j uniformly random.

- Determine max positive transfer possible: delta_max_pos = min(1 - h_j, h_i). That's the amount we can move from i to j (i decreases, j increases). We choose sign randomly; then choose δ uniformly from [0, step] where step is min(delta_max_pos, step) for positive, or similar for negative.

def propose_move(h, step):
    i, j = np.random.choice(N, size=2, replace=False)
    if np.random.rand() < 0.5:
        max_delta = min(step, h[i], 1 - h[j])
        if max_delta <= 1e-12: return None
        delta = np.random.rand() * max_delta
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    else:
        max_delta = min(step, h[j], 1 - h[i])
        if max_delta <= 1e-12: return None
        delta = np.random.rand() * max_delta
        new_h = h.copy()
        new_h[j] -= delta
        new_h[i] += delta
    return new_h

def propose_gradient_move(h, step):
    N = len(h)
    # compute convolution correlation to find worst lag
    corr = np.correlate(h, 1 - h, mode='full')
    lag_index = np.argmax(corr)  # index from 0 to 2*N-2
    L = lag_index - (N - 1) # shift amount; could be negative
    # use absolute shift? The correlation is symmetric: max may be at positive shift. use L's magnitude.
    # Determine neighbor sums for each i
    neighbor_sum = np.zeros(N)
    if L >= 0:
        # i+L valid for i in [0,N-1-L]
        neighbor_sum[:N-L] += h[L:]  # i+L term: for i from 0 to N-1-L, neighbor = h[i+L]
        neighbor_sum[L:] += h[:N-L]  # i-L term: for i from L to N-1, neighbor = h[i-L]
    else:
        # L negative => shift left
        L_pos = -L
        # i+L valid for i in [L_pos,N-1]
        neighbor_sum[L_pos:] += h[:N-L_pos]
        # i-L valid for i in [0,N-1-L_pos]
        neighbor_sum[:N-L_pos] += h[L_pos:]
    derivative = 1.0 - neighbor_sum
    # choose donor indices where derivative > 0 (neighbor_sum <1) and h > 0
    donor_candidates = np.where((derivative > 0) & (h > 0))[0]
    receiver_candidates = np.where((derivative < 0) & (h < 1))[0]
    if donor_candidates.size == 0 or receiver_candidates.size == 0:
        # fallback random move
        return propose_move(h, step)
    i = np.random.choice(donor_candidates)
    j = np.random.choice(receiver_candidates)
    # maximum delta: cannot make h_i negative or h_j >1
    max_delta = min(step, h[i], 1 - h[j])
    if max_delta <= 0:
        return None
    delta = np.random.rand() * max_delta
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

import time, math, random
import numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def _score_and_corr(h):
    """Return (C5 bound, full correlation array)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / h.size
    return corr.max() * dx, corr


def _propose_move(h, max_delta):
    """Transfer a small amount from one index to another, preserving sum."""
    N = h.size
    i, j = np.random.choice(N, 2, replace=False)
    # direction: i -> j  (i loses, j gains)  or opposite
    if np.random.rand() < 0.5:
        # i -> j
        delta = min(max_delta, h[i], 1.0 - h[j])
        if delta <= 0:
            return None
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        return new_h
    else:
        # j -> i
        delta = min(max_delta, h[j], 1.0 - h[i])
        if delta <= 0:
            return None
        new_h = h.copy()
        new_h[j] -= delta
        new_h[i] += delta
        return new_h


def _propose_gradient_move(h, corr, max_delta):
    """Move mass guided by the worst lag (max correlation)."""
    N = h.size
    # shift corresponding to the maximal correlation entry
    shift = corr.argmax() - (N - 1)          # can be negative
    # compute sum of neighbours at distance |shift|
    neigh_sum = np.zeros(N, dtype=float)
    if shift >= 0:
        neigh_sum[: N - shift] += h[shift:]
        neigh_sum[shift:] += h[: N - shift]
    else:
        s = -shift
        neigh_sum[s:] += h[: N - s]
        neigh_sum[: N - s] += h[s:]

    deriv = 1.0 - neigh_sum                 # dS/dh_i for the worst lag
    donors = np.where((deriv > 0) & (h > 1e-12))[0]
    recvs = np.where((deriv < 0) & (h < 1.0 - 1e-12))[0]

    if donors.size == 0 or recvs.size == 0:
        return _propose_move(h, max_delta)

    i = np.random.choice(donors)
    j = np.random.choice(recvs)
    delta = min(max_delta, h[i], 1.0 - h[j])
    if delta <= 0:
        return None
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def _perturb(h, n_moves, max_delta):
    """Apply a few random transfers to escape a plateau."""
    out = h.copy()
    for _ in range(n_moves):
        cand = _propose_move(out, max_delta)
        if cand is not None:
            out = cand
    return out


def _optimize_local(h0, time_budget, start_time):
    """Hill‑climbing / simulated‑annealing on a fixed grid size."""
    N = h0.size
    cur_h = h0.copy()
    cur_score, cur_corr = _score_and_corr(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    step = 0.5               # maximal transferred amount
    stagn = 0                # iterations without improvement
    it = 0

    while time.time() - start_time < time_budget:
        # adapt step size slowly
        if it and it % 3000 == 0:
            step = max(1e-4, step * 0.9)

        # mostly gradient‑guided moves
        if np.random.rand() < 0.7:
            cand = _propose_gradient_move(cur_h, cur_corr, step)
        else:
            cand = _propose_move(cur_h, step)

        if cand is None:
            continue

        cand_score, cand_corr = _score_and_corr(cand)

        # Accept only if not worse (simple greedy)
        if cand_score < cur_score - 1e-12:
            cur_h, cur_score, cur_corr = cand, cand_score, cand_corr
            stagn = 0
            if cand_score < best_score - 1e-12:
                best_h, best_score = cand.copy(), cand_score
        else:
            stagn += 1

        # Restart after a long stall
        if stagn > 5000:
            cur_h = _perturb(best_h, n_moves=12, max_delta=step)
            cur_score, cur_corr = _score_and_corr(cur_h)
            stagn = 0

        it += 1

    return best_h, best_score
