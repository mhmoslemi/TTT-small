import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def compute_c5(h):
    """C5 value for a step vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _project_box_sum(v, target):
    """
    Project vector v onto the box [0,1] with the linear equality
    sum(v) = target (exact to machine precision).
    Uses a binary search on the Lagrange multiplier.
    """
    # Clip once – the binary search will keep the result inside the box.
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def gradient_shift(h, k):
    """Sub‑gradient of the overlap for a given non‑negative shift k."""
    n = h.shape[0]
    if k < 0:
        k = -k
    mask = np.arange(n) <= n - 1 - k
    left = np.zeros(n)
    if k < n:
        left[: n - k] = h[k:]
    right = np.zeros(n)
    if k < n:
        right[k:] = h[: n - k]
    return mask.astype(float) - left - right


def _optimize_once(h, time_limit, target_c5=None):
    """
    Gradient‑descent on the worst shift.
    Returns the best vector seen during the allocated time.
    """
    n = h.shape[0]
    target_sum = n * 0.5
    h = _project_box_sum(h, target_sum)

    best_h = h.copy()
    best_val = compute_c5(best_h)

    lr = 0.1
    min_lr = 1e-7
    start = time.time()

    while time.time() - start < time_limit:
        # current objective and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        cur_val = corr[idx] * (2.0 / n)
        if cur_val < best_val:
            best_val, best_h = cur_val, h.copy()
            if target_c5 is not None and best_val <= target_c5:
                break

        shift = idx - (n - 1)
        g = gradient_shift(h, shift)

        # trial step
        hn = _project_box_sum(h - lr * g, target_sum)
        corr_n = np.correlate(hn, 1.0 - hn, mode="full")
        val_n = np.max(corr_n) * (2.0 / n)

        if val_n <= cur_val:
            h = hn
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.5
            if lr < min_lr:
                # small random shake to escape possible stall
                noise = np.random.uniform(-0.02, 0.02, size=n)
                h = _project_box_sum(h + noise, target_sum)
                lr = 0.1
    return best_h, best_val


def upscale(h, factor):
    """Linear interpolation to a finer grid (factor must be integer)."""
    n = h.shape[0]
    new_n = n * factor
    old_x = np.linspace(0.0, 2.0, n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, h)
    return _project_box_sum(new_h, new_n * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points).

    h_values – 1‑D array with values in [0,1] and sum = n_points/2,
    c5_bound – the computed C5 upper bound,
    n_points – length of the discretisation.
    """
    np.random.seed(seed)
    start_all = time.time()

    # ------------------------------------------------------------------
    # initialise a starting vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = h0.shape[0]
        h0 = _project_box_sum(h0, n0 * 0.5)
    else:
        # default: 84 points, simple linear ramp (already satisfies the integral)
        n0 = 84
        h0 = np.linspace(1.0, 0.0, n0, dtype=float)
        h0 = _project_box_sum(h0, n0 * 0.5)

    # ------------------------------------------------------------------
    # first optimisation stage on the initial resolution
    # ------------------------------------------------------------------
    elapsed = time.time() - start_all
    remaining = max(budget_s - elapsed, 0.0)
    stage_time = remaining * 0.55  # keep reserve for later up‑scaling
    best_h, best_val = _optimize_once(h0, stage_time, target_c5=0.3808)

    # ------------------------------------------------------------------
    # optional up‑scaling steps while time permits
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start_all
        remain = budget_s - elapsed
        if remain < 20.0:
            break
        # limit to a reasonable size (< 1000 points)
        if best_h.shape[0] * 2 > 1000:
            break
        # prepare finer grid
        finer = upscale(best_h, 2)
        # allocate a fraction of the remaining budget
        stage_time = remain * 0.45
        best_h, best_val = _optimize_once(finer, stage_time, target_c5=0.3808)

    # ------------------------------------------------------------------
    # final evaluation (uses the same routine as the checker)
    # ------------------------------------------------------------------
    try:
        c5_bound = evaluate_erdos_solution(best_h)  # pre‑imported helper
    except Exception:  # fallback if the helper is unavailable
        c5_bound = compute_c5(best_h)

    return best_h, float(c5_bound), best_h.shape[0]
