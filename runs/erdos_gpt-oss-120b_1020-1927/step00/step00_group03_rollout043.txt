def softmax_gradient(h, tau=0.1):
    c = np.correlate(h, 1-h, mode='full')
    shifts = np.arange(-len(h)+1, len(h))
    # compute weights
    max_c = np.max(c)
    # subtract max for numerical stability
    w = np.exp((c - max_c) / tau)
    w_sum = w.sum()
    # gradient is sum over L of (w / w_sum) * grad_corr(L)

    for idx, L in enumerate(shifts):
        weight = w[idx] / w_sum
        if weight == 0:
            continue
        g += weight * gradient_of_corr(h, L)
    return g
