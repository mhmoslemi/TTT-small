import time
import numpy as np

def _project_box_sum(y: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project y onto the set {x | 0 <= x_i <= 1, sum(x) = target_sum}
    using a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(y) - 1.0       # ensures all entries are clipped to 1 (sum = n)
    hi = np.max(y) + 1.0        # ensures all entries are clipped to 0 (sum = 0)
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)


def _subgradient_max_overlap(h: np.ndarray) -> np.ndarray:
    """
    Return a subgradient of   f(h) = max_k  sum_i h_i (1 - h_{i+k})
    at the current point h. The subgradient is taken from the shift k
    that attains the current maximum.
    """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(corr.argmax())
    L = idx - (n - 1)                 # shift corresponding to the worst overlap

    g = np.ones_like(h)               # derivative of the term Σ h_i
    if L >= 0:
        # i+L in range
        g[: n - L] -= h[L:]
        # i-L in range
        g[L:] -= h[: n - L]
    else:
        L = -L
        # i+L  (i-L) in range
        g[L:] -= h[: n - L]
        g[: n - L] -= h[L:]

    # factor dx does not affect the subgradient direction
    return g, corr.max() * dx, L


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # choose number of points – 200 is a good trade‑off
    n = 200
    target_sum = n / 2.0
    dx = 2.0 / n

    # initialise h ---------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float).ravel()
        if h0.size != n:
            # resample to required size by simple linear interpolation
            x_old = np.linspace(0, 2, h0.size, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, h0)
        else:
            h = h0.copy()
        # enforce exact sum and box constraints
        h = np.clip(h, 0.0, 1.0)
        h = _project_box_sum(h, target_sum)
    else:
        # random binary start (exact sum)
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, size=n // 2, replace=False)
        h[ones] = 1.0

    best_h = h.copy()
    best_score = np.inf

    # subgradient descent --------------------------------------------------
    alpha0 = 0.8          # initial step size
    iter_cnt = 0
    while True:
        # time check
        if time.time() - start > budget_s * 0.85:
            break

        g, cur_score, worst_shift = _subgradient_max_overlap(h)
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()

        # diminishing step size
        step = alpha0 / np.sqrt(iter_cnt + 1.0)
        h_new = h - step * g
        h = _project_box_sum(h_new, target_sum)

        iter_cnt += 1

    # occasional random perturbations to escape possible plateaus
    while time.time() - start < budget_s * 0.97:
        # jitter around the best solution found so far
        noise = rng.normal(scale=0.03, size=n)
        cand = best_h + noise
        cand = np.clip(cand, 0.0, 1.0)
        cand = _project_box_sum(cand, target_sum)

        corr = np.correlate(cand, 1.0 - cand, mode="full")
        cand_score = corr.max() * dx
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand.copy()

    # final bound computation (exact)
    final_corr = np.correlate(best_h, 1.0 - best_h, mode="full")
    final_score = final_corr.max() * dx

    return best_h.tolist(), float(final_score), n
