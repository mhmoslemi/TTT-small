def project_box_simplex(v, target_sum, tol=1e-9, max_iter=100):
    low = -1.0
    high = 2.0
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) <= tol:
            return h
        if s > target_sum:
            low = lam    # need larger lambda to reduce sum (since subtract lambda more)
        else:
            high = lam
    # return final
    return np.clip(v - (low+high)/2.0, 0.0, 1.0)

def project_box_simplex(v, target_sum, tol=1e-9, max_iter=100):
    low = -1.0
    high = 2.0
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) <= tol:
            return h
        if s > target_sum:
            low = lam

            high = lam

import time
import math
import numpy as np

def _cmax(h):
    """Maximum overlap integral for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) in full mode
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _proj_box_simplex(v, target, tol=1e-12, max_iter=60):
    """
    Euclidean projection of vector v onto the set {h | 0 <= h_i <= 1, sum h = target}.
    Implements the water‐filling (dual) algorithm by binary searching the dual scalar.
    """
    low, high = -1.0, 2.0  # bounds that certainly contain the optimal multiplier
    for _ in range(max_iter):
        lam = (low + high) * 0.5
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) <= tol:
            return h
        # sum decreases monotonically with increasing lam
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(v - (low + high) * 0.5, 0.0, 1.0)

def _gradient_worst(h, shift):
    """Sub‑gradient of the worst correlation C(shift) w.r.t. h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        # forward shift
        i1 = np.arange(0, n - shift)
        g[i1] += (1.0 - h[i1 + shift])
        i2 = np.arange(shift, n)
        g[i2] -= h[i2 - shift]
    else:
        s = -shift
        # backward shift (equivalent to forward shift with sign flipped)
        i1 = np.arange(s, n)
        g[i1] += (1.0 - h[i1 - s])
        i2 = np.arange(0, n - s)
        g[i2] -= h[i2 + s]
    return g

def _subgradient_optimize(h0, time_limit):
    """
    Simple projected sub‑gradient descent on max_{k} C(k).
    Returns the best vector encountered and its C‑value.
    """
    n = h0.shape[0]
    target_sum = n / 2.0
    dx = 2.0 / n

    h = h0.copy()
    best_h = h.copy()
    best_val = _cmax(best_h)

    it = 0
    t_start = time.time()
    while time.time() - t_start < time_limit:
        # current correlation and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        idx_max = int(np.argmax(corr))
        cur_val = float(corr[idx_max])
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            # early exit if we already beat the target bound
            if best_val <= 0.3808:
                break

        shift = idx_max - (n - 1)            # integer shift corresponding to idx_max
        grad = _gradient_worst(h, shift)     # sub‑gradient of the worst term

        # diminishing step size
        step = 0.7 / math.sqrt(it + 1.0)

        # gradient step and projection
        h_candidate = _proj_box_simplex(h - step * grad, target_sum)

        # evaluate candidate
        cand_val = _cmax(h_candidate)

        # simple acceptance rule (always accept improvements)
        if cand_val <= cur_val or np.random.rand() < 0.001:
            h = h_candidate

        it += 1

        # occasional random perturbation to escape possible stagnation
        if it % 5000 == 0:
            if np.random.rand() < 0.2:
                # inject small random noise while keeping feasibility
                noise = (np.random.rand(n) - 0.5) * 0.04
                h_noise = np.clip(h + noise, 0.0, 1.0)
                h = _proj_box_simplex(h_noise, target_sum)

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimises a step function h on [0,2] satisfying
        0 <= h <= 1,    sum(h) = n/2
    to minimise   C_5 = max_k ∫ h(x)(1-h(x+k))dx .
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    # ------------------------------------------------------------------

        h0 = np.full(n, 0.5, dtype=np.float64)
