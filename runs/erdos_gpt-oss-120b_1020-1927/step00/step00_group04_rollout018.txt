    n = len(a)
    grad = np.ones_like(a)
    if k >= 0:
        # subtract a_{j+k} for j in [0, n-1-k]

    h = init_h.copy()
    n = len(h)
    target_sum = n/2
    best_h = h
    best_val = evaluate(h)

        # compute correlation

        max_corr = corr.max()
        # early stopping if no improvement after many iterations?

        # project onto feasible region
        h_new = project(h_new)
        h = h_new

            best_h = h.copy()

def project_to_feasible(a):
    # a: array
    n = a.shape[0]
    target = n/2
    a = np.clip(a, 0.0, 1.0)
    # adjust sum to target
    # water-filling algorithm
    # We'll iterate up to few times
    # At each iteration compute diff = target - sum(a)
    # If diff is zero within tolerance, break.
    # Determine 'free' indices where a is not at bounds that can accept more (if diff>0) or lose more (if diff<0)
    # If diff>0: we need to add to values <1. If diff<0: we need to subtract from values >0.
    # delta = diff / num_free
    # Adjust and clip again.
    max_iter = 1000
    for _ in range(max_iter):
        current_sum = a.sum()
        diff = target - current_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = a < 1.0 - 1e-12
        else:
            free = a > 1e-12
        num_free = np.count_nonzero(free)
        if num_free == 0:
            # can't adjust, break
            break
        delta = diff / num_free
        a[free] += delta
        # clip after adjustment
        a = np.clip(a, 0.0, 1.0)
    # after final ensure sum close:
    # final scaling to target if still off due to numerical errors (should be small)
    if abs(a.sum() - target) > 1e-6:
        # final fix: add uniform offset then clip
        offset = (target - a.sum()) / n
        a += offset
        a = np.clip(a, 0.0, 1.0)
        # If still off, just accept if diff small
    return a

def grad_for_shift(h, k):
    n = len(h)
    grad = np.ones_like(h)
    if k >= 0:
        if k != 0:
            grad[:n - k] -= h[k:]  # subtract a_{j+k}
            grad[k:] -= h[:n - k]  # subtract a_{j-k}
        else:
            grad -= 2 * h
    else:
        kk = -k
        grad[kk:] -= h[:n - kk]
        grad[:n - kk] -= h[kk:]
    return grad
```

def optimize_h(start_h, max_iters, step0, time_limit):
    h = start_h.copy()
    n = len(h)
    best = h.copy()
    best_val = evaluate(best)
    start_time = time.time()
    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = corr.max()
        # compute shifts where near max
        idxs = np.where(corr >= max_val - 1e-12)[0]
        # gradient sum
        grad = np.zeros_like(h)
        for idx in idxs:
            k = idx - (n - 1)
            grad += grad_for_shift(h, k)
        grad /= len(idxs)
        # step schedule
        step = step0 * (0.99 ** i)
        h = h - step * grad
        h = project(h)
        val = evaluate(h)
        if val < best_val - 1e-12:
            best_val = val
            best = h.copy()
    return best, best_val
```

def project_to_feasible(a):
    n = a.shape[0]
    target = n / 2.0
    a = np.clip(a, 0.0, 1.0)
    # Simple water-filling iterative.
    for _ in range(100):  # iteration limit
        current_sum = a.sum()
        diff = target - current_sum
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            free = a < 1.0 - 1e-12
        else:
            free = a > 1e-12
        num_free = np.count_nonzero(free)
        if num_free == 0:
            # cannot adjust further
            break
        delta = diff / num_free
        a[free] += delta
        a = np.clip(a, 0.0, 1.0)
    # final correction: if still off due to numerical, adjust uniformly small
    err = target - a.sum()
    if abs(err) > 1e-8:
        a += err / n
        a = np.clip(a, 0.0, 1.0)
    # ensure sum exactly target within tiny tolerance
    return a

def evaluate(h: np.ndarray) -> float:
    """Return the C5 value for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx


def grad_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the correlation term for shift k."""
    n = h.shape[0]
    grad = np.ones_like(h)
    if k == 0:
        grad -= 2.0 * h
        return grad
    if k > 0:
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:                         # k < 0
        kk = -k
        grad[kk:] -= h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return grad


def project_to_feasible(a: np.ndarray) -> np.ndarray:
    """Project a vector onto [0,1]^n with sum_i a_i = n/2."""
    n = a.shape[0]
    target = n * 0.5
