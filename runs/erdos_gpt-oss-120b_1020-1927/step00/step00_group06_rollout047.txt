def project(v, target):
    lo = -np.max(v) - 1  # ensure v+lo <= -1, all clipped to 0
    hi = 1 - np.min(v) + 1  # ensure v+hi >= 2, all clipped to 1
    for _ in range(50):
        mid = (lo + hi) / 2
        w = np.clip(v + mid, 0, 1)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo+hi)/2, 0, 1)
    # In case of rounding errors, adjust slight scaling:
    # Compute residual
    res = target - w.sum()
    if abs(res) > 1e-8:
        # distribute residual among non-saturated entries
        free = (w > 0) & (w < 1)
        cnt = free.sum()
        if cnt > 0:
            w[free] += res / cnt
            w = np.clip(w, 0, 1)  # saturate again
    return w

import time
import math
import numpy as np

def _project_onto_bounds_and_sum(v, target):
    """
    Project vector v onto the set {w | 0 <= w_i <= 1, sum(w) = target}
    using a simple binary search on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0      # guarantees all clipped to 0
    hi = 1.0 - np.min(v) + 1.0 # guarantees all clipped to 1
    for _ in range(50):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for possible rounding error
    resid = target - w.sum()
    if abs(resid) > 1e-12:
        free = (w > 0.0) & (w < 1.0)
        cnt = free.sum()
        if cnt > 0:
            w[free] += resid / cnt
            w = np.clip(w, 0.0, 1.0)
        else:
            # fallback – uniform scaling
            w *= target / w.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

def _c5_and_shift(vec):
    """
    Returns (C5 value, shift achieving the max, raw max correlation).
    """
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    max_corr = np.max(corr)
    shift = int(np.argmax(corr) - (len(vec) - 1))
    c5 = max_corr * (2.0 / len(vec))
    return c5, shift, corr

def _gradient_shift(vec, shift):
    """
    Gradient of the overlap term for a given shift.
    d/dh_j [ sum_i h_i (1 - h_{i+shift}) ] = 1 - h_{j+shift} - h_{j-shift}
    (terms outside the range are omitted).
    """
    n = len(vec)
    g = np.ones(n)
    if shift >= 0:
        k = shift
        if k < n:
            g[:n - k] -= vec[k:]
            g[k:]     -= vec[:n - k]
    else:
        k = -shift
        if k < n:
            g[k:]     -= vec[:n - k]
            g[:n - k] -= vec[k:]
    return g

    # ------------------------------------------------------------------
    final_c5, _, _ = _c5_and_shift(best_h)
    return best_h.tolist(), float(final_c5), n
