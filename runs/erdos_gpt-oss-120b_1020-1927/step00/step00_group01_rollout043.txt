import numpy as np, time

def max_corr(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx] * dx
    lag = max_idx - (len(h) - 1)
    return max_val, lag

def compute_grad(h, lag, dx):
    N = h.size
    if lag >= 0:
        term1 = np.concatenate([h[lag:], np.zeros(lag)])
        term2 = np.concatenate([np.zeros(lag), h[:N-lag]])
    else:
        l = -lag
        term1 = np.concatenate([np.zeros(l), h[:N-l]])
        term2 = np.concatenate([h[l:], np.zeros(l)])
    return dx * (1 - term1 - term2)

def proj_box_sum(v, target, max_iter=60):
    lo = -1.0
    hi = 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
```

import numpy as np, time, math

def _proj_box_sum(v, target, iters=60):
    """Project v onto the box [0,1] with sum(v)=target."""
    lo, hi = -1.0, 1.0
    for _ in range(iters):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _max_corr_and_lag(h, dx):
    """Return (max_value, lag) for the overlap objective."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = corr[idx] * dx
    lag = idx - (len(h) - 1)
    return max_val, lag


def _grad_for_lag(h, lag, dx):
    """Subgradient of the max‑lag term."""
    n = h.size
    if lag >= 0:
        term1 = np.concatenate([h[lag:], np.zeros(lag)])
        term2 = np.concatenate([np.zeros(lag), h[: n - lag]])
    else:
        l = -lag
        term1 = np.concatenate([np.zeros(l), h[: n - l]])
        term2 = np.concatenate([h[l:], np.zeros(l)])
    return dx * (1.0 - term1 - term2)


def _subgradient_step(h, lag, dx, step):
    """One projected subgradient step."""
    g = _grad_for_lag(h, lag, dx)
    target = h.size * 0.5
    return _proj_box_sum(h - step * g, target)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Find a step function h on [0,2] with integral 1 that minimizes
    max_k ∫ h(x)(1−h(x+k))dx.
    Returns (h_array, bound_value, n_points).
    """
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise problem size and a feasible start vector
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)  # provided parent, if any
    except Exception:
        h0 = None

    if h0 is None:
        N = 100                     # default resolution
        h_cur = np.full(N, 0.5)     # constant 0.5 (feasible)
    else:
        N = h0.size
        # up‑sample parent if we later change N, otherwise keep it
        h_cur = h0.copy()

    target_sum = N * 0.5
    h_cur = _proj_box_sum(h_cur, target_sum)
    dx = 2.0 / N

    # ------------------------------------------------------------------
    # keep the best solution seen
    # ------------------------------------------------------------------
    cur_val, cur_lag = _max_corr_and_lag(h_cur, dx)
    best_h = h_cur.copy()
    best_val = cur_val

    # ------------------------------------------------------------------
    # simulated annealing phase (fast, cheap for N≈100)
    # ------------------------------------------------------------------
    T = 0.05                     # initial temperature
    decay = 0.9999               # exponential cooling
    max_iters = int(5e5)         # generous upper bound
    it = 0
    while time.time() - start < budget_s - 1.0 and it < max_iters:
        # propose a transfer of mass between two coordinates
        i, j = np.random.choice(N, 2, replace=False)

        # direction: try to move from j to i
        inc_i = 1.0 - h_cur[i]    # how much i can increase
        dec_j = h_cur[j]          # how much j can give
        transfer = min(inc_i, dec_j)

        if transfer <= 0.0:
            # try opposite direction
            inc_i = h_cur[i]
            dec_j = 1.0 - h_cur[j]
            transfer = min(inc_i, dec_j)
            if transfer <= 0.0:
                it += 1
                continue
            delta = np.random.rand() * transfer
            h_new = h_cur.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            delta = np.random.rand() * transfer
            h_new = h_cur.copy()
            h_new[i] += delta
            h_new[j] -= delta

        # evaluate new state
        new_val, new_lag = _max_corr_and_lag(h_new, dx)

        # accept if better or with Metropolis probability
        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
            h_cur, cur_val, cur_lag = h_new, new_val, new_lag

        # keep global best
        if cur_val < best_val:
            best_val, best_h = cur_val, h_cur.copy()

        # temperature schedule
        T *= decay
        it += 1

    # ------------------------------------------------------------------
    # refinement with projected subgradient descent from the best point
    # ------------------------------------------------------------------
    h_ref = best_h.copy()
    cur_val, cur_lag = _max_corr_and_lag(h_ref, dx)

    best_h = _proj_box_sum(best_h, target_sum)
    final_val, _ = _max_corr_and_lag(best_h, dx)
