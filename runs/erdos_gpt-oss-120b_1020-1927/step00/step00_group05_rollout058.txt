import numpy as np, time

def project_capped_simplex(v, total):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > total:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return w

def compute_c_max(h, dx):
    cor = np.correlate(h, 1.0 - h, mode='full')
    return cor.max() * dx, cor

def compute_gradient(h, shift, dx):
    N = len(h)
    if shift >= 0:
        ip = np.empty(N, dtype=h.dtype)
        ip[:N - shift] = h[shift:]
        ip[N - shift:] = 0.0
        im = np.empty(N, dtype=h.dtype)
        im[:shift] = 0.0
        im[shift:] = h[:N - shift]
    else:
        s = -shift
        ip = np.empty(N, dtype=h.dtype)
        ip[:s] = 0.0
        ip[s:] = h[:N - s]
        im = np.empty(N, dtype=h.dtype)
        im[:N - s] = h[s:]
        im[N - s:] = 0.0
    grad = dx * (1.0 - ip - im)  # gradient of C(s)
    return grad
```

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random
    np.random.seed(seed)
    start = time.time()
    N = 200  # maybe adjust based on budget
    dx = 2.0 / N
    total = N / 2.0  # sum required
    # initial h: triangular shape
    x = (np.arange(N) + 0.5) * dx
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    # ensure sum constraint via projection (should already satisfy)
    h = project_capped_simplex(h, total)
    best_h = h.copy()
    best_c, _ = compute_c_max(h, dx)
    # parameters
    lr0 = 0.5  # step size
    it = 0
    no_improve = 0
    while time.time() - start < budget_s - 0.5:
        cmax, cor = compute_c_max(h, dx)
        # update best
        if cmax < best_c:
            best_c = cmax
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # find shift achieving max
        s_idx = np.argmax(cor)
        shift = s_idx - (N - 1)  # integer shift
        grad = compute_gradient(h, shift, dx)
        # adaptive learning rate decreasing with iteration
        lr = lr0 / np.sqrt(it + 1)
        # gradient descent step
        h_new = h - lr * grad
        # project
        h_new = project_capped_simplex(h_new, total)
        h = h_new
        it += 1
        # occasional random perturb to escape
        if no_improve > 2000:
            # restart with small noise around best
            noise = np.random.uniform(-0.05, 0.05, size=N)
            h = np.clip(best_h + noise, 0.0, 1.0)
            h = project_capped_simplex(h, total)
            no_improve = 0
    # final compute best_c bound
    final_c, _ = compute_c_max(best_h, dx)
    return best_h, final_c, N

import time
import numpy as np

def _project_capped_simplex(v: np.ndarray, total: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set
        { x : 0 <= x_i <= 1, sum_i x_i = total }.
    Uses the fact that the projection has the form
        x_i = clip(v_i - λ, 0, 1)
    for a suitable λ.  λ is found by binary search.
    """
    lo = v.min() - 1.0          # makes all clip(v-λ)=1  -> sum = len(v)
    hi = v.max()                # makes all clip(v-λ)=0  -> sum = 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > total:
            lo = mid
        else:
            hi = mid
    λ = (lo + hi) * 0.5
    return np.clip(v - λ, 0.0, 1.0)


def _c_max(h: np.ndarray, dx: float):
    """return (max_overlap_integral, full correlation array)"""
    cov = np.correlate(h, 1.0 - h, mode="full")
    return cov.max() * dx, cov


def _gradient_for_shift(h: np.ndarray, shift: int, dx: float):
    """
    Gradient of C(shift) = sum_i h_i (1 - h_{i+shift}) dx
    w.r.t. each component of h.
    """
    N = h.shape[0]
    if shift >= 0:
        # h[i+shift] (zero when out of range)
        hi = np.empty(N, dtype=h.dtype)
        hi[: N - shift] = h[shift:]
        hi[N - shift:] = 0.0
        # h[i-shift] (zero when out of range)
        lo = np.empty(N, dtype=h.dtype)
        lo[:shift] = 0.0
        lo[shift:] = h[: N - shift]
    else:
        s = -shift
        hi = np.empty(N, dtype=h.dtype)
        hi[:s] = 0.0
        hi[s:] = h[: N - s]
        lo = np.empty(N, dtype=h.dtype)
        lo[: N - s] = h[s:]
        lo[N - s:] = 0.0
    # ∂C/∂h_i = dx * (1 - h_{i+shift} - h_{i-shift})
    return dx * (1.0 - hi - lo)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where
        h_values : numpy array with values in [0,1] and sum(h)*dx = 1
        c5_bound : the achieved max_k  ∫ h(x)(1-h(x+k))dx
        n_points : length of the discretisation (len(h_values))
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------ #
    # initialise discretisation ------------------------------------------------
    # Prefer to up‑sample any supplied initial_h_values, otherwise start
    # from a triangular shape (the classic feasible construction).
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        m = init.size
        # target number of points – a modest increase over the parent size
        N = max(200, int(m * 2.5))
        dx = 2.0 / N
        # mid‑point positions for the parent grid and the new grid
        x_parent = (np.arange(m) + 0.5) * (2.0 / m)
        x_new = (np.arange(N) + 0.5) * dx
        # linear interpolation (outside the interval we keep zero)
        h = np.interp(x_new, x_parent, init, left=0.0, right=0.0)
    else:
        N = 200
        dx = 2.0 / N
        x = (np.arange(N) + 0.5) * dx
        h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))        # triangular seed

    # enforce feasibility exactly
    total = N / 2.0                      # because sum(h)*dx must be 1
    h = _project_capped_simplex(h, total)

    best_h = h.copy()
    best_c, _ = _c_max(h, dx)

    # ------------------------------------------------------------------ #
    # optimization loop – subgradient descent on the worst shift ------------
    it = 0
    no_improve = 0
    # initial learning rate; will be decreased with √iteration
    base_lr = 0.5

    while time.time() - start_time < budget_s - 0.5:
        # evaluate current objective
        cur_c, corr = _c_max(h, dx)

        # keep the best encountered vector
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # identify the shift attaining the maximum
        shift_idx = int(np.argmax(corr))
        shift = shift_idx - (N - 1)          # signed integer shift

        # sub‑gradient of the active term
        grad = _gradient_for_shift(h, shift, dx)

        # diminishing step size
        lr = base_lr / np.sqrt(it + 1.0)

        # gradient step and projection back onto the feasible set
        h = h - lr * grad
        h = _project_capped_simplex(h, total)

        it += 1

        # occasional random perturbation to escape plateaus
        if no_improve >= 3000:
            noise = np.random.uniform(-0.03, 0.03, size=N)
            h = np.clip(best_h + noise, 0.0, 1.0)
            h = _project_capped_simplex(h, total)
            no_improve = 0

    # final evaluation of the best vector found
    final_c, _ = _c_max(best_h, dx)

    return best_h, float(final_c), N
