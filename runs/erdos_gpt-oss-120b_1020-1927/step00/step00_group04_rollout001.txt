    n = len(h)

    term1 = np.zeros_like(h)
    term2 = np.zeros_like(h)
    if shift >= 0:

        if t < n:
            term1[t:] = w[:n-t]
            term2[:n-t] = h[t:]

import time
import numpy as np

def _project_box_sum(v, target):
    """Project vector v onto the set {x in [0,1]^n, sum(x)=target}."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        if x.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and initial vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n0 = init.shape[0]
        n = max(200, n0 * 5)                     # a finer grid
        xs0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(xs, xs0, init)            # linear interpolation
    else:
        n = 200
        h = np.full(n, 0.5)

    target_sum = n / 2.0
    h = _project_box_sum(h, target_sum)

    # ------------------------------------------------------------------
    # Helper utilities
    # ------------------------------------------------------------------
    def _corr_unscaled(vec):
        return np.correlate(vec, 1.0 - vec, mode="full")

    def _c5(vec):
        """Scaled objective: max_k ∫ h(x)(1−h(x+k))dx."""
        return np.max(_corr_unscaled(vec)) * (2.0 / n)

    def _grad_shift(vec, shift):
        """Gradient of the unscaled correlation for a given integer shift."""
        w = 1.0 - vec
        g = np.zeros_like(vec)
        if shift >= 0:
            if shift < n:
                g[: n - shift] = w[shift:]
                g[shift:] -= vec[: n - shift]
        else:
            t = -shift
            if t < n:
                g[t:] = w[: n - t]
                g[: n - t] -= vec[t:]
        return g

    def _random_binary():
        """Generate a binary feasible vector."""
        vec = np.zeros(n, dtype=float)
        idx = np.random.choice(n, n // 2, replace=False)
        vec[idx] = 1.0
        return vec

    # ------------------------------------------------------------------
    # Initialise best solution
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_val = _c5(best_h)

    # try a few random starters immediately
    for _ in range(5):
        cand = _random_binary()
        val = _c5(cand)
        if val < best_val:
            best_val, best_h = val, cand.copy()

    # ------------------------------------------------------------------
    # Main optimisation loop (subgradient descent with occasional restarts)
    # ------------------------------------------------------------------
    max_inner = 4000          # iterations per local descent
    while time.time() - start_time < budget_s * 0.95:
        # start point: jittered best or fresh binary
        if np.random.rand() < 0.6:
            cur = best_h + np.random.normal(scale=0.04, size=n)
            cur = np.clip(cur, 0.0, 1.0)
            cur = _project_box_sum(cur, target_sum)
        else:
            cur = _random_binary()

        cur_val = _c5(cur)
        stagnant = 0

        for it in range(max_inner):
            # time guard
            if time.time() - start_time >= budget_s * 0.95:
                break

            # shift that attains the current maximum
            corr = _corr_unscaled(cur)
            max_idx = np.argmax(corr)
            shift = max_idx - (n - 1)

            # gradient for that shift
            g = _grad_shift(cur, shift)

            # simple diminishing step size
            step = 0.5 / np.sqrt(it + 1)

            improved = False
            for _ in range(7):
                cand = _project_box_sum(cur - step * g, target_sum)
                cand_val = _c5(cand)
                if cand_val < cur_val - 1e-12:
                    cur, cur_val = cand, cand_val
                    improved = True
                    stagnant = 0
                    break
                step *= 0.5

            if not improved:
                stagnant += 1
                # occasional random perturbation to escape flat regions
                if stagnant % 300 == 0:
                    break
                if np.random.rand() < 0.02:
                    cur = cur + np.random.normal(scale=0.02, size=n)
                    cur = np.clip(cur, 0.0, 1.0)
                    cur = _project_box_sum(cur, target_sum)
                    cur_val = _c5(cur)

        # update global best
        if cur_val < best_val:
            best_val, best_h = cur_val, cur.copy()
            if best_val <= 0.3808:
                break

    # ------------------------------------------------------------------
    # Return the best found construction
    # ------------------------------------------------------------------
    final_bound = _c5(best_h)
    return best_h, final_bound, n
