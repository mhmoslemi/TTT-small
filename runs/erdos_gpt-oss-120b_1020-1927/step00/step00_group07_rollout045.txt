import time
import numpy as np

def _evaluate(h, dx):
    """return max overlap C5 and the full overlap vector."""
    ov = np.correlate(h, 1.0 - h, mode="full") * dx
    return ov.max(), ov

def _project_sum_box(y, target):
    """
    Euclidean projection of y onto {x : sum(x)=target, 0≤x≤1}.
    Solution has form x_i = clip(y_i - λ, 0, 1) for some λ.
    Find λ by binary search.
    """
    lo = np.min(y) - 1.0          # low enough → all clipped to 1
    hi = np.max(y)                # high enough → all clipped to 0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def _subgradient_descent(h0, dx, target_sum, max_iters, step0,
                         start_time, time_budget):
    """
    Projected sub‑gradient descent on the objective
        max_k  ∫ h(x)(1‑h(x+k))dx .
    Returns the best point seen and its objective value.
    """
    n = h0.size
    h = h0.copy()
    best_h = h.copy()
    best_val = np.inf

    for t in range(1, max_iters + 1):
        # time guard
        if time.time() - start_time > time_budget:
            break

        # compute overlap vector
        O = np.correlate(h, 1.0 - h, mode="full") * dx
        cur_val = O.max()
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        # active shifts (those attaining the max within tolerance)
        tol = cur_val * 1e-9 + 1e-12
        act = np.where(O >= cur_val - tol)[0] - (n - 1)
        cnt = act.size
        if cnt == 0:
            # shouldn't happen, but protect against division by zero
            break

        # gradient = cnt * 1 - Σ_s (h shifted by +s + h shifted by –s)
        grad = np.full(n, cnt, dtype=np.float64)

        for s in act:
            if s >= 0:
                # shift +s
                if s < n:
                    grad[: n - s] -= h[s:]
                    # shift –s
                    grad[s:] -= h[: n - s]
                # else s == n (no overlap), nothing to subtract
            else:
                s_abs = -s
                if s_abs < n:
                    grad[s_abs:] -= h[: n - s_abs]
                    grad[: n - s_abs] -= h[s_abs:]
        # step size schedule
        step = step0 / np.sqrt(t)
        h = h - step * grad
        # project back to feasible set
        h = _project_sum_box(h, target_sum)
    return best_h, best_val
