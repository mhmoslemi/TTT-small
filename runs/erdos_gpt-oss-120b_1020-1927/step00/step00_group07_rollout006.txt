def compute_max_overlap(h):
    n = len(h)
    max_val = 0.0
    for k in range(-n+1, n):
        i_start = max(0, -k)
        i_end = min(n, n - k)
        if i_start >= i_end:
            continue
        slice_h = h[i_start:i_end]
        slice_comp = 1 - h[i_start + k: i_end + k]  # shift
        val = np.dot(slice_h, slice_comp)
        if val > max_val:
            max_val = val
    return max_val * dx

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random, itertools, math
    start = time.time()
    np.random.seed(seed)
    random.seed(seed)
    # attempt to read global initial_h_values
    try:
        init_vals = np.array(initial_h_values, dtype=float)
        n = init_vals.shape[0]
    except Exception:
        n = 80
        init_vals = None
    dx = 2.0 / n

    def score_of(arr):
        # arr can be np.ndarray of float (0/1)
        # Use np.correlate
        return np.max(np.correlate(arr, 1 - arr, mode='full')) * dx

    # start with initial solution
    if init_vals is not None:
        # round to binary maybe
        arr = (init_vals > 0.5).astype(np.int8)
        # adjust to have exact half ones
        target = n // 2
        cur = arr.sum()
        if cur > target:
            idx = np.where(arr == 1)[0]
            np.random.shuffle(idx)
            arr[idx[:cur - target]] = 0
        elif cur < target:
            idx = np.where(arr == 0)[0]
            np.random.shuffle(idx)
            arr[idx[:target - cur]] = 1
        h = arr
    else:
        # random balanced
        h = np.zeros(n, dtype=np.int8)
        h[:n//2] = 1
        np.random.shuffle(h)

    best_h = h.copy()
    best_score = score_of(best_h)

    # enumeration of periodic patterns
    # get even divisors
    divisors = [p for p in range(2, n+1) if n % p == 0 and p % 2 == 0]
    # sort small to large to start fast
    divisors.sort()
    # limit enumeration to manageable combos
    max_comb = 50000  # threshold
    for p in divisors:
        if time.time() - start > budget_s * 0.7:
            break
        ones = p // 2
        comb_cnt = math.comb(p, ones)
        if comb_cnt > max_comb:
            continue
        # enumerate
        # Use itertools.combinations
        for comb in itertools.combinations(range(p), ones):
            block = np.zeros(p, dtype=np.int8)
            block[list(comb)] = 1
            full = np.tile(block, n // p)
            sc = score_of(full)
            if sc < best_score - 1e-12:
                best_score = sc
                best_h = full.copy()
        # helpful to maybe also try reversed block? Already enumerated.

    # Simulated annealing from best_h
    h = best_h.copy()
    # Keep best globally
    # Precompute index arrays for ones and zeros
    # define temperature schedule
    T = 5.0
    cooling = 0.9995  # slow cooling
    min_T = 1e-4
    iter_without_improve = 0
    last_improve_iter = 0
    iter_count = 0

    while time.time() - start < budget_s * 0.95:
        # choose t number of swaps
        # adapt t based on temperature maybe
        max_swap = max(1, int(T * 2))  # not exceed n/2
        t = np.random.randint(1, max_swap + 1)

        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) < t or len(zeros_idx) < t:

        swap_zero = np.random.choice(zeros_idx, t, replace=False)
        h_new = h.copy()
        h_new[swap_one] = 0
        h_new[swap_zero] = 1
        new_score = score_of(h_new)
        delta = new_score - best_score
        accept = False
        if new_score < best_score:
            accept = True
        else:
            if T > 0:

                if random.random() < prob:

        if accept:

                best_h = h_new.copy()

                h[swap_zero] = 1

    # final local improvement via pairwise swaps

import numpy as np
import random
import itertools
import math
import time

def compute_score(arr: np.ndarray, dx: float) -> float:
    """Maximum overlap integral for a step function given by *arr*."""
    # np.correlate follows the evaluation used by the harness.
    return np.max(np.correlate(arr, 1 - arr, mode="full")) * dx

def random_balanced(n: int) -> np.ndarray:
    """Return a binary array of length *n* with exactly n/2 ones."""
    a = np.zeros(n, dtype=np.int8)
    a[: n // 2] = 1
    np.random.shuffle(a)
    return a

def even_divisors(n: int):
    """All even divisors of *n* (including n itself)."""
    return [d for d in range(2, n + 1) if n % d == 0 and d % 2 == 0]

def enumerate_periodic(n: int, dx: float, time_limit: float,
                      best_h: np.ndarray, best_score: float):
    """Exhaustively try periodic patterns with small periods."""
    start = time.time()
    for p in even_divisors(n):
        # skip very large combination spaces
        ones_per_block = p // 2
        comb_cnt = math.comb(p, ones_per_block)
        if comb_cnt > 50000:
            continue
        # enumerate all blocks of length p with exactly p/2 ones
        for comb in itertools.combinations(range(p), ones_per_block):
            if time.time() - start > time_limit:
                return best_h, best_score
            block = np.zeros(p, dtype=np.int8)
            block[list(comb)] = 1
            full = np.tile(block, n // p)
            sc = compute_score(full, dx)
            if sc < best_score:
                best_score = sc
                best_h = full.copy()
    return best_h, best_score

def simulated_annealing(h: np.ndarray, dx: float, time_limit: float,
                        best_h: np.ndarray, best_score: float):
    """Simple SA that swaps equal numbers of 0/1 while preserving balance."""
    start = time.time()
    n = len(h)
    T = 5.0                         # initial temperature
    cooling = 0.9995
    min_T = 1e-4
    no_improve = 0
    while time.time() - start < time_limit:
        # choose a random number of simultaneous swaps
        max_swap = max(1, int(T * 2))
        t = np.random.randint(1, max_swap + 1)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < t or len(zeros) < t:
            continue

        o_sel = np.random.choice(ones, t, replace=False)
        z_sel = np.random.choice(zeros, t, replace=False)

        h_new = h.copy()
        h_new[o_sel] = 0
        h_new[z_sel] = 1

        new_score = compute_score(h_new, dx)
        delta = new_score - best_score
        if new_score < best_score:
            accept = True
        else:
            accept = (random.random() < math.exp(-delta / T) if T > 0 else False)

        if accept:
            h = h_new
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        T = max(T * cooling, min_T)

        if no_improve > 20000:
            # restart near the current best
            h = best_h.copy()
            no_improve = 0
    return best_h, best_score

def pairwise_improve(h: np.ndarray, dx: float, time_limit: float,
                    best_h: np.ndarray, best_score: float):
    """Deterministic pair‑swap hill climbing until no immediate improvement."""
    start = time.time()
    n = len(h)
    while time.time() - start < time_limit:
        improved = False
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        np.random.shuffle(ones)
        np.random.shuffle(zeros)

        for i in ones:
            for j in zeros:
                if time.time() - start >= time_limit:
                    break
                cand = best_h.copy()
                cand[i] = 0
                cand[j] = 1
                sc = compute_score(cand, dx)
                if sc < best_score - 1e-12:
                    best_h = cand
                    best_score = sc
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_score

    # ----------------------------------------------------------------------

            h[idx[: cur - target]] = 0
        elif cur < target:
            idx = np.where(h == 0)[0]
            np.random.shuffle(idx)
            h[idx[: target - cur]] = 1
    else:
        h = random_balanced(n)

    best_h = h.copy()
    best_score = compute_score(best_h, dx)

    # ----------------------------------------------------------------------
    # Time budgeting
    # ----------------------------------------------------------------------
    start = time.time()
    # 20 % for exhaustive periodic search, 60 % for SA, 20 % for final descent
    enum_budget = budget_s * 0.20
    sa_budget   = budget_s * 0.60
    pair_budget = budget_s * 0.20

    # ----------------------------------------------------------------------
    # 1. Exhaustive search over short periods (if cheap enough)
    # ----------------------------------------------------------------------
    best_h, best_score = enumerate_periodic(
        n, dx, enum_budget, best_h, best_score)

    # ----------------------------------------------------------------------
    # 2. Simulated annealing on the full binary vector
    # ----------------------------------------------------------------------
    best_h, best_score = simulated_annealing(
        best_h, dx, sa_budget, best_h, best_score)

    # ----------------------------------------------------------------------
    # 3. Deterministic pair‑swap refinement
    # ----------------------------------------------------------------------
    best_h, best_score = pairwise_improve(
        best_h, dx, pair_budget, best_h, best_score)

    # Ensure the returned array respects the constraints exactly.
    best_h = best_h.astype(float)           # convert to float for the harness
    return best_h, best_score, n
