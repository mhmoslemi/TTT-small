def subgradient_step(h, step):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, h, mode='full')
    min_val = np.min(corr)
    # Determine argmin indices within tolerance
    tol = 1e-12 * (np.abs(min_val) + 1)
    arg_idx = np.where(np.abs(corr - min_val) <= tol)[0]
    grad = np.zeros_like(h)
    for idx in arg_idx:
        shift = idx - (n - 1)
        if shift >= 0:
            # forward contributions
            grad[:n-shift] += -dx * h[shift:]
            # backward contributions
            grad[shift:] += -dx * h[:n-shift]
        else:
            s = -shift
            grad[s:] += -dx * h[:n-s]
            grad[:n-s] += -dx * h[s:]
    # step update (gradient descent)
    h_new = h - step * grad
    # project.
    target = n / 2.0
    h_proj = project_box_sum(h_new, target)
    return h_proj, grad

def upscale(vec, n_target):
    n_src = len(vec)
    if n_target == n_src:
        return vec.copy()
    # linspace positions
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_target, endpoint=False)
    # periodic interpolation:
    v_up = np.interp(x_tgt, x_src, vec, period=2)

import time
import numpy as np

def _project_onto_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {x | 0<=x<=1, sum(x)=target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - hi, 0.0, 1.0)


def _corr_and_score(h: np.ndarray):
    """Return (full autocorrelation of h, C5 score)."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, h, mode="full")
    min_val = np.min(corr)
    score = 1.0 - min_val * dx
    return corr, score


def _subgradient_step(h: np.ndarray, corr: np.ndarray, step: float) -> np.ndarray:
    """One projected subgradient step for C5."""
    n = h.size
    dx = 2.0 / n
    min_val = np.min(corr)
    tol = 1e-12 * (abs(min_val) + 1.0)
    idxs = np.where(np.abs(corr - min_val) <= tol)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        shift = idx - (n - 1)
        if shift >= 0:
            grad[: n - shift] += -dx * h[shift:]
            grad[shift:] += -dx * h[: n - shift]
        else:
            s = -shift
            grad[s:] += -dx * h[: n - s]
            grad[: n - s] += -dx * h[s:]
    h_new = h - step * grad
    return _project_onto_box_sum(h_new, n / 2.0)


def _random_exchange(h: np.ndarray) -> np.ndarray:
    """Move a random amount of mass from one index to another."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    if np.random.rand() < 0.5:
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0:
            return h
        delta = np.random.uniform(0.0, max_move)
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    else:
        max_move = min(h[j], 1.0 - h[i])
        if max_move <= 0:
            return h
        delta = np.random.uniform(0.0, max_move)
        new_h = h.copy()
        new_h[j] -= delta
        new_h[i] += delta
    return _project_onto_box_sum(new_h, n / 2.0)


def _upscale(vec: np.ndarray, n_target: int) -> np.ndarray:
    """Linear‑interpolate vec (defined on [0,2]) to n_target points."""
    n_src = vec.size
    if n_target == n_src:
        return vec.copy()
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    up = np.interp(x_tgt, x_src, vec, period=2.0)
    return _project_onto_box_sum(up, n_target / 2.0)


def _c5_score_direct(h: np.ndarray) -> float:
    """Direct evaluation used for the final reported value."""
    n = h.size
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(vals))


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize the Erdős C₅ constant.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s

    # ------------------------------------------------------------------ #
    # 1. Starting vector ------------------------------------------------ #
    init = kwargs.get("initial_h_values")
    if init is not None:
        init_vec = np.asarray(init, dtype=float)
    else:
        # fallback: a uniform half‑density vector
        init_vec = np.full(48, 0.5, dtype=float)

    # Ensure feasibility
    init_vec = _project_onto_box_sum(init_vec, init_vec.size / 2.0)

    # ------------------------------------------------------------------ #
    # 2. Candidate discretisations -------------------------------------- #
    n0 = init_vec.size
    cand_ns = [n0]
    if n0 * 2 <= 200:
        cand_ns.append(n0 * 2)
    if n0 * 3 <= 200:
        cand_ns.append(n0 * 3)

    best_h = init_vec.copy()
    best_score = _c5_score_direct(best_h)
    best_n = best_h.size

    # ------------------------------------------------------------------ #
    # 3. Main optimisation loop ---------------------------------------- #
    for n in sorted(set(cand_ns)):
        if time.time() - start_time > max_time - 5.0:
            break

        # initialise for this grid size
        if n == n0:
            h = init_vec.copy()
        else:
            h = _upscale(init_vec, n)

        corr, score = _corr_and_score(h)
        # local best for this grid
        local_best_h = h.copy()
        local_best_score = score

        step = 0.5
        it = 0
        # limits chosen to keep overall runtime well within the budget
        while True:
            if time.time() - start_time > max_time - 2.0:
                break
            # projected subgradient step
            h_new = _subgradient_step(h, corr, step)
            corr_new, score_new = _corr_and_score(h_new)

            if score_new < score:
                # accept improvement
                h, corr, score = h_new, corr_new, score_new
                step = min(step * 1.02, 5.0)  # gently increase step
                if score_new < local_best_score - 1e-12:
                    local_best_h, local_best_score = h_new.copy(), score_new
            else:
                # reject – shrink step
                step *= 0.5

            # occasional random exchange to escape flat regions
            if it % 50 == 0 and np.random.rand() < 0.3:
                h_rand = _random_exchange(h)
                _, rand_score = _corr_and_score(h_rand)
                if rand_score < score:
                    h, corr, score = h_rand, *(_corr_and_score(h_rand))

            # update global best
            if score < best_score - 1e-12:
                best_h, best_score, best_n = h.copy(), score, n

            it += 1
            if step < 1e-6 or it >= 25000:
                break

    # ------------------------------------------------------------------ #
    # 4. Final reporting ------------------------------------------------ #
    final_score = _c5_score_direct(best_h)
    return best_h, final_score, best_h.size
