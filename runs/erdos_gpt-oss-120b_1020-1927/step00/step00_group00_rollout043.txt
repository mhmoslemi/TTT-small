def hill_climb_binary(n, rng, time_limit):
    dx = 2.0 / n
    # start with random bitset of n/2 ones.
    ones = rng.choice(n, n//2, replace=False)
    h = np.zeros(n, dtype=np.float64)
    h[ones] = 1.0
    best_h = h.copy()
    best_val = c5_of_h(h, dx)
    # start variables
    start = time.time()
    it = 0
    no_improve = 0
    max_no_improve = 20000
    while time.time() - start < time_limit:
        # choose random swap
        # pick i from ones
        if best_val == 0:
            break
        idx_one = rng.choice(np.where(h == 1)[0])
        idx_zero = rng.choice(np.where(h == 0)[0])
        # swap
        h[idx_one] = 0
        h[idx_zero] = 1
        # evaluate
        val = c5_of_h(h, dx)
        if val < best_val - 1e-12:
            best_val = val
            best_h = h.copy()
            # optional reset counters
            no_improve = 0
        else:
            # revert
            h[idx_one] = 1
            h[idx_zero] = 0
            no_improve += 1
        it += 1
        # if no improvement for many iterations, random restart
        if no_improve >= max_no_improve:
            # restart random ones/h:
            h[:] = 0
            ones = rng.choice(n, n//2, replace=False)
            h[ones] = 1.0
            # Evaluate restart solution: If it better update
            val = c5_of_h(h, dx)
            if val < best_val:
                best_val = val
                best_h = h.copy()
            no_improve = 0
    return best_h, best_val

import time, numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def continuous_local_search(h0, rng, time_limit):
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_val = evaluate_c5(h)
    start = time.time()
    # initial step size
    step = 0.2
    # schedule: gradually reduce step each 10% time
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        # adapt step
        if elapsed > time_limit * 0.5:
            step = 0.05
        if elapsed > time_limit * 0.8:
            step = 0.01
        # pick two indices
        i, j = rng.choice(n, size=2, replace=False)
        # compute max allowable delta
        max_delta = min(step, 1 - h[i], h[j])
        min_delta = max(-step, -h[i], h[j] - 1)
        if max_delta <= min_delta:
            continue
        delta = rng.uniform(min_delta, max_delta)
        h[i] += delta
        h[j] -= delta
        # ensure bounds just in case
        # evaluate
        val = evaluate_c5(h)
        if val < best_val - 1e-12:
            best_val = val
            best_h = h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta
    return best_h, best_val

def binary_hill_climb(n, rng, time_limit):
    dx = 2.0 / n
    # create random binary vector
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    # compute C5
    best_h = h.copy()
    best_val = evaluate_c5(h)
    current_val = best_val
    start = time.time()
    no_improve = 0
    while time.time() - start < time_limit:
        # pick random swap
        idx_one = rng.choice(np.flatnonzero(h == 1))
        idx_zero = rng.choice(np.flatnonzero(h == 0))
        # propose swap
        h[idx_one] = 0.0
        h[idx_zero] = 1.0
        val = evaluate_c5(h)
        if val < current_val - 1e-12:
            current_val = val
            # if better than best
            if val < best_val:
                best_val = val
                best_h = h.copy()
            no_improve = 0
        else:
            # revert
            h[idx_one] = 1.0
            h[idx_zero] = 0.0
            no_improve += 1
        # random restart after many failures
        if no_improve > 20000:
            h[:] = 0
            ones = rng.choice(n, n//2, replace=False)
            h[ones] = 1.0
            current_val = evaluate_c5(h)
            no_improve = 0
    return best_h, best_val

def simulated_annealing(h0, rng, time_limit):
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    current_val = evaluate_c5(h)
    best_h = h.copy()
    best_val = current_val
    start = time.time()
    # schedule
    temp_initial = 1.0
    temp_final = 1e-4
    # simple exponential cooling over allowed iterations
    # determine total iterations approx based on time budget: estimate per iteration ~ 0.00002 sec, compute iters = int( (time_limit * 0.9) / 0.00005 ) maybe.
    # but we can just loop until time runs out.
    while time.time() - start < time_limit:
        # compute progress fraction
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = temp_initial * (temp_final / temp_initial) ** frac
        # propose move
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        # compute delta bounds
        delta_min = max(-h[i], -(1 - h[j]))
        delta_max = min(1 - h[i], h[j])
        if delta_max <= delta_min:
            continue
        delta = rng.uniform(delta_min, delta_max)
        # apply
        h[i] += delta
        h[j] -= delta
        val = evaluate_c5(h)
        # if improvement accept
        if val < current_val - 1e-12:
            current_val = val
            if val < best_val:
                best_val = val
                best_h = h.copy()
        else:
            # accept with probability exp(-(val - current_val)/T)
            if rng.random() < np.exp(-(val - current_val) / max(T, 1e-12)):
                current_val = val
            else:
                # revert
                h[i] -= delta
                h[j] += delta
    return best_h, best_val
```

import time
import numpy as np

def _c5_value(h):
    """C5 for a step function given by the sample vector h."""
    n = len(h)
    dx = 2.0 / n
    # convolution of h with (1-h); mode='full' gives all integer shifts
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _random_feasible(n, rng):
    """
    Produce a feasible vector of length n (0<=h_i<=1, sum(h)*dx = 1).
    The simple construction starts from a binary vector of weight n//2
    and, if n is odd, adds a 0.5 entry.
    """
    h = np.zeros(n, dtype=np.float64)
    half = n // 2
    ones = rng.choice(n, half, replace=False)
    h[ones] = 1.0
    if n % 2:          # odd n – need one entry of 0.5 to hit the exact mass
        remaining = list(set(range(n)) - set(ones))
        h[rng.choice(remaining)] = 0.5
    return h


def _simulated_annealing(start_h, rng, timelimit):
    """
    Simple simulated‑annealing on the feasible polytope.
    At each step we transfer a small amount δ from one coordinate
    to another while respecting the box constraints.
    """
    h = start_h.copy()
    n = len(h)
    best_h = h.copy()
    best_val = _c5_value(h)
    cur_val = best_val

    start = time.time()
    # temperature schedule (linear in time)
    T_start = 2e-2
    T_end = 1e-5

    # maximal step size for a transfer; shrinks slowly with time
    max_step = 0.5

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= timelimit - 0.05:
            break
        # linear cooling
        frac = elapsed / timelimit
        T = T_start * (1 - frac) + T_end * frac
        # choose a source (high) and a sink (low) index
        high_idx = np.where(h > 0.5)[0]
        low_idx = np.where(h < 0.5)[0]
        if high_idx.size == 0 or low_idx.size == 0:
            # fall back to completely random pair
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
            if i == j:
                continue

        # maximum amount we may move from i to j while staying feasible
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= 0:
            continue
        # propose a transfer (always from high to low)
        delta = rng.uniform(0.0, min(max_step, delta_max))

        # apply move
        h[i] -= delta
        h[j] += delta

        new_val = _c5_value(h)

        # better solution ?
        if new_val < cur_val - 1e-12:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
                cur_val = new_val
            else:
                # revert the move
                h[i] += delta
                h[j] -= delta

        # slowly tighten the step size as we approach the end
        if elapsed > 0.7 * timelimit:
            max_step = max_step * 0.5

    return best_h, best_val
