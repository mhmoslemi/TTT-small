def project_to_sum(v, target):
    low, high = -2.0, 2.0
    for _ in range(60):

import time, math
import numpy as np

def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the box [0,1]^n intersect {sum = target}.

    The projection is the Euclidean projection and can be obtained by a
    binary search for a shift τ such that sum(clamp(v+τ,0,1)) = target.
    """
    low, high = -2.0, 2.0
    for _ in range(60):
        mid = (low + high) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        if w.sum() < target:
            low = mid
        else:
            high = mid
    w = np.clip(v + high, 0.0, 1.0)

    # tiny fix to hit the target exactly (within floating error)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = np.where((w > 0.0) & (w < 1.0))[0]
        if free.size:
            i = free[0]
            w[i] = np.clip(w[i] + diff, 0.0, 1.0)
        else:
            # all entries are at a bound – adjust first entry safely
            w[0] = np.clip(w[0] + diff, 0.0, 1.0)
    return w


def _c5_and_conv(h: np.ndarray):
    """Return (C5 value, full correlation array) for a step function h."""
    n = h.size
    conv = np.correlate(h, 1.0 - h, mode="full")       # no scaling yet
    c5 = conv.max() * (2.0 / n)                       # multiply by dx = 2/n
    return c5, conv

    Optimize a step function h on [0,2] to minimise
          C5 = max_k ∫ h(x)(1‑h(x+k))dx .
    Returns (h_values, c5_bound, n_points).
    """

    np.random.seed(seed)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_h = None

    #  helper for a single optimisation run (gradient‑descent on the worst lag)
    # ------------------------------------------------------------------
    def _optimize(n: int, iters: int, step0: float, start_vec: np.ndarray = None):
        nonlocal best_h, best_c5, best_n

        target_sum = n / 2.0                     # because ∫h = 1, dx = 2/n
        if start_vec is None:

            h = _project_to_sum(v, target_sum)

        for it in range(iters):
            # time guard
            if time.time() - start > budget_s * 0.95:

            # correlation and current C5
            conv = np.correlate(h, 1.0 - h, mode="full")
            c5 = conv.max() * (2.0 / n)

            # keep the best ever seen

                best_h = h.copy()

            idx = conv.argmax()
            lag = idx - (n - 1)          # lag = idx - centre
