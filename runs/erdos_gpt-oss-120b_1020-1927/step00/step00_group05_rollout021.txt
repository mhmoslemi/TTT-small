    n = len(h)

        return None, None  # no move

import numpy as np
import time
import random
import math

def _compute_c5(h: np.ndarray) -> float:
    """Return the maximum overlap integral for the step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _random_transfer(h: np.ndarray) -> np.ndarray | None:
    """Move a small amount of mass from one coordinate to another, preserving sum."""
    n = h.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return None
    # feasible epsilon so both remain in [0,1]
    eps_low = max(-h[i], h[j] - 1.0)          # cannot make i negative nor j exceed 1
    eps_high = min(1.0 - h[i], h[j])          # cannot exceed bound for i nor go below 0 for j
    if eps_low >= eps_high:
        return None
    eps = eps_low + (eps_high - eps_low) * np.random.rand()
    # limit step size to keep changes small (helps stability)
    max_step = 0.05
    eps = max(-max_step, min(max_step, eps))
    if abs(eps) < 1e-12:
        return None
    nh = h.copy()
    nh[i] += eps
    nh[j] -= eps
    return nh


def _targeted_move(h: np.ndarray) -> np.ndarray | None:
    """
    Perform a move that directly tries to improve the worst (largest) overlap.
    It picks the shift where the overlap is maximal, then transfers a little mass
    from a pair that contributes little to that shift to a pair that can contribute more.
    """
    n = h.shape[0]
    # find worst shift
    corr = np.correlate(h, 1.0 - h, mode="full")
    worst_idx = int(np.argmax(corr))
    shift = worst_idx - (n - 1)          # integer shift (can be negative)

    # indices that appear in the overlap for this shift
    if shift >= 0:
        i_vals = np.arange(0, n - shift)
    else:
        i_vals = np.arange(-shift, n)
    if i_vals.size < 2:
        return None

    # capacity to increase a pair (i,i+shift) while staying in [0,1]
    inc_cap = np.minimum(1.0 - h[i_vals], 1.0 - h[i_vals + shift])
    # capacity to decrease a pair (a,a+shift)
    dec_cap = np.minimum(h[i_vals], h[i_vals + shift])

    # choose a target pair that can be increased
    good_targets = np.where(inc_cap > 1e-5)[0]
    if good_targets.size == 0:
        return None
    tgt = np.random.choice(good_targets)
    i = i_vals[tgt]
    j = i + shift
    inc_max = inc_cap[tgt]

    # choose a donor pair that can be decreased
    good_donors = np.where(dec_cap > 1e-5)[0]
    # ensure donor pair differs from target pair
    good_donors = good_donors[good_donors != tgt]
    if good_donors.size == 0:
        return None
    donor = np.random.choice(good_donors)
    a = i_vals[donor]
    b = a + shift
    dec_max = dec_cap[donor]

    # feasible epsilon : can't change more than either capacity
    eps_max = min(inc_max, dec_max)
    if eps_max <= 1e-7:
        return None
    eps = eps_max * np.random.rand() * 0.5   # be modest
    if eps < 1e-8:
        return None

    nh = h.copy()
    nh[i] += eps
    nh[j] += eps
    nh[a] -= eps
    nh[b] -= eps
    # numerical safety
    np.clip(nh, 0.0, 1.0, out=nh)
    return nh
