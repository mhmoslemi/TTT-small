def project_capped_simplex(v, weight):
    n = len(v)
    # Clip to make sure sum feasible: if weight < 0 or > n, impossible; but weight = n/2 within [0,n].
    # Define helper function sum_at_lambda(lambda):
    # return np.sum(np.minimum(np.maximum(v - lambda, 0), 1))
    # Use binary search.
    lo = np.min(v) - 1  # guarantee all v - lo >= 1 => x_i = 1
    hi = np.max(v)  # guarantee all v - hi <= 0 => x_i=0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > weight:
            lo = mid
        else:
            hi = mid
    # final projection
    return np.clip(v - (lo+hi)/2, 0, 1)

def hill_climb_binary(n, weight, time_budget):
    start = time.time()

    h = np.zeros(n, dtype=np.float64)
    ones_idx = np.random.choice(n, weight, replace=False)
    h[ones_idx] = 1.0
    best_h = h.copy()
    best_max = evaluate_overlap(h)
    # Precompute convolution for initial for speed: We'll recompute after each swap using np.correlate.

    while time.time() - start < time_budget:
        # random swap attempt
        # pick i among positions with 1, j with 0
        ones_positions = np.where(h > 0.5)[0]
        zeros_positions = np.where(h < 0.5)[0]
        if len(ones_positions) == 0 or len(zeros_positions) == 0:
            break
        i = np.random.choice(ones_positions)
        j = np.random.choice(zeros_positions)
        # propose swap
        h_candidate = h.copy()
        h_candidate[i] = 0.0
        h_candidate[j] = 1.0

            best_h = h.copy()
            best_max = cand_max
        # optionally keep if better than current but not best? for hill rest.
    return best_h, best_max

def gradient_descent_continuous(h_initial, time_budget):

    h = h_initial.copy()
    # ensure constraints projection
    h = project_capped_simplex(h)
    best_h = h.copy()
    best_max = evaluate_overlap(h)
    n = len(h)
    ndx = 2.0 / n
    weight = n/2
    alpha = 10.0
    eta0 = 0.5
    iter = 0
    while time.time() - start < time_budget:
        # compute conv
        conv = np.correlate(h, 1-h, mode='full') * ndx
        max_f = conv.max()
        if max_f < best_max:
            best_max = max_f
            best_h = h.copy()
        # compute softmax weights for max approximation
        # shift indices
        max_val = conv.max()
        expVals = np.exp(alpha * (conv - max_val))
        Z = expVals.sum()
        w = expVals / Z
        # compute gradient
        grad = np.full(n, ndx)  # start with ndx
        # vectorized loops
        for idx, shift in enumerate(range(-(n-1), n)):
            w_shift = w[idx] * ndx
            if w_shift == 0:
                continue
            s = shift
            if s >= 0:
                # forward
                if n - s > 0:
                    grad[:n-s] -= w_shift * h[s:]
                    grad[s:]   -= w_shift * h[:n-s]
            else:
                sp = -s
                if n - sp > 0:
                    grad[sp:] -= w_shift * h[:n-sp]
                    grad[:n-sp] -= w_shift * h[sp:]
        # step size schedule
        eta = eta0 / (1 + 0.001 * iter)
        h_new = h - eta * grad
        # project onto constraints
        h = project_capped_simplex(h_new)
        # schedule for alpha
        if iter % 500 == 0 and iter > 0:
            alpha *= 1.2
            if alpha > 200:
                alpha = 200
        iter += 1
        if iter >= 20000:
            break
    return best_h, best_max

def hill_climb_binary(n, weight, time_limit):
    start = time.time()
    # create random start Balanced
    indices = np.arange(n)
    np.random.shuffle(indices)
    h = np.zeros(n)
    h[indices[:weight]] = 1.0
    best_h = h.copy()
    best_val = overlap_max(h)
    # Pre compute indices list of ones and zeros to avoid scanning each time.
    # But after each swap we update sets.
    ones = set(np.where(h == 1)[0])
    zeros = set(np.where(h == 0)[0])
    # iteration
    while time.time() - start < time_limit:
        i = np.random.choice(list(ones))
        j = np.random.choice(list(zeros))
        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        val = overlap_max(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
            # update sets
            ones.remove(i)
            ones.add(j)
            zeros.remove(j)
            zeros.add(i)
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
        # optionally occasional random swap acceptance?
    return best_h, best_val

import time
import numpy as np

def _evaluate_overlap(h):
    """Return max_{shift} ∫ h(x)*(1-h(x+shift)) dx for the discrete grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # zero‑padding correlation (numpy does this automatically)
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(conv.max())

def _project_onto_capped_simplex(v, target):
    """
    Euclidean projection of v onto
        { x : 0 ≤ x_i ≤ 1,  sum x_i = target }.
    Uses a simple bisection on the Lagrange multiplier λ:
        x_i = clip(v_i - λ, 0, 1)
    """
    lo = np.min(v) - 1.0          # ensures all clip to 1
    hi = np.max(v)                # ensures all clip to 0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _subgradient_descent(h0, max_iters, time_limit):
    """
    Minimise the worst overlap using a sub‑gradient method.
    Returns the best (h, value) discovered.
    """
    n = h0.shape[0]
    dx = 2.0 / n
    target = n / 2.0
    h = _project_onto_capped_simplex(h0, target)
    best_h = h.copy()
    best_val = _evaluate_overlap(h)

    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        # current overlap profile
        conv = np.correlate(h, 1.0 - h, mode='full') * dx
        cur_val = float(conv.max())
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        # indices attaining the maximum (allow small tolerance)
        idxs = np.where(np.abs(conv - cur_val) < 1e-12)[0]
        idx = np.random.choice(idxs)      # pick one worst shift
        shift = idx - (n - 1)            # integer shift value

        # sub‑gradient of that worst term
        g = np.full(n, dx)               # derivative of Σ h_i term
        s = shift
        if s >= 0:
            if n - s > 0:
                g[:n - s] -= dx * h[s:]
                g[s:]     -= dx * h[:n - s]
        else:
            sp = -s
            if n - sp > 0:
                g[sp:]    -= dx * h[:n - sp]
                g[:n - sp] -= dx * h[sp:]

        # diminishing step size
        eta = 0.6 / (1.0 + it * 0.001)
        h_new = h - eta * g
        h = _project_onto_capped_simplex(h_new, target)

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with integral 1 that minimises
    the maximum overlap integral C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_budget = float(budget_s)

    # ------------------------------------------------------------------
    # Determine the discretisation size.
    # Prefer to reuse the provided initial_h_values when present.
    try:
        init = np.array(initial_h_values, dtype=float)   # provided by the harness
        n = init.shape[0]
    except NameError:
        # fallback: use a modest size (70 points) when no starter is given
        n = 70
        init = None

    dx = 2.0 / n
    target_sum = n / 2.0                     # because ∫ h = 1

    # ------------------------------------------------------------------
    # Helper to obtain a fresh random feasible start.
    def random_feasible():
        v = np.random.rand(n)
        return _project_onto_capped_simplex(v, target_sum)

    # ------------------------------------------------------------------
    # Keep the best solution seen.
    best_h = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # Allocate time: we run several independent sub‑gradient runs.
    # Each run gets an equal slice of the remaining budget (minimum 5 s).
    runs = 0
    while True:
        elapsed = time.time() - start_time
        remaining = time_budget - elapsed
        if remaining < 5.0:
            break

        # pick a start: alternate between the given initial vector and a random one
        if runs == 0 and init is not None:
            h0 = init.copy()
        else:
            h0 = random_feasible()

        # time slice for this trial
        slice_time = max(5.0, remaining / 5.0)

        # run sub‑gradient descent (capped to a few thousand iterations)
        max_iters = 5000
        h_candidate, val_candidate = _subgradient_descent(
            h0, max_iters=max_iters, time_limit=slice_time
        )

        if val_candidate < best_val:
            best_val = val_candidate
            best_h = h_candidate.copy()

        runs += 1

    # ------------------------------------------------------------------
    # A short local refinement: small random perturbations accepted if they
    # improve the objective.
    if best_h is not None:
        h = best_h.copy()
        val = best_val
        # give the refinement about the last second of the budget
        while time.time() - start_time < time_budget - 0.5:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            eps = (np.random.rand() - 0.5) * 0.08   # perturb up to ±0.04
            hi, hj = h[i], h[j]
            hi_new = np.clip(hi + eps, 0.0, 1.0)
            hj_new = np.clip(hj - eps, 0.0, 1.0)
            delta = (hi_new - hi) + (hj_new - hj)
            # keep the integral exactly; if clipping broke it, reject
            if abs(delta) > 1e-9:
                continue
            h_candidate = h.copy()
            h_candidate[i] = hi_new
            h_candidate[j] = hj_new
            cand_val = _evaluate_overlap(h_candidate)
            if cand_val + 1e-12 < val:
                h = h_candidate
                val = cand_val
        best_h = h
        best_val = val

    # ------------------------------------------------------------------
    # Final sanity check (project once more to guarantee feasibility)
    best_h = _project_onto_capped_simplex(best_h, target_sum)

    # Return as a plain list for compatibility
    return (best_h.tolist(), best_val, n)
