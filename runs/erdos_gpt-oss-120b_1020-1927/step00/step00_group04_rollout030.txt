def project_sum(h, target):
    # ensure sum close to target by scaling then adjusting mid.
    sumh = h.sum()
    if sumh == 0:
        # avoid division by zero
        h[:] = target / len(h)
        return h
    scale = target / sumh
    h *= scale
    # after scaling may exceed bounds due to rounding; clip
    np.clip(h, 0, 1, out=h)
    # correct any residual error
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff to a middle element that has slack
        # find index not at bounds
        for idx in [len(h)//2, 0, -1]:
            if (diff > 0 and h[idx] < 1) or (diff < 0 and h[idx] > 0):
                h[idx] += diff
                h[idx] = min(max(h[idx], 0.0), 1.0)
                break
    return h

import time
import random
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / h.size)   # dx = 2/n


def _project_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Scale (and if necessary adjust) h so that sum(h) == target, keeping 0≤h≤1."""
    s = h.sum()
    if s > 0:
        h *= target / s
    else:
        h[:] = target / h.size
    np.clip(h, 0.0, 1.0, out=h)

    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny remainder to a non‑boundary entry
        for idx in (h.size // 2, 0, h.size - 1):
            if (diff > 0 and h[idx] < 1.0) or (diff < 0 and h[idx] > 0.0):
                h[idx] = min(max(h[idx] + diff, 0.0), 1.0)
                break
    return h


def _improve_by_worst_shift(h: np.ndarray, time_limit: float) -> np.ndarray:
    """Deterministic descent: repeatedly act on the shift that gives the current maximum."""
    n = h.size
    target = n / 2.0
    start = time.time()
    while time.time() - start < time_limit:
        conv = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(conv.argmax())
        k = idx_max - (n - 1)                # shift (positive → h[i] with h[i+k])
        # overlapping index range for this shift
        if k >= 0:
            i_range = np.arange(0, n - k)
        else:
            i_range = np.arange(-k, n)
        diffs = h[i_range] - h[i_range + k]   # hi - hj
        # we try to reduce the disparity hi > hj (move mass from hi to hj)
        pos = np.where(diffs > 1e-12)[0]
        if pos.size == 0:
            # nothing to improve in this direction → stop
            break
        i_sel = pos[np.argmax(diffs[pos])]
        i = i_range[i_sel]
        j = i + k
        hi, hj = h[i], h[j]
        # maximum delta that keeps both entries in [0,1]
        max_delta = min(hi - hj, 1.0 - hj, hi)
        if max_delta <= 1e-12:
            break
        delta = 0.5 * max_delta                # take a modest step
        h[i] = hi - delta
        h[j] = hj + delta
        # sum stays exactly constant, only clipping for safety
        np.clip(h, 0.0, 1.0, out=h)
    return h
