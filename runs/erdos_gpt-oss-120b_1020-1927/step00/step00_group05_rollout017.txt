import time
import numpy as np
import random

def _evaluate(h):
    """
    Helper to compute C5 using the provided evaluation method.
    Assumes a global `evaluate_erdos_solution` is already imported.
    """
    return evaluate_erdos_solution(h)


def _mip_solution(n, max_seconds=900):
    """
    Solve a mixed‑integer linear program that finds a binary vector h of length n
    with exactly n/2 ones minimizing the maximal overlap.
    Returns (h_array, t_opt) where t_opt is the integer maximal overlap.
    If the MILP does not finish successfully, returns (None, None).
    """
    import cvxpy as cp

    H = n // 2                     # required number of ones
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)

    constraints = [cp.sum(h) == H, t >= 0]

    # Build constraints for each shift s (positive and negative are handled together)
    for s in range(1, n):
        ns = n - s                 # length of overlapping region
        hi = h[:ns]                # h_i   for i = 0 .. n-s-1
        hj = h[s:]                 # h_{i+s}
        y = cp.Variable(ns, boolean=True)   # y_i = h_i * h_{i+s}

        # y_i <= h_i , y_i <= h_{i+s}
        constraints += [y <= hi, y <= hj]
        # y_i >= h_i + h_{i+s} - 1   (McCormick for binary product)
        constraints += [y >= hi + hj - 1]

        # A_s = sum(hi) - sum(y)   (shift +s)
        constraints += [cp.sum(hi) - cp.sum(y) <= t]
        # A_{-s} = sum(hj) - sum(y)  (shift -s)
        constraints += [cp.sum(hj) - cp.sum(y) <= t]

    prob = cp.Problem(cp.Minimize(t), constraints)

    # Try SCIP first, fall back to GLPK_MI if unavailable
    try:
        prob.solve(solver=cp.SCIP, verbose=False, max_seconds=max_seconds)
    except Exception:
        prob.solve(solver=cp.GLPK_MI, verbose=False)

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, None

    h_val = np.rint(h.value).astype(int)   # ensure exact 0/1
    t_val = int(round(t.value))
    return h_val, t_val


def _random_binary(n):
    """Return a random binary vector of length n with exactly n/2 ones."""
    half = n // 2
    arr = np.zeros(n, dtype=int)
    arr[:half] = 1
    np.random.shuffle(arr)
    return arr


def _hill_climb(h, best_c5, budget, start_time):
    """
    Simple hill‑climbing that swaps a 1 and a 0.
    Stops when time expires.
    """
    n = len(h)
    dx = 2.0 / n
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1 - h)

    while time.time() - start_time < budget:
        i = random.choice(ones)
        j = random.choice(zeros)

        # propose swap
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1

        c5_new = _evaluate(h_new)

        if c5_new + 1e-12 < best_c5:
            # accept improvement
            h = h_new
            best_c5 = c5_new
            # update index lists efficiently
            ones[ones == i] = j
            zeros[zeros == j] = i
        # occasional random restart to avoid stagnation
        if random.random() < 0.001:
            # random re‑shuffle a small fraction of entries
            idx = np.random.choice(n, size=n // 20, replace=False)
            sub = h[idx]
            np.random.shuffle(sub)
            h[idx] = sub
            ones = np.flatnonzero(h)
            zeros = np.flatnonzero(1 - h)

    return h, best_c5
