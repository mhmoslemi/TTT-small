import numpy as np, time, math, random

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def compute_grad(h, shift):
    n = len(h)
    grad = np.ones(n)
    if shift >= 0:
        if shift != 0:
            grad[:n-shift] -= h[shift:]          # term 1 - h[i+shift]
            grad[shift:] -= h[:n-shift]          # term - h[i-shift]
        else:
            grad -= 2 * h
    else:
        d = -shift
        grad[d:] -= h[:n-d]
        grad[:n-d] -= h[d:]
    return grad

def max_corr_and_shift(h):
    n = len(h)
    conv = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(conv)
    max_val = conv[max_idx]
    shift = max_idx - (n - 1)
    return max_val, shift

def hill_climb(initial_h, time_limit, seed=0):
    rng = np.random.RandomState(seed)
    n = len(initial_h)
    dx = 2.0 / n
    best_h = initial_h.copy()
    best_obj = evaluate(best_h)
    cur_h = best_h.copy()
    cur_obj = best_obj

    start = time.time()
    # parameters
    T0 = 0.01
    # you may adapt T schedule based on remaining time
    # loop
    iteration = 0
    while True:
        now = time.time()
        if now - start > time_limit - 0.5:
            break
        iteration += 1

        # compute shift of max correlation
        max_val, shift = max_corr_and_shift(cur_h)

        # compute gradient
        grad = compute_grad(cur_h, shift)

        # find positive and negative gradient indices
        pos_idx = np.where(grad > 1e-9)[0]
        neg_idx = np.where(grad < -1e-9)[0]

        # decide move type
        if len(pos_idx) > 0 and len(neg_idx) > 0 and rng.rand() < 0.8:
            i = rng.choice(pos_idx)
            j = rng.choice(neg_idx)
        else:
            i, j = rng.choice(n, size=2, replace=False)
            # ensure direction correct: move from i->j
        # compute max possible delta
        max_delta = min(cur_h[i], 1 - cur_h[j])
        if max_delta <= 0:
            continue
        # step size
        delta = max_delta * rng.rand() * 0.5
        if delta <= 0:
            continue

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_obj = evaluate(new_h)

        # temperature schedule overtime
        # e.g., T = T0 * (1 - iteration / max_iter)
        elapsed = now - start
        remain = time_limit - elapsed
        # approximate total iterations expected: we can base on time.
        # Let's compute T decreasing linearly from T0 to 1e-6 over total allowed time.

            cur_obj = new_obj

                best_obj = cur_obj
        # occasional random perturbation:
        if iteration % 20000 == 0:

def random_feasible(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1.0
    # add small noise
    noise = rng.normal(scale=0.1, size=n)
    h += noise
    np.clip(h, 0, 1, out=h)
    # re-balance sum
    total = h.sum()
    target = n * 0.5
    diff = target - total
    # distribute diff uniformly across all entries not at bounds (or simply add diff/n to each and clip)
    h += diff / n
    np.clip(h, 0, 1, out=h)
    # after this sum may still be off small due to clipping; adjust first non-bound entry
    total2 = h.sum()
    if abs(total2 - target) > 1e-8:
        # adjust using one index
        for idx2 in range(n):
            if 0 < h[idx2] < 1:
                h[idx2] += target - total2
                break
    return h

import time
import math
import numpy as np

# ---------- helper functions ----------
def _objective(h):
    """Return the C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _max_corr_and_shift(h):
    """Maximum correlation value and corresponding integer shift."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(conv))
    shift = idx - (n - 1)
    return conv[idx], shift


def _grad_shift(h, shift):
    """
    Gradient of the correlation (for the shift that attains the max)
    with respect to each entry of h.
    """
    n = h.shape[0]
    grad = np.ones(n)
    if shift >= 0:
        if shift != 0:
            grad[: n - shift] -= h[shift:]
            grad[shift:] -= h[: n - shift]
        else:
            grad -= 2.0 * h
    else:
        d = -shift
        grad[d:] -= h[: n - d]
        grad[: n - d] -= h[d:]
    return grad


def _random_binary_feasible(n, rng):
    """Generate a binary feasible vector (exactly n/2 ones)."""
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    # add small jitter and re‑balance to stay feasible
    noise = rng.normal(scale=0.08, size=n)
    h = np.clip(h + noise, 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    h += diff / n  # small uniform shift
    np.clip(h, 0.0, 1.0, out=h)
    # final tiny correction if needed
    cur = h.sum()
    if abs(cur - target) > 1e-12:
        for i in range(n):
            if 0.0 < h[i] < 1.0:
                h[i] += target - cur
                break
    return h


def _hill_climb(start_h, time_limit, seed):
    """
    Simple simulated‑annealing / hill‑climbing on the feasible set.
    The sum constraint is preserved exactly by moving mass between
    two coordinates.
    """
    rng = np.random.RandomState(seed)
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_val = _objective(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    T0 = 0.015  # initial temperature for occasional uphill steps

    while True:
        now = time.time()
        if now - start >= time_limit - 0.5:
            break
        it += 1

        # shift that currently gives the worst (largest) value
        _, shift = _max_corr_and_shift(cur_h)
        grad = _grad_shift(cur_h, shift)

        pos = np.where(grad > 1e-9)[0]
        neg = np.where(grad < -1e-9)[0]

        if len(pos) and len(neg) and rng.rand() < 0.85:
            i = rng.choice(pos)
            j = rng.choice(neg)
        else:
            i, j = rng.choice(n, 2, replace=False)

        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.rand() * 0.5
        if delta <= 0.0:
            continue

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_val = _objective(new_h)

        # temperature linearly decays to near zero
        elapsed = now - start
        T = T0 * (1.0 - elapsed / time_limit) + 1e-12
        if new_val < cur_val - 1e-12 or rng.rand() < math.exp(-(new_val - cur_val) / T):
            cur_h = new_h
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_h = cur_h.copy()
                best_val = cur_val

        # occasional perturbation to avoid stagnation
        if it % 25000 == 0:
            pert = rng.normal(scale=0.02, size=n)
            tmp = best_h + pert
            np.clip(tmp, 0.0, 1.0, out=tmp)
            target = n * 0.5
            diff = target - tmp.sum()
            tmp += diff / n
            np.clip(tmp, 0.0, 1.0, out=tmp)
            cur_h = tmp
            cur_val = _objective(cur_h)

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)

    # initial construction (provided by the harness if any)
