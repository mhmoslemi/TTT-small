def compute_c5(h):
    n = len(h)
    nfft = 1 << (2*n-1).bit_length()
    # compute FFT of h and reversed complement

def compute_c5(h):

    H = np.fft.rfft(h, nfft)

def propose_move(h, step):
    n = len(h)

    if i == j:
        return None

    if delta_low >= delta_high:

    # propose uniform random within [-step, step] limited by feasible range
    delta = np.random.uniform(-step, step)
    delta = max(delta_low, min(delta, delta_high))
    return i, j, delta

def anneal(h0, budget):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_val = best_val
    step = 0.2
    iter = 0

        # schedule
        elapsed = time.time() - start
        frac = elapsed / budget
        T = max(1e-4, 0.1 * (1 - frac))  # linear cooling
        # propose
        mv = propose_move(h, step)
        if mv is None:

        i,j,delta = mv

        h[i] += delta
        h[j] -= delta
        val = compute_c5(h)

            h[j] += delta
        iter+=1
        # adjust step every some iterations

            step = max(0.01, step * 0.9)  # reduce step

import numpy as np
import time

def _project_to_constraints(h):
    """Project h onto the set {0<=h_i<=1, sum(h)=len(h)/2}."""
    n = h.size
    target = n / 2.0
    # binary search for a shift lambda such that sum(clip(h+lam)) = target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h = np.clip(h + lam, 0.0, 1.0)
    # tiny correction if rounding left a residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt:
            h[mask] += residual / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def _c5_value(h):
    """Return C5 = dx * max_k convolution(h, 1-h)."""
    n = h.size
    dx = 2.0 / n
    # length needed for linear convolution = 2*n-1
    nfft = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft(np.flip(1.0 - h), n=nfft)
    conv = np.fft.irfft(H * G, n=nfft)[: 2 * n - 1]
    return dx * float(np.max(conv.real))

def _random_binary(n):
    """Binary vector of length n with exactly n/2 ones."""
    half = n // 2
    h = np.zeros(n, dtype=float)
    idx = np.random.choice(n, half, replace=False)
    h[idx] = 1.0
    return h

def _propose_move(h, step):
    """Return (i, j, delta) for a feasible mass transfer, or None."""
    n = h.size
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return None
    hi, hj = h[i], h[j]
    # feasible delta so that 0 <= hi+delta <= 1, 0 <= hj-delta <= 1
    delta_min = -min(hi, 1.0 - hj)      # decreasing i, increasing j
    delta_max =  min(1.0 - hi, hj)      # increasing i, decreasing j
    if delta_min >= delta_max:
        return None
    delta = np.random.uniform(-step, step)
    delta = max(delta_min, min(delta, delta_max))
    if abs(delta) < 1e-14:
        return None
    return i, j, delta

def _anneal(h0, budget):
    """Simulated annealing on h0 respecting the constraints."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_value(h)
    cur_val = best_val
    step = 0.3
    it = 0
    no_improve = 0

    while time.time() - start < budget:
        mv = _propose_move(h, step)
        if mv is None:
            continue
        i, j, delta = mv
        h[i] += delta
        h[j] -= delta
        new_val = _c5_value(h)

        # temperature decays linearly
        elapsed = time.time() - start
        frac = elapsed / budget
        T = max(1e-5, 0.2 * (1.0 - frac))

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if np.random.rand() < np.exp((cur_val - new_val) / T):
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert
            h[i] -= delta
            h[j] += delta

        it += 1
        if it % 5000 == 0:
            # slowly shrink step size
            step = max(0.01, step * 0.92)
            if no_improve > 50000:
                # restart from best found so far
                h = best_h.copy()
                cur_val = best_val
                no_improve = 0

        if best_val < 0.3808:   # target reached – stop early
            break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_total = time.time()

    # choose a reasonable number of sample points
    n = 256  # even, gives target sum n/2
    # obtain a starting vector
    starts = []
    init = globals().get("initial_h_values")
    if init is not None:
        # upsample initial to n points via linear interpolation
        src_x = np.linspace(0.0, 2.0, init.size, endpoint=False)
        tgt_x = np.linspace(0.0, 2.0, n, endpoint=False)
        h_up = np.interp(tgt_x, src_x, init.astype(float))
        h_up = _project_to_constraints(h_up)
        starts.append(h_up)

    # add a few random binary starts
    for _ in range(3):
        starts.append(_random_binary(n))

    # allocate time roughly equally among starts
    remaining = budget_s - (time.time() - start_total)
    if remaining <= 0:
        # fallback – use first start
        best_h = starts[0]
        best_val = _c5_value(best_h)
        return best_h.tolist(), best_val, n

    per_start = remaining / len(starts)
    best_overall_h = None
    best_overall_val = float('inf')

    for h0 in starts:
        # give each start a slice of the remaining budget
        alloc = per_start * 0.97  # tiny reserve for finalisation
        h_opt, val_opt = _anneal(h0, alloc)
        if val_opt < best_overall_val:
            best_overall_val = val_opt
            best_overall_h = h_opt.copy()
        if time.time() - start_total > budget_s * 0.97:
            break

    # final projection to guarantee feasibility (numeric drift)
    best_overall_h = _project_to_constraints(best_overall_h)
    best_overall_val = _c5_value(best_overall_h)

    return best_overall_h.tolist(), best_overall_val, n
