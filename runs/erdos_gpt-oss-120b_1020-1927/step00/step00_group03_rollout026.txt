def project_box_sum(v, target):
    # solve for lambda via bisection:
    # note: function sum_clip(lambda) monotonic decreasing in lambda.
    lo = -1.0  # because v_i - lambda could exceed 1 easily;
    hi = 1.0   # Actually safe bound large.
    # Expand bounds to ensure sum at lo > target and at hi < target.
    # We'll expand until condition holds.
    def sum_clip(lam):
        return np.clip(v - lam, 0.0, 1.0).sum()
    # Expand hi until sum <= target
    while sum_clip(hi) > target:
        hi *= 2.0
        if hi > 1e9:
            break
    while sum_clip(lo) < target:
        lo -= 2.0
        if lo < -1e9:
            break
    for _ in range(60):
        lam = (lo + hi) / 2.0
        s = sum_clip(lam)
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0.0, 1.0)
    # It should sum close to target, but small numerical error may remain; adjust by scaling residual among non-bounded entries.
    # However due to rounding, we can adjust a bit:
    diff = target - x.sum()

import numpy as np, time

def project_box_sum(v, target):
    # Bisection on lambda to get projection onto [0,1] and sum constraint.
    low = -1.0
    high = 1.0
    # adjust bounds
    def s(l):
        return np.clip(v - l, 0.0, 1.0).sum()
    # Expand high if sum > target
    while s(high) > target:
        high *= 2.0
    while s(low) < target:
        low -= 2.0
    for _ in range(60):
        mid = (low + high) * 0.5
        sm = s(mid)
        if sm > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    # small residual distribution
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    # use evaluate if present for precise result
    try:
        return evaluate_erdos_solution(h)
    except NameError:
        corr = np.correlate(h, 1.0 - h, mode='full')
        return corr.max() * dx

def gradient_step(h, step):
    n = h.size
    target = n / 2.0
    corr = np.correlate(h, h, mode='full')
    m = corr.min()
    eps = 1e-12
    ks = np.where(corr <= m + eps)[0] - (n - 1)
    g = np.zeros_like(h)
    for k in ks:
        if k >= 0:
            left = slice(0, n - k)
            right = slice(k, n)
        else:
            k_abs = -k
            left = slice(k_abs, n)
            right = slice(0, n - k_abs)
        g[left] += h[right]
        g[right] += h[left]
    h_new = h + step * g
    return project_box_sum(h_new, target)

def hill_climb(h, iters=2000, step0=0.4, decay=0.97):
    step = step0
    for i in range(iters):
        # we could adapt step, but simple schedule
        h = gradient_step(h, step)
        step *= decay
    return h

def simulated_annealing(h, max_iter=15000, init_temp=0.02, max_delta=0.05):
    n = h.size
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_h = h.copy()
    cur_val = best_val
    for it in range(max_iter):
        # linear cooling
        temp = init_temp * (1 - it / max_iter)
        # decide direction random
        i, j = np.random.randint(0, n, size=2)
        while i == j:
            j = np.random.randint(0, n)
        # choose direction
        if np.random.rand() < 0.5:
            delta = min(max_delta, cur_h[i], 1.0 - cur_h[j])
            if delta <= 0:
                continue
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            delta = min(max_delta, cur_h[j], 1.0 - cur_h[i])
            if delta <= 0:
                continue
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        new_val = compute_c5(new_h)

            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    return best_h, best_val

def make_step_shape(n, a_frac):
    # a_frac ∈ [0,0.5] fraction of total length
    dx = 2.0 / n
    a = a_frac
    if a >= 0.5:
        a = 0.5
    # compute b, handle a=1 case
    if a >= 1.0 - 1e-12:
        b = 0.0
    else:

    x = (np.arange(n) + 0.5) * dx
    h = np.where((x <= a) | (x >= 2 - a), 1.0, b)
    # project to ensure exact sum
    target = n / 2.0
    h = project_box_sum(h, target)
    return h

import time
import random
import numpy as np

# -------------------------------------------------------------
# Projection of a vector onto the intersection of a box and a
# hyperplane {x : sum(x) = target, 0 <= x_i <= 1}
def _proj_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    lo, hi = -1.0, 1.0

    def _s(lam):
        return np.clip(v - lam, 0.0, 1.0).sum()

    # expand bounds until they bracket the solution
    while _s(hi) > target:
        hi *= 2.0
    while _s(lo) < target:
        lo -= 2.0

    for _ in range(60):
        lam = (lo + hi) * 0.5
        s = _s(lam)
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    # tiny numerical residue – distribute it over free variables
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

# Gradient step that raises the smallest self‑correlation
def _gradient_step(h: np.ndarray, step_sz: float) -> np.ndarray:
    n = h.size
    target = n * 0.5
    corr = np.correlate(h, h, mode='full')
    m = corr.min()
    eps = 1e-12
    ks = np.where(corr <= m + eps)[0] - (n - 1)   # lag values attaining the minimum
    g = np.zeros_like(h)

    for k in ks:
        if k >= 0:
            left = slice(0, n - k)
            right = slice(k, n)
        else:
            kabs = -k
            left = slice(kabs, n)
            right = slice(0, n - kabs)
        g[left] += h[right]
        g[right] += h[left]

    h_new = h + step_sz * g
    return _proj_box_sum(h_new, target)


def _hill_climb(start_h: np.ndarray, iters: int = 800,
                step0: float = 0.4, decay: float = 0.98) -> np.ndarray:
    h = start_h.copy()
    step = step0
    for _ in range(iters):
        h = _gradient_step(h, step)
        step *= decay
    return h

# Simple simulated‑annealing refinement (preserves the sum exactly)
def _simulated_anneal(start_h: np.ndarray,
                      max_iter: int = 12000,
                      init_temp: float = 0.02,
                      max_delta: float = 0.05):
    n = start_h.size
    best_h = cur_h = start_h.copy()
    best_val = cur_val = _c5_value(cur_h)

    for it in range(max_iter):
        temp = init_temp * (1.0 - it / max_iter)

        i, j = np.random.randint(0, n, size=2)
        while i == j:
            j = np.random.randint(0, n)

        # pick a direction and a feasible delta
        if np.random.rand() < 0.5:
            delta = min(max_delta, cur_h[i], 1.0 - cur_h[j])
            if delta <= 0:
                continue
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            delta = min(max_delta, cur_h[j], 1.0 - cur_h[i])
            if delta <= 0:
                continue
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta

        new_val = _c5_value(new_h)
        accept = (new_val < cur_val) or (np.random.rand() <
                                         np.exp((cur_val - new_val) / max(temp, 1e-12)))
        if accept:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val

    return best_h, best_val

# Helper to build a two‑level step shape (high on the ends, low inside)
def _make_step_shape(n: int, a_frac: float) -> np.ndarray:
    # a_frac = length of each high end segment (0 <= a <= 0.5)
    a = max(0.0, min(0.5, a_frac))
    # interior value b obtained from the area constraint
    if a >= 1.0 - 1e-12:

    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.where((x <= a) | (x >= 2.0 - a), 1.0, b)
    # small projection to meet the exact sum requirement
    return _proj_box_sum(h, n * 0.5)

    final_val = _c5_value(best_h)
