def anneal(h, budget, start_time):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_score = np.max(np.correlate(h, 1-h, mode='full')) * dx
    cur_h = h.copy()
    cur_score = best_score
    T0 = 0.05
    iteration = 0
    while time.time() - start_time < budget:
        # propose move
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j: continue
        # max delta
        max_inc = min(0.2, 1 - cur_h[i])
        max_dec = min(0.2, cur_h[j])
        if max_inc <= 0 or max_dec <=0: continue
        delta = np.random.rand() * min(max_inc, max_dec)
        # Random direction: we could also decrease i, increase j but we already choose sign.
        # Apply
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # evaluate
        new_score = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
        # acceptance
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / (T0 * (0.99**iteration))):
            cur_h, cur_score = new_h, new_score
            if cur_score < best_score:
                best_h, best_score = cur_h.copy(), cur_score
        iteration += 1
    return best_h, best_score

import time, math, random
import numpy as np

def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] with sum(v)=target."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _random_feasible(n: int, target: float) -> np.ndarray:
    """Uniform random point in [0,1]^n with prescribed sum."""
    v = np.random.rand(n)
    return _project_to_sum(v, target)


def _c5_score(h: np.ndarray) -> float:
    """Compute the overlap constant for a sampled function h."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a step function h (as a list), its C5 bound, and the number of points.
    The algorithm starts from the supplied `initial_h_values` (if present),
    projects it to a feasible vector of length 200, then performs a simple
    simulated‑annealing hill‑climb while respecting the constraints.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ---- parameters ---------------------------------------------------------
    n_points = 200                     # < 1000 as required
    target_sum = n_points / 2.0        # because ∫ h = 1  and dx = 2/n
    dx = 2.0 / n_points

    # ---- initial feasible vector --------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n0 = init.size
        # linear interpolation to the target grid
        src_x = np.linspace(0.0, 2.0, n0, endpoint=False)
        dst_x = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.interp(dst_x, src_x, init)
        h = np.clip(h, 0.0, 1.0)
        h = _project_to_sum(h, target_sum)
    except Exception:
        # fallback: completely random feasible vector
        h = _random_feasible(n_points, target_sum)

    best_h = h.copy()
    best_score = _c5_score(best_h)

    start_time = time.time()
    time_limit = budget_s * 0.97  # leave a small margin

    # ---- annealing parameters ------------------------------------------------
    T0 = 0.05            # initial temperature
    decay = 0.9995       # exponential decay factor per iteration
    max_delta = 0.18     # maximal mass transferred in a single move

    # main optimisation loop
    cur_h = best_h.copy()
    cur_score = best_score
    T = T0
    it = 0
    while time.time() - start_time < time_limit:
        # pick two distinct indices
        i, j = np.random.randint(n_points), np.random.randint(n_points)
        if i == j:
            continue

        # how much can be moved from j to i while staying in [0,1] ?
        max_inc = min(max_delta, 1.0 - cur_h[i])
        max_dec = min(max_delta, cur_h[j])
        if max_inc <= 0.0 or max_dec <= 0.0:
            continue
        delta = np.random.rand() * min(max_inc, max_dec)

        # propose the new vector
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        # evaluate
        new_score = _c5_score(new_h)

        # Metropolis acceptance
        if new_score < cur_score or random.random() < math.exp(-(new_score - cur_score) / T):
            cur_h, cur_score = new_h, new_score
            if cur_score < best_score:
                best_h, best_score = cur_h.copy(), cur_score

        # temperature schedule
        T = max(1e-7, T * decay)
        it += 1

        # occasional restart to avoid stagnation
        if it % 50000 == 0:
            if random.random() < 0.5:
                # jump back to the best found so far
                cur_h = best_h.copy()
                cur_score = best_score
                T = T0
            else:
                # fresh random feasible start
                cur_h = _random_feasible(n_points, target_sum)
                cur_score = _c5_score(cur_h)
                T = T0

    # final clean‑up: ensure feasibility exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_sum(best_h, target_sum)
    final_score = _c5_score(best_h)

    # return as required
    return best_h.tolist(), float(final_score), n_points
