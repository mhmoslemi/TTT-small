def c5_from_h(h):
    n = len(h)
    # compute correlation using np.correlate

def local_optimize(h, max_iters, time_limit):
    n = len(h)
    cur_h = h.copy()
    cur_val = c5_from_h(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

        if time.time() - start > time_limit:

        # temperature schedule

        # pick indices

import time
import numpy as np
import random

def _c5_value(h):
    """Return C5 for a discrete step function h (array of 0/1 or floats)."""
    n = h.shape[0]
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    return max_corr * (2.0 / n)


def _targeted_optimize(start_h, max_iters, time_limit):
    """
    Simple targeted hill‑climb / simulated annealing.
    Swaps a 1 with a 0 that participates in the current worst shift.
    """
    cur = start_h.copy()
    n = cur.shape[0]
    cur_val = _c5_value(cur)
    best = cur.copy()
    best_val = cur_val
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        # compute full correlation to locate the worst shift
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        max_c = corr.max()
        max_pos = np.where(corr == max_c)[0]
        shift_idx = random.choice(max_pos.tolist())
        shift = shift_idx - (n - 1)          # actual integer shift

        # indices that contribute to the worst overlap
        if shift >= 0:
            mask = (cur[:n - shift] == 1) & (cur[shift:] == 0)
            bad_one = np.where(mask)[0]
        else:
            s = -shift
            mask = (cur[s:] == 1) & (cur[:n - s] == 0)
            bad_one = np.where(mask)[0] + s

        # if nothing contributes (unlikely) fall back to random swap
        if bad_one.size == 0:
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            i = random.choice(ones.tolist())
            j = random.choice(zeros.tolist())
        else:
            i = random.choice(bad_one.tolist())
            zeros = np.where(cur == 0)[0]
            # avoid swapping the same position
            if zeros.size == 1 and zeros[0] == i:
                continue
            j = random.choice(zeros.tolist())

        # propose swap
        cur[i], cur[j] = 0, 1
        new_val = _c5_value(cur)

        # accept only if not worsening (pure descent)
        if new_val <= cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
        else:
            # revert
            cur[i], cur[j] = 1, 0

    return best, best_val
