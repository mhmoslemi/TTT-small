import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant for a given step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def overlapping_indices(n: int, lag: int):
    """Return index arrays i, j such that j = i + lag and both lie in [0,n)."""
    if lag >= 0:
        i = np.arange(0, n - lag, dtype=int)
        j = i + lag
    else:
        d = -lag
        i = np.arange(d, n, dtype=int)
        j = i - d
    return i, j


def make_targeted_move(h: np.ndarray, lag: int, rng: np.random.Generator, max_frac: float = 0.5):
    """
    Generate a candidate vector by moving a small amount of mass from a donor
    to a recipient that is expected to lower the worst lag `lag`.
    Returns (candidate_vector, (donor, recipient, delta)) or (None, None)
    if no suitable pair is found.
    """
    if lag == 0:
        return None, None
    n = h.size
    i_idx, j_idx = overlapping_indices(n, lag)
    if i_idx.size == 0:
        return None, None

    shift_vals = h[j_idx]                     # h[i+lag] for the overlapped i's
    donor_mask = (h[i_idx] > 1e-12) & (shift_vals < 0.5)
    recpt_mask = (h[i_idx] < 1.0 - 1e-12) & (shift_vals > 0.5)

    donors = i_idx[donor_mask]
    recpts = i_idx[recpt_mask]
    if donors.size == 0 or recpts.size == 0:
        return None, None

    donor = rng.choice(donors)
    recpt = rng.choice(recpts)

    max_delta = min(h[donor], 1.0 - h[recpt])
    if max_delta <= 0:
        return None, None
    delta = max_delta * max_frac * rng.random()
    if delta < 1e-12:
        delta = max_delta * max_frac

    cand = h.copy()
    cand[donor] -= delta
    cand[recpt] += delta
    return cand, (donor, recpt, delta)


def propose_random_move(h: np.ndarray, rng: np.random.Generator, max_frac: float = 0.5):
    """Move a random amount of mass from a random positive entry to a random non‑full entry."""
    n = h.size
    attempts = 0
    while attempts < 10:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i != j and h[i] > 1e-12 and h[j] < 1.0 - 1e-12:
            break
        attempts += 1
    else:
        return None

    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0:
        return None
    delta = max_delta * max_frac * rng.random()
    if delta < 1e-12:
        delta = max_delta * max_frac

    cand = h.copy()
    cand[i] -= delta
    cand[j] += delta
    return cand


def random_balanced_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a balanced vector of length n with ∑h = n/2 (fractional values allowed)."""
    h = np.zeros(n, dtype=float)
    half = n // 2
    ones = rng.choice(n, half, replace=False)
    h[ones] = 1.0
    if n % 2 == 1:
        remaining = np.where(h == 0)[0]
        h[rng.choice(remaining)] = 0.5
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise starting vector
    if "initial_h_values" in globals():
        cur = np.array(initial_h_values, dtype=float).copy()
    else:
        # a modest fallback size
        cur = random_balanced_vector(100, rng)

    n = cur.size
    target_sum = n / 2.0

    # ensure the sum constraint (tiny adjustments)
    diff = target_sum - cur.sum()
    if abs(diff) > 1e-12:
        for idx in range(n):
            low = -cur[idx]
            high = 1.0 - cur[idx]
            add = np.clip(diff, low, high)
            cur[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    cur = np.clip(cur, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Simulated‑annealing style local search
    best_h = cur.copy()
    best_c = compute_c5(best_h)

    cur_h = cur.copy()
    cur_c = best_c

    start = time.time()
    # use ~97 % of the budget for optimisation, keep a tiny margin for return
    time_limit = budget_s * 0.97

    while time.time() - start < time_limit:
        # worst lag for the current configuration
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(corr))
        lag = max_idx - (n - 1)

        # try a directed move that should lower the worst lag
        cand, _ = make_targeted_move(cur_h, lag, rng, max_frac=0.4)

        # fall back to a completely random mass transfer if directed move fails
        if cand is None:
            cand = propose_random_move(cur_h, rng, max_frac=0.6)

        if cand is None:
            continue

        cand_c = compute_c5(cand)

        # keep the best ever seen
        if cand_c < best_c:
            best_h = cand.copy()
            best_c = cand_c

        # accept/reject for the current walk (simple annealing)
        elapsed = time.time() - start
        temperature = max(0.001, 0.01 * (1.0 - elapsed / time_limit))
        if cand_c < cur_c or rng.random() < np.exp(-(cand_c - cur_c) / temperature):
            cur_h, cur_c = cand, cand_c

    # ------------------------------------------------------------------
    # final housekeeping: clip and enforce the sum exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    final_diff = target_sum - best_h.sum()
    if abs(final_diff) > 1e-12:
        for idx in range(best_h.size):
            low = -best_h[idx]
            high = 1.0 - best_h[idx]
            add = np.clip(final_diff, low, high)
            best_h[idx] += add
            final_diff -= add
            if abs(final_diff) < 1e-12:
                break

    final_c5 = compute_c5(best_h)
    return best_h, float(final_c5), best_h.size
