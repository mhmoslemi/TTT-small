def optimize_h(initial_h, max_iters, time_limit):
    n = len(initial_h)
    h = initial_h.copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    indices_one = np.flatnonzero(h == 1)
    indices_zero = np.flatnonzero(h == 0)
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        # Temperature gradually
        T = max(0.001, 0.01 * (1 - it / max_iters))
        # propose swap
        i = np.random.choice(indices_one)
        j = np.random.choice(indices_zero)
        # apply swap
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        accept = False
        if delta <= 0:
            accept = True
        else:
            if np.random.rand() < np.exp(-delta / T):
                accept = True
        if accept:
            current_c5 = new_c5
            # update lists
            # remove i from ones, add to zeros; similarly for j

            indices_one = np.setdiff1d(indices_one, np.array([i]), assume_unique=True)

                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
        # optional: maybe restart if stagnated

import numpy as np
import time
import random
import math

def compute_c5(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1-h(x+k))dx for the discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation with zero padding (np.correlate implements this)
    conv = np.correlate(h, 1 - h, mode="full")
    return float(conv.max() * dx)


def random_binary_vector(n: int) -> np.ndarray:
    """Binary vector of length n with exactly n/2 ones (n must be even)."""
    arr = np.zeros(n, dtype=int)
    arr[: n // 2] = 1
    np.random.shuffle(arr)
    return arr


def _worst_shift_and_mismatches(h: np.ndarray):
    """Find a shift attaining the maximal overlap and the list of indices
    i where h[i]=1 and h[i+shift]=0 (i.e. contributing to the worst value)."""
    n = h.shape[0]
    conv = np.correlate(h, 1 - h, mode="full")
    max_val = conv.max()
    idx = int(np.argmax(conv))                 # index in full correlation
    shift = idx - (n - 1)                      # actual integer shift (negative possible)

    if shift >= 0:
        left = h[: n - shift]
        right = h[shift:]
        mask = (left == 1) & (right == 0)
        mismatches = np.where(mask)[0]         # indices in the original array
    else:
        s = -shift
        left = h[s:]
        right = h[: n - s]
        mask = (left == 1) & (right == 0)
        mismatches = np.where(mask)[0] + s     # offset back to original indexing
    return shift, mismatches, max_val


def hill_climb(start_h: np.ndarray, max_iters: int, deadline: float):
    """Local optimisation starting from start_h.
    Returns a (best_h, best_c5) pair found before deadline."""
    h = start_h.copy()
    n = h.shape[0]
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        shift, mismatches, _ = _worst_shift_and_mismatches(h)

        # choose a 1 which actually contributes to the worst shift if possible
        if mismatches.size:
            i = int(np.random.choice(mismatches))
        else:
            ones = np.flatnonzero(h)
            i = int(np.random.choice(ones))

        zeros = np.flatnonzero(1 - h)
        j = int(np.random.choice(zeros))

        # perform the swap
        h[i], h[j] = h[j], h[i]

        new_c5 = compute_c5(h)

        # simulated‑annealing style acceptance
        if new_c5 <= cur_c5:
            accept = True
        else:
            # temperature decays linearly; scale chosen heuristically
            T = max(1e-4, cur_c5 * 0.02 * (1.0 - it / max_iters))
            prob = math.exp(-(new_c5 - cur_c5) / T)
            accept = prob > random.random()

        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert the swap
            h[i], h[j] = h[j], h[i]

    return best_h, best_c5


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C5 bound and the number of points."""
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # initialise a base vector (use the supplied construction if present)
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values", None)
    if init is not None:
        base = np.array(init, dtype=int)
        base = (base > 0.5).astype(int)          # enforce strict 0/1
    else:
        # fallback: a simple balanced binary vector of length 70
        n0 = 70
        base = np.concatenate([np.ones(n0 // 2, dtype=int),
                               np.zeros(n0 // 2, dtype=int)])
        np.random.shuffle(base)

    # enlarge the problem size by repeating the base pattern.
    # repeating preserves the integral constraint automatically.
    repeat_factor = 4                               # 4 → 280 points for a 70‑point base
    h_start = np.tile(base, repeat_factor)
    n = h_start.shape[0]

    # ensure we really have exactly n/2 ones (it should by construction)
    assert h_start.sum() == n // 2

    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 0.5)   # tiny safety margin

    # ------------------------------------------------------------------
    # main optimisation loop – a mixture of random restarts and refinement
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # half the time start from the current best, half from a fresh random vector
        if random.random() < 0.5:
            start_vec = best_h.copy()
        else:
            start_vec = random_binary_vector(n)

        # a modest number of iterations; the inner loop respects the global deadline
        local_h, local_c5 = hill_climb(start_vec, max_iters=3000, deadline=deadline)

        if local_c5 < best_c5:
            best_c5 = local_c5
            best_h = local_h.copy()
            if best_c5 <= 0.3808:
                break

    # ------------------------------------------------------------------
    # final answer
    # ------------------------------------------------------------------
    final_c5 = compute_c5(best_h)
    return best_h.tolist(), float(final_c5), int(n)
