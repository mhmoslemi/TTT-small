    corr = np.correlate(h, 1 - h, mode='full')

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: compute the Erdős overlap constant C5 for a given vector h
# ----------------------------------------------------------------------
def _c5_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


# ----------------------------------------------------------------------
# Fallback: simple simulated‑annealing hill climber for binary vectors
# ----------------------------------------------------------------------
def _binary_hill_climb(n, target_sum, time_budget, rng, start_vec=None):
    """Return a binary vector of length n with exactly target_sum ones."""
    # initialise
    if start_vec is None:
        h = np.zeros(n, dtype=np.int8)
        h[:target_sum] = 1
        rng.shuffle(h)
    else:
        h = start_vec.copy().astype(np.int8)

    best_h = h.copy()
    best_score = _c5_value(best_h.astype(float))

    cur_h = h.copy()
    cur_score = best_score

    start = time.time()
    # temperature schedule (linear cooling)
    T0 = 0.02
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        T = max(0.0, T0 * (1.0 - elapsed / time_budget))

        # pick a random 1 and a random 0 and swap them
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        cand = cur_h.copy()
        cand[i] = 0
        cand[j] = 1

        cand_score = _c5_value(cand.astype(float))

        # accept if better or by Metropolis rule
        if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / max(T, 1e-9)):
            cur_h = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_h = cand.copy()
                best_score = cand_score

    return best_h.astype(float), best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    remaining = lambda: max(0.0, budget_s - (time.time() - start_time))

    # ------------------------------------------------------------------
    # 1) Try an exact mixed‑integer formulation (CBC)
    # ------------------------------------------------------------------
    try:
        import cvxpy as cp

        # use an even length to keep the integral exactly satisfiable with 0/1 values
        n = 100
        w = n // 2                     # required number of ones
        dx = 2.0 / n

        # decision variables
        h = cp.Variable(n, boolean=True)
        t = cp.Variable(integer=True)   # the maximal overlap (integer, will be integral)

        constraints = [cp.sum(h) == w, t >= 0]

        # auxiliary AND variables z_{i,j} = h_i ∧ h_j  (only for i<j)
        z = {}
        for i in range(n):
            for j in range(i + 1, n):
                z[(i, j)] = cp.Variable(boolean=True)
                constraints += [
                    z[(i, j)] <= h[i],
                    z[(i, j)] <= h[j],
                    z[(i, j)] >= h[i] + h[j] - 1,
                ]

        # overlap constraints for every non‑zero shift
        for k in range(1, n):
            # first part of the correlation (sum of h over the overlapping prefix)
            sum_prefix = cp.sum(h[: n - k])
            sum_suffix = cp.sum(h[k:])          # overlapping suffix
            # product part : sum of h_i h_{i+k} over the overlap
            prod = cp.sum([z[(i, i + k)] for i in range(n - k)])

            constraints += [sum_prefix - prod <= t]
            constraints += [sum_suffix - prod <= t]

        prob = cp.Problem(cp.Minimize(t), constraints)

        # give CBC a time limit so we never exceed the global budget
        time_limit = remaining()
        prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)

        if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
            t_val = int(round(t.value))
            # extract binary solution (round to 0/1 to protect against numerical tolerance)
            h_sol = np.rint(np.array([v.value for v in h])).astype(float)
            # repair any tiny sum deviation (should not happen, but be safe)
            diff = w - int(round(h_sol.sum()))
            if diff != 0:
                idx = np.where(h_sol == 0)[0] if diff > 0 else np.where(h_sol == 1)[0]
                if len(idx) >= abs(diff):
                    h_sol[idx[:abs(diff)]] = 1.0 if diff > 0 else 0.0
            c5 = _c5_value(h_sol)
            # If the MILP gave a bound good enough, return immediately
            if c5 <= 0.38085:
                return (h_sol.tolist(), c5, n)

            # otherwise keep this solution as a starting point for local search
            best_h = h_sol
            best_score = c5
        else:
            # fall back to random start if the solver failed
            best_h = None
            best_score = float("inf")
    except Exception:
        # any problem importing/solving – go straight to heuristic
        best_h = None
        best_score = float("inf")

    # ------------------------------------------------------------------
    # 2) Heuristic improvement (simulated annealing) – always executed
    # ------------------------------------------------------------------
    # use up to the remaining time (at least a few seconds)
    heuristic_time = max(0.0, remaining() - 5.0)  # keep a tiny reserve
    if heuristic_time > 2.0:
        # start vector: either the exact solution (if any) or a fresh random binary one
        start_vec = None
        if best_h is not None and best_h.shape[0] == 100:
            start_vec = np.rint(best_h).astype(np.int8)
        h_opt, opt_score = _binary_hill_climb(
            n=100,
            target_sum=50,
            time_budget=heuristic_time,
            rng=rng,
            start_vec=start_vec,
        )
        if opt_score < best_score:
            best_h = h_opt
            best_score = opt_score

    # ------------------------------------------------------------------
    # 3) Prepare final answer
    # ------------------------------------------------------------------
    if best_h is None:
        # As a last resort, return a simple balanced alternating pattern
        n = 100
        best_h = np.zeros(n, dtype=float)
        best_h[::2] = 1.0  # 1,0,1,0,...
        best_score = _c5_value(best_h)

    # final sanity checks
    best_h = np.clip(best_h, 0.0, 1.0)
    # enforce exact integral constraint (should already hold)
    target_sum = best_h.shape[0] / 2
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-8:
        # adjust by adding the needed fraction to the first element (tiny correction)
        best_h[0] += (target_sum - cur_sum)

    final_c5 = _c5_value(best_h.astype(float))
    return (best_h.tolist(), final_c5, best_h.shape[0])
