def run(seed=42, budget_s=1100, **kwargs):
    import time, random, numpy as np
    random.seed(seed)
    np.random.seed(seed)
    # Determine n_points
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=np.float64)
        n = init.shape[0]
        # Ensure binary
        if np.any((init > 0) & (init < 1)):
            # Convert to binary by choosing top half values as 1.
            h = binary_from_continuous(init)
        else:
            h = init.copy()
    else:
        n = 500
        if n % 2 != 0:
            n += 1
        # Sturmian initial pattern
        alpha = (np.sqrt(5) - 1) / 2
        frac = (np.arange(n) * alpha) % 1
        h = (frac < 0.5).astype(np.float64)
        # adjust sum to exactly n/2 if not equal
        diff = int(h.sum() - n//2)
        if diff > 0:
            # too many ones, turn diff ones to zero at random
            ones = np.where(h == 1)[0]
            turn = np.random.choice(ones, diff, replace=False)
            h[turn] = 0
        elif diff < 0:
            zeros = np.where(h == 0)[0]
            turn = np.random.choice(zeros, -diff, replace=False)
            h[turn] = 1
    # Ensure sum is n/2
    # if not exactly due to rounding we fix
    total_ones = int(np.sum(h))
    target = n // 2
    if total_ones > target:
        # turn some ones to zero
        ones = np.where(h == 1)[0]
        turn = np.random.choice(ones, total_ones - target, replace=False)
        h[turn] = 0
    elif total_ones < target:
        zeros = np.where(h == 0)[0]
        turn = np.random.choice(zeros, target - total_ones, replace=False)
        h[turn] = 1

    best_h = h.copy()
    best_score = max_overlap(best_h)
    current_h = h.copy()
    current_score = best_score

    # maintain lists of indices for ones and zeros
    ones_idx = list(np.where(current_h > 0.5)[0])
    zeros_idx = list(np.where(current_h < 0.5)[0])

    start = time.time()
    time_limit = budget_s * 0.95  # leave margin

    # annealing parameters
    T0 = 0.02
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        # sample a swap pair
        i = random.choice(ones_idx)
        j = random.choice(zeros_idx)
        # perform swap
        current_h[i], current_h[j] = current_h[j], current_h[i]
        new_score = max_overlap(current_h)

        # temperature

            ones_idx.remove(i)
            zeros_idx.append(i)

            # no change to lists
        # optional break if best_score at target lower bound maybe
        if best_score <= 0.3808:
            # we reached target, break early
            break

    # Return
    return (best_h.tolist(), float(best_score), n)

import time, random, math
import numpy as np

def binary_from_continuous(v: np.ndarray) -> np.ndarray:
    """Return a binary (0/1) vector with exactly half the entries equal to 1,
    chosen from the largest values of v."""
    n = v.shape[0]
    half = n // 2
    # indices of the half largest entries
    idx = np.argpartition(v, -half)[-half:]
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h

def overlap_array(h: np.ndarray) -> np.ndarray:
    """Full correlation of h with (1‑h) using FFT, multiplied by dx."""
    n = h.shape[0]
    b = 1.0 - h
    # length for zero‑padding (next power of two)
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, L)
    B = np.fft.rfft(b, L)
    conv = np.fft.irfft(H * np.conj(B), L)[: 2 * n - 1]
    conv *= 2.0 / n            # dx = 2 / n
    return conv

def max_overlap(h: np.ndarray) -> float:
    """Maximum overlap value for a given h."""
    return float(overlap_array(h).max())
