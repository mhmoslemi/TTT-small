def project_capsimple(v, target, ub=1.0):
    # Use water-filling binary search.
    lo = np.min(v) - ub  # maybe -ub
    hi = np.max(v) + 1e-8  # small offset.
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0, ub)
        s = x.sum()

        max_def = np.max(defects)

        conv_wh = np.convolve(w, h, mode='full')
        term2 = conv_wh[:n]

def project_capsimple(v, target_sum, ub=1.0):
    # v: array.
    # Goal: minimize ||x - v||^2 s.t. 0 <= x_i <= ub, sum x_i = target_sum.
    # Use water-filling with binary search.
    lo = -ub*2  # consider low enough
    hi = ub*2
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0, ub)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0, ub)

import time
import math
import numpy as np

def _project_to_capped_simplex(v: np.ndarray, target: float, ub: float = 1.0) -> np.ndarray:
    """
    Project vector `v` onto the set {x | 0 <= x_i <= ub, sum(x) = target}
    using a simple water‑filling (binary search on the threshold).
    """
    lo = -ub * 2.0
    hi = ub * 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, ub)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    thr = (lo + hi) / 2.0
    return np.clip(v - thr, 0.0, ub)


def _c5_max(h: np.ndarray, dx: float) -> float:
    """return max_k ∫ h(x)(1‑h(x+k))dx approximated by discrete correlation."""
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that minimises
    max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # target number of points (≤1000 for speed)
    n = 200
    dx = 2.0 / n
    target_sum = n / 2.0  # because sum(h)*dx = 1

    # generate a few diverse starting vectors
    starters = []

    x = (np.arange(n) + 0.5) * dx
    starters.append(np.minimum(x, 2.0 - x))

    # 3) random

    # 4) use parent construction if available

        starters.append(parent.copy())
    except NameError:
        pass  # no parent provided

    best_h = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # core optimisation – smooth max via log‑sum‑exp and projected gradient
    for init_h in starters:

        # hyper‑parameters
