def get_c5(h):
    n = len(h)
    total = np.sum(h)
    corr = np.correlate(h, h, mode='full')
    overlap = total - corr
    c5 = np.max(overlap) * (2.0 / n)
    return c5

import time, random, numpy as np

def compute_c5(arr, total, dx):
    corr = np.correlate(arr, arr, mode='full')
    min_corr = np.min(corr)
    return (total - min_corr) * dx

def shifted_view(arr, s):
    n = arr.shape[0]
    shifted = np.zeros_like(arr)
    if s >= 0:
        # shift right: arr[i+s] -> shifted[i]
        if s < n:
            shifted[:n - s] = arr[s:]
    else:
        # s negative: shift left
        k = -s
        if k < n:
            shifted[k:] = arr[:n - k]
    return shifted

def run(seed=42, budget_s=1100):
    import random, time, numpy as np
    random.seed(seed)
    np.random.seed(seed)

    # Initialize h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        # force binary
        if not np.all(np.logical_or(h0 == 0, h0 == 1)):
            h0 = (h0 > 0.5).astype(float)
        target = n // 2
        cur = int(h0.sum())
        if cur > target:
            idx = np.where(h0 == 1)[0]
            to_zero = np.random.choice(idx, cur - target, replace=False)
            h0[to_zero] = 0
        elif cur < target:
            idx = np.where(h0 == 0)[0]
            to_one = np.random.choice(idx, target - cur, replace=False)
            h0[to_one] = 1
        h = h0.copy()
    else:
        n = 200
        h = np.zeros(n, dtype=float)
        ones = np.random.choice(n, n // 2, replace=False)
        h[ones] = 1.0

    total = n / 2.0
    dx = 2.0 / n

    current_c5 = compute_c5(h, total, dx)
    best_h = h.copy()
    best_c5 = current_c5

    start = time.time()
    no_improve = 0
    max_no_improve = 5000   # restart after this many failed attempts

    while time.time() - start < budget_s * 0.95:
        corr = np.correlate(h, h, mode='full')
        idx_min = int(np.argmin(corr))
        shift = idx_min - (n - 1)

        h_shift = shifted_view(h, shift)
        ones_candidates = np.where((h == 1) & (h_shift == 0))[0]
        zeros_candidates = np.where((h == 0) & (h_shift == 1))[0]

        if len(ones_candidates) == 0 or len(zeros_candidates) == 0:
            # fallback: random swap between any 1 and any 0
            i = int(np.random.choice(np.where(h == 1)[0]))
            j = int(np.random.choice(np.where(h == 0)[0]))
        else:
            i = int(np.random.choice(ones_candidates))
            j = int(np.random.choice(zeros_candidates))

        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]

        new_c5 = compute_c5(h_new, total, dx)

        if new_c5 < current_c5 - 1e-12:
            h = h_new
            current_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                # restart with a fresh random configuration
                h = np.zeros(n, dtype=float)
                ones = np.random.choice(n, n // 2, replace=False)
                h[ones] = 1.0
                current_c5 = compute_c5(h, total, dx)
                no_improve = 0

    final_c5 = compute_c5(best_h, total, dx)
    # Convert result to Python list to avoid returning numpy arrays if not desired
    return (best_h.tolist(), float(final_c5), n)

import time
import random
import numpy as np

def compute_c5(arr: np.ndarray, total: float, dx: float) -> float:
    """
    Return the overlap bound C5 for a given step function.
    """
    corr = np.correlate(arr, arr, mode="full")
    min_corr = np.min(corr)                      # worst (smallest) auto‑correlation
    return float((total - min_corr) * dx)


def shifted_view(arr: np.ndarray, s: int) -> np.ndarray:
    """
    Return an array of the same length as `arr` such that
    out[i] = arr[i + s] if the index is inside the range,
    otherwise 0.  This implements the zero‑extension used in the
    definition of the overlap integral.
    """
    n = arr.shape[0]
    out = np.zeros_like(arr)
    if s >= 0:
        if s < n:
            out[: n - s] = arr[s:]
    else:
        k = -s
        if k < n:
            out[k:] = arr[: n - k]
    return out


def _initial_binary(h: np.ndarray) -> np.ndarray:
    """
    Convert a possibly non‑binary initial vector to a binary one
    preserving the total mass (exactly half the entries are 1).
    """
    n = h.shape[0]
    # round to {0,1}
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    cur = int(h_bin.sum())
    if cur > target:
        idx = np.where(h_bin == 1)[0]
        to_zero = np.random.choice(idx, cur - target, replace=False)
        h_bin[to_zero] = 0.0
    elif cur < target:
        idx = np.where(h_bin == 0)[0]
        to_one = np.random.choice(idx, target - cur, replace=False)
        h_bin[to_one] = 1.0
    return h_bin


def run(seed: int = 42, budget_s: int = 1100):
    """
    Search for a step function h on [0,2] (discretised) that minimises the
    Erdős overlap bound C5.  The function returns a triple
       (h_values, c5_bound, n_points)
    where `h_values` is a list of floats in [0,1] satisfying the constraints.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------ #
    # initialise h
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        # force a proper binary start – this is a very good seed for the search
        h = _initial_binary(h0)
    else:
        n = 200                      # even, modest size
        h = np.zeros(n, dtype=float)
        h[np.random.choice(n, n // 2, replace=False)] = 1.0

    total = n / 2.0                 # ∑h[i] (exact by construction)
    dx = 2.0 / n                    # grid spacing

    current_c5 = compute_c5(h, total, dx)
    best_h = h.copy()
    best_c5 = current_c5

    start_time = time.time()
    no_improve = 0
    max_no_improve = 6000          # restart after this many fruitless attempts
    step_max = 0.2                  # maximal amount moved in a fractional step
    eps = 1e-12

    while time.time() - start_time < budget_s * 0.95:
        # ------------------------------------------------------------------
        # find the shift that gives the worst (smallest) auto‑correlation
        corr = np.correlate(h, h, mode="full")
        idx_min = int(np.argmin(corr))
        shift = idx_min - (n - 1)         # shift in index units

        # ------------------------------------------------------------------
        # try a *fractional* improvement move that increases the correlation
        # for the identified worst shift
        h_shift = shifted_view(h, shift)

        donor_mask = (h > eps) & (h_shift < eps)        # h[i] >0 but partner ≈0
        receiver_mask = (h < 1 - eps) & (h_shift > eps)  # h[j] <1 and partner >0

        donors = np.where(donor_mask)[0]
        receivers = np.where(receiver_mask)[0]

        made_move = False
        if donors.size and receivers.size:
            i = int(np.random.choice(donors))
            j = int(np.random.choice(receivers))
            # amount that can be moved without breaking bounds
            delta = min(h[i], 1.0 - h[j], step_max)
            if delta > eps:
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                new_c5 = compute_c5(h_new, total, dx)
                if new_c5 < current_c5 - 1e-12:
                    h = h_new
                    current_c5 = new_c5
                    made_move = True
                    no_improve = 0
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h_new.copy()
        # ------------------------------------------------------------------
        # if the fractional move did not succeed, fall back to a binary swap
        if not made_move:
            # find mismatched positions for the worst shift (binary view)
            h_shift_bin = shifted_view(h, shift)
            ones_mismatch = np.where((h == 1) & (h_shift_bin == 0))[0]
            zeros_mismatch = np.where((h == 0) & (h_shift_bin == 1))[0]

            if ones_mismatch.size and zeros_mismatch.size:
                i = int(np.random.choice(ones_mismatch))
                j = int(np.random.choice(zeros_mismatch))
            else:
                # generic random swap between any 1 and any 0
                i = int(np.random.choice(np.where(h == 1)[0]))
                j = int(np.random.choice(np.where(h == 0)[0]))

            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]

            new_c5 = compute_c5(h_new, total, dx)
            if new_c5 < current_c5 - 1e-12:
                h = h_new
                current_c5 = new_c5
                no_improve = 0
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
            else:
                no_improve += 1

        # ------------------------------------------------------------------
        # occasional restart to avoid stagnation
        if no_improve >= max_no_improve:
            h = np.zeros(n, dtype=float)
            h[np.random.choice(n, n // 2, replace=False)] = 1.0
            current_c5 = compute_c5(h, total, dx)
            no_improve = 0

    # ---------------------------------------------------------------------- #
    final_c5 = compute_c5(best_h, total, dx)
    # sanity check – the linear constraint must hold to numerical precision
    assert abs(best_h.sum() * dx - 1.0) < 1e-9

    return (best_h.tolist(), float(final_c5), n)
