def simulated_annealing_binary(n, time_limit, h_init=None):
    dx = 2.0 / n

    # Initialize h
    if h_init is not None:
        h = np.array(h_init, dtype=float)
        # round to 0/1
        h = (h > 0.5).astype(float)
        # adjust number of ones
        ones = int(h.sum())
        target = n // 2
        if ones > target:
            # set some 1 ->0
            idx = np.where(h==1)[0]
            np.random.shuffle(idx)
            h[idx[:ones-target]] = 0
        elif ones < target:
            idx = np.where(h==0)[0]
            np.random.shuffle(idx)
            h[idx[:target-ones]] = 1
    else:
        # random binary with exactly n/2 ones
        h = np.zeros(n, dtype=float)
        h[:n//2] = 1
        np.random.shuffle(h)

    # precompute complement
    # compute current c5
    cur_c5 = compute_c5(h, n)  # compute via correlation times dx
    best_h = h.copy()
    best_c5 = cur_c5

    # get lists of indices of zeros and ones
    ones_idx = np.where(h==1)[0]
    zeros_idx = np.where(h==0)[0]

    start = time.time()
    # set temperature schedule
    T = 0.02  # maybe high enough
    # final temperature maybe near 0
    # define cooling_factor per iteration; or linear schedule based on elapsed time.

    it = 0
    # maximum iterations maybe large.
    while time.time() - start < time_limit:
        # propose swap
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        # create new h swap (we can just compute new h via copy)
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1

        new_c5 = compute_c5(h_new, n)

        delta = new_c5 - cur_c5

        if delta < 0 or np.random.rand() < math.exp(-delta / T):

            cur_c5 = new_c5

def continuous_refine(h, time_limit):
    n = len(h)
    dx = 2.0 / n
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    # Use simple random pair modifications
    step = 0.2
    while time.time() - start < time_limit:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = (np.random.rand() - 0.5) * 2 * step  # uniform in [-step, step]
        # ensure bounds
        hi = h[i] + delta
        hj = h[j] - delta
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h_new = h.copy()
        h_new[i] = hi
        h_new[j] = hj
        new_c5 = compute_c5(h_new)
        # Accept if better or with small probability
        if new_c5 < cur_c5 or np.random.rand() < 0.001:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        # Reduce step gradually
        elapsed = time.time() - start
        frac = elapsed / time_limit
        step = 0.2 * (1 - frac) + 0.001
    return best_h, best_c5
```

def simulated_annealing_binary(n, time_limit, init_h=None):
    import numpy as np, math, time
    dx = 2.0 / n
    # init binary
    if init_h is not None:
        h = (init_h > 0.5).astype(float)
        # adjust to have exactly n/2 ones
        count = int(h.sum())
        target = n // 2
        if count > target:
            # turn excess ones to zeros
            ones_idx = np.where(h == 1)[0]
            np.random.shuffle(ones_idx)
            h[ones_idx[:count-target]] = 0
        elif count < target:
            zeros_idx = np.where(h == 0)[0]
            np.random.shuffle(zeros_idx)
            h[zeros_idx[:target-count]] = 1
    else:
        h = np.zeros(n, dtype=float)
        h[:n//2] = 1
        np.random.shuffle(h)
    # create list indices
    ones_list = [int(i) for i in np.where(h == 1)[0]]
    zeros_list = [int(i) for i in np.where(h == 0)[0]]
    def compute_c5_local(arr):
        corr = np.correlate(arr, 1 - arr, mode='full')
        return np.max(corr) * dx
    cur_c5 = compute_c5_local(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    # start temperature
    T0 = 0.02
    while time.time() - start < time_limit:
        # propose swap
        i = np.random.choice(ones_list)
        j = np.random.choice(zeros_list)
        # swap
        h[i] = 0
        h[j] = 1
        # update index lists fast
        ones_list.remove(i)
        zeros_list.append(i)
        zeros_list.remove(j)
        ones_list.append(j)
        new_c5 = compute_c5_local(h)
        delta = new_c5 - cur_c5
        elapsed = time.time() - start
        # linear cooling
        T = T0 * (1 - elapsed / time_limit)
        if delta < 0 or np.random.rand() < math.exp(-delta / max(T, 1e-8)):
            cur_c5 = new_c5
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
            # revert lists
            zeros_list.remove(i)
            ones_list.append(i)
            ones_list.remove(j)
            zeros_list.append(j)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5
```

import time, math, numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def init_from_parent(parent, n):
    if parent is None:
        return None
    p = np.array(parent, dtype=float)
    src = np.linspace(0.0, 2.0, p.size, endpoint=False)
    dst = np.linspace(0.0, 2.0, n, endpoint=False)
    interp = np.interp(dst, src, p)
    bin_h = (interp > 0.5).astype(float)
    target = n // 2
    cur = int(bin_h.sum())
    if cur > target:
        idx = np.where(bin_h == 1)[0]
        np.random.shuffle(idx)
        bin_h[idx[:cur - target]] = 0.0
    elif cur < target:
        idx = np.where(bin_h == 0)[0]
        np.random.shuffle(idx)
        bin_h[idx[:target - cur]] = 1.0
    return bin_h

def simulated_annealing_binary(n, limit, init_h=None):
    dx = 2.0 / n
    if init_h is not None:
        h = init_h.copy()
        cnt = int(h.sum())
        target = n // 2
        if cnt > target:
            idx = np.where(h == 1)[0]
            np.random.shuffle(idx)
            h[idx[:cnt - target]] = 0.0
        elif cnt < target:
            idx = np.where(h == 0)[0]
            np.random.shuffle(idx)
            h[idx[:target - cnt]] = 1.0
    else:
        h = np.zeros(n, dtype=float)
        h[: n // 2] = 1.0
        np.random.shuffle(h)
    ones = [int(i) for i in np.where(h == 1)[0]]
    zeros = [int(i) for i in np.where(h == 0)[0]]
    best_h = h.copy()
    best_c5 = compute_c5(h)

    cur_c5 = best_c5
    start = time.time()
    T0 = 0.05
    while time.time() - start < limit:
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        ones.remove(i)
        zeros.append(i)
        zeros.remove(j)
        ones.append(j)

        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / limit)
        accept = delta < 0 or np.random.rand() < math.exp(-delta / max(T, 1e-12))
        if accept:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
            zeros.remove(i)
            ones.append(i)
            ones.remove(j)
            zeros.append(j)
        # occasional temperature boost to avoid stagnation
        if np.random.rand() < 0.001:
            T0 = max(T0 * 0.9, 0.005)
    return best_h, best_c5

def continuous_refine(h, limit):
    n = h.shape[0]
    dx = 2.0 / n
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    start = time.time()
    step = 0.2
    while time.time() - start < limit:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = (np.random.rand() - 0.5) * 2.0 * step
        hi, hj = cur[i] + delta, cur[j] - delta
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        cand = cur.copy()
        cand[i], cand[j] = hi, hj
        cand_c5 = compute_c5(cand)
        if cand_c5 < cur_c5 or np.random.rand() < 0.001:
            cur, cur_c5 = cand, cand_c5
            if cur_c5 < best_c5:
                best, best_c5 = cur.copy(), cur_c5
        elapsed = time.time() - start
        step = 0.2 * (1.0 - elapsed / limit) + 1e-4
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_total = time.time()
    total_budget = budget_s

    # choose working resolution
    n = 200
    parent_vec = None
    if 'initial_h_values' in globals():
        parent_vec = initial_h_values
    init_bin = init_from_parent(parent_vec, n)

    best_h = None
    best_c5 = float('inf')
    # allocate time for several SA runs
    while True:
        elapsed = time.time() - start_total
        remain = total_budget - elapsed
        if remain < 10.0:
            break
        run_time = min(25.0, remain * 0.75)
        h_candidate, c5_candidate = simulated_annealing_binary(n, run_time,
                                                              init_bin if best_h is None else None)
        if c5_candidate < best_c5:
            best_c5, best_h = c5_candidate, h_candidate.copy()
        # after first successful run discard init to force randomness
        init_bin = None

    # final gentle continuous refinement if time permits
    elapsed = time.time() - start_total
    remain = total_budget - elapsed
    if remain > 5.0 and best_h is not None:
        best_h, best_c5 = continuous_refine(best_h, remain)

    # final safety repair
    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = n / 2
    cur_sum = best_h.sum()
    diff = cur_sum - target_sum
    if abs(diff) > 1e-9:
        free = np.where((best_h > 0.0) & (best_h < 1.0))[0]
        if free.size:
            adj = diff / free.size
            best_h[free] -= adj
        else:
            best_h *= target_sum / cur_sum
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
