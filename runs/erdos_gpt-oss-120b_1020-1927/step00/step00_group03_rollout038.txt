def project_onto_box_simplex(x, target_sum, max_iter=10):
    # clip to [0,1]
    x = np.clip(x, 0, 1)
    for _ in range(max_iter):
        s = x.sum()
        diff = target_sum - s
        if abs(diff) < 1e-9:
            break
        free_mask = (x > 0) & (x < 1)
        free_indices = np.where(free_mask)[0]
        if free_indices.size == 0:
            # adjust all equally? just shift all due to rounding
            x += diff / len(x)
            x = np.clip(x, 0, 1)
            continue
        delta = diff / free_indices.size
        x[free_indices] += delta
        x = np.clip(x, 0, 1)
    # After loop, adjust final scaling for any leftover error
    s = x.sum()
    if abs(s - target_sum) > 1e-6:
        # distribute difference uniformly across non-bound entries
        for _ in range(10):
            free = (x > 0) & (x < 1)
            cnt = free.sum()
            if cnt == 0: break
            delta = (target_sum - x.sum())/cnt
            x[free] += delta
            x = np.clip(x, 0, 1)
    return x

def project_onto_box_simplex(v, target_sum, max_iter=100):
    # ensure box
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = v.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        # indices that can move (not at boundaries)
        free_mask = (v > 0) & (v < 1)
        free = np.where(free_mask)[0]
        if free.size == 0:
            # distribute diff equally among all (may violate bounds)
            v += diff / v.size
            v = np.clip(v, 0.0, 1.0)
            continue
        delta = diff / free.size
        v[free] += delta
        # clip again
        v = np.clip(v, 0.0, 1.0)
    # final adjust due to floating error:
    # use simple redistribution of residual to free indices
    cur_sum = v.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-9:
        # adjust uniformly in small increments
        # find indices not at bounds
        free_mask = (v > 0) & (v < 1)
        free = np.where(free_mask)[0]
        if free.size > 0:
            v[free] += diff / free.size
            v = np.clip(v, 0.0, 1.0)
    return v

import time
import numpy as np

def _compute_c5(h):
    """Return max_k ∫ h(x)(1‑h(x+k)) dx for the discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear correlation (zero padding outside [0,2])
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float((corr.max() * dx)), corr  # also return full correlation for reuse


def _worst_shift_and_mismatches(h, corr):
    """Given h and the full correlation, return the worst shift,
    the indices where h_i=1 & h_{i+s}=0 (A) and where h_i=0 & h_{i+s}=1 (B)."""
    n = h.shape[0]
    idx = int(np.argmax(corr))               # index in full correlation array
    s = idx - (n - 1)                        # actual integer shift

    if s >= 0:
        left = np.arange(n - s)
        right = left + s
    else:
        s_abs = -s
        left = np.arange(s_abs, n)
        right = left - s_abs

    # mismatched positions for this shift
    left_vals = h[left]
    right_vals = h[right]

    A = left[(left_vals == 1) & (right_vals == 0)]  # 1 → 0
    B = left[(left_vals == 0) & (right_vals == 1)]  # 0 → 1
    return s, A, B


def _random_binary_vector(n, rng):
    """Generate a binary vector of length n with exactly n/2 ones."""
    ones = n // 2
    idx = rng.permutation(n)
    v = np.zeros(n, dtype=float)
    v[idx[:ones]] = 1.0
    return v


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Choose a modest even length; use the length of the provided
    # initial_h_values when possible, otherwise fall back to 70.
    N = 70
    if "initial_h_values" in globals():
        cand = np.asarray(initial_h_values)
        Ncand = cand.shape[0]
        if Ncand % 2 == 0 and Ncand > 0:

    # Ensure N is even (needed for exact half‑mass constraint)
    if N % 2:
        N += 1

    best_h = None
    best_val = float("inf")
    target_val = 0.3808                     # we try to beat this

    # Main optimisation loop – hill‑climbing with targeted swaps.
    while time.time() - start_time < budget_s * 0.95:
        # Start from a fresh random binary vector.
        h = _random_binary_vector(N, rng)
        cur_val, corr = _compute_c5(h)

        # Local search for this start point.
        no_progress = 0
        while True:

            if cur_val < best_val:

                best_h = h.copy()
                if best_val <= target_val:

            s, A, B = _worst_shift_and_mismatches(h, corr)

                j = rng.choice(B)

                new_val, new_corr = _compute_c5(h)
                if new_val <= cur_val:

            else:
