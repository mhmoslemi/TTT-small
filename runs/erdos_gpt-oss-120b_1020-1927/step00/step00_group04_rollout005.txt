def compute_c5(h):
    # using FFT for speed
    n = len(h)
    # compute correlation of h with (1 - h), linear convolution
    # Equivalent to full correlation: conv of h and rev(1-h)
    # Use FFT with zero padding length L = next_pow2(2*n)
    L = 1 << ((2*n-1).bit_length())
    fft_h = np.fft.rfft(h, n=L)

def project_to_sum(x, target):
    # project onto [0,1] box and sum constraint (Euclidean projection)
    # Using water-filling algorithm: find lambda such that sum(clamp(x+lambda,0,1)) = target.
    lo = -1.0
    hi = 1.0
    # Expand bounds
    while np.clip(x + lo, 0.0, 1.0).sum() > target:
        lo *= 2
    while np.clip(x + hi, 0.0, 1.0).sum() < target:
        hi *= 2
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

import time
import numpy as np

def _project_to_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of x onto the box [0,1] with sum(x)=target."""
    lo, hi = -2.0, 2.0
    # expand interval until the desired sum lies between the two ends
    while np.clip(x + lo, 0.0, 1.0).sum() > target:
        lo *= 2.0
    while np.clip(x + hi, 0.0, 1.0).sum() < target:
        hi *= 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def _eval_c5(h: np.ndarray) -> float:
    """Return the C5 value for a given step function h (dx = 2/len(h))."""
    n = h.size
    raw = np.correlate(h, 1.0 - h, mode="full")
    return raw.max() * (2.0 / n)


def _gradient_worst_shift(h: np.ndarray, lag: int) -> np.ndarray:
    """Gradient of the worst‑shift objective w.r.t. h."""
    n = h.size
    if lag >= 0:
        h_minus = np.concatenate((np.zeros(lag), h[: n - lag]))
        h_plus = np.concatenate((h[lag:], np.zeros(lag)))
    else:
        L = -lag
        h_minus = np.concatenate((h[L:], np.zeros(L)))          # h_{i+L}
        h_plus = np.concatenate((np.zeros(L), h[: n - L]))      # h_{i-L}
    return 1.0 - (h_minus + h_plus)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step‑function h, its C5 bound and the number of samples."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # -------------------------------------------------
    # Choose resolution and produce an initial candidate
    # -------------------------------------------------
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
        n0 = base.size
        n = max(120, n0 * 3)
        if n % 2:
            n += 1
        # linear interpolation to the larger grid
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x, x0, base)
    else:
        n = 200
        h0 = np.full(n, 0.5)

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _project_to_sum(h0, n / 2.0)

    # -------------------------------------------------
    # Assemble a small set of diverse starting points
    # -------------------------------------------------
    starts = [h0]
    for _ in range(4):
        sigma = rng.uniform(0.2, 0.7)
        xs = np.linspace(0.0, 2.0, n, endpoint=False)
        g = np.exp(-0.5 * ((xs - 1.0) / sigma) ** 2)
        g /= g.max()                     # scale to max 1
        g = np.clip(g, 0.0, 1.0)
        g = _project_to_sum(g, n / 2.0)
        starts.append(g)

    best_h = None
    best_c5 = np.inf

    # -------------------------------------------------
    # Main optimisation loop (gradient descent on worst shift)
    # -------------------------------------------------
    for h_start in starts:
        if time.time() - start > budget_s * 0.94:
            break
        h = h_start.copy()
        cur_c5 = _eval_c5(h)

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        step = 0.2
        it = 0
        while time.time() - start < budget_s * 0.94:
            # worst shift
            raw = np.correlate(h, 1.0 - h, mode="full")
            max_idx = int(np.argmax(raw))
            lag = max_idx - (n - 1)

            # gradient for that shift
            grad = _gradient_worst_shift(h, lag)

            # tentative update
            h_new = h - step * grad
            np.clip(h_new, 0.0, 1.0, out=h_new)
            h_new = _project_to_sum(h_new, n / 2.0)

            new_c5 = _eval_c5(h_new)

            if new_c5 < cur_c5 - 1e-12:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                step = min(step * 1.08, 1.0)
            else:
                step *= 0.5
                if step < 1e-6:
                    break

            it += 1
            if it % 500 == 0 and rng.random() < 0.15:
                # occasional random jitter to escape flat regions
                jitter = rng.uniform(-step, step, size=n)
                h_tmp = h + jitter
                np.clip(h_tmp, 0.0, 1.0, out=h_tmp)
                h_tmp = _project_to_sum(h_tmp, n / 2.0)
                c_tmp = _eval_c5(h_tmp)
                if c_tmp < cur_c5:
                    h = h_tmp
                    cur_c5 = c_tmp
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()

        # fine‑tuning with a smaller fixed step
        step_small = 0.03
        for _ in range(800):
            if time.time() - start > budget_s * 0.96:
                break
            raw = np.correlate(h, 1.0 - h, mode="full")
            lag = int(np.argmax(raw)) - (n - 1)
            grad = _gradient_worst_shift(h, lag)
            h_new = h - step_small * grad
            np.clip(h_new, 0.0, 1.0, out=h_new)
            h_new = _project_to_sum(h_new, n / 2.0)
            new_c5 = _eval_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                step_small *= 0.9
                if step_small < 1e-7:
                    break

    # -------------------------------------------------
    # Final sanity projection and return
    # -------------------------------------------------
    if best_h is None:
        best_h = h0
        best_c5 = _eval_c5(best_h)

    best_h = _project_to_sum(best_h, n / 2.0)
    best_c5 = _eval_c5(best_h)

    return (best_h.tolist(), float(best_c5), int(n))
