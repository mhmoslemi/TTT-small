import time
import numpy as np
import random

def _cost(h, dx):
    """Return the C5 value for a given step function `h`."""
    # np.correlate computes the discrete cross‑correlation (no reversal)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * dx)


def _random_feasible(n, target, rng):
    """Generate a random feasible vector (values in [0,1], sum = target)."""
    # start from uniform 0.5 and then perform random swaps
    h = np.full(n, 0.5, dtype=float)
    cur = target - h.sum()
    # distribute the residual uniformly without breaking bounds
    if abs(cur) > 1e-12:
        # give the whole residual to the first non‑boundary entry
        for i in range(n):
            lo, hi = 0.0, 1.0
            if cur > 0 and h[i] < hi:
                add = min(cur, hi - h[i])
                h[i] += add
                cur -= add
            elif cur < 0 and h[i] > lo:
                sub = min(-cur, h[i] - lo)
                h[i] -= sub
                cur += sub
            if abs(cur) < 1e-12:
                break
    # a few random pairwise swaps to break symmetry
    for _ in range(3 * n):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        lo = max(-h[i], -(1.0 - h[j]))
        hi = min(1.0 - h[i], h[j])
        if lo >= hi:
            continue
        delta = rng.random() * (hi - lo) + lo
        h[i] += delta
        h[j] -= delta
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a step function h (as a list), its C5 value, and n_points.
    The algorithm is a simple simulated‑annealing search with pairwise
    adjustments that preserve the sum constraint.
    """
    rng = np.random.default_rng(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # initialise --------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = h0.size
        target = n / 2.0
        # enforce box constraints and exact sum (tiny numerical adjustments)
        h0 = np.clip(h0, 0.0, 1.0)
        diff = h0.sum() - target
        if abs(diff) > 1e-12:
            # move the surplus/deficit to the first non‑boundary entry
            for i in range(n):
                if diff > 0 and h0[i] > 0.0:
                    take = min(diff, h0[i])
                    h0[i] -= take
                    diff -= take
                elif diff < 0 and h0[i] < 1.0:
                    give = min(-diff, 1.0 - h0[i])
                    h0[i] += give
                    diff += give
                if abs(diff) < 1e-12:
                    break
        cur_h = h0
    else:
        n = 95
        target = n / 2.0
        cur_h = np.full(n, 0.5, dtype=float)

    dx = 2.0 / n

    cur_c = _cost(cur_h, dx)
    best_h = cur_h.copy()
    best_c = cur_c

    # ------------------------------------------------------------------
    # annealing parameters -----------------------------------------------
    start = time.time()
    T0, T_end = 0.7, 1e-6          # temperature schedule
    step0, step_end = 0.35, 1e-5   # magnitude of a single pair swap

    # counters for stagnation detection
    last_improve = 0
    stall_limit = int(5e4)        # after this many iterations without improvement
    it = 0

    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.95:
            break

        # linear schedules
        frac = elapsed / budget_s
        T = T0 * (1.0 - frac) + T_end * frac
        step = step0 * (1.0 - frac) + step_end * frac

        # --------------------------------------------------------------
        # pick a move --------------------------------------------------
        if rng.random() < 0.7:
            # targeted move: use the shift that currently attains the max
            corr_full = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            lag = int(np.argmax(corr_full))
            shift = lag - (n - 1)          # integer shift (positive → right)
            # choose a random overlapping index
            if shift >= 0:
                idx = rng.integers(0, n - shift)
                i, j = idx, idx + shift
            else:
                idx = rng.integers(-shift, n)

            hi, hj = cur_h[i], cur_h[j]
            if hi > hj:

                    continue
                delta = rng.random() * max_delta
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
            else:
                # fallback to a generic random pair swap
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                lo = max(-cur_h[i], -(1.0 - cur_h[j]))
                hi = min(1.0 - cur_h[i], cur_h[j])
                if lo >= hi:
                    continue
                delta = rng.random() * (hi - lo) + lo
                delta = np.clip(delta, -step, step)
                new_h = cur_h.copy()
                new_h[i] += delta
                new_h[j] -= delta
        else:
            # plain random pair swap
            i, j = rng.integers(0, n, size=2)

            hi = min(1.0 - cur_h[i], cur_h[j])

            for _ in range(5):
                i, j = rng.integers(0, n, size=2)

                hi = min(1.0 - cur_h[i], cur_h[j])
                if lo >= hi:
                    continue
                delta = rng.random() * (hi - lo) + lo
                cur_h[i] += delta
                cur_h[j] -= delta
            cur_c = _cost(cur_h, dx)
            last_improve = it

        it += 1

    # ------------------------------------------------------------------
    # final feasibility tweak (tiny rounding errors) --------------------
    final_sum = best_h.sum()
    diff = final_sum - target
    if abs(diff) > 1e-12:
        # adjust a single non‑boundary entry
        for idx in range(n):

                diff = best_h.sum() - target
                if abs(diff) < 1e-12:
                    break

    best_c = _cost(best_h, dx)

    # return as plain Python list to obey the required signature
    return (best_h.tolist(), float(best_c), n)
