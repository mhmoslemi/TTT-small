def project_onto_constraints(v):
    # ensure 0 <= v <=1 and sum(v) = target.
    # Use simple iterative: Equivalent to solving argmin_{w} ||w - v||^2 subject to 0<=w<=1, sum(w)=target.
    # This can be solved by Lagrange multiplier lambda and clipping: w_i = clip(v_i - lambda, 0, 1). Choose lambda s.t. sum_i w_i = target.
    # Use binary search on lambda as sum_i clip(v_i - lambda, 0, 1) monotonic decreasing in lambda.
    target = n/2
    lo, hi = -1, 2   # broad enough range for lambda
    for _ in range(50):
        lam = (lo + hi)/2
        w = np.clip(v - lam, 0, 1)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo+hi)/2, 0, 1)
    # final adjust small discrepancy: we may need to distribute remainder by adding to some components not at bounds.
    diff = target - w.sum()
    if abs(diff) > 1e-9:
        # adjust uniformly on free components (not at bounds)
        free = (w > 1e-12) & (w < 1-1e-12)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0, 1)
        else:
            # fallback: just add diff to first component within bounds
            w[0] = np.clip(w[0] + diff, 0, 1)
    return w

import time
import numpy as np

def compute_c5(h):
    """Maximum overlap value for a given step function."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def gradient_shift(h, shift):
    """Gradient of the correlation for a single shift (full‑range indexing)."""
    n = h.size
    comp = 1.0 - h
    g = np.zeros_like(h)
    if shift >= 0:
        L = n - shift
        # derivative w.r.t. the first argument of the product
        g[:L] += comp[shift:shift + L]          # + (1‑h_{j+shift})
        # derivative w.r.t. the second argument
        g[shift:shift + L] += -h[:L]            # - h_{j}
    else:                                   # shift < 0
        s = -shift
        L = n - s
        g[s:] += comp[:L]                      # + (1‑h_{j-s})
        g[:L] += -h[s:]                       # - h_{j+s}
    return g

def project_box_sum(v, target):
    """Euclidean projection of v onto [0,1]^n ∩ {sum = target}."""
    lo, hi = -2.0, 2.0                     # safe search interval for λ
    for _ in range(50):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny residual correction
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 1e-12) & (w < 1.0 - 1e-12)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            # fallback: adjust the first non‑saturated entry
            for i in range(w.size):
                if 0.0 <= w[i] + diff <= 1.0:
                    w[i] += diff
                    break
    return w

def optimize(h0, time_limit):
    """Simple sub‑gradient descent with occasional random rescue."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(h)
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        cur_val = corr[idx] * (2.0 / n)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        shift = idx - (n - 1)
        g = gradient_shift(h, shift)

        # basic diminishing step size
        step = 0.6 / np.sqrt(it)
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.1):
            h_candidate = project_box_sum(h - step * factor * g, target)
            new_val = compute_c5(h_candidate)
            if new_val < cur_val - 1e-9:
                h = h_candidate
                improved = True
                break
        if not improved:
            # random pair‑swap perturbation (helps escape plateaus)
            if np.random.rand() < 0.07:
                i, j = np.random.choice(n, 2, replace=False)
                delta = np.random.uniform(-0.07, 0.07)

                    h_tmp[i], h_tmp[j] = vi, vj
                    # sum unchanged by construction
                    if compute_c5(h_tmp) < cur_val:
                        h = h_tmp
    return best_h, best_val
