        alpha = (math.sqrt(5) - 1) / 2.0           # golden–ratio conjugate
        idx = np.arange(n)
        frac = (idx * alpha) % 1.0
        a = (frac < 0.5).astype(float)
        return fix_balance(a)

    def threshold_initial():
        if "initial_h_values" in globals():
            a = np.array(initial_h_values, dtype=float)
            a = (a >= 0.5).astype(float)
            return fix_balance(a)
        return random_binary()

    init_fns = [random_binary, sturmian, threshold_initial]

    # ------------------------------------------------------------
    # simulated annealing core
    # ------------------------------------------------------------
    best_h = None
    best_val = float("inf")
    start = time.time()
    time_limit = budget_s * 0.95   # keep a safety margin

    for make_start in init_fns:
        h_cur = make_start()
        cur_val = max_overlap(h_cur)

        # keep local best as well
        local_best_h = h_cur.copy()
        local_best_val = cur_val

        # annealing schedule parameters
        max_iters = 200_000

        for it in range(max_iters):
            if time.time() - start > time_limit:
                break

            # temperature (geometric cooling to near zero)
            frac = it / max_iters
            T = 1.0 * (1e-5) ** frac

            # pick a random 1↔0 swap
            ones = np.flatnonzero(h_cur > 0.5)
            zeros = np.flatnonzero(h_cur <= 0.5)
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)

            # perform swap
            h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
            new_val = max_overlap(h_cur)
            delta = new_val - cur_val

            # Metropolis acceptance
            if delta <= 0 or (T > 0 and np.random.rand() < math.exp(-delta / T)):
                cur_val = new_val
                if new_val < local_best_val:
                    local_best_val = new_val
                    local_best_h = h_cur.copy()
                if new_val < best_val:
                    best_val = new_val
                    best_h = h_cur.copy()
            else:
                # revert the swap
                h_cur[i], h_cur[j] = h_cur[j], h_cur[i]

            # occasional cheap restart to avoid dead‑ends
            if it % 5000 == 0 and random.random() < 0.02:
                h_cur = make_start()
                cur_val = max_overlap(h_cur)

        # keep the best discovered in this run
        if local_best_val < best_val:
            best_val = local_best_val
            best_h = local_best_h.copy()

        if time.time() - start > time_limit:
            break

    # fallback – should never happen but keep safety net
    if best_h is None:
        best_h = random_binary()
        best_val = max_overlap(best_h)

    # final verification (re‑compute to avoid round‑off)
    final_c5 = max_overlap(best_h)

    return (best_h, final_c5, n)
