    while sum_clip(lo) < target_sum:
        lo -= 1.0
    while sum_clip(hi) > target_sum:

def optimize_local(h0, rng, max_time=30.0, total_start_time=None, global_budget=None):

    n = len(h0)
    S = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    iter = 0
    start = time.time()
    # parameters
    alpha0 = 0.5
    decay = 0.999  # or 0.99
    # We'll also run random perturbation after some passes.
    while True:
        elapsed = time.time() - start
        # check time
        if total_start_time is not None:
            total_elapsed = time.time() - total_start_time
            if global_budget is not None and total_elapsed > global_budget - 5.0:
                break
        if max_time is not None and elapsed > max_time:
            break
        # compute c5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # compute correlation and argmax shift(s)
        corr = np.correlate(h, 1-h, mode='full')
        max_val = np.max(corr)
        max_indices = np.where(corr == max_val)[0]
        # pick a random shift among them
        idx = rng.choice(max_indices)
        s = idx - (n - 1)  # shift value
        # compute gradient for shift s
        grad = gradient_for_shift(h, s)
        # perform step with decreasing alpha
        alpha = alpha0 * (decay ** iter)
        # linear step
        h_new = h - alpha * grad
        # project
        h_new = project_onto_constraints(h_new, S)
        # compute new C5
        new_c5 = compute_c5(h_new)
        # accept if improvement; else maybe randomly accept with small prob
        if new_c5 <= c5:
            h = h_new
        else:
            # occasional random accept to escape local minima: probability decays
            prob = np.exp(-(new_c5 - c5) / (alpha + 1e-12))
            if rng.random() < prob:
                h = h_new
        # random perturb occasionally to avoid stagnation
        if iter % 200 == 0 and iter > 0:
            # random perturb: select k indices and add small random
            k = max(1, n // 20)  # perturb about 5% of entries
            pert = rng.uniform(-0.1, 0.1, size=k)
            idxs = rng.choice(n, size=k, replace=False)
            h_temp = h.copy()
            h_temp[idxs] += pert
            # project
            h_temp = project_onto_constraints(h_temp, S)
            # ensure within bounds
            h_temp = np.clip(h_temp, 0, 1)
            if compute_c5(h_temp) < c5:
                h = h_temp
        iter += 1
    return best_h, best_c5

def optimize_local(h0, rng, max_time=30.0, total_start_time=None, global_budget=None):
    import time, numpy as np
    n = int(h0.shape[0])
    S = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    start = time.time()
    iter = 0
    alpha0 = 0.5
    decay = 0.999
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed > max_time:
            break
        if total_start_time is not None and global_budget is not None:
            if now - total_start_time > global_budget - 1.0:
                break
        # compute current objective and worst shift(s)
        corr = np.correlate(h, 1 - h, mode='full')
        maxc = np.max(corr)
        # evaluate current C5
        cur_val = maxc * (2.0 / n)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        # identify indices achieving max
        idxs = np.where(corr == maxc)[0]
        idx = rng.choice(idxs)
        s = idx - (n - 1)
        grad = gradient_for_shift(h, s)
        # step size
        alpha = alpha0 * (decay ** iter)
        # propose new
        h_new = h - alpha * grad
        # project
        h_new = project_onto_constraints(h_new, S)
        # evaluate new
        new_corr = np.correlate(h_new, 1 - h_new, mode='full')
        new_val = np.max(new_corr) * (2.0 / n)
        # accept if improvement
        if new_val <= cur_val:
            h = h_new
        else:
            # occasional acceptance to escape local minima
            prob = math.exp(-(new_val - cur_val) / (alpha + 1e-12))
            if rng.random() < prob:
                h = h_new
        # occasional random perturb to diversify
        if iter % 200 == 0 and iter > 0:
            # generate new candidate by random start and take few gradient steps?
            h_pert = random_start(n, rng)
            # optionally do few gradient steps from h_pert for a short burst
            # But we can just replace h = h_pert if it improves
            val_pert = compute_c5(h_pert)
            if val_pert < cur_val:
                h = h_pert
        iter += 1
    return best_h, best_val

def project_onto_constraints(y, target):
    # binary search for lambda such that sum clip(y - lam,0,1) = target
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    # ensure bounds give sums on opposite sides
    def sum_clip(lam):
        return np.clip(y - lam, 0.0, 1.0).sum()
    # expand lo if sum at lo < target
    while sum_clip(lo) < target:
        lo -= 1.0
    while sum_clip(hi) > target:
        hi += 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = sum_clip(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y - lam, 0.0, 1.0)
    # small adjustment to match target exactly (optional)
    diff = x.sum() - target
    if abs(diff) > 1e-8:
        # distribute diff only on free (not at 0 or 1)
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            adjust = diff / free.sum()
            x[free] -= adjust
            # clip again to maintain bounds
            x = np.clip(x, 0.0, 1.0)
    return x

import numpy as np
import time
import math

def _compute_c5(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1-h(x+k)) dx for the discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_onto_constraints(y: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project y onto the set {x | 0≤x_i≤1, sum x_i = target_sum}.
    Uses a simple bisection on the Lagrange multiplier λ such that
    x_i = clip(y_i - λ, 0, 1) and sum x_i = target_sum.
    """
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0

    def _s(lam):
        return np.clip(y - lam, 0.0, 1.0).sum()

    # Ensure the interval bounds straddle the target sum
    while _s(lo) < target_sum:
        lo -= 1.0
    while _s(hi) > target_sum:
        hi += 1.0

    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = _s(mid)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    x = np.clip(y - lam, 0.0, 1.0)

    # Small correction to hit the target exactly (helps the equality constraint)
    diff = x.sum() - target_sum
    if abs(diff) > 1e-9:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            adj = diff / free.sum()
            x[free] -= adj
            x = np.clip(x, 0.0, 1.0)
    return x


def _random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible h on n points."""
    y = rng.random(n)  # uniform in [0,1)
    return _project_onto_constraints(y, n / 2.0)


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Subgradient of the correlation for a given shift (maximising shift)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    for i in range(n):
        j = i + shift
        if 0 <= j < n:
            g[i] += (1.0 - h[j])
        k = i - shift
        if 0 <= k < n:
            g[i] -= h[k]
    return g


def _optimize_local(h0: np.ndarray,
                    rng: np.random.Generator,
                    max_time: float,
                    total_start: float,
                    global_budget: float) -> tuple[np.ndarray, float]:
    """
    Simple projected gradient descent aimed at reducing the worst‑shift overlap.
    Stops when max_time (seconds) is exhausted or the global budget is about to run out.
    """
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = _compute_c5(h)

    start = time.time()
    it = 0
    alpha0 = 0.5
    decay = 0.999

    while True:
        now = time.time()
        if now - start > max_time:
            break
        if now - total_start > global_budget - 2.0:  # keep a tiny safety margin
            break

        # current correlation and max shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        cur_val = max_corr * (2.0 / n)

        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        # pick a shift attaining the maximum (randomly if several)
        max_idxs = np.where(corr == max_corr)[0]
        idx = rng.choice(max_idxs)
        shift = idx - (n - 1)

        # gradient for that shift
        g = _gradient_for_shift(h, shift)

        # step size schedule
        alpha = alpha0 * (decay ** it)

        # tentative move then projection
        h_new = _project_onto_constraints(h - alpha * g, target)

        new_corr = np.correlate(h_new, 1.0 - h_new, mode="full")
        new_val = np.max(new_corr) * (2.0 / n)

        # accept if it improves; otherwise use a small Metropolis probability
        if new_val <= cur_val:
            h = h_new
        else:
            prob = math.exp(-(new_val - cur_val) / (alpha + 1e-12))
            if rng.random() < prob:
                h = h_new

        # occasional random perturbation to escape plateaus
        if it % 250 == 0 and it > 0:
            cand = _random_start(n, rng)
            cand_val = _compute_c5(cand)
            if cand_val < cur_val:
                h = cand

        it += 1

    return best_h, best_val


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """

    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine the list of grid sizes to explore
    n_candidates = []
    if 'initial_h_values' in globals():
        n0 = int(initial_h_values.shape[0])
        n_candidates.append(n0)
        # try a couple of multiples (up to ≈200 points, still cheap)

            n_candidates.append(n0 * 2)
        if n0 * 4 <= 200:
            n_candidates.append(n0 * 4)
    else:
        n_candidates = [47, 94, 188]

    best_h = None
    best_val = np.inf
    best_n = None

    remaining = len(n_candidates)
    for n in n_candidates:
        # time budgeting per grid size
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed
        if time_left < 5.0:
            break
        per_grid_budget = max(5.0, time_left / remaining)

        # construct an initial feasible vector for this grid size
        if 'initial_h_values' in globals() and n == len(initial_h_values):
            h0 = initial_h_values.copy()
            h0 = _project_onto_constraints(h0, n / 2.0)
        elif 'initial_h_values' in globals() and n % len(initial_h_values) == 0:
            factor = n // len(initial_h_values)
            h0 = np.repeat(initial_h_values, factor)
            # repetition already respects the integral constraint
        else:
            h0 = _random_start(n, rng)

        # local optimisation
        h_opt, val_opt = _optimize_local(
            h0, rng,
            max_time=per_grid_budget,

            global_budget=budget_s

        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()

        remaining -= 1
        if time.time() - start_time > budget_s - 1.0:

    # fallback if nothing succeeded
    if best_h is None:
        n = 50
        best_h = np.full(n, 0.5)
        best_val = _compute_c5(best_h)
        best_n = n

    return best_h, float(best_val), int(best_n)
