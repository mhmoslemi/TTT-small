def subgradient(h):
    n = len(h)
    conv = np.correlate(h, 1 - h, mode='full')
    max_val = conv.max()
    # tolerance relative
    eps = 1e-12
    active = np.where(np.abs(conv - max_val) < 1e-9)[0]  # indices
    grads = np.zeros(n)
    for idx in active:
        s = idx - (n - 1)  # shift
        # compute for each i
        # term1: (1 - h_{i+s}) for i in range (0..n-1), where i+s in range
        # term2: h_{i - s} for i in range (0..n-1), where i-s in range
        # So for each i:
        # add (1 - h[i+s]) if i+s < n else 0
        # subtract h[i - s] if i - s >= 0 else 0
        # Use vectorized approach.
        # We can compute shift contributions via slicing.
        # For positive shift s>=0:
        #   for i in 0..n-1-s: term1 add (1 - h[i+s])
        #   for i in s..n-1: term2 subtract h[i-s]
        # For negative shift s<0: symmetrical.
        # We'll implement using loops? Use vectorization.
        if s >= 0:
            # term1 for i=0..n-1-s
            grads[:n-s] += (1 - h[s:])
            # term2 for i=s..n-1
            grads[s:] -= h[:n-s]
        else:
            # s negative => shift to left

            # term1 for i=s_abs..n-1 (i+s >=0)

            grads[:n-s_abs] -= h[s_abs:]
    # Average if multiple active shifts
    grads /= len(active)
    return grads

import time, random, math
import numpy as np

def _evaluate_vec(h):
    """C5 value for a vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx

def _subgrad_and_max(h):
    """Return subgradient (average over all active shifts) and raw max correlation."""
    n = h.shape[0]
    conv = np.correlate(h, 1 - h, mode='full')
    mx = conv.max()
    # active indices (possible ties)
    act = np.where(np.abs(conv - mx) < 1e-12)[0]
    grads = np.zeros(n, dtype=np.float64)
    for idx in act:
        s = idx - (n - 1)          # shift corresponding to this index
        if s >= 0:
            grads[:n - s] += (1 - h[s:])   # term (1 - h_{i+s})
            grads[s:]   -= h[:n - s]       # term -h_{i-s}
        else:
            sa = -s
            grads[sa:]   += (1 - h[:n - sa])
            grads[:n - sa] -= h[sa:]
    if act.size:
        grads /= act.size
    return grads, mx

def _repair(h):
    """Clip to [0,1] and enforce the integral exactly."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # try to distribute on non‑bound entries
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            per = diff / mask.sum()
            h[mask] += per
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _upscale(h, n_new):
    """Linearly interpolate h to length n_new."""
    n_old = h.shape[0]
    if n_new == n_old:
        return h.copy()
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    x_old = (np.arange(n_old) + 0.5) * dx_old
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return _repair(h_new)

    """Simple subgradient descent with pairwise transfers."""
    h = _repair(h0)
