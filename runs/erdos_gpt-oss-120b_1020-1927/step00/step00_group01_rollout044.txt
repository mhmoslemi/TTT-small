    import numpy as np, time, random
    random.seed(seed)
    np.random.seed(seed)
    # Gather initial
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        n_default = 100
        h_init = np.concatenate([np.ones(n_default//2), np.zeros(n_default - n_default//2)])
        np.random.shuffle(h_init)
    # ensure sum
    n = len(h_init)
    target_sum = n/2.0
    cur = h_init.astype(float)
    # adjust sum if needed
    cur_sum = cur.sum()
    if abs(cur_sum - target_sum) > 1e-12:

                cur[i] += diff
                break

            # clip and adjust residual
            cur = np.clip(cur,0,1)
            cur_sum = cur.sum()

                    break

        return np.max(np.correlate(arr, 1 - arr, mode='full')) * dx
    best_h = cur.copy()
    best_score = score(best_h)
    cur_h = cur.copy()
    cur_score = best_score

import numpy as np
import time
import random

def compute_c5(arr: np.ndarray) -> float:
    """Return the C5 value for a given step function sampled array."""
    n = arr.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives all integer shifts
    return np.max(np.correlate(arr, 1.0 - arr, mode="full")) * dx


def _project_to_target_sum(arr: np.ndarray, target: float) -> np.ndarray:
    """
    Adjust ``arr`` (all entries in [0,1]) so that its sum equals ``target``.
    The adjustment is performed by a simple redistribution that respects bounds.
    """
    cur = arr.copy()
    cur_sum = cur.sum()
    diff = target - cur_sum
    if abs(diff) < 1e-12:
        return cur
    # try to fix by adding the whole diff to a single feasible entry
    for i in range(cur.shape[0]):
        if 0.0 <= cur[i] + diff <= 1.0:
            cur[i] += diff
            return cur
    # scale uniformly then clip and fix any remaining discrepancy
    if cur_sum != 0.0:
        cur *= target / cur_sum
    cur = np.clip(cur, 0.0, 1.0)
    cur_sum = cur.sum()
    residual = target - cur_sum
    for i in range(cur.shape[0]):
        if 0.0 <= cur[i] + residual <= 1.0:
            cur[i] += residual
            break
    return cur

    Optimize the Erdős minimum overlap constant C5.
    Returns (h_values, c5_bound, n_points).

        np.random.shuffle(h)

    n = h.shape[0]
    target_sum = n / 2.0
    h = _project_to_target_sum(h, target_sum)

    # ------------------------------------------------------------------
    # simulated‑annealing style local search
    # ------------------------------------------------------------------
    cur_h = h.copy()
    best_h = cur_h.copy()
    cur_score = compute_c5(cur_h)
    best_score = cur_score

    start_time = time.time()
    # temperature schedule parameters
    T_start = 0.02
    T_end = 1e-6
    # main loop – stop a little before the budget expires
    time_limit = budget_s * 0.9
    while True:

        if elapsed >= time_limit:

        T = (1.0 - frac) * T_start + frac * T_end

        # propose a transfer of mass from index j to i
