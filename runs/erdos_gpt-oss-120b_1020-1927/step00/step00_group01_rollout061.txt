    # find λ such that sum(clip(v - λ, lower, upper)) = target_sum
    # Use bisection
    lo = -1e6
    hi = 1e6
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    # after loop, x approximates solution
    x = np.clip(v - (lo+hi)/2.0, lower, upper)

    diff = target_sum - x.sum()

def compute_grad_at_max_shift(h):
    n = len(h)
    corr = compute_correlation_vals(h)
    max_val = corr.max()
    # find lags achieving max: use first
    idx = np.flatnonzero(corr == max_val)[0]  # first occurrence
    s = idx - (n-1)
    grad = np.ones_like(h)
    if s >= 0:
        grad[:n-s] -= h[s:]
        grad[s:]   -= h[:n-s]
    else:
        s_abs = -s
        grad[s_abs:] -= h[:n - s_abs]
        grad[:n - s_abs] -= h[s_abs:]
    return grad

def project_onto_bounds_and_sum(v, S, lower=0.0, upper=1.0):
    # solve min ||x-v||^2 s.t. lower <= x <= upper, sum x = S
    # Use water-filling via binary search on lambda
    n = len(v)
    # initial lambda bounds
    lo = -1e6
    hi = 1e6
    # ensure feasible range: compute sum clipped at extreme
    # Actually we can compute min possible sum: sum(l) = n*lower
    # max possible sum: sum(u) = n*upper
    # Guarantee S within that range (it is because S = n/2 and lower=0 upper=1)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo+hi)/2.0, lower, upper)
    # to ensure exact sum, adjust small residual
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        # distribute diff to free variables (not at bound)
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
        else:
            # all at bounds, adjust by adding diff to first element
            x[0] += diff
        # clip again
        x = np.clip(x, lower, upper)
    return x

def gradient_at_max(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()
    idx = np.flatnonzero(corr == max_val)[0]  # first
    s = idx - (n - 1)
    grad = np.ones_like(h)
    if s >= 0:
        # i where i+s < n -> i in [0, n-s-1]
        grad[:n-s] -= h[s:]
        # i where i-s >=0 -> i in [s, n-1]
        grad[s:] -= h[:n-s]
    else:
        s_abs = -s
        # i+s >=0 -> i in [s_abs, n-1] (i - s_abs >=0)
        grad[s_abs:] -= h[:n - s_abs]
        # i-s >=0 -> i in [0, n - s_abs -1] (i + s_abs < n)
        grad[:n - s_abs] -= h[s_abs:]
    return grad

def project_box_sum(v, target, lower=0.0, upper=1.0):
    # shift v to satisfy sum via water-filling.
    # start with v clipped to bounds to guarantee feasibility region.
    # Then find λ.
    lo = -1e6
    hi = 1e6
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, lower, upper)
    # adjust rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        # Add diff to any free variable(s)
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
        else:
            # Add to the first variable (which must be at bound but we can adjust)
            x[0] += diff
        # clip again
        x = np.clip(x, lower, upper)
    return x

        grad[s:] -= h[:n - s]
    else:
        s_abs = -s
        grad[s_abs:] -= h[:n - s_abs]
        grad[:n - s_abs] -= h[s_abs:]
    return grad, max_val

def _project_onto_box_sum(v, target_sum, lo=0.0, hi=1.0):
    """Project vector v onto {x : lo<=x<=hi, sum(x)=target_sum}."""
    # Clip initial bounds to guarantee feasibility region contains solution
    # Use water‑filling: find λ such that sum(clamp(v-λ, lo, hi)) = target_sum
    left, right = -1e6, 1e6
    for _ in range(60):
        mid = (left + right) * 0.5
        x = np.clip(v - mid, lo, hi)
        s = x.sum()
        if s > target_sum:
            left = mid
        else:
            right = mid
    x = np.clip(v - (left + right) * 0.5, lo, hi)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        free = (x > lo + 1e-12) & (x < hi - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
        else:
            x[0] += diff
        x = np.clip(x, lo, hi)
    return x


def _c5_value(h):
    """Return the C5 value for a discrete step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def _grad_max_corr(h):
    """Gradient of the (unnormalized) maximal correlation term."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    idx = np.argmax(corr)                     # first occurrence of the max
    shift = idx - (n - 1)                     # integer lag corresponding to max
    grad = np.ones_like(h)
    if shift >= 0:
        grad[: n - shift] -= h[shift:]
        grad[shift:]        -= h[: n - shift]
    else:
        s = -shift
        grad[s:]            -= h[: n - s]
        grad[: n - s]       -= h[s:]
    return grad, max_val

    while True:
        if time.time() - start > time_limit:
            break

        grad, _ = _grad_max_corr(h)
        v = h - step * grad
        h = _project_onto_box_sum(v, target)

        cur_val = _c5_value(h)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()

        step = max(step * decay, min_step)
        it += 1

        # occasional random transfer to escape plateaus
        if it % 200 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            max_delta = min(0.1, h[i], 1.0 - h[j])
            if max_delta > 0:
                delta = max_delta * np.random.rand()
                cand = h.copy()
                cand[i] -= delta
                cand[j] += delta
                cand_val = _c5_value(cand)
                if cand_val < cur_val:
                    h = cand
                    cur_val = cand_val
                    if cur_val < best_val:
                        best_val = cur_val
                        best_h = h.copy()

        if it >= 10000:
            break
        if step <= min_step and it > 500 and best_val - cur_val < 1e-12:
            break
    return best_h, best_val


def _initial_u_shape(n):
    """U‑shaped deterministic start: h(x)=|x-1| on [0,2]."""
    dx = 2.0 / n
    x = np.arange(n) * dx + dx * 0.5
    h = np.abs(x - 1.0)
    h = _project_onto_box_sum(h, n / 2.0)
    return h


def _random_feasible(n):
    v = np.random.random(n)
    return _project_onto_box_sum(v, n / 2.0)


def _upsample(h_small, n_target):
    """Linear interpolation of a smaller vector to a larger size."""
    n_small = len(h_small)
    dx_small = 2.0 / n_small
    dx_target = 2.0 / n_target
    xs = np.arange(n_small) * dx_small + dx_small * 0.5
    xt = np.arange(n_target) * dx_target + dx_target * 0.5
    return np.interp(xt, xs, h_small)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_val = np.inf
    best_n = None

    # candidate resolutions (all < 1000 points)
    n_candidates = [120, 200, 320]

    # include the parent size if available
    if "initial_h_values" in globals():
        n_parent = len(initial_h_values)
        n_candidates = [n_parent] + n_candidates

    for n in n_candidates:
        if time.time() - start_time > budget_s - 0.5:
            break

        # create a list of initial vectors for this n
        inits = []

        if "initial_h_values" in globals() and n == len(initial_h_values):
            # use the parent directly (projected to guarantee constraints)
            parent = np.asarray(initial_h_values, dtype=float).copy()
            parent = _project_onto_box_sum(parent, n / 2.0)
            inits.append(parent)

        # upsample parent if possible
        if "initial_h_values" in globals():
            up = _upsample(initial_h_values, n)
            up = _project_onto_box_sum(up, n / 2.0)
            inits.append(up)

        # deterministic U‑shape
        inits.append(_initial_u_shape(n))

        # a few random starts
        for _ in range(2):
            inits.append(_random_feasible(n))

        for h0 in inits:
            if time.time() - start_time > budget_s - 0.5:
                break
            # allocate a slice of the remaining budget to this optimisation run
            time_left = budget_s - (time.time() - start_time) - 0.1
            h_opt, val_opt = _gd_optimize(h0, time_left, start_time)
            if val_opt < best_val:
                best_val = val_opt
                best_h = h_opt.copy()
                best_n = n

    # Fallback – uniform half‑density if nothing succeeded
    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5)
        best_h = _project_onto_box_sum(best_h, n / 2.0)
        best_val = _c5_value(best_h)
        best_n = n

    # final validation (tiny rounding adjustments may be needed)
    best_h = _project_onto_box_sum(best_h, best_n / 2.0)
    best_val = _c5_value(best_h)

    return best_h, float(best_val), int(best_n)
