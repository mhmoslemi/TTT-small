    h = h0.copy()
    n = len(h)
    target = n/2
    best_h = h.copy()
    best_c5 = compute_c5(h)

        max_corr = corr.max()

        idxs = np.flatnonzero(corr == max_corr)
        grad = np.zeros_like(h)

            lag = idx - (n - 1)
            if lag >= 0:

                # we can vectorize using slicing
                # Prepare arrays of length n filled with zeros
                # contribution for positions i = 0..n-1-lag
                grad[:n-lag] += 1 - h[lag:]
                # contribution negative: - h[i] for i such that i+lag = j (i = j - lag)
                # i from 0..n-1-lag maps to j = i+lag from lag..n-1
                grad[lag:] -= h[:n-lag]

                grad[ell:] += 1 - h[:n-ell]

        # optional early stop

def subgrad_opt(initial_h, max_iter, step0, deadline):
    h = initial_h.copy()
    n = len(h)
    target = n/2.0
    best_h = h.copy()
    best_val = compute_c5(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        idxs = np.flatnonzero(corr == max_corr)
        grad = np.zeros_like(h)
        for idx in idxs:
            lag = idx - (n - 1)
            if lag >= 0:
                # up to n - lag
                # term plus:
                grad[:n - lag] += 1 - h[lag:]
                # term minus:
                grad[lag:] -= h[:n - lag]
            else:
                ell = -lag
                # term plus for positions ell..n-1
                grad[ell:] += 1 - h[:n - ell]
                # term minus for positions 0..n-ell-1
                grad[:n - ell] -= h[ell:]
        # step size
        step = step0 / np.sqrt(it + 1) / (np.linalg.norm(grad) + 1e-12)
        h = project_box_sum(h - step * grad, target)
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val
```

def project_box_sum(v, target_sum):
    lo = -1.0
    hi = 2.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

import numpy as np
import time
import random

# ---------- helper functions -------------------------------------------------

def project_box_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {0 ≤ x ≤ 1, sum(x)=target_sum}."""
    # binary search for Lagrange multiplier λ so that sum(clamp(v-λ,0,1)) = target_sum
    lo, hi = -1.0, 2.0                 # guarantees sum at lo ≥ target, at hi ≤ target
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant for a given discretised h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * float(corr.max())


def random_init(n: int) -> np.ndarray:
    """Uniform random vector in [0,1] with prescribed sum n/2."""
    v = np.random.rand(n)
    return project_box_sum(v, n / 2.0)


def upsample_h(h_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation of a step function onto a finer grid."""
    n_small = len(h_small)
    x_small = np.linspace(0.0, 2.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 2.0, n_big, endpoint=False)
    h_big = np.interp(x_big, x_small, h_small)
    return project_box_sum(h_big, n_big / 2.0)


def subgrad_opt(initial_h: np.ndarray, max_iter: int, step0: float, deadline: float):
    """Projected sub‑gradient descent on max‑overlap objective."""
    h = initial_h.copy()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(h)

    for it in range(max_iter):
        if time.time() > deadline:
            break

        # correlation and active lags
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        active = np.flatnonzero(corr == max_corr)

        grad = np.zeros_like(h)
        for idx in active:
            lag = idx - (n - 1)           # shift index (negative allowed)

            if lag >= 0:
                # contribution from i (h_i part)
                grad[: n - lag] += 1.0 - h[lag:]
                # contribution from i+lag (the -h_i term)
                grad[lag:] -= h[: n - lag]
            else:
                ell = -lag
                grad[ell:] += 1.0 - h[: n - ell]
                grad[: n - ell] -= h[ell:]

        # step size inversely proportional to iteration and gradient magnitude
        norm = np.linalg.norm(grad)
        if norm < 1e-12:
            # gradient vanished – perform a tiny random nudge
            h += (np.random.rand(n) - 0.5) * 1e-4
            h = project_box_sum(h, target)
            continue

        step = step0 / np.sqrt(it + 1) / norm
        h = project_box_sum(h - step * grad, target)

        cur = compute_c5(h)
        if cur < best_val:
            best_val = cur
            best_h = h.copy()

    return best_h, best_val


def local_search(h: np.ndarray, cur_val: float, deadline: float, max_trials: int = 2000):
    """Simple pairwise exchange hill‑climbing that preserves the sum."""
    n = len(h)
    cur_h = h.copy()
    cur_c5 = cur_val

    while max_trials > 0 and time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # random direction of transfer
        if np.random.rand() < 0.5:
            # move from i to j
            max_eps = min(cur_h[i], 1.0 - cur_h[j])
            if max_eps <= 0.0:
                max_trials -= 1
                continue
            eps = np.random.rand() * max_eps
            new_h = cur_h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            # move from j to i
            max_eps = min(cur_h[j], 1.0 - cur_h[i])
            if max_eps <= 0.0:
                max_trials -= 1
                continue
            eps = np.random.rand() * max_eps
            new_h = cur_h.copy()
            new_h[j] -= eps
            new_h[i] += eps

        new_val = compute_c5(new_h)
        if new_val < cur_c5 - 1e-12:
            cur_h = new_h
            cur_c5 = new_val
        max_trials -= 1

    return cur_h, cur_c5


def optimise_one(n: int, init_h: np.ndarray, deadline: float):
    """Run sub‑gradient descent + local search for a given grid size."""
    # parameters tuned for speed / quality
    max_iter = 4000
    step0 = 0.8
    h_opt, val = subgrad_opt(init_h, max_iter, step0, deadline)

    # a short local improvement phase
    if time.time() < deadline:
        h_opt, val = local_search(h_opt, val, deadline, max_trials=1500)

    return h_opt, val


# ---------- main entry point -------------------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C5 bound and the number of points used."""
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s * 0.99   # keep a tiny slack

    # base grid size (provided or fallback)
    n0 = len(initial_h_values) if 'initial_h_values' in globals() else 47

    # start from the given construction if available
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values, dtype=float)
        h_base = project_box_sum(h_base, n0 / 2.0)
    else:
        h_base = random_init(n0)

    best_h = h_base
    best_val = compute_c5(best_h)

    # ----- many restarts on the original grid ---------------------------------
    while time.time() < deadline * 0.6:
        h0 = random_init(n0)
        h_opt, val = optimise_one(n0, h0, deadline)
        if val < best_val:
            best_val, best_h = val, h_opt
        if best_val < 0.3808:
            break

    # ----- if time permits, try a doubled resolution -------------------------
    if best_val >= 0.3808 and time.time() < deadline * 0.9:
        n1 = n0 * 2
        if n1 <= 200:
            # initialise from upsampled best solution
            h_up = upsample_h(best_h, n1)

            while time.time() < deadline * 0.95:
                # random or upsampled start
                if np.random.rand() < 0.4:
                    h0 = h_up.copy()
                else:
                    h0 = random_init(n1)
                h_opt, val = optimise_one(n1, h0, deadline)
                if val < best_val:
                    best_val, best_h = val, h_opt
                    h_up = upsample_h(best_h, n1)  # keep a fresh upsample for next loop
                if best_val < 0.3808:
                    break

    # final projection just in case of tiny numeric drift
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_val = compute_c5(best_h)

    return best_h, best_val, len(best_h)
