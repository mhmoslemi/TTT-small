def project_onto_constraints(h):
    # clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    # adjust to satisfy sum using simple water-filling algorithm.
    total = h.sum()
    if np.isclose(total, target):
        return h
    # Use iterative approach: while diff not zero:
    diff = target - total
    # Determine indices that are not at bounds:
    while abs(diff) > 1e-9:
        # Identify free indices where adjustments can be made (i.e., h_i not at 0 if diff>0, not at 1 if diff<0)
        free = None
        if diff > 0:
            free = np.where(h < 1 - 1e-12)[0]
            if free.size == 0:
                break
            # max increase per free entry:
            max_inc = np.min(1 - h[free])
            inc = min(diff / free.size, max_inc)
            h[free] += inc
            diff -= inc * free.size
        else:
            free = np.where(h > 1e-12)[0]
            if free.size == 0:
                break
            max_dec = np.min(h[free])
            dec = min(-diff / free.size, max_dec)
            h[free] -= dec
            diff += dec * free.size
        # loop
    # final clip:
    h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np

def _project_onto_constraints(x, target_sum):
    """Project vector x onto the box [0,1] and the hyperplane sum(x)=target_sum."""
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - x.sum()
    # Simple water‑filling loop (fast for our small sizes)
    while abs(diff) > 1e-12:
        if diff > 0:
            free = np.where(x < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            inc = min(diff / free.size, np.min(1.0 - x[free]))
            x[free] += inc
            diff -= inc * free.size
        else:
            free = np.where(x > 1e-12)[0]
            if free.size == 0:
                break
            dec = min(-diff / free.size, np.min(x[free]))
            x[free] -= dec
            diff += dec * free.size
    return np.clip(x, 0.0, 1.0)


def _c5_value(h):
    """Evaluate C₅ for a step function given by vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _gradient_worst_shift(h):
    """Return the gradient of the worst shift (the one attaining max C₅)."""
    n = h.shape[0]
    corr_full = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr_full))
    shift = max_idx - (n - 1)           # integer shift causing the maximum
    grad = np.ones(n)
    if shift >= 0:
        # positions where i+shift is inside the array
        grad[: n - shift] -= h[shift:]
        # positions where i-shift is inside the array
        grad[shift:] -= h[: n - shift]
    else:
        s = -shift
        grad[s:] -= h[: n - s]
        grad[: n - s] -= h[s:]
    return grad, shift


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function giving a small C₅ value.

    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # initialise the vector (use the provided parent if there is one)
    # ------------------------------------------------------------------
    try:
        h0 = np.asarray(initial_h_values, dtype=float)   # parent supplied by the harness
        n0 = h0.shape[0]
    except Exception:          # no parent – fall back to a random binary start
        n0 = 45
        h0 = np.zeros(n0, dtype=float)
        h0[np.random.choice(n0, n0 // 2, replace=False)] = 1.0

    # up‑sample once (factor 2) – gives more freedom without blowing up runtime
    factor = 2
    h = np.repeat(h0, factor)
    n = h.shape[0]
    target_sum = n / 2.0
    h = _project_onto_constraints(h, target_sum)

    # ------------------------------------------------------------------
    # keep the best solution seen so far
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_score = _c5_value(best_h)

    # ------------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------------
    max_iter = 300000                # hard iteration cap (time check will stop earlier)
    step = 0.6                       # initial step size for gradient step
    decay = 0.9995                   # exponential decay of step size
    restart_interval = 7000          # how often to inject a random perturbation
    perturb_chance = 0.12            # probability of a full restart at a restart point

    # ------------------------------------------------------------------
    # main optimisation loop (projected gradient descent with occasional noise)
    # ------------------------------------------------------------------
    it = 0
    while it < max_iter:
        # time guard – leave a second margin for clean exit
        if time.time() - start_time > budget_s - 1.0:
            break

        # gradient of the worst shift
        grad, _ = _gradient_worst_shift(h)

        # gradient step + projection
        h_new = h - step * grad
        h_new = _project_onto_constraints(h_new, target_sum)

        # evaluate
        new_score = _c5_value(h_new)

        # keep the global best
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()

        # accept improvement (or stay) – we only move if we improve
        # (this is a simple hill‑climb; occasional noise helps escape plateaus)
        if new_score <= _c5_value(h):
            h = h_new

        # decay step size
        step *= decay

        # occasional random perturbation to explore new basins
        if (it + 1) % restart_interval == 0:
            if np.random.rand() < perturb_chance:
                # complete restart from the current best with a shake
                noise = (np.random.rand(n) - 0.5) * 0.3
                h = _project_onto_constraints(best_h + noise, target_sum)
                step = 0.6          # reset step size after a restart
            else:
                # gentle shake around the current point
                noise = (np.random.rand(n) - 0.5) * 0.1
                h = _project_onto_constraints(h + noise, target_sum)

        it += 1

    # ------------------------------------------------------------------
    # final evaluation and output
    # ------------------------------------------------------------------
    final_score = _c5_value(best_h)

    return (best_h.tolist(), float(final_score), n)
