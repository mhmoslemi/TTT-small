import time
import math
import random
import numpy as np

def compute_c5(h):
    """Return the C5 value for a step function h sampled on a uniform grid."""
    h = np.asarray(h, dtype=np.float64)
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv.max() * (2.0 / h.size)

def project_onto_constraints(v):
    """Project vector v onto the box [0,1]^n and the hyperplane sum(v)=n/2."""
    v = np.clip(v, 0.0, 1.0)
    n = v.size
    target = n / 2.0
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    diff = target - s
    if diff > 0:
        free = np.where(v < 1.0 - 1e-12)[0]
    else:
        free = np.where(v > 1e-12)[0]
    if free.size == 0:
        # fallback: adjust first element as much as possible
        v[0] = max(min(v[0] + diff, 1.0), 0.0)
        return v
    inc = diff / free.size
    v[free] += inc
    v = np.clip(v, 0.0, 1.0)
    # final tiny correction if rounding left a residue
    residual = target - v.sum()
    if abs(residual) > 1e-12:
        for idx in free:
            adj = max(min(v[idx] + residual, 1.0), 0.0) - v[idx]
            v[idx] += adj
            residual -= adj
            if abs(residual) < 1e-12:
                break
    return v

def propose_neighbor(h, step):
    """Generate a neighboring feasible vector by moving mass between two coordinates."""
    n = h.size
    i = np.random.randint(n)
    j = np.random.randint(n)
    while j == i:
        j = np.random.randint(n)
    hi, hj = h[i], h[j]
    max_inc = min(1.0 - hi, hj)       # increase i, decrease j
    max_dec = min(hi, 1.0 - hj)       # decrease i, increase j
    if np.random.rand() < 0.5:
        # try positive delta
        dmax = min(max_inc, step)
        if dmax <= 0.0:
            return None
        delta = np.random.rand() * dmax
    else:
        # try negative delta
        dmax = min(max_dec, step)
        if dmax <= 0.0:
            return None
        delta = -np.random.rand() * dmax
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    # enforce bounds (should already hold)
    new_h[i] = max(min(new_h[i], 1.0), 0.0)
    new_h[j] = max(min(new_h[j], 1.0), 0.0)
    return new_h
