def project_to_simplex_capped(v, target):
    """Project onto Euclidean distance onto intersection of box [0,1] and sum=target."""
    # Use algorithm: find tau such that sum_i clip(v_i - tau, 0, 1) = target.
    # Since sum_i clip(v_i - tau) is monotone decreasing in tau.
    lo = -np.max(v) - 1  # ensure v - lo is large positive, clip to 1 => sum = N
    hi = np.max(v) + 1   # ensure v - hi is large negative, clip to 0 => sum = 0
    for _ in range(60):
        mid = (lo + hi) / 2
        w = np.clip(v - mid, 0, 1)
        s = np.sum(w)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, 0, 1)

import time
import numpy as np


def _project_to_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto the set {x | 0≤x_i≤1, sum x_i = target}."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):                     # binary search for the correct offset
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(w)
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(v - tau, 0.0, 1.0)


def _evaluate(h: np.ndarray) -> float:
    """Return C₅ for the discretised step function h."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _max_shift_info(h: np.ndarray):
    """Maximum correlation value, corresponding integer shift, and the full correlation array."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_raw = corr[idx]
    shift = idx - (h.shape[0] - 1)   # zero shift is at index n-1
    return max_raw, shift, corr


def _hill_step(h: np.ndarray, step: float):
    """Try a deterministic move that reduces the current worst‑overlap shift."""
    n = h.shape[0]
    max_raw, shift, _ = _max_shift_info(h)
    cur_val = max_raw * (2.0 / n)

    if shift == 0:                     # no useful move when the worst shift is zero
        return False, h, cur_val

    # indices participating in the worst shift
    if shift > 0:
        i_start, i_end = 0, n - shift
    else:
        i_start, i_end = -shift, n
    i_idx = np.arange(i_start, i_end)
    j_idx = i_idx + shift

    # contribution of each overlapping pair to the worst value
    contrib = h[i_idx] * (1.0 - h[j_idx])
    local_max = int(np.argmax(contrib))
    i = i_idx[local_max]
    j = j_idx[local_max]

    # move mass from i → j (reduces h_i·(1‑h_j))
    delta = min(step, h[i], 1.0 - h[j])
    if delta > 1e-12:
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_val = _evaluate(h_new)
        if new_val < cur_val - 1e-12:
            return True, h_new, new_val

    # try the opposite direction (j → i)
    delta2 = min(step, h[j], 1.0 - h[i])
    if delta2 > 1e-12:
        h_new = h.copy()
        h_new[j] -= delta2
        h_new[i] += delta2
        new_val = _evaluate(h_new)
        if new_val < cur_val - 1e-12:
            return True, h_new, new_val

    # finally try a few smaller steps in the preferred direction
    for factor in (0.5, 0.25, 0.125):
        d = delta * factor
        if d <= 1e-12:
            continue
        h_test = h.copy()
        h_test[i] -= d
        h_test[j] += d
        new_val = _evaluate(h_test)
        if new_val < cur_val - 1e-12:
            return True, h_test, new_val

    return False, h, cur_val


def _random_move(h: np.ndarray, max_step: float):
    """Create a feasible random transfer of mass between two entries."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = max_step * np.random.rand()
    if np.random.rand() < 0.5:
        delta = min(delta, h[i], 1.0 - h[j])
        if delta <= 1e-12:
            return None
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        return h_new
    else:
        delta = min(delta, h[j], 1.0 - h[i])
        if delta <= 1e-12:
            return None
        h_new = h.copy()
        h_new[j] -= delta
        h_new[i] += delta
        return h_new


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a discretised step function h on [0,2] to minimise the Erdős C₅ overlap constant.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # -------------------------------------------------
    #  initialise problem size and a feasible starting point
    # -------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)  # may be supplied by the harness
        n = init.shape[0]
    except Exception:
        n = 100                     # a modest size when no parent is provided
        init = None

    target_sum = n / 2.0
    if init is not None:
        cur_h = _project_to_capped_simplex(init, target_sum)
    else:
        cur_h = _project_to_capped_simplex(np.random.rand(n), target_sum)

    cur_val = _evaluate(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    # -------------------------------------------------
    #  optimisation parameters
    # -------------------------------------------------
    step = 0.20               # maximal transferred mass in one move
    min_step = 0.001
    no_improve = 0
    max_no_improve = 3000    # after this many failed attempts we restart
    decay_factor = 0.9995    # slowly shrink the step after successful moves

    # -------------------------------------------------
    #  main optimisation loop
    # -------------------------------------------------
    while time.time() - start_time < budget_s - 0.5:
        # 70 % of the time perform a directed “hill‑step”, otherwise a random move
        if np.random.rand() < 0.7:
            improved, new_h, new_val = _hill_step(cur_h, step)
        else:
            cand = None
            for _ in range(5):
                cand = _random_move(cur_h, step)
                if cand is not None:
                    break
            if cand is None:                                 # fallback: small Gaussian perturbation
                cand = _project_to_capped_simplex(cur_h + step * np.random.randn(n), target_sum)
            new_val = _evaluate(cand)
            improved = new_val < cur_val - 1e-12
            new_h = cand

        if improved:
            cur_h, cur_val = new_h, new_val
            no_improve = 0
            step = max(min_step, step * decay_factor)
            if cur_val < best_val:
                best_h, best_val = cur_h.copy(), cur_val
        else:
            no_improve += 1
            if no_improve % 500 == 0:                     # gently increase step when stuck
                step = min(0.5, step * 1.1)

        # hard restart after a long barren spell
        if no_improve >= max_no_improve:
            cur_h = _project_to_capped_simplex(np.random.rand(n), target_sum)
            cur_val = _evaluate(cur_h)
            no_improve = 0
            step = max(min_step, step * 0.5)

    # final safety projection and evaluation
    best_h = _project_to_capped_simplex(best_h, target_sum)
    best_val = _evaluate(best_h)

    return (best_h.tolist(), best_val, n)
