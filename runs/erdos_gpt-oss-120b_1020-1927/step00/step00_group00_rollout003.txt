def project_onto_simplex_and_box(x, T):
    # return h = clip(x + lam, 0, 1) where lam chosen s.t. sum(h) = T.
    # Use bisection on lam.
    low = -1.0
    high = 1.0
    for _ in range(50):
        lam = (low + high) / 2.0
        h = np.clip(x + lam, 0, 1)
        s = np.sum(h)
        if s < T:
            low = lam
        else:
            high = lam
    h = np.clip(x + (low+high)/2.0, 0, 1)
    # may not be exactly T due to rounding; return h
    return h

    """Euclidean projection of x onto {low<=z<=high, sum(z)=target}."""
    lo, hi = -10.0, 10.0
    # enlarge interval until the target is bracketed
    for _ in range(100):
        s_lo = np.sum(np.clip(x + lo, low, high))
        s_hi = np.sum(np.clip(x + hi, low, high))
        if s_lo > target:
            lo -= 10.0
        elif s_hi < target:
            hi += 10.0
        else:
            break
    # binary search for lambda
    for _ in range(60):
        lam = (lo + hi) / 2.0
        s = np.sum(np.clip(x + lam, low, high))
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(x + (lo + hi) / 2.0, low, high)


def random_feasible_h(n):
    """Uniform random point in the hyper‑cube intersecting the sum‑constraint."""
    x = np.random.rand(n)
    return project_onto_box_hyperplane(x, 0.0, 1.0, n / 2.0)


def random_exchange_small(h, max_delta=0.05):
    """Exchange a small amount of mass between two coordinates, preserving the sum."""
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]
    delta_min = max(-hi, hj - 1.0)
    delta_max = min(1.0 - hi, hj)
    if delta_max <= delta_min:
        return None
    delta = np.random.uniform(delta_min, delta_max)
    if abs(delta) > max_delta:
        delta = max_delta if delta > 0 else -max_delta
        delta = max(delta, delta_min)
        delta = min(delta, delta_max)
    h2 = h.copy()
    h2[i] += delta
    h2[j] -= delta
    return h2


def hill_climb(h, time_deadline, steps=2000, start_delta=0.02):
    """Simple deterministic hill climbing with decreasing step size."""
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    delta = start_delta
    for _ in range(steps):
        if time.time() > time_deadline:
            break
        i, j = np.random.choice(n, 2, replace=False)
        hi, hj = cur[i], cur[j]
        d_min = max(-hi, hj - 1.0)
        d_max = min(1.0 - hi, hj)
        if d_max <= d_min:
            continue
        d = np.random.uniform(d_min, d_max)
        if abs(d) > delta:
            d = delta if d > 0 else -delta
            d = max(d, d_min)
            d = min(d, d_max)
        cand = cur.copy()
        cand[i] += d
        cand[j] -= d
        cand_c5 = compute_c5(cand)
        if cand_c5 < cur_c5 - 1e-12:
            cur, cur_c5 = cand, cand_c5
            delta = start_delta          # reset step size after improvement
        else:
            delta *= 0.995               # slowly shrink step size
            if delta < 1e-6:
                delta = start_delta
    return cur, cur_c5


# ---------- main optimisation routine ----------
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = np.inf
    best_n = None

    # use the provided construction if any
    try:
        init_arr = np.array(initial_h_values, dtype=float)
        n0 = init_arr.size
        init_arr = project_onto_box_hyperplane(init_arr, 0.0, 1.0, n0 / 2.0)
        c5_init = compute_c5(init_arr)
        best_h, best_c5, best_n = init_arr, c5_init, n0
    except NameError:
        init_arr = None

    # list of discretisations to explore
    grid_sizes = [45, 90, 180]
    for n in grid_sizes:
        if time.time() - start_time > budget_s * 0.95:
            break
        target_sum = n / 2.0
        bounds = [(0.0, 1.0)] * n
        eq_cons = {'type': 'eq', 'fun': lambda h, T=target_sum: np.sum(h) - T}
        max_restarts = 80
        for _ in range(max_restarts):
            if time.time() - start_time > budget_s * 0.95:
                break

            # initialise a feasible point
            if init_arr is not None and n == init_arr.size:
                h0 = init_arr.copy()
                h0 = random_exchange_small(h0, max_delta=0.03) or h0
            elif init_arr is not None and n > init_arr.size:
                # linear interpolation to a finer grid
                x_new = np.linspace(0, 2, n, endpoint=False)
                x_old = np.linspace(0, 2, init_arr.size, endpoint=False)
                h0 = np.interp(x_new, x_old, init_arr)
                h0 = project_onto_box_hyperplane(h0, 0.0, 1.0, target_sum)
            else:
                h0 = random_feasible_h(n)

            # local optimisation with SLSQP
            maxiter = 400 if n <= 90 else 800
            res = minimize(
                lambda h: np.max(np.correlate(h, 1 - h, mode='full')),
                h0,
                method='SLSQP',
                bounds=bounds,
                constraints=[eq_cons],
                options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
            )
            if not res.success:
                cand = h0
            else:
                cand = project_onto_box_hyperplane(res.x, 0.0, 1.0, target_sum)

            cand_c5 = compute_c5(cand)
            if cand_c5 < best_c5:
                best_h, best_c5, best_n = cand, cand_c5, n

            # further refinement with cheap hill climbing
            deadline = start_time + budget_s * 0.95
            refined, refined_c5 = hill_climb(cand, deadline, steps=1500, start_delta=0.02)
            if refined_c5 < best_c5:
                best_h, best_c5, best_n = refined, refined_c5, n

    # final safety projection and exact evaluation (if helper available)
    if best_h is not None:
        best_h = project_onto_box_hyperplane(best_h, 0.0, 1.0, best_n / 2.0)
        try:
            best_c5 = float(evaluate_erdos_solution(best_h))
        except Exception:
            best_c5 = compute_c5(best_h)

    # fallback (should never happen)
    if best_h is None:
        best_n = 45
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    return best_h.tolist(), best_c5, int(best_n)
