def simulated_annealing(h_init, rng, time_budget):
    h = h_init.copy()
    cur_score = compute_c5(h)
    best_h = h.copy(); best_score = cur_score
    T0 = 0.05  # initial temperature range of score difference
    start_time = time.time()
    iter = 0
    while time.time() - start_time < time_budget:
        # propose swap
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h > 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        i0 = rng.choice(zeros)
        i1 = rng.choice(ones)
        # swap
        h[i0], h[i1] = 1.0, 0.0
        new_score = compute_c5(h)
        delta = new_score - cur_score
        # acceptance criteria
        # temperature schedule linear decreasing
        T = max(T0 * (1 - (time.time() - start_time)/time_budget), 0.001)
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            # revert swap
            h[i0], h[i1] = 0.0, 1.0
        iter += 1
    return best_h, best_score

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    L = l_fft  # precomputed
    # pad arrays to length L
    h_pad = np.zeros(L, dtype=np.float64)
    h_pad[:n] = h
    g_pad = np.zeros(L, dtype=np.float64)
    g_pad[:n] = 1.0 - h
    # reverse g_pad up to n:
    g_rev = np.zeros(L, dtype=np.float64)
    g_rev[:n] = (1.0 - h)[::-1]
    # compute convolution via rfft
    H = np.fft.rfft(h_pad)
    G = np.fft.rfft(g_rev)
    conv = np.fft.irfft(H * G, n=L)[:2 * n - 1]
    return dx * np.max(conv)

def simulated_annealing_binary(h_start, max_iters, rng, time_limit):
    h = h_start.copy()
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    cur_score = compute_c5(h)
    best_score = cur_score
    best_h = h.copy()
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        i0 = rng.choice(zeros)
        i1 = rng.choice(ones)
        # swap
        h[i0] = 1.0
        h[i1] = 0.0
        new_score = compute_c5(h)
        # acceptance
        # Cooling schedule: linear dec.
        t = max(0.001, (max_iters - it) / max_iters)  # T ~ 1 -> 0.001
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / t):
            cur_score = new_score
            # update zeros/ones arrays
            zeros[zeros == i0] = i1
            ones[ones == i1] = i0
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            # revert swap
            h[i0] = 0.0
            h[i1] = 1.0
        # optionally flush arrays after many updates to keep dtype
    return best_h, best_score

import numpy as np
import time

def _next_pow2(x: int) -> int:
    """small utility to get next power of two >= x."""
    return 1 << (x - 1).bit_length()


def _correlation_h_one_minus_h(h: np.ndarray, L: int) -> np.ndarray:
    """
    Linear correlation of h with (1-h) using FFT.
    Returns array of length 2*len(h)-1.
    """
    n = h.shape[0]
    # reverse (1-h) for correlation
    rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, n=L)
    R = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(H * R, n=L)[: 2 * n - 1]
    # eliminate tiny negative rounding errors
    conv[conv < 0] = 0.0
    return conv


def _project_to_feasible(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Orthogonal projection of v onto the set
        {x : 0 <= x_i <= 1, sum(x) = target_sum}.
    Uses a simple binary search for the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
