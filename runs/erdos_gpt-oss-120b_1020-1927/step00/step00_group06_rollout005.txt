import time
import numpy as np

def project_onto_simplex_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the hypercube [0,1]^n with sum equal to target."""
    # Bisection on λ where x_i = clip(v_i - λ, 0, 1)
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def compute_c5(h: np.ndarray, dx: float) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k)) dx for the discretised step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def propose_local(h: np.ndarray, rng: np.random.Generator, sigma: float) -> np.ndarray:
    """Transfer a small amount of mass from one coordinate to another."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    max_delta = min(1.0 - h[i], h[j])
    if max_delta <= 0.0:
        return h.copy()
    delta = sigma * max_delta * rng.random()
    nh = h.copy()
    nh[i] += delta
    nh[j] -= delta
    return nh


def propose_global(h: np.ndarray, rng: np.random.Generator, sigma_big: float) -> np.ndarray:
    """Add Gaussian noise to all coordinates (later projected)."""
    return h + sigma_big * rng.standard_normal(h.shape[0])


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and starting point
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        n = init_arr.shape[0]
    else:
        init_arr = None
        n = 95  # fallback size

    dx = 2.0 / n
    target_sum = n / 2.0

    if init_arr is not None and init_arr.size == n:
        cur_h = project_onto_simplex_bounds(init_arr, target_sum)
    else:
        cur_h = project_onto_simplex_bounds(rng.random(n), target_sum)

    best_h = cur_h.copy()
    best_val = compute_c5(best_h, dx)

    # ------------------------------------------------------------------
    # Simulated‑annealing parameters
    # ------------------------------------------------------------------
    temperature = 0.12
    sigma = 0.35         # typical step size for local moves
    sigma_big = 0.70     # step size for occasional global moves
    no_improve = 0
    max_no_improve = 8000

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while True:
        if time.time() - start > budget_s - 0.5:
            break

        # Decide move type
        if rng.random() < 0.02:               # global perturbation
            v = propose_global(cur_h, rng, sigma_big)
            new_h = project_onto_simplex_bounds(v, target_sum)
        else:                                 # local transfer
            new_h = propose_local(cur_h, rng, sigma)

        new_val = compute_c5(new_h, dx)

        # Metropolis acceptance
        if new_val < best_val or rng.random() < np.exp(
            -(new_val - best_val) / max(temperature, 1e-12)

            cur_h = new_h
            if new_val < best_val:
                best_h = new_h
                best_val = new_val
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Cool down / adapt step sizes
        temperature *= 0.99995
        sigma = max(sigma * 0.99999, 0.01)
        sigma_big = max(sigma_big * 0.9999, 0.15)

        # Random restart if stuck
        if no_improve > max_no_improve:
            cur_h = project_onto_simplex_bounds(rng.random(n), target_sum)
            no_improve = 0

    # Final projection to guarantee feasibility
    best_h = project_onto_simplex_bounds(best_h, target_sum)
    best_val = compute_c5(best_h, dx)
