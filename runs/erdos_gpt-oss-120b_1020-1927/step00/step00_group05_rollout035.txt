def project_capped_simplex(v, S):
    n = len(v)
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    # Ensure that at lo total > S and at hi total < S
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        total = w.sum()
        if total > S:
            lo = lam
        else:
            hi = lam
    # final
    lam = (lo + hi) / 2.0
    w = np.clip(v - lam, 0.0, 1.0)
    return w

def gradient_step(h, eta, S):
    n = len(h)
    conv = np.correlate(h, h, mode='full')
    k_idx = np.argmin(conv)
    shift = k_idx - (n - 1)
    g = np.zeros_like(h)
    if shift >= 0:
        if shift > 0:
            g[:n-shift] += h[shift:]
            g[shift:] += h[:n-shift]
        else:
            # shift=0: gradient = 2*h
            g[:] = 2 * h
    else:
        k = -shift
        g[:n-k] += h[k:]
        g[k:] += h[:n-k]
    # update and project
    h_new = h + eta * g
    h_new = project_capped_simplex(h_new, S)
    return h_new

import time
import numpy as np

def _project_capped_simplex(v, target_sum):
    """Project vector v onto {0<=x<=1, sum(x)=target_sum}."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def _c5_value(h):
    """Return the C5 value for step function h (array of samples)."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / n)

def _min_product_shift(h):
    """Return index shift (as integer) that minimises the product sum Σ h_i h_{i+sh}."""
    conv = np.convolve(h, h, mode="full")  # same as correlation for real vectors
    # argmin gives the worst shift (including zero)
    return np.argmin(conv) - (h.shape[0] - 1)

def _gradient_step(h, eta, target_sum):
    """Move h a little in the direction that improves the worst shift."""
    n = h.shape[0]
    shift = _min_product_shift(h)
    g = np.zeros_like(h)

    if shift >= 0:
        if shift == 0:
            g[:] = 2.0 * h
        else:
            g[: n - shift] += h[shift:]
            g[shift:] += h[: n - shift]
    else:
        k = -shift
        g[: n - k] += h[k:]
        g[k:] += h[: n - k]

    h_new = h + eta * g
    h_new = _project_capped_simplex(h_new, target_sum)
    return h_new

def _binarise_top(h):
    """Return a binary vector with exactly n/2 ones (choose the largest entries)."""
    n = h.shape[0]
    half = n // 2
    idx = np.argpartition(-h, half - 1)[:half]
    bin_h = np.zeros_like(h)
    bin_h[idx] = 1.0
    return bin_h
