import time
import numpy as np

def _shift_zero(arr, s):
    """Return arr shifted by s with zero padding (out‑of‑range entries = 0)."""
    n = arr.shape[0]
    if s == 0:
        return arr.copy()
    if s > 0:
        out = np.empty(n, dtype=arr.dtype)
        out[:s] = 0.0
        out[s:] = arr[:n - s]
        return out
    # s < 0
    s = -s
    out = np.empty(n, dtype=arr.dtype)
    out[-s:] = 0.0
    out[:n - s] = arr[s:]
    return out


def _c5_value(h, dx):
    """Maximum of the overlap integral for the current h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()


def _initial_tent(n):
    """Triangular (tent) shape on [0,2] with total integral 1."""
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1e-12
    h = np.minimum(x, 2.0 - x)
    dx = 2.0 / n
    total = h.sum() * dx
    factor = 1.0 / total
    h = np.clip(h * factor, 0.0, 1.0)
    # enforce exact sum after clipping
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
        else:
            h *= target / h.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the step function h to minimise the Erdos C5 constant.

    Returns:
        h_values : np.ndarray   – the optimal discretised step function
        c5_bound : float        – the achieved C5 value
        n_points : int          – number of discretisation points
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------
    # 1.  Determine the discretisation size and an initial feasible h
    # ------------------------------------------------------------
    try:
        # initial_h_values is provided by the harness (if any)
        h = np.asarray(initial_h_values, dtype=np.float64)
        n = h.size
    except Exception:          # no supplied initial vector – make a tent
        n = 200                 # a modest size (fast, yet expressive)
        h = _initial_tent(n)

    dx = 2.0 / n
    target_sum = n / 2.0

    # ensure feasibility (clip then correct the integral)
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    if not np.isclose(cur_sum, target_sum, atol=1e-9):
        diff = target_sum - cur_sum
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
        else:
            h *= target_sum / cur_sum
        h = np.clip(h, 0.0, 1.0)

    # ------------------------------------------------------------
    # 2.  Helper to compute C5 and the set of worst shifts
    # ------------------------------------------------------------
    def evaluate(h_vec):
        """Return (max_corr, list_of_worst_shifts)."""
        corr = np.correlate(h_vec, 1.0 - h_vec, mode="full") * dx
        maxv = corr.max()
        # indices where the max is attained (within tiny tolerance)
        idxs = np.where(np.isclose(corr, maxv, atol=1e-14))[0]
        shifts = idxs - (n - 1)               # integer shift values
        return maxv, shifts

    best_h = h.copy()
    best_c5 = _c5_value(best_h, dx)

    # ------------------------------------------------------------
    # 3.  Main optimisation loop – projected coordinate ascent on the worst shift
    # ------------------------------------------------------------
    max_step = 0.25          # initial maximal transfer amount
    min_step = 1e-4
    decay = 0.9995           # geometric decay of step size
    it = 0
    no_improve = 0

    while time.time() - start_time < budget_s * 0.94:
        it += 1
        # current objective and worst shifts
        cur_c5, worst_shifts = evaluate(h)

        # gradient that tells which coordinates help to lower the current worst value
        grad = np.zeros_like(h)
        for s in worst_shifts:

            grad += 1.0 - (left + right)

        dec_idx = int(np.argmax(grad))      # decrease here (most positive derivative)

            delta = min(delta, 1.0 - h[i], h[j])
            if delta > 0:

                h[j] -= delta

        # admissible transfer amount respecting the box constraints
        delta = max_step * rng.random()
        delta = min(delta, 1.0 - h[inc_idx], h[dec_idx])

            for _ in range(5):

            elapsed = time.time() - start_time

            if rng.random() < np.exp(-(trial_c5 - cur_c5) / temperature):
                h = h_trial
                cur_c5 = trial_c5

        # step‑size schedule
        max_step = max(min_step, max_step * decay)

    # ------------------------------------------------------------
    # 4.  Final post‑processing – enforce constraints exactly
    # ------------------------------------------------------------
    final_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = final_h.sum()
    if not np.isclose(cur_sum, target_sum, atol=1e-9):
        diff = target_sum - cur_sum
        free = (final_h > 0.0) & (final_h < 1.0)

            final_h[free] += diff / free.sum()
        else:

        final_h = np.clip(final_h, 0.0, 1.0)

    final_c5 = _c5_value(final_h, dx)
    return final_h, float(final_c5), n
