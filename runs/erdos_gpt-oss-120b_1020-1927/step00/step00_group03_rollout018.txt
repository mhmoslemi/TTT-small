def proj_cap_simplex(v, target):
    # Use binary search for lambda.
    low = np.min(v) - 1.0  # ensure low enough
    high = np.max(v) + 1.0 # ensure high enough
    # we search for lambda such that sum(clip(v - lambda, 0, 1)) = target
    for _ in range(100):
        mid = (low + high)/2
        x = np.clip(v - mid, 0, 1)
        s = np.sum(x)
        if s > target:
            low = mid
        else:
            high = mid
    return np.clip(v - (low+high)/2, 0, 1)

import time
import numpy as np

# ---------- projection onto the capped simplex ----------
def _proj_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {x : sum(x)=target, 0<=x<=1} using a binary search on the dual."""
    lo = np.min(v) - 1.0          # low enough so that all entries are clipped to 1
    hi = np.max(v) + 1.0          # high enough so that all entries are clipped to 0
    for _ in range(60):          # enough for double precision
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


# ---------- smooth max correlation and its gradient ----------
def _smooth_max_and_grad(h: np.ndarray, gamma: float):
    """Return a smooth approximation of max_k Corr(k) and its sub‑gradient."""
    n = h.size
    # raw correlations (no dx factor)
    r = np.correlate(h, 1.0 - h, mode="full")
    # stable soft‑max weights
    r_max = r.max()
    shifted = r - r_max                         # now max(shifted)=0
    exp_shift = np.exp(gamma * shifted)
    w = exp_shift / exp_shift.sum()             # weights sum to 1

    # gradient = sum_k w_k * grad Corr(k)
    g = np.zeros_like(h)
    mid = n - 1
    for shift in range(-n + 1, n):
        weight = w[shift + mid]
        if weight == 0.0:
            continue
        if shift >= 0:
            # forward contribution (1 - h[i+shift])
            g[: n - shift] += weight * (1.0 - h[shift:])
            # backward contribution (-h[i-shift])
            g[shift:] -= weight * h[: n - shift]
        else:
            s = -shift
            g[s:] += weight * (1.0 - h[: n - s])
            g[: n - s] -= weight * h[s:]
    return r.max(), g


# ---------- main optimisation routine ----------
