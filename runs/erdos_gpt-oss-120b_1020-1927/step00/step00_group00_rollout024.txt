import time
import numpy as np
from scipy.optimize import minimize

# ------------------- evaluation helpers ------------------- #
def _c5_from_h(h: np.ndarray) -> float:
    """Exact C5 evaluation used by the harness."""
    n = h.size
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def _all_shifts(h: np.ndarray) -> np.ndarray:
    """Return the overlap for every integer shift (full correlation)."""
    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx  # length 2n‑1

# ------------------- smooth max objective ------------------- #
def _smooth_max_val(h: np.ndarray, alpha: float) -> float:
    """Log‑sum‑exp smooth approximation of max(shift overlap)."""
    D = _all_shifts(h)
    m = np.max(D)
    # stable log‑sum‑exp
    ex = np.exp(alpha * (D - m))
    return m + np.log(np.sum(ex)) / alpha


def _smooth_max_grad(h: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of the smooth‑max objective."""
    n = h.size
    dx = 2.0 / n

    D = _all_shifts(h)
    m = np.max(D)
    ex = np.exp(alpha * (D - m))
    w = ex / np.sum(ex)                     # weights, sum to 1, length 2n‑1

    # forward[i] = Σ_m w_m * h_{i+m},  backward[i] = Σ_m w_m * h_{i‑m}
    forward = np.zeros(n, dtype=np.float64)
    backward = np.zeros(n, dtype=np.float64)

    # w index 0 → shift = -(n‑1)
    for idx, wv in enumerate(w):
        if wv == 0.0:
            continue
        shift = idx - (n - 1)               # actual integer shift
        if shift >= 0:
            # forward contribution: h[i+shift]
            forward[: n - shift] += wv * h[shift:]
            # backward contribution: h[i‑shift]
            backward[shift:] += wv * h[: n - shift]
        else:
            s = -shift                       # positive value
            # forward: h[i‑s]
            forward[s:] += wv * h[: n - s]
            # backward: h[i+s]
            backward[: n - s] += wv * h[s:]

    grad = dx * (1.0 - (forward + backward))
    return grad


# ------------------- core optimizer ------------------- #
def _optimize_h(init_h: np.ndarray, max_iter: int, alpha: float) -> np.ndarray:
    """SLSQP optimisation of the smooth‑max objective."""
    n = init_h.size
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}

    def fun(x):
        return _smooth_max_val(x, alpha)

    def jac(x):
        return _smooth_max_grad(x, alpha)

    res = minimize(
        fun,
        init_h,
        method="SLSQP",
        jac=jac,
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    # clip to stay inside the box (numeric drift)
    x = np.clip(res.x, 0.0, 1.0)

    # tiny correction to enforce the exact integral (sum = n/2)
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        # distribute the remaining mass uniformly over non‑saturated components
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size:
            x[free] += diff / free.size
            x = np.clip(x, 0.0, 1.0)
    return x


# ------------------- main driver ------------------- #
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with a low C₅ bound."""
    start = time.time()
    np.random.seed(seed)

    # starting point: use provided construction if available
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=np.float64)
    else:
        # fallback: uniform Dirichlet of length 45
        n0 = 45
        base = np.random.dirichlet(np.ones(n0)) * (n0 / 2.0)

    best_h = None
    best_score = np.inf
    best_n = None

    # try several resolutions (repeat the base step function)
    for factor in (1, 2, 4):
        if time.time() - start > budget_s:
            break
        n = base.size * factor
        # repeat the base pattern if factor > 1
        h_repeat = np.repeat(base, factor) if factor > 1 else base.copy()

        # candidate starting vectors for this resolution
        starts = [(h_repeat, "repeat")]
        # a couple of random Dirichlet starts
        for _ in range(2):
            rnd = np.random.dirichlet(np.ones(n)) * (n / 2.0)
            starts.append((rnd, "rand"))

        for h0, tag in starts:
            if time.time() - start > budget_s:
                break

            # allocate iterations proportional to remaining time
            remaining = budget_s - (time.time() - start)
            max_iter = max(200, int(min(2500, remaining * 2.5)))  # very generous upper bound

            try:
                opt_h = _optimize_h(h0, max_iter=max_iter, alpha=12.0)
                score = _c5_from_h(opt_h)
                if score < best_score:
                    best_score = score
                    best_h = opt_h.copy()
                    best_n = n
            except Exception:
                # any numerical trouble – just skip this attempt
                continue

    # fallback to the base pattern if nothing succeeded
    if best_h is None:
        best_h = base.copy()
        best_score = _c5_from_h(best_h)
        best_n = base.size

    return (best_h, float(best_score), int(best_n))
