    best_score = compute_c5(best_h)
    cur_h = h.copy()
    cur_score = best_score

        # Choose random number of swaps t between 1 and max_swaps
        # keep half t ones/t zeros.
        # Let's usually choose t = 1 with prob 0.8 else t=2

        i_swap = rng.choice(ones_idx, size=t, replace=False)
        j_swap = rng.choice(zeros_idx, size=t, replace=False)
        h_new = cur_h.copy()
        h_new[i_swap] = 0
        h_new[j_swap] = 1
        new_score = compute_c5(h_new)
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / temperature):
            cur_h = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score
                no_improve = 0
        else:
            no_improve += 1
        # update temperature schedule
        temperature = schedule(it)
        # early break if no_improve > patience
        if no_improve > patience:
            break
    return best_h, best_score

import time
import numpy as np

def _compute_c5(h):
    """Return the C5 value for binary vector h."""
    n = h.shape[0]
    # correlation of h with (1-h)
    c = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode="full")
    return np.max(c) * (2.0 / n)


def _mismatched_indices(h, shift):
    """Return two arrays:
       - positions i where h[i]=1 and h[i+shift]=0  (unmatched ones)
       - positions i where h[i]=0 and h[i+shift]=1  (unmatched zeros)
       Only indices with both i and i+shift inside the array are considered.
    """
    n = h.shape[0]
    if shift >= 0:
        i_cur = np.arange(n - shift)
        i_sh = i_cur + shift
    else:               # negative shift
        k = -shift
        i_cur = np.arange(k, n)
        i_sh = i_cur - k
    hi = h[i_cur]
    hs = h[i_sh]
    mis_one = i_cur[(hi == 1) & (hs == 0)]
    mis_zero = i_cur[(hi == 0) & (hs == 1)]
    return mis_one, mis_zero


def _greedy_improve(start_h, rng, max_iters=3000):
    """Greedy local search that repeatedly reduces the worst shift."""
    cur_h = start_h.copy()
    n = cur_h.shape[0]
    best_h = cur_h.copy()
    best_score = _compute_c5(best_h)

    for _ in range(max_iters):
        # correlation gives the mismatch count for every lag
        corr = np.correlate(cur_h.astype(np.float64), 1.0 - cur_h.astype(np.float64), mode="full")
        max_val = corr.max()
        if max_val == 0:      # perfect solution
            break

        # choose a lag attaining the maximum (randomly among them)
        idx_candidates = np.where(corr == max_val)[0]
        idx = rng.choice(idx_candidates)
        shift = idx - (n - 1)          # convert index to integer shift

        # find a mismatched 1 and a mismatched 0 for this shift
        mis_one, mis_zero = _mismatched_indices(cur_h, shift)
        if mis_one.size == 0 or mis_zero.size == 0:
            # cannot improve this lag – give up on this start
            break

        i = rng.choice(mis_one)
        j = rng.choice(mis_zero)
        # perform the swap (i is a 1, j is a 0)
        cur_h[i] = 0
        cur_h[j] = 1

        new_score = _compute_c5(cur_h)
        if new_score <= best_score:
            # accept the move (always accept improvements)
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert – the move did not help
            cur_h[i] = 1
            cur_h[j] = 0

    return best_h, best_score
