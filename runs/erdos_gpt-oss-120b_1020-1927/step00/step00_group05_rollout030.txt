def project_sum_box(x, total, lo=0., hi=1., max_iter=100):
    y = np.clip(x, lo, hi)
    for _ in range(max_iter):
        # compute diff to meet sum
        diff = total - y.sum()
        if abs(diff) < 1e-12:
            break
        # adjust by distributing diff equally among non-saturated elements
        free = (y > lo + 1e-12) & (y < hi - 1e-12)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            # cannot adjust further
            break
        delta = diff / n_free
        y[free] += delta
        # clip again
        y = np.clip(y, lo, hi)
    return y

import time
import numpy as np

def _project_sum_box(v, total, lo=0.0, hi=1.0, maxit=60):
    """
    Euclidean projection of v onto the intersection of the box [lo,hi]^n
    and the hyperplane sum(v)=total.  Uses a simple bisection on the
    Lagrange multiplier for the equality constraint.
    """
    # ensure feasible interval for lambda: at lambda=-2 all entries become hi,
    # at lambda=+2 all entries become lo.
    lam_low, lam_high = -2.0, 2.0
    for _ in range(maxit):
        lam_mid = (lam_low + lam_high) * 0.5
        y = np.clip(v - lam_mid, lo, hi)
        s = y.sum()
        if s > total:
            lam_low = lam_mid
        else:
            lam_high = lam_mid
    return np.clip(v - (lam_low + lam_high) * 0.5, lo, hi)

def _score(h, dx):
    """C5 value for a given h (max overlap integral)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # 1. Initialise h
    # ------------------------------------------------------------------ #
    try:
        init_h = np.asarray(initial_h_values, dtype=float)
        n = init_h.size
    except Exception:  # initial_h_values not provided
        n = 70                     # a modest size, fast to evaluate
        init_h = None

    dx = 2.0 / n
    target_sum = n / 2.0           # because integral must be 1

    if init_h is None:
        # random feasible start
        h = rng.random(n)
        h = np.clip(h, 0.0, 1.0)
        h = _project_sum_box(h, target_sum)
    else:
        h = init_h.copy()
        # safety: enforce constraints (might be already satisfied)
        h = np.clip(h, 0.0, 1.0)
        h = _project_sum_box(h, target_sum)

    # ------------------------------------------------------------------ #
    # 2. Baseline evaluation
    # ------------------------------------------------------------------ #
    best_h = h.copy()
    best_score = _score(best_h, dx)
    current_score = best_score

    # ------------------------------------------------------------------ #
    # 3. Optimisation loop (projected sub‑gradient + random swaps)
    # ------------------------------------------------------------------ #
    start = time.time()
    max_time = budget_s * 0.95  # keep a safety margin
    it = 0
    no_improve = 0
    # basic step‑size schedule
    base_alpha = 0.12

    while time.time() - start < max_time:
        # -------------------------------------------------------------- #
        #   a) locate the worst shift (max overlap integral)
        # -------------------------------------------------------------- #
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        idx_max = int(corr_full.argmax())
        L = idx_max - (n - 1)          # signed shift index

        # -------------------------------------------------------------- #
        #   b) gradient of the worst‑case term (sub‑gradient)
        # -------------------------------------------------------------- #
        i = np.arange(n)
        plus = i + L
        mask_plus = (plus >= 0) & (plus < n)
        term1 = np.where(mask_plus, 1.0 - h[plus], 1.0)

        minus = i - L
        mask_minus = (minus >= 0) & (minus < n)
        term2 = np.where(mask_minus, h[minus], 0.0)

        grad = term1 - term2          # ∂O_L/∂h_i

        # -------------------------------------------------------------- #
        #   c) gradient step (moving opposite to the sub‑gradient)
        # -------------------------------------------------------------- #
        alpha = base_alpha / (1.0 + it * 0.0005)
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _project_sum_box(h_new, target_sum)

        # -------------------------------------------------------------- #
        #   d) evaluate
        # -------------------------------------------------------------- #
        new_score = _score(h_new, dx)

        # accept if it does not worsen the current state too much
        # (allow a tiny increase to escape shallow plateaus)
        if new_score <= current_score * 1.001:
            h = h_new
            current_score = new_score

        # store globally best solution
        if new_score + 1e-12 < best_score:
            best_score = new_score
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # -------------------------------------------------------------- #
        #   e) occasional random swaps / perturbations
        # -------------------------------------------------------------- #
        if it % 200 == 0:
            # small random exchange of mass between two positions
            i1, i2 = rng.integers(0, n, size=2, endpoint=False)
            eps = rng.uniform(-0.04, 0.04)
            cand = h.copy()
            if 0.0 <= cand[i1] + eps <= 1.0 and 0.0 <= cand[i2] - eps <= 1.0:
                cand[i1] += eps
                cand[i2] -= eps
                # projection only needed for rounding errors
                cand = _project_sum_box(cand, target_sum)
                cand_score = _score(cand, dx)
                if cand_score < best_score:
                    best_score = cand_score
                    best_h = cand.copy()
                if cand_score < current_score:
                    h = cand
                    current_score = cand_score

        # -------------------------------------------------------------- #
        #   f) restart from best with a fresh jitter if stuck too long
        # -------------------------------------------------------------- #
        if no_improve > 6000:
            jitter = rng.normal(scale=0.06, size=n)
            h = best_h + jitter
            h = np.clip(h, 0.0, 1.0)
            h = _project_sum_box(h, target_sum)
            current_score = _score(h, dx)
            no_improve = 0

        it += 1

        # early stop if we already beat the target record comfortably
        if best_score < 0.3808:
            break

    # ------------------------------------------------------------------ #
    # 4. Return result
    # ------------------------------------------------------------------ #
    # Ensure final vector satisfies constraints exactly (tiny numeric drift)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_sum_box(best_h, target_sum)

    final_score = _score(best_h, dx)
    return best_h, float(final_score), int(n)
