import time, random, math
import numpy as np

def _project_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] with exact sum = target."""
    n = v.shape[0]
    lower, upper = 0.0, 1.0
    if target <= n * lower:
        return np.full(n, lower, dtype=v.dtype)
    if target >= n * upper:
        return np.full(n, upper, dtype=v.dtype)

    lo = np.min(v - upper)          # all entries become upper when λ ≤ lo
    hi = np.max(v - lower)          # all entries become lower when λ ≥ hi
    for _ in range(60):            # binary search for λ
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, lower, upper)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, lower, upper)


def _c5_fft(h: np.ndarray) -> float:
    """C5 value using FFT based linear correlation."""
    n = h.shape[0]
    g = 1.0 - h
    L = 2 * n
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(g[::-1], n=L)          # reverse g for correlation
    conv = np.fft.irfft(H * G, n=L)
    corr = conv[: 2 * n - 1]                # linear (full) correlation
    return np.max(corr) * (2.0 / n)


def _gradient_worst_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Gradient of the overlap for the worst lag."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if lag >= 0:
        i = np.arange(n - lag)
        g[i] += 1.0 - h[i + lag]
        g[i + lag] -= h[i]
    else:
        d = -lag
        i = np.arange(n - d)
        g[i + d] += 1.0 - h[i]
        g[i] -= h[i + d]
    return g


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function giving a small C₅ bound."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # number of samples – a power of two helps FFT speed
    N = 256
    target_sum = N / 2.0

    # initialise from a parent construction if available, otherwise random
    if "initial_h_values" in globals():
        try:
            init = np.asarray(initial_h_values, dtype=float)
            m = init.shape[0]
            # linear interpolation to the desired resolution
            old_x = np.linspace(0.0, 2.0, m, endpoint=False)
            new_x = np.linspace(0.0, 2.0, N, endpoint=False)
            h = np.interp(new_x, old_x, init)
        except Exception:
            h = np.random.rand(N)
    else:
        h = np.random.rand(N)

    h = _project_capped_simplex(h, target_sum)

    best_h = h.copy()
    best_c5 = _c5_fft(h)

    cur_h = h.copy()
    cur_c5 = best_c5

    max_iters = 500_000
    step_start, step_end = 0.15, 0.001

    for it in range(max_iters):
        if time.time() - start > budget_s * 0.94:
            break

        # correlation to identify worst lag
        n = cur_h.shape[0]
        g = 1.0 - cur_h
        L = 2 * n
        H = np.fft.rfft(cur_h, n=L)
        G = np.fft.rfft(g[::-1], n=L)
        conv = np.fft.irfft(H * G, n=L)
        corr = conv[: 2 * n - 1]
        worst_idx = int(np.argmax(corr))
        lag = worst_idx - (n - 1)

        # gradient step aimed at reducing the worst overlap
        step = step_start - (step_start - step_end) * (it / max_iters)
        grad = _gradient_worst_lag(cur_h, lag)
        trial = cur_h - step * grad
        trial = _project_capped_simplex(trial, target_sum)

        trial_c5 = _c5_fft(trial)

        if trial_c5 <= cur_c5:
            cur_h, cur_c5 = trial, trial_c5
            if trial_c5 < best_c5:
                best_h, best_c5 = trial.copy(), trial_c5

        # occasional jitter to escape shallow local minima
        if it % 1000 == 0 and it:
            jitter = (np.random.rand(N) - 0.5) * 0.02
            noisy = cur_h + jitter
            noisy = _project_capped_simplex(noisy, target_sum)
            noisy_c5 = _c5_fft(noisy)
            if noisy_c5 < best_c5:
                best_h, best_c5 = noisy.copy(), noisy_c5
                cur_h, cur_c5 = noisy, noisy_c5

    # final tidy‑up
    best_h = _project_capped_simplex(best_h, target_sum)
    final_c5 = _c5_fft(best_h)
    return best_h, final_c5, N
