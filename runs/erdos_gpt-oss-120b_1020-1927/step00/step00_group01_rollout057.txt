def worst_lag_and_gradient(h):
    n = h.shape[0]
    # compute correlation using FFT
    rev = (1 - h)[::-1]
    m = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:2*n-1]
    idx = int(np.argmax(conv))
    lag = idx - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        # positive lag: shift right
        g[:n - lag] += (1 - h[lag:])
        g[lag:] -= h[:n - lag]
    else:
        l = -lag
        g[l:] += (1 - h[:n - l])
        g[:n - l] -= h[l:]
    return lag, g, conv.max()
```

def project_to_cap(v, total, cap=1.0, tol=1e-10):
    # binary search for theta
    lo = -cap*2
    hi = cap*2
    for _ in range(60):
        theta = (lo + hi) / 2.0
        x = np.clip(v - theta, 0.0, cap)
        s = x.sum()
        if s > total:
            lo = theta
        else:
            hi = theta
    x = np.clip(v - (lo+hi)/2.0, 0.0, cap)
    # adjust any tiny sum error
    diff = total - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff proportionally to entries not at bounds
        mask = (x > 0) & (x < cap)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0, cap)
        else:
            # fallback: adjust a random index not at bound
            idxs = np.where(x < cap)[0]
            if idxs.size > 0:
                x[idxs[0]] = min(cap, max(0, x[idxs[0]] + diff))
    return x

import numpy as np
import time
import random
import math


def project_to_cap(v, total, cap=1.0, tol=1e-12):
    """Project v onto the capped simplex {0≤x≤cap, sum x = total}."""
    lo, hi = -cap * 2.0, cap * 2.0
    for _ in range(60):
        th = (lo + hi) / 2.0
        x = np.clip(v - th, 0.0, cap)
        s = x.sum()
        if s > total:
            lo = th
        else:
            hi = th
    x = np.clip(v - (lo + hi) / 2.0, 0.0, cap)
    diff = total - x.sum()
    if abs(diff) > tol:
        mask = (x > 0.0) & (x < cap)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, cap)
        else:
            # fallback to a single adjustable entry
            idx = np.where(x < cap)[0]
            if idx.size:
                i = idx[0]
                x[i] = min(cap, max(0.0, x[i] + diff))
    return x


def compute_c5(h, dx):
    """Return the overlap bound C5 for the step function described by h."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    m = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[: 2 * n - 1]
    return conv.max() * dx


def worst_lag_and_gradient(h):
    """
    Return (lag, gradient, max_corr) for the current h.
    lag is integer shift (negative allowed) where correlation is maximal.
    gradient is the sub‑gradient of the maximal correlation w.r.t. h.
    """
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    m = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    lag = idx - (n - 1)

    g = np.zeros(n, dtype=np.float64)
    if lag >= 0:
        # positive lag
        g[: n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[: n - lag]
    else:
        l = -lag
        g[l:] += (1.0 - h[: n - l])
        g[: n - l] -= h[l:]
    return lag, g, conv.max()


def random_binary(n):
    """Return a random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def local_search(h_init, n, dx, start_time, time_budget):
    """Simple simulated‑annealing / sub‑gradient descent on the given dimension."""
    h = h_init.copy()
    cur_val = compute_c5(h, dx)
    best_h = h.copy()
    best_val = cur_val

    temp_start = 0.012
    temp_end = 0.0005
    no_improve = 0
    last_best = best_val

    while time.time() - start_time < time_budget:
        # sub‑gradient for the currently worst lag
        lag, grad, _ = worst_lag_and_gradient(h)

        pos = np.where(grad > 1e-12)[0]
        neg = np.where(grad < -1e-12)[0]

        deterministic = False
        move_type = None
        delta = 0.0
        i = j = -1

        if pos.size and neg.size:
            i = pos[np.argmax(grad[pos])]
            j = neg[np.argmin(grad[neg])]
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max > 1e-12:
                delta = 0.5 * delta_max
                h[i] -= delta
                h[j] += delta
                deterministic = True
                move_type = "det"
        if not deterministic:
            i, j = np.random.choice(n, 2, replace=False)
            delta = (np.random.rand() * 2.0 - 1.0) * 0.02
            lo = -min(h[i], 1.0 - h[j])
            hi = min(1.0 - h[i], h[j])
            delta = np.clip(delta, lo, hi)
            if abs(delta) < 1e-14:
                continue
            h[i] += delta
            h[j] -= delta
            move_type = "rand"

        new_val = compute_c5(h, dx)

        # temperature schedule (linear in elapsed fraction)
        elapsed = time.time() - start_time
        prog = elapsed / time_budget
        T = temp_start * (1.0 - prog) + temp_end * prog

        accept = False
        if new_val <= cur_val:
            accept = True
        else:
            if T > 0.0:
                prob = math.exp(-(new_val - cur_val) / T)
                if random.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h.copy()
                no_improve = 0
                last_best = best_val
        else:
            # revert the move
            if move_type == "det":
                h[i] += delta
                h[j] -= delta
            else:
                h[i] -= delta
                h[j] += delta

        # occasional restart if stuck
        if best_val == last_best:
            no_improve += 1
        else:
            no_improve = 0
            last_best = best_val

        if no_improve >= 50000:
            # restart from a fresh random binary configuration
            h = random_binary(n)
            cur_val = compute_c5(h, dx)
            no_improve = 0

    return best_h, best_val

    """
    np.random.seed(seed)

    start = time.time()
    max_time = budget_s * 0.95
