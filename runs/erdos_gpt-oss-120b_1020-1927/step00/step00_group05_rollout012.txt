def project_capped_simplex(v, target):

    lower = 0.0
    upper = 1.0
    # ensure target feasible: n*lower <= target <= n*upper

def project_capped_simplex(v, target):
    # project onto [0,1] and sum target
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

def gradient_subgrad(h):
    n = h.size
    # compute raw correlation
    corr_raw = np.correlate(h, 1 - h, mode='full')
    max_val = corr_raw.max()
    active_idx = np.where(corr_raw == max_val)[0]  # all indices at max
    # compute gradient sum
    grad = np.zeros(n)
    for idx in active_idx:
        L = idx - (n - 1)
        if L >= 0:
            ov = n - L
            grad[:ov] += (1 - h[L:])
            grad[L:] += -h[:ov]
        else:

            ov = n - Lp
            grad[Lp:] += (1 - h[:ov])
            grad[:ov] += -h[Lp:]
    # average
    grad /= len(active_idx)
    # multiply by dx
    dx = 2.0 / n
    grad *= dx
    return grad

import numpy as np
import time
import math

def _project_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {x | 0<=x<=1, sum(x)=target}."""
    lo = v.min() - 1.0          # all entries become 1 after subtracting lo
    hi = v.max()                # all entries become 0 after subtracting hi
    for _ in range(60):        # enough for double precision
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)


def _subgradient(h: np.ndarray) -> np.ndarray:
    """Subgradient of max_k ∫ h(x)(1-h(x+k))dx for the current h."""
    n = h.size
    # raw correlation (no dx factor, same lag ordering as np.correlate)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = corr.max()
    eps = 1e-12
    active = np.where(corr >= max_val - eps)[0]

    grad = np.zeros_like(h)
    for idx in active:
        L = idx - (n - 1)          # lag
        if L >= 0:
            ov = n - L
            grad[:ov] += (1.0 - h[L:])
            grad[L:]  += -h[:ov]
        else:
            Lp = -L
            ov = n - Lp
            grad[Lp:] += (1.0 - h[:ov])
            grad[:ov] += -h[Lp:]
    grad /= len(active)            # average over active lags
    grad *= (2.0 / n)              # dx factor
    return grad


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---------- initialise problem size ----------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = h0.size
        target_sum = n / 2.0
        h0 = _project_capped_simplex(h0, target_sum)
    else:
        n = 200                     # fallback size, still <1k
        target_sum = n / 2.0
        rnd = rng.random(n)
        h0 = _project_capped_simplex(rnd, target_sum)

    # ---------- two parallel search threads ----------
    # 1) continuous (fractional) vector
    h_frac = h0.copy()
    c5_frac = evaluate_erdos_solution(h_frac)

    # 2) binary vector (exactly target_sum ones, n even)
    half = int(round(target_sum))
    perm = rng.permutation(n)
    h_bin = np.zeros(n, dtype=float)
    h_bin[perm[:half]] = 1.0
    c5_bin = evaluate_erdos_solution(h_bin)

    # ---------- best seen ----------
    best_h = h_frac.copy()
    best_c5 = c5_frac
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    iter_cnt = 0

    # ---------- helper for binary swaps ----------
    def _indices_one_zero(arr):
        ones = np.where(arr > 0.5)[0]
        zeros = np.where(arr <= 0.5)[0]
        return ones, zeros

    # ---------- main optimisation loop ----------
    while time.time() - start < budget_s * 0.95:
        if rng.random() < 0.5:
            # ----- continuous subgradient step -----
            grad = _subgradient(h_frac)
            step = 1.0 / math.sqrt(iter_cnt + 1)           # diminishing steps
            cand = h_frac - step * grad
            cand = _project_capped_simplex(cand, target_sum)
            c5_cand = evaluate_erdos_solution(cand)

            if c5_cand < c5_frac - 1e-12:
                h_frac, c5_frac = cand, c5_cand
                if c5_cand < best_c5:
                    best_c5, best_h = c5_cand, cand.copy()
            else:
                # occasional uphill move (simulated‑annealing style)
                temp = 0.02 / math.sqrt(iter_cnt + 1)
                if rng.random() < math.exp(-(c5_cand - c5_frac) / max(temp, 1e-12)):
                    h_frac, c5_frac = cand, c5_cand
        else:
            # ----- binary swap move -----
            ones, zeros = _indices_one_zero(h_bin)
            if ones.size == 0 or zeros.size == 0:
                continue
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cand = h_bin.copy()
            cand[i] = 0.0
            cand[j] = 1.0
            c5_cand = evaluate_erdos_solution(cand)

            if c5_cand < c5_bin - 1e-12:
                h_bin, c5_bin = cand, c5_cand
                if c5_cand < best_c5:
                    best_c5, best_h = c5_cand, cand.copy()
            else:
                temp = 0.015 / math.sqrt(iter_cnt + 1)
                if rng.random() < math.exp(-(c5_cand - c5_bin) / max(temp, 1e-12)):
                    h_bin, c5_bin = cand, c5_cand

        # ----- occasional random perturbation of the current best -----
        if iter_cnt % 500 == 0:
            pert = best_h + rng.normal(scale=0.02, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            pert = _project_capped_simplex(pert, target_sum)
            c5_pert = evaluate_erdos_solution(pert)
            if c5_pert < best_c5:
                best_c5, best_h = c5_pert, pert.copy()
                # also feed back to the continuous thread
                h_frac, c5_frac = pert.copy(), c5_pert

        iter_cnt += 1

    # final projection to guarantee feasibility and final evaluation
    best_h = _project_capped_simplex(best_h, target_sum)
    best_c5 = evaluate_erdos_solution(best_h)

    return best_h, float(best_c5), n
