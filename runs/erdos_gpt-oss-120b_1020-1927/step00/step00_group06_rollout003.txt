        else:
            excess = s - target_sum
            idx = np.where(v > 0.0)[0]
            while excess > 1e-12 and idx.size:
                dec = excess / idx.size
                slack = v[idx]
                dec_limited = np.minimum(dec, slack)
                v[idx] -= dec_limited
                excess -= dec_limited.sum()
                idx = idx[v[idx] > 1e-12]
        return v

    h = repair(h)

    # ------------------------------------------------------------
    # Objective computation (same as evaluate_erdos_solution)
    # ------------------------------------------------------------
    def compute_c5(vec):
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        return np.max(corr) * dx

    # ------------------------------------------------------------
    # Initial best
    # ------------------------------------------------------------
    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Current state for simulated annealing
    cur_h = best_h.copy()
    cur_score = best_score

    # ------------------------------------------------------------
    # Simulated annealing parameters
    # ------------------------------------------------------------
    max_temp = 0.20          # start temperature
    min_temp = 1e-6          # end temperature
    max_step = 0.30          # maximal transfer amount at start
    min_step = 0.001         # minimal transfer amount at the end
    no_improve = 0           # counter for stagnation
    last_repair = 0          # iteration of last global repair
    start_time = time.time()
    it = 0

    # ------------------------------------------------------------
    # Main loop
    # ------------------------------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break

        it += 1
        frac = elapsed / (budget_s * 0.95)          # 0 → 1 over the allowed time
        temp = max_temp * (1.0 - frac) + min_temp * frac
        step = max_step * (1.0 - frac) + min_step * frac

        # ---- propose a move (mass transfer between two indices) ----
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # allowed delta range respecting bounds and step size
        delta_max = min(step, cur_h[i], 1.0 - cur_h[j])
        delta_min = -min(step, cur_h[j], 1.0 - cur_h[i])
        if delta_max < delta_min:
            continue
        delta = np.random.uniform(delta_min, delta_max)

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # no need for repair here – sum is preserved and bounds respected by construction
        new_score = compute_c5(new_h)

        # ---- acceptance ----
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / (temp + 1e-12)):
            cur_h = new_h
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
        else:
            no_improve += 1

        # ---- occasional global repair to counter floating‑point drift ----
        if it - last_repair >= 5000:
            cur_h = repair(cur_h)
            cur_score = compute_c5(cur_h)
            best_h = repair(best_h)
            best_score = compute_c5(best_h)
            last_repair = it
            no_improve = 0

        # ---- restart if stuck for a long time ----
        if no_improve > 200000:
            # initialise a fresh candidate near the current best
            cur_h = best_h + np.random.normal(scale=step, size=n)
            cur_h = repair(cur_h)
            cur_score = compute_c5(cur_h)
            no_improve = 0

    # ------------------------------------------------------------
    # Finalisation
    # ------------------------------------------------------------
    best_h = repair(best_h)
    best_score = compute_c5(best_h)

    return best_h.astype(np.float32), float(best_score), n
