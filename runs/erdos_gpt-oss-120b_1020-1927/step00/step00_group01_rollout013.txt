def project_to_box_and_sum(y, target_sum):
    # Solve min ||x - y||^2 s.t. 0 <= x <= 1, sum(x) = target_sum.
    # Use water-filling.
    # Approach: binary search for λ such that sum(clip(y - λ, 0, 1)) = target_sum.
    # Since sum is decreasing in λ.
    lo = -1.0  # λ low bound
    hi = 1.0   # λ high bound
    # Actually λ may exceed range of y; we need extended bounds.
    # We can set lo = min(y)-1, hi = max(y)+1.
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo+hi)/2.0, 0.0, 1.0)
    # after clipping may not exactly meet sum because of rounding. Slight adjust: adjust interior points a bit.
    # We can compute residual and distribute among points not at bounds.
    residual = target_sum - x.sum()
    if abs(residual) > 1e-10:
        # Distribute residual evenly among indices with 0< x <1
        mask = (x > 1e-12) & (x < 1-1e-12)
        if mask.any():
            add = residual / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)  # reclip
        else:
            # if no interior points, adjust all (should not happen for reasonable parameters)
            x += residual / n
            x = np.clip(x, 0.0, 1.0)
    return x

def project_to_box_and_sum(v, target):
    # Works for v as numpy array. Returns new array.
    # water-filling with Lagrange multiplier.
    # Seek λ such that sum(clip(v - λ, 0, 1)) = target.
    lo = np.min(v) - 1.0  # lower bound
    hi = np.max(v) + 1.0  # high bound
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo+hi)*0.5, 0.0, 1.0)
    # adjust rounding errors:
    # If sum deviates slight from target due to floating error, adjust inside range
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        # add diff to interior points proportionally
        interior = (x > 0) & (x < 1)
        count = interior.sum()
        if count > 0:
            x[interior] += diff / count
            x = np.clip(x, 0.0, 1.0)
        else:
            # if no interior points (unlikely), distribute uniformly among all
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x

def project_to_constraints(v, target):
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo+hi)*0.5, 0.0, 1.0)
    # adjust: may not match exactly due to floating rounding
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        interior = (x > 1e-12) & (x < 1-1e-12)
        cnt = interior.sum()
        if cnt > 0:

    return x

def max_corr_and_lag(h, dx):
    corr = np.correlate(h, 1-h, mode='full') * dx
    max_idx = np.argmax(corr)
    return corr[max_idx], max_idx

```
def grad_of_lag(h, lag, dx):
    n = len(h)
    g = np.zeros_like(h)
    if lag >= 0:
        i = np.arange(0, n-lag)
        j = i + lag
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:
        i = np.arange(-lag, n)
        j = i + lag
        g[i] += 1 - h[j]
        g[j] -= h[i]
    return g * dx
```

import time
import math

# ---------- projection onto the box [0,1] and the hyperplane sum = target ----------
def _project_box_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto {x: 0<=x<=1, sum(x)=target_sum} using a water‑filling search."""
    lo = np.min(v) - 1.0          # makes all entries clipped to 1
    hi = np.max(v) + 1.0          # makes all entries clipped to 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for rounding errors
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        interior = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = interior.sum()
        if cnt:
            x[interior] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback – distribute uniformly (should rarely happen)
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x


# ---------- correlation utilities ----------
def _max_corr_and_lag(h: np.ndarray, dx: float):
    """Return (max value, lag index) of the full correlation of h with (1‑h)."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx


def _grad_of_lag(h: np.ndarray, lag: int, dx: float):
    """Gradient of the lag‑th term C(L) = ∫ h(x)(1‑h(x+L))dx."""
    n = h.size
    g = np.zeros_like(h)
    if lag >= 0:
        i = np.arange(0, n - lag)
        j = i + lag
    else:                     # lag < 0
        i = np.arange(-lag, n)
        j = i + lag
    # derivative w.r.t. h[i]
    g[i] += 1.0 - h[j]
    # derivative w.r.t. h[j]
    g[j] -= h[i]
    return g * dx


# ---------- core optimizer ----------
def _optimize_from(start_h: np.ndarray, target_sum: float,
                   max_iters: int, start_time: float, budget: float):
    """Projected sub‑gradient descent for a single start point."""
    h = start_h.copy()
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_val = np.max(np.correlate(h, 1.0 - h, mode="full") * dx)

    for it in range(1, max_iters + 1):
        # time guard
        if time.time() - start_time > budget * 0.93:
            break

        # current worst lag
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        max_idx = int(np.argmax(corr))
        cur_val = float(corr[max_idx])
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        lag = max_idx - (n - 1)
        g = _grad_of_lag(h, lag, dx)

        # diminishing step size
        step = 0.30 / math.sqrt(it)
        y = h - step * g
        h = _project_box_sum(y, target_sum)

    return best_h, best_val

            break
        cand_h, cand_val = _optimize_from(
            s, target_sum, max_iters_per_start, start_time, budget_s
        )
        if cand_val < best_overall_val:
            best_overall_val = cand_val
            best_overall_h = cand_h.copy()

    # final safety projection and evaluation
    if best_overall_h is None:
        best_overall_h = np.full(n, 0.5)
    best_overall_h = _project_box_sum(best_overall_h, target_sum)
    final_c5 = float(np.max(np.correlate(best_overall_h,
                                        1.0 - best_overall_h,
                                        mode="full") * dx))

    # return plain Python list as required by the harness
    return best_overall_h.tolist(), final_c5, n
