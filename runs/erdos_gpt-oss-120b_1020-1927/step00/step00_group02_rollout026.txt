import numpy as np, time, math

def compute_c5(h, dx):
    # max overlap integral
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_onto_capped_simplex(v, target_sum, ub=1.0, max_iter=100):
    # Project v onto the set {x : 0 <= x_i <= ub, sum x_i = target_sum}
    # Uses bisection on lambda.
    # Ensure feasible: 0 <= target_sum <= ub * len(v)
    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, ub)
        s = x.sum()
        if s < target_sum:
            hi = lam
        else:
            lo = lam
        if abs(s - target_sum) < 1e-12:
            break
    # final projection
    x = np.clip(v - lam, 0.0, ub)
    # due to numerical errors, may need small adjustment
    diff = target_sum - x.sum()
    if abs(diff) > 1e-10:
        # distribute diff over non-bound elements
        mask = (x > 1e-12) & (x < ub - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, ub)
    return x

def random_feasible_v(rng, m, target_sum):

    v0 = rng.rand(m)
    v = project_onto_capped_simplex(v0, target_sum, ub=1.0)

    h = best_h.copy()
    cur_c5 = best_c5
    best_h_fine = h.copy()
    best_c5_fine = cur_c5

    max_iter_fine = 300_000
    T0_f, T_min_f = 0.005, 1e-6
    step_max_f, step_min_f = 0.05, 5e-4
    last_imp_f = 0

    for itf in range(max_iter_fine):
        if time.time() - start > budget_s * 0.99:
            break
        T = T0_f * (1 - itf / max_iter_fine) + T_min_f
        step = step_max_f * (1 - itf / max_iter_fine) + step_min_f

        i, j = rng.randint(n), rng.randint(n - 1)
        if j >= i:
            j += 1
        hi, hj = h[i], h[j]
        lower = max(-hi, hj - 1.0)
        upper = min(1.0 - hi, hj)
        lower = max(lower, -step)
        upper = min(upper, step)
        if lower >= upper:
            continue
        delta = rng.uniform(lower, upper)
        h[i] = hi + delta
        h[j] = hj - delta
        new_c5 = compute_c5(h, dx)
        dE = new_c5 - cur_c5
        if dE < 0 or rng.rand() < math.exp(-dE / T):
            cur_c5 = new_c5
            if new_c5 < best_c5_fine:
                best_c5_fine = new_c5
                best_h_fine = h.copy()
                last_imp_f = itf
        else:
            h[i], h[j] = hi, hj

        if itf - last_imp_f > 50_000:
            # small random jitter around current best to escape flats
            h = best_h_fine.copy()
            for _ in range(10):
                a, b = rng.randint(n), rng.randint(n - 1)
                if b >= a:
                    b += 1
                ha, hb = h[a], h[b]
                eps = 0.02
                low = max(-ha, hb - 1.0)
                up = min(1.0 - ha, hb)
                low = max(low, -eps)
                up = min(up, eps)
                if low < up:
                    d = rng.uniform(low, up)
                    h[a] = ha + d
                    h[b] = hb - d
            cur_c5 = compute_c5(h, dx)

    # -------------------------------------------------------------
    # Choose the best result obtained
    if best_c5_fine < best_c5:
        final_h = best_h_fine
        final_c5 = best_c5_fine
    else:
        final_h = best_h
        final_c5 = best_c5

    # Adjust for tiny integral mismatch (should be negligible)
    desired_sum = n / 2.0
    diff = desired_sum - final_h.sum()
    if abs(diff) > 1e-9:
        for idx in range(n):
            if diff > 0 and final_h[idx] < 1.0:
                add = min(diff, 1.0 - final_h[idx])
                final_h[idx] += add
                diff -= add
            elif diff < 0 and final_h[idx] > 0.0:
                sub = min(-diff, final_h[idx])
                final_h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
        final_c5 = compute_c5(final_h, dx)

    return final_h.tolist(), float(final_c5), n
