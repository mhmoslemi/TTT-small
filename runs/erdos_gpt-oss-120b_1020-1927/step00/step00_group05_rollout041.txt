def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    ov = np.correlate(h, 1 - h, mode='full')

def shifted(arr, shift):
    n = arr.shape[0]
    if shift >= 0:
        if shift >= n:
            return np.zeros_like(arr)
        out = np.empty_like(arr)
        out[:n-shift] = arr[shift:]
        out[n-shift:] = 0.0
        return out
    else:
        shift = -shift
        if shift >= n:
            return np.zeros_like(arr)
        out = np.empty_like(arr)
        out[shift:] = arr[:n-shift]
        out[:shift] = 0.0
        return out

def project_sum_bounds(x, target_sum):
    # Clip to bounds first.
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if abs(s - target_sum) < 1e-12:
        return x
    # Iterative redistribution (water-filling)
    for _ in range(20):
        diff = target_sum - s
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        x[free] += delta
        # Clip again
        x = np.clip(x, 0.0, 1.0)
        s = x.sum()
        if abs(s - target_sum) < 1e-12:
            break
    # In rare case of tiny leftover, adjust a random free element.
    if abs(x.sum() - target_sum) > 1e-9:
        diff = target_sum - x.sum()
        # distribute to any index within bounds
        idx = np.random.choice(len(x))
        x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)

import time
import math
import numpy as np


def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated by correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    ov = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(ov) * dx)


def shifted(arr: np.ndarray, s: int) -> np.ndarray:
    """Shift arr by s positions, filling emptied entries with 0 (no wrap)."""
    n = arr.shape[0]
    if s >= 0:
        if s >= n:
            return np.zeros_like(arr)
        out = np.empty_like(arr)
        out[: n - s] = arr[s:]
        out[n - s :] = 0.0
        return out
    else:
        s = -s
        if s >= n:
            return np.zeros_like(arr)
        out = np.empty_like(arr)
        out[s:] = arr[: n - s]
        out[:s] = 0.0
        return out


def project_sum_bounds(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the box [0,1]^n while keeping the sum equal to target."""
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if abs(s - target) < 1e-12:
        return x
    # water‑filling style redistribution
    for _ in range(20):
        diff = target - s
        free = (x > 0.0) & (x < 1.0)
        cnt = int(free.sum())
        if cnt == 0:
            break
        delta = diff / cnt
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) < 1e-12:
            break
    # tiny correction if needed
    if abs(x.sum() - target) > 1e-9:
        diff = target - x.sum()
        idx = np.random.randint(0, x.size)
        x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x
