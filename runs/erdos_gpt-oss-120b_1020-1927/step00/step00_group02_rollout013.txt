import time
import numpy as np

def _project_onto_bounds_sum(v, target_sum, eps=1e-12):
    """Project vector v onto the set {0<=x_i<=1 , sum x_i = target_sum}."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _full_corr(h):
    """Cross‑correlation of h with (1‑h) using zero padding (full mode)."""
    n = h.shape[0]
    L = 2 * n - 1
    pad_h = np.concatenate([h, np.zeros(n - 1, dtype=h.dtype)])
    pad_g = np.concatenate([1.0 - h, np.zeros(n - 1, dtype=h.dtype)])
    H = np.fft.rfft(pad_h)
    G = np.fft.rfft(pad_g)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    return corr.real


def _c5_of(h):
    """Return C5 for a given discretised h."""
    n = h.shape[0]
    dx = 2.0 / n
    return _full_corr(h).max() * dx
