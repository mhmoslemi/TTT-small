import numpy as np, time, random, math

def build_shape(p, n):
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    raw = 1.0 - np.abs(x - 1.0) ** p
    # ensure non-negative (should be)
    raw = np.maximum(raw, 0.0)
    scale = (n / 2.0) / raw.sum()
    h = raw * scale
    np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def search_exponent(n, p_min, p_max, steps):
    best_h = None
    best_c5 = np.inf
    best_p = None
    for p in np.linspace(p_min, p_max, steps):
        h = build_shape(p, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_p = p
    # refine around best_p
    if best_p is not None:
        low = max(p_min, best_p - (p_max - p_min) / steps)
        high = min(p_max, best_p + (p_max - p_min) / steps)
        for p in np.linspace(low, high, steps):
            h = build_shape(p, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                best_p = p
    return best_h, best_c5, best_p

def hill_climb(initial_h, time_limit):
    n = len(initial_h)
    dx = 2.0 / n
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    current_h = best_h.copy()
    current_c5 = best_c5
    step = 0.05  # max adjustment per move
    start = time.time()
    while time.time() - start < time_limit:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        hi, hj = current_h[i], current_h[j]
        lower = max(-hi, hj - 1.0)
        upper = min(1.0 - hi, hj)
        if upper <= lower:
            continue
        # random delta within allowed interval, limited by step size
        delta = (upper - lower) * np.random.rand() + lower
        # clip to step size
        if delta > step:
            delta = step
        elif delta < -step:
            delta = -step
        new_h = current_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        new_c5 = compute_c5(new_h)
        if new_c5 < current_c5:
            current_h, current_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5
        # optionally shrink step size over time
        if (time.time() - start) > time_limit * 0.8:
            step = max(step * 0.9, 1e-3)
    return best_h, best_c5

import time
import random
import numpy as np


def build_shape(p: float, n: int) -> np.ndarray:
    """Create a symmetric step function h on [0,2] using exponent p."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx               # mid‑point sampling
    raw = 1.0 - np.abs(x - 1.0) ** p
    raw = np.maximum(raw, 0.0)                  # safety
    scale = (n / 2.0) / raw.sum()               # enforce ∑h = n/2
    h = raw * scale
    np.clip(h, 0.0, 1.0, out=h)
    return h


def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx for the discrete vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def search_exponent(n: int, p_min: float, p_max: float, steps: int):
    """Coarse + fine search over exponent p."""
    best_h = None
    best_c5 = np.inf
    best_p = None
    # coarse grid
    for p in np.linspace(p_min, p_max, steps):
        h = build_shape(p, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h, best_p = c5, h, p
    # local refinement around the best p
    if best_p is not None:
        delta = (p_max - p_min) / steps
        lo = max(p_min, best_p - delta)
        hi = min(p_max, best_p + delta)
        for p in np.linspace(lo, hi, steps):
            h = build_shape(p, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h, best_p = c5, h, p
    return best_h, best_c5, best_p


def hill_climb(start_h: np.ndarray, time_limit: float):

    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    step = 0.05                       # maximal absolute change per move
    start = time.time()
    while time.time() - start < time_limit:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        hi, hj = cur_h[i], cur_h[j]
        lower = max(-hi, hj - 1.0)          # keep both entries in [0,1]
        upper = min(1.0 - hi, hj)
        if upper <= lower:
            continue
        delta = (upper - lower) * np.random.rand() + lower
        # enforce step size
        if delta > step:
            delta = step
        elif delta < -step:
            delta = -step
        new_h = cur_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5:
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5
        # slowly shrink step as we approach the time limit
        if (time.time() - start) > 0.8 * time_limit:
            step = max(step * 0.9, 1e-3)
    return best_h, best_c5


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of points."""
    random.seed(seed)
    np.random.seed(seed)

    # Desired number of sample points (even for exact mass constraint)
    n = 200
    best_h = None
    best_c5 = np.inf

    # If a parent construction is present, evaluate it first
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        # Project onto the feasible affine subspace if needed
        target_sum = h0.size / 2.0
        cur_sum = h0.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            scale = target_sum / cur_sum
            h0 = np.clip(h0 * scale, 0.0, 1.0)
        best_h, best_c5 = h0, compute_c5(h0)

    # --- 1. Exponent family search ---------------------------------
    start = time.time()
    remaining = budget_s - (time.time() - start) - 5.0   # keep margin for later phases
    if remaining > 0.0:
        h_exp, c_exp, _ = search_exponent(n, 1.0, 2.5, 151)
        if c_exp < best_c5:
            best_h, best_c5 = h_exp, c_exp

    # --- 2. Local refinement via hill climbing --------------------
    remaining = max(0.0, budget_s - (time.time() - start) - 1.0)
    if remaining > 0.1 and best_h is not None:
        h_opt, c_opt = hill_climb(best_h, remaining)
        if c_opt < best_c5:
            best_h, best_c5 = h_opt, c_opt

    # --- 3. Final sanity projection --------------------------------
    # Ensure the exact integral constraint (possible tiny drift)
    target = best_h.size / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        # adjust a random entry within the bounds
        for idx in np.random.permutation(best_h.size):
            cand = best_h[idx] + diff
            if 0.0 <= cand <= 1.0:
                best_h[idx] = cand
                break

    final_c5 = compute_c5(best_h)
    return best_h, float(final_c5), best_h.size
