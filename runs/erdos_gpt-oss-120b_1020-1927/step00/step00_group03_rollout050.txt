def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    rng = np.random.default_rng(seed)

    # ----- configuration -----
    N = 256                     # number of samples (must be even)
    total_time = budget_s * 0.96
    start_time = time.time()
    end_time = start_time + total_time

    # temperature schedule for simulated annealing
    T0 = 20.0
    T_end = 1e-3
    alpha = -math.log(T_end / T0) / total_time if total_time > 0 else 0.0

    # ----- helper: max mismatch (C5 numerator) -----
    def max_mismatch(seq):
        """Return max_k sum_i h_i (1‑h_{i+k}) for binary seq."""
        a = seq.astype(np.float64)
        b_rev = (1.0 - a)[::-1]
        size = 2 * N
        fa = np.fft.rfft(a, size)
        fb = np.fft.rfft(b_rev, size)
        conv = np.fft.irfft(fa * fb, size)
        # first 2*N‑1 entries correspond to full correlation
        return int(np.rint(conv[: 2 * N - 1].max()))

    # ----- initialise best solution -----
    best_seq = None
    best_M = N + 1  # large initial value

    # Use parent construction if available
    if "initial_h_values" in globals():

                interp = np.interp(x_new, x_old, init)
                seq = (interp > 0.5).astype(np.int8)
                # enforce exact half‑ones constraint
                diff = int(seq.sum() - N // 2)
                if diff > 0:
