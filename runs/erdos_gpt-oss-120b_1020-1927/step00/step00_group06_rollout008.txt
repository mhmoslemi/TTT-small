def gradient_for_shift(h, k):
    N = len(h)
    grad = np.zeros(N, dtype=np.float64)
    if k >= 0:
        # forward shift
        # contribution when i in [0,N-1-k] as first index: derivative of h_i term = 1 - h_{i+k}
        i_first = np.arange(0, N - k)
        grad[i_first] += 1 - h[i_first + k]
        # contribution as second index: i = i'+k, i' in [0,N-1-k]; derivative = -h_{i'}
        i_second = i_first + k
        grad[i_second] += -h[i_first]
    else:
        # negative shift: shift left; equivalently treat as positive shift on reversed?
        # For k negative, define shift = -k > 0, but derivative formulas differ:
        # The correlation sum: sum_i h_i * (1 - h_{i+k}) where k negative => i+k < i, i+k may be smaller.
        # Equivalent to sum_i h_{i - shift} * (1 - h_i) for shift = -k positive.
        # So we can treat similarly with shift = -k but swapping roles of i.
        s = -k
        # For i in [s, N-1] (since i-k >=0)
        # Let's derive similar symmetrical transposed formula:
        # Original expression: sum_i h_i * (1 - h_{i - s}) for i >= s
        # Equivalent to sum_{j=0}^{N-1-s} h_{j + s} * (1 - h_{j}) with j = i - s.
        # So for j from 0..N-1-s:
        # term = h_{j+s} * (1 - h_j)
        # derivative wrt h_{j+s} : 1 - h_j
        # derivative wrt h_j : -h_{j+s}
        # So we implement:
        j = np.arange(0, N - s)
        grad[j + s] += 1 - h[j]
        grad[j] += -h[j + s]
    return grad

def project_onto_capped_simplex(v, S):
    # as described
    # uses binary search
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(60):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > S:
            low = mid
        else:
            high = mid
    x = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    # Adjust any small sum error
    diff = S - x.sum()
    if abs(diff) > 1e-8:
        # adjust interior indices
        interior = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if interior.size > 0:
            add = diff / interior.size
            x[interior] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            # if no interior, distribute to all with capacity
            for i in range(len(x)):
                cap_up = 1 - x[i]
                cap_down = x[i]
                if diff > 0 and cap_up > 0:
                    delta = min(diff, cap_up)
                    x[i] += delta
                    diff -= delta
                elif diff < 0 and cap_down > 0:
                    delta = min(-diff, cap_down)
                    x[i] -= delta
                    diff += delta
                if abs(diff) < 1e-12:
                    break
    return x

def gradient_for_shift(h, k):

import numpy as np

def _project_capped_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project v onto {0≤x≤1, sum x = S} using a simple binary search."""
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > S:
            low = mid
        else:
            high = mid
    x = np.clip(v - (low + high) * 0.5, 0.0, 1.0)

    # tiny correction for numerical drift
    diff = S - x.sum()
    if abs(diff) > 1e-9:
        interior = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if interior.size:
            add = diff / interior.size
            x[interior] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            # distribute greedily
            for i in range(len(x)):
                cap_up = 1.0 - x[i]
                cap_down = x[i]
                if diff > 0 and cap_up > 0:
                    d = min(diff, cap_up)
                    x[i] += d
                    diff -= d
                elif diff < 0 and cap_down > 0:
                    d = min(-diff, cap_down)
                    x[i] -= d
                    diff += d
                if abs(diff) < 1e-12:
                    break
    return x


def _worst_shift(h: np.ndarray):
    N = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    k = idx - (N - 1)
    val = corr[idx]
    return k, val


def _grad_for_shift(h: np.ndarray, k: int):
    """Gradient of the overlap for the shift that attains the maximum."""
    N = h.size
    g = np.zeros(N, dtype=np.float64)
    if k >= 0:
        limit = N - k
        i = np.arange(limit)
        g[i] += 1.0 - h[i + k]
        g[i + k] += -h[i]
    else:
        s = -k
        limit = N - s
        j = np.arange(limit)
        g[j + s] += 1.0 - h[j]
        g[j] += -h[j + s]
    return g

    except NameError:
