import time
import numpy as np

def _project_to_sum_and_box(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the feasible set {0<=y<=1, sum(y)=target}."""
    lo, hi = -2.0, 2.0                     # bounds for the Lagrange multiplier
    for _ in range(60):                    # enough for double precision
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def _max_overlap(h: np.ndarray) -> tuple[float, np.ndarray]:
    """Return max_{k} ∫ h(x)(1‑h(x+k))dx using the discretised correlation."""
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max()), corr


def _shift_sum(h: np.ndarray, k: int) -> np.ndarray:
    """Vector   s_i = h_{i+k}+h_{i‑k} (out‑of‑range terms are 0)."""
    n = h.size
    if k == 0:
        return 2.0 * h
    s = np.zeros_like(h)
    s[: n - k] += h[k:]
    s[k:] += h[: n - k]
    return s


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function for the Erdős C₅ problem.

    Returns (h_values, c5_bound, n_points) where
        h_values : np.ndarray of shape (n_points,)
        c5_bound : float = max_k ∫ h(x)(1‑h(x+k))dx
        n_points : int = length of the discretisation.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------ #
    # 1.  Determine problem size and initialise a feasible starting vector
    # ------------------------------------------------------------------ #
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = h0.size
        target = n / 2.0
        h0 = _project_to_sum_and_box(h0, target)
    else:
        n = 200                     # default size if no parent vector is given
        target = n / 2.0
        h0 = _project_to_sum_and_box(np.random.rand(n), target)

    # ------------------------------------------------------------------ #
    # 2.  Helper utilities
    # ------------------------------------------------------------------ #
    def random_feasible():
        return _project_to_sum_and_box(np.random.rand(n), target)

    best_h = h0.copy()
    best_score, _ = _max_overlap(best_h)

    # ------------------------------------------------------------------ #
    # 3.  Main optimisation – projected sub‑gradient descent
    # ------------------------------------------------------------------ #
    start_time = time.time()
    max_total_iters = 20000          # generous ceiling, real stop is time‑based
    step0 = 0.8                      # initial step length
    top_k = 5                        # number of shifts used for a sub‑gradient

    # a few restarts (including the supplied initial vector)
    restart_vectors = [h0] + [random_feasible() for _ in range(5)]

    for h_start in restart_vectors:
        h = h_start.copy()
        for it in range(1, max_total_iters + 1):
            # current objective
            cur_val, corr = _max_overlap(h)
            if cur_val < best_score:
                best_score, best_h = cur_val, h.copy()

            # time check
            if time.time() - start_time > budget_s - 0.5:
                break

            # indices of the largest few correlations
            if top_k >= corr.size:
                idxs = np.arange(corr.size)
            else:
                idxs = np.argpartition(-corr, top_k - 1)[:top_k]

            # corresponding (absolute) shifts
            shifts = np.abs(idxs - (n - 1)).astype(int)

            # sub‑gradient: average of the individual shift‑sums
            g = np.zeros_like(h)
            for k in shifts:
                g += _shift_sum(h, k)
            g /= float(len(shifts))

            # step (decreasing like 1/√t)
            step = step0 / np.sqrt(it)

            # projected update
            h = _project_to_sum_and_box(h + step * g, target)

            # occasional tiny random jitter to escape plateaux
            if it % 250 == 0:
                eps = 0.01
                h = _project_to_sum_and_box(h + eps * (np.random.rand(n) - 0.5), target)

        if time.time() - start_time > budget_s - 0.5:
            break

    # ------------------------------------------------------------------ #
    # 4.  Fine‑grained local search around the best candidate found so far
    # ------------------------------------------------------------------ #
    h = best_h.copy()
    while time.time() - start_time < budget_s - 0.5:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # we only move mass from a larger entry to a smaller one
        if h[i] >= 1.0 or h[j] <= 0.0:
            continue
        max_inc = 1.0 - h[i]
        max_dec = h[j]
        delta = min(max_inc, max_dec) * 0.5 * np.random.rand()
        if delta <= 0.0:
            continue
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_val, _ = _max_overlap(h_new)
        cur_val, _ = _max_overlap(h)
        if new_val < cur_val:
            h = h_new
            if new_val < best_score:
                best_score, best_h = new_val, h_new.copy()

    return best_h, float(best_score), n
