def project_onto_box_sum(y, S, tol=1e-10, max_iter=100):
    lo = -max(y) - 1.0  # ensure s >= S
    hi = max(y) + 1.0   # ensure s <= S
    # Actually we want s decreasing in lambda
    # At large negative lambda, y_i - lambda large => clip to 1 => sum = n
    # At large positive lambda, y_i - lambda negative => clip to 0 => sum = 0
    # So we can set lo = -np.max(y) - 1, hi = np.max(y) + 1
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        clipped = np.clip(y - mid, 0, 1)
        s = clipped.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

def gradient_for_shift(h, shift, dx):

    if shift >= 0:
        # term1: 1 - h[shift:] for first n-shift entries, 1 for last shift entries
        term1 = np.empty(n)

        if shift > 0:

import time
import math

import numpy as np

# ------------------------------------------------------------
# Helper: project a vector onto the hyperplane sum=h_target
# intersected with the box [0,1]^n.
def _project_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    # binary search for lambda such that sum(clip(v - λ,0,1)) = target
    lo, hi = -1e6, 1e6
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


# ------------------------------------------------------------
# Gradient of the correlation term for a given shift.
def _grad_for_shift(h: np.ndarray, shift: int, dx: float) -> np.ndarray:
    n = h.shape[0]
    if shift >= 0:
        # (1 - h[i+shift]) part
        t1 = np.empty(n)
        t1[: n - shift] = 1.0 - h[shift:]
        if shift > 0:
            t1[n - shift :] = 1.0
        # h[i-shift] part
        t2 = np.empty(n)
        if shift > 0:
            t2[:shift] = 0.0
            t2[shift:] = h[: n - shift]
        else:                         # shift == 0
            t2[:] = h
    else:
        s = -shift
        # (1 - h[i-shift]) part
        t1 = np.empty(n)
        if s > 0:
            t1[:s] = 1.0
            t1[s:] = 1.0 - h[: n - s]
        else:
            t1[:] = 1.0 - h
        # h[i+ s] part
        t2 = np.empty(n)
        if s > 0:
            t2[: n - s] = h[s:]
            t2[n - s :] = 0.0
        else:
            t2[:] = 0.0
    return (t1 - t2) * dx

# Compute C5 for a candidate vector.
def _c5_value(h: np.ndarray) -> float:
    try:
        # the harness provides this helper; use it if available
        return evaluate_erdos_solution(h)
    except Exception:
        # fallback (identical to the official evaluator)
        dx = 2.0 / h.shape[0]
        return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


# ------------------------------------------------------------
# Core sub‑gradient descent on a single start vector.
def _descend(
    h_start: np.ndarray,
    target_sum: float,
    max_iter: int,
    best_h: np.ndarray,
    best_val: float,
    start_time: float,
    time_budget: float,
) -> tuple[np.ndarray, float]:
    h = h_start.copy()
    n = h.shape[0]
    dx = 2.0 / n
    for it in range(1, max_iter + 1):
        # time guard
        if time.time() - start_time > time_budget * 0.95:
            break

        conv = np.correlate(h, 1.0 - h, mode="full")
        cur_max = conv.max()
        # all shifts attaining (almost) the max
        active = np.where(conv >= cur_max - 1e-12)[0]

        grad = np.zeros_like(h)
        for idx in active:
            shift = idx - (n - 1)
            grad += _grad_for_shift(h, shift, dx)
        grad /= len(active)

        # simple diminishing step size
        step = 1.0 / math.sqrt(it)
        y = h - step * grad
        h = _project_sum_box(y, target_sum)

        val = _c5_value(h)
        if val < best_val - 1e-12:
            best_val = val
            best_h = h.copy()

        # occasional jitter to escape flat regions
        if it % 5000 == 0:
            jitter = best_h + np.random.randn(n) * 0.01
            jitter = np.clip(jitter, 0.0, 1.0)
            h = _project_sum_box(jitter, target_sum)

    return best_h, best_val

    (h_values, c5_bound, n_points)
    """
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # --------------------------------------------------------------------
    # Build an initial discretisation.
    init = globals().get("initial_h_values")
    if init is not None:
        base_h = np.asarray(init, dtype=float)
        base_n = base_h.shape[0]
    else:
        base_n = 70
        base_h = np.full(base_n, 0.5, dtype=float)

    # Work at twice the resolution (capped at 500 points for speed)
    factor = 2
    n = base_n * factor
    if n > 500:
        n = base_n
    if init is not None:
        x_base = np.linspace(0.0, 2.0, base_n, endpoint=False)
        x_fine = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_fine, x_base, base_h)
    else:
        h0 = np.random.rand(n)

    target_sum = n / 2.0
    h0 = _project_sum_box(h0, target_sum)

    best_h = h0.copy()
    best_val = _c5_value(best_h)

    # --------------------------------------------------------------------
    # First descent from the interpolated start.
    max_iter_start = min(20000, int(budget_s * 0.4 / 0.001))
    best_h, best_val = _descend(
        h_start=best_h,
        target_sum=target_sum,
        max_iter=max_iter_start,
        best_h=best_h,
        best_val=best_val,
        start_time=start,
        time_budget=budget_s,
    )

    # --------------------------------------------------------------------
    # Random restarts while time permits.
    while time.time() - start < budget_s * 0.9:
        remaining = budget_s - (time.time() - start)
        if remaining < 5.0:
            break
        # generate a fresh feasible random vector
        rnd = np.random.rand(n)
        rnd = _project_sum_box(rnd, target_sum)

        max_iter = min(15000, int(remaining * 0.5 / 0.001))
        best_h, best_val = _descend(
            h_start=rnd,
            target_sum=target_sum,
            max_iter=max_iter,
            best_h=best_h,
            best_val=best_val,
            start_time=start,
            time_budget=budget_s,
        )

    final_val = _c5_value(best_h)
    return best_h.tolist(), float(final_val), n
