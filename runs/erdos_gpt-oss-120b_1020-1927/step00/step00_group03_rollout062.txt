def hill_climb(h, time_limit):
    N = len(h)
    best_h = h.copy()
    best_val, _ = compute_c5(h)
    start = time.time()
    it_without_impr = 0
    while time.time() - start < time_limit:
        cur_val, corr = compute_c5(h)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
            it_without_impr = 0
        else:
            it_without_impr += 1
        # Determine shift
        shift_idx = np.argmax(corr)
        shift = shift_idx - (N-1)
        # find mismatched ones and candidate zeros
        mismatched, candidates = mismatch_and_candidates(h, shift)
        if mismatched.size and candidates.size:
            i = np.random.choice(mismatched)
            j = np.random.choice(candidates)
        else:
            # random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i]
        new_val, _ = compute_c5(h)
        if new_val <= cur_val:
            # accept
            pass
        else:
            # reject revert
            h[i], h[j] = h[j], h[i]
        # optionally restart if stuck
        if it_without_impr > 5000:
            # reinitialize random binary to avoid stuck
            # maybe keep best_h and start from best_h
            h = best_h.copy()
            it_without_impr = 0
    return best_h, best_val

def build_binary_pattern(N, L, G):

    # G: length 6 (sum 1)

        block_lengths.append(L[i])
    block_lengths.append(G[5])

    # factor = N / 2 (since total length 2)
    factor = N / 2.0
    steps = np.rint(np.array(block_lengths) * factor).astype(int)
    # Ensure steps are non-negative
    steps[steps < 0] = 0
    # Adjust total steps to N
    diff = N - steps.sum()
    # Distribute diff among zero blocks (even indices)
    zero_indices = np.arange(0, len(steps), 2)  # 0,2,4,...10
    zi = 0
    while diff != 0 and zi < len(zero_indices):
        idx = zero_indices[zi]
        if diff > 0:
            steps[idx] += 1
            diff -= 1
        else:
            if steps[idx] > 0:
                steps[idx] -= 1
                diff += 1
        zi = (zi + 1) % len(zero_indices) if diff != 0 else zi

    h = np.zeros(N, dtype=np.float64)

            if end > N:  # safety

        pos += cnt
        if pos >= N:
            break
    # In case length mismatches
    if pos < N:
        # Fill remaining with zeros

    # Ensure sum constraint exactly N/2
    cur_sum = h.sum()

import time
import numpy as np


def _compute_c5(h):
    """Return C5 value and raw correlation array for a binary step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * (2.0 / n), corr


def _mismatch_and_candidates(h, offset):
    """Indices of ones that contribute to the maximal shift
    and zero positions that can be swapped without increasing that shift."""
    n = len(h)
    if offset >= 0:
        d = offset
        if d >= n:
            mism = np.where(h == 1)[0]
            cand = np.empty(0, dtype=int)
            return mism, cand
        # mismatched ones (inside range)
        idx1 = np.where((h[: n - d] == 1) & (h[d:] == 0))[0]
        # mismatched ones (out of range)
        idx2 = np.where(h[n - d :] == 1)[0] + (n - d)
        mism = np.concatenate((idx1, idx2)) if idx2.size else idx1
        # candidate zeros that would match after swapping
        cand = np.where((h[: n - d] == 0) & (h[d:] == 1))[0]
    else:
        d = -offset
        if d >= n:
            mism = np.where(h == 1)[0]
            cand = np.empty(0, dtype=int)
            return mism, cand
        # mismatched ones (inside range)
        idx1 = np.where((h[d:] == 1) & (h[: n - d] == 0))[0] + d
        # mismatched ones (out of range)
        idx2 = np.where(h[:d] == 1)[0]
        mism = np.concatenate((idx1, idx2)) if idx2.size else idx1
        # candidate zeros that would match after swapping
        cand = np.where((h[d:] == 0) & (h[: n - d] == 1))[0] + d
    return mism, cand


def _build_binary_pattern(N, L, G):
    """
    Build a binary array of length N representing 5 intervals of ones.
    L – lengths of the five intervals (sum = 1),
    G – lengths of the six gaps (sum = 1).  Total length = 2.
    """
    # block order: G0, L0, G1, L1, …, G4, L4, G5
    blocks = []
    for i in range(5):
        blocks.append(G[i])   # zeros
        blocks.append(L[i])   # ones
    blocks.append(G[5])       # trailing zeros

    factor = N / 2.0                 # 2 units of length correspond to N steps
    steps = np.rint(np.array(blocks) * factor).astype(int)
    steps[steps < 0] = 0

    # Correct total number of steps to N by adjusting zero blocks (even indices)
    diff = N - steps.sum()
    zero_idx = np.arange(0, len(steps), 2)  # positions of zero blocks
    zi = 0
    while diff != 0:
        idx = zero_idx[zi % len(zero_idx)]
        if diff > 0:
            steps[idx] += 1
            diff -= 1
        else:
            if steps[idx] > 0:
                steps[idx] -= 1
                diff += 1
        zi += 1

    # Assemble the binary array
    h = np.zeros(N, dtype=np.float64)
    pos = 0
    for k, cnt in enumerate(steps):
        if cnt <= 0:
            continue
        if k % 2 == 1:                    # odd block → ones
            end = pos + cnt
            if end > N:
                end = N
            h[pos:end] = 1.0
        pos += cnt
        if pos >= N:
            break

    # Enforce exact mass constraint (sum = N/2)
    target = N // 2
    cur = int(h.sum())
    delta = target - cur
    if delta > 0:
        zeros = np.where(h == 0)[0]
        choose = np.random.choice(zeros, delta, replace=False)
        h[choose] = 1.0
    elif delta < 0:
        ones = np.where(h == 1)[0]
        choose = np.random.choice(ones, -delta, replace=False)
        h[choose] = 0.0
    return h


def _hill_climb(start_h, time_limit):
    """Local improvement by swapping a 1 with a 0 that reduces the worst shift."""
    h = start_h.copy()
    best_h = h.copy()
    best_val, _ = _compute_c5(h)
    start = time.time()
    no_improve = 0

    while time.time() - start < time_limit:
        cur_val, corr = _compute_c5(h)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # shift with maximal overlap
        max_idx = int(np.argmax(corr))
        offset = max_idx - (len(h) - 1)

        mism, cand = _mismatch_and_candidates(h, offset)

        if mism.size and cand.size:
            i = mism[np.random.randint(mism.size)]
            j = cand[np.random.randint(cand.size)]
        else:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = ones[np.random.randint(ones.size)]
            j = zeros[np.random.randint(zeros.size)]

        # try the swap
        h[i], h[j] = h[j], h[i]
        new_val, _ = _compute_c5(h)

        if new_val <= cur_val + 1e-12:
            # keep the swap
            pass
        else:
            # reject
            h[i], h[j] = h[j], h[i]

        # occasional restart from the best found so far
        if no_improve > 5000:
            h = best_h.copy()
            no_improve = 0

    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a small Erdős C₅ value."""
    np.random.seed(seed)

    # Choose a moderate grid size – large enough for accuracy yet cheap.
    N = 512

    # -----------------------------------------------------------------
    # 1. Quick random sampling of 5‑interval patterns
    # -----------------------------------------------------------------
    start_time = time.time()
    total_budget = float(budget_s)

    # Allocate roughly 20 % of the budget to the random phase
    rand_time = min(60.0, 0.2 * total_budget)
    end_rand = start_time + rand_time

    best_h = None
    best_val = np.inf

    while time.time() < end_rand:
        # Dirichlet‑like draws for interval lengths and gaps
        L = np.random.gamma(1.0, size=5)
        L /= L.sum()          # sum(L) = 1  (total mass of ones)
        G = np.random.gamma(1.0, size=6)
        G /= G.sum()          # sum(G) = 1  (total length of gaps)

        h = _build_binary_pattern(N, L, G)
        val, _ = _compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()

    # -----------------------------------------------------------------
    # 2. Local optimisation starting from the best sampled pattern
    # -----------------------------------------------------------------
    remaining = start_time + total_budget - time.time()
    if remaining > 0.5:
        if best_h is None:
            # fallback: a random balanced binary vector
            best_h = np.zeros(N, dtype=np.float64)
            idx = np.random.choice(N, N // 2, replace=False)
            best_h[idx] = 1.0
            best_val, _ = _compute_c5(best_h)

        best_h, best_val = _hill_climb(best_h, remaining)

    # Final evaluation (uses the same routine as the harness)
    final_val, _ = _compute_c5(best_h)

    return best_h, final_val, N
