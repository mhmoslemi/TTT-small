import time
import numpy as np

# Helper: project any vector onto the box [0,1] and the hyperplane sum = target
def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    # water-filling: find theta so that sum_i clip(v_i - theta,0,1) = target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        theta = (lo + hi) * 0.5
        p = np.clip(v - theta, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = theta
        else:
            hi = theta
    theta = (lo + hi) * 0.5
    return np.clip(v - theta, 0.0, 1.0)


# Plain evaluation used as a fallback when the harness does not provide it
def _eval_h(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    - n_points : number of samples used
    """

    # ------------------------------------------------------------------ #
    # 1. Determine problem size and an initial feasible vector
    # ------------------------------------------------------------------ #
    try:
        # initial_h_values is provided by the harness (if any)
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = h0.size

        # no starter – pick a modest size and build a triangular shape

    start_time = time.time()
