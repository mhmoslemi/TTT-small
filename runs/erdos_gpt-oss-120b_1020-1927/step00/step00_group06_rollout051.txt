def random_feasible_vector(n):

import time, math, numpy as np

def _project_to_sum_and_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {x: 0≤x_i≤1, sum x_i = target} by a simple
    greedy filling (sufficient for our fallback)."""
    v = np.clip(v, 0.0, 1.0)
    n = v.shape[0]
    idx = np.argsort(-v)               # descending
    out = np.zeros_like(v)
    whole = int(np.floor(target))
    frac = target - whole
    if whole > 0:
        out[idx[:whole]] = 1.0
    if frac > 1e-12 and whole < n:
        out[idx[whole]] = frac
    return out

def _c5_value(h: np.ndarray) -> float:
    """Return C_5 for a discrete step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) using zero padding
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _random_feasible(n: int) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum = n/2."""
    target = n / 2.0
    whole = int(np.floor(target))
    frac = target - whole
    v = np.zeros(n, dtype=float)
    # choose positions for the whole ones
    if whole > 0:
        ones_idx = np.random.choice(n, whole, replace=False)
        v[ones_idx] = 1.0
    # possibly one fractional entry
    if frac > 1e-12:
        remaining = np.setdiff1d(np.arange(n), np.where(v == 1.0)[0])
        idx = np.random.choice(remaining)
        v[idx] = frac
    return v
