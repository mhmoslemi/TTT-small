def project_bound_simplex(x, target):
    # Clip to 0-1 after subtracting offset lambda.
    lo = -1.0
    hi = 1.0

def project_onto_simplex_box(x, target):
    # Find lambda such that sum(clip(x - lambda,0,1)) = target
    lo = -2.0 # ensure lower enough
    hi = 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        y = np.clip(x - lam, 0.0, 1.0)
        s = np.sum(y)
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo+hi)/2.0, 0.0, 1.0)

def subgradient_optimize(n, init_h, time_budget):
    h = init_h.copy()
    target = n/2
    start = time.time()
    best_h = h.copy()
    best_val = compute_overlap(h)
    iteration = 0
    while time.time() - start < time_budget:
        # compute convolution
        sum_h = np.sum(h)
        conv = np.convolve(h, h, mode='full')
        v = sum_h - conv
        # identify worst shift (max)
        idx = np.argmax(v)
        L = idx - (n-1)   # shift amount, integer from -(n-1) to (n-1)
        # compute subgradient dv/dh
        # dv_i = 1 - (h[i+L] if in range else 0) - (h[i-L] if in range else 0)
        grad = np.ones(n)
        if L >= 0:
            grad[:n-L] -= h[L:]       # term h[i+L] for i = 0..n-L-1
            grad[L:]  -= h[:n-L]      # term h[i-L] for i = L..n-1
        else:
            Lp = -L
            grad[Lp:] -= h[:n-Lp]
            grad[:n-Lp] -= h[Lp:]
        # step size schedule
        step = 0.5 / math.sqrt(iteration+1)  # tune
        h_new = h - step * grad
        # ensure constraints via projection
        h_new = project_onto_simplex_box(h_new, target)
        # evaluate
        val = compute_overlap(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        # Optionally accept improvement only if val better else maybe keep? We'll keep h = h_new anyway (subgradient).
        h = h_new
        iteration += 1
        if iteration % 100 == 0 and time.time() - start > time_budget * 0.9:
            break
    return best_h, best_val

import numpy as np, time, math

def _proj_simplex_box(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto {0<=y<=1, sum(y)=target} by binary search on the offset λ."""
    lo, hi = -1.0, 2.0                      # λ bounds are generous
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Exact C5 value for a discrete step function."""
    n = h.size
    total = h.sum()
    # correlation returns Σ_i h_i h_{i+L} for all L (full mode)
    corr = np.correlate(h, h, mode="full")
    v = total - corr                         # v[L] = Σ h_i (1-h_{i+L})
    return (2.0 / n) * v.max()


def _soft_gradient(h: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the smooth max  (log‑sum‑exp) of v = Σh(1‑h_shift)."""
    n = h.size
    total = h.sum()
    corr = np.correlate(h, h, mode="full")
    v = total - corr                         # length 2n‑1, L = idx-(n‑1)

    # soft‑max weights
    z = beta * v
    m = np.max(z)
    exp_z = np.exp(z - m)
    w = exp_z / exp_z.sum()                 # Σ w = 1

    # e1 = Σ_L w_L * h_{j+L}, e2 = Σ_L w_L * h_{j‑L}
    w_rev = w[::-1]
    conv1 = np.convolve(w_rev, h, mode="full")
    conv2 = np.convolve(w, h, mode="full")
    off = len(w) - 1                         # offset = 2n‑2
    e1 = conv1[off:off + n]
    e2 = conv2[off:off + n]

    grad = np.ones(n) - e1 - e2
    return grad


def _gradient_descent(h0: np.ndarray,
                     target_sum: float,
                     max_seconds: float,
                     beta: float = 10.0) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the smoothed max objective."""
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_value(h)
    start = time.time()
    it = 0
    while time.time() - start < max_seconds:
        grad = _soft_gradient(h, beta)
        gn = np.linalg.norm(grad)
        if gn == 0.0:
            break
        # try a handful of step lengths (scaled by norm)
        improved = False
        for scale in (0.8, 0.4, 0.2, 0.1, 0.05, 0.02, 0.01):
            step = scale / gn
            h_new = _proj_simplex_box(h - step * grad, target_sum)
            val = _c5_value(h_new)
            if val + 1e-9 < best_val:
                h = h_new
                best_h = h_new.copy()
                best_val = val
                improved = True
                break
        if not improved:
            # no progress – shrink step size more aggressively
            step = 0.005 / gn
            h = _proj_simplex_box(h - step * grad, target_sum)
        it += 1
        if best_val <= 0.3808:               # early exit if record beaten
            break
    return best_h, best_val


def _subgradient_optimize(h0: np.ndarray,
                          target_sum: float,
                          max_seconds: float) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max (non‑smooth)."""
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_value(h)
    start = time.time()
    it = 0
    n = h.size
    while time.time() - start < max_seconds:
        total = h.sum()
        corr = np.correlate(h, h, mode="full")
        v = total - corr
        idx = int(v.argmax())
        L = idx - (n - 1)                     # shift amount
        grad = np.ones(n)
        if L >= 0:
            grad[:n - L] -= h[L:]
            grad[L:]     -= h[:n - L]
        else:
            p = -L
            grad[p:]    -= h[:n - p]
            grad[:n - p] -= h[p:]
        step = 0.4 / math.sqrt(it + 1.0)
        h_new = _proj_simplex_box(h - step * grad, target_sum)
        val = _c5_value(h_new)
        if val + 1e-9 < best_val:
            best_h = h_new.copy()
            best_val = val
            h = h_new
        else:
            h = h_new
        it += 1
        if best_val <= 0.3808:
            break
    return best_h, best_val
