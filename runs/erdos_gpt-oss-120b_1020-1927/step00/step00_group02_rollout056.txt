def propose_move(h, rng):
    n = len(h)
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if j >= i:
        j += 1
    # ensure i,j differ
    hi = h[i]; hj = h[j]
    # compute potential delta: we can move from j to i: increase i <=1 and decrease j >=0
    max_inc = min(1.0 - hi, hj)
    max_dec = min(hi, 1.0 - hj)  # moving opposite direction: increase j, decrease i
    # choose direction randomly
    if rng.random() < 0.5:
        # move from j to i
        max_delta = max_inc
        if max_delta <= 0:
            return None
        delta = rng.random() * max_delta
        return i, j, delta

            return None

import time, math
import numpy as np

# ---------------------------------------------------------
# Helper: compute the C5 value for a step function h (array)
def compute_c5(h, dx):
    """Return max_k ∫ h(x)(1-h(x+k)) dx for discrete h."""
    # correlation h with (1-h) using full mode (zero padding)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

# ---------------------------------------------------------
# Construct a binary vector with two blocks of ones.
def two_block_vector(len1, gap1, n):
    """
    n must be even, half = n//2.
    len1 : length of the first block of ones (samples)
    gap1 : length of zeros between the two blocks (samples)
    Returns binary array h of length n with exactly n/2 ones.
    """
    half = n // 2
    len2 = half - len1                # second block size
    # gap2 = half - gap1 automatically (unused, zeros at the end)
    h = np.zeros(n, dtype=np.float64)
    if len1:
        h[0:len1] = 1.0
    start2 = len1 + gap1
    if len2:
        h[start2:start2 + len2] = 1.0
    return h

# ---------------------------------------------------------
# Exhaustive search over all two‑block configurations.
def exhaustive_two_block_search(n, dx):
    half = n // 2
    best_val = float("inf")
    best_h = None
    # iterate integer lengths; inclusive of the end points
    for len1 in range(half + 1):
        for gap1 in range(half + 1):
            h = two_block_vector(len1, gap1, n)
            val = compute_c5(h, dx)
            if val < best_val:
                best_val = val
                best_h = h.copy()
    return best_h, best_val

# ---------------------------------------------------------
# Simple hill‑climbing (binary swaps) limited by a time budget.
def hill_climb(start_h, time_budget, rng, dx):
    n = len(start_h)
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val
    start = time.time()
    # pre‑compute locations of ones / zeros for faster updates
    while time.time() - start < time_budget:
        # locate current ones and zeros
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform the swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_val = compute_c5(cur_h, dx)

        if new_val < cur_val:
            # keep the move
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0

    return best_h, best_val

# ---------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points).
    h_values is a list of floats in [0,1] with sum exactly n_points/2.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a working resolution.
    # 200 points gives a good trade‑off between speed and granularity.
    n_points = 200
    if n_points % 2:
        n_points += 1
    dx = 2.0 / n_points
    half = n_points // 2

    # ------------------------------------------------------------------
    # 1) Exhaustive search over the two‑block family (fast, gives a solid start)
    best_h, best_val = exhaustive_two_block_search(n_points, dx)

    # ------------------------------------------------------------------
    # 2) Local improvement by hill‑climbing from the best configuration.
    elapsed = time.time() - kwargs.get("start_time", time.time())
    remaining = budget_s * 0.94 - elapsed  # keep a small safety margin
    if remaining > 2.0:
        best_h, best_val = hill_climb(best_h, remaining, rng, dx)

    # ------------------------------------------------------------------
    # 3) Additional random restarts (if time permits) to explore other patterns.
    elapsed = time.time() - kwargs.get("start_time", time.time())
    remaining = budget_s * 0.94 - elapsed
    # each restart gets at most a few seconds; continue while we have slack.
    while remaining > 3.0:
        # create a fresh random binary vector with exactly half ones
        h0 = np.zeros(n_points, dtype=np.float64)
        ones = rng.choice(n_points, size=half, replace=False)
        h0[ones] = 1.0
        # allocate a slice of the remaining budget
        slice_budget = min(remaining / 2.0, 5.0)
        h_opt, val_opt = hill_climb(h0, slice_budget, rng, dx)
        if val_opt < best_val:
            best_h, best_val = h_opt.copy(), val_opt
        # update remaining time
        elapsed = time.time() - kwargs.get("start_time", time.time())
        remaining = budget_s * 0.94 - elapsed

    # ------------------------------------------------------------------
    # Final safety check: enforce exact integral constraint.
    total = best_h.sum()
    target = half
    if total != target:
        diff = int(target - total)
        if diff > 0:
            zeros = np.where(best_h == 0.0)[0]
            best_h[zeros[:diff]] = 1.0
        elif diff < 0:
            ones = np.where(best_h == 1.0)[0]
            best_h[ones[: -diff]] = 0.0

    # Recompute the bound to be absolutely consistent with the returned vector.
    final_val = compute_c5(best_h, dx)

    # Return as plain Python list for the harness.
    return best_h.tolist(), float(final_val), n_points
