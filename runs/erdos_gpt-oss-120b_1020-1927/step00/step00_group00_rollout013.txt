def _max_lag_info(h, corr=None):
    n = h.shape[0]
    if corr is None:
        corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    if shift >= 0:
        i = np.arange(shift, n)
        prod = h[i] * h[i - shift]
    else:
        i = np.arange(0, n + shift)
        prod = h[i] * h[i - shift]
    if prod.size == 0:
        i_max_local = 0
    else:
        i_max_local = i[np.argmax(prod)]
    return shift, i_max_local

import time, math
import numpy as np

def _bound(h: np.ndarray) -> float:
    """Return C5 for a given discretisation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full").max() * dx


def _sa_optimize(
    h0: np.ndarray,
    time_budget: float,
    rng: np.random.Generator,
    start_T: float = 2e-3,
    end_T: float = 1e-7,
    max_step: float = 0.12,
    prob_target: float = 0.25,
):
    """Simple simulated‑annealing on the feasible set."""
    h = h0.copy()
    n = h.shape[0]
    dx = 2.0 / n
    # initial correlation and objective
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_val = corr.max() * dx
    best_h = h.copy()
    best_val = cur_val

    start = time.time()
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_budget:
            break

        # linear schedule for step size and temperature
        frac = elapsed / time_budget
        step = max_step * (1.0 - frac) + 0.001 * frac
        T = start_T * (1.0 - frac) + end_T * frac

        # information about the worst lag (the one attaining the max)
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)

        # ----- propose a move -------------------------------------------------
        if rng.random() < prob_target:
            # targeted: decrease a point that participates in the largest product
            if shift >= 0:
                i_range = np.arange(shift, n)
                prod = h[i_range] * h[i_range - shift]
            else:
                i_range = np.arange(0, n + shift)
                prod = h[i_range] * h[i_range - shift]
            if prod.size == 0:

            i_max = i_range[int(np.argmax(prod))]

            # choose a partner to receive the mass
            for _ in range(10):
