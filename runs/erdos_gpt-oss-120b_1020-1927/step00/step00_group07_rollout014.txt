def proj_capped_simplex(v, S, max_iter=100):
    # binary search for lambda to enforce sum S after clipping after translation.
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi)/2.0
        proj = np.clip(v - lam, 0, 1)
        cur = proj.sum()
        if cur > S:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0, 1)

import numpy as np, time, random, math

def _proj_capped_simplex(v, target_sum, eps=1e-12, iters=60):
    """Project v onto the set {x : 0≤x_i≤1, sum x_i = target_sum}."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(iters):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _evaluate_max_corr(h, dx):
    """Maximum of ∫ h(x)(1-h(x+k))dx approximated by discrete correlation."""
    return np.max(np.correlate(h, 1 - h, mode="full") * dx)


def run(seed=42, budget_s=1100, **_):

    np.random.seed(seed)

    # ------------------------------------------------------------------

    else:
        n = 200                                    # fallback size

        h = _proj_capped_simplex(v, n / 2.0)

    dx = 2.0 / n
    target_sum = n / 2.0
    h = _proj_capped_simplex(h, target_sum)      # enforce feasibility

    best_h = h.copy()
    best_val = _evaluate_max_corr(best_h, dx)

    start_time = time.time()
    max_iters = 30000
    alpha = 0.5
    alpha_max = 30.0

    step_coeff = 0.4
    T0 = 0.05                                   # temperature for occasional uphill moves
    shift_indices = np.arange(-(n - 1), n)       # d = shift
    half = n - 1

    for it in range(max_iters):

        # correlation values for current h
