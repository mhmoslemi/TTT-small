    n = len(h)
    dx = 2.0 / n
    best_val = max_overlap(h, dx)
    best_h = h.copy()
    rng = np.random.default_rng(0)
    iterations = 0
    while time.time() - start_time < budget:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j: continue
        # random direction sign
        direction = rng.choice([1, -1])
        if direction == 1:
            max_delta = min(1 - h[i], h[j])
        else:
            max_delta = min(h[i], 1 - h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random()
        h_new = h.copy()
        if direction == 1:
            h_new[i] += delta
            h_new[j] -= delta
        else:
            h_new[i] -= delta
            h_new[j] += delta
        val = max_overlap(h_new, dx)
        if val < best_val - 1e-12:
            best_val = val
            best_h = h_new.copy()
            h = h_new
        else:
            # occasionally accept worse move with low prob to escape
            if rng.random() < 0.001:
                h = h_new
        iterations += 1
    return best_h, best_val

                d = rng.random() * mx
                v = cur.copy()
                v[i] -= d
                v[j] += d

        # projection onto the feasible set
        cand = _project_sum_bounds(v, target)
        cand_val = _max_overlap(cand)

        # SA acceptance
        elapsed = time.time() - start
        t = elapsed / time_limit
        T = T0 * (1 - t) + T1 * t
        if cand_val < cur_val or rng.random() < math.exp(-(cand_val - cur_val) / max(T, 1e-12)):
            cur, cur_val = cand, cand_val
            if cand_val < best_val:
                best, best_val = cand.copy(), cand_val

    return best, best_val

def _improve_binary(h, time_limit):
    """Simple hill‑climb on a binary (0/1) vector."""
    start = time.time()
    n = h.shape[0]
    cur = h.astype(float)
    cur_val = _max_overlap(cur)
    best = cur.copy()
    best_val = cur_val
    rng = np.random.default_rng(54321)

    while time.time() - start < time_limit:
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = cur.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_val = _max_overlap(cand)
        if cand_val < cur_val:
            cur, cur_val = cand, cand_val
            if cand_val < best_val:
                best, best_val = cand.copy(), cand_val
        elif rng.random() < 0.001:          # tiny chance to escape
            cur, cur_val = cand, cand_val
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. obtain an initial vector (if supplied) and make it feasible
    # ------------------------------------------------------------------
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
    except NameError:
        # fall back: even length binary with half ones
        n0 = 100
        h0 = np.zeros(n0, dtype=float)
        h0[: n0 // 2] = 1.0
        rng.shuffle(h0)

    n0 = h0.shape[0]
    target0 = n0 / 2.0
    if not np.isclose(h0.sum(), target0):
        h0 = _project_sum_bounds(h0, target0)

    best_h, best_val = h0.copy(), _max_overlap(h0)

    start_time = time.time()
    remaining = budget_s

    # ------------------------------------------------------------------
    # 2. Continuous SA/gradient optimisation on the original grid
    # ------------------------------------------------------------------
    t_phase = min(remaining * 0.45, 300)               # at most 5 min
    h_opt, val_opt = _improve_continuous(best_h, t_phase)
    if val_opt < best_val:
        best_h, best_val = h_opt, val_opt
    remaining = budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # 3. Try a larger even-sized binary construction (replicated pattern)
    # ------------------------------------------------------------------
    if remaining > 30:
        n_big = min(4 * n0, 1000)                     # ensure ≤1000
        if n_big % 2 == 1:
            n_big += 1
        # binary start: random half‑ones
        h_big = np.zeros(n_big, dtype=float)
        h_big[: n_big // 2] = 1.0
        rng.shuffle(h_big)
        t_big = remaining * 0.45
        h_big_opt, val_big_opt = _improve_binary(h_big, t_big)
        if val_big_opt < best_val:
            best_h, best_val = h_big_opt, val_big_opt
        remaining = budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # 4. Final refinement on the current best vector
    # ------------------------------------------------------------------
    if remaining > 5:
        h_ref, val_ref = _improve_continuous(best_h, remaining)
        if val_ref < best_val:
            best_h, best_val = h_ref, val_ref

    # final safety projection (tiny numerical drift)
    n_final = best_h.shape[0]
    target_final = n_final / 2.0
    if not np.isclose(best_h.sum(), target_final):
        best_h = _project_sum_bounds(best_h, target_final)

    best_val = _max_overlap(best_h)   # final exact evaluation
    return best_h.tolist(), float(best_val), n_final
