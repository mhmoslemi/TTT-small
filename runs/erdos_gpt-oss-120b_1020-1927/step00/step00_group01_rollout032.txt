def optimize_once(n, seed, max_iters):
    np.random.seed(seed)
    # initial binary pattern
    half = n//2
    x = np.zeros(n)
    indices = np.random.permutation(n)
    x[indices[:half]] = 1.0
    if n % 2 == 1:
        x[indices[half]] = 0.5
    # No need for projection since already feasible.

    best_x = x.copy()
    best_obj = compute_obj(best_x, 2.0/n)

    for it in range(max_iters):
        grad = compute_grad(x)
        # choose step size schedule
        # Use diminishing step size.
        lr = 0.5 / np.sqrt(it+1)  # (or 0.3)
        x_raw = x - lr * grad
        # project
        x = project_onto_capped_simplex(x_raw, s=n/2, ub=1.0, tol=1e-9)
        # compute obj
        obj = compute_obj(x, 2.0/n)
        if obj < best_obj:
            best_obj = obj
            best_x = x.copy()
        # Optionally do random swap to escape
        if it % 200 == 0 and it>0:
            # small random change
            i,j = np.random.choice(n, 2, replace=False)
            delta = min(0.1, x[i], 1-x[j])
            if delta>1e-12:
                x_temp = x.copy()
                x_temp[i] -= delta
                x_temp[j] += delta
                # No need to project because sum same and bounds satisfied
                x = x_temp
    return best_x, best_obj

def project_onto_capped_simplex(v, s, ub=1.0, tol=1e-9):
    # Solve min ||y - v||^2 s.t. sum(y) = s, 0 <= y <= ub
    # Use binary search on tau.
    # Starting bounds for tau:
    # For tau-> -inf, y_i = ub => sum = n*ub
    # For tau-> +inf, y_i = 0 => sum = 0
    # So range.
    lower = -1e5
    upper = 1e5
    for _ in range(60):
        tau = (lower + upper) / 2.0
        y = np.clip(v - tau, 0, ub)
        total = np.sum(y)
        if total > s:
            lower = tau
        else:
            upper = tau
    y = np.clip(v - (lower+upper)/2.0, 0, ub)
    # ensure sum close to s; if not due to numeric tolerance, adjust via small correction.
    # Compute diff = s - sum(y). Distribute diff among non-bound elements proportionally.
    diff = s - np.sum(y)
    if abs(diff) > 1e-8:
        # Adjust by adding diff / count to elements not at bounds
        free = (y > 0 + 1e-12) & (y < ub - 1e-12)
        if np.any(free):
            y[free] += diff / np.sum(free)
            y = np.clip(y, 0, ub)
        else:
            # all at bounds, just distribute small diff equally
            y += diff / len(y)
            y = np.clip(y, 0, ub)
    return y

import numpy as np, time

def compute_corr(x):
    return np.correlate(x, 1 - x, mode='full')

def compute_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = compute_corr(x)
    return dx * np.max(corr)

def max_corr_and_shift(x):
    n = len(x)
    corr = compute_corr(x)
    max_idx = np.argmax(corr)
    shift = max_idx - (n - 1)
    return max_idx, shift

def compute_grad(x):
    n = len(x)
    _, shift = max_corr_and_shift(x)
    grad = np.zeros(n)
    if shift >= 0:
        s = shift
        # (1 - x[i+s]) part for i=0..n-s-1
        grad[:n-s] += (1 - x[s:])
        # -x[i-s] part for i=s..n-1
        grad[s:] -= x[:n-s]
    else:
        s = -shift
        # (1 - x[i-s]) for i=s..n-1
        grad[s:] += (1 - x[:n-s])
        # -x[i+s] for i=0..n-s-1
        grad[:n-s] -= x[s:]
    # multiply by dx
    grad *= (2.0 / n)
    return grad

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def _correlate(x: np.ndarray) -> np.ndarray:
    """Full correlation of x with (1-x)."""
    return np.correlate(x, 1.0 - x, mode="full")


def _objective(x: np.ndarray) -> float:
    """C5 bound for a given step vector."""
    n = x.size
    dx = 2.0 / n
    return dx * np.max(_correlate(x))


def _max_shift(x: np.ndarray):
    """Return (index, shift) where the correlation attains its maximum."""
    n = x.size
    corr = _correlate(x)
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # zero‑based shift (negative or positive)
    return idx, shift


def _gradient(x: np.ndarray) -> np.ndarray:
    """Gradient of the max‑correlation objective (subgradient)."""
    n = x.size
    _, shift = _max_shift(x)
    g = np.zeros_like(x)

    if shift >= 0:
        s = shift
        # (1 - x[i+s]) contribution
        g[: n - s] += 1.0 - x[s:]
        # -x[i-s] contribution
        g[s:] -= x[: n - s]
    else:
        s = -shift
        # (1 - x[i-s]) contribution
        g[s:] += 1.0 - x[: n - s]
        # -x[i+s] contribution
        g[: n - s] -= x[s:]

    g *= (2.0 / n)                # multiply by dx
    return g


def _project_capped_simplex(v: np.ndarray, target_sum: float, ub: float = 1.0, tol: float = 1e-9) -> np.ndarray:
    """
    Euclidean projection of v onto {y | sum(y)=target_sum, 0<=y<=ub}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    n = v.size
    # Quick check (already feasible)
    if abs(v.sum() - target_sum) <= tol and np.all(v >= -tol) and np.all(v <= ub + tol):
        return np.clip(v, 0.0, ub)

    lo, hi = -1e5, 1e5
    for _ in range(60):
        tau = (lo + hi) * 0.5
        y = np.clip(v - tau, 0.0, ub)
        s = y.sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    y = np.clip(v - (lo + hi) * 0.5, 0.0, ub)

    # Final correction for tiny numerical drift
    diff = target_sum - y.sum()
    if abs(diff) > 1e-8:
        free = (y > 0.0 + 1e-12) & (y < ub - 1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, ub)
        else:
            # fallback: distribute uniformly (should not happen often)
            y += diff / n
            y = np.clip(y, 0.0, ub)
    return y

def _optimize_one_run(n: int, rng_seed: int, max_iters: int = 12000) -> tuple:
    """
    Perform a projected (sub)gradient descent run on a length‑n vector.
    Returns the best vector found and its objective value.
    """
    rng = np.random.RandomState(rng_seed)
    half = n // 2
    x = np.zeros(n, dtype=float)

    # Initialise with a binary pattern (half 1’s, half 0’s, possibly a 0.5)
    perm = rng.permutation(n)
    x[perm[:half]] = 1.0
    if n % 2:
        x[perm[half]] = 0.5                     # exact sum = n/2
    # Ensure feasibility (tiny numerical errors might appear)
    x = _project_capped_simplex(x, target_sum=n / 2.0)

    best_x = x.copy()
    best_val = _objective(x)

    # Diminishing step‑size schedule
    base_lr = 0.9

    for it in range(max_iters):
        grad = _gradient(x)
        lr = base_lr / np.sqrt(it + 1.0)          # 1/√t decay
        x_new = x - lr * grad
        x = _project_capped_simplex(x_new, target_sum=n / 2.0)

        cur_val = _objective(x)
        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()

        # occasional random mass‑swap to escape plateaus
        if (it + 1) % 250 == 0:
            i, j = rng.choice(n, 2, replace=False)
            # move a small chunk from i to j respecting bounds
            delta = min(0.08, x[i], 1.0 - x[j])
            if delta > 1e-12:
                x[i] -= delta
                x[j] += delta
                # projection not needed – constraints stay intact

    return best_x, best_val
