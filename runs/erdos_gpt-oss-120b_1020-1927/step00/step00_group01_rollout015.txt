    best_h = h.copy()
    best_val = compute_C5(h)
    cur_val = best_val

    start = time.time()
    max_iter = 10000000 # high upper bound
    i = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # schedule
        ratio = i / max_iter
        if ratio > 1:
            ratio = 1
        T = (0.01) * (1 - ratio) + 1e-6 * ratio  # linear interpolation
        max_step = 0.3 * (1 - ratio) + 0.001 * ratio
        move = random_move(h, rng, max_step)
        if move[0] is None:
            i += 1
            continue
        i_idx, j_idx, delta = move

        h_new = h.copy()
        h_new[i_idx] += delta
        h_new[j_idx] -= delta
        # Ensure values are within bounds after rounding errors
        h_new[i_idx] = np.clip(h_new[i_idx], 0.0, 1.0)
        h_new[j_idx] = np.clip(h_new[j_idx], 0.0, 1.0)

            cur_val = new_val
            if new_val < best_val:

        i += 1
    return best_h, best_val

def project_onto_box_simplex(y, target, tol=1e-10, max_iter=100):
    # binary search lambda to satisfy sum constraints with x = clip(y - lam, 0, 1)
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    x = np.clip(y - (lo + hi) / 2, 0.0, 1.0)
    # Adjust small residual to meet target precisely
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute to non-saturated entries
        idx = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if len(idx) > 0:
            x[idx] += diff / len(idx)
            x = np.clip(x, 0.0, 1.0)
    return x

    idx = np.argmin(autocorr)

    n = len(h)
    # compute contributions c_i = (h[i+lag] if i+lag in range) + (h[i-lag] if i-lag in range)
    c = np.zeros(n)

        c[:n-lag] += h[lag:]  # h[i+lag] corresponds to h[i+lag] for i from 0..n-lag-1 (i+lag in range)

import time
import numpy as np

def _project_box_simplex(y, target, tol=1e-10, max_iter=80):
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        idx = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if idx.size:
            x[idx] += diff / idx.size
            x = np.clip(x, 0.0, 1.0)
    return x

def _random_initial(n, rng):
    y = rng.random(n)
    return _project_box_simplex(y, n / 2.0)

def _upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    y = np.interp(new_x, old_x, h)
    return _project_box_simplex(y, new_n / 2.0)

def _max_overlap(h):
    n = len(h)
    s = h.sum()
    ac = np.correlate(h, h, mode='full')
    min_ac = ac.min()
    return (s - min_ac) * (2.0 / n)

def _random_move(h, rng, max_step):
    n = len(h)
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    hi = h[i]
    hj = h[j]
    lo = max(-hi, hj - 1.0)
    hi_bound = min(1.0 - hi, hj)
    if hi_bound <= lo:
        return None, None, None
    lo2 = max(lo, -max_step)
    hi2 = min(hi_bound, max_step)
    if lo2 >= hi2:
        return None, None, None
    delta = rng.uniform(lo2, hi2)
    return i, j, delta

def _targeted_move(h, rng, max_step):
    n = len(h)
    ac = np.correlate(h, h, mode='full')
    idx = ac.argmin()
    lag = idx - (n - 1)
    c = np.zeros(n)
    if lag == 0:
        c = 2 * h
    elif lag > 0:
        L = lag
        c[:n - L] += h[L:]
        c[L:] += h[:n - L]
    else:
        L = -lag
        c[L:] += h[:n - L]
        c[:n - L] += h[L:]
    i_max = int(np.argmax(c))

        return None, None, None
    hi = h[i_max]

def _anneal(h0, budget, rng, target_bound=0.3808):
    h = h0.copy()
    best_h = h.copy()
    best_val = _max_overlap(h)
    cur_val = best_val
