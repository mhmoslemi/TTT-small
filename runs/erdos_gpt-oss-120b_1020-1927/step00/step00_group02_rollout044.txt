def _compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n

    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def _project_to_feasible(h):

    h += diff / n
