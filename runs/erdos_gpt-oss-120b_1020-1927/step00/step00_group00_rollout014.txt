import numpy as np

def _project_onto_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {x | 0<=x_i<=1 , sum x = target}."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        p = np.clip(v - mid, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _smooth_max_step(h: np.ndarray, alpha: float, eta: float, target: float) -> (np.ndarray, float):
    """One iteration: smooth‑max of overlaps, gradient step, projection."""
    n = h.size
    dx = 2.0 / n
    ov = np.correlate(h, 1.0 - h, mode="full") * dx          # length 2n‑1
    m = ov.max()
    w = np.exp(alpha * (ov - m))
    w /= w.sum()                                            # soft weights, sum=1

    # gradient of smooth max = 1 - Σ_d w_d * (h shifted forward + h shifted backward)
    grad = np.ones_like(h)
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        d = idx - (n - 1)                                   # shift offset
        if d >= 0:
            nd = n - d
            if nd:
                grad[:nd] -= weight * h[d:]                 # forward term
                grad[d:] -= weight * h[:nd]                 # backward term
        else:
            da = -d
            nd = n - da
            if nd:
                grad[da:] -= weight * h[:nd]                # forward (negative) term
                grad[:nd] -= weight * h[da:]                # backward (negative) term

    h_new = h - eta * grad
    h_new = _project_onto_box_sum(h_new, target)

    new_ov = np.correlate(h_new, 1.0 - h_new, mode="full") * dx
    return h_new, new_ov.max()
