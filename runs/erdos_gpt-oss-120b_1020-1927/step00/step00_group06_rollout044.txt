        s = w.sum()
        if s > target_sum:
            lo = t
        else:
            hi = t
    return np.clip(v - t, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """
    Compute C_5 = max_k ∫ h(x)*(1-h(x+k)) dx for the discretized step function.
    Uses FFT to obtain the full linear correlation efficiently.
    """
    n = h.shape[0]
    dx = 2.0 / n
    # Length for linear convolution (next power of two for speed)
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    corr = corr[: 2 * n - 1]            # keep the linear part
    return dx * np.max(corr)


def _mutate(h: np.ndarray, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """
    Produce a new feasible vector from h.
    Performs a few pairwise mass transfers that keep the total sum unchanged,
    and occasionally adds a small Gaussian perturbation coupled with a projection.
    """
    n = h.shape[0]
    h_new = h.copy()
    # Random number of pairwise swaps
    swaps = rng.integers(1, 6)
    for _ in range(swaps):
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = h_new[i], h_new[j]
        # feasible epsilon range so that both stay in [0,1]
        lo = max(-hi, hj - 1.0)
        hi_eps = min(1.0 - hi, hj)
        if lo < hi_eps:
            eps = rng.uniform(lo, hi_eps)
            h_new[i] = hi + eps
            h_new[j] = hj - eps
    # Occasionally inject a small global noise and re‑project
    if rng.random() < 0.1:
        noisy = h_new + rng.normal(scale=0.02, size=n)
        h_new = _project_capped_simplex(noisy, target_sum)
    return h_new


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] that yields a low C_5 value.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine discretisation size
    # ------------------------------------------------------------------
    # Prefer a moderately fine grid; 120 points is a good trade‑off.
    n_points = 120
    target_sum = n_points / 2.0

    # ------------------------------------------------------------------
    # 2. Initialise a feasible population
    # ------------------------------------------------------------------
    try:
        # Scale the supplied initial construction (if any) to the chosen grid.
        init = np.asarray(initial_h_values, dtype=float)   # type: ignore[name-defined]
        if init.ndim != 1:
            raise ValueError
        x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h0 = np.interp(x_new, x_old, init)
    except Exception:
        # No usable initial construction – start from random.
        h0 = rng.uniform(size=n_points)

    h0 = _project_capped_simplex(h0, target_sum)

    # Elite solution (global best)
    best_h = h0.copy()
    best_score = _c5_value(best_h)

    # Population for evolutionary search
    pop_size = 30
    population = [best_h]
    for _ in range(pop_size - 1):
        rand_vec = rng.uniform(size=n_points)
        population.append(_project_capped_simplex(rand_vec, target_sum))

    scores = [_c5_value(h) for h in population]

    # Keep population sorted (best first)
    order = np.argsort(scores)
    population = [population[i] for i in order]
    scores = [scores[i] for i in order]

    # ------------------------------------------------------------------
    # 3. Evolutionary loop respecting the time budget
    # ------------------------------------------------------------------
    start = time.time()
    elapsed = 0.0
    generations_without_improve = 0
    max_stagnant = 200      # generations before a partial restart

    while elapsed < budget_s:
        # ----- generate offspring ---------------------------------------
        offspring = []
        # Mutate the top half of the current population
        num_parents = max(1, pop_size // 2)
        for i in range(num_parents):
            child = _mutate(population[i], rng, target_sum)
            offspring.append(child)
        # Occasionally inject a completely random individual
        if rng.random() < 0.05:
            rand_vec = rng.uniform(size=n_points)
            offspring.append(_project_capped_simplex(rand_vec, target_sum))

        # ----- evaluate -------------------------------------------------
        offspring_scores = [_c5_value(h) for h in offspring]

        # ----- selection ------------------------------------------------
        all_pop = population + offspring
        all_scores = scores + offspring_scores
        idx = np.argsort(all_scores)[:pop_size]
        population = [all_pop[i] for i in idx]
        scores = [all_scores[i] for i in idx]

        # ----- update global best ---------------------------------------
        if scores[0] < best_score - 1e-12:
            best_score = scores[0]
            best_h = population[0].copy()
            generations_without_improve = 0
        else:
            generations_without_improve += 1

        # ----- occasional diversity boost -------------------------------
        if generations_without_improve >= max_stagnant:
            # Replace the worst half with fresh random feasible vectors
            for i in range(pop_size // 2, pop_size):
                rand_vec = rng.uniform(size=n_points)
                population[i] = _project_capped_simplex(rand_vec, target_sum)
                scores[i] = _c5_value(population[i])
            generations_without_improve = 0

        # ----- time check ------------------------------------------------
        elapsed = time.time() - start

    # ------------------------------------------------------------------
    # 4. Refine the best candidate with a simple hill‑climber
    # ------------------------------------------------------------------
    current = best_h.copy()
    current_score = best_score
    while time.time() - start < budget_s:
        cand = _mutate(current, rng, target_sum)
        cand_score = _c5_value(cand)
        if cand_score < current_score:
            current, current_score = cand, cand_score
            if cand_score < best_score:
                best_h, best_score = cand.copy(), cand_score

    # ------------------------------------------------------------------
    # 5. Final post‑processing
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_capped_simplex(best_h, target_sum)
    best_score = _c5_value(best_h)

    return (best_h, float(best_score), n_points)
