def step_move(h):
    n = len(h)
    # compute overlap array
    c, max_val, max_idx = compute_overlap(h)
    k = max_idx - (n-1)
    # compute shifted arrays for gradient-like
    if k >= 0:
        h_p = np.concatenate((h[k:], np.zeros(k)))
        h_m = np.concatenate((np.zeros(k), h[:n-k]))
    else:
        k_abs = -k
        h_p = np.concatenate((np.zeros(k_abs), h[:n-k_abs]))
        h_m = np.concatenate((h[k_abs:], np.zeros(k_abs)))
    g = 1 - (h_p + h_m)
    dec = np.where(g > 1e-12)[0]
    inc = np.where(g < -1e-12)[0]
    if len(dec) == 0 or len(inc) == 0:
        # fallback random move
        i, j = np.random.choice(n, 2, replace=False)
        delta_max = min(h[i], 1 - h[j])
        if delta_max <= 0:
            return h, max_val
        delta = delta_max * np.random.rand()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # compute new objective
        new_max = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
        if new_max < max_val:
            return h_new, new_max
        else:
            return h, max_val
    # choose random i from dec, j from inc
    i = dec[np.random.randint(len(dec))]
    j = inc[np.random.randint(len(inc))]
    delta_max = min(h[i], 1 - h[j])
    if delta_max <= 0:
        return h, max_val
    # optionally limit delta by magnitude of gradient difference
    delta = delta_max * (np.random.rand() * 0.8 + 0.1) # between 0.1 and 0.9 of max
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    # compute new objective
    new_max = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
    # accept if improvement
    if new_max < max_val:
        return h_new, new_max
    else:
        # occasionally accept worse move with small probability
        prob = np.exp((max_val - new_max) / (0.001 + 1e-12)) # temperature maybe 0.001

            return h_new, new_max

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (top level, no closures)
# ------------------------------------------------------------
def _correlation_max(h, dx):
    """Return max overlap value and index of the lag that attains it."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_idx = int(np.argmax(corr))
    return float(corr[max_idx]), max_idx


def _shift_array(arr, k):
    """Return arr shifted by k positions (zero padding, not periodic)."""
    n = arr.size
    if k >= 0:
        out = np.empty(n, dtype=arr.dtype)
        out[: n - k] = arr[k:]
        out[n - k :] = 0.0
        return out
    else:  # k < 0
        k = -k
        out = np.empty(n, dtype=arr.dtype)
        out[k:] = arr[: n - k]
        out[:k] = 0.0
        return out


def _gradient_for_shift(h, k):
    """Sub‑gradient of the worst overlap w.r.t. h for a given shift k."""
    hp = _shift_array(h, k)          # h_{i+k} (zero where out‑of‑range)
    hm = _shift_array(h, -k)         # h_{i-k}
    # d/ dh_i  ( h_i * (1 - h_{i+k}) + h_{i-k} * (1 - h_i) )  = 1-(h_{i+k}+h_{i-k})
    return 1.0 - (hp + hm)


def _make_feasible(h, target_sum):
    """Clip to [0,1] and then adjust the total sum to target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        # degenerate – distribute uniformly
        h[:] = target_sum / h.size
        return h
    # rescale to hit the sum (may violate bounds by a tiny amount)
    factor = target_sum / cur
    h *= factor
    # repair any out‑of‑bounds caused by scaling
    excess = h - 1.0
    deficit = -h
    if np.any(excess > 0):
        over = excess[excess > 0]
        h[excess > 0] = 1.0
        target_sum -= over.sum()
    if np.any(deficit > 0):
        under = deficit[deficit > 0]
        h[deficit > 0] = 0.0
        target_sum += under.sum()
    # final fine‑tune by distributing the remaining discrepancy
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # add/subtract from the most flexible positions
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            # fallback: spread uniformly
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
        else:
            step = diff / free.size
            h[free] += step
            h = np.clip(h, 0.0, 1.0)
    return h


def _single_move(h, k_max, rng):
    """
    Propose a move that tries to reduce the worst overlap for shift k_max.
    Returns (new_h, improved) – `improved` is True if the move lowered the objective.
    """
    n = h.size
    g = _gradient_for_shift(h, k_max)

    dec = np.where(g > 1e-12)[0]   # decreasing h reduces the worst overlap
    inc = np.where(g < -1e-12)[0]  # increasing h reduces the worst overlap

    if dec.size == 0 or inc.size == 0:
        # fallback: random feasible swap
        i, j = rng.choice(n, 2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            return h, False
        delta = max_delta * rng.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        return h_new, True

    i = int(rng.choice(dec))
    j = int(rng.choice(inc))
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0:
        return h, False
    # move a random fraction (10‑90%) of the feasible amount
    delta = max_delta * (0.1 + 0.8 * rng.random())
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new, True


def _optimize(h_start, dx, target_sum, budget, rng):
    """
    Local optimisation based on targeted pairwise moves.
    Returns the best vector found and its achieved bound.
    """
    best_h = h_start.copy()
    best_val, best_idx = _correlation_max(best_h, dx)

    h = best_h.copy()
    start_time = time.time()
    it = 0
    # simple linear cooling schedule
    T0 = 0.005
    while time.time() - start_time < budget:
        # identify the current worst shift
        cur_val, cur_idx = _correlation_max(h, dx)
        k = cur_idx - (h.size - 1)   # actual lag (negative … positive)
        # propose a move
        h_candidate, ok = _single_move(h, k, rng)
        if not ok:
            it += 1
            continue
        cand_val, _ = _correlation_max(h_candidate, dx)

        # Metropolis acceptance
        if cand_val < cur_val:
            h = h_candidate
            if cand_val < best_val:
                best_h, best_val = h_candidate.copy(), cand_val
        else:
            # temperature falls linearly to zero
            elapsed = time.time() - start_time
            T = max(0.0, T0 * (1.0 - elapsed / budget))
            prob = np.exp(-(cand_val - cur_val) / (T + 1e-12))
            if rng.random() < prob:
                h = h_candidate

        # occasional jitter to escape plateaus
        if it % 5000 == 0 and it > 0:
            jitter = (rng.random(h.size) - 0.5) * 0.02   # +-1%
            h_tmp = h + jitter
            h_tmp = _make_feasible(h_tmp, target_sum)
            tmp_val, _ = _correlation_max(h_tmp, dx)
            if tmp_val < best_val:
                best_h, best_val = h_tmp.copy(), tmp_val
                h = h_tmp
        it += 1

    return best_h, best_val
