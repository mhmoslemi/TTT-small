def gradient_for_shift(h, s):
    N = len(h)
    grad = np.zeros(N)
    if s >= 0:
        # forward term: i from 0 to N-1-s
        grad[:N-s] += 1 - h[s:]
        # backward term: i from s to N-1
        grad[s:] -= h[:N-s]
    else:
        # negative shift: set t = -s >0 (positive)
        t = -s
        # forward term: i from t to N-1, since i+s = i - t >=0
        grad[t:] += 1 - h[:N-t]
        # backward term: i from 0 to N-1-t, i - s = i + t
        grad[:N-t] -= h[t:]
    return grad

def project_onto_box_and_sum(v, target, max_iter=60):
    lo = -np.max(v) - 1  # ensure all v+lo <=0 so clip->0
    hi = 1 - np.min(v)   # ensure all v+hi >=1; clip->1
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0, 1)
        s = w.sum()
        if abs(s - target) < 1e-12:
            return w
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2.0, 0, 1)

import numpy as np
import random
import time
import math

def _proj_box_sum(v: np.ndarray, target: float, max_iter: int = 60) -> np.ndarray:
    """Project v onto the set {w | 0<=w<=1, sum(w)=target}."""
    lo = -np.max(v) - 1.0   # makes v+lo <= 0 -> clipped to 0, sum = 0
    hi = 1.0 - np.min(v)    # makes v+hi >= 1 -> clipped to 1, sum = len(v)
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if abs(s - target) <= 1e-12:
            return w
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Return C5 for a given h (max overlap integral)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _gradient_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Subgradient of the overlap term for a given shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        s = shift
        # forward term: i where i+s in range
        g[: n - s] += 1.0 - h[s:]
        # backward term: i where i-s in range
        g[s:] -= h[: n - s]
    else:
        t = -shift
        # forward term (i+t in range)
        g[t:] += 1.0 - h[: n - t]
        # backward term (i-t in range)
        g[: n - t] -= h[t:]
    return g


def _step_gradient(h: np.ndarray, shift: int, alpha: float, target: float) -> np.ndarray:
    """Take a projected gradient step for the worst shift."""
    grad = _gradient_shift(h, shift)
    new_raw = h - alpha * grad
    return _proj_box_sum(new_raw, target)


def _pair_move(h: np.ndarray) -> np.ndarray | None:
    """Generate a feasible vector by moving mass between two coordinates."""
    n = h.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    lower = max(-hi, hj - 1.0)
    upper = min(1.0 - hi, hj)
    if lower >= upper:
        return None
    delta = np.random.uniform(lower, upper)
    nh = h.copy()
    nh[i] = hi + delta
    nh[j] = hj - delta
    return nh

        h = np.random.rand(n)
    n = h.shape[0]
    target = n / 2.0
    # enforce feasibility
    h = np.clip(h, 0.0, 1.0)
    h = _proj_box_sum(h, target)

    best_h = h.copy()
    best_score = _c5_value(h)

    start = time.time()
    it = 0
    stagnation = 0
    # annealing / step‑size schedules
    temp0 = 0.1
    alpha0 = 0.5

    while time.time() - start < budget_s - 0.4:
        it += 1
        # current evaluation
        cur_score = _c5_value(h)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1

        # decide move type
        if random.random() < 0.18:      # gradient–guided move
            # find a shift attaining the current maximum
            corr = np.correlate(h, 1.0 - h, mode="full")
            dx = 2.0 / n
            corr = corr * dx
            max_val = corr.max()
            cand = np.where(np.abs(corr - max_val) < 1e-12)[0]
            shift = int(random.choice(cand.tolist())) - (n - 1)
            alpha = alpha0 * (0.9995 ** it)
            nh = _step_gradient(h, shift, alpha, target)
            new_score = _c5_value(nh)
            # simple Metropolis acceptance
            if new_score <= cur_score or random.random() < 1e-3:
                h = nh
        else:                           # random pair move (simulated annealing)
            nh = _pair_move(h)
            if nh is None:
                continue
            new_score = _c5_value(nh)
            # temperature schedule
            temp = temp0 * (0.9999 ** it)
            if new_score < cur_score or random.random() < math.exp(-(new_score - cur_score) / max(temp, 1e-12)):
                h = nh

        # occasional diversify if stuck
        if stagnation > 25000:
            # perturb best solution and re‑project
            nh = best_h + np.random.normal(scale=0.12, size=n)
            nh = np.clip(nh, 0.0, 1.0)
            h = _proj_box_sum(nh, target)
            stagnation = 0

    # final bound – use the provided evaluator if available
    if "evaluate_erdos_solution" in globals():
        final_c5 = float(evaluate_erdos_solution(best_h.tolist()))
    else:
        final_c5 = best_score

    return best_h.tolist(), final_c5, n
