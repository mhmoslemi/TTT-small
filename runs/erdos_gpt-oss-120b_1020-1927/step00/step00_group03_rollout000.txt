def project_cap_simplex(v, S):
    # ensures 0 <= v <= 1, sum = S.
    v = np.clip(v, 0, 1)
    # iterative water-filling
    max_iter = 100
    for _ in range(max_iter):
        total = v.sum()
        diff = total - S
        if abs(diff) < 1e-12:
            break
        # indices not on bound according to sign of diff
        if diff > 0:
            # need to reduce sum: reduce among those >0
            free = v > 0
            # but also avoid reducing those at 0 (they cannot go below)
        else:
            # need to increase sum: increase where v < 1
            free = v < 1
        count = free.sum()
        if count == 0:
            break
        adjust = diff / count
        # modify v for free indices
        v[free] -= adjust
        # clip again
        v = np.clip(v, 0, 1)
    return v

import time, math, random
import numpy as np

def _max_overlap(h: np.ndarray) -> float:
    """C5 value for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def _worst_shifts(h: np.ndarray) -> np.ndarray:
    """All integer shifts that attain the current maximum overlap."""
    n = h.shape[0]
    c = np.correlate(h, 1 - h, mode="full")
    mx = c.max()
    idx = np.where(c == mx)[0]
    # shift = index - (n-1)
    return idx - (n - 1)


def _neighbor_sum(h: np.ndarray, d: int) -> np.ndarray:
    """For each position i return h[i+d] + h[i-d] (out‑of‑range terms are 0)."""
    n = h.shape[0]
    ns = np.zeros_like(h)
    if d >= 0:
        ns[: n - d] += h[d:]
        ns[d:] += h[: n - d]
    else:
        dd = -d
        ns[dd:] += h[: n - dd]
        ns[: n - dd] += h[dd:]
    return ns


def _step_move(h: np.ndarray, d: int, delta: float) -> np.ndarray:
    """
    Transfer a small amount `delta` of mass from a low‑gradient position
    to a high‑gradient one for the current worst shift `d`.
    The total sum stays unchanged.
    """
    n = h.shape[0]
    ns = _neighbor_sum(h, d)

    inc_mask = h < 1.0 - 1e-12
    dec_mask = h > 1e-12
    inc_idx = np.where(inc_mask)[0]
    dec_idx = np.where(dec_mask)[0]

    if inc_idx.size == 0 or dec_idx.size == 0:
        return h.copy()

    # maximize the increase of the product term
    i_up = inc_idx[np.argmax(ns[inc_idx])]
    i_down = dec_idx[np.argmin(ns[dec_idx])]

    move = min(delta, 1.0 - h[i_up], h[i_down])
    if move <= 0.0:
        return h.copy()

    h_new = h.copy()
    h_new[i_up] += move
    h_new[i_down] -= move
    return h_new


def _project_cap_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection onto {x | 0<=x_i<=1, sum x_i = target_sum}.
    Simple water‑filling with a few iterations – sufficient for our size.
    """
    x = np.clip(v, 0.0, 1.0)
    for _ in range(50):
        cur = x.sum()
        diff = cur - target_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = x > 0.0
        else:
            free = x < 1.0
        cnt = free.sum()
        if cnt == 0:
            break
        adj = diff / cnt
        x[free] -= adj
        x = np.clip(x, 0.0, 1.0)
    return x


def _optimize(initial_h: np.ndarray, budget: float):
    """Heuristic optimiser returning a (near)‑optimal step function."""
    n = initial_h.shape[0]
    S = n / 2.0
    h = _project_cap_simplex(initial_h, S)

    cur_val = _max_overlap(h)
    best_h = h.copy()
    best_val = cur_val

    start = time.time()
    T0, Tend = 0.5, 1e-5          # temperature schedule
    delta = 0.03                  # basic transfer amount

    # ---- simulated annealing phase ----
    while time.time() - start < budget * 0.94:
        elapsed = time.time() - start
        T = T0 * (1 - elapsed / budget) + Tend * (elapsed / budget)

        # pick one of the worst shifts
        d = int(random.choice(_worst_shifts(h).tolist()))
        h_candidate = _step_move(h, d, delta)

        # if the move does nothing, try a random swap as a cheap perturbation
        if np.allclose(h_candidate, h):
            i, j = np.random.choice(n, 2, replace=False)
            move = min(delta, h[i], 1.0 - h[j])
            if move > 0:
                h_candidate = h.copy()
                h_candidate[i] -= move
                h_candidate[j] += move

        h_candidate = _project_cap_simplex(h_candidate, S)
        cand_val = _max_overlap(h_candidate)

        # Metropolis acceptance
        if cand_val < cur_val or random.random() < math.exp(-(cand_val - cur_val) / max(T, 1e-12)):
            h = h_candidate
            cur_val = cand_val

        # keep the global best encountered
        if cand_val < best_val:
            best_val = cand_val
            best_h = h_candidate.copy()

        # occasional step‑size reduction to fine‑tune
        if random.random() < 0.001:
            delta = max(delta * 0.9, 1e-4)

    # ---- deterministic local improvement ----
    for _ in range(3000):
        d = int(random.choice(_worst_shifts(best_h).tolist()))
        h2 = _step_move(best_h, d, delta * 0.5)
        h2 = _project_cap_simplex(h2, S)
        v2 = _max_overlap(h2)
        if v2 + 1e-12 < best_val:
            best_h, best_val = h2, v2

    return best_h, best_val


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the time budget."""
    random.seed(seed)
    np.random.seed(seed)

    # obtain a feasible starting point
    try:
        init = np.array(initial_h_values, dtype=float)  # provided by the harness
    except NameError:
        # fallback: uniform 0.5 (satisfies the integral exactly)
        init = np.full(70, 0.5, dtype=float)

    h_opt, bound = _optimize(init, budget_s)

    # final safety projection (numeric drift)
    h_opt = _project_cap_simplex(np.clip(h_opt, 0.0, 1.0), len(h_opt) / 2.0)

    return (h_opt.tolist(), float(bound), len(h_opt))
