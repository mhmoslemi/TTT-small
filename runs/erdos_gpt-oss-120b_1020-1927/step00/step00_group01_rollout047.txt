def gradient(h):
    corr = np.correlate(h, 1-h, mode='full')
    max_val = np.max(corr)
    idx_max = np.where(corr == max_val)[0]
    n = len(h)
    grad = np.zeros(n)
    for idx in idx_max:
        k = idx - (n - 1)  # lag
        if k >= 0:
            # positive lag L = k
            L = k
            # (1 - h[j+L]) term
            # for j where j+L < n
            grad[:n-L] += (1 - h[L:])  # j from 0..n-L-1; (1-h_{j+L})
            # -h_{j-L} term
            grad[L:] += -h[:n-L] # j from L..n-1; j-L from 0..n-L-1
        else:
            # negative lag = -L, i.e., shift left. Equivalent to lag L negative
            # correlation for negative k uses reversed direction; but we can treat same formula with k negative,
            # or just use symmetry: corr(-k) = sum_{i} h[i] * (1 - h[i - L])
            L = -k
            # For correlation with lag -L, sum over i = L..n-1: h[i] * (1 - h[i - L])
            # Derivative contributions:
            # For derivative w.r.t h_j when j = i (first factor) -> (1 - h_{j - L}) if j - L >=0
            # When j = i - L (second factor) -> -h_{j + L}
            # So:
            # term1: (1 - h[j-L]) for j>=L
            grad[L:] += (1 - h[:n-L])  # j = L..n-1, j-L = 0..n-L-1
            # term2: -h[j+L] for j+L < n => j <= n-L-1
            grad[:n-L] += -h[L:]  # j from 0..n-L-1
    grad = grad / len(idx_max) * dx
    return grad

    def project_onto_bounds_and_sum(v, target_sum):
        # water-filling: project onto [0,1] intersection with hyperplane sum = target_sum.
        # This works because we can solve for lambda such that sum(min(1, max(0, v_i + lambda))) = target_sum.
        # We'll use a simple bisection on lambda.
        lo = -1.0  # lower bound
        hi = 1.0   # upper bound
        # Might need wider bounds depending on v. Use range [-2,2]
        lo = -2.0
        hi = 2.0
        for _ in range(100):
            mid = (lo+hi)/2.0
            proj = np.clip(v + mid, 0.0, 1.0)
            s = proj.sum()
            if s < target_sum:
                lo = mid
            else:
                hi = mid
        return proj

    def make_feasible_random(n):
        # generate random vector and project
        v = np.random.rand(n)
        return project_onto_bounds_and_sum(v, n/2.0)

    # Objective and gradient functions for a given n (dx).
    def make_obj_and_grad(n):
        dx = 2.0 / n
        def objective(h):
            corr = np.correlate(h, 1-h, mode='full')
            return dx * np.max(corr)
        def gradient(h):
            corr = np.correlate(h, 1-h, mode='full')
            max_val = np.max(corr)
            idx_max = np.flatnonzero(corr == max_val)
            # Compute gradient contributions
            grad = np.zeros_like(h)
            half = n - 1
            for idx in idx_max:
                k = idx - half  # lag
                if k >= 0:
                    L = k
                    if L < n:

                        grad[:n-L] += (1 - h[L:])

                    # -h[j+L] for j = 0..n-1-L
                    grad[:n-L] += -h[L:]
            grad = grad / len(idx_max) * dx

    for n in n_candidates:

    # Return as required

            return dx * np.max(corr)

def _project(v, target):
    """Shift vector by a constant λ and clip to [0,1] so sum equals target."""
    lo, hi = -2.0, 2.0

        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)

            lo = mid
        else:
            hi = mid

    dx = 2.0 / n

        corr = np.correlate(h, 1 - h, mode='full')
