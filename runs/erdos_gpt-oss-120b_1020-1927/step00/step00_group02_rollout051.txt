def project_onto_capped_simplex(v, s, low=0., high=1., tol=1e-10):
    n = len(v)
    # Clip v to [low, high] initially to bound.
    # We need to find lambda such that sum(max(low, min(high, v - lambda))) = s.
    # Use binary search between low_lambda and high_lambda.
    lo = -np.max(v) - 10  # enough negative to make all high
    hi = np.max(v) + 10   # enough positive to make all low
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, low, high)
        total = proj.sum()
        if total > s:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    return np.clip(v - (lo+hi)/2.0, low, high)

def evaluate(h):
    n = h.shape[0]
    dx = 2.0 / n
    return dx * np.max(np.correlate(h, 1 - h, mode='full'))

def project_onto_capped_simplex(v, s, lo=0.0, hi=1.0, max_iter=100, tol=1e-10):
    v = np.clip(v, lo, hi)
    n = v.size
    # quick check
    total = v.sum()
    if abs(total - s) <= tol:
        return v
    # binary search for lambda
    lo_lambda = -np.max(v) - 10.0
    hi_lambda = np.max(v) + 10.0
    for _ in range(max_iter):
        mid = (lo_lambda + hi_lambda) / 2.0
        proj = np.clip(v - mid, lo, hi)
        total = proj.sum()
        if total > s:
            lo_lambda = mid
        else:
            hi_lambda = mid
        if hi_lambda - lo_lambda < tol:
            break
    return np.clip(v - (lo_lambda + hi_lambda) / 2.0, lo, hi)

def compute_gradient_maxlag(h):
    # returns gradient vector for h based on lag(s) achieving max correlation
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = corr.max()
    max_indices = np.where(corr == max_corr)[0]
    # compute signed lags
    lags = max_indices - (n - 1)
    g = np.zeros_like(h)
    idx = np.arange(n)
    # Iterate over lags
    for L in lags:
        # term_minus: h_{i - L}
        i_minus = idx - L
        mask_minus = (i_minus >= 0) & (i_minus < n)
        term_minus = np.zeros_like(h)
        term_minus[mask_minus] = h[i_minus[mask_minus]]
        # term_plus: h_{i + L}
        i_plus = idx + L
        mask_plus = (i_plus >= 0) & (i_plus < n)
        term_plus = np.zeros_like(h)
        term_plus[mask_plus] = h[i_plus[mask_plus]]
        g += 1.0 - term_minus - term_plus
    return g

import numpy as np
import time
import math

def evaluate(h: np.ndarray) -> float:
    """Return the C5 value for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return dx * np.max(np.correlate(h, 1.0 - h, mode="full"))


def project_onto_capped_simplex(
    v: np.ndarray, s: float, lo: float = 0.0, hi: float = 1.0,
    max_iter: int = 100, tol: float = 1e-10
) -> np.ndarray:
    """
    Project vector v onto the set {x : lo <= x_i <= hi, sum(x)=s}.
    Uses a simple binary‑search on the offset λ such that
    x_i = clamp(v_i - λ, lo, hi) and the sum constraint holds.
    """
    v = np.clip(v, lo, hi)
    total = v.sum()
    if abs(total - s) <= tol:
        return v

    # λ lower/upper bounds: far enough to saturate at hi/lo
    lo_lambda = -np.max(v) - 10.0
    hi_lambda = np.max(v) + 10.0
    for _ in range(max_iter):
        mid = (lo_lambda + hi_lambda) * 0.5
        proj = np.clip(v - mid, lo, hi)
        total = proj.sum()
        if total > s:
            lo_lambda = mid
        else:
            hi_lambda = mid
        if hi_lambda - lo_lambda < tol:
            break
    λ = (lo_lambda + hi_lambda) * 0.5
    return np.clip(v - λ, lo, hi)


def compute_gradient_maxlag(h: np.ndarray) -> np.ndarray:
    """
    Gradient of the *max* correlation lag(s) with respect to h.
    For each lag L attaining the maximal correlation we add
    ∂Corr_L/∂h_i = 1 - h_{i-L} - h_{i+L} (out‑of‑range terms treated as 0).
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = corr.max()
    max_idxs = np.where(corr == max_corr)[0]
    lags = max_idxs - (n - 1)          # signed lags
    g = np.zeros_like(h)
    idx = np.arange(n)

    for L in lags:
        i_minus = idx - L
        mask_minus = (i_minus >= 0) & (i_minus < n)
        term_minus = np.zeros_like(h)
        term_minus[mask_minus] = h[i_minus[mask_minus]]

        i_plus = idx + L
        mask_plus = (i_plus >= 0) & (i_plus < n)
        term_plus = np.zeros_like(h)
        term_plus[mask_plus] = h[i_plus[mask_plus]]

        g += 1.0 - term_minus - term_plus
    return g


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise a step function for the Erdős C₅ problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise from the provided construction (if any)
    # ------------------------------------------------------------------
    try:
        base = np.array(initial_h_values, dtype=np.float64)  # type: ignore
        n0 = base.size
        repeat_factor = 4                     # increase resolution
        h = np.repeat(base, repeat_factor).astype(np.float64)
    except Exception:
        # No supplied construction – start from a random feasible point
        n = 120
        h = rng.random(n)
        h = project_onto_capped_simplex(h, n / 2.0)

    n = h.size
    target_sum = n / 2.0

    # Small jitter to break the perfect repetition symmetry
    jitter = rng.uniform(-0.04, 0.04, size=n)
    h = np.clip(h + jitter, 0.0, 1.0)
    h = project_onto_capped_simplex(h, target_sum)

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing / gradient‑guided hill climbing
    # ------------------------------------------------------------------
    current_h = h.copy()
    best_h = h.copy()
    best_val = evaluate(best_h)
    current_val = best_val

    start_time = time.time()
    total_time = budget_s * 0.95               # keep a safety margin
    T0 = 5e-4                                   # initial temperature
    max_delta_factor_initial = 0.2
    no_improve = 0
    max_no_improve = 20000                     # restart after stagnation

    while time.time() - start_time < total_time:
        elapsed = time.time() - start_time
        T = T0 * max(0.0, 1.0 - elapsed / total_time)

        # --------------------------------------------------------------
        #   Choose a move: gradient‑guided (70% chance) or random pair
        # --------------------------------------------------------------
        use_grad = rng.random() < 0.7
        i_up = i_down = -1

        if use_grad:
            g = compute_gradient_maxlag(current_h)
            i_up = int(np.argmax(g))          # positive gradient → decrease
            i_down = int(np.argmin(g))        # negative gradient → increase
            if g[i_up] <= 0 and g[i_down] >= 0:
                use_grad = False               # no clear descent direction

        if not use_grad:
            # pick a random feasible donor and receiver
            donors = np.where(current_h > 1e-12)[0]
            receivers = np.where(current_h < 1.0 - 1e-12)[0]
            if donors.size == 0 or receivers.size == 0:
                break
            i_up = int(rng.choice(donors))
            i_down = int(rng.choice(receivers))
            if i_up == i_down:
                continue

        # --------------------------------------------------------------
        #   Transfer a small amount of mass from i_up to i_down
        # --------------------------------------------------------------
        max_delta = min(current_h[i_up], 1.0 - current_h[i_down])

        new_h = current_h.copy()
        new_h[i_up] -= delta
        new_h[i_down] += delta
        new_val = evaluate(new_h)

        # --------------------------------------------------------------
        #   Acceptance rule (always accept improvement, otherwise Metropolis)
        # --------------------------------------------------------------
        accept = False
        if new_val < current_val - 1e-12:
            accept = True
        elif T > 0.0 and rng.random() < math.exp((current_val - new_val) / T):
            accept = True

        if accept:
            current_h = new_h
            current_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = new_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # --------------------------------------------------------------
        #   Restart if we have been stuck for a while
        # --------------------------------------------------------------
        if no_improve >= max_no_improve:
            perturb = rng.uniform(-0.05, 0.05, size=n)
            cand = best_h + perturb
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_onto_capped_simplex(cand, target_sum)
            current_h = cand
            current_val = evaluate(current_h)
            no_improve = 0

    # ------------------------------------------------------------------
    # 3. Final projection & evaluation
    # ------------------------------------------------------------------
    best_h = project_onto_capped_simplex(best_h, target_sum)
    best_val = evaluate(best_h)

    return best_h.tolist(), float(best_val), int(n)
