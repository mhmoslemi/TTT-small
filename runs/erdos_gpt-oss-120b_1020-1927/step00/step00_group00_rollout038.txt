    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    # enforce bounds [0,1] and renorm sum
    h_new = np.clip(h_new, 0, 1)
    # adjust sum to n_new/2 using scaling of differences (but can't exceed bounds)
    # We'll simply scale total by factor to meet sum, then clip again, then adjust small correction.
    target_sum = n_new / 2.0
    current_sum = h_new.sum()
    if current_sum != 0:
        h_new = h_new * (target_sum / current_sum)

    # Rebalance to exact sum

            diff -= add

    iter = 0
    max_iter = 200000

    base_step = 0.05

        # determine worst shift
        cvec = np.correlate(h, 1-h, mode='full')
        cmax_idx = np.argmax(cvec)
        s = cmax_idx - (n - 1)
        # compute shifted h
        if s >= 0:
            h_shift = np.concatenate((np.zeros(s), h[:n-s]))
        else:
            h_shift = np.concatenate((h[-s:], np.zeros(-s)))
        # Compute candidate indices
        # pick a with large h_shift and h>eps
        # pick b with small h_shift and h<1-eps
        # Use random sampling to avoid deterministic loops
        # we can find top candidates using argpartition
        d = h_shift
        # indices where h>1e-6, else not.
        cand_a = np.where(h > 1e-6)[0]
        if len(cand_a) == 0:
            continue
        # among these choose a with max d
        a_idx = cand_a[np.argmax(d[cand_a])]
        # for b choose where h<1-1e-6
        cand_b = np.where(h < 1 - 1e-6)[0]
        if len(cand_b) == 0:
            continue
        b_idx = cand_b[np.argmin(d[cand_b])]
        # Compute diff in d
        diff = d[a_idx] - d[b_idx]
        if diff <= 0:
            # no improvement possible; maybe random move
            # random transfer
            i, j = np.random.choice(n, 2, replace=False)
            max_transfer = min(h[i], 1 - h[j])
            if max_transfer <= 0:
                continue
            delta = max_transfer * np.random.rand()
            h[i] -= delta
            h[j] += delta
            continue
        # propose move delta
        max_allowed = min(h[a_idx], 1 - h[b_idx], base_step)
        # also limit delta to guarantee positive improvement: need delta*diff * dx > 0.
        delta = max_allowed * np.random.rand()
        if delta <= 0:
            continue
        # apply
        h[a_idx] -= delta
        h[b_idx] += delta
        # compute new C5
        new_c5 = compute_c5(h)
        if new_c5 < best_c5 - 1e-10:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # revert with probability maybe
            # Accept if improves else revert
            # Let's revert for simplicity
            h[a_idx] += delta
            h[b_idx] -= delta
        # Occasionally random perturbations
        if iter % 5000 == 0:
            # random move to escape
            i, j = np.random.choice(n, 2, replace=False)
            max_transfer = min(h[i], 1 - h[j])
            delta = max_transfer * np.random.rand()
            h[i] -= delta
            h[j] += delta
            # maybe accept or revert if not improve
            new_c5 = compute_c5(h)
            if new_c5 < best_c5:

                best_h = h.copy()
            else:

    # final best_h may not exactly satisfy sum due to rounding; adjust

    target_sum = n/2.0

        for i in range(n):

            adj = np.sign(diff) * min(abs(diff), max_adj)

            diff -= adj

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def shifted_h(h, shift):
    n = len(h)
    if shift >= 0:
        return np.concatenate((np.zeros(shift), h[:n - shift]))
    else:
        shift = -shift
        return np.concatenate((h[shift:], np.zeros(shift)))

def move_worst_shift(h):
    n = len(h)
    # find worst shift (max C5)
    cvec = np.correlate(h, 1 - h, mode='full')

    d = h_shift
    # indices where we can move mass
    # choose a where d large and h>0
    cand_a = np.where((h > 1e-12) & (d > 1e-12))[0]
    cand_b = np.where((h < 1 - 1e-12) & (d < 1 - 1e-12))[0]

        new_h[i] -= delta
        new_h[j] += delta
        return new_h

import numpy as np
import time
import math

def _c5_value(h: np.ndarray) -> float:
    """return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid"""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate gives sum_i h[i]*(1‑h[i+lag]), exactly the required integral times dx
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)


def _shifted(h: np.ndarray, shift: int) -> np.ndarray:
    """zero‑padded shift of the vector (positive shift → right)"""
    n = h.shape[0]
    if shift >= 0:
        return np.concatenate((np.zeros(shift), h[: n - shift]))
    s = -shift
    return np.concatenate((h[s:], np.zeros(s)))


def _move_worst_shift(h: np.ndarray) -> np.ndarray:
    """move a little mass from a low‑overlap position to a high‑overlap one
    for the shift that currently attains the maximum overlap integral."""
    n = h.shape[0]
    # locate the shift giving the current worst (largest) C5
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (n - 1)

    # the shifted copy of h for that shift
    hs = _shifted(h, shift)

    # source positions: h positive but shifted value small (low overlap)
    src = np.where((h > 1e-12) & (hs < 1 - 1e-12))[0]
    # target positions: room to increase and shifted value large (high overlap)
    dst = np.where((h < 1 - 1e-12) & (hs > 1e-12))[0]

    if src.size == 0 or dst.size == 0:
        # fallback: simple random mass exchange
        a, b = np.random.choice(n, 2, replace=False)
        max_delta = min(h[a], 1 - h[b])
        if max_delta <= 0:
            return h
        delta = max_delta * np.random.rand()
        nh = h.copy()
        nh[a] -= delta
        nh[b] += delta
        return nh

    a = src[np.argmin(hs[src])]      # lowest shifted partner
    b = dst[np.argmax(hs[dst])]      # highest shifted partner
    if a == b:
        return h
    max_delta = min(h[a], 1 - h[b])
    if max_delta <= 0:
        return h
    delta = max_delta * np.random.rand()
    nh = h.copy()
    nh[a] -= delta
    nh[b] += delta
    return nh

    # ------------------------------------------------------------

    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:

                                         1.0 - h[i] if diff > 0 else h[i])
            h[i] += adjust
            diff -= adjust
            if abs(diff) < 1e-12:
                break
