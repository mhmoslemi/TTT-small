def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def solve_milp(N, time_limit):
    import cvxpy as cp, numpy as np
    m = N // 2
    h = cp.Variable(boolean=True, shape=N)
    maxC = cp.Variable()
    constraints = [cp.sum(h) == m]
    for k in range(1, N):
        length = N - k
        y = cp.Variable(boolean=True, shape=length)
        constraints += [y <= h[:length],
                        y <= h[k:],
                        y >= h[:length] + h[k:] - 1]
        Ck = cp.sum(h[:length]) - cp.sum(y)
        constraints.append(Ck <= maxC)
    constraints.append(maxC >= 0)
    prob = cp.Problem(cp.Minimize(maxC), constraints)
    # try solvers
    for solver in (cp.CBC, cp.GLPK_MI, cp.GLPK):
        try:
            prob.solve(solver=solver, verbose=False, max_seconds=time_limit)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                h_val = np.round(h.value).astype(int)
                maxC_val = maxC.value
                return h_val, maxC_val
        except (cp.error.SolverError, Exception):
            continue
    return None, None
```

def milp_solve(N):
    m = N // 2
    h = cp.Variable(N, boolean=True)
    maxC = cp.Variable()
    constraints = [cp.sum(h) == m, maxC >= 0]

    for k in range(1, N):
        length = N - k
        y = cp.Variable(length, boolean=True)
        # constraints linking y, h
        constraints.append(y <= h[:length])
        constraints.append(y <= h[k:])
        constraints.append(y >= h[:length] + h[k:] - 1)
        Ck = cp.sum(h[:length]) - cp.sum(y)

    prob = cp.Problem(cp.Minimize(maxC), constraints)
    # try solvers

                h_val = np.rint(h.value).astype(int)
                maxC_val = maxC.value
                return h_val, maxC_val
        except Exception:
            continue
    return None, None

import time
import random
import numpy as np
import cvxpy as cp

def compute_c5(h):
    """Integral bound for a step function h given as a vector."""
    h = np.asarray(h, dtype=float)
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def milp_solve(n):
    """Mixed‑integer linear formulation that minimises max_k Σh_i(1‑h_{i+k})."""
    half = n // 2
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()                     # maximal overlap count (integer)
    cons = [cp.sum(h) == half, t >= 0]

    for k in range(1, n):
        L = n - k
        y = cp.Variable(L, boolean=True)  # y[i] = h[i]·h[i+k]
        cons.append(y <= h[:L])
        cons.append(y <= h[k:])
        cons.append(y >= h[:L] + h[k:] - 1)
        Ck = cp.sum(h[:L]) - cp.sum(y)    # Σh_i – Σh_i h_{i+k}
        cons.append(Ck <= t)

    prob = cp.Problem(cp.Minimize(t), cons)

    # Try a few common solvers; ECOS_BB is almost always present.
    for solver in (cp.ECOS_BB, cp.CBC, cp.GLPK_MI, cp.GLPK):
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                h_val = np.rint(h.value).astype(int)
                t_val = t.value
                return h_val, t_val
        except Exception:
            continue
    return None, None


def local_search(start_h, start_score, time_budget):
    """Simple hill‑climbing (swap a 1 with a 0) respecting the integral constraint."""
    n = start_h.shape[0]
    best_h = start_h.copy()
    best_score = start_score
    cur_h = start_h.copy()
    ones = np.where(cur_h == 1)[0]
    zeros = np.where(cur_h == 0)[0]
    end = time.time() + time_budget
    while time.time() < end:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # perform the swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_score = compute_c5(cur_h)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_h = cur_h.copy()
            # update index lists after improvement
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
        else:
            # revert the swap
            cur_h[i] = 1
            cur_h[j] = 0
    return best_h, best_score

        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.shape[0]
        h0 = (h0 >= 0.5).astype(int)         # force binary
    else:
        n = 80
        h0 = np.zeros(n, dtype=int)
        h0[: n // 2] = 1
        np.random.shuffle(h0)

    # guarantee the integral condition exactly
    if h0.sum() != n // 2:
        idx = np.arange(n)
        np.random.shuffle(idx)
        h0[:] = 0
        h0[idx[: n // 2]] = 1

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    start = time.time()

    # ------------------------------------------------------------------
    # try a global MILP optimisation (if time permits)
    # ------------------------------------------------------------------
    if budget_s - (time.time() - start) > 30:
        h_opt, _ = milp_solve(n)
        if h_opt is not None:
            score_opt = compute_c5(h_opt)
            if score_opt < best_score - 1e-12:
                best_h, best_score = h_opt, score_opt

    # ------------------------------------------------------------------
    # finish with a quick randomized local search
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 2:
        # spend most of the remaining budget on local improvement
        best_h, best_score = local_search(
            best_h, best_score, time_budget=remaining * 0.9
        )

    # ------------------------------------------------------------------
    # final sanity checks and return
    # ------------------------------------------------------------------
    # ensure constraints are met (up to tiny numeric noise)
    assert abs(best_h.sum() - n // 2) <= 1e-9
    assert best_h.min() >= 0 and best_h.max() <= 1

    return (best_h.astype(float), float(best_score), int(n))
