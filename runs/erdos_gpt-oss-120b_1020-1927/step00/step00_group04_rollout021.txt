def smooth_objective(h, alpha=10.0):
    # h may be np array
    n = len(h)
    # compute autocorrelation h*h correlation (full)
    A_full = np.correlate(h, h, mode='full')
    # get non-negative shifts
    A = A_full[n-1:]   # length n
    # compute smooth min approx
    # we want f = (1/alpha) * logsumexp(-alpha * A)
    # note that for large alpha, -alpha* A is negative large numbers, we risk overflow but logsumexp handles.
    return (1.0/alpha) * logsumexp(-alpha * A)

def smooth_gradient(h, alpha=10.0):
    n = len(h)
    A_full = np.correlate(h, h, mode='full')
    A = A_full[n-1:]  # length n
    # compute softmax weights w_s = exp(-α A_s) / sum exp(-α A)
    # use logsumexp for stability
    log_exp = -alpha * A
    # subtract max for numerical stability
    max_log = np.max(log_exp)
    w_unnorm = np.exp(log_exp - max_log)
    denom = np.sum(w_unnorm)
    w = w_unnorm / denom
    # now compute gradient = - sum_s w_s * dA_s/dh
    grad = np.zeros_like(h)
    # shift s = 0
    ws = w[0]
    # derivative of A0 = 2 * h
    grad -= ws * 2.0 * h
    # shifts s = 1 .. n-1
    for s in range(1, n):
        ws = w[s]
        # for i = 0..n-1-s: gradient contribution -ws * h[i+s]
        grad[:n-s] -= ws * h[s:]
        # for i = s..n-1: gradient contribution -ws * h[i-s]
        grad[s:] -= ws * h[:n-s]
    return grad

def local_swap_hill_climb(h, max_iter, rng):
    # random swap of values (including possible nonbinary)
    n = len(h)
    best = h.copy()
    best_val = evaluate(best)
    for _ in range(max_iter):
        # choose two indices i,j
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # compute allowed delta: ensure 0 <= h[i] +/- delta <= 1 and 0 <= h[j] -/+ delta <= 1
        # we exchange some mass from j to i.
        # Choose delta = min(upper_i, lower_j) * rng.random()
        max_inc_i = 1.0 - best[i]
        max_dec_j = best[j]  # how much we can decrease
        max_dec_i = best[i]
        max_inc_j = 1.0 - best[j]
        # random direction: either move from j to i or i to j
        if rng.random() < 0.5:
            # transfer from j to i
            delta_max = min(max_inc_i, max_dec_j)
            if delta_max <= 0: continue
            delta = delta_max * rng.random()
            new = best.copy()
            new[i] += delta
            new[j] -= delta
        else:
            # transfer from i to j
            delta_max = min(max_inc_j, max_dec_i)
            if delta_max <= 0: continue
            delta = delta_max * rng.random()
            new = best.copy()
            new[i] -= delta
            new[j] += delta
        # compute value
        val = evaluate(new)
        if val < best_val - 1e-12:
            best = new
            best_val = val
    return best, best_val

def optimize_start(start_vec, rng, time_limit):
    n = len(start_vec)
    # first gradient descent using SLSQP
    # Define wrapper for objective and gradient that count evaluations? Not essential.
    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n/2,
                   'jac': lambda h: np.ones_like(h)})
    bounds = [(0.0, 1.0)] * n
    # Choose alpha maybe 20
    alpha = 15.0
    def fun(h):
        return (1.0/alpha) * logsumexp(-alpha * np.correlate(h, h, mode='full')[n-1:])
    def jac(h):
        # compute gradient as earlier using same alpha
        A_full = np.correlate(h, h, mode='full')
        A = A_full[n-1:]  # length n
        log_exp = -alpha * A
        max_log = np.max(log_exp)
        w_unnorm = np.exp(log_exp - max_log)
        denom = np.sum(w_unnorm)
        w = w_unnorm / denom
        grad = np.zeros_like(h)
        # shift s=0
        ws = w[0]
        grad -= ws * 2.0 * h
        for s in range(1, n):
            ws = w[s]
            grad[:n-s] -= ws * h[s:]
            grad[s:] -= ws * h[:n-s]
        return grad
    # Use minimize
    res = minimize(fun, start_vec, method='SLSQP', jac=jac,
                   bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})
    h_opt = res.x
    # Clip and project to ensure constraints exactly
    # Project onto sum and bounds
    h_opt = np.clip(h_opt, 0.0, 1.0)
    # To ensure sum exactly, project
    target = n/2
    h_opt = project_onto_sum_and_bounds(h_opt, target)
    # Evaluate
    best_val = evaluate(h_opt)
    # Local hill climb more refinement
    # define small number of iterations depending on remaining time
    start_time = time.time()
    # dynamic loop
    while time.time() - start_time < time_limit:
        # try random swap hill climbing
        h_opt, val = local_swap_hill_climb(h_opt, max_iter=200, rng=rng)
        if val < best_val:
            best_val = val
        else:
            # maybe break if no improvement after some attempts? we can loop anyway limited by time
            pass
    return h_opt, best_val

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_onto_sum_and_bounds(v, target):
    # bisection for lambda
    # v: arbitrary real numbers
    low = -np.max(v) - 2.0
    high = np.max(v) + 2.0
    for _ in range(50):
        mid = (low + high) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = mid
        else:
            high = mid
    w = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    # adjust small diff
    diff = target - w.sum()
    if abs(diff) > 1e-8:
        mask = (w > 0) & (w < 1)
        if np.any(mask):
            w[mask] += diff / np.sum(mask)
            w = np.clip(w, 0.0, 1.0)
    return w

def smooth_objective_and_grad(h, alpha):
    n = h.shape[0]
    # compute autocorrelation full
    A_full = np.correlate(h, h, mode='full')
    A = A_full[n-1:]   # length n
    # compute objective
    obj = (1.0/alpha) * logsumexp(-alpha * A)
    # gradient
    log_exp = -alpha * A
    max_log = np.max(log_exp)
    w_unnorm = np.exp(log_exp - max_log)
    denom = np.sum(w_unnorm)
    w = w_unnorm / denom
    grad = np.zeros_like(h)
    # shift s=0
    ws = w[0]
    grad -= ws * 2.0 * h
    for s in range(1, n):
        ws = w[s]
        # grad[i] for i=0..n-s-1 subtract ws * h[i+s]
        grad[:n-s] -= ws * h[s:]
        # grad[i] for i=s..n-1 subtract ws * h[i-s]
        grad[s:] -= ws * h[:n-s]
    return obj, grad

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ---------- evaluation ----------
def evaluate(h: np.ndarray) -> float:
    """max overlap integral for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


# ---------- projection onto simplex with box ----------
def project_to_sum_and_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1]^n while forcing sum(v)=target."""
    # bisection on lambda for the capped simplex
    low = -np.max(v) - 2.0
    high = np.max(v) + 2.0
    for _ in range(60):
        mid = (low + high) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            low = mid
        else:
            high = mid
    w = np.clip(v - (low + high) * 0.5, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-10:
        mask = (w > 0.0) & (w < 1.0)
        if np.any(mask):
            w[mask] += diff / mask.sum()
            w = np.clip(w, 0.0, 1.0)
    return w


# ---------- smooth objective (penalized) ----------
def _obj_penalized(h: np.ndarray, n: int, alpha: float, target: float, lam: float) -> float:
    """Smooth approximation of max overlap + quadratic penalty for sum constraint."""
    # autocorrelation part (full length, take non‑negative shifts)
    A = np.correlate(h, h, mode="full")[n - 1 :]
    smooth_min = (1.0 / alpha) * logsumexp(-alpha * A)
    penalty = lam * (np.sum(h) - target) ** 2
    return smooth_min + penalty


def _grad_penalized(
    h: np.ndarray, n: int, alpha: float, target: float, lam: float
) -> np.ndarray:
    """Gradient of the penalized smooth objective."""
    A_full = np.correlate(h, h, mode="full")
    A = A_full[n - 1 :]
    # softmax weights for the smooth min
    log_exp = -alpha * A
    max_log = np.max(log_exp)
    w_unnorm = np.exp(log_exp - max_log)
    denom = np.sum(w_unnorm)
    w = w_unnorm / denom

    grad = np.zeros_like(h)
    ws = w[0]
    grad -= ws * 2.0 * h
    for s in range(1, n):
        ws = w[s]
        grad[: n - s] -= ws * h[s:]
        grad[s:] -= ws * h[: n - s]

    # penalty gradient
    sum_diff = np.sum(h) - target
    grad += 2.0 * lam * sum_diff * np.ones_like(h)
    return grad


# ---------- local improvement ----------
def _local_transfer(h: np.ndarray, rng: np.random.Generator, time_budget: float) -> np.ndarray:
    """Simple random transfer moves keeping the sum constraint."""
    n = h.shape[0]
    target = n / 2.0
    best = h.copy()
    best_val = evaluate(best)
    start = time.time()
    while time.time() - start < time_budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = best[i], best[j]

        # move mass from j to i
        max_inc = 1.0 - hi
        max_dec = hj
        if max_inc > 0 and max_dec > 0:
            delta = rng.random() * min(max_inc, max_dec)
            cand = best.copy()
            cand[i] += delta
            cand[j] -= delta
            val = evaluate(cand)
            if val < best_val - 1e-12:
                best, best_val = cand, val
                continue

        # move mass from i to j
        max_inc = 1.0 - hj
        max_dec = hi
        if max_inc > 0 and max_dec > 0:
            delta = rng.random() * min(max_inc, max_dec)
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            val = evaluate(cand)
            if val < best_val - 1e-12:
                best, best_val = cand, val
                continue

        # occasional random jitter
        if rng.random() < 0.01:
            jitter = rng.uniform(-0.05, 0.05, size=n)
            cand = np.clip(best + jitter, 0.0, 1.0)
            cand = project_to_sum_and_bounds(cand, target)
            val = evaluate(cand)
            if val < best_val:
                best, best_val = cand, val
    return best


# ---------- optimisation from a start ----------
def _optimize_start(start_vec: np.ndarray, rng: np.random.Generator, total_time: float):
    """Run a penalised L‑BFGS‑B optimisation followed by cheap local moves."""
    n = start_vec.shape[0]
    target = n / 2.0
    alpha = 15.0          # smoothing parameter
    lam = 200.0           # penalty strength

    # ----- continuous L‑BFGS‑B -----
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        _obj_penalized,
        start_vec,
        method="L-BFGS-B",
        jac=_grad_penalized,
        bounds=bounds,
        args=(n, alpha, target, lam),
        options={"maxiter": 250, "ftol": 1e-9, "disp": False},
    )
    h_opt = np.clip(res.x, 0.0, 1.0)
    h_opt = project_to_sum_and_bounds(h_opt, target)

    # ----- cheap local tweaks -----
    remaining = max(0.0, total_time - (time.time() - _run_start_time))
    if remaining > 0.0:
        h_opt = _local_transfer(h_opt, rng, remaining)

    return h_opt, evaluate(h_opt)


# ---------- runner ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) with a hopefully low C₅."""
    global _run_start_time
    _run_start_time = time.time()
    rng = np.random.default_rng(seed)

    # base construction
    if "initial_h_values" in globals():
        base_h = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: random binary of length 42
        n0 = 42
        base_h = np.zeros(n0, dtype=float)
        base_h[rng.choice(n0, size=n0 // 2, replace=False)] = 1.0
    base_n = base_h.shape[0]
    base_h = project_to_sum_and_bounds(base_h, base_n / 2.0)

    # generate a modest list of start vectors (different resolutions)
    scale_factors = [1, 2, 3]  # duplicate the base pattern
    starts = []
    for f in scale_factors:
        n = base_n * f
        # duplicated pattern (preserves sum automatically)
        dup = np.repeat(base_h, f)
        starts.append(dup)

        # a couple of random binary starts
        for _ in range(2):
            v = np.zeros(n, dtype=float)
            v[rng.choice(n, size=n // 2, replace=False)] = 1.0
            starts.append(v)

        # a couple of random continuous starts
        for _ in range(2):
            v = rng.random(n)
            v = project_to_sum_and_bounds(v, n / 2.0)
            starts.append(v)

    best_h = None
    best_val = np.inf

    # give each start a modest slice of the remaining budget
    remaining_budget = budget_s
    for sv in starts:
        elapsed = time.time() - _run_start_time
        if elapsed >= budget_s - 1.0:
            break
        # allocate up to 0.6 s per start (or whatever is left)
        time_for_this = min(0.6, budget_s - elapsed)
        h_candidate, val = _optimize_start(sv, rng, time_for_this)
        if val < best_val:
            best_val, best_h = val, h_candidate

    # fallback if nothing succeeded
    if best_h is None:
        best_h = base_h
        best_val = evaluate(best_h)

    # final safety projection
    n_final = best_h.shape[0]
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum_and_bounds(best_h, n_final / 2.0)
    best_val = evaluate(best_h)

    return best_h, best_val, n_final
