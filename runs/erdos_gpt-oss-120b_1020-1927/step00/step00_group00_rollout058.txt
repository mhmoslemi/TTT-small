def simulated_annealing(h, time_budget):
    start = time.time()
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_h = h.copy()
    cur_score = best_score
    n = len(h)
    T0 = 0.05
    step_max = 0.4
      
    while time.time() - start < time_budget:
        # update temperature based on ratio of time left
        elapsed = time.time() - start
        t_frac = elapsed / time_budget
        T = T0 * (1 - t_frac) + 1e-6
        step = step_max * (1 - t_frac) + 1e-6
        # propose move
        i, j = np.random.randint(0, n, 2)
        # get allowed delta range
        lower = max(-cur_h[i], cur_h[j] - 1)
        upper = min(1 - cur_h[i], cur_h[j])
        if lower >= upper:
            continue
        d = lower + np.random.rand() * (upper - lower) * min(1, step) # Wait step scaling...
        # We want to restrict magnitude of d to approx step.
        # We can implement as d = np.clip(np.random.uniform(-step, step), lower, upper)
        d = np.clip((np.random.rand()*2-1)*step, lower, upper)
        new_h = cur_h.copy()
        new_h[i] += d
        new_h[j] -= d
        new_score = compute_c5(new_h)
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / T):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score
    return best_h, best_score
```

import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def project_to_constraints(h):
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # add diff to first free entry
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if idx.size:
            i = idx[0]
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    # final clip
    return np.clip(h, 0.0, 1.0)

def worst_lag_and_gradient(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)
    g = np.zeros(n, dtype=h.dtype)
    inds = np.arange(n)
    overlap = (inds + lag >= 0) & (inds + lag < n)
    g[overlap] = 1.0
    i = inds[overlap]
    g[i] -= h[i + lag]
    g[i + lag] -= h[i]
    return lag, g

def optimize_h(initial_h, time_budget, rng):
    n = initial_h.shape[0]
    cur_h = initial_h.copy()
    cur_h = project_to_constraints(cur_h)
    cur_score = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    start = time.time()
    while time.time() - start < time_budget:
        lag, grad = worst_lag_and_gradient(cur_h)
        pos = np.where(grad > 1e-12)[0]
        neg = np.where(grad < -1e-12)[0]
        if pos.size and neg.size:
            i = int(rng.choice(pos))
            j = int(rng.choice(neg))
        else:
            i, j = rng.choice(n, 2, replace=False)
        # step size decays
        t = (time.time() - start) / time_budget
        max_move = 0.5 * (1 - t) + 1e-8
        max_delta = min(max_move, cur_h[i], 1 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = rng.rand() * max_delta
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)
        if new_score < cur_score or rng.rand() < math.exp(-(new_score - cur_score) / (0.02 + 0.1 * t)):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()
        # occasionally random shake
        if rng.rand() < 0.005:
            a, b = rng.choice(n, 2, replace=False)
            lower = max(-cur_h[a], cur_h[b] - 1)
            upper = min(1 - cur_h[a], cur_h[b])
            if lower < upper:
                d = lower + rng.rand() * (upper - lower)
                new_h2 = cur_h.copy()
                new_h2[a] += d
                new_h2[b] -= d
                s2 = compute_c5(new_h2)
                if s2 < cur_score or rng.rand() < 0.5:
                    cur_h = new_h2
                    cur_score = s2
                    if s2 < best_score:
                        best_score = s2
                        best_h = new_h2.copy()
    return best_h, best_score

def upscale(h, factor):
    n = h.shape[0]
    new_n = int(n * factor)
    if new_n == n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, h)
    # keep within bounds
    new_h = np.clip(new_h, 0.0, 1.0)

    target = new_n / 2.0
    diff = target - new_h.sum()

import time, math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step function."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def project_to_constraints(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 exactly (up to eps)."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # adjust a single feasible entry
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if idx.size:
            i = idx[0]
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)


def worst_lag_and_gradient(h: np.ndarray):
    """Return lag attaining the maximal overlap and the gradient of that overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = int(np.argmax(corr))
    lag = max_idx - (n - 1)

    g = np.zeros(n, dtype=h.dtype)
    idx = np.arange(n)
    overlap = (idx + lag >= 0) & (idx + lag < n)
    g[overlap] = 1.0
    i = idx[overlap]
    g[i] -= h[i + lag]
    g[i + lag] -= h[i]
    return lag, g


def optimize_h(initial_h: np.ndarray, time_budget: float, rng: np.random.RandomState):
    """Local simulated‑annealing style optimisation."""
    n = initial_h.shape[0]
    cur_h = project_to_constraints(initial_h.copy())
    cur_score = compute_c5(cur_h)
    best_h, best_score = cur_h.copy(), cur_score

    start = time.time()
    while time.time() - start < time_budget:
        lag, grad = worst_lag_and_gradient(cur_h)

        pos = np.where(grad > 1e-12)[0]
        neg = np.where(grad < -1e-12)[0]
        if pos.size and neg.size:
            i = int(rng.choice(pos))
            j = int(rng.choice(neg))
        else:
            i, j = rng.choice(n, 2, replace=False)

        # decaying move size
        t = (time.time() - start) / time_budget
        max_move = 0.5 * (1.0 - t) + 1e-9
        max_delta = min(max_move, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = rng.rand() * max_delta

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = compute_c5(new_h)

        temp = max(0.05 * (1.0 - t), 1e-8)
        if new_score < cur_score or rng.rand() < math.exp(-(new_score - cur_score) / temp):
            cur_h, cur_score = new_h, new_score
            if new_score < best_score:
                best_h, best_score = new_h.copy(), new_score

        # occasional random shake to escape plateaus
        if rng.rand() < 0.005:
            a, b = rng.choice(n, 2, replace=False)
            lo = max(-cur_h[a], cur_h[b] - 1.0)
            hi = min(1.0 - cur_h[a], cur_h[b])
            if lo < hi:
                d = lo + rng.rand() * (hi - lo)
                new_h2 = cur_h.copy()
                new_h2[a] += d
                new_h2[b] -= d
                s2 = compute_c5(new_h2)
                if s2 < cur_score or rng.rand() < 0.5:
                    cur_h, cur_score = new_h2, s2
                    if s2 < best_score:
                        best_h, best_score = new_h2.copy(), s2
    return best_h, best_score


def upscale(h: np.ndarray, factor: float) -> np.ndarray:
    """Linear up‑sample by `factor` and adjust to keep the integral constraint."""
    n = h.shape[0]
    new_n = max(2, int(round(n * factor)))
    if new_n == n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, h)
    new_h = np.clip(new_h, 0.0, 1.0)

    target = new_n / 2.0
    diff = target - new_h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(new_h < 1.0)[0]
        else:
            idx = np.where(new_h > 0.0)[0]
        if idx.size:
            i = idx[0]
            new_h[i] = np.clip(new_h[i] + diff, 0.0, 1.0)
    return np.clip(new_h, 0.0, 1.0)

    # 2. First optimisation phase (original resolution)
    # ------------------------------------------------------------------
    start_all = time.time()
    phase1 = budget_s * 0.6

    # ------------------------------------------------------------------
    elapsed = time.time() - start_all
    remaining = budget_s - elapsed
    if remaining > 5.0:
        refined = upscale(best_h, 2.0)               # double the points

    # 4. Final projection and score computation
    # ------------------------------------------------------------------
    best_h = project_to_constraints(best_h)
