        idx = np.argsort(init_h)[::-1]

        n = 80

    # ------------------------------------------------------------------
    def compute_c5(h: np.ndarray) -> float:
        """Maximum overlap integral for a binary step function."""
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def random_binary() -> np.ndarray:

        h = np.zeros(n, dtype=int)
        h[rng.choice(n, size=half, replace=False)] = 1
        return h

    def perturb(h: np.ndarray, k: int) -> np.ndarray:

            if ones.size == 0 or zeros.size == 0:
                break

            j = rng.choice(zeros)
            h_new[i] = 0
            h_new[j] = 1
