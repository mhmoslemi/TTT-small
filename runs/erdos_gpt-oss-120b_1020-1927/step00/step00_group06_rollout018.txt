import time, math
import numpy as np

def _compute_c5(h, dx):
    """Return max_k ∫ h(x)(1−h(x+k))dx using discrete correlation."""
    # np.correlate gives sum_i h[i] * (1-h[i+k]) over overlapping indices
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def _random_feasible(N, rng):
    """Generate a random vector in [0,1]^N with sum exactly N/2."""
    target = N / 2.0
    # start from uniform values and shift to the correct mean
    v = rng.random(N)
    v = v - (v.mean() - 0.5)               # makes average close to 0.5
    v = np.clip(v, 0.0, 1.0)

    # adjust the sum by redistribution on free (non‑saturated) entries
    for _ in range(10):
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = v < 1.0
            if not free.any():
                break
            add = diff / free.sum()
            v[free] = np.minimum(v[free] + add, 1.0)
        else:
            free = v > 0.0
            if not free.any():
                break
            sub = (-diff) / free.sum()
            v[free] = np.maximum(v[free] - sub, 0.0)
    # tiny correction to enforce the sum precisely
    residual = target - v.sum()
    if abs(residual) > 1e-10:
        # adjust one random free entry
        idxs = np.where((v > 0) & (v < 1))[0]
        if idxs.size == 0:
            idxs = np.arange(N)
        i = rng.choice(idxs)
        v[i] = np.clip(v[i] + residual, 0.0, 1.0)
    return v
