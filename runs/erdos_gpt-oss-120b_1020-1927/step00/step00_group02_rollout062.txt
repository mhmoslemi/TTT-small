import time
import numpy as np

# ------------------------------------------------------------
# Helper: project onto the set {0<=h_i<=1, sum h_i = target}
def _project_box_simplex(v, target):
    """Euclidean projection onto box [0,1] with a linear sum constraint."""
    # Binary search for the Lagrange multiplier λ such that
    # sum_i clip(v_i - λ, 0, 1) = target
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


# ------------------------------------------------------------
# Compute C5 for a given step vector h
def _c5_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


# ------------------------------------------------------------
# Gradient of a single shift correlation term
def _grad_shift(h, shift):
    n = h.shape[0]
    grad = np.zeros_like(h)

    idx = np.arange(n)
    idx_plus = idx + shift
    mask_plus = (idx_plus >= 0) & (idx_plus < n)
    if mask_plus.any():
        grad[idx[mask_plus]] += (1.0 - h[idx_plus[mask_plus]])

    idx_minus = idx - shift
    mask_minus = (idx_minus >= 0) & (idx_minus < n)
    if mask_minus.any():
        grad[idx[mask_minus]] += -h[idx_minus[mask_minus]]
    return grad
