def compute_overlaps(h):
    corr = np.correlate(h, 1-h, mode='full')
    # correlation array length 2n-1, indices from -(n-1) to (n-1).
    # For shift k >=0, we need corr[n-1 + k] (starting at index n-1)
    return corr[n-1:]   # length n

def compute_c5(h):
    overlaps = compute_overlaps(h)
    dx = 2.0 / len(h)
    return np.max(overlaps) * dx

import numpy as np, time, math

def compute_overlaps(h):
   n = len(h)
   L = 2*n - 1
   # Fast using FFT
   H = np.fft.fft(h, L)
   rev = np.fft.fft((1-h)[::-1], L)
   conv = np.fft.ifft(H * rev).real
   return conv[n-1:]

def compute_c5(h):
   ov = compute_overlaps(h)
   return np.max(ov) * (2.0 / len(h))

def gradient(h, k):
   n = len(h)
   term1 = np.zeros(n)
   term2 = np.zeros(n)
   if k < n:
       term1[:n - k] = h[k:]
       term2[k:] = h[:n - k]
   g = 1 - term1 - term2
   return g

def step_limit(h, d):
   pos = d > 1e-12
   if pos.any():
       step_pos = np.min((1 - h[pos]) / d[pos])
   else:
       step_pos = np.inf
   neg = d < -1e-12
   if neg.any():
       step_neg = np.min((-h[neg]) / d[neg])
   else:
       step_neg = np.inf
   return min(step_pos, step_neg)

def project_sum(h, target):
    # Clip then equal adjust for sum
    h = np.clip(h, 0, 1)
    s = h.sum()
    diff = target - s
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        free = (h > 0) & (h < 1)
        nfree = free.sum()
        if nfree == 0:
            break
        delta = diff / nfree
        h[free] += delta
        h = np.clip(h, 0, 1)
        s = h.sum()
        diff = target - s
    return h

import time
import numpy as np

# ---------- utility functions ----------

def compute_overlaps(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h) for all non‑negative shifts."""
    n = h.shape[0]
    L = 2 * n - 1                      # length of linear convolution
    H = np.fft.fft(h, L)
    R = np.fft.fft((1.0 - h)[::-1], L)
    conv = np.fft.ifft(H * R).real
    return conv[n - 1:]                # shifts 0 … n‑1

def compute_c5_fft(h: np.ndarray) -> float:
    """C5 using the FFT based correlation (same as the official metric)."""
    return np.max(compute_overlaps(h)) * (2.0 / h.shape[0])

def compute_c5_exact(h: np.ndarray) -> float:
    """C5 using the same np.correlate that the judge uses."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr[n - 1:]) * (2.0 / n)

def gradient_wrt_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the overlap for a fixed shift k (ignoring bounds)."""
    n = h.shape[0]
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    if k < n:
        term1[: n - k] = h[k:]
        term2[k:] = h[: n - k]
    return 1.0 - term1 - term2          # derivative of S - sum_i h_i h_{i+k}

def step_limit(h: np.ndarray, d: np.ndarray) -> float:
    """Largest α≥0 such that 0≤h+αd≤1."""
    pos = d > 1e-12
    neg = d < -1e-12
    limit_pos = np.min((1.0 - h[pos]) / d[pos]) if np.any(pos) else np.inf
    limit_neg = np.min((-h[neg]) / d[neg]) if np.any(neg) else np.inf
    return min(limit_pos, limit_neg)

def project_to_sum_and_bounds(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the box [0,1]^n while keeping the total sum = target_sum."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        mask = (v > 0.0) & (v < 1.0)
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

# ---------- core optimizer ----------

def optimize_one_start(h_start: np.ndarray,
                       max_iters: int,
                       time_deadline: float,
                       start_time: float) -> tuple[np.ndarray, float]:
    """Run a simple descent that tries to push down the worst shift."""
    n = h_start.shape[0]
    target_sum = n / 2.0
    h = project_to_sum_and_bounds(h_start, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5_fft(h)

    alpha = 0.12                # initial step size
    stagnation = 0

    for it in range(max_iters):
        if time.time() - start_time > time_deadline:
            break

        overlaps = compute_overlaps(h)
        cur_c5 = np.max(overlaps) * (2.0 / n)

        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
            stagnation = 0
            alpha = min(alpha * 1.05, 0.25)
        else:
            stagnation += 1

        # worst shift
        k = int(np.argmax(overlaps))
        g = gradient_wrt_shift(h, k)
        lam = g.mean()                 # makes direction sum‑zero
        direction = -g + lam

        lim = step_limit(h, direction)
        if lim <= 0.0:                 # cannot move – try a tiny random swap
            i, j = np.random.choice(n, 2, replace=False)
            eps = 0.01 * np.random.rand()
            if h[i] + eps <= 1.0 and h[j] - eps >= 0.0:
                h[i] += eps
                h[j] -= eps
            continue

        step = min(alpha, 0.9 * lim)
        h_new = h + step * direction
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum_and_bounds(h_new, target_sum)

        new_c5 = compute_c5_fft(h_new)

        if new_c5 < cur_c5:
            h = h_new
            alpha = min(alpha * 1.05, 0.25)
        else:
            alpha *= 0.7
            # occasional random perturbation to escape plateaus
            if np.random.rand() < 0.04:
                i, j = np.random.choice(n, 2, replace=False)
                eps = 0.02 * np.random.rand()
                if h[i] + eps <= 1.0 and h[j] - eps >= 0.0:
                    h[i] += eps
                    h[j] -= eps

        if stagnation > 2500:
            break

    # final clean‑up pass (few random swaps)
    for _ in range(120):
        if time.time() - start_time > time_deadline:
            break
        i, j = np.random.choice(n, 2, replace=False)
        eps = (np.random.rand() - 0.5) * 0.03
        if h[i] + eps > 1.0 or h[i] + eps < 0.0:
            continue
        if h[j] - eps > 1.0 or h[j] - eps < 0.0:
            continue
        h_tmp = h.copy()
        h_tmp[i] += eps
        h_tmp[j] -= eps
        c5_tmp = compute_c5_fft(h_tmp)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h_tmp.copy()
            h = h_tmp

    return best_h, best_c5
