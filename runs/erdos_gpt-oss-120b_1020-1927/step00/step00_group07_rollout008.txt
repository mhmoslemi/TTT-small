def project_onto_sum_box(y, target):
    lo = -1.0
    hi = 1.0
    for _ in range(40):
        lam = (lo + hi) / 2.0
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    x = np.clip(y + (lo+hi)/2.0, 0.0, 1.0)
    # possibly adjust small error
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff over non-boundary elements if any
        idx = np.where((x > 0) & (x < 1))[0]
        if idx.size > 0:
            x[idx] += diff / idx.size
            x = np.clip(x, 0.0, 1.0)
    return x

    n = len(h)

    n = len(h)
    dx = 2.0/n
    best_h = h.copy()

        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        idx_max = np.argmax(corr)

import time, math
import numpy as np

def _compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate returns the full convolution (length 2n‑1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project_sum_box(y, target):
    """
    Project vector y onto the set {x | 0≤x_i≤1, Σx_i = target}.
    Uses a simple water‑filling (binary search on the shift λ).
    """
    lo, hi = -2.0, 2.0          # wide enough interval for the shift
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    x = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for rounding errors
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size:
            x[free] += diff / free.size
            x = np.clip(x, 0.0, 1.0)
    return x


def _subgradient_optimize(h, target, time_limit, start):
    """
    Deterministic sub‑gradient descent for a fixed amount of time.
    Returns the best vector found and its C5 value.
    """
    n = h.shape[0]
    best_h = h.copy()
    best_val = _compute_c5(best_h)
    cur = h.copy()
    cur_val = best_val
    step0 = 0.5

    while time.time() - start < time_limit:
        # correlation and worst lag
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        idx = np.argmax(corr)
        lag = idx - (n - 1)

        # gradient of the worst shift (∂/∂h_i a_k)
        grad = np.ones_like(cur)
        if lag > 0:
            grad[:-lag] -= cur[lag:]      # h_{i+lag}
            grad[lag:]  -= cur[:-lag]     # h_{i‑lag}
        elif lag < 0:
            k = -lag
            grad[k:]   -= cur[:-k]
            grad[:-k]  -= cur[k:]
        else:                               # lag == 0
            grad -= 2.0 * cur

        # decreasing step size (linear schedule)
        elapsed = time.time() - start
        t = (elapsed) / (time_limit - start)
        step = step0 * (1.0 - t) + 1e-6

        # take a step and project back onto the feasible set
        y = cur - step * grad
        cur = _project_sum_box(y, target)

        cur_val = _compute_c5(cur)
        if cur_val < best_val:
            best_val = cur_val
            best_h = cur.copy()

    return best_h, best_val, cur

    # ----- initialise h ----------------------------------------------------
    if 'initial_h_values' in globals():

        # fallback: 80 points, half ones, half zeros
        n0 = 80
        h0 = np.zeros(n0, dtype=float)
        h0[: n0 // 2] = 1.0
    n = h0.shape[0]
    target_sum = n / 2.0

    # enforce exact sum and box constraints
    h0 = np.clip(h0, 0.0, 1.0)
    diff = target_sum - h0.sum()
    if abs(diff) > 1e-12:
        # cheap correction on the first component that can absorb the error
        for i in range(n):
            adj = np.clip(h0[i] + diff, 0.0, 1.0) - h0[i]
            if abs(adj - diff) < 1e-12:
                h0[i] += adj
                break

    start = time.time()
    # ----- Phase 1: sub‑gradient descent ------------------------------------
    phase1_time = budget_s * 0.5
    best_h, best_val, cur_h = _subgradient_optimize(
        h0, target_sum, start + phase1_time, start)

    cur_val = _compute_c5(cur_h)

    # ----- Phase 2: simulated annealing ------------------------------------
    # remaining budget (a little margin left for safety)
    remaining = budget_s - (time.time() - start) - 0.1
    if remaining > 0.5:

        while time.time() - start < budget_s * 0.97:
            # linear temperature decay
            frac = (time.time() - start) / budget_s
            T = T0 * (1.0 - frac) + Tf * frac

            i, j = rng.choice(n, size=2, replace=False)

            # direction: move mass from j to i (or opposite) while staying feasible
            if rng.random() < 0.5:

                delta = rng.random() * max_delta
                new_i = cur_h[i] + delta
                new_j = cur_h[j] - delta
            else:

    # final safety projection (tiny numerical drift)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        for i in range(n):
            cand = best_h[i] + diff
            if 0.0 <= cand <= 1.0:
                best_h[i] = cand
                break

    final_val = _compute_c5(best_h)
    return best_h.tolist(), float(final_val), n
