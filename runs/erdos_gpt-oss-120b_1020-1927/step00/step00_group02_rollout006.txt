import time
import numpy as np

def _compute_c5(vec: np.ndarray) -> float:
    """Return the C5 value for a step function given by `vec`."""
    n = vec.shape[0]
    dx = 2.0 / n
    # correlation of h with (1‑h); np.correlate implements
    # sum_i h[i] * (1‑h[i+lag]) for each lag
    corr = np.correlate(vec, 1.0 - vec, mode="full") * dx
    return float(corr.max())

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h, its C5 bound and its number of points.

    The returned h satisfies 0 ≤ h[i] ≤ 1 and sum(h)*dx = 1.
    """
    start_time = time.time()
    rng = np.random.RandomState(seed)

    # ------------------------------------------------------------------
    # 1. Initial vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        # fallback: uniform 0.5 (guarantees the integral constraint)
        n_fallback = 200
        h = np.full(n_fallback, 0.5, dtype=np.float64)

    n = h.size
    target_sum = n / 2.0
    # enforce the integral constraint exactly (tiny numerical drift)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # Simple proportional scaling followed by a tiny correction
        scale = target_sum / cur_sum
        h = np.clip(h * scale, 0.0, 1.0)
        diff = target_sum - h.sum()
        # distribute the remaining diff to the first non‑saturated entry
        for i in range(n):
            lo = 0.0 - h[i]
            hi = 1.0 - h[i]
            if lo <= diff <= hi:
                h[i] += diff
                break

    # ------------------------------------------------------------------
    # 2. Helpers for proposing feasible moves
    # ------------------------------------------------------------------
    def _feasible_delta(i, j, delta):
        """Clip `delta` so that both h[i]+delta and h[j]-delta stay in [0,1]."""
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if lo > hi:
            return None
        delta = max(lo, min(hi, delta))
        return delta

    def _propose_swap(vec, step_scale):
        """Return a new vector obtained by moving a small amount from one
        coordinate to another while preserving the total sum."""
        i, j = rng.randint(0, n, size=2)
        while i == j:
            j = rng.randint(0, n)
        # random delta centred at zero, limited by step_scale
        delta = (rng.rand() - 0.5) * 2.0 * step_scale
        lo = max(-vec[i], vec[j] - 1.0)
        hi = min(1.0 - vec[i], vec[j])
        if lo >= hi:
            return None, None, None
        delta = max(lo, min(hi, delta))
        new_vec = vec.copy()
        new_vec[i] += delta
        new_vec[j] -= delta
        return new_vec, i, j

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing / hill‑climbing loop
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    # start from the same point for the current state
    cur_h = best_h.copy()
    cur_c5 = best_c5

    max_step = 0.3                     # maximum magnitude of a single transfer
    it = 0
    # schedule parameters
    decay_rate = 4e-4                  # exponential decay of step size
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 0.5:
            break

        # step size shrinks slowly as iteration count grows
        step_scale = max_step * np.exp(-decay_rate * it)

        cand_h, i, j = _propose_swap(cur_h, step_scale)
        if cand_h is None:
            it += 1
            continue

        cand_c5 = _compute_c5(cand_h)

        T = max(1e-5, 0.05 * (1.0 - elapsed / budget_s))

        accept = False
        if cand_c5 < cur_c5:
            accept = True
        else:
            # Metropolis acceptance
            if rng.rand() < np.exp(-(cand_c5 - cur_c5) / T):
                accept = True

        if accept:
            cur_h, cur_c5 = cand_h, cand_c5
            if cand_c5 < best_c5 - 1e-12:
                best_h, best_c5 = cand_h.copy(), cand_c5

        it += 1

    # 4. Final sanity check & tiny correction
    # ------------------------------------------------------------------
    # Ensure the sum constraint holds (tiny numerical errors may appear)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-9:

        free = np.where((best_h > 1e-12) & (best_h < 1.0 - 1e-12))[0]

            best_h[free] += inc

            for idx in range(n):

                hi = 1.0 - best_h[idx]
                if lo <= diff <= hi:
                    best_h[idx] += diff
                    break

    final_c5 = _compute_c5(best_h)

    # Return plain Python list for compatibility with the harness
    return (best_h.tolist(), final_c5, n)
