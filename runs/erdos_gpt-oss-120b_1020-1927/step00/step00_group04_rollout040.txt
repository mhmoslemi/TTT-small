    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2, alpha >= 0]  # alpha maybe free
    # For each shift k from 1 to n-1:

def _project_to_box_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector v onto the set {x | 0 <= x_i <= 1, sum x_i = target_sum}.
    Uses a simple bisection on the offset λ in x_i = clip(v_i + λ, 0, 1).
    """
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = w.sum() - target_sum
    if abs(diff) > 1e-10:
        free = (w > 1e-12) & (w < 1 - 1e-12)
        if free.any():
            w[free] -= diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w


def _c5_value(h: np.ndarray) -> float:
    """Return the Erdős C₅ value for step vector h."""
    try:
        # evaluate_erdos_solution is provided by the harness (if present)
        return evaluate_erdos_solution(h)
    except Exception:  # fallback to explicit computation
        n = h.shape[0]
        dx = 2.0 / n
        return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _max_c5_shift(h: np.ndarray, dx: float):
    """Return (max_value, shift_index) where the max occurs (full correlation)."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = int(np.argmax(corr))
    return corr[idx], idx


def _gradient_step(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Gradient of the C₅ term that attains the maximum.
    For shift k (positive or negative) the derivative is:
        ∂C₅/∂h_j = 1 - (h_{j+k} + h_{j‑k})   (missing terms omitted)
    """
    n = h.shape[0]
    if shift == 0:
        # derivative of sum_i h_i (1‑h_i) = 1 - 2 h_i
        return 1.0 - 2.0 * h
    grad = np.ones_like(h)
    if shift > 0:
        k = shift
        # terms where j+k is inside the array
        grad[: n - k] -= h[k:]
        # terms where j‑k is inside the array
        grad[k:] -= h[: n - k]
    else:  # shift < 0
        k = -shift
        grad[k:] -= h[: n - k]
        grad[: n - k] -= h[k:]
    return grad


def _optimize_start(h0: np.ndarray, target_sum: float,
                    max_iters: int, time_limit: float, start_time: float):
    """Projected gradient descent on the C₅ maximum (sub‑gradient method)."""
    n = h0.shape[0]
    dx = 2.0 / n
    h = _project_to_box_simplex(h0, target_sum)
    best_h = h.copy()
    best_val = _c5_value(h)

    # simple decay schedule for learning rate
    for it in range(max_iters):
        # current C₅ and the shift at which it is attained
        cur_val, idx = _max_c5_shift(h, dx)
        shift = idx - (n - 1)

        # sub‑gradient for that shift
        grad = _gradient_step(h, shift)

        # step size (decreasing)
        lr = 0.5 / (1.0 + it * 0.001)

        # take a step and re‑project
        h_new = _project_to_box_simplex(h - lr * grad, target_sum)

        new_val = _c5_value(h_new)
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()

        # periodic termination check
        if (it & 127) == 0 and (time.time() - start_time) > time_limit * 0.95:
            break

        h = h_new

    return best_h, best_val


# ----- main entry point ----------------------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] that yields a low
    Erdős C₅ bound. Returns (list_of_h, c5_value, n_points).
    """
    np.random.seed(seed)
    start = time.time()

    # decide discretisation size
    try:
        n_initial = len(initial_h_values)
    except Exception:
        n_initial = 0
    n = max(200, n_initial * 4)        # at least 200 points, more if a parent exists
    target_sum = n / 2.0                # required integral = 1

    # ---- generate starting vectors ---------------------------------------
    starts = []

    # parent if available (resample / project to current n)
    if n_initial:
        try:
            parent = np.asarray(initial_h_values, dtype=float)
            if parent.size != n:
                x_old = np.linspace(0.0, 2.0, parent.size, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n,       endpoint=False)
                parent = np.interp(x_new, x_old, parent)
            starts.append(_project_to_box_simplex(parent, target_sum))
        except Exception:
            pass

    # deterministic analytic seeds
    x = np.linspace(0.0, 2.0, n, endpoint=False)

    # inverted cosine: high at the ends, low in the centre (integral = 1)
    h_cos = (1.0 + np.cos(np.pi * x)) / 2.0
    starts.append(_project_to_box_simplex(h_cos, target_sum))

    # triangular “U” shape (peaks at the ends, zero in middle)
    h_tri = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    starts.append(_project_to_box_simplex(h_tri, target_sum))

    # a few random feasible starts
    for _ in range(5):
        rnd = np.random.rand(n)
        starts.append(_project_to_box_simplex(rnd, target_sum))

    # ---- optimisation loop ------------------------------------------------
    best_h = None
    best_c5 = float('inf')
    max_iters_per = 3000

    for h0 in starts:
        if (time.time() - start) > budget_s * 0.95:
            break
        h_opt, c5_opt = _optimize_start(
            h0, target_sum, max_iters_per, budget_s, start
        )
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # fallback (should never happen)
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_h = _project_to_box_simplex(best_h, target_sum)
        best_c5 = _c5_value(best_h)

    # final safety projection and evaluation
    best_h = _project_to_box_simplex(best_h, target_sum)
    best_c5 = _c5_value(best_h)

    return (best_h.tolist(), float(best_c5), int(n))
