import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return C5 = dx * max correlation of h with (1-h)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(corr))


def _min_product_shift(h: np.ndarray):
    """Find shift giving the minimal product sum (i.e. worst overlap)."""
    prod = np.correlate(h, h, mode="full")
    idx = int(np.argmin(prod))
    shift = idx - (h.shape[0] - 1)
    return shift, float(prod[idx])


def _neighbor_sum(h: np.ndarray, shift: int) -> np.ndarray:
    """For a given shift, return the vector of neighbour‑sums
    neigh[i] = h[i+shift] + h[i-shift] (terms outside the range are omitted)."""
    n = h.shape[0]
    neigh = np.zeros(n, dtype=float)
    if shift >= 0:
        # i+shift term
        neigh[: n - shift] += h[shift:]
        # i-shift term
        neigh[shift:] += h[: n - shift]
    else:
        s = -shift
        # i+shift = i-s
        neigh[s:] += h[: n - s]
        # i-shift = i+s
        neigh[: n - s] += h[s:]
    return neigh


def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust the vector (keeping bounds) so that its sum equals target."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Try to distribute the difference greedily.
    # Add to entries that are not at the upper bound,
    # subtract from entries that are not at the lower bound.
    if diff > 0:
        # need to increase total mass
        for i in np.argsort(-h):  # start from the smallest values
            cap = 1.0 - h[i]
            if cap <= 0:
                continue
            add = min(cap, diff)
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        # need to decrease total mass
        for i in np.argsort(h):  # start from the largest values
            cap = h[i]
            if cap <= 0:
                continue
            sub = min(cap, -diff)
            h[i] -= sub
            diff += sub
            if diff >= -1e-12:
                break
    return h

    np.random.seed(seed)
