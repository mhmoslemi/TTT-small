import numpy as np, random, time, math

def _project(vec, target_sum):
    """Project vec onto the box [0,1] and the hyperplane sum(vec)=target_sum."""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        w = np.clip(vec - lam, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(vec - (lo + hi) * 0.5, 0.0, 1.0)


def _c5(vec):
    """Return the max overlap integral for a given step vector."""
    N = len(vec)
    dx = 2.0 / N
    corr = np.correlate(vec, 1.0 - vec, mode="full") * dx
    return float(corr.max())


def _c5_and_shift(vec):
    """Return (max_value, shift_index, full_correlation_array)."""
    N = len(vec)
    dx = 2.0 / N
    corr = np.correlate(vec, 1.0 - vec, mode="full") * dx
    idx = int(np.argmax(corr))
    shift = idx - (N - 1)
    return float(corr[idx]), shift, corr

            if s_worst < N:
                g[: N - s_worst] = 1.0 - cur_h[s_worst:]          # ∂/∂h_i from the first factor
                g[s_worst:] -= cur_h[: N - s_worst]               # ∂/∂h_i from the second factor
        else:
            sa = -s_worst
            if sa < N:
                g[sa:] = 1.0 - cur_h[: N - sa]
                g[: N - sa] -= cur_h[sa:]

        # learning‑rate schedule
        frac = it / max_iters
        lr = base_lr * (1.0 - frac) * 0.5
        if lr < 1e-7:
            lr = 1e-7

        # step opposite to the gradient to *decrease* the maximal correlation
        v = cur_h - lr * g
        new_h = _project(v, target)
        new_c5 = _c5(new_h)

        # accept if improvement or with a small probabilistic rise (simulated annealing)
        if new_c5 < cur_c5 - 1e-12:
            cur_h, cur_c5 = new_h, new_c5
            no_improve = 0
        else:
            temp = 0.01 * (1.0 - frac) + 1e-5
            prob = math.exp(-(new_c5 - cur_c5) / temp) if new_c5 > cur_c5 else 1.0
            if random.random() < prob:
                cur_h, cur_c5 = new_h, new_c5
                no_improve = 0
            else:
                no_improve += 1

        # record global best
        if cur_c5 < best_c5:
            best_h, best_c5 = cur_h.copy(), cur_c5
            no_improve = 0

        it += 1

        # occasional random jitter to escape flat regions
        if it % 5000 == 0:
            noise = (np.random.rand(N) - 0.5) * 0.02
            cur_h = _project(cur_h + noise, target)
            cur_c5 = _c5(cur_h)

        # restart if stuck too long
        if no_improve > 20_000:
            x = (np.arange(N) + 0.5) * (2.0 / N)
            cur_h = np.sin(np.pi * x / 4.0) ** 2
            cur_h = _project(cur_h, target)
            cur_c5 = _c5(cur_h)
            no_improve = 0

    final_c5 = _c5(best_h)
    return (best_h, float(final_c5), N)
```
