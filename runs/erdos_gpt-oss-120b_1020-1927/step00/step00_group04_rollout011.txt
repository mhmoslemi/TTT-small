def max_corr_and_grad(h):
    # compute correlation
    conv = np.correlate(h, 1-h, mode='full')
    max_val = conv.max()
    # identify active shifts
    max_indices = np.where(conv == max_val)[0]  # indices
    # offsets
    offsets = max_indices - (len(h)-1)
    # compute gradient (for each active shift)
    n = len(h)
    grad = np.zeros(n)
    for d in offsets:
        # vectorized:
        idx = np.arange(n)
        idx_plus = idx + d
        idx_minus = idx - d
        term_plus = np.where((idx_plus >= 0) & (idx_plus < n), h[idx_plus], 0.0)
        term_minus = np.where((idx_minus >= 0) & (idx_minus < n), h[idx_minus], 0.0)
        grad += (1.0 - term_plus - term_minus)
    grad = grad / len(offsets)  # average
    return max_val, grad
```

def optimize_h(initial_h, max_iter=200, maxfun=2000):
    n = len(initial_h)
    target = n/2
    # define objective and gradient closure functions
    def fun(x):
        maxv = np.correlate(x, 1-x, mode='full').max()
        return maxv * (2.0/n)
    def jac(x):
        conv = np.correlate(x, 1-x, mode='full')
        maxv = conv.max()
        active_idx = np.where(conv == maxv)[0]
        offsets = active_idx - (n-1)
        grad = np.zeros(n)
        idx = np.arange(n)
        for d in offsets:
            idx_plus = idx + d
            idx_minus = idx - d
            term_plus = np.where((idx_plus >=0) & (idx_plus < n), x[idx_plus], 0.0)
            term_minus = np.where((idx_minus >=0) & (idx_minus < n), x[idx_minus], 0.0)
            grad += (1.0 - term_plus - term_minus)
        grad = grad / len(offsets)
        return grad * (2.0/n)
    cons = ({'type':'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones_like(x)})
    bounds = [(0.0, 1.0)] * n
    res = minimize(fun, initial_h, method='SLSQP', jac=jac, bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol':1e-12, 'disp':False})
    # After optimization, due to numeric errors, project to feasible
    h_opt = project_to_simplex_box(res.x) # ensure sum exactly target after rounding.
    return h_opt, fun(h_opt)

def local_search(h, iterations, step_scale):
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(best_h)
    for it in range(iterations):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # choose epsilon
        eps = (np.random.rand() - 0.5) * step_scale
        # adjust ensure not exceeding bounds:
        if eps > 0:
            eps = min(eps, 1 - best_h[j])
            eps = min(eps, best_h[i])  # mass to move from i to j cannot exceed h_i

        new_h[i] -= eps
        new_h[j] += eps
        # compute val
        val = compute_c5(new_h)
        if val < best_val:
            best_h = new_h
            best_val = val

import time
import numpy as np
from scipy.optimize import minimize

# ----- utility functions -----------------------------------------------------

def compute_c5(h):
    """Return C5 for a step function sampled in h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv.max() * dx


def max_corr_and_grad(h):
    """
    Return (C5, gradient) for the current h.
    Gradient is the sub‑gradient of the max correlation term.
    """
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    maxv = conv.max()
    # active shifts (tolerance for floating point ties)
    tol = 1e-12
    active = np.where(np.abs(conv - maxv) <= tol)[0]
    offsets = active - (n - 1)

    idx = np.arange(n)
    grad = np.zeros(n)
    for d in offsets:
        ip = idx + d
        im = idx - d
        term_p = np.where((ip >= 0) & (ip < n), h[ip], 0.0)
        term_m = np.where((im >= 0) & (im < n), h[im], 0.0)
        grad += (1.0 - term_p - term_m)
    grad = grad / len(offsets)

    dx = 2.0 / n
    return maxv * dx, grad * dx


def objective_func(h):
    """Objective for the optimizer (C5)."""
    val, _ = max_corr_and_grad(h)
    return val


def gradient_func(h):
    """Gradient of the objective (sub‑gradient)."""
    _, g = max_corr_and_grad(h)
    return g


def equality_constraint(h):
    """Equality constraint: integral = 1 (i.e. sum = n/2)."""
    n = h.shape[0]
    return np.sum(h) - n / 2.0


def equality_jacobian(h):
    """Jacobian of the equality constraint (vector of ones)."""
    return np.ones_like(h)


def project_to_box_simplex(x):
    """
    Project x onto the intersection of [0,1]^n and the hyperplane sum = n/2.
    Uses a simple bisection on the shift λ such that
        y = clip(x + λ, 0, 1)  and  sum(y) = target.
    """
    n = x.shape[0]
    target = n / 2.0
    low = -np.max(x) - 1.0
    high = 1.0 - np.min(x) + 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(x + (low + high) * 0.5, 0.0, 1.0)


def upscale_h(old_h, new_n):
    """Linearly interpolate old_h (length m) to new_n points on [0,2)."""
    m = old_h.shape[0]
    old_x = np.linspace(0.0, 2.0, m, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    return project_to_box_simplex(new_h)


def random_h(n):
    """Generate a random feasible h (sum = n/2, values in [0,1])."""
    rnd = np.random.rand(n)
    return project_to_box_simplex(rnd)


def optimize_h(start_h, max_iter=200):
    """
    Run SLSQP from start_h, respecting bounds and the integral constraint.
    Returns a feasible h and its C5 value.
    """
    n = start_h.shape[0]
    cons = {'type': 'eq',
            'fun': equality_constraint,
            'jac': equality_jacobian}
    bounds = [(0.0, 1.0)] * n
    opts = {'maxiter': max_iter, 'ftol': 1e-12, 'disp': False}
    res = minimize(objective_func, start_h, method='SLSQP',
                   jac=gradient_func, bounds=bounds,
                   constraints=[cons], options=opts)
    h_opt = project_to_box_simplex(res.x)
    return h_opt, compute_c5(h_opt)


def local_search(h, time_limit):
    """
    Simple hill‑climbing: move a small amount of mass between two positions.
    Runs until time_limit seconds have elapsed.
    """
    start = time.time()
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(best_h)
    step_scale = 0.2
    while time.time() - start < time_limit:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = (np.random.rand() - 0.5) * step_scale
        if eps > 0:
            eps = min(eps, best_h[i])          # cannot take more than present at i
            eps = min(eps, 1.0 - best_h[j])    # cannot exceed 1 at j
        else:
            eps = -eps
            eps = min(eps, best_h[j])
            eps = min(eps, 1.0 - best_h[i])
            eps = -eps
        if eps == 0.0:
            continue
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        val = compute_c5(cand)
        if val < best_val:
            best_h = cand
            best_val = val
            step_scale = max(step_scale * 0.99, 0.01)
    return best_h, best_val


# ----- main driver ---------------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    t0 = time.time()
    np.random.seed(seed)

    # --- initial construction ------------------------------------------------
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values, dtype=float)
    else:
        # simple fallback: half ones, half zeros (ensures integral = 1)
        n0 = 42
        h_start = np.concatenate([np.ones(n0 // 2), np.zeros(n0 - n0 // 2)])
    h_start = project_to_box_simplex(h_start)

    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    best_n = best_h.shape[0]

    # --- resolution schedule -------------------------------------------------
    schedule = [42, 84, 168, 336, 672]          # growing grids
    if best_n not in schedule:
        schedule = [best_n] + [s for s in schedule if s > best_n]

    # --- iterative refinement ------------------------------------------------
    for n in schedule:
        # time guard
        if time.time() - t0 > budget_s - 10:
            break

        # upscale if needed
        if n != best_n:
            cur_h = upscale_h(best_h, n)
        else:
            cur_h = best_h.copy()

        cur_h = project_to_box_simplex(cur_h)

        # number of random starts – modest for large n
        if n <= 84:
            starts = 12
        elif n <= 168:
            starts = 8
        else:
            starts = 5

        for s in range(starts):
            if time.time() - t0 > budget_s - 8:
                break
            if s == 0:
                init = cur_h.copy()
            elif s == 1:
                # small jitter of current best
                jitter = 0.03 * np.random.randn(n)
                init = project_to_box_simplex(cur_h + jitter)
            else:
                init = random_h(n)

            opt_h, opt_val = optimize_h(init, max_iter=150)

            if opt_val < best_c5 - 1e-7:
                best_h, best_c5, best_n = opt_h, opt_val, n

        # quick refinement on the current best at this resolution
        if time.time() - t0 < budget_s - 30:
            refined_h, refined_val = optimize_h(best_h, max_iter=200)
            if refined_val < best_c5 - 1e-7:
                best_h, best_c5 = refined_h, refined_val
                best_n = n

    # --- final local search (time‑bounded) -----------------------------------
    remaining = budget_s - (time.time() - t0) - 2
    if remaining > 0.5:
        best_h, best_c5 = local_search(best_h, remaining)

    # final sanity projection
    best_h = project_to_box_simplex(best_h)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_h.shape[0]
