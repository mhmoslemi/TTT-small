def project_to_feasible(h, n, tol=1e-9, max_iter=100):
    target = n/2
    for _ in range(max_iter):
        h = np.clip(h, 0, 1)
        current = h.sum()
        diff = target - current
        if abs(diff) < tol:
            break
        # Determine indices not at bounds
        free = (h > 0 + 1e-12) & (h < 1 - 1e-12)
        if free.any():
            # Distribute diff equally among free indices
            h[free] += diff / np.count_nonzero(free)
        else:
            # All at bounds (rare). Adjust by scaling (might push over)
            h = h * (target / current)
        # Ensure after adding diff, we might exceed bounds, so continue loop.
    # Final clip
    h = np.clip(h, 0, 1)
    # final adjust if sum still off due to rounding:
    # small residual, adjust first interior element if possible
    residual = target - h.sum()
    if abs(residual) > 1e-8:
        # attempt to add residual to a random non-bound index
        idx = np.where((h > 0) & (h < 1))[0]
        if idx.size > 0:
            i = idx[0]
            h[i] += residual
            h[i] = np.clip(h[i], 0, 1)
    return h

import time, math, numpy as np

# ---------------------------------------------------------------
# Helper functions ------------------------------------------------
# ---------------------------------------------------------------

def compute_c5(h):
    """Return the overlap constant C5 for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def project_to_feasible(h, target_sum):
    """
    Clip to [0,1] and adjust to satisfy sum(h) == target_sum.
    Simple iterative water‑filling.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        nfree = free.sum()
        if nfree == 0:          # all at a bound – scale everything
            h *= target_sum / cur
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / nfree
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual fix
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        idx = np.where((h > 0.0) & (h < 1.0))[0]
        if idx.size:
            h[idx[0]] = np.clip(h[idx[0]] + residual, 0.0, 1.0)
    return h


def random_feasible(rng, n):
    """Generate a random vector in [0,1]^n with exact sum n/2."""
    target = n / 2.0
    v = rng.random(n)
    v = project_to_feasible(v, target)
    return v


def upsample_to_n(arr, n):
    """Linear interpolation of a 1‑D array onto n equally spaced points."""
    m = len(arr)
    if m == n:
        return np.array(arr, dtype=float)
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, arr)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ---- grid size ------------------------------------------------
    n = 100                     # even – convenient for binary sum
    target_sum = n / 2.0
    dx = 2.0 / n

    # ---- initial guess --------------------------------------------
    if "initial_h_values" in globals() and initial_h_values is not None:
        h_cur = upsample_to_n(np.array(initial_h_values, dtype=float), n)
    else:
        h_cur = rng.random(n)
    h_cur = project_to_feasible(h_cur, target_sum)

    best_h = h_cur.copy()
    best_val = compute_c5(best_h)

    # ---- simulated‑annealing parameters ---------------------------
    start_time = time.time()
    max_iters = 3_000_000                    # used only for the cooling schedule
    it = 0
    T_start = 0.02
    T_end   = 5e-7

    while True:
        # time check
        if time.time() - start_time > budget_s * 0.96:
            break

        # temperature (geometric cooling)
        T = T_start * (T_end / T_start) ** (it / max_iters)

        # --- decide move type ------------------------------------------------
        if rng.random() < 0.85:               # generic two‑index mass‑swap
            i, j = rng.choice(n, size=2, replace=False)
            hi, hj = h_cur[i], h_cur[j]

            low  = max(-hi,     hj - 1.0)      # keep both in [0,1]
            high = min(1.0 - hi, hj)

            if low >= high:                     # no feasible delta, skip
                it += 1
                continue

            delta = rng.uniform(low, high)       # mass moved from j to i
            h_new = h_cur.copy()
            h_new[i] += delta
            h_new[j] -= delta

        else:                                 # targeted move on the worst shift
            # current correlation and worst lag
            corr_full = np.correlate(h_cur, 1.0 - h_cur, mode="full")
            lag = np.argmax(corr_full) - (n - 1)   # lag in [-n+1, n-1]

            if lag >= 0:
                # pairs (i, i+lag)
                idx = np.where((h_cur[: n - lag] > 0.0) &
                               (h_cur[lag:] < 1.0))[0]
                if idx.size == 0:
                    it += 1
                    continue
                i = idx[rng.integers(idx.size)]
                j = i + lag
            else:
                d = -lag
                # pairs (i+d, i)
                idx = np.where((h_cur[d:] > 0.0) &
                               (h_cur[: n - d] < 1.0))[0]
                if idx.size == 0:
                    it += 1
                    continue
                i = idx[rng.integers(idx.size)] + d
                j = i - d

            max_move = min(h_cur[i], 1.0 - h_cur[j])
            if max_move <= 1e-12:
                it += 1
                continue
            delta = rng.uniform(0.0, max_move)
            h_new = h_cur.copy()
            h_new[i] -= delta
            h_new[j] += delta

        # evaluate

        # Metropolis acceptance
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()

        if new_val < compute_c5(h_cur) or rng.random() < math.exp(-(new_val - compute_c5(h_cur)) / (T + 1e-12)):
            h_cur = h_new

    # final projection & value

    best_val = compute_c5(best_h)

    # optional final check with the provided evaluator (if any)

    except Exception:

    return best_h.tolist(), float(best_val), int(n)
