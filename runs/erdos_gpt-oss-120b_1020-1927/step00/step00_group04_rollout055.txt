def hill_climb(h_initial, max_iter, start_eps, end_eps, temp_start, temp_end):
    N = len(h_initial)
    target_sum = N/2
    h = h_initial.copy()
    best_h = h.copy()
    best_val = compute_c5_nd_raw(h)
    # schedule for epsilon and temperature:
    for it in range(max_iter):
        # linear schedule
        t = it / max_iter
        eps = start_eps * (1 - t) + end_eps * t
        T = temp_start * (1 - t) + temp_end * t
        # pick two distinct indices
        i, j = np.random.choice(N, size=2, replace=False)
        # propose epsilon shift from i to j (positive means moving mass from i to j)
        max_move_from_i = min(eps, h[i])  # can't go below 0
        max_move_to_i = min(eps, 1 - h[i]) # not used
        # we need epsilon <= both availability at i and capacity at j
        max_move = min(eps, h[i], 1 - h[j])
        # also can move opposite direction: from j to i
        max_move_rev = min(eps, h[j], 1 - h[i])
        # choose direction randomly
        if np.random.rand() < 0.5:
            move = max_move * (np.random.rand()) # random up to max_move
            if move == 0: continue
            new_h = h.copy()
            new_h[i] -= move
            new_h[j] += move
        else:
            move = max_move_rev * (np.random.rand())
            if move == 0: continue
            new_h = h.copy()
            new_h[i] += move
            new_h[j] -= move
        # evaluate
        val = compute_c5_nd_raw(new_h)
        delta = val - best_val
        if delta < 0:
            h = new_h
            best_h = new_h
            best_val = val
        else:
            # accept with probability exp(-delta/T)
            if T > 0 and np.random.rand() < np.exp(-delta / T):
                h = new_h
        # optionally track early stop
    return best_h, best_val

    corr = np.correlate(h, 1 - h, mode='full')

    def project_box_sum(v, target):

            s = proj.sum()

        max_corr = np.max(corr)
        dx = 2.0 / len(vals)
        return max_corr * dx

    # subgradient step function
    def subgrad_step(vals, step):
        N = len(vals)
        corr = np.correlate(vals, 1.0 - vals, mode='full')
        idx = np.argmax(corr)
        k = idx - (N - 1)
        g = np.zeros_like(vals)

        return new

    def random_swap(vals, eps, rng):

            proj = np.clip(v + mid, 0.0, 1.0)
            s = proj.sum()

            else:

    # ----------------------------------------------------------------------
    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * (2.0 / h.size)

    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------
    def subgrad_step(h, step):
        n = h.size
        corr = np.correlate(h, 1.0 - h, mode="full")

        g = np.zeros_like(h)

            g[:end] += (1.0 - h[k:])
            g[k:]  -= h[:end]

            g[m:]  += (1.0 - h[:end])
            g[:end] -= h[m:]
