import numpy as np
import time
from scipy.optimize import minimize

BETA = 20.0
LAMBDA = 1000.0

def compute_conv_max(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx, conv * dx

def actual_bound(h):
    # use preimported if exists
    try:
        return evaluate_erdos_solution(h)
    except Exception:
        val, _ = compute_conv_max(h)
        return val

def smooth_objective_and_grad(h):
    n = h.shape[0]
    dx = 2.0 / n
    # full convolution values scaled
    conv = np.correlate(h, 1 - h, mode='full') * dx
    max_c = np.max(conv)
    shifted = conv - max_c
    # stable exponent
    expb = np.exp(BETA * shifted)
    sum_expb = np.sum(expb)
    w = expb / sum_expb  # sum to 1

    # smooth max value
    smooth_max = max_c + (1.0 / BETA) * np.log(sum_expb)

    # penalty for sum constraint
    target = n * 0.5
    sum_h = np.sum(h)
    penalty = LAMBDA * (sum_h - target) ** 2

    obj = smooth_max + penalty

    # gradient
    # base derivative from smooth max part
    # gradient of sum w_i * ∂C_i/∂h
    # first term from ∂C_i/∂h includes constant 1 term (dx)
    base = dx * np.ones(n)

    # compute convolutions for shifted contributions
    # forward contribution: sum_i w_i * h_{s+i}
    conv_fwd = np.convolve(w, h, mode='full')
    conv_rev = np.convolve(w[::-1], h, mode='full')
    offset = n - 1
    contrib_fwd = conv_fwd[offset:offset + n]
    contrib_rev = conv_rev[offset:offset + n]

    grad = base - dx * (contrib_fwd + contrib_rev)

    # gradient of penalty term
    grad += 2.0 * LAMBDA * (sum_h - target) * np.ones(n)

    return obj, grad

def random_feasible_h(n, rng):
    target = n * 0.5
    # start with zeros
    h = np.zeros(n, dtype=float)
    remaining = target
    # Simple random fill
    while remaining > 1e-12:
        i = rng.integers(0, n)
        cap = 1.0 - h[i]
        if cap <= 0:
            continue
        delta = min(cap, remaining) * rng.random()
        h[i] += delta
        remaining -= delta
        # occasionally break to avoid infinite loop
        if remaining < 1e-12:
            break
    return h

def project_to_sum_and_bounds(h, target):
    # Simple iterative projection: first clip, then fix sum by adding/removing mass
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    if abs(diff) < 1e-10:
        return h
    # distribute diff over indices that are not at bounds
    # Determine indices with slack
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        # Choose random index that can absorb diff
        if diff > 0:
            candidates = np.where(h < 1.0)[0]
        else:
            candidates = np.where(h > 0.0)[0]
        if candidates.size == 0:
            break
        i = np.random.choice(candidates)
        capacity = (1.0 - h[i]) if diff > 0 else h[i]
        delta = min(abs(diff), capacity) * (1 if diff > 0 else -1)
        h[i] += delta
        diff -= delta
    return h

def objective_wrapper(x):
    obj, grad = smooth_objective_and_grad(x)
    return obj, grad

import numpy as np
import time
from scipy.optimize import minimize

# smoothing and penalty parameters (can be tuned)
BETA = 20.0          # smoothness for the max
LAMBDA = 1000.0      # penalty weight for the sum constraint


def _actual_bound(h):
    """Return the exact C5 bound for a step vector h."""
    try:
        # pre‑imported helper (may not exist in all environments)
        return evaluate_erdos_solution(h)          # type: ignore
    except Exception:
        n = h.shape[0]
        dx = 2.0 / n
        conv = np.correlate(h, 1.0 - h, mode="full")
        return np.max(conv) * dx


def _smooth_obj_grad(h):
    """
    Smooth approximation of the max overlap together with its gradient.
    Uses the soft‑max formulation:
        f(h) = (1/β)·log( Σ_k exp(β·C_k) ) + λ·(Σh – n/2)²
    where C_k = dx·∑_i h_i (1–h_{i+k}) .
    Returns (f, grad) .
    """
    n = h.shape[0]
    dx = 2.0 / n

    # C_k values (scaled by dx)
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(conv)
    shifted = conv - max_c                     # for numerical stability

    # soft‑max weights
    expb = np.exp(BETA * shifted)
    sum_expb = np.sum(expb)
    w = expb / sum_expb                        # Σ w = 1

    # smooth max term
    smooth_max = max_c + (1.0 / BETA) * np.log(sum_expb)

    # penalty for the linear constraint Σh = n/2
    target = n * 0.5
    sum_h = np.sum(h)
    penalty = LAMBDA * (sum_h - target) ** 2

    obj = smooth_max + penalty

    # ----- gradient -----
    # derivative of Σ w·C_k part:
    # ∂C_k/∂h_i = dx·(1 – (h_{i+k}+h_{i‑k}))
    base = dx * np.ones(n)                     # contribution of the constant 1

    # forward term Σ_i w_i·h_{s‑i}
    conv_fwd = np.convolve(w, h, mode="full")
    # backward term Σ_i w_i·h_{s+i}
    conv_rev = np.convolve(w[::-1], h, mode="full")
    off = n - 1
    contrib_fwd = conv_fwd[off:off + n]        # Σ_i w_i·h_{s‑i}
    contrib_rev = conv_rev[off:off + n]        # Σ_i w_i·h_{s+i}

    grad = base - dx * (contrib_fwd + contrib_rev)

    # gradient of the penalty λ·(Σh–target)²
    grad += 2.0 * LAMBDA * (sum_h - target) * np.ones(n)

    return obj, grad


def _obj_wrapper(x):
    """Wrapper returning (value, gradient) for scipy.optimize."""
    return _smooth_obj_grad(x)


def _project_to_sum_and_bounds(h, target):
    """
    Simple projection that respects 0≤h≤1 and Σh = target.
    First clips to the box, then distributes the remaining discrepancy
    (if any) onto non‑saturated coordinates.
    """
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target - cur
    if np.abs(diff) < 1e-12:
        return h

    # distribute diff greedily
    for _ in range(1000):
        if np.abs(diff) < 1e-12:
            break
        if diff > 0:
            cand = np.where(h < 1.0)[0]
        else:
            cand = np.where(h > 0.0)[0]
        if cand.size == 0:   # nothing can move any more
            break
        i = np.random.choice(cand)
        cap = (1.0 - h[i]) if diff > 0 else h[i]
        delta = min(abs(diff), cap) * (1.0 if diff > 0 else -1.0)
        h[i] += delta
        diff -= delta
    return h


def _random_feasible(n):
    """Generate a random vector in [0,1] with Σh = n/2."""
    target = n * 0.5
    h = np.zeros(n, dtype=float)
    remaining = target
    while remaining > 1e-12:
        i = np.random.randint(0, n)
        slack = 1.0 - h[i]
        if slack <= 0:
            continue
        delta = min(slack, remaining) * np.random.rand()
        h[i] += delta
        remaining -= delta
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] for the Erdos C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ----- choose discretisation size -----
    # Use a finer grid than the supplied 80‑point solution when possible.
    desired_n = 120                     # can be changed – must stay < 1000
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size == desired_n:
            h0 = init.copy()
        else:
            # linear interpolation onto the new grid
            old_x = np.linspace(0.0, 2.0, init.size, endpoint=False)
            new_x = np.linspace(0.0, 2.0, desired_n, endpoint=False)
            h0 = np.interp(new_x, old_x, init)
    else:
        h0 = _random_feasible(desired_n)

    target_sum = desired_n * 0.5
    h0 = _project_to_sum_and_bounds(h0, target_sum)

    best_h = h0.copy()
    best_score = _actual_bound(best_h)

    bounds = [(0.0, 1.0)] * desired_n
    max_iter_per_opt = 250

    # ----- main optimisation loop -----
    while time.time() - start_time < budget_s - 5.0:
        # start from current best with a small random perturbation
        pert = (np.random.rand(desired_n) - 0.5) * 0.2
        start_vec = best_h + pert
        start_vec = np.clip(start_vec, 0.0, 1.0)
        start_vec = _project_to_sum_and_bounds(start_vec, target_sum)

        res = minimize(_obj_wrapper, start_vec, method="L-BFGS-B",
                       jac=True, bounds=bounds,
                       options={"maxiter": max_iter_per_opt, "ftol": 1e-12, "disp": False})
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = _project_to_sum_and_bounds(h_opt, target_sum)

        score = _actual_bound(h_opt)
        if score < best_score:
            best_score = score
            best_h = h_opt.copy()
            # stop early if we already beat the target record
            if best_score <= 0.3808:
                break

        # ----- a few cheap local swaps to escape tiny plateaus -----
        for _ in range(15):
            i, j = np.random.randint(0, desired_n, size=2)
            if i == j:
                continue
            max_transfer = min(best_h[i], 1.0 - best_h[j])
            if max_transfer <= 0:
                continue
            delta = max_transfer * np.random.rand() * 0.5
            trial = best_h.copy()
            trial[i] -= delta
            trial[j] += delta
            s = _actual_bound(trial)
            if s < best_score:
                best_h = trial
                best_score = s
                if best_score <= 0.3808:
                    break

        # check remaining time
        if time.time() - start_time > budget_s - 2.0:
            break

    return (best_h.tolist(), float(best_score), desired_n)
