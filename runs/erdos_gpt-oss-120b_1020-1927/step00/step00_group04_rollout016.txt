def compute_c5(h):

    corr = np.correlate(h, 1 - h, mode='full') * dx

def project_onto_box_simplex(v, s):
    # this projects onto set {x in [0,1]^n: sum x = s}
    # using algorithm from Duchi et al. with clipping.
    # Steps:
    # 1. Clip v to [0,1] produce w.
    # 2. If sum(w) == s, return w.
    # 3. If sum(w) > s: we need to reduce some values >0.
    #    Solve projection onto simplex in [0,1] that may require removing mass from values >0.
    #    Equivalent to solving for lambda such that sum(clip(v - lambda, 0, 1)) = s.
    # Use bisection on lambda.
    n = len(v)
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0, 1)
        tot = w.sum()
        if tot > s:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0, 1)
```

import numpy as np, time, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def project_box_simplex(v, target):
    # Project onto [0,1] box and sum constraint.
    # Use bisection on lambda for clipping.
    # Ensure target between 0 and len(v)
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0, 1)
        total = w.sum()
        if total > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2.0, 0, 1)
    # Minor adjustment to meet target precisely via scaling of interior values.
    # Not strictly necessary.
    return w

def local_search(start_h, deadline, rng):
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    n = len(cur_h)
    start_time = time.time()
    T0 = 0.02  # initial temperature
    while time.time() < deadline:
        # time fraction
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 0.0
        T = max(1e-9, T0 * (1 - frac))
        # propose move
        i, j = rng.integers(0, n, size=2, replace=False)
        if rng.random() < 0.2:
            # random reassign one value, compensate the other
            new_i = rng.random()
            delta = new_i - cur_h[i]
            new_j = cur_h[j] - delta
        else:
            # small perturb
            step = 0.08
            delta = (rng.random() * 2 - 1) * step
            new_i = cur_h[i] + delta
            new_j = cur_h[j] - delta
        if not (0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0):
            continue
        new_h = cur_h.copy()
        new_h[i] = new_i
        new_h[j] = new_j
        new_c = compute_c5(new_h)
        if new_c < cur_c or rng.random() < math.exp(-(new_c - cur_c) / T):
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_h = new_h.copy()
                best_c = new_c
    return best_h, best_c

def upscale_h(h, new_n):

    # Ensure values within [0,1]

import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step function h."""
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives all integer shifts
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def project_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {x : 0<=x<=1, sum(x)=target}."""
    # Bisection on lambda for the mapping x_i = clip(v_i - λ, 0, 1)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return w


def upscale_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear periodic interpolation of h to a larger grid."""
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    # periodic interpolation (period = 2)
    return np.clip(np.interp(new_x, old_x, h, period=2.0), 0.0, 1.0)


def _gradient_step(h: np.ndarray, target_sum: float, lr: float) -> np.ndarray:
    """One gradient descent step for the active (worst) shift."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_idx = np.argmax(corr)
    s = max_idx - (n - 1)            # shift corresponding to the maximum C_k
    grad = np.ones(n)                # derivative of Σh_i part
    if s >= 0:
        if s < n:
            grad[: n - s] -= h[s:]          # h_{i+s}
            grad[s:] -= h[: n - s]          # h_{i-s}
    else:
        sp = -s
        if sp < n:
            grad[sp:] -= h[: n - sp]        # h_{i+s} with s<0
            grad[: n - sp] -= h[sp:]        # h_{i-s} with s<0
    grad *= dx
    new_h = h - lr * grad
    # project back to feasible set
    return project_box_simplex(new_h, target_sum)


def local_search(start_h: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple:
    """
    Simulated‑annealing / gradient hybrid local search.
    Returns the best (h, C5) found before `deadline`.
    """
    n = len(start_h)
    target = n / 2.0
    cur_h = start_h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    start_time = time.time()
    T0 = 0.02  # initial temperature for SA part

    while time.time() < deadline:
        # linear cooling schedule (relative to this run)
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 0.0
        T = max(1e-12, T0 * (1.0 - frac))

        if rng.random() < 0.3:   # gradient move
            lr = 0.3 * (1.0 - frac) + 0.02   # slowly shrink step size
            cand_h = _gradient_step(cur_h, target, lr)
        else:                     # random swap / small perturbation
            i, j = rng.integers(0, n, size=2, replace=False)
            if rng.random() < 0.2:
                # random reassignment of i, compensate at j
                new_i = rng.random()
                delta = new_i - cur_h[i]
                new_j = cur_h[j] - delta
            else:
                # tiny symmetric perturbation
                step = 0.08
                delta = (rng.random() * 2.0 - 1.0) * step
                new_i = cur_h[i] + delta
                new_j = cur_h[j] - delta
            if not (0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0):
                continue
            cand_h = cur_h.copy()
            cand_h[i] = new_i
            cand_h[j] = new_j

        cand_c = compute_c5(cand_h)
        # Metropolis acceptance
        if cand_c < cur_c or rng.random() < math.exp(-(cand_c - cur_c) / T):
            cur_h, cur_c = cand_h, cand_c
            if cand_c < best_c:
                best_h, best_c = cand_h.copy(), cand_c

    return best_h, best_c


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function improving the Erdős C₅ bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------

    for n_pts in resolutions:

            target = n_pts / 2.0

        # run a few independent local searches

        while time.time() < slice_deadline:
            # jitter a little to diversify starts
            cur = local_best_h.copy()
            for _ in range(3):

                delta = (rng.random() * 2.0 - 1.0) * 0.02

            if cand_c < local_best_c:

        # keep the best across resolutions
        if local_best_c < best_c:
            best_h, best_c = local_best_h, local_best_c

    # ------------------------------------------------------------------
    # final fine‑tuning on the current best until time runs out
    final_deadline = start_time + budget_s * 0.99
    n = len(best_h)
    target = n / 2.0
    while time.time() < final_deadline:
        i, j = rng.integers(0, n, size=2, replace=False)
        delta = (rng.random() * 2.0 - 1.0) * 0.01

        new_j = best_h[j] - delta

    # final projection to guarantee exact feasibility
    if abs(best_h.sum() - target) > 1e-9:
        best_h = project_box_simplex(best_h, target)
        best_c = compute_c5(best_h)

    n_points = len(best_h)
    return best_h, float(best_c), n_points
