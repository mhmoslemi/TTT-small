def simulated_annealing(h0, time_limit):
    start = time.time()
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    n = h.shape[0]
    # parameters
    max_step = 0.5  # max delta per move; maybe adjust adaptively
    # temperature schedule
    T0 = cur_score * 0.1  # initial
    Tf = cur_score * 1e-4
    it = 0
    
    while True:
        now = time.time()
        if now - start > time_limit:
            break

        # propose move
        i, j = np.random.randint(0, n, size=2)
        while i == j:
            j = np.random.randint(0, n)
        # compute allowed delta
        lower = max(-h[i], h[j] - 1.0)
        upper = min(1.0 - h[i], h[j])
        # bound step magnitude
        step = max_step
        low = max(lower, -step)
        high = min(upper, step)
        if low >= high:
            continue
        delta = np.random.uniform(low, high)

        # apply move
        h[i] += delta
        h[j] -= delta
        new_score = compute_c5(h)
        delta_score = new_score - cur_score

        # temperature schedule

                best_h = h.copy()
        else:

            h[j] += delta

        it += 1
        # optionally update max_step adaptively maybe decreasing

import time, math
import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k)) dx approximated on a grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _rand_two_move(h: np.ndarray, step: float):
    """Propose a move that changes two entries while keeping the sum unchanged."""
    n = h.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return None
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    lo = max(lo, -step)
    hi = min(hi, step)
    if lo >= hi:
        return None
    d = np.random.uniform(lo, hi)
    h[i] += d
    h[j] -= d
    return i, j, d

def _undo_two_move(h: np.ndarray, i: int, j: int, d: float):
    h[i] -= d
    h[j] += d

def _targeted_move(h: np.ndarray, lag: int, step: float):
    """Move mass between two overlapping pairs for a given lag."""
    n = h.shape[0]
    if lag == 0 or lag >= n:
        return None
    idx = np.arange(n - lag)

    # pairs we may increase
    mask_up = (h[idx] < 1.0) & (h[idx + lag] < 1.0)
    if not np.any(mask_up):
        return None
    cand_up = idx[mask_up]
    sum_up = h[cand_up] + h[cand_up + lag]
    i_up = cand_up[np.argmax(sum_up)]

    # pairs we may decrease
    mask_dn = (h[idx] > 0.0) & (h[idx + lag] > 0.0)
    if not np.any(mask_dn):
        return None
    cand_dn = idx[mask_dn]
    sum_dn = h[cand_dn] + h[cand_dn + lag]
    i_dn = cand_dn[np.argmin(sum_dn)]

    # allowable delta (same amount added to both elements of a pair)
    d_up = min(1.0 - h[i_up], 1.0 - h[i_up + lag])
    d_dn = min(h[i_dn], h[i_dn + lag])
    d_max = min(d_up, d_dn, step)
    if d_max <= 0.0:
        return None
    d = np.random.uniform(0.5 * d_max, d_max)

    h[i_up] += d
    h[i_up + lag] += d
    h[i_dn] -= d
    h[i_dn + lag] -= d
    return i_up, i_dn, lag, d

def _undo_targeted_move(h: np.ndarray, i_up: int, i_dn: int, lag: int, d: float):
    h[i_up] -= d
    h[i_up + lag] -= d
    h[i_dn] += d
    h[i_dn + lag] += d

    best_score = cur_score

    # ----- simulated‑annealing parameters ---------------------------------
    max_step = 0.3                         # maximum delta per move
    T = 0.02                               # initial temperature
    decay = 0.99995                        # geometric cooling
    it = 0
    last_improve = 0
    restart_after = 200000                # iterations without improvement

    # ----- main optimisation loop -----------------------------------------
    while True:
        if time.time() - start_time > budget_s * 0.96:
            break

        # lag that currently attains the maximal correlation (worst overlap)
        idx_max = int(np.argmax(cur_corr))
        lag = abs(idx_max - (n - 1))

        # decide move type
        if lag != 0 and np.random.rand() < 0.7:
            mv = _targeted_move(h, lag, max_step)
            if mv is None:
                mv = _rand_two_move(h, max_step)
                if mv is None:
                    continue
                i, j, d = mv
                is_target = False
            else:
                i_up, i_dn, lag_used, d = mv
                is_target = True
        else:
            mv = _rand_two_move(h, max_step)
            if mv is None:
                continue
            i, j, d = mv
            is_target = False

        # evaluate new state
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_score = float(np.max(new_corr) * (2.0 / n))
        delta = new_score - cur_score

        # accept / reject (simulated annealing)
        accept = False
        if delta < 0:
            accept = True
        else:
            if np.random.rand() < math.exp(-delta / max(T, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            cur_corr = new_corr
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_h = h.copy()
                last_improve = it
        else:
            # revert move
            if is_target:
                _undo_targeted_move(h, i_up, i_dn, lag_used, d)
            else:
                _undo_two_move(h, i, j, d)

        # schedule updates
        T = max(T * decay, 1e-6)
        if it % 50000 == 0 and it > 0:
            max_step = max(max_step * 0.9, 1e-3)

        # restart if stuck
        if it - last_improve > restart_after:
            # inject a burst of random large moves
            for _ in range(100):
                mv = _rand_two_move(h, max_step * 5)
                if mv is None:
                    continue
                i, j, d = mv
                new_corr = np.correlate(h, 1.0 - h, mode='full')
                new_score = float(np.max(new_corr) * (2.0 / n))
                if new_score < cur_score:
                    cur_score, cur_corr = new_score, new_corr
                else:
                    _undo_two_move(h, i, j, d)
            last_improve = it

        it += 1

    # ----- final clean‑up -------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-9:
        free = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
        ids = np.where(free)[0]
        if ids.size:
            adj = diff / ids.size
            best_h[ids] += adj
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            best_h[0] += diff
            best_h = np.clip(best_h, 0.0, 1.0)

    final_score = _c5_of(best_h)
    return best_h, final_score, n
