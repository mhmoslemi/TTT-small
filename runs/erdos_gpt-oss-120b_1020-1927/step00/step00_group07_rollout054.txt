        if k2 < N:
            g[k2:] += 1 - h[:N - k2]
            g[:N - k2] -= h[k2:]
    return g

def project_onto_box_simplex(y, target):
    # Using binary search lambda.
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # After binary search, due to rounding may not be exactly target, we can adjust a small amount.
    # We'll adjust by distributing residual onto interior points.
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # Add diff uniformly to non-bounded elements (where 0 < x <1)
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            # Clip again
            x = np.clip(x, 0.0, 1.0)
        else:
            # All at bounds, shift small random.
            pass
    return x

def optimize_one_restart(N, init_h, max_iters, time_limit, start_time):
    h = init_h.copy()
    target = N / 2.0
    best_h = h.copy()
    best_score, _, _ = compute_c5(h)
    # Local improvements
    steps = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001]
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        maxc_val = corr[max_idx]
        shift = max_idx - (N - 1)
        # gradient
        g = gradient_for_shift(h, shift)
        # try steps
        improved = False
        for α in steps:
            y = h - α * g
            y = project_onto_box_simplex(y, target)
            score, _, _ = compute_c5(y)
            if score < best_score - 1e-12:
                best_score = score
                best_h = y.copy()
                h = y
                improved = True
                break
        if not improved:
            # random perturb if stuck
            # pick random i,j and move delta
            i, j = np.random.choice(N, 2, replace=False)
            # max delta such that stays within [0,1]
            max_delta = min(1.0 - h[i], h[j])

                if score < best_score - 1e-12:
                    best_score = score
                    best_h = y.copy()

    return best_h, best_score

def gradient_for_shift(h, shift):
    N = h.shape[0]
    g = np.zeros(N)
    if shift >= 0:
        k = shift
        if k < N:
            g[:N - k] += 1 - h[k:]   # term1
            g[k:] -= h[:N - k]       # term2
    else:
        k2 = -shift
        if k2 < N:
            g[k2:] += 1 - h[:N - k2]
            g[:N - k2] -= h[k2:]
    return g

import time
import random
import numpy as np

def compute_c5(h):
    """Return C5 = dx * max correlation, the raw max, and the full correlation."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = np.max(corr)
    return maxc * dx, maxc, corr

def gradient_for_shift(h, shift):
    """Subgradient of the correlation at the given shift (maximising shift)."""
    N = h.size
    g = np.ones(N, dtype=np.float64)   # derivative of Σ h_i
    if shift >= 0:
        k = shift
        if k < N:
            g[:N - k] -= h[k:]            # subtract h_{i+k}
            g[k:]     -= h[:N - k]        # subtract h_{i-k}
    else:
        k = -shift
        if k < N:
            g[k:]     -= h[:N - k]        # subtract h_{i-k}
            g[:N - k] -= h[k:]            # subtract h_{i+k}
    return g

def project_onto_box_simplex(y, target, eps=1e-12):
    """
    Euclidean projection of y onto {x | 0≤x_i≤1, Σx_i = target}.
    Uses a binary search on the Lagrange multiplier λ.
    """
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    # adjust tiny residual to hit the target exactly
    diff = target - x.sum()
    if abs(diff) > eps:
        mask = (x > eps) & (x < 1.0 - eps)
        if mask.any():
            add = diff / mask.sum()
            x[mask] += add
            x = np.clip(x, 0.0, 1.0)
    return x

def gradient_descent_opt(h, max_iters, deadline, steps, penalty_lambda):
    N = h.size
    target = N * 0.5
    best_h = h.copy()
    best_score, _, _ = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # current worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (N - 1)
        grad = gradient_for_shift(cur_h, shift)
        if penalty_lambda != 0.0:
            grad += penalty_lambda * (1.0 - 2.0 * cur_h)
        improved = False
        for a in steps:
            cand = cur_h - a * grad
            cand = project_onto_box_simplex(cand, target)
            cand_score, _, _ = compute_c5(cand)
            if cand_score < cur_score - 1e-14:
                cur_h, cur_score = cand, cand_score
                improved = True
                break
        if not improved:
            # random mass transfer between two positions
            i, j = np.random.choice(N, 2, replace=False)
            delta = min(1.0 - cur_h[i], cur_h[j])
            if delta > 0:
                mag = delta * np.random.rand() * 0.5
                cand = cur_h.copy()
                cand[i] += mag
                cand[j] -= mag
                cand = project_onto_box_simplex(cand, target)
                cand_score, _, _ = compute_c5(cand)
                if cand_score < cur_score - 1e-14:
                    cur_h, cur_score = cand, cand_score
        if cur_score < best_score:
            best_h, best_score = cur_h.copy(), cur_score
    return best_h, best_score

def simulated_annealing_binary(N, init_h, max_iters, deadline):
    """Binary (0/1) hill‑climb with a simple annealing schedule."""
    target = N * 0.5
    h = init_h.copy()
    cur_score, _, _ = compute_c5(h)
    best_h, best_score = h.copy(), cur_score
    T0 = max(1e-3, cur_score * 0.2)          # initial temperature
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # pick a 1 and a 0 to swap
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_score, _, _ = compute_c5(h_new)
        if new_score < cur_score:
            h, cur_score = h_new, new_score
            if new_score < best_score:
                best_h, best_score = h_new.copy(), new_score
        else:
            # temperature linearly cools to 0.01·T0
            T = T0 * max(0.01, 1.0 - it / max_iters)
            if T > 0.0 and np.exp(-(new_score - cur_score) / T) > np.random.rand():
                h, cur_score = h_new, new_score
    return best_h, best_score

    best_h, best_score = h0.copy(), compute_c5(h0)[0]

    # ---- gradient descent from the (possibly) smooth start ----------
    steps = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001]
    gd_h, gd_score = gradient_descent_opt(
        h0, max_iters=5000, deadline=deadline, steps=steps, penalty_lambda=0.5)
    if gd_score < best_score:
        best_h, best_score = gd_h, gd_score

    # ---- binary simulated annealing restarts -------------------------
    # allocate remaining budget roughly equally among a handful of restarts
    remaining = deadline - time.time()
    if remaining > 10.0:
        restarts = 12
        it_per_restart = 3000
        for _ in range(restarts):
            if time.time() > deadline - 5.0:
                break
            # random binary start
            rnd = np.random.rand(N)
            bin_h = project_onto_box_simplex(rnd, N * 0.5)
            bin_h = (bin_h > 0.5).astype(float)   # force exact 0/1
            bin_h = project_onto_box_simplex(bin_h, N * 0.5)  # ensure exact count
            sa_h, sa_score = simulated_annealing_binary(N, bin_h, it_per_restart, deadline)
            if sa_score < best_score:
                best_h, best_score = sa_h, sa_score

    # ---- final fine‑tuned gradient descent ---------------------------
    if time.time() < deadline - 5.0:
        fine_h, fine_score = gradient_descent_opt(
            best_h, max_iters=2000, deadline=deadline,
            steps=[0.02, 0.01, 0.005, 0.001, 0.0005],
            penalty_lambda=0.3)
        if fine_score < best_score:
            best_h, best_score = fine_h, fine_score

    # ensure feasibility exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_simplex(best_h, N * 0.5)
    best_score = compute_c5(best_h)[0]

    return best_h, best_score, N
