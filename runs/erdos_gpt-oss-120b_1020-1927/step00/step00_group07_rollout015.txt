import time, math
import numpy as np

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a binary step function h on [0,2] with ∫h = 1 that
    minimizes   C5 = max_k ∫ h(x)·(1-h(x+k)) dx .
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Initialise a feasible binary vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        # Use the supplied construction as a starting point.
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        target_sum = n // 2
        # Convert to binary (threshold) and enforce exact sum.
        h = (init > 0.5).astype(float)
        cur_sum = int(h.sum())
        if cur_sum < target_sum:
            zeros = np.flatnonzero(h == 0)
            pick = rng.choice(zeros, target_sum - cur_sum, replace=False)
            h[pick] = 1.0
        elif cur_sum > target_sum:
            ones = np.flatnonzero(h == 1)
            pick = rng.choice(ones, cur_sum - target_sum, replace=False)
            h[pick] = 0.0
    else:
        # No parent solution – start from a random binary vector.
        n = 80
        target_sum = n // 2
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, target_sum, replace=False)] = 1.0

    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Scoring routine
    # ------------------------------------------------------------------
    def c5_score(arr: np.ndarray) -> float:
        """Return the C5 value for a binary array `arr`."""
        return np.max(np.correlate(arr, 1.0 - arr, mode="full")) * dx

    cur_score = c5_score(h)
    best_h = h.copy()
    best_score = cur_score

    # Maintain index lists for fast sampling.
    ones_idx = np.flatnonzero(h == 1)
    zeros_idx = np.flatnonzero(h == 0)

    # ------------------------------------------------------------------
    # Simulated annealing / hill climbing
    # ------------------------------------------------------------------
    T = 0.02                 # initial temperature
    cooling = 0.99995        # exponential cooling factor
    max_no_improve = 5000    # restart after this many non‑improving steps
    no_improve = 0
    iteration = 0

    while time.time() - start < budget_s:
        iteration += 1
        # Cool down temperature.
        T = max(1e-6, T * cooling)

        # Choose how many bits to swap (mostly 1, sometimes larger).
        if rng.random() < 0.1:
            max_swap = min(10, len(ones_idx), len(zeros_idx))
            t = rng.integers(2, max_swap + 1)
        else:
            t = 1

        # Randomly select indices to flip.
        if len(ones_idx) < t or len(zeros_idx) < t:
            continue
        swap_one = rng.choice(ones_idx, t, replace=False)
        swap_zero = rng.choice(zeros_idx, t, replace=False)

        new_h = h.copy()
        new_h[swap_one] = 0.0
        new_h[swap_zero] = 1.0

        new_score = c5_score(new_h)
        delta = new_score - cur_score

        # Metropolis acceptance.
        if delta < -1e-12 or rng.random() < math.exp(-delta / T):
            h = new_h
            cur_score = new_score
            # Update index lists.
            ones_idx = np.flatnonzero(h == 1)
            zeros_idx = np.flatnonzero(h == 0)

        # Track the global best.
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_h = h.copy()
            no_improve = 0
            # Early exit if we already beat the target bound.
            if best_score <= 0.3808:
                break
        else:
            no_improve += 1

        # Restart if stuck.
        if no_improve >= max_no_improve:
            h = np.zeros(n, dtype=float)
            h[rng.choice(n, n // 2, replace=False)] = 1.0
            cur_score = c5_score(h)
            ones_idx = np.flatnonzero(h == 1)
            zeros_idx = np.flatnonzero(h == 0)
            no_improve = 0

    # ------------------------------------------------------------------
    # Final greedy refinement (pure hill climbing)
    # ------------------------------------------------------------------
    # Use the best solution found so far as the starting point.
    h = best_h.copy()
    cur_score = best_score
    ones_idx = np.flatnonzero(h == 1)
    zeros_idx = np.flatnonzero(h == 0)

    while time.time() - start < budget_s:
        improved = False
        for _ in range(500):   # limited attempts per refinement round
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
            swap_one = rng.choice(ones_idx, 1)[0]
            swap_zero = rng.choice(zeros_idx, 1)[0]
            h_candidate = h.copy()
            h_candidate[swap_one] = 0.0
            h_candidate[swap_zero] = 1.0
            cand_score = c5_score(h_candidate)
            if cand_score < cur_score - 1e-12:

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_h = h.copy()
            if best_score <= 0.3808:
                break

    # ------------------------------------------------------------------
    # Return the solution.
    # ------------------------------------------------------------------
    return best_h.tolist(), float(best_score), n
