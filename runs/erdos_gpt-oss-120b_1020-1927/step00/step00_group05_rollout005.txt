def compute_c5(h, dx):
    n = len(h)
    L = 1 << ((2*n-1).bit_length())
    # compute fft of h and reversed (1-h)
    fh = np.fft.fft(h, L)
    f_rev = np.fft.fft((1-h)[::-1], L)
    conv = np.fft.ifft(fh * f_rev).real
    # take central part: full length 2*n-1: from 0 to 2*n-2 inclusive
    corr = conv[:2*n-1]
    value = np.max(corr) * dx
    return value
```

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    L = 1 << ((2*n - 1).bit_length())
    # Use rfft for real values
    H = np.fft.rfft(h, n=L)
    R = np.fft.rfft((1 - h)[::-1], n=L)
    conv = np.fft.irfft(H * R, n=L)
    # conv length L, we need first 2n-1 entries
    conv = conv[:2*n-1]
    return conv.max() * dx
```

def compute_c5_fft(h):
    n = h.shape[0]
    dx = 2.0 / n
    L = 1 << ((2 * n - 1).bit_length())
    # compute FFT
    H = np.fft.rfft(h, n=L)
    B = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(H * B, n=L)
    # conv length = L
    conv = conv[:2*n - 1]
    return conv.max() * dx
```

import time
import numpy as np

# ----------------------------------------------------------------------
# fast evaluation of C5 for a binary/real step function on a uniform grid
# ----------------------------------------------------------------------
def _c5_fft(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # length for a non‑circular convolution
    L = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    B = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(H * B, n=L)[: 2 * n - 1]
    return float(conv.max() * dx)

    n: int,
    init_h: np.ndarray | None = None,
    seed: int = 0,
    time_limit: float = 5.0,
) -> tuple[np.ndarray, float]:
