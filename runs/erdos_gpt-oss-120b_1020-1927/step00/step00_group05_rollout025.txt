    # correlation of h with (1-h)

import numpy as np, time, math

def _compute_c5(h: np.ndarray) -> float:
    """C5 for a given step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(corr.max() * dx)


def _max_overlap_and_lag(h: np.ndarray):
    """Return (C5 value, lag that attains it)."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    max_corr = float(corr[idx])
    lag = idx - (n - 1)          # lag ∈ [-(n-1), n-1]
    return max_corr * (2.0 / n), lag


def _project_onto_constraints(h: np.ndarray) -> np.ndarray:
    """Project onto 0≤h≤1 and sum(h)=n/2 (Euclidean projection)."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0            # search interval for the Lagrange multiplier
    for _ in range(30):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def _random_binary(n: int) -> np.ndarray:

    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, n // 2, replace=False)
    a[idx] = 1.0
    return a

        init_arr = np.array(initial_h_values, dtype=np.float64)
        n = init_arr.size
        has_init = True
    except Exception:
        init_arr = None
        n = 70                      # a safe default when no parent is present
        has_init = False

    dx = 2.0 / n
    best_h = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 1. Gradient descent from parent construction (if any)
    # --------------------------------------------------------------
    if has_init:
        h = _project_onto_constraints(init_arr.copy())
        cur_val = _compute_c5(h)
        if cur_val < best_val:
            best_h, best_val = h.copy(), cur_val

        for it in range(2000):
            if time.time() - start > budget_s - 5:
                break
            corr = np.correlate(h, 1 - h, mode="full")
            idx = int(np.argmax(corr))
            lag = idx - (n - 1)

            g = np.zeros_like(h)
            if lag >= 0:
                g[: n - lag] += 1 - h[lag:]
                g[lag:] += -h[: n - lag]
            else:
                k = -lag
                g[k:] += 1 - h[: n - k]
                g[: n - k] += -h[k:]

            grad = g * dx
            step = 0.5 / math.sqrt(it + 1)
            h_new = _project_onto_constraints(h - step * grad)
            val_new = _compute_c5(h_new)

            # simple improvement rule (rare uphill moves allowed)
            if val_new < cur_val or np.random.rand() < 0.001:
                h, cur_val = h_new, val_new
                if val_new < best_val:
                    best_h, best_val = h_new.copy(), val_new

    # --------------------------------------------------------------
    # 2. Simulated‑annealing on binary vectors
    # --------------------------------------------------------------
    while time.time() - start < budget_s - 5:
        h = _random_binary(n)
        cur_val, cur_lag = _max_overlap_and_lag(h)
        if cur_val < best_val:
            best_h, best_val = h.copy(), cur_val

        # annealing parameters
        steps = 8000
        for s in range(steps):
            if time.time() - start > budget_s - 5:
                break
            T = 0.02 * (1.0 - s / steps)          # linear cooling

            # most moves target the current worst shift
            if np.random.rand() < 0.7:
                lag = cur_lag
                if lag >= 0:
                    mism = np.where((h[: n - lag] == 1) & (h[lag:] == 0))[0]
                else:
                    k = -lag
                    mism = np.where((h[k:] == 1) & (h[: n - k] == 0))[0] + k
                if mism.size:
                    i = int(np.random.choice(mism))
                    zeros = np.where(h == 0)[0]
                    if zeros.size == 0:
                        continue
                    j = int(np.random.choice(zeros))
                else:
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    if ones.size == 0 or zeros.size == 0:
                        continue
                    i = int(np.random.choice(ones))
                    j = int(np.random.choice(zeros))
            else:
                i = np.random.randint(n)
                j = np.random.randint(n)
                if h[i] == h[j]:
                    continue

            # propose the swap
            h[i], h[j] = h[j], h[i]
            new_val, new_lag = _max_overlap_and_lag(h)
            delta = new_val - cur_val
            accept = False
            if delta <= 0:
                accept = True
            elif T > 0 and np.random.rand() < math.exp(-delta / T):
                accept = True

            if accept:
                cur_val, cur_lag = new_val, new_lag
                if new_val < best_val:
                    best_h, best_val = h.copy(), new_val
                if best_val < 0.38:           # early stop if already good enough
                    break
            else:
                # revert
                h[i], h[j] = h[j], h[i]

        if best_val < 0.38:
            break

    # --------------------------------------------------------------
    # 3. Fractional refinement (sub‑gradient steps)
    # --------------------------------------------------------------
    if best_h is not None:
        h = best_h.copy()
        for it in range(2000):
            if time.time() - start > budget_s - 5:
                break
            corr = np.correlate(h, 1 - h, mode="full")
            idx = int(np.argmax(corr))
            lag = idx - (n - 1)

            g = np.zeros_like(h)
            if lag >= 0:
                g[: n - lag] += 1 - h[lag:]
                g[lag:] += -h[: n - lag]
            else:
                k = -lag
                g[k:] += 1 - h[: n - k]
                g[: n - k] += -h[k:]

            grad = g * dx
            step = 0.05 / math.sqrt(it + 1)
            h_new = _project_onto_constraints(h - step * grad)
            val_new = _compute_c5(h_new)

            if val_new < best_val:
                best_h, best_val = h_new.copy(), val_new
            h = h_new

        best_h = _project_onto_constraints(best_h)
        best_val = _compute_c5(best_h)

    # --------------------------------------------------------------
    # 4. Fallback (should never trigger)
    # --------------------------------------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5, dtype=np.float64)
        best_val = _compute_c5(best_h)

    return (best_h, float(best_val), n)
