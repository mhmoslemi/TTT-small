def gradient_step(h, shift):
    n = len(h)
    grad = np.zeros_like(h)
    s = shift
    # compute overlapping region indexes
    if s >= 0:
        ov_len = n - s
        # 1 term
        grad[:ov_len] += 1.0
        # -h_{i+s}
        grad[:ov_len] -= h[s:]
        # -h_{i-s} for i>=s (i from s to n-1)
        grad[s:] -= h[:ov_len]
    else:
        s_abs = -s
        ov_len = n - s_abs
        grad[:ov_len] += 1.0
        grad[:ov_len] -= h[s_abs:]
        grad[s_abs:] -= h[:ov_len]
    return grad

import time
import math
import numpy as np

def _project_to_box_sum(v, target):
    """Project vector v onto {0 <= x <= 1, sum(x)=target}."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny adjustment to hit the target sum more accurately
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny difference uniformly over non‑bound entries
        mask = (w > 0.0) & (w < 1.0)
        if mask.any():
            w[mask] += diff / mask.sum()
            w = np.clip(w, 0.0, 1.0)
    return w


def _gradient_step(h, shift):
    """Compute gradient of O(shift)=∑h_i(1−h_{i+shift}) w.r.t. h."""
    n = h.size
    grad = np.zeros_like(h)
    if shift >= 0:
        ov = n - shift
        grad[:ov] += 1.0               # sum term
        grad[:ov] -= h[shift:]         # -h_{i+shift}
        grad[shift:] -= h[:ov]         # -h_{i-shift}
    else:
        s = -shift
        ov = n - s
        grad[s:] += 1.0                # sum term for i≥s
        grad[s:] -= h[:ov]             # -h_{i-s}= -h_{i+shift}
        grad[:ov] -= h[s:]             # -h_{i+shift}= -h_{i-s}
    return grad


def _c5_of_h(h):
    """Compute C5 = max_k ∫ h(x)(1−h(x+k))dx  (discretised)."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(conv.max() * (2.0 / n))


def _random_feasible_a(m, target):
    """Random feasible a (length m) with 0≤a_i≤1 and sum=target."""
    while True:
        a = np.random.rand(m)
        a = a / a.sum() * target
        if (a <= 1.0).all():
            return a


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ---- problem size -------------------------------------------------
    block_cnt = 10                     # number of constant pieces
    n = 200                            # total samples (must be multiple of block_cnt)
    block_len = n // block_cnt
    target_sum = n / 2.0               # ∑h = n/2

    # ---- initialise a (piecewise constants) -------------------------

        # interpolate to the chosen resolution

        pos_dst = np.linspace(0.0, 2.0, n, endpoint=False)
        h_interp = np.interp(pos_dst, pos_src, init_h)
        a = h_interp.reshape(block_cnt, block_len).mean(axis=1)
        a = _project_to_box_sum(a, block_cnt / 2.0)   # sum must be m/2
    else:
        a = _random_feasible_a(block_cnt, block_cnt / 2.0)

        final_score = best_score

    final_h = _project_to_box_sum(final_h, target_sum)
    final_score = _c5_of_h(final_h)

    return (final_h.tolist(), float(final_score), n)
