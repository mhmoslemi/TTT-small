import numpy as np, time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def gradient_for_shift(h, k):
    n = len(h)
    g = np.ones(n)
    if k >= 0:
        if k < n:
            g[:n - k] -= h[k:]
            g[k:] -= h[:n - k]
    else:
        kn = -k
        if kn < n:
            g[kn:] -= h[:n - kn]
            g[:n - kn] -= h[kn:]
    return g

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the max overlap integral for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _gradient_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of I_k = ∫ h(x)*(1-h(x+k))dx w.r.t. the discretised h."""
    n = h.shape[0]
    g = np.ones_like(h)
    if k >= 0:
        if k < n:
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
    else:
        kn = -k
        if kn < n:
            g[kn:] -= h[: n - kn]
            g[: n - kn] -= h[kn:]
    return g


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x | 0≤x_i≤1, Σx_i = target_sum}.
    Uses a simple binary search on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0          # ensures all entries are clipped to 0
    hi = 1.0 - np.min(v) + 1.0     # ensures all entries are clipped to 1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum exactly n/2."""
    v = rng.random(n)
    v = _project_onto_simplex(v, n / 2.0)
    return v

    """
    Optimize the step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # --------------------------------------------------------------------
    # initialise h
    try:
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.shape[0]
    except Exception:                     # fallback if no parent exists
        n = 70
        h = np.full(n, 0.5, dtype=np.float64)   # already satisfies the integral
    target_sum = n / 2.0

    # keep the best solution seen so far
    best_h = h.copy()
    best_val = _compute_c5(best_h)

    # --------------------------------------------------------------------
    # core optimisation routine (gradient + random perturbations)
    def _optimise_one_resolution(h_vec: np.ndarray, max_iters: int):
        nonlocal best_h, best_val, start
        step = 0.2
        n_loc = h_vec.shape[0]
        target = n_loc / 2.0

        for it in range(max_iters):
            if time.time() - start > budget_s - 0.3:
                break

            # correlation and worst shift(s)
            corr = np.correlate(h_vec, 1.0 - h_vec, mode="full")
            max_corr = corr.max()
            cur_val = max_corr * (2.0 / n_loc)

            # early exit if already good enough
            if cur_val <= 0.3808:
                best_h[:] = h_vec
                best_val = cur_val
                return True

            # all shifts attaining the maximum (subgradient aggregation)
            max_idxs = np.where(corr == max_corr)[0]
            shifts = max_idxs - (n_loc - 1)
            g = np.zeros_like(h_vec)
            for k in shifts:
                g += _gradient_for_shift(h_vec, int(k))
            g /= len(shifts)

        # optimise at the finer level
        stop = _optimise_one_resolution(best_h, max_iters=2000)
        if stop:

    # --------------------------------------------------------------------
    # final output
    final_n = best_h.shape[0]
    final_c5 = _compute_c5(best_h)
    return (best_h, float(final_c5), final_n)
