import time, numpy as np, cvxpy as cp, math

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def solve_binary(N, max_seconds):
    # builds and solves MILP for binary minimax overlap
    h = cp.Variable(N, boolean=True)
    # map pair to index
    pair_to_idx = {}
    y = []
    for i in range(N):
        for j in range(i+1, N):
            pair_to_idx[(i,j)] = len(y)
            y.append(cp.Variable(boolean=True))
    y = np.array(y)
    constraints = [cp.sum(h) == N/2]

    # product constraints
    for (i,j), idx in pair_to_idx.items():
        constraints.append(y[idx] <= h[i])
        constraints.append(y[idx] <= h[j])
        constraints.append(y[idx] >= h[i] + h[j] - 1)

    t = cp.Variable()
    constraints.append(t >= 0)
    # shift constraints
    for s in range(1, N):
        # sum of y[i,i+s] for i in range(N-s)
        sum_y = cp.sum([ y[pair_to_idx[(i,i+s)]] for i in range(N - s) ])
        constraints.append(N/2 - sum_y <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    try:
        prob.solve(solver=cp.ECOS_BB, verbose=False, max_iters=400000)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False)
        except Exception:
            return None, None, None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, None, prob.status
    # round h
    h_val = np.rint(np.array(h.value)).astype(float)
    # enforce exact sum
    target_sum = N//2
    cur_sum = int(h_val.sum())
    if cur_sum != target_sum:
        diff = target_sum - cur_sum
        if diff > 0:
            zeros = np.where(h_val == 0)[0]
            idx = np.random.choice(zeros, diff, replace=False)
            h_val[idx] = 1
        else:
            ones = np.where(h_val == 1)[0]
            idx = np.random.choice(ones, -diff, replace=False)
            h_val[idx] = 0
    t_val = float(t.value)
    return h_val, t_val, prob.status

    # Local search (simulated annealing) as fallback

def project_onto_simplex_box(v, target):
    # Solve min ||x - v||^2 s.t. sum x = target, 0 <= x <= 1
    # Use water-filling binary search on λ.
    lo = -np.max(v)  # ensures all <=0 after shift => 0 after clipping
    hi = 1 - np.min(v)  # ensures all >=1 after shift =>1 after clipping
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0, 1)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) / 2.0, 0, 1)
```

import time, math, numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the hyperplane sum(x)=target with 0<=x<=1
# ----------------------------------------------------------------------
def _proj_sum_box(v, target, iters=60):
    """Project vector v onto {0<=x<=1, sum(x)=target}."""
    lo = -np.max(v)                # forces all v+λ ≤ 0 → after clipping all 0
    hi = 1.0 - np.min(v)           # forces all v+λ ≥ 1 → after clipping all 1
    for _ in range(iters):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


# ----------------------------------------------------------------------
# Compute the (scaled) overlap constant C5 for a candidate h
# ----------------------------------------------------------------------
def _c5_value(h):
    N = h.shape[0]
    dx = 2.0 / N
    # linear correlation with zero padding (mode='full')
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

# Core optimizer: projected sub‑gradient descent on the raw (unscaled)
# maximum overlap w_max = max_k sum_i h_i (1-h_{i+k}).
# ----------------------------------------------------------------------
def _optimize(N, h0, target_raw, time_budget):
    """Return (best_h, best_raw) within time_budget seconds."""
    start = time.time()
    best_h = None
    best_raw = np.inf

    # number of inner steps per restart – adapt to time left
    while time.time() - start < time_budget * 0.9:
        # initialise: random perturbation of h0 (or fresh random)
        if h0 is None:
            h = np.random.rand(N)
        else:
            h = h0 + 0.02 * np.random.randn(N)   # small noise
        h = np.clip(h, 0.0, 1.0)
        h = _proj_sum_box(h, N / 2.0)

        # parameters for the diminishing step size
        step0 = 0.8
        max_it = 5000
        raw_cur = np.max(np.correlate(h, 1.0 - h, mode='full'))

        for it in range(max_it):
            # correlation and worst shift
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx = int(np.argmax(corr))
            shift = idx - (N - 1)                # signed shift
            # sub‑gradient of the worst term
            g = np.ones(N)
            if shift >= 0:
                s = shift
                if s < N:
                    g[: N - s] -= h[s:]
                    g[s:]   -= h[: N - s]
            else:
                s = -shift
                if s < N:
                    g[s:]   -= h[: N - s]
                    g[: N - s] -= h[s:]

            # step size

                        return best_h, best_raw
                # stop if out of time

        # end of one restart
