def compute_c5(x):
    n = len(x)
    dx = 2.0 / n
    return np.max(np.correlate(x, 1 - x, mode='full')) * dx

Method:

```
def project_onto_capped_simplex(v, S):
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(v - mid, 0.0, 1.0)
        s = p.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
```

Now gradient function via subgradient:

```
def subgradient(h):
    n = len(h)
    L = np.correlate(h, 1 - h, mode='full')
    max_val = L.max()
    mask = np.isclose(L, max_val, atol=1e-12)
    idxs = np.where(mask)[0]
    g = np.zeros_like(h)
    for idx in idxs:
        ell = idx - (n - 1)
        if ell >= 0:
            if ell < n:
                g[:n-ell] += (1 - h[ell:])
                g[ell:] -= h[:n-ell]
        else:
            Labs = -ell
            if Labs < n:
                g[Labs:] += (1 - h[:n-Labs])
                g[:n-Labs] -= h[Labs:]
    if len(idxs) > 0:
        g /= len(idxs)
    g *= (2.0 / n)  # dx factor
    return g, max_val * (2.0 / n)
```

def optimize_from_h(h_start, time_limit):
    start = time.time()
    h = h_start.copy()
    n = len(h)
    S = n / 2.0
    dx = 2.0 / n
    L = np.correlate(h, 1 - h, mode='full')
    best_c5 = L.max() * dx
    best_h = h.copy()
    iteration = 0
    step0 = 1.0
    while time.time() - start < time_limit:
        iteration += 1
        g, cur_c5 = subgradient(h)
        eta = step0 / math.sqrt(iteration)
        v = h - eta * g
        h_new = project_onto_capped_simplex(v, S)
        # evaluate new C5
        Lnew = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = Lnew.max() * dx
        # accept if not worse (allow small tolerance)
        if c5_new <= cur_c5 + 1e-12:
            h = h_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
        else:
            # maybe accept with small prob (simulated annealing)
            # Temperature schedule
            temp = max(1e-3, 0.2 / math.sqrt(iteration))
            if np.random.rand() < math.exp(-(c5_new - cur_c5) / temp):
                h = h_new
        # occasional random perturbation to escape
        if iteration % 5000 == 0:
            # add small noise
            perturb = np.random.uniform(-0.02, 0.02, size=n)
            h = project_onto_capped_simplex(h + perturb, S)
    return best_h, best_c5

import numpy as np
import time
import math
import random

def _project_onto_capped_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {0 <= x_i <= 1, sum x_i = target_sum}."""
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        p = np.clip(v - mid, 0.0, 1.0)
        s = p.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Exact integral approximation used by the evaluator."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def _subgrad(h: np.ndarray):
    """Return a subgradient of C5 and the current C5 value."""
    n = h.shape[0]
    L = np.correlate(h, 1 - h, mode="full")
    max_val = L.max()
    active = np.isclose(L, max_val, atol=1e-12)
    idxs = np.where(active)[0]

    g = np.zeros_like(h)
    for idx in idxs:
        ell = idx - (n - 1)
        if ell >= 0:
            # (1 - h_{i+ell}) term
            g[: n - ell] += (1 - h[ell:])
            # -h_{i-ell} term
            g[ell:] -= h[: n - ell]
        else:
            Labs = -ell
            g[Labs:] += (1 - h[: n - Labs])
            g[: n - Labs] -= h[Labs:]
    if len(idxs) > 0:
        g /= len(idxs)
    g *= (2.0 / n)                     # factor dx
    return g, max_val * (2.0 / n)      # return gradient and current C5


def _optimize_from_start(h0: np.ndarray, timelimit: float):
    """Projected subgradient descent with a simple annealing scheme."""
    start = time.time()
    h = h0.copy()
    n = h.shape[0]
    S = n / 2.0
    dx = 2.0 / n

    # initial record
    cur_c5 = _c5_value(h)
    best_h = h.copy()
    best_c5 = cur_c5

    it = 0
    base_step = 1.0
    while time.time() - start < timelimit:
        it += 1
        grad, cur_c5 = _subgrad(h)

        step = base_step / math.sqrt(it)
        v = h - step * grad
        h_new = _project_onto_capped_simplex(v, S)

        new_c5 = _c5_value(h_new)

        # simple descent acceptance (with occasional probabilistic uphill moves)
        if new_c5 <= cur_c5 + 1e-12:
            h = h_new
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h_new.copy()
        else:
            temp = max(1e-3, 0.2 / math.sqrt(it))
            if np.random.rand() < math.exp(-(new_c5 - cur_c5) / temp):
                h = h_new

        # periodic jitter to try escape from plateaus
        if it % 5000 == 0:
            jitter = np.random.uniform(-0.02, 0.02, size=n)
            h = _project_onto_capped_simplex(h + jitter, S)

    return best_h, best_c5


def _generate_svc_h(n_points: int, iterations: int = 10) -> np.ndarray:
    """Construct a fat Cantor (Smith–Volterra–Cantor) set of measure 1."""
    xs = np.linspace(0.0, 2.0, n_points, endpoint=False)
    intervals = [(0.0, 2.0)]
    L0 = 2.0
    for k in range(iterations):
        rem = L0 / (4.0 ** (k + 1))
        new_int = []
        for a, b in intervals:
            length = b - a
            if rem >= length:
                new_int.append((a, b))
                continue
            side = (length - rem) / 2.0
            new_int.append((a, a + side))
            new_int.append((b - side, b))
        intervals = new_int

    h = np.zeros(n_points, dtype=float)
    for a, b in intervals:
        mask = (xs >= a) & (xs < b)
        h[mask] = 1.0
    return _project_onto_capped_simplex(h, n_points / 2.0)


def _random_binary_h(n: int) -> np.ndarray:
    h = np.zeros(n, dtype=float)
    idx = np.random.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to lower the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # target discretisation
    n_points = 512

    # collect starting candidates
    starts = []

    # 1) use the provided parent if available (upsampled if needed)
    try:

        if parent.shape[0] != n_points:

                repeat = n_points // parent.shape[0]
                parent = np.repeat(parent, repeat)
            else:
                xs_old = np.linspace(0.0, 2.0, parent.shape[0], endpoint=False)
                xs_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
                parent = np.interp(xs_new, xs_old, parent, period=2.0)
        parent = _project_onto_capped_simplex(parent, n_points / 2.0)
        starts.append(parent)
    except Exception:
        pass

    # 2) fat Cantor (Smith–Volterra–Cantor) construction
    starts.append(_generate_svc_h(n_points, iterations=10))

    # 3) random binary vector
    starts.append(_random_binary_h(n_points))

    # 4) random continuous vector projected onto the feasible simplex
    rnd = np.random.rand(n_points)
    starts.append(_project_onto_capped_simplex(rnd, n_points / 2.0))

    # Remove exact duplicates (unlikely but cheap)
    unique_starts = []
    seen = set()
    for s in starts:
        key = tuple(np.round(s, 8))
        if key not in seen:

            unique_starts.append(s)

        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 5.0:

        allowed = max(5.0, remaining * 0.5)
        h_opt, c5_opt = _optimize_from_start(s, allowed)
        if c5_opt < best_c5:

            best_h = h_opt.copy()

    # Final safety check (re‑evaluate)
    if best_h is None:
        best_h = np.zeros(n_points, dtype=float)
        best_c5 = _c5_value(best_h)

    final_c5 = _c5_value(best_h)
    best_c5 = float(final_c5)

    return best_h.tolist(), best_c5, n_points
