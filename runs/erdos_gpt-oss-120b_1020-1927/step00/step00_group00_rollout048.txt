import numpy as np

def _c5_of(h):
    """Return C5 for a step function defined by array h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode="full") * dx).max()


def _project_to_sum(h, target):
    """
    Project the vector h onto the hyperplane sum(h)=target while keeping
    each entry inside [0,1].  Simple water‑filling iteration.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            inc_cap = (1.0 - h).sum()
            if inc_cap <= 0:
                break
            add = diff * (1.0 - h) / inc_cap
            h = np.clip(h + add, 0.0, 1.0)
        else:
            dec_cap = h.sum()
            if dec_cap <= 0:
                break
            sub = (-diff) * h / dec_cap
            h = np.clip(h - sub, 0.0, 1.0)
    # small uniform fix if needed
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        idx = np.where(free)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h
