    pos = int(np.argmax(grad))
    neg = int(np.argmin(grad))
    if pos == neg or grad[pos] <= 0 or grad[neg] >= 0:
        return None
    dec = v[pos]               # amount we can decrease
    inc = 1.0 - v[neg]          # amount we can increase
    delta = min(dec, inc, step)
    if delta <= 0:
        return None
    nv = v.copy()
    nv[pos] -= delta
    nv[neg] += delta
    return nv

def _swap_move(v, step):
    n = v.size
    i, j = np.random.choice(n, 2, replace=False)
    if np.random.rand() < 0.5:
        dec, inc = v[i], 1.0 - v[j]
        if dec == 0 or inc == 0:
            return None
        delta = np.random.rand() * min(dec, inc, step)
        if delta == 0:
            return None
        nv = v.copy()
        nv[i] -= delta
        nv[j] += delta
        return nv
    else:
        dec, inc = v[j], 1.0 - v[i]
        if dec == 0 or inc == 0:
            return None
        delta = np.random.rand() * min(dec, inc, step)
        if delta == 0:
            return None
        nv = v.copy()
        nv[j] -= delta
        nv[i] += delta
        return nv

def _reseed_noise(base, scale):
    nv = base + np.random.uniform(-scale, scale, size=base.size)
    np.clip(nv, 0.0, 1.0, out=nv)
    target = base.size / 2.0
    diff = nv.sum() - target
    if abs(diff) > 1e-12:
        mask = (nv > 0) & (nv < 1)
        cnt = mask.sum()
        if cnt:
            nv[mask] -= diff / cnt
        else:
            nv[0] -= diff
        np.clip(nv, 0.0, 1.0, out=nv)
    return nv

def _local_optimize(start_vec, alloc_secs):
    n = start_vec.size
    best = start_vec.copy()
    best_sc = _score_vec(best)
    cur = best.copy()
    cur_sc = best_sc
    base_step = 0.2
    start = time.time()
    last_imp = start
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= alloc_secs:
            break
        step = base_step * (1.0 - elapsed / alloc_secs)
        # decide move type
        if np.random.rand() < 0.6:
            cand = _gradient_move(cur, step)
        else:
            cand = _swap_move(cur, step)
        if cand is None:
            continue
        cand_sc = _score_vec(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                last_imp = now
        else:
            # annealing acceptance
            temp = max(1e-4, (alloc_secs - elapsed) / alloc_secs)
            if np.random.rand() < math.exp(-(cand_sc - cur_sc) / temp):
                cur, cur_sc = cand, cand_sc
        # occasional restart from best with noise
        if now - last_imp > 30.0 and np.random.rand() < 0.1:
            cur = _reseed_noise(best, 0.1)
            cur_sc = _score_vec(cur)
            last_imp = now
    return best, best_sc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    # Initialise vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.size
        target = n / 2.0
        diff = h0.sum() - target
        if abs(diff) > 1e-12:
            h0[0] -= diff
        np.clip(h0, 0.0, 1.0, out=h0)
        # final tiny re‑balance if needed
        diff = h0.sum() - target
        if abs(diff) > 1e-12:
            mask = (h0 > 0) & (h0 < 1)
            cnt = mask.sum()
            if cnt:
                h0[mask] -= diff / cnt
            else:
                h0[0] -= diff
            np.clip(h0, 0.0, 1.0, out=h0)
    else:
        n = 45
        h0 = np.full(n, 0.5, dtype=np.float64)

    start_time = time.time()
    remaining = budget_s - (time.time() - start_time)
    # main optimisation at current resolution
    best_h, best_sc = _local_optimize(h0, remaining * 0.7)

    # possibly refine at doubled resolution if time permits
    elapsed = time.time() - start_time
    rem = budget_s - elapsed
    if rem > 20.0 and best_h.size <= 80:
        h2 = np.repeat(best_h, 2)
        best_h2, best_sc2 = _local_optimize(h2, rem)
        best_h, best_sc = best_h2, best_sc2

    # final sanity check
    n = best_h.size
    target = n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target) > 1e-8:
        # adjust first entry
        best_h[0] += target - cur_sum
        np.clip(best_h, 0.0, 1.0, out=best_h)

    final_sc = _score_vec(best_h)
    return best_h.tolist(), float(final_sc), int(n)
