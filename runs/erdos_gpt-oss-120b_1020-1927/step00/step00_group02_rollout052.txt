def project(v, target):
    # ensures 0 <= v_i <= 1 and sum = target.
    # solve lam via binary search
    low = -np.max(v) - 1  # ensures v - lam > 1

        s = proj.sum()
        if s > target:
            low = lam
        else:

def compute_gradient(h, shift):
    N = h.shape[0]
    grad = np.ones(N)
    if shift == 0:
        grad -= 2*h
        return grad
    if shift > 0:
        # forward neighbor for i< N - shift: subtract h[i+shift]
        grad[:N-shift] -= h[shift:]
        # backward neighbor for i >= shift: subtract h[i-shift]
        grad[shift:] -= h[:N-shift]
    else:
        s = -shift
        # shift negative => same as positive s but reverse roles
        grad[s:] -= h[:N-s]
        grad[:N-s] -= h[s:]
    return grad
```

    np.random.seed(seed)
    random.seed(seed)
    t0 = time.time()
    remaining = lambda: time.time() - t0

    best_h = None
    best_val = np.inf

    def project(v, target):
        lo = -2.0
        hi = 2.0
        for _ in range(60):
            lam = (lo + hi)/2.0
            proj = np.clip(v - lam, 0.0, 1.0)
            s = proj.sum()
            if s > target:
                lo = lam
            else:
                hi = lam
        return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

    def compute_c5(h):
        N = h.shape[0]
        dx = 2.0 / N
        corr = np.correlate(h, 1 - h, mode='full')
        return dx * np.max(corr)

    def compute_gradient(h, shift):
        N = h.shape[0]
        grad = np.ones(N)
        if shift == 0:
            grad -= 2 * h
            return grad
        if shift > 0:
            # subtract forward neighbor
            grad[:N-shift] -= h[shift:]
            # subtract backward neighbor
            grad[shift:] -= h[:N-shift]
        else:
            s = -shift
            grad[s:] -= h[:N-s]
            grad[:N-s] -= h[s:]
        return grad

    # function for local refinement (pair swapping)
    def local_refine(h, cur_val, steps, target):
        N = h.shape[0]
        for _ in range(steps):
            i, j = np.random.randint(0, N, size=2)
            if i == j:
                continue
            # choose epsilon range based on current magnitude
            eps = (np.random.rand() - 0.5) * 0.1  # [-0.05,0.05]
            if 0 <= h[i] + eps <= 1 and 0 <= h[j] - eps <= 1:
                h_new = h.copy()
                h_new[i] += eps
                h_new[j] -= eps
                val = compute_c5(h_new)
                if val < cur_val:
                    h = h_new
                    cur_val = val
        return h, cur_val

    # optional function to upsample
    def upsample(h, new_N):
        old_N = len(h)
        x_old = np.linspace(0, 2, old_N, endpoint=False)
        x_new = np.linspace(0, 2, new_N, endpoint=False)
        h_new = np.interp(x_new, x_old, h)
        # ensure sum via projection
        return project(h_new, new_N/2)

    # stage sizes
    stage_Ns = [180, 360, 720]  # progressive refinement
    # time allocation per stage based on remaining
    for stage_idx, N in enumerate(stage_Ns):
        # decide remaining time
        elapsed = remaining()
        if elapsed > budget_s * 0.95:
            break
        target_sum = N / 2.0
        # initialize starting points
        initial_points = []
        # if best_h from previous stage, upsample and use
        if best_h is not None:
            initial_points.append(upsample(best_h, N))
        else:
            # also use global initial if exists
            if 'initial_h_values' in globals():
                init_arr = np.array(initial_h_values, copy=False)
                old_N = init_arr.shape[0]
                if old_N != N:
                    # interpolate to N
                    x_old = np.linspace(0,2,old_N,endpoint=False)
                    x_new = np.linspace(0,2,N,endpoint=False)
                    init_res = np.interp(x_new, x_old, init_arr)
                    initial_points.append(project(init_res, target_sum))
                else:
                    initial_points.append(project(init_arr, target_sum))
        # random restarts
        num_random = 3
        for _ in range(num_random):
            v = np.random.rand(N)
            initial_points.append(project(v, target_sum))

        # iterate over starting points
        for h0 in initial_points:
            h = h0.copy()
            cur_val = compute_c5(h)
            # keep best within this stage
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                if best_val <= 0.3808:
                    # early exit whole run
                    return best_h.tolist(), float(best_val), N
            # optimization parameters
            max_iter = 8000
            step0 = 0.2
            for it in range(max_iter):
                if remaining() > budget_s * 0.95:
                    break
                # compute correlation and shift
                corr = np.correlate(h, 1 - h, mode='full')
                max_idx = np.argmax(corr)
                cur_val = (2.0 / N) * corr[max_idx]
                if cur_val < best_val:
                    best_val = cur_val
                    best_h = h.copy()
                # break if target reached
                if best_val <= 0.3808:
                    return best_h.tolist(), float(best_val), N
                shift = max_idx - (N - 1)
                grad = compute_gradient(h, shift)
                step = step0 / math.sqrt(1 + it)
                h_new = h - step * grad
                h_new = project(h_new, target_sum)
                # compute new objective
                new_corr = np.correlate(h_new, 1 - h_new, mode='full')
                new_val = (2.0 / N) * new_corr.max()
                # accept if improved
                if new_val <= cur_val:
                    h = h_new
                else:
                    # occasional probabilistic acceptance
                    T = max(0.0, (max_iter - it) / max_iter) * 0.01
                    if np.random.rand() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                        h = h_new
                # occasional noise injection
                if it % 2000 == 0 and it > 0:
                    h = project(h + np.random.normal(scale=0.01, size=N), target_sum)
            # do local refinement after gradient loop
            cur_val = compute_c5(h)
            h, cur_val = local_refine(h, cur_val, steps=2000, target=target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                if best_val <= 0.3808:
                    return best_h.tolist(), float(best_val), N
        # after trying all starts for this stage, continue to next higher N

    # final step: ensure best_h is projected at final N (maybe last stage's N)
    if best_h is None:
        # fallback to uniform
        N = 180
        best_h = np.full(N, 0.5)
        best_h = project(best_h, N/2)
    final_val = compute_c5(best_h)
    return best_h.tolist(), float(final_val), len(best_h)

import numpy as np
import math
import time
import random

def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto [0,1]^n and onto the hyperplane sum(v)=target."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx for the discrete vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _gradient_max_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """∇_i g_shift where g_shift = Σ h_i(1-h_{i+shift})."""
    n = h.shape[0]
    grad = np.ones(n)
    if shift == 0:
        grad -= 2.0 * h
        return grad
    if shift > 0:
        # forward neighbour
        grad[: n - shift] -= h[shift:]
        # backward neighbour
        grad[shift:] -= h[: n - shift]
    else:  # shift < 0
        s = -shift
        grad[s:] -= h[: n - s]
        grad[: n - s] -= h[s:]
    return grad


def _local_pair_swap(h: np.ndarray, cur_val: float, steps: int) -> (np.ndarray, float):
    """Swap a small amount between two coordinates while preserving the sum."""
    n = h.shape[0]
    for _ in range(steps):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = (np.random.rand() - 0.5) * 0.1  # [-0.05,0.05]
        if 0.0 <= h[i] + eps <= 1.0 and 0.0 <= h[j] - eps <= 1.0:
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps
            val = _c5_value(h_new)
            if val < cur_val:
                h, cur_val = h_new, val
    return h, cur_val


def _upsample(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear up‑sampling of a step function."""
    old_n = h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, new_n / 2.0)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h, its C5 bound, and the used number of points."""
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # progressive refinements: coarse → medium → fine
    stages = [180, 360, 720]

    best_h = None
    best_val = np.inf

    # obtain the provided construction if any
    init_parent = globals().get("initial_h_values")
    if init_parent is not None:
        init_parent = np.asarray(init_parent, dtype=float)

    for N in stages:
        if time_left() < 1.0:
            break
        target = N / 2.0

        # possible starts for this stage
        starts = []

        # up‑sample previous best
        if best_h is not None:
            starts.append(_upsample(best_h, N))

        # up‑sample the given initial construction
        if init_parent is not None:
            if init_parent.shape[0] != N:
                x_old = np.linspace(0.0, 2.0, init_parent.shape[0], endpoint=False)
                x_new = np.linspace(0.0, 2.0, N, endpoint=False)
                h_res = np.interp(x_new, x_old, init_parent)
                starts.append(_project_box_sum(h_res, target))
            else:
                starts.append(_project_box_sum(init_parent.copy(), target))

        # a few pure random starts
        for _ in range(3):
            v = np.random.rand(N)
            starts.append(_project_box_sum(v, target))

        # optimise each start
        for h in starts:
            if time_left() < 1.0:

            cur_val = _c5_value(h)
            if cur_val < best_val:

                if best_val <= 0.3808:

    # final safety projection and recomputation
    best_h = _project_box_sum(np.asarray(best_h, dtype=float), len(best_h) / 2.0)
    best_val = _c5_value(best_h)
