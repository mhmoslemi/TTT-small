def project_onto_bounds_and_sum(y, S):
    lo = -np.max(y) - 1 # ensure clip set at lower bound 1 maybe?

import time
import numpy as np

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Construct a simple monotone step‑function (mid‑point linear ramp)
    that satisfies the constraints and yields a very small C₅ bound.
    """
    np.random.seed(seed)        # deterministic, not strictly needed
    start = time.time()

    # Choose a modest number of points (well under the 1 000 limit)
    n_points = 200                     # can be increased without breaking limits
    dx = 2.0 / n_points

    # Linear ramp sampled at the mid‑points of each subinterval:
    # h_i = (i + 0.5) / n_points  for i = 0,…,n_points‑1
    # This satisfies 0 ≤ h_i ≤ 1 and ∑h_i * dx = 1 exactly.
    h = (np.arange(n_points, dtype=np.float64) + 0.5) / n_points

    # Compute the overlap bound exactly as the evaluator does.
    # np.correlate performs the zero‑padded correlation; we then
    # multiply by dx to obtain the integral approximation.
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c5_bound = float(np.max(corr))

    # Just in case the time budget is extremely tight (unlikely here),
    # we could stop early, but the work above is negligible.
    _ = time.time() - start  # dummy to respect the budget variable

    return h, c5_bound, n_points
