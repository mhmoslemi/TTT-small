import numpy as np
import scipy.optimize as optimize

def generate_hexagonal_grid(num_rows, num_cols, square_size=1.0):
    """Generate center coordinates for a hexagonal packing in a square."""
    grid = []
    for row in range(num_rows):
        y = (row + 0.5) * (np.sqrt(3) / 2) * square_size / (num_rows - 1)
        for col in range(num_cols):
            x = col * square_size / (num_cols - 1)
            if row % 2 == 0:
                x += square_size / (2 * (num_cols - 1))
            grid.append([x, y])
    return np.array(grid)

def find_gaps(centers, radii, square_size=1.0):
    """Find gaps near the square's boundaries and corners where new circles can be placed."""
    gaps = []
    for x, y in np.random.uniform(0, 1, (100, 2)):
        r = 0.01
        if not any(np.hypot(x - c[0], y - c[1]) < (radii[i] + r) for i, c in enumerate(centers)):
            gaps.append((x, y, r))
    return gaps

def objective_function(params, centers, square_size=1.0):
    """Objective function to maximize the sum of radii."""
    new_centers = params[:26, :2]
    new_radii = params[26, :]
    total_radius = np.sum(new_radii)
    # Enforce that all circles are inside the square
    if any(np.any(new_centers[i] < 0 - 1e-12) or np.any(new_centers[i] > 1 + 1e-12) for i in range(26)):
        return -np.inf
    # Enforce that circles do not overlap or cross square boundary
    for i in range(26):
        for j in range(i + 1, 26):
            dist = np.hypot(new_centers[i, 0] - new_centers[j, 0], new_centers[i, 1] - new_centers[j, 1])
            if dist < new_radii[i] + new_radii[j] - 1e-12:
                return -np.inf
    return total_radius

def run_packing():

    params = np.vstack([centers, radii.reshape(1, -1)])  # [x1, y1, x2, y2, ..., r1, r2, ...]

    result = optimize.minimize(
