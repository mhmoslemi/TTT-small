import cvxpy as cp

def run_packing():

    objective = cp.Maximize(cp.sum(radii))
    
    # Constraints: No circle overlaps, and all circles fit in [0,1]^2
    constraints = []
    for i in range(num_circles):
        constraints.append(radii[i] >= 0)
