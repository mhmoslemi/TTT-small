from scipy.spatial import distance

        centers.append([x, 0.95])
        radii.append(0.05)

    # Place circles along the left and right edges
    for y in np.linspace(0.1, 0.9, 6):
        centers.append([0.05, y])
        radii.append(0.05)

        centers.append([0.95, y])
        radii.append(0.05)

    # Place circles in a hexagonal grid near the center

            x = 0.25 + col * 0.2
            y = 0.4 + row * 0.2

            radii.append(0.15)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Define the objective function: maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize sum

    # Define the constraints: non-overlapping and boundary conditions
    def constraint(x):
        # x is a flattened array: radii first, then centers
        r = x[:len(radii)]
        centers = x[len(radii):].reshape((len(centers), 2))
        for i in range(len(r)):
            x_i, y_i = centers[i]
            r_i = r[i]
            if x_i - r_i < -1e-8 or x_i + r_i > 1 + 1e-8 or y_i - r_i < -1e-8 or y_i + r_i > 1 + 1e8:
                return -1  # Violates boundary

                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-8:
                    return -1  # Overlap

    # Initial guess: radii as initial values, centers as initial positions
    x0 = np.concatenate((radii, centers.flatten()))

    result = minimize(objective, x0, bounds=bounds, constraints={"type": "ineq", "fun": constraint})

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
