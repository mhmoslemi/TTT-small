import scipy.optimize as optimize

            centers.append([0.1, 0.9])

        centers.append([x, y])
        radii.append(0.1)

    for i in range(8):
        x = 0.5 + i * 0.125
        y = 0.5 + i * 0.125

        centers.append([x, y])
        radii.append(0.1)
    
    # Concatenate to proper shapes
    centers = np.array(centers)

    def penalty_function(x):
        # x is a flattened vector: [c1x, c1y, r1, c2x, c2y, r2, ..., cnx, cny, rn]

        # First, check if any circle is outside the square
        for i in range(n):

            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
