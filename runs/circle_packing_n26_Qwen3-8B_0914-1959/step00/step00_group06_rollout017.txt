import scipy.optimize as optimize


def run_packing():

    rows = 5
    cols = 5
    radius = 0.05  # Initial guess for radius

    centers = []
    radii = []

    for row in range(rows):
        for col in range(cols):
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            if x > 1 - 1e-6 or y > 1 - 1e-6:
                break
            centers.append([x, y])
            radii.append(radius)

    centers = np.array(centers)
    radii = np.array(radii)

    # Objective function to maximize total area
    def objective(x):
        # x is a flattened array of [r_0, r_1, ..., r_25]
        radii = x
        total_area = np.sum(np.pi * radii**2)
        return -total_area  # We minimize for optimization

    # Constraints
    def constraint_center(x, i):
        r = x[i]
        if r < 0:
            return r
        return r

    def constraint_boundary(x, i):
        r = x[i]
        if r < 0:
            return r
        x_i, y_i = centers[i]
        if (x_i - r < -1e-12) or (x_i + r > 1 + 1e-12) or (y_i - r < -1e-12) or (y_i + r > 1 + 1e-12):
            return r
        return 0

    def constraint_overlap(x, i, j):
        r_i = x[i]
        r_j = x[j]
        if r_i < 0 or r_j < 0:
            return r_i + r_j
        delta_x = centers[i, 0] - centers[j, 0]
        delta_y = centers[i, 1] - centers[j, 1]
        dist_sq = delta_x**2 + delta_y**2

            return r_i + r_j
        return 0

    # Create constraints and bounds
    bounds = [(0, 1) for _ in range(26)]
    cons = []
    for i in range(26):

    result = optimize.minimize(

        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
    )

    centers = np.array(centers)
    radii = result.x
