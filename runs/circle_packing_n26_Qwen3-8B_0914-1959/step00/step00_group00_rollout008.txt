import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial guess: hexagonal packing with 6 rows of circles
    # Pack 5 rows of 5, 5, 5, 5, 5 circles (adjust to fit into 1x1 square)
    n = 26
    rows = 5
    cols = 5
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Create a hexagonal packing guess
    for i in range(n):
        row = i // cols
        col = i % cols
        x = 0.5 + (col - cols / 2) * (1 / (cols - 1)) if cols > 1 else 0.5
        y = 0.5 + (row - rows / 2) * (1 / (rows - 1)) if rows > 1 else 0.5
        dist = np.sqrt((col - (cols - 1) / 2) ** 2 + (row - (rows - 1) / 2) ** 2)
        radius = 0.1 * (1 / (1 + dist))
        centers[i] = [x, y]
        radii[i] = radius

    # Define the loss function (to be maximized)
    def objective(p):
        # Unpack parameters
        centers_packed = p[:n * 2].reshape(n, 2)
        radii_packed = p[n * 2:].reshape(n)

        # Initial loss (negative of sum of radii)
        loss = -np.sum(radii_packed)

        # Add penalty for boundary violations
        for i in range(n):
            x, y = centers_packed[i]
            r = radii_packed[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                loss += 1e6 * r

        # Add penalty for overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_packed[i, 0] - centers_packed[j, 0]
                dy = centers_packed[i, 1] - centers_packed[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii_packed[i] + radii_packed[j] - 1e-12:
                    loss += 1e6 * (radii_packed[i] + radii_packed[j])

        return loss

    # Optimization bounds
    bounds = []
    for _ in range(n * 2):
        bounds.append((0, 1))
    for _ in range(n):
        bounds.append((0, 0.5))  # Maximum radius is 0.5

    # Initial guess: center and radius for each circle
    initial_guess = np.concatenate([centers.flatten(), radii])

    # Perform optimization
    result = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        method='L-BFGS-B',
        options={'disp': False}
    )

    # Extract final centers and radii
    centers_final = result.x[:n * 2].reshape(n, 2)

    sum_radii = np.sum(radii_final)

    return centers_final, radii_final, sum_radii
