import numpy as np
import cvxpy as cp

def run_packing():
    # --- Step 1: Initial placement of 26 circles in a square grid (for exploration)
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):
        x = i // np.sqrt(num_circles)
        y = i % np.sqrt(num_circles)
        centers[i] = [x / np.sqrt(num_circles), y / np.sqrt(num_circles)]

    # --- Step 2: Define variables and constraints
    radii = cp.Variable(num_circles)
    obj = cp.Maximize(cp.sum(radii))
    constraints = []

    # --- Constraint 1: Circle within the unit square
    for i in range(num_circles):
        x, y = centers[i]
        constraints.append(x - radii[i] >= -1e-12)
        constraints.append(x + radii[i] <= 1 + 1e-12)
        constraints.append(y - radii[i] >= -1e-12)
        constraints.append(y + radii[i] <= 1 + 1e-12)

    # --- Constraint 2: Non-overlapping between circles
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # --- Step 3: Solve the optimization problem
    problem = cp.Problem(obj, constraints)
    problem.solve(solver=cp.CVXOPT, verbose=False)  # Use CVXOPT as it's a numerical solver

    # --- Step 4: Return results
    return centers, radii.value, np.sum(radii.value)
