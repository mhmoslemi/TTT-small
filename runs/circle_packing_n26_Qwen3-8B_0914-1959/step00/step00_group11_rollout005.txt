import numpy as np
import scipy.optimize as opt

def run_packing():
    # Step 1: Generate a hexagonal grid of circle centers
    n = 26
    # Number of rows in the hexagonal grid
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))

    # Generate hexagonal grid
    centers_hex = []
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * i
            y = i
            centers_hex.append([x, y])

    # Reduce the grid slightly to fit within the unit square
    scale = 1.0 / (np.max(centers_hex, axis=0) + 1e-6)
    centers_hex = np.array(centers_hex) * scale

    radii = np.full(n, 1.0 / (np.max(centers_hex, axis=0) + 1e-6))

    # Step 3: Use optimization to maximize sum of radii under constraints
    # Define the variables: the radii
    x0 = radii

    # Define the constraints for non-overlap
    def non_overlap_constraint(x, centers):
        n = len(x)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                lhs = dist - (x[i] + x[j])
                constraints.append(lhs >= 1e-6)
        return constraints

    constraints = non_overlap_constraint(x0, centers_hex)

    # Define the bounds for radii
    bounds = [(0.0, 1.0) for _ in range(n)]

    # Objective: maximize the sum of radii
    def objective(x):
        return -np.sum(x)

    # Run the optimization
    result = opt.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints
    )

    # Extract the optimal radii and centers

    sum_radii = np.sum(radii_opt)
