    rows, cols = 5, 6
    centers = []

    # Octave spacing: Adjust this parameter to get a better approximation of unit square
    spacing = 0.2

    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing * (3**0.5) / 2  # Hexagonal layout
            if i % 2 == 1:
                x += spacing / 2  # Offset for staggered rows
            centers.append([x, y])

    centers = np.array(centers)

    # Initial guess for radii: All equal and within bounds
    max_radius = min(1 - 0.1, 0.2)  # Reserve 0.1 buffer around square
    radii = np.full(len(centers), max_radius)

    # Objective: Maximize sum of (π * r_i^2), which is equivalent to maximizing sum(r_i^2)
    def objective(radii):
        return -np.sum(radii ** 2)  # Minimize negative sum to maximize positive sum

    # Constraints:
    # - Boundaries: (x - r >= 0) and (x + r <= 1)
    # - No overlap: dist(c_i, c_j) >= r_i + r_j

        constraints = []

        # Boundary constraints: Reserving 0.1 from each side

            r = radii[i]
            x, y = centers[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False

        # Ensure radii are non-negative
        for r in radii:
            if r < 0:
                return False

        # No overlaps between circles

                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j]:
                    return False

    radii_flat = radii.flatten()

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
