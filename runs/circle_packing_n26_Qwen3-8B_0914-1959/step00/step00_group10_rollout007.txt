import cvxpy as cp

def run_packing():

        constraints.append(y >= 0)

            dist = cp.sqrt(dx**2 + dy**2)

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    result = prob.solve()

    # Extract the optimal centers and radii
