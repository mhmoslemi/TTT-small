from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from typing import Tuple, List, Union

def generate_hexagonal_centers(n: int) -> np.ndarray:
    """Generate centers for a hexagonal packing of n circles."""
    row_count = int(np.sqrt(n))
    col_count = int(np.ceil(n / row_count))
    delta = np.sqrt(3) / 2  # distance between rows in a hexagonal grid
    centers = np.zeros((n, 2))
    idx = 0

            x = c + 0.5 * (r % 2)
            y = r * delta

                idx += 1
    return centers
