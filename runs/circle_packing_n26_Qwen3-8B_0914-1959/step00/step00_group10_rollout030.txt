import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define grid spacing and number of circles
    num_circles = 26
    grid_size = 5  # 5x5 grid with 1 circle per cell
    grid = np.linspace(0.1, 0.9, grid_size)

    # Generate grid positions
    centers = []
    for x in grid:
        for y in grid:
            centers.append([x, y])
    centers = np.array(centers)
    centers = np.vstack([centers, [0.5, 0.5]])  # Adjust for corner circles

    # Initial radii
    radii = np.full(num_circles, 0.05)

    # Optimization variables: radii
    bounds = [(0, 0.1) for _ in range(num_circles)]

    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraint_overlap(i, j, radii):
        x1, y1 = centers[i]
        x2, y2 = centers[j]
        dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
        return dist - (radii[i] + radii[j]) >= 1e-8

    def constraint_boundary(i, radii):
        x, y = centers[i]
        return x - radii[i] >= -1e-8 and x + radii[i] <= 1 + 1e-8 and y - radii[i] >= -1e-8 and y + radii[i] <= 1 + 1e-8

    # Define constraints in a list
    constraints = []
    for i in range(num_circles):

            constraints.append({
                'type': 'ineq',
                'fun': lambda r, i=i, j=j: constraint_overlap(i, j, r)
            })
        constraints.append({
            'type': 'ineq',
            'fun': lambda r, i=i: constraint_boundary(i, r)
        })

    # Optimize
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    # Extract results
    final_radii = result.x
    final_centers = centers[:len(final_radii)]
    final_sum = np.sum(final_radii)
