from scipy.spatial.distance import pdist, squareform

def validate_and_sum(centers, radii):
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    n = len(centers)
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-8 or x + r > 1 + 1e-8 or
            y - r < -1e-8 or y + r > 1 + 1e-8):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    return True, sum(radii)

def objective(params, n_circles=26):
    centers = params[:2*n_circles].reshape(n_circles, 2)
    radii = params[2*n_circles:]
    total = 0
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        total += r

            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx ** 2 + dy ** 2)
            if dist < radii[i] + radii[j] - 1e-8:
                return float('inf')  # Penalize overlaps
    return -total  # Maximize sum of radii

def hexagonal_grid(n_circles=26, width=1.0, height=1.0):
    cols = int(np.sqrt(n_circles * 2 / 3))
    rows = (n_circles + cols - 1) // cols
    centers = []
    for row in range(rows):
        for col in range(cols):
            x = col / (cols - 1) * width
            y = row / (rows - 1) * height
            if row % 2 == 1:
                x += width / (2 * (cols - 1))
            centers.append([x, y])
    return np.array(centers[:n_circles])
