def compute_overlap(centers, radii):
    n = len(centers)
    overlaps = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])

                overlaps.append((i, j))
    return overlaps
