import cvxpy as cp

def run_packing():
    n = 26

    # Generate a more optimal starting configuration
    # We'll use a more balanced distribution across the square
    # Starting with a rectangular grid and then fine-tuning

    x = np.linspace(0.1, 0.9, 5)
    y = np.linspace(0.1, 0.9, 5)
    x, y = np.meshgrid(x, y)
    x = x.flatten()
    y = y.flatten()

    # Add more circles to reach 26
    # Add 6 more circles in a more balanced location
    additional_x = np.random.uniform(0.1, 0.9, 6)
    additional_y = np.random.uniform(0.1, 0.9, 6)

    # Combine all centers
    centers = np.column_stack((x, y))
    centers = np.vstack((centers, np.column_stack((additional_x, additional_y))))
    centers = centers[:26]

    # Decision variables
    radii = cp.Variable(n)

    # Constraints
    constraints = []

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        constraints.append(x - radii[i] >= -1e-8)
        constraints.append(x + radii[i] <= 1 + 1e-8)
        constraints.append(y - radii[i] >= -1e-8)
        constraints.append(y + radii[i] <= 1 + 1e-8)

    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-8)

    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Formulate and solve the problem
    problem = cp.Problem(objective, constraints)
    problem.solve(max_iter=1000)

    centers_opt = centers
