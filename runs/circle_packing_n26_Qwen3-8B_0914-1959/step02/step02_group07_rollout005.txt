from scipy.optimize import minimize

def run_packing():

    # Letting the circle radii be optimized via the optimization process

    # Generate a random set of initial points within the unit square

    # Define the objective function to maximize the sum of the radii
    def objective(params):
        centers_flat = params[:n * 2]
        radii_flat = params[n * 2:]
        centers = centers_flat.reshape(n, 2)
        radii = radii_flat

        # Check if any circle is outside the unit square
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-8 or x + r > 1 + 1e-8 or

        # Check for overlapping circles
        for i in range(n):

                dx = centers[i, 0] - centers[j, 0]

                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-8:

        # Return -sum to maximize
        return -np.sum(radii)
