import numpy as np
from pymoo.core.problem import Problem
from pymoo.algorithms.nsga2 import NSGA2
from pymoo.optimize import minimize
from pymoo.operators.mutation.pmh import PMHMutation
from pymoo.operators.crossover.sbx import SBXCrossover
from pymoo.operators.selection.tournament import TournamentSelection
from pymoo.core.termination import SingleObjectiveTermination
from pymoo.operators.sampling.latin import LatinHypercubeSampling

class CirclePackingProblem(Problem):
    def __init__(self, n_circles):
        super().__init__(
            n_obj=1,
            n_constr=0,
            n_vars=2 * n_circles + n_circles,
            x_l=np.array([0.0] * (2 * n_circles) + [0.0] * n_circles),
            x_u=np.array([1.0] * (2 * n_circles) + [1.0] * n_circles),
            type=np.float64
        )
        self.n_circles = n_circles
        self.radii = np.full(n_circles, 0.01)

    def _evaluate(self, x, out, **kwargs):
        centers = x[:2 * self.n_circles].reshape(self.n_circles, 2)
        radii = x[2 * self.n_circles:]

        # Enforce bounds
        for i in range(self.n_circles):
            x_i, y_i = centers[i]
            r_i = radii[i]

            if x_i - r_i < -1e-8 or x_i + r_i > 1 + 1e-8:
                out["F"] = np.inf
                return
            if y_i - r_i < -1e-8 or y_i + r_i > 1 + 1e-8:
                out["F"] = np.inf
                return

        # Enforce non-overlapping
        for i in range(self.n_circles):
            for j in range(i + 1, self.n_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-8:
                    out["F"] = np.inf
                    return

        out["F"] = -np.sum(radii)  # Maximize sum of radii

def run_packing():
    n_circles = 26
    problem = CirclePackingProblem(n_circles)

    algorithm = NSGA2(
        pop_size=150,
        crossover=SBXCrossover(prob=0.9, eta=20),
        mutation=PMHMutation(eta=20, prob=0.1),
        selection=TournamentSelection(k=3)
    )

    termination = SingleObjectiveTermination(xtol=1e-8, atol=1e-8, max_iter=200)

    result = minimize(problem, algorithm, termination, verbose=False)

    centers = result.X[:2 * n_circles].reshape(n_circles, 2)
