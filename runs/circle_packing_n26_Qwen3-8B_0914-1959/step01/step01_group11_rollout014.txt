import cvxpy as cp

def run_packing():

    num_circles = 26
    grid = []
    for i in range(5):
        for j in range(6):
            x = j / 5
            y = i / 5
            grid.append((x, y))

    if len(grid) < num_circles:
        grid.append((0.5, 0.5))  # Add one more circle in the center

    centers_var = cp.Variable((num_circles, 2))
    radii_var = cp.Variable(num_circles)

    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(radii_var))

    # Constraints
    constraints = []

    # Boundary constraints (each circle must be inside the unit square)
    for i in range(num_circles):
        constraints.append(centers_var[i, 0] - radii_var[i] >= 0)
        constraints.append(centers_var[i, 0] + radii_var[i] <= 1)
        constraints.append(centers_var[i, 1] - radii_var[i] >= 0)
        constraints.append(centers_var[i, 1] + radii[i] <= 1)

    radii_var.value = radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

            if dist < radii[i] + radii[j] - 1e-12:
