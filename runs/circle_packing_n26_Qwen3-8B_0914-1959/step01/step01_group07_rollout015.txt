def generate_hexagonal_grid(num_circles, square_size=1.0):

def compute_distances(points):
    """Compute pairwise distances between all points."""
    n = points.shape[0]
    dists = np.zeros((n, n))
    for i in range(n):

            dy = points[i, 1] - points[j, 1]
            dists[i, j] = dists[j, i] = np.sqrt(dx * dx + dy * dy)
    return dists
