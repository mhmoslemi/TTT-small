import numpy as np
import math

def run_packing():

    n_rows = 6
    n_cols = 4

    vertical_spacing = r * math.sqrt(3)
    horizontal_spacing = 2 * r

    # Create a grid of centers
    centers = []
    for i in range(n_rows):
        y = i * vertical_spacing
        for j in range(n_cols):
            x = j * horizontal_spacing
            centers.append([x, y])

    # Convert to numpy array
    centers = np.array(centers)

    # All circles have the same radius
