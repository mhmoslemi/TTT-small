import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial guess for centers: grid-based with 5 rows and 5 columns
    num_circles = 26
    grid = np.linspace(0, 1 - 1e-10, 5)
    centers = np.array([
        [x, y] for x in grid for y in grid
    ])
    centers = centers[:num_circles]

    # Initial estimated radii based on grid spacing
    def grid_spacing(x, y):
        return np.min(np.abs(x - grid[np.abs(grid - x).argmin()])) * 0.9

    radii = np.array([grid_spacing(x, y) for x, y in centers])

    # Optimization: maximize sum of radii subject to constraints
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints_func(radii):
        # Ensure non-negative radii
        if np.any(radii < 0):
            return False

        # Check non-overlapping constraint
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Prepare optimization variables (radii)
    bounds = [(0, 1) for _ in range(num_circles)]
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[{'type': 'callback', 'fun': constraints_func}]
    )

    # Extract final centers and radii
    final_centers = centers
    final_radii = result.x

    total_sum = np.sum(final_radii)
    return (final_centers, final_radii, total_sum)
