from scipy.optimize import minimize
from itertools import combinations

def run_packing():

    num_circles = 26

        for i in range(26):
            x_i, y_i = pos[i]
            r_i = rad[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return np.inf
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return np.inf

    result = minimize(objective, x0, bounds=bounds, method='L-BFGS-B', options={'maxiter': 500})
