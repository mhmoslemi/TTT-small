    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def solve_lp_direction(a, solver):
    """
    Given fixed a, solve the LP:
      minimize t
      subject to: conv(a, b)[k] <= t for all k
                  sum(b) = 1
                  b >= 0
    Returns the optimal b.
    """
    n = len(a)
    num_lags = 2 * n - 1
    
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    
    constraints = [cp.sum(b_var) == 1]
    
    # Build constraint matrix efficiently
    # For lag k: sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[k-j] * b[j] <= t
    # We build this as a sparse matrix
    
    # For efficiency, build the constraint matrix directly
    # Row k has nonzeros at positions j where 0 <= j < n and 0 <= k-j < n
    # i.e., max(0, k-n+1) <= j <= min(n-1, k)
    # coefficient at position j is a[k-j]
    
    rows = []
    cols = []
    vals = []
    
    for k in range(num_lags):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A_mat = cp.sp_matrix((vals, (rows, cols)), shape=(num_lags, n))
    
    constraints.append(A_mat @ b_var <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def frank_wolfe_accelerated(a0, time_limit, solver, max_iters=200, verbose=True, rng_seed=42):
    """
    Accelerated Frank-Wolfe with full LP subproblem.
    """
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_obj = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 30
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t > 1:
            beta = (t - 2.0) / (t + 0.5)
            y = x + beta * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Solve LP to get direction
        direction = solve_lp_direction(y, solver)
        if direction is None:
            direction = x.copy()
        
        # Frank-Wolfe step
        gamma = 2.0 / (t + 2.0)
        x_new = (1.0 - gamma) * x + gamma * direction
        x_new = np.maximum(x_new, 0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / sx
        
        x_prev = x
        x = x_new
        
        # Evaluate
        obj = fast_autocorr(x).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_x = x.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_obj
            print(f"    FW t={t} ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
        
        if stall > max_stall:
            # Restart with perturbation
            pert = rng.normal(0, 0.01, n) * best_x
            x = np.maximum(best_x + pert, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            stall = 0
    
    return best_x, best_obj

Great, so the formula ∇g_k(a)^T b = 2*conv(a,b)[k] holds in general.

So the Frank-Wolfe subproblem is:
minimize t
subject to: 2*conv(a,b)[k] + g_k(a) <= t for all k = 0,...,2n-2
           sum(b) = 1
           b >= 0

Where g_k(a) = conv(a,a)[k] (the current autocorrelation values).

This is a linear program in (b, t) with:
- n+1 variables
- (2n-1) + 1 constraints (inequality + equality)
- The constraint matrix for the inequality part is 2*A(a) where A(a) is the (2n-1) × n matrix with A(a)[k,j] = a[k-j] if valid.

This is exactly what I had before, just with the right-hand side being t - g_k(a) instead of t.

So the LP is:
minimize t
subject to: (2*A(a) @ b) + g(a) <= t * ones
           sum(b) = 1
           b >= 0

Where g(a) = conv(a,a).

This can be rewritten as:
minimize t
subject to: 2*A(a) @ b - t*ones <= -g(a)
           sum(b) = 1
           b >= 0
           t is free (or t >= 0)

In CVXPY:
```python
b_var = cp.Variable(n, nonneg=True)
t_var = cp.Variable()
constraints = [cp.sum(b_var) == 1]
# 2 * conv(a,b)[k] + g_k(a) <= t for all k
# i.e., 2 * (A(a) @ b)[k] <= t - g(a)[k]
constraints.append(2 * A_mat @ b_var + g_a <= t_var)
prob = cp.Problem(cp.Minimize(t_var), constraints)
```

This should work! And it's a proper Frank-Wolfe step.

Now, the key question is: how fast does this converge? Since the problem is non-convex (the original objective is not convex), Frank-Wolfe is only guaranteed to converge to a stationary point. But in practice, it should work well.

1. The LP might be infeasible if the current point is "bad" (unlikely since b=a is always feasible)
2. The LP solver might be slow for large n
3. Nesterov acceleration might not help for non-convex problems

For point 2: with n=2400, the LP has ~5.76M nonzeros. ECOS can handle this. SCS might be faster.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full-constraint Frank-Wolfe with LP subproblem (all 2n-1 lag constraints)
2. Nesterov acceleration
3. Coordinate descent polishing with peak-aware transfers
4. Adaptive length selection (try multiple n values)
5. Diverse restarts

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(200, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    print("  LP solver (fallback): SCS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_conv_matrix_coeffs(a):
    """Build the sparse coefficient matrix for conv(a, b).
    Returns (rows, cols, vals) for the (2n-1) x n matrix where
    M[k,j] = a[k-j] if 0<=k-j<n, else 0.
    """
    n = len(a)
    num_lags = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(num_lags):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return np.array(rows), np.array(cols), np.array(vals)


def solve_lp_direction(a, solver):
    """
    Solve the Frank-Wolfe LP subproblem:
      minimize t
      subject to: 2*conv(a,b)[k] + conv(a,a)[k] <= t for all k
                  sum(b) = 1
                  b >= 0
    Returns optimal b.
    """
    n = len(a)
    num_lags = 2 * n - 1
    
    g_a = fast_autocorr(a)  # conv(a,a)
    
    rows, cols, vals = build_conv_matrix_coeffs(a)
    A_mat = cp.sp_matrix((vals * 2.0, (rows, cols)), shape=(num_lags, n))
    
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(b_var) == 1,
        A_mat @ b_var + g_a <= t_var
    ]
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-10:
        return None
    return b_val / b_sum


def frank_wolfe_lp(a0, time_limit, solver, max_iters=300, verbose=True, rng_seed=42):
    """
    Frank-Wolfe with exact LP subproblem and Nesterov acceleration.
    """
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_obj = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 25
    restart_count = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t > 1:
            beta = (t - 2.0) / (t + 0.5)
            y = x + beta * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Solve LP to get Frank-Wolfe direction
        direction = solve_lp_direction(y, solver)
        if direction is None:
            # Fallback: use current point
            direction = x.copy()
        
        # Frank-Wolfe step with Nesterov
        gamma = 2.0 / (t + 2.0)
        x_new = (1.0 - gamma) * x + gamma * direction
        x_new = np.maximum(x_new, 0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / sx
        
        x_prev = x
        x = x_new
        
        # Evaluate
        obj = fast_autocorr(x).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_x = x.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and (t % 10 == 0 or stall > max_stall - 5):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_obj
            print(f"    FW t={t} ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
        
        if stall > max_stall:
            stall = 0
            restart_count += 1
            # Perturb from best
            noise_scale = 0.005 + 0.005 * min(restart_count, 5)
            pert = rng.normal(0, noise_scale, n) * best_x
            x = np.maximum(best_x + pert, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
    
    return best_x, best_obj


def coordinate_descent_polish(a0, time_limit, verbose=True, rng_seed=123):
    """
    Coordinate descent: move mass from high-peak-contribution to low-peak-contribution positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = np.argsort(ac)[::-1][:min(20, len(ac))]
        
        improved = False
        
        # Compute peak sensitivity for each position
        peak_sens = np.zeros(n)
        for k in peak_k[:10]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            peak_sens[idx] += a[j_idx]
        
        # Find the position with highest sensitivity that has significant mass
        hot_candidates = np.where(peak_sens > 0)[0]
        if len(hot_candidates) == 0:
            stall += 1
            continue
        
        # Sort by sensitivity (highest first)
        hot_sorted = hot_candidates[np.argsort(peak_sens[hot_candidates])[::-1]]
        
        # Cold positions (low sensitivity)
        cold_candidates = np.where(peak_sens < np.percentile(peak_sens[hot_candidates], 20))[0]
        if len(cold_candidates) == 0:
            cold_candidates = np.argsort(peak_sens)[:max(1, n // 5)]
        
        n_hot_try = min(len(hot_sorted), 30)
        n_cold_try = min(len(cold_candidates), 30)
        
        for src in hot_sorted[:n_hot_try]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            
            for dst in cold_candidates[:n_cold_try]:
                if dst == src:
                    continue
                if a[dst] > a[src]:
                    continue
                
                for frac in [0.002, 0.005, 0.01, 0.02]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Escape: random perturbation
                n_perturb = max(5, n // 50)
                for _ in range(n_perturb):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.01, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0)
                        if a.sum() > 1e-15:
                            a = a / a.sum()
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True
                            else:
                                a[i] += amt
                                a[j] -= amt
                                a = np.maximum(a, 0)
                                if a.sum() > 1e-15:
                                    a = a / a.sum()
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full-constraint Frank-Wolfe with LP subproblem (all 2n-1 lag constraints)
2. Principled coordinate descent using subgradient-based pair scoring
3. Nesterov-accelerated FW with adaptive restarts
4. Multi-length exploration (sweep n to find optimal length)
5. Diverse restarts to escape local minima
Key improvement: exact Frank-Wolfe direction via full LP (vs. top-k approximation)
and a more principled coordinate descent using pair-scoring from active-set analysis.

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(100, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    print("  LP solver (fallback): SCS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_conv_rows(a):
    """Build sparse representation of conv(a, b) constraint matrix.
    Returns (rows, cols, vals) for matrix M where M[k,j] = a[k-j] if valid.
    Shape: (2n-1) x n
    """
    n = len(a)
    num_lags = 2 * n - 1
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(num_lags):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        count = j_end - j_start
        rows[idx:idx + count] = k
        cols[idx:idx + count] = np.arange(j_start, j_end)
        vals[idx:idx + count] = a[k - np.arange(j_start, j_end)]
        idx += count
    return rows[:idx], cols[:idx], vals[:idx]


def solve_lp_direction(a, solver, max_lags=None):
    """
    Solve Frank-Wolfe LP subproblem:
      minimize t
      subject to: 2*conv(a,b)[k] + conv(a,a)[k] <= t for all k
                  sum(b) = 1, b >= 0
    Returns optimal b (normalised to sum 1).
    """
    n = len(a)
    num_lags = 2 * n - 1
    g_a = fast_autocorr(a)

    if max_lags is not None and max_lags < num_lags:
        top_idx = np.argsort(g_a)[::-1][:max_lags]
        top_idx = np.sort(top_idx)
    else:
        top_idx = np.arange(num_lags)

    n_lags_used = len(top_idx)

    rows = np.empty(n * n_lags_used, dtype=np.int32)
    cols = np.empty(n * n_lags_used, dtype=np.int32)
    vals = np.empty(n * n_lags_used, dtype=np.float64)
    idx = 0
    for k in top_idx:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        count = j_end - j_start
        if count == 0:
            continue
        rows[idx:idx + count] = lambda i, ks=k_int: np.full(count, i) if False else i
        rows[idx:idx + count] = np.full(count, int(np.where(top_idx == k)[0][0]))
        cols[idx:idx + count] = np.arange(j_start, j_end)
        vals[idx:idx + count] = a[k_int - np.arange(j_start, j_end)]
        idx += count

    A_mat = cp.sp_matrix((vals * 2.0, (rows[:idx], cols[:idx])), shape=(n_lags_used, n))
    g_subset = g_a[top_idx]

    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()

    constraints = [
        cp.sum(b_var) == 1,
        A_mat @ b_var + g_subset <= t_var
    ]

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-10:
        return None
    return b_val / b_sum


def frank_wolfe_lp(a0, time_limit, solver, max_iters=300, verbose=True, rng_seed=42,
                   use_acceleration=True, max_lags_reduction=None):
    """
    Frank-Wolfe with LP subproblem. Optionally uses Nesterov acceleration.
    max_lags_reduction: if set, only use top-K lags in the LP (for speed).
    """
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_obj = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 20
    restart_count = 0
    iters_done = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Determine extrapolated point
        if use_acceleration and t > 1:
            beta = (t - 2.0) / (t + 0.5)
            y = x + beta * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Solve LP for Frank-Wolfe direction
        direction = solve_lp_direction(y, solver, max_lags=max_lags_reduction)
        if direction is None:
            direction = x.copy()

        # Frank-Wolfe step
        if use_acceleration:
            gamma = 2.0 / (t + 2.0)
        else:
            gamma = 2.0 / (t + 1.0)
        x_new = (1.0 - gamma) * x + gamma * direction
        x_new = np.maximum(x_new, 0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / sx

        x_prev = x
        x = x_new
        iters_done += 1

        # Evaluate
        obj = fast_autocorr(x).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_x = x.copy()
            stall = 0
        else:
            stall += 1

        if verbose and (t % 15 == 0 or stall > max_stall - 3):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_obj
            print(f"    FW t={t} ({elapsed:.1f}s): J={J:.6f}, stall={stall}, restarts={restart_count}")

        if stall > max_stall:
            stall = 0
            restart_count += 1
            noise_scale = 0.003 + 0.003 * min(restart_count, 10)
            pert = rng.normal(0, noise_scale, n) * best_x
            x = np.maximum(best_x + pert, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()

    return best_x, best_obj


def coordinate_descent_v2(a0, time_limit, verbose=True, rng_seed=123):
    """
    Principled coordinate descent: score each (src, dst) pair by estimated
    improvement in max autocorrelation, then greedily apply best moves.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        # Active set: lags within 99.5% of max
        active_threshold = 0.995 * M_curr
        active_lags = np.where(ac >= active_threshold)[0]
        if len(active_lags) == 0:
            active_lags = np.array([int(np.argmax(ac))])
        # Limit to top 15 active lags
        if len(active_lags) > 15:
            active_lags = active_lags[np.argsort(ac[active_lags])[::-1][:15]]

        # Compute pair score: for moving eps from i to j,
        # delta_max ~= eps * max_{k in active} (a[k-j] - a[k-i])
        # (with a[m] = 0 for invalid m)
        # We want negative delta (decrease in max), so score = -(a[k-j] - a[k-i])
        # Higher score = better move

        # Build sensitivity vector: for each position p, how much it contributes to active peaks
        # sens[p] = sum_{k in active} a[k-p] (clipped to valid range)
        sens = np.zeros(n)
        for k in active_lags:
            k_int = int(k)
            for p in range(max(0, k_int - n + 1), min(n, k_int + 1)):
                sens[p] += a[k_int - p]

        # For a move from src to dst: improvement ~ sens[src] - sens[dst]
        # (higher sens means more contribution to peaks, so removing from there helps)
        # Score for pair (src, dst) = sens[src] - sens[dst]

        # Find best pairs
        # src should have high sens, dst should have low sens
        sorted_by_sens = np.argsort(sens)[::-1]
        n_src = min(50, n)
        n_dst = min(50, n)

        src_candidates = sorted_by_sens[:n_src]
        dst_candidates = sorted_by_sens[-n_dst:]

        best_improvement = -1
        best_move = None
        best_frac = 0.0

        for src in src_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            src_sens = sens[src]
            if src_sens <= 0:
                continue

            for dst in dst_candidates:
                if dst == src:
                    continue
                dst_sens = sens[dst]
                score = src_sens - dst_sens
                if score <= 0:
                    continue
                # Estimate improvement
                # Try small fractions
                for frac in [0.003, 0.008, 0.015]:
                    amt = a[src] * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    improvement = best_m - M_trial
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_move = (src, dst, frac, a_trial.copy())

        if best_move is not None and best_improvement > 1e-15:
            src, dst, frac, a_trial = best_move
            a = a_trial
            best_m = fast_autocorr(a).max()
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                # Escape mechanism: random perturbation
                n_perturb = max(5, n // 40)
                found = False
                for _ in range(n_perturb):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.01, 0.06)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0)
                        if a.sum() > 1e-15:
                            a = a / a.sum()
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                found = True
                                break
                            else:
                                a[i] += amt
                                a[j] -= amt
                                a = np.maximum(a, 0)
                                if a.sum() > 1e-15:
                                    a = a / a.sum()

                if not found:
                    # Larger structural perturbation
                    n_swap = max(3, n // 100)
                    for _ in range(n_swap):
                        i = int(rng.integers(n))
                        j = int(rng.integers(n))
                        if i != j and a[i] > a[j]:
                            amt = (a[i] - a[j]) * 0.3
                            a[i] -= amt
                            a[j] += amt
                    a = np.maximum(a, 0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def subgradient_mirror_descent(a0, time_limit, c0=3.0, threshold_frac=0.99,
                               verbose=True, rng_seed=77):
    """
    Mirror descent with exact active-set subgradient (fast per-iteration).
    Used as a complement to the LP-based FW method.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_x = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 3000
    c_current = c0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        M = ac.max()
        if M < 1e-30:
            break

        # Exact subgradient from active set
        active = np.where(ac >= threshold_frac * M)[0]
        if len(active) == 0:
            active = np.array([int(ac.argmax())])
        if len(active) > 50:
            active = active[np.argsort(ac[active])[::-1][:50]]

        g = np.zeros(n)
        for k in active:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            g[idx] += 2.0 * y[j_idx]

        # Normalise gradient
        gn = np.linalg.norm(g)
        if gn > 1e-15:
            g = g / gn

        # Mirror step in log-space
        eta = c_current / (np.sqrt(t) + 1.0) * np.sqrt(n)
        log_a = np.log(np.maximum(y, 1e-15)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_x = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            pert = rng.normal(0, 0.005, n) * best_x
            x = np.maximum(best_x + pert, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            c_current *= 1.15
            if threshold_frac > 0.98:
                threshold_frac = max(0.98, threshold_frac - 0.003)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}, stall={stall}, c={c_current:.2f}")

    return best_x, best_m

        try_update(a_out, f"fw_full_n{n0}")

    # === Phase 2: Mirror descent with exact subgradient (fast iterations) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Mirror descent n={n_best} ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_out, _ = subgradient_mirror_descent(norm_sum1(best_seq), t2, c0=3.0,
                                               threshold_frac=0.99, verbose=True, rng_seed=3)
        try_update(a_out, f"md_n{n_best}")

    # === Phase 3: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord descent n={n_best} ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_out, _ = coordinate_descent_v2(norm_sum1(best_seq), t3, verbose=True, rng_seed=5)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 4: FW-LP with acceleration (different trajectory) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: FW-LP (accelerated) n={n_best} ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        a_out, _ = frank_wolfe_lp(norm_sum1(best_seq), t4, solver,
                                   max_iters=150, verbose=True, rng_seed=7,
                                   use_acceleration=True)
        try_update(a_out, f"fwaccel_n{n_best}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 5: Multi-length ===")
        lengths_to_try = []
        for delta in [200, -200, 400, -400]:
            nl = n_best + delta
            if 1000 <= nl <= 4000:
                lengths_to_try.append(nl)

        for nl in lengths_to_try[:3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            a_try = interpolate_sequence(best_seq, nl)
            t5 = min(remaining * 0.35, 120.0)
            if t5 > 25:
                print(f"  Trying n={nl}")
                a_out, _ = frank_wolfe_lp(norm_sum1(a_try), t5, solver,
                                           max_iters=80, verbose=True, rng_seed=9)
                try_update(a_out, f"fw_n{nl}")

    # === Phase 6: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final coord n={n_best} ===")
    if remaining > 30:
        t6 = min(remaining * 0.35, 120.0)
        if t6 > 20:
            a_out, _ = coordinate_descent_v2(norm_sum1(best_seq), t6, verbose=True, rng_seed=11)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 7: Final FW-LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t7 = min(remaining * 0.50, 100.0)
        if t7 > 20:
            a_out, _ = frank_wolfe_lp(norm_sum1(best_seq), t7, solver,
                                       max_iters=50, verbose=True, rng_seed=13,
                                       use_acceleration=False)
            try_update(a_out, f"fw_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
