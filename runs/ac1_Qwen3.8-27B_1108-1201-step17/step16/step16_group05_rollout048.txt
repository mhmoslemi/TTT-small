import numpy as np
import time

def fast_conv(a):
    """Compute linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # First project onto sum(x) = target_sum with x >= 0
    # Scale to sum 1, solve, then scale back
    v1 = v / target_sum if target_sum > 0 else v
    u = np.sort(v1)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    result = np.maximum(v1 - theta, 0.0)
    return result * target_sum

def conv_subgradient(a, k):
    """Compute gradient of (a*a)[k] w.r.t. a.
    grad[j] = 2*a[k-j] if 0 <= k-j < n, else 0."""
    n = len(a)
    grad = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n, k + 1)
    for j in range(j_start, j_end):
        idx = k - j
        if 0 <= idx < n:
            grad[j] = 2.0 * a[idx]
    return grad

def conv_subgradient_fast(a, k):
    """Vectorized version of conv_subgradient."""
    n = len(a)
    grad = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n, k + 1)
    idx = k - np.arange(j_start, j_end)
    valid = (idx >= 0) & (idx < n)
    if valid.any():
        grad[j_start + np.where(valid)[0]] = 2.0 * a[idx[valid]]
    return grad

def subgradient_method(a0, time_limit, n_active=5, verbose=True, seed=42):
    """Nesterov-accelerated subgradient method with active set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    c = fast_conv(a)
    best_m = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 2000
    c0 = 2.0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        # Compute convolution and find active lags
        c_y = fast_conv(y)
        M_y = c_y.max()
        
        # Adaptive active set: lags within a fraction of the max
        # Start with more active lags (smoother) and tighten over time
        frac = min(1.0, t / 5000.0)
        threshold = 1.0 - 0.3 * frac  # 0.7 -> 1.0 over 5000 iterations
        active_mask = c_y >= M_y * threshold
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(c_y)])
        
        # Cap active set size
        if len(active_idx) > 50:
            # Keep the top 50
            top_vals = c_y[active_idx]
            order = np.argsort(top_vals)[::-1][:50]
            active_idx = active_idx[order]
        
        # Compute subgradient (average over active lags)
        grad = np.zeros(n)
        for k in active_idx:
            grad += conv_subgradient_fast(y, int(k))
        grad /= len(active_idx)
        
        # Step size
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            gn = 1e-15
        eta = c0 / (np.sqrt(t) + 1.0) / gn
        
        # Projected subgradient step
        x_new = project_simplex(y - eta * grad, target_sum=1.0)
        
        x_prev = x
        x = x_new
        
        # Track best
        c_x = fast_conv(x)
        M_x = c_x.max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart on stall
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.5)
            noise = rng.normal(0, 0.003, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-15)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      SubGrad restart t={t} ({elapsed:.0f}s): J={J:.6f}")
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      SubGrad t={t} ({elapsed:.0f}s): J={J:.6f}, active={len(active_idx)}")
    
    return best_a, best_m

def kaiser_window(n, alpha):
    """Kaiser window of length n with parameter alpha."""
    if n == 1:
        return np.array([1.0])
    m = np.arange(n)
    center = (n - 1) / 2.0
    x = (m - center) / center
    from scipy.special import i0
    return np.i0(alpha * np.sqrt(np.maximum(1 - x**2, 0))) / np.i0(alpha)

def blackman_window(n):
    """Blackman window."""
    m = np.arange(n)
    return 0.42 - 0.5 * np.cos(2 * np.pi * m / (n - 1)) + 0.08 * np.cos(4 * np.pi * m / (n - 1))

def raised_cosine(n, beta=0.5):
    """Raised cosine window: f(x) = 0.5*(1 + cos(2*pi*x)) for x in [0,1]."""
    x = np.linspace(0, 1, n)
    return 0.5 * (1 + np.cos(2 * np.pi * x))

def optimal_continuous_shape(n):
    """Find optimal continuous shape using parametric optimization."""
    from scipy.optimize import minimize
    
    # Parameterize f as a sum of cosines: f(x) = sum_j c_j * cos(j*pi*x) for x in [0,1]
    # with f >= 0 constraint
    # Use a small number of harmonics
    J = 8
    x = np.linspace(0, 1, 400)
    
    def obj(params):
        c = params
        f = np.zeros_like(x)
        for j in range(J):
            f += c[j] * np.cos(j * np.pi * x)
        # Ensure non-negativity by squaring (or clipping)
        f = np.maximum(f, 0)
        s = f.sum() * (x[1] - x[0])
        if s < 1e-10:
            return 1e10
        f = f / s  # normalize to integral 1
        # Compute convolution (approximate)
        c_conv = np.convolve(f, f)
        return c_conv.max()
    
    # Initial guess: raised cosine
    c0 = np.zeros(J)
    c0[0] = 1.0
    c0[1] = 0.3
    
    result = minimize(obj, c0, method='Nelder-Mead', 
                      options={'maxiter': 5000, 'xatol': 1e-8, 'fatol': 1e-10})
    
    c_opt = result.x

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
