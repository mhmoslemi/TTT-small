def compute_subgradient_fft(a, ac, temperature):
    """Compute soft-max subgradient using FFT-based Hankel matrix-vector product.
    
    Returns G where G[m] = 2 * sum_j a[j] * w[m+j], w = softmax(ac/temperature).
    """
    n = len(a)
    # Compute soft-max weights
    log_w = ac / temperature
    log_w -= log_w.max()
    w = np.exp(log_w)
    W = w.sum()
    if W < 1e-30:
        w = np.ones(len(ac)) / len(ac)
    else:
        w = w / W
    
    # Compute G[m] = 2 * sum_j a[j] * w[m+j] using FFT
    N = 1 << (n + len(w) - 1 - 1).bit_length()
    A_fft = np.fft.rfft(a, N)
    W_fft = np.fft.rfft(w, N)
    G = 2.0 * np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    
    return G

A good schedule: start high (e.g., 0.1 * max_val) and decay to low (e.g., 0.001 * max_val).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Quick LP Frank-Wolfe for global optimization
2. Nesterov accelerated mirror descent with FFT-accelerated soft-max subgradient
3. Coordinate descent polish with exact 1D line search
Key innovation: O(n log n) subgradient computation via FFT-based Hankel matvec,
enabling 10x+ more iterations than O(n^2) approach.
"""

import numpy as np
import time
import cvxpy as cp

def fast_autocorr(a):
    """Compute full linear autocorrelation (a * a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate the sequence using the global evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to non-negative vector with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgradient_fft(a, ac, temperature):
    """Compute soft-max subgradient of max_k (conv(a,a))[k] in O(n log n).
    
    G[m] = 2 * sum_j a[j] * w[m+j] where w = softmax(ac / temperature)
    Uses FFT-based Hankel matrix-vector product.
    """
    n = len(a)
    log_w = ac / temperature
    log_w -= log_w.max()
    w = np.exp(log_w)
    W = w.sum()
    if W < 1e-30:
        w = np.ones(len(ac)) / len(ac)
    else:
        w = w / W
    
    # G[m] = 2 * sum_{j=0}^{n-1} a[j] * w[m+j]
    # Use FFT cross-correlation: ifft(FFT(a) * conj(FFT(w)))[:n]
    N = 1 << (n + len(w) - 1 - 1).bit_length()
    A_fft = np.fft.rfft(a, N)
    W_fft = np.fft.rfft(w, N)
    G = 2.0 * np.fft.irfft(A_fft * np.conj(W_fft), N)[:n]
    return G

def nesterov_mirror_descent(a0, time_limit, init_temp_frac=0.1, final_temp_frac=0.002,
                            c0=5.0, verbose=True, seed=42):
    """Nesterov accelerated mirror descent with FFT subgradient."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 5000
    running_sum = np.zeros(n)
    count = 0
    best_stall_limit = 5000
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature (fraction of max autocorrelation)
        frac = min(1.0, t / 10000.0)
        temp_frac = init_temp_frac * (1 - frac) + final_temp_frac * frac
        temp_frac = max(temp_frac, 1e-4)
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation and subgradient
        ac = fast_autocorr(y)
        max_ac = ac.max()
        temperature = temp_frac * max_ac
        G = compute_subgradient_fft(y, ac, temperature)
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_y = np.log(y)
        log_new = log_y - eta * G
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        # Track best
        M = fast_autocorr(x).max()
        if M < best_m - 1e-16:
            best_m = M
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Ergodic average
        running_sum += x
        count += 1
        if count % 1000 == 0:
            avg = running_sum / count
            avg = np.maximum(avg, 0.0)
            s_avg = avg.sum()
            if s_avg > 1e-15:
                avg = avg / s_avg
                M_avg = fast_autocorr(avg).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg.copy()
        
        # Periodic restart on stall
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(2000, max_stall // 2)
            pert = rng.normal(0, 0.003, n)
            x_r = best_a * (1.0 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            c0 = max(2.0, c0 * 0.7)
            # Reset ergodic
            running_sum = np.zeros(n)
            count = 0
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): score={score:.8f}, stall={stall_count}, c0={c0:.2f}")
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, max_constraints=400, verbose=True):
    """LP-based Frank-Wolfe for global optimization."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Select solver
    solver = None
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.SCS
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Select active constraints
        threshold = M_curr * 0.95
        active_lags = np.where(ac >= threshold)[0]
        if len(active_lags) > max_constraints:
            active_lags = np.argsort(ac)[::-1][:max_constraints]
        
        # Build constraint matrix
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((len(active_lags), n))
        for row_idx, k in enumerate(active_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]
        
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Frank-Wolfe update with Nesterov momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): score={2*n*best_m:.8f}, active={len(active_lags)}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=True, seed=789):
    """Coordinate descent with multi-target mass transfer."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 3
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_moves = min(100, n // 2)
        
        for _ in range(n_moves):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            if a[i] < 1e-14:
                continue
            n_targets = min(12, n)
            targets = rng.integers(0, n, size=n_targets)
            
            best_trial_m = best_m
            best_move = None
            
            for dst in targets:
                if dst == i:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[i] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[dst] += amt
                    if a_trial[i] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_trial_m - 1e-17:
                        best_trial_m = m_trial
                        best_move = (i, dst, frac)
            
            if best_move is not None:
                i, dst, frac = best_move
                amt = a[i] * frac
                a[i] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_trial_m
                best_a = a.copy()
                improved = True
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
        
        if verbose and stall % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.1f}s): score={2*n*best_m:.8f}, stall={stall}")
    
    return best_a, best_m

def interpolated_init(n, parent=None):
    """Create initialization, optionally interpolating from parent."""
    if parent is not None and len(parent) > 0:
        p = np.asarray(parent, dtype=np.float64)
        p = np.maximum(p, 0)
        s = p.sum()
        if s < 1e-15:
            p = np.ones(len(p))
            s = len(p)
        p = p / s
        old_x = np.linspace(0, 1, len(p))
        new_x = np.linspace(0, 1, n)
        result = np.interp(new_x, old_x, p)
        result = np.maximum(result, 0)
        ss = result.sum()
        if ss > 1e-15:
            return result / ss
    # Default: raised cosine (good low-autocorrelation shape)
    x = np.linspace(0, 2 * np.pi, n)
    result = 0.5 * (1 + np.cos(x))
    result = np.maximum(result, 1e-10)
    return result / result.sum()
