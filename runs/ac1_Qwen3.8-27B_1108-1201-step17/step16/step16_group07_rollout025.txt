**Approach A: Iterate on the autocorrelation domain**
Instead of optimizing in the "time" domain (the sequence a), optimize in the "autocorrelation" domain. Given a target autocorrelation c, find a nonnegative a that produces it. This is a matrix factorization problem (a*a = c means a is a "square root" of c under convolution).

**Approach B: Use the KKT conditions more directly**

**Approach C: Alternating minimization with regularization**

**Approach D: Use the structure of the problem**
The problem is: minimize max_k c[k] subject to a >= 0, sum(a) = 1 (wlog).
This is equivalent to: find the smallest M such that c[k] <= M for all k, a >= 0, sum(a) = 1.
This is a feasibility problem in (a, M), and M is the objective.

We can use bisection on M and check feasibility. The feasibility check is: does there exist a >= 0, sum(a) = 1, such that (a*a)[k] <= M for all k?

This is an LP! (The constraints (a*a)[k] <= M are bilinear in a, so it's NOT an LP in a.)

Wait, (a*a)[k] = sum_i a[i]*a[k-i] is quadratic in a. So the feasibility problem is a quadratic constraint, not linear.

However, the previous approach uses a "linearization" trick: fix a = a_fixed, then solve for b such that (a_fixed * b)[k] <= M. This is linear in b.

So the approach is:
1. Fix a
2. Find b that maximizes sum(b) subject to (a*b)[k] <= M for all k, b >= 0
3. Update a towards b

This is a "linearized" or "ascent" method. The issue is that it only uses first-order information (the linearization of the quadratic constraint).

**Approach E: Use second-order information**

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    csv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (csv - S))[0][-1]
    theta = (csv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_l2_gradient(a, n):
    """Compute gradient of ||conv(a,a)||_2^2 w.r.t. a, efficiently via FFT."""

    # cross[j] = sum_{m=0}^{n-1} a[m] * c[j+m]

    # FFT(a) * FFT(c) gives circular conv of a and c

    # = sum_m a[m]*c[m+j] = (a * c)[j] in the "one-sided" sense
    # Using FFT: if we compute IFFT(FFT(a_rev) * FFT(c)), we get:
    # result[k] = sum_m a_rev[m] * c[k-m] = sum_m a[n-1-m] * c[k-m]
    # Let m' = n-1-m: result[k] = sum_{m'} a[m'] * c[k-n+1+m']

    diag_idx = np.arange(n)

def l2_flatten(a0, time_limit, verbose=False, seed=0):
    """Minimize ||conv(a,a)||_2^2 subject to a >= 0, sum(a) = sqrt(2n) via Nesterov projected gradient."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s

    best_a = a.copy()
    best_l2 = np.sum(fast_autocorr(a) ** 2)
    best_max = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    t = 0
    step = 1.0 / np.sqrt(n)
    patience = 0
    max_patience = 2000

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, c = compute_l2_gradient(y, n)
        # Also track max
        max_c = c.max()

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * (y - step * grad)
        y_new = project_simplex(y_new, norm_val)

        momentum = min((t - 1.0) / (t + 2.0), 0.9)
        a_new = y_new + momentum * (y_new - y_prev)
        a_new = project_simplex(a_new, norm_val)

        y_prev = y
        y = y_new
        a = a_new

        # Evaluate
        conv_a = fast_autocorr(a)
        l2_a = np.sum(conv_a ** 2)
        max_a = conv_a.max()

        if l2_a < best_l2 - 1e-10:
            best_l2 = l2_a
            best_a = a.copy()
            patience = 0
            step = min(step * 1.002, 5.0 / n)
        else:
            patience += 1
            if patience % 50 == 0:
                step *= 0.95
            if patience > max_patience:
                if verbose:
                    print(f"    L2Flatten: stalled at t={t}, best_l2={best_l2:.6f}, best_max={best_max:.8f}")
                break

        # Update best_max tracking (we care about max, not just L2)
        if max_a < best_max:
            best_max = max_a

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    L2Flatten t={t} ({elapsed:.0f}s): max={max_a:.8f}, l2={l2_a:.4f}, step={step:.6e}")

    return best_a, best_max

        a = a * norm_val / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 2.0 / n
    patience = 0
    max_patience = 3000
    t = 0
    lambda_l2 = 0.05

    while time.monotonic() - t0 < time_limit:
        t += 1
        c = fast_autocorr(a)
        M = c.max()

        # Subgradient of max(c): pick top peaks
        n_peaks = max(5, min(20, n // 60))
        top_k = np.argsort(c)[-n_peaks:]
        weights = np.exp(-10.0 * (M - c[top_k]) / max(M * 0.01, 1e-10))
        weights /= weights.sum()

        # Subgradient of max(c)
        sg = np.zeros(n)
        for w, k in zip(weights, top_k):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            idx = np.arange(lo, hi)
            sg[lo:hi] += w * a[k - idx]
            if 2 * lo <= k <= 2 * hi - 2:
                # Diagonal correction (minor)
                pass

        # L2 gradient
        l2_grad, _ = compute_l2_gradient(a, n)
        l2_scale = lambda_l2 * np.sum(c ** 2) / max(M, 1e-10)

        # Combined gradient
        grad = sg + l2_scale * l2_grad
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        a_trial = a - step * grad * np.sqrt(n)
        a_trial = project_simplex(a_trial, norm_val)

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.003, 10.0 / n)
        else:
            patience += 1
            if patience % 30 == 0:
                step *= 0.95
            if patience > max_patience:
                break

        # Adaptive lambda
        if t % 1000 == 0:
            lambda_l2 = max(0.01, lambda_l2 * 0.8)

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Combined t={t} ({elapsed:.0f}s): score={score_trial:.8f}, step={step:.6e}")

    return best_a, best_score


def subgradient_at_k(a, k, n):
    """Compute subgradient of (a*a)[k] w.r.t. a at index k."""
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    idx = np.arange(lo, hi)
    grad[lo:hi] += a[k - idx]
    return grad


def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)

    if size == 0:
        return None, None

    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0, pressure_thresh=0.03):
    """Nesterov-accelerated LP best-response with pressure-based active set."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    t = 0
    stall_count = 0
    stall_threshold = 80
    conv_size = 2 * n - 1

    # Pressure tracking: how "pressed" each constraint is
    pressure = np.zeros(conv_size)
    pressure_alpha = 0.1  # EMA coefficient for pressure

    while time.monotonic() - t0 < time_limit:
        t += 1

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Update pressure: high where constraints are near-tight
        tightness = conv_y / max(M_curr, 1e-10)
        pressure = (1 - pressure_alpha) * pressure + pressure_alpha * tightness

        # Active set: top pressure constraints + top autocorrelation values
        n_active = min(int(1.2 * n), conv_size)
        # Combine pressure and current values
        combined_score = 0.6 * pressure + 0.4 * tightness
        active_set = np.argsort(combined_score)[-n_active:]

        # Always include the top max values
        n_top = max(3, n // 100)
        top_k = np.argsort(conv_y)[-n_top:]
        active_set = np.unique(np.concatenate([active_set, top_k]))

        # Cap
        if len(active_set) > int(1.8 * n):
            keep = np.argsort(combined_score[active_set])[-int(1.8 * n):]
            active_set = active_set[keep]

        b, duals = lp_best_response(y, M_curr, active_set, solver)

        if b is None:
            # Fallback: use only top constraints
            active_set_small = np.argsort(conv_y)[-max(5, n // 50):]
            b, duals = lp_best_response(y, M_curr, active_set_small, solver)
            if b is None:
                # Skip this iteration
                y = y_prev  # revert
                stall_count += 1
                if stall_count > stall_threshold:
                    y = best_a.copy()
                    y_prev = best_a.copy()
                    a = best_a.copy()
                    stall_count = 0
                    pressure = np.zeros(conv_size)
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = project_simplex(y_new, norm_val)

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.91)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            a_new = project_simplex(a_new, norm_val)

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        score_a = fast_autocorr(a).max()
        cur_best = min(score_y, score_a)
        if cur_best < best_score - 1e-14:
            best_score = cur_best
            best_a = (y if score_y < score_a else a).copy()
            stall_count = 0
        else:
            stall_count += 1

        # Ergodic average
        if t % 30 == 0:
            a_avg = 0.5 * best_a + 0.5 * a
            a_avg = project_simplex(a_avg, norm_val)
            score_avg = fast_autocorr(a_avg).max()
            if score_avg < best_score - 1e-14:
                best_score = score_avg
                best_a = a_avg.copy()
                stall_count = 0

        # Restart if stalled
        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            pressure = np.zeros(conv_size)
            stall_count = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP restart at t={t} ({elapsed:.0f}s): best={best_score:.10f}")

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s): best={best_score:.10f}, stalls={stall_count}")

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, verbose=False, seed=0):
    """Fast coordinate descent with FFT-based updates."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    max_no_improve = 6
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        conv = fast_autocorr(a)

        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            s_others = norm_val - aj

            lo = max(0, j)
            hi = min(size, j + n)
            if lo >= hi:
                continue

            # Try a geometric grid of values around current
            typical = norm_val / n
            candidates = [0.0]
            if aj > 1e-8:
                candidates += [aj * 0.5, aj * 0.75, aj * 1.25, aj * 1.5, aj * 2.0]
            candidates += [typical * 0.3, typical * 0.7, typical, typical * 1.5, typical * 2.0, typical * 3.0]
            candidates = sorted(set([max(0.0, min(c, norm_val * 0.3)) for c in candidates]))

            best_x = aj
            best_obj = conv.max()

            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue

                new_conv = conv.copy()
                delta = x - aj
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta

                scale = norm_val / new_sum
                new_max = new_conv.max() * scale * scale

                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm_val / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        conv = fast_autocorr(a)
                        improved = True
                    else:
                        a = best_a.copy()
                        conv = fast_autocorr(a)

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence."""
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False


def generate_structured_init(n, mode=0, rng=None):
    """Generate structured initial sequences."""
    if rng is None:
        rng = np.random.default_rng(42)
    if mode == 0:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif mode == 1:
        # Hann-like
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a ** 0.7  # Sharpen
    elif mode == 2:
        # Uniform with random perturbation
        a = np.ones(n) * (1 + 0.3 * rng.standard_normal(n))
    elif mode == 3:
        # Two-lobe
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.4 * np.cos(4 * np.pi * x) + 0.2 * np.cos(8 * np.pi * x)
    elif mode == 4:
        # Exponential decay from center
        center = n // 2
        idx = np.abs(np.arange(n) - center)
        a = np.exp(-idx / (n * 0.3))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a / a.sum()


def upsample_sequence(a, new_n, rng):
    """Upsample a sequence to a new length using linear interpolation + noise."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    # Add small noise
    noise = 1.0 + 0.05 * rng.standard_normal(new_n)
    a_new = a_interp * np.maximum(noise, 0.5)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
