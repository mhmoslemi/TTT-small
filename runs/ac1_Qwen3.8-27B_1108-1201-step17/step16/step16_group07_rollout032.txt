        y_prev = y
        y = y_new
        a = a_trial if score < best_score + 1e-12 else a_next
        
        # Restart if stalled
        if stall_count > stall_threshold:
            a = best_a.copy()
            y = best_a.copy()
            y_prev = best_a.copy()
            step = norm_val / (n * t_param * 0.3)
            stall_count = 0
            t_param = t_param * 1.3  # Increase sharpness
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NS t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.4f}, t_param={t_param:.1f}")
    
    return best_a, best_score

def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        if k == 0:
            T[idx, :] = 2.0 * a_fixed
        else:
            T[idx, 0:n-k] = a_fixed[k:n]
            T[idx, k:n] = a_fixed[0:n-k]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

def nesterov_lp(a0, time_limit, solver, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response with adaptive active set."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    conv_size = 2 * n - 1
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a * norm_val / a.sum()
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    y_prev = a.copy()
    active_set = None
    stall_count = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        
        # Adaptive active set: tighten over time
        if active_set is None or t % 30 == 0:
            progress = min(1.0, t / 500.0)
            tol = 0.10 * (1.0 - progress) + 0.01 * progress
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            n_top = max(5, min(50, n // 40))
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k.astype(int)]))
            if len(active_set) > 80:
                conv_at_active = conv_y[active_set]
                keep = np.argsort(conv_at_active)[-80:]
                active_set = active_set[keep]
        
        b, duals = lp_best_response(y, M_curr, active_set, solver)
        
        if b is None:
            if t % 10 == 0:
                active_set = np.argsort(conv_y)[-30:]
                b, duals = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.90)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a
        
        y_prev = y
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        
        if stall_count > 60:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def freq_perturb(a, rng, scale=0.3):
    """Perturb in frequency domain for diverse restarts."""
    n = len(a)
    A = rfft(a, 2 * n)
    # Add random phase to non-DC components
    phase = rng.uniform(0, 2 * np.pi, A.shape)
    amp_factor = 1.0 + scale * rng.standard_normal(A.shape)
    amp_factor = np.maximum(amp_factor, 0.1)
    A_new = A * amp_factor * np.exp(1j * phase * scale)
    # Keep DC component
    A_new[0] = A[0]
    a_new = irfft(A_new, 2 * n)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def coordinate_descent(a0, time_limit, verbose=False, seed=0):
    """Fast cycle coordinate descent."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm_val / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    max_no_improve = 5
    
    while time.monotonic() - t0 < time_limit:
        if no_improve >= max_no_improve:
            break
        improved = False
        conv = fast_autocorr(a)
        current_max = conv.max()
        
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            aj = a[j]
            if aj < 1e-15:
                # Try turning on
                candidates = [norm_val/n * 0.5, norm_val/n, norm_val/n * 2]
            else:
                typical = norm_val / n
                candidates = [0.0, aj*0.5, aj*1.5, typical*0.5, typical, typical*2]
            
            best_x = aj
            best_obj = current_max
            
            for x in candidates:
                x = max(0.0, min(x, norm_val))
                if abs(x - aj) < 1e-12:
                    continue
                delta = x - aj
                new_conv = conv.copy()
                lo = j
                hi = min(size, j + n)
                new_conv[lo:hi] += 2.0 * delta * a[lo-j:hi-j]
                if 2*j < size:
                    new_conv[2*j] += delta*delta
                # Scale for normalization (approximate)
                new_sum = norm_val  # We're normalized, so changing one element changes sum
                # Actually let's just compute the raw max and compare
                new_max_raw = new_conv.max()
                if new_max_raw < best_obj:
                    best_obj = new_max_raw
                    best_x = x
            
            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s = a.sum()
                if s > 1e-15:
                    a = a * norm_val / s
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                        conv = fast_autocorr(a)
                        current_max = conv.max()
                    else:
                        a = best_a.copy()
                        conv = fast_autocorr(a)
                        current_max = conv.max()
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
    
    return best_a, best_score

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score
