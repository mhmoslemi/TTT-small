Step 2 is a quadratic minimization: minimize a^T (sum_k q[k] * M_k) a subject to a >= 0, sum(a) = 1. The matrix Q = sum_k q[k] * M_k is a weighted sum of antidiagonal matrices. This is a QCQP with a single quadratic constraint... wait, it's minimize a^T Q a subject to a >= 0, sum(a) = 1. This is a Quadratic Programming problem (QP), which is convex if Q is PSD.

Is Q = sum_k q[k] * M_k PSD? M_k has eigenvalues on the unit circle (it's a partial permutation matrix). A weighted sum of such matrices with positive weights... it's not necessarily PSD. For example, if q is concentrated on a single k, Q = M_k which is not PSD.

So step 2 is a non-convex QP in general. We can solve it with local methods (gradient descent, BFGS, etc.).

This "game-theoretic" approach is interesting but might be overkill. Let me stick with the simpler subgradient approach.

import numpy as np

def fast_autocorr(a):
    # FFT-based autocorrelation (linear convolution)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    a_list = [float(max(0, min(1000, x))) for x in a]
    if sum(a_list) < 0.01:
        return float('inf')
    return float(evaluate_sequence(a_list))

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def proj_simplex(v):
    # Project v onto {x >= 0, sum(x) = 1}
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Standard algorithm
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_topk(a, ac, top_k=10):
    # Return subgradient using top-k lags
    n = len(a)
    # Find top-k lags
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    # Uniform weights (or proportional to values)
    w = np.ones(k_max) / k_max
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        for i in range(n):
            j = k_int - i
            if 0 <= j < n:
                grad[i] += 2.0 * wi * a[j]
    return grad

def nesterov_mirror_descent_v2(a0, time_limit, top_k=10, c0=3.0, verbose=True):
    # Nesterov mirror descent with exact subgradient
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute autocorrelation
        ac = fast_autocorr(y)
        
        # Compute subgradient
        g = subgradient_topk(y, ac, top_k)
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        a_new = np.exp(log_a - log_a.max())
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        # Check improvement
        M = fast_autocorr(x).max()
        if M < best_m:
            best_m = M
            best_a = x.copy()
        
        if verbose and t % 5000 == 0:
            print(f"    NMD2 t={t}: J={2*n*best_m:.6f}")
    
    return best_a, best_m

def frequency_smoothing(a0, n_iters=100, alpha=0.1):
    # Frequency-domain flattening
    n = len(a0)
    a = norm_sum1(a0)
    N = 1 << (2*n-2).bit_length()  # FFT size >= 2n-1
    
    for it in range(n_iters):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        # Compute target magnitude (geometric mean or arithmetic mean)
        # Use arithmetic mean of non-DC magnitudes
        if len(mag) > 1:
            mean_non_dc = mag[1:].mean()
            # Blend current magnitude with target
            new_mag = np.empty_like(mag)
            new_mag[0] = mag[0]  # Keep DC (preserves sum)
            new_mag[1:] = (1-alpha) * mag[1:] + alpha * mean_non_dc
            # Reconstruct with new magnitude, same phase
            phase = np.angle(A)
            A_new = new_mag * np.exp(1j * phase)
            a_new = np.fft.irfft(A_new, N)[:n]
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-15:
                a = a_new / s
            else:
                break
    
    return a

def coordinate_descent_v2(a0, time_limit, verbose=True):
    # Coordinate descent with 2-swap
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    while time.monotonic() - t0 < time_limit and stall < 20:
        improved = False
        for _ in range(min(100, n)):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                amt = min(a[i], a[j]) * frac
                if amt < 1e-15:
                    continue
                # Try both directions
                for sign in [1, -1]:
                    a_trial = a.copy()
                    a_trial[i] -= sign * amt
                    a_trial[j] += sign * amt
                    if a_trial[i] < -1e-15 or a_trial[j] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    m = fast_autocorr(a_trial).max()
                    if m < best_m - 1e-15:
                        a = a_trial
                        best_m = m
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                amt = a[i] * 0.01
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
        
        if verbose and stall % 5 == 0:
            print(f"    CD2 ({time.monotonic()-t0:.1f}s): J={2*n*best_m:.6f}")
    
    return best_a, best_m
