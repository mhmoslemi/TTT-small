Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Key insight: For non-negative sequences, max autocorrelation is always at lag 0

The uniform sequence achieves exactly 2. This program verifies this and explores

import numpy as np

def fast_autocorr(a):

def mirror_descent_subgrad(a, time_limit, c0=2.0, verbose=True, rng_seed=42):
    """
    Mirror descent using the exact subgradient at the peak lag.
    For non-negative a, peak is at lag 0, so subgrad[i] = 2*n*a[i].
    This drives toward uniform. We still run it to confirm.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n, 2.0
