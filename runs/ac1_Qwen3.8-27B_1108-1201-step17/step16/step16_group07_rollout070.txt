```python
"""
Optimization for minimizing
    2*n*max(conv(a,a)) / (sum(a))^2
over nonnegative finite sequences.

The program keeps the sequence normalized to sum(a) = sqrt(2n), so the
objective is simply max(conv(a,a)).  It combines:
  1. LP-based best response with Nesterov-style averaging and line search.
  2. Exact incremental coordinate descent with one-dimensional resampling.
  3. A lightweight smoothed-subgradient method for occasional escape.
  4. Resampling to a slightly larger length when time remains.

All helper functions are top-level.  No lambdas or closures are used.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

try:
    import cvxpy as cp
except Exception:
    cp = None

_LP_SOLVER = None


def get_lp_solver():
    """Pick one available CVXPY LP solver, if any."""
    global _LP_SOLVER
    if _LP_SOLVER is not None:
        return _LP_SOLVER
    if cp is None:
        _LP_SOLVER = None
        return None

    for name in ["CLARABEL", "PDLP", "SCS", "ECOS"]:
        s = getattr(cp, name, None)
        if s is None:
            continue
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(
                cp.Minimize(cp.sum(x)),
                [cp.sum(x) == 1, x <= 0.4]
            )
            prob.solve(solver=s, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                _LP_SOLVER = s
                print(f"  LP solver: {name}")
                return _LP_SOLVER
        except Exception:
            continue

    _LP_SOLVER = None
    return None


def fast_autocorr(a):
    """Full convolution a * a returned as a real array."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    out = irfft(A * A, fft_size)[:size]
    return np.asarray(out, dtype=np.float64)


def normalize_to(a):
    """Clip to nonnegative finite values and scale to sum sqrt(2n)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    if n == 0:
        return a.copy()
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.full(n, norm / n, dtype=np.float64)
    return a * (norm / s)


def score_of(a):
    """Internal score for a normalized sequence: max convolution."""
    if a is None or len(a) == 0:
        return float("inf")
    return float(fast_autocorr(a).max())


def try_update(best_seq, best_score, cand, label, t0):

def pressure_for_order(a, conv, n_peaks):
    """
    Heuristic pressure on coordinates from the largest lags.
    Used only to order coordinates in one CD pass.
    """
    a = np.asarray(a, dtype=np.float64)
    conv = np.asarray(conv, dtype=np.float64)
    n = a.size
    W = np.zeros(n, dtype=np.float64)
    if n == 0 or conv.size == 0:
        return W

    n_peaks = int(min(max(1, n_peaks), conv.size))
    idx = np.argsort(conv)[-n_peaks:]
    vals = conv[idx]
    vmax = float(vals.max())
    vmin = float(vals.min())
    width = max(vmax - vmin, 1e-8)
    weights = np.exp(-4.0 * (vmax - vals) / width)
    ws = weights.sum()
    if ws <= 0:
        weights = np.ones(n_peaks, dtype=np.float64) / n_peaks
    else:
        weights = weights / ws

    for k, w in zip(idx, weights):
        k = int(k)
        w = float(w)
        if w <= 0:
            continue
        l = max(0, k - n + 1)
        r = min(n, k + 1)
        if l < r:
            W[l:r] += 2.0 * w * a[k - np.arange(l, r)]
    return W


def choose_active_set(conv, n, iteration, stall):
    """Select lag constraints for the LP best-response problem."""
    conv = np.asarray(conv, dtype=np.float64)
    m = conv.size
    if m == 0:
        return np.arange(0, dtype=int)

    M = float(conv.max())
    if M <= 0.0:
        return np.arange(m, dtype=int)

    progress = min(1.0, iteration / 600.0)
    tol = 0.10 * (1.0 - progress) + 0.004 * progress
    if stall > 20:
        tol = min(tol, 0.02)

    threshold = M * (1.0 - tol)
    active = np.where(conv >= threshold)[0]

    base_top = max(10, n // 30)
    growth = int(min(90, iteration // 50))
    n_top = min(m, base_top + growth)
    top = np.argsort(conv)[-n_top:]

    active = np.unique(np.concatenate((active, top)))

    max_active = min(m, max(800, int(1.25 * n)))
    if active.size > max_active:
        keep = np.argsort(conv[active])[-max_active:]
        active = active[keep]

    return active.astype(int, copy=False)


def lp_best_response(a_fixed, M, active_set, solver):
    """
    Solve:
        max sum(b)
        s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0.
    """
    if solver is None or cp is None:
        return None

    a_fixed = np.asarray(a_fixed, dtype=np.float64)
    n = a_fixed.size
    active_set = np.asarray(active_set, dtype=int)
    m = active_set.size
    if n == 0 or m == 0:
        return None
    if not np.isfinite(M) or M <= 0.0:
        return None

    T = np.zeros((m, n), dtype=np.float64)
    for idx in range(m):
        k = int(active_set[idx])
        l = max(0, k - n + 1)
        r = min(n, k + 1)
        if l < r:
            T[idx, l:r] = a_fixed[k - np.arange(l, r)]

    b = cp.Variable(n, nonneg=True)
    constraints = [T @ b <= M, b <= 1e4]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None

    val = getattr(b, "value", None)
    if val is None:
        return None
    val = np.asarray(val, dtype=np.float64)
    val = np.where(np.isfinite(val), val, 0.0)
    val = np.maximum(val, 0.0)
    if val.sum() < 1e-14:
        return None
    return val


def lp_nesterov(a0, time_limit, solver, seed=0, verbose=False):
    """
    LP best-response search with Nesterov-style y-iterate and line search.
    """
    a = normalize_to(a0)
    n = a.size
    if n == 0 or time_limit <= 0:
        return a.copy(), float("inf")

    y = a.copy()
    y_prev = a.copy()

    conv = fast_autocorr(a)
    current = float(conv.max())
    best = current
    best_a = a.copy()

    t0 = time.monotonic()
    active = None
    stall = 0
    conv_size = 2 * n - 1

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M = float(conv_y.max())
        if not np.isfinite(M) or M <= 0:
            break

        if active is None or t % 25 == 0:
            active = choose_active_set(conv_y, n, t, stall)

        g = lp_best_response(y, M, active, solver)
        if g is None and t % 4 == 0:
            g = lp_best_response(y, M, np.arange(conv_size), solver)
        if g is None:
            continue

        g = normalize_to(g)

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_to(y_new)

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.92)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to(a_new)

        # Line search from current a toward a_new.
        best_cand = a.copy()
        best_sc = current
        for tau in (1.0, 0.8, 0.6, 0.4, 0.25, 0.1):
            cand = a + tau * (a_new - a)
            cand = np.maximum(cand, 0.0)
            cand = normalize_to(cand)
            sc = score_of(cand)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc = sc
                best_cand = cand

        a = best_cand
        current = best_sc
        y_prev = y
        y = y_new

        if current < best - 1e-12:
            best = current
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 60:
            a = best_a.copy()
            y = best_a.copy()
            y_prev = best_a.copy()
            current = best
            active = None
            stall = 0

        if verbose and t % 120 == 0:
            elapsed = time.monotonic() - t0
            print(
                f"    LP iter={t} ({elapsed:.0f}s): "
                f"best={best:.10f} current={current:.10f}"
            )

    return best_a, best


def score_candidates(base, v, j, M_out, candidates, aj, norm, best_local, best_x, chunk=14):
    """
    Score a list of candidate raw values for coordinate j without mutating arrays.
    The candidate score includes the required global renormalization.
    """
    if candidates is None or candidates.size == 0:
        return best_x, best_local

    start = 0
    L = int(candidates.size)
    while start < L:
        cc = np.asarray(candidates[start:start + chunk], dtype=np.float64)
        if cc.size == 0:
            break
        ds = cc - aj
        valid = (np.abs(ds) > 1e-10)
        if np.any(valid):
            c2 = cc[valid]
            d2 = ds[valid]
            denom = norm + d2
            ok = denom > 1e-8
            if np.any(ok):
                c2 = c2[ok]
                d2 = d2[ok]
                denom = denom[ok]
                scaled = (norm / denom) ** 2

                aff = base[None, :] + (2.0 * d2)[:, None] * v[None, :]
                aff[:, j] += d2 * d2

                if M_out > 0.0:
                    np.maximum(aff, M_out, out=aff)

                mm = aff.max(axis=1)
                sc = mm * scaled
                imin = int(np.argmin(sc))
                if sc[imin] < best_local - 1e-12:
                    best_local = float(sc[imin])
                    best_x = float(c2[imin])
        start += chunk

    return best_x, best_local


def coordinate_descent_exact(a0, time_limit, seed=0, grid=32, refine=28,
                             chunk=14, recompute_every=18, verbose=False):
    """
    Exact incremental coordinate descent.

    For each coordinate, the raw replacement value is chosen by 1D search.
    After each accepted move the whole vector is rescaled to sum sqrt(2n),
    and the convolution is updated accordingly.
    """
    a = np.asarray(a0, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    a = normalize_to(a)
    n = a.size
    if n == 0 or time_limit <= 0:
        return a.copy(), float("inf")

    norm = np.sqrt(2.0 * n)
    nc = 2 * n - 1
    typical = norm / n

    conv = fast_autocorr(a)
    current = float(conv.max())
    best = current
    best_a = a.copy()

    t0 = time.monotonic()
    pass_count = 0
    no_global = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        pass_count += 1

        # Periodically recompute to remove drift.
        if (pass_count % recompute_every == 0) or (no_global > 0 and current > best * 1.001):
            conv = fast_autocorr(a)
            current = float(conv.max())
            if np.isfinite(current) and current < best - 1e-12:
                best = current
                best_a = a.copy()

        if pass_count % 4 == 1 and pass_count > 3:
            press = pressure_for_order(a, conv, max(20, n // 80))
            order = np.argsort(-press)
        else:
            order = rng.permutation(n)

        global_improved = False

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = float(a[j])
            lo = j
            hi = j + n

            if j > 0 and (j + n) < nc:
                M_out = max(float(conv[:j].max()), float(conv[j + n:].max()))
            elif j > 0:
                M_out = float(conv[:j].max())
            elif (j + n) < nc:
                M_out = float(conv[j + n:].max())
            else:
                M_out = 0.0

            base = conv[lo:hi]
            v = a[:n]

            sqrt_cur = np.sqrt(max(current, 1e-6) + 1e-9)
            x_max = max(
                2.5 * aj if aj > 1e-14 else 0.0,
                4.0 * typical,
                2.0 * sqrt_cur,
                0.02 * norm
            )
            x_cap = max(0.1 * norm, 5.0 * sqrt_cur)
            x_max = min(x_max, x_cap, 0.5 * norm)
            if x_max <= 1e-12:
                x_max = typical

            c = np.linspace(0.0, x_max, grid)
            extra = np.array([
                0.0,
                0.25 * aj, 0.5 * aj, 0.75 * aj, 1.25 * aj, 1.5 * aj, 2.0 * aj,
                0.5 * typical, 1.0 * typical, 2.0 * typical, 4.0 * typical, 10.0 * typical,
                0.5 * sqrt_cur, 1.0 * sqrt_cur, 1.5 * sqrt_cur
            ], dtype=np.float64)
            candidates = np.concatenate((c, extra))
            candidates = candidates[
                (candidates >= 0.0) &
                (candidates <= x_max * 1.5) &
                np.isfinite(candidates)
            ]

            best_x, best_local = score_candidates(
                base, v, j, M_out, candidates, aj, norm, current, aj, chunk
            )

            if abs(best_x - aj) > 1e-10:
                if best_x <= 1e-12:
                    rlo = 0.0
                    rhi = max(aj * 0.5, 2.0 * typical)
                else:
                    rlo = max(0.0, best_x * 0.6)
                    rhi = best_x * 1.6
                if rhi <= rlo:
                    rhi = rlo + max(typical, 1e-6)

                fine = np.linspace(rlo, rhi, refine)
                fine = np.concatenate((fine, [0.0, best_x]))
                fine = np.unique(fine)

                best_x, best_local = score_candidates(
                    base, v, j, M_out, fine, aj, norm, best_local, best_x, chunk
                )

            if abs(best_x - aj) > 1e-10 and best_local < current - 1e-12:
                delta = best_x - aj
                new_sum = norm + delta
                if new_sum > 1e-8:
                    scale = norm / new_sum
                    s2 = scale * scale

                    # conv for the un-renormalized vector.
                    conv[lo:hi] += 2.0 * delta * a[:n]
                    conv[2 * j] += delta * delta

                    # Renormalize whole sequence and whole convolution.
                    conv *= s2
                    a *= scale
                    a[j] = scale * best_x

                    cur_actual = float(conv.max())
                    if np.isfinite(cur_actual) and cur_actual < current - 1e-12:
                        current = cur_actual
                        if current < best - 1e-12:
                            best = current
                            best_a = a.copy()
                            global_improved = True
                    else:
                        # Revert exactly.
                        conv /= s2
                        a /= scale
                        a[j] = aj
                        conv[lo:hi] -= 2.0 * delta * a[:n]
                        conv[2 * j] -= delta * delta

                    # Defensive finite check.
                    if (not np.all(np.isfinite(a))) or (not np.all(np.isfinite(conv))):
                        a = best_a.copy()
                        conv = fast_autocorr(a)
                        current = float(conv.max())

        if global_improved:
            no_global = 0
        else:
            no_global += 1
            if no_global > 25:
                scale_noise = 0.005 + 0.003 * min(4.0, no_global / 10.0)
                a = best_a * (1.0 + scale_noise * rng.standard_normal(n))
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-12:
                    a = a * (norm / s)
                else:
                    a = best_a.copy()
                conv = fast_autocorr(a)
                current = float(conv.max())
                no_global = 0

        if verbose and pass_count % 25 == 0:
            elapsed = time.monotonic() - t0
            print(
                f"    CD pass={pass_count} ({elapsed:.0f}s): "
                f"best={best:.10f} current={current:.10f}"
            )

    return best_a, best


def subgradient_smooth(a0, time_limit, seed=0, verbose=False):
    """
    Smoothed subgradient / multiplicative flattening method.
    Used only as a shorter polishing/escape phase.
    """
    a = np.asarray(a0, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    a = normalize_to(a)
    n = a.size
    if n == 0 or time_limit <= 0:
        return a.copy(), float("inf")

    conv = fast_autocorr(a)
    current = float(conv.max())
    best = current
    best_a = a.copy()

    t0 = time.monotonic()
    step = 0.6
    stall = 0
    rng = np.random.default_rng(seed)

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        current = float(conv.max())
        if not np.isfinite(current):
            break
        if current < best - 1e-12:
            best = current
            best_a = a.copy()
            stall = 0

        m = conv.size
        if m == 0:
            break

        n_peaks = max(10, min(m, n // 25))
        idx = np.argsort(conv)[-n_peaks:]
        vals = conv[idx]
        vmax = float(vals.max())
        width = max(vmax - float(vals.min()), 1e-8)
        weights = np.exp(-3.0 * (vmax - vals) / width)
        ws = weights.sum()
        if ws <= 0:
            weights = np.ones(n_peaks) / n_peaks
        else:
            weights = weights / ws

        W = np.zeros(n, dtype=np.float64)
        for k, w in zip(idx, weights):
            k = int(k)
            w = float(w)
            if w <= 0.0:
                continue
            l = max(0, k - n + 1)
            r = min(n, k + 1)
            if l < r:
                W[l:r] += 2.0 * w * a[k - np.arange(l, r)]

        Wp = W - W.mean()
        cand = a * np.exp(-step * Wp)
        cand = np.maximum(cand, 0.0)
        cand = normalize_to(cand)

        sc = score_of(cand)
        if np.isfinite(sc) and sc < current - 1e-12:
            a = cand
            current = sc
            if current < best - 1e-12:
                best = current
                best_a = a.copy()
                stall = 0
            step = min(step * 1.05, 20.0)
        else:
            step *= 0.97
            stall += 1

        if stall > 300:
            break

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(
                f"    SG t={t} ({elapsed:.0f}s): "
                f"best={best:.10f} current={current:.10f}"
            )

    return best_a, best


def resample_sequence(a, target_n):
    """Linearly resample a nonnegative sequence to a new length."""
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    if n == 0 or target_n == 0:
        return a.copy()
    if n == target_n:
        return normalize_to(a)
    if n < 2 or target_n < 2:
        out = np.full(target_n, a.mean(), dtype=np.float64)
        return normalize_to(out)

    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, target_n)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0)
    return normalize_to(out)


def propose_candidate():
    """
    Main search entry point.  Runs within 1800 seconds and returns a list
    of nonnegative floats minimizing the given evaluation function.
    """
    t0 = time.monotonic()
    budget = 1770.0

    best_seq = None
    best_score = float("inf")

    # Load the supplied parent construction if available.
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.maximum(p, 0.0)
            if p.sum() > 1e-12:
                best_seq = normalize_to(p)
                best_score = score_of(best_seq)
                # Use the real evaluator for the authoritative score.
                try:
                    rs = evaluate_sequence([float(x) for x in best_seq])
                    if np.isfinite(rs):
                        best_score = float(rs)
                except Exception:
                    pass
                print(f"Parent loaded: n={best_seq.size}, score={best_score:.8f}")
    except Exception:
        best_seq = None

    # Fallback initial shape.
    if best_seq is None:
        n = 1600
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.0)
        best_seq = normalize_to(a)
        best_score = score_of(best_seq)
        print(f"Default init: n={best_seq.size}, score={best_score:.8f}")

    solver = get_lp_solver()

    # ---- Phase 1: main LP + coordinate descent ----
    rem = budget - (time.monotonic() - t0)
    t_lp = min(rem * 0.30, 430.0)
    if rem > 320 and solver is not None and t_lp > 30:
        a_in = normalize_to(np.maximum(best_seq, 0.0))
        a_out, _ = lp_nesterov(a_in, t_lp, solver, seed=0, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t0)

    rem = budget - (time.monotonic() - t0)
    t_cd = min(rem * 0.28, 340.0)
    if rem > 220 and t_cd > 30:
        a_out, _ = coordinate_descent_exact(best_seq, t_cd, seed=1, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "cd1", t0)

    # ---- Phase 2: second LP + coordinate descent ----
    rem = budget - (time.monotonic() - t0)
    t_lp = min(rem * 0.18, 260.0)
    if rem > 260 and solver is not None and t_lp > 30:
        a_in = normalize_to(np.maximum(best_seq, 0.0))
        a_out, _ = lp_nesterov(a_in, t_lp, solver, seed=123, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp2", t0)

    rem = budget - (time.monotonic() - t0)
    t_cd = min(rem * 0.24, 320.0)
    if rem > 220 and t_cd > 30:
        a_out, _ = coordinate_descent_exact(best_seq, t_cd, seed=7, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "cd2", t0)

    # ---- Phase 3: optionally resample to larger length ----
    rem = budget - (time.monotonic() - t0)
    n_cur = len(best_seq)
    if rem > 560:
        target_n = 2048 if n_cur <= 1600 else (2560 if n_cur < 2560 else n_cur)
        if target_n != n_cur:
            work = resample_sequence(best_seq, target_n)
            # If it already beats the incumbent, update; otherwise keep optimizing it.
            best_seq, best_score = try_update(best_seq, best_score, work, "resample", t0)

            rem = budget - (time.monotonic() - t0)
            t_lp = min(rem * 0.18, 160.0)
            if rem > 260 and solver is not None and t_lp > 20:
                a_out, _ = lp_nesterov(work, t_lp, solver, seed=555, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_big", t0)

            # Use the best candidate of this length for CD.
            if best_seq is not None and len(best_seq) == target_n:
                work = np.asarray(best_seq, dtype=np.float64).copy()
            else:
                work = np.asarray(work, dtype=np.float64).copy()

            rem = budget - (time.monotonic() - t0)
            t_cd = min(rem * 0.32, 330.0)
            if rem > 180 and t_cd > 30:
                a_out, _ = coordinate_descent_exact(work, t_cd, seed=101, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "cd_big", t0)

    # ---- Final polishing phases ----
    rem = budget - (time.monotonic() - t0)
    t_sg = min(rem * 0.25, 120.0)
    if rem > 160 and t_sg > 30:
        a_out, _ = subgradient_smooth(best_seq, t_sg, seed=42, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sg_final", t0)

    rem = budget - (time.monotonic() - t0)
    t_cd = min(rem * 0.55, 260.0)
    if rem > 120 and t_cd > 30:
        a_out, _ = coordinate_descent_exact(best_seq, t_cd, seed=313, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "cd_final1", t0)

    rem = budget - (time.monotonic() - t0)
    t_cd = min(rem * 0.70, 160.0)
    if rem > 80 and t_cd > 20:
        a_out, _ = coordinate_descent_exact(best_seq, t_cd, seed=1234, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "cd_final2", t0)

    # Final authoritative evaluation.
    if best_seq is None:
        return [1.0]

    final = np.asarray(best_seq, dtype=np.float64)
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.maximum(final, 0.0)
    s = final.sum()
    if s < 1e-8:
        return [1.0]
    final = normalize_to(final)

    try:
        fs = evaluate_sequence([float(x) for x in final])
        if np.isfinite(fs):
            best_score = float(fs)
    except Exception:
        pass

    print(
        f"FINAL: score={best_score:.8f}, n={final.size}, "
        f"time={time.monotonic() - t0:.0f}s"
    )
    return [float(x) for x in final]
```
