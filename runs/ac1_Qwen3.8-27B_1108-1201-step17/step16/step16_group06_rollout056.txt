def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

    s = a.sum()
    if s < 1e-15:

def build_sparse_conv_matrix(n):
    """Build sparse matrix T (2n-1 x n) where (T @ b)[k] = sum_j T[k,j]*b[j] = (y*b)[k]
    when T[k,j] = y[j]. We return the sparsity structure (row, col) indices.
    """
    if n in _sparse_cache:
        return _sparse_cache[n]
    
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    
    row_arr = np.array(rows, dtype=np.int32)
    col_arr = np.array(cols, dtype=np.int32)
    _sparse_cache[n] = (row_arr, col_arr)
    return (row_arr, col_arr)


def convolve_sparse(n, y, b):
    """Compute (y * b) using the sparse structure."""

    data = y[col_arr]
    T = sp.csr_matrix((data, (row_arr, col_arr)), shape=(2*n-1, n))
    return T @ b

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov-accelerated update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y_prev = y.copy()
        y = y_new
        x = x_new
        iter_count += 1
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}, M_curr={M_curr:.8f}")
    
    return best_a, best_m


def nesterov_subgradient(a0, time_limit, beta=3.0, verbose=True, seed=42, max_iters=100000000):
    """Nesterov-accelerated subgradient descent on the simplex.
    Uses weighted combination of top peaks for a smoother gradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = 2.0
    stall_count = 0
    max_stall = 5000
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        
        # Weighted gradient from top peaks
        sorted_idx = np.argsort(ac)[::-1]
        top_k = min(30, len(ac))
        top_vals = np.maximum(ac[sorted_idx[:top_k]], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(top_k) / top_k
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, sorted_idx[:top_k]):
            k_int = int(k)
            # d/d a[i] of conv[k] = 2 * a[k-i] if 0 <= k-i < n
            idx = k_int - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            if valid.any():
                grad[valid] += 2.0 * wi * a[idx[valid]]
        
        eta = gamma / (np.sqrt(t) + 1.0)
        
        # Mirror descent step (multiplicative updates on the simplex)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > max_stall:
            stall_count = 0
            # Restart with perturbation
            perturbation = rng.normal(0, 0.01, n)
            x_restarted = best_a * (1.0 + perturbation)
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NGD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def coordinate_descent_polish(a0, time_limit, verbose=True, seed=123):
    """Peak-targeted coordinate descent: move mass from peak-contributing
    positions to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_idx = np.argmax(ac)
        k_int = int(peak_idx)
        
        # Sensitivity: how much each position contributes to top peaks
        top_peaks = np.argsort(ac)[::-1][:20]
        sensitivity = np.zeros(n)
        for pk in top_peaks:
            pk_int = int(pk)
            idx = pk_int - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            sensitivity[valid] += a[idx[valid]]
        
        # Find source (high sensitivity) and destination (low sensitivity)
        src_candidates = np.argsort(sensitivity)[::-1][:20]
        dst_candidates = np.argsort(sensitivity)[:30]
        
        improved = False
        for src in src_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue
            for dst in dst_candidates:
                if dst == src:
                    continue
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall < max_stall:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordPD ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    row_arr, col_arr = get_sparse_structure(n)
    data = y[col_arr]
    return sp.csr_matrix((data, (row_arr, col_arr)), shape=(2 * n - 1, n))


def lp_frank_wolfe(a0, time_limit, solver_name, verbose=True, max_iters=100000, relax=1.0):
    """LP-based Frank-Wolfe with Nesterov acceleration and sparse matrix."""
    import cvxpy as cp
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    x_prev = a.copy()
    iter_count = 0
    consecutive_fail = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max() * relax
        if M_curr < 1e-14:
            break

        # Build constraint matrix
        T = build_conv_matrix_sparse(n, y)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            consecutive_fail += 1
            if consecutive_fail > 10:
                break
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            consecutive_fail += 1
            if consecutive_fail > 10:
                break
            continue

        consecutive_fail = 0
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov-accelerated FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        # Also try intermediate points between y and g_dir
        if iter_count % 5 == 0:
            for frac in [0.3, 0.5, 0.7]:
                trial = (1 - frac) * norm_sum1(a0) + frac * g_dir
                trial = np.maximum(trial, 0)
                st = trial.sum()
                if st > 1e-15:
                    trial = trial / st
                    mt = fast_autocorr(trial).max()
                    if mt < best_m:
                        best_m = mt
                        best_a = trial.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}")

    return best_a, best_m


def nesterov_subgradient(a0, time_limit, beta=3.0, verbose=True, seed=42):
    """Nesterov-accelerated mirror descent with peak-weighted subgradients."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = 3.0
    stall_count = 0
    max_stall = 8000
    top_k = 50

    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        sorted_idx = np.argsort(ac)[::-1]
        k_use = min(top_k, len(ac))
        top_vals = np.maximum(ac[sorted_idx[:k_use]], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_use) / k_use
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, sorted_idx[:k_use]):
            k_int = int(k)
            idx = k_int - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            if valid.any():
                grad[valid] += 2.0 * wi * a[idx[valid]]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > max_stall:
            stall_count = 0
            perturbation = rng.normal(0, 0.015, n)
            x_restarted = best_a * (1.0 + perturbation)
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NGD t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=True, seed=123):
    """Peak-targeted coordinate descent for fine-tuning."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        peak_idx = np.argmax(ac)
        k_int = int(peak_idx)

        # Compute sensitivity to top peaks
        top_peaks = np.argsort(ac)[::-1][:25]
        sensitivity = np.zeros(n)
        for pk in top_peaks:
            pk_int = int(pk)
            idx = pk_int - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            sensitivity[valid] += a[idx[valid]]

        src_candidates = np.argsort(sensitivity)[::-1][:30]
        dst_candidates = np.argsort(sensitivity)[:40]

        improved = False
        for src in src_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue
            for dst in dst_candidates:
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall < max_stall and stall % 3 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordPD ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator combining LP-FW, subgradient, and coordinate descent.
    
    Strategy: Spend most time on LP-FW (the most effective method),
    interspersed with fast subgradient steps and coordinate descent polish.
    Explore a few different sequence lengths.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: LP-FW main (40% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_cur = len(best_seq)
        print(f"\n=== Phase 1: LP-FW n={n_cur} (main workhorse) ===")
        t1 = min(remaining * 0.40, 600.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t1, solver, verbose=True)
        try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 2: Subgradient (10% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: Nesterov subgradient n={n_cur} ===")
        t2 = min(remaining * 0.10, 150.0)
        a_out, _ = nesterov_subgradient(norm_sum1(best_seq), t2, beta=3.0, verbose=True, seed=42)
        try_update(a_out, f"sgd_n{n_cur}")

    # === Phase 3: LP-FW round 2 (20% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: LP-FW round 2 n={n_cur} ===")
        t3 = min(remaining * 0.25, 350.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t3, solver, verbose=True)
        try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 4: Coordinate descent (8% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: Coord descent n={n_cur} ===")
        t4 = min(remaining * 0.08, 100.0)
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t4, verbose=True, seed=42)
        try_update(a_out, f"cd_n{n_cur}")

    # === Phase 5: Try alternative lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    test_ns = []
    for delta in [300, 500, -300, 700]:
        nt = n_best + delta
        if 1000 <= nt <= 4500 and nt != n_best:
            test_ns.append(nt)

    for nt in test_ns[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n=== Phase 5: Trying n={nt} ===")
        a_inter = interpolate_sequence(best_seq, nt)
        # Subgradient warmup at new length
        t_a = min(remaining * 0.12, 60.0)
        a_out, _ = nesterov_subgradient(norm_sum1(a_inter), t_a, beta=3.0, verbose=False, seed=nt % 1000)
        try_update(a_out, f"sgd_n{nt}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t_b = min(remaining * 0.25, 200.0)
            a_out, _ = lp_frank_wolfe(norm_sum1(a_inter), t_b, solver, verbose=False)
            try_update(a_out, f"lp_n{nt}")

    # === Phase 6: Final LP-FW on best sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_best = len(best_seq)
        print(f"\n=== Phase 6: Final LP n={n_best} ===")
        t6 = min(remaining * 0.50, 250.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t6, solver, verbose=True)
        try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Final coord polish n={n_best} ===")
        t7 = min(remaining * 0.70, 120.0)
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t7, verbose=True, seed=777)
        try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
