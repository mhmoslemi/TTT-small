def smooth_max_conv(a, tau):
    """Smooth approximation of max(conv(a,a)) using log-sum-exp."""
    c = fast_autocorr(a)
    return (1.0 / tau) * logsumexp(tau * c)

def smooth_max_conv_grad(a, tau):
    """Compute smooth max convolution value and gradient."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)[:size]
    
    # Smooth max
    max_C = C.max()
    shifted = tau * (C - max_C)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    f = max_C + np.log(sum_exp) / tau
    
    # Gradient: f = (1/tau) * log(sum_k exp(tau*C_k))
    # df/da_i = (1/tau) * sum_k exp(tau*C_k)/sum_exp * dC_k/da_i
    # dC_k/da_i = 2*a[k-i] if 0 <= k-i < n, else a[k-i] if k-i==i (i.e., k=2i) ... 
    # Actually: C_k = sum_m a[m]*a[k-m]
    # dC_k/da_i: terms where m=i: a[i]*a[k-i] (if valid), terms where k-m=i i.e. m=k-i: a[k-i]*a[i] (if valid)
    # So dC_k/da_i = a[k-i] + a[k-i] = 2*a[k-i] if 0<=k-i<n
    # (since i is always in [0,n) and we need k-i in [0,n))
    
    # Therefore: df/da_i = (1/tau) * sum_k weights[k] * 2*a[k-i] where weights[k] = exp(tau*C_k)/sum_exp
    #            = (2/tau) * sum_{j: 0<=j<n, 0<=i+j<size} weights[i+j] * a[j]
    # This is a cross-correlation!
    
    weights = exp_shifted / sum_exp
    W = np.fft.rfft(weights, fft_size)
    grad = (2.0 / tau) * np.fft.irfft(W * A, fft_size)[:n]
    
    return f, grad

def objective_and_grad(a, tau):
    """Objective and gradient for L-BFGS-B (minimizing n * smooth_max_conv)."""
    n = len(a)
    f, g = smooth_max_conv_grad(a, tau)
    # We want to minimize n * f (since score = 2 * n * max_conv / sum^2, and we normalize sum=1)
    return n * f, n * g

def active_set_optimize(a0, time_limit, n_active_max=50):
    """Iteratively identify active lags and optimize against them."""
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for outer in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_autocorr(a)
        M = c.max()
        
        # Identify active lags (within epsilon of max)
        active = np.where(c >= M * 0.999)[0]
        if len(active) > n_active_max:
            active = np.argsort(c)[::-1][:n_active_max]
        
        # Build constraint matrix: for each active lag k, the constraint is
        # sum_i a[i] * a[k-i] <= M (where k-i is in valid range)
        # This is quadratic, so we linearize: at current a, the constraint is
        # sum_i (a[k-i] * da[i] + a[i] * da[k-i]) + C_k <= M
        # i.e., sum_i 2*a[k-i]*da[i] <= M - C_k (for the active constraint k)
        # But this is a linearization, not exact.
        
        # Alternative: use the subgradient of max at the active set
        # subgrad_i = sum_{k in active} weight_k * 2*a[k-i] / len(active)
        
        weights = np.ones(len(active)) / len(active)
        subgrad = np.zeros(n)
        for w, k in zip(weights, active):
            k_int = int(k)
            for i in range(n):
                j = k_int - i
                if 0 <= j < n:
                    subgrad[i] += w * 2.0 * a[j]
        
        # Take a step in the negative subgradient direction
        # Project onto {x >= 0, sum(x) = 1}
        lr = 0.01 / (1 + outer * 0.1)
        x_new = a - lr * subgrad
        x_new = np.maximum(x_new, 0)
        sx = x_new.sum()
        if sx > 1e-15:
            x_new = x_new / sx
        
        m = fast_autocorr(x_new).max()
        if m < best_m:
            best_m = m
            best_a = x_new.copy()
            a = x_new
        else:
            # Line search: try smaller steps
            for factor in [0.5, 0.25, 0.1]:
                x_try = a - lr * factor * subgrad
                x_try = np.maximum(x_try, 0)
                sx = x_try.sum()
                if sx > 1e-15:
                    x_try = x_try / sx
                m_try = fast_autocorr(x_try).max()
                if m_try < best_m:
                    best_m = m_try
                    best_a = x_try.copy()
                    a = x_try
                    break
    
    return best_a, best_m

def diverse_initialization(n, rng):
    """Generate diverse initial sequences."""
    x = np.linspace(0, 1, n)
    init_type = rng.integers(0, 5)
    
    if init_type == 0:
        # Uniform
        a = np.ones(n)
    elif init_type == 1:
        # Triangle
        a = np.maximum(1 - 2 * np.abs(x - 0.5), 0)
    elif init_type == 2:
        # Arcsine-like (high at edges)
        a = 1 / np.sqrt(np.maximum(x * (1 - x), 1e-10))
    elif init_type == 3:
        # Bimodal

        # Random
        a = np.abs(rng.normal(0, 1, n))
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

def coordinate_polish(a0, time_limit, rng_seed=42):
    """Local coordinate descent polish."""
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 20
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_sweeps = min(100, n // 3)
        
        for _ in range(n_sweeps):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                for direction in [1, -1]:
                    src, dst = (i, j) if direction == 1 else (j, i)
                    if a[src] < 1e-12:
                        continue
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    m = fast_autocorr(a_trial).max()
                    if m < best_m - 1e-14:
                        a = a_trial
                        best_m = m
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def fast_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_score(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()

def smooth_obj_grad(x, tau):
    """
    Objective: n * smooth_max(conv(normalized_x, normalized_x))
    where smooth_max(y) = (1/tau) * log(sum(exp(tau * y)))
    """
    n = len(x)
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-15:
        return 1e10, np.zeros(n)
    a = a / s  # Normalize
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    C = np.fft.irfft(A * A, fft_size)[:size]
    
    # Smooth max
    max_C = C.max()

    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    f = n * (max_C + np.log(sum_exp) / tau)

    # df/da_i = (2/tau) * sum_j weights[i+j] * a[j]  (cross-correlation)

def lbgfs_anneal(a0, time_budget, tau_schedule=None):
    """L-BFGS-B with annealing schedule."""
    if tau_schedule is None:
        tau_schedule = [300, 200, 150, 100, 70, 50, 35, 25, 20, 15, 12, 10, 8, 7, 6, 5, 4, 3, 2.5, 2, 1.5, 1.2, 1.0]
    
    n = len(a0)
    a = a0.copy()
    t0 = time.monotonic()
    bounds = [(0.0, None)] * n
    
    for tau in tau_schedule:
        if time.monotonic() - t0 > time_budget:
            break
        
        # Allocate iterations based on remaining schedule
        n_remaining = len(tau_schedule) - tau_schedule.index(tau)
        time_per = time_budget / n_remaining
        max_iter = max(100, min(5000, int(time_per * 300)))
        
        try:
            result = minimize(
                smooth_obj_grad,
                a,
                args=(tau,),
                jac=True,
                method='L-BFGS-B',
                bounds=bounds,
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-15,
                    'gtol': 1e-10,
                    'maxls': 50
                }
            )
            a_new = np.maximum(result.x, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
            a = a_new
            sc = eval_score(a)
            print(f"    tau={tau}: score={sc:.8f}, iters={result.nit}, time={time.monotonic()-t0:.1f}s")
        except Exception as e:
            print(f"    tau={tau}: error: {e}")
    
    return a
