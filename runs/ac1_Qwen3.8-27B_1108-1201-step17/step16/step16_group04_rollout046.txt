def fast_conv(a):
    """Compute linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_objective(a):
    """Compute the objective value for a normalized (sum=1) sequence."""
    n = len(a)
    c = fast_conv(a)
    max_c = c.max()
    return 2.0 * n * max_c  # since sum(a) = 1


def init_arc_sine(n, temperature=1.0):
    """Arc-sine-like initialization (edge-concentrated)."""
    j = np.arange(n, dtype=np.float64)
    x = (j + 0.5) / n  # in (0, 1)
    # Arc-sine density: 1/sqrt(x*(1-x))
    # Clip to avoid infinities at edges
    x_clipped = np.clip(x, 0.01, 0.99)
    a = 1.0 / np.sqrt(x_clipped * (1.0 - x_clipped))
    a = a ** temperature
    return normalize(a)


def init_bimodal(n, alpha=3.0):
    """Bimodal initialization with peaks at n/4 and 3n/4."""
    j = np.arange(n, dtype=np.float64)
    x = j / (n - 1) if n > 1 else np.array([0.5])
    a = np.exp(-alpha * ((x - 0.25)**2 + (x - 0.75)**2))
    # Add small edge mass
    a[0] += 0.1
    a[-1] += 0.1
    return normalize(a)


def init_uniform(n):
    return np.ones(n) / n


def init_random_sparse(n, sparsity=0.3, rng=None):
    """Random sparse initialization."""
    if rng is None:
        rng = np.random.default_rng(42)
    a = np.zeros(n)
    k = max(2, int(n * sparsity))
    idx = rng.choice(n, size=k, replace=False)
    a[idx] = 1.0
    # Blur slightly
    kernel = np.ones(3) / 3
    a = np.convolve(a, kernel, mode='same')
    return normalize(a)


def init_fourier(n, n_harmonics=20, rng=None):
    """Fourier-based initialization with random phases."""
    if rng is None:
        rng = np.random.default_rng(123)
    j = np.arange(n, dtype=np.float64)
    a = np.ones(n) * 0.5
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(0.1, 1.0) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * j / n + phase)
    return normalize(a)

        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        c = fast_conv(y)
        M = c.max()
        obj = 2.0 * n * M
        if obj < best_obj:
            best_obj = obj
            best_a = y.copy()
        
        # Active constraints
        active_lags = np.argsort(c)[::-1][:current_top_k]
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(b_var) == 1]  # normalize direction
        
        for k in active_lags:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            j_idx = np.arange(js, je + 1)
            coeff = y[k_int - j_idx]
            constraints.append(coeff @ b_var[j_idx] <= M)
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        b_val = b_val / b_sum
        
        # Step size
        alpha = 2.0 / (t + 2.0)
        x_new = (1.0 - alpha) * y + alpha * b_val
        x_new = np.maximum(x_new, 0.0)
        x_new = x_new / x_new.sum()
        
        x_prev = y
        x = x_new
        
        mx = fast_conv(x).max()
        obj_x = 2.0 * n * mx
        if obj_x < best_obj:
            best_obj = obj_x
            best_a = x.copy()
        
        if t % 100 == 0 and current_top_k < top_k_max:
            current_top_k = min(current_top_k + 5, top_k_max)
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW t={t} ({elapsed:.1f}s): obj={best_obj:.6f}, top_k={current_top_k}")
    
    return best_a, best_obj
