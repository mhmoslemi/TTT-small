def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

        a_list = [float(x) for x in a]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def subgradient_direction(a, C, top_k=5):
    """Compute subgradient of max_k C[k] where C = conv(a,a).
    Returns gradient vector for the sum of top_k active constraints."""
    n = len(a)
    top_idx = np.argsort(C)[::-1][:top_k]
    top_vals = C[top_idx]
    # Weight by proximity to max
    max_val = C.max()
    # Use exponential weighting
    w = np.exp(-10 * (max_val - top_vals) / max(1e-15, max_val))
    w = w / w.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # grad[i] += 2 * wi * a[k-i] for valid i
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def project_simplex(v, s=1.0):
    """Project v onto {x: x >= 0, sum(x) = s}."""
    n = len(v)
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    cols = []
    vals = []

            cols.append(j)

    # Total non-zeros: sum_{k=0}^{2n-2} (min(n-1,k) - max(0,k-n+1) + 1)
    # = n^2 (approximately, since the "bandwidth" grows to n and then shrinks)

    # ...

    # Total: 1+2+...+(n-1)+n+(n-1)+...+2+1 = n^2

    a_new = a + best_alpha * direction
    a_new = np.maximum(a_new, 0)
