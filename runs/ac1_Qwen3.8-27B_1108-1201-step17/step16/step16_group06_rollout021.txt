def fast_autocorr(a):
    """Compute full autocorrelation (a convolved with a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum 1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgradient(a, ac, top_k=50, beta=3.0):
    """
    Compute weighted subgradient of max_k (a*a)[k].
    Uses softmax weighting over top-k peaks.
    """
    n = len(a)
    # Weight peaks by softmax of their values
    max_ac = ac.max()
    # Inverse: weight higher peaks more
    shifted = ac - max_ac  # <= 0
    w = np.exp(-beta * (-shifted))  # = exp(beta * (max_ac - ac))
    # Actually, we want to weight the PEAKS more, so:
    # w_k proportional to exp(beta * ac_k) for the top peaks
    # But this overflows, so use: w_k proportional to exp(beta * (ac_k - max_ac))
    # which puts the most weight on the max
    
    # Simpler: use top-k with linear decay
    k = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k]
    top_vals = ac[top_idx]
    # Weights: linearly decreasing from 1 to 1/k
    weights = np.linspace(1.0, 0.1, k)
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for wi, pk in zip(weights, top_idx):
        pk_int = int(pk)
        # grad[m] += 2 * wi * a[pk-m] for valid m
        m_start = max(0, pk_int - n + 1)
        m_end = min(n, pk_int + 1)
        for m in range(m_start, m_end):
            j = pk_int - m
            if 0 <= j < n:
                grad[m] += 2.0 * wi * a[j]
    return grad

def compute_subgradient_fast(a, ac, top_k=50):
    """
    Faster subgradient computation using vectorized operations.
    grad[m] = 2 * sum_k w_k * a[k-m] for valid k-m
    = 2 * (w * a_reversed)[m] (with appropriate indexing)
    """
    n = len(a)
    k = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k]
    top_vals = ac[top_idx]
    # Weights: softmax with temperature
    max_v = top_vals.max()
    w = np.exp(top_vals - max_v)
    w = w / w.sum()
    
    # For each peak position pk, the gradient contribution is:
    # grad[m] += 2 * w_i * a[pk - m] for m in [max(0, pk-n+1), min(n-1, pk)]
    # This is equivalent to: grad += 2 * (w_convolve_with_shifted_a)
    # 
    # More efficiently: for each peak pk, the vector [a[pk-m] for valid m] is a 
    # "window" of a (possibly reversed). We can accumulate these.
    
    grad = np.zeros(n)
    for i in range(k):
        pk = int(top_idx[i])
        wi = w[i] * 2.0
        # Contribution: wi * a[pk-m] for m in [max(0,pk-n+1), min(n-1,pk)]
        m_lo = max(0, pk - n + 1)
        m_hi = min(n - 1, pk)
        # j = pk - m, so j ranges from pk - m_hi to pk - m_lo
        j_lo = pk - m_hi
        j_hi = pk - m_lo
        # The values a[j_lo], a[j_lo+1], ..., a[j_hi] go into grad[m_hi], grad[m_hi-1], ..., grad[m_lo]
        # i.e., grad[m_lo:m_hi+1] += wi * a[j_lo:j_hi+1][::-1]
        if m_lo <= m_hi:
            grad[m_lo:m_hi+1] += wi * a[j_lo:j_hi+1][::-1]
    
    return grad

def nesterov_mirror_descent(a0, time_limit, verbose=True, top_k_init=30, top_k_max=200, beta_init=2.0):
    """
    Nesterov-accelerated mirror descent on the simplex.
    Minimizes max_k (a*a)[k] subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()       # Nesterov's x (the "current" point)
    y = a.copy()       # Nesterov's y (the "lookahead" point)
    x_prev = a.copy()
    
    gamma = 1.0  # step size parameter
    stall = 0
    max_stall = 2000
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation at y
        ac = fast_autocorr(y)
        
        # Compute subgradient at y (using y's autocorrelation for peak selection,
        # but the gradient formula uses y)
        top_k = min(top_k_init + t // 100, top_k_max)
        grad = compute_subgradient_fast(y, ac, top_k)
        
        # Mirror descent step: log(a_new) = log(y) - eta * grad
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        log_a -= log_a.max()  # for numerical stability
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Track best
        M = fast_autocorr(x).max()
        if M < best_m - 1e-14:
            best_m = M
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.3)
            # Perturb from best
            rng = np.random.default_rng(t)
            pert = rng.normal(0, 0.05, n) * best_a
            x_restarted = best_a + pert
            x_restarted = normalize(x_restarted)
            x = x_restarted
            x_prev = x_restarted
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m
