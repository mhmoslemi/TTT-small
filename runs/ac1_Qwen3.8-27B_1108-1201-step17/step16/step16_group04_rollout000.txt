import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_crossconv(a, b):
    """Compute linear cross-convolution a * b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def evaluate_safe(a):
    try:

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def spectral_equalize(a, n_iter=30, eps=1e-10):
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = np.abs(A)**2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.7 * a + 0.3 * a_new
    return a

def solve_lp_direction(y, M, active_lags, solver):
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)
    
    if not constraints:
        return np.ones(n) / n
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def lp_fw_cutting_plane(a0, time_limit, solver, top_k_init=80, top_k_max=250, verbose=True):
    """LP-FW with cutting plane constraint generation and line search."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    current_top_k = top_k_init
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        if t == 1:
            y = x.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()
        
        # Select active constraints
        active_lags = list(np.argsort(ac)[::-1][:current_top_k])
        
        # Solve LP with cutting planes
        b = solve_lp_direction(y, M_curr, active_lags, solver)
        if b is None:
            continue
        
        # Check for violated constraints (cutting plane)
        yb = fast_crossconv(y[:n], b[:n])  # conv(y, b)
        # yb has length 2n-1
        violated = np.where(yb > M_curr + 1e-10)[0]
        if len(violated) > 0:
            # Add most violated constraints
            worst = violated[np.argsort(yb[violated])[::-1][:min(10, len(violated))]]
            active_lags.extend(worst.tolist())
            b = solve_lp_direction(y, M_curr, active_lags, solver)
            if b is None:
                continue
        
        # Line search
        best_alpha = 2.0 / (t + 2.0)  # Default FW step
        best_obj = fast_autocorr((1-best_alpha)*y + best_alpha*b).max()
        
        # Try a few alternative step sizes
        for alpha in [best_alpha * 0.5, best_alpha * 1.5, best_alpha * 2.0, 0.05, 0.1, 0.2]:
            if alpha <= 0 or alpha > 2.0:
                continue
            cand = (1.0 - alpha) * y + alpha * b
            cand = np.maximum(cand, 0.0)
            sc = cand.sum()
            if sc < 1e-15:
                continue
            cand = cand / sc
            obj = fast_autocorr(cand).max()
            if obj < best_obj:
                best_obj = obj
                best_alpha = alpha
        
        x_new = (1.0 - best_alpha) * y + best_alpha * b
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx
        
        x_prev = y
        x = x_new
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if t % 50 == 0 and current_top_k < top_k_max:
            current_top_k = min(current_top_k + 5, top_k_max)
        
        if verbose and t % 20 == 0:

def logsumexp_gradient(a, beta=200.0):

    n = len(a)
    ac = fast_autocorr(a)
    # Subtract max for numerical stability
    ac_shifted = ac - ac.max()
    weights = np.exp(beta * ac_shifted)
    weights = weights / weights.sum()
    
    # Gradient of sum_k w_k * (a*a)[k] with respect to a
    # (a*a)[k] = sum_j a[j] * a[k-j]
    # d/d a[i] (a*a)[k] = a[k-i] + a[i-k] (when both are in range)
    grad = np.zeros(n)
    for k in range(len(ac)):
        wk = weights[k]
        if wk < 1e-15:
            continue
        # d/d a[i] sum_j a[j]*a[k-j] for i in [0, n-1]
        # = a[k-i] (if 0 <= k-i < n) + a[i-k] (if 0 <= i-k < n)
        # Note: if k-i == i-k (i.e., 2i = k), it's 2*a[i]... 
        # Actually d/d a[i] (sum_j a[j] a[k-j]) = a[k-i] + a[i-k] 
        # where we interpret a[j] = 0 for j out of [0,n-1]
        # But wait, when j = i in the first term, we get a[k-i] (the other factor)
        # and when k-j = i, i.e., j = k-i, we get a[j] = a[k-i]

def adam_smoothed_gradient(a0, time_limit, beta=100.0, verbose=True):
    """Adam optimizer on the log-sum-exp smoothed objective."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps_adam = 0.8, 0.999, 1e-8
    lr = 0.01
    t_adam = 0
    stall_count = 0
    
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        t_adam += 1
        
        grad, M_curr = logsumexp_gradient(a, beta=beta)
        
        # Normalize gradient by n to keep scale reasonable
        grad = grad / n
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t_adam)
        v_hat = v / (1 - beta2**t_adam)
        
        # Mirror descent style: multiplicative update
        # a_new[i] = a[i] * exp(-lr * m_hat[i] / (sqrt(v_hat[i]) + eps))
        update = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a * np.exp(-update)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        a_new = a_new / s
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        a = a_new
        
        # Reduce beta as we progress (sharpen the max)
        if it % 500 == 0:
            beta = max(20.0, beta * 0.95)
        
        # Learning rate decay
        lr = 0.01 / (1.0 + 0.001 * t_adam)
        
        # Restart if stalled
        if stall_count > 2000:
            stall_count = 0
            pert = rng.normal(0, 0.003, n)
            a = best_a * (1.0 + pert)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Adam t={it} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}, lr={lr:.6f}")
    
    return best_a, best_m

def coordinate_polish_v2(a0, time_limit, verbose=True, rng_seed=42):
    """Improved coordinate descent targeting the peak."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))
        
        # Find the coordinate that most contributes to the peak
        # (a*a)[peak_k] = sum_j a[j] * a[peak_k - j]
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        if contrib.max() < 1e-20:
            break
        
        # Try moving mass from the highest contributor to a better location
        src = idx[np.argmax(contrib)]
        
        # Score each possible destination
        # Moving mass from src to dst changes:
        # - Removes contribution from all lags involving src
        # - Adds contribution from all lags involving dst
        # We want to minimize the new peak
        
        # Only consider destinations far from the peak
        in_peak_region = np.zeros(n, dtype=bool)
        for dk in range(-5, 6):
            k2 = peak_k + dk
            if 0 <= k2 < len(ac):
                s2 = max(0, k2 - n + 1)
                e2 = min(n - 1, k2)
                if s2 <= e2:
                    in_peak_region[s2:e2+1] = True
        
        candidates = np.where(~in_peak_region)[0]
        if len(candidates) < 100:
            candidates = np.arange(n)
        
        # Sample a subset
        perm = rng.permutation(len(candidates))
        chosen = candidates[perm][:min(200, len(candidates))]
        
        best_dst = -1
        best_M = float('inf')
        
        for dst in chosen:
            if dst == src:
                continue
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_M:
                    best_M = M_trial
                    best_dst = dst
                    best_frac = frac
                    best_a_trial = a_trial.copy()
        
        if best_dst >= 0 and best_M < M_curr - 1e-15:
            a = best_a_trial
            best_m = best_M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Random perturbation
            if a.sum() > 1e-15:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord v2 ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m
