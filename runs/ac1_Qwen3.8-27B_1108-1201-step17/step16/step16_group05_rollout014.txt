    if s < 1e-15:
        return np.full(len(a), target / len(a))
    return target * a / s

def _project_simplex(v, target=1.0):
    n = len(v)
    v = v.astype(np.float64).copy()
    # Project to sum=target, v>=0
    v = v / target  # work with sum=1
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (css - 1))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (css[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0) * target

def _grad_smoothed(a, beta, top_k=300):
    """Gradient of smoothed max of autocorrelation."""
    n = len(a)
    ac = _fast_autocorr(a)
    top_idx = np.argsort(ac)[::-1][:top_k]
    top_vals = ac[top_idx]
    sh = beta * (top_vals - top_vals[0])
    w = np.exp(sh)
    w /= w.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n, ki + 1)
        if lo < hi:
            j = np.arange(lo, hi)
            grad[j] += 2.0 * wi * a[ki - j]
    return grad, ac.max()

def _apg(a0, t_limit, beta_lo=5.0, beta_hi=30.0, top_k=300, c=2.0, seed=0):
    """Accelerated projected gradient on the simplex."""
    n = len(a0)
    a = _normalize(a0)
    best = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 2000
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > t_limit:
            break
        frac = min(1.0, t / 10000.0)
        cur_beta = beta_lo + (beta_hi - beta_lo) * (1 - frac)
        
        # Nesterov
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = _project_simplex(y)
        
        g, _ = _grad_smoothed(y, cur_beta, top_k)
        eta = c / (np.sqrt(t) + 1.0)
        x_new = _project_simplex(y - eta * g)
        
        x_prev = x
        x = x_new
        
        M = _fast_autocorr(x).max()
        if M < best_m - 1e-16:
            best_m = M
            best = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.5)
            noise = rng.normal(0, 0.003, n) * best
            x = _normalize(best + noise)
            x_prev = x.copy()
    
    return best, best_m

def _lp_fw(a0, t_limit, solver, verbose=True, max_iters=5000):
    """LP Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = _normalize(a0)
    best = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x_mom = a.copy()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > t_limit:
            break
        
        ac = _fast_autocorr(y)
        M = ac.max()
        
        # Build T matrix (Toeplitz-like)
        # T[k, i] = y[k-i] if 0 <= k-i < n
        # We only need constraints where ac[k] is close to M
        # Use active set: constraints with ac[k] > M * 0.85
        threshold = M * 0.85
        active = np.where(ac > threshold)[0]
        
        b_var = cp.Variable(n, nonneg=True)
        T_active = np.zeros((len(active), n))
        for row_idx, k in enumerate(active):
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n, ki + 1)
            T_active[row_idx, lo:hi] = y[ki - np.arange(lo, hi)]
        
        constraints = [T_active @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        s_point = b_val / b_sum
        
        # Nesterov FW
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * s_point
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new /= sy
        
        beta_t = (t - 1.0) / (t + 2.0)
        x_mom = y_new + beta_t * (y_new - y)
        x_mom = np.maximum(x_mom, 0.0)
        sx = x_mom.sum()
        if sx > 1e-15:
            x_mom /= sx
        else:
            x_mom = y_new.copy()
        
        y = y_new
        
        for cand in [y, x_mom]:
            mc = _fast_autocorr(cand).max()
            if mc < best_m - 1e-16:
                best_m = mc
                best = cand.copy()
        
        if verbose and t % 5 == 0:
            J = 2.0 * n * best_m
            print(f"      LP-FW it={t} ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    
    return best, best_m

def _coord_descent(a0, t_limit, seed=0):
    """Peak-aware coordinate descent."""
    n = len(a0)
    a = _normalize(a0)
    best = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < t_limit and stall < max_stall:
        ac = _fast_autocorr(a)
        M = ac.max()
        # Find peak lags
        peak_lags = np.argsort(ac)[::-1][:10]
        
        # Compute contribution of each position to peaks
        contrib = np.zeros(n)
        for k in peak_lags:
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n, ki + 1)
            j = np.arange(lo, hi)
            contrib[j] += a[j] * a[ki - j]
        
        # Try to move mass from high-contrib to low-impact positions
        high_src = np.argsort(contrib)[::-1][:20]
        improved = False
        
        for src in high_src:
            if time.monotonic() - t0 > t_limit:
                break
            if a[src] < 1e-10:
                continue
            
            # Find best destination
            best_dst = -1
            best_score = best_m
            # Sample destinations
            n_try = min(n, 1000)
            candidates = rng.choice(n, n_try, replace=False) if n > n_try else np.arange(n)
            
            for dst in candidates:
                if dst == src:
                    continue
                # Quick estimate: check if moving to dst reduces peak
                amt = min(a[src] * 0.01, 1e-4)
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                M_trial = _fast_autocorr(a_trial).max()
                if M_trial < best_score:
                    best_score = M_trial
                    best_dst = dst
                    best_amt = amt
            
            if best_dst >= 0:
                a[src] -= best_amt
                a[best_dst] += best_amt
                a = np.maximum(a, 0)
                a /= a.sum()
                M_new = _fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best = a.copy()
                    improved = True
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.001, 0.01)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a /= a.sum()
                M_new = _fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best = a.copy()
                    stall = 0
    
    return best, best_m

def _interpolate(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, np.maximum(a_norm, 0))
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-15:
        result = np.ones(new_n) / new_n
    else:
        result /= s
    return result

def _get_parent():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p.copy()
    except:
        pass
    return None

def _score(a):
    """Compute the objective value."""
    n = len(a)
    ac = _fast_autocorr(a)
    max_b = ac.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)
