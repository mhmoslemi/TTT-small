The simplex projection algorithm:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_subgradient(a, ac, top_k, beta):
    """Compute subgradient using top-k lags with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def smooth_max_objective(a, ac, tau):
    """Smooth approximation of max(ac) using log-sum-exp."""
    scaled = ac * tau
    mx = scaled.max()
    if mx > 500:
        scaled = scaled - mx
    log_sum = np.log(np.exp(scaled).sum())
    return (mx + log_sum) / tau

def nesterov_smooth_max(a0, time_limit, verbose=True, rng_seed=42):
    """Nesterov accelerated mirror descent with adaptive smooth-max subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = 3.0
    stall_count = 0
    max_stall = 2000
    # Adaptive parameters
    top_k = 30
    beta_p = 6.0
    running_avg = a.copy()
    avg_count = 1

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Geometric temperature decay
        tau = 0.05 * (0.9995 ** t)
        tau = max(tau, 0.001)

        # Adaptive top-k: increase when stalling
        if stall_count > 500:
            top_k = min(top_k * 1.5, 300)
            beta_p = max(beta_p * 0.8, 2.0)
            max_stall = max(500, max_stall // 2)

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Compute subgradient
        g = compute_subgradient(y, ac, top_k, beta_p)

        # Nesterov step with backtracking
        eta = gamma / (np.sqrt(t) + 1.0)
        # Try a few step sizes (backtracking)
        best_a_new = None
        best_obj = smooth_max_objective(y, ac, tau)
        for factor in [1.0, 0.5, 0.25, 0.1]:
            if eta * factor < 1e-10:
                break
            log_a = np.log(y) - eta * factor * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = simplex_project(y)
            else:
                a_new = a_new / s_new

            ac_new = fast_autocorr(a_new)
            obj_new = smooth_max_objective(a_new, ac_new, tau)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a_new = a_new.copy()
                break

        if best_a_new is not None:
            x_prev = x
            x = best_a_new
        else:
            x_prev = x
            # Fallback: small step
            x = simplex_project(y - 0.001 * eta * g)

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Periodic restarting
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(300, max_stall // 2)
            pert = rng.normal(0, 0.01, n)
            x_r = best_a * (1 + pert)
            x_r = simplex_project(x_r)
            x = x_r
            x_prev = x_r
            gamma = max(0.5, gamma * 0.7)
            top_k = max(20, top_k // 2)
            beta_p = min(beta_p * 1.3, 10.0)

        # Running average check
        running_avg += x
        avg_count += 1
        if avg_count % 1000 == 0:
            avg = running_avg / avg_count
            avg = simplex_project(avg)
            M_avg = fast_autocorr(avg).max()
            if M_avg < best_m:
                best_m = M_avg
                best_a = avg.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NSM t={t} ({elapsed:.1f}s): J={J:.6f}, tau={tau:.5f}, topk={top_k}, stall={stall_count}")

    return best_a, best_m

def frank_wolfe_linesearch(a0, time_limit, solver, max_constraints=200, verbose=True):
    """Frank-Wolfe with growing constraints and backtracking line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Maintain a growing set of important lags
    constraint_lags = set()
    y = a.copy()

    for t in range(1, 5000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Update constraint set: lags within 0.97 of max
        threshold = M_curr * 0.97
        active_lags = np.where(ac >= threshold)[0]
        if len(active_lags) > max_constraints:
            active_lags = np.argsort(ac)[::-1][:max_constraints]

        # Build constraint matrix
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((len(active_lags), n))
        for row_idx, k in enumerate(active_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]

        constraints = [T @ b_var <= M_curr]
        t_lp = time.monotonic()
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        lp_time = time.monotonic() - t_lp
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Backtracking line search
        alpha = 2.0 / (t + 2.0)
        best_alpha = alpha
        best_m_trial = fast_autocorr(y).max()
        for trial_alpha in [alpha, alpha * 1.5, alpha * 2.0, alpha * 0.5]:
            y_trial = (1.0 - trial_alpha) * y + trial_alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            sy = y_trial.sum()
            if sy < 1e-15:
                continue
            y_trial = y_trial / sy
            m_trial = fast_autocorr(y_trial).max()
            if m_trial < best_m_trial:
                best_m_trial = m_trial
                best_alpha = trial_alpha
                if m_trial < best_m - 0.1 * (M_curr - best_m):
                    break

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new = y_new / sy

        my = fast_autocorr(y_new).max()
        if my < best_m:
            best_m = my
            best_a = y_new.copy()
        y = y_new

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW iter={t} ({elapsed:.1f}s): J={J:.6f}, alpha={best_alpha:.4f}, lp={lp_time:.2f}s")

    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=True, rng_seed=123):

            candidates = rng.integers(0, n, size=n_tries)
            
            for dst in candidates:
                if dst == i:

                # Golden section search for optimal fraction
                lo = 0.0

                gr = (math.sqrt(5) - 1) / 2

                # Evaluate at a few points for speed

                    if s_t < 1e-15:

                    a_trial = a_trial / s_t

                    if s_t > 1e-15:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

                break
            t_alt = min(remaining * 0.25, 100.0)
            if t_alt < 20:
                continue
            a_long = interpolate_sequence(best_seq, n_alt)
            a_in = norm_sum1(a_long)
            a_out, _ = nesterov_smooth_max(a_in, t_alt, verbose=False, rng_seed=n_alt)
            try_update(a_out, f"NSM_n{n_alt}")

        # Final FW on best
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and len(best_seq) <= 3500:
            t_final = min(remaining * 0.4, 100.0)
            if t_final > 20:
                print(f"\n=== Phase 6: Final FW n={len(best_seq)} (budget={t_final:.0f}s) ===")
                a_in = norm_sum1(best_seq)
                a_out, _ = frank_wolfe_linesearch(a_in, t_final, solver, max_constraints=200, verbose=False)
                try_update(a_out, f"FW_final_{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
