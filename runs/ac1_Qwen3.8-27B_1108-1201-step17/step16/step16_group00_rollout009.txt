            if a[i] < 1e-10:
                continue
            
            # Pick j: prefer positions NOT in the peak-lag support
            # (to reduce peak contribution)
            # Compute "peak pressure" at each position
            peak_pressure = np.zeros(n)
            for k in top_idx[:min(20, K)]:
                ki = int(k)
                for m in range(max(0, ki - n + 1), min(n, ki + 1)):
                    peak_pressure[m] += ac[k]
            
            # Pick j from low-pressure positions
            if rng.random() < 0.7:
                # Weighted random from low-pressure positions
                weights = 1.0 / (peak_pressure + 0.01)
                weights /= weights.sum()
                j = int(rng.choice(n, p=weights))
            else:
                j = int(rng.integers(n))
            
            if j == i:
                continue
            
            s = a[i] + a[j]
            if s < 1e-10:
                continue
            
            # Compute the optimal t using top-K lags
            # For each top lag k, compute delta_ac[k] as A*t^2 + B*t + C
            A_coeffs = np.zeros(K)
            B_coeffs = np.zeros(K)
            C_coeffs = np.zeros(K)
            
            ai = a[i]
            aj = a[j]
            
            for idx_k in range(K):
                k = int(top_idx[idx_k])
                A = 0.0
                B = 0.0
                C = 0.0
                
                # term_i: a_new[i] * a_new[k-i] - a_old[i] * a_old[k-i]
                if 0 <= k - i < n:
                    ki = k - i
                    if ki == i:
                        # t*t - ai*ai
                        A += 1.0
                        B += 0.0
                        C += -ai * ai
                    elif ki == j:
                        # t*(s-t) - ai*aj = -t^2 + s*t - ai*aj
                        A += -1.0
                        B += s
                        C += -ai * aj
                    else:
                        # t*a[ki] - ai*a[ki] = (t-ai)*a[ki]
                        B += a[ki]
                        C += -ai * a[ki]
                
                # term_j: a_new[j] * a_new[k-j] - a_old[j] * a_old[k-j]
                if 0 <= k - j < n:
                    kj = k - j
                    if kj == i:
                        # (s-t)*t - aj*ai = -t^2 + s*t - ai*aj
                        A += -1.0
                        B += s
                        C += -ai * aj
                    elif kj == j:
                        # (s-t)^2 - aj*aj = t^2 - 2s*t + s^2 - aj^2
                        A += 1.0
                        B += -2.0 * s
                        C += s * s - aj * aj
                    else:
                        # (s-t)*a[kj] - aj*a[kj] = (s-t-aj)*a[kj] = (ai-t)*a[kj]
                        B += -a[kj]
                        C += s * a[kj] - aj * a[kj]
                
                A_coeffs[idx_k] = A
                B_coeffs[idx_k] = B
                C_coeffs[idx_k] = C
            
            # Now find t in [0, s] minimizing max_k (ac[top_idx[k]] + A_k*t^2 + B_k*t + C_k)
            # Use golden section search (function is convex in t)
            
            golden = (np.sqrt(5.0) - 1.0) / 2.0
            
            def obj_at(t_val):
                vals = top_vals + A_coeffs * t_val**2 + B_coeffs * t_val + C_coeffs
                return vals.max()
            
            # Golden section search
            lo = 0.0
            hi = s
            # Initial pattern
            c = hi - golden * (hi - lo)
            d = lo + golden * (hi - lo)
            fc = obj_at(c)
            fd = obj_at(d)
            
            for _ in range(60):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - golden * (hi - lo)
                    fc = obj_at(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + golden * (hi - lo)
                    fd = obj_at(d)
                
                if hi - lo < 1e-14 * s:
                    break
            
            t_opt = (lo + hi) / 2.0
            t_opt = max(0.0, min(s, t_opt))
            
            # Apply the update
            a[i] = t_opt
            a[j] = s - t_opt
            a = np.maximum(a, 0.0)
            ss = a.sum()
            if ss < 1e-15:
                a = best_a.copy()
                continue
            a = a / ss
            
            # Full verification
            ac_new = fast_autocorr(a)
            M_new = ac_new.max()
            
            if M_new < M_curr - 1e-15:
                M_curr = M_new
                improved = True
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                    no_improve = 0
            else:
                # Revert
                a[i] = ai
                a[j] = aj
                a = a / a.sum()
        
        if not improved:
            no_improve += 1
        
        iter_count += 1
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Pair iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, no_improve={no_improve}")
    
    return best_a, best_M


def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad
