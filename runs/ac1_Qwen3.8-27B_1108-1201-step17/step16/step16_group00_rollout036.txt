For j=0: m = n-1. conv_result[n-1] = sum_i a[i'] * w[i'] (first n terms of w).
For j=n-1: m = 2n-2. conv_result[2n-2] = sum_i a[i'] * w[n-1+i'] (last n terms of w).

So grad[j] = 2 * conv_result[n-1+j] for j=0..n-1.
grad = 2 * conv_result[n-1 : 2*n-1]

def fast_conv(a, b=None):
    """Compute full linear convolution using FFT."""
    if b is None:
        b = a
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def compute_subgrad_full(a, beta=8.0, min_weight=1e-12):
    """
    Compute soft subgradient of max(conv(a,a)) using all lags with exp weighting.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    ac = fast_autocorr(a)
    # Soft-max weights
    shifted = ac - ac.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(ac) / len(ac)
    else:
        w = w / w_sum
    # Mask tiny weights to save computation
    w[w < min_weight] = 0.0
    if w.sum() < 1e-15:
        w = np.ones_like(ac) / len(ac)
        w = w / w.sum()
    # grad[j] = 2 * sum_m w[m] * a[m-j] for valid m-j
    # = 2 * convolve(a[::-1], w, mode='full')[n-1 : 2*n-1]
    # Use FFT
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev_padded = np.zeros(fft_size)
    a_rev_padded[:n] = a[::-1]
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    conv = np.fft.irfft(np.fft.rfft(a_rev_padded) * np.fft.rfft(w_padded), fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, ac.max()


def compute_subgrad_topk(a, top_k=80, beta=5.0):
    """Compute subgradient using only top-k peaking lags."""
    n = len(a)
    ac = fast_autocorr(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    shifted = top_vals - top_vals.max()
    w_top = np.exp(beta * shifted)
    w_sum = w_top.sum()
    if w_sum < 1e-30:
        w_top = np.ones(k_max) / k_max
    else:
        w_top = w_top / w_sum
    # Build full weight vector
    w = np.zeros(len(ac))
    for wi, idx in zip(w_top, top_idx):
        w[int(idx)] = wi
    # Same FFT computation
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev_padded = np.zeros(fft_size)
    a_rev_padded[:n] = a[::-1]
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    conv = np.fft.irfft(np.fft.rfft(a_rev_padded) * np.fft.rfft(w_padded), fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, ac.max()


def mirror_descent_optimize(a0, time_limit, verbose=True, beta_init=8.0,
                            beta_min=3.0, c0=5.0, restart_every=6000,
                            seed=42):
    """
    Nesterov-accelerated mirror descent on the simplex with adaptive soft-subgradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_beta = beta_init
    iterations = 0

    while time.monotonic() - t0 < time_limit:
        iterations += 1
        beta_t = (iterations - 1.0) / (iterations + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute subgradient
        if iterations < 200 or iterations % 100 == 0:
            g, M_curr = compute_subgrad_full(y, beta=cur_beta)
        else:
            g, M_curr = compute_subgrad_topk(y, top_k=80, beta=cur_beta)

        # Mirror descent step
        eta = gamma / (np.sqrt(iterations) + 1.0)
        log_y = np.log(y)
        log_a_new = log_y - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x.copy()
        x = a_new

        # Evaluate
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive beta: increase sharpness over time
        cur_beta = max(beta_min, beta_init - 0.001 * iterations)

        # Restart if stalled
        if stall_count > restart_every:
            stall_count = 0
            # Perturb around best
            pert_scale = max(0.01, 0.05 * (1 - stall_count / (2 * restart_every)))
            pert = rng.normal(0, pert_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            # Also try a random restart
            if rng.random() < 0.3:
                x_r = rng.dirichlet(np.ones(n) * 0.5)
                x = x_r
                x_prev = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    MD restart iter={iterations} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.2f}")

        if verbose and iterations % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD iter={iterations} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def lp_frank_wolfe_optimize(a0, time_limit, solver, verbose=True,
                            full_constraints=True, top_K=500):
    """
    LP-based Frank-Wolfe with optionally full constraints and Nesterov acceleration.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    prev_y = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Build constraint matrix
        # T[m, j] = y[m-j] if 0 <= m-j < n, else 0
        # We want: T @ b <= M_curr * ones
        if full_constraints:
            n_constraints = 2 * n - 1
            # Use FFT to build the constraint: conv(y, b)[m] <= M for all m
            # T is a (2n-1) x n matrix where T[m,j] = y[m-j] if valid
            T = np.zeros((2 * n - 1, n))
            for j in range(n):
                T[j:j + n, j] = y
            K_sel = 2 * n - 1
        else:
            K_sel = min(top_K, 2 * n - 1)
            # Select most binding constraints
            # For the current y, the most binding are where conv(y,y) is largest
            top_idx = np.argsort(ac)[::-1][:K_sel]
            T = np.zeros((K_sel, n))
            for col_idx, m in enumerate(top_idx):
                m_int = int(m)
                j_start = max(0, m_int - n + 1)
                j_end = min(n - 1, m_int)
                js = np.arange(j_start, j_end + 1)
                T[col_idx, js] = y[m_int - js]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr * np.ones(K_sel),
                        cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        prev_y = y
        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-{full_constraints and 'FULL' or 'TOPK'} iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_descent_optimize(a0, time_limit, verbose=True, seed=123):
    """
    Coordinate descent with global sensitivity analysis for move selection.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 200
    peels = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Find positions contributing to peak lags
        n_top = min(30, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_top]

        # For each peak lag, find the most contributing source position
        improvement_found = False

        for k in top_lags[:min(10, n_top)]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Compute destination sensitivity
            # For each position d, compute how much moving mass there would affect peak lags
            n_sens = min(15, n_top)
            sens = np.zeros(n)
            for k2 in top_lags[:n_sens]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                for d in range(s2, e2 + 1):
                    partner = k2i - d
                    sens[d] += a[partner]

            # Find best destination: low sensitivity, far from src
            candidates = np.where((sens < sens.mean()) & (np.abs(np.arange(n) - src) > n // 8))[0]
            if len(candidates) == 0:
                candidates = np.where(sens < sens.mean())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            # Score by inverse distance
            dist = np.abs(candidates - src).astype(float)
            dist[dist == 0] = 1
            scores = sens[candidates] / (1.0 + dist / n)
            best_dst = candidates[np.argmin(scores)]

            if best_dst == src:
                continue

            # Try multiple transfer fractions
            found_improve = False
            for frac in [0.02, 0.01, 0.005, 0.05, 0.002, 0.08, 0.03, 0.015]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improvement_found = True
                    found_improve = True
                    stall = 0
                    break

            if found_improve:
                peels += 1
                if verbose and peels % 50 == 0:
                    elapsed = time.monotonic() - t0
                    J = 2.0 * n * best_m
                    print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, peels={peels}")

        if not improvement_found:
            stall += 1
            # Random perturbation
            if stall % 10 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

            if verbose and stall % 50 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}, peels={peels}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def make_smooth_init(n, seed=0, freqs=None):
    """Create a smooth non-negative initialization via Fourier series."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    if freqs is None:
        freqs = [1, 2, 3, 4, 5, 6, 7]
    for f in freqs:
        coeff = rng.normal(0, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_symmetric_init(n, seed=0):
    """Create a symmetric smooth initialization."""
    a = make_smooth_init(n, seed=seed)
    a_sym = 0.5 * (a + a[::-1])
    return norm_sum1(a_sym)


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Multi-phase optimization combining mirror descent, LP-FW, and coordinate descent.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_symmetric_init(n_base, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Mirror descent with multiple inits ===
    print(f"\n=== Phase 1: Mirror Descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.30, 500.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        # Symmetric smooth
        inits.append(make_symmetric_init(n0, seed=7))
        labels.append("sym_smooth")

        # Different frequency inits
        inits.append(make_smooth_init(n0, seed=13, freqs=[1, 2, 3]))
        labels.append("low_freq")

        # Blend parent with smooth
        if parent is not None:
            blend = 0.7 * norm_sum1(parent) + 0.3 * make_symmetric_init(n0, seed=3)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Remove duplicates
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 50)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(20.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 15:
                break
            print(f"  MD init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = mirror_descent_optimize(init, t_avail, verbose=False,
                                                beta_init=8.0, beta_min=4.0,
                                                c0=5.0, restart_every=5000,
                                                seed=42 + idx)
            try_update(a_out, f"md_{lbl}")

    # === Phase 2: LP Frank-Wolfe (full constraints) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW Full (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 40 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_optimize(a_in, t2, solver, verbose=True,
                                            full_constraints=True)
        try_update(a_out, f"lp_full_n{n_best}")

    # === Phase 3: Mirror descent polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: MD Polish ===")
    n_best = len(best_seq)
    t3 = min(remaining * 0.12, 160.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_optimize(a_in, t3, verbose=False,
                                            beta_init=10.0, beta_min=5.0,
                                            c0=3.0, restart_every=8000,
                                            seed=999)
        try_update(a_out, f"md_polish_n{n_best}")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coordinate Descent ===")
    n_best = len(best_seq)
    t4 = min(remaining * 0.12, 160.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_optimize(a_in, t4, verbose=True, seed=42)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Second LP-FW round ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP-FW Round 2 ===")
    n_best = len(best_seq)
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_optimize(a_in, t5, solver, verbose=True,
                                            full_constraints=True)
        try_update(a_out, f"lp_r2_n{n_best}")

    # === Phase 6: Try slightly different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    candidate_lengths = [n_best - 200, n_best + 200, n_best + 400]
    candidate_lengths = [l for l in candidate_lengths if 1000 <= l <= 5000]

    for new_n in candidate_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n=== Phase 6: Length n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.25, 120.0)
        if t6a > 20:
            a_out, _ = mirror_descent_optimize(a_long, t6a, verbose=False,
                                                beta_init=8.0, beta_min=4.0,
                                                c0=4.0, restart_every=4000,
                                                seed=777)
            try_update(a_out, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 3500:
            t6b = min(remaining * 0.30, 120.0)
            if t6b > 20:
                a_out, _ = lp_frank_wolfe_optimize(a_long, t6b, solver, verbose=True,
                                                    full_constraints=True)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Coord (n={n_best}) ===")
    if remaining > 20:
        t7 = min(remaining * 0.4, 150.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_descent_optimize(a_in, t7, verbose=True, seed=555)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 8: Final LP if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30 and n_best <= 3500:
        print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
        t8 = min(remaining * 0.6, 120.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_optimize(a_in, t8, solver, verbose=True,
                                                full_constraints=True)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_symmetric_init(n_fb, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
