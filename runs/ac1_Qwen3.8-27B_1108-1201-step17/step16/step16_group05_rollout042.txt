import numpy as np
import time

def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    ac = np.fft.irfft(A * A, fft_size)[:size]
    return ac

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_grad_topk(a, ac, k=100):
    """Compute exact gradient of sum_{j in top_k} w_j * ac[j] where w_j = softmax(ac)."""
    n = len(a)
    # Get top k lags
    top_idx = np.argsort(ac)[::-1][:k]
    top_vals = ac[top_idx]
    # Softmax weights (numerically stable)
    shifted = top_vals - top_vals[0]
    exp_vals = np.exp(shifted)  # beta=1, we can adjust
    weights = exp_vals / exp_vals.sum()
    
    g = np.zeros(n)
    for wi, ki in zip(weights, top_idx):
        ki = int(ki)
        m_start = max(0, ki - n + 1)
        m_end = min(n - 1, ki)
        if m_start > m_end:
            continue
        m_range = np.arange(m_start, m_end + 1)
        j_range = ki - m_range
        g[m_range] += 2.0 * wi * a[j_range]
    return g, ac.max()

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

        beta_temps = [1.0, 2.0, 5.0, 10.0, 20.0, 30.0, 50.0, 100.0]
    
    stall = 0
    max_stall = 500
    eta = 0.1  # Initial step size
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Compute smoothed gradient
        # Use adaptive beta: start smooth, get sharper over time
        frac = min(1.0, t / 5000.0)
        beta = 5.0 + 50.0 * frac  # From 5 to 55
        
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = ac[top_idx]
        shifted = beta * (top_vals - top_vals[0])
        exp_vals = np.exp(shifted - shifted.max())
        weights = exp_vals / exp_vals.sum()
        
        g = np.zeros(n)
        for wi, ki in zip(weights, top_idx):
            ki = int(ki)
            m_start = max(0, ki - n + 1)
            m_end = min(n - 1, ki)
            if m_start > m_end:

            m_range = np.arange(m_start, m_end + 1)
            j_range = ki - m_range
            g[m_range] += 2.0 * wi * a[j_range]

        log_update = -eta * g

        else:

            best_a = a_new.copy()

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_grad_softmax(a, ac, beta, top_k=200):
    """Compute gradient of softmax-weighted top-k autocorrelation values."""
    n = len(a)
    num_lags = len(ac)
    k_eff = min(top_k, num_lags)
    top_idx = np.argsort(ac)[::-1][:k_eff]
    top_vals = ac[top_idx]
    shifted = beta * (top_vals - top_vals[0])
    shifted = shifted - shifted.max()
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    g = np.zeros(n)
    for wi, ki in zip(weights, top_idx):
        ki = int(ki)
        m_start = max(0, ki - n + 1)
        m_end = min(n - 1, ki)
        if m_start > m_end:
            continue
        m_range = np.arange(m_start, m_end + 1)
        j_range = ki - m_range
        g[m_range] += 2.0 * wi * a[j_range]
    return g, ac.max()


def mirror_descent(a0, time_limit, top_k=200, beta_start=3.0, beta_end=60.0,
                   eta_start=1.5, verbose=True, seed=42):
    """Mirror descent with adaptive temperature and step size."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    eta = eta_start
    stall = 0
    max_stall = 800
    best_eta = eta_start

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / (time_limit * 60.0))
        cur_beta = beta_start + (beta_end - beta_start) * frac

        ac = fast_autocorr(a)
        g, M_curr = compute_grad_softmax(a, ac, cur_beta, top_k=top_k)

        # Normalize gradient for stable step
        g_inf = np.max(np.abs(g))
        if g_inf < 1e-18:
            break
        g_normed = g / g_inf

        # Multiplicative update
        log_update = -eta * g_normed
        log_update = np.clip(log_update, -20.0, 20.0)
        a_new = a * np.exp(log_update)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        M_new = fast_autocorr(a_new).max()

        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
            a = a_new
            # Slightly increase eta on success
            eta = min(eta * 1.005, 4.0)
            best_eta = eta
        else:
            stall += 1
            eta = max(eta * 0.97, 0.05)
            # Don't accept the step
            a = a

        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.3)
            eta = 0.5
            # Perturb from best
            noise_scale = 0.005 + 0.01 * (1.0 - min(1.0, (t % 10000) / 10000.0))
            noise = rng.normal(0, noise_scale, n) * best_a
            a = np.maximum(best_a + noise, 1e-12)
            a = a / a.sum()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    MD restart t={t} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.3f}")

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.3f}, beta={cur_beta:.1f}")

    # Final eval
    final_m = fast_autocorr(best_a).max()
    return best_a, final_m


def accelerated_projected_gradient(a0, time_limit, beta_start=20.0, beta_end=5.0,
                                   top_k=150, c0=1.5, verbose=False, seed=42):
    """Nesterov-accelerated projected gradient on the simplex (backup method)."""

            break

        cur_beta = beta_start + (beta_end - beta_start) * frac

        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        g, _ = compute_grad_softmax(y, fast_autocorr(y), cur_beta, top_k=top_k)

        if g_inf < 1e-18:
            break
        g = g / g_inf

        eta = gamma / (np.sqrt(t) + 1.0)

        sx = x_new.sum()
        if sx < 1e-15:

        else:
            x_new = x_new / sx

        x_prev = x
        x = x_new

            stall_count = 0
            max_stall = int(max_stall * 1.3)
            noise = rng.normal(0, 0.005, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0

def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Precompute Toeplitz matrix
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a

    y = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        # Recompute T with current y
        for j in range(n):
            T[:, j] = 0.0
            T[j:j + n, j] = y[j:n - j] if j + n <= 2 * n - 1 else y[j:]
            # Fix: T[k, j] = y[k-j] if 0 <= k-j < n
        # Actually rebuild properly
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new

        mc = fast_autocorr(y).max()
        if mc < best_m - 1e-16:
            best_m = mc
            best_a = y.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, seed=123):
    """Peak-aware coordinate descent: move mass from high-peak contributors to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 150

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(30, len(ac))]
        improved = False

        # Compute peak contribution at each position
        peak_contrib = np.zeros(n)
        for k in top_k[:15]:
            ki = int(k)
            m_start = max(0, ki - n + 1)
            m_end = min(n - 1, ki)
            if m_start > m_end:
                continue
            m_range = np.arange(m_start, m_end + 1)
            j_range = ki - m_range
            contrib = a[m_range] * a[j_range]
            peak_contrib[m_range] += contrib

        # Find high-contribution sources
        sources = np.argsort(peak_contrib)[::-1][:40]

        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            # Score destinations: prefer positions with low autocorrelation impact
            n_try = min(n, 400)
            candidates = np.random.choice(n, n_try, replace=False) if n > n_try else np.arange(n)

            best_dst = -1
            best_score = 0.0

            for dst in candidates:
                if dst == src:
                    continue
                # Estimate impact of adding to dst
                impact = 0.0
                for k in top_k[:10]:
                    ki = int(k)
                    d1 = ki - dst
                    if 0 <= d1 < n:
                        impact += a[d1]
                    d2 = dst - ki
                    if 0 <= d2 < n:
                        impact += a[d2]
                score = -impact
                if score > best_score:
                    best_score = score
                    best_dst = dst

            if best_dst < 0:
                continue

            # Try moving mass
            for frac in [0.003, 0.008, 0.015, 0.03, 0.06]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.003, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def spectral_init(n, mode=0):
    """Generate diverse spectral-informed initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        # Sine window (good baseline)
        a = np.sin(np.pi * x) ** 2
    elif mode == 1:
        # Hann window
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    elif mode == 2:
        # Bimodal
        a = np.exp(-40.0 * (x - 0.25) ** 2) + np.exp(-40.0 * (x - 0.75) ** 2)
    elif mode == 3:
        # Quartic centered
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0) ** 2
    elif mode == 4:
        # Tri-modal
        a = (np.exp(-50.0 * (x - 0.15) ** 2) +
             np.exp(-50.0 * (x - 0.5) ** 2) +
             np.exp(-50.0 * (x - 0.85) ** 2))
    elif mode == 5:
        # Raised cosine with ripple
        base = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
        a = base * (1.0 + 0.3 * np.sin(4.0 * np.pi * x) ** 2)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    else:
        n_base = 2400
        best_seq = spectral_init(n_base, mode=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: Mirror descent (main workhorse) ===
    rem = remaining_time()
    print(f"\n=== Phase 1: Mirror Descent n={n0} ===")
    t1 = min(rem * 0.35, 550.0)
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent(
            a_in, t1, top_k=250, beta_start=3.0, beta_end=50.0,
            eta_start=1.5, verbose=True, seed=42)
        try_update(a_out, "md_n0")

    # === Phase 2: APG from mirror descent result (different geometry) ===
    rem = remaining_time()
    print(f"\n=== Phase 2: APG n={len(best_seq)} ===")
    t2 = min(rem * 0.15, 200.0)
    if t2 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = accelerated_projected_gradient(
            a_in, t2, beta_start=25.0, beta_end=8.0,
            top_k=200, c0=2.0, verbose=False, seed=77)
        try_update(a_out, "apg_n0")

    # === Phase 3: LP Frank-Wolfe ===
    rem = remaining_time()
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP-FW n={n_cur} ===")
    t3 = min(rem * 0.15, 250.0)
    if t3 > 40 and n_cur <= 3000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True)
        try_update(a_out, f"lpfw_n{n_cur}")

    # === Phase 4: Coordinate polish ===
    rem = remaining_time()
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: Coord n={n_cur} ===")
    t4 = min(rem * 0.10, 120.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=True, seed=42)
        try_update(a_out, f"coord_n{n_cur}")

    # === Phase 5: Mirror descent with different init (bimodal) ===
    rem = remaining_time()
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: MD bimodal init n={n_cur} ===")
    t5 = min(rem * 0.12, 150.0)
    if t5 > 30:
        init_bim = spectral_init(n_cur, mode=2)
        # Blend with current best
        blend = 0.5 * norm_sum1(best_seq) + 0.5 * init_bim
        a_out, _ = mirror_descent(
            blend, t5, top_k=200, beta_start=5.0, beta_end=40.0,
            eta_start=1.0, verbose=False, seed=123)
        try_update(a_out, "md_bimodal")

    # === Phase 6: Try longer length ===
    rem = remaining_time()
    n1 = min(n0 + 400, 2800)
    if n1 > n0 and rem > 150:
        n_cur = len(best_seq)
        n_target = min(n1, n_cur + 400)
        if n_target > n_cur:
            print(f"\n=== Phase 6: Scale to n={n_target} + MD ===")
            a_long = interpolate_sequence(best_seq, n_target)
            a_in = norm_sum1(a_long)
            t6 = min(rem * 0.20, 250.0)
            if t6 > 40:
                a_out, _ = mirror_descent(
                    a_in, t6, top_k=250, beta_start=3.0, beta_end=40.0,
                    eta_start=1.0, verbose=True, seed=99)
                try_update(a_out, f"md_n{n_target}")

                # Quick coord at longer length
                rem = remaining_time()
                if rem > 40:
                    t6b = min(rem * 0.3, 80.0)
                    if t6b > 15:
                        a_out2, _ = coordinate_polish(a_out, t6b, verbose=False, seed=55)
                        try_update(a_out2, f"coord_n{n_target}")

    # === Phase 7: Final mirror descent polish at best length ===
    rem = remaining_time()
    n_best = len(best_seq)
    print(f"\n=== Phase 7: MD polish n={n_best} ===")
    t7 = min(rem * 0.15, 150.0)
    if t7 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent(
            a_in, t7, top_k=300, beta_start=15.0, beta_end=80.0,
            eta_start=0.8, verbose=False, seed=314)
        try_update(a_out, f"md_polish_n{n_best}")

    # === Phase 8: Final coordinate polish ===
    rem = remaining_time()
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord n={n_best} ===")
    t8 = min(rem * 0.3, 100.0)
    if t8 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t8, verbose=True, seed=271)
        try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 9: One more LP if time allows ===
    rem = remaining_time()
    n_best = len(best_seq)
    if n_best <= 2500 and rem > 40:
        print(f"\n=== Phase 9: Final LP n={n_best} ===")
        t9 = min(rem * 0.5, 80.0)
        if t9 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t9, solver, verbose=False)
            try_update(a_out, f"lp_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
