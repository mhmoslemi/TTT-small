def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v, s_target=1.0):
    """Project v onto the simplex {a >= 0, sum(a) = s_target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # Algorithm from Duchi et al. 2011
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s_target))[0][-1]
    theta = (cssv[rho] - s_target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def logsumexp_smoothed_grad(a, ac, beta, top_k=200):
    """Compute gradient of log-sum-exp smoothed max of autocorrelation values."""
    n = len(a)
    # Use only top-k values to save computation
    k = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k]
    top_vals = ac[top_idx]
    
    # Stable log-sum-exp
    max_val = top_vals.max()
    shifted = top_vals - max_val
    weights = np.exp(beta * shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        # Uniform fallback
        return 2.0 * a, 0.0
    
    p = weights / w_sum
    
    # Gradient: 2 * sum_k p_k * Q_k * a
    grad = np.zeros(n)
    for pi, ki in zip(p, top_idx):
        k_int = int(ki)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start >= j_end:
            continue
        idx = np.arange(j_start, j_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * pi * a[j_idx]
    
    # The smoothed objective value
    ls = max_val + np.log(w_sum) / beta
    
    return grad, ls

def solve_lp_direction(y, M, active_lags, solver):
    """Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0."""
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    
    constraints = []
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)
    
    if not constraints:
        return None
        
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def lp_fw(a0, time_limit, solver, top_k_init=80, top_k_max=250, verbose=True):

    current_top_k = top_k_init

        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        if t % 100 == 0 and current_top_k < top_k_max:

        if verbose and t % 20 == 0:

                # Simpler: just compute the actual max after the trial move for top candidates
                pass
            
            # Simpler approach: try a few random destinations with exact evaluation
            n_tries = min(50, len(candidates))
            chosen = rng.choice(candidates, n_tries, replace=False)
            
            best_try_m = M_curr
            best_try_a = None
            for dst in chosen:
                if dst == src:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_try_m:
                    best_try_m = M_trial
                    best_try_a = a_trial
                    best_dst = dst
            
            if best_try_a is not None and best_try_m < best_m - 1e-15:
                a = best_try_a
                best_m = best_try_m
                best_a = a.copy()
                improved = True
                stall = 0
                break
            elif best_try_a is not None and best_try_m < M_curr:
                # Accept even if not better than global best (to escape local minima)
                a = best_try_a
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            # Perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / max(a.sum(), 1e-15)
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def spectral_equalize(a, n_iter=20, eps=1e-8):
    """Flatten power spectrum via iterative DFT correction."""
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = A.real ** 2 + A.imag ** 2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.5 * a + 0.5 * a_new
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
