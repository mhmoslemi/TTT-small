def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)

    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*ac_k)).
    Returns grad[i] = (4/beta) * sum_j p[i+j] * a[j]
    where p[k] = exp(beta*ac[k]) / sum_l exp(beta*ac[l])
    """
    n = len(a)
    max_ac = ac.max()
    exp_ac = np.exp(beta * (ac - max_ac))
    p = exp_ac / exp_ac.sum()

    # cross[i] = sum_j p[i+j] * a[j] = (p * reverse(a))[i+n-1]
    q = a[::-1]
    conv_len = len(p) + len(q) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    Q = np.fft.rfft(q, fft_size)
    conv = np.fft.irfft(P * Q, fft_size)[:conv_len]
    grad = (4.0 / beta) * conv[n - 1: n - 1 + n]
    return grad


def exact_subgradient_avg(a, ac, threshold_frac=0.99):
    """
    Compute the averaged (minimum-norm approximation) subgradient of max(ac).
    Averages the individual subgradients over all active lags.
    """
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)

    active = np.where(ac >= threshold_frac * M)[0]
    n_active = len(active)
    if n_active == 0:
        active = np.array([int(np.argmax(ac))])
        n_active = 1

    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]

    g /= n_active
    return g


def mirror_update(a, grad, eta):
    """Mirror descent step on the simplex using negative entropy."""
    log_ratio = -eta * grad
    # Shift to avoid underflow
    log_ratio_shifted = log_ratio - log_ratio.max()
    weights = np.exp(log_ratio_shifted)
    a_new = a * weights
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new / a_new.sum()
    return a_new


def lse_phase(a0, time_limit, beta_start=3.0, beta_end=80.0, c_step=2.0,
              verbose=True, rng_seed=1):
    """
    LSE smooth gradient descent with beta annealing and Nesterov acceleration.
    """
    n = len(a0)
    a = a0.copy()
    ac = fast_conv_self(a)
    beta_curr = ac.max()  # will be overridden
    best_a = a.copy()
    best_obj = 2.0 * n * ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 300

    total_iters_est = 20000
    t_iter = 0

    while time.monotonic() - t0 < time_limit:
        t_iter += 1
        # Annealing schedule for beta
        frac = min(t_iter / float(total_iters_est), 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac

        # Nesterov extrapolation
        if t_iter > 2:
            beta_t = (t_iter - 1.5) / (t_iter + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        else:
            y = x.copy()

        # Compute LSE gradient
        ac_y = fast_conv_self(y)
        grad = lse_gradient(y, ac_y, beta)

        # Step size with backtracking
        eta = c_step / (beta * np.sqrt(t_iter + 1.0)) * np.sqrt(n)
        a_new = mirror_update(y, grad, eta)

        # Check improvement (using true objective, not LSE)
        ac_new = fast_conv_self(a_new)
        obj_new = 2.0 * n * ac_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            # Backtracking: reduce step
            eta *= 0.7
            a_new = mirror_update(y, grad, eta)
            ac_new = fast_conv_self(a_new)
            obj_new = 2.0 * n * ac_new.max()
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                stall = 0

        x_prev = x
        x = a_new

        # Restart if stuck
        if stall > max_stall:
            stall = 0
            noise = rng.normal(0, 0.01, n) * best_a
            x = best_a + noise
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()

        if verbose and t_iter % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE t={t_iter} ({elapsed:.1f}s): obj={best_obj:.6f}, beta={beta:.1f}, stall={stall}")

    return best_a, best_obj


def exact_amd_phase(a0, time_limit, c0=2.0, threshold_frac=0.99,
                    verbose=True, rng_seed=2):
    """
    Nesterov-accelerated exact subgradient with averaged (min-norm) selection.
    """
    n = len(a0)
    a = a0.copy()
    ac = fast_conv_self(a)
    best_a = a.copy()
    best_obj = 2.0 * n * ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 3000
    thresh = threshold_frac
    c_current = c0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute subgradient
        ac_y = fast_conv_self(y)
        g = exact_subgradient_avg(y, ac_y, threshold_frac=thresh)

        # Step size
        eta = c_current / (np.sqrt(t) + 1.0) * np.sqrt(n)

        # Mirror update
        a_new = mirror_update(y, g, eta)

        x_prev = x
        x = a_new

        # Evaluate
        ac_new = fast_conv_self(x)
        obj_new = 2.0 * n * ac_new.max()
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive restart
        if stall > max_stall:
            stall = 0
            noise_scale = 0.005 + 0.01 * np.sqrt(min(stall, max_stall * 3) / max_stall)
            noise = rng.normal(0, noise_scale, n) * best_a
            x = best_a + noise
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            # Tighten threshold over time
            if thresh > 0.999:
                thresh = max(0.999, thresh - 0.002)
            c_current = c_current * 1.15

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AMD t={t} ({elapsed:.1f}s): obj={best_obj:.6f}, thresh={thresh:.4f}, stall={stall}")

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=True, rng_seed=3):
    """
    Sensitivity-aware coordinate descent with multi-scale mass transfer.
    """
    n = len(a0)
    a = a0.copy()
    ac = fast_conv_self(a)
    best_a = a.copy()
    best_obj = 2.0 * n * ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv_self(a)
        M = ac.max()
        peak_idx = np.argsort(ac)[::-1][:min(25, len(ac))]

        # Compute sensitivity: how much each position contributes to peaks
        peak_sens = np.zeros(n)
        for k in peak_idx[:15]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            peak_sens[idx] += a[j_idx]

        # Find high-sensitivity (source) and low-sensitivity (destination) positions
        # Sources: positions with high a[i] AND high sensitivity
        source_score = a * peak_sens
        top_sources = np.argsort(source_score)[::-1][:min(50, n)]

        # Destinations: low sensitivity, low value
        in_peak = np.zeros(n, dtype=bool)
        for k in peak_idx[:15]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            in_peak[i_start:i_end] = True

        cold = np.where(~in_peak)[0]
        if len(cold) > 300:
            cold = rng.choice(cold, size=300, replace=False)

        improved = False

        for src in top_sources[:30]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            # Sort destinations by sensitivity (lowest first)
            sorted_dst = cold[np.argsort(peak_sens[cold])[:min(50, len(cold))]]

            for dst in sorted_dst[:25]:
                if dst == src:
                    continue
                if a[dst] >= a[src] * 1.5:
                    continue

                # Try multiple transfer fractions
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv_self(a_trial).max()
                    obj_trial = 2.0 * n * M_trial
                    if obj_trial < best_obj - 1e-12:
                        a = a_trial
                        best_obj = obj_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

            if improved:
                break

        if not improved:
            stall += 1
            # Random exploration: try random transfers
            n_rand = max(5, n // 200)
            for _ in range(n_rand):
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-8:
                    continue
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                old_i = a[i]
                old_j = a[j]
                a[i] = old_i - amt
                a[j] = old_j + amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv_self(a).max()
                    obj_new = 2.0 * n * M_new
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                    else:
                        a[i] = old_i
                        a[j] = old_j

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord ({elapsed:.1f}s): obj={best_obj:.6f}, stall={stall}")

    return best_a, best_obj


def fine_coordinate_descent(a0, time_limit, verbose=True, rng_seed=4):
    """
    Very fine-grained coordinate descent: tries tiny moves in many directions.
    """
    n = len(a0)
    a = a0.copy()
    ac = fast_conv_self(a)
    best_a = a.copy()
    best_obj = 2.0 * n * ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 100

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv_self(a)
        M = ac.max()
        # Focus on the top few peak lags
        peak_idx = np.argsort(ac)[::-1][:10]

        improved = False

        # For each peak, try to reduce it by modifying the most contributing pairs
        for k in peak_idx[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            if len(idx) == 0:
                continue
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]

            # Sort by contribution (highest first)
            order = np.argsort(contrib)[::-1]

            for pi in order[:20]:
                src_i = idx[pi]
                dst_j = j_idx[pi]
                if a[src_i] < 1e-10:
                    continue

                # Try moving mass from src_i to several candidate destinations
                # Candidates: nearby positions with low sensitivity
                positions_to_check = []
                # Nearby positions
                lo = max(0, src_i - 20)
                hi = min(n, src_i + 21)
                for pos in range(lo, hi):
                    if pos != src_i and pos != int(dst_j):
                        positions_to_check.append(pos)
                # Random positions
                positions_to_check.extend(int(x) for x in rng.integers(n, size=20))

                for dst in positions_to_check[:40]:
                    if dst == src_i or a[dst] > a[src_i]:
                        continue
                    for frac in [0.001, 0.002, 0.005, 0.01, 0.02]:
                        amt = a[src_i] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_conv_self(a_trial).max()
                        obj_trial = 2.0 * n * M_trial
                        if obj_trial < best_obj - 1e-12:
                            a = a_trial
                            best_obj = obj_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1

        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCoord ({elapsed:.1f}s): obj={best_obj:.6f}, stall={stall}")

    return best_a, best_obj


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with LSE smooth phase, exact AMD, and coordinate descent.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2 + 0.1 * np.cos(2 * np.pi * x) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: LSE smooth optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE smooth n={n0} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 30:
        a_out, _ = lse_phase(norm_sum1(best_seq), t1, beta_start=2.0, beta_end=60.0,
                             c_step=2.5, verbose=True, rng_seed=1)
        try_update(a_out, f"lse_n{n0}")

    # === Phase 2: Exact AMD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Exact AMD n={n_best} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 30:
        a_out, _ = exact_amd_phase(norm_sum1(best_seq), t2, c0=2.0, threshold_frac=0.99,
                                   verbose=True, rng_seed=2)
        try_update(a_out, f"amd_n{n_best}")

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord descent n={n_best} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 20:
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t3, verbose=True, rng_seed=3)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 4: Fine coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Fine coord n={n_best} (remaining={remaining:.0f}s) ===")

        try_update(a_out, f"finecoord_n{n_best}")

    # === Phase 5: LSE again (different params) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LSE restart n={n_best} (remaining={remaining:.0f}s) ===")

                             c_step=1.5, verbose=True, rng_seed=5)
        try_update(a_out, f"lse2_n{n_best}")

    # === Phase 6: Exact AMD (tighter threshold) ===

        if remaining > 60 and n_short < n_best_now:
            print(f"  Trying n={n_short}")
            a_short = interpolate_sequence(best_seq, n_short)
            t7b = min(remaining * 0.20, 80.0)
            if t7b > 20:

                                       verbose=False, rng_seed=11)
            try_update(a_out, f"amd_subpolish_n{n_best}")

        if t10 > 5:
            a_out, _ = coordinate_descent(norm_sum1(best_seq), t10, verbose=False, rng_seed=12)
            try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
