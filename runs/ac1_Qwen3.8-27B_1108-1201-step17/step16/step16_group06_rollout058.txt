(2*Q_k*a)[p] = 2 * sum_j Q_k[p,j] * a[j] = 2 * a[k-p] (if 0 ≤ k-p < n), else 0.

So ∇_a conv(a,a)[k] = 2 * (a[k], a[k-1], ..., a[max(0,k-n+1)])^T (padded with zeros) = 2 * reverse(a)[k-n+1 : k+1] (with appropriate bounds).

More precisely, the p-th component of the gradient is 2*a[k-p] if 0 ≤ k-p < n, else 0.

So the gradient of conv(a,a)[k] at a is: grad[p] = 2*a[k-p] for p in range(max(0, k-n+1), min(n, k+1)), and 0 otherwise.

Now, for the subgradient of f(a) = max_k conv(a,a)[k], we take a weighted average of the gradients at the active peaks.

If the active set is {k_1, ..., k_m} (where conv(a,a)[k_i] = f(a)), then any convex combination of their gradients is a subgradient.

For optimization, we can use:
- subgrad = sum_i w_i * grad(a, k_i) where w_i >= 0, sum w_i = 1.
- A good choice is w_i proportional to exp(β * conv(a,a)[k_i]) for large β (this concentrates on the maximum).

The update (for mirror descent with entropy):
a_new[p] ∝ a[p] * exp(-eta * subgrad[p])

This is the "exponentiated gradient" or "multiplicative weights" update.

Now, the key question is: how fast does this converge? For a max of N smooth functions, the subgradient method with step size eta_t = O(1/sqrt(t)) gives O(1/sqrt(t)) convergence. With Nesterov acceleration, we can get O(1/t).

But since we're dealing with a non-convex problem (max of quadratics), convergence is not guaranteed. However, in practice, it works well when the "landscape" is relatively smooth.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

6. Multi-length exploration
7. Final coordinate descent polish

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgradient(a, ac, top_k=20, beta=3.0):
    """Compute soft subgradient of max(conv) using top-k peaks."""
    n = len(a)
    # Get top-k peak indices
    top_idx = np.argsort(ac)[::-1][:top_k]
    top_vals = ac[top_idx]
    # Softmax weights
    w = np.exp(beta * (top_vals - top_vals[0]))
    w /= w.sum()
    # Accumulate subgradient
    grad = np.zeros(n)
    for wt, k in zip(w, top_idx):
        k_int = int(k)
        # grad[p] += 2 * wt * a[k-p] for valid p
        # p ranges from max(0, k-n+1) to min(n-1, k)
        p_start = max(0, k_int - n + 1)
        p_end = min(n - 1, k_int)
        if p_start > p_end:
            continue
        # a[k-p] for p in [p_start, p_end]
        indices = np.arange(p_start, p_end + 1)
        a_indices = k_int - indices
        grad[indices] += 2.0 * wt * a[a_indices]
    return grad

def multiplicative_weights_optimize(a0, time_limit, top_k=30, beta=3.0, verbose=True):
    """Nesterov-accelerated multiplicative weights for minimizing max(conv)."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = 1.0  # step size parameter
    stall = 0
    max_stall = 2000
    
    for t in range(1, 10**9):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute autocorrelation and subgradient
        ac = fast_autocorr(y)
        M = ac.max()
        grad = compute_subgradient(y, ac, top_k=top_k, beta=beta)
        
        # Adaptive step size
        # eta = gamma / (t^0.5 + 1)
        eta = gamma / (np.sqrt(t) + 1.0)
        
        # Multiplicative weights update
        log_a = np.log(y) - eta * grad
        # Subtract max for numerical stability
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Perturbation to escape local minima
        if stall > max_stall:
            stall = 0
            # Random perturbation
            scale = 0.05 + 0.05 * (1.0 - best_m / 2.0)  # smaller as we improve
            noise = rng.normal(0, scale, n)
            x_pert = best_a * np.exp(noise)
            x_pert = np.maximum(x_pert, 1e-10)
            x_pert = x_pert / x_pert.sum()
            x = x_pert
            x_prev = x_pert
            best_m = fast_autocorr(x).max()
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MW t={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}, stall={stall}")
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):

        # Get top peaks
        top_count = min(20, len(ac))

        sensitivity = np.zeros(n)

        high_sens = np.argsort(sensitivity)[::-1][:10]

def block_swap(a, block_size, rng):
    """Try swapping mass between blocks to reduce peak."""
    n = len(a)
    a = a.copy()
    num_blocks = n // block_size
    if num_blocks < 2:
        return a
    
    # Compute peak
    ac = fast_autocorr(a)
    peak_idx = np.argmax(ac)
    
    # Determine which blocks contribute to the peak
    k_int = int(peak_idx)
    p_start = max(0, k_int - n + 1)
    p_end = min(n - 1, k_int)
    
    # Try moving mass from contributing blocks to non-contributing blocks
    contributing_blocks = set()
    for p in range(p_start, p_end + 1):
        block = p // block_size
        contributing_blocks.add(block)
    
    non_contributing = [b for b in range(num_blocks) if b not in contributing_blocks]
    if not non_contributing:
        return a
    
    # Move a small fraction from contributing to non-contributing
    for src_block in list(contributing_blocks)[:3]:
        src_start = src_block * block_size
        src_end = (src_block + 1) * block_size
        for dst_block in non_contributing[:3]:
            dst_start = dst_block * block_size
            dst_end = (dst_block + 1) * block_size
            
            # Move mass
            for frac in [0.01, 0.05, 0.1]:
                a_trial = a.copy()
                src_mass = a_trial[src_start:src_end].sum()
                if src_mass < 1e-10:
                    continue
                amt = src_mass * frac
                # Reduce source proportionally
                a_trial[src_start:src_end] *= (1 - frac)
                # Add to destination proportionally
                dst_mass = a_trial[dst_start:dst_end].sum()
                if dst_mass < 1e-15:
                    a_trial[dst_start:dst_end] += amt / block_size
                else:
                    a_trial[dst_start:dst_end] += a_trial[dst_start:dst_end] * (amt / dst_mass)
                
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                M_trial = fast_autocorr(a_trial).max()
                M_curr = ac.max()
                if M_trial < M_curr:
                    a = a_trial
                    ac = fast_autocorr(a)
                    break
    
    return a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
