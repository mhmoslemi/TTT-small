"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. APG with power-smoothed max objective (adaptive top_k and exponent)
2. Mirror descent with multi-peak multiplicative update
3. Direct coordinate moves targeting individual peaks

5. Multi-restart with diverse initializations
"""

import numpy as np
import time


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def power_smoothed_grad(a, top_k, p):
    c = fast_conv(a)
    n = len(a)
    K = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:K]
    top_vals = c[top_idx]
    obj = (np.mean(top_vals ** p)) ** (1.0 / p)
    weights = top_vals ** (p - 1)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= wsum
    grad = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return obj, grad


def apg_power(a0, time_limit, top_k=500, p_start=15.0, p_end=60.0, c0=2.0, seed=42, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 1500
    cur_top_k = top_k

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        frac = min(1.0, t / 12000.0)
        cur_p = p_start + (p_end - p_start) * frac
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)
        obj, g = power_smoothed_grad(y, cur_top_k, cur_p)
        eta = gamma / (np.sqrt(t) + 1.0)
        x_new = project_simplex(y - eta * g)
        x_prev = x
        x = x_new
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.25)
            cur_top_k = min(cur_top_k + 50, 2000)
            noise = rng.normal(0, 0.003, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      APG-pow restart t={t} ({elapsed:.0f}s): J={J:.6f}, top_k={cur_top_k}, p={cur_p:.0f}")
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      APG-pow t={t} ({elapsed:.0f}s): J={J:.6f}, p={cur_p:.0f}, top_k={cur_top_k}")
    return best_a, best_m


def mirror_descent(a0, time_limit, n_peaks=30, eta0=0.3, seed=42, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    eta = eta0
    stall = 0

    for it in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_conv(a)
        K = min(n_peaks, len(c))
        top_idx = np.argsort(c)[::-1][:K]
        top_vals = c[top_idx]
        w = np.exp((top_vals - top_vals[0]) * 3.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(K) / K
        else:
            w /= wsum
        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            g[idx] += wi * a[j_idx]
        a_new = a * np.exp(-eta * g)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            a = a_new
            if stall > 50:
                eta *= 0.7
                stall = 0
                if eta < 0.0005:
                    noise = rng.normal(0, 0.002, n) * best_a
                    a = np.maximum(best_a + noise, 1e-12)
                    a = a / a.sum()
                    eta = eta0 * 0.2
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      MD iter={it} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.5f}")
    return best_a, best_m


def coord_peak_reduce(a0, time_limit, n_peaks=10, seed=42, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        c = fast_conv(a)
        top_idx = np.argsort(c)[::-1][:n_peaks]
        improved = False
        for k_star in top_idx:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k_star)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            products = a[idx] * a[j_idx]
            if np.all(products < 1e-20):
                continue
            src_pos = idx[np.argmax(products)]
            candidates = np.setdiff1d(np.arange(n), idx)
            if len(candidates) == 0:
                continue
            scores = np.zeros(len(candidates))
            for i_face, j_val in enumerate(candidates):
                if 0 <= k_int - j_val < n:
                    scores[i_face] = a[j_val] * a[k_int - j_val]
            for k2 in top_idx[:4]:
                k2_int = int(k2)
                for i_face, j_val in enumerate(candidates):
                    if 0 <= k2_int - j_val < n:
                        scores[i_face] += 0.05 * a[j_val] * a[k2_int - j_val]
            dst_pos = candidates[np.argmin(scores)]
            for frac in [0.001, 0.003, 0.008, 0.02, 0.04]:
                amt = a[src_pos] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src_pos] -= amt
                a_trial[dst_pos] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.001, 0.008)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CoordPeak ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
