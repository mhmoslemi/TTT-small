    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += wi * a[j_idx]
    
    return 2.0 * g

def exact_subgradient_all(a, c, threshold_frac=0.95):
    """Subgradient using all lags within threshold_frac of max."""
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    
    active = np.where(c >= threshold_frac * c_max)[0]
    m = len(active)
    if m == 0:
        active = np.array([int(c.argmax())])
        m = 1
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += a[j_idx]
    
    return 2.0 * g / m  # average over active set

def nesterov_subgrad_descent(a0, time_limit, alpha=50.0, top_k=30, 
                                c0=1.0, momentum_decay=0.9,
                                verbose=False, rng_seed=42):
    """
    Nesterov accelerated subgradient descent with smoothed subgradient.
    Optimizes max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    c = fast_conv(a)
    best_a = a.copy()
    best_obj = 2.0 * n * c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    # Nesterov state
    x_prev = a.copy()
    x = a.copy()
    v = np.zeros(n)  # velocity for momentum
    
    # BB step size state
    a_prev_vec = a.copy()
    g_prev = None
    gamma = c0 / np.sqrt(n)
    
    stall = 0
    max_stall = 3000
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta = (1.0 + np.sqrt(1.0 + 4.0 * (t-1)**2)) / 2.0
        # Actually, standard Nesterov: y = x + (t-1)/(t+2) * (x - x_prev)
        beta = (t - 1.0) / (t + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute subgradient
        c_y = fast_conv(y)
        g = smoothed_subgradient(y, c_y, alpha=alpha, top_k=top_k)
        
        # BB step size
        if g_prev is not None:
            dy = y - a_prev_vec
            dg = g - g_prev
            s_dd = dy @ dg
            s_dg = dg @ dg
            if s_dd > 1e-20 and s_dg > 1e-20:
                gamma_bb = s_dd / s_dg
                if gamma_bb > 0 and gamma_bb < 10.0 * gamma:
                    gamma = 0.5 * gamma + 0.5 * gamma_bb
            a_prev_vec = y.copy()
            g_prev = g.copy()
        
        # Step
        step = gamma / (np.sqrt(t) + 1.0) * np.sqrt(n)
        x_new = y - step * g
        x_new = np.maximum(x_new, 0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx
        
        x_prev = x
        x = x_new
        
        # Evaluate
        c_x = fast_conv(x)
        obj = 2.0 * n * c_x.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stuck
        if stall > max_stall:
            stall = 0
            # Perturb best
            noise = rng.normal(0, 0.01, n)
            x_restart = best_a * (1.0 + noise)
            x_restart = np.maximum(x_restart, 1e-10)
            sr = x_restart.sum()
            if sr > 1e-15:
                x = x_restart / sr
            else:
                x = best_a.copy()
            x_prev = x.copy()
            # Adjust parameters
            alpha = min(alpha * 1.5, 200.0)
            gamma = gamma * 1.1
            if verbose:
                print(f"      Restart at t={t}, alpha={alpha:.1f}, gamma={gamma:.4f}")
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      t={t} ({elapsed:.1f}s): obj={best_obj:.6f}, stall={stall}")
    
    return best_a, best_obj

def coordinate_polish(a0, time_limit, verbose=False, rng_seed=77):
    """
    Coordinate-wise polishing: for each position, find the optimal value
    given all other positions fixed.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    c = fast_conv(a)
    best_obj = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 50
    order = np.arange(n)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M = c.max()
        
        # Randomize order
        rng.shuffle(order)
        improved_this_pass = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try changing a[i]
            # The contribution of a[i] to c[k] is: sum over j where k=i+j or k=i-j (need to be careful)
            # Actually c[k] = sum_j a[j]*a[k-j], so a[i] appears in c[k] when k-i >= 0 and k-i < n,
            # i.e., c[k] includes a[i]*a[k-i] when 0 <= k-i < n.
            # Also c[k] includes a[k-i]*a[i] when 0 <= i < n (same thing by symmetry of sum).
            # So the terms involving a[i] are: for each valid j, c[i+j] gets a[i]*a[j], and c[j-i] gets... 
            # Actually, c[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
            # a[i] appears when j=i (giving a[i]*a[k-i] in c[k] for k in [i, i+n-1])
            # and when k-j = i, i.e., j = k-i (giving a[k-i]*a[i] in c[k] for k in [i, i+n-1])
            # Wait, these are the same set! c[k] contains the term a[i]*a[k-i] if 0 <= k-i < n,
            # AND the term a[k-i]*a[i] if 0 <= k-i < n... no that's double counting.
            
            # Let me be more careful. c = convolve(a, a), so:
            # c[k] = sum_j a[j] * a[k-j] (with valid indices 0<=j<n, 0<=k-j<n)
            # The term involving a[i] specifically is when j=i: a[i]*a[k-i] (valid when 0<=k-i<n, i.e., i<=k<=i+n-1)
            # OR when k-j=i, i.e., j=k-i: a[k-i]*a[i] (valid when 0<=k-i<n, i.e., i<=k<=i+n-1)
            # These are the SAME terms! So a[i] contributes a[i]*a[k-i] to c[k] for k in [i, i+n-1]
            # (where k-i is in [0, n-1]).
            # But wait, when i = k-i, i.e., k=2i, we only count it once.
            
            # Actually for convolution, c[k] = sum_j a[j]*a[k-j]. The term a[i]*a[i] appears
            # when j=i and k-j=i, i.e., k=2i. For k != 2i, the terms a[i]*a[k-i] and a[k-i]*a[i]
            # are different terms in the sum (they correspond to j=i and j=k-i respectively).
            # So the total contribution of a[i] to c[k] is:
            #   a[i]*a[k-i] (from j=i) + a[k-i]*a[i] (from j=k-i) = 2*a[i]*a[k-i] if k != 2i and both valid
            #   a[i]^2 (from j=i=k-i) if k = 2i
            
            # For simplicity, let's just try binary search on a[i] to minimize max(c).
            # This is expensive (O(n log n) per coordinate due to FFT), so let's be selective.
            
            # Quick check: is a[i] at a "hot" position?
            # Compute the "sensitivity" of the peak to a[i]
            peak_k = c.argmax()
            k_int = int(peak_k)
            if 0 <= k_int - i < n:
                # a[i] contributes to the peak
                pass
            else:
                continue  # Skip if a[i] doesn't affect the peak
            
            # Try a few values
            current_val = a[i]
            if current_val < 1e-15:
                continue
            
            # Try reducing
            for frac in [0.1, 0.3, 0.5, 0.7]:
                new_val = current_val * (1.0 - frac)
                a_trial = a.copy()
                a_trial[i] = new_val
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_t = fast_conv(a_trial).max()
                if M_t < best_obj - 1e-15:
                    a = a_trial
                    best_obj = M_t
                    best_a = a.copy()
                    improved_this_pass = True
                    break
            
            # Try increasing (if it was low)
            if not improved_this_pass and current_val < 0.1 / n:
                for frac in [0.5, 1.0, 2.0]:
                    new_val = current_val * (1.0 + frac)
                    a_trial = a.copy()
                    a_trial[i] = new_val
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_t = fast_conv(a_trial).max()
                    if M_t < best_obj - 1e-15:
                        a = a_trial
                        best_obj = M_t
                        best_a = a.copy()
                        improved_this_pass = True
                        break
        
        if not improved_this_pass:
            stall += 1
            # Random perturbation to escape
            n_perturb = max(5, n // 50)
            idx = rng.choice(n, n_perturb, replace=False)
            for i in idx:
                a[i] *= (1.0 + rng.normal(0, 0.05))
                a[i] = max(a[i], 1e-10)
            a = a / a.sum()
            M_new = fast_conv(a).max()
            if M_new < best_obj:
                best_obj = M_new
                best_a = a.copy()
                stall = 0
        else:
            stall = 0
    
    return best_a, best_obj

def sinusoidal_init(n, mode=0, rng=None):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        # Basic sinusoidal
        a = np.sin(np.pi * x)**2 + 0.1 * np.cos(2*np.pi*x)**2
    elif mode == 1:
        # Two-lobe
        a = np.sin(np.pi * x)**4 + 0.05 * np.sin(2*np.pi*x)**2
    elif mode == 2:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2*np.pi*x))
        a = a**2
    elif mode == 3:
        # Gaussian-like
        a = np.exp(-4.0 * (x - 0.5)**2) + 0.1
    elif mode == 4:
        # Multi-frequency
        a = 1.0 + 0.3*np.cos(2*np.pi*3*x) + 0.2*np.cos(2*np.pi*7*x + 1.0)
    elif mode == 5:
        # Uniform with noise
        a = np.ones(n) + 0.3 * np.sin(2*np.pi*2*x)
    else:
        # Random
        if rng is None:
            rng = np.random.default_rng(mode)
        a = rng.random(n) + 0.1
    a = np.maximum(a, 0.01)
    return a / a.sum()
