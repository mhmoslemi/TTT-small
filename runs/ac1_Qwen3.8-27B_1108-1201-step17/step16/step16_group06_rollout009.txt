    """Compute gradient of log-sum-exp penalty.
    grad_j = 2n * ((w *star* p)[j] + (w * p)[j])
    where *star* is cross-correlation and * is convolution.
    """
    # FFT size
    size = 2 * n - 1
    fft_n = 1 << (size - 1).bit_length()
    
    # Convolution w * p
    W = np.fft.rfft(w, fft_n)
    P = np.fft.rfft(p, fft_n)
    conv_wp = np.fft.irfft(W * P, fft_n)[:n]  # first n components
    
    # Cross-correlation w *star* p[j] = sum_i w[i+j] * p[i]
    # This equals (w * flip(p))[j + n - 1] where flip(p)[i] = p[n-1-i]

    cross = np.fft.irfft(W * PF, fft_n)  # full result, length fft_n

    cross_corr = cross[n-1:2*n-1]  # length n
    
    return 2 * n * (conv_wp + cross_corr)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def cosine_init(n, freq=1.0, alpha=0.5):
    """Create a cosine-modulated initialization."""
    x = np.arange(n, dtype=np.float64)
    envelope = np.sin(np.pi * (x + 0.5) / n)
    modulation = 1.0 + alpha * np.cos(2.0 * np.pi * freq * x / n)
    p = envelope * modulation
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / n
    return p / s


def random_smooth_init(n, rng_seed=0, n_bumps=3):
    """Create a random smooth initialization using random Fourier features."""
    rng = np.random.default_rng(rng_seed)
    x = np.linspace(0.0, 1.0, n)
    p = np.ones(n, dtype=np.float64)
    for _ in range(n_bumps):
        freq = rng.uniform(1.0, 5.0)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        p += 0.3 * np.cos(2.0 * np.pi * freq * x + phase)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / n
    return p / s


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator combining LSE mirror descent and LP-FW."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    # Load parent
    n_default = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
            n_default = len(parent)
    except Exception:
        pass

    if best_seq is None:
        best_seq = cosine_init(n_default, freq=1.0, alpha=0.3)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_default}, score={best_score:.6f}")

    print(f"Starting: n={len(best_seq)}, score={best_score:.6f}")

    solver = get_solver()

    # === Phase 1: Quick multi-init LSE-MD exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 1: Multi-init exploration ===")
        n_cur = len(best_seq)
        inits = {
            'parent': norm_sum1(best_seq),
            'cos1': cosine_init(n_cur, 1.0, 0.3),
            'cos3': cosine_init(n_cur, 3.0, 0.5),
            'rand1': random_smooth_init(n_cur, 42, 5),
            'rand2': random_smooth_init(n_cur, 123, 3),
            'sin2': norm_sum1(np.sin(np.pi * np.arange(n_cur) / n_cur) ** 2),
        }
        t_per_init = min(remaining * 0.12, 25.0)
        for name, init in inits.items():
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            a_out, _ = lse_mirror_descent(init, t_per_init, lam0=5.0, verbose=False, rng_seed=hash(name) % 10000)
            try_update(a_out, f"init_{name}")

    # === Phase 2: Main LSE-MD (longer run, higher lambda) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: Main LSE-MD n={n_cur} ===")
        t2 = min(remaining * 0.35, 500.0)
        a_out, _ = lse_mirror_descent(norm_sum1(best_seq), t2, lam0=2.0, lam_mult=1.06, lam_max=500.0, verbose=True, rng_seed=77)
        try_update(a_out, "lse_main")

    # === Phase 3: LP-FW refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: LP-FW n={n_cur} ===")
        t3 = min(remaining * 0.25, 350.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t3, solver, verbose=True)
        try_update(a_out, "lp_fw")

    # === Phase 4: LSE-MD after LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: LSE-MD round 2 n={n_cur} ===")
        t4 = min(remaining * 0.25, 300.0)
        a_out, _ = lse_mirror_descent(norm_sum1(best_seq), t4, lam0=20.0, lam_mult=1.05, lam_max=1000.0, verbose=True, rng_seed=314)
        try_update(a_out, "lse_r2")

    # === Phase 5: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 5: LP-FW round 2 n={n_cur} ===")
        t5 = min(remaining * 0.3, 250.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t5, solver, verbose=True)
        try_update(a_out, "lp_fw_r2")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_best = len(best_seq)
        print(f"\n=== Phase 6: Multi-length exploration ===")
        test_ns = [n_best + 200, n_best + 400, n_best - 200, n_best + 600]
        for nt in test_ns:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50 or nt < 400 or nt > 4000:
                continue
            print(f"  Trying n={nt}...")
            a_inter = interpolate_sequence(best_seq, nt)
            t_per = min(remaining * 0.2, 80.0)
            a_out, _ = lse_mirror_descent(norm_sum1(a_inter), t_per, lam0=5.0, verbose=False, rng_seed=nt % 10000)
            try_update(a_out, f"multin_lse_n{nt}")

    # === Phase 7: Final LP-FW polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Final LP-FW n={n_best} ===")
        t7 = min(remaining * 0.6, 200.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t7, solver, verbose=True)
        try_update(a_out, "lp_final")

    # === Phase 8: Final LSE-MD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Final LSE-MD n={n_best} ===")
        t8 = min(remaining * 0.7, 150.0)
        a_out, _ = lse_mirror_descent(norm_sum1(best_seq), t8, lam0=50.0, lam_mult=1.03, lam_max=2000.0, verbose=False, rng_seed=999)
        try_update(a_out, "lse_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence! Returning default.")
        return [1.0]

    return [float(x) for x in best_seq]
