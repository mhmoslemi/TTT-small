If we add a small constant c to all n+k positions (including the original n):
- New sum = sum_a + (n+k)*c
- New conv values increase by terms involving c
- It's not clear this helps.

What about interpolating to a longer length? If we interpolate (upsample) the sequence:
- The "effective" n increases
- The convolution structure changes
- It might or might not help

I think it's better to focus on optimizing at a fixed n rather than trying to change n. The parent is n=2400, so let me try n=2400 primarily, with maybe one alternative length.

Actually, let me reconsider. The score is 2n * max_b / (sum_a)^2. For a given "shape" of the sequence (normalized to sum 1), the score is 2n * max(conv(p,p)). If we can find a sequence where max(conv(p,p)) = c/n for some constant c < 1.5, then any n gives score = 2c. The key is the "shape" - how well the autocorrelation is flattened.

For very large n, we can approach the theoretical limit (which is > 1 based on our earlier analysis). The question is what's achievable at n=2400.

Let me just focus on n=2400 (the parent length) and optimize hard.

def fast_autocorr(a):
    """Compute linear convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def spectral_flattening(a0, n_iters=200, verbose=False):
    """Gerchberg-Saxton style spectral flattening for initialization.
    Uses |A| flattening as a proxy for flat autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0)
    fft_size = 1 << ((2 * n - 1) - 1).bit_length()
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    for it in range(n_iters):
        # Zero-pad and FFT
        a_pad = np.zeros(fft_size)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        
        # Target: flat magnitude
        target_mag = np.sqrt(np.sum(a**2) * fft_size / len(np.fft.rfftfreq(fft_size)))
        # Actually, we want |A|^2 to be flat, so |A| = sqrt(sum(a^2) * fft_size) / sqrt(n_freqs)
        # By Parseval: sum |A[k]|^2 = fft_size * sum(a_pad[i]^2) = fft_size * sum(a[i]^2)
        # Number of freq bins: fft_size//2 + 1
        n_freqs = fft_size // 2 + 1
        target_mag = np.sqrt(fft_size * np.sum(a**2) / n_freqs)
        
        # Replace magnitude, keep phase
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        
        # But DC component should be sum(a)
        A_new[0] = abs(A_new[0])  # keep DC positive
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check improvement
        m = fast_autocorr(a_new).max()
        if m < best_m:
            best_m = m
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 50 == 0:
            print(f"    GS iter={it}: max_conv={m:.8f}")
    
    return best_a, best_m

def nesterov_mirror_descent(a0, time_limit, initial_temp=0.01, final_temp=0.001,
                            c0=3.0, verbose=True, rng_seed=42):
    """Nesterov accelerated mirror descent with log-sum-exp subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    max_stall = 5000
    # Track running averages with different windows
    avg_short = a.copy()
    avg_short_count = 1
    avg_long = a.copy()
    avg_long_count = 1
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Exponential temperature decay
        frac = 1.0 - np.exp(-t / 3000.0)
        temp = final_temp + (initial_temp - final_temp) * (1.0 - frac)
        temp = max(temp, 1e-5)
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation of y
        ac = fast_autocorr(y)
        
        # Compute subgradient using log-sum-exp
        log_ac = (ac - ac.max()) / temp
        weights = np.exp(log_ac)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            # Fallback: only use the argmax
            k_max = int(np.argmax(ac))
            weights = np.zeros_like(ac)
            weights[k_max] = 1.0
        else:
            weights /= w_sum
        
        # Compute gradient
        grad = np.zeros(n)
        for k in range(len(ac)):
            w = weights[k]
            if w < 1e-12:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * w * y[j_idx]
        
        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Update best
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Running averages
        avg_short += x
        avg_short_count += 1
        avg_long += x
        avg_long_count += 1
        
        if avg_short_count % 200 == 0:
            avg = avg_short / avg_short_count
            avg = np.maximum(avg, 0.0)
            s_avg = avg.sum()
            if s_avg > 1e-15:
                avg = avg / s_avg
                M_avg = fast_autocorr(avg).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg.copy()
        
        if avg_long_count % 1000 == 0:
            avg = avg_long / avg_long_count
            avg = np.maximum(avg, 0.0)
            s_avg = avg.sum()
            if s_avg > 1e-15:
                avg = avg / s_avg
                M_avg = fast_autocorr(avg).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg.copy()
        
        # Periodic restart
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(2000, max_stall // 2)
            pert = rng.normal(0, 0.003, n)
            x_r = best_a * (1.0 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            gamma = max(1.0, gamma * 0.7)
            # Reset short average
            avg_short = x_r.copy()
            avg_short_count = 1
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.6f}, temp={temp:.5f}, stall={stall_count}")
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, max_constraints=500, verbose=True):
    """LP Frank-Wolfe with constraint warm-starting."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    
    for t in range(1, 5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Select constraints: use a larger set for tighter bound
        # Use all lags where ac[k] > 0.85 * M_curr
        threshold = M_curr * 0.85
        active_lags = np.where(ac >= threshold)[0]
        if len(active_lags) > max_constraints:
            # Keep the top ones
            active_lags = np.argsort(ac)[::-1][:max_constraints]
        
        # Build LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = []
        
        # Add constraint: sum(b) = 1 (or we maximize sum(b) with conv constraint)
        # Actually, let's use the formulation: max sum(b) s.t. T@b <= M_curr, b >= 0
        T = np.zeros((len(active_lags), n))
        for row_idx, k in enumerate(active_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]
        
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Frank-Wolfe update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        y = y_new
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}, n_act={len(active_lags)}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=True, rng_seed=777):
    """Coordinate descent with smart candidate selection."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 10
    sweep = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        sweep += 1
        sweep_improved = False
        
        # Process coordinates in random order
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            a_i = a[i]
            if a_i < 1e-12:
                continue
            
            # Try moving mass to the "least helpful" position
            # Heuristic: move to where it reduces the peak most
            n_tries = min(6, n)
            candidates = rng.integers(0, n, size=n_tries)
            
            best_trial_m = best_m
            best_move = None
            
            for dst in candidates:
                if dst == i:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a_i * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[dst] += amt
                    if a_trial[i] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_trial_m - 1e-16:
                        best_trial_m = m_trial
                        best_move = (dst, amt)
            
            if best_move is not None:
                dst, amt = best_move
                a[i] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_trial_m
                best_a = a.copy()
                sweep_improved = True
        
        if sweep_improved:
            stall = 0
        else:
            stall += 1
            # Escape: random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD sweep={sweep} ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def binary_sequence_init(n, rng):
    """Create a random binary-like initialization."""
    # Random 0/1 sequence with ~30% ones
    a = (rng.random(n) < 0.3).astype(float)
    if a.sum() < 1:
        a = (rng.random(n) < 0.5).astype(float)
    return a / a.sum()

def barker_like_init(n, rng):
    """Create a Barker-inspired initialization using structured patterns."""
    # Use a multi-scale pattern
    a = np.zeros(n)
    # Add some structured components
    for scale in [3, 7, 11, 13, 17, 23]:
        pattern = np.sign(np.cos(2 * np.pi * rng.integers(1, scale) * np.arange(n) / n))
        a += pattern * (1.0 / scale)
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a / a.sum()
