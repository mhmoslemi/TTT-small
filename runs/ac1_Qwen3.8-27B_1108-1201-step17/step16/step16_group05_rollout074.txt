```
def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0: return np.array([0.0])
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]
```

```
def project_simplex(v):
    v=np.asarray(v,dtype=np.float64)
    n=len(v)
    if n==0: return v
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    rho=np.nonzero(u * np.arange(1,n+1) > (cssv - 1.0))[0]
    if len(rho)==0:
       return np.ones(n)/n
    rho=rho[-1]
    theta=(cssv[rho]-1.0)/(rho+1.0)
```

```
def fast_conv(a):
```

```
def peak_gradient(a, top_k=120, peak_sharp=3.0):
    a=np.asarray(a,dtype=np.float64)
    n=a.size
    c=fast_conv(a)
    k_eff=min(max(int(top_k),1), c.size)
    if k_eff >= c.size:
       idx=np.arange(c.size)
       vals=c.astype(np.float64)
    else:
       idx=np.argsort(c)[::-1][:k_eff]
       vals=c[idx]
    vals=vals.astype(np.float64)
    span=vals[0]-vals[-1]
    if span > 1e-15:
       z=(vals - vals[-1]) / span
       weights=np.exp(peak_sharp * (1.0 - z))
    else:
       weights=np.ones(k_eff, dtype=np.float64)
    weights=weights / weights.sum()
    v=np.zeros(n, dtype=np.float64)
    for wi,k in zip(weights, idx):

       end=min(n, k + 1)
       if start >= end: continue
       ii=np.arange(start,end)
       jj=k - ii
       v[ii] += 2.0 * float(wi) * a[jj]
    return v, float(c.max())
```

```
def interpolate_sequence(f, new_n):
    f=np.asarray(f,dtype=np.float64).ravel()
    old_n=f.size
    if old_n==0: return np.ones(new_n)/new_n
    s=f.sum()
    if s < 1e-15: return np.ones(new_n)/new_n
    f=f/s
    old_x=np.linspace(0.0,1.0,old_n)
    new_x=np.linspace(0.0,1.0,new_n)
    out=np.interp(new_x, old_x, f)
    out=np.maximum(out,0.0)
    ss=out.sum()
    if ss < 1e-15: return np.ones(new_n)/new_n
    return out/ss
```

```python
"""
Optimize a non-negative sequence a to minimize
    2 * n * max(convolve(a, a)) / (sum(a))^2.

Strategy:
- Start from the supplied previous construction when available.
- Use FFT-based exact linear convolution for fast objective evaluations.
- Alternate among three complementary local optimizers:
  1. dense projected gradient with line search on a weighted-peak gradient;
  2. Frank-Wolfe-like simplex moves toward low linearized peak-gradient points;
  3. targeted donor/destination mass swaps derived from the peak gradient.
- Periodically explore nearby sequence lengths and structured boundary-skewed
  initializations, then return the best officially evaluated sequence.
"""

import numpy as np
import time


def fast_conv(a):
    """Fast full linear convolution a * a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def approx_score(a):
    """Fast objective using FFT convolution."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    s = a.sum()
    if s < 1e-15 or not np.isfinite(s):
        return float("inf")
    c = fast_conv(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))


def evaluate_safe(a):
    """Official evaluation wrapper with error handling."""
    try:
        arr = normalize_sum1(a)
        if arr.size == 0:
            return float("inf")
        sc = evaluate_sequence([float(x) for x in arr])
        return float(sc)
    except Exception:

def consider_candidate(candidate, best_seq, best_score, label, start_time):
    """Update best state if candidate appears to improve the official score."""
    if candidate is None:
        return best_seq, best_score
    a = normalize_sum1(candidate)
    if a.size == 0:
        return best_seq, best_score

    approx = approx_score(a)
    if (not np.isfinite(best_score)) or approx < best_score - 1e-10:
        sc = evaluate_safe(a)
        if np.isfinite(sc) and (not np.isfinite(best_score) or sc < best_score - 1e-12):
            best_seq = a.copy()
            best_score = sc
            elapsed = time.monotonic() - start_time
            print(f"[{label}] new best official = {sc:.7f}, n={a.size}, t={elapsed:.1f}s")
    return best_seq, best_score

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n, dtype=np.float64) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def peak_gradient(a, top_k=120, peak_sharp=3.0):
    """
    Gradient of a softly weighted maximum of the convolution.

    Peaks are selected near the maximum and weighted by their relative height.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    c = fast_conv(a)
    k_eff = min(max(int(top_k), 1), c.size)

    if k_eff >= c.size:
        idx = np.arange(c.size)
        vals = c.astype(np.float64)
    else:
        idx = np.argsort(c)[::-1][:k_eff]
        vals = c[idx].astype(np.float64)

    span = vals[0] - vals[-1]
    if span > 1e-15:
        z = (vals - vals[-1]) / span
        weights = np.exp(peak_sharp * (1.0 - z))
    else:
        weights = np.ones(k_eff, dtype=np.float64)

    weights = weights / weights.sum()
    v = np.zeros(n, dtype=np.float64)

    for wi, k in zip(weights, idx):
        k = int(k)
        start = max(0, k - n + 1)
        end = min(n, k + 1)
        if start >= end:
            continue
        ii = np.arange(start, end)
        jj = k - ii
        v[ii] += 2.0 * float(wi) * a[jj]

    return v, float(c.max())


def peak_direction(a, top_k=120, peak_sharp=3.0, coord_sharp=15.0):
    """Simplex direction toward coordinates with low peak-gradient."""
    a = np.asarray(a, dtype=np.float64)
    v, m = peak_gradient(a, top_k, peak_sharp)
    n = a.size
    span = v.max() - v.min()
    if span > 1e-15:
        z = (v - v.min()) / span
        b = np.exp(-coord_sharp * z)
    else:
        b = np.ones(n, dtype=np.float64)
    b = b / b.sum()
    return b, m


def plateau_frank_wolfe(a0, time_limit, top_k=120, peak_sharp=3.0,
                        coord_sharp=15.0, max_stall=500,
                        verbose=False, seed=1):
    """
    Frank-Wolfe-like optimizer.

    At each step, a direction b is a softmax over low linearized peak gradient.
    Then a line search is done on the simplex from x toward b.
    """
    if time_limit <= 0.0:
        a = normalize_sum1(a0)
        return a, float(fast_conv(a).max())

    a = normalize_sum1(a0)
    n = a.size
    best = a.copy()
    best_m = float(fast_conv(a).max())

    x = a.copy()
    x_prev = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    cur_top_k = int(top_k)
    cur_peak_sharp = float(peak_sharp)
    cur_coord_sharp = float(coord_sharp)
    cur_max_stall = int(max_stall)

    base_alphas = (0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5)

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)

        b, _ = peak_direction(y, cur_top_k, cur_peak_sharp, cur_coord_sharp)

        # If the direction is nearly degenerate, perturb it.
        if np.linalg.norm(b - x) < 1e-5:
            r = rng.random(n)
            b = project_simplex(0.9 * x + 0.1 * r)

        m_x = float(fast_conv(x).max())
        best_trial = x
        best_trial_m = m_x

        trial_alphas = list(base_alphas)
        trial_alphas.append(2.0 / (t + 2.0))

        for al in trial_alphas:
            if al <= 0.0 or al > 1.0:
                continue
            trial = (1.0 - al) * x + al * b
            if trial.min() < -1e-12:
                trial = np.maximum(trial, 0.0)
                s = trial.sum()
                if s < 1e-15:
                    continue
                trial = trial / s
            m = float(fast_conv(trial).max())
            if m < best_trial_m - 1e-15:
                best_trial_m = m
                best_trial = trial

        x_prev = x
        x = best_trial
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15 or not np.isfinite(s):
            x = best.copy()
            x_prev = best.copy()
        else:
            x = x / s

        if best_trial_m < best_m - 1e-15:
            best_m = best_trial_m
            best = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > cur_max_stall:
            stall = 0
            cur_max_stall = int(cur_max_stall * 1.25)
            r = rng.random(n)
            x = normalize_sum1(0.8 * best + 0.2 * r)
            x_prev = x.copy()
            cur_top_k = min(cur_top_k + 30, 400)
            cur_coord_sharp = min(cur_coord_sharp * 1.25, 60.0)
            cur_peak_sharp = float(rng.uniform(1.5, 6.0))
            if verbose:
                print(f"    plateau restart t={t}, approx={2.0 * n * best_m:.7f}")

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    plateau t={t}, t={elapsed:.1f}s, approx={2.0 * n * best_m:.7f}")

    return best, float(best_m)


def gradient_line_search(a0, time_limit, top_k=150, peak_sharp=2.5,
                         verbose=False, seed=2):
    """
    Dense projected ascent/descent with line search on a weighted-peak gradient.
    """
    if time_limit <= 0.0:
        a = normalize_sum1(a0)
        return a, float(fast_conv(a).max())

    a = normalize_sum1(a0)
    n = a.size
    best = a.copy()
    best_m = float(fast_conv(a).max())

    x = a.copy()
    x_prev = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    cur_top_k = int(top_k)
    cur_peak_sharp = float(peak_sharp)
    cur_max_stall = int(max(600, time_limit * 3))

    factors = (0.01, 0.03, 0.1, 0.3, 1.0, 3.0, 10.0)

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)

        g, _ = peak_gradient(y, cur_top_k, cur_peak_sharp)

        m_x = float(fast_conv(x).max())
        m_y = float(fast_conv(y).max())

        if m_x <= m_y:
            best_trial = x
            best_trial_m = m_x
        else:
            best_trial = y
            best_trial_m = m_y

        gmax = float(np.max(np.abs(g))) if g.size else 0.0
        if gmax > 1e-12:
            for f in factors:
                lam = f / gmax
                trial = project_simplex(y - lam * g)
                m = float(fast_conv(trial).max())
                if m < best_trial_m - 1e-15:
                    best_trial_m = m
                    best_trial = trial
        else:
            trial = normalize_sum1(y + rng.normal(0.0, 1e-4, n))
            m = float(fast_conv(trial).max())
            if m < best_trial_m - 1e-15:
                best_trial_m = m
                best_trial = trial

        x_prev = x
        x = best_trial
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15 or not np.isfinite(s):
            x = best.copy()
            x_prev = best.copy()
        else:
            x = x / s

        if best_trial_m < best_m - 1e-15:
            best_m = best_trial_m
            best = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > cur_max_stall:
            stall = 0
            cur_max_stall = int(cur_max_stall * 1.25)
            r = rng.random(n)
            x = normalize_sum1(0.8 * best + 0.2 * r)
            x_prev = x.copy()
            cur_top_k = min(cur_top_k + 40, 500)
            cur_peak_sharp = float(rng.uniform(1.0, 5.0))
            if verbose:
                print(f"    grad restart t={t}, approx={2.0 * n * best_m:.7f}")

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    grad t={t}, t={elapsed:.1f}s, approx={2.0 * n * best_m:.7f}")

    return best, float(best_m)


def swap_polish(a0, time_limit, top_k=50, peak_sharp=3.5, verbose=False, seed=3):
    """
    Coordinate / mass-swap polish.

    Move mass from coordinates with high peak-gradient to coordinates with
    low peak-gradient, checked by exact FFT objective.
    """
    if time_limit <= 0.0:
        a = normalize_sum1(a0)
        return a, float(fast_conv(a).max())

    a = normalize_sum1(a0)
    n = a.size
    best = a.copy()
    best_m = float(fast_conv(a).max())

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 35
    cur_top_k = int(top_k)
    cur_peak_sharp = float(peak_sharp)
    fracs = (0.01, 0.03, 0.1, 0.3)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        v, _ = peak_gradient(a, cur_top_k, cur_peak_sharp)

        donor_scores = np.full(n, -1e30, dtype=np.float64)
        mask = a > 1e-8
        donor_scores[mask] = v[mask]

        donors = np.argsort(donor_scores)[::-1][:8]
        dests = np.argsort(v)[:8]

        improved = False

        for i in donors:
            i = int(i)
            if a[i] < 1e-9:
                continue

            for j in dests:
                j = int(j)
                if i == j:
                    continue

                for frac in fracs:
                    amt = a[i] * frac
                    if amt < 1e-18:
                        continue

                    trial = a.copy()
                    trial[i] -= amt
                    trial[j] += amt
                    trial = np.maximum(trial, 0.0)
                    s = trial.sum()
                    if s < 1e-15:
                        continue
                    trial = trial / s

                    m = float(fast_conv(trial).max())
                    if m < best_m - 1e-16:
                        a = trial
                        best_m = m
                        best = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1

            # Small random escaper move.
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j and a[i] > 1e-7:
                amt = float(a[i] * rng.uniform(0.005, 0.05))
                trial = a.copy()
                trial[i] -= amt
                trial[j] += amt
                trial = np.maximum(trial, 0.0)
                s = trial.sum()
                if s > 1e-15:
                    trial = trial / s
                    m = float(fast_conv(trial).max())
                    if m < best_m:
                        a = trial
                        best_m = m
                        best = a.copy()
                        stall = 0

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    swap t={elapsed:.1f}s, approx={2.0 * n * best_m:.7f}, stall={stall}")

    return best, float(best_m)


def interpolate_sequence(f, new_n):
    """Interpolate a non-negative sequence to a new length and renormalize."""
    f = np.asarray(f, dtype=np.float64).ravel()
    old_n = f.size
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / max(1, int(new_n))
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / max(1, int(new_n))
    f = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, f)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / max(1, int(new_n))
    return out / ss


def make_structured_candidates(n, reference):
    """
    Generate diverse non-negative initializations.

    The most useful shapes are skewed toward one boundary, because the
    nonlinear objective is based on convolution, not symmetric autocorrelation.
    """
    cands = []
    if n <= 0:
        return cands

    x = np.linspace(0.0, 1.0, n)
    eps = 1.0 / (n + 1.0)

    # Power-law singularities near boundaries, with several regularization levels.
    for alpha in (0.35, 0.45, 0.5, 0.55, 0.65):
        left = np.power(x + eps, -alpha)
        right = np.power(1.0 - x + eps, -alpha)
        cands.append(normalize_sum1(left))
        cands.append(normalize_sum1(right))
        for w in (0.3, 0.5, 0.7):
            mixed = w * left + (1.0 - w) * right
            cands.append(normalize_sum1(mixed))

    # Exponential boundary layers.
    for gamma in (1.0, 3.0, 10.0, 30.0):
        cands.append(normalize_sum1(np.exp(-gamma * x)))
        cands.append(normalize_sum1(np.exp(-gamma * (1.0 - x))))

    # Polynomial ramps.
    for p in (0.5, 1.0, 2.0):
        cands.append(normalize_sum1(x ** p))
        cands.append(normalize_sum1((1.0 - x) ** p))

    # Mildly symmetric / U-shaped candidates.
    cands.append(normalize_sum1(1.0 + 5.0 * (2.0 * x - 1.0) ** 2))
    cands.append(normalize_sum1((1.0 - (2.0 * x - 1.0) ** 2) ** 2))

    # Reference sequence, if available.
    if reference is not None:
        ref = np.asarray(reference, dtype=np.float64).ravel()
        if ref.size > 0 and ref.sum() > 1e-15:
            cands.append(interpolate_sequence(ref, n))

    return cands


def top_candidates_by_score(cands, k):
    """Pick k lowest approximate-score candidates without using sorting on arrays."""
    k = max(0, int(k))
    if k == 0:
        return []
    top = []
    for c in cands:
        if c is None:
            continue
        c = normalize_sum1(c)
        sc = approx_score(c)
        if not np.isfinite(sc):
            sc = float("inf")

        if len(top) < k:
            top.append((sc, c))
        else:
            worst = 0
            for i in range(1, len(top)):
                if top[i][0] > top[worst][0]:
                    worst = i
            if sc < top[worst][0]:
                top[worst] = (sc, c)
    return top

                elif n_cur > 2800:
                    n_try = max(2400, n_cur - 200)
                else:
                    n_try = 3000

                if n_try != n_cur and 800 <= n_try <= 4000:
                    a_init = interpolate_sequence(best_seq, n_try)
                    t_block = min(150.0, rem * 0.35)
                    if t_block > 20.0:
                        a_out, _ = plateau_frank_wolfe(
                            a_init,
                            t_block,
                            top_k=120,
                            peak_sharp=3.0,
                            coord_sharp=15.0,
                            max_stall=600,
                            verbose=False,
                            seed=seed_base
                        )
                        best_seq, best_score = consider_candidate(
                            a_out, best_seq, best_score, f"length_{n_try}", t_global
                        )

            rem2 = deadline - time.monotonic()
            if rem2 > 80.0:
                cand_n = int(best_seq.size)
                ref = parent if parent is not None else best_seq
                cands = make_structured_candidates(cand_n, ref)
                top = top_candidates_by_score(cands, 2)

                for sc, c in top:
                    rem3 = deadline - time.monotonic()
                    if rem3 < 40.0:
                        break
                    if np.isfinite(best_score) and sc > best_score + 1e-5:
                        continue

                    t_block = min(60.0, rem3 * 0.45)
                    if t_block > 15.0:
                        a_out, _ = plateau_frank_wolfe(
                            c,
                            t_block,
                            top_k=100,
                            peak_sharp=3.0,
                            coord_sharp=15.0,
                            max_stall=400,
                            verbose=False,
                            seed=seed_base + 7
                        )
                        best_seq, best_score = consider_candidate(
                            a_out, best_seq, best_score, f"struct_{cand_n}", t_global
                        )

        cycle += 1
        if cycle % 4 == 0:
            elapsed = time.monotonic() - t_global
            print(f"cycle={cycle}, elapsed={elapsed:.1f}s, official={best_score:.7f}, n={best_seq.size}")

    # Final polish with whatever time remains.
    rem = deadline - time.monotonic()
    if rem > 30.0:
        a_out, _ = plateau_frank_wolfe(
            best_seq,
            min(rem * 0.7, 120.0),
            top_k=160,
            peak_sharp=2.5,
            coord_sharp=20.0,
            max_stall=1000,
            verbose=False,
            seed=99991
        )
        best_seq, best_score = consider_candidate(
            a_out, best_seq, best_score, "final_plateau", t_global
        )

    rem = deadline - time.monotonic()
    if rem > 15.0:
        a_out, _ = swap_polish(
            best_seq,
            min(rem * 0.8, 60.0),
            top_k=40,
            peak_sharp=3.0,
            verbose=False,
            seed=99997
        )
        best_seq, best_score = consider_candidate(
            a_out, best_seq, best_score, "final_swap", t_global
        )

    # Final official validation and sanitization.
    if best_seq is None:
        best_seq = np.ones(2400, dtype=np.float64) / 2400

    final_seq = normalize_sum1(best_seq)
    final_score = evaluate_safe(final_seq)

    if np.isfinite(final_score):
        best_seq = final_seq
        best_score = final_score
    else:
        # Extremely defensive fallback.
        best_seq = np.ones(2400, dtype=np.float64) / 2400
        best_score = evaluate_safe(best_seq)

    elapsed = time.monotonic() - t_global
    print(f"FINAL: official={best_score:.7f}, n={best_seq.size}, elapsed={elapsed:.1f}s")

    return [float(x) for x in best_seq]
```
