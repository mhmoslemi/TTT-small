def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_to(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def active_subgradient(a, ac, threshold_frac):
    """Compute subgradient from lags near the maximum."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n), 1
    active = np.where(ac >= threshold_frac * M)[0]
    if len(active) == 0:
        active = np.array([int(ac.argmax())])
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    gn = np.linalg.norm(g)
    if gn > 1e-15:
        g = g / gn
    return g, len(active)


def mirror_update(y, g, eta):
    """Mirror descent update in log-space, projected to simplex."""
    log_a = np.log(np.maximum(y, 1e-18)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s

def lp_direction(a, ac, top_k=80, M_target=None):
    """
    Solve LP to get Frank-Wolfe direction.
    max sum(b) s.t. conv(a, b)[k] <= M_target for top-k lags, b >= 0.
    Returns normalized direction or None on failure.
    """
    if not _HAVE_CP:
        return None
    solver = get_lpsolver()
    if solver is None:
        return None
    n = len(a)
    if M_target is None:
        M_target = ac.max()
    if M_target < 1e-30:
        return None

    top_lags = np.argsort(ac)[::-1][:min(top_k, len(ac))]

    b_var = cp.Variable(n, nonneg=True)
    constraints = []

    for k in top_lags:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        coeff = np.zeros(n)
        for ii, j in zip(idx, j_idx):
            coeff[j] += a[ii]
        constraints.append(coeff @ b_var <= M_target)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    direction = b_val / b_sum

    # Normalize to unit L2 for use as subgradient-like direction
    ln = np.linalg.norm(direction)
    if ln > 1e-15:
        direction = direction / ln
    return direction

def make_initialization(n, variant, rng):
    """Create diverse initializations."""
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    if variant == 0:
        # Sinusoidal
        a = np.sin(np.pi * x) ** 2
    elif variant == 1:
        # Linear ramp
        a = x + 0.1
    elif variant == 2:
        # Exponential decay
        a = np.exp(-3.0 * x)
    elif variant == 3:
        # Bathtub
        a = (x - 0.5) ** 2 + 0.1
    elif variant == 4:
        # Gaussian off-center
        a = np.exp(-30.0 * (x - 0.3) ** 2) + 0.05
    elif variant == 5:
        # Double peak
        a = np.exp(-40.0 * (x - 0.25) ** 2) + np.exp(-40.0 * (x - 0.75) ** 2) + 0.02
    elif variant == 6:
        # Cosine
        a = 0.5 + 0.4 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif variant == 7:
        # Power law
        a = x ** 2.0 + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s

    # === Phase 5: LP-based Frank-Wolfe polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if _HAVE_CP and n_best <= 2500:
        print(f"\n=== Phase 5: LP polish n={n_best} (remaining={remaining:.0f}s) ===")
        t5 = min(remaining * 0.12, 150.0)
        if t5 > 20:
            a_out, _ = lp_polish(norm_sum1(best_seq), t5, verbose=True, max_iters=min(50, int(t5/2)))
            try_update(a_out, f"lp_polish_n{n_best}")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 6: Multi-length (remaining={remaining:.0f}s) ===")
        for delta in [300, -300, 600, -600]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            n_try = n_best + delta
            if n_try < 800 or n_try > 3500:
                continue
            a_long = interpolate_to(best_seq, n_try)
            t6 = min(remaining * 0.25, 120.0)
            if t6 > 20:
                a_out, _ = amd_adaptive(norm_sum1(a_long), t6, c0=3.0,
                                         init_thresh=0.995, lo_thresh=0.985,
                                         lp_interval=500, verbose=True, rng_seed=10+delta)
                try_update(a_out, f"amd_n{n_try}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord n={n_best} (remaining={remaining:.0f}s) ===")
    if remaining > 20:
        t7 = min(remaining * 0.30, 150.0)
        if t7 > 10:
            a_out, _ = coord_descent_optimal(norm_sum1(best_seq), t7, verbose=True, rng_seed=7)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 8: Final AMD sub-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 8: AMD sub-polish n={n_best} (remaining={remaining:.0f}s) ===")
        t8 = min(remaining * 0.35, 100.0)
        if t8 > 10:
            a_out, _ = amd_adaptive(norm_sum1(best_seq), t8, c0=1.5,
                                     init_thresh=0.9995, lo_thresh=0.998,
                                     lp_interval=200, verbose=False, rng_seed=8)
            try_update(a_out, f"amd_subn{n_best}")

    # === Phase 9: Final LP if feasible ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if _HAVE_CP and n_best <= 2000 and remaining > 20:
        print(f"\n=== Phase 9: Final LP n={n_best} (remaining={remaining:.0f}s) ===")
        t9 = min(remaining * 0.40, 60.0)
        if t9 > 10:
            a_out, _ = lp_polish(norm_sum1(best_seq), t9, verbose=True, max_iters=min(30, int(t9/2)))
            try_update(a_out, f"lp_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence! Returning fallback.")
        n_fb = 2400
        x_fb = np.linspace(0, 1, n_fb, endpoint=False)
        best_seq = np.sin(np.pi * x_fb) ** 2 + 0.1
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()

    return [float(x) for x in best_seq]
