"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for a non-negative sequence a.

Approach (different from previous SOTA):
  - Unified accelerated mirror (Nesterov) descent in log-space with a
    VECTORIZED soft top-k subgradient whose (k, beta) is annealed from a
    "wide" view (many lags, low power) to a "narrow" view (few lags, high power).
  - Multi-strategy restarts to escape local minima: Gaussian, anti-peak,
    random-mass-swap, partial reversal, and a "dual-projected" re-seed.
  - A surgical peak-aware coordinate descent that does tiny mass transfers
    specifically aimed at the top few autocorrelation lags.
  - Length exploration around the parent length (interpolation + re-optimise),
    since the achievable bound is length-dependent.
  - Strict wall-clock accounting so the best finite sequence is always returned.
All helper functions are top-level; no closures, no lambdas.
"""

import numpy as np
import time


def fast_autocorr(a):
    """Linear autocorrelation conv(a,a) via FFT, shape (2n-1,)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    # Clean tiny negative round-off
    out[out < 0.0] = 0.0
    return out


def norm_sum1(a):
    """Non-negative normalisation to sum 1 (robust)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Resample a normalised non-negative sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return np.asarray(f, dtype=np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def gradient_topk_vec(a, ac, top_k, beta):
    """
    Soft subgradient of max-style objective using top_k lags weighted by
    value**beta. Vectorised per lag. Returns gradient w.r.t. a (not divided
    by anything; caller sets step size).
    """
    n = len(a)
    k_max = min(int(top_k), len(ac))
    order = np.argsort(ac)[::-1][:k_max]
    vals = ac[order]
    if vals[0] < 1e-30:
        return np.zeros(n)
    w = vals ** beta
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / ws
    g = np.zeros(n)
    for wi, k in zip(w, order):
        if wi < 1e-12:
            continue
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        jidx = k - idx
        g[idx] += (2.0 * wi) * a[jidx]
    return g


def gradient_active_vec(a, ac, threshold_frac):
    """Exact active-set subgradient (lags within threshold_frac of max)."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    mask = ac >= threshold_frac * M
    idxs = np.where(mask)[0]
    if len(idxs) == 0:
        idxs = np.array([int(np.argmax(ac))])
    g = np.zeros(n)
    for k in idxs:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        jidx = k - idx
        g[idx] += 2.0 * a[jidx]
    return g


def objective_normalized(a):
    """Return (max_autocorr, J) for a sum-1 non-negative a. J = 2*n*max."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    M = fast_autocorr(a).max()
    return M, 2.0 * n * M


def restart_gauss(best_a, rng, scale=0.008):
    noise = rng.normal(0.0, scale, len(best_a)) * best_a
    x = np.maximum(best_a + noise, 1e-10)
    return x / x.sum()


def restart_anti_peak(best_a, ac, rng, frac=0.015):
    n = len(best_a)
    peak_k = np.argsort(ac)[::-1][:6]
    red = np.zeros(n)
    for k in peak_k:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        jidx = k - idx
        red[idx] += best_a[jidx]
        red[jidx] += best_a[idx]
    m = red.max()
    if m > 1e-20:
        x = best_a - frac * red / m
        x = np.maximum(x, 1e-10)
        x = x / x.sum()
        # combine with a light gaussian jitter
        x = x + rng.normal(0.0, 0.002, n) * x
        x = np.maximum(x, 1e-10)
        return x / x.sum()
    return best_a.copy()


def restart_swap(best_a, rng, n_swaps=None):
    n = len(best_a)
    if n_swaps is None:
        n_swaps = max(5, n // 40)
    x = best_a.copy()
    for _ in range(n_swaps):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j or x[i] < 1e-10:
            continue
        amt = x[i] * rng.uniform(0.05, 0.20)
        x[i] -= amt
        x[j] += amt
    x = np.maximum(x, 1e-10)
    return x / x.sum()


def restart_partial_reversal(best_a, rng):
    n = len(best_a)
    lo = n // 5
    hi = 4 * n // 5
    split = int(rng.integers(lo, max(lo + 1, hi)))
    x = best_a.copy()
    x[split:] = best_a[split:][::-1]
    x = np.maximum(x, 1e-10)
    return x / x.sum()


def choose_restart(best_a, ac, rng, counter):
    strat = int(counter % 5)
    if strat == 0:
        return restart_gauss(best_a, rng, scale=0.006 + 0.010 * (counter % 3) / 2.0)
    if strat == 1:
        return restart_anti_peak(best_a, ac, rng)
    if strat == 2:
        return restart_swap(best_a, rng)
    if strat == 3:
        return restart_partial_reversal(best_a, rng)
    # 4: mix of two
    a1 = restart_gauss(best_a, rng, 0.010)
    a2 = restart_swap(best_a, rng)
    x = 0.5 * a1 + 0.5 * a2
    x = np.maximum(x, 1e-10)
    return x / x.sum()


def amd_phase(a0, time_limit, mode, verbose=True, rng_seed=42,
              c0=3.0, max_stall=2500):
    """
    Accelerated mirror descent. mode in {'wide','narrow','adaptive'}.
    Returns (best_a, best_M).
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_M = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    restart_counter = 0

    # per-mode anneal target
    if mode == 'wide':
        k_start, k_end = 60.0, 15.0
        b_start, b_end = 2.0, 6.0
    elif mode == 'narrow':
        k_start, k_end = 8.0, 3.0
        b_start, b_end = 15.0, 40.0
    else:  # adaptive: wide -> narrow
        k_start, k_end = 40.0, 4.0
        b_start, b_end = 3.0, 25.0

    total_steps = max_stall * 4 + 20000  # rough horizon for anneal

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # anneal hyperparameters
        frac = min(t / float(total_steps), 1.0)
        k_cur = k_start + (k_end - k_start) * frac
        b_cur = b_start + (b_end - b_start) * frac

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = gradient_topk_vec(y, ac, top_k=max(3, int(round(k_cur))), beta=b_cur)
        gn = np.linalg.norm(g)
        if gn < 1e-20:
            g = gradient_active_vec(y, ac, 0.999)
            gn = np.linalg.norm(g)

        eta = c0 / (np.sqrt(t) + 1.0) * np.sqrt(n)
        if gn > 1e-20:
            eta = eta / gn

        log_y = np.log(y)
        log_new = log_y - eta * g
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-16:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall >= max_stall:
            ac_best = fast_autocorr(best_a)
            x = choose_restart(best_a, ac_best, rng, restart_counter)
            restart_counter += 1
            x_prev = x
            stall = 0
            c0 *= 1.05
            if verbose:
                el = time.monotonic() - t0
                print("    AMD-restart n=%d (%ds) strat=%d c=%.2f"
                      % (n, int(el), restart_counter, c0))

        if verbose and t % 4000 == 0:
            el = time.monotonic() - t0
            print("    AMD t=%d (%ds) J=%.6f k=%.1f b=%.1f stall=%d"
                  % (t, int(el), 2.0 * n * best_M, k_cur, b_cur, stall))

    return best_a, best_M


def surgical_coord(a0, time_limit, verbose=True, rng_seed=777, max_stall=150):
    """
    Peak-aware surgical coordinate descent: tiny mass transfers aimed at the
    top few autocorrelation lags, moving mass out of positions that feed the
    peaks toward low-sensitivity positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_M = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        peak_k = np.argsort(ac)[::-1][:8]

        # peak contribution map (how strongly reducing a[i] cuts the peaks)
        peak_contrib = np.zeros(n)
        for k in peak_k[:6]:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            jidx = k - idx
            peak_contrib[idx] += a[jidx]
            peak_contrib[jidx] += a[idx]

        # cold positions: low peak contribution, small current value
        peak_contrib_sorted = np.argsort(peak_contrib)
        cold = peak_contrib_sorted[:max(64, n // 4)]

        improved = False
        # only consider the most peak-driving source positions
        src_pool = np.argsort(peak_contrib)[::-1][:max(64, n // 3)]

        for src in src_pool:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-9:
                continue
            # choose a few cold destinations, prefer small-value ones
            cands = cold[np.argsort(a[cold])[:40]]
            for dst in cands:
                if dst == src or a[dst] > a[src] * 1.5:
                    continue
                for fr in (0.0015, 0.004, 0.010, 0.02):
                    amt = a[src] * fr
                    if amt < 1e-20:
                        continue
                    at = a.copy()
                    at[src] -= amt
                    at[dst] += amt
                    at = np.maximum(at, 0.0)
                    st = at.sum()
                    if st < 1e-15:
                        continue
                    at = at / st
                    Mt = fast_autocorr(at).max()
                    if Mt < best_M - 1e-17:
                        a = at
                        best_M = Mt
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            stall = 0
            continue

        stall += 1
        # escape: a few random transfers accepting only improvements
        esc_ok = False
        for _ in range(max(3, n // 80)):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-9:
                continue
            amt = a[i] * rng.uniform(0.01, 0.06)
            a[i] -= amt
            a[j] += amt
            a = np.maximum(a, 0.0)
            st = a.sum()
            if st < 1e-15:
                continue
            a = a / st
            Mt = fast_autocorr(a).max()
            if Mt < best_M:
                best_M = Mt
                best_a = a.copy()
                stall = 0
                esc_ok = True
                break
            else:
                a[i] += amt
                a[j] -= amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
        if esc_ok:
            continue

        if verbose and stall % 40 == 0:
            el = time.monotonic() - t0
            print("    Surg n=%d (%ds) J=%.6f stall=%d"
                  % (n, int(el), 2.0 * n * best_M, stall))

    return best_a, best_M


def _consider(a, best_seq, best_score, label, t_global):
    """Try to register a candidate as the new best. Returns (best_seq,best_score)."""
    if a is None:
        return best_seq, best_score
    try:
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    except Exception:
        return best_seq, best_score
    s = a.sum()
    if not np.isfinite(s) or s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        sc = float(evaluate_sequence([float(v) for v in a_norm]))
    except Exception:
        return best_seq, best_score
    if np.isfinite(sc) and sc < best_score:
        el = time.monotonic() - t_global
        print("  *** NEW BEST [%s]: %.6f (n=%d, t=%ds)"
              % (label, sc, len(a_norm), int(el)))
        return a_norm.copy(), sc
    return best_seq, best_score

        if tF > 15:
            out, _ = surgical_coord(norm_sum1(best_seq), tF,
                                    verbose=True, rng_seed=606, max_stall=120)
            best_seq, best_score = _consider(out, best_seq, best_score, "surgF", t_global)

    # ---- Phase G: last AMD micro-polish with whatever time is left ----
    r = remaining()
    if r > 20:
        tG = min(r * 0.60, 90.0)
        if tG > 10:
            out, _ = amd_phase(norm_sum1(best_seq), tG, 'narrow',
                               verbose=False, rng_seed=707, c0=1.5,
                               max_stall=1200)
            best_seq, best_score = _consider(out, best_seq, best_score, "amdG", t_global)

    # ---- final validation ----
    if best_seq is None:
        print("ERROR: no valid sequence")
        return [1.0]
    try:
        fs = float(evaluate_sequence([float(v) for v in best_seq]))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass
    print("FINAL: score=%.6f n=%d time=%ds"
          % (best_score, len(best_seq), int(time.monotonic() - t_global)))
    return [float(v) for v in best_seq]
