def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def objective_value(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)

def compute_soft_grad(a, ac, top_k, beta):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad

def compute_logsumexp_grad(a, ac, top_k, temp):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    vals = ac[top_idx] / temp
    max_v = vals.max()
    shifted = vals - max_v
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)

        if i_start > i_end:

        idx = np.arange(i_start, i_end + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad

def smooth_obj_value(a, top_k, temp):
    ac = fast_autocorr(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac_norm = ac / (s * s)
    top_idx = np.argsort(ac_norm)[::-1][:top_k]
    vals = ac_norm[top_idx] / temp
    max_v = vals.max()
    lse = max_v + np.log(np.sum(np.exp(vals - max_v)))

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_ramp(n, power=1.0, direction='left'):
    if direction == 'left':
        x = np.arange(n, 0, -1, dtype=np.float64)
    else:
        x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)

def make_exp_decay(n, rate=1.0, direction='left'):
    if direction == 'left':
        x = np.arange(n - 1, -1, -1, dtype=np.float64)
    else:
        x = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * x / n)
    return norm_sum1(a)

def make_bimodal(n, c1=0.15, c2=0.85, w=0.12):
    x = np.linspace(0, 1, n)
    a = np.exp(-(x - c1)**2 / (2*w**2)) + np.exp(-(x - c2)**2 / (2*w**2)) + 0.01
    return norm_sum1(a)

def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in [1, 2, 3, 4, 5, 7, 11, 13]:
        coeff = rng.normal(0, 0.07)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def make_af(n, strength=0.3):
    """Anti-symmetric: high on left, low on right. Exploits center lag = sum a[j]*a[n-1-j]."""
    x = np.linspace(0, 1, n)
    a = 1.0 + strength * np.cos(np.pi * x)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def make_af_quad(n, strength=0.5):
    x = np.linspace(0, 1, n)
    a = 1.0 + strength * (2*x - 1)
    a = np.maximum(a, 0.05)
    return norm_sum1(a)

        a_out, _ = coord_2d(a_in, t4, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_best}")

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    lengths_to_try = [n_best + 400, n_best - 400, n_best + 800]
    lengths_to_try = [l for l in lengths_to_try if 500 <= l <= 5000]
    lengths_to_try = list(dict.fromkeys(lengths_to_try))[:3]

    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n=== Phase 5: n={new_n} ===")
        a_long = interpolate_seq(best_seq, new_n)
        t5a = min(remaining * 0.25, 120.0)
        if t5a > 20:
            a_out, _ = gradient_descent(a_long, t5a, top_k_init=60, beta_init=5.0,
                                         use_lse=True, c0=3.0, restart_every=5000, verbose=False)
            try_update(a_out, f"gd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 3500:
            t5b = min(remaining * 0.20, 100.0)
            if t5b > 20:
                a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t5b, solver, verbose=False, top_K=200)
                try_update(a_out, f"lp_n{new_n}")

    # Phase 6: Final LP
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 3500:
        t6 = min(remaining * 0.30, 150.0)
        if t6 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True, top_K=400)
            try_update(a_out, f"lp_final_n{n_best}")

    # Phase 7: Final coordinate + gradient
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Polish (n={n_best}) ===")
    if remaining > 30:
        t7a = min(remaining * 0.4, 80.0)
        if t7a > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_2d(a_in, t7a, verbose=False, rng_seed=999)
            try_update(a_out, f"coord_final_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.5, 60.0)
            if t7b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = gradient_descent(a_in, t7b, top_k_init=100, beta_init=3.0,
                                             use_lse=True, c0=2.0, restart_every=6000, verbose=False)
                try_update(a_out, f"gd_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 2400
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x)**2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
