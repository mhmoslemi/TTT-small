def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def project_simplex(v, target=1.0):
    n = len(v)
    if n == 0:
        return v
    mu = np.zeros(n)
    sv = np.sort(v)[::-1]
    csum = np.cumsum(sv)
    rho = np.nonzero(sv * np.arange(1, n + 1) > (csum - target))[0][-1]
    mu = (csum[rho] - target) / (rho + 1.0)
    return np.maximum(v - mu, 0.0)


def fista_smooth(a0, time_limit, beta_start=5.0, beta_end=80.0, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    t_k = 1.0
    step = 1.0
    total_iters = max(1, int(time_limit))

    for it in range(max(total_iters, 10000)):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (it + 50.0))
        beta = beta_start * (beta_end / beta_start) ** progress

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x + ((t_k - 1.0) / t_new) * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        c = fast_autocorr(y)
        bc = c - c.max()
        exp_c = np.exp(beta * bc)
        p = exp_c / exp_c.sum()

        L = 1 << (3 * n - 2 - 1).bit_length()
        P = np.fft.rfft(p, L)
        A = np.fft.rfft(y, L)
        corr = np.fft.irfft(P * np.conj(A), L)[:n]
        grad = 2.0 * corr

        for _ in range(10):
            z = y - step * grad
            x_new = project_simplex(z, 1.0)
            c_new = fast_autocorr(x_new)
            b_new = c_new.max()
            b_cur = c.max()
            if b_new <= b_cur + step * (0.5 * step * np.dot(grad, grad) + 1e-12):
                break
            step *= 0.5

        x_prev = x
        x = x_new
        t_k = t_new

        m_val = fast_autocorr(x).max()
        if m_val < best_m - 1e-15:
            best_m = m_val
            best_a = x.copy()

        if verbose and it % 2000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA it={it} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}, beta={beta:.1f}", flush=True)

    return best_a, best_m

            vals.extend([vj] * n)

            y = a.copy()
            xs = a.copy()
            continue

        T = sp.csc_matrix((vals, (rows, cols)), shape=(num_rows, n))
        T = T.tocsr()

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            xs_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            xs_new = y_new + beta_t * (y_new - y)
            xs_new = np.maximum(xs_new, 0.0)

            else:

        x_prev = xs
        xs = xs_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(xs).max()
        if mx < best_m:
            best_m = mx
            best_a = xs.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW it={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}", flush=True)

    return best_a, best_m


def peak_targeted_cd(a0, time_limit, verbose=True, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        peak_idx = int(np.argmax(ac))

        i_start = max(0, peak_idx - n + 1)
        i_end = min(n - 1, peak_idx)
        if i_start >= i_end:
            stall += 1
            continue

        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[peak_idx - idx]
        if contrib.max() < 1e-20:
            stall += 1
            continue

        top_peaks = np.argsort(ac)[::-1][:10]
        sensitivity = np.zeros(n)
        for pk in top_peaks:
            pk_int = int(pk)
            for pos in range(n):
                d = pk_int - pos
                if 0 <= d < n:
                    sensitivity[pos] += a[d]
        sensitivity = np.maximum(sensitivity, 1e-15)

        improved = False
        src_ranking = np.argsort(contrib)[::-1]
        for src_rank in src_ranking[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[src_rank])
            if a[src] < 1e-10:
                continue

            candidates = np.argsort(sensitivity)[:min(100, n // 5)]
            for dst in candidates:
                if dst == src:
                    continue
                for frac in [0.003, 0.008, 0.015, 0.03, 0.06, 0.12, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-20:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_a = a.sum()
                    if s_a > 1e-15:
                        a = a / s_a
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = max(0, stall - 1)
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CD ({elapsed:.1f}s): J={J:.6f}", flush=True)

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss
