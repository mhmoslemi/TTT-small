def fast_autocorr(a):
    """Compute autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def compute_subgradient(a, ac, top_k=5):
    """Compute subgradient using top-k lags with equal weight."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    w = np.ones(k_max) / k_max
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def compute_soft_subgradient(a, ac, temperature):
    """Compute soft subgradient using log-sum-exp weighting."""
    n = len(a)
    log_ac = np.log(np.maximum(ac, 1e-30) / temperature)
    log_ac -= log_ac.max()
    weights = np.exp(log_ac)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        top_idx = np.argsort(ac)[::-1][:min(10, len(ac))]
        weights = np.zeros_like(ac)
        weights[top_idx] = 1.0
        w_sum = weights.sum()
    weights /= w_sum
    
    grad = np.zeros(n)
    for k in range(len(ac)):
        w = weights[k]

def fast_autocorr(a):
    """Compute autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def compute_soft_gradient(a, ac, tau, top_k=50):
    """Compute gradient of logsumexp approximation using top-k lags."""
    n = len(a)
    b_max = ac.max()
    shifted = (ac - b_max) / tau
    # Compute weights only for top_k lags
    top_idx = np.argsort(ac)[::-1][:top_k]
    top_shifted = shifted[top_idx]
    weights = np.exp(top_shifted - top_shifted.max())
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights /= w_sum
    grad = np.zeros(n)
    for k, w in zip(top_idx, weights):
        k = int(k)
        if w < 1e-12:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        grad[idx] += 2.0 * w * a[j_idx]
    return grad

def compute_hard_subgradient(a, ac, top_k=5):
    """Compute subgradient using top-k lags with equal weight (hard max)."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    w = np.ones(k_max) / k_max
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def nmd_backtracking(a0, time_limit, top_k=5, c0=3.0, verbose=True, seed=42):
    """Nesterov mirror descent with backtracking line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    max_stall = 3000

        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        if t < 500 or (t % 1000 < 500):
            g = compute_hard_subgradient(y, ac, top_k)
        else:
            g = compute_soft_gradient(y, ac, tau=ac.max() * 0.01, top_k=20)
        # Backtracking line search
        eta_t = eta
        best_a_new = y.copy()
        best_obj = ac.max()
        for bs in range(15):
            log_a = np.log(y) - eta_t * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:

            obj_new = fast_autocorr(a_new).max()
            if obj_new <= best_obj - 1e-15 * max(1.0, best_obj):
                best_obj = obj_new
                best_a_new = a_new.copy()
                break
            eta_t *= 0.5
            if eta_t < 1e-12:
                break
        x_prev = x
        x = best_a_new
        # Adaptive step size growth

            eta = min(eta * 1.02, 100.0)
        else:
            eta = max(eta * 0.95, 0.5)
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:

        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(500, max_stall // 2)
