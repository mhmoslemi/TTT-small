import numpy as np
import cvxpy as cp
import time

# Global solver cache
_solver_name = None

def _get_solver():
    global _solver_name
    if _solver_name is not None:
        return _solver_name
    for s in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_name = s

        except Exception:
            continue
    _solver_name = cp.SCS
    return _solver_name


def _fast_autocorr(a):
    """Linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _eval_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def _normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _grad_from_weights(w, a):
    """Compute gradient g[j] = 2*sum_m a[m]*w[j+m] using correlate."""
    return 2.0 * np.correlate(w, a, mode='valid')


def _mirror_step(a, g, lr):
    """Mirror descent step in log domain."""
    log_a = np.log(np.maximum(a, 1e-30))
    log_new = log_a - lr * g
    mx = log_new.max()
    a_new = np.exp(log_new - mx)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _spectral_equalize(a, n_iter=20, eps=1e-8):
    """Flatten power spectrum."""
    a = _normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = np.abs(A)**2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.5 * a + 0.5 * a_new
    return a


def _primal_dual(a0, time_limit, decay=0.97, power=4.0, lr0=1.0, 
                 top_frac=0.1, verbose=False, tag=""):
    """
    Entropy-regularized primal-dual method.
    
    Maintains weight vector w over lags, updated to emphasize peaks.
    Uses mirror descent (exponential family) for the primal update.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    w = np.ones(2 * n - 1) * 0.1  # initialize with small uniform weights
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    k_top = max(5, int(top_frac * (2 * n - 1)))
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = _fast_autocorr(a)
        M = c.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
        
        # Update weights: emphasize top peaks
        # Use rank-based weighting
        sorted_idx = np.argsort(c)[::-1]
        top_k = sorted_idx[:k_top]
        
        # Decay old weights
        w *= decay
        # Boost top-k peaks
        for idx in top_k:
            w[idx] += 1.0
        
        # Also add a smooth "heat" based on peak ratio
        peak_ratio = c / (M + 1e-30)
        w += 0.1 * (peak_ratio ** power)
        
        # Normalize weights
        w_max = w.max()
        if w_max > 1e-10:
            w = w / w_max
        
        # Compute gradient
        g = _grad_from_weights(w, a)
        
        # Adaptive learning rate
        lr = lr0 / (1.0 + 0.005 * t)
        
        # Mirror descent
        a = _mirror_step(a, g, lr)
        a = np.maximum(a, 1e-15)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PD[{tag}] t={t} ({elapsed:.1f}s): J={J:.6f}, M={M:.6e}")
    
    return best_a, best_m


def _lp_fw_step(a, solver, top_k=80):
    """
    Single LP Frank-Wolfe step.
    Returns new point or None on failure.
    """
    n = len(a)
    c = _fast_autocorr(a)
    M = c.max()
    
    # Select top-k active lags
    active = np.argsort(c)[::-1][:top_k]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(b_var) == 1.0]
    
    for k in active:
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = a[ki - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def _lp_fw(a0, time_limit, solver, top_k=80, verbose=False, tag=""):
    """Accelerated LP Frank-Wolfe."""
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov
        if t == 1:
            y = x.copy()
        else:
            beta = (t - 1.0) / (t + 1.5)
            y = x + beta * (x - x_prev)
            y = np.maximum(y, 1e-15)
            y = y / y.sum()
        
        c = _fast_autocorr(y)
        M = c.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = y.copy()
        
        # LP direction
        direction = _lp_fw_step(y, solver, top_k)
        if direction is None:
            continue
        
        alpha = 2.0 / (t + 2.0)
        x_new = (1.0 - alpha) * y + alpha * direction
        x_new = np.maximum(x_new, 0.0)
        x_new = x_new / max(x_new.sum(), 1e-15)
        
        x_prev = y
        x = x_new
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            mx = _fast_autocorr(x).max()
            if mx < best_m:
                best_m = mx
                best_a = x.copy()
            J = 2.0 * n * best_m
            print(f"    LP-FW[{tag}] t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def _coord_descent(a0, time_limit, verbose=False, tag="", rng_seed=42):
    """
    Peak-shaving coordinate descent.
    Tries to reduce the top peaks by moving mass from contributing
    positions to less sensitive positions.
    """
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = _fast_autocorr(a)
        M = c.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
        
        # Focus on top peaks
        top_peaks = np.argsort(c)[::-1][:15]
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            idx = np.arange(j_start, j_end + 1)
            contrib = a[idx] * a[ki - idx]
            
            if contrib.max() < 1e-18:
                continue
            
            # Find the most contributing pair
            best_pair_idx = np.argmax(contrib)
            src = idx[best_pair_idx]
            partner = ki - src
            
            # Try moving mass from src to a position that doesn't affect top peaks
            # Score each candidate destination
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:8]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2+1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)
            
            # Sample a subset of candidates
            n_sample = min(100, len(candidates))
            chosen = candidates[rng.choice(len(candidates), n_sample, replace=False)]
            
            best_dst = -1
            best_delta = 0
            
            for dst in chosen:
                if dst == src or dst == partner:
                    continue
                # Estimate impact: how much would c[top_peaks] increase?
                sens = 0.0
                for k2 in top_peaks[:5]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    d2 = dst - k2i
                    if 0 <= d1 < n:
                        sens += a[d1]
                    if 0 <= d2 < n and d2 != d1:
                        sens += a[d2]
                expected_reduction = contrib[best_pair_idx]
                if expected_reduction - 0.1 * sens > best_delta:
                    best_delta = expected_reduction - 0.1 * sens
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            # Try different fractions
            for frac in [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2]:
                amt = a[src] * frac
                if amt < 1e-15:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                if a_trial[src] < -1e-12:
                    break
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    break
                a_trial = a_trial / st
                M_trial = _fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st

                    if M_new < best_m:

                        best_a = a.copy()
                        stall = 0
                    else:

                        a[j] -= amt

            else:
                a = best_a.copy()

def _interpolate(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_n = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_n)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss
