    Gradient of f_beta(a) = (1/beta) * log(sum_k exp(beta * (a*a)[k])) w.r.t. a.
    Result: 2 * sum_k w_k * a[k-i] for each i, where w_k proportional to exp(beta*(a*a)[k]).
    Computed via FFT cross-correlation.
    """
    ac = fast_conv(a)
    max_ac = ac.max()
    log_w = beta * (ac - max_ac)
    log_w -= log_w.max()
    w = np.exp(log_w)
    w /= w.sum()
    
    n = len(a)
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_sz = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, fft_sz)
    Ar = np.fft.rfft(a_rev, fft_sz)
    conv = np.fft.irfft(W * Ar, fft_sz)[:conv_len]
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad

def simplex_project(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.ones(1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_idx = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordPolish ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1 normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with adaptive time allocation and multi-strategy approach."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 100.0, 1700.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2 + 0.1
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Smooth optimization from parent (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: SmoothOpt n={n0} (budget remaining: {remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        a_in = np.maximum(best_seq, 1e-15)
        a_in = a_in / a_in.sum()
        a_out, _ = smooth_optimize(a_in, t1, beta_start=5.0, beta_end=8000.0,
                                   lr0=0.08, verbose=True, seed=42)
        try_update(a_out, f"smooth_n{n0}_s42")

    # === Phase 2: Second smooth pass with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: SmoothOpt pass 2 n={len(best_seq)} ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_in = np.maximum(best_seq, 1e-15)
        a_in = a_in / a_in.sum()
        a_out, _ = smooth_optimize(a_in, t2, beta_start=50.0, beta_end=20000.0,
                                   lr0=0.03, verbose=True, seed=7, max_iters=150000)
        try_update(a_out, f"smooth_n{len(best_seq)}_s7")

    # === Phase 3: LP Frank-Wolfe polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LP Frank-Wolfe n={n_cur} ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 60:
        a_in = np.maximum(best_seq, 0.0)
        a_in = a_in / a_in.sum()
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True)
        try_update(a_out, f"lp_n{n_cur}")

    # === Phase 4: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: CoordPolish n={n_cur} ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_in = np.maximum(best_seq, 0.0)
        a_in = a_in / a_in.sum()
        a_out, _ = coordinate_polish(a_in, t4, verbose=True, seed=42)
        try_update(a_out, f"coord_n{n_cur}")

    # === Phase 5: Try different sequence lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 200:
        for delta in [200, -200, 400, -400, 600]:
            n_try = n_cur + delta
            if n_try < 800 or n_try > 6000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            print(f"\n=== Phase 5: Length n={n_try} (remaining: {remaining:.0f}s) ===")
            a_long = interpolate_sequence(best_seq, n_try)
            t5 = min(remaining * 0.35, 200.0)
            if t5 > 30:
                a_out, _ = smooth_optimize(a_long, t5, beta_start=10.0, beta_end=5000.0,
                                           lr0=0.05, verbose=False, seed=100 + n_try % 100)
                try_update(a_out, f"smooth_n{n_try}")
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 30:
                    t5b = min(remaining * 0.3, 80.0)
                    if t5b > 15:
                        a_out2, _ = coordinate_polish(a_out, t5b, verbose=False, seed=200)
                        try_update(a_out2, f"coord_n{n_try}")

    # === Phase 6: Final LP pass on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if n_cur <= 3000 and remaining > 80:
        print(f"\n=== Phase 6: Final LP n={n_cur} (remaining: {remaining:.0f}s) ===")
        t6 = min(remaining * 0.5, 200.0)
        if t6 > 30:
            a_in = np.maximum(best_seq, 0.0)
            a_in = a_in / a_in.sum()
            a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True)
            try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final Coord n={n_cur} (remaining: {remaining:.0f}s) ===")
    if remaining > 20:
        t7 = min(remaining * 0.8, 100.0)
        if t7 > 10:
            a_in = np.maximum(best_seq, 0.0)
            a_in = a_in / a_in.sum()
            a_out, _ = coordinate_polish(a_in, t7, verbose=True, seed=314)
            try_update(a_out, f"coord_final_n{n_cur}")

    # === Phase 8: One more smooth pass if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 8: Final Smooth n={n_cur} (remaining: {remaining:.0f}s) ===")
        t8 = min(remaining * 0.7, 120.0)
        if t8 > 20:
            a_in = np.maximum(best_seq, 1e-15)
            a_in = a_in / a_in.sum()
            a_out, _ = smooth_optimize(a_in, t8, beta_start=200.0, beta_end=50000.0,
                                       lr0=0.01, verbose=True, seed=777, max_iters=100000)
            try_update(a_out, f"smooth_final_n{n_cur}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    # Ensure we return with good margin before timeout
    elapsed = time.monotonic() - t_global
    print(f"Returning after {elapsed:.1f}s (budget was {budget_s:.0f}s)")
    return [float(x) for x in best_seq]
