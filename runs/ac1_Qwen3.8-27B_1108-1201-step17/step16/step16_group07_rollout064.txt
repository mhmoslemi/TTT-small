"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent via log-sum-exp with FFT-based cross-correlation (primary, O(n log n)/step)
2. LP best-response with Nesterov acceleration (refinement)
3. Multi-peak subgradient with ergodic averaging
4. Coordinate descent (final polish)
5. Multi-restart with diverse initializations

Key innovation: Log-sum-exp smoothing of the max function enables fast gradient-based

Adaptive epsilon annealing focuses optimization from broad to narrow.

def get_solver():
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    if not HAS_CVXPY:
        return None
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):

                print(f"  LP solver: {s}")

        except Exception:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def smooth_gradient_fft(a, p, n, fft_size):
    """Compute g[j] = 2 * sum_{i=0}^{n-1} a[i] * p[j+i] via FFT cross-correlation."""
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)
    corr = irfft(A * P, fft_size)
    return 2.0 * corr[:n]

def normalize_to(a, n):
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s


def lp_best_response(a_fixed, M, active_set, solver):
    if solver is None:
        return None, None
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)
    if size == 0:
        return None, None
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def smooth_gradient_descent(a0, time_limit, verbose=False, seed=0):

    rng = np.random.default_rng(seed)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1).bit_length()
    v = np.zeros(n)
    t_iter = 0
    step = 1.0
    patience = 0
    max_patience = 300
    h = a.copy()

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            noise_scale = 0.03 + 0.04 * (patience / max_patience)
            noise = noise_scale * rng.standard_normal(n)
            a = best_a * (1.0 + noise)
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a * norm / s_a
            else:
                a = best_a.copy()
            v = np.zeros(n)
            step = 0.8
            patience = 0
            continue

        t_iter += 1
        conv = fast_autocorr(a)
        max_conv = conv.max()
        if max_conv < 1e-12:
            break

        progress = min(1.0, elapsed / (time_limit * 0.85))

        eps = max(eps, 1e-10)

        shifted = (conv - max_conv) / eps
        shifted = np.clip(shifted, -60, 60)
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = p / p_sum

        g = smooth_gradient_fft(a, p, n, fft_size)
        g_mean = g.mean()
        g_proj = g - g_mean

        momentum = min(0.88, 0.4 + 0.0004 * t_iter)
        v = momentum * v + (1.0 - momentum) * g_proj

        a_trial = a - step * v
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-15:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.008, 4.0)
        else:
            patience += 1
            step *= 0.975
            if step < 1e-8:
                break

        h = h * (t_iter / (t_iter + 1)) + a * (1.0 / (t_iter + 1))
        if t_iter % 3000 == 0:
            h_norm = normalize_to(h, n)
            score_h = fast_autocorr(h_norm).max()

            print(f"    SmoothGD iter={t_iter} ({elapsed:.0f}s): best={best_score:.10f}, eps={eps:.6f}, step={step:.4f}")

    return best_a, best_score


def subgradient_descent(a0, time_limit, verbose=False, seed=0):
    """Multi-peak subgradient with ergodic averaging."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.4 / n ** 0.3
    patience = 0
    max_patience = 500
    h = a.copy()
    t_iter = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        t_iter += 1
        conv = fast_autocorr(a)
        M_curr = conv.max()

        n_peaks = max(5, min(20, n // 40))
        thresh_factor = 0.85 + 0.1 * ((t_iter % 4) / 4.0)
        thresh = M_curr * thresh_factor
        active = np.where(conv >= thresh)[0]
        if len(active) == 0:
            active = np.argsort(conv)[-n_peaks:]
        elif len(active) > n_peaks * 5:
            order = np.argsort(conv[active])
            active = active[order[-n_peaks * 5:]]

        weights = np.exp(-12.0 * (M_curr - conv[active]) / max(M_curr * 0.04, 1e-10))
        weights = weights / weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, active):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                g[lo:hi] += w * 2.0 * a[k - np.arange(lo, hi)]

        g_mean = g.mean()
        g = g - g_mean
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn

        a_trial = a - step * g * n ** 0.3
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.003, 2.5)
        else:
            patience += 1
            step *= 0.985
            if step < 1e-10:
                break

        h = h * (t_iter / (t_iter + 1)) + a * (1.0 / (t_iter + 1))
        if t_iter % 2000 == 0:
            h_norm = normalize_to(h, n)
            score_h = fast_autocorr(h_norm).max()
            if score_h < best_score:
                best_score = score_h
                best_a = h_norm.copy()

        if verbose and t_iter % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={t_iter} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):

        y_prev = y
        y = y_new
        a = a_new

        h = h * (n_iters / (n_iters + 1)) + a * (1.0 / (n_iters + 1))

def coordinate_descent(a0, time_limit, verbose=False, seed=0):
    """Coordinate descent with candidate evaluation and FFT verification."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        conv = fast_autocorr(a)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            aj = a[j]
            lo = max(0, j)
            hi = min(size, j + n)
            if lo >= hi:
                continue

            typical = norm / n
            candidates = [0.0, aj * 0.4, aj * 0.7, aj * 1.0, aj * 1.3, aj * 1.7,
                          typical * 0.5, typical, typical * 1.5, typical * 2.5]
            candidates = sorted(set([max(0.0, min(c, norm * 0.25)) for c in candidates]))

            best_x = aj
            best_obj = conv.max()

            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                delta = x - aj
                new_conv = conv.copy()
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta
                new_max = new_conv.max()
                if new_max < best_obj - 1e-14:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                    else:
                        a = best_a.copy()

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def generate_initialization(n, method, rng):
    """Generate diverse initializations for multi-restart."""
    if method == 0:
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
    elif method == 1:
        x = np.linspace(0.0, 1.0, n)
        freq = rng.uniform(1.5, 3.5)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.35 * np.cos(2 * np.pi * freq * x + phase)
    elif method == 2:
        a = 0.5 + 0.25 * rng.standard_normal(n)
    elif method == 3:
        period = rng.integers(n // 8, n // 3)
        x = np.arange(n)
        a = 0.4 + 0.5 * np.cos(2 * np.pi * x / period)
    elif method == 4:
        a = 0.4 + 0.3 * ((-1) ** np.arange(n)) + 0.15 * rng.standard_normal(n)
    elif method == 5:
        x = np.arange(n) / n
        a = 0.3 + 0.6 * np.abs(np.sin(np.pi * x * rng.uniform(1, 4)))
    elif method == 6:
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.4 * np.cos(2 * np.pi * 2.5 * x) + 0.2 * np.cos(2 * np.pi * 7.3 * x + 0.5)
    elif method == 7:
        x = np.linspace(0, 1, n)
        a = 0.4 + 0.3 * np.cos(2 * np.pi * 1.3 * x) + 0.2 * np.cos(2 * np.pi * 5.7 * x + 1.1) + 0.1 * np.cos(2 * np.pi * 11.2 * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def update_best(best_seq, best_score, a_new, label, t_global):
    """Try to update the best sequence. Returns (best_seq, best_score, improved)."""
    if a_new is None:
        return best_seq, best_score, False
    a_new = np.maximum(np.asarray(a_new, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a_norm = a_new / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a_norm)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False

        a_norm = normalize_to(best_seq, n)
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True, seed=1)
        best_seq, best_score, _ = update_best(best_seq, best_score, a_out, f"lp1_n{n}", t_global)

    # === Phase 3: Smooth GD round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        print(f"\n=== Phase 3: Smooth GD r2 n={n}, budget={t3:.0f}s ===")
        a_norm = normalize_to(best_seq, n)
        a_out, _ = smooth_gradient_descent(a_norm, t3, verbose=True, seed=77)
        best_seq, best_score, _ = update_best(best_seq, best_score, a_out, f"smooth2_n{n}", t_global)

    # === Phase 4: Subgradient with ergodic averaging ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 20:
        print(f"\n=== Phase 4: SubGrad n={n}, budget={t4:.0f}s ===")
        a_norm = normalize_to(best_seq, n)
        a_out, _ = subgradient_descent(a_norm, t4, verbose=True, seed=3)
        best_seq, best_score, _ = update_best(best_seq, best_score, a_out, f"subg_n{n}", t_global)

    # === Phase 5: LP round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 30 and solver is not None:
        print(f"\n=== Phase 5: LP r2 n={n}, budget={t5:.0f}s ===")
        a_norm = normalize_to(best_seq, n)
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True, seed=999)
        best_seq, best_score, _ = update_best(best_seq, best_score, a_out, f"lp2_n{n}", t_global)

    # === Phase 6: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 15:
        print(f"\n=== Phase 6: CD n={n}, budget={t6:.0f}s ===")
        a_norm = normalize_to(best_seq, n)
        a_out, _ = coordinate_descent(a_norm, t6, verbose=True, seed=5)
        best_seq, best_score, _ = update_best(best_seq, best_score, a_out, f"cd_n{n}", t_global)

    # === Phase 7: Multi-restart with different initializations ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_restarts = 3
    t7_total = min(remaining * 0.25, 300.0)
    t7_each = t7_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        t7 = min(t7_each, remaining - 10)
        if t7 < 20:
            break
        print(f"\n=== Phase 7r{r}: Restart init={r+1}, budget={t7:.0f}s ===")
        n_cur = len(best_seq)
        init = generate_initialization(n_cur, r + 1, rng)
        # Blend with current best for a warm start
        blend = 0.3 * init + 0.7 * best_seq
        blend = np.maximum(blend, 0.0)
        blend = blend / blend.sum()
        a_norm = normalize_to(blend, n_cur)
        a_out, _ = smooth_gradient_descent(a_norm, t7, verbose=False, seed=100 + r)
        best_seq, best_score, _ = update_best(best_seq, best_score, a_out, f"restart{r}_n{n_cur}", t_global)

    # === Phase 8: Final CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final CD n={n_cur} ===")
        t8 = min(remaining * 0.6, 60.0)
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent(a_norm, t8, verbose=False, seed=77)
        best_seq, best_score, _ = update_best(best_seq, best_score, a_out, f"cdFinal_n{n_cur}", t_global)

    # Final subgradient micro-polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 20:
        t9 = min(remaining * 0.5, 30.0)
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = subgradient_descent(a_norm, t9, verbose=False, seed=314)
        best_seq, best_score, _ = update_best(best_seq, best_score, a_out, f"subgFinal_n{n_cur}", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
