def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def golden_section_max(f, a, b, tol=1e-6, max_iter=60):
    """Golden section search to maximize f on [a, b]."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
        else:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
    return (a + b) / 2

def fw_line_search(ac_a, ac_ag, ac_g):
    """
    Find alpha in [0,1] that minimizes max_k [(1-a)^2 * ac_a[k] + 2a(1-a) * ac_ag[k] + a^2 * ac_g[k]].
    Uses the fact that the objective is convex in alpha.
    """
    def obj(alpha):
        ac = (1 - alpha) ** 2 * ac_a + 2 * alpha * (1 - alpha) * ac_ag + alpha ** 2 * ac_g
        return ac.max()
    
    # Check endpoints and a few interior points
    best_alpha = 0.0
    best_val = obj(0.0)
    
    # Try golden section
    alpha_gs = golden_section_max(lambda x: -obj(x), 0.0, 1.0, tol=1e-8, max_iter=50)
    val_gs = obj(alpha_gs)
    if val_gs < best_val:
        best_alpha = alpha_gs
        best_val = val_gs
    
    # Also try some fixed points
    for test_alpha in [0.5, 0.3, 0.7, 0.1, 0.9]:
        v = obj(test_alpha)
        if v < best_val:
            best_val = v
            best_alpha = test_alpha
    
    return best_alpha, best_val

def fw_lp_direction(a, M, solver):
    """
    Solve the LP: maximize sum(b) subject to conv(a, b) <= M (elementwise), b >= 0.
    Returns the normalized direction b/sum(b).
    """
    n = len(a)
    num_lags = 2 * n - 1
    
    # Build sparse constraint matrix A where A[k, j] = a[k-j] if 0 <= k-j <= n-1
    rows = []
    cols = []
    vals = []
    
    # For each lag k (0 to 2n-2), the valid j values are max(0, k-n+1) to min(n-1, k)
    for k in range(num_lags):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    
    A_mat = cp.SparseMatrix((np.array(vals, dtype=np.float64), (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))), shape=(num_lags, n))
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    
    return b_val / b_sum

def soft_subgradient(a, ac, top_k=50, beta=6.0):
    """Soft subgradient using top-k lags with exponential weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = np.exp(beta * (top_vals / top_vals[0] - 1.0))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * wi * a[j_idx]
    return g

def mirror_update(y, g, eta):
    log_a = np.log(np.maximum(y, 1e-15)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Quick AMD warm-up to escape initial transients
2. Full-constraint LP-based Frank-Wolfe with golden-section line search (main workhorse)
3. AMD with soft subgradient for diversification and escaping local minima
4. Aggressive peak-aware coordinate descent for final polish
Key improvement: uses ALL lags in the LP for exact Frank-Wolfe directions,
with efficient O(n) line search via precomputed cross-correlations.
"""

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is None:
        for s in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS]:
            try:
                x = cp.Variable(4, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
                prob.solve(solver=s)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    _global_solver = s
                    print(f"  LP solver: {s}")
                    return s
            except Exception:
                continue
        _global_solver = cp.SCS
        print(f"  LP solver (fallback): SCS")
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_crosscorr(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def precompute_lp_indices(n):
    num_lags = 2 * n - 1
    rows = []
    cols = []
    for k in range(num_lags):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), num_lags

def build_lp_sparse(a, rows, cols, num_lags, n):
    k_vals = rows
    j_vals = cols
    i_vals = k_vals - j_vals
    vals = a[i_vals].astype(np.float64)
    return cp.SparseMatrix((vals, (rows, cols)), shape=(num_lags, n))

def solve_lp_direction(a, M, rows, cols, num_lags, n, solver, warm_start=None):
    n2 = n
    A_mat = build_lp_sparse(a, rows, cols, num_lags, n2)
    b_var = cp.Variable(n2, nonneg=True)
    if warm_start is not None:
        b_var.value = warm_start
    constraints = [A_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, 0.0
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, 0.0
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = float(b_val.sum())
    if b_sum < 1e-15:
        return None, 0.0
    return b_val / b_sum, b_sum

def fw_line_search(ac_a, ac_ag, ac_g):
    def obj(alpha):
        ac = (1.0 - alpha) ** 2 * ac_a + 2.0 * alpha * (1.0 - alpha) * ac_ag + alpha ** 2 * ac_g
        return ac.max()
    best_alpha = 0.0
    best_val = obj(0.0)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_lo = 0.0
    a_hi = 1.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = obj(c)
    fd = obj(d)
    for _ in range(50):
        if a_hi - a_lo < 1e-9:
            break
        if fc < fd:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = obj(d)
        else:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = obj(c)
    alpha_gs = (a_lo + a_hi) / 2.0
    val_gs = obj(alpha_gs)
    if val_gs < best_val:
        best_alpha = alpha_gs
        best_val = val_gs
    for test_alpha in [0.1, 0.3, 0.5, 0.7, 0.9]:
        v = obj(test_alpha)
        if v < best_val:
            best_val = v
            best_alpha = test_alpha
    return best_alpha, best_val

def soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = np.exp(beta * (top_vals / top_vals[0] - 1.0))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * wi * a[j_idx]
    return g

def mirror_update(y, g, eta):
    log_a = np.log(np.maximum(y, 1e-15)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s

def amd_soft(a0, time_limit, c0=2.5, top_k=50, beta=5.0, max_stall=400, verbose=True, rng_seed=7):

    x_prev = a.copy()
    stall_count = 0
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        ac = fast_autocorr(y)
        g = soft_subgradient(y, ac, top_k=top_k, beta=beta)
        eta = c0 / (np.sqrt(t) + 1.0) * np.sqrt(n)
        a_new = mirror_update(y, g, eta)
        x_prev = x
        x = a_new
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:

        if stall_count > max_stall:

            perturbation = rng.normal(0, 0.008, n) * best_a

            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD-soft t={t} ({elapsed:.0f}s): J={J:.6f}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coordinate descent n={n_best} ===")
    t5 = min(remaining * 0.25, 250.0)
    if t5 > 15:
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t5, verbose=True, rng_seed=5)
        try_update(a_out, f"coord_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 6: Final FW polish n={n_best} ===")
        t6 = min(remaining * 0.6, 200.0)
        if t6 > 20:
            a_out, _ = frank_wolfe_full(norm_sum1(best_seq), t6, solver, verbose=False, rng_seed=6)
            try_update(a_out, f"fw_final_n{n_best}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            print(f"\n=== Phase 7: Final coord micro-polish ===")
            t7 = min(remaining * 0.5, 100.0)
            if t7 > 10:
                a_out, _ = coordinate_descent(norm_sum1(best_seq), t7, verbose=False, rng_seed=7)
                try_update(a_out, f"coord_micro_n{n_best}")

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    return [float(x) for x in best_seq]
