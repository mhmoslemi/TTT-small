The key insight for improvement: instead of using M = max(conv(a,a)), we could use a slightly smaller M (like 0.99 * max) to force the LP to find a more diverse direction. Or we could use multiple M values and combine.

Another idea: instead of a single LP, solve multiple LPs with different "focus" lags and combine the results.

Let me also think about the "peak shaving" approach more carefully:

Given current a with peak at lag k*, the convolution at k* is sum_j a[j]*a[k*-j]. To reduce this:
- We can reduce a[j] for the j that contributes most to the peak
- But we need to add that mass somewhere else (to maintain sum=1)
- The best place to add it is where it minimally increases other convolution values

This is essentially a 2-point move: reduce a[i] and increase a[j]. The change in conv at lag k is:
- delta_conv[k] = -a[i]*a[k-i] - a[k-i]*a[i] + a[j]*a[k-j] + a[k-j]*a[j] (approximately, for small changes)

Actually for a unit transfer from position p to position q:
- New a[p] = a[p] - eps, a[q] = a[q] + eps
- Change in conv[k] = -eps*(a[k-p] + a[p-k]) + eps*(a[k-q] + a[q-k]) = eps * ((a[k-q] + a[q-k]) - (a[k-p] + a[p-k]))

So we want to find (p, q) such that for the peak lag k*:
(a[k*-q] + a[q-k*]) - (a[k*-p] + a[p-k*]) < 0

i.e., position q has smaller "contribution potential" at lag k* than position p.

And we also need to check that other lags don't increase too much.

This is a good local search strategy. Let me implement it efficiently.

For efficiency, I can precompute for each position i and each of the top-K lags, the "contribution value" c(i, k) = a[k-i] + a[i-k] (with out-of-range as 0). Then for a transfer from p to q, the change at lag k is eps * (c(q,k) - c(p,k)).

We want: for the peak lag k*, c(q,k*) < c(p,k*), and for all other important lags k, c(q,k) - c(p,k) should be small (ideally negative).

Score for destination q given source p: sum over top lags k of w(k) * (c(q,k) - c(p,k))
We want q that minimizes this score.

This can be done efficiently if we precompute the score for all q.

import numpy as np

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:

                _global_solver = s

        except Exception:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * np.conj(A), fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def get_soft_grad(a, ac, top_k=30, beta=4.0):
    """Compute soft subgradient: weighted combination of gradients at top lags."""
    n = len(a)
    num_lags = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:num_lags]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # grad[i] += 2 * wi * (a[k-i] if 0<=k-i<n else 0) + 2 * wi * (a[i-k] if 0<=i-k<n else 0)
        # First part: a[k-i] for i in [max(0,k-n+1), min(k, n-1)]
        i_lo = max(0, k_int - n + 1)
        i_hi = min(k_int, n - 1)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += 2.0 * wi * a[k_int - idx]
        # Second part: a[i-k] for i in [k, n-1]
        i_lo2 = max(k_int, 0)
        i_hi2 = n - 1
        if i_lo2 <= i_hi2:
            idx2 = np.arange(i_lo2, i_hi2 + 1)
            grad[idx2] += 2.0 * wi * a[idx2 - k_int]
    return grad

def amd_step(a0, time_limit, top_k=40, beta=4.0, c0=3.0, restart_every=5000, verbose=False):
    """Nesterov-accelerated mirror descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    cur_k = top_k
    cur_beta = beta

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = get_soft_grad(y, ac, top_k=cur_k, beta=cur_beta)

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.01, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_k = min(cur_k + 10, 80)
            cur_beta = max(cur_beta - 0.3, 2.0)

    return best_a, best_m

def lp_fw_step(a0, time_limit, solver, top_K=200, verbose=False):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Build constraint matrix for selected constraints
        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y

        margins = M_curr - T_full @ y
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]
        M_sel = np.full(K_sel, M_curr)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m

def peak_shave(a0, time_limit, verbose=False, rng_seed=42):
    """Targeted peak-shaving local search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        peak_k = int(top_k[0])

        # Precompute contribution scores for all positions
        # score[q] = sum over top lags k of (a[k-q] + a[q-k])_valid
        # This tells us how much adding mass at q would increase each peak lag
        scores = np.zeros(n)
        for k in top_k[:min(15, n_top)]:
            k_int = int(k)
            for i in range(n):
                s = 0.0
                j1 = k_int - i
                if 0 <= j1 < n:
                    s += a[j1]
                j2 = i - k_int
                if 0 <= j2 < n:
                    s += a[j2]
                scores[i] += s

        # Find the source: position in the peak lag with highest contribution
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        if js > je:
            stall += 1
            continue
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        src = int(idx[np.argmax(contrib)])

        # Find destination: lowest score, not the source
        scores[src] = np.inf
        dst = int(np.argmin(scores))
        if dst == src or a[src] < 1e-12:
            stall += 1
            continue

        # Try multiple fractions
        improved = False
        for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-16:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            # Random perturbation
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.002, 0.01)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
