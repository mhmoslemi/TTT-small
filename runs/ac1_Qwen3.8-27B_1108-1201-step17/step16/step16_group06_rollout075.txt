"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed Nesterov gradient descent (main method)
2. Multiplicative-weights Frank-Wolfe (secondary method)
3. Peak-targeted coordinate descent (final polish)
4. Multi-length exploration with interpolation

import numpy as np
import time

def fast_conv(a, b):
    """Compute full convolution using FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:n]

def fast_autocorr(a):
    """Compute conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).flatten()
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def logsumexp_gradient_descent(a0, time_limit=600, beta_start=20, beta_end=50000,

    a is normalized to sum 1.

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Nesterov state
    x = a.copy()
    x_prev = a.copy()
    y = a.copy()
    
    total_iters = 200000
    
    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal beta
        progress = min(1.0, (t - 1) / (total_iters * 0.9))
        if progress < 0.5:
            # Linear ramp up
            beta = beta_start + (beta_end * 0.1 - beta_start) * (progress / 0.5)
        else:
            # Exponential ramp to beta_end
            frac = (progress - 0.5) / 0.5
            beta = beta_end * 0.1 * (beta_end / (beta_end * 0.1)) ** frac
            beta = min(beta, beta_end)
        
        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            mom = (np.sqrt(1.0 + 4.0 * (t-1)**2) - 1.0) / (2.0 * (t-1))
            y = x + mom * (x - x_prev)
            y = np.maximum(y, 1e-30)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        
        # Compute autocorrelation of y
        ac = fast_autocorr(y)
        M = ac.max()
        if M < 1e-14:
            break
        
        # Compute softmax weights
        x_ac = ac / M  # normalize so max is 1
        # Use beta * x_ac to avoid overflow
        shifted = beta * (x_ac - 1.0)  # max is 0, so exp won't overflow
        weights = np.exp(shifted)

        if w_sum < 1e-30:
            weights = np.ones(len(ac)) / len(ac)
        else:
            weights = weights / w_sum

        Y = np.fft.rfft(y, fft_size)

        # convolution: ifft(FFT(w) * FFT(y))
        conv_part = np.fft.irfft(W * Y, fft_size)[:n]
        
        grad = cross_corr + conv_part
        
        # Project gradient onto simplex tangent space
        grad = grad - grad.mean()

def mw_frank_wolfe(a0, time_limit=600, verbose=True):
    """
    Frank-Wolfe with multiplicative weights inner solver.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    x_prev = a.copy()
    
    max_iters = 50000
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current M
        ac_y = fast_autocorr(y)
        M = ac_y.max()
        if M < 1e-14:
            break
        
        # Solve inner LP using multiplicative weights
        # max sum(b) s.t. (y*b)[i] <= M, b >= 0
        b = np.ones(n) / n
        n_mw = 100  # MW iterations per FW step
        
        for mw_t in range(n_mw):
            v = fast_conv(y, b)  # (y*b)[i], length 2n-1
            excess = np.maximum(v - M, 0.0)
            if excess.max() < 1e-12 * M:
                break
            
            # Compute pressure: p[j] = y[j] * sum_{i=j}^{min(j+n-1, 2n-2)} excess[i]
            cum = np.zeros(len(excess) + 1)
            cum[1:] = np.cumsum(excess)
            p = np.zeros(n)
            for j in range(n):
                i_end = min(j + n, len(excess))
                p[j] = y[j] * (cum[i_end] - cum[j])
            
            # Update b
            log_update = -np.clip(p / M, 0, 2.0)
            b = b * np.exp(log_update)
            b = np.maximum(b, 1e-30)
            s_b = b.sum()
            if s_b < 1e-15:
                b = np.ones(n) / n
            else:
                b = b / s_b
        
        # Nesterov-accelerated FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = b.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x_prev = x
        x = x_new
        
        # Check
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  MW-FW t={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed Nesterov mirror descent (main method, smooth gradient via FFT)
2. Multiplicative-weights Frank-Wolfe (secondary, many fast iterations)
3. Peak-targeted coordinate descent (final polish)
4. Multi-length exploration with interpolation
"""

import numpy as np
import time

def fast_conv(a, b):
    """Compute full convolution using FFT."""
    na, nb = len(a), len(b)
    n = na + nb - 1
    fft_size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:n]

def fast_autocorr(a):
    """Compute conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64).flatten()
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_lse_gradient(y, beta):
    """
    Compute gradient of (1/beta)*log(sum exp(beta*ac[i])) w.r.t. y.
    ac = conv(y,y), y normalized to sum 1.
    Returns gradient vector of length n.
    """
    n = len(y)
    ac = fast_autocorr(y)
    M = ac.max()
    if M < 1e-14:
        return np.zeros(n)
    
    # Softmax weights
    shifted = beta * (ac / M - 1.0)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(len(ac)) / len(ac)
    else:
        weights = weights / w_sum
    
    # Gradient: grad[j] = sum_i w[i]*(y[i-j] + y[j-i])
    # = cross_corr[j] + conv[j]
    # cross_corr[j] = sum_k y[k]*w[k+j] = ifft(FFT(y)*conj(FFT(w)))[j]
    # conv[j] = sum_k w[k]*y[j-k] = ifft(FFT(w)*FFT(y))[j]
    
    N = 3 * n - 1
    fft_size = 1 << (N - 1).bit_length()
    Y = np.fft.rfft(y, fft_size)
    W = np.fft.rfft(weights, fft_size)
    
    cross_corr = np.fft.irfft(Y * np.conj(W), fft_size)[:n]
    conv_part = np.fft.irfft(W * Y, fft_size)[:n]
    
    grad = cross_corr + conv_part
    return grad

def lse_nesterov_descent(a0, time_limit, beta_start, beta_end, verbose, total_iters):
    """
    Log-sum-exp smoothed Nesterov mirror descent on the simplex.
    Uses multiplicative (mirror) updates for stability.
    """
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    
    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Beta annealing: fast ramp then slow exponential
        progress = min(1.0, (t - 1) / (total_iters - 1))
        if progress < 0.3:
            beta = beta_start * (beta_end / beta_start) ** (progress / 0.3 * 0.3)
        else:
            frac = (progress - 0.3) / 0.7
            beta = beta_start * (beta_end / beta_start) ** (0.3 + 0.7 * frac)
            beta = min(beta, beta_end)
        
        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            mom = (np.sqrt(1.0 + 4.0 * (t - 1) ** 2) - 1.0) / (2.0 * (t - 1))
            y = x + mom * (x - x_prev)
            y = np.maximum(y, 1e-30)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        
        # Compute gradient
        grad = compute_lse_gradient(y, beta)
        if np.all(np.abs(grad) < 1e-15):
            break
        
        # Project onto tangent space of simplex
        grad = grad - grad.mean()
        
        # Step size: use 1/(beta * ||a||_inf) as Lipschitz bound approximation
        ac_y = fast_autocorr(y)
        M_y = ac_y.max()
        grad_norm = np.sqrt((grad ** 2).sum())
        if grad_norm < 1e-15:
            break
        # Adaptive learning rate
        lr = 2.0 / (grad_norm * (1.0 + np.sqrt(t)))
        lr = min(lr, 0.5 / (beta * M_y + 1e-10))
        
        # Mirror descent update (multiplicative)
        log_y = np.log(y + 1e-30) - lr * grad
        max_log = log_y.max()
        x_new = np.exp(log_y - max_log)
        x_new = np.maximum(x_new, 1e-30)
        x_new = x_new / x_new.sum()
        
        x_prev = x
        x = x_new
        
        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_m:
            best_m = M_new
            best_a = x.copy()
        
        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LSE-ND t={t}/{total_iters} ({elapsed:.0f}s): J={J:.6f}, M={best_m:.8f}, beta={beta:.0f}")
    
    return best_a, best_m

def mw_frank_wolfe(a0, time_limit, verbose, max_fw_iters, n_mw_iters):
    """
    Frank-Wolfe with multiplicative weights inner LP solver.
    Much faster per iteration than exact LP.
    """
    n = len(a0)
    y = norm_sum1(a0)
    x = y.copy()
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    for t in range(1, max_fw_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac_y = fast_autocorr(y)
        M = ac_y.max()
        if M < 1e-14:
            break
        
        # Solve inner LP: max sum(b) s.t. (y*b)[i] <= M, b >= 0
        # Using multiplicative weights
        b = np.ones(n) / n
        
        for mw_t in range(n_mw_iters):
            v = fast_conv(y, b)
            excess = np.maximum(v - M, 0.0)
            if excess.max() < 1e-12 * M:
                break
            
            # Pressure: p[j] = y[j] * sum_{i=j}^{min(j+n-1,2n-2)} excess[i]
            cum = np.zeros(len(excess) + 1)
            cum[1:] = np.cumsum(excess)
            p = np.zeros(n)
            for j in range(n):
                i_end = min(j + n, len(excess))
                p[j] = y[j] * (cum[i_end] - cum[j])
            
            # Multiplicative update
            scale = np.clip(p / M, 0.0, 3.0)
            b = b * np.exp(-scale)
            b = np.maximum(b, 1e-30)
            s_b = b.sum()
            if s_b < 1e-15:
                b = np.ones(n) / n
            else:
                b = b / s_b
        
        # Nesterov FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = b.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MW-FW t={t} ({elapsed:.0f}s): J={J:.6f}, M={best_m:.8f}")
    
    return best_a, best_m

def peak_coord_descent(a0, time_limit, verbose, seed):
    """
    Peak-targeted coordinate descent: move mass from high-peak-contributing
    positions to low-impact positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top peak positions
        top_k = min(5, len(ac))
        peak_order = np.argsort(ac)[::-1][:top_k]
        
        # Compute sensitivity: how much each position contributes to top peaks
        sens = np.zeros(n)
        for p in peak_order:
            p_int = int(p)
            lo = max(0, p_int - n + 1)
            hi = min(n - 1, p_int)
            for j in range(lo, hi + 1):
                d1 = p_int - j
                d2 = j - p_int
                c = 0.0
                if 0 <= d1 < n:
                    c += a[d1]
                if 0 <= d2 < n:
                    c += a[d2]
                sens[j] += ac[p_int] * c
        
        # Rank positions
        src_rank = np.argsort(sens)[::-1]
        dst_rank = np.argsort(sens)
        
        improved = False
        for si in range(min(15, n)):
            if time.monotonic() - t0 > time_limit:
                break
            src = int(src_rank[si])
            if a[src] < 1e-12:
                continue
            
            for di in range(min(80, n)):
                if time.monotonic() - t0 > time_limit:
                    break
                dst = int(dst_rank[di])
                if dst == src:
                    continue
                
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    s_t = a_t.sum()
                    if s_t < 1e-15:
                        continue
                    a_t = a_t / s_t
                    M_t = fast_autocorr(a_t).max()
                    if M_t < best_m - 1e-16:
                        a = a_t
                        best_m = M_t
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            n_pert = max(5, n // 80)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.003, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                M_c = fast_autocorr(a).max()
                if M_c < best_m:
                    best_m = M_c
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordPD ({elapsed:.0f}s): J={J:.6f}, M={best_m:.8f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Resample sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate():
    """
    Main search: multi-phase optimization combining LSE-ND, MW-FW, and coordinate descent
    with multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = 1720.0
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    # Helper to try updating (inlined to avoid closure)
    def _do_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    
    # Phase 1: LSE-ND main (45% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 1: LSE-ND n={n0} ===")
        t1 = min(remaining * 0.45, 700.0)
        a_out, _ = lse_nesterov_descent(norm_sum1(best_seq), t1, 20.0, 50000.0, True, 150000)
        _do_update(a_out, "lse_p1")
    
    # Phase 2: MW-FW
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: MW-FW n={n_cur} ===")
        t2 = min(remaining * 0.18, 250.0)
        a_out, _ = mw_frank_wolfe(norm_sum1(best_seq), t2, True, 30000, 80)
        _do_update(a_out, f"mwfw_{n_cur}")
    
    # Phase 3: LSE-ND round 2 with higher beta
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: LSE-ND r2 n={n_cur} ===")
        t3 = min(remaining * 0.22, 350.0)
        a_out, _ = lse_nesterov_descent(norm_sum1(best_seq), t3, 100.0, 100000.0, True, 200000)
        _do_update(a_out, "lse_p3")
    
    # Phase 4: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: CoordPD n={n_cur} ===")
        t4 = min(remaining * 0.10, 100.0)
        a_out, _ = peak_coord_descent(norm_sum1(best_seq), t4, True, 42)
        _do_update(a_out, f"coord_{n_cur}")
    
    # Phase 5: Multi-n exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 80:
        print(f"\n=== Phase 5: Multi-n exploration ===")
        n_cands = []
        for d in [300, 600, 900, -300, -600]:
            nt = n_best + d
            if 1000 <= nt <= 4000 and nt != n_best:
                n_cands.append(nt)
        
        for nt in n_cands[:2]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            print(f"  Trying n={nt}...")
            a_inter = interpolate_sequence(best_seq, nt)
            t_per = min(remaining * 0.30, 200.0)
            a_out, _ = lse_nesterov_descent(norm_sum1(a_inter), t_per, 50.0, 30000.0, False, 100000)
            _do_update(a_out, f"mn_{nt}")
            # Quick coord polish on new length
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30 and len(best_seq) == nt:
                t_cp = min(remaining * 0.15, 60.0)
                a_out2, _ = peak_coord_descent(norm_sum1(best_seq), t_cp, False, nt % 1000)
                _do_update(a_out2, f"mn_coord_{nt}")
    
    # Phase 6: Final LSE-ND polish (very high beta)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 6: Final LSE-ND n={n_best} ===")
        t6 = min(remaining * 0.40, 150.0)
        a_out, _ = lse_nesterov_descent(norm_sum1(best_seq), t6, 500.0, 200000.0, False, 100000)
        _do_update(a_out, "lse_final")
    
    # Phase 7: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Final CoordPD n={n_best} ===")
        t7 = min(remaining * 0.60, 100.0)
        a_out, _ = peak_coord_descent(norm_sum1(best_seq), t7, True, 777)
        _do_update(a_out, "coord_final")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
