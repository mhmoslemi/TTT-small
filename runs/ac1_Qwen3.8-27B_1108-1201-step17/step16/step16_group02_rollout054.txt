def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def hard_subgradient(a, ac, top_k=5):
    """Compute subgradient using top-k lags with equal weight."""
    n = len(a)
    top_idx = np.argsort(ac)[::-1][:min(top_k, len(ac))]
    grad = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * a[j_idx] / top_k  # normalized
    return grad

def mixed_subgradient(a, ac, top_k=10, soft_weight=0.3, temperature=0.01):
    """Mix of hard and soft subgradient."""
    n = len(a)
    # Hard part
    g_hard = hard_subgradient(a, ac, top_k)
    # Soft part
    log_ac = np.log(np.maximum(ac, 1e-30) / temperature)
    log_ac -= log_ac.max()
    weights = np.exp(log_ac)
    w_sum = weights.sum()
    weights /= w_sum
    g_soft = np.zeros(n)
    active = weights > 1e-10
    for k in np.where(active)[0]:
        k_int = int(k)
        w = weights[k]
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g_soft[idx] += 2.0 * w * a[j_idx]
    return (1 - soft_weight) * g_hard + soft_weight * g_soft

def lp_fw_linesearch(a0, time_limit, solver, max_iter=2000, verbose=True):

    t0 = time.monotonic()

            break

        M_curr = ac.max()
        
        # Select active constraints
        threshold = M_curr * 0.95
        active_lags = np.where(ac >= threshold)[0]
        if len(active_lags) > 300:
            active_lags = np.argsort(ac)[::-1][:300]
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((len(active_lags), n))
        for row_idx, k in enumerate(active_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]
        
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Line search: find best alpha in [0, 1]
        best_alpha = 0
        best_trial_m = best_m
        
        # Try a grid of alphas

            y = np.maximum(y, 0.0)
