**Use the circular frame potential** (with zero-padding to a large size). This is approximately the same as the linear one for the interior, and the gradient is easy to compute:

Wait, `B^2 * conj(B) = B * |B|^2`. And since b is real, `B[n-k] = conj(B[k])`, so `B^2 * conj(B)` has the property that its IFFT is real. Similarly `B^3` has IFFT that's real.

Let me verify: `B^3[k+j]`... no, let me think differently.
`B^3[n-k] = B[n-k]^3 = conj(B[k])^3 = conj(B[k]^3) = conj(B^3[k])`. ✓
`B^2*conj(B)[n-k] = B[n-k]^2 * conj(B[n-k]) = conj(B[k])^2 * B[k] = B[k]^2 * conj(B[k]) = conj(B^2*conj(B)[k])`...

And `conj(B^2*conj(B)[k]) = conj(B[k]^2 * conj(B[k])) = conj(B[k])^2 * B[k]`.

So `B^2*conj(B)[n-k] = conj((B^2*conj(B))[k])`. ✓

So both IFFT(B^3) and IFFT(B^2*conj(B)) are real (up to numerical error). Good.

def lp_frank_wolfe(a0, time_limit, solver_name, verbose=False, max_iters=50000):
    """LP Frank-Wolfe with Nesterov acceleration."""
    import cvxpy as cp
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()

            break

        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Build constraint: (y * g)[k] <= M_curr for all k
        # (y * g)[k] = sum_j y[j] * g[k-j]
        # For fixed y, this is linear in g.
        # T is (2n-1) x n matrix: T[k,j] = y[j] if 0 <= k-j < n, else 0
        num_rows = 2 * n - 1
        T = np.zeros((num_rows, n))
        for j in range(n):
            start = j
            end = min(num_rows, j + n)
            T[start:end, j] = y[j]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        else:
            y_new = y_new / sy

            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_conv(y).max()
        if my < best_m:

        mx = fast_conv(x).max()
        if mx < best_m:
