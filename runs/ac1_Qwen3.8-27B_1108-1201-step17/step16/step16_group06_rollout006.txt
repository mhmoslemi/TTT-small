def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def build_sparse_T(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    import scipy.sparse as sp
    return sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def compute_subgradient(a, k_active):
    """Compute subgradient of max autocorrelation at active peaks."""
    n = len(a)
    g = np.zeros(n)
    for k in k_active:
        k = int(k)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] += a[j]
    # Scale: the subgradient of (a*a)[k] w.r.t. a[i] is 2*a[k-i] if valid
    # Actually from our derivation: d(a*a)[k]/da[i] = 2*a[k-i] if 0<=k-i<n
    # But we summed over active k's, so we need to be careful
    # Let me redo: if f(a) = max_k (a*a)[k], and the active set is K,
    # then a subgradient is (1/|K|) * sum_{k in K} grad (a*a)[k]
    # where grad_k[i] = 2*a[k-i] if 0 <= k-i < n
    # So g[i] = (1/|K|) * sum_{k in K} 2*a[k-i] (if valid)
    # Which is what we have above (without the factor of 2, let's add it)
    g *= 2.0 / len(k_active)
    return g

def mirror_descent_subgrad(a0, time_limit, tol=0.01, verbose=False, max_iters=1000000, eta0=1.0):

    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = eta0
    stall = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Find all near-active peaks (within tol of max)
        threshold = M * (1.0 - tol)
        active_idx = np.where(ac >= threshold)[0]
        
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(ac)])
        
        # Compute subgradient
        g = compute_subgradient(a, active_idx)
        
        # Mirror descent step
        # log a_new = log a - eta * g, then normalize
        min_a = np.min(a)
        if min_a < 1e-15:

            a_safe = np.maximum(a, 1e-15)
        else:
            a_safe = a
        
        log_a = np.log(a_safe)
        # Scale g to be comparable to log values
        g_scale = np.max(np.abs(g))
        if g_scale < 1e-15:
            break
        g_normalized = g / g_scale
        
        log_a_new = log_a - eta * g_normalized
        max_log = np.max(log_a_new)
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 1e-15)

            eta *= 0.8
            stall += 1

                stall = 0
                # Small perturbation

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}, eta={eta:.4f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
