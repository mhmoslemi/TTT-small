    active = np.where(conv >= threshold * M)[0]
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += a[j_idx]
    return g


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_ramp(n, power=1.0):
    """Create a power ramp initialization."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return a / a.sum()


def make_sine_ramp(n, power=1.0):
    """Create a sine-based ramp (0 at start, 1 at end)."""
    x = np.linspace(0, np.pi / 2, n)
    a = np.sin(x) ** (2 * power)
    return a / a.sum()


def make_parent():
    """Try to load parent sequence."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p / p.sum()
    except Exception:
        pass
    return None


def evaluate_safe_local(a):
    """Safely evaluate the objective."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def amd_optimize(a0, time_limit, c0=3.0, threshold=0.995, verbose=True, rng_seed=42):
    """
    Accelerated Mirror Descent to minimize max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    n = len(a0)
    a = norm_sum1(a0)
    conv = fast_conv(a)
    best_a = a.copy()
    best_m = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 5000
    c_current = c0

    for t in range(1, 5000000):
        # Time check (every 100 iterations for efficiency)
        if t % 100 == 0 and time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute subgradient
        conv_y = fast_conv(y)
        g = compute_subgradient(y, conv_y, threshold)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn

        # Adaptive step size
        eta = c_current / (np.sqrt(t) + 1.0) * np.sqrt(n)

        # Mirror (multiplicative) update
        log_a = np.log(np.maximum(y, 1e-15)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / sn

        x_prev = x
        x = a_new

        # Evaluate progress (every 10 iterations to save time)
        if t % 10 == 0:
            M_new = fast_conv(x).max()
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = x.copy()
                stall = 0
            else:
                stall += 1

        # Restart if stalled
        if stall > max_stall:
            stall = 0
            # Perturb from best
            n_pert = max(10, n // 40)
            idx = rng.choice(n, size=n_pert, replace=False)
            multipliers = rng.uniform(0.7, 1.3, n_pert)
            x_restarted = best_a.copy()
            x_restarted[idx] *= multipliers
            x_restarted = np.maximum(x_restarted, 1e-15)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted.copy()
            c_current = c_current * 1.2
            if c_current > 15.0:
                c_current = 15.0

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}, c={c_current:.2f}")

    return best_a


def coordinate_descent(a0, time_limit, verbose=True, rng_seed=123):
    """
    Coordinate descent: move mass from high-contributing positions to reduce peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    conv = fast_conv(a)
    best_a = a.copy()
    best_m = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 50

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        peak_lags = np.argsort(conv)[::-1][:min(10, len(conv))]
        peak_max = conv[peak_lags[0]]

        # Compute sensitivity: for each position, how much it contributes to peaks
        peak_sens = np.zeros(n)
        for k in peak_lags[:8]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            peak_sens[idx] += a[j_idx]

        # Try moving mass from high-sensitivity to low-sensitivity positions
        # Sort positions by sensitivity
        high_sens = np.argsort(peak_sens)[::-1][:30]
        low_sens = np.argsort(peak_sens)[:30]

        improved = False
        for src in high_sens:
            if a[src] < 1e-10:
                continue
            for dst in low_sens:
                if dst == src or a[dst] > a[src]:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    m_trial = fast_conv(a_trial).max()
                    if m_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                # Recompute peak lags after improvement
                continue

        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            n_pert = max(5, n // 80)
            sources = rng.choice(n, size=n_pert, replace=False)
            for i in sources:
                if a[i] > 1e-8:
                    j = int(rng.integers(n))
                    if j != i:
                        frac = rng.uniform(0.02, 0.08)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        if a.sum() > 1e-15:
                            a = a / a.sum()
                            m_new = fast_conv(a).max()
                            if m_new < best_m:
                                best_m = m_new
                                best_a = a.copy()
                                stall = 0
                            else:
                                a[i] += amt
                                a[j] -= amt
                                a = np.maximum(a, 0.0)
                                if a.sum() > 1e-15:
                                    a = a / a.sum()

        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    return norm_sum1(a_new)
