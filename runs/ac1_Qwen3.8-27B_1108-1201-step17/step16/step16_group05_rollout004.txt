def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lg_frank_wolfe(a0, time_limit, solver, verbose=True, max_iters=None):

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if max_iters is None:

    x_mom = a.copy()
    iter_count = 0

        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Build constraint matrix using only top constraints (for efficiency)
        # Number of active constraints that matter
        n_top = min(2 * n - 1, 3 * n)

            iter_count += 1

            continue

        if sx < 1e-15:
            x_mom = y_new.copy()

            x_mom = x_mom / sx

    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find the argmax lag(s)
        top_lags = np.argsort(ac)[::-1][:15]
        
        # Compute contribution of each coordinate to the top lags
        contrib = np.zeros(n)
        for k in top_lags[:8]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                j = k_int - i
                if 0 <= j < n:
                    # Contribution of a[i] to c[k]
                    weight = a[j]
                    if i != j:
                        weight *= 2.0
                    contrib[i] += weight * (ac[k_int] / M_curr)
        
        # Find high-contribution sources
        sources = np.argsort(contrib)[::-1][:min(50, n)]
        
        improved = False
        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            
            # Find best destination: one that minimizes the increase in max autocorrelation
            # Score destination by how much it would increase peak lags
            n_try = min(n, 200)
            candidates = np.random.choice(n, n_try, replace=False) if n > n_try else np.arange(n)
            
            best_dst = -1
            best_score = float('inf')
            
            for dst in candidates:
                if dst == src:
                    continue
                # Estimate impact of moving mass to dst
                impact = 0.0
                for k in top_lags[:6]:
                    k_int = int(k)
                    d = k_int - dst
                    if 0 <= d < n:
                        impact += a[d]
                        if dst != d:
                            impact += a[d]
                    d2 = dst - k_int
                    if 0 <= d2 < n:
                        impact += a[d2]
                        if dst != d2:
                            impact += a[d2]
                if impact < best_score:
                    best_score = impact
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            # Try moving mass
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-17:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            if n > 2:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.001, 0.005)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()
        
        if verbose and stall % 20 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm, left=0.0, right=0.0)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def get_parent_sequence():
    """Try to load the parent sequence."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p.copy()
    except Exception:
        pass
    return None


def diverse_initializations(n, rng, best_seq=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.linspace(0, 1, n)
    
    # 1. Sine
    s1 = np.sin(np.pi * x) ** 2
    inits.append(s1 / s1.sum())
    
    # 2. Ramp (asymmetric)
    s2 = x + 0.1
    inits.append(s2 / s2.sum())
    
    # 3. Quadratic (concentrated at one end)
    s3 = x ** 3 + 0.01
    inits.append(s3 / s3.sum())
    
    # 4. Bimodal
    s4 = np.exp(-50 * (x - 0.3) ** 2) + np.exp(-50 * (x - 0.7) ** 2)
    inits.append(s4 / s4.sum())
    
    # 5. Quartic bump
    s5 = np.maximum(1.0 - (2*x - 1)**2, 0) ** 4
    ss = s5.sum()
    if ss > 0:
        inits.append(s5 / ss)
    
    # 6. Cosine-based asymmetric
    s6 = np.cos(np.pi * x * 0.5) ** 2 + 0.3 * x
    inits.append(s6 / s6.sum())
    
    # 7. Blend with best if available
    if best_seq is not None:
        bs = interpolate_sequence(best_seq, n)
        inits.append(0.7 * bs + 0.3 * inits[1])
        inits[-1] = norm_sum1(inits[-1])
    
    # 8. Random smooth
    noise = rng.normal(0, 1, n)
    # Smooth it
    kernel = np.exp(-np.arange(-20, 21)**2 / 50.0)
    kernel /= kernel.sum()
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = np.maximum(smooth, 0) + 0.01
    inits.append(smooth / smooth.sum())
    
    return inits
