        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def spectral_equalize(a, n_iter=50, eps=1e-10):
    """Flatten power spectrum via iterative DFT correction with non-negativity."""
    a = normalize(a).copy()
    n = len(a)
    for i in range(n_iter):
        A = np.fft.fft(a)
        P = np.abs(A) ** 2
        # Target: flat spectrum (excluding DC)
        P_mean = P[1:].mean()
        target = np.concatenate([[P[0]], np.full(n-1, P_mean)])
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        # Adaptive blending: more aggressive early, more conservative late
        blend = 0.7 if i < 10 else (0.5 if i < 30 else 0.3)
        a = (1-blend) * a + blend * a_new
    return a


def multiplicative_peak_shave(a0, time_limit, top_k=80, eta0=1.0, verbose=True):
    """
    Multiplicative update to reduce peak of convolution.
    a[i] <- a[i] * (1 - eta * contribution[i])
    where contribution measures how much position i contributes to top peaks.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    eta = eta0
    stall = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Compute contribution of each position to top peaks
        top_idx = np.argsort(ac)[::-1][:top_k]
        # Weight by peak value (softmax-like)
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals / top_vals.sum()
        
        # Contribution: for position i, how much does it contribute to top peaks?
        # For peak at lag k, position i contributes a[i] * a[k-i] (if valid)
        # So contribution[i] = sum_k w[k] * a[k-i] (if k-i valid)
        contrib = np.zeros(n)
        for wk, k in zip(w, top_idx):
            k_int = int(k)
            # For position i, the partner is at k-i (must be in [0, n-1])
            # So i must be in [k-(n-1), k] = [k-n+1, k] intersected with [0, n-1]
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                partners = k_int - np.arange(i_start, i_end + 1)
                contrib[i_start:i_end+1] += wk * a[partners]
        
        # Multiplicative update
        max_contrib = contrib.max()
        if max_contrib < 1e-20:
            break
        # Adaptive eta: decrease if stalling
        if stall > 100:
            eta *= 0.95
            stall = 0
        
        a_new = a * (1.0 - eta * contrib / max_contrib)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        # Small random perturbation to avoid getting stuck
        if t % 500 == 0:
            pert = rng.normal(0, 0.001, n)
            a_new = a_new * (1.0 + pert)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
        
        a = a_new
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MPS t={t} ({elapsed:.1f}s): J={J:.6f}, eta={eta:.4f}")
    
    return best_a, best_m


def solve_lp_direction_fast(y, M, active_lags, solver):
    """
    Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0, sum(b)=1.
    Returns normalized direction or None.
    """
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    
    constraints = [cp.sum(b_var) == 1]
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def lp_fw_improved(a0, time_limit, solver, top_k_init=80, top_k_max=300, verbose=True):
    """
    Improved LP Frank-Wolfe with:
    - Nesterov momentum
    - Backtracking line search
    - Adaptive constraint set (top-K + random)
    - Periodic full-constraint verification
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(456)
    
    x = a.copy()
    x_prev = a.copy()
    current_top_k = top_k_init
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        
        # Compute autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()
        
        # Select active constraints: top-K + some random
        n_constraints = min(current_top_k + 20, len(ac))
        top_indices = np.argsort(ac)[::-1][:current_top_k]
        # Add some random constraints for diversification
        n_random = min(20, len(ac) - current_top_k)
        if n_random > 0:
            remaining = np.setdiff1d(np.arange(len(ac)), top_indices)
            random_indices = remaining[rng.integers(0, len(remaining), n_random)]
            active_lags = np.concatenate([top_indices, random_indices])
        else:
            active_lags = top_indices
        
        # Solve LP
        direction = solve_lp_direction_fast(y, M_curr, active_lags, solver)
        if direction is None:
            continue
        
        # Backtracking line search
        alpha = 2.0 / (t + 2.0)
        best_alpha = alpha
        best_obj = M_curr
        
        for _ in range(5):
            x_trial = (1.0 - alpha) * y + alpha * direction
            x_trial = np.maximum(x_trial, 0.0)
            sx = x_trial.sum()
            if sx < 1e-15:
                break
            x_trial = x_trial / sx
            obj = fast_autocorr(x_trial).max()
            if obj < best_obj:
                best_obj = obj
                best_alpha = alpha
            alpha *= 0.5
        
        x_new = (1.0 - best_alpha) * y + best_alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx
        
        x_prev = y
        x = x_new
        
        # Check improvement
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        # Adapt top_k
        if t % 30 == 0 and current_top_k < top_k_max:
            current_top_k = min(current_top_k + 10, top_k_max)
        
        # Periodic restart with perturbation
        if t % 5000 == 0:
            pert = rng.normal(0, 0.01, n)
            x_pert = best_a * (1.0 + pert)
            x_pert = np.maximum(x_pert, 1e-10)
            x_pert = x_pert / x_pert.sum()
            # Keep the better of current and perturbed
            if fast_autocorr(x_pert).max() < best_m:
                x = x_pert
                x_prev = x_pert.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW+ t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}")
    
    return best_a, best_m


def coordinate_descent_v2(a0, time_limit, verbose=True, rng_seed=42):
    """
    Improved coordinate descent:
    - Identifies top peaks
    - For each peak, finds the best (src, dst, amount) move
    - Uses sensitivity analysis
    - Multiple scales of moves
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(30, len(ac))]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find the position with maximum contribution to this peak
            src_relative = np.argmax(contrib)
            src = idx[src_relative]
            partner = k_int - src
            
            # Strategy: move mass from src to a position that doesn't contribute to any top peak
            # Compute "danger" for each position: how much would adding mass there increase top peaks?
            danger = np.zeros(n)
            for k2 in top_k[:15]:
                k2i = int(k2)
                # If we add mass at position d, it contributes to peak k2 through:
                # a[d] * a[k2-d] (if k2-d valid) and a[k2-d] * a[d] (same thing for convolution)
                # Actually for conv: adding delta at position d increases conv[k] by delta * a[k-d] (if valid)
                # So danger[d] = sum_k2 w[k2] * a[k2-d] (if valid)
                d_start = max(0, k2i - n + 1)
                d_end = min(n - 1, k2i)
                if d_start <= d_end:
                    partners_d = k2i - np.arange(d_start, d_end + 1)
                    danger[d_start:d_end+1] += a[partners_d]
            
            # Find safe destinations (low danger)
            # Exclude positions that are src or partner
            danger[src] = np.inf
            if 0 <= partner < n:
                danger[partner] = np.inf
            
            safe_positions = np.where(danger < danger.max() * 0.3)[0]
            if len(safe_positions) < 20:
                safe_positions = np.argsort(danger)[:min(200, n)]
            
            # Try moving mass from src to various safe positions
            best_move = None
            best_score = best_m
            
            for dst in safe_positions[:min(50, len(safe_positions))]:
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_score - 1e-16:
                        best_score = M_trial
                        best_move = (src, dst, amt)
            
            if best_move is not None:
                src_m, dst_m, amt_m = best_move
                a[src_m] -= amt_m
                a[dst_m] += amt_m
                a = np.maximum(a, 0.0)
                a = a / max(a.sum(), 1e-15)
                best_m = best_score
                best_a = a.copy()
                improved = True
                stall = 0
        
        if not improved:
            stall += 1
            # Try random perturbation
            n_tries = 10
            for _ in range(n_tries):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            break
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / max(a.sum(), 1e-15)
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord v2 ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def spectral_equalize(a, n_iter=50, eps=1e-10):
    """Flatten power spectrum via iterative DFT correction with non-negativity."""
    a = normalize(a).copy()
    n = len(a)
    for i in range(n_iter):
        A = np.fft.fft(a)
        P = np.abs(A) ** 2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        blend = 0.7 if i < 10 else (0.5 if i < 30 else 0.3)
        a = (1.0 - blend) * a + blend * a_new
    return a


def multiplicative_peak_shave(a0, time_limit, top_k=100, eta0=1.0, verbose=True):
    """
    Multiplicative update: a[i] <- a[i] * (1 - eta * contrib[i])
    where contrib measures how much position i feeds into top convolution peaks.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    eta = eta0
    stall = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(a)
        M = ac.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals / top_vals.sum()

        contrib = np.zeros(n)
        for wk, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                partners = k_int - np.arange(i_start, i_end + 1)
                contrib[i_start:i_end + 1] += wk * a[partners]

        max_contrib = contrib.max()
        if max_contrib < 1e-20:
            break

        if stall > 200:
            eta *= 0.9
            stall = 0
        elif stall > 50:
            eta *= 0.97

        a_new = a * (1.0 - eta * contrib / max_contrib)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

            a_new = a_new * (1.0 + pert)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()

        a = a_new

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MPS t={t} ({elapsed:.1f}s): J={J:.6f}, eta={eta:.4f}")

def solve_lp_direction(y, M, active_lags, solver):
    """
    Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0, sum(b)=1.
    """
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)

    constraints = [cp.sum(b_var) == 1]
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_gaussian_peak(n, center_frac, width_frac):
    """Create a smooth Gaussian-like peak."""
    x = np.linspace(0.0, 1.0, n)
    result = np.exp(-0.5 * ((x - center_frac) / width_frac) ** 2)
    s = result.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return result / s

    print(f"\n=== Phase 1: Spectral Equalization (budget={remaining:.0f}s) ===")
    a_spec = spectral_equalize(best_seq, n_iter=60)
    _check_update(a_spec, "spectral")

    # === Phase 2: Multiplicative Peak Shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: MPS n={len(best_seq)}, budget={remaining:.0f}s ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_in = normalize(best_seq)
        a_out, _ = multiplicative_peak_shave(a_in, t2, top_k=120, eta0=0.9, verbose=True)
        _check_update(a_out, f"mps_n{len(a_out)}")

    # === Phase 3: LP-FW (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW+ n={n_best}, budget={remaining:.0f}s ===")
    t3 = min(remaining * 0.35, 550.0)
    if t3 > 30:
        a_in = normalize(best_seq)
        a_out, _ = lp_fw_improved(a_in, t3, solver, top_k_init=100, top_k_max=300, verbose=True)
        _check_update(a_out, f"lpfw_n{n_best}")

    # === Phase 4: Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord v2 n={n_best}, budget={remaining:.0f}s ===")
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 20:
        a_in = normalize(best_seq)
        a_out, _ = coordinate_descent_v2(a_in, t4, verbose=True, rng_seed=42)
        _check_update(a_out, f"coord_n{n_best}")

    # === Phase 5: LP-FW refined (higher K) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW refined n={n_best}, budget={remaining:.0f}s ===")
    t5 = min(remaining * 0.25, 350.0)
    if t5 > 30:
        a_in = normalize(best_seq)
        a_out, _ = lp_fw_improved(a_in, t5, solver, top_k_init=150, top_k_max=400, verbose=True)
        _check_update(a_out, f"lpfw2_n{n_best}")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    n_alt = min(n_best + 600, 3600)
    if n_alt != n_best and remaining > 250:
        print(f"\n=== Phase 6: Alt length n={n_alt}, budget={remaining:.0f}s ===")
        a_long = interpolate_sequence(best_seq, n_alt)
        a_long = spectral_equalize(a_long, n_iter=25)
        _check_update(a_long, f"interp_n{n_alt}")

        t6a = min(remaining * 0.4, 250.0)
        if t6a > 30:
            a_in = normalize(a_long)
            a_out, _ = lp_fw_improved(a_in, t6a, solver, top_k_init=100, top_k_max=250, verbose=True)
            _check_update(a_out, f"lpfw_n{n_alt}")

        # Also try a shorter length
        remaining = time_budget - (time.monotonic() - t_global)
        n_short = max(1600, n_best - 400)
        if n_short != n_best and remaining > 200:
            print(f"  Sub-phase 6b: Short length n={n_short}")
            a_short = interpolate_sequence(best_seq, n_short)
            a_short = spectral_equalize(a_short, n_iter=20)
            _check_update(a_short, f"interp_n{n_short}")
            t6b = min(remaining * 0.35, 200.0)
            if t6b > 30:
                a_in = normalize(a_short)
                a_out, _ = lp_fw_improved(a_in, t6b, solver, top_k_init=80, top_k_max=200, verbose=True)
                _check_update(a_out, f"lpfw_n{n_short}")

    # === Phase 7: Final MPS + Coord ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Polish n={n_best}, budget={remaining:.0f}s ===")

    t7a = min(remaining * 0.25, 150.0)
    if t7a > 15:
        a_in = normalize(best_seq)
        a_out, _ = multiplicative_peak_shave(a_in, t7a, top_k=150, eta0=0.5, verbose=False)
        _check_update(a_out, f"mps_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7b = min(remaining * 0.35, 150.0)
    if t7b > 15:
        a_in = normalize(best_seq)
        a_out, _ = coordinate_descent_v2(a_in, t7b, verbose=True, rng_seed=777)
        _check_update(a_out, f"coord_final_n{n_best}")

    # === Phase 8: Last LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 3200 and remaining > 80:
        t8 = min(remaining * 0.6, 200.0)
        if t8 > 30:
            a_in = normalize(best_seq)
            a_out, _ = lp_fw_improved(a_in, t8, solver, top_k_init=200, top_k_max=500, verbose=True)
            _check_update(a_out, f"lpfw_veryfinal_n{n_best}")

    # === Phase 9: Last coordinate ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        t9 = min(remaining * 0.85, 80.0)
        if t9 > 10:
            a_in = normalize(best_seq)
            a_out, _ = coordinate_descent_v2(a_in, t9, verbose=False, rng_seed=1234)
            _check_update(a_out, f"coord_veryfinal_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
