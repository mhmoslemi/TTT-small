"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 with nonnegative a using:
1. Fast FFT-based weighted-gradient line search with annealed active-peak weights.
2. Log-space softmax parameterization to preserve nonnegativity and normalization.
3. Multi-start, diverse smooth initializations, parent interpolation, length exploration.
4. Final localized pair-transfer polishing.
"""

import numpy as np
import time

# Phase parameters: (top fraction of conv lags, concentration, time fraction)
PHASES_FULL = (
    (1.00, 0.2, 0.10),
    (0.50, 1.0, 0.15),
    (0.20, 5.0, 0.20),
    (0.06, 20.0, 0.25),
    (0.015, 80.0, 0.30),
)
PHASES_MEDIUM = (
    (0.80, 0.5, 0.30),
    (0.20, 10.0, 0.35),
    (0.05, 50.0, 0.35),
)
PHASES_SHARP = (
    (0.10, 3.0, 0.25),
    (0.03, 25.0, 0.35),
    (0.01, 100.0, 0.40),
)


def fft_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_pos(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def conv_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    N = fft_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[:2 * n - 1]


def normalized_and_max(a):
    a = normalize_pos(a)
    c = conv_fft(a)
    cmax = float(c.max()) if c.size else 0.0
    return a, max(0.0, cmax)


def update_best_from_max(a, max_c, best_seq, best_score, label="", t0=0.0):
    if a is None:
        return best_seq, best_score
    a = normalize_pos(a)
    if a.size == 0:
        return best_seq, best_score
    if not np.isfinite(max_c):
        cmax = float(conv_fft(a).max())
    else:
        cmax = float(max_c)
    if not np.isfinite(cmax):
        return best_seq, best_score
    cmax = max(0.0, cmax)
    sc = 2.0 * a.size * cmax
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t0
        if label:
            print(f"  update {label}: score={sc:.8f} n={a.size} t={elapsed:.0f}s")
        else:
            print(f"  update: score={sc:.8f} n={a.size} t={elapsed:.0f}s")
        return a.copy(), sc
    return best_seq, best_score


def update_best_from_sequence(a, best_seq, best_score, label="", t0=0.0):
    if a is None:
        return best_seq, best_score
    a, cmax = normalized_and_max(a)
    return update_best_from_max(a, cmax, best_seq, best_score, label, t0)


def resample_sequence(a, n):
    n = int(n)
    if n <= 0:
        return normalize_pos(a)
    a = normalize_pos(a)
    if a.size == n:
        return a
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    b = np.interp(x_new, x_old, a)
    return normalize_pos(b)


def clean_sequence(a):
    a = normalize_pos(a)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = a.copy()
    a[a < 1e-300] = 0.0
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def init_uniform(n):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    return np.ones(n, dtype=np.float64) / float(n)


def init_gaussian(n, sigma=0.20, center=0.5):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    sigma = max(1e-3, float(sigma))
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return normalize_pos(a)


def init_triangular(n):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 - np.abs(2.0 * x - 1.0)
    a = np.maximum(a, 1e-12)
    return normalize_pos(a)


def init_exp(n, rate=1.5, reverse=False):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    rate = max(1e-3, float(rate))
    if reverse:
        a = np.exp(rate * (x - 1.0))
    else:
        a = np.exp(-rate * x)
    return normalize_pos(a)


def init_two_gauss(n):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((x - 0.35) / 0.12) ** 2) + np.exp(-0.5 * ((x - 0.65) / 0.12) ** 2) + 1e-12
    return normalize_pos(a)


def init_sine(n):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return normalize_pos(a)


def random_smooth_init(n, seed=1, cutoff_frac=0.1):
    n = int(n)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    rng = np.random.default_rng(seed)
    N = fft_pow2(n)
    r = rng.normal(size=n)
    R = np.fft.rfft(r, N)
    keep = max(2, int(cutoff_frac * (N // 2)))
    if keep < R.size:
        R[keep:] = 0.0
    r = np.real(np.fft.irfft(R, N)[:n])
    r = r - r.mean()
    a = np.exp(0.5 * r)
    return normalize_pos(a)


def noisy_log_sequence(a, seed=1, noise=0.05):
    a = normalize_pos(a)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    rng = np.random.default_rng(seed)
    x = np.log(np.maximum(a, 1e-300))
    x += rng.normal(0.0, float(noise), a.size)
    x = x - x.mean()
    y = np.exp(x - x.max())
    s = float(y.sum())
    if s <= 1e-300:
        return a
    return y / s


def make_initials(n, parent, seed=1):
    n = int(n)
    inits = []
    if n <= 0:
        return [np.ones(1, dtype=np.float64)]
    rng = np.random.default_rng(seed)

    if parent is not None:
        p = normalize_pos(parent)
        if p.size != n:
            p = resample_sequence(p, n)
        inits.append(p)
        for noise in (0.02, 0.05, 0.10, 0.20):
            x = np.log(np.maximum(p, 1e-300))
            x += rng.normal(0.0, float(noise), n)
            x = x - x.mean()
            y = np.exp(x - x.max())
            s = float(y.sum())
            if s > 1e-300:
                inits.append(y / s)

    inits.append(init_gaussian(n, 0.18, 0.30))
    inits.append(init_gaussian(n, 0.20, 0.50))
    inits.append(init_gaussian(n, 0.18, 0.70))
    inits.append(init_gaussian(n, 0.35, 0.45))
    inits.append(init_exp(n, 1.0))
    inits.append(init_exp(n, 2.0, reverse=True))
    inits.append(init_two_gauss(n))
    inits.append(init_sine(n))
    inits.append(init_triangular(n))
    inits.append(random_smooth_init(n, seed + 101, 0.08))
    inits.append(random_smooth_init(n, seed + 102, 0.03))
    inits.append(random_smooth_init(n, seed + 103, 0.20))
    return inits


def grad_weighted_topk(a, top_k, conc):
    a = normalize_pos(a)
    n = a.size
    m = 2 * n - 1
    c = conv_fft(a)
    cmax = float(c.max()) if m else 0.0
    k = int(top_k)
    if k < 1:
        k = 1
    if k > m:
        k = m

    if k >= m:
        idx = np.arange(m, dtype=np.int64)
        vals = c
    else:
        idx = np.argpartition(c, m - k).astype(np.int64)[-k:]
        vals = c[idx]

    if k > 1:
        vmax = float(vals.max())
        vmin = float(vals.min())
        rang = vmax - vmin
        if conc is None or not np.isfinite(conc) or conc <= 0.0 or rang <= 1e-300:
            wtop = np.ones(k, dtype=np.float64) / float(k)
        else:
            scaled = (vmax - vals) / rang
            scaled = np.clip(scaled, 0.0, 1.0)
            wtop = np.exp(-float(conc) * scaled)
            sw = float(wtop.sum())
            if sw <= 1e-300:
                wtop = np.ones(k, dtype=np.float64) / float(k)
            else:
                wtop = wtop / sw
    else:
        wtop = np.ones(1, dtype=np.float64)

    wsum = float(np.dot(wtop, vals))

    if k <= 32:
        g = np.zeros(n, dtype=np.float64)
        for t in range(k):
            wi = float(wtop[t])
            if wi == 0.0:
                continue
            kk = int(idx[t])
            js = kk - (n - 1)
            if js < 0:
                js = 0
            je = kk
            if je > n - 1:
                je = n - 1
            if js <= je:
                pos = np.arange(js, je + 1, dtype=np.int64)
                g[js:je + 1] += (2.0 * wi) * a[kk - pos]
    else:
        w_dense = np.zeros(m, dtype=np.float64)
        w_dense[idx] = wtop
        a_rev = a[::-1]
        N = fft_pow2(3 * n - 2)
        W = np.fft.rfft(w_dense, N)
        AR = np.fft.rfft(a_rev, N)
        convw = np.fft.irfft(W * AR, N)
        g = 2.0 * convw[n - 1:2 * n - 1]

    g = np.where(np.isfinite(g), g, 0.0)
    return g, c, cmax, wsum, idx, wtop


def grad_ls_run(a0, time_limit, phases, seed=1, verbose=False):
    a_in = normalize_pos(a0)
    n = a_in.size
    if n == 0:
        return a_in, 0.0
    if time_limit <= 0:
        cmax = float(conv_fft(a_in).max())
        return a_in, max(0.0, cmax)

    t0 = time.monotonic()
    x = np.log(np.maximum(a_in, 1e-300))
    x = x - x.mean()
    a = a_in
    c = conv_fft(a)
    cmax = float(c.max())
    best_a = a.copy()
    best_c = max(0.0, cmax)
    best_x = x.copy()

    rng = np.random.default_rng(seed)
    total_frac = 0.0
    for ph in phases:
        total_frac += float(ph[2])
    if total_frac <= 0.0:
        total_frac = 1.0

    cum = 0.0
    restart_count = 0
    last_imp_iter = 0
    global_iter = 0
    len_c = 2 * n - 1

    for pi in range(len(phases)):
        top_frac, conc, frac = phases[pi]
        cum += float(frac)
        phase_end = t0 + time_limit * (cum / total_frac)

        top_k = int(round(float(top_frac) * len_c))
        if top_k < 8:
            top_k = 8
        if top_k > len_c:
            top_k = len_c

        tau = 0.5
        stall = 0

        if verbose:
            print(f"    grad_ls phase {pi}: top_frac={top_frac:.3f}, conc={conc}, top_k={top_k}")

        while time.monotonic() < phase_end:
            if time.monotonic() > t0 + time_limit:
                break
            global_iter += 1

            xm = x - x.max()
            xm = np.maximum(xm, -700.0)
            a = np.exp(xm)
            s = float(a.sum())
            if not np.isfinite(s) or s <= 1e-300:
                x = best_x.copy()
                continue
            a = a / s

            g, c, cmax, wsum, idx, wtop = grad_weighted_topk(a, top_k, conc)
            cmax = max(0.0, cmax)

            if cmax < best_c:
                best_c = cmax
                best_a = a.copy()
                best_x = x - x.mean()
                last_imp_iter = global_iter

            if cmax > best_c + max(1e-7, 1e-4 * best_c):
                noise = min(0.30, 0.04 + 0.02 * restart_count)
                x = best_x + rng.normal(0.0, noise, n)
                x = x - x.mean()
                tau = 0.20
                restart_count += 1
                stall = 0
                continue

            if global_iter - last_imp_iter > 300000:
                noise = min(0.35, 0.05 + 0.03 * restart_count)
                x = best_x + rng.normal(0.0, noise, n)
                x = x - x.mean()
                restart_count += 1
                last_imp_iter = global_iter
                stall = 0
                tau = 0.30

            gm = float(g.mean())
            r = g - gm
            std = float(r.std())
            if not np.isfinite(std) or std < 1e-14:
                r = rng.normal(size=n)
            else:
                r = r / std

            r = r - r.mean()
            rstd = float(r.std())
            if not np.isfinite(rstd) or rstd < 1e-14:
                r = rng.normal(size=n)
                r = r - r.mean()
                rstd = float(r.std())
            if rstd <= 1e-14:
                continue
            r = r / rstd

            accepted = False
            for fac in (1.0, 0.5, 0.25):
                tau_try = tau * fac
                if tau_try < 1e-5:
                    break

                x_new = x - tau_try * r
                x_new = x_new - x_new.max()
                x_new = np.maximum(x_new, -700.0)
                a_new = np.exp(x_new)
                sn = float(a_new.sum())
                if not np.isfinite(sn) or sn <= 1e-300:
                    continue
                a_new = a_new / sn

                c_new = conv_fft(a_new)
                nm = max(0.0, float(c_new.max()))

                accept = False
                if nm < cmax - 1e-15:
                    accept = True
                else:
                    if nm <= cmax + max(1e-13, 1e-8 * cmax):
                        wdot = float(np.dot(wtop, c_new[idx]))
                        if wdot < wsum - 1e-15:
                            accept = True

                if accept:
                    x = x_new
                    x = x - x.mean()
                    a = a_new
                    cmax = nm
                    if cmax < best_c:
                        best_c = cmax
                        best_a = a_new.copy()
                        best_x = x - x.mean()
                        last_imp_iter = global_iter
                    tau = min(tau_try * 1.25, 1.5)
                    accepted = True
                    stall = 0
                    break

            if not accepted:
                tau = max(tau * 0.7, 1e-4)
                stall += 1
                if stall > 5000:
                    noise = min(0.35, 0.04 + 0.02 * restart_count)
                    x = best_x + rng.normal(0.0, noise, n)
                    x = x - x.mean()
                    tau = 0.20
                    stall = 0
                    restart_count += 1
                elif stall % 500 == 0:
                    x = x + rng.normal(0.0, 0.01, n)
                    x = x - x.mean()
                    tau = 0.10

            if verbose and global_iter % 20000 == 0:
                print(f"      grad_ls iter={global_iter}, score={2.0*n*best_c:.8f}, cmax={best_c:.10f}")

    if verbose:
        print(f"    grad_ls done, score={2.0*n*best_c:.8f}")
    return best_a, max(0.0, best_c)


def pair_polish(a0, time_limit, seed=777, verbose=False):
    a = normalize_pos(a0)
    n = a.size
    if n == 0 or time_limit <= 0:
        cmax = float(conv_fft(a).max()) if n else 0.0
        return a, max(0.0, cmax)

    t0 = time.monotonic()
    c = conv_fft(a)
    cmax = max(0.0, float(c.max()))
    best_a = a.copy()
    best_c = cmax
    rng = np.random.default_rng(seed)

    top_frac = 0.03
    top_k = max(16, int(round(top_frac * (2 * n - 1))))
    epoch = 0
    no_imp = 0
    restart_count = 0

    while time.monotonic() - t0 < time_limit:
        epoch += 1
        if time.monotonic() - t0 > time_limit:
            break

        a = normalize_pos(a)
        g, c, cmax, wsum, idx, wtop = grad_weighted_topk(a, top_k, 30.0)
        cmax = max(0.0, cmax)

        if cmax < best_c:
            best_c = cmax
            best_a = a.copy()

        if cmax > best_c + max(1e-7, 1e-4 * best_c):
            a = best_a.copy()
            cmax = best_c
            no_imp = 0
            continue

        order = np.argsort(g)
        low = order[:8]
        high = order[::-1][:16]

        high_list = []
        for h in high:
            hi = int(h)
            if a[hi] > 1e-12 and len(high_list) < 8:
                high_list.append(hi)

        improved = False
        for si in high_list:
            if improved:
                break
            base_amt = a[si]
            if base_amt <= 1e-12:
                continue
            for di in low:
                di = int(di)
                if si == di:
                    continue
                for frac in (0.10, 0.35):
                    a2 = a.copy()
                    amt = base_amt * frac
                    if a2[si] >= amt:
                        a2[si] -= amt
                        a2[di] += amt
                        s2 = float(a2.sum())
                        if s2 > 1e-300:
                            a2 = a2 / s2
                            c2 = conv_fft(a2)
                            nm = max(0.0, float(c2.max()))
                            accept = False
                            if nm < cmax - 1e-15:
                                accept = True
                            elif nm <= cmax + max(1e-13, 1e-8 * cmax):
                                wdot = float(np.dot(wtop, c2[idx]))
                                if wdot < wsum - 1e-15:
                                    accept = True
                            if accept:
                                a = a2
                                cmax = nm
                                improved = True
                                if nm < best_c:
                                    best_c = nm
                                    best_a = a2.copy()
                                break
                if improved:
                    break

        if not improved:
            si = int(rng.integers(n))
            di = int(rng.integers(n))
            if si != di and a[si] > 1e-12:
                a2 = a.copy()
                amt = a[si] * 0.02
                if a2[si] >= amt:
                    a2[si] -= amt
                    a2[di] += amt
                    s2 = float(a2.sum())
                    if s2 > 1e-300:
                        a2 = a2 / s2
                        nm = max(0.0, float(conv_fft(a2).max()))
                        if nm < cmax - 1e-14:
                            a = a2
                            cmax = nm
                            if nm < best_c:
                                best_c = nm
                                best_a = a2.copy()
            no_imp += 1

        if no_imp > 2000:
            noise = min(0.30, 0.03 + 0.02 * restart_count)
            x = np.log(np.maximum(a, 1e-300))
            x += rng.normal(0.0, noise, n)
            x = x - x.mean()
            a = np.exp(x - x.max())
            s = float(a.sum())
            if s <= 1e-300:
                a = best_a.copy()
            else:
                a = a / s
            no_imp = 0
            restart_count += 1

        if verbose and epoch % 500 == 0:
            print(f"      pair epoch={epoch}, score={2.0*n*best_c:.8f}")

    return best_a, max(0.0, best_c)


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    total_budget = min(float(budget_s) - 80.0, 1720.0)
    best_seq = None
    best_score = float("inf")

    # Load parent if available.
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all():
            p = np.clip(p, 0.0, 1000.0)
            if p.sum() > 1e-10:
                a, cmax = normalized_and_max(p)
                best_seq = a
                best_score = 2.0 * a.size * max(0.0, cmax)
                print(f"parent: n={a.size}, score={best_score:.8f}")
    except Exception:
        best_seq = None

    if best_seq is None or best_seq.size == 0:
        n = 2400
        best_seq = init_gaussian(n, 0.25, 0.5)
        _, cmax = normalized_and_max(best_seq)
        best_score = 2.0 * best_seq.size * max(0.0, cmax)
        print(f"default init: n={best_seq.size}, score={best_score:.8f}")

    try:
        n0 = best_seq.size

        # Multi-start medium optimization.
        inits = make_initials(n0, best_seq, seed=17)
        selected = []
        dup_tol = 0.02
        for a in inits:
            if len(selected) >= 8:
                break
            a = normalize_pos(a)
            if a.size != n0:
                continue
            dup = False
            for b in selected:
                if a.size == b.size:
                    diff = float(np.linalg.norm(a - b) / max(1, a.size))
                    if diff < dup_tol:
                        dup = True
                        break
            if not dup:
                selected.append(a)

        for idx, a in enumerate(selected):
            rem = total_budget - (time.monotonic() - t_global)
            if rem < 80:
                break
            alloc = 50.0 if idx == 0 else 25.0
            if len(selected) > 6:
                alloc *= 0.7
            alloc = min(alloc, max(20.0, rem * 0.18))
            if alloc < 15:
                break
            out, maxc = grad_ls_run(a, alloc, PHASES_MEDIUM, seed=100 + idx, verbose=False)
            best_seq, best_score = update_best_from_max(out, maxc, best_seq, best_score,
                                                        f"multi{idx}_n{a.size}", t_global)

        # Long run from current best.
        rem = total_budget - (time.monotonic() - t_global)
        if rem > 400:
            alloc = min(0.50 * rem, 650.0)
            out, maxc = grad_ls_run(best_seq, alloc, PHASES_FULL, seed=201, verbose=True)
            best_seq, best_score = update_best_from_max(out, maxc, best_seq, best_score,
                                                        f"long_n{best_seq.size}", t_global)

        # Perturbed long run.
        rem = total_budget - (time.monotonic() - t_global)
        if rem > 300:
            a_pert = noisy_log_sequence(best_seq, seed=202, noise=0.05)
            alloc = min(0.25 * rem, 300.0)
            out, maxc = grad_ls_run(a_pert, alloc, PHASES_FULL, seed=202, verbose=False)
            best_seq, best_score = update_best_from_max(out, maxc, best_seq, best_score,
                                                        f"longpert_n{a_pert.size}", t_global)

        # Try a longer length if time remains.
        rem = total_budget - (time.monotonic() - t_global)
        n_long = int(min(5200, max(best_seq.size + 1200, 3200)))
        if rem > 250 and abs(n_long - best_seq.size) > 200:
            a_long = resample_sequence(best_seq, n_long)
            alloc = min(0.30 * rem, 350.0)
            ph = PHASES_FULL if alloc > 150 else PHASES_MEDIUM
            out, maxc = grad_ls_run(a_long, alloc, ph, seed=301, verbose=False)
            best_seq, best_score = update_best_from_max(out, maxc, best_seq, best_score,
                                                        f"longlen_n{n_long}", t_global)

        # Sharp refinement from slightly perturbed best.
        rem = total_budget - (time.monotonic() - t_global)
        if rem > 120:
            a_pert = noisy_log_sequence(best_seq, seed=402, noise=0.02)
            alloc = min(0.30 * rem, 200.0)
            out, maxc = grad_ls_run(a_pert, alloc, PHASES_SHARP, seed=402, verbose=True)
            best_seq, best_score = update_best_from_max(out, maxc, best_seq, best_score,
                                                        f"sharp_n{a_pert.size}", t_global)

        # Pair polish.
        rem = total_budget - (time.monotonic() - t_global)
        if rem > 80:
            alloc = min(0.35 * rem, 250.0)
            out, maxc = pair_polish(best_seq, alloc, seed=501, verbose=True)
            best_seq, best_score = update_best_from_max(out, maxc, best_seq, best_score,
                                                        f"pair_n{best_seq.size}", t_global)

        # Final mini-pair from a tiny perturbation.
        rem = total_budget - (time.monotonic() - t_global)
        if rem > 50:
            a_pert = noisy_log_sequence(best_seq, seed=502, noise=0.01)
            alloc = min(0.40 * rem, 100.0)
            out, maxc = pair_polish(a_pert, alloc, seed=503, verbose=False)
            best_seq, best_score = update_best_from_max(out, maxc, best_seq, best_score,
                                                        f"pair2_n{a_pert.size}", t_global)

    except Exception as exc:
        print(f"search error: {exc}")

    if best_seq is None or best_seq.size == 0:
        return [1.0]

    best_seq = clean_sequence(best_seq)

    try:
        ev = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(ev):
            print(f"final_evaluated: score={ev:.8f}, n={len(best_seq)}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
