def fast_autocorr(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_peak_gradient(a, top_k=30):
    """Compute weighted subgradient using top-k peaks."""
    n = len(a)
    c = fast_autocorr(a)
    num_lags = len(c)
    k_eff = min(top_k, num_lags)
    top_idx = np.argsort(c)[::-1][:k_eff]
    top_vals = c[top_idx]
    # Softmax weights
    shifted = top_vals - top_vals[0]
    exp_vals = np.exp(10 * shifted)  # beta=10 for sharper focus
    weights = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        # grad[m] = 2 * a[k-m] if 0 <= k-m < n
        m_start = max(0, k_int - n + 1)
        m_end = min(n, k_int + 1)
        if m_start >= m_end:
            continue
        m_idx = np.arange(m_start, m_end)
        a_idx = k_int - m_idx
        grad[m_idx] += 2.0 * wi * a[a_idx]
    return grad, c.max()

def project_simplex(v):

    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def minimize_flatness(a0, time_limit, verbose=True, seed=42):

    momentum = 0.9
    v = np.zeros(n)
    eta = 0.1
    eta_scale = 1.0
    stall = 0
    max_stall = 10000
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:

        # Compute gradient

        # Momentum
        v = momentum * v + (1 - momentum) * grad

        a_new = project_simplex(a - step * v)

        else:
            stall += 1
        
        a = a_new
        
        # Reduce step size on stall
