_global_solver = None

def get_solver():

        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:

def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def compute_objective(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2 directly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * fast_autocorr(a).max() / (s * s)

def make_symmetric(a):
    """Make sequence symmetric about its center."""
    n = len(a)
    a_sym = np.zeros(n, dtype=np.float64)
    for i in range(n):
        a_sym[i] = (a[i] + a[n-1-i]) / 2.0
    return a_sym

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def lp_fw_optimize(a0, time_limit, solver, verbose=True, symmetric=False, 
                   backtrack_points=None):
    """LP-based Frank-Wolfe with backtracking line search."""
    n = len(a0)
    a = norm_sum1(a0)
    if symmetric:
        a = make_symmetric(a)
        a = norm_sum1(a)
    
    best_a = a.copy()
    best_obj = compute_objective(a)
    t0 = time.monotonic()
    
    if backtrack_points is None:
        backtrack_points = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    
    n_iters = 0
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Build constraint matrix
        T = np.zeros((2*n-1, n), dtype=np.float64)
        for i in range(n):
            T[i:i+n, i] = a
        
        ac = fast_autocorr(a)
        M = ac.max()
        if M < 1e-15:
            break
        
        # Solve LP
        b = cp.Variable(n, nonneg=True)
        constraints = [T @ b <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
        s_b = b_val.sum()
        if s_b < 1e-15:
            break
        b_val = b_val / s_b
        
        if symmetric:
            b_val = make_symmetric(b_val)
            b_val = norm_sum1(b_val)
        
        # Backtracking line search
        cur_obj = compute_objective(a)
        best_trial = None
        best_trial_obj = cur_obj
        
        for alpha in backtrack_points:
            a_trial = (1.0 - alpha) * a + alpha * b_val
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            obj_trial = compute_objective(a_trial)
            if obj_trial < best_trial_obj - 1e-12:
                best_trial_obj = obj_trial
                best_trial = a_trial.copy()
        
        if best_trial is not None:
            a = best_trial
            if best_trial_obj < best_obj:
                best_obj = best_trial_obj
                best_a = a.copy()
        
        n_iters += 1
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={n_iters} ({elapsed:.0f}s): obj={best_obj:.6f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    LP-FW done ({elapsed:.0f}s): {n_iters} iters, obj={best_obj:.6f}")
    
    return best_a, best_obj

def soft_subgradient(a, ac, top_k=50, beta=5.0):
    """Compute soft subgradient of max of autocorrelation."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n, dtype=np.float64)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def mirror_descent_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0,
                            restart_every=5000, verbose=True):
    """Nesterov mirror descent with adaptive soft subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = compute_objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        g = soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)
        
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = compute_objective(x)
        if M_new < best_obj - 1e-15:
            best_obj = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 100)
            cur_beta = max(cur_beta - 0.3, 2.0)
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={t} ({elapsed:.0f}s): obj={best_obj:.6f}, stall={stall_count}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    MD done ({elapsed:.0f}s): obj={best_obj:.6f}")
    
    return best_a, best_obj

def coordinate_descent_optimize(a0, time_limit, verbose=True, rng_seed=42):

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()

    stall = 0

        ac = fast_autocorr(a)

        peak_lags = np.argsort(ac)[::-1][:10]
        
        improved = False
        for k in peak_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_contrib_idx = np.argmax(contrib)
            src = idx[max_contrib_idx]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination
            # Sensitivity: for each position d, compute how much it contributes to peaks
            n_sens = min(5, len(peak_lags))
            scores = np.zeros(n)
            for k2 in peak_lags[:n_sens]:
                k2i = int(k2)
                for d in range(n):
                    d1 = k2i - d
                    if 0 <= d1 < n:
                        scores[d] += a[d1]
                    d2 = d - k2i
                    if 0 <= d2 < n:
                        scores[d] += a[d2]
            
            candidates = np.where(scores == scores.min())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            dst = candidates[int(rng.integers(len(candidates)))]
            
            if dst == src:
                continue
            
            # Try moving a small fraction
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:

                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                obj_trial = compute_objective(a_trial)
                if obj_trial < best_obj - 1e-10:
                    a = a_trial
                    best_obj = obj_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))

            if i != j and a[i] > 1e-10:

                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    obj_new = compute_objective(a)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 50 == 0 and stall > 0:

        elapsed = time.monotonic() - t0

def make_power_law(n, power=1.0):
    """Create a power-law shaped sequence."""
    x = np.linspace(0.01, 1.0, n)
    a = x ** power
    return norm_sum1(a)

def make_gaussian(n, sigma=0.2, center=0.5):
    """Create a Gaussian-shaped sequence."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return norm_sum1(a)

def make_bimodal(n, sep=0.3, width=0.1):
    """Create a bimodal sequence."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5 + sep) / width) ** 2) + \
        np.exp(-0.5 * ((x - 0.5 - sep) / width) ** 2) + 0.01
    return norm_sum1(a)

def make_smooth_random(n, seed=0):
    """Create a smooth random sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in [1, 2, 3, 4, 5, 7]:
        coeff = rng.normal(0, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

    except Exception:
        return float('inf')


def compute_objective(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * fast_autocorr(a).max() / (s * s)


def make_symmetric(a):
    n = len(a)
    a_sym = np.zeros(n, dtype=np.float64)
    for i in range(n):
        a_sym[i] = 0.5 * (a[i] + a[n - 1 - i])
    return a_sym


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_constraint_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        T[i:i + n, i] = a
    return T

def soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n, dtype=np.float64)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def mirror_descent_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0,
                            restart_every=5000, verbose=True, seed=42):

    x_prev = a.copy()

    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        obj_new = compute_objective(x)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 100)
            cur_beta = max(cur_beta - 0.3, 2.0)

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD t={t} ({elapsed:.0f}s): obj={best_obj:.6f}, stall={stall_count}")

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"      MD done ({elapsed:.0f}s): obj={best_obj:.6f}")

            if src_val < 1e-12:
                continue

            n_sens = min(4, len(peak_lags))
            scores = np.zeros(n)
            for k2 in peak_lags[:n_sens]:
                k2i = int(k2)
                for d in range(n):
                    d1 = k2i - d
                    if 0 <= d1 < n:
                        scores[d] += a[d1]
                    d2 = d - k2i
