```python
def fast_self_conv(a):

    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]

def compute_gradient(w, a):
    """grad[i] = sum_k w[k] * (a[k-i] + a[i-k])"""
    n = len(a)
    N = 1 << (3 * n - 3).bit_length()  # enough for conv(w, a) without aliasing

    W = np.fft.rfft(w, N)

    conv_wa = np.fft.irfft(W * A, N)
    term2 = conv_wa[:n]
    
    # term1[i] = (w_rev * a)[2n-2-i] for i in [0, n-1]
    conv_wra = np.fft.irfft(Wr * A, N)
    term1 = conv_wra[n-1:2*n-1][::-1]
    
    return term1 + term2

def softmax_params_to_a(x):
    """Convert soft parameters to probability vector."""
    # Use log-sum-exp for numerical stability
    xmax = x.max()
    ex = np.exp(x - xmax)
    return ex / ex.sum()

def a_to_softmax_params(a):
    """Convert probability vector to soft parameters (log)."""
    a = np.maximum(a, 1e-30)
    return np.log(a)

def objective_and_grad(x, beta):
    """Compute objective and gradient in x-space (softmax parameterization)."""
    a = softmax_params_to_a(x)
    n = len(a)
    b = fast_self_conv(a)
    
    # Log-sum-exp of b
    bmax = b.max()
    log_sum_exp = bmax + np.log(np.sum(np.exp(beta * (b - bmax))))
    F = log_sum_exp / beta
    
    # Weights
    w = np.exp(beta * (b - bmax))
    w /= w.sum()
    
    # Gradient w.r.t. a
    grad_a = compute_gradient(w, a)
    
    # Gradient w.r.t. x: grad_x[i] = grad_a[i] * a[i] - a[i] * sum_j grad_a[j] * a[j]
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def fast_self_conv(a):
    """Compute a * a (linear convolution) using FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0]**2])
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]

def compute_grad_weights(w, a):
    """Compute grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) via FFT."""
    n = len(a)
    nw = len(w)  # = 2n-1
    # Need FFT size >= nw + n - 1 = 3n - 2
    N = 1 << (3 * n - 3).bit_length()
    
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    Wr = np.fft.rfft(w[::-1], N)
    
    # term2[i] = (w * a)[i] for i in [0, n-1]
    conv_wa = np.fft.irfft(W * A, N)
    term2 = conv_wa[:n]
    
    # term1[i] = (w_rev * a)[nw-1-i] for i in [0, n-1]
    conv_wra = np.fft.irfft(Wr * A, N)
    term1 = conv_wra[n-1:nw][::-1]
    
    return term1 + term2

def softmax_a(x):
    """Stable softmax."""
    x_shifted = x - x.max()
    ex = np.exp(x_shifted)
    return ex / ex.sum()

def objective_and_grad(x, beta, top_k=None):
    """
    Compute F(x) = (1/beta) * log(sum_k in top_k exp(beta * b_k))
    and its gradient w.r.t. x.
    """
    a = softmax_a(x)
    n = len(a)
    b = fast_self_conv(a)
    
    if top_k is not None and top_k < len(b):
        # Only use top_k lags
        idx = np.argsort(b)[::-1][:top_k]
        b_use = b[idx]
    else:
        b_use = b
    
    bmax = b_use.max()
    exp_vals = np.exp(beta * (b_use - bmax))
    lse = bmax + np.log(exp_vals.sum())
    F = lse / beta
    
    # Weights (over all lags, but zero for non-top if top_k specified)
    if top_k is not None and top_k < len(b):
        w = np.zeros(len(b))
        w[idx] = exp_vals / exp_vals.sum()
    else:
        w = np.exp(beta * (b - bmax))
        w /= w.sum()
    
    grad_a = compute_grad_weights(w, a)
    grad_x = a * (grad_a - np.dot(grad_a, a))
    
    return F, grad_x
```

```python
    """
    x has length n-1. a = [x_0, ..., x_{n-2}, 1 - sum(x_0..x_{n-2})].
    Minimize F(a) - mu * sum(log(a_i)) subject to a_i > 0.
    """

    a = np.empty(n)
    a[:n-1] = x
    a[n-1] = 1.0 - x.sum()
    
    # Check feasibility
    if a.min() <= 1e-15:
        return 1e10, np.zeros_like(x)
    
    b = fast_self_conv(a)
    bmax = b.max()
    exp_vals = np.exp(beta * (b - bmax))
    lse = bmax + np.log(exp_vals.sum())
    F = lse / beta
    
    if mu > 0:
        F -= mu * np.sum(np.log(a))
    
    w = exp_vals / exp_vals.sum()
    grad_a = compute_grad_weights(w, a)
    
    if mu > 0:
        grad_a -= mu / a
    
    # Convert to x-space: a_n = 1 - sum(x)
    # da_i/dx_j = delta_{ij} for i < n-1
    # da_n/dx_j = -1
    # dF/dx_j = grad_a[j] - grad_a[n-1] for j = 0, ..., n-2
    grad_x = grad_a[:n-1] - grad_a[n-1]
    
    return F, grad_x
```

`term2[m] = sum_k w[k] * a[m-k]` (valid: 0<=k<5, 0<=m-k<3)
- `m=0`: `w[0]*a[0] = 0.1*1 = 0.1`
- `m=1`: `w[1]*a[0] + w[0]*a[1] = 0.2*1 + 0.1*2 = 0.4`
- `m=2`: `w[2]*a[0] + w[1]*a[1] + w[0]*a[2] = 0.3*1 + 0.2*2 + 0.1*3 = 0.8`

`term1[m] = sum_k w[k] * a[k-m]` (valid: 0<=k<5, 0<=k-m<3)
- `m=0`: `w[0]*a[0] + w[1]*a[1] + w[2]*a[2] = 0.1*1 + 0.2*2 + 0.3*3 = 1.4`
- `m=1`: `w[1]*a[0] + w[2]*a[1] + w[3]*a[2] = 0.2*1 + 0.3*2 + 0.2*3 = 1.4`
- `m=2`: `w[2]*a[0] + w[3]*a[1] + w[4]*a[2] = 0.3*1 + 0.2*2 + 0.1*3 = 0.8`

`grad = term1 + term2 = [1.5, 1.8, 1.6]`
