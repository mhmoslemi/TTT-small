def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def objective_from_autocorr(a):
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (a.sum() ** 2)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def make_u_shaped(n, sharpness=2.0):
    x = np.linspace(0, 1, n)
    a = (x ** sharpness) + ((1.0 - x) ** sharpness) + 0.01
    return norm_sum1(a)


def make_edge_concentrated(n, decay=0.02):

    a = (1.0 - x ** sharpness) * x ** sharpness + 0.01

def compute_exact_subgrad(a, ac, n_max=5):
    n = len(a)
    max_val = ac.max()
    top_lags = np.argsort(ac)[::-1][:n_max]
    weights = ac[top_lags] / ac[top_lags].sum()
    grad = np.zeros(n, dtype=np.float64)
    for w, k in zip(weights, top_lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start >= i_end:
            idx = np.array([i_start])
        else:
            idx = np.arange(i_start, i_end + 1)
        j_idx = k_int - idx
        grad[idx] += 2.0 * w * a[j_idx]
    return grad


def compute_soft_subgradient(a, ac, top_k=80, beta=4.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n, dtype=np.float64)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def simplex_project(v, target_sum=1.0):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 0:
        return v
    mu = np.zeros(n)
    for _ in range(50):
        mu = np.maximum(0.0, v - mu)
        s = mu.sum()
        if abs(s - target_sum) < 1e-12:
            break
        mu = v - (s - target_sum) / n
        mu = np.maximum(mu, 0.0)
    return np.maximum(v - (mu.sum() - target_sum) / n, 0.0)


def amd_optimize(a0, time_limit, top_k=60, beta=4.0, c0=3.5, restart_every=6000, verbose=False, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    momentum = 0.0
    momentum_rate = 0.999

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        momentum = momentum_rate * momentum + (1.0 - momentum_rate) * beta_t
        y = x + momentum * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            pert_scale = min(0.05, 0.002 * (stall_count // restart_every + 1))
            pert = rng.normal(0, pert_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 20, 150)
            cur_beta = max(cur_beta - 0.3, 1.5)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}, top_k={cur_top_k}, beta={cur_beta:.1f}")

        if verbose and t % 10000 == 0:

            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=300, max_constraints=None):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    if max_constraints is None:
        max_constraints = min(2 * n - 1, max(top_K, 500))
    top_K = min(top_K, max_constraints)

    y = a.copy()
    x = a.copy()
    prev_y = a.copy()

    T_full = np.zeros((2 * n - 1, n))

    def rebuild_T(vec):
        T_full[:] = 0.0
        for j in range(n):
            T_full[j:j + n, j] = vec

    rebuild_T(y)

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        rebuild_T(y)
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        constraint_vals = T_full @ y
        margins = M_curr - constraint_vals
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]
        M_sel = np.full(K_sel, M_curr)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        prev_y = y
        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123, n_peak_lags=15):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500
    iter_count = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_peak_lags]

        improved_this_round = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:

                    if 0 <= d1 < n:
                        sensitivity[d] += a[d1]
            # Also account for the self-contribution
            sensitivity += 2.0 * a * 0.1

            # Find positions with lowest sensitivity that are not the source
            sensitivity[src] = np.inf
            # Prefer positions far from src
            dst = np.argmin(sensitivity)

            if dst == src or sensitivity[dst] > 1e10:
                # Try random destination

                if len(candidates) > 0:
                    dst = candidates[rng.integers(len(candidates))]
                else:
                    dst = int(rng.integers(n))
                    if dst == src:
                        dst = (dst + 1) % n

            # Try multiple fractions
            found = False
            for frac in [0.002, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved_this_round = True
                    found = True
                    stall = 0
                    break

            if not found:
                # Try reverse direction: move from dst to src (if dst has mass)
                if a[dst] > 1e-10:
                    for frac in [0.005, 0.01, 0.02]:
                        amt = a[dst] * frac
                        a_trial = a.copy()
                        a_trial[dst] -= amt
                        a_trial[src] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        a_trial = a_trial / a_trial.sum()
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved_this_round = True
                            found = True
                            stall = 0
                            break

        if not improved_this_round:
            stall += 1
            # Random perturbation to escape
            n_moves = rng.integers(1, 4)
            for _ in range(n_moves):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved_this_round = True

        iter_count += 1
        if verbose and (stall % 100 == 0 or iter_count % 50 == 0):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def subgradient_optimize(a0, time_limit, verbose=False, rng_seed=77):
    """Projected subgradient with momentum and backtracking."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    momentum_vec = np.zeros(n)
    momentum_alpha = 0.9
    eta = 0.01
    backtrack_factor = 0.7
    max_backtracks = 10
    stall = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        g = compute_exact_subgrad(a, ac, n_max=3)

        momentum_vec = momentum_alpha * momentum_vec + (1.0 - momentum_alpha) * g

        # Backtracking line search
        step = eta
        accepted = False
        for _ in range(max_backtracks):
            a_trial = a - step * momentum_vec
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                a_trial = a.copy()
            else:
                a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            M_curr = ac.max()
            if M_trial < M_curr - 1e-14:
                a = a_trial
                accepted = True
                break
            step *= backtrack_factor

        if not accepted:
            stall += 1
            if stall > 500:
                # Restart with perturbation
                pert = rng.normal(0, 0.01, n) * a
                a = a + pert
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                momentum_vec = np.zeros(n)
                eta = 0.01
                stall = 0
            continue
        else:
            stall = 0
            # Slowly increase step size
            eta = min(eta * 1.001, 0.05)

        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Subgrad t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def two_point_coordinate_descent(a0, time_limit, verbose=False, rng_seed=555):
    """Optimize pairs of coordinates simultaneously."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:10]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if len(idx) < 2:
                continue

            # Find top 2 contributing pairs
            top2_idx = np.argsort(contrib)[::-1][:2]
            for ti in top2_idx:
                if time.monotonic() - t0 > time_limit:
                    break
                i1 = idx[ti]
                i2 = k_int - i1
                if i1 == i2:
                    continue
                if a[i1] < 1e-10 or a[i2] < 1e-10:
                    continue

                # Try to rebalance: move mass from larger to smaller
                if a[i1] > a[i2]:
                    src, dst = i1, i2
                else:
                    src, dst = i2, i1

                # Also try moving both to a third location
                for third in [dst, int(rng.integers(n))]:
                    for frac in [0.01, 0.03, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[third] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-15:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0

        if not improved:
            stall += 1
            # Random two-point move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m
