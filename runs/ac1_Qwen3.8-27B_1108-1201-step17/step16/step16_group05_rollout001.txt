def fast_autocorr(a):

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum() if f.sum() > 1e-15 else np.ones(old_n) / old_n
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_impact_scores(a, top_lags):
    """For each position j, compute sum of a[j+k] + a[j-k] over top lags k."""
    n = len(a)
    scores = np.zeros(n)

        if k == 0:
            scores += 2.0 * a
        else:
            # a[j+k] valid for j = 0..n-1-k
            if k < n:
                scores[:n - k] += a[k:]
                # a[j-k] valid for j = k..n-1
                scores[k:] += a[:n - k]
    return scores

            break

        # Go to frequency domain
        A = np.fft.rfft(a)
        power = np.abs(A) ** 2
        mean_power = power.mean()

        # Compute autocorrelation peaks
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Identify peak lags
        top_k = min(10, len(ac))
        peak_lags = np.argsort(ac)[::-1][:top_k]

        # For each peak lag k, the autocorrelation is IDFT of power at that frequency
        # We want to reduce power at frequencies that align with peak lags
        # Simple approach: reduce power at peaks of the spectrum
        # and increase at valleys, then project back

        # Flatten: new_power = mean + alpha * (power - mean) * decay
        # Where decay is smaller for frequencies that contribute to peaks
        alpha = 0.1  # Flattening strength
        new_power = power.copy()
        new_power = alpha * power + (1 - alpha) * mean_power

        # Reconstruct
        new_mag = np.sqrt(new_power)
        new_phase = np.angle(A)
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = np.fft.irfft(A_new, n)

        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      Spectral ({elapsed:.0f}s) it={it}: J={J:.6f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      Spectral done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def apg_optimize(a0, time_limit, beta_start=30.0, beta_end=8.0,
                 top_k=200, c0=2.0, verbose=True, seed=42):
    """Nesterov-accelerated projected gradient with adaptive smoothed-max."""
    n = len(a0)
    a = normalize_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 4000
    current_top_k = top_k

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / 30000.0)
        cur_beta = beta_start + (beta_end - beta_start) * frac

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)

        # Compute smoothed gradient
        ac = fast_autocorr(y)
        k_eff = min(current_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_eff]
        top_vals = ac[top_idx]
        shifted = top_vals - top_vals[0]
        exp_vals = np.exp(cur_beta * shifted)
        w_sum = exp_vals.sum()
        if w_sum < 1e-30:
            weights = np.ones(k_eff) / k_eff
        else:
            weights = exp_vals / w_sum

        grad = np.zeros(n)
        for wi, k in zip(weights, top_idx):
            k_int = int(k)
            if k_int == 0:
                grad += 2.0 * wi * y
            elif k_int < n:
                seg = n - k_int
                grad[:seg] += 2.0 * wi * y[k_int:k_int + seg]
                grad[k_int:k_int + seg] += 2.0 * wi * y[:seg]

        # Step
        eta = c0 / (np.sqrt(t) + 1.0)
        x_new = project_simplex(y - eta * grad)

        x_prev = x
        x = x_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.4)
            noise = rng.normal(0, 0.003, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            current_top_k = min(current_top_k + 30, 400)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      APG restart t={t} ({elapsed:.0f}s): J={J:.6f}, tk={current_top_k}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      APG t={t} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.1f}")

    return best_a, best_m


def coarse_to_fine(a_best, n_final, time_limit, verbose=True):
    """Multi-scale: optimize at coarse resolution, then refine at fine."""
    # Start at coarse
    n_coarse = max(200, n_final // 10)
    a_coarse = interpolate_to_length(a_best, n_coarse)
    t_coarse = min(time_limit * 0.3, 120.0)

    if t_coarse > 20:
        a_coarse, _ = apg_optimize(a_coarse, t_coarse, beta_start=25.0,
                                    beta_end=10.0, top_k=100, verbose=False)

    # Interpolate to medium
    n_med = max(400, n_final // 3)
    a_med = interpolate_to_length(a_coarse, n_med)
    rem = time_limit - (time.monotonic() - (time.monotonic() - t_coarse))
    t_med = min(rem * 0.4, 150.0)

    if t_med > 20:
        a_med, _ = apg_optimize(a_med, t_med, beta_start=25.0,
                                 beta_end=10.0, top_k=150, verbose=False)

    # Interpolate to fine and polish
    a_fine = interpolate_to_length(a_med, n_final)
    t_fine = max(0, time_limit - t_coarse - t_med - 10)
    if t_fine > 15:
        a_fine, _ = peak_shave_coordinate(a_fine, t_fine, verbose=verbose)

    return a_fine


def get_parent_sequence():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p.copy()
    except Exception:
        pass
    return None


def propose_candidate(budget_s=1800.0):
    """Main search: multi-length peak-shaving with spectral and gradient methods."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    print(f"Time budget: {time_budget:.0f}s")

    # Get initial sequence
    parent = get_parent_sequence()
    if parent is not None:
        best_seq = normalize_sum1(parent)
        best_score = evaluate_safe(best_seq)
        n0 = len(parent)
        print(f"Parent: n={n0}, score={best_score:.6f}")
    else:
        n0 = 2400
        x = np.linspace(0, 1, n0)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n0}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: Try multiple lengths with quick APG ===
    rem = remaining_time()
    print(f"\n=== Phase 1: Multi-length search ===")
    n_candidates = list(range(max(1600, n0 - 800), n0 + 1601, 400))
    # Include original and some variations
    n_candidates = sorted(set(n_candidates))
    t_per = min(rem * 0.20 / max(len(n_candidates), 1), 60.0)
    
    for n_try in n_candidates:
        rem = remaining_time()
        if rem < 30:
            break
        t_alloc = min(t_per, rem - 10)
        if t_alloc < 15:
            break
        a_init = interpolate_to_length(best_seq, n_try)
        a_out, _ = apg_optimize(a_init, t_alloc, beta_start=25.0,
                                 beta_end=10.0, top_k=150, verbose=False)
        try_update(a_out, f"apg_n{n_try}")

    # === Phase 2: Peak shaving on best ===
    rem = remaining_time()
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Peak shaving n={n_best} ===")
    t2 = min(rem * 0.25, 350.0)
    if t2 > 30:
        a_out, _ = peak_shave_coordinate(best_seq, t2, verbose=True, seed=42)
        try_update(a_out, f"peakshave_n{n_best}")

        # Second peak shaving pass with different seed
        rem = remaining_time()
        t2b = min(rem * 0.12, 150.0)
        if t2b > 20:
            a_out2, _ = peak_shave_coordinate(a_out, t2b, verbose=False, seed=999)
            try_update(a_out2, f"peakshave2_n{len(a_out)}")

    # === Phase 3: Spectral flatten ===
    rem = remaining_time()
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Spectral n={n_best} ===")
    t3 = min(rem * 0.10, 100.0)
    if t3 > 20:
        a_out, _ = spectral_flatten(best_seq, t3, verbose=True)
        try_update(a_out, f"spectral_n{n_best}")

        # Follow up with peak shaving on spectral result
        rem = remaining_time()
        if rem > 30:
            t3b = min(rem * 0.10, 100.0)
            if t3b > 15 and len(a_out) == n_best:
                a_out2, _ = peak_shave_coordinate(a_out, t3b, verbose=False, seed=777)
                try_update(a_out2, f"spec+peak_n{n_best}")

    # === Phase 4: APG with high beta (peak-focused) ===
    rem = remaining_time()
    n_best = len(best_seq)
    print(f"\n=== Phase 4: High-beta APG n={n_best} ===")
    t4 = min(rem * 0.18, 250.0)
    if t4 > 30:
        a_out, _ = apg_optimize(best_seq, t4, beta_start=50.0,
                                 beta_end=15.0, top_k=300,
                                 c0=2.5, verbose=True, seed=123)
        try_update(a_out, f"apg_hi_n{n_best}")

    # === Phase 5: Coarse-to-fine at longer length ===
    rem = remaining_time()
    n_long = min(n_best + 1200, 4000)
    if n_long > n_best and rem > 200:
        print(f"\n=== Phase 5: Coarse-to-fine n={n_best}->{n_long} ===")
        t5 = min(rem * 0.25, 300.0)
        a_out = coarse_to_fine(best_seq, n_long, t5, verbose=True)
        try_update(a_out, f"coarse_fine_n{n_long}")

        # Polish the result
        rem = remaining_time()
        if rem > 40:
            t5b = min(rem * 0.15, 150.0)
            if t5b > 20:
                a_out2, _ = peak_shave_coordinate(a_out, t5b, verbose=True, seed=555)
                try_update(a_out2, f"cf_peak_n{n_long}")

    # === Phase 6: Final rounds of peak shaving on global best ===
    rem = remaining_time()
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_best} ===")
    t6 = min(rem * 0.30, 200.0)
    if t6 > 20:
        a_out, _ = peak_shave_coordinate(best_seq, t6, verbose=True, seed=2024)
        try_update(a_out, f"final_peak_n{n_best}")

    # === Phase 7: Another attempt with different strategy ===
    rem = remaining_time()
    if rem > 60:
        print(f"\n=== Phase 7: Alternate approach ===")
        n_best = len(best_seq)
        # Try: blend best with a smooth bump and re-optimize
        x = np.linspace(0, 1, n_best)
        bump = np.exp(-50 * (x - 0.5) ** 2)
        bump = bump / bump.sum()
        for blend_w in [0.1, 0.2, 0.3]:
            rem = remaining_time()
            if rem < 40:
                break
            a_blend = (1 - blend_w) * normalize_sum1(best_seq) + blend_w * bump
            a_blend = normalize_sum1(a_blend)
            t7 = min(rem * 0.3, 80.0)
            if t7 > 20:
                a_out, _ = peak_shave_coordinate(a_blend, t7, verbose=False, seed=3000)
                try_update(a_out, f"blend{blend_w}_n{n_best}")

    # === Final validation ===
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    # Ensure all values are valid
    result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
    return result
