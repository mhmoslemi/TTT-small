    if ss < 1e-15:

        # This is 2*beta * (w * a) (cross-correlation)
        g = 2.0 * beta * np.convolve(w, a, mode='full')[:n]
        
        # Wait, let me verify: sum_k w_k * a[k-j] where k ranges over all lags
        # k-j must be in [0, n-1], so k in [j, j+n-1]
        # This is the cross-correlation of w and a
        # np.convolve(w, a) gives (w*a)[m] = sum_k w[k]*a[m-k]
        # We want sum_k w[k]*a[k-j] = sum_k w[k]*a[(k-j)]
        # Let m = k-j, then k = m+j: sum_m w[m+j]*a[m]
        # This is the cross-correlation: np.correlate(w, a, 'full') or similar
        
        # Actually, let me just use the formula directly:
        # g[j] = 2*beta * sum_{k: 0<=k-j<n} w[k] * a[k-j]
        # = 2*beta * sum_{m=0}^{n-1} w[m+j] * a[m] (where m = k-j)
        # This is the (j)-th element of the cross-correlation of w and a
        
        # np.correlate(w, a, mode='full')[n-1+j] = sum_m w[m] * a[m-j]... 
        # Hmm, let me just use the direct computation for correctness
        
        # Actually the efficient way: 
        # g[j] = 2*beta * sum_{k=j}^{j+n-1} w[k] * a[k-j] (for k in [0, 2n-2])
        # = 2*beta * (w_shifted * a)[j] where w_shifted is w zero-padded
        # This is a convolution: g = 2*beta * convolve(w, a, 'full')[n-1 : n-1+n]
        # Wait no. Let me think again.
        
        # ac[k] = sum_i a[i]*a[k-i] for valid i
        # d(ac[k])/d(a[j]) = 2*a[k-j] if 0<=k-j<n
        # g[j] = sum_k beta*w[k]*d(ac[k])/d(a[j]) = 2*beta*sum_k w[k]*a[k-j] (for 0<=k-j<n)
        
        # Let's define: for each j, g[j] = 2*beta * sum_{k=max(j,0)}^{min(2n-2, j+n-1)} w[k]*a[k-j]
        # = 2*beta * sum_{m=0}^{n-1} w[j+m]*a[m]
        
        # This is the cross-correlation (w circ a)[j] = sum_m w[j+m]*a[m]
        # In numpy: np.correlate(w_padded, a, 'full') where w_padded = w
        
        # Actually: np.correlate(w, a, mode='same') or 'full'
        # np.correlate(w, a, 'full')[k] = sum_m w[m] * a[m - (k - (len(w)-1))]... this is confusing
        
        # Let me just use FFT:
        # g[j] = 2*beta * sum_{m=0}^{n-1} w[j+m] * a[m]
        # = 2*beta * IFFT(FFT(w) * conj(FFT(a)))[j]  (cross-correlation)
        
        # Actually, cross-correlation R[j] = sum_m w[j+m]*a[m] = (w * a_flipped)[j + n - 1]
        # where * denotes convolution and a_flipped = a[::-1]
        
        # Let me just use a direct FFT approach:
        # R[j] = sum_m w[j+m]*a[m] for m=0..n-1
        # This equals the (j)-th element of the "reverse cross-correlation"
        # = IFFT(FFT(w) * conj(FFT(a)))[j] (with proper zero-padding)
        
        fft_size = 1 << (2*n - 1).bit_length() if 2*n-1 > 1 else 1
        W = np.fft.rfft(w, fft_size)
        A = np.fft.rfft(a, fft_size)
        # Cross-correlation: sum_m w[m+j]*a[m]
        # = sum_m w[m+j]*a[m] = IFFT(W * conj(A))[j]  (with zero-padding to avoid circular)
        cross = np.fft.irfft(W * np.conj(A), fft_size)[:2*n-1]
        # cross[k] = sum_m w[m]*a[m-k] for appropriate range
        # Hmm, I need to be careful about the definition.
        
        # Let me just do it the simple way with convolve:
        # g[j] = 2*beta * sum_{m=0}^{n-1} w[j+m]*a[m]
        # = 2*beta * np.convolve(w, a, 'full')[n-1+j]  ... let me verify
        
        # np.convolve(w, a, 'full')[k] = sum_i w[i]*a[k-i]
        # For k = n-1+j: sum_i w[i]*a[n-1+j-i]
        # Let m = n-1-j+i, then i = m+n-1-j: sum_m w[m+n-1-j]*a[m]
        # Hmm, that's sum_m w[m+(n-1-j)]*a[m]
        # I want sum_m w[j+m]*a[m]
        # So I need n-1-j = j, i.e., j = (n-1)/2. That doesn't work in general.
        
        # OK let me just use the other order:
        # np.convolve(a, w, 'full')[k] = sum_i a[i]*w[k-i]
        # For k = j: sum_i a[i]*w[j-i]
        # Let m = i: sum_m a[m]*w[j-m] = sum_m w[j-m]*a[m]
        # I want sum_m w[j+m]*a[m]
        # So I need w[j-m] = w[j+m], which means I need to use flipped w.
        
        # Alternative: g[j] = 2*beta * sum_m a[m]*w[j+m]
        # = 2*beta * np.convolve(a, w[::-1], 'full')[?]
        # np.convolve(a, w_flipped, 'full')[k] = sum_i a[i]*w_flipped[k-i] = sum_i a[i]*w[n-1-(k-i)] = sum_i a[i]*w[n-1-k+i]
        # For k = j+n-1: sum_i a[i]*w[i]... no that's not right either.
        
        # You know what, let me just compute it directly for clarity and correctness:
        g = np.zeros(n)
        for j in range(n):
            # g[j] = 2*beta * sum_{m=0}^{n-1} w[j+m]*a[m]
            # where w has length 2n-1, so w[j+m] is valid for j+m < 2n-1
            max_m = min(n, 2*n-1-j)
            g[j] = 2.0 * beta * np.dot(w[j:j+max_m], a[:max_m])
        
        # This is O(n^2) per iteration which is too slow for n=2400
        # Let me use FFT properly
        
        # The correct FFT computation:
        # g[j] = 2*beta * sum_{m=0}^{n-1} w[j+m]*a[m]
        # Let's define v[m] = a[m] for m=0..n-1, 0 otherwise
        # Then g[j] = 2*beta * sum_{m=0}^{n-1} w[j+m]*v[m]
        # = 2*beta * (w * v_flipped)[(2n-2) - j]  ... no
        # Actually: cross-correlation (w ⊞ v)[j] = sum_m w[j+m]*v[m]
        # This is the "reverse cross-correlation" or "cross-correlation with sign flip"
        # In FFT terms: IFFT(FFT(w) * conj(FFT(v)))[j] gives sum_m w[m]*conj(v[m-j])
        # For real signals: sum_m w[m]*v[m-j]
        # That's not quite what I want. I want sum_m w[j+m]*v[m].
        
        # Let's define u[k] = w[k] for k=0..2n-2
        # sum_m u[j+m]*v[m] = sum_m u[j+m]*v[m]
        # Let l = j+m, then m = l-j: sum_l u[l]*v[l-j]
        # = (u * v_flipped)[l] evaluated at... 
        # v_flipped[m] = v[n-1-m]
        # (u * v_flipped)[k] = sum_l u[l]*v_flipped[k-l] = sum_l u[l]*v[n-1-k+l]
        # Set n-1-k+l = l-j => n-1-k = -j => k = n-1+j
        # So (u * v_flipped)[n-1+j] = sum_l u[l]*v[l-j]
        # And I want sum_l u[l]*v[l-j] which is exactly what I have!
        
        # So: g[j]/(2*beta) = (w * a_flipped)[n-1+j] where a_flipped = a[::-1]
        # Wait, v = a, v_flipped = a[::-1]
        # (w * a_flipped)[n-1+j] = sum_l w[l]*a_flipped[n-1+j-l] = sum_l w[l]*a[n-1-(n-1+j-l)] = sum_l w[l]*a[l-j]
        # = sum_l w[l]*a[l-j]
        # I want sum_m w[j+m]*a[m]. Let l = j+m, m = l-j:
        # = sum_l w[l]*a[l-j] ✓
        
        # So: g = 2*beta * np.convolve(w, a[::-1], 'full')[n-1 : n-1+n]
        
        g = 2.0 * beta * np.convolve(w, a[::-1], mode='full')[n-1:n-1+n]
        # But wait, I computed y above, not a. Let me use y for the current iterate.
        
        # Actually I should use y (the Nesterov point) for the gradient computation
        # Let me redo with y instead of a
        g = 2.0 * beta * np.convolve(w, y[::-1], mode='full')[n-1:n-1+n]
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_y = np.log(y + 1e-30)
        new_log = log_y - eta * g
        new_log -= new_log.max()
        a_new = np.exp(new_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = y
        x = a_new
        
        # Check improvement
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 5000:
            stall = 0
            pert = rng.normal(0, 0.003, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LSE-MD t={t} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}")
    
    return best_a, best_m


def gerchberg_saxton(a0, time_limit, n_iter_max=5000, verbose=True):
    """
    Gerchberg-Saxton style iteration:
    1. Compute FFT of a
    2. Modify power spectrum to be flatter
    3. Reconstruct with original phase
    4. Clip to non-negative
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a)
        P = A.real**2 + A.imag**2
        # Target: flatten P
        P_target = np.median(P) * np.ones_like(P)
        # Blend
        blend = 0.3
        P_new = (1-blend)*P + blend*P_target
        # New amplitude
        amp_new = np.sqrt(P_new)
        # Keep original phase
        phase = np.angle(A)
        A_new = amp_new * np.exp(1j*phase)
        a_new = np.real(np.fft.irfft(A_new, n))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
        a = a_new
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    GS it={it} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def solve_lp_direction(y, M, active_lags, solver):
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(b_var) == 1]
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var) * 0.0), constraints)
    # Actually, with sum(b)=1, we want to minimize max_k (y*b)[k]
    # But that's a second variable. Let's use the original formulation:
    # max sum(b) s.t. (y*b)[k] <= M for active k, b >= 0
    # This gives a direction that "stays within" the current peak level
    
    b_var2 = cp.Variable(n, nonneg=True)
    constraints2 = []
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints2.append(y_coeff @ b_var2[j_idx] <= M)
    
    prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), constraints2)
    try:
        prob2.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob2.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def lp_franck_wolfe(a0, time_limit, solver, top_k_init=80, top_k_max=250, verbose=True):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    current_top_k = top_k_init
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        if t == 1:
            y = x.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()
        
        active_lags = np.argsort(ac)[::-1][:current_top_k]
        direction = solve_lp_direction(y, M_curr, active_lags, solver)
        if direction is None:
            # Fallback: use soft subgradient direction
            top_idx = np.argsort(ac)[::-1][:current_top_k]
            top_vals = np.maximum(ac[top_idx], 1e-30)
            w = top_vals ** 5.0
            w = w / w.sum()
            g = np.zeros(n)
            for wi, k in zip(w, top_idx):
                k_int = int(k)
                i_start = max(0, k_int - n + 1)
                i_end = min(n, k_int + 1)
                idx = np.arange(i_start, i_end)
                j_idx = k_int - idx
                g[idx] += 2.0 * wi * y[j_idx]
            # Mirror descent step
            eta = 3.0 / (np.sqrt(t) + 1.0)
            log_y = np.log(y + 1e-30)
            new_log = log_y - eta * g
            new_log -= new_log.max()
            direction = np.exp(new_log)
            direction = np.maximum(direction, 1e-30)
            direction = direction / direction.sum()
        
        alpha = 2.0 / (t + 2.0)
        x_new = (1.0 - alpha) * y + alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx
        
        x_prev = y
        x = x_new
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if t % 100 == 0 and current_top_k < top_k_max:
            current_top_k = min(current_top_k + 10, top_k_max)
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}")
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(30, len(ac))]
        improved = False
        
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find best destination
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)
            perm = rng.permutation(len(candidates))[:min(300, len(candidates))]
            chosen = candidates[perm]
            
            best_dst = -1
            best_score = float('inf')
            
            for dst in chosen:
                if dst == src:
                    continue
                sens = 0.0
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    d1 = k2i - dst
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    sens += add
                if sens < best_score:
                    best_score = sens
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            dst = best_dst
            found = False
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04, 0.06]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if not found and not improved:
                pass
        
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / max(a.sum(), 1e-15)
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def water_filling(a0, time_limit, verbose=True):
    """
    Water-filling: iteratively equalize the top autocorrelation peaks
    by redistributing mass from contributing positions.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
        
        # Get top peaks
        sorted_idx = np.argsort(ac)[::-1]
        top_peaks = sorted_idx[:min(20, len(ac))]
        peak_vals = ac[top_peaks]
        avg_peak = peak_vals.mean()
        
        # For each top peak, try to reduce it
        for k in top_peaks[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            if ac[k_int] < avg_peak * 0.9:
                continue
                
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            total = contrib.sum()
            if total < 1e-15:
                continue
            
            # Find the most contributing pair and reduce it
            max_i = np.argmax(contrib)
            src_pos = idx[max_i]
            partner = k_int - src_pos
            
            # Try to move mass from src_pos to a random non-contributing position
            in_this_peak = np.zeros(n, dtype=bool)
            in_this_peak[js:je+1] = True
            
            # Also mark positions that contribute to other top peaks
            for k2 in top_peaks[:10]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_this_peak[s2:e2+1] = True
            
            free = np.where(~in_this_peak)[0]
            if len(free) == 0:
                continue
            
            dst = free[rng.integers(len(free))]
            
            for frac in [0.005, 0.01, 0.02]:
                amt = a[src_pos] * frac
                if amt < 1e-18:
                    break
                a_trial = a.copy()
                a_trial[src_pos] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    break
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    break
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    WaterFill ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m
