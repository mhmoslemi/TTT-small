import time

def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_objective_and_grad(w, t):

    Compute the log-sum-exp smooth objective and its gradient w.r.t. w,
    where a = softmax(w).
    """
    n = len(w)
    # Compute a = softmax(w)
    w_shifted = w - w.max()
    exp_w = np.exp(w_shifted)
    a = exp_w / exp_w.sum()
    
    # Compute C = conv(a, a)
    C = fast_conv(a)
    
    # Compute log-sum-exp
    tC = t * C
    tC_shifted = tC - tC.max()
    log_sum_exp = tC.max() + np.log(np.sum(np.exp(tC_shifted)))
    f = log_sum_exp / t
    
    # Compute p_k = exp(t*C[k]) / sum exp(t*C[k])
    p = np.exp(tC_shifted) / np.sum(np.exp(tC_shifted))

    # Compute df/dw_j = a_j * (df_da_j - sum_i df_da_i * a_i)
    direction = np.dot(df_da, a)
    df_dw = a * (df_da - direction)
    
    return f, df_dw, a

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax_fn(w):
    w_shifted = w - w.max()
    exp_w = np.exp(w_shifted)
    s = exp_w.sum()
    return exp_w / s

def smooth_obj_grad(w, t):
    n = len(w)
    a = softmax_fn(w)
    C = fast_conv(a)
    tC = t * C
    tC_max = tC.max()
    exp_tC = np.exp(tC - tC_max)
    sum_exp = exp_tC.sum()
    f = (tC_max + np.log(sum_exp)) / t
    p = exp_tC / sum_exp
    
    fft_size = 1 << (2*n - 1 - 1).bit_length() if n > 1 else 2
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(P * np.conj(A), fft_size)[:n]
    
    df_da = 2.0 * cross
    direction = np.dot(df_da, a)
    df_dw = a * (df_da - direction)
    
    return f, df_dw

def lbfgs_anneal(a0, time_budget, t_schedule=None, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    w = np.log(a + 1e-30)
    
    if t_schedule is None:
        t_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0]
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t_start = time.monotonic()
    
    for t in t_schedule:
        elapsed = time.monotonic() - t_start
        if elapsed > time_budget:
            break
        remaining = time_budget - elapsed
        if remaining < 5:
            break
        
        # Adaptive max iterations
        max_iter = min(300, max(20, int(remaining * 30)))
        
        try:
            result = minimize(smooth_obj_grad, w, args=(t,),
                            method='L-BFGS-B', jac=True,
                            options={'maxiter': max_iter, 'gtol': 1e-12, 'maxfun': max_iter * 3,
                                    'ftol': 1e-14})
            if result.x is not None:
                w = result.x
        except Exception:
            pass
        
        a_curr = norm_sum1(softmax_fn(w))
        M_curr = fast_conv(a_curr).max()
        if M_curr < best_M:
            best_M = M_curr
            best_a = a_curr.copy()
        
        if verbose:
            elapsed = time.monotonic() - t_start
            print(f"    L-BFGS t={t:.1f}: J={2*n*best_M:.6f}, nit={result.nit if hasattr(result, 'nit') else '?'} ({elapsed:.1f}s)")
    
    return best_a, best_M

def nesterov_smooth(a0, time_budget, t_schedule=None, verbose=True, seed=42):
    n = len(a0)
    w = np.log(norm_sum1(a0) + 1e-30)
    best_a = norm_sum1(a0).copy()
    best_M = fast_conv(best_a).max()
    t_start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    if t_schedule is None:
        t_schedule = [1.0, 5.0, 10.0, 50.0, 100.0]
    
    for t in t_schedule:
        elapsed = time.monotonic() - t_start
        if elapsed > time_budget:
            break
        if time_budget - elapsed < 3:
            break
        
        w_prev = w.copy()
        step = 0
        stall = 0
        lr = 1.0 / (n ** 0.5 + 1.0)
        
        max_steps = 50000
        while step < max_steps and time.monotonic() - t_start < time_budget:
            # Batch: do 10 steps, then check
            for _ in range(10):
                f, grad = smooth_obj_grad(w, t)
                if step == 0:
                    w_nest = w
                else:
                    beta_t = (1 + np.sqrt(1 + 4*step*step)) / 2
                    w_nest = w + ((beta_t - 1) / beta_t) * (w - w_prev)
                    w_nest = np.maximum(w_nest, -50)  # prevent extreme values
                f_n, grad_n = smooth_obj_grad(w_nest, t)
                w_prev = w
                w = w_nest - lr * grad_n
                step += 1
            
            a_curr = norm_sum1(softmax_fn(w))
            M_curr = fast_conv(a_curr).max()
            if M_curr < best_M - 1e-14:
                best_M = M_curr
                best_a = a_curr.copy()
                stall = 0
            else:
                stall += 1
            
            if stall > 200:
                # Restart with perturbation
                w = np.log(best_a + 1e-30) + rng.normal(0, 0.01, n)
                w_prev = w.copy()
                stall = 0
                lr = 1.0 / (n ** 0.5 + 1.0)
                step = 0
            elif stall > 100:
                lr *= 0.9  # decay learning rate
        
        if verbose:
            elapsed = time.monotonic() - t_start
            print(f"    NSD t={t:.1f}: J={2*n*best_M:.6f} ({elapsed:.1f}s)")
    
    return best_a, best_M

def lp_frank_wolfe(a0, time_budget, solver, verbose=True, max_iters=200):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t_start = time.monotonic()
    
    y = a.copy()
    
    # Precompute T matrix structure
    T_data = []
    T_rows = []
    T_cols = []
    for j in range(n):
        for i in range(n):
            T_rows.append(j + i)
            T_cols.append(j)
            T_data.append(y[i])
    # This is O(n^2) which is too slow for n=2400. Use dense for now.
    # Actually, let's just build it directly.
    
    for it in range(max_iters):
        elapsed = time.monotonic() - t_start
        if elapsed > time_budget:
            break
        
        M_curr = fast_conv(y).max()
        if M_curr < 1e-14:
            break
        
        # Build T matrix: T[k, j] = y[k-j] if 0 <= k-j < n, else 0
        T = np.zeros((2*n-1, n))
        for j in range(n):
            T[j:j+n, j] = y
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum
        
        # Line search
        best_alpha = 0.5
        best_eval = fast_conv((1-best_alpha)*y + best_alpha*direction).max()
        for alpha in [0.1, 0.2, 0.3, 0.7, 0.8, 0.9]:
            trial = np.maximum((1-alpha)*y + alpha*direction, 0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            trial_m = fast_conv(trial).max()
            if trial_m < best_eval:
                best_eval = trial_m
                best_alpha = alpha
        
        y_new = np.maximum((1-best_alpha)*y + best_alpha*direction, 0)
        y_new = y_new / y_new.sum()
        y = y_new
        
        M_y = fast_conv(y).max()
        if M_y < best_M:
            best_M = M_y
            best_a = y.copy()
        
        if verbose and (it+1) % 5 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    LP iter={it+1} ({elapsed:.1f}s): J={2*n*best_M:.6f}")
    
    return best_a, best_M

def coordinate_polish(a0, time_budget, verbose=False, seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t_start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t_start < time_budget and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:min(10, len(ac))]
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t_start > time_budget:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find destination
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            n_cand = min(len(candidates), 300)
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]
            
            best_dst = -1
            best_score = float('inf')
            for dst in chosen:
                if dst == int(src):
                    continue
                j2 = k_int - int(dst)
                if 0 <= j2 < n:
                    sens = a[int(dst)] * a[j2]
                else:
                    sens = 0
                if sens < best_score:
                    best_score = sens
                    best_dst = int(dst)
            
            if best_dst < 0:
                continue
            dst = best_dst
            src_int = int(src)
            
            found = False
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src_int] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src_int] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M - 1e-16:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break
            
            if not found and stall > 100:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_conv(a).max()
                        if M_new < best_M:
                            best_M = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a = best_a.copy()
        
        if not improved:
            stall += 1
    
    if verbose:
        elapsed = time.monotonic() - t_start
        J = 2.0 * n * best_M
        print(f"    Coord polish ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_M
