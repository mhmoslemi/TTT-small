4. **Second-order methods** (use the Hessian of the active constraint)

**Active set method**: At the optimum, there are only a few lags `k*` where `(a*a)_k* = max`. At these lags, the constraint `(a*a)_k <= t` is active. The KKT conditions give:
- For each active lag k*: `(a*a)_k* = t`
- The gradient of `(a*a)_k*` is in the span of {1, active gradients}
- `a_i >= 0` and `a_i * (gradient component) = 0`

This suggests an active set algorithm:
1. Start with all lags active
2. Identify the "most violated" constraints
3. Solve the KKT system for the active set
4. Update the active set
5. Iterate

But with `2n-1` constraints, this could be expensive. However, at the optimum, only `O(1)` or `O(log n)` constraints are typically active.

**Practical approach: Alternating between smoothing and active set**

1. Use log-sum-exp smoothing with large β to approximate the max
2. Use L-BFGS to minimize the smoothed objective (subject to `a >= 0`, `sum a = 1`)
3. Increase β gradually
4. At the end, use an active set method to polish

For the L-BFGS step, we need to handle the constraint `a >= 0`. We can use:
- Log-barrier: add `-μ * sum log(a_i)` to the objective
- Or use the "log-parameterization": `a_i = exp(z_i) / sum_j exp(z_j)` (softmax), which automatically satisfies `a >= 0` and `sum a = 1`

The softmax parameterization is elegant: we optimize over `z ∈ R^n` without constraints, and `a = softmax(z)` automatically lies in the simplex.

The gradient with respect to `z` is:
`∂f/∂z_i = sum_j (∂f/∂a_j) * (δ_{ij} - a_i) * a_j`

This is the "natural gradient" or "Fisher information metric" scaled gradient.

```python
def softmax_objective(z, beta, top_k=100):
    """Compute the smoothed objective and its gradient w.r.t. z."""
    n = len(z)
    # Compute a = softmax(z)
    z_max = z.max()
    a = np.exp(z - z_max)
    a = a / a.sum()
    
    # Compute convolution
    c = fast_autocorr(a)
    
    # Compute smoothed max
    c_max = c.max()
    if top_k is not None:
        idx = np.argsort(c)[::-1][:top_k]
        c_work = c[idx]
        exp_c = np.exp(beta * (c_work - c_max))
        w = exp_c / exp_c.sum()
        f_val = c_max + np.log(exp_c.sum()) / beta
    else:
        exp_c = np.exp(beta * (c - c_max))
        w_full = exp_c / exp_c.sum()
        f_val = c_max + np.log(exp_c.sum()) / beta
        w = None  # use full
    
    # Compute gradient w.r.t. a
    if top_k is not None:
        grad_a = np.zeros(n)
        for wi, k in zip(w, idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            idx_j = np.arange(i_start, i_end + 1)
            mirror = k_int - idx_j
            valid = (mirror >= 0) & (mirror < n)
            contrib = np.zeros(len(idx_j))
            contrib[valid] = a[mirror[valid]]
            double_mask = valid & (mirror != idx_j)
            contrib[double_mask] *= 2
            grad_a[idx_j] += wi * contrib
        grad_a /= beta
    # Gradient w.r.t. z: grad_z_i = sum_j grad_a_j * (delta_ij - a_i) * a_j
    # = a_i * grad_a_i - a_i * sum_j a_j * grad_a_j
    dot = a @ grad_a
    grad_z = a * (grad_a - dot)
    
    return f_val, grad_z
```

This is clean and efficient. Each evaluation is O(n log n + k*n) where k is top_k.

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, non-negative, with floor."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-15)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def softmax_entity(z):
    """Compute a = softmax(z) stably."""
    z = z - z.max()
    a = np.exp(z)
    a = a / a.sum()
    return a


def smoothed_obj_grad_z(z, beta, top_k=100):
    """
    Compute log-sum-exp smoothed objective and gradient w.r.t. z (softmax params).
    f(z) = (1/beta) * log(sum_k exp(beta * c_k)) where c = a*a, a = softmax(z)
    Returns (f_value, grad_z).
    """
    n = len(z)
    a = softmax_entity(z)
    c = fast_autocorr(a)
    c_max = c.max()
    
    if c_max <= 0:
        return 0.0, np.zeros(n)
    
    # Top-k selection for gradient
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_work = c[idx]
    else:
        idx = np.arange(len(c))
        c_work = c
    
    # Log-sum-exp for objective value (use full for accuracy)
    # But for gradient, use top-k
    exp_work = np.exp(beta * (c_work - c_max))
    w = exp_work / exp_work.sum()
    
    # Objective: use full sum for accuracy
    all_exp = np.exp(beta * (c - c_max))
    f_val = c_max + np.log(all_exp.sum()) / beta
    
    # Gradient w.r.t. a: grad_a = (1/beta) * sum_k w_k * dc_k/da
    grad_a = np.zeros(n)
    for wi, k in zip(w, idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx_j = np.arange(i_start, i_end + 1)
        mirror = k_int - idx_j
        valid = (mirror >= 0) & (mirror < n)
        contrib = np.zeros(len(idx_j))
        contrib[valid] = a[mirror[valid]]
        # Double where k != 2*i (mirror != self)
        double_mask = valid & (mirror != idx_j)
        contrib[double_mask] *= 2.0
        grad_a[idx_j] += wi * contrib
    
    grad_a /= beta
    
    # Gradient w.r.t. z: grad_z = a * (grad_a - a . grad_a)
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    
    return f_val, grad_z

    except Exception:

    if result.x is not None:

    return z_init

        z_new = lbfgs_optimize_z(z, beta, max_iter=max_iter, top_k=tk, time_limit=t_limit)
        z = z_new
        a_new = softmax_entity(z)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"      beta={beta}: max_c={m_new:.2e}, obj={2*n*m_new:.6f} ({elapsed:.1f}s)")
        
        a = a_new
    
    return best_a


def spectral_equalize(a, n_iter=30, eps=1e-10):
    """Flatten power spectrum via iterative DFT correction."""
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = np.abs(A) ** 2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.5 * a + 0.5 * a_new
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum and non-negativity."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 1e-15)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def random_polish(a_init, time_limit, rng_seed=42, verbose=False):
    """
    Random coordinate perturbation polish with simulated annealing acceptance.
    Works in a-space with simplex projection.
    """
    n = len(a_init)
    a = normalize(a_init)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    temp = 0.001
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        # Random perturbation: move mass from one coordinate to another
        n_perturb = rng.integers(1, 4)
        a_trial = a.copy()
        
        for _ in range(n_perturb):
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst:
                continue
            frac = rng.uniform(0.001, 0.05)
            amt = a_trial[src] * frac
            a_trial[src] -= amt
            a_trial[dst] += amt
        
        a_trial = np.maximum(a_trial, 1e-15)
        s = a_trial.sum()
        a_trial = a_trial / s
        
        m_trial = fast_autocorr(a_trial).max()
        
        if m_trial < best_m:
            a = a_trial
            best_a = a_trial.copy()
            best_m = m_trial
            stall = 0
            # Cool down
            temp = max(temp * 0.99, 0.0001)
        else:
            # Metropolis acceptance (rare, for exploration)
            delta = m_trial - best_m
            if rng.random() < np.exp(-delta / (temp * best_m + 1e-30)):
                a = a_trial
            stall += 1
            temp = max(temp * 0.995, 0.0001)
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      polish: obj={2*n*best_m:.6f} ({elapsed:.1f}s, stall={stall})")
    
    return best_a


def active_set_refine(a_init, time_limit, verbose=False):
    """
    Refine using active-set method: equalize the top lags.
    At optimum, top lags should all have the same value t.
    Uses iterated linearization + projection.
    """
    n = len(a_init)
    a = normalize(a_init)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for iteration in range(200):
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        c = fast_autocorr(a)
        M = c.max()
        
        # Identify active set: lags within a small fraction of max
        threshold = M * 0.999
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Try to equalize: for each active lag, compute the "excess"
        # and adjust a to reduce the max
        # Strategy: for the lag with the largest value, find the direction
        # that most reduces it while least affecting others
        
        k_max = int(active[np.argmax(c[active])])
        i_start = max(0, k_max - n + 1)
        i_end = min(n - 1, k_max)
        idx_j = np.arange(i_start, i_end + 1)
        mirror = k_max - idx_j
        valid = (mirror >= 0) & (mirror < n)
        
        # Direction: reduce the "hot" products
        # For each pair (j, k_max-j), we can reduce by moving mass away from one
        hotness = np.zeros(n)
        for pos, j in enumerate(idx_j):
            if valid[pos]:
                m = mirror[pos]
                if j == m:
                    hotness[j] += a[j]  # self-product
                else:
                    hotness[j] += 0.5 * a[m]
                    hotness[m] += 0.5 * a[j]
        
        # Move mass from hot positions to cool positions
        total_hot = hotness.sum()
        if total_hot < 1e-15:
            break
        
        # Find cool positions (low hotness)
        cool_threshold = np.percentile(hotness[hotness > 0], 50) if np.sum(hotness > 0) > 10 else 0
        cool_positions = np.where(hotness < cool_threshold)[0]
        
        if len(cool_positions) == 0:
            cool_positions = np.where(hotness == hotness.min())[0]
        
        # Try a small adjustment
        step = 0.01
        a_new = a.copy()
        
        # Move a small fraction of mass from top hot positions to cool ones
        hot_positions = np.argsort(hotness)[::-1][:20]
        n_move = min(5, len(hot_positions), len(cool_positions))
        
        for i in range(n_move):
            src = hot_positions[i]
            if a[src] < 1e-12:
                continue
            # Find best cool destination
            best_dst = -1
            best_gain = 0
            for dst in cool_positions[:50]:
                if dst == src:
                    continue
                # Gain: reduction in c[k_max] from moving mass
                # dc/d(a_src) * (-delta) + dc/d(a_dst) * delta
                # We want to minimize c[k_max], so maximize the reduction
                gain = hotness[src] - hotness[dst]
                if gain > best_gain:
                    best_gain = gain
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            amt = a[src] * step
            a_new[src] -= amt
            a_new[best_dst] += amt
        
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m * (1 - 1e-12):
            a = a_new
            best_a = a_new.copy()
            best_m = m_new
        
        # Check convergence
        if iteration > 10:
            c_check = fast_autocorr(a)
            active_check = np.where(c_check >= c_check.max() * 0.999)[0]
            if len(active_check) <= 3 and np.ptp(c_check[active_check]) < 1e-10 * c_check.max():
                if verbose:
                    print(f"      active-set converged at iter {iteration}")
                break
    
    return best_a
