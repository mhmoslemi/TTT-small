def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_score(a):

    n = len(a)
    s = a.sum()

    return 2.0 * n * ac.max() / (s * s)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_peak_gradient(a, ac, top_k, beta):
    """
    Compute gradient of smoothed max of autocorrelation.
    Uses top-k lags with softmax weighting.
    """
    n = len(a)
    k_eff = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_eff]
    top_vals = ac[top_idx]
    shifted = top_vals - top_vals[0]
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        # dc_k/da[i] = 2*a[k-i] if 0 <= k-i < n
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        partner = k_int - idx
        grad[idx] += 2.0 * wi * a[partner]
    return grad, top_idx, weights

def spectral_refine(a0, time_limit, verbose=True, seed=77):
    """
    Frequency-domain refinement: iteratively adjust the spectrum to
    reduce autocorrelation peaks.
    
    Idea: for the peak lag k*, the contribution in frequency domain is
    related to the coherence of |A(f)|^2 at frequency 2*pi*k*/N.
    We can reduce this by adding a small "anti-sinusoid" to the spectrum.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    rng = np.random.default_rng(seed)
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))
        
        # Compute spectrum
        A = np.fft.rfft(a, fft_size)
        mag2 = A.real ** 2 + A.imag ** 2
        # Note: A is complex, |A|^2 = A * conj(A)
        mag2 = np.abs(A) ** 2
        
        # The autocorrelation at lag k* is:
        # c[k*] = (1/N) * sum_f |A(f)|^2 * exp(2*pi*i*f*k*/N)
        # To reduce c[k*], we want to reduce |A(f)|^2 where the cosine term is positive
        # and increase where it's negative (but we can't increase without increasing sum)
        
        # Simpler approach: compute the "phase mismatch" and adjust
        # c[k*] = sum_i a[i]*a[k*-i]
        # Gradient tells us to reduce a[i] where a[k*-i] is large
        # In frequency domain, this is like deconvolving the peak
        
        # Practical approach: add a small perturbation in frequency domain
        # that reduces the peak
        freq_response = np.exp(2j * np.pi * np.arange(len(A)) * peak_k / fft_size)
        
        # The contribution to c[k*] from each frequency bin is proportional to
        # mag2[j] * real(freq_response[j])
        contrib = mag2 * freq_response.real
        
        # Reduce bins with positive contribution, increase (slightly) with negative
        # This is a "spectral notching" approach
        alpha = 0.01 * (1.0 + t / 1000.0)
        delta = -alpha * contrib / (mag2.sum() + 1e-15)
        
        # Modify magnitude (not phase)
        new_mag2 = mag2 + delta
        new_mag2 = np.maximum(new_mag2, 0.0)
        
        # Reconstruct with same phase
        phase = np.angle(A)
        new_A = np.sqrt(new_mag2) * np.exp(1j * phase)
        new_a = np.fft.irfft(new_A, fft_size)[:n]
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-15:
            continue
        new_a = new_a / s
        
        M_new = fast_autocorr(new_a).max()
        if M_new < best_M - 1e-15:
            a = new_a
            best_M = M_new
            best_a = a.copy()
        else:
            # Try a smaller step
            alpha2 = alpha * 0.1
            delta2 = -alpha2 * contrib / (mag2.sum() + 1e-15)
            new_mag2_2 = np.maximum(mag2 + delta2, 0.0)
            new_A2 = np.sqrt(new_mag2_2) * np.exp(1j * phase)
            new_a2 = np.fft.irfft(new_A2, fft_size)[:n]
            new_a2 = np.maximum(new_a2, 0.0)
            s2 = new_a2.sum()
            if s2 > 1e-15:
                new_a2 = new_a2 / s2
                M_new2 = fast_autocorr(new_a2).max()
                if M_new2 < best_M - 1e-15:
                    a = new_a2
                    best_M = M_new2
                    best_a = a.copy()
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"    Spectral refine ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_M


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        t_r = min(t_per_restart, rem - 10)
        if t_r < 15:
            break
        
        # Create perturbed starting point
        strategy = r % 4
        if strategy == 0:
            # Small Gaussian noise
            noise = rng.normal(0, 0.02, len(best_seq)) * best_seq
            a_start = np.maximum(best_seq + noise, 1e-10)
            a_start = a_start / a_start.sum()
        elif strategy == 1:
            # Frequency shift
            n = len(best_seq)
            shift = int(rng.integers(1, max(2, n // 20)))
            a_start = np.zeros(n)
            a_start[shift:] = best_seq[:-shift]
            a_start[:shift] = best_seq[n-shift:]
            a_start = a_start / a_start.sum()
        elif strategy == 2:
            # Blend with different shape
            x = np.linspace(0, 1, len(best_seq))
            alt = np.exp(-20 * (x - rng.uniform(0.3, 0.7)) ** 2)
            a_start = 0.7 * best_seq + 0.3 * alt / alt.sum() * best_seq.sum()
            a_start = np.maximum(a_start, 1e-10)
            a_start = a_start / a_start.sum()
        else:
            # Antisymmetric perturbation (reduce center peak)
            n = len(best_seq)
            a_start = best_seq.copy()
            mid = n // 2
            # Reduce center, increase edges
            center_mask = np.zeros(n)
            center_mask[max(0, mid - n//8):min(n, mid + n//8)] = 1
            edge_mask = 1 - center_mask
            a_start = a_start * (1 - 0.1 * center_mask + 0.15 * edge_mask)
            a_start = np.maximum(a_start, 1e-10)
            a_start = a_start / a_start.sum()
        
        a_out, _ = apg_optimize(
            a_start, t_r,
            beta_start=30.0, beta_end=5.0,
            top_k_start=100, top_k_end=500,
            gamma=2.0, verbose=False, seed=1000 + r)
        try_update(a_out, f"restart{r}")
    
    # === Phase 6: Multi-length exploration (10% of time) ===
    rem = remaining_time()
    if rem > 60:
        print(f"\n=== Phase 6: Multi-length ===")
        n0 = len(best_seq)
        target_lengths = []
        # Try slightly longer
        if n0 < 3500:
            target_lengths.append(min(n0 + 400, 3600))
        # Try slightly shorter (sometimes shorter is better due to 2n factor)
        if n0 > 1500:
            target_lengths.append(max(n0 - 400, 1200))
        
        t_per_len = min(rem * 0.10, 100.0) / max(1, len(target_lengths))
        
        for n_new in target_lengths:
            rem = remaining_time()
            if rem < 40:
                break
            t_nl = min(t_per_len, rem - 10)
            if t_nl < 20:
                break
            
            print(f"  Trying n={n_new} (current best n={len(best_seq)})")
            a_long = interpolate_sequence(best_seq, n_new)
            a_out, _ = apg_optimize(
                a_long, t_nl,
                beta_start=30.0, beta_end=5.0,
                top_k_start=100, top_k_end=600,
                gamma=2.0, verbose=False, seed=500)
            try_update(a_out, f"len{n_new}")
            
            # Quick CD at new length
            rem = remaining_time()
            if rem > 30:
                a_out2, _ = pair_move_cd(a_out, min(rem * 0.3, 40.0), verbose=False, seed=600)
                try_update(a_out2, f"cd_len{n_new}")
    
    # === Phase 7: Final coordinate polish (remaining time) ===
    rem = remaining_time()
    print(f"\n=== Phase 7: Final CD n={len(best_seq)} ===")
    if rem > 20:
        t7 = min(rem * 0.8, 120.0)
        if t7 > 10:
            a_out, _ = pair_move_cd(best_seq, t7, verbose=True, seed=314)
            try_update(a_out, f"final_cd_n{len(best_seq)}")
    
    # Final validation
    if best_seq is not None:
        fs = safe_evaluate(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    # Ensure all values are in valid range
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
