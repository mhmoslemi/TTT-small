    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm1(a):
    """Normalize so sum = 1, clamp to nonnegative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def make_power_ramp(n, p=1.0):
    """Create a power ramp: a[i] = (i+1)^p, normalized to sum 1."""
    a = np.arange(1, n+1, dtype=np.float64) ** p
    return a / a.sum()

def make_two_sided(n, p=1.0, q=0.5):
    """Create a two-sided shape: a[i] = (i+1)^p * (n-i)^q, normalized."""
    a = (np.arange(1, n+1, dtype=np.float64) ** p) * (np.arange(n, 0, -1, dtype=np.float64) ** q)
    return a / a.sum()

def make_exponential(n, rate=1.0):
    """Create an exponential ramp: a[i] = exp(rate*i/(n-1)), normalized."""
    a = np.exp(rate * np.arange(n, dtype=np.float64) / max(n-1, 1))
    return a / a.sum()

def smooth_max(ac, temperature=0.01):
    """Compute smooth approximation of max using log-sum-exp."""
    shifted = ac / temperature
    m = shifted.max()
    return temperature * (m + np.log(np.exp(shifted - m).sum()))

def smooth_max_grad(ac, a, temperature=0.01):
    """Compute gradient of smooth max of convolution w.r.t. a.
    
    d/d(a[i]) sum_k ac[k] = sum_k d(ac[k])/d(a[i])
    ac[k] = sum_j a[j]*a[k-j]
    d(ac[k])/d(a[i]) = a[k-i] + a[i-k] (when indices valid)
    
    For smooth max: grad = (1/T) * exp(ac/T) / sum(exp(ac/T)) * d(ac)/d(a)
    """
    n = len(a)
    shifted = ac / temperature
    m = shifted.max()
    weights = np.exp(shifted - m)
    weights = weights / weights.sum()
    
    # Gradient of ac w.r.t. a[i]:
    # ac[k] = sum_j a[j]*a[k-j]
    # d(ac[k])/d(a[i]) = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
    # But since ac = conv(a,a), and conv is "symmetric" in the sense that
    # d(conv(a,a)/da[i]) = a shifted: it's a[i] convolved with delta, 
    # i.e., d(ac)/d(a[i]) = a shifted by i in both directions
    # Actually: ac[k] = sum_j a[j]*a[k-j]
    # d(ac[k])/d(a[i]) = a[k-i] (from the first factor, j=i) + a[k-i] (from second factor, k-j=i, j=k-i)
    # Wait: if we differentiate w.r.t. a[i]:
    #   The term a[i]*a[k-i] appears when j=i (first factor)
    #   The term a[j]*a[i] appears when k-j=i, i.e., j=k-i (second factor)
    # So d(ac[k])/d(a[i]) = a[k-i] + a[k-i] = 2*a[k-i] (if 0<=k-i<n)
    # Actually only one of these is valid at a time... no wait.
    # ac[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
    # If j=i is valid (0<=i<n and 0<=k-i<n): contributes a[i]*a[k-i], derivative w.r.t. a[i] is a[k-i]
    # If k-j=i is valid, i.e., j=k-i is valid (0<=k-i<n and 0<=k-(k-i)<n=i<n): contributes a[k-i]*a[i], derivative w.r.t. a[i] is a[k-i]
    # But these are the SAME term when j=i and k-j=i, i.e., k=2i. In that case it's a[i]^2 and derivative is 2*a[i].
    # In general, if j=i and k-j=i are both valid AND distinct (i ≠ k-i, i.e., k≠2i), we get 2*a[k-i].
    # If k=2i, we get a[i]*a[i] = a[i]^2, derivative = 2*a[i] = 2*a[k-i]. Same formula!
    # So: d(ac[k])/d(a[i]) = 2*a[k-i] if 0<=k-i<n, else 0.
    # Wait, that's not right either. Let me be more careful.
    # 
    # ac[k] = sum_j a[j]*a[k-j] (over valid j)
    # Treat a as a vector. The derivative of ac[k] w.r.t. a[i]:
    #   = sum_j [delta_{j,i}*a[k-j] + a[j]*delta_{k-j,i}]
    #   = a[k-i] (if i is valid, j=i) + a[k-i] (if k-i is valid, j=k-i)
    #   But wait: in the first term, j=i is valid means 0<=i<n AND 0<=k-i<n
    #   In the second term, j=k-i is valid means 0<=k-i<n AND 0<=k-(k-i)<n, i.e., 0<=i<n
    #   So: first term valid iff 0<=k-i<n (and i is always in [0,n-1])
    #       second term valid iff 0<=k-i<n (and i is always in [0,n-1])
    #   Both terms give a[k-i] when valid.
    #   So d(ac[k])/d(a[i]) = 2*a[k-i] if 0<=k-i<n, else 0.
    #   EXCEPT when j=i in the first term AND j=k-i in the second term are the Same j (i.e., i=k-i, k=2i).
    #   In that case, the term a[i]*a[i] = a[i]^2 is differentiated once, giving 2*a[i].
    #   But our formula gives 2*a[k-i] = 2*a[i]. Same!
    #   So the formula d(ac[k])/d(a[i]) = 2*a[k-i] * 1[0<=k-i<n] is always correct.
    
    # So: grad[i] = sum_k weights[k] * 2*a[k-i] * 1[0<=k-i<n]
    #            = 2 * sum_k weights[k] * a[k-i] * 1[0<=k-i<n]
    # This is 2 * (weights * a) convolved... it's the cross-correlation of weights with a, times 2.
    # Specifically: grad[i] = 2 * sum_{k: 0<=k-i<n} weights[k] * a[k-i]
    #                  = 2 * sum_{j: 0<=j<n} weights[i+j] * a[j]  (substituting j=k-i)
    
    # This is 2 * (reverse cross-correlation of weights and a)
    # = 2 * np.convolve(weights, a[::-1])[n-1-i] ... let me think.
    # Actually: sum_j weights[i+j]*a[j] for j=0..min(n-1, len(weights)-1-i)
    # This is the cross-correlation at lag i.
    # cross_corr[i] = sum_j a[j] * weights[i+j]
    # = sum_j a[j] * weights[n-1-(n-1-i-j)] ... hmm
    # Let me just compute it directly using FFT since n can be up to 5000.
    
    # grad[i] = 2 * sum_{j=0}^{n-1} a[j] * weights[i+j] * 1[i+j < len(weights)]
    # = 2 * cross_correlation(a, weights)[i]  (with appropriate definition)
    
    # Using FFT: cross_correlation can be computed as ifft(conj(fft(a)) * fft(weights))
    # But we need a specific cross-correlation. Let me use np.correlate or direct FFT.
    
    # Actually, let's just use the convolution identity:
    # sum_j a[j]*weights[i+j] = (a * reverse(weights))[i + len(weights)-1] ... this is getting complicated.
    # Let me just compute it with a matrix-free approach.
    
    # For efficiency, note that:
    # grad[i] = 2 * sum_{j=0}^{n-1} a[j] * weights[i+j] * 1[0 <= i+j < 2n-1]
    # This is a valid cross-correlation. We can compute it as:
    # Let W = weights (length 2n-1), A = a (length n)
    # Then grad[i]/2 = sum_{j=0}^{n-1} A[j] * W[i+j]
    # = (A * W_full)[i] where * denotes cross-correlation
    # Cross-correlation: (A ⋆ W)[i] = sum_j A[j] * W[i+j] (one-sided)
    # = conj part of ifft(fft(A) * conj(fft(W)))... 
    
    # Simplest: use np.correlate in 'full' mode
    # np.correlate(W, A, 'full')[n-1+i] = sum_j W[j+i]*A[j] for valid j
    # Hmm, let me just use direct computation with FFT.
    
    # Actually the cleanest way: 
    # grad[i]/2 = sum_j a[j] * weights[i+j]
    # Let's define this as the "autocorrelation-like" product.
    # We can compute it as: ifft(fft(a, N) * fft(pad(weights), N)) ... no.
    
    # Let me just use the direct formula with a loop over the top weights:
    # Since weights is concentrated (exponential decay from the max), 
    # only the top ~100 weights contribute significantly.
    
    # For efficiency with large n, use the top-K weights approach:
    top_k = min(200, len(ac))
    top_idx = np.argsort(ac)[::-1][:top_k]
    top_weights = weights[top_idx]
    
    grad = np.zeros(n)
    for wi, k in zip(top_weights, top_idx):
        k = int(k)
        # grad[i] += 2 * wi * a[k-i] for 0 <= k-i < n
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    
    return grad

def projected_grad_desc(a0, time_limit, temperature=0.01, lr=None, verbose=True):
    """Projected gradient descent with smooth max objective.
    
    Minimize: smooth_max(conv(a,a)) subject to a>=0, sum(a)=1.
    """
    n = len(a0)
    a = norm1(a0.copy())
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if lr is None:
        lr = 0.01
    
    # Adaptive temperature: start high, decrease
    T = temperature * 5
    T_min = temperature * 0.1
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
        
        # Compute gradient
        g = smooth_max_grad(ac, a, temperature=T)
        
        # Gradient step
        # We want to minimize, so go in negative gradient direction
        # But we need to maintain sum(a) = 1 and a >= 0
        # Projected gradient: a_new = a - lr * g, then project to simplex
        
        # To maintain sum=1: subtract the mean of (lr*g) to center
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / s
        
        # Check if improved
        ac_trial = fast_autocorr(a_trial)
        M_trial = ac_trial.max()
        
        if M_trial < M:
            a = a_trial
            # Decrease temperature slowly
            T = max(T * 0.999, T_min)
        else:
            # Backtrack
            lr *= 0.8
            if lr < 1e-8:
                lr = 0.01
                # Perturb to escape
                a = a + np.random.default_rng(it).normal(0, 0.01, n)
                a = np.maximum(a, 0)
                a = a / a.sum()
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    PGD it={it} ({elapsed:.0f}s): obj={J:.6f}, M={best_M:.6e}, lr={lr:.4f}, T={T:.4f}")
    
    return best_a, best_M

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def coordinate_refine(a0, time_limit, verbose=True):
    """Coordinate-wise refinement: move mass from high-convolution positions to low ones."""
    n = len(a0)
    a = norm1(a0.copy())
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 100:
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Find positions that most contribute to the peak
        peak_k = int(np.argmax(ac))
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        
        # Contributions to the peak
        contrib = np.zeros(n)
        for i in range(i_start, i_end):
            j = peak_k - i
            contrib[i] = a[i] * a[j]
        
        # Find the position with highest contribution
        if contrib.max() < 1e-20:
            break
        
        src = int(np.argmax(contrib))
        
        # Find best destination: minimize increase in max convolution
        # Try moving a small amount to a random position
        improved = False
        for _ in range(50):
            dst = int(rng.integers(n))
            if dst == src:
                continue
            amt = a[src] * 0.01
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_M:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * 0.005
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Coord ({elapsed:.0f}s): obj={J:.6f}, stall={stall}")
    
    return best_a, best_M

def soft_subgradient_descent(a0, time_limit, top_k=80, beta=4.0, verbose=True):
    """Mirror descent with soft subgradient in probability simplex.
    
    Uses the log-barrier / exponential multiplicative weights approach.
    This is similar to the previous AMD but with tuned hyperparameters.
    """
    n = len(a0)
    a = norm1(a0.copy())
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    x = a.copy()
    x_prev = a.copy()
    c0 = 5.0
    eta_base = 0.5
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        ac = fast_autocorr(y)
        
        # Soft subgradient
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            g[idx] += 2.0 * wi * y[j_idx]
        
        eta = eta_base / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = x.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    MMD t={t} ({elapsed:.0f}s): obj={J:.6f}")
    
    return best_a, best_M
