            d = c
            fd = fc

            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2

def lp_direction_scipy(a):
    n = len(a)
    # min M
    # s.t. A[k,:] @ b <= M for all k  →  A @ b - I*M <= 0  →  [A, -I] @ [b; M] <= 0
    #      sum(b) = 1  →  [1...1, 0] @ [b; M] = 1
    #      b >= 0, M >= 0  →  -b <= 0, -M <= 0
    
    # Total variables: n + 1
    # c = [0, ..., 0, 1] (minimize M)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A @ b - M <= 0 for each k
    # [A, -e] @ [b; M] <= 0 where e = ones(2n-1)
    # Also: -b <= 0, -M <= 0
    
    A_mat = np.zeros((2*n-1, n))
    for j in range(n):
        A_mat[j:j+n, j] = a
    
    # Full A_ub: (2n-1 + n + 1) x (n+1)
    # First 2n-1 rows: [A_mat, -ones]
    # Next n rows: [-I_n, 0]
    # Last row: [0, -1]
    
    A_ub = np.zeros((2*n-1 + n + 1, n+1))
    A_ub[:2*n-1, :n] = A_mat
    A_ub[:2*n-1, n] = -1.0
    A_ub[2*n-1:2*n-1+n, :n] = -np.eye(n)
    A_ub[-1, n] = -1.0
    
    b_ub = np.zeros(2*n-1 + n + 1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
    
    if res.status != 0:
        return None
    
    b_val = np.maximum(res.x[:n], 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

```python
def line_search_quadratic(ac_a, ab, ac_b, max_iter=80):
    """Minimize max_k [(1-t)^2*ac_a[k] + 2t(1-t)*ab[k] + t^2*ac_b[k]] over t in [0,1]."""
    def f(t):
        return np.max((1-t)**2 * ac_a + 2*t*(1-t) * ab + t**2 * ac_b)
    
    # Use ternary search (since f is convex in t)
    lo, hi = 0.0, 1.0
    for _ in range(max_iter):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if f(m1) < f(m2):
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-10:
            break
    return (lo + hi) / 2
```

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def convolve_fn(a, b):
    """Full convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def normalize_nonneg(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_objective(a):
    ac = fft_autocorr(a)
    return 2.0 * len(a) * ac.max()


def evaluate_safe_local(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def gen_raised_cos(n, p=1.0):
    x = np.linspace(0, 1, n)
    base = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    a = base ** p
    a = np.maximum(a, 0.0)
    return normalize_nonneg(a)


def gen_beta(n, alpha=2.0, beta=2.0):
    x = np.linspace(1e-8, 1.0 - 1e-8, n)
    a = x ** alpha * (1.0 - x) ** beta
    return normalize_nonneg(a)


def gen_gaussian(n, sigma=0.30):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    return normalize_nonneg(a)


def gen_parent(n):
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0 or p.sum() < 1e-10:
            return None
        if len(p) == n:
            return normalize_nonneg(p)
        p_norm = p / p.sum()
        old_x = np.linspace(0.0, 1.0, len(p_norm))
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, p_norm)
        return normalize_nonneg(a)
    except Exception:
        return None


def interpolate_to_n(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    return normalize_nonneg(a_new)


def lp_direction(a):
    """
    Solve: min M s.t. conv(a,b)[k] <= M for all k, sum(b)=1, b>=0.
    Returns normalized b or None.
    """
    n = len(a)
    # Build A_mat: A_mat[k, j] = a[k-j] if 0 <= k-j < n
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_mat[j:j + n, j] = a

    # Variables: x = [b_0, ..., b_{n-1}, M], total n+1
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

    # Inequality: A_mat @ b - M <= 0  →  [A_mat | -1] @ x <= 0
    # Also: -b <= 0, -M <= 0
    n_ineq = (2 * n - 1) + n + 1
    A_ub = np.zeros((n_ineq, n + 1))
    A_ub[:2 * n - 1, :n] = A_mat
    A_ub[:2 * n - 1, n] = -1.0
    A_ub[2 * n - 1:2 * n - 1 + n, :n] = -np.eye(n)
    A_ub[-1, n] = -1.0
    b_ub = np.zeros(n_ineq)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs',
                      options={"presolve": True})
    except Exception:
        return None

    if res.status != 0:
        return None

    b_val = np.maximum(res.x[:n], 0.0)
    s = b_val.sum()
    if s < 1e-12:
        return None
    return b_val / s


def subgradient_direction(a, n_top=20):
    """
    Heuristic direction: minimize weighted sum of top autocorrelation lags.
    g = sum_{k in top} w_k * 2 * (a shifted by k), normalized to simplex.
    """
    n = len(a)
    ac = fft_autocorr(a)
    top_idx = np.argsort(ac)[::-1][:n_top]
    w = ac[top_idx]
    w = w / w.sum()

    # Subgradient of max at active lags: g[i] = 2 * sum_k w_k * a[k-i] (valid)
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        for i in range(i_start, i_end):
            j = ki - i
            if 0 <= j < n:
                g[i] += 2.0 * wi * a[j]

    # Convert to Frank-Wolfe direction: min g^T b over simplex → pick min g[i]
    # But that gives a vertex. Instead, use softmax of -g for a smooth direction.
    g_max = g.max()
    exp_arg = -(g - g_max) / max(g.std(), 1e-15)
    exp_arg = np.clip(exp_arg, -50, 50)
    b = np.exp(exp_arg)
    return normalize_nonneg(b)


def spectral_direction(a, n_zeros=50):
    """
    Direction that zeros out the largest mid-band power spectrum bins.
    This encourages a flatter autocorrelation.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    power = np.abs(A) ** 2

    # Keep DC and the lowest few bins, zero out the largest mid-band bins
    n_half = len(A)
    protect = max(10, n // 200)
    power_masked = power.copy()
    power_masked[:protect] = 0  # protect low frequencies
    power_masked[-1] = 0  # protect Nyquist

    # Find the largest bins to zero
    top_bins = np.argsort(power_masked)[::-1][:n_zeros]
    power_masked[top_bins] = 0

    # Reconstruct
    A_new = A.copy()
    A_new[top_bins] = 0
    b = np.fft.irfft(A_new, fft_size)[:n]
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s < 1e-12:
        return None
    return b / s


def smooth_direction(a, sigma_frac=0.02):
    """Gaussian-smoothed version of a as a direction."""
    n = len(a)
    sigma = max(1.0, sigma_frac * n)
    # Vectorized Gaussian kernel
    x = np.arange(n)
    kernel = np.exp(-0.5 * ((x - n // 2) / sigma) ** 2)
    b = np.convolve(a, kernel, mode='same')
    return normalize_nonneg(b)


def ternary_line_search(ac_a, ab, ac_b, max_iter=60):
    """
    Minimize f(t) = max_k [(1-t)^2*ac_a[k] + 2*t*(1-t)*ab[k] + t^2*ac_b[k]]
    over t in [0, 1]. f is convex in t.
    """
    def f(t):
        return np.max((1.0 - t) ** 2 * ac_a + 2.0 * t * (1.0 - t) * ab + t ** 2 * ac_b)

    lo, hi = 0.0, 1.0
    for _ in range(max_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = f(m1)
        f2 = f(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-10:
            break
    t_opt = (lo + hi) / 2.0
    return t_opt, f(t_opt)


def lp_optimizer(a_init, time_budget, verbose=True, use_momentum=True, legacy_momentum=0.5):
    """
    Main LP-based optimizer with momentum and line search.
    """
    n = len(a_init)
    a = normalize_nonneg(a_init.copy())
    best_a = a.copy()
    best_obj = compute_objective(a)

    t0 = time.monotonic()
    prev_dir = a.copy()
    stall = 0
    max_stall = 50
    iter_count = 0
    lp_fail_count = 0

    while time.monotonic() - t0 < time_budget:
        iter_count += 1

        # Feasibility check
        if a.sum() < 1e-8:
            a = best_a.copy()

        # Get direction
        direction = None
        method = "lp"

        if lp_fail_count < 5:
            b = lp_direction(a)
            if b is not None:
                direction = b
                lp_fail_count = 0
            else:
                lp_fail_count += 1

        if direction is None:
            # Alternate methods
            if iter_count % 3 == 1:
                direction = subgradient_direction(a, n_top=30)
                method = "subgrad"
            elif iter_count % 3 == 2:
                direction = spectral_direction(a, n_zeros=max(20, n // 100))
                method = "spectral"
                if direction is None:
                    direction = smooth_direction(a)
                    method = "smooth"
            else:
                direction = smooth_direction(a, sigma_frac=0.015)
                method = "smooth"

        if direction is None:
            direction = a.copy()

        # Apply momentum to direction
        if use_momentum:
            direction = normalize_nonneg((1.0 - legacy_momentum) * direction + legacy_momentum * prev_dir)

        # Precompute for line search
        ac_a = fft_autocorr(a)
        ab = convolve_fn(a, direction)
        ac_b = fft_autocorr(direction)

        # Line search
        t_opt, f_opt = ternary_line_search(ac_a, ab, ac_b, max_iter=30)

        # Validate
        if t_opt <= 1e-8:
            stall += 1
            if stall > max_stall:
                # Perturb and restart
                noise = np.random.default_rng(iter_count).normal(0, 0.01, n) * best_a
                a = normalize_nonneg(best_a + noise)
                direction = smooth_direction(a)
                prev_dir = direction
                stall = 0
            continue

        a_new = (1.0 - t_opt) * a + t_opt * direction
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            a = best_a.copy()
            continue
        a_new = a_new / s

        obj_new = compute_objective(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                a = best_a + np.random.default_rng(iter_count + 1000).normal(0, 0.005, n) * best_a
                a = normalize_nonneg(a)
                best_obj = compute_objective(a)
                best_a = a.copy()
                stall = 0
            a = a_new if obj_new < compute_objective(a) else a

        a = a_new
        prev_dir = direction

        if verbose and iter_count % 20 == 1:
            elapsed = time.monotonic() - t0
            print(f"    LP opt [{method}] iter={iter_count} ({elapsed:.1f}s): obj={best_obj:.6f}, t={t_opt:.4f}, stall={stall}")

    return best_a, best_obj


def add_correction(a_init, time_budget, verbose=False):
    """
    Try to add a small non-negative correction c to reduce peak autocorrelation.
    Minimize max_k conv(a+c, a+c) over c >= 0, ||c||_1 <= eps, (a+c)/sum(a+c) normalized.
    Uses iterative peak-reduction heuristic.
    """
    n = len(a_init)
    a = normalize_nonneg(a_init.copy())
    best_a = a.copy()
    best_obj = compute_objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for it in range(200):
        if time.monotonic() - t0 > time_budget:
            break
        ac = fft_autocorr(a)
        peak_k = int(ac.argmax())
        M = ac.max()

        # Find the pair (i, j) with i+j such that conv index is peak_k
        # conv[a,a][k] = sum_i a[i]*a[k-i], so the largest contributor
        ki = peak_k
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        contrib = a[idx] * a[j_idx]

        if contrib.max() < 1e-20:
            break

        max_local = np.argmax(contrib)
        src_i = idx[max_local]
        dst_j = j_idx[max_local]

        # Move a small fraction from src to a random low-contribution position
        if a[src_i] > 1e-10:
            # Find a good destination (low autocorrelation sensitivity)
            candidates = rng.choice(n, size=min(50, n), replace=False)
            best_dst = None
            best_trial_obj = best_obj
            for dst in candidates:
                if dst == src_i:
                    continue
                for frac in [0.005, 0.01, 0.02]:
                    amt = a[src_i] * frac
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    if a_trial.sum() < 1e-15:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    obj_trial = compute_objective(a_trial)
                    if obj_trial < best_trial_obj:
                        best_trial_obj = obj_trial
                        best_dst = (dst, frac)

            if best_dst is not None:
                dst, frac = best_dst
                amt = a[src_i] * frac
                a[src_i] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                if best_trial_obj < best_obj:
                    best_obj = best_trial_obj
                    best_a = a.copy()

    return best_a, best_obj


def coordinate_polish(a_init, time_budget, verbose=False):
    """Aggressive coordinate descent polish."""
    n = len(a_init)
    a = normalize_nonneg(a_init.copy())
    best_a = a.copy()
    best_obj = compute_objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    max_stall = 80

    while time.monotonic() - t0 < time_budget and stall < max_stall:
        ac = fft_autocorr(a)
        peak_k = int(ac.argmax())
        M = ac.max()

        ki = peak_k
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        contrib = a[idx] * a[j_idx]

        if contrib.max() < 1e-20:
            break

        max_local = int(np.argmax(contrib))
        src = idx[max_local]

        # Try moving from src to various destinations
        n_try = min(30, n)
        if a[src] > 1e-10:
            dests = np.argsort(a)[:n_try]  # lowest values first
            improved = False
            for dst in dests:
                if dst == src:
                    continue
                for frac in [0.003, 0.008, 0.015, 0.03]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    if a_trial.sum() < 1e-15:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    obj_trial = compute_objective(a_trial)
                    if obj_trial < best_obj - 1e-14:
                        a = a_trial
                        best_obj = obj_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

            if not improved:
                # Random perturbation
                sources = rng.choice(n, size=max(2, n // 200), replace=False)
                any_improved = False
                for s_idx in sources:
                    if a[s_idx] < 1e-10:
                        continue
                    d_idx = int(rng.integers(n))
                    if d_idx == s_idx:
                        continue
                    frac = rng.uniform(0.01, 0.04)
                    amt = a[s_idx] * frac
                    a[s_idx] -= amt
                    a[d_idx] += amt
                    a_tmp = np.maximum(a, 0.0)
                    if a_tmp.sum() > 1e-15:
                        a_tmp = a_tmp / a_tmp.sum()
                        if compute_objective(a_tmp) < best_obj:
                            a = a_tmp
                            best_obj = compute_objective(a)
                            best_a = a.copy()
                            any_improved = True
                            break
                    else:
                        a[s_idx] += amt
                        a[d_idx] -= amt

                if not any_improved:
                    stall += 1
                    # Larger escape
                    n_esc = max(5, n // 100)
                    for _ in range(n_esc):
                        si = int(rng.integers(n))
                        di = int(rng.integers(n))
                        if si != di and a[si] > 1e-8:
                            amt = a[si] * rng.uniform(0.02, 0.06)
                            a[si] -= amt
                            a[di] += amt
                            a = np.maximum(a, 0.0)
                            if a.sum() > 1e-15:
                                a = a / a.sum()
                                if compute_objective(a) < best_obj:
                                    best_obj = compute_objective(a)
                                    best_a = a.copy()
                                    stall = 0
                                    break
                            else:
                                a[si] += amt
                                a[di] -= amt
            else:
                stall = 0
        else:
            stall += 1

        if verbose and stall % 15 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord polish ({elapsed:.1f}s): obj={best_obj:.6f}, stall={stall}")

    return best_a, best_obj


def propose_candidate(budget_s=1800.0):
    """
    Main search: multi-init, multi-length LP optimization with momentum,
    spectral shaping, and coordinate polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    rng = np.random.default_rng(42)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe_local(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Generate initial candidates
    candidates = []  # list of (a, label)

    # Parent
    for n_try in [2400, 2800, 3200]:
        p = gen_parent(n_try)
        if p is not None:
            candidates.append((p, f"parent_n{n_try}"))

    # Raised cosine
    for p_exp in [1.0, 1.5, 2.0, 3.0]:
        for n_try in [2400, 2800]:
            a = gen_raised_cos(n_try, p=p_exp)
            candidates.append((a, f"rc_p{p_exp}_n{n_try}"))

    # Beta
    for alpha in [1.5, 2.0, 2.5, 3.0]:
        a = gen_beta(2400, alpha=alpha, beta=alpha)
        candidates.append((a, f"beta_{alpha}_n2400"))

    # Gaussian
    for sigma in [0.20, 0.25, 0.30, 0.35]:
        a = gen_gaussian(2400, sigma=sigma)
        candidates.append((a, f"gauss_s{sigma}_n2400"))

    # Evaluate initial scores to prioritize
    scored = []
    for a, label in candidates:
        try:
            sc = evaluate_safe_local(a)
            scored.append((sc, a, label))
        except Exception:
            scored.append((float('inf'), a, label))

    scored.sort(key=lambda x: x[0])
    print(f"Generated {len(scored)} candidates, best initial: {scored[0][0]:.6f} ({scored[0][2]})")

    # Deduplicate by n (keep best per n)
    seen_n = {}
    for sc, a, label in scored:
        n = len(a)
        if n not in seen_n or sc < seen_n[n][0]:
            seen_n[n] = (sc, a, label)

    final_candidates = sorted(seen_n.values(), key=lambda x: x[0])

    # Time allocation: give more time to better candidates
    n_candidates = len(final_candidates)
    total_time = time_budget - 60.0  # reserve 60s for polish

    # Allocate time proportional to (1/rank) for top candidates
    weights = []
    for i, (sc, a, label) in enumerate(final_candidates):
        if i == 0:
            w = 3.0
        elif i == 1:
            w = 2.0
        elif i < 4:
            w = 1.5
        elif i < 7:
            w = 1.0
        else:
            w = 0.5
        weights.append(w)

    total_w = sum(weights)
    times = [w / total_w * total_time for w in weights]

    # Limit number of candidates to optimize (top 8)
    n_opt = min(len(final_candidates), 8)

    for i in range(n_opt):
        sc, a, label = final_candidates[i]
        t_alloc = times[i]
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget - 30:
            break
        if t_alloc < 20:
            t_alloc = 20

        print(f"\n=== Optimizing candidate {i+1}/{n_opt}: {label} (initial={sc:.6f}, time={t_alloc:.0f}s) ===")

        # Run LP optimizer
        a_out, obj_out = lp_optimizer(a, t_alloc, verbose=True, use_momentum=True, legacy_momentum=0.4)
        try_update(a_out, f"lp_{label}")

    # === Polish phase ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    print(f"\n=== Polish phase ({remaining:.0f}s remaining) ===")

    if best_seq is not None and remaining > 30:
        n_best = len(best_seq)

        # Coordinate polish
        t_coord = min(remaining * 0.4, 120.0)
        if t_coord > 10:
            print(f"  Coordinate polish (n={n_best}, {t_coord:.0f}s)")
            a_out, _ = coordinate_polish(best_seq, t_coord, verbose=True)
            try_update(a_out, f"coord_polish_n{n_best}")

        # LP polish
        elapsed = time.monotonic() - t_global
        remaining = time_budget - elapsed
        if remaining > 30:
            t_lp = min(remaining * 0.4, 150.0)
            if t_lp > 15:
                print(f"  LP polish (n={n_best}, {t_lp:.0f}s)")
                a_out, _ = lp_optimizer(best_seq, t_lp, verbose=True, use_momentum=False, legacy_momentum=0)
                try_update(a_out, f"lp_polish_n{n_best}")

        # Try slightly different lengths
        elapsed = time.monotonic() - t_global
        remaining = time_budget - elapsed
        if remaining > 60:
            for delta in [200, -200, 400]:
                n_try = n_best + delta
                if n_try < 800 or n_try > 4000:
                    continue
                a_interp = interpolate_to_n(best_seq, n_try)
                t_try = min(remaining * 0.25, 100.0)
                if t_try < 15:
                    continue
                print(f"  Trying n={n_try} ({t_try:.0f}s)")
                a_out, _ = lp_optimizer(a_interp, t_try, verbose=False, use_momentum=True, legacy_momentum=0.4)
                try_update(a_out, f"lp_n{n_try}")
                elapsed = time.monotonic() - t_global
                remaining = time_budget - elapsed
                if remaining < 30:
                    break

        # Final coord polish
        elapsed = time.monotonic() - t_global
        remaining = time_budget - elapsed
        if best_seq is not None and remaining > 15:
            t_final = min(remaining * 0.6, 60.0)
            if t_final > 5:
                print(f"  Final coordinate polish ({t_final:.0f}s)")
                a_out, _ = coordinate_polish(best_seq, t_final, verbose=False)
                try_update(a_out, "final_coord")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Returning fallback.")
        # Fallback: uniform
        best_seq = np.ones(2400) / 2400.0
        best_score = evaluate_safe_local(best_seq)
        print(f"  Fallback score: {best_score:.6f}")

    return [float(x) for x in best_seq]
