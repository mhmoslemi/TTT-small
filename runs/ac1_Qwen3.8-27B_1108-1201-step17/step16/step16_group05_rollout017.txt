def fast_autocorr(a):
    """Compute linear autocorrelation (convolution with itself) via FFT."""

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mirror_descent_smoothed(a0, time_limit, beta_start=30.0, beta_end=5.0,
                            top_k=200, verbose=True, seed=42):

    Mirror descent (multiplicative weights) on the simplex for min max autocorrelation.
    Uses log-sum-exp smoothing of the max.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Precompute for speed
    stall = 0
    max_stall = 2000
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta
        frac = min(1.0, t / 30000.0)
        cur_beta = beta_start + (beta_end - beta_start) * frac
        
        # Compute smoothed gradient
        g, M_curr = compute_gradient_smoothed(a, cur_beta, top_k=top_k)
        
        # Mirror descent update: a_i <- a_i * exp(-eta * g_i) / Z
        # Step size: eta ~ 1/sqrt(t) for optimal convergence
        eta = 1.0 / np.sqrt(t + 1.0)
        
        # To avoid underflow, use log-space
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - eta * g
        log_a_new -= log_a_new.max()  # numerical stability
        a_new = np.exp(log_a_new)
        a_new /= a_new.sum()
        
        # Check improvement
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            a = a_new  # Still update a to explore
            stall += 1
        
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.4)
            # Restart with perturbation
            noise = rng.normal(0, 0.003, n) * best_a
            a_re = np.maximum(best_a + noise, 1e-15)
            a = a_re / a_re.sum()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      MD restart t={t} ({elapsed:.0f}s): J={J:.6f}")
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      MD t={t} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.1f}")
    
    return best_a, best_m


def accelerated_projected_gradient_v2(a0, time_limit, beta_start=25.0, beta_end=6.0,

        eta = 2.0 / (np.sqrt(t) + 1.0)

            print(f"      APG t={t} ({elapsed:.0f}s): J={J:.6f}")

def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with Nesterov acceleration - optimized version."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Build constraint matrix using the full convolution structure
        # T @ b = y * b (convolution), T is (2n-1) x nToeplitz-like
        # For efficiency, only use top constraints
        n_lags = 2 * n - 1
        top_k_constraint = min(n_lags, 3 * n)
        
        # Build the (2n-1) x n matrix where row k gives (y*b)[k]
        # (y*b)[k] = sum_j y[j] * b[k-j] for valid j
        # So T[k, j] = y[j] if 0 <= k-j < n, else 0
        # T[k, j] = y[j] * [0 <= k-j < n] = y[j] * [k-n+1 <= j <= k]
        
        b_var = cp.Variable(n, nonneg=True)
        
        # Build T efficiently
        T = np.zeros((n_lags, n))
        for j in range(n):
            lo = max(0, j - n + 1)
            hi = min(n_lags - 1, j)
            # Wait, T[k, j] = y[j] if 0 <= k-j < n
            # k-j >= 0 => k >= j
            # k-j < n => k < j+n => k <= j+n-1
            # So for fixed j: k ranges from j to j+n-1, clipped to [0, n_lags-1]
            k_lo = max(0, j)
            k_hi = min(n_lags - 1, j + n - 1)
            T[k_lo:k_hi+1, j] = y[j]
        
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new
        
        mc = fast_autocorr(y).max()
        if mc < best_m - 1e-15:
            best_m = mc
            best_a = y.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def coordinate_polish_v2(a0, time_limit, verbose=True, seed=123):
    """
    Coordinate descent focused on reducing peak autocorrelation.
    Strategy: identify positions that contribute most to peak lags,
    move their mass to positions with lower peak contribution.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top peak lags
        top_k = np.argsort(ac)[::-1][:min(30, len(ac))]
        
        # Compute per-position contribution to each peak lag
        # c[k] = sum_i a[i]*a[k-i], position j contributes a[j]*a[k-j] to c[k]
        peak_contrib = np.zeros(n)
        for k in top_k[:15]:
            k_int = int(k)
            j_lo = max(0, k_int - n + 1)
            j_hi = min(k_int, n - 1)
            if j_lo > j_hi:
                continue
            js = np.arange(j_lo, j_hi + 1)
            peak_contrib[js] += 2.0 * a[js] * a[k_int - js]
        
        # Try to move mass from high-contribution positions to low-contribution ones
        sources = np.argsort(peak_contrib)[::-1][:50]
        improved = False
        
        for src in sources[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            
            # Find best destination: minimize increase in peak autocorrelation
            # A position dst is good if it doesn't contribute much to peak lags
            best_dst = -1
            best_score = 0.0
            
            # Sample candidate destinations
            n_cand = min(n, 1000)
            if n_cand < n:
                candidates = rng.choice(n, n_cand, replace=False)
            else:
                candidates = np.arange(n)
            
            for dst in candidates:
                if dst == src:
                    continue
                # Score: how much would adding mass at dst increase peaks?
                # If we add delta at dst, c[k] increases by delta * (a[k-dst] + a[dst-k]) for each k
                # We want to minimize the max increase over peak lags
                max_increase = 0.0
                for k in top_k[:10]:
                    k_int = int(k)
                    inc = 0.0
                    if 0 <= k_int - dst < n:
                        inc += a[k_int - dst]
                    if 0 <= dst - k_int < n:
                        inc += a[dst - k_int]
                    max_increase = max(max_increase, inc)
                score = -max_increase  # We want to minimize max_increase
                if score > best_score:
                    best_score = score
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            # Try moving mass
            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            n_try = min(100, n)
            for _ in range(n_try):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-8:
                    continue
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        break
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      CD done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror descent (multiplicative weights) with adaptive smoothed-max objective
2. Nesterov-accelerated projected gradient with adaptive top-k
3. LP Frank-Wolfe with active-set constraint reduction
4. Peak-aware coordinate descent for final polish
5. Multi-length exploration and diverse initializations
"""

import numpy as np
import cvxpy as cp
import time

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_smoothed(a, beta, top_k=200):
    n = len(a)
    ac = fast_autocorr(a)
    M_curr = ac.max()
    k_eff = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_eff]
    top_vals = ac[top_idx]
    shifted = top_vals - top_vals[0]
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        j_lo = max(0, k_int - n + 1)
        j_hi = min(k_int, n - 1)
        if j_lo > j_hi:
            continue
        js = np.arange(j_lo, j_hi + 1)
        partner = k_int - js
        grad[js] += 2.0 * wi * a[partner]
    return grad, M_curr


def mirror_descent_smoothed(a0, time_limit, beta_start=30.0, beta_end=5.0,
                            top_k=200, verbose=True, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 2000
    current_top_k = top_k

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        frac = min(1.0, t / 30000.0)
        cur_beta = beta_start + (beta_end - beta_start) * frac
        g, M_curr = compute_gradient_smoothed(a, cur_beta, top_k=current_top_k)
        eta = 1.0 / np.sqrt(t + 1.0)
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new /= a_new.sum()
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        a = a_new
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.4)
            noise = rng.normal(0, 0.003, n) * best_a
            a_re = np.maximum(best_a + noise, 1e-15)
            a = a_re / a_re.sum()
            current_top_k = min(current_top_k + 20, 350)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      MD restart t={t} ({elapsed:.0f}s): J={J:.6f}, tk={current_top_k}")
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      MD t={t} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.1f}")
    return best_a, best_m


def accelerated_projected_gradient(a0, time_limit, beta_start=25.0, beta_end=6.0,
                                   top_k=200, verbose=True, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 3000
    current_top_k = top_k

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        frac = min(1.0, t / 25000.0)
        cur_beta = beta_start + (beta_end - beta_start) * frac
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        g, _ = compute_gradient_smoothed(y, cur_beta, top_k=current_top_k)
        eta = 2.0 / (np.sqrt(t) + 1.0)
        x_new = project_simplex(y - eta * g)
        x_prev = x
        x = x_new
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.3)
            noise = rng.normal(0, 0.004, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x = x_re / x_re.sum()
            x_prev = x.copy()
            current_top_k = min(current_top_k + 30, 400)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      APG restart t={t} ({elapsed:.0f}s): J={J:.6f}, tk={current_top_k}")
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      APG t={t} ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Use only top constraints for speed
        n_lags = 2 * n - 1
        n_cons = min(n_lags, 2 * n)
        top_constraint = np.argsort(ac)[::-1][:n_cons]

        b_var = cp.Variable(n, nonneg=True)

        # Build T for only selected lags
        T = np.zeros((n_cons, n))
        for idx, k in enumerate(top_constraint):
            k_int = int(k)
            j_lo = max(0, k_int - n + 1)
            j_hi = min(k_int, n - 1)
            if j_lo <= j_hi:
                T[idx, j_lo:j_hi + 1] = y[k_int - np.arange(j_lo, j_hi + 1)]

        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new

        mc = fast_autocorr(y).max()
        if mc < best_m - 1e-15:
            best_m = mc
            best_a = y.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def get_parent_sequence():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

def try_update_seq(best_state, a, label, t_global):
    """Try to update best state with candidate sequence a."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
