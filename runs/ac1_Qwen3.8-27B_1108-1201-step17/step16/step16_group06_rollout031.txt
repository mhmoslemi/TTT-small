    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def eval_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_list = [float(x) for x in a]
    try:
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


defObjective_val(a):
    a = np.asarray(a, dtype=np.float64)
    ac = fast_autocorr(a)
    return 2.0 * len(a) * ac.max() / (a.sum() ** 2)


def objective_val(a):
    a = np.asarray(a, dtype=np.float64)
    ac = fast_autocorr(a)
    return 2.0 * len(a) * ac.max() / (a.sum() ** 2)


def build_T_dense(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a[j]
    return T


def lp_frank_wolfe(a0, time_limit, solver_name, verbose=True, max_iters=100000):
    import cvxpy as cp
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    x_prev = a.copy()

    # Pre-allocate LP variables
    b_var = cp.Variable(n, nonneg=True)

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        T = build_T_dense(y)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Backtracking line search
        best_t = 0.1
        best_f = objective_val(y)
        alpha = 2.0 / (t + 2.0)

        # Try Nesterov step and a few alternatives
        candidates_t = [alpha, 2.0 * alpha, 0.5 * alpha, 1.0 / (t + 1.0), 0.5 / (t + 1)]
        candidates_t = np.clip(candidates_t, 0.01, 1.0)

        for tc in candidates_t:
            y_trial = (1.0 - tc) * y + tc * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            sy = y_trial.sum()
            if sy < 1e-15:
                continue
            y_trial = y_trial / sy
            m_trial = fast_autocorr(y_trial).max()
            f_trial = 2.0 * n * m_trial
            if f_trial < best_f:
                best_f = f_trial
                best_t = tc

        # Nesterov update
        y_new = (1.0 - best_t) * y + best_t * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.10f}")

    return best_a, best_m


def smooth_gradient_polish(a0, time_limit, verbose=True):
    """Mirror descent with softmax-smoothed objective on the probability simplex."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    beta_start = 5.0
    beta_end = 80.0
    max_steps = 50000
    gamma = 3.0

    for t in range(1, max_steps + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive beta: start moderate, increase over time
        frac = min(1.0, t / (max_steps * 0.8))
        beta = beta_start + (beta_end - beta_start) * frac

        ac = fast_autocorr(a)
        # Softmax weights on the convolution values
        ac_shifted = ac - ac.max()
        w = np.exp(beta * ac_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(ac)) / len(ac)
        else:
            w = w / w_sum

        # Gradient of smoothed objective w.r.t. a
        # d(c(k))/d(a[i]) = a[k-i] + a[i-k] (when valid)
        grad = np.zeros(n)
        # Use FFT-based approach: the gradient is related to convolution of w with a
        # Specifically: grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
        # This is 2 * (w * a_flipped) convolved appropriately
        # Let's compute directly using the relationship:
        # sum_k w[k] * a[k-i] = (w * a)[i] reversed... 
        # Actually: sum_k w[k]*a[k-i] for valid k,i is a cross-correlation
        # Let's use FFT: cross-corr(w, a)[i] = sum_k w[k]*a[k+i]
        # And we need sum_k w[k]*a[k-i] = cross-corr(w, a)[-i] = cross-corr(a, w)[i]
        
        # Compute w * a (convolution) and w * a_flipped
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        W = np.fft.rfft(w, fft_size)
        A = np.fft.rfft(a, fft_size)
        
        # conv(w, a)[m] = sum_k w[k] * a[m-k]
        # We want sum_k w[k] * a[k-i] = conv(w, a)[k-i] summed... 
        # Actually: sum_k w[k] * a[k-i] for fixed i
        # Let j = k-i, so k = j+i: sum_j w[j+i] * a[j]
        # = (w_shifted * a)[i] where w_shifted depends on i
        # This is the cross-correlation: <w, shift_i(a)>
        
        # Simpler: grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
        # = sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k]
        # First term: let m = k-i, sum_m w[m+i]*a[m] = cross_corr(w, a, lag=-i)
        # Second term: let m = i-k, sum_m w[i-m]*a[m] = cross_corr(w, a, lag=i)
        # So grad[i] = cross_corr(w,a,-i) + cross_corr(w,a,i)
        # And cross_corr(w,a,lag) = ifft(fft(w) * conj(fft(a))) shifted
        # For real signals: cross_corr = ifft(conj(fft(w)) * fft(a)) ... let me just do it directly
        
        # Direct computation using convolution:
        # sum_k w[k]*a[k-i]: this is (w * a_flip)[n-1-i] where a_flip = a[::-1]
        # Actually let me just use the full convolution and index appropriately
        wa = np.fft.irfft(W * A, fft_size)[:2 * len(w) - 1]
        # wa[m] = sum_k w[k]*a[m-k]
        # sum_k w[k]*a[k-i] = sum_k w[k]*a[k-i]... 
        # Let's use: for a fixed i, sum over k of w[k]*a[k-i]
        # = sum over k where 0<=k-i<n of w[k]*a[k-i]
        # = sum over j where 0<=j<n, k=j+i of w[j+i]*a[j]
        # This is the cross-correlation of w and a at lag i
        # cross_corr(w, a, i) = sum_j w[j+i]*a[j]
        # = ifft(fft(w) * conj(fft(a_spaced)))[i] ... 
        
        # Let me just do it the straightforward way with FFT cross-correlation
        # cross_corr(w, a)[d] = sum_j w[j+d]*a[j] for d = -(n-1), ..., n-1
        # Using FFT: cross_corr = ifft(conj(fft(w)) * fft(a)) (with proper padding)
        # Actually for cross-correlation: <w, R^{-d}a> where R is reversal
        # cross_corr(w,a,d) = sum_j w[j+d]*a[j] = (w * a_rev)[n-1+d] where a_rev = a[::-1]
        
        a_rev = a[::-1]
        Wa = np.fft.irfft(W * np.fft.rfft(a_rev, fft_size), fft_size)[:2 * n - 1]
        # Wa[m] = sum_j w[j] * a_rev[m-j] = sum_j w[j] * a[n-1-(m-j)] = sum_j w[j] * a[n-1-m+j]
        # Let k = n-1-m+j, j = k-n+1+m: Wa[m] = sum_k w[k-n+1+m] * a[k]
        # So cross_corr(w,a,d) = sum_j w[j+d]*a[j]
        # = Wa[n-1-d]... let me verify with a small example.
        # w = [w0, w1], a = [a0, a1], n=2
        # cross_corr(w,a,0) = w[0]*a[0] + w[1]*a[1]
        # cross_corr(w,a,1) = w[1]*a[0]
        # cross_corr(w,a,-1) = w[0]*a[1]
        # Wa: fft(w) * fft(a_rev), a_rev = [a1, a0]
        # conv(w, a_rev) = [w0*a1, w0*a0+w1*a1, w1*a0]
        # Wa[0] = w0*a1, Wa[1] = w0*a0+w1*a1, Wa[2] = w1*a0
        # cross_corr(d=0) = w0*a0+w1*a1 = Wa[1] = Wa[n-1-0] = Wa[n-1]
        # cross_corr(d=1) = w1*a0 = Wa[2] = Wa[n-1+1]... = Wa[n+d-1]
        # cross_corr(d=-1) = w0*a1 = Wa[0] = Wa[n-1-1] = Wa[n-1+d]
        # So cross_corr(w,a,d) = Wa[n-1+d]
        
        # grad[i] = cross_corr(w,a,-i) + cross_corr(w,a,i) = Wa[n-1-i] + Wa[n-1+i]
        valid = np.arange(n)
        idx1 = n - 1 - valid  # n-1-i for i in [0, n-1]
        idx2 = n - 1 + valid  # n-1+i for i in [0, n-1]
        idx1 = idx1[idx1 < len(Wa)]
        idx2 = idx2[idx2 < len(Wa)]
        v1 = valid[valid < len(idx1)]
        v2 = valid[valid < len(idx2)]
        grad[v1] += Wa[idx1]
        grad[v2] += Wa[idx2]

        # Mirror descent step (multiplicative weights on simplex)
        eta = gamma / (np.sqrt(t) + 1.0)
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
        a = a_new

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothGrad t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Targeted coordinate descent on peak-contributing positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top peaks
        top_k = min(30, len(ac))
        peak_indices = np.argsort(ac)[::-1][:top_k]
        
        # Compute sensitivity: for each position i, how much it contributes to top peaks
        sensitivity = np.zeros(n)
        for pk in peak_indices[:15]:
            pk_int = int(pk)
            for pos in range(n):
                d1 = pk_int - pos
                if 0 <= d1 < n:
                    sensitivity[pos] += a[d1]
                d2 = pos - pk_int
                if 0 <= d2 < n:
                    sensitivity[pos] += a[d2]
        
        # Try moving mass from high-sensitivity to low-sensitivity positions
        improved = False
        n_src = min(20, n)
        n_dst = min(30, n)
        src_candidates = np.argsort(sensitivity)[::-1][:n_src]
        dst_candidates = np.argsort(sensitivity)[:n_dst]
        
        for src in src_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue
            for dst in dst_candidates:
                if int(dst) == int(src):
                    continue
                # Try several transfer fractions
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                    amt = a[int(src)] * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[int(src)] -= amt
                    a_trial[int(dst)] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 50 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()
        
        if verbose and stall % 200 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordPD stall={stall} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lp_frank_wolfe_fast_stops(a0, time_limit, solver_name, verbose=False):
    """LP-FW that stops early if no improvement for a while."""
    import cvxpy as cp
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    x_prev = a.copy()

    b_var = cp.Variable(n, nonneg=True)
    no_improve_count = 0
    max_no_improve = 300

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve_count > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        T = build_T_dense(y)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            no_improve_count += 1
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve_count += 1
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            no_improve_count += 1
            continue
        g_dir = b_val / b_sum

        # Backtracking
        alpha = 2.0 / (t + 2.0)
        best_t = alpha
        best_m_y = fast_autocorr(y).max()
        for tc in [alpha, 1.5 * alpha, 0.5 * alpha, min(1.0, 2.0 / t)]:
            y_trial = (1.0 - tc) * y + tc * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            sy = y_trial.sum()
            if sy < 1e-15:
                continue
            y_trial = y_trial / sy
            m_trial = fast_autocorr(y_trial).max()
            if m_trial < best_m_y:
                best_m_y = m_trial
                best_t = tc

        y_new = (1.0 - best_t) * y + best_t * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        mx = fast_autocorr(x).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            no_improve_count = 0
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = eval_score(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = eval_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: LP-FW main optimization (45% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 1: LP-FW main n={n0} ===")
        t1 = min(remaining * 0.45, 800.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t1, solver, verbose=True)
        try_update(a_out, f"lp_main_n{n0}")

    # === Phase 2: Smooth gradient polish (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: Smooth gradient polish n={n_cur} ===")
        t2 = min(remaining * 0.15, 200.0)
        a_out, _ = smooth_gradient_polish(norm_sum1(best_seq), t2, verbose=True)
        try_update(a_out, f"smooth_n{n_cur}")

    # === Phase 3: LP-FW round 2 (20% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: LP-FW round 2 n={n_cur} ===")
        t3 = min(remaining * 0.20, 350.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t3, solver, verbose=True)
        try_update(a_out, f"lp_r2_n{n_cur}")

    # === Phase 4: Coordinate descent polish (10% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: Coordinate polish n={n_cur} ===")
        t4 = min(remaining * 0.10, 150.0)
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t4, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_cur}")

    # === Phase 5: Multi-n exploration (10% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 5: Multi-n exploration ===")
        n_best = len(best_seq)
        deltas = [200, 400, -200, 600, -400]
        tried = 0
        for delta in deltas:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40 or tried >= 3:
                break
            nt = n_best + delta
            if 800 <= nt <= 4000:
                print(f"  Trying n={nt}...")
                a_inter = interpolate_sequence(best_seq, nt)
                t5 = min(remaining * 0.15, 120.0)
                a_out, _ = lp_frank_wolfe_fast_stops(norm_sum1(a_inter), t5, solver, verbose=False)
                try_update(a_out, f"multin_n{nt}")
                tried += 1

    # === Phase 6: Final LP-FW with remaining time ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 6: Final LP-FW n={n_best} ===")
        t6 = remaining * 0.8
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t6, solver, verbose=True)
        try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Final coord polish n={n_best} ===")
        t7 = remaining * 0.9
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t7, verbose=False, rng_seed=777)
        try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
