```python
def pnorm_gradient(a, ac, p=50):
    """Compute gradient of (sum ac^p / len(ac))^(1/p) w.r.t. a."""
    n = len(a)
    ac_clipped = np.maximum(ac, 1e-30)
    ac_p = ac_clipped ** p
    total = ac_p.sum()
    if total < 1e-30:
        return np.zeros(n)
    # d(pnorm)/da[j] = (total)^((1/p)-1) / (2n-1) * sum_k p * ac_k^(p-1) * 2*a[k-j] * 1[j in S(k)]
    # Simplify: proportional to sum_k ac_k^(p-1) * 2*a[k-j] * 1[j in S(k)]
    weights = ac_clipped ** (p-1)
    # Normalize weights for numerical stability
    weights = weights / weights.max()
    
    grad = np.zeros(n)
    for k in range(len(ac)):
        w = weights[k]
        if w < 1e-10:
            continue
        k_int = k
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += w * a[j_idx]
    return 2.0 * grad
```

```python
    # Compute weights
    log_weights = (p-1) * np.log(ac_clipped)
    log_weights -= log_weights.max()
    weights = np.exp(log_weights)
    
    # Only keep top-k weights
    if top_k < len(weights):
        top_idx = np.argsort(weights)[::-1][:top_k]
        weights_sparse = np.zeros_like(weights)
        weights_sparse[top_idx] = weights[top_idx]
        weights = weights_sparse
    
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    
    grad = np.zeros(n)
    for k in np.where(weights > 1e-10)[0]:
        w = weights[k]
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += w * a[j_idx]
    
    return 2.0 * grad / wsum
```

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_gradient_pnorm(a, ac, p=40, top_k=120):
    """Compute gradient of p-norm of ac (smooth approximation to max)."""
    n = len(a)
    ac_clipped = np.maximum(ac, 1e-30)
    log_w = (p - 1) * np.log(ac_clipped)
    log_w -= log_w.max()
    weights = np.exp(log_w)
    
    # Sparse: keep top-k
    if top_k < len(weights):
        idx = np.argsort(weights)[::-1][:top_k]
        ws = np.zeros(len(weights))
        ws[idx] = weights[idx]
        weights = ws
    
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights /= wsum
    
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for k in active:
        ki = int(k)
        i_lo = max(0, ki - n + 1)
        i_hi = min(n, ki + 1)
        if i_lo >= i_hi:
            continue
        idx = np.arange(i_lo, i_hi)
        grad[idx] += weights[k] * a[ki - idx]
    return 2.0 * grad

def compute_gradient_max(a, ac, top_k=20):
    """Compute subgradient using top-k lags uniformly weighted."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    grad = np.zeros(n)
    for k in top_idx:
        ki = int(k)
        i_lo = max(0, ki - n + 1)
        i_hi = min(n, ki + 1)
        if i_lo >= i_hi:
            continue
        idx = np.arange(i_lo, i_hi)
        grad[idx] += (1.0 / k_max) * a[ki - idx]
    return 2.0 * grad

def nesterov_mirror_descent(a0, time_limit, initial_temp=0.01, final_temp=0.001,

    """Nesterov mirror descent with p-norm smoothed objective."""
    n = len(a0)
    a = normalize(a0)

    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 2000
    rng = np.random.default_rng(42)
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive p: start moderate, increase to get closer to max
        frac = min(1.0, t / 5000.0)
        p_val = p_start + (p_end - p_start) * frac
        temp = initial_temp * (1 - frac) + final_temp * frac
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        ac = fast_autocorr(y)

        else:
            g = compute_gradient_max(y, ac, top_k=30)
        
        # Mirror descent update: a_new[i] ∝ y[i] * exp(-eta * g[i])
        eta = 2.0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > max_stall:
            stall = 0
            max_stall = max(500, max_stall // 2)
            pert = rng.normal(0, 0.01, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)

def simulated_annealing(a0, time_limit, verbose=True, rng_seed=999):
    """Simulated annealing for local optimization."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    T_start = 0.05
    T_end = 0.001
    current_m = best_m
    
    for step in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        frac = min(1.0, step / 10000.0)
        T = T_start * (T_end / T_start) ** frac
        
        # Propose a move: transfer mass between two random coordinates
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if a[i] < 1e-10:
            continue
        
        amt = a[i] * rng.uniform(0.001, 0.05)
        a[i] -= amt
        a[j] += amt
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        
        m_new = fast_autocorr(a).max()
        delta = m_new - current_m
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_m = m_new
            if m_new < best_m:
                best_m = m_new
                best_a = a.copy()
        else:
            # Reject: undo
            a[j] -= amt
            a[i] += amt
            a = a / a.sum()
        
        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SA step={step} ({elapsed:.1f}s): J={J:.6f}, T={T:.5f}")
    
    return best_a, best_m
