def interp_to(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n

    old_x = np.linspace(0.0, 1.0, old_n)

    f_interp = np.interp(new_x, old_x, f_norm)

    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def build_conv_matrix(y):

def spectral_init(n, rng):
    """Create a sequence from random phases: |IFFT(exp(i*phi))|^2 type construction."""
    phases = rng.uniform(0, 2 * np.pi, n)
    spectrum = np.exp(1j * phases)
    b = np.real(np.fft.ifft(spectrum))
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s < 1e-10:
        b = np.ones(n) / n
    else:
        b = b / s
    return b

def multi_spectral_init(n, rng):
    """Average multiple random-phase constructions for smoother result."""
    n_ave = 20
    b_sum = np.zeros(n)
    for _ in range(n_ave):
        b_sum += spectral_init(n, rng)
    return b_sum / n_ave

            try_update(a_out, "lp_r2_n" + str(n_best))

    # Phase 9: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        print("\n=== Phase 9: Final coord polish n=" + str(n_best) + " ===")
        t9 = min(remaining * 0.5, 120.0)
        a_out = coordinate_polish(normalize(best_seq), t9, verbose=False, seed=777)
        try_update(a_out, "coord_final_n" + str(n_best))

    # Phase 10: Final LP if time remaining
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        print("\n=== Phase 10: Final LP n=" + str(n_best) + " ===")
        t10 = min(remaining * 0.8, 150.0)
        a_out = lp_frank_wolfe(normalize(best_seq), t10, solver, verbose=True)
        try_update(a_out, "lp_final_n" + str(n_best))

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print("\nFINAL: score=" + str(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(time.monotonic() - t_global)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
