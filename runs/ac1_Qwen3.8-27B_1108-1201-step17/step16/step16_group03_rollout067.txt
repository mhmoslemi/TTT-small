```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2.

Strategy (differs from prior art mainly in the LP-FW engine and orchestration):
  1. Multi-start: diverse bell-shaped / window initializations across a few lengths,
     briefly warmed with Nesterov mirror descent to pick a promising basin.
  2. Nesterov accelerated mirror descent (fast workhorse) on the simplex, using the
     exact subgradient of the dominant (and occasionally top-k) convolution peaks,
     with adaptive step size and stochastic restarts.
  3. LP-based Frank-Wolfe with an ADAPTIVE constraint set: a fast reduced LP that
     enforces only the top-K active peaks for the bulk of the run (10-50x faster per
     iter), switching to the full (2n-1)-constraint LP for the finishing phase.
     Direction LP:  min t  s.t.  T@b <= t, sum(b)=1, b>=0.
  4. Coordinate / 2-opt mass-transfer polish targeting the active peaks.
  5. Multi-length exploration via monotone linear interpolation + re-optimization.

All helpers are top-level (no closures, no lambdas). Global mutable state is carried
in the module-level list G to satisfy the no-closure rule.
"""

import numpy as np
import cvxpy as cp
import time

# ----------------------------------------------------------------------------
# Global solver cache and shared mutable state (avoids closures / lambdas)
# ----------------------------------------------------------------------------
_global_solver = None
_G = {"best_seq": None, "best_score": float("inf"), "t0": 0.0, "budget": 1700.0}


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(16, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ("optimal", "optimal_inaccurate"):
                _global_solver = s
                print("  [setup] LP solver:", s)
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  [setup] LP solver (fallback): ECOS")
    return _global_solver


# ----------------------------------------------------------------------------
# Numerical helpers
# ----------------------------------------------------------------------------
def fast_conv(a):
    """Full linear convolution a*a via FFT (real symmetric result)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float("inf")


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_from_max(m, n):
    return 2.0 * n * m


def build_conv_matrix(a, peak_indices=None):
    """T with (T @ b)[k] = (a*b)[k]; only the requested rows are built."""
    n = len(a)
    if peak_indices is None:
        idx = np.arange(2 * n - 1)
    else:
        idx = np.asarray(peak_indices)[:, 0] if peak_indices.ndim == 2 else np.asarray(peak_indices)
    T = np.zeros((len(idx), n))
    for row in range(len(idx)):
        k = int(idx[row])
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        width = je - js
        if width > 0:
            T[row, js:je] = a[k - js:k - js + width]
    return T


def make_init(n, shape):
    x = np.linspace(0.0, 1.0, n)
    if shape == "uniform":
        a = np.ones(n)
    elif shape == "gaussian":
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    elif shape == "sine":
        a = np.sin(np.pi * x) ** 2
    elif shape == "raised_cosine":
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif shape == "flat_top":
        a = 0.54 + 0.46 * np.cos(2.0 * np.pi * x)
    elif shape == "triangular":
        a = 1.0 - np.abs(2.0 * x - 1.0)
    elif shape == "dolph":
        # cheap Dolph–Chebyshev-ish envelope: raised cosine squared
        a = (0.5 * (1.0 + np.cos(2.0 * np.pi * x))) ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    a = norm_sum1(a)
    return a


def interpolate_sequence(f, new_n):
    """Monotone-preserving linear resample of a normalized sequence to new_n."""
    f = np.asarray(f, dtype=np.float64)
    old_n = len(f)
    f = f / f.sum()
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a = np.interp(new_x, old_x, f)
    return norm_sum1(a)


# ----------------------------------------------------------------------------
# State update helper (no closure: uses module-level _G)
# ----------------------------------------------------------------------------
def try_update(a, label):
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < _G["best_score"]:
        _G["best_score"] = sc
        _G["best_seq"] = a_norm.copy()
        elapsed = time.monotonic() - _G["t0"]
        print("  *** NEW BEST [%s]: %.6f (n=%d)  t=%.0fs" % (label, sc, len(a), elapsed))


def remaining_time():
    return _G["budget"] - (time.monotonic() - _G["t0"])


# ----------------------------------------------------------------------------
# Optimizer 1: Nesterov accelerated mirror descent on the simplex
# ----------------------------------------------------------------------------
def peak_subgrad(y, k):
    n = len(y)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n, k + 1)
    for j in range(js, je):
        jj = k - j
        if 0 <= jj < n:
            grad[j] += 2.0 * y[jj]
    return grad


def nesterov_md(a0, time_limit, verbose=False, seed=42, c0=5.0, restart_every=4000):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    ac0 = fast_conv(x)
    best_m = ac0.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    t = 0
    stall = 0
    use_topk = False

    while time.monotonic() - t0 < time_limit:
        t += 1
        # Nesterov momentum extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_conv(y)
        top_idx = np.argsort(ac)[::-1]

        # Alternate single-peak (tight) and top-3 subgradients
        if use_topk:
            grad = peak_subgrad(y, int(top_idx[0])) + peak_subgrad(y, int(top_idx[1]))
            if len(top_idx) > 2:
                grad += peak_subgrad(y, int(top_idx[2]))
            eta = (c0 / (np.sqrt(t) + 1.0)) * 0.6
        else:
            grad = peak_subgrad(y, int(top_idx[0]))
            eta = c0 / (np.sqrt(t) + 1.0)
        use_topk = (t % 500) < 100  # 1-in-5 blocks use multi-peak

        # Mirror (entropic) update
        log_a = np.log(y) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Stochastic restart to escape plateaus
        if stall > restart_every:
            pert = 0.004 + 0.002 * (stall % 5)
            x_r = best_a + rng.normal(0.0, pert, n) * best_a
            x_r = norm_sum1(x_r)
            x = x_r
            x_prev = x_r
            stall = 0

        if verbose and t % 4000 == 0:
            el = time.monotonic() - t0
            print("    [NMD] t=%d (%.0fs) score=%.6f stall=%d" % (t, el, 2.0 * n * best_m, stall))

    return best_a, best_m


# ----------------------------------------------------------------------------
# Optimizer 2: LP-based Frank-Wolfe with adaptive (reduced/full) constraint set
# ----------------------------------------------------------------------------
def lp_fw(a0, time_limit, solver, verbose=True, reduced_k=150,
          warm_iters=8, full_after_frac=0.80):
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac0 = fast_conv(y)
    best_m = ac0.max()
    t0 = time.monotonic()
    it = 0

    b_var = cp.Variable(n, nonneg=True)

    while time.monotonic() - t0 < time_limit:
        it += 1
        frac = (time.monotonic() - t0) / time_limit
        use_full = (it <= warm_iters) or (frac >= full_after_frac)

        ac = fast_conv(y)
        if use_full:
            idx = np.arange(2 * n - 1)
        else:
            idx = np.argsort(ac)[::-1][:reduced_k]
        T = build_conv_matrix(y, idx)

        t_var = cp.Variable()
        constraints = [cp.sum(b_var) == 1, T @ b_var <= t_var]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ("optimal", "optimal_inaccurate"):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum

        # Polyak-style-like adaptive step: blend with diminishing step
        alpha = 2.0 / (it + 2.0)
        # try a couple of candidate alphas and keep the best (cheap line search)
        y_new = None
        best_trial_m = fast_conv(y).max()
        for a_cand in (alpha, min(1.0, alpha * 2.0), min(1.0, alpha + 0.1)):
            trial = (1.0 - a_cand) * y + a_cand * direction
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            mt = fast_conv(trial).max()
            if mt < best_trial_m:
                best_trial_m = mt
                y_new = trial
        if y_new is None:
            y_new = y

        y = y_new
        M_new = y_new.sum() and fast_conv(y).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = y.copy()

        if verbose and it % 40 == 0:
            el = time.monotonic() - t0
            print("    [LP-FW] it=%d (%.0fs) score=%.6f full=%s" %
                  (it, el, 2.0 * n * best_m, use_full))

    return best_a, best_m

# Optimizer 3: coordinate / 2-opt mass-transfer polish
# ----------------------------------------------------------------------------
def coord_descend(a0, time_limit, verbose=False, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        top_peaks = np.argsort(ac)[::-1][:min(16, 2 * n - 1)]
        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_i = int(k)
            js = max(0, k_i - n + 1)
            je = min(n, k_i + 1)
            idx = np.arange(js, je)
            prod = a[idx] * a[k_i - idx]
            if prod.size == 0 or prod.max() < 1e-20:
                continue
            src = int(idx[int(np.argmax(prod))])

            # mark indices that contribute to the top peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:8]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True

            cands = np.where(~in_peak)[0]
            if cands.size == 0:
                cands = np.arange(n)
            nc = min(cands.size, 250)
            chosen = cands[rng.permutation(cands.size)[:nc]]

            best_dst = -1
            best_sens = float("inf")
            for dst in chosen:
                if dst == src:
                    continue
                sens = 0.0
                for k2 in top_peaks[:6]:
                    j2 = int(k2) - dst
                    if 0 <= j2 < n:
                        sens += a[j2]
                if sens < best_sens:
                    best_sens = sens
                    best_dst = dst
            if best_dst < 0:
                continue
            dst = best_dst

            for frac in (0.002, 0.008, 0.02, 0.05):
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                at = a.copy()
                at[src] -= amt
                at[dst] += amt
                at = np.maximum(at, 0.0)
                st = at.sum()
                if st < 1e-15:
                    continue
                at = at / st
                mt = fast_conv(at).max()
                if mt < best_m - 1e-16:
                    a = at
                    best_m = mt
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 120:
                for _ in range(12):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-10:
                        continue
                    amt = a[i] * rng.uniform(0.002, 0.02)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    mn = fast_conv(a).max()
                    if mn < best_m:
                        best_m = mn
                        best_a = a.copy()
                        stall = 0
                        break

    if verbose:
        el = time.monotonic() - t0
        print("    [Coord] (%.0fs) score=%.6f" % (el, 2.0 * n * best_m))
    return best_a, best_m

def propose_candidate(budget_s=1800.0):

    _G["budget"] = min(float(budget_s) - 40.0, 1760.0)
    _G["best_seq"] = None

    solver = get_solver()
    print("[main] time budget: %.0fs" % _G["budget"])

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)

            p_norm = p / p.sum()
            ps = evaluate_safe(p_norm)
            if np.isfinite(ps):
                _G["best_seq"] = p_norm.copy()
                _G["best_score"] = ps
                print("[main] parent n=%d score=%.6f" % (len(p), ps))
    except Exception:
        pass

    if _G["best_seq"] is None:
        n_base = 2400
        _G["best_seq"] = make_init(n_base, "raised_cosine")
        s = evaluate_safe(_G["best_seq"])
        _G["best_score"] = np.inf if not np.isfinite(s) else s
        print("[main] default init n=%d score=%.6f" % (n_base, _G["best_score"]))

    n0 = len(_G["best_seq"])

    # ---- Phase 1: multi-start basin scouting (short NMD on varied shapes/lengths)
    rem = remaining_time()
    t_p1 = min(rem * 0.10, 110.0)
    if t_p1 > 20:
        print("\n[main] Phase 1: multi-start scouting (%.0fs)" % t_p1)
        lens = [n0, max(1200, n0 - 400), min(4000, n0 + 500)]
        shapes = ["gaussian", "sine", "dolph", "flat_top", "raised_cosine"]

        for length in lens:

                continue

                    break
            if attempts >= 12:
                break

    # ---- Phase 2: Nesterov MD on best (workhorse)

    n_b = len(_G["best_seq"])
    t_p2 = min(rem * 0.22, 330.0)
    print("\n[main] Phase 2: NMD n=%d (%.0fs)" % (n_b, t_p2))

        o, _ = nesterov_md(norm_sum1(_G["best_seq"]), t_p2, verbose=True, seed=42, c0=5.0)

    # ---- Phase 3: LP-FW reduced (the key accelerator)
    rem = remaining_time()
    n_b = len(_G["best_seq"])
    t_p3 = min(rem * 0.34, 520.0)
    print("\n[main] Phase 3: LP-FW reduced n=%d (%.0fs)" % (n_b, t_p3))
    if t_p3 > 60:
        o, _ = lp_fw(norm_sum1(_G["best_seq"]), t_p3, solver, verbose=True,
                     reduced_k=150, warm_iters=8, full_after_frac=0.85)
        try_update(o, "lpfd_n%d" % n_b)

    # ---- Phase 4: NMD polish
    rem = remaining_time()
    n_b = len(_G["best_seq"])
    t_p4 = min(rem * 0.12, 150.0)
    print("\n[main] Phase 4: NMD polish n=%d (%.0fs)" % (n_b, t_p4))
    if t_p4 > 20:
        o, _ = nesterov_md(norm_sum1(_G["best_seq"]), t_p4, verbose=False, seed=99, c0=6.0)
        try_update(o, "nmdp_n%d" % n_b)

    # ---- Phase 5: full-constraint LP-FW
    rem = remaining_time()
    n_b = len(_G["best_seq"])
    t_p5 = min(rem * 0.24, 360.0)
    print("\n[main] Phase 5: LP-FW full n=%d (%.0fs)" % (n_b, t_p5))
    if t_p5 > 60:
        o, _ = lp_fw(norm_sum1(_G["best_seq"]), t_p5, solver, verbose=True,
                     reduced_k=200, warm_iters=1000, full_after_frac=0.0)
        try_update(o, "lpff_n%d" % n_b)

    # ---- Phase 6: coordinate polish
    rem = remaining_time()
    n_b = len(_G["best_seq"])
    t_p6 = min(rem * 0.10, 110.0)
    print("\n[main] Phase 6: coord n=%d (%.0fs)" % (n_b, t_p6))
    if t_p6 > 15:
        o, _ = coord_descend(norm_sum1(_G["best_seq"]), t_p6, verbose=True, seed=42)
        try_update(o, "coord_n%d" % n_b)

    # ---- Phase 7: neighbouring lengths ----
    rem = remaining_time()
    n_b = len(_G["best_seq"])
    if rem > 150:
        print("\n[main] Phase 7: neighbouring lengths (%.0fs)" % rem)
        for delta in (200, -200, 400, -400, 600):
            n_try = n_b + delta
            if n_try < 800 or n_try > 4500:
                continue
            rem = remaining_time()
            if rem < 90:
                break
            a_long = interpolate_sequence(_G["best_seq"], n_try)
            sub = min(rem * 0.4, 130.0)
            if sub < 30:
                break
            o, _ = nesterov_md(a_long, sub, verbose=False, seed=55 + n_try % 100)
            try_update(o, "nmd_n%d" % n_try)
            rem = remaining_time()
            if rem > 70 and len(_G["best_seq"]) == n_try:
                o2, _ = lp_fw(norm_sum1(_G["best_seq"]), min(rem * 0.4, 90.0), solver,
                              verbose=False, reduced_k=150, full_after_frac=0.7)
                try_update(o2, "lp_n%d" % n_try)

    # ---- Phase 8: final full LP polish ----
    rem = remaining_time()
    n_b = len(_G["best_seq"])
    print("\n[main] Phase 8: final LP n=%d (%.0fs)" % (n_b, rem))
    if rem > 40:
        t8 = min(rem * 0.85, 220.0)
        if t8 > 25:
            o, _ = lp_fw(norm_sum1(_G["best_seq"]), t8, solver, verbose=True,
                         reduced_k=200, warm_iters=1000, full_after_frac=0.0)
            try_update(o, "lpfin_n%d" % n_b)

    # ---- final validation ----
    if _G["best_seq"] is not None:
        fs = evaluate_safe(_G["best_seq"])
        if np.isfinite(fs):
            _G["best_score"] = fs
            _G["best_seq"] = norm_sum1(_G["best_seq"])
        print("\n[main] FINAL score=%.6f n=%d time=%.0fs" %
              (_G["best_score"], len(_G["best_seq"]), time.monotonic() - _G["t0"]))
        return [float(x) for x in _G["best_seq"]]

    print("\n[main] ERROR: no valid sequence")
    return [1.0]
```
