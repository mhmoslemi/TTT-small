def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_scaled_score(a):
    """Compute 2n*max_conv/(sum)^2 directly."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * len(a) * ac.max() / (s * s)


def logsumexp_obj_grad(z, n, tau):
    """Compute logsumexp-smoothed objective and gradient w.r.t. z (log-domain)."""
    z_s = z - z.max()
    a = np.exp(z_s)
    s = a.sum()
    a = a / s

    ac = fast_autocorr(a)

    tau_ac = tau * ac
    max_v = tau_ac.max()
    exp_vals = np.exp(tau_ac - max_v)
    lse = max_v + np.log(exp_vals.sum())
    obj = (2.0 * n / tau) * lse

    w = exp_vals / exp_vals.sum()

    grad_a = np.zeros(n)
    for k in range(len(ac)):
        wk = w[k]
        if wk < 1e-14:
            continue
        k_int = k
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad_a[idx] += 2.0 * wk * a[j_idx]

    grad_a *= (4.0 * n / tau)
    inner = np.dot(grad_a, a)
    grad_z = a * (grad_a - inner)
    return obj, grad_z


def do_lse_step(z, n, tau, max_iter):
    """Run L-BFGS-B for logsumexp objective. Returns optimized z."""
    from scipy.optimize import minimize

    def fun(zv):
        o, g = logsumexp_obj_grad(zv, n, tau)
        return o

    def fun_g(zv):
        o, g = logsumexp_obj_grad(zv, n, tau)
        return o, g

    try:
        res = minimize(fun_g, z, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                'iprint': -1, 'maxls': 40})
        return res.x
    except Exception:
        return z


def logsumexp_optimize(a0, time_limit, tau_start, tau_end, n_phases, verbose):
    """Progressive LogSumExp optimization with increasing tau."""

            break
        tau = tau_schedule[phase]

                if i1 != i2 and a[i1] > 1e-8:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i1] * frac
                    a[i1] -= amt
                    a[i2] += amt
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    sc = compute_scaled_score(a)
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i1] += amt
                        a[i2] -= amt
                        a = np.maximum(a, 0.0)
                        s = a.sum()

        if verbose and sweep_count % 3 == 1:
            elapsed = time.monotonic() - t0
            print(f"    ECD sweep={sweep_count} ({elapsed:.1f}s): score={best_score:.8f} stall={stall}")

    return best_a, best_score


def lp_frank_wolfe(a0, time_limit, solver, n_max_constraints, verbose):
    """LP Frank-Wolfe with broad constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()

    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        threshold = M_curr * 0.985
        active_lags = np.where(ac >= threshold)[0]
        if len(active_lags) > n_max_constraints:
            active_lags = np.argsort(ac)[::-1][:n_max_constraints]
        if len(active_lags) == 0:
            active_lags = np.array([int(np.argmax(ac))])

        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((len(active_lags), n))
        for row_idx, k in enumerate(active_lags):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                cols = np.arange(i_start, i_end)
                T[row_idx, cols] = y[k_int - cols]

        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new = y_new / sy
        else:
            y_new = y.copy()
        y = y_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if verbose and t % 15 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={2.0*n*best_m:.8f}")

    return best_a, best_m


def nmd_short(a0, time_limit, initial_temp, final_temp, c0, verbose):
    """Short Nesterov Mirror Descent for polishing."""
    from scipy.optimize import minimize
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    max_stall = 1500

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / 3000.0)
        temp = initial_temp * (1 - frac) + final_temp * frac
        temp = max(temp, 1e-5)

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = x.copy()

        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Top-k subgradient
        k_max = min(60, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** 4.0
        w_sum = w.sum()
        if w_sum > 1e-30:
            w = w / w_sum
        else:
            w = np.ones(k_max) / k_max

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = y.copy()

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(500, max_stall // 2)
            pert = rng.normal(0, 0.004, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            gamma = max(0.5, gamma * 0.5)

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NMD t={t} ({elapsed:.1f}s): J={2.0*n*best_m:.8f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search: LogSumExp-LBFGS -> ECD -> LP-FW -> ECD polish -> Multi-length.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Total budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    n_base = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base} score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        x = np.linspace(0, 1, n_base)
        best_seq = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Init: n={n_base} score={best_score:.8f}")

    # === Phase 1: LogSumExp L-BFGS-B (35%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 500.0)
    print(f"\n=== P1: LSE-LBFGS n={len(best_seq)} budget={t1:.0f}s ===")
    if t1 > 30:
        a_out = logsumexp_optimize(best_seq, t1, 100.0, 4000.0, 5, True)[0]
        a_out = np.maximum(a_out, 0.0)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  P1 NEW BEST: {best_score:.8f}")

    # === Phase 2: ECD (30%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 450.0)
    print(f"\n=== P2: ECD n={len(best_seq)} budget={t2:.0f}s ===")
    if t2 > 30:
        # Run ECD with multiple seeds
        t2_per = t2 / 3
        for seed in [42, 123, 789]:
            if time.monotonic() - t_global > time_budget:
                break
            a_out = exact_coord_descent(best_seq, t2_per, 120, True, seed)[0]
            a_out = np.maximum(a_out, 0.0)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  P2(#{seed}) NEW BEST: {best_score:.8f}")

    # === Phase 3: LP-FW (20%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== P3: LP-FW n={len(best_seq)} budget={t3:.0f}s ===")
    if t3 > 30:
        a_out = lp_frank_wolfe(best_seq, t3, solver, 400, True)[0]
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  P3 NEW BEST: {best_score:.8f}")

    # === Phase 4: NMD short polish (10%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 120.0)
    print(f"\n=== P4: NMD polish n={len(best_seq)} budget={t4:.0f}s ===")
    if t4 > 20:
        a_out = nmd_short(best_seq, t4, 0.005, 0.001, 2.0, True)[0]
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  P4 NEW BEST: {best_score:.8f}")

    # === Phase 5: ECD final polish (5%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.05, 60.0)
    print(f"\n=== P5: ECD final n={len(best_seq)} budget={t5:.0f}s ===")
    if t5 > 15:
        a_out = exact_coord_descent(best_seq, t5, 80, False, 555)[0]
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  P5 NEW BEST: {best_score:.8f}")

    # === Phase 6: Multi-length (10%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_alt = len(best_seq) + 300
        print(f"\n=== P6: Alt length n={n_alt} budget={remaining:.0f}s ===")
        a_alt = interpolate_sequence(best_seq, n_alt)
        a_alt = norm_sum1(a_alt)

        t6a = min(remaining * 0.4, 150.0)
        if t6a > 20:
            a_out = logsumexp_optimize(a_alt, t6a, 50.0, 3000.0, 3, True)[0]
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  P6-LSE NEW BEST: {best_score:.8f}")

        remaining = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining * 0.4, 120.0)
        if t6b > 15 and len(best_seq) <= 3000:
            a_out = exact_coord_descent(best_seq, t6b, 80, False, 314)[0]
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  P6-ECD NEW BEST: {best_score:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)} time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
