def normalize_to(a, n):

def subgradient_at_k(a, k, n):
    grad = np.zeros(n)
    lo1 = max(0, k - n + 1)
    hi1 = min(n, k + 1)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        grad[lo1:hi1] += a[k - idx]
    lo2 = max(0, k)
    hi2 = min(n, k + n)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[lo2:hi2] += a[idx - k]
    return grad


def lp_best_response(a_fixed, M, active_set, solver, extra_constraints=None):
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)

    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    if extra_constraints is not None:
        constraints.append(extra_constraints)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def smooth_max_and_grad(a, tau):
    """Compute log-sum-exp approximation to max(conv(a,a)) and its gradient."""
    n = len(a)
    conv = fast_autocorr(a)
    # Log-sum-exp for numerical stability
    c_max = conv.max()
    log_sum = c_max / tau + np.log(np.sum(np.exp((conv - c_max) / tau)))
    smooth_val = tau * log_sum
    # Softmax weights
    w = np.exp((conv - c_max) / tau)
    w /= w.sum()
    # Gradient: grad_i = 2 * sum_{m=0}^{n-1} w_{i+m} * a_m
    # Use FFT cross-correlation
    N_fft = 1 << (2 * (2 * n - 1) - 1).bit_length()
    W = rfft(w, N_fft)
    A = rfft(a, N_fft)
    # cross-correlation: sum_m w[i+m]*a[m] = IFFT(FFT(w) * conj(FFT(a)))
    corr = irfft(W * np.conj(A), N_fft)[:2 * n - 1]
    grad = 2.0 * corr[:n]
    return smooth_val, grad, w, conv

def smoothed_gradient_optimize(a0, time_limit, verbose=False, seed=0):
    """
    Adam-based optimization of the log-sum-exp smoothed max.
    Uses tau annealing: start smooth, end sharp.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Adam hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5 / np.sqrt(n)

    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    no_improve = 0
    max_no_improve = 500

    # Tau annealing: start large (smooth), decrease to small (sharp)
    tau_start = 2.0
    tau_end = 0.05
    tau = tau_start

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        # Anneal tau
        progress = min(1.0, t / 3000.0)
        tau = tau_start * (tau_end / tau_start) ** progress

        # Compute smooth objective and gradient
        smooth_val, grad, w, conv = smooth_max_and_grad(a, tau)

        # Project gradient onto the constraint hyperplane (sum = norm)
        # Remove mean component to maintain sum constraint
        grad_centered = grad - np.mean(grad)

        # Adam update
        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad_centered
        v = beta2 * v + (1 - beta2) * grad_centered ** 2
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)
        update = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - update
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        # Evaluate true objective
        score_new = fast_autocorr(a_new).max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
            lr = min(lr * 1.01, 2.0 / np.sqrt(n))
        else:
            no_improve += 1
            lr *= 0.995
            if lr < 1e-6:
                break

        a = a_new

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothGrad t={t} ({elapsed:.0f}s): true_score={best_score:.10f}, tau={tau:.4f}, lr={lr:.6f}")

    return best_a, best_score

                    continue

                # Compute new convolution value at affected positions
                delta = x - aj
                # Affected: conv[lo:hi] changes by 2*delta*a[lo-j:hi-j]
                # Plus correction at 2j
                new_conv_local = conv[lo:hi].copy()
                new_conv_local += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv_local[2 * j - lo] += delta * delta

                # New max is max of (unaffected conv, affected conv) * scale^2
                unaffected_max = np.delete(conv, np.arange(lo, hi)).max() if hi - lo < size else conv.max()
                affected_max = new_conv_local.max()
                new_max_raw = max(unaffected_max, affected_max)

                scale = norm / new_sum
                new_max = new_max_raw * scale * scale

                if new_max < best_obj - 1e-14:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        conv = fast_autocorr(a)
                        improved = True
                    else:
                        a = best_a.copy()
                        conv = fast_autocorr(a)

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_a_score if False else best_a, best_score


def dual_guided_polish(a0, time_limit, solver, verbose=False):
    """
    Use LP dual variables to guide subgradient updates.
    KKT-based: at optimum, weighted subgradient should be constant on support.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    patience = 0
    max_patience = 300
    step = 0.3 / n ** 0.3

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Get active set
        tol = 0.03
        active_set = np.where(conv >= M_curr * (1.0 - tol))[0]
        if len(active_set) < 3:
            active_set = np.argsort(conv)[-max(3, n // 100):]

        # Solve LP to get duals
        b, duals = lp_best_response(a, M_curr, active_set, solver)
        if b is None:
            patience += 1
            continue

        # If we have duals, use them for weighting
        if duals is not None and len(duals) == len(active_set):
            weights = np.maximum(duals, 0.0)
            if weights.sum() < 1e-15:
                weights = np.ones(len(weights))
            weights = weights / weights.sum()
        else:
            weights = np.exp(-5.0 * (M_curr - conv[active_set]) / max(M_curr * 0.02, 1e-10))
            weights = weights / weights.sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, active_set):
            grad += w * subgradient_at_k(a, int(k), n)

        # On the support, we want grad to be constant
        support_thresh = 0.003 * a.max()
        support = np.where(a > support_thresh)[0]
        if len(support) < 2:
            break

        grad_mean = np.mean(grad[support])
        direction = grad - grad_mean
        direction[support] -= 0.05 * np.std(grad[support])

        a_trial = a - step * direction
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.008, 1.0)
        else:
            patience += 1
            step *= 0.97
            if step < 1e-10:
                break

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualPolish t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score


def perturb_and_optimize(best_seq, time_limit, solver, rng, seed):
    """Perturb the best sequence and optimize."""
    n = len(best_seq)
    scale = 0.005 + 0.01 * rng.random()
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.3)
    perturbed = best_seq * noise
    s = perturbed.sum()
    if s < 1e-15:
        return best_seq
    perturbed = perturbed / s
    a_norm = normalize_to(perturbed, n)
    a_out, _ = nesterov_lp_optimize(a_norm, time_limit, solver, verbose=False, seed=seed)
    return a_out


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence."""
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False


def multi_subgradient(a0, time_limit, verbose=False, seed=42):
    """Multi-peak subgradient with adaptive step size."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.8 / n ** 0.35
    patience = 0
    max_patience = 500
    rng = np.random.default_rng(seed)

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        n_peaks = max(4, min(15, n // 60))
        top_indices = np.argsort(conv)[-n_peaks:]

        if t % 10 == 0:
            threshold = M_curr * (0.80 + 0.15 * rng.random())
            candidates = np.where(conv >= threshold)[0]
            if len(candidates) > 0:
                k_rand = candidates[rng.integers(len(candidates))]
                top_indices = np.append(top_indices.astype(int), k_rand)

        weights = np.exp(-6.0 * (M_curr - conv[top_indices.astype(int)]) / max(M_curr * 0.03, 1e-10))
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            grad += w * subgradient_at_k(a, int(k), n)

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        a_trial = a - step * grad * n ** 0.35
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.006, 2.0)
        else:
            patience += 1
            step *= 0.988
            if step < 1e-10:
                break

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiSubgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def propose_candidate():
    """
    Main search: combines LP best-response, smoothed gradient descent,
    dual-guided polish, coordinate descent, and multi-restart.
    """
    rng = np.random.default_rng(2025)
    t_global = time.monotonic()
    time_budget = 1750.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)

    # Brief length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Length Exploration (budget={min(remaining*0.04, 60):.0f}s) ===")
        t_le = min(remaining * 0.04, 60.0)
        for n_try in [n_base - 20, n_base + 20, n_base + 50]:
            if time.monotonic() - t_global > time_budget - t_le:
                break
            # Resample parent to new length
            t_new = np.linspace(0, 1, n_try)
            t_old = np.linspace(0, 1, len(best_seq))
            a_try = np.interp(t_new, t_old, best_seq)
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
                sc_try = evaluate_safe(a_try)
                if np.isfinite(sc_try) and sc_try < best_score:
                    best_score = sc_try
                    best_seq = a_try.copy()
                    print(f"  Length {n_try}: score={sc_try:.8f} (improved!)")
                else:
                    print(f"  Length {n_try}: score={sc_try:.8f}")

    n_base = len(best_seq)
    print(f"Starting main optimization: n={n_base}, score={best_score:.8f}")

    # === Phase 1: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
        a_norm = normalize_to(best_seq, n_base)
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, seed=0)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp1_n{n_base}", t_global)

    # === Phase 2: Smoothed gradient descent (new method) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 40:
        print(f"\n=== Phase 2: Smoothed Gradient n={n_cur}, budget={t2:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = smoothed_gradient_optimize(a_norm, t2, verbose=True, seed=1)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"smooth_n{n_cur}", t_global)

    # === Phase 3: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.22, 300.0)
    if t3 > 40:
        print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur}, budget={t3:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, seed=1234)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp2_n{n_cur}", t_global)

    # === Phase 4: Dual-guided polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 30:
        print(f"\n=== Phase 4: Dual Polish n={n_cur}, budget={t4:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = dual_guided_polish(a_norm, t4, solver, verbose=True)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"dual_n{n_cur}", t_global)

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20:
        print(f"\n=== Phase 5: CD n={n_cur}, budget={t5:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent_fast(a_norm, t5, verbose=True, seed=5)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd_n{n_cur}", t_global)

    # === Phase 6: Multi-subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 20:
        print(f"\n=== Phase 6: MultiSubgrad n={n_cur}, budget={t6:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = multi_subgradient(a_norm, t6, verbose=True, seed=7)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"subg_n{n_cur}", t_global)

    # === Phase 7: Perturbation restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_restarts = 3
    t7_total = min(remaining * 0.15, 150.0)
    t7_each = t7_total / max(1, n_restarts)
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t7 = min(t7_each, remaining - 15)
        if t7 < 15:
            break
        print(f"\n=== Phase 7r{r}: Perturbation n={n_cur}, budget={t7:.0f}s ===")
        a_out = perturb_and_optimize(best_seq, t7, solver, rng, seed=500 + r * 100)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"pert{r}_n{n_cur}", t_global)

    # === Phase 8: Final smooth gradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        t8 = min(remaining * 0.3, 80.0)
        print(f"\n=== Phase 8: Final Smooth n={n_cur}, budget={t8:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = smoothed_gradient_optimize(a_norm, t8, verbose=False, seed=99)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"smoothFinal_n{n_cur}", t_global)

    # === Phase 9: Final LP micro-optimize ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        t9 = min(remaining * 0.3, 80.0)
        print(f"\n=== Phase 9: Final LP n={n_cur}, budget={t9:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t9, solver, verbose=False, seed=777)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lpFinal_n{n_cur}", t_global)

    # === Phase 10: Final CD micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 20:
        t10 = min(remaining * 0.4, 50.0)
        print(f"\n=== Phase 10: Final CD n={n_cur}, budget={t10:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent_fast(a_norm, t10, verbose=False, seed=42)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cdFinal_n{n_cur}", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
