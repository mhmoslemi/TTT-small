def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_uniform(n):
    return np.ones(n) / n

def make_triangular(n):
    x = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(x - (n-1)/2.0) / ((n-1)/2.0)
    return a / a.sum()

def make_exponential(n, rate=1.0):
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * x / n)
    return a / a.sum()

def make_bimodal(n, centers=(0.2, 0.8), width=0.05):
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.01
    for c in centers:
        a += np.exp(-((x - c) / width) ** 2)
    return a / a.sum()

def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 8):
        coeff = rng.normal(0, 0.15)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()

def make_concentrated(n, center_frac=0.5, width=0.1):
    """Most mass concentrated at center."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center_frac) / width) ** 2)
    return a / a.sum()

def make_edges(n, width=0.15):
    """Mass concentrated at edges."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x) / width) ** 2) + np.exp(-((x - 1) / width) ** 2)
    return a / a.sum()

def lp_frank_wolfe(a0, time_limit, top_K=200, verbose=False):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Find a solver
    solver = None
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS
    
    y = a.copy()
    x_nest = a.copy()
    iter_count = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Update T matrix (lazy)
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Build constraint matrix for top-K constraints
        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y
        
        margins = M_curr - T_full @ y
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]
        
        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x_nest = x_new
        iter_count += 1
        
        for candidate in [y, x_nest]:
            mc = fast_autocorr(candidate).max()
            if mc < best_m:
                best_m = mc
                best_a = candidate.copy()
        
        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m
