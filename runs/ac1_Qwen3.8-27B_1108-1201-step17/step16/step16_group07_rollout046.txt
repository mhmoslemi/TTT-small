    # and k-i in S: max(0, k-n+1) <= k-i <= min(n-1, k) iff max(0, k-n+1) - (k-n+1) <= i - (k-n+1) ... 
    # Let me just use: k-i in S iff 0 <= k-i < n and 0 <= k-(k-i) < n = 0 <= i < n
    # So k-i in S iff 0 <= k-i < n (and 0<=i<n is already required for us to care)
    # i.e., k-n < i <= k (combined with 0<=i<n)
    
    # So for each i in [0, n-1]:
    #   d/d a[i] (a*a)[k] = a[k-i] * 1{max(0,k-n+1) <= i <= min(n-1,k)} 
    #                      + a[i] * 1{0 <= k-i < n}
    
    # The first condition: max(0,k-n+1) <= i <= min(n-1, k) -- this is the "direct" term
    # The second condition: 0 <= k-i < n, i.e., k-n < i <= k -- this is the "cross" term
    
    # For the "direct" term: i ranges from max(0,k-n+1) to min(n-1,k)
    #   grad[i] += a[k-i]
    # For the "cross" term: i ranges from max(0, k-n+1) to min(n-1, k)
    #   Wait, k-n < i <= k and 0 <= i < n gives max(0, k-n+1) <= i <= min(n-1, k)
    #   Hmm, k-n < i means i >= k-n+1 (for integer i), and i <= k, and 0 <= i < n.
    #   So the range is the same! max(0, k-n+1) to min(n-1, k).
    #   
    # Wait, that means for the same range of i, we add both a[k-i] and a[i]?
    # Let me verify with a small example.
    # n=3, a=[a0,a1,a2], k=1: (a*a)[1] = a[0]*a[1] + a[1]*a[0] = 2*a[0]*a[1]
    # d/d a[0] = 2*a[1], d/d a[1] = 2*a[0], d/d a[2] = 0
    # Using my formula:
    # S = {j: max(0,1-3+1) <= j <= min(2,1)} = {j: 0 <= j <= 1} = {0, 1}
    # For i=0: a[k-i]*1{i in S} + a[i]*1{k-i in S} = a[1]*1 + a[0]*1{0<=1<3} = a[1] + a[0]
    # But the correct answer is 2*a[1]. So something is wrong.
    
    # Ah I see the issue. When i=0 and k=1:
    # "direct" term: a[k-i] = a[1], and i=0 is in S={0,1}. So we add a[1]. ✓
    # "cross" term: a[i] = a[0], and k-i = 1. Is k-i=1 in S? S={0,1}, yes. So we add a[0]. 
    # But the correct derivative is 2*a[1], not a[1]+a[0].
    
    # I think I'm making an error. Let me redo this.
    # (a*a)[k] = sum_{j in S} a[j] * a[k-j]
    # d/d a[i] = sum_{j in S} [delta(j,i) * a[k-j] + a[j] * delta(k-j, i)]
    # First part: sum_{j in S} delta(j,i) * a[k-j] = a[k-i] if i in S, else 0
    # Second part: sum_{j in S} a[j] * delta(k-j, i) = a[k-i] if k-i in S, else 0
    #                                                         (because delta(k-j,i) means j = k-i, 
    #                                                          and we need k-i in S)
    # So d/d a[i] = a[k-i] * 1{i in S} + a[k-i] * 1{k-i in S}
    #              = a[k-i] * (1{i in S} + 1{k-i in S})
    
    # Ah! Both terms give a[k-i], not a[k-i] + a[i]!
    # Let me verify: n=3, k=1, i=0:
    # a[k-i] = a[1], i=0 in S={0,1}? Yes. k-i=1 in S={0,1}? Yes.
    # d/d a[0] = a[1] * (1+1) = 2*a[1]. ✓
    
    # For i=1: a[k-i] = a[0], i=1 in S? Yes. k-i=0 in S? Yes.
    # d/d a[1] = a[0] * (1+1) = 2*a[0]. ✓
    
    # For i=2: i=2 not in S. k-i = -1 not in S.
    # d/d a[2] = 0. ✓
    
    # Great! So the correct formula is:
    # d/d a[i] (a*a)[k] = a[k-i] * (1{i in S} + 1{k-i in S})
    # where S = {j: max(0,k-n+1) <= j <= min(n-1, k)}
    
    # Note that i in S means max(0,k-n+1) <= i <= min(n-1, k)
    # k-i in S means max(0,k-n+1) <= k-i <= min(n-1, k)
    #   iff max(0,k-n+1) - k <= -i <= min(n-1,k) - k
    #   iff k - min(n-1,k) <= i <= k - max(0,k-n+1)
    #   if k < n: k - (n-1) <= i <= k - (k-n+1) = n-1, i.e., k-n+1 <= i <= n-1
    #     combined with 0<=i<n: max(0, k-n+1) <= i <= n-1
    #   if k >= n: k - (n-1) <= i <= k - 0 = k, i.e., k-n+1 <= i <= k
    #     combined with 0<=i<n: k-n+1 <= i <= n-1
    #   In both cases: max(0, k-n+1) <= i <= min(n-1, k) ... wait that's the same as S!
    
    # Hmm, let me recheck. k-i in S means:
    # max(0, k-n+1) <= k-i <= min(n-1, k)
    # => max(0, k-n+1) - k <= -i <= min(n-1, k) - k
    # => k - min(n-1, k) <= i <= k - max(0, k-n+1)
    
    # Case 1: k < n
    #   min(n-1,k) = k, max(0,k-n+1) = 0 (since k-n+1 <= k < n means k-n+1 could be negative)
    #   Actually max(0, k-n+1): if k < n-1, then k-n+1 < 0, so max = 0
    #   If k = n-1, then k-n+1 = 0, max = 0
    #   So for k < n: k - k <= i <= k - 0 => 0 <= i <= k
    #   Combined with 0<=i<n: 0 <= i <= min(n-1, k) = k (since k<n)
    #   So: 0 <= i <= k
    
    # Case 2: k >= n
    #   min(n-1,k) = n-1, max(0,k-n+1) = k-n+1 (since k>=n means k-n+1>=1)
    #   k - (n-1) <= i <= k - (k-n+1) = n-1
    #   Combined with 0<=i<n: k-n+1 <= i <= n-1
    
    # So:
    # If k < n: i in S is [0, k], and k-i in S is [0, k]. They're the same!
    #   d/d a[i] = a[k-i] * 2 for i in [0, k], and 0 otherwise.
    # If k >= n: i in S is [k-n+1, n-1], and k-i in S is [k-n+1, n-1]. They're the same!
    #   d/d a[i] = a[k-i] * 2 for i in [k-n+1, n-1], and 0 otherwise.
    
    # Wait, that's not right either. Let me recheck.
    # For k < n: i in S means 0 <= i <= min(n-1, k) = k.
    #   k-i in S means 0 <= k-i <= min(n-1, k) = k, i.e., 0 <= k-i <= k, i.e., 0 <= i <= k.
    #   Same range! So d/d a[i] = 2*a[k-i] for 0 <= i <= k, 0 otherwise.
    
    # For k >= n (specifically n <= k <= 2n-2): i in S means k-n+1 <= i <= n-1.
    #   k-i in S means k-n+1 <= k-i <= n-1, i.e., k-(n-1) <= i <= k-(k-n+1) = n-1.
    #   So k-n+1 <= i <= n-1. Same range!
    #   d/d a[i] = 2*a[k-i] for k-n+1 <= i <= n-1, 0 otherwise.
    
    # So in all cases, the subgradient is:
    # grad[i] = 2 * a[k-i] if i is in the valid range, 0 otherwise
    # where the valid range is [max(0, k-n+1), min(n-1, k)]
    
    # Let me verify with n=3, k=1:
    # Range: [max(0,1-3+1), min(2,1)] = [0, 1]
    # grad[0] = 2*a[1], grad[1] = 2*a[0], grad[2] = 0. ✓
    
    # n=3, k=2: (a*a)[2] = a[0]*a[2] + a[1]*a[1] + a[2]*a[0]
    # Range: [max(0,2-3+1), min(2,2)] = [0, 2]
    # grad[0] = 2*a[2], grad[1] = 2*a[1], grad[2] = 2*a[0]
    # Check: d/d a[0] of (a0*a2 + a1*a1 + a2*a0) = a2 + a2 = 2*a2 ✓
    #         d/d a[1] of ... = 2*a1 ✓
    #         d/d a[2] of ... = a0 + a0 = 2*a0 ✓
    
    # So the formula is simply: grad[i] = 2*a[k-i] for i in [max(0,k-n+1), min(n-1, k)]
    
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)  # exclusive upper bound
    if lo < hi:
        grad[lo:hi] = 2.0 * a[k - np.arange(lo, hi)]
    return grad

def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    
    if size == 0:
        return None, None
    
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response with adaptive active-set management."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    conv_size = 2 * n - 1
    n_iters = 0
    last_violation_check = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set
        if active_set is None or t % 30 == 0:
            # Tighten over time
            progress = min(1.0, t / 800.0)
            tol = 0.12 * (1.0 - progress) + 0.002 * progress
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            # Always include top peaks
            n_top = max(10, n // 50)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            # Cap for efficiency (but be generous)
            max_active = int(2.0 * n)
            if len(active_set) > max_active:
                conv_at_active = conv_y[active_set]
                keep = np.argsort(conv_at_active)[-max_active:]
                active_set = active_set[keep]

        # Periodic violation check
        if t - last_violation_check > 20:
            last_violation_check = t
            # Check if any constraint is significantly violated
            conv_y_check = conv_y  # already computed
            violated = np.where(conv_y_check > M_curr * 1.001)[0]
            if len(violated) > 0:
                active_set = np.unique(np.concatenate([active_set, violated]))
                if len(active_set) > int(2.5 * n):
                    conv_at_active = conv_y[active_set]
                    keep = np.argsort(conv_at_active)[-int(2.5 * n):]
                    active_set = active_set[keep]

        b, duals = lp_best_response(y, M_curr, active_set, solver)

        if b is None:
            # Try with just the top peaks
            top_n = max(20, n // 30)
            top_k = np.argsort(conv_y)[-top_n:]
            b, duals = lp_best_response(y, M_curr, top_k, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update with adaptive momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            # Adaptive momentum: reduce if oscillating
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.95)
            # Detect oscillation
            if t > 10:
                osc = np.linalg.norm(y_new - y) + np.linalg.norm(y - y_prev)
                if osc > 2.0 * norm / np.sqrt(n):
                    momentum *= 0.5
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average check
        if t % 40 == 0:
            a_damped = 0.6 * best_a + 0.4 * a
            a_damped = np.maximum(a_damped, 0.0)
            s_d = a_damped.sum()
            if s_d > 1e-15:
                a_damped = a_damped * norm / s_d
                score_d = fast_autocorr(a_damped).max()
                if score_d < best_score:
                    best_score = score_d
                    best_a = a_damped.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > 60:
            # Try a perturbation before restarting
            pert = rng.standard_normal(n) * 0.02 * best_a
            a_pert = best_a + pert
            a_pert = np.maximum(a_pert, 0.0)
            s_p = a_pert.sum()
            if s_p > 1e-15:
                a_pert = a_pert * norm / s_p
                score_p = fast_autocorr(a_pert).max()
                if score_p < best_score:
                    best_a = a_pert.copy()
                    best_score = score_p
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def mirror_descent_polish(a0, time_limit, verbose=False, seed=42):
    """Mirror descent (multiplicative weights) for the non-negative simplex."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = 1.0 / np.sqrt(n)
    patience = 0
    max_patience = 300
    rng = np.random.default_rng(seed)

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Find active peaks
        n_peaks = max(5, min(20, n // 50))
        top_indices = np.argsort(conv)[-n_peaks:]

        # Compute weighted subgradient
        weights = np.exp(-10.0 * (M_curr - conv[top_indices]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            grad += w * subgradient_at_k(a, int(k), n)

        # Mirror descent: a_new[i] = a[i] * exp(-eta * grad[i] / max_grad)
        max_grad = np.max(np.abs(grad))
        if max_grad < 1e-15:
            break
        exp_factor = np.exp(-eta * grad / max_grad)
        a_new = a * exp_factor
        a_new = np.maximum(a_new, 1e-10)  # Keep positive for numerical stability
        a_new = a_new * norm / a_new.sum()

        score_new = fast_autocorr(a_new).max()
        if score_new < best_score - 1e-14:
            a = a_new
            best_score = score_new
            best_a = a_new.copy()
            patience = 0
            eta = min(eta * 1.01, 5.0 / np.sqrt(n))
        else:
            patience += 1
            eta *= 0.98
            if eta < 1e-6:
                break

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDescent t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def coordinate_descent(a0, time_limit, verbose=False, seed=0):
    """Exact coordinate descent with efficient autocorrelation updates."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0
    max_no_improve = 5

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        conv = fast_autocorr(a)

        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            lo = max(0, j)
            hi = min(size, j + n)

            # Try several candidate values
            candidates = [0.0, aj * 0.5, aj * 1.5, norm / n]
            # Add perturbation
            candidates.append(aj + (rng.random() - 0.5) * 0.1 * norm / n)
            candidates = [max(0.0, c) for c in candidates]

            best_x = aj
            best_obj = best_score

            for x in candidates:
                if abs(x - aj) < 1e-10:
                    continue

                new_conv = conv.copy()
                delta = x - aj
                if lo < hi:
                    new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta

                new_max = new_conv.max()
                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj) > 1e-10:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        conv = fast_autocorr(a)
                        improved = True
                    else:
                        a = best_a.copy()
                        conv = fast_autocorr(a)

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def full_lp_polish(a0, time_limit, solver, verbose=False):
    """Solve LP with ALL constraints for the best response, then iterate a few times."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    conv_size = 2 * n - 1

    # Try solving with all constraints
    for it in range(5):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Use all constraints (or a large subset if too many)
        if conv_size <= 4000:
            active = np.arange(conv_size)
        else:
            # Use top 2000
            top_k = np.argsort(conv)[-min(2000, conv_size):]
            active = top_k

        b, _ = lp_best_response(a, M_curr, active, solver)
        if b is None:
            break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm / b_sum

        # Convex combination
        for t_blend in [0.5, 0.7, 0.9, 0.3]:
            a_trial = (1 - t_blend) * a + t_blend * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            score_t = fast_autocorr(a_trial).max()
            if score_t < best_score:
                best_score = score_t
                best_a = a_trial.copy()

        a = best_a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    FullLP iter={it} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False
