```python
        # So:

        # From a[j+k]: j ranges from 0 to min(n-1, n-1-k) = n-1-k (if k>0), contribution is a[j+k]
        if k == 0:
            grad += 2.0 * w * a
        else:
            # j from k to n-1: grad[j] += w * a[j-k]
            grad[k:] += w * a[:n-k]
            # j from 0 to n-1-k: grad[j] += w * a[j+k]
            grad[:n-k] += w * a[k:]
    return grad
```

```python
def fast_autocorr(a):
    """Compute autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_grad_weights(a, ac, beta):
    """
    Compute the smoothed gradient and the softmax weights.
    beta = 1/temperature
    Returns: (grad, weights, smoothed_value)
    """
    n = len(a)
    # Softmax weights with numerical stability
    scaled = beta * ac
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    w_sum = exp_scaled.sum()
    if w_sum < 1e-300:
        # Fallback: uniform over top 20
        top_k = min(20, len(ac))
        top_idx = np.argsort(ac)[::-1][:top_k]
        weights = np.zeros_like(ac)
        weights[top_idx] = 1.0 / top_k
    else:
        weights = exp_scaled / w_sum
    
    smoothed_val = np.log(w_sum) / beta  # = max + log(sum exp)/beta
    
    # Compute gradient
    grad = np.zeros(n)
    # Only use lags with significant weight
    threshold = 1e-12
    active = np.where(weights > threshold)[0]
    
    for k in active:
        w = beta * weights[k]  # d(beta * c_k)/da = beta * dc_k/da
        # Actually: smoothed = (1/beta)*log(sum exp(beta*c_k))
        # d/d a_j = sum_k weights[k] * dc_k/da_j
        # So grad_j = sum_k weights[k] * dc_k/da_j
        w_use = weights[k]
        if k == 0:
            grad += 2.0 * w_use * a
        else:
            grad[k:] += w_use * a[:n-k]
            grad[:n-k] += w_use * a[k:]
    
    return grad, weights

def smoothed_mirror_descent(a0, time_limit, beta_start=200.0, beta_end=2000.0,
                            c0=1.0, verbose=True):
    """
    Mirror descent on the smoothed objective with FISTA acceleration.
    Uses multiplicative updates (negative entropy mirror).
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    x = a.copy()
    x_prev = a.copy()
    t_fista = 1.0
    stall_count = 0
    
    n_iter = 0
    while time.monotonic() - t0 < time_limit:
        n_iter += 1
        # Adaptive beta (temperature decreases over time)
        frac = min(1.0, n_iter / 5000.0)
        beta = beta_start + (beta_end - beta_start) * frac
        
        # FISTA lookahead
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista))
        y = x + ((t_fista - 1.0) / t_new) * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute gradient of smoothed objective at y
        ac = fast_autocorr(y)
        grad, weights = compute_grad_weights(y, ac, beta)
        
        # Step size: for mirror descent with Lipschitz L, use eta = 1/(L*t)
        # L ~ 2*n (rough bound), use conservative step
        eta = c0 / (np.sqrt(n_iter) * np.sqrt(n))
        
        # Mirror descent update: x_new = exp(y - eta*grad) / sum(exp(y - eta*grad))
        log_x = np.log(y) - eta * grad
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new
        
        x_prev = x
        x = x_new
        t_fista = t_new
        
        # Check improvement
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Also check the FISTA iterate
        if n_iter > 10:
            x_fista = x + ((t_fista - 1.0) / (t_fista + 1.0)) * (x - x_prev)
            x_fista = np.maximum(x_fista, 0)
            sf = x_fista.sum()
            if sf > 1e-15:
                x_fista = x_fista / sf
                Mf = fast_autocorr(x_fista).max()
                if Mf < best_m:
                    best_m = Mf
                    best_a = x_fista.copy()
                    stall_count = 0
        
        # Restart if stalled
        if stall_count > 2000:
            stall_count = 0
            pert = rng.normal(0, 0.01, n)
            x_r = best_a * (1.0 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            t_fista = 1.0
            c0 = max(0.3, c0 * 0.5)
        
        if verbose and n_iter % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SMD iter={n_iter} ({elapsed:.1f}s): J={J:.8f}, beta={beta:.0f}, stall={stall_count}")
    
    return best_a, best_m

def lp_fw_cutting_planes(a0, time_limit, solver, verbose=True):
    """
    Frank-Wolfe with LP oracle using constraint generation.
    At each iteration:
    1. Solve LP with current constraint set to get direction b
    2. Move towards b
    3. Add violated constraints to the set
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Initial constraint set: top 30 lags
    ac0 = fast_autocorr(a)
    active_lags = set(np.argsort(ac0)[::-1][:30].tolist())
    
    for t in range(1, 200):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current autocorrelation
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Build constraint matrix for active lags
        lags_sorted = sorted(active_lags)
        n_constraints = len(lags_sorted)
        
        # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
        A_mat = np.zeros((n_constraints, n))
        for row, k in enumerate(lags_sorted):
            # A[row, j] = a[k-j] for valid j
            # k-j in [0, n-1] => j in [k-n+1, k] intersect [0, n-1]
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            if j_start < j_end:
                A_mat[row, j_start:j_end] = a[k - np.arange(j_start, j_end)]
        
        # LP: max sum(b) s.t. A_mat @ b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [A_mat @ b_var <= M_curr]
        objective = cp.Maximize(cp.sum(b_var))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum
        
        # Frank-Wolfe step
        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * direction
        a_new = np.maximum(a_new, 0.0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sa
        
        # Check improvement
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Update constraint set: check for violated constraints
        ac_new = fast_autocorr(a)
        # Find lags where ac_new[k] > M_curr (violated the old bound)
        # Actually, we should check all lags not in active set
        all_lags = range(2 * n - 1)
        violated = []
        for k in all_lags:
            if k not in active_lags:
                # Check if this constraint would be violated by the LP solution
                # conv(a, b_val)[k] <= M_curr should hold
                # But we're checking if the NEW point a_new violates
                pass
        
        # Simpler: just add the top lags of the new autocorrelation
        top_lags = np.argsort(ac_new)[::-1][:50]
        for k in top_lags:
            active_lags.add(int(k))
        
        # Limit constraint set size
        if len(active_lags) > 200:
            # Keep only the most important ones
            lag_importance = {k: ac_new[k] for k in active_lags}

def gradient_local_search(a0, time_limit, beta=500.0, verbose=True, rng_seed=99):
    """
    Local search guided by the gradient of the smoothed objective.
    At each step:
    1. Compute gradient
    2. Find best (src, dst) pair: src has high gradient, dst has low
    3. Transfer mass with bisection line search
    """
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        # Compute gradient
        ac = fast_autocorr(a)
        grad, _ = compute_grad_weights(a, ac, beta)
        
        # Find top "increase" and "decrease" candidates
        # We want to decrease coordinates with high grad and increase those with low grad
        # But we also need to maintain the simplex constraint (sum = 1)
        
        # Strategy: pick a random src from top-50% gradient, random dst from bottom-50%
        half = n // 2
        top_half = np.argsort(grad)[half:]  # high gradient (should decrease)
        bot_half = np.argsort(grad)[:half]  # low gradient (should increase)
        
        improved = False
        
        for _ in range(min(30, n // 10)):
            if time.monotonic() - t0 > time_limit:
                break
            
            src = int(rng.choice(top_half))
            dst = int(rng.choice(bot_half))
            
            if src == dst or a[src] < 1e-14 or a[dst] < 1e-14:
                continue
            
            # Bisection: find best fraction to transfer
            lo, hi = 0.0, min(1.0, a[src] * 0.9)
            best_frac = 0.0
            best_trial_m = best_m
            
            for _ in range(8):  # bisection iterations
                mid = (lo + hi) / 2.0
                amt = a[src] * mid
                if amt < 1e-18:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    break
                a_trial = a_trial / st
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_trial_m:
                    best_trial_m = m_trial
                    best_frac = mid
                    lo = mid
                else:
                    hi = mid
            
            if best_frac > 0 and best_trial_m < best_m:
                amt = a[src] * best_frac
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_trial_m
                best_a = a.copy()
                improved = True
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape local minimum
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
                else:
                    # Revert
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
        
        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    GLS ({elapsed:.1f}s): J={J:.8f}, stall={stall}")
    
    return best_a, best_m

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
```

def get_lp_solver():

            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)

def compute_grad_weights(a, ac, beta):
    n = len(a)
    scaled = beta * ac
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    w_sum = exp_scaled.sum()
    if w_sum < 1e-300:

        top_idx = np.argsort(ac)[::-1][:top_k]

        weights[top_idx] = 1.0 / top_k
    else:
        weights = exp_scaled / w_sum
    grad = np.zeros(n)
    threshold = max(1e-14, 1.0 / (len(ac) * 1000.0))
    active = np.where(weights > threshold)[0]

        w = weights[k]
        j_start = max(0, int(k) - n + 1)
        j_end = min(n, int(k) + 1)
        if j_start >= j_end:
            continue
        idx = int(k) - np.arange(j_start, j_end)
        grad[j_start:j_end] += 2.0 * w * a[idx]
    return grad, weights

    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    x = a.copy()
    x_prev = a.copy()
    t_fista = 1.0
    stall_count = 0
    n_iter = 0
    ergodic_sum = np.zeros(n)
    ergodic_count = 0

    while time.monotonic() - t0 < time_limit:
        n_iter += 1
        frac = min(1.0, n_iter / 8000.0)
        beta = beta_start + (beta_end - beta_start) * frac
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista))
        momentum = (t_fista - 1.0) / t_new
        y = x + momentum * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        ac = fast_autocorr(y)
        grad, weights = compute_grad_weights(y, ac, beta)
        eta = c0 / (np.sqrt(n_iter) * np.sqrt(n))
        log_x = np.log(y) - eta * grad
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new
        x_prev = x
        x = x_new
        t_fista = t_new
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        ergodic_sum += x
        ergodic_count += 1
        if ergodic_count % 200 == 0:
            ergodic = ergodic_sum / ergodic_count
            ergodic = np.maximum(ergodic, 0.0)
            se = ergodic.sum()
            if se > 1e-15:
                ergodic = ergodic / se
                Me = fast_autocorr(ergodic).max()
                if Me < best_m:
                    best_m = Me
                    best_a = ergodic.copy()
                    stall_count = 0
        if stall_count > 3000:
            stall_count = 0
            pert = rng.normal(0, 0.008, n)
            x_r = best_a * (1.0 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            t_fista = 1.0
            c0 = max(0.2, c0 * 0.6)
        if verbose and n_iter % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SMD iter={n_iter} ({elapsed:.1f}s): J={J:.8f}, beta={beta:.0f}, stall={stall_count}")
    return best_a, best_m
