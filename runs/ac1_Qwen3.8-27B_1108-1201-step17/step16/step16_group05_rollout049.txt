def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = y
    return T

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)

        # Compute gradient of smoothed max
        ac = fast_autocorr(y)
        num_lags = len(ac)
        k_eff = min(top_k, num_lags)
        top_idx = np.argsort(ac)[::-1][:k_eff]
        top_vals = ac[top_idx]
        shifted = top_vals - top_vals[0]
        exp_vals = np.exp(beta * shifted)
        weights = exp_vals / exp_vals.sum()

        grad = np.zeros(n)
        for wi, k in zip(weights, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        # Step
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            gn = 1e-15
        eta = gamma / (np.sqrt(t) + 1.0)

        x_new = project_simplex(y - eta * grad)
        x_prev = x
        x = x_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.5)
            noise = rng.normal(0, 0.003, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      APG t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def pair_swap_cd(a0, time_limit, verbose=True, seed=99):
    """Coordinate descent that swaps mass between mirror pairs to reduce center lag."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 30:
        ac = fast_autocorr(a)
        k_peak = int(np.argmax(ac))
        M_curr = ac[k_peak]

        # For the peak lag, find the pair with max product
        i_start = max(0, k_peak - n + 1)
        i_end = min(n, k_peak + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_peak - idx
        products = a[idx] * a[j_idx]

        if len(products) == 0:
            stall += 1
            break

        max_pi = np.argmax(products)
        src_i = idx[max_pi]
        src_j = j_idx[max_pi]

        # Try moving mass from the larger of (src_i, src_j) to a random position
        if a[src_i] >= a[src_j]:
            donor = src_i
        else:
            donor = src_j

        if a[donor] < 1e-10:
            stall += 1
            break

        improved = False
        # Try several destination candidates
        n_cand = min(200, n)
        cands = rng.choice(n, n_cand, replace=False)

        for dst in cands:
            if dst == donor or dst == src_i or dst == src_j:
                continue
            for frac in [0.002, 0.005, 0.01, 0.02]:
                amt = a[donor] * frac
                if amt < 1e-14:
                    break
                a_trial = a.copy()
                a_trial[donor] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-14:
                    break
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1

        if verbose and stall % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      PairSwap ({elapsed:.1f}s): J={J:.6f}, k={k_peak}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      PairSwap done ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def get_parent_sequence():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p.copy()
    except Exception:
        pass
    return None


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)
    rng = np.random.default_rng(2024)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # Load parent
    parent = get_parent_sequence()
    if parent is not None:
        p_norm = norm_sum1(parent)
        p_score = evaluate_safe(p_norm)
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
        best_seq = p_norm.copy()
        best_score = p_score
    else:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: LP-FW (main workhorse) - 50% of time ===
    rem = remaining_time()
    print(f"\n=== Phase 1: LP-FW n={n0} ===")
    t1 = rem * 0.50
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_enhanced(a_in, t1, solver, verbose=True)
        try_update(a_out, f"lpfw_n{n0}")

    # === Phase 2: Peak-targeted coordinate descent - 15% ===
    rem = remaining_time()
    print(f"\n=== Phase 2: PeakCD n={len(best_seq)} ===")
    t2 = rem * 0.15
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_targeted_cd(a_in, t2, verbose=True, seed=42)
        try_update(a_out, f"peakcd_n{len(best_seq)}")

    # === Phase 3: LP-FW second round - 15% ===
    rem = remaining_time()
    print(f"\n=== Phase 3: LP-FW round 2 n={len(best_seq)} ===")
    t3 = rem * 0.15
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_enhanced(a_in, t3, solver, verbose=True)
        try_update(a_out, f"lpfw2_n{len(best_seq)}")

    # === Phase 4: High-beta APG for polish - 10% ===
    rem = remaining_time()
    print(f"\n=== Phase 4: APG high-beta n={len(best_seq)} ===")
    t4 = rem * 0.10
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = apg_high_beta(a_in, t4, beta=80.0, top_k=400, c0=3.0, seed=77, verbose=False)
        try_update(a_out, f"apg_hi_n{len(best_seq)}")

    # === Phase 5: Pair-swap CD - 8% ===
    rem = remaining_time()
    print(f"\n=== Phase 5: PairSwap n={len(best_seq)} ===")
    t5 = rem * 0.08
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = pair_swap_cd(a_in, t5, verbose=True, seed=123)
        try_update(a_out, f"pairswap_n{len(best_seq)}")

    # === Phase 6: Try different length + LP-FW - 5% ===
    rem = remaining_time()
    n1 = min(n0 + 400, 3000)
    if n1 > n0 and rem > 100:
        print(f"\n=== Phase 6: Scale to n={n1} + LP-FW ===")
        a_long = interpolate_sequence(best_seq, n1)
        a_in = norm_sum1(a_long)
        t6 = rem * 0.05
        if t6 > 30:
            a_out, _ = lp_frank_wolfe_enhanced(a_in, t6, solver, verbose=True)
            try_update(a_out, f"lpfw_n{n1}")

    # === Phase 7: Final LP-FW at best - remaining time ===
    rem = remaining_time()
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW n={n_best} ===")
    if rem > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_enhanced(a_in, rem * 0.6, solver, verbose=True)
        try_update(a_out, f"lpfw_final_n{n_best}")

    # === Phase 8: Final peak-targeted CD ===
    rem = remaining_time()
    n_best = len(best_seq)
    if rem > 10:
        print(f"\n=== Phase 8: Final PeakCD n={n_best} ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_targeted_cd(a_in, rem * 0.7, verbose=True, seed=314)
        try_update(a_out, f"peakcd_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
