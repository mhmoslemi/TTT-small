```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS with log-sum-exp smoothing (annealed beta) for fast global optimization
2. Peak-targeting mirror descent (Nesterov-accelerated) for fine-tuning
3. Coordinate descent for local polishing
4. Multi-length exploration (2400 to 8000)
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_autocorr(a):
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def spectral_equalize(a, n_iter=30):
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = A.real**2 + A.imag**2
        C = 1.0 / np.sqrt(P + 1e-12)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.5 * a + 0.5 * a_new
    return a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
```

```python
def lbfgs_smoothed(a0, time_limit, beta_schedule='auto'):

    a = normalize(a0)
    theta = np.log(a + 1e-30)  # log parameterization

    # θ = log(a) works: softmax(log(a)) = a / sum(a) = a (if sum(a)=1)
    
    t0 = time.monotonic()

    best_obj = fast_autocorr(a).max()
    best_a = a.copy()
    
    for beta, n_iter in zip(betas, iters_per_beta):
        if time.monotonic() - t0 > time_limit:
            break
        
        def objective_and_gradient(theta):
            # a = softmax(theta)
            theta_shifted = theta - theta.max()
            exp_theta = np.exp(theta_shifted)
            a = exp_theta / exp_theta.sum()
            
            # Compute autocorrelation
            c = fast_autocorr(a)
            
            # Compute log-sum-exp

            max_bc = beta_c.max()
            log_sum_exp = max_bc + np.log(np.sum(np.exp(beta_c - max_bc)))
            f = 2.0 * n * log_sum_exp / beta
            
            # Compute weights
            w = np.exp(beta_c - max_bc) / np.sum(np.exp(beta_c - max_bc))
            
            # Compute d_i = sum_m a[m] * w[i+m]
```

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-length search (interpolate parent to larger n)
2. L-BFGS with log-sum-exp smoothed objective and beta annealing
3. Nesterov subgradient with mirror descent (exact max, top-K peaks)

import numpy as np
import time
from scipy.optimize import minimize


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]

def spectral_equalize(a, n_iter=25):
    """Flatten power spectrum via iterative DFT correction."""
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = A.real ** 2 + A.imag ** 2
        C = 1.0 / np.sqrt(P + 1e-14)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.5 * a + 0.5 * a_new
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def softmax(theta):
    """Numerically stable softmax."""
    t = theta - theta.max()
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(theta) / len(theta)
    return e / s


def compute_smoothed_objective_and_grad(theta, beta, n, fft_size):
    """
    Compute log-sum-exp smoothed objective and gradient for L-BFGS.
    a = softmax(theta), f = (2n/beta) * log(sum exp(beta * c_k))
    grad_theta_i = 4n * a_i * (d_i - mean_d)
    where d_i = sum_m a_m * w_{i+m}, w_k = exp(beta*c_k) / sum(exp(beta*c))
    """
    a = softmax(theta)
    
    # Autocorrelation
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
    
    # Log-sum-exp
    beta_c = beta * c
    max_bc = beta_c.max()
    exp_bc = np.exp(beta_c - max_bc)
    sum_exp = exp_bc.sum()
    log_sum_exp = max_bc + np.log(sum_exp)
    f = 2.0 * n * log_sum_exp / beta
    
    # Weights
    w = exp_bc / sum_exp
    
    # Compute d_i = sum_m a[m] * w[i+m]
    # d[i] = conv_full(a, w_rev)[2n-2-i]
    Wr = np.fft.fft(w[::-1], fft_size)
    conv = np.fft.ifft(A * Wr).real
    d = conv[2 * n - 2:2 * n - 2 + n][::-1]
    
    # Gradient
    mean_d = np.dot(a, d)
    grad = 4.0 * n * a * (d - mean_d)
    
    if not np.all(np.isfinite(grad)):
        grad = np.zeros_like(theta)
    if not np.isfinite(f):
        f = 1e10
        grad = np.zeros_like(theta)
    
    return f, grad


def lbfgs_optimize(a0, time_limit, max_beta_mult=100):
    """L-BFGS with annealed beta schedule."""
    n = len(a0)
    a = normalize(a0)
    theta = np.log(a + 1e-30)
    fft_size = 1 << (3 * n - 2).bit_length()
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    # Beta schedule: mult of n
    beta_mults = [1, 2, 5, 10, 20, 50, max_beta_mult]
    iters_per_level = [300, 250, 200, 150, 150, 100, 100]
    
    for beta_mult, n_iter in zip(beta_mults, iters_per_level):
        if time.monotonic() - t0 > time_limit:
            break
        beta = beta_mult * n
        
        def fun(theta):
            f, g = compute_smoothed_objective_and_grad(theta, beta, n, fft_size)
            return f, g
        
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        
        # Estimate time per iteration
        t1 = time.monotonic()
        f0, g0 = fun(theta)
        t_per_iter = max(time.monotonic() - t1, 0.001)
        max_iter = max(10, min(n_iter, int(remaining / (t_per_iter * 1.5))))
        
        result = minimize(
            fun,
            theta,
            method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': max_iter,
                'gtol': 1e-7,
                'ftol': 1e-14,
                'maxfun': max_iter * 3,
                'iprint': -1
            }
        )
        
        theta_new = np.asarray(result.x, dtype=np.float64)
        a_new = softmax(theta_new)
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
        
        theta = theta_new
        a = a_new
        J = 2.0 * n * best_m
        print(f"      L-BFGS beta={beta:.0f}: J={J:.6f}, status={result.status if hasattr(result, 'status') else '?'}")
    
    # Final: recompute with best theta
    a_final = softmax(theta)
    a_final = np.maximum(a_final, 0.0)
    a_final = a_final / a_final.sum()
    M_final = fast_autocorr(a_final).max()
    if M_final < best_m:
        best_m = M_final
        best_a = a_final.copy()
    
    return best_a, best_m


def subgradient_polish(a0, time_limit, rng_seed=123):
    """Nesterov subgradient with mirror descent, targeting top-K peaks."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    theta = np.log(a + 1e-30)
    fft_size = 1 << (2 * n - 2).bit_length()
    
    top_k = 20
    c0 = 5.0
    stall = 0
    max_stall = 500
    
    x = a.copy()
    x_prev = a.copy()
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation and gradient
        A = np.fft.rfft(y, fft_size)
        c = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
        
        # Top-K peaks with softmax weighting
        k_select = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k_select]
        top_vals = np.maximum(c[top_idx], 1e-30)
        w = top_vals ** 4.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_select) / k_select
        else:
            w = w / w_sum
        
        # Gradient in a-space: grad_a = 2n * sum_k w_k * 2 * a_{shifted by k}
        # grad_a_i = 2n * sum_k w_k * (a_{k-i}^+ + a_{i-k}^+)
        # = 2n * sum_k w_k * 2 * a_{|k-i|} (when valid)
        # Actually: grad_a_i = 2n * sum_k w_k * 2 * a_{k-i} * 1[0<=k-i<=n-1]
        # = 4n * sum_m a_m * (sum_k w_k * 1[k = i+m]) = 4n * sum_m a_m * w_{i+m}
        # Same as before: d_i = sum_m a_m * w_{i+m}
        
        # Compute d via FFT
        w_full = np.zeros(2 * n - 1)
        w_full[top_idx] = w
        Wr = np.fft.fft(w_full[::-1], fft_size)
        conv = np.fft.ifft(A * Wr).real
        d = conv[2 * n - 2:2 * n - 2 + n][::-1]
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_y = np.log(y + 1e-30)
        theta_new = log_y - eta * 4.0 * n * (d - np.dot(y, d))
        a_new = softmax(theta_new)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        x_prev = y
        x = a_new
        
        M_new = c.max()  # Approximate (using y's autocorr)
        M_x = fast_autocorr(x).max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall:
            stall = 0
            pert = rng.normal(0, 0.003, n)
            x_pert = best_a * (1.0 + pert)
            x_pert = np.maximum(x_pert, 1e-12)
            x_pert = x_pert / x_pert.sum()
            x = x_pert
            x_prev = x_pert
            theta = np.log(x_pert + 1e-30)
    
    # Final check
    M_final = fast_autocorr(x).max()
    if M_final < best_m:
        best_m = M_final
        best_a = x.copy()
    
    J = 2.0 * n * best_m
    print(f"      Subgrad: J={J:.6f}")
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=42):
    """Peak-targeted coordinate descent."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        A = np.fft.rfft(a, fft_size)
        c = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
        M_curr = c.max()
        top_k = np.argsort(c)[::-1][:15]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            
            # Find the pair (i, k-i) with maximum contribution to this peak
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            
            # Try moving a small amount from src to random other positions
            n_tries = min(50, n // 10)
            candidates = rng.integers(0, n, size=n_tries)
            
            best_dst = -1
            best_frac = 0
            best_M = M_curr
            
            for dst in candidates:
                dst = int(dst)
                if dst == src:
                    continue
                # Try a few fractions
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M:
                        best_M = M_trial
                        best_dst = dst
                        best_frac = frac
            
            if best_dst >= 0:
                amt = a[src] * best_frac
                a[src] -= amt
                a[best_dst] += amt
                a
