import numpy as np
import scipy.optimize as scipy_opt
import time


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def objective_value(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    ac = fast_autocorr(a)
    max_ac = ac.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_ac / (s * s)


def build_constraint_matrix(y, active_lags):
    """Build LP constraint matrix where row k has y[k-j] at position j."""
    n = len(y)
    K = len(active_lags)
    A = np.zeros((K, n))
    for i, k in enumerate(active_lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        j = np.arange(j_start, j_end + 1)
        A[i, j] = y[k - j]
    return A


def solve_lp_fast(y, M, active_lags):
    """Solve max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0.
    Returns normalized b/sum(b) or None on failure."""
    n = len(y)
    A_ub = build_constraint_matrix(y, active_lags)
    if A_ub.shape[0] == 0:
        return None
    b_ub = np.full(A_ub.shape[0], M)
    c = -np.ones(n)
    try:
        res = scipy_opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                                bounds=[(0.0, None)] * n, method='highs',
                                options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    b_val = np.maximum(res.x, 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def spectral_equalize(a, n_iter=25, eps=1e-10, blend=0.4):
    """Flatten power spectrum via iterative DFT correction."""
    a = normalize(a).copy()
    n = len(a)
    for _ in range(n_iter):
        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.fft(a, fft_size)
        P = np.abs(A) ** 2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new, fft_size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = (1 - blend) * a + blend * a_new
    return a


def smoothed_gradient(a, ac, top_k=50, beta=6.0):
    """Compute gradient of smoothed max using top-K peaks with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def nesterov_mirror_descent(a0, time_limit, top_k=80, beta=5.0, c0=5.0,
                            verbose=True, rng_seed=999):
    """Nesterov-accelerated mirror descent on the simplex."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute gradient
        ac = fast_autocorr(y)
        g = smoothed_gradient(y, ac, top_k=top_k, beta=beta)

        # Mirror descent step (exponential family / softmax)
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y + 1e-30) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = y
        x = a_new

        # Check improvement
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Restart if stalled
        if stall_count > 2000:
            stall_count = 0
            pert = rng.normal(0, 0.003, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def lp_frWK_main(a0, time_limit, top_k_init=30, top_k_max=300,
                 verbose=True, rng_seed=42):
    """LP Frank-Wolfe with Nesterov acceleration, momentum, and adaptive constraints."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    direction_momentum = a.copy()
    momentum_alpha = 0.7
    current_top_k = top_k_init
    stall = 0
    max_stall = 500

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break

        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        # Compute autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()

        # Select active constraints: top-K + some random
        k_sel = min(current_top_k, len(ac))
        top_lags = np.argsort(ac)[::-1][:k_sel]
        n_rand = max(5, k_sel // 8)
        rand_lags = rng.integers(0, len(ac), size=n_rand)
        active_lags = np.unique(np.concatenate([top_lags, rand_lags]))

        direction = solve_lp_fast(y, M_curr, active_lags)
        if direction is None:
            stall += 1
            continue

        # Momentum on direction
        direction_momentum = (1 - momentum_alpha) * direction_momentum + momentum_alpha * direction
        direction_momentum = np.maximum(direction_momentum, 0.0)
        dm_sum = direction_momentum.sum()
        if dm_sum < 1e-15:
            direction_momentum = direction.copy()
        else:
            direction_momentum = direction_momentum / dm_sum

        alpha = 2.0 / (t + 2.0)

        x_new = (1.0 - alpha) * y + alpha * direction_momentum
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx

        x_prev = y
        x = x_new

        # Check improvement
        mx = fast_autocorr(x).max()
        if mx < best_m - 1e-15:
            best_m = mx
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adapt top_k
        if t % 100 == 0 and current_top_k < top_k_max:
            current_top_k = min(current_top_k + 10, top_k_max)

        # Restart if stalled
        if stall > 200 and stall % 50 == 0:
            pert = rng.normal(0, 0.002, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new.copy()
            direction_momentum = x_new.copy()
            stall = 0

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Peak-targeted coordinate descent with exact line search."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # Find best destination
            best_dst = -1
            best_score = float('inf')

            # Sample candidates from low-contribution regions
            n_cand = min(300, n)
            cand = rng.integers(0, n, size=n_cand)

            for dst in cand:
                if dst == src:
                    continue
                sens = 0.0
                for k2 in top_k[:5]:
                    k2i = int(k2)
                    d = k2i - dst
                    if 0 <= d < n:
                        sens += a[d]
                    d2 = dst - k2i
                    if 0 <= d2 < n and d2 != d:
                        sens += a[d2]
                if sens < best_score:
                    best_score = sens
                    best_dst = dst

            if best_dst < 0:
                continue

            dst = best_dst
            found = False
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    break
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    break
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found:
                pass

        if not improved:
            stall += 1
            # Global perturbation
            for _ in range(10):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.005)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            break
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / max(a.sum(), 1e-15)

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def spectral_polish(a0, time_limit, verbose=True, rng_seed=555):
    """Polish in the spectral domain: iteratively reduce peaks by adjusting phase."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    fft_size = 1 << (2 * n - 1).bit_length()

    for step in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < best_m:
            best_m = M_curr
            best_a = a.copy()

        # Find peak lags
        top_lags = np.argsort(ac)[::-1][:20]

        # Compute FFT
        A = np.fft.fft(a, fft_size)

        # Compute current IFFT(|A|^2) and gradient wrt A
        # We want to reduce peaks: modify |A|^2 to reduce IFFT peaks
        # Gradient of IFFT(|A|^2)[k] wrt A[m] is complex;
        # Instead, use a heuristic: reduce |A| at frequencies that contribute to peaks

        # Weighted approach: compute contribution of each frequency to each peak
        # |A[m]|^2 * exp(i*2*pi*m*k/fft_size) contributes to peak at k
        # We want to reduce this for top peaks

        # Simple approach: for each top peak, compute the "frequency content" that
        # feeds into it, and reduce those frequencies
        reduce_mask = np.zeros(fft_size)
        for k in top_lags:
            k_int = int(k)
            # Frequency m contributes to time k via exp(i*2*pi*m*k/fft_size)
            # The "energy" at frequency m is |A[m]|^2
            # We want to reduce frequencies where the phase aligns with the peak
            phase = 2.0 * np.pi * np.arange(fft_size) * k_int / fft_size
            alignment = np.cos(phase)  # positive = constructive
            reduce_mask += np.maximum(alignment, 0) * np.abs(A) ** 2

        # Reduce |A| where reduce_mask is high
        if reduce_mask.max() > 0:
            reduction = 1.0 - 0.1 * (reduce_mask / reduce_mask.max())
            A_new = A * np.sqrt(np.maximum(reduction, 0.5))
            a_new = np.real(np.fft.ifft(A_new, fft_size))[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            a = 0.7 * a + 0.3 * a_new

        # Add small random perturbation
        if step % 100 == 0:
            pert = rng.normal(0, 0.001, n)
            a = a * (1.0 + pert)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()

        if verbose and step % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SpecPolish step={step} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase strategy."""
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)

    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Try to load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize(parent)
            best_score = p_score
            parent_loaded = True
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Spectral pre-conditioning ===
    print(f"\n=== Phase 1: Spectral Equalization (n={n0}) ===")
    a_spec = spectral_equalize(best_seq, n_iter=30, blend=0.5)
    try_update(a_spec, "spectral")

    # === Phase 2: Main LP-FW (largest time allocation) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.45, 700.0)
    if t2 > 60:
        a_in = normalize(best_seq)
        a_out, _ = lp_frWK_main(a_in, t2, top_k_init=30, top_k_max=300, verbose=True, rng_seed=42)
        try_update(a_out, f"lpfw_n{len(a_out)}")

    # === Phase 3: Nesterov Mirror Descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: NMD Polish (n={n_best}, budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_in = normalize(best_seq)
        a_out, _ = nesterov_mirror_descent(a_in, t3, top_k=80, beta=5.0, c0=5.0, verbose=True, rng_seed=999)
        try_update(a_out, f"nmd_n{len(a_out)}")

    # === Phase 4: LP-FW again (refined, more constraints) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW Refined (n={n_best}, budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 350.0)
    if t4 > 60:
        a_in = normalize(best_seq)
        a_out, _ = lp_frWK_main(a_in, t4, top_k_init=80, top_k_max=400, verbose=True, rng_seed=777)
        try_update(a_out, f"lpfw2_n{len(a_out)}")

    # === Phase 5: Spectral Polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Spectral Polish (n={n_best}, budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_in = normalize(best_seq)
        a_out, _ = spectral_polish(a_in, t5, verbose=True, rng_seed=555)
        try_update(a_out, f"specpolish_n{len(a_out)}")

    # === Phase 6: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coord Descent (n={n_best}, budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 200.0)
    if t6 > 20:
        a_in = normalize(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{len(a_out)}")

    # === Phase 7: Try a different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    n_alt = n_best + 400
    if n_alt != n_best and remaining > 200:
        print(f"\n=== Phase 7: Alt Length (n={n_alt}, budget={remaining:.0f}s) ===")
        a_long = interpolate_sequence(best_seq, n_alt)
        a_long = spectral_equalize(a_long, n_iter=20, blend=0.5)
        try_update(a_long, f"interp_n{n_alt}")

        t7a = min(remaining * 0.5, 300.0)
        if t7a > 60:
            a_in = normalize(a_long)
            a_out, _ = lp_frWK_main(a_in, t7a, top_k_init=40, top_k_max=250, verbose=True, rng_seed=123)
            try_update(a_out, f"lpfw_n{n_alt}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t7b = min(remaining * 0.4, 100.0)
            if t7b > 20:
                a_in = normalize(best_seq)
                a_out, _ = nesterov_mirror_descent(a_in, t7b, top_k=60, beta=4.0, c0=4.0, verbose=False, rng_seed=31415)
                try_update(a_out, f"nmd_alt_n{len(a_out)}")

    # === Phase 8: NMD on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 8: NMD Final (n={n_best}, budget={remaining:.0f}s) ===")
        t8 = min(remaining * 0.30, 150.0)
        if t8 > 30:
            a_in = normalize(best_seq)
            a_out, _ = nesterov_mirror_descent(a_in, t8, top_k=100, beta=6.0, c0=6.0, verbose=True, rng_seed=27182)
            try_update(a_out, f"nmd_final_n{len(a_out)}")

    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 9: Final Coord (n={n_best}, budget={remaining:.0f}s) ===")
        t9 = min(remaining * 0.60, 120.0)
        if t9 > 15:
            a_in = normalize(best_seq)
            a_out, _ = coordinate_polish(a_in, t9, verbose=True, rng_seed=31415)
            try_update(a_out, f"coord_final_n{len(a_out)}")

    # === Phase 10: Very final short LP-FW burst ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 3000 and remaining > 40:
        print(f"\n=== Phase 10: Final LP-FW Burst (n={n_best}, budget={remaining:.0f}s) ===")
        t10 = min(remaining * 0.70, 80.0)
        if t10 > 20:
            a_in = normalize(best_seq)
            a_out, _ = lp_frWK_main(a_in, t10, top_k_init=100, top_k_max=350, verbose=False, rng_seed=99999)
            try_update(a_out, f"lpfw_burst_n{len(a_out)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
