```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Accelerated Smoothed Frank-Wolfe with log-sum-exp gradient (FFT-based, O(n log n) per iteration)
2. Adaptive temperature (beta) schedule for smooth-to-sharp peak focusing
3. Iterative LP-based Successive Convex Approximation (SCA) with active constraint selection
4. Coordinate descent polish with peak-targeted moves
5. Multi-length exploration and structured restarts
"""

import numpy as np
import time
import cvxpy as cp

_fft_cache = {}

def _next_pow2(n):
    return 1 << (n - 1).bit_length()

def fast_conv(a, b=None):
    """Compute linear convolution via FFT with caching of FFT sizes."""
    if b is None:
        b = a
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = _next_pow2(size)
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def fast_autocorr(a):
    """Compute linear autocorrelation conv(a,a) via FFT."""
    return fast_conv(a, a)

def compute_smoothed_gradient(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*c[k])) where c=conv(a,a).
    Returns gradient vector of length n. O(n log n) via FFT.
    """
    n = len(a)
    c = fast_autocorr(a)
    # Log-sum-exp trick
    max_c = c.max()
    exp_bc = np.exp(beta * (c - max_c))
    w = exp_bc / exp_bc.sum()
    # g[m] = 2 * sum_j a[j] * w[m+j]
    # = 2 * conv(a, w_rev)[2n-2-m] where w_rev = w[::-1]
    w_rev = w[::-1]
    conv_aw = fast_conv(a, w_rev)
    # conv_aw has length 3n-2, we need indices 2n-2, 2n-3, ..., n-1
    g = 2.0 * conv_aw[n-1 : 2*n-1][::-1]
    return g

def smoothed_objective(a, beta):
    """Compute (1/beta)*log(sum exp(beta*c[k])) as a smooth surrogate for max(c)."""
    c = fast_autocorr(a)
    max_c = c.max()
    return max_c + np.log(np.sum(np.exp(beta * (c - max_c)))) / beta

def normalize(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def accel_smoothed_fw(a0, time_limit, verbose=True):
    """
    Accelerated Smoothed Frank-Wolfe with adaptive beta.
    Minimizes max(conv(a,a)) subject to a>=0, sum(a)=1.
    """
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    x_mm = best_a.copy()  # minimizer of the average (for AKW)
    
    beta = 5.0
    beta_growth = 1.003
    stall_count = 0
    best_iter = 0
    rng = np.random.default_rng(12345)
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            beta_nest = (t - 1.0) / (t + 1.5)
            y = x + beta_nest * (x - x_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        
        # Compute smoothed gradient
        g = compute_smoothed_gradient(y, beta)
        g = np.maximum(g, 0.0)  # gradient should be non-negative for non-negative a
        
        # FW step: find q = argmin g (direction to move toward)
        q = int(np.argmin(g))
        
        # Also find p = argmax g (coordinate to decrease)
        p = int(np.argmax(g))
        
        # Pairwise step: move mass from p to q
        if p == q:
            # Use standard FW
            v = np.zeros(n)
            v[q] = 1.0
            alpha = 2.0 / (t + 2.0)
            x_new = (1.0 - alpha) * y + alpha * v
        else:
            # Pairwise: a[p] decreases, a[q] increases
            alpha = 2.0 / (t + 2.0)
            x_new = y.copy()
            move = alpha * x_new[p]
            x_new[p] -= move
            x_new[q] += move
        
        x_new = np.maximum(x_new, 0.0)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new
        
        # Evaluate improvement
        c_new = fast_autocorr(x_new)
        val_new = c_new.max()
        
        if val_new < best_val:
            best_val = val_new
            best_a = x_new.copy()
            best_iter = t
            stall_count = 0
        else:
            stall_count += 1
        
        # Update AKW minimizer
        alpha_mm = 2.0 / (t + 2.0)
        x_mm = (1.0 - alpha_mm) * x_mm + alpha_mm * x_new
        
        x_prev = y
        x = x_new
        
        # Adaptive beta: increase slowly
        beta *= beta_growth
        if beta > 500.0:
            beta = 500.0
        
        # Restart if stalled for too long
        if stall_count > 5000:
            stall_count = 0
            # Perturb from best
            pert = rng.normal(0, 0.01, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new.copy()
            x_mm = x_new.copy()
            beta = 5.0
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            obj = 2.0 * n * best_val
            print(f"    SFW t={t} ({elapsed:.1f}s): obj={obj:.6f}, beta={beta:.1f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        obj = 2.0 * n * best_val
        print(f"    SFW done ({elapsed:.1f}s): obj={obj:.6f}")
    
    return best_a, best_val

def solve_sca_lp(a_t, M_init, n_active, solver):
    """
    Solve one SCA LP iteration:
    min t s.t. linearized constraints, sum(a)=1, a>=0
    Linearized constraint for lag k:
      sum_i [a_t[i]*a[k-i] + a[i]*a_t[k-i]] - c_t[k] <= t
    which simplifies to:
      2 * sum_i a_t[i]*a[k-i] - c_t[k] <= t  (using symmetry)
    Wait, that's not right. Let me redo:
      L_k(a) = sum_i [a_t[i]*a[k-i] + a[i]*a_t[k-i]] - c_t[k]
    The constraint is L_k(a) <= t, i.e.:
      sum_i a_t[i]*a[k-i] + sum_i a[i]*a_t[k-i] <= t + c_t[k]
    
    Note: sum_i a_t[i]*a[k-i] is a linear form in a.
    And sum_i a[i]*a_t[k-i] is also linear in a (it's the same thing by reversal).
    
    Actually: sum_i a_t[i]*a[k-i] = sum_j a[j] * a_t[k-j] (substituting j=k-i)
    And: sum_i a[i]*a_t[k-i] = sum_j a[j] * a_t[k-j] (same expression!)
    
    So L_k(a) = 2 * sum_j a[j] * a_t[k-j] - c_t[k] <= t
    i.e., sum_j a[j] * (2*a_t[k-j]) <= t + c_t[k]
    """
    n = len(a_t)
    c_t = fast_autocorr(a_t)
    M_curr = c_t.max()
    
    # Select active lags (top n_active)
    active = np.argsort(c_t)[::-1][:n_active]
    
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1, t_var >= 0]
    
    for k in active:
        k_int = int(k)
        # Coefficient for a[j] is 2*a_t[k-j] for valid j (0<=k-j<n)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        coeffs = 2.0 * a_t[k_int - j_idx]
        rhs = float(c_t[k_int])
        constraints.append(coeffs @ a_var[j_idx] <= t_var + rhs)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    a_val = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_val.sum()
    if s < 1e-15:
        return None, None
    a_val = a_val / s
    t_val = float(t_var.value)
    
    return a_val, t_val

def sca_polish(a0, time_limit, n_active=100, solver=None, verbose=True):
    """Iterative SCA (linearized constraints) polish."""
    if solver is None:
        solver = _get_solver()
    
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for it in range(1, 500):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_new, t_val = solve_sca_lp(a, best_val, n_active, solver)
        if a_new is None:
            break
        
        # Check if we improved
        val_new = fast_autocorr(a_new).max()
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
        
        # Damped update for stability
        a = 0.5 * a + 0.5 * a_new
        a = normalize(a)
        
        # Check convergence
        if abs(best_val - val_new) < 1e-12:
            break
        
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            obj = 2.0 * n * best_val
            print(f"    SCA it={it} ({elapsed:.1f}s): obj={obj:.6f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        obj = 2.0 * n * best_val
        print(f"    SCA done ({elapsed:.1f}s): obj={obj:.6f}")
    
    return best_a, best_val

            if i != j and a[i] > 1e-10:

def _find_safe_dst(a, top_lags, src, n):
    """Find destination coordinates that are 'safe' (not involved in top peaks)."""
    in_peak = np.zeros(n, dtype=bool)
    in_peak[src] = True
    for k in top_lags[:8]:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        i_idx = np.arange(i_start, i_end + 1)
        j_idx = k_int - i_idx
        valid = (j_idx >= 0) & (j_idx < n)
        in_peak[i_idx[valid]] = True
        in_peak[j_idx[valid]] = True
    
    candidates = np.where(~in_peak)[0]
    if len(candidates) < 10:
        candidates = np.where(a < np.median(a))[0]
        candidates = candidates[candidates != src]
    return candidates

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def tapered_init(n, taper_power=2.0):
    """Create a tapered initialization (reduces edge peaks)."""
    x = np.arange(n) / max(n - 1, 1)
    # Taper: reduced at edges, full in middle
    taper = np.sin(np.pi * x) ** taper_power
    return taper / taper.sum()

def random_binary_init(n, density=0.5, seed=0):
    """Random binary initialization with given density."""
    rng = np.random.default_rng(seed)
    a = (rng.random(n) < density).astype(np.float64)
    if a.sum() < 1:
        a[0] = 1.0
    return a / a.sum()

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    n_alt = n_best + 300 if n_best <= 2400 else n_best - 300
    n_alt = max(1200, min(n_alt, 3200))
    if n_alt != n_best and remaining > 120:
        print(f"\n=== Phase 5: Alt length n={n_alt}, budget={remaining:.0f}s ===")
        a_long = interpolate_sequence(best_seq, n_alt)
        # Apply taper to reduce edges
        x = np.arange(n_alt) / max(n_alt - 1, 1)
        taper = 0.7 + 0.3 * np.sin(np.pi * x) ** 2
        a_long = a_long * taper
        a_long = normalize(a_long)
        
        t5a = min(remaining * 0.40, 200.0)
        if t5a > 30:
            a_out, _ = accel_smoothed_fw(a_long, t5a, verbose=True)
            try_update(a_out, f"sfw_n{n_alt}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            n_best2 = len(best_seq)
            t5b = min(remaining * 0.3, 120.0)
            if t5b > 20:
                a_in = normalize(best_seq)
                a_out, _ = sca_polish(a_in, t5b, n_active=100, solver=solver, verbose=True)
                try_update(a_out, f"sca2_n{n_best2}")
    
    # === Phase 6: SCA with more active constraints ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 2800 and remaining > 60:
        print(f"\n=== Phase 6: SCA (more constraints) n={n_best}, budget={remaining:.0f}s ===")
        t6 = min(remaining * 0.35, 200.0)
        if t6 > 30:
            a_in = normalize(best_seq)
            a_out, _ = sca_polish(a_in, t6, n_active=150, solver=solver, verbose=True)
            try_update(a_out, f"sca3_n{n_best}")
    
    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Final Coord n={n_best} ===")
        t7 = min(remaining * 0.50, 120.0)
        if t7 > 10:
            a_in = normalize(best_seq)
            a_out, _ = coordinate_polish(a_in, t7, verbose=True, rng_seed=777)
            try_update(a_out, f"coord_final_n{n_best}")
    
    # === Phase 8: One more SFW with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 8: SFW final n={n_best} ===")
        t8 = min(remaining * 0.70, 150.0)
        if t8 > 15:
            # Perturb slightly for diversity
            a_in = normalize(best_seq)
            pert = rng.normal(0, 0.003, n_best)
            a_in = np.maximum(a_in * (1 + pert), 1e-8)
            a_in = a_in / a_in.sum()
            a_out, _ = accel_smoothed_fw(a_in, t8, verbose=True)
            try_update(a_out, f"sfw_pert_n{n_best}")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
```
