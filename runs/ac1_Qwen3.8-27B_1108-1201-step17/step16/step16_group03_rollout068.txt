"""

import numpy as np
import time


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def fast_corr_valid(w, a):
    """c[i] = sum_j w[i+j]*a[j] for i=0..len(a)-1; w has len >= len(a)."""
    n = len(a)
    m = len(w)
    needed = m + n - 1
    fft_size = 1 << (needed - 1).bit_length()
    if fft_size < 2:
        fft_size = 2
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    full = np.fft.irfft(W * np.conj(A), fft_size)
    return full[:m - n + 1]


def smooth_conv_grad(a, beta):
    """Returns (lse_max_value, gradient) of smoothed conv max."""
    ac = fast_conv(a)
    x_max = ac.max()
    shifted = beta * (ac - x_max)
    e = np.exp(shifted)
    s_e = e.sum()
    if s_e < 1e-30:
        M_soft = x_max
        w = np.zeros_like(ac)
        w[np.argmax(ac)] = 1.0
    else:
        M_soft = x_max + np.log(s_e) / beta
        w = e / s_e
    grad = 2.0 * fast_corr_valid(w, a)
    return M_soft, grad


def hard_subgradient(a, top_k=1):
    """Returns (max_val, subgradient) using hard max with top_k peaks."""
    ac = fast_conv(a)
    n = len(a)
    M_curr = ac.max()
    top_idx = np.argsort(ac)[::-1][:top_k]
    grad = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            j = k_int - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    if top_k > 1:
        grad /= float(top_k)
    return M_curr, grad


def mirror_step(a, grad, eta):
    log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def nesterov_smooth(a0, time_limit, beta_start, beta_end, c0, verbose, seed):

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 3000

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta = beta_start * (beta_end / beta_start) ** progress

        y = x + momentum * (x - x_prev)

        sy = y.sum()

            y = x.copy()
        else:
            y = y / sy

            M_soft, grad = smooth_conv_grad(y, beta)

        x_prev = x
        x = x_new

        M_hard = fast_conv(x).max()

        else:
            stall += 1

            stall = 0

            J = 2.0 * n * best_m
            print(f"    SmoothND t={t} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}, stall={stall}")

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_init(n, kind, seed):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'sine':
        a = np.sin(np.pi * x) ** 2
    elif kind == 'triangle':
        a = 1.0 - np.abs(2.0 * x - 1.0)
    elif kind == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    elif kind == 'u_shaped':
        a = (2.0 * x - 1.0) ** 2 + 0.1
    elif kind == 'random':
        a = rng.random(n) + 0.01
    elif kind == 'cosine':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif kind == 'bimodal':
        a = np.exp(-30.0 * (x - 0.25) ** 2) + np.exp(-30.0 * (x - 0.75) ** 2)
    elif kind == 'flat_mid':
        a = np.ones(n)
        if n >= 10:
            a[:n // 4] = 0.5
            a[3 * n // 4:] = 0.5
    else:
        a = np.sin(np.pi * x) ** 2
    return norm_sum1(a)

def attempt_update(a, best_seq, best_score, t_global, label):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_seq = a_norm.copy()
        best_score = sc
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def get_lp_solver():
    try:
        import cvxpy as cp
        for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
            try:
                x = cp.Variable(4, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
                prob.solve(solver=s, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    print(f"  LP solver: {s}")
                    return s
            except Exception:
                continue
    except Exception:
        pass
    return None

        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
