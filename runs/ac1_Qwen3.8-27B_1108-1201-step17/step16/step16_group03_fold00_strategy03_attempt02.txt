def hard_subgradient(y, n, k_star):
    """Computes the exact subgradient of the convolution peak."""
    grad = np.zeros(n, dtype=np.float64)

def softmax_topk(y, ac, top_k):
    """Soft subgradient: weighted average of top-k peak gradients."""
    n = len(y)
    top_idx = np.argsort(ac, axis=-1, kind='mergesort', order=''])

    # Handle case where no valid peaks exist
    if top_k == 0:
        return np.zeros(n)

    # Find the strongest positive peaks (avoid sorting large arrays multiple times)
    valid_peaks = (ac[top_idx[top_k-1] if top_k > 0 else top_idx] > 1e-10) * top_idx[0:top_k]
    top_vals = ac[valid_peaks[top_k-1:top_k] if top_k <= len(valid_peaks) else valid_peaks]
    top_vals = np.maximum(top_vals, 1e-15)
    top_vals[top_vals < 1e-10] = 1e-15  # Ensure positive weights

    # Smooth the top_k gradient with a prominence-based weight
    beta = 4.0
    sorted_peaks = np.argsort(top_vals)[::-1]
    indices = []
    weights = []
    for i in range(min(top_k, n//2)):
