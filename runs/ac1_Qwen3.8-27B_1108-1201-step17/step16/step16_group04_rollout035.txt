def lp_fw_step(a, M, active_lags, solver):
    """Solve LP for FW direction."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(b_var) == 1]
    for k in active_lags:
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        coeff = a[ki - j_idx]
        constraints.append(coeff @ b_var[j_idx] <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s
