import numpy as np
import time

def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def logsumexp(x, beta=1.0):
    """Compute log(sum(exp(beta * x))) / beta stably."""
    bx = beta * np.asarray(x, dtype=np.float64)
    m = bx.max()
    if m > 100:
        return m / beta  # overflow protection, just return max
    return (m + np.log(np.exp(bx - m).sum())) / beta

def compute_gradient_smoothed(a, ac, beta, top_k=None):
    """Compute gradient of (1/beta)*log(sum(exp(beta*ac[k]))).
    
    Returns gradient with respect to a.
    """
    n = len(a)
    if top_k is not None:
        if top_k < len(ac):
            top_idx = np.argsort(ac)[::-1][:top_k]
            ac_work = np.zeros_like(ac)
            ac_work[top_idx] = ac[top_idx]
            ac = ac_work
    
    beta_ac = beta * ac
    max_val = beta_ac.max()
    weights = np.exp(beta_ac - max_val)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        # Fallback: use only the argmax
        k_max = int(np.argmax(ac))
        grad = np.zeros(n)
        i_start = max(0, k_max - n + 1)
        i_end = min(n, k_max + 1)
        idx = np.arange(i_start, i_end)
        grad[idx] = 2.0 * a[k_max - idx]
        return grad
    weights /= w_sum
    
    grad = np.zeros(n)
    for k in range(len(ac)):
        w = weights[k]
        if w < 1e-15:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        grad[idx] += 2.0 * w * a[k - idx]
    return grad

def compute_gradient_topk(a, ac, k_list):
    """Compute gradient of max over specific lags (uniform weighting)."""
    n = len(a)
    grad = np.zeros(n)
    for k in k_list:

        i_end = min(n, k + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        grad[idx] += (2.0 / len(k_list)) * a[k - idx]
    return grad
