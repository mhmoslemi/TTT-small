Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed L-BFGS with continuation on smoothing parameter
2. DFT-based spectral equalization for initialization and periodic correction
3. Multi-length exploration via interpolation
4. Peak-targeted coordinate descent for final polish

import numpy as np
from scipy.optimize import minimize
import time

def _fft_size(n):
    """Next power of 2 >= n."""
    return 1 << (n - 1).bit_length()

def _compute_conv(a):
    """Compute linear auto-convolution via FFT."""
    n = len(a)
    N = _fft_size(2 * n - 1)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2*n-1]
