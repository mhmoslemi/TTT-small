def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize_to_sum(a, target_sum):
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target_sum / len(a)
    return a * target_sum / s


def cross_correlation(w, a):
    """Compute (w ⊛ a)[j] = sum_k w[k]*a[k-j] for j=0..n-1, where w has length 2n-1."""
    n = len(a)
    needed = len(w) + n - 1
    N = 1 << (needed - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    return irfft(np.conj(W) * A, N)[:n]


def compute_subgradient(a, conv, mu=5.0, n_peaks=20):
    """Compute weighted subgradient of max(conv) using soft-max peaks."""
    n = len(a)
    M = conv.max()
    if M < 1e-15:
        return np.zeros(n)
    
    # Soft-max weights over peaks
    n_top = min(n_peaks, len(conv))
    top_idx = np.argsort(conv)[-n_top:]
    vals = conv[top_idx]
    
    # Exponential weighting (heavier on peaks closer to max)
    scale = max(M * 0.03, 1e-10)
    weights = np.exp(-(M - vals) / scale)
    w_sum = weights.sum()
    if w_sum < 1e-15:
        weights = np.ones(len(weights)) / len(weights)
    else:
        weights = weights / w_sum
    
    # Build full weight vector
    w_full = np.zeros(2 * n - 1)
    w_full[top_idx] = weights
    
    # Cross-correlation via FFT
    g = 2.0 * cross_correlation(w_full, a)
    return g


def acgm_optimize(a0, time_limit, verbose=False, mu=5.0, n_peaks=20, seed=0):
    """
    Accelerated Conditional Gradient (Nesterov-accelerated Frank-Wolfe).
    Minimizes max autocorrelation on the simplex {a>=0, sum(a)=sqrt(2n)}.
    Each iteration is O(n log n) via FFT.

        if time.monotonic() - t0 > time_limit:
            break

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))

        z = np.maximum(z, 0.0)
        
        # Compute subgradient at z
        conv_z = fast_autocorr(z)
        g = compute_subgradient(z, conv_z, mu=mu, n_peaks=n_peaks)
        
        # Frank-Wolfe linear oracle: v = target_sum * e_{argmin g}
        min_idx = np.argmin(g)
        
        # Adaptive step size: Polyak-like when possible
        # Gap: f(b_curr) - min_{v in simplex} <g, v> = f(b_curr) - target_sum * min(g)
        f_curr = fast_autocorr(b_curr).max()
        gap = f_curr - target_sum * np.min(g)
        
        if gap > 1e-14:
            # Use adaptive step
            denom = np.dot(g, b_curr) - target_sum * g[min_idx]
            if denom > 1e-14:
                alpha = min(1.0, gap / denom)
            else:
                alpha = 2.0 / (t_new + 1.0)
        else:
            alpha = 0.0
        
        # Fallback: standard Nesterov step
        if alpha < 1e-10:
            alpha = 2.0 / (t_new + 1.0)
        
        # Update
        v = np.zeros(n)
        v[min_idx] = target_sum

        b_curr = b_next
        t = t_new
        n_iters += 1
        
        # Restart if stalled for too long

            b_prev = best_a.copy()
            b_curr = best_a.copy()
            t = 1.0
            stall = 0
        
        if verbose and n_iters % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ACGM iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # Binary search for threshold
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv[-1] - target_sum))[0][-1:]
    if len(rho) == 0:
        return np.ones(n) * target_sum / n
    theta = (cssv[rho[0]] - target_sum) / (rho[0] + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_at_k(a, k, n):
    """Compute subgradient of (a*a)[k] w.r.t. a."""
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        grad[lo:hi] += 2.0 * a[k - np.arange(lo, hi)]
    return grad


def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    import cvxpy as cp
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)
    
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

        print(f"\n=== Phase 6: CD n={n_cur}, budget={t6:.0f}s ===")
        a_norm = normalize_to_sum(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = coordinate_descent(a_norm, t6, verbose=True, seed=5)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd_n{n_cur}", t_global)
    
    # === Phase 7: LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.10, 120.0)
    if t7 > 40:
        print(f"\n=== Phase 7: LP polish n={n_cur}, budget={t7:.0f}s ===")
        a_norm = normalize_to_sum(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = lp_nesterov_optimize(a_norm, t7, solver, verbose=False, seed=123)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp2_n{n_cur}", t_global)
    
    # === Phase 8: Final CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        t8 = min(remaining * 0.5, 60.0)
        print(f"\n=== Phase 8: Final CD n={n_cur}, budget={t8:.0f}s ===")
        a_norm = normalize_to_sum(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = coordinate_descent(a_norm, t8, verbose=False, seed=77)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cdFinal_n{n_cur}", t_global)
    
    # === Phase 9: Perturbation restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 100:
        n_restarts = 2
        t9_each = min(remaining * 0.3 / n_restarts, 80.0)
        for r in range(n_restarts):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            t9 = min(t9_each, remaining - 20)
            if t9 < 30:
                break
            print(f"\n=== Phase 9r{r}: Perturb+ACGM n={n_cur}, budget={t9:.0f}s ===")
            a_out = perturb_and_res_optimize(best_seq, t9, solver, rng, seed=500 + r)
            best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"pert{r}_n{n_cur}", t_global)
    
    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
