                # conv_new[peak_idx] = conv[peak_idx] + 2*(x-alpha)*a[j_offset] (if j_offset != m)

                elif a[j_offset] > 1e-10:

                    # conv[peak_idx] + 2*(x-alpha)*a[j_offset] = peak_val * 0.9
                    x_target = (peak_val * 0.9 - conv[peak_idx]) / (2 * a[j_offset]) + alpha
                    if x_target > 0:
                        candidates.append(x_target)

                    # (to "flatten" the top)
                    sorted_conv = np.sort(conv)[::-1]

                        if x_target2 > 0:

            # Geometric spacings
            for mult in [0.1, 0.3, 0.5, 0.7, 1.5, 2.0, 3.0, 5.0]:
                candidates.append(alpha * mult)

            candidates = [max(0.0, min(c, norm)) for c in candidates]

                # Diagonal correction

                # Full max: max of unchanged and changed
                new_max = max(np.max(new_affected), np.max(np.delete(conv, np.arange(m, min(m+n, 2*n-1)))))
                
                if new_max < best_fx:
                    best_fx = new_max
                    best_x = x
            
            if best_x != alpha and best_fx < best_score - 1e-12:

                s_new = a.sum()

                    actual_score = fast_autocorr(a).max()
                    if actual_score < best_score - 1e-12:
                        best_score = actual_score
                        best_a = a.copy()
                        improved_in_pass = True
        
        if not improved_in_pass:
            break
        
        if verbose and iteration % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD-Opt iter={iteration} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

_global_solver = None

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize_to(a, n):
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s


def cross_corr(a, b):
    """Compute cross-correlation sum_k a[k]*b[k-m] for m=0..len(b)-1 using FFT."""
    na = len(a)
    nb = len(b)
    N = 1 << (na + nb - 1 + 1).bit_length()
    A = rfft(a, N)
    B = rfft(b, N)
    result = irfft(A * np.conj(B), N)
    return result[:nb]


def l2_calibration(a0, time_limit, seed=0, verbose=False, max_iters=2000):
    """Minimize sum(conv^2) using gradient descent with cross-correlation gradient."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:

        a = a * norm / s0

    t0 = time.monotonic()

    patience = 0
    max_patience = 300

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
