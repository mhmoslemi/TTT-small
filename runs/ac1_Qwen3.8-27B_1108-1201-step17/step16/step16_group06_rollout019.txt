"""
Minimizes 2*n*max(autocorr(a))/sum(a)^2 using:

3. Peak-pair coordinate descent for final polish

import numpy as np
import time

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def proj_simplex(v, target=1.0):
    """Project vector v onto the simplex {x: sum(x) = target, x >= 0}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target))[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgrad_autocorr(a, k):
    """Compute subgradient of max autocorrelation at peak index k."""
    n = len(a)
    s = np.zeros(n)
    if k >= 0:
        s[:n-k] += a[k:]
    else:
        s[-k:] += a[:n+k]
    if k >= 0:
        s[k:] += a[:n-k]
    else:
        s[:n+k] += a[-k:]
    return s

def fista_soft(a0, time_limit, top_k=5, temperature=1.0, verbose=True):
    """FISTA with soft subgradient (weighted average over top-k peaks)."""
    n = len(a0)
    a = proj_simplex(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    
    y = a.copy()
    x_prev = a.copy()
    t = 1
    step = 0.5  # 1/L, L=2
    
    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac_y = fast_autocorr(y)
        
        # Soft subgradient: weight top-k peaks
        top_indices = np.argsort(ac_y)[::-1][:top_k]
        top_vals = ac_y[top_indices]
        # Softmax weights
        w = np.exp(top_vals / max(temperature, 1e-10) - top_vals.max())
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, ki in zip(w, top_indices):
            grad += wi * subgrad_autocorr(y, int(ki))
        
        # FISTA update
        beta = (t - 1.0) / (t + 1.5)
        z = y - step * grad
        x_new = proj_simplex(z)
        y = x_new + beta * (x_new - x_prev)
        x_prev = x_new
        t += 1
        
        # Track best
        mx = fast_autocorr(x_new).max()
        if mx < best_m:
            best_m = mx
            best_a = x_new.copy()
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA it={it} ({elapsed:.1f}s): obj={2*n*best_m:.6f}")
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver_name, max_iters=5000, verbose=True):
    """LP-based Frank-Wolfe method."""
    import cvxpy as cp
    n = len(a0)
    a = proj_simplex(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    x_prev = a.copy()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-15:
            break
        
        # Build constraint matrix C[k, i] = y[k-i] if 0 <= k-i < n
        num_lags = 2 * n - 1
        C = np.zeros((num_lags, n))
        for i in range(n):
            # For lag k, C[k,i] = y[k-i] if 0 <= k-i < n
            # i.e., k-i in [0, n-1], so k in [i, i+n-1]
            k_start = i
            k_end = i + n - 1
            C[k_start:k_end+1, i] = y[:n]  # y[0], y[1], ..., y[n-1]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [C @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov-accelerated update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}")
    
    return best_a, best_m

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
