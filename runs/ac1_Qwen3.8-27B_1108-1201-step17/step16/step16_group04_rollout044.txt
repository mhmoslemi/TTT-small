def fast_autocorr(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def score_from_maxb(a):
    """Compute the objective from the sequence (assumed normalized to sum=1)."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def spectral_balance(a, n_iter=50, blend=0.3):
    """Iteratively flatten the power spectrum."""
    a = normalize(a).copy()
    n = len(a)
    for it in range(n_iter):
        A = np.fft.fft(a)
        P = np.abs(A) ** 2
        # Target: flat power spectrum
        P_target = P.sum() / n
        # Iterate: adjust phases to flatten
        # a_new = ifft(A * (P_target / P)^0.5)  -- but this changes a, not just phase
        # Better: a_new = ifft(A / |A| * |A_target|^0.5) where |A_target| = const
        A_flat = A / (np.abs(A) + 1e-15) * np.sqrt(P_target)
        a_candidate = np.real(np.fft.ifft(A_flat))
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s < 1e-15:
            break
        a_candidate = a_candidate / s
        a = (1.0 - blend) * a + blend * a_candidate
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    return a


def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def lp_direction(y, M, active_lags, solver):
    """
    Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0.
    Returns normalized direction or None.
    """
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

        s = v.sum()
        if s < 1e-15:
            return float('inf')
        v = v / s
        ac = fast_autocorr(v)
        return ac.max()
    
    # Ternary search
    lo, hi = 0.0, 1.0
    for _ in range(30):

        f1 = f_alpha(m1)

        if f1 < f2:

        else:
