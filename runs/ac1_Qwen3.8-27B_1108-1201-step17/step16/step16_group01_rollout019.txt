def exact_subgradient(a, ac, threshold_frac=0.995):
    """Compute exact subgradient from active set."""

def log_space_update(a, g, eta):
    """Mirror descent update in log-space."""
    log_a = np.log(np.maximum(a, 1e-18)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def backtracking_logspace(a, g, eta_max, c=0.5, rho=0.7, max_iters=30):
    """Backtracking line search in log-space to find a valid step size."""
    f_curr = fast_conv(a).max()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return a.copy(), eta_max
    g_norm = g / gn
    eta = eta_max
    for _ in range(max_iters):
        a_trial = log_space_update(a, g_norm, eta)
        f_trial = fast_conv(a_trial).max()
        if f_trial <= f_curr - 0.1 * eta * gn * (f_curr + 1e-10):
            return a_trial, eta
        eta *= rho
        if eta < 1e-12:
            break
    return a.copy(), eta

            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac_y = fast_conv(y)
        # Use exact subgradient early, softmax later for exploration
        if t < 10000:
            g = exact_subgradient(y, ac_y, threshold_frac=threshold_frac)
        else:
            T = T_switch + 0.1 * np.sin(t * 0.001)
            g = softmax_subgradient(y, ac_y, T=max(T, 0.05), top_k=60)

        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        eta = c0 / (np.sqrt(t) + 1.0) * np.sqrt(n)
        a_new = log_space_update(y, g, eta)

        x_prev = x
        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > max_stall:
            stall_count = 0
            # Mix of time and frequency domain perturbation
            if rng.random() < 0.5:
                perturbation = rng.normal(0, 0.015, n) * best_a
                x_restarted = best_a + perturbation
            else:
                fft_best = np.fft.rfft(best_a)
                phase_noise = rng.normal(0, 0.03, len(fft_best))
                fft_pert = fft_best * np.exp(1j * phase_noise)
                x_restarted = np.fft.irfft(fft_pert, n)
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD-HYB t={t} ({elapsed:.1f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def aggressive_coord_descent(a0, time_limit, verbose=True, rng_seed=123, rng=None):
    """Aggressive coordinate descent with adaptive step sizes and peak tracking."""
    if rng is None:
        rng = np.random.default_rng(rng_seed)
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_conv(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 80

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        peak_k = np.argsort(ac)[::-1][:min(20, len(ac))]
        improved = False

        # Build peak sensitivity map
        peak_sens = np.zeros(n)
        for k in peak_k[:12]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            peak_sens[idx] += a[j_idx]

        # Find hot positions (high sensitivity) and try to reduce them
        hot_threshold = np.percentile(peak_sens, 85)
        hot_positions = np.where(peak_sens > hot_threshold)[0]
        cold_positions = np.where(peak_sens <= hot_threshold)[0]

        if len(cold_positions) == 0:
            cold_positions = np.where(peak_sens == peak_sens.min())[0]

        rng.shuffle(hot_positions)
        rng.shuffle(cold_positions)

        n_hot = min(len(hot_positions), 50)
        n_cold = min(len(cold_positions), 60)

        for src in hot_positions[:n_hot]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue

            for dst in cold_positions[:n_cold]:
                if dst == src:
                    continue
                # Try multiple transfer fractions
                for frac in [0.01, 0.03, 0.08, 0.15, 0.3]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                continue

        if not improved:
            stall += 1
            # Escape mechanism: random multi-coordinate perturbation
            n_perturb = max(5, n // 80)
            indices = rng.choice(n, size=n_perturb, replace=False)
            for i in indices:
                if a[i] > 1e-8:
                    j = int(rng.integers(n))
                    if j != i:
                        frac = rng.uniform(0.03, 0.15)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
            a = np.maximum(a, 0.0)
            if a.sum() > 1e-15:
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    # Revert (approximately by re-normalizing from best)
                    a = best_a + rng.normal(0, 0.005, n) * best_a
                    a = np.maximum(a, 1e-10)
                    a = a / a.sum()

        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def newton_coord_descent(a0, time_limit, verbose=True, rng_seed=77, rng=None):
    """Coordinate descent with Newton-motivated step sizes for peak reduction."""
    if rng is None:
        rng = np.random.default_rng(rng_seed)
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_conv(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 60

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        k_peak = int(np.argmax(ac))
        M_curr = ac[k_peak]
        if M_curr < 1e-20:
            break

        # Get the gradient direction for the peak lag
        i_start = max(0, k_peak - n + 1)
        i_end = min(n, k_peak + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_peak - idx
        grad_peak = 2.0 * a[j_idx]

        # Find the position with maximum contribution to the peak
        contrib = a[idx] * a[j_idx]
        max_contrib_idx = idx[np.argmax(contrib)]

        # Newton-like step: move mass from high-contrib to low-sensitivity position
        # Estimate "curvature" by trying a few step sizes
        peak_sens = np.zeros(n)
        top_lags = np.argsort(ac)[::-1][:10]
        for k2 in top_lags:
            k2i = int(k2)
            s2 = max(0, k2i - n + 1)
            e2 = min(n, k2i + 1)
            if s2 >= e2:
                continue
            idx2 = np.arange(s2, e2)
            j2 = k2i - idx2
            peak_sens[idx2] += a[j2]

        cold_mask = peak_sens < np.percentile(peak_sens, 30)
        cold_positions = np.where(cold_mask)[0]
        if len(cold_positions) == 0:
            cold_positions = np.array([int(rng.integers(n))])

        improved = False
        # Try moving from the peak contributor to various cold positions
        candidates = cold_positions[np.argsort(peak_sens[cold_positions])[:40]]
        src = max_contrib_idx

        for dst in candidates:
            if dst == src:
                continue
            # Binary search for optimal transfer amount
            lo, hi = 0.0, a[src] * 0.9
            for _ in range(6):
                mid = (lo + hi) / 2
                a_trial = a.copy()
                a_trial[src] -= mid
                a_trial[dst] += mid
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    lo = mid
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    lo = mid
                else:
                    hi = mid
            if improved:
                break

        if not improved:
            stall += 1
            # Random escape
            n_esc = max(3, n // 100)
            for _ in range(n_esc):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.05, 0.2)
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            if a.sum() > 1e-15:
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a = best_a + rng.normal(0, 0.01, n) * best_a
                    a = np.maximum(a, 1e-10)
                    a = a / a.sum()

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Newton-Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def multi_temperature_sweep(a0, time_limit, n_temps=5, verbose=True, rng_seed=55, rng=None):
    """Run AMD with multiple temperatures in parallel (sequentially) and keep best."""
    if rng is None:
        rng = np.random.default_rng(rng_seed)
    n = len(a0)
    best_a = norm_sum1(a0).copy()
    best_m = fast_conv(best_a).max()
    t0 = time.monotonic()
    time_per_run = time_limit / n_temps

    temperatures = [0.1, 0.3, 0.5, 1.0, 2.0]
    c_values = [2.5, 3.0, 4.0, 5.0, 3.5]

    for i in range(n_temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        t_run = min(time_per_run, remaining)
        if t_run < 5:
            break

        a_in = best_a + rng.normal(0, 0.005, n) * best_a
        a_in = np.maximum(a_in, 1e-10)
        a_in = a_in / a_in.sum()

        a_out, m_out = amd_softmax(a_in, t_run, c0=c_values[i], T0=temperatures[i],
                                   T_decay=0.9997, top_k=60, verbose=False,
                                   rng_seed=rng_seed + i, rng=rng)
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    Multi-T run {i} ({elapsed:.1f}s): J={J:.6f}, T={temperatures[i]}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with diverse methods and adaptive time allocation.
    """
    global _global_start_time
    _global_start_time = time.monotonic()
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    rng_master = np.random.default_rng(2024)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    if best_seq is None:
        # Default initialization: raised cosine squared
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.cos(np.pi * (x - 0.5)) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting optimization from n={n0}, score={best_score:.6f}")
    print(f"Time budget: {time_budget:.0f}s")

    # === Phase 1: Quick AMD with exact subgradient (establish baseline) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: AMD exact (n={n0}) ===")
    t1 = min(remaining * 0.15, 180.0)
    if t1 > 20:
        a_out, _ = amd_hybrid(norm_sum1(best_seq), t1, c0=3.0, threshold_frac=0.99,
                              verbose=True, rng_seed=1, rng=rng_master)
        try_update(a_out, "amd_hyb1")

    # === Phase 2: AMD with softmax subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: AMD softmax (n={n_best}) ===")
    t2 = min(remaining * 0.20, 250.0)
    if t2 > 30:
        a_out, _ = amd_softmax(norm_sum1(best_seq), t2, c0=3.5, T0=0.4,
                               T_decay=0.9997, top_k=80, verbose=True,
                               rng_seed=2, rng=rng_master)
        try_update(a_out, "amd_sm2")

    # === Phase 3: Aggressive coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Aggressive coord (n={n_best}) ===")
    t3 = min(remaining * 0.15, 150.0)
    if t3 > 15:
        a_out, _ = aggressive_coord_descent(norm_sum1(best_seq), t3,
                                            verbose=True, rng_seed=3, rng=rng_master)
        try_update(a_out, "coord3")

    # === Phase 4: Newton coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Newton coord (n={n_best}) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        a_out, _ = newton_coord_descent(norm_sum1(best_seq), t4,
                                        verbose=True, rng_seed=4, rng=rng_master)
        try_update(a_out, "newton4")

    # === Phase 5: Multi-temperature sweep ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-T sweep (n={n_best}) ===")
    t5 = min(remaining * 0.15, 180.0)
    if t5 > 30:
        a_out, _ = multi_temperature_sweep(norm_sum1(best_seq), t5, n_temps=4,
                                           verbose=True, rng_seed=5, rng=rng_master)
        try_update(a_out, "multit5")

    # === Phase 6: AMD polish with very tight threshold ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: AMD polish (n={n_best}) ===")
    t6 = min(remaining * 0.12, 120.0)
    if t6 > 15:
        a_out, _ = amd_hybrid(norm_sum1(best_seq), t6, c0=2.0, threshold_frac=0.998,
                              verbose=True, rng_seed=6, rng=rng_master)
        try_update(a_out, "amd_polish6")

    # === Phase 7: Coordinate descent final ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Coord final (n={n_best}) ===")
    t7 = min(remaining * 0.10, 100.0)
    if t7 > 10:
        a_out, _ = aggressive_coord_descent(norm_sum1(best_seq), t7,
                                            verbose=True, rng_seed=7, rng=rng_master)
        try_update(a_out, "coord_final7")

    # === Phase 8: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 8: Length exploration ===")
        lengths_to_try = []
        # Try slightly longer
        for delta in [200, 400]:
            nl = n_best + delta
            if nl <= 4000:
                lengths_to_try.append(nl)
        # Try slightly shorter
        for delta in [200, 400]:
            ns = n_best - delta
            if ns >= 1200:
                lengths_to_try.append(ns)

        time_per_length = (remaining - 30) / max(len(lengths_to_try), 1)
        for nl in lengths_to_try:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            t_len = min(time_per_length, remaining - 20)
            if t_len < 20:
                break
            print(f"  Trying n={nl} (time={t_len:.0f}s)")
            a_long = interpolate_sequence(best_seq, nl)
            a_out, _ = amd_softmax(norm_sum1(a_long), t_len, c0=3.0, T0=0.3,
                                   T_decay=0.9997, top_k=60, verbose=False,
                                   rng_seed=hash(nl) % 10000, rng=rng_master)
            try_update(a_out, f"len_{nl}")

    # === Phase 9: Final AMD with backtracking ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 9: AMD backtracking (n={n_best}) ===")
        t9 = min(remaining * 0.30, 120.0)
        if t9 > 15:
            a_out, _ = amd_softmax(norm_sum1(best_seq), t9, c0=4.0, T0=0.2,
                                   T_decay=0.9998, top_k=50, use_backtrack=True,
                                   verbose=True, rng_seed=9, rng=rng_master)
            try_update(a_out, "amd_bt9")

    # === Phase 10: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 10: Final coord polish (n={n_best}) ===")
        t10 = min(remaining * 0.50, 80.0)
        if t10 > 10:
            a_out, _ = newton_coord_descent(norm_sum1(best_seq), t10,
                                            verbose=True, rng_seed=10, rng=rng_master)
            try_update(a_out, "final_coord10")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
