def weights_and_phi(c, sigma):
    M = float(c.max())
    if M <= 0 or sigma <= 0:
        w = np.ones_like(c)/len(c)
        return M, float(c.mean()), w
    z = (c - M) / sigma
    # clip lower to avoid underflow? exp handles.
    expz = np.exp(z)
    s = float(expz.sum())
    if s <= 0 or not np.isfinite(s):
       w = np.zeros_like(c); imap=int(np.argmax(c)); w[imap]=1.0; return M, M, w
    w = expz/s

"""
Search for non-negative step-function heights minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2
by combining:
  1. Spectral square-root / magnitude-balance initializers for several lengths.
  2. Smoothinged natural-gradient descent in the exponential (softmax) family.
  3. Frank-Wolfe-style linear-programming polish using active convolution peaks.
  4. Final local variants (symmetrization, power transforms, blends).
"""

import numpy as np

import math

def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_nonneg(a):
    a = np.atleast_1d(np.asarray(a, dtype=np.float64)).astype(np.float64, copy=True)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def conv_full(a):
    a = normalize_nonneg(a)
    n = len(a)
    if n == 1:
        return np.array([float(a[0] * a[0])])
    size = 2 * n - 1
    L = next_pow2(size)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:size]
    c = np.asarray(c, dtype=np.float64)
    c = np.maximum(c, 0.0)
    return c


def score_from_c(c, n):
    return float(2.0 * n * np.max(c))


def evaluate_safe(a):
    try:
        arr = np.asarray(a, dtype=np.float64)
        arr = np.clip(arr, 0.0, 1000.0)
        arr = arr[np.isfinite(arr)]
        if arr.size == 0:
            return float("inf")
        arr = arr / arr.sum()
        out = evaluate_sequence([float(x) for x in arr])
        return float(out)
    except Exception:
        return float("inf")


def candidate_tuple_from_array(a):
    a = normalize_nonneg(a)
    n = len(a)
    c = conv_full(a)
    sc = score_from_c(c, n)
    return sc, a, n


def weights_and_phi(c, sigma):
    c = np.asarray(c, dtype=np.float64)
    M = float(c.max())
    if M <= 0.0 or sigma <= 0.0 or not np.isfinite(M) or not np.isfinite(sigma):
        w = np.ones_like(c, dtype=np.float64) / float(len(c))
        return M, float(c.mean()), w
    z = (c - M) / sigma
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros_like(c, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return M, M, w
    w = e / s
    phi = M + sigma * math.log(s)
    return M, float(phi), w


def grad_from_weights(w, a):
    a = normalize_nonneg(a)
    w = np.asarray(w, dtype=np.float64)
    n = len(a)
    if n == 1:
        return np.array([2.0 * float(w[0]) * float(a[0])])
    need = 3 * n - 2
    L = next_pow2(need)
    ah = np.zeros(L, dtype=np.float64)
    ah[:n] = a
    wh = np.zeros(L, dtype=np.float64)
    wh[:len(w)] = w[::-1]
    F = np.fft.rfft(ah) * np.fft.rfft(wh)
    out = np.fft.irfft(F, L)
    start = 2 * n - 2
    g = out[start : n - 2 : -1]
    g = 2.0 * g
    g = np.asarray(g, dtype=np.float64)
    return np.maximum(g, 0.0)


def interp_sequence(parent, new_n):
    parent = normalize_nonneg(parent)
    old_n = len(parent)
    new_n = int(max(1, new_n))
    if old_n == new_n:
        return parent.copy()
    if old_n == 1:
        return np.full(new_n, 1.0 / float(new_n), dtype=np.float64)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, parent)
    return normalize_nonneg(out)


def cosine_profile(n):
    n = int(max(1, n))
    x = np.linspace(0.0, 1.0, n)
    a = 0.25 + 0.75 * np.cos(np.pi * x) ** 2
    return normalize_nonneg(a)


def spectral_root_init(n, n_candidates=40, L_factor=4):
    n = int(max(1, n))
    if n <= 0:
        return np.ones(1, dtype=np.float64)

    L = next_pow2(max(int(L_factor) * n, 1024))
    target = np.zeros(L, dtype=np.float64)
    target[:max(1, 2 * n - 1)] = 1.0 / float(max(1, 2 * n - 1))

    C = np.fft.fft(target)
    mag = np.abs(C) + 1e-18
    phase = np.angle(C)
    root = np.sqrt(mag) * np.exp(0.5j * phase)

    half = L // 2
    A = np.zeros(L, dtype=np.complex128)
    A[0] = root[0]
    if half > 0:
        idx = np.arange(1, half)
        if idx.size > 0:
            A[idx] = root[idx]
            A[L - idx] = np.conj(root[idx])
        A[half] = root[half].real

    r = np.fft.ifft(A).real
    rs = float(r.sum())
    if abs(rs) > 1e-15 and np.isfinite(rs):
        r = r / rs

    best_a = None
    best_score = float("inf")
    base = np.arange(n, dtype=np.int64)

    for sign in (1.0, -1.0):
        rs2 = r * sign
        rd = np.concatenate((rs2, rs2))
        pos = np.maximum(rd, 0.0)
        neg = np.maximum(-rd, 0.0)
        cum_pos = np.concatenate(([0.0], np.cumsum(pos)))
        cum_neg = np.concatenate(([0.0], np.cumsum(neg)))
        window_pos = cum_pos[n:L + n] - cum_pos[:L]
        window_neg = cum_neg[n:L + n] - cum_neg[:L]
        heuristic = window_pos - 10.0 * window_neg
        order = np.argsort(heuristic)[::-1]
        per_sign = max(4, int(n_candidates) // 2)

        for s_idx in order[:per_sign]:
            s = int(s_idx)
            idxs = (s + base) % L
            w = np.maximum(rd[idxs], 0.0)
            ws = float(w.sum())
            if ws <= 1e-15:
                continue
            w = w / ws
            c = conv_full(w)
            sc = score_from_c(c, n)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = w.copy()

    if best_a is None:
        return np.ones(n, dtype=np.float64) / float(n)
    return best_a


def spectral_flat_update(a, n_iters=10, strength=0.35):
    a = normalize_nonneg(a)
    n = len(a)
    if n == 0:
        return a

    L = next_pow2(2 * n)
    target = np.zeros(L, dtype=np.float64)
    target[:max(1, 2 * n - 1)] = 1.0 / float(max(1, 2 * n - 1))
    T = np.fft.fft(target)
    target_mag = np.sqrt(np.abs(T) + 1e-30)
    if target_mag[0] > 1e-30:
        target_mag = target_mag / target_mag[0]

    best_a = a.copy()
    best_score, _, _ = candidate_tuple_from_array(a)

    for _ in range(int(max(1, n_iters))):
        A = np.fft.fft(a, L)
        cur = np.abs(A)
        ratio = target_mag / np.maximum(cur, 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        A_new = A * np.power(ratio, float(strength))
        a_new = np.fft.ifft(A_new).real
        a_new = np.maximum(a_new, 0.0)
        s = float(a_new.sum())
        if s <= 1e-15:
            break
        a_new = a_new / s
        blend = (1.0 - float(strength)) * a + float(strength) * a_new
        blend = normalize_nonneg(blend)

        sc, blend, _ = candidate_tuple_from_array(blend)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = blend.copy()
        a = blend

    return best_a


def symmetrize_sequence(a):
    a = normalize_nonneg(a)
    return normalize_nonneg(0.5 * (a + a[::-1]))


def power_transform_sequence(a, p):

    if p <= 0.0:

    b = a ** p

def build_active_set(c, w, n, K, cover):

    w = np.asarray(w, dtype=np.float64)

    top_c = np.argsort(c)[::-1][:K]
    top_w = np.argsort(w)[::-1][:K]

def solve_lp_direction_scipy(a, M, active, U):
    if not _HAVE_SCIPY:
        return None
    a = normalize_nonneg(a)
    n = len(a)
    active = np.asarray(active, dtype=np.int64).ravel()
    active = active[(active >= 0) & (active <= 2 * n - 2)]
    if active.size == 0:
        return None
    active = np.unique(active)

    row_parts = []
    col_parts = []
    data_parts = []

    for k in active:
        ki = int(k)
        js = max(0, ki - n + 1)
        je = min(n - 1, ki)
        if js > je:
            continue
        cols = np.arange(js, je + 1, dtype=np.int32)
        coeffs = a[ki - cols]
        if coeffs.size == 0 or float(np.max(coeffs)) <= 0.0:
            continue
        row_parts.append(np.full(cols.size, len(row_parts), dtype=np.int32))
        col_parts.append(cols)
        data_parts.append(np.asarray(coeffs, dtype=np.float64))

    if not row_parts:
        return None

    row = np.concatenate(row_parts)
    col = np.concatenate(col_parts)
    data = np.concatenate(data_parts)
    n_rows = len(row_parts)

    A = _sp_sparse.coo_matrix((data, (row, col)), shape=(n_rows, n)).tocsr()
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(n_rows, float(M) + 1e-12, dtype=np.float64)
    bounds = [(0.0, float(U))] * n

    try:
        res = _linprog(
            c_obj,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "output_flag": False}
        )
    except Exception:
        return None

    if res is None or getattr(res, "status", 1) != 0 or res.x is None:
        return None

    b = np.asarray(res.x, dtype=np.float64)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return None
    return b / s


def get_cvxpy_solver():
    global _cvxpy_solver
    if _cvxpy_solver is not None:
        return _cvxpy_solver
    if not _HAVE_CVXPY:
        return None

    for name in ["CLARABEL", "ECOS", "PDLP", "SCS", "ECOS_BB"]:
        try:
            solver = getattr(cp, name)
        except Exception:
            continue
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=solver, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                _cvxpy_solver = solver
                return solver
        except Exception:
            continue

    try:
        _cvxpy_solver = cp.ECOS
        return _cvxpy_solver
    except Exception:
        _cvxpy_solver = None
        return None


def solve_lp_direction_cvxpy(a, M, active, U):
    if not _HAVE_CVXPY:
        return None
    solver = get_cvxpy_solver()
    if solver is None:
        return None

    a = normalize_nonneg(a)
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [b_var <= float(U)]

    for k in active:
        ki = int(k)
        js = max(0, ki - n + 1)
        je = min(n - 1, ki)
        if js > je:
            continue
        cols = np.arange(js, je + 1)
        coeffs = np.asarray(a[ki - cols], dtype=np.float64)
        if coeffs.size == 0 or float(np.max(coeffs)) <= 0.0:
            continue
        constraints.append(coeffs @ b_var[cols] <= float(M) + 1e-12)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    b = np.asarray(b_var.value, dtype=np.float64)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return None
    return b / s


def solve_lp_direction(a, M, active, U):
    b = solve_lp_direction_scipy(a, M, active, U)
    if b is not None:
        return b
    return solve_lp_direction_cvxpy(a, M, active, U)


def smooth_ng_optimize(a, time_limit, sigma_start=0.5, sigma_end=0.02, alpha0=0.06, verbose=True):
    a = normalize_nonneg(a)
    n = len(a)
    if n == 0 or time_limit <= 0:
        return a, float("inf")

    t0 = time.monotonic()
    c = conv_full(a)
    score = score_from_c(c, n)
    best_a = a.copy()
    best_score = score

    alpha = float(alpha0)
    momentum = np.zeros(n, dtype=np.float64)
    stall = 0
    rng = np.random.default_rng((int(time.monotonic() * 1000) % 2147483647) + 1)
    last_print = t0

    while time.monotonic() - t0 < float(time_limit):
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(1e-9, float(time_limit)))
        if frac < 0.25:
            sf = float(sigma_start)
        elif frac < 0.50:
            sf = 0.20
        elif frac < 0.75:
            sf = 0.08
        else:
            sf = float(sigma_end)

        M_prov, phi, w = weights_and_phi(c, max(1e-15, sf * max(1e-15, float(c.max()))))
        if M_prov <= 0.0:
            break

        g = grad_from_weights(w, a)
        gmax = float(g.max())
        if not np.isfinite(gmax) or gmax <= 1e-30:
            break

        d = g / gmax
        momentum = 0.85 * momentum + d
        dmax = float(momentum.max())
        if dmax > 1e-30:
            d = momentum / dmax

        accepted = False
        sigma = max(1e-15, sf * max(1e-15, float(c.max())))

        for aa in (alpha, 0.5 * alpha, 0.25 * alpha):
            if aa <= 1e-12:
                continue
            trial = a * np.exp(-aa * d)
            trial = np.maximum(trial, 0.0)
            s = float(trial.sum())
            if s <= 1e-15:
                continue
            trial = trial / s
            ct = conv_full(trial)
            sc_t = score_from_c(ct, n)

                score = sc_t
                if sc_t < best_score - 1e-12:

                c = conv_full(a)

                alpha = float(alpha0)
                stall = 0

        if verbose and time.monotonic() - last_print > 20.0:
            print(f"    smooth-n: best={best_score:.8f} current={score:.8f} t={time.monotonic()-t0:.1f}s", flush=True)

            a_sp = spectral_flat_update(best_a, n_iters=18, strength=0.35)
            sc_sp, a_sp, _ = candidate_tuple_from_array(a_sp)
            if np.isfinite(sc_sp) and sc_sp < best_score:
                best_score = sc_sp
                best_a = a_sp
                print(f"Spectral polish: {best_score:.8f}", flush=True)

    rem = total_budget - (time.monotonic() - t_start)
    if rem > 180.0:
        n_best = len(best_a)
        if n_best <= 4096:
            cap = 750.0
        elif n_best <= 6144:
            cap = 550.0
        elif n_best <= 9000:
            cap = 420.0
        else:
            cap = 320.0
        t_smooth = min(cap, rem * 0.45)
        if t_smooth > 60.0:
            a_new, sc_new = smooth_ng_optimize(
                best_a, t_smooth,
                sigma_start=0.50, sigma_end=0.02, alpha0=0.06, verbose=True
            )
            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_a = a_new
                print(f"Smooth primary: {best_score:.8f}", flush=True)

    rem = total_budget - (time.monotonic() - t_start)
    if rem > 180.0 and (_HAVE_SCIPY or _HAVE_CVXPY):
        n_best = len(best_a)
        K = 450 if n_best <= 4096 else 260
        cap = 450.0 if n_best <= 5000 else 320.0
        t_lp = min(cap, rem * 0.35)
        if t_lp > 60.0:
            a_new, sc_new = lp_fw_optimize(
                best_a, t_lp,
                K=K, cover_count=220, sigma_factor=0.10, verbose=True
            )
            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_a = a_new
                print(f"LP polish: {best_score:.8f}", flush=True)

    rem = total_budget - (time.monotonic() - t_start)
    if rem > 150.0:
        n_best = len(best_a)
        cap = 350.0 if n_best <= 5000 else 250.0
        t_smooth2 = min(cap, rem * 0.40)
        if t_smooth2 > 60.0:
            a_new, sc_new = smooth_ng_optimize(
                best_a, t_smooth2,
                sigma_start=0.15, sigma_end=0.01, alpha0=0.03, verbose=False
            )
            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_a = a_new
                print(f"Smooth narrow: {best_score:.8f}", flush=True)

    rem = total_budget - (time.monotonic() - t_start)
    if len(top_indices) > 1 and rem > 250.0:
        sec_a = cand_arrays[top_indices[1]]
        sec_score = float(cand_scores[top_indices[1]])
        if np.isfinite(sec_score) and sec_score <= best_score + 5e-4:
            t_sec = min(260.0, rem * 0.35)
            if t_sec > 60.0:
                a_sec, sc_sec = smooth_ng_optimize(
                    sec_a, t_sec,
                    sigma_start=0.30, sigma_end=0.02, alpha0=0.05, verbose=False
                )
                if np.isfinite(sc_sec) and sc_sec < best_score:
                    best_score = sc_sec
                    best_a = a_sec
                    print(f"Secondary length smooth: {best_score:.8f}", flush=True)

            rem = total_budget - (time.monotonic() - t_start)
            if rem > 90.0 and (_HAVE_SCIPY or _HAVE_CVXPY):
                t_sec2 = min(90.0, rem * 0.25)
                if t_sec2 > 30.0:
                    a_sec2, sc_sec2 = lp_fw_optimize(
                        best_a, t_sec2, K=220, cover_count=160, sigma_factor=0.08, verbose=False
                    )
                    if np.isfinite(sc_sec2) and sc_sec2 < best_score:
                        best_score = sc_sec2
                        best_a = a_sec2
                        print(f"Secondary length LP: {best_score:.8f}", flush=True)

    rem = total_budget - (time.monotonic() - t_start)
    if rem > 60.0 and (_HAVE_SCIPY or _HAVE_CVXPY):
        t_final_lp = min(120.0, rem * 0.35)
        if t_final_lp > 30.0:
            a_new, sc_new = lp_fw_optimize(
                best_a, t_final_lp, K=200, cover_count=150, sigma_factor=0.08, verbose=False
            )
            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_a = a_new
                print(f"Final LP: {best_score:.8f}", flush=True)

    final_a = best_a
    final_score = best_score

    a_t = symmetrize_sequence(final_a)
    sc_t, a_t, _ = candidate_tuple_from_array(a_t)
    if np.isfinite(sc_t) and sc_t < final_score:
        final_score = sc_t
        final_a = a_t

    for p in (0.50, 0.70, 0.85, 1.15, 1.40, 1.80, 2.50):
        a_t = power_transform_sequence(final_a, p)
        sc_t, a_t, _ = candidate_tuple_from_array(a_t)
        if np.isfinite(sc_t) and sc_t < final_score:
            final_score = sc_t
            final_a = a_t

    try:
        n_f = len(final_a)
        a_root = spectral_root_init(n_f, n_candidates=8, L_factor=4)
        a_t = normalize_nonneg(0.5 * final_a + 0.5 * a_root)
        sc_t, a_t, _ = candidate_tuple_from_array(a_t)
        if np.isfinite(sc_t) and sc_t < final_score:
            final_score = sc_t
            final_a = a_t
    except Exception:
        pass

    final_a = normalize_nonneg(final_a)
    exact_score = evaluate_safe(final_a)
    if np.isfinite(exact_score):
        if exact_score < final_score:
            final_score = exact_score
    else:
        if np.isfinite(parent_score):
            final_a = normalize_nonneg(parent)
            final_score = parent_score
        else:
            final_a = np.ones(2048, dtype=np.float64)
            final_score, final_a, _ = candidate_tuple_from_array(final_a)

    final_a = np.asarray(final_a, dtype=np.float64)
    final_a = np.clip(final_a, 0.0, 1000.0)
    s = float(final_a.sum())
    if s <= 1e-15:
        final_a = np.ones(2048, dtype=np.float64)
        s = float(final_a.sum())
    final_a = final_a / s

    print(f"FINAL candidate: local_score={final_score:.8f} n={len(final_a)} time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_a]
