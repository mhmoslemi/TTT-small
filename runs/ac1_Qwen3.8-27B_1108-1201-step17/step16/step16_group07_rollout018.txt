def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize_to(a, n):
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s

def build_constraint_matrix(y, active_set, n):
    """Build the constraint matrix T where T[k_index, j] gives (y*b)[k] coefficient for b_j."""
    size = len(active_set)
    T = np.zeros((size, n))
    for idx, k in enumerate(active_set):
        k = int(k)
        # (y*b)[k] = sum_j y[j] * b[k-j] for valid j, k-j
        # = sum_j b[j] * y[k-j] for valid k-j
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            j_indices = np.arange(lo, hi)
            # b[j] coefficient is y[k-j]
            T[idx, lo:hi] = y[k - j_indices]
    return T

def lp_best_response(y, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_set, b >= 0."""
    n = len(y)
    T = build_constraint_matrix(y, active_set, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False, eps=1e-10)
    except Exception:
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    return b_val, None

def get_solver():

            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

def nesterov_lp(a0, time_limit, solver, verbose=False, seed=0, active_tol_start=0.1, active_tol_end=0.005):

    else:

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t_start = time.monotonic()
    n_iters = 0
    conv_size = 2 * n - 1
    stall_count = 0
    best_ergodic = a.copy()
    ergodic_a = a.copy()
    
    # Track ergodic average
    for t in range(1, 10000000):
        if time.monotonic() - t_start > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set: tolerance decreases over time
        progress = min(1.0, t / max(100, int(time_limit / 0.5)))
        tol = active_tol_start * (1 - progress) + active_tol_end * progress
        base_threshold = M_curr * (1.0 - tol)
        active_set = np.where(conv_y >= base_threshold)[0]
        
        # Always include top peaks
        n_top = max(10, n // 50)
        top_k = np.argsort(conv_y)[-n_top:]
        active_set = np.unique(np.concatenate([active_set, top_k]))
        
        # Cap for efficiency
        max_constraints = int(2.0 * n)
        if len(active_set) > max_constraints:
            # Keep the most important ones
            conv_at_active = conv_y[active_set]
            keep = np.argsort(conv_at_active)[-max_constraints:]
            active_set = active_set[keep]

        b = lp_best_response(y, M_curr, active_set, solver)
        if b is None:
            # Fallback: use all constraints
            active_set = np.arange(conv_size)
            b = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best (check both y and a and ergodic)
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Ergodic average (updated every iteration for speed)
        ergodic_a = ergodic_a + a_new / (t + 1)
        if t % 30 == 0:
            erg_score = fast_autocorr(ergodic_a).max()
            if erg_score < best_score:
                best_score = erg_score
                best_a = ergodic_a.copy()
                stall_count = 0

            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            ergodic_a = best_a.copy()
            stall_count = 0

            elapsed = time.monotonic() - t_start
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}, M={M_curr:.6f}")

    return best_a, best_score

        # And the second term: sum_k p[k] * a[j-k] = sum_k a[k] * p[j-k] = (a * p)[j] (convolution)
        
        # So grad[j] = cross_corr(p, a)[j] + conv(a, p)[j]
        
        # cross_corr(p, a)[j] = sum_k a[k] * p[k+j]
        # conv(a, p)[j] = sum_k a[k] * p[j-k]
        
        # Using FFT with padding to length >= 2*(2n-1) + (n-1) = 5n-3, let's use next power of 2.
        
        fft_len = 1 << (5 * n).bit_length()
        P = rfft(np.concatenate([p, np.zeros(fft_len - len(p))]), fft_len)
        A = rfft(np.concatenate([a, np.zeros(fft_len - n)]), fft_len)
        
        # cross_corr(p, a)[j] = sum_k a[k] p[k+j]
        # = ifft(conj(A) * P)[:n] ... let me verify
        # ifft(conj(A) * P)[j] = sum_k a[k] * p[k-j] ... no that's not right either.
        
        # Let me think clearly:
        # FFT: X = FFT(x), Y = FFT(y)
        # ifft(X * conj(Y)) = cross-correlation of x and y = sum_k x[k+l] * conj(y[k])
        # ifft(conj(X) * Y) = sum_k y[k+l] * conj(x[k])
        # ifft(X * Y) = convolution of x and y = sum_k x[k] * y[l-k]
        
        # I want: sum_k a[k] * p[k+j]
        # This is: (a ⋆ p)(-j) where (⋆) is cross-correlation
        # = ifft(conj(FFT(p)) * FFT(a)) at index... 
        
        # Ugh, let me just use scipy.signal or compute directly.
        # For n=1600, direct computation is O(n * (2n-1)) = O(2n^2) ≈ 5M ops, which is fine.
        
        conv_size_local = 2 * n - 1
        # First term: sum_k p[k] * a[k-j] for k-j in [0, n-1]
        # = sum_{k=j}^{j+2n-2} p[k] * a[k-j] (with a[k-j]=0 if k-j >= n)
        # = sum_{k=j}^{min(j+n-1, 2n-2)} p[k] * a[k-j]
        
        # Second term: sum_k p[k] * a[j-k] for j-k in [0, n-1]
        # = sum_{k=max(0,j-n+1)}^{min(j, 2n-2)} p[k] * a[j-k]
        
        # Combined: grad[j] = sum_{k} p[k] * (a[k-j] + a[j-k]) (with a[i]=0 for i out of range)
        
        # This is actually just the convolution of p with a "reflected" version...
        # Let me define r[i] = a[i] for i in [0,n), 0 otherwise.
        # grad[j] = sum_k p[k] * r[k-j] + sum_k p[k] * r[j-k]
        # = (p ⊛ r)(j) + (p * r)(j) where ⊛ is cross-correlation and * is convolution
        # But cross-correlation and convolution are related: (p ⊛ r)(j) = (p * r_rev)(j) where r_rev[i] = r[-i]
        
        # I'll just compute it directly using numpy's efficient operations.
        # grad[j] = dot(p[j:j+n], a) + dot(p[0:j+1], a[j:-1:-1]) ... this is O(n^2) total
        
        # For n=1600, this is 1600 * 1600 = 2.56M multiplications per gradient, which is fast.
        
        # But doing it in a loop is slow. Let me vectorize:
        # Create a Toeplitz-like matrix... no that's O(n^2) memory.
        
        # Actually, the simplest vectorized approach:
        # grad = np.zeros(n)
        # for j in range(n):
        #     # first term: sum_k p[k] * a[k-j]
        #     k_lo = max(0, j)
        #     k_hi = min(conv_size_local, j + n)
        #     if k_lo < k_hi:
        #         grad[j] += np.dot(p[k_lo:k_hi], a[k_lo-j:k_hi-j])
        #     # second term: sum_k p[k] * a[j-k]
        #     k_lo2 = max(0, j - (n-1))
        #     k_hi2 = min(conv_size_local, j + 1)
        #     if k_lo2 < k_hi2:
        #         grad[j] += np.dot(p[k_lo2:k_hi2], a[j-k_hi2+1:j-k_lo2+1])
        
        # This is O(n^2) but with numpy dot products it should be OK for n=1600.
        # Total ops: about n * n/2 * 2 = n^2 ≈ 2.5M per gradient. At 1GHz that's ~3ms per gradient.
        # If we do 10000 gradients in 1000 seconds, that's 100 per second, so 10ms per gradient budget. OK.
        
        # Actually, let me use a cleaner FFT-based approach.
        # (a*a)[k] = sum_i a[i] a[k-i]
        # sum_k p[k] (a*a)[k] = sum_k p[k] sum_i a[i] a[k-i] = sum_i a[i] sum_k p[k] a[k-i]
        # = sum_i a[i] * (p ⊛ a)(i)  [where (p ⊛ a)(i) = sum_k p[k] a[k-i]]
        
        # And (p ⊛ a)(i) = ifft(FFT(p) * conj(FFT(a)))[i] ... let me verify.
        # Standard: ifft(FFT(x) * conj(FFT(y)))[n] = sum_k x[k+n] y[k] (cross-correlation)
        # So ifft(FFT(p) * conj(FFT(a)))[i] = sum_k p[k+i] a[k]
        # That's sum_k p[k+i] a[k], which is (p ⊛ a)(i) where the correlation is defined as sum_k p[k+i] a[k].
        # But I want sum_k p[k] a[k-i] = sum_k p[k+i] a[k] (substituting k->k+i). Yes! Same thing.
        
        # So: grad = a * ifft(FFT(p_padded) * conj(FFT(a_padded)))[:n]
        # where p_padded has length >= 3n-2 (to avoid circular correlation artifacts)
        
        # Wait, I need to be careful. Let me use full FFT (not rfft) for correctness.
        
        # Actually, for the gradient, I showed:
        # grad[j] = sum_k p[k] * (a[k-j] + a[j-k])
        # First part: sum_k p[k] a[k-j] = (p ⊛ a)(j) = ifft(FFT(p) * conj(FFT(a)))[j]
        # Second part: sum_k p[k] a[j-k] = (p * a)(j) = ifft(FFT(p) * FFT(a))[j]
        
        # So grad = ifft(FFT(p) * conj(FFT(a)))[:n] + ifft(FFT(p) * FFT(a))[:n]
        
        # With appropriate zero-padding to length >= 3n-2.
        
        from scipy.fft import fft, ifft
        pad_len = 1 << (3 * n - 2).bit_length()
        p_pad = np.zeros(pad_len, dtype=np.complex128)
        p_pad[:conv_size_local] = p
        a_pad = np.zeros(pad_len, dtype=np.complex128)
        a_pad[:n] = a
        
        P_f = fft(p_pad)
        A_f = fft(a_pad)
        
        grad = np.real(ifft(P_f * np.conj(A_f))[:n]) + np.real(ifft(P_f * A_f)[:n])
        
        # Project gradient to tangent space of {sum(a) = norm, a >= 0}
        # Tangent space: sum of directions = 0
        grad_mean = np.mean(grad)
        grad_tangent = grad - grad_mean
        
        # Take a step
        a_trial = a + step_size * grad_tangent  # We want to DECREASE f, so go against gradient
        # Wait, f_μ is increasing in the direction of grad, so we want to go AGAINST grad
        a_trial = a - step_size * grad_tangent
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial
        
        score = fast_autocorr(a_trial).max()
        if score < best_score - 1e-14:
            a = a_trial
            best_score = score
            best_a = a_trial.copy()
            patience = 0
            n_iters += 1
            # Adaptive step size
            step_size = min(step_size * 1.01, 3.0 / np.sqrt(n))
        else:
            patience += 1
            step_size *= 0.95
            if step_size < 1e-12:
                break
        
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    SmoothGD iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}, mu={mu:.4f}")

    return best_a, best_score

def coordinate_descent_polish(a0, time_limit, verbose=False, seed=0):
    """Coordinate descent for final polish."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t_start = time.monotonic()
    size = 2 * n - 1
    
    n_passes = 0
    no_improve = 0
    max_no_improve = 3

    while time.monotonic() - t_start < time_limit:
        n_passes += 1
        improved = False
        conv = fast_autocorr(a).copy()
        
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t_start > time_limit:
                break
            
            aj = a[j]
            
            # Try a few candidate values
            typical = norm / n
            candidates = [0.0, aj * 0.5, aj * 1.5, typical * 0.5, typical, typical * 1.5, typical * 2.0]
            candidates = sorted(set([max(0.0, min(c, norm * 0.3)) for c in candidates]))
            
            best_x = aj
            best_obj = conv.max()
            
            for x in candidates:
                if abs(x - aj) < 1e-10:
                    continue
                delta = x - aj
                new_conv = conv.copy()
                lo = max(0, j)
                hi = min(size, j + n)
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta
                
                # Since we're keeping sum = norm, scaling factor is 1 (sum doesn't change)
                # Wait, it does change: new_sum = norm - aj + x
                new_sum = norm - aj + x
                if new_sum < 1e-15:
                    continue
                scale = norm / new_sum
                new_max = new_conv.max() * scale * scale
                
                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x
            
            if abs(best_x - aj) > 1e-10:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        conv = fast_autocorr(a)
                        improved = True
                    else:
                        a = best_a.copy()
                        conv = fast_autocorr(a)
        
        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

    return best_a, best_score
