    exponent -= exponent.max()  # numerical stability
    b = a * np.exp(exponent)
    s = b.sum()
    if s < 1e-300:

    return b / s

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def power_sum_objective(a, p):
    r = fast_autocorr(a)
    return (np.sum(r ** p)) ** (1.0 / p)


def power_sum_gradient(a, p):
    """Gradient of (sum_k r_k^p)^(1/p) w.r.t. a, where r = conv(a,a)."""
    n = len(a)
    r = fast_autocorr(a)
    r_p = np.sum(r ** p)
    if r_p < 1e-300:
        return np.zeros(n), r
    norm_factor = r_p ** (1.0 / p - 1.0)
    r_pm1 = r ** (p - 1)
    # Cross-correlation: v[j] = sum_{m=0}^{n-1} r_pm1[j+m] * a[m]
    N = 1 << (3 * n - 2).bit_length()
    R = np.fft.rfft(r_pm1, N)
    A = np.fft.rfft(a, N)
    v = np.fft.irfft(R * np.conj(A), N)[:n]
    grad = 2.0 * norm_factor * v
    return grad, r


def mirror_descent(a0, time_limit, p_schedule=(4, 8, 16, 32, 64, 128, 256, 512),
                   eta0=1.0, verbose=True, seed=42):
    """Mirror descent with power-sum smoothing and p-annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = power_sum_objective(a, p_schedule[-1])
    best_score_proxy = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    for p in p_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        p_start_time = time.monotonic()
        p_time_limit = min(time_limit - (p_start_time - t0), 180.0)

        eta = eta0 / max(1.0, np.sqrt(p))
        iters_since_improve = 0
        max_iters_no_improve = 50000
        p_obj_at_start = power_sum_objective(a, p)

        for it in range(1, 10000000):
            if time.monotonic() - t0 > time_limit:
                break
            if time.monotonic() - p_start_time > p_time_limit:
                break

            grad, r = power_sum_gradient(a, p)
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break

            # Mirror descent step: a_new = a * exp(-eta * grad / ||grad||) / Z
            scaled = eta * grad / gn
            scaled -= scaled.max()
            b = a * np.exp(-scaled)
            s = b.sum()
            if s < 1e-30:
                break
            b = b / s

            # Check improvement
            obj_new = (np.sum(fast_autocorr(b) ** p)) ** (1.0 / p)
            if obj_new < p_obj_at_start - 1e-12:
                a = b
                p_obj_at_start = obj_new
                iters_since_improve = 0
                # Reduce eta slightly on improvement (slower for stability)
                eta *= 0.999
            else:
                iters_since_improve += 1
                eta *= 0.95
                if eta < 1e-10:
                    break

            # Update best
            M = fast_autocorr(a).max()
            if M < best_score_proxy - 1e-14:
                best_score_proxy = M
                best_a = a.copy()
                if verbose and (it % 50000 == 0):
                    elapsed = time.monotonic() - t0
                    print(f"      MD p={p} it={it} ({elapsed:.0f}s): proxy={2*n*best_score_proxy:.6f}")

            if iters_since_improve > max_iters_no_improve:
                # Restart with perturbation
                a = norm_sum1(best_a + rng.normal(0, 0.002, n) * best_a)
                eta = eta0 / max(1.0, np.sqrt(p))
                iters_since_improve = 0
                p_obj_at_start = power_sum_objective(a, p)
                break

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"      MD p={p} done ({elapsed:.0f}s): proxy={2*n*best_score_proxy:.6f}")

    return best_a, 2.0 * n * best_score_proxy


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_m=50):
    """LP-based Frank-Wolfe with Nesterov acceleration and adaptive constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_mom = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Build constraint matrix using top constraints only for speed
        # Full matrix would be (2n-1) x n, use row slicing
        n_constraints = len(ac)
        b_var = cp.Variable(n, nonneg=True)

        # Use full Toeplitz matrix (for correctness)
        # T[k, j] = y[k-j] if 0 <= k-j < n, else 0
        T = np.zeros((n_constraints, n))
        for j in range(n):
            valid_k = np.arange(j, min(j + n, n_constraints))
            T[valid_k, j] = y[valid_k - j]

        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            eta = 1.0 / (t + 1)
            y = (1 - eta) * y + eta * a
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        beta_t = (t - 1.0) / (t + 2.0)
        if t > 1:
            x_mom = y_new + beta_t * (y_new - y)
            x_mom = np.maximum(x_mom, 0.0)
            sx = x_mom.sum()
            if sx < 1e-15:
                x_mom = y_new.copy()
            else:
                x_mom = x_mom / sx
        else:
            x_mom = y_new.copy()

        y = y_new

        for candidate in [y, x_mom]:
            mc = fast_autocorr(candidate).max()
            if mc < best_m - 1e-14:
                best_m = mc
                best_a = candidate.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, 2.0 * n * best_m


def coordinate_polish(a0, time_limit, verbose=True, seed=123):
    """Coordinate descent: move mass from peak contributors to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 150

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_idx = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False

        # Find peak contributors
        peak_contrib = np.zeros(n)
        for k in top_idx[:8]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            peak_contrib[idx] += a[idx] * a[j_idx]

        sources = np.argsort(peak_contrib)[::-1][:20]
        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            best_dst = -1
            best_delta = 0.0
            n_try = min(n, 300)
            candidates = np.random.choice(n, n_try, replace=False) if n > n_try else np.arange(n)

            for dst in candidates:
                if dst == src:
                    continue
                impact = 0.0
                for k in top_idx[:6]:
                    k_int = int(k)
                    d = k_int - dst
                    if 0 <= d < n:
                        impact += a[d]
                    d2 = dst - k_int
                    if 0 <= d2 < n:
                        impact += a[d2]
                score = -impact
                if score > best_delta:
                    best_delta = score
                    best_dst = dst

            if best_dst < 0:
                continue

            for frac in [0.001, 0.003, 0.008, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.001, 0.005)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, 2.0 * n * best_m


def interpolated_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def get_parent_sequence():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p.copy()
    except Exception:
        pass
    return None


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    rng = np.random.default_rng(2024)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    parent = get_parent_sequence()
    if parent is not None:
        p_norm = norm_sum1(parent)
        p_score = evaluate_safe(p_norm)
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
        best_seq = p_norm.copy()
        best_score = p_score
    else:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: Mirror Descent with p-annealing (main workhorse) ===
    rem = remaining_time()
    print(f"\n=== Phase 1: Mirror Descent n={n0} ===")
    t1 = min(rem * 0.35, 500.0)
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        p_sched = (4, 8, 16, 32, 64, 128, 256, 512)
        a_out, _ = mirror_descent(a_in, t1, p_schedule=p_sched, eta0=2.0, verbose=True, seed=42)
        try_update(a_out, f"md_n{n0}")

    # === Phase 2: LP Frank-Wolfe ===
    rem = remaining_time()
    print(f"\n=== Phase 2: LP-FW n={len(best_seq)} ===")
    t2 = min(rem * 0.22, 300.0)
    if t2 > 60 and len(best_seq) <= 3000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True)
        try_update(a_out, f"lpfw_n{len(best_seq)}")

    # === Phase 3: Mirror Descent round 2 (finer p) ===
    rem = remaining_time()
    print(f"\n=== Phase 3: MD round 2 n={len(best_seq)} ===")
    t3 = min(rem * 0.15, 200.0)
    if t3 > 40:
        a_in = norm_sum1(best_seq)
        p_sched_fine = (32, 64, 128, 256, 512, 1024)
        a_out, _ = mirror_descent(a_in, t3, p_schedule=p_sched_fine, eta0=0.5, verbose=True, seed=77)
        try_update(a_out, f"md2_n{len(best_seq)}")

    # === Phase 4: Coordinate polish ===
    rem = remaining_time()
    print(f"\n=== Phase 4: Coord n={len(best_seq)} ===")
    t4 = min(rem * 0.12, 150.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=True, seed=42)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 5: Try longer length ===
    rem = remaining_time()
    n1 = min(n0 + 400, 3000)
    if n1 > n0 and rem > 150:
        print(f"\n=== Phase 5: Scale to n={n1} ===")
        a_long = interpolated_sequence(best_seq, n1)
        a_in = norm_sum1(a_long)
        t5 = min(rem * 0.15, 200.0)
        if t5 > 60:
            a_out, _ = mirror_descent(a_in, t5, p_schedule=(8, 32, 128, 512), eta0=1.5, verbose=True, seed=99)
            try_update(a_out, f"md_n{n1}")
            rem = remaining_time()
            if rem > 40 and n1 <= 3000:
                t5b = min(rem * 0.3, 80.0)
                if t5b > 30:
                    a_out2, _ = lp_frank_wolfe(a_out, t5b, solver, verbose=False)
                    try_update(a_out2, f"lpfw_n{n1}")

    # === Phase 6: Final LP at best length ===
    rem = remaining_time()
    n_best = len(best_seq)
    if n_best <= 2800 and rem > 80:
        print(f"\n=== Phase 6: Final LP n={n_best} ===")
        t6 = min(rem * 0.25, 150.0)
        if t6 > 40:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Diverse restarts ===
    rem = remaining_time()
    if rem > 120:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Diverse inits n={n_best} ===")
        x = np.linspace(0, 1, n_best)
        inits = []
        # Bimodal
        bim = np.exp(-40 * (x - 0.25) ** 2) + np.exp(-40 * (x - 0.75) ** 2)
        inits.append(bim / bim.sum())
        # Wide Gaussian
        gaus = np.exp(-8 * (x - 0.5) ** 2)
        inits.append(gaus / gaus.sum())
        # Random smooth
        n_fourier = 20
        freqs = rng.integers(1, 50, n_fourier)
        phases = rng.uniform(0, 2 * np.pi, n_fourier)
        random_smooth = np.ones(n_best)
        for f_idx, ph in zip(freqs, phases):
            random_smooth += np.abs(np.cos(2 * np.pi * f_idx * x + ph)) ** 2
        inits.append(random_smooth / random_smooth.sum())
        # Blend with best
        blend = 0.5 * norm_sum1(best_seq) + 0.5 * inits[0]
        inits.append(norm_sum1(blend))

        t_per = min(rem * 0.18, 100.0)
        for idx, init in enumerate(inits[:4]):
            rem = remaining_time()
            if rem < 50:
                break
            t7 = min(t_per, rem - 30)
            if t7 > 30:
                a_out, _ = mirror_descent(init, t7, p_schedule=(8, 32, 128, 512),
                                          eta0=1.0, verbose=False, seed=200 + idx)
                try_update(a_out, f"diverse{idx}")
                # Quick coord polish
                rem = remaining_time()
                if rem > 30:
                    a_out2, _ = coordinate_polish(a_out, min(rem * 0.3, 30.0), verbose=False, seed=300 + idx)
                    try_update(a_out2, f"diverse{idx}_coord")

    # === Phase 8: Final coordinate polish ===
    rem = remaining_time()
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord n={n_best} ===")
    if rem > 40:
        t8 = min(rem * 0.7, 180.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, verbose=True, seed=314)
            try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
