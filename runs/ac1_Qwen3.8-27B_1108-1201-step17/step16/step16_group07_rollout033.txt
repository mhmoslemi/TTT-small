def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)
    
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    return b_val, None

def pairwise_local_search(a0, time_limit, verbose=False, seed=0):
    """Targeted pairwise optimization: transfer mass between pairs."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 50
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        conv = fast_autocorr(a)
        M = conv.max()
        
        # Find positions with high "cost" (contribute to peaks)
        # The cost of position j is approximately sum_k w[k] * a[k-j]
        # Just use the gradient as a proxy
        n_peaks = max(5, n // 40)
        top_peaks = np.argsort(conv)[-n_peaks:]
        
        # For each peak, the positions that "feed" it are j where a[k-j] is large
        # So positions j that are "close" (in index space) to peak positions are costly
        
        # Strategy: for each position j, compute its "importance" 
        # (how much reducing it would help)
        # importance[j] ~ sum over active peaks k of a[k-j]
        imp = np.zeros(n)
        for k in top_peaks:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                idx = k - np.arange(lo, hi)
                imp[lo:hi] += a[idx]
        
        # Try to move mass from high-importance positions to low-importance ones
        # Sort by importance
        order = np.argsort(imp)
        
        improved = False
        for _ in range(n // 2):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick a source (high imp) and target (low imp)
            src = order[rng.integers(n // 2, n)]
            dst = order[rng.integers(0, n // 2)]
            
            if src == dst:
                continue
            
            # Try transferring some mass from src to dst
            a_src = a[src]
            if a_src < 1e-10:
                continue
            
            # Try different transfer amounts
            for frac in [0.1, 0.25, 0.5, 0.75, 0.9]:
                delta = a_src * frac
                a[src] = a_src - delta
                a[dst] += delta
                s = a.sum()
                # sum is preserved, so no renormalization needed
                score = fast_autocorr(a).max()
                if score < best_score - 1e-12:
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[src] = a_src
                    a[dst] -= delta
            
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    PLS ({elapsed:.1f}s): best={best_score:.10f}, no_improve={no_improve}")

def coordinate_descent(a0, time_limit, verbose=False, seed=0):

            # So changing a[j] by delta changes conv[lo:hi] by 2*delta*a[k-j]
            lo = max(0, j)
            hi = min(size, j + n)
            
            candidates = [0.0, aj * 0.5, aj * 1.5, aj + norm/n * 0.5, aj - norm/n * 0.5]
            candidates = [max(0.0, c) for c in candidates]
            
            best_x = aj
            best_obj = conv.max()
            
            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                new_conv = conv.copy()
                delta = x - aj
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta
                
                new_max = new_conv.max()
                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x
            
            if abs(best_x - aj) > 1e-12:

                new_score = fast_autocorr(a).max()
                if new_score < best_score - 1e-13:
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                    conv = fast_autocorr(a)
                else:
                    a = best_a.copy()
                    conv = fast_autocorr(a)
        
        if not improved:
            no_improve += 1
    
    return best_a, best_score

        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=False, seed=0):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    n_iters = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        if active_set is None or t % 20 == 0:
            progress = min(1.0, t / 300.0)
            tol = 0.12 * (1.0 - progress) + 0.002 * progress
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            n_top = max(3, n // 100)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            if len(active_set) > int(0.8 * n):
                conv_at_active = conv_y[active_set]
                keep = np.argsort(conv_at_active)[-int(0.8 * n):]
                active_set = active_set[keep]

        b = lp_best_response(y, M_curr, active_set, solver)
        if b is None:
            active_set = np.argsort(conv_y)[-max(5, n // 50):]
            b = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.8)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        if stall_count > 80:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def pairwise_local_search(a0, time_limit, verbose=False, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve_rounds = 0
    max_no_improve = 30

    while time.monotonic() - t0 < time_limit and no_improve_rounds < max_no_improve:
        conv = fast_autocorr(a)
        M = conv.max()
        n_peaks = max(5, n // 40)
        top_peaks = np.argsort(conv)[-n_peaks:]

        imp = np.zeros(n)
        for k in top_peaks:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                idx = k - np.arange(lo, hi)
                imp[lo:hi] += a[idx]

        order = np.argsort(imp)
        improved = False
        n_trials = n

        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break

            src = order[rng.integers(n // 3, n)]
            dst = order[rng.integers(0, n // 3)]
            if src == dst:
                continue

            a_src = a[src]
            if a_src < 1e-10:
                continue

            for frac in [0.2, 0.5, 0.8]:
                delta = a_src * frac
                a[src] = a_src - delta
                a[dst] += delta
                score = fast_autocorr(a).max()
                if score < best_score - 1e-12:
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[src] = a_src
                    a[dst] -= delta

            if improved:
                break

        if not improved:
            no_improve_rounds += 1
        else:
            no_improve_rounds = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    PLS ({elapsed:.1f}s): best={best_score:.10f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=False, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    max_no_improve = 5

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        improved = False
        conv = fast_autocorr(a)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            lo = max(0, j)
            hi = min(size, j + n)

            typical = norm / n
            candidates = [0.0, aj * 0.5, aj * 1.5, max(0.0, aj - typical * 0.3), aj + typical * 0.3, typical]
            candidates = sorted(set([min(max(0.0, c), norm * 0.3) for c in candidates]))

            best_x = aj
            best_obj = conv.max()

            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                new_conv = conv.copy()
                delta = x - aj
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta

                new_max = new_conv.max()
                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                new_score = fast_autocorr(a).max()
                if new_score < best_score - 1e-13:
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                    conv = fast_autocorr(a)
                else:
                    a = best_a.copy()
                    conv = fast_autocorr(a)

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.1f}s): best={best_score:.10f}, no_improve={no_improve}")

    return best_a, best_score


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False


def propose_candidate():
    """
    Main search combining mirror descent, LP refinement,
    pairwise local search, and coordinate descent.
    """
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = 1770.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_cur = len(best_seq)
    print(f"Starting with n={n_cur}, score={best_score:.8f}")

    # Phase 1: Mirror Descent
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 550.0)

        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 3: Pairwise Local Search
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Pairwise Local Search, budget={t3:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = pairwise_local_search(a_norm, t3, verbose=True, seed=7)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "pls1", t_global)

    # Phase 4: Mirror Descent round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Mirror Descent r2, budget={t4:.0f}s ===")

        a_out, _ = mirror_descent(a_norm, t4, verbose=True, seed=999)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "md2", t_global)

    # Phase 5: LP round 2

    n_cur = len(best_seq)

        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "lp2", t_global)

    # Phase 6: Pairwise + CD final
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 60:
        t6a = min(remaining * 0.5, 120.0)
        t6b = remaining - t6a - 10
        if t6b < 20:
            t6a = remaining * 0.6
            t6b = remaining - t6a - 5

        if t6a > 20:
            print(f"\n=== Phase 6a: PLS final, budget={t6a:.0f}s ===")

        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
