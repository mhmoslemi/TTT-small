def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    return _global_solver


def fast_autocorr(a):
    """Compute self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgradient(a, ac, top_k, beta):
    """Compute soft subgradient focusing on top-K peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, top_k_init=10, top_k_max=100, beta_init=10.0, beta_min=2.0, c0=5.0, verbose=False):
    """Adaptive AMD with progressive refinement."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 3000
    cur_k = top_k_init
    cur_beta = beta_init

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, top_k=cur_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive parameters
        if stall % 1000 == 0:
            # Gradually increase top_k and decrease beta
            progress = min(1.0, stall / max_stall)
            cur_k = int(top_k_init + progress * (top_k_max - top_k_init))
            cur_beta = beta_init - progress * (beta_init - beta_min)

        if stall > max_stall:
            # Restart with structured perturbation
            stall = 0
            # Try different perturbation strategies
            strategy = rng.integers(4)
            if strategy == 0:
                # Gaussian noise
                pert = rng.normal(0, 0.03, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                # Multiplicative noise
                x_r = best_a * np.exp(rng.normal(0, 0.05, n))
            elif strategy == 2:
                # Blend with linear ramp
                ramp = np.arange(1, n+1, dtype=np.float64)
                ramp = norm_sum1(ramp)
                x_r = (1 - 0.3) * best_a + 0.3 * ramp
            else:
                # Blend with uniform
                x_r = best_a * 0.7 + 0.3 * np.ones(n) / n
            
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart ({elapsed:.0f}s): J={J:.6f}, k={cur_k}, beta={cur_beta:.1f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def peak_directed_cd(a0, time_limit, verbose=False):
    """Coordinate descent that specifically targets peak-contributing pairs."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top peaks
        n_peaks = min(10, len(ac))
        peak_indices = np.argsort(ac)[::-1][:n_peaks]
        
        improved = False
        
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find the pair with maximum contribution
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination: minimize increase in all peaks
            # Score each position by how much it would contribute to peaks
            scores = np.zeros(n)
            for k2 in peak_indices:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                # For position d, its contribution to peak k2 is a[k2-d] if valid
                for d in range(n):
                    d1 = k2i - d
                    if 0 <= d1 < n:
                        scores[d] += a[d1]
            
            # Mask out positions that are sources for any peak
            peak_positions = set()
            for k2 in peak_indices:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                for d in range(s2, e2+1):
                    peak_positions.add(d)
            
            # Prefer positions outside peak regions with low scores
            mask = np.ones(n, dtype=bool)
            for p in peak_positions:
                mask[p] = False
            
            if mask.any():
                candidates = np.where(mask)[0]
            else:
                candidates = np.arange(n)
            
            best_dst = candidates[np.argmin(scores[candidates])]
            
            if best_dst == src:
                continue
            
            # Try different fractions
            for frac in [0.02, 0.05, 0.1, 0.15, 0.25, 0.4]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PCD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        J = 2.0 * n * best_m
        print(f"    PCD final: J={J:.6f}")
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, top_K=200, verbose=False):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    prev_y = a.copy()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        # Update constraint matrix
        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select top-K most binding constraints
        constraint_vals = T_full @ y
        margins = M_curr - constraint_vals
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False, max_iter=5000)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        prev_y = y
        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_initialization(n, kind, seed=0):
    rng = np.random.default_rng(seed)
    if kind == 'linear':
        a = np.arange(1, n + 1, dtype=np.float64)
        return norm_sum1(a)
    elif kind == 'quadratic':
        a = np.arange(1, n + 1, dtype=np.float64) ** 2
        return norm_sum1(a)
    elif kind == 'sinusoidal':
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
        return norm_sum1(a)
    elif kind == 'bimodal':
        x = np.linspace(0, 1, n)
        a = np.exp(-100 * (x - 0.1) ** 2) + np.exp(-100 * (x - 0.9) ** 2) + 0.01
        return norm_sum1(a)
    elif kind == 'random_smooth':
        x = np.linspace(0, 1, n)
        a = np.ones(n) * 0.5
        for freq in [1, 2, 3, 4, 5, 7, 10]:
            coeff = rng.normal(0, 0.08)
            phase = rng.uniform(0, 2 * np.pi)
            a += coeff * np.sin(2 * np.pi * freq * x + phase)
        a = np.maximum(a, 0.0)
        return norm_sum1(a)
    elif kind == 'power_law':
        a = 1.0 / np.arange(1, n + 1, dtype=np.float64) ** 0.5
        return norm_sum1(a)
    else:
        return np.ones(n) / n

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_soft_subgradient(a, ac, top_k, beta):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def compute_smooth_subgradient(a, ac, temperature):
    n = len(a)
    shifted = ac - ac.max()
    if temperature < 1e-10:
        weights = np.zeros(len(ac))
        weights[np.argmax(ac)] = 1.0
    else:
        weights = np.exp(shifted / temperature)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.zeros(len(ac))
            weights[np.argmax(ac)] = 1.0
        else:
            weights = weights / w_sum
    grad = np.zeros(n)
    for k in range(len(ac)):
        if weights[k] < 1e-12:
            continue
        k_int = k
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * weights[k] * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, top_k_init=5, top_k_max=80, beta_init=12.0,
                 beta_min=2.0, c0=5.0, use_smooth=False, smooth_temp=0.1, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 2000
    cur_k = top_k_init
    cur_beta = beta_init
    best_stall = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        if use_smooth:
            temp = max(smooth_temp * (1.0 + 10.0 / (t + 1.0)), 0.01)
            g = compute_smooth_subgradient(y, ac, temperature=temp)
        else:
            g = compute_soft_subgradient(y, ac, top_k=cur_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
            best_stall = 0
        else:
            stall += 1
            best_stall += 1

        if stall % 500 == 0:
            progress = min(1.0, stall / max_stall)
            cur_k = int(top_k_init + progress * (top_k_max - top_k_init))
            cur_beta = beta_init - progress * (beta_init - beta_min)

        if stall > max_stall:
            stall = 0
            strategy = int(rng.integers(5))
            if strategy == 0:
                pert = rng.normal(0, 0.02, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                x_r = best_a * np.exp(rng.normal(0, 0.04, n))
            elif strategy == 2:
                ramp = np.arange(1, n + 1, dtype=np.float64)
                ramp = norm_sum1(ramp)
                frac = float(rng.uniform(0.2, 0.5))
                x_r = (1.0 - frac) * best_a + frac * ramp
            elif strategy == 3:
                x_r = best_a * (1.0 + 0.1 * rng.normal(0, 1, n))
            else:
                mid = n // 2
                x_r = best_a.copy()
                x_r[:mid] = x_r[:mid] * 0.8 + 0.2 * x_r[mid:2*mid]
                x_r[mid:2*mid] = x_r[mid:2*mid] * 0.8 + 0.2 * x_r[:mid]

            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()

            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart ({elapsed:.0f}s): J={J:.6f}, k={cur_k}, beta={cur_beta:.1f}")

        if verbose and t % 4000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def peak_directed_cd(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(88888)
    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_peaks = min(8, len(ac))
        peak_indices = np.argsort(ac)[::-1][:n_peaks]

        improved = False

        for pi in range(n_peaks):
            if time.monotonic() - t0 > time_limit:
                break
            k = int(peak_indices[pi])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            scores = np.zeros(n)
            for k2 in peak_indices:
                k2i = int(k2)
                for d in range(n):
                    d1 = k2i - d
                    if 0 <= d1 < n:
                        scores[d] += a[d1]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in peak_indices:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                peak_mask[s2:e2 + 1] = True

            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) == 0:
                non_peak = np.arange(n)

            best_dst = int(non_peak[np.argmin(scores[non_peak])])
            if best_dst == src:
                best_dst = int((src + 1) % n)

            found = False
            for frac in [0.03, 0.06, 0.10, 0.15, 0.25, 0.40]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and pi == 0 and stall > 100:
                j = int(rng.integers(n))
                if j != src:
                    frac = float(rng.uniform(0.01, 0.05))
                    amt = src_val * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = float(rng.uniform(0.005, 0.03))
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 80 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PCD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        J = 2.0 * n * best_m
        print(f"    PCD done: J={J:.6f}")
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, top_K=250, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    T_full = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_full[j:j + n, j] = y

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        T_full[:] = 0.0
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        constraint_vals = T_full @ y
        margins = M_curr - constraint_vals
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False, max_iter=3000)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 8 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_init_linear(n):
    a = np.arange(1, n + 1, dtype=np.float64)
    return norm_sum1(a)


def make_init_quadratic(n):
    a = np.arange(1, n + 1, dtype=np.float64) ** 2
    return norm_sum1(a)


def make_init_sinusoidal(n, seed=0):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    return norm_sum1(a)


def make_init_bimodal(n):
    x = np.linspace(0, 1, n)
    a = np.exp(-80.0 * (x - 0.08) ** 2) + np.exp(-80.0 * (x - 0.92) ** 2) + 0.01
    return norm_sum1(a)


def make_init_random_smooth(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in [1, 2, 3, 4, 5, 7, 11]:
        coeff = rng.normal(0, 0.07)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_init_power(n, p=0.5):
    a = np.arange(1, n + 1, dtype=np.float64) ** p
    return norm_sum1(a)


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def update_best(best_seq, best_score, candidate, label, t_global):
    if candidate is None or len(candidate) == 0:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score

            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")

            best_score = p_score

        pass

        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
