    return _global_solver


def fast_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_ab(a, b):
    """Full linear convolution of a with b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    B = np.fft.rfft(b.astype(np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_power_ramp(n, alpha):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** alpha
    return normalize(a)


def make_bimodal_ramp(n, frac=0.3):
    """Two ramps: one at left, one at right with different heights."""
    a = np.zeros(n)
    left = np.arange(1, int(n * frac) + 1) ** 1.5
    right = np.arange(1, n - int(n * frac)) ** 2.0
    a[:len(left)] = left
    a[n - len(right):] = right * 1.5
    return normalize(a)


def make_quadratic_tail(n, center=0.85, width=0.25):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / width) ** 2) + 0.05 * x
    return normalize(a)


def compute_soft_gradient(a, conv_val, beta=8.0, top_k=80):
    """
    Soft subgradient of max(conv(a,a)) w.r.t. a.
    d/da[j] of conv(a,a)[m] = 2*a[m-j] for max(0,m-n+1) <= j <= min(n-1,m), else 0.
    """
    n = len(a)
    k_max = min(top_k, len(conv_val))
    top_idx = np.argsort(conv_val)[::-1][:k_max]
    top_vals = np.maximum(conv_val[top_idx], 1e-30)
    w = top_vals ** beta
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / ws

    grad = np.zeros(n)
    for wi, m in zip(w, top_idx):
        m_int = int(m)
        i_start = max(0, m_int - n + 1)
        i_end = min(n - 1, m_int)
        if i_start > i_end:
            continue
        j_idx = np.arange(i_start, i_end + 1)
        grad[j_idx] += 2.0 * wi * a[m_int - j_idx]
    return grad


def gradient_optimize(a0, time_limit, beta=8.0, top_k=80, c0=3.0,
                      restart_every=6000, verbose=True):
    """Mirror descent with soft subgradient and adaptive restarts."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_conv = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_beta = beta
    cur_top_k = top_k

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        # Compute gradient
        cv = fast_conv(y)
        g = compute_soft_gradient(y, cv, top_k=cur_top_k, beta=cur_beta)

        # Mirror descent step (exponential family)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_conv - 1e-18:
            best_conv = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive restart
        if stall_count > restart_every:
            stall_count = 0
            # Try different perturbations
            pert_type = rng.integers(3)
            if pert_type == 0:
                pert = rng.normal(0, 0.03, n) * best_a
            elif pert_type == 1:
                # Smooth perturbation
                pert = np.zeros(n)
                center = int(rng.integers(n // 4, 3 * n // 4))
                width = int(n * rng.uniform(0.1, 0.3))
                idx = np.arange(max(0, center - width), min(n, center + width))
                pert[idx] = 0.05 * np.exp(-0.5 * ((np.arange(len(idx)) - len(idx) // 2) / (width // 3 + 1)) ** 2)
            else:
                # Shift the mass distribution
                pert = rng.normal(0, 0.02, n)
                # Apply a smooth filter
                kernel = np.ones(10) / 10
                pert = np.convolve(pert, kernel, mode='same')

            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_beta = max(cur_beta - 0.3, 3.0)
            cur_top_k = min(cur_top_k + 20, 200)
            gamma = max(gamma * 0.9, 1.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_conv
                print(f"    GD restart t={t} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.1f}, top_k={cur_top_k}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_conv
            print(f"    GD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_conv


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=150):
    """LP-based Frank-Wolfe with constraint selection."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_conv = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    prev_best = best_conv

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        cv = fast_conv(y)
        M_curr = cv.max()
        if M_curr < 1e-18:
            break

        # Build constraint matrix for y * b
        # conv(y, b)[m] = sum_i y[i] * b[m-i]
        # We need: for selected m, sum_i y[i]*b[m-i] <= M_curr
        # This is a linear constraint on b.
        
        # Select top-K most binding constraints
        K_sel = min(top_K, 2 * n - 1)
        # All constraints are binding at level M_curr, so select the ones
        # where the margin is smallest (i.e., closest to M_curr)
        # Actually all are <= M_curr by definition, so select the largest ones
        val = cv.copy()
        sel_idx = np.argsort(val)[::-1][:K_sel]

        # Build the constraint matrix A such that A @ b <= M_curr
        # A[m, j] = y[m - j] if 0 <= m-j < n, else 0
        A = np.zeros((K_sel, n))
        for i, m in enumerate(sel_idx):
            m_int = int(m)
            j_start = max(0, m_int - n + 1)
            j_end = min(n - 1, m_int)
            if j_start <= j_end:
                A[i, j_start:j_end + 1] = y[m_int - np.arange(j_start, j_end + 1)]

        b_var = cp.Variable(n, nonneg=True)
        b_sum = cp.sum(b_var)
        constraints = [A @ b_var <= M_curr, b_sum == 1]
        prob = cp.Problem(cp.Maximize(b_sum), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_s = b_val.sum()
        if b_s < 1e-15:
            continue
        g_dir = b_val / b_s

        # Frank-Wolfe step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new / y_new.sum()

        # Nesterov
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            x_new = x_new / x_new.sum()

        y = y_new
        x = x_new

        # Track best
        my = fast_conv(y).max()
        if my < best_conv:
            best_conv = my
            best_a = y.copy()
        mx = fast_conv(x).max()
        if mx < best_conv:
            best_conv = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_conv
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

        # Stall detection
        if best_conv == prev_best and t % 500 == 0 and t > 1000:
            # Perturb to escape
            rng_local = np.random.default_rng(t)
            y = y + 0.01 * rng_local.normal(0, 1, n) * y
            y = np.maximum(y, 0)
            y = y / y.sum()
            prev_best = best_conv

    return best_a, best_conv


def swap_optimize(a0, time_limit, verbose=True, seed=42):
    """Pairwise swap coordinate descent: move mass from one position to another."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_conv = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        cv = fast_conv(a)
        # Identify peak positions
        n_top = min(30, 2 * n - 1)
        top_positions = np.argsort(cv)[::-1][:n_top]

        improved = False

        # For each peak, try to reduce it by moving mass
        for m_peak in top_positions[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            m_int = int(m_peak)
            # Indices that contribute to this peak: j in [max(0,m-n+1), min(n-1,m)]
            j_start = max(0, m_int - n + 1)
            j_end = min(n - 1, m_int)
            if j_start > j_end:
                continue

            # Contribution at each j: a[j] * a[m-j]
            j_idx = np.arange(j_start, j_end + 1)
            contrib = a[j_idx] * a[m_int - j_idx]
            
            # Find the largest contributor
            max_contrib_idx = np.argmax(contrib)
            src = j_idx[max_contrib_idx]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Try moving a small fraction to a "quiet" position
            # Find positions with low total contribution to all peaks
            quiet_score = np.zeros(n)
            for m2 in top_positions[:10]:
                m2i = int(m2)
                s2 = max(0, m2i - n + 1)
                e2 = min(n - 1, m2i)
                idx2 = np.arange(s2, e2 + 1)
                # Contribution if we add mass at position d: a[m2-d] + a[d-m2] (if valid)
                for d in range(n):
                    c = 0.0
                    if 0 <= m2i - d < n:
                        c += a[m2i - d]
                    if 0 <= d - m2i < n:
                        c += a[d - m2i]
                    quiet_score[d] += c

            # Candidate destinations: low quiet_score, not the source
            quiet_score[src] = np.inf
            # Also avoid positions near the source in peak regions
            dst = int(np.argmin(quiet_score))
            if dst == src or quiet_score[dst] > 10 * quiet_score.min():
                dst = int(rng.integers(n))
                if dst == src:
                    dst = (dst + 1) % n

            # Try different fractions
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_conv - 1e-18:
                    a = a_trial
                    best_conv = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 != i2 and a[i1] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i1] * frac
                a[i1] -= amt
                a[i2] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_conv:
                    best_conv = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_conv
            print(f"    Swap ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_conv
    if verbose:
        print(f"    Swap final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_conv


def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length, preserving the shape."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    s = a_interp.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / s


def propose_candidate(budget_s=1800.0):
    """Main search: multi-phase optimization for minimizing J = 2n*max(conv)/(sum)^2."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize(parent)
            best_score = p_score
    except Exception:
        pass

    # === Phase 1: Multi-initialization exploration ===
    print(f"\n=== Phase 1: Multi-init exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)

    # Build candidate initializations at several lengths
    inits = []
    labels = []
    lengths_to_try = [3200, 4000, 5000]

    if parent is not None:
        inits.append(normalize(parent))
        labels.append(f"parent_n{len(parent)}")

    for n in lengths_to_try:
        # Power ramps with various exponents
        for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
            inits.append(make_power_ramp(n, alpha))
            labels.append(f"ramp_a{alpha}_n{n}")
        # Bimodal
        inits.append(make_bimodal_ramp(n))
        labels.append(f"bimodal_n{n}")
        # Quadratic tail
        inits.append(make_quadratic_tail(n))
        labels.append(f"qtail_n{n}")

    # Evaluate all initializations quickly
    init_scores = []
    for a, lbl in zip(inits, labels):
        cv = fast_conv(a)
        sc = 2.0 * len(a) * cv.max()
        init_scores.append(sc)
        print(f"    {lbl}: J={sc:.6f}")

    # Sort by score and pick top ones for optimization
    sorted_idx = np.argsort(init_scores)
    
    # Allocate time: top initializations get more time
    n_top = min(8, len(inits))
    top_inits = [inits[i] for i in sorted_idx[:n_top]]
    top_labels = [labels[i] for i in sorted_idx[:n_top]]

    # Phase 1a: Quick gradient optimization on each top init
    t_per_quick = max(20.0, remaining * 0.25 / n_top)
    for idx, (a_init, lbl) in enumerate(zip(top_inits, top_labels)):
        elapsed_so_far = time.monotonic() - t_global
        if elapsed_so_far > time_budget - 60:
            break
        t_avail = min(t_per_quick, time_budget - 60 - (time.monotic() - t_global) if False else time_budget - 60 - elapsed_so_far)
        if t_avail < 15:
            break
        print(f"  Quick GD '{lbl}' (t={t_avail:.0f}s)...")
        a_out, _ = gradient_optimize(a_init, t_avail, beta=8.0, top_k=60, c0=3.0,
                                      restart_every=4000, verbose=False)
        try_update(a_out, f"gd_{lbl}")

    # Phase 1b: Thorough gradient optimization on the best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1b: Thorough GD on best (n={len(best_seq)}) ===")
    t1b = min(remaining * 0.20, 300.0)
    if t1b > 30:
        a_in = normalize(best_seq)
        a_out, _ = gradient_optimize(a_in, t1b, beta=10.0, top_k=100, c0=4.0,
                                      restart_every=8000, verbose=True)
        try_update(a_out, f"gd_thorough_n{len(a_in)}")

    # === Phase 2: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30 and n_best <= 5000:
        a_in = normalize(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, top_K=100)
        try_update(a_out, f"lp_n{n_best}")

    # === Phase 3: More gradient optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: GD round 2 (n={n_best}) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_in = normalize(best_seq)
        # Different hyperparameters
        a_out, _ = gradient_optimize(a_in, t3, beta=6.0, top_k=50, c0=5.0,
                                      restart_every=3000, verbose=False)
        try_update(a_out, f"gd2_n{n_best}")

    # === Phase 4: Swap optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Swap optimization (n={n_best}) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 15:
        a_in = normalize(best_seq)
        a_out, _ = swap_optimize(a_in, t4, verbose=True, seed=777)
        try_update(a_out, f"swap_n{n_best}")

    # === Phase 5: Try other lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    alt_lengths = [l for l in [3500, 4500, 5000, 6000] if l != n_best and l <= 6000]
    
    for new_n in alt_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n=== Phase 5: n={new_n} ===")
        a_long = interpolate_to_length(best_seq, new_n)
        
        # Also try a fresh power ramp at this length
        a_ramp = make_power_ramp(new_n, 1.5)
        
        # Pick the better starting point
        cv1 = fast_conv(a_long).max()
        cv2 = fast_conv(a_ramp).max()
        if cv1 < cv2:
            a_start = a_long
            lbl_start = "interp"
        else:
            a_start = a_ramp
            lbl_start = "ramp"
        
        t5 = min(remaining * 0.30, 200.0)
        if t5 > 30:
            a_out, _ = gradient_optimize(a_start, t5, beta=8.0, top_k=80, c0=3.5,
                                          restart_every=5000, verbose=False)
            try_update(a_out, f"gd_{lbl_start}_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and new_n <= 5000:
            t5b = min(remaining * 0.15, 100.0)
            if t5b > 20:
                a_in = normalize(best_seq)
                a_out, _ = lp_frank_wolfe(a_in, t5b, solver, verbose=False, top_K=80)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 6: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 5000:
        t6 = min(remaining * 0.30, 180.0)
        if t6 > 20:
            a_in = normalize(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True, top_K=200)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final swap polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final swap (n={n_best}) ===")
    if remaining > 20:
        t7 = min(remaining * 0.40, 120.0)
        if t7 > 10:
            a_in = normalize(best_seq)
            a_out, _ = swap_optimize(a_in, t7, verbose=True, seed=999)
            try_update(a_out, f"swap_final_n{n_best}")

    # === Phase 8: One final gradient pass with fine-tuned params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 8: Final GD (n={n_best}) ===")
        t8 = min(remaining * 0.5, 60.0)
        if t8 > 10:
            a_in = normalize(best_seq)
            a_out, _ = gradient_optimize(a_in, t8, beta=12.0, top_k=50, c0=2.0,
                                          restart_every=2000, verbose=False)
            try_update(a_out, f"gd_final_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        n_fb = 3200
        best_seq = make_power_ramp(n_fb, 1.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    # Safety: ensure we return within time
    elapsed = time.monotonic() - t_global
    if elapsed > budget_s - 30:
        print(f"Warning: close to time limit ({elapsed:.0f}s/{budget_s:.0f}s)")

    return [float(x) for x in best_seq]
