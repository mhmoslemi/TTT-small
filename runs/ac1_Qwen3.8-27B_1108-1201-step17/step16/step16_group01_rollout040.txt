def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize_nonneg(a):

    s = a.sum()
    if s < 1e-15:

def structured_init(n, seed=42):

def compute_subgradient(a, ac, threshold_frac=0.99):
    """
    Compute the subgradient of max(ac) with respect to a.
    Uses all lags where ac[k] >= threshold_frac * max(ac).
    Returns (gradient, number_of_active_lags).
    """
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n), 0
    
    active = np.where(ac >= threshold_frac * M)[0]
    if len(active) == 0:
        active = np.array([int(ac.argmax())])
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += a[j_idx]
    
    return 2.0 * g, len(active)

def bb_step_size(a_prev, a_curr, g_prev, g_curr, g_obj_prev, g_obj_curr):
    """
    Barzilai-Borwein step size.
    Returns a positive step size.
    """
    s = a_curr - a_prev
    y = g_curr - g_prev
    
    s_norm_sq = np.dot(s, s)
    y_norm_sq = np.dot(y, y)
    
    if s_norm_sq < 1e-30 or y_norm_sq < 1e-30:
        return 1e-8
    
    sy = np.dot(s, y)
    if abs(sy) < 1e-30:
        return 1e-8
    
    # BB step sizes
    gamma_s = s_norm_sq / sy
    gamma_y = sy / y_norm_sq
    
    # Use the one with positive numerator/denominator
    gamma = 1e-8
    if gamma_s > 0:
        gamma = gamma_s
    elif gamma_y > 0:
        gamma = gamma_y
    else:
        gamma = 1e-8
    
    # Clip to reasonable range
    gamma = np.clip(gamma, 1e-10, 1.0)
    
    return gamma

def accelerated_descent(a0, time_limit, c0=3.0, threshold_frac=0.99, 
                       use_bb=True, verbose=True, rng_seed=42):
    """
    Accelerated gradient descent with:
    - Exact active-set subgradient

    - Nesterov momentum

    ac = fast_autocorr(a)
    best_a = a.copy()

    x = a.copy()
    x_prev = a.copy()
    g_prev = None
    a_prev_for_bb = None
    
    stall_count = 0
    max_stall = 3000
    current_threshold = threshold_frac
    c_current = c0
    eta = c0 * np.sqrt(n)
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        # Compute autocorrelation and subgradient
        ac = fast_autocorr(y)
        g, n_active = compute_subgradient(y, ac, threshold_frac=current_threshold)

            eta_new = bb_step_size(a_prev_for_bb, y, g_prev, g, 0, 0)
            # Damp the BB step size for stability
            eta = 0.5 * eta + 0.5 * eta_new
        else:

        a_prev_for_bb = y.copy()
        g_prev = g.copy()

def peak_shaving(a0, time_limit, verbose=True, rng_seed=99):
    """
    Peak shaving: iteratively identify the peak lag and try to reduce it
    by moving mass from high-contributing positions to low-contributing ones.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))
        
        # Find positions that contribute to the peak
        k_int = peak_k
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        contrib = a[idx] * a[j_idx]
        
        if contrib.max() < 1e-20:
            break
        
        # Find the top contributors to the peak
        top_contrib_idx = np.argsort(contrib)[::-1][:min(20, len(idx))]
        
        improved = False
        
        for tc in top_contrib_idx:
            if time.monotonic() - t0 > time_limit:
                break
            
            src_i = idx[tc]
            src_j = j_idx[tc]
            
            # We want to reduce a[src_i] or a[src_j]
            # Move mass to a position that doesn't contribute to the peak
            hot_set = set(idx.tolist())
            hot_set.update(j_idx.tolist())
            
            cold_positions = [i for i in range(n) if i not in hot_set]
            if len(cold_positions) == 0:
                cold_positions = list(range(n))
            
            # Pick a cold position with small value
            if len(cold_positions) > 50:
                cold_positions = list(rng.choice(cold_positions, size=50, replace=False))
            
            sorted_cold = sorted(cold_positions, key=lambda i: a[i])[:20]
            
            for dst in sorted_cold:
                if dst == src_i or dst == src_j:
                    continue
                
                # Try to move mass from src_i to dst
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src_i] * frac
                    if amt < 1e-15:
                        continue
                    
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random escape
            n_perturb = max(5, n // 50)
            for _ in range(n_perturb):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            break
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0)
                            if a.sum() > 1e-15:
                                a = a / a.sum()
        
        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            peak_k = int(np.argmax(fast_autocorr(a)))
            print(f"    PS ({elapsed:.1f}s): J={J:.6f}, stall={stall}, peak_k={peak_k}")
    
    return best_a, best_m

def coordinate_descent_v2(a0, time_limit, verbose=True, rng_seed=123):
    """

    - Use binary search on transfer amount
    - Try both directions (increase and decrease)

    n = len(a0)

            break
        
        # Find top peak lags
        top_lags = np.argsort(ac)[::-1][:min(10, len(ac))]
        
        # Compute peak sensitivity for each position
        peak_sens = np.zeros(n)
        in_peak_count = np.zeros(n)

            i_end = min(n, k_int + 1)

def spectral_postprocess(a0, time_limit, verbose=True, n_iters=100):
    """
    Try to flatten the power spectrum to reduce autocorrelation peaks.
    Works by: in frequency domain, try to make |FFT(a)|^2 more uniform,
    then project back to non-negative time domain.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # FFT size
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute power spectrum
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # Try to flatten: scale power by 1/sqrt(power) to make it uniform
        # But we need to maintain the DC component
        target_power = np.ones_like(power)
        target_power[0] = power[0]  # Keep DC
        
        # Interpolate between current and target
        alpha = 0.1
        new_power = (1 - alpha) * power + alpha * target_power
        
        # Compute new amplitude (keep phase)
        phase = A / np.abs(A + 1e-30)
        new_A = np.sqrt(new_power) * phase
        
        # Inverse FFT
        a_new = np.fft.irfft(new_A, fft_size)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller alpha
            alpha = 0.01
            new_power = (1 - alpha) * power + alpha * target_power
            new_A = np.sqrt(new_power) * phase
            a_new = np.fft.irfft(new_A, fft_size)[:n]
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            M_new = fast_autocorr(a_new).max()
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = a_new.copy()
                a = a_new
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spec ({elapsed:.1f}s): J={J:.6f}, iter={it}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Structured diverse initializations (sinusoidal, sparse, random)
2. Accelerated gradient descent with Barzilai-Borwein adaptive step sizes
3. Peak shaving: targeted mass transfer to reduce specific autocorrelation peaks
4. Spectral domain post-processing to flatten power spectrum
5. Binary-search coordinate descent polish
6. Multi-length exploration with diverse restarts
Key innovations: BB step sizes for faster convergence, spectral flattening,
and targeted peak reduction beyond generic subgradient methods.

import numpy as np

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        result = evaluate_sequence(a_list)
        return float(result)
    except Exception:
        return float('inf')


def normalize_nonneg(a):
    """Clamp to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def generate_diverse_inits(n, seed=42):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    inits = []

    # Type 1: sinusoidal mixture
    a1 = np.ones(n)
    for _ in range(6):
        freq = rng.integers(1, max(2, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.25)
        a1 += amp * np.cos(2 * np.pi * freq * x / n + phase)
    a1 = np.maximum(a1, 0)
    a1 = a1 / a1.sum()
    inits.append(a1)

    # Type 2: random with smoothing
    a2 = rng.random(n)
    kernel = np.ones(7) / 7
    a2 = np.convolve(a2, kernel, mode='same')
    a2 = np.maximum(a2, 0)

    a3 = np.ones(n) * 0.1

    peak_positions = rng.choice(n, size=n_peaks, replace=False)
    for p in peak_positions:
        a3[p] += rng.uniform(0.5, 2.0)
    a3 = np.maximum(a3, 0)
    a3 = a3 / a3.sum()

    inits.append(a4)

    # Type 5: cosine window (tapered)

    a5 = np.maximum(a5, 0)
    a5 = a5 / a5.sum()
    inits.append(a5)

def compute_subgradient(a, ac, threshold_frac=0.99):
    """
    Compute subgradient of max(ac) w.r.t. a using active set.
    Returns (gradient, n_active).
    """
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n), 0

    active = np.where(ac >= threshold_frac * M)[0]
    if len(active) == 0:
        active = np.array([int(ac.argmax())])

    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += a[j_idx]

    return 2.0 * g, len(active)


def accelerated_descent(a0, time_limit, c0=3.0, threshold_frac=0.99,

    best_m = ac.max()

    x_prev = a.copy()
    g_prev = None
    x_prev_bb = None

    max_stall = 3000
    current_threshold = threshold_frac
    c_current = c0
    eta = c0 * np.sqrt(n)

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        # Compute autocorrelation and subgradient
        ac = fast_autocorr(y)
        g, n_active = compute_subgradient(y, ac, threshold_frac=current_threshold)

        # Step size: BB or fixed schedule
        if use_bb and t > 3 and g_prev is not None and x_prev_bb is not None:
            s_vec = y - x_prev_bb
            y_vec = g - g_prev
            s_sq = np.dot(s_vec, s_vec)

            if s_sq > 1e-25 and abs(s_y) > 1e-25:

            elif y_sq > 1e-25 and abs(s_y) > 1e-25:

                eta = 0.3 * eta + 0.7 * gamma_bb
            else:
                eta = c_current / (np.sqrt(t) + 1.0) * np.sqrt(n)
        else:
            eta = c_current / (np.sqrt(t) + 1.0) * np.sqrt(n)

        x_prev_bb = y.copy()
        g_prev = g.copy()

        # Gradient step with projection
        x_new = y - eta * g
        x_new = np.maximum(x_new, 0.0)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new

        x_prev = x
        x = x_new

        if M_new < best_m - 1e-15:

def spectral_flatten(a0, time_limit, verbose=True, n_iters=300):
    """
    Flatten the power spectrum to reduce autocorrelation peaks.
    Alternates between spectral shaping and non-negativity projection.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        phase = A / (np.abs(A) + 1e-30)

        # Target: flatten power spectrum (keep DC)
        target_power = np.ones_like(power)
        target_power[0] = power[0]

        # Adaptive alpha: start aggressive, decrease
        alpha = max(0.005, 0.15 * (1.0 - it / n_iters))

        new_power = (1.0 - alpha) * power + alpha * target_power
        new_power = np.maximum(new_power, 0)
        new_A = np.sqrt(new_power) * phase

        a_new = np.fft.irfft(new_A, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            alpha2 = alpha * 0.1
            new_power2 = (1.0 - alpha2) * power + alpha2 * target_power
            new_power2 = np.maximum(new_power2, 0)
            new_A2 = np.sqrt(new_power2) * phase
            a_new2 = np.fft.irfft(new_A2, fft_size)[:n]
            a_new2 = np.maximum(a_new2, 0)
            s2 = a_new2.sum()
            if s2 < 1e-15:
                continue
            a_new2 = a_new2 / s2
            M_new2 = fast_autocorr(a_new2).max()
            if M_new2 < best_m - 1e-15:
                best_m = M_new2
                best_a = a_new2.copy()
                a = a_new2

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spec ({elapsed:.1f}s): J={J:.6f}, iter={it}")

    return best_a, best_m


def coordinate_descent_v2(a0, time_limit, verbose=True, rng_seed=123):
    """
    Coordinate descent with binary search on transfer amount.
    Focuses on peak-sensitive positions.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 250

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < 1e-25:
            break

        top_lags = np.argsort(ac)[::-1][:min(8, len(ac))]

        # Compute peak sensitivity
        peak_sens = np.zeros(n)
        for k in top_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            if len(idx) > 0:
                peak_sens[idx] += a[j_idx]

        hot_order = np.argsort(peak_sens)[::-1]
        cold_order = np.argsort(peak_sens)

        improved = False
        n_hot = min(len(hot_order), 35)

        for src in hot_order[:n_hot]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            n_cold = min(len(cold_order), 35)
            for dst in cold_order[:n_cold]:
                if dst == src or a[dst] > a[src]:
                    continue

                # Binary search on transfer amount
                lo = 0.0
                hi = a[src]
                found = False
                for _ in range(7):
                    mid = (lo + hi) / 2.0
                    if mid < 1e-18:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= mid
                    a_trial[dst] += mid
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        break
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        lo = mid
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                    else:
                        hi = mid

                if found:
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Escape
            if stall % 15 == 0:
                n_perturb = max(3, n // 80)
                for _ in range(n_perturb):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.02, 0.08)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0)
                        if a.sum() > 1e-15:
                            a = a / a.sum()
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True
                                break
                            else:
                                a[i] += amt
                                a[j] -= amt
                                a = np.maximum(a, 0)
                                if a.sum() > 1e-15:
                                    a = a / a.sum()

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD2 ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Uses diverse initializations, accelerated descent with BB steps,
    peak shaving, spectral flattening, and coordinate descent polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')
    best_tracker = {'seq': None, 'score': float('inf')}

    def try_update(a, label):
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_tracker['score']:
            best_tracker['score'] = sc
            best_tracker['seq'] = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s}")

    # Load parent
    n_base = 2400
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            n_base = len(p)
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={n_base}, score={p_score:.6f}")
            best_tracker['seq'] = parent.copy()
            best_tracker['score'] = p_score
    except Exception:
        pass

    if best_tracker['seq'] is None:
        x = np.linspace(0, 1, n_base)
        init_seq = np.cos(np.pi * x / n_base) ** 2 + 0.05
        init_seq = init_seq / init_seq.sum()
        best_tracker['seq'] = init_seq
        best_tracker['score'] = evaluate_safe(init_seq)
        print(f"Default init: n={n_base}, score={best_tracker['score']:.6f}")

    # === Phase 1: Quick AMD from parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_tracker['seq'])
    print(f"\n=== Phase 1: AMD from parent (n={n_cur}) ===")
    t1 = min(remaining * 0.12, 180.0)
    if t1 > 20:
        a_out, _ = accelerated_descent(best_tracker['seq'], t1, c0=3.0,
                                       threshold_frac=0.99, use_bb=True,
                                       verbose=True, rng_seed=1)
        try_update(a_out, f"amd1_n{n_cur}")

    # === Phase 2: Main AMD (longer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_tracker['seq'])
    print(f"\n=== Phase 2: Main AMD (n={n_cur}) ===")
    t2 = min(remaining * 0.25, 420.0)
    if t2 > 30:
        a_out, _ = accelerated_descent(best_tracker['seq'], t2, c0=4.0,
                                       threshold_frac=0.995, use_bb=True,
                                       verbose=True, rng_seed=2)
        try_update(a_out, f"amd2_n{n_cur}")

    # === Phase 3: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_tracker['seq'])
    print(f"\n=== Phase 3: Peak shaving (n={n_cur}) ===")
    t3 = min(remaining * 0.18, 300.0)
    if t3 > 20:
        a_out, _ = peak_shaving(best_tracker['seq'], t3, verbose=True, rng_seed=3)
        try_update(a_out, f"ps_n{n_cur}")

    # === Phase 4: Spectral flattening ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_tracker['seq'])
    print(f"\n=== Phase 4: Spectral (n={n_cur}) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        a_out, _ = spectral_flatten(best_tracker['seq'], t4, verbose=True, n_iters=400)
        try_update(a_out, f"spec_n{n_cur}")

    # === Phase 5: AMD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_tracker['seq'])
    print(f"\n=== Phase 5: AMD polish (n={n_cur}) ===")
    t5 = min(remaining * 0.18, 280.0)
    if t5 > 20:
        a_out, _ = accelerated_descent(best_tracker['seq'], t5, c0=2.0,
                                       threshold_frac=0.998, use_bb=True,
                                       verbose=True, rng_seed=5)
        try_update(a_out, f"amd5_n{n_cur}")

    # === Phase 6: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_tracker['seq'])
    print(f"\n=== Phase 6: Coord descent (n={n_cur}) ===")
    t6 = min(remaining * 0.15, 220.0)
    if t6 > 15:
        a_out, _ = coordinate_descent_v2(best_tracker['seq'], t6, verbose=True, rng_seed=6)
        try_update(a_out, f"cd_n{n_cur}")

    # === Phase 7: Diverse restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 7: Diverse restarts ===")
        inits = generate_diverse_inits(len(best_tracker['seq']), seed=77)
        for i, init in enumerate(inits[:3]):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t7 = min(remaining * 0.30, 120.0)
            if t7 > 25:
                print(f"  Restart {i+1}")
                a_out, _ = accelerated_descent(init, t7, c0=3.5,
                                               threshold_frac=0.99, use_bb=True,
                                               verbose=False, rng_seed=10 + i)
                try_update(a_out, f"restart{i+1}")

    # === Phase 8: Multi-length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_tracker['seq'])
    if remaining > 80:
        print(f"\n=== Phase 8: Multi-length ===")
        for n_try in [n_cur + 300, n_cur + 500, n_cur - 300]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            if n_try < 600 or n_try > 4000:
                continue
            a_interp = interpolate_sequence(best_tracker['seq'], n_try)
            t8 = min(remaining * 0.35, 100.0)
            if t8 > 20:
                print(f"  n={n_try}")
                a_out, _ = accelerated_descent(a_interp, t8, c0=3.0,
                                               threshold_frac=0.99, use_bb=True,
                                               verbose=False, rng_seed=20)
                try_update(a_out, f"ml_n{n_try}")

    # === Phase 9: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_tracker['seq'])
    if remaining > 30:
        print(f"\n=== Phase 9: Final polish (n={n_cur}) ===")
        t9 = min(remaining * 0.40, 100.0)
        if t9 > 15:
            a_out, _ = accelerated_descent(best_tracker['seq'], t9, c0=1.5,
                                           threshold_frac=0.999, use_bb=True,
                                           verbose=False, rng_seed=30)
            try_update(a_out, f"final_ad_n{n_cur}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t9b = min(remaining * 0.50, 70.0)
            if t9b > 10:
                a_out, _ = coordinate_descent_v2(best_tracker['seq'], t9b,
                                                 verbose=False, rng_seed=31)
                try_update(a_out, f"final_cd_n{n_cur}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t9c = min(remaining * 0.50, 50.0)
            if t9c > 5:
                a_out, _ = peak_shaving(best_tracker['seq'], t9c,
                                        verbose=False, rng_seed=32)
                try_update(a_out, f"final_ps_n{n_cur}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t9d = min(remaining * 0.50, 40.0)
            if t9d > 5:
                a_out, _ = spectral_flatten(best_tracker['seq'], t9d,
                                            verbose=False, n_iters=100)
                try_update(a_out, f"final_spec_n{n_cur}")

    # Final
    if best_tracker['seq'] is not None:
        fs = evaluate_safe(best_tracker['seq'])
        if np.isfinite(fs):
            best_score = fs
            best_seq = best_tracker['seq']
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0] * n_base

    return [float(x) for x in best_seq]
