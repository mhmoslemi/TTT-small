def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize_to(a, n):
    """Normalize a to have sum = sqrt(2n)."""
    norm_val = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm_val / n
    return a * norm_val / s


def subgradient_at_k(a, k, n):
    """Compute subgradient of (a*a)[k] w.r.t. a."""
    grad = np.zeros(n)
    lo1 = max(0, k - n + 1)
    hi1 = min(n, k + 1)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        grad[lo1:hi1] += a[k - idx]
    lo2 = max(0, k)
    hi2 = min(n, k + n)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[lo2:hi2] += a[idx - k]
    return grad


def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)
    if size == 0:
        return None, None

    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def smooth_lp_best_response(a_fixed, M, beta, solver):
    """
    Solve smoothed LP: max sum(b) s.t. (1/beta)*log(sum_k exp(beta*(a*b)[k])) <= M, b >= 0.
    Uses only a subset of peaks for efficiency.
    """
    n = len(a_fixed)
    conv = fast_autocorr(a_fixed)
    conv_size = len(conv)
    M_curr = conv.max()
    if M_curr < 1e-15:
        return None

    # Select top peaks for the log-sum-exp
    n_top = max(10, min(conv_size, n // 4))
    top_k = np.argsort(conv)[-n_top:]

    T = np.zeros((n_top, n))
    for idx in range(n_top):
        k = int(top_k[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    # log-sum-exp constraint: (1/beta)*log(sum exp(beta*(b^T T_k))) <= M
    # Equivalent to: sum exp(beta * (T @ b)[k]) <= exp(beta * M)
    # This is convex. Use cp.log_sum_exp if available, otherwise approximate.
    linear_terms = T @ b_var
    # cp.log_sum_exp exists in cvxpy
    constraint = cp.log_sum_exp(beta * linear_terms) <= beta * M
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [constraint])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def smooth_alternating(a0, time_limit, solver, verbose=False, seed=0):
    """
    Alternating optimization with log-sum-exp smoothing.
    Alternates between:
    1. LP best response (exact) for b given a
    2. Smooth gradient step for a given the "direction" from step 1
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    conv_size = 2 * n - 1
    step = 0.3
    patience = 0
    max_patience = 300

    # Beta schedule: start smooth, get sharper
    beta_start = 20.0
    beta_end = 500.0
    total_expected = max(100, int(time_limit / 2.0))

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        progress = min(1.0, t / total_expected)
        beta = beta_start * (beta_end / beta_start) ** progress

        conv = fast_autocorr(a)
        M_curr = conv.max()
        if M_curr < 1e-12:
            break

        # Step 1: LP best response (exact max constraint)
        active_thr = M_curr * 0.85
        active_set = np.where(conv >= active_thr)[0]
        n_top = max(10, n // 50)
        top_k = np.argsort(conv)[-n_top:]
        active_set = np.unique(np.concatenate([active_set, top_k]))
        if len(active_set) > int(1.0 * n):
            keep = np.argsort(conv[active_set])[-int(1.0 * n):]
            active_set = active_set[keep]

        b, _ = lp_best_response(a, M_curr, active_set, solver)
        if b is None:
            patience += 1
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            patience += 1
            continue
        g_lp = b * norm_val / b_sum

        # Step 2: Smooth gradient step on a
        # Compute softmax weights over autocorrelation peaks
        weights_raw = np.exp(beta * (conv - M_curr))
        w_sum = weights_raw.sum()
        if w_sum < 1e-15:
            weights = np.ones(conv_size) / conv_size
        else:
            weights = weights_raw / w_sum

        # Only use top K weights for efficiency
        n_use = max(5, min(20, conv_size // 100))
        top_w = np.argsort(weights)[-n_use:]
        # Renormalize
        w_used = weights[top_w] / weights[top_w].sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(w_used, top_w):
            grad += w * subgradient_at_k(a, int(k), n)

        # Project gradient: remove mean component (to preserve sum)
        support = np.where(a > 0.01 * a.max())[0]
        if len(support) > 1:
            grad_mean = grad[support].mean()
            grad[support] -= grad_mean

        # Gradient step
        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm_val / s_trial

        # Also try the LP direction directly
        a_lp = g_lp.copy()

        # Pick the better
        score_grad = fast_autocorr(a_trial).max()
        score_lp = fast_autocorr(a_lp).max()

        if score_grad <= score_lp:
            a_new = a_trial
            new_score = score_grad
        else:
            # Blend
            a_blend = 0.5 * a_trial + 0.5 * a_lp
            a_blend = np.maximum(a_blend, 0.0)
            s_bl = a_blend.sum()
            if s_bl > 1e-15:
                a_blend = a_blend * norm_val / s_bl
            score_blend = fast_autocorr(a_blend).max()
            if score_blend < new_score:
                a_new = a_blend
                new_score = score_blend
            else:
                a_new = a_lp
                new_score = score_lp

        if new_score < best_score - 1e-12:
            a = a_new
            best_score = new_score
            best_a = a.copy()
            patience = 0
            step = min(step * 1.01, 1.5)
        else:
            # Try smaller step
            for frac in [0.5, 0.25, 0.1]:
                a_small = a - step * frac * grad
                a_small = np.maximum(a_small, 0.0)
                s_sm = a_small.sum()
                if s_sm < 1e-15:
                    continue
                a_small = a_small * norm_val / s_sm
                sc_sm = fast_autocorr(a_small).max()
                if sc_sm < best_score:
                    a = a_small
                    best_score = sc_sm
                    best_a = a.copy()
                    patience = 0
                    break
            else:
                patience += 1
                step *= 0.97
                if step < 1e-8:
                    break

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothAlt t={t} ({elapsed:.0f}s): best={best_score:.10f}, beta={beta:.0f}")

    return best_a, best_score

    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    max_no_improve = 6

    while time.monotonic() - t0 < time_limit:
        if no_improve >= max_no_improve:
            break
        improved = False
        conv = fast_autocorr(a)

        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            if aj < 1e-10:
                continue

            lo = max(0, j)
            hi = min(size, j + n)

            # Try a few candidate values
            typical = norm_val / n
            candidates = []
            if aj > 1e-10:
                candidates.extend([aj * 0.5, aj * 0.8, aj * 1.2, aj * 1.5])
            candidates.extend([0.0, typical * 0.5, typical, typical * 1.5, typical * 2.0])
            candidates = sorted(set([max(0.0, min(c, norm_val * 0.3)) for c in candidates]))

            best_x = aj
            best_obj = conv.max()

            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                delta = x - aj
                new_conv = conv.copy()
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta
                # Account for normalization change
                s_new = norm_val + delta  # approximate (we renormalize)
                if s_new < 1e-10:
                    continue
                scale = norm_val / s_new
                new_max = new_conv.max() * scale * scale
                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s = a.sum()
                if s > 1e-15:
                    a = a * norm_val / s
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-12:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                    else:
                        a = best_a.copy()

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def perturb_and_reoptimize(best_seq, time_limit, solver, rng, seed):
    """Perturb and re-optimize from perturbed state."""
    n = len(best_seq)
    # Structured perturbation: add a small sinusoid
    freq = rng.integers(1, n // 4)
    phase = rng.uniform(0, 2 * np.pi)
    amp = 0.05 + 0.05 * rng.random()
    x = np.arange(n)
    perturbation = 1.0 + amp * np.cos(2 * np.pi * freq * x / n + phase)
    perturbation = np.maximum(perturbation, 0.1)
    perturbed = best_seq * perturbation
    s = perturbed.sum()
    if s < 1e-15:
        return best_seq
    perturbed = perturbed / s
    a_norm = normalize_to(perturbed, n)
    a_out, _ = nesterov_lp_optimize(a_norm, time_limit, solver, verbose=False, seed=seed)
    return a_out


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence."""
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False


def propose_candidate():
    """
    Main search combining LP best-response, smooth alternating optimization,
    multi-length exploration, and coordinate descent polish.
    """
    rng = np.random.default_rng(2025)
    t_global = time.monotonic()
    time_budget = 1770.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)
    print(f"Starting optimization with n={n_base}, score={best_score:.8f}")

    # === Phase 1: LP Nesterov (main optimization) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
        a_norm = normalize_to(best_seq, n_base)
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, seed=0)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp1_n{n_base}", t_global)

    # === Phase 2: Smooth alternating optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 40:
        print(f"\n=== Phase 2: Smooth alternating n={n_cur}, budget={t2:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = smooth_alternating(a_norm, t2, solver, verbose=True, seed=42)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"smooth_n{n_cur}", t_global)

    # === Phase 3: LP Nesterov round 2 (from improved state) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40:
        print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur}, budget={t3:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, seed=777)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp2_n{n_cur}", t_global)

    # === Phase 4: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        print(f"\n=== Phase 4: Multi-length exploration, budget={remaining:.0f}s ===")
        t4 = min(remaining * 0.25, 300.0)
        a_out, sc_out = multi_length_explore(best_seq, t4, solver, verbose=True)
        if a_out is not None and sc_out < best_score:
            best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "multi_len", t_global)

    # === Phase 5: Smooth alternating round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        print(f"\n=== Phase 5: Smooth alternating r2 n={n_cur}, budget={t5:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = smooth_alternating(a_norm, t5, solver, verbose=False, seed=999)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"smooth2_n{n_cur}", t_global)

    # === Phase 6: Perturbation restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_restarts = 3
    t6_total = min(remaining * 0.25, 250.0)
    t6_each = t6_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        t6 = min(t6_each, remaining - 20)
        if t6 < 20:
            break
        print(f"\n=== Phase 6r{r}: Perturbation restart n={n_cur}, budget={t6:.0f}s ===")
        a_out = perturb_and_reoptimize(best_seq, t6, solver, rng, seed=1000 + r)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"pert{r}_n{n_cur}", t_global)

    # === Phase 7: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 7: CD polish n={n_cur}, budget={min(remaining*0.4, 100):.0f}s ===")
        t7 = min(remaining * 0.4, 100.0)
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent_polish(a_norm, t7, verbose=False, seed=55)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd_n{n_cur}", t_global)

    # === Phase 8: Final LP micro-optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final LP n={n_cur}, budget={min(remaining*0.5, 80):.0f}s ===")
        t8 = min(remaining * 0.5, 80.0)
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t8, solver, verbose=False, seed=31415)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"final_lp_n{n_cur}", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
