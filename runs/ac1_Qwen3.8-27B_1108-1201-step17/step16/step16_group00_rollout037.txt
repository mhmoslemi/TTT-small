**Algorithm for coordinate descent:**
1. Compute autocorrelation and identify active set $S$ (top 20 lags)
2. For each position $i$ (in order of "importance" = sensitivity to active lags):

   b. Find the $j$ that most reduces the max over $S$
   c. Do a 1D line search for the optimal $\delta$
   d. Execute the move
3. Repeat until no improvement

- For each active lag $k$, the "good" positions (to move mass to) are those where $a[k-j] + a[j-k]$ is small
- So for each active lag, I identify the top "bad" positions and top "good" positions, and only consider pairs from these sets

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_constraint_matrix(y):
    n = len(y)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = y
    return A


def solve_lp_direction(y, solver):
    n = len(y)
    A = build_constraint_matrix(y)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None, None
    b_val = b_val / b_sum
    M_val = float(M_var.value)
    return b_val, M_val

        # Nesterov step

        if s < 1e-15:

            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[ki - idx]

            if contrib.max() < 1e-20:
                continue

            # Sort sources by contribution (highest first)
            src_order = idx[np.argsort(contrib)[::-1]][:5]

            for src in src_order:
                if a[src] < 1e-12:
                    continue

                # Find best target: low sensitivity, not source
                sens_copy = sens.copy()
                sens_copy[src] = np.inf
                # Also avoid positions too close to source
                for j in range(max(0, src - 2), min(n, src + 3)):
                    sens_copy[j] = np.maximum(sens_copy[j], sens_copy[src - 1] if src > 0 else sens.copy()[src])

                dst_candidates = np.argsort(sens_copy)[:10]
                dst = int(dst_candidates[0])
                if dst == src:
                    dst = int(dst_candidates[1]) if len(dst_candidates) > 1 else int(rng.integers(n))

                # 1D line search
                best_delta = 0.0
                best_obj = ac.max()

                # Precompute check_lags including special lags for this pair
                cl = list(top_idx[:8])
                special = [2 * src, 2 * dst, src + dst]
                for s_val in special:
                    if 0 <= s_val < 2 * n - 1 and s_val not in cl:
                        cl.append(s_val)
                # Also add a few random lags as safety check
                for _ in range(3):
                    rnd_k = int(rng.integers(2 * n - 1))
                    if rnd_k not in cl:
                        cl.append(rnd_k)

                delta_max = a[src]
                # Try geometrically spaced deltas
                deltas = np.concatenate([
                    np.array([0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]),
                    np.array([delta_max * f for f in [0.01, 0.05, 0.1]])
                ])
                deltas = deltas[(deltas > 1e-8) & (deltas < delta_max)]

                for delta in sorted(deltas):
                    obj = ac_change_eval(a, ac, src, dst, delta, cl)
                    if obj < best_obj:
                        best_obj = obj
                        best_delta = delta

                if best_delta > 1e-8 and best_obj < M_curr - 1e-14:
                    a[src] -= best_delta
                    a[dst] += best_delta
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    improved = True

        if not improved:
            # Try random perturbation
            if stall > 30 or round_count > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()

        round_count += 1

        if verbose and round_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord round={round_count} ({elapsed:.0f}s): J={2.0*n*best_m:.6f}")

    return best_a, best_m


def hybrid_optimize(a0, time_limit, solver, verbose=True):
    """Alternating LP-FW and coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Phase A: LP-FW
    t_a = time_limit * 0.6
    a, m = lp_fw_nesterov(a, t_a, solver, verbose=verbose)
    if 2.0 * n * m < 2.0 * n * best_m:
        best_m = m
        best_a = a.copy()

    # Phase B: Coordinate descent
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 10:
        a, m = coord_descent_guided(a, remaining, verbose=verbose)
        if 2.0 * n * m < 2.0 * n * best_m:
            best_m = m
            best_a = a.copy()

    return best_a, best_m
