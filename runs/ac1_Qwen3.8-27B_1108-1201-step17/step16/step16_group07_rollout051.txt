def fast_conv(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]

def fast_grad(a, weights):
    """Compute gradient of weighted sum of convolution values.
    grad[j] = 2 * sum_m a[m] * weights[j+m] (valid m, j+m)
    Uses FFT for O(n log n) computation."""
    n = len(a)
    N = 1 << (2 * n - 1 - 1).bit_length()  # >= 2n-1
    W = rfft(weights, N)
    A = rfft(a, N)
    cross = irfft(np.conj(W) * A, N)
    grad = np.zeros(n)
    grad[0] = cross[0]
    if n > 1:
        grad[1:] = cross[N-1 : N-n : -1]
    grad *= 2.0
    return grad
