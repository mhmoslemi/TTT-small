    size = len(k_indices)

    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:

        pass

    else:
        a = a * norm / s0

    t0 = time.monotonic()
    active_set = None
    conv_size = 2 * n - 1
    n_iters = 0
    stall_count = 0

    y_prev = a.copy()
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        
        # Adaptive active set
        if active_set is None or t % 20 == 0:
            progress = min(1.0, t / 500.0)
            # Start with 15% tolerance, end with 2%
            tol = 0.15 * (1.0 - progress) + 0.02 * progress
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            # Always include top peaks
            n_top = max(10, n // 50)

                        improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0
    
    return best_a, best_score

Let me think of another approach: **iterative water-filling**.

The idea is:
1. Compute the autocorrelation
2. The positions where the autocorrelation is near the maximum are the "bottleneck"
3. We want to redistribute the "energy" of the sequence to reduce the peak
4. This is analogous to water-filling in information theory

A simpler version:
1. Identify the top M peaks in the autocorrelation
2. For each peak position k, the subgradient tells us which elements of a to reduce
3. Reduce those elements and increase others (those that don't contribute to the peaks)
4. Iterate

def subgradient_at_k_fast(a, k, n):
    grad = np.zeros(n)
    lo1 = max(0, k - n + 1)
    hi1 = min(n, k + 1)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        grad[lo1:hi1] += a[k - idx]
    lo2 = max(0, k)
    hi2 = min(n, k + n)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[lo2:hi2] += a[idx - k]
    return grad


def lp_best_response(a_fixed, M, active_set, solver):
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)

    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        tol = 0.04
        active_set = np.where(conv >= M_curr * (1.0 - tol))[0]
        if len(active_set) < 5:
            active_set = np.argsort(conv)[-max(5, n // 80):]
        if len(active_set) > 2 * n:
            active_set = active_set[np.argsort(conv[active_set])[-2 * n:]]

        b, duals = lp_best_response(a, M_curr, active_set, solver)
        if b is None:
            patience += 2
            continue

        if duals is not None and len(duals) == len(active_set):
            weights = np.maximum(duals, 0.0)
            if weights.sum() < 1e-15:
                weights = np.ones(len(weights))
            weights = weights / weights.sum()
        else:
            weights = np.exp(-6.0 * (M_curr - conv[active_set]) / max(M_curr * 0.03, 1e-10))
            weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, active_set):
            grad += w * subgradient_at_k_fast(a, int(k), n)

        support = np.where(a > 0.003 * a.max())[0]
        if len(support) < 2:
            break

        grad_mean = np.mean(grad[support])
        direction = grad.copy()
        direction[support] -= grad_mean
        direction[support] -= 0.15 * np.std(grad[support])

        a_trial = a - step * direction * n ** 0.25
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.008, 2.0)
        else:
            patience += 1
            step *= 0.96
            if step < 1e-8:
                break

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualPolish t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False


def propose_candidate():
    """
    Main search combining LP best-response, gradient projection,
    coordinate descent, and spectral smoothing for autocorrelation minimization.
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = 1760.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)
    print(f"Starting optimization with n={n_base}, score={best_score:.8f}")

    # === Phase 1: LP Nesterov (primary, 50% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.50, 800.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
        a_norm = normalize_to(best_seq, n_base)
        a_out, _ = lp_nesterov_v2(a_norm, t1, solver, verbose=True, seed=0)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp1_n{n_base}", t_global)

    # === Phase 2: Gradient projection (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.10, 150.0)
    if t2 > 20:
        print(f"\n=== Phase 2: Gradient Projection n={n_cur}, budget={t2:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = gradient_projection_v2(a_norm, t2, verbose=True, seed=1)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"gradproj_n{n_cur}", t_global)

    # === Phase 3: LP Nesterov round 2 (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur}, budget={t3:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = lp_nesterov_v2(a_norm, t3, solver, verbose=True, seed=123)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp2_n{n_cur}", t_global)

    # === Phase 4: Dual-guided polish (8% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.08, 120.0)
    if t4 > 15:
        print(f"\n=== Phase 4: Dual polish n={n_cur}, budget={t4:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = dual_guided_polish(a_norm, t4, solver, verbose=False)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"dual_n{n_cur}", t_global)

    # === Phase 5: Coordinate descent (7% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.07, 100.0)
    if t5 > 10:
        print(f"\n=== Phase 5: CD n={n_cur}, budget={t5:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent_v2(a_norm, t5, verbose=True, seed=5)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd_n{n_cur}", t_global)

    # === Phase 6: Spectral smoothing + short LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 6: Spectral smoothing ===")
        for alpha in [2.0, 4.0, 6.0]:
            a_smooth = spectral_smooth(best_seq, alpha=alpha)
            _, _ = try_update(best_seq, best_score, a_smooth, f"spec_a{alpha}", t_global)
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining <= 40:
                break

    # === Phase 7: Perturbation restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 80:
        n_restarts = 2
        t7_each = min(remaining * 0.12, 100.0)
        for r in range(n_restarts):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            t7 = min(t7_each, remaining - 20)
            if t7 < 20:
                break
            print(f"\n=== Phase 7r{r}: Perturbation restart n={n_cur}, budget={t7:.0f}s ===")
            scale = 0.008 + 0.01 * rng.random()
            noise = 1.0 + scale * rng.standard_normal(n_cur)
            noise = np.maximum(noise, 0.3)
            perturbed = best_seq * noise
            s = perturbed.sum()
            if s > 1e-15:
                perturbed = perturbed / s
                a_norm = normalize_to(perturbed, n_cur)
                a_out, _ = lp_nesterov_v2(a_norm, t7, solver, verbose=False, seed=500 + r * 7)
                best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"pert{r}_n{n_cur}", t_global)

    # === Phase 8: Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 25:
        t8 = min(remaining * 0.5, 60.0)
        print(f"\n=== Phase 8: Final CD n={n_cur} ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent_v2(a_norm, t8, verbose=False, seed=77)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cdFinal_n{n_cur}", t_global)

    # === Phase 9: Final gradient projection micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 15:
        t9 = min(remaining * 0.6, 40.0)
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = gradient_projection_v2(a_norm, t9, verbose=False, seed=314)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"gpFinal_n{n_cur}", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
