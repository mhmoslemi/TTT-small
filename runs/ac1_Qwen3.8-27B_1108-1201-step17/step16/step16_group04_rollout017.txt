"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral shaping (Gerchberg-Saxton style power flattening) for initialization
2. Adaptive log-sum-exp mirror descent with Nesterov acceleration (fast, many iterations)
3. Two-phase beta annealing (coarse -> fine)
4. Peak-targeted coordinate descent with adaptive moves
5. Multi-length and multi-initialization exploration
"""

import numpy as np
import time


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def spectral_shape(a, n_iter=50, alpha=0.5):
    """Flatten power spectrum via iterative DFT correction (Gerchberg-Saxton)."""
    a = normalize(a.copy())
    n = len(a)
    for it in range(n_iter):
        A = np.fft.fft(a)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-15)
        target = 1.0 / np.sqrt(mag_safe)
        phase = A / mag_safe
        A_new = phase * target
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = (1.0 - alpha) * a + alpha * a_new
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_grad_sparse(a, ac, beta, top_k=300):
    """Compute sparse gradient of log-sum-exp smoothed max autocorrelation.
    Returns (grad, ac_max) where grad[j] proportional to d(max_smooth)/d(a[j])."""
    n = len(a)
    ac_max = ac.max()
    # Stable log-sum-exp weights
    exp_ac = np.exp(beta * (ac - ac_max))
    total = exp_ac.sum()
    if total < 1e-300:
        # All weights are ~0, use uniform
        top_idx = np.argsort(ac)[::-1][:min(top_k, len(ac))]
        w_sparse = np.ones(len(top_idx)) / len(top_idx)
    else:
        if top_k < len(ac):
            top_idx = np.argsort(ac)[::-1][:top_k]
            w_sparse = exp_ac[top_idx] / total
            w_sparse_sum = w_sparse.sum()
            if w_sparse_sum > 1e-300:
                w_sparse = w_sparse / w_sparse_sum
            else:
                w_sparse = np.ones(len(top_idx)) / len(top_idx)
        else:
            top_idx = np.arange(len(ac))
            w_sparse = exp_ac / total

    # Compute gradient: grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
    grad = np.zeros(n)
    for idx_pos in range(len(top_idx)):
        w_k = w_sparse[idx_pos]
        if w_k < 1e-18:
            continue
        k = int(top_idx[idx_pos])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # For j in [js, je], k-j is in [0, n-1]
        # grad[j] += 2 * w_k * a[k-j]
        other_idx = k - np.arange(js, je + 1)
        grad[js:je + 1] += 2.0 * w_k * a[other_idx]

    return grad, ac_max


def lse_mirror_descent(a0, time_limit, beta_start=10.0, beta_end=500.0,
                       eta0=2.0, top_k=300, restart_period=5000,
                       verbose=True, rng_seed=123):
    """
    Nesterov-accelerated mirror descent with log-sum-exp smoothing.
    Adaptive beta (geometric increase) and eta (1/sqrt(t) decay).
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    ac_best = fast_autocorr(a)
    best_m = ac_best.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    n_total_iter = 0
    stall = 0

    # Estimate total iterations for beta scheduling
    while True:
        if time.monotonic() - t0 > time_limit:
            break

        t = n_total_iter + 1

        # Nesterov momentum
        if t > 1:
            mu = (t - 1.0) / (t + 1.5)
            y = x + mu * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-18)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive beta: geometric increase
        progress = min(1.0, t / max(t / 3.0, 100))  # ramp up over first third
        beta = beta_start * (beta_end / beta_start) ** progress

        # Compute gradient
        grad, ac_max = compute_grad_sparse(y, ac, beta, top_k)

        # Step size: eta = eta0 / sqrt(t)
        eta = eta0 / (np.sqrt(t) + 1.0)

        # Mirror descent update (exponential family)
        log_a = np.log(y + 1e-30) - eta * grad
        log_max = log_a.max()
        a_new = np.exp(log_a - log_max)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = y
        x = a_new
        n_total_iter += 1

        # Restart if stalled
        if stall > restart_period:
            stall = 0
            pert = rng.normal(0, 0.003, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-12)
            sx = x_new.sum()
            if sx > 1e-15:
                x_new = x_new / sx
                x = x_new
                x_prev = x_new.copy()

        if verbose and n_total_iter % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LSE-MD it={n_total_iter} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}")

    # Final check
    ac_f = fast_autocorr(x)
    if ac_f.max() < best_m:
        best_m = ac_f.max()
        best_a = x.copy()

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=42):
    """Peak-targeted coordinate descent with adaptive transfer sizes."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(30, len(ac))]
        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # Find best destination: minimize impact on top peaks
            # Score each candidate by how much it would add to top peaks
            best_dst = -1
            best_score = float('inf')

            # Sample candidates
            n_cand = min(500, n)
            candidates = rng.integers(0, n, n_cand)

            for dst in candidates:
                dst_int = int(dst)
                if dst_int == int(src):
                    continue
                sens = 0.0
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    d1 = k2i - dst_int
                    d2 = dst_int - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n and d2 != d1:
                        add += a[d2]
                    sens += add
                if sens < best_score:
                    best_score = sens
                    best_dst = dst_int

            if best_dst < 0:
                continue

            dst = best_dst
            found = False
            for frac in [0.005, 0.01, 0.02, 0.04, 0.07, 0.1, 0.15, 0.25]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        s = a.sum()
                        if s > 1e-15:
                            a = a / s

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      Coord ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

                                     eta0=3.0, top_k=400, restart_period=3000,
                                     verbose=verbose, rng_seed=1)
        if m1 < best_m:
            best_m = m1
            best_a = a1.copy()
        a = best_a.copy()

    # Phase 2: Medium
    remaining = time_limit - (time.monotonic() - t0)
    t2 = remaining * 0.35
    if t2 > 10:
        if verbose:
            print(f"      Phase 2 (medium, beta=30-200): {t2:.0f}s")
        a2, m2 = lse_mirror_descent(a, t2, beta_start=30.0, beta_end=200.0,
                                     eta0=2.0, top_k=300, restart_period=4000,
                                     verbose=verbose, rng_seed=2)
        if m2 < best_m:
            best_m = m2
            best_a = a2.copy()
        a = best_a.copy()

    # Phase 3: Fine
    remaining = time_limit - (time.monotonic() - t0)
    t3 = remaining * 0.25
    if t3 > 5:
        if verbose:
            print(f"      Phase 3 (fine, beta=100-1000): {t3:.0f}s")
        a3, m3 = lse_mirror_descent(a, t3, beta_start=100.0, beta_end=1000.0,
                                     eta0=1.5, top_k=200, restart_period=5000,
                                     verbose=verbose, rng_seed=3)
        if m3 < best_m:
            best_m = m3
            best_a = a3.copy()

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator:
    1. Load parent or create initial sequence
    2. Spectral shaping for better initialization
    3. Multi-phase LSE mirror descent (coarse -> fine)
    4. Coordinate polish
    5. Multi-length exploration
    6. Final polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1750.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # --- Load parent ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 2400
        # Sine-squared initialization
        x = np.linspace(0, 1, n_base)
        a_init = np.sin(np.pi * x) ** 2
        a_init = a_init / a_init.sum()
        best_seq = a_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"\nStarting: n={n0}, best_score={best_score:.6f}, budget={time_budget:.0f}s")

    # === Step 1: Spectral shaping to improve initialization ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        a_spec = spectral_shape(normalize(best_seq), n_iter=80, alpha=0.5)
        try_update(a_spec, "spectral_shape")

    # === Step 2: Main optimization on original length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t_main = min(remaining * 0.40, 600.0)
    if t_main > 30:
        print(f"\n=== Main LSE optimization: n={n_best}, budget={t_main:.0f}s ===")
        a_in = normalize(best_seq)
        a_out, _ = multi_start_lse(a_in, t_main, n_targets=n_best, verbose=True)
        try_update(a_out, f"lse_main_n{n_best}")

    # === Step 3: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_coord = min(remaining * 0.12, 150.0)
    if t_coord > 10:
        print(f"\n=== Coord polish: n={len(best_seq)}, budget={t_coord:.0f}s ===")
        a_in = normalize(best_seq)
        a_out, _ = coordinate_polish(a_in, t_coord, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # === Step 4: Another LSE pass (different beta schedule) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_lse2 = min(remaining * 0.20, 250.0)
    if t_lse2 > 20:
        print(f"\n=== LSE pass 2: n={len(best_seq)}, budget={t_lse2:.0f}s ===")
        a_in = normalize(best_seq)
        a_out, _ = lse_mirror_descent(a_in, t_lse2, beta_start=50.0, beta_end=500.0,
                                       eta0=2.5, top_k=250, restart_period=4000,
                                       verbose=True, rng_seed=99)
        try_update(a_out, f"lse2_n{len(best_seq)}")

    # === Step 5: Try larger length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_alt1 = min(n_best + 600, 5000)
    if n_alt1 != n_best and remaining > 200:
        print(f"\n=== Alt length n={n_alt1}, budget={remaining:.0f}s ===")
        a_long = interpolate_sequence(normalize(best_seq), n_alt1)
        a_long = spectral_shape(a_long, n_iter=40, alpha=0.5)

        t5a = min(remaining * 0.4, 300.0)
        if t5a > 20:
            a_out, _ = multi_start_lse(a_long, t5a, n_targets=n_alt1, verbose=True)
            try_update(a_out, f"lse_n{n_alt1}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t5b = min(remaining * 0.3, 100.0)
            if t5b > 10:
                a_in = normalize(best_seq)
                a_out, _ = coordinate_polish(a_in, t5b, verbose=False, rng_seed=77)
                try_update(a_out, f"coord_alt_n{len(best_seq)}")

    # === Step 6: Try much larger length if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_alt2 = min(n_best + 1500, 8000)
    if n_alt2 != n_best and n_alt2 != n_alt1 and remaining > 250:
        print(f"\n=== Large length n={n_alt2}, budget={remaining:.0f}s ===")
        a_large = interpolate_sequence(normalize(best_seq), n_alt2)
        a_large = spectral_shape(a_large, n_iter=30, alpha=0.5)

        t6 = min(remaining * 0.5, 300.0)
        if t6 > 20:
            a_out, _ = multi_start_lse(a_large, t6, n_targets=n_alt2, verbose=True)
            try_update(a_out, f"lse_n{n_alt2}")

    # === Step 7: Final LSE fine-tuning on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    t_final = min(remaining * 0.35, 200.0)
    if t_final > 15:
        print(f"\n=== Final LSE: n={n_final}, budget={t_final:.0f}s ===")
        a_in = normalize(best_seq)
        a_out, _ = lse_mirror_descent(a_in, t_final, beta_start=200.0, beta_end=2000.0,
                                       eta0=1.0, top_k=150, restart_period=6000,
                                       verbose=True, rng_seed=456)
        try_update(a_out, f"lse_final_n{n_final}")

    # === Step 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t8 = min(remaining * 0.7, 80.0)
        if t8 > 10:
            print(f"\n=== Final Coord: n={len(best_seq)}, budget={t8:.0f}s ===")
            a_in = normalize(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, verbose=True, rng_seed=1234)
            try_update(a_out, f"coord_very_final_n{len(best_seq)}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
