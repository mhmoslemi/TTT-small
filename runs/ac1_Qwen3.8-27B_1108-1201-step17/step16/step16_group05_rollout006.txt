    n = len(v)

def smoothed_max_grad(a, beta, top_k=200):
    """
    Compute gradient of (1/beta)*log(sum(exp(beta*r[k])) for top-k lags)
    where r = conv(a,a)
    """
    n = len(a)
    r = fast_autocorr(a)
    k_eff = min(top_k, len(r))
    top_idx = np.argsort(r)[::-1][:k_eff]
    top_vals = r[top_idx]
    shifted = top_vals - top_vals[0]
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad, r.max()
