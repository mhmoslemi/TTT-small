    """Clip sequence values to between 0 and 1000, ensure sum >= 0.01."""
    arr = np.asarray(sequence, dtype=np.float64)

        if i % 100 == 0 and i > 0:
            base_points = np.linspace(0, 1, n)
            interpolated_points = np.linspace(0, 1, new_n)
            interpolated_seq = np.interp(interpolated_points, base_points, best_seq)
            interpolated_seq = to_normalized(interpolated_seq)
            time_used = time.monotonic()
            if time_used - time0 > 60 or time_remaining < 0.005:
                time_remaining = 0.005
            new_score = validate_solution(interpolated_seq, time_remaining)
            if new_score < best_score:
                best_score = new_score
                best_seq = interpolated_seq
            current_top_k = min(top_k, new_n)
        elif new_n != n:
            base_points = np.linspace(0, 1, n)
            interpolated_points = np.linspace(0, 1, new_n)
            interpolated_seq = np.interp(interpolated_points, base_points, best_seq)
            interpolated_seq = to_normalized(interpolated_seq)
            new_score = validate_solution(interpolated_seq, 0.05)
            if new_score < best_score:
                best_score = new_score
                best_seq = interpolated_seq
            current_top_k = min(top_k, new_n)

        # Adaptive constraint generation
        active_lag_indices = np.argsort(autocorr)[-1*current_top_k:]
        b_direction = solve_lp_direction(y, M_curr, active_lag_indices)

        # Early termination if no improvement
        if not b_direction or b_direction.sum() < 1e-15:
            continue

        # Apply accelerated gradient step
        alpha = 0.5 * cp.Minimize(cp.sum_squares(y - b_direction) + 0.01 * cp.norm(y, 1))
        # Use l-BFGS-B with callable function, but time-limited simplex is safer
        beta = min(alpha, 0.5)
        x1 = 0.9 * best_seq + 0.1 * b_direction
        x1 = clip_nonnegative(x1)
        x1_sum = np.sum(x1)
        if x1_sum < 0.01:
            x1 = clip_nonnegative(0.5 * best_seq + 0.5 * b_direction)
            x1_sum = np.sum(x1)

        x1 = x1 / x1_sum

        # Combine parent and child via explicit crossover (structurally, sequence-wise)
        x_new = (0.9 * best_seq + 0.1 * x1)
        x_new = (x_new + np.random.randn(len(x_new)) * 0.001).clip(0, None)
        x_new_sum = np.sum(x_new)
        if x_new_sum < 0.01:
            x_new = x_new * (x_new_sum + 0.01 * eta)

        x_new_sum = np.sum(x_new)
        failed_evaluations = 0
        # Evaluate until successful or time permits
        while time.monotonic() - time0 < time_limit:
            score_new = validate_solution(x_new, 0.01)
            if score_new <= best_score * 0.99:
                best_score = score_new
                best_seq = x_new.copy()
                time_remaining = time_limit - (time.monotonic() - time0)
                break
            x_new = clip_nonnegative(x_new * 1.01)
            x_new = to_normalized(x_new)

def crossover_sequences(seq1, seq2, alpha=0.5):
    """Produce offspring by mixing two normalized sequences from a selected segment."""
    len1 = len(seq1)
    len2 = len(seq2)
    min_len = min(len1, len2)
    max_len = max(len1, len2)
    new_len = np.clip(int(min_len + 0.5 * (max_len - min_len)), min_len*0.8, max_len*1.2)
    # Interpolate length
    base_points = np.linspace(0, 1, len1)
    new_points = np.linspace(0, 1, new_len)
    new_seq = np.interp(new_points, base_points, (1-alpha)*seq1 + alpha*seq2)
    new_seq = clip_nonnegative(new_seq)
    new_seq_sum = np.sum(new_seq)
    if new_seq_sum < 0.01:
        new_seq = clip_nonnegative(new_seq * (new_seq_sum + 0.01 * (new_len/new_seq_sum)))
    return new_seq, new_len
