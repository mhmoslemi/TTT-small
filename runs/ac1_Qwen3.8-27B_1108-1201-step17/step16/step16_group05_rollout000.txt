def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        log_ratio = -eta * g
        # Clip to prevent overflow
        log_ratio = np.clip(log_ratio, -50.0, 50.0)
        a_new = a * np.exp(log_ratio)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
            a_new = np.maximum(a_new, 0.0)
            s_new2 = a_new.sum()
            if s_new2 > 1e-15:
                a_new = a_new / s_new2

        a = a_new
        M_new = fast_autocorr(a).max()

        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a.copy()
            stall = 0
            # Slightly increase step size on improvement
            best_eta = min(best_eta * 1.01, c0 * 2.0)
        else:
            stall += 1
            # Decrease step size on stall
            if stall % 100 == 0:
                best_eta *= 0.95

        # Periodic restart
        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.4)
            noise_scale = 0.003 * cur_top_k / top_k_end
            noise = rng.normal(0, noise_scale, n) * best_a
            a_re = np.maximum(best_a + noise, 1e-15)
            a_re = a_re / a_re.sum()
            a = a_re
            best_eta = c0
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      CMD restart t={t} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CMD t={t} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.1f}, eta={eta:.6f}")

    return best_a, best_m


def subgradient_projected_grad(a0, time_limit, beta_start=25.0, beta_end=6.0,
                               top_k_start=150, top_k_end=350,
                               c0=2.0, verbose=True, seed=42):
    """
    Nesterov-accelerated projected gradient with adaptive smoothed-max.
    Uses backtracking-inspired step size adjustment.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 3000
    step_mult = 1.0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / 25000.0)
        cur_beta = beta_start + (beta_end - beta_start) * frac
        cur_top_k = int(top_k_start + (top_k_end - top_k_start) * frac)

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)

        g, M_curr, _, _ = compute_smoothed_grad(y, cur_beta, top_k=cur_top_k)

        eta = gamma * step_mult / (np.sqrt(t) + 1.0)

        x_new = project_simplex(y - eta * g)

        x_prev = x
        x = x_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall = 0
            step_mult = min(step_mult * 1.005, 2.0)
        else:
            stall += 1
            if stall % 500 == 0:
                step_mult = max(step_mult * 0.9, 0.3)

        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.3)
            noise = rng.normal(0, 0.004, n) * best_a
            x_re = np.maximum(best_a + noise, 1e-12)
            x_re = x_re / x_re.sum()
            x = x_re
            x_prev = x_re
            step_mult = 1.0
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      APG restart t={t} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      APG t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe_fast(a0, time_limit, solver, verbose=True, max_constraints=None):
    """
    LP Frank-Wolfe with constraint selection (only top-K lags as constraints).
    Much faster than using all 2n-1 constraints.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    b_var = cp.Variable(n, nonneg=True)

    if max_constraints is None:
        max_constraints = min(2 * n - 1, 4 * n)

    K = min(200, 2 * n - 1)  # Start with fewer constraints
    b_prev = None

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Select top-K constraints
        top_k = np.argsort(ac)[::-1][:K]
        M_top = ac[top_k].max()

        # Build constraint matrix for selected lags
        # T[k, j] = a[j] if the lag k involves position j
        b_var = cp.Variable(n, nonneg=True)
        T = np.zeros((K, n))
        for ii, k in enumerate(top_k):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            T[ii, idx] = a[j_idx]

        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            K = max(100, K // 2)
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            K = max(100, K // 2)
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov Frank-Wolfe update
        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sa
        a = a_new

        M_new = fast_autocorr(a).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a.copy()
            # Gradually increase K as we make progress
            K = min(K + 5, max_constraints)
        else:
            K = max(150, K - 2)

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}")

    return best_a, best_m


def coordinate_polish_fast(a0, time_limit, verbose=True, seed=42):
    """
    Fast coordinate descent: move mass from high-peak-contributing positions
    to low-impact positions. Uses precomputed impact scores.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 80

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        # Compute peak contribution per position
        peak_contrib = np.zeros(n)
        for k in top_k[:15]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            peak_contrib[idx] += a[idx] * a[j_idx]

        # Find best move: high-contrib source -> low-impact dest
        sources = np.argsort(peak_contrib)[::-1][:50]
        improved = False

        # Precompute impact scores for all positions
        impact = np.zeros(n)
        for k in top_k[:10]:
            k_int = int(k)
            for dst in range(n):
                d = k_int - dst
                if 0 <= d < n:
                    impact[dst] += a[d]
                d2 = dst - k_int
                if 0 <= d2 < n:
                    impact[dst] += a[d2]

        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            # Find best destination: low impact, not the source
            impact[src] = np.inf  # Exclude source
            dst_candidates = np.argsort(impact)[:200]
            best_dst = -1
            best_delta = 0.0

            for dst in dst_candidates:
                if dst == src or a[dst] + a[src] * 0.05 > 1.0:
                    continue
                # Quick estimate
                amt = a[src] * 0.02
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    best_delta = best_m - M_trial
                    best_dst = dst
                    if best_delta > 1e-6:
                        break

            if best_dst >= 0:
                # Try multiple fractions
                for frac in [0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[best_dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      CD done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def get_parent_sequence():
    """Try to load parent sequence from global."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p.copy()
    except Exception:
        pass
    return None


def diverse_perturbation(a, rng, scale=0.01):
    """Create a diverse perturbation of a sequence."""
    n = len(a)
    a_pert = a.copy()
    # Smooth noise
    noise = rng.normal(0, scale, n)
    # Apply a low-pass filter to the noise
    kernel_size = min(n // 10, 50)
    if kernel_size > 1:
        kernel = np.ones(kernel_size) / kernel_size
        noise = np.convolve(noise, kernel, mode='same')
    a_pert = a_pert + noise * a_pert
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Uses entropic mirror descent as primary optimizer,
    LP Frank-Wolfe for structural improvements,
    and coordinate descent for final polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    rng = np.random.default_rng(2024)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Load parent or create default ===
    parent = get_parent_sequence()
    if parent is not None:
        p_norm = norm_sum1(parent)
        p_score = evaluate_safe(p_norm)
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
        best_seq = p_norm.copy()
        best_score = p_score
    else:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Entropic Mirror Descent (main workhorse) ===
    rem = remaining_time()
    print(f"\n=== Phase 1: CMD n={n0}, time={min(rem*0.35, 500.0):.0f}s ===")
    t1 = min(rem * 0.35, 500.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = entropic_mirror_descent(
            a_in, t1, beta_start=35.0, beta_end=8.0,
            top_k_start=80, top_k_end=300,
            c0=1.5, verbose=True, seed=42)
        try_update(a_out, f"cmd_n{n0}")

    # === Phase 2: Nesterov APG (different dynamics) ===
    rem = remaining_time()
    print(f"\n=== Phase 2: APG n={len(best_seq)} ===")
    t2 = min(rem * 0.20, 280.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_projected_grad(
            a_in, t2, beta_start=25.0, beta_end=6.0,
            top_k_start=150, top_k_end=350,
            c0=2.5, verbose=True, seed=77)
        try_update(a_out, f"apg_n{len(best_seq)}")

    # === Phase 3: LP Frank-Wolfe (structural improvement) ===
    rem = remaining_time()
    n_curr = len(best_seq)
    print(f"\n=== Phase 3: LP-FW n={n_curr} ===")
    t3 = min(rem * 0.18, 250.0)
    if t3 > 30 and n_curr <= 3500:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_fast(
            a_in, t3, solver, verbose=True, max_constraints=600)
        try_update(a_out, f"lpfw_n{n_curr}")

    # === Phase 4: CMD polish after LP ===
    rem = remaining_time()
    print(f"\n=== Phase 4: CMD polish n={len(best_seq)} ===")
    t4 = min(rem * 0.12, 150.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = entropic_mirror_descent(
            a_in, t4, beta_start=20.0, beta_end=10.0,
            top_k_start=200, top_k_end=400,
            c0=0.8, verbose=False, seed=99)
        try_update(a_out, f"cmd_polish_n{len(best_seq)}")

    # === Phase 5: Coordinate descent ===
    rem = remaining_time()
    print(f"\n=== Phase 5: Coord n={len(best_seq)} ===")
    t5 = min(rem * 0.10, 120.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish_fast(a_in, t5, verbose=True, seed=42)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 6: Try slightly different lengths ===
    rem = remaining_time()
    n_best = len(best_seq)
    if rem > 150:
        for delta_n in [200, -200, 400]:
            n_try = n_best + delta_n
            if n_try < 500 or n_try > 4000:
                continue
            rem = remaining_time()
            if rem < 80:
                break
            print(f"\n=== Phase 6: Try n={n_try} ===")
            a_long = interpolate_sequence(best_seq, n_try)
            t6 = min(rem * 0.3, 120.0)
            if t6 > 30:
                a_out, _ = entropic_mirror_descent(
                    norm_sum1(a_long), t6, beta_start=25.0, beta_end=8.0,
                    top_k_start=100, top_k_end=250,
                    c0=1.2, verbose=False, seed=200 + delta_n)
                try_update(a_out, f"cmd_n{n_try}")
                rem = remaining_time()
                if rem > 30:
                    t6b = min(rem * 0.3, 50.0)
                    if t6b > 15:
                        a_out2, _ = coordinate_polish_fast(
                            norm_sum1(a_out), t6b, verbose=False, seed=300)
                        try_update(a_out2, f"coord_n{n_try}")

    # === Phase 7: Diverse restarts ===
    rem = remaining_time()
    if rem > 100:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Diverse restarts n={n_best} ===")
        n_restarts = 3
        t_per = min(rem * 0.6 / n_restarts, 80.0)
        for r in range(n_restarts):
            rem = remaining_time()
            if rem < 40:
                break
            t7 = min(t_per, rem - 20)
            if t7 < 20:
                break
            # Create diverse perturbation
            a_pert = diverse_perturbation(norm_sum1(best_seq), rng, scale=0.02 + 0.01 * r)
            a_out, _ = entropic_mirror_descent(
                a_pert, t7, beta_start=30.0, beta_end=10.0,
                top_k_start=100, top_k_end=250,
                c0=1.0, verbose=False, seed=500 + r)
            try_update(a_out, f"restart{r}_cmd")
            # Quick coord polish
            rem = remaining_time()
            if rem > 25:
                a_out2, _ = coordinate_polish_fast(
                    norm_sum1(a_out), min(rem * 0.3, 30.0), verbose=False, seed=600 + r)
                try_update(a_out2, f"restart{r}_coord")

    # === Phase 8: Final LP polish ===
    rem = remaining_time()
    n_best = len(best_seq)
    if n_best <= 3000 and rem > 60:
        print(f"\n=== Phase 8: Final LP n={n_best} ===")
        t8 = min(rem * 0.3, 120.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_fast(
                a_in, t8, solver, verbose=True, max_constraints=800)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 9: Final coordinate polish ===
    rem = remaining_time()
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final coord n={n_best} ===")
    if rem > 20:
        t9 = min(rem * 0.5, 80.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish_fast(a_in, t9, verbose=True, seed=314)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 10: One more CMD pass with very high beta ===
    rem = remaining_time()
    if rem > 30:
        print(f"\n=== Phase 10: Final CMD (high beta) ===")
        t10 = min(rem * 0.6, 60.0)
        if t10 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = entropic_mirror_descent(
                a_in, t10, beta_start=50.0, beta_end=20.0,
                top_k_start=300, top_k_end=500,
                c0=0.5, verbose=False, seed=999)
            try_update(a_out, f"cmd_final_highbeta")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
