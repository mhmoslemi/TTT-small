```
def fast_autocorr(a):
    a=np.asarray(a,dtype=np.float64)
    n=a.size
    size=2*n-1
    f=1 << (size-1).bit_length()
    A=np.fft.rfft(a,f)
    return np.fft.irfft(A*A,f)[:size]
```

```
def softmax_weights(values, beta):
    v=np.asarray(values,dtype=np.float64)
    if v.size==0: return v
    v=v-v.max()
    e=np.exp(beta*v)
    s=e.sum()
    if s<=0 or not np.isfinite(s): return np.ones_like(v)/v.size
    return e/s
```

```
def hard_top_gradient(a, top_idx, weights):
    n=a.size
    grad=np.zeros(n,dtype=np.float64)
    for ww, kk in zip(weights, top_idx):
       w=float(ww); k=int(kk)
       if w <=0 or not np.isfinite(w): continue
       ist=max(0, k-n+1)
       ien=min(n-1, k)
       if ist <= ien:
          idx=np.arange(ist, ien+1, dtype=np.int64)
          partners=k-idx
          grad[ist:ien+1] += (2.0*w)*a[partners]
    return grad
```

```python
"""
Search for low-peak non-negative step functions minimizing
    2*n*max(conv(a,a))/(sum(a))^2.

The program combines:
1. A top-lag exponentiated-gradient (mirror descent) method using exact
   subgradients from the currently worst lags, with Nesterov-style momentum,
   adaptive step size, and noise-restart escapes.
2. A fast exact local search using one-coordinate mass transfers.  The change
   of the full autocorrelation for a transfer i -> j is computed in O(n)
   algebraically, allowing many exact candidate moves without FFT each time.
3. Length interpolation/sweeping to escape resolution-specific local minima.

The final sequence is returned as a Python list of floats.
"""

import numpy as np
import time


def time_left(t_start, budget_s):
    """Return remaining time under a second budget."""
    return max(0.0, float(budget_s) - (time.monotonic() - t_start))


def normalize_sequence(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    if a.size == 0:
        return a
    s = float(a.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def fast_autocorr(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_direct(a):
    """Direct objective matching the evaluator for valid non-negative inputs."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01 or not np.isfinite(s):
        return float("inf")
    ac = fast_autocorr(a)
    m = float(ac.max())
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m / (s * s))


def safe_evaluate(a):
    """Safely call the externally provided evaluator."""
    try:
        arr = np.asarray(a, dtype=np.float64)
        lst = [float(x) for x in arr]
        return float(evaluate_sequence(lst))
    except Exception:
        return float("inf")


def top_k_indices(values, k):
    """Indices of the k largest values, without a full sort when possible."""
    v = np.asarray(values, dtype=np.float64)
    n = v.size
    if n == 0 or k <= 0:
        return np.empty(0, dtype=np.int64)
    if k >= n:
        return np.argsort(v)[::-1]
    part = np.argpartition(v, -k)[-k:]
    order = np.argsort(v[part])[::-1]
    return part[order]


def softmax_weights(values, beta):
    """Softmax weights for a small set of peak-lag values."""
    v = np.asarray(values, dtype=np.float64)
    if v.size == 0:
        return np.empty(0, dtype=np.float64)
    v = v - float(np.max(v))
    e = np.exp(float(beta) * v)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    return e / s


def hard_top_gradient(a, top_idx, weights):
    """
    Weighted exact subgradient for the top lags.

    For lag k, d(conv[k])/d(a_i) = 2*a[k-i] if 0 <= k-i < n, else 0.
    """
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    for ww, kk in zip(weights, top_idx):
        w = float(ww)
        k = int(kk)
        if w <= 0.0 or not np.isfinite(w):
            continue
        ist = max(0, k - n + 1)
        ien = min(n - 1, k)
        if ist <= ien:
            idx = np.arange(ist, ien + 1, dtype=np.int64)
            partners = k - idx
            grad[ist:ien + 1] += (2.0 * w) * a[partners]
    return grad


def toplag_mirror_descent(a0, time_limit, eta_init=0.25, top_start=2,
                          top_end=14, beta_weights=8.0,
                          verbose=False, seed=1):
    """
    Exponentiated gradient descent on the simplex using gradients from the
    currently worst lags.  Momentum and restarts are included.
    """
    n = len(a0)
    if n == 0:
        return a0, float("inf")

    a = normalize_sequence(a0)
    ac = fast_autocorr(a)
    M_cur = float(ac.max())
    best_a = a.copy()
    best_M = M_cur
    best_obj = 2.0 * n * best_M

    t0 = time.monotonic()
    total_budget = max(0.0, float(time_limit))
    rng = np.random.default_rng(seed)

    a_prev = a.copy()
    k = 1
    eta = float(eta_init)
    eta_min = 0.01
    eta_max = 3.0

    no_best = 0
    rejects = 0
    restart_std = 0.005
    max_iters = 50000000

    for it in range(1, max_iters + 1):
        if total_budget > 0.0:
            if time.monotonic() - t0 > total_budget:
                break

        top_count = top_start + (it // 2000) + (no_best // 15000)
        top_count = min(int(top_end), int(ac.size), max(1, int(top_count)))

        if k > 1:
            tf = float(k)
            tt = (1.0 + np.sqrt(1.0 + 4.0 * tf * tf)) / 2.0
            alpha = (tf - 1.0) / tt
            y = a + alpha * (a - a_prev)
            y = np.maximum(y, 1e-12)
            sy = float(y.sum())
            if sy <= 1e-15 or not np.isfinite(sy):
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a.copy()

        ac_y = fast_autocorr(y)
        chk = float(ac_y.max())
        if not np.isfinite(chk):
            y = a.copy()
            ac_y = fast_autocorr(y)

        top_idx = top_k_indices(ac_y, top_count)
        wtop = softmax_weights(ac_y[top_idx], beta_weights)
        grad = hard_top_gradient(y, top_idx, wtop)

        maxg = float(grad.max())
        if maxg <= 1e-30 or not np.isfinite(maxg):
            a_new = y.copy()
        else:
            g = grad / maxg
            g = np.maximum(g, 0.0)
            ex = np.exp(-eta * g)
            a_new = y * ex
            a_new = np.maximum(a_new, 1e-15)
            s_new = float(a_new.sum())
            if s_new <= 1e-15 or not np.isfinite(s_new):
                a_new = y.copy()
            else:
                a_new = a_new / s_new

        ac_new = fast_autocorr(a_new)
        M_new = float(ac_new.max())

        if np.isfinite(M_new) and M_new <= M_cur + 1e-13:
            a_prev = a
            a = a_new
            M_cur = M_new
            k += 1

            if M_new < best_M - 1e-15:
                best_M = M_new
                best_a = a.copy()
                best_obj = 2.0 * n * best_M
                no_best = 0
                eta = min(eta_max, eta * 1.05)
            else:
                no_best += 1
                if no_best > 2000:
                    eta = max(eta_min, eta * 0.995)
        else:
            eta = max(eta_min, eta * 0.5)
            k = 1
            a_prev = a.copy()
            rejects += 1
            no_best += 1

        if no_best > 12000 or rejects > 150:
            if restart_std < 0.004:
                restart_std = 0.01 + 0.03 * float(rng.random())
            else:
                restart_std = max(0.004, restart_std * 0.8)

            noise = rng.normal(0.0, restart_std, n) * best_a
            a = best_a + noise
            a = normalize_sequence(a)
            a_prev = a.copy()
            k = 1
            M_cur = float(fast_autocorr(a).max())
            no_best = 0
            rejects = 0
            eta = min(eta_max, max(eta_init * 0.75, eta * 0.8))

        if verbose and it % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TMD it={it} t={elapsed:.0f}s "
                  f"cur={2.0 * n * M_cur:.6f} best={best_obj:.6f} "
                  f"eta={eta:.4f} top={top_count}")

    return best_a, best_obj


def transfer_new_ac(ac, a, i, j, delta):
    """
    Exact new autocorrelation after moving mass delta from i to j:
        a[i] -= delta, a[j] += delta, sum unchanged.
    """
    n = a.size
    new_ac = ac.copy()
    two_d = 2.0 * float(delta)
    d2 = float(delta) * float(delta)

    new_ac[i:i + n] -= two_d * a
    new_ac[j:j + n] += two_d * a

    if 2 * i < 2 * n - 1:
        new_ac[2 * i] += d2
    if 2 * j < 2 * n - 1:
        new_ac[2 * j] += d2
    ipj = i + j
    if ipj < 2 * n - 1:
        new_ac[ipj] -= 2.0 * d2

    return new_ac


def coordinate_polish(a0, time_limit, verbose=False, seed=123,
                      top_count=8, source_limit=25, dest_limit=10):
    """
    Fast exact local search using one-coordinate mass transfers.
    Candidate sources are high-mass / high-peak-contribution coordinates;
    candidate destinations are low marginal-influence coordinates.
    """
    n = len(a0)
    if n == 0:
        return a0, float("inf")

    a = normalize_sequence(a0)
    ac = fast_autocorr(a)
    M_cur = float(ac.max())
    best_a = a.copy()
    best_M = M_cur
    best_obj = 2.0 * n * best_M

    t0 = time.monotonic()
    total_budget = max(0.0, float(time_limit))
    rng = np.random.default_rng(seed)

    fracs = (0.003, 0.01, 0.03, 0.08, 0.2)

    if n > 5000:
        top_count = min(top_count, 5)
        source_limit = min(source_limit, 12)
        dest_limit = min(dest_limit, 6)
    elif n > 3500:
        top_count = min(top_count, 8)
        source_limit = min(source_limit, 18)
        dest_limit = min(dest_limit, 8)

    stall = 0
    rnd = 0

    while True:
        if total_budget > 0.0:
            if time.monotonic() - t0 > total_budget:
                break

        rnd += 1
        M_cur = float(ac.max())
        obj_cur = 2.0 * n * M_cur
        if obj_cur < best_obj - 1e-15:
            best_obj = obj_cur
            best_M = M_cur
            best_a = a.copy()

        tc = min(int(top_count), int(ac.size), 20)
        top_idx = top_k_indices(ac, tc)
        wtop = softmax_weights(ac[top_idx], 15.0)

        remove_score = np.zeros(n, dtype=np.float64)
        add_cost = np.zeros(n, dtype=np.float64)

        for ww, kk in zip(wtop, top_idx):
            w = float(ww)
            k = int(kk)
            if w <= 0.0 or not np.isfinite(w):
                continue
            ist = max(0, k - n + 1)
            ien = min(n - 1, k)
            if ist <= ien:
                idx = np.arange(ist, ien + 1, dtype=np.int64)
                partners = k - idx
                remove_score[ist:ien + 1] += w * (a[ist:ien + 1] * a[partners])
                add_cost[ist:ien + 1] += w * a[partners]

        mcount = min(30, n)
        if mcount == n:
            mass_idx = np.arange(n, dtype=np.int64)
        else:
            mass_idx = np.argpartition(-a, mcount - 1)[:mcount]

        rcount = min(50, n)
        if rcount == n:
            rem_idx = np.arange(n, dtype=np.int64)
        else:
            rem_idx = np.argpartition(-remove_score, rcount - 1)[:rcount]

        sources = np.unique(np.concatenate((mass_idx, rem_idx)))
        if sources.size > 0:
            sources = sources[np.argsort(-remove_score[sources])]
        else:
            sources = np.arange(n, dtype=np.int64)

        eff_source = min(int(source_limit), int(sources.size))

        dcount = min(100, n)
        if dcount == n:
            dest_pool = np.arange(n, dtype=np.int64)
        else:
            dest_pool = np.argpartition(add_cost, dcount - 1)[:dcount]
        dest_pool = dest_pool[np.argsort(add_cost[dest_pool])]
        eff_dest = min(int(dest_limit), int(dest_pool.size))

        threshold = best_M if best_M < M_cur else M_cur
        improved = False

        for si in range(eff_source):
            i = int(sources[si])
            ai = float(a[i])
            if ai <= 1e-12:
                continue
            for dj in range(eff_dest):
                j = int(dest_pool[dj])
                if i == j:
                    continue
                for frac in fracs:
                    delta = float(frac) * ai
                    if delta > 0.9 * ai:
                        delta = 0.9 * ai
                    if delta <= 0.0:
                        continue
                    new_ac = transfer_new_ac(ac, a, i, j, delta)
                    Mn = float(new_ac.max())
                    if Mn < threshold - 1e-16:
                        a[i] -= delta
                        if a[i] < 0.0:
                            a[i] = 0.0
                        a[j] += delta
                        ac = new_ac
                        M_cur = Mn
                        if Mn < best_M - 1e-16:
                            best_M = Mn
                            best_obj = 2.0 * n * best_M
                            best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            random_trials = 100 if n <= 4000 else 40
            for _rr in range(random_trials):
                if total_budget > 0.0:
                    if time.monotonic() - t0 > total_budget:
                        break

                if sources.size > 0:
                    i = int(sources[int(rng.integers(sources.size))])
                else:
                    i = int(rng.integers(n))

                if dest_pool.size > 0:
                    j = int(dest_pool[int(rng.integers(dest_pool.size))])
                else:
                    j = int(rng.integers(n))

                if i == j:
                    j = int(rng.integers(n))
                    if i == j:
                        continue

                ai = float(a[i])
                if ai <= 1e-12:
                    continue

                delta = ai * float(rng.uniform(0.002, 0.25))
                if delta > 0.9 * ai:
                    delta = 0.9 * ai

                new_ac = transfer_new_ac(ac, a, i, j, delta)
                Mn = float(new_ac.max())
                if Mn < threshold - 1e-16:
                    a[i] -= delta
                    if a[i] < 0.0:
                        a[i] = 0.0
                    a[j] += delta
                    ac = new_ac
                    M_cur = Mn
                    if Mn < best_M - 1e-16:
                        best_M = Mn
                        best_obj = 2.0 * n * best_M
                        best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if not improved:
                stall += 1

                if stall in (20, 50, 100, 200):
                    best_work = best_a.copy()
                    best_work_M = best_M
                    for std in (0.003, 0.01, 0.03, 0.08):
                        cand = best_a + rng.normal(0.0, std, n) * best_a
                        cand = normalize_sequence(cand)
                        mc = float(fast_autocorr(cand).max())
                        if mc < best_work_M - 1e-16:
                            best_work_M = mc
                            best_work = cand
                    a = best_work.copy()
                    ac = fast_autocorr(a)
                    M_cur = float(ac.max())
                    if M_cur < best_M - 1e-16:
                        best_M = M_cur
                        best_obj = 2.0 * n * best_M
                        best_a = a.copy()
                    stall = 0

                if stall > 300:
                    break

        if verbose and rnd % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    coord round={rnd} t={elapsed:.0f}s best={best_obj:.6f}")

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    f = np.asarray(f, dtype=np.float64)
    old_n = f.size
    new_n = int(new_n)
    if old_n == 0 or new_n <= 0:
        return f.copy()
    if old_n == new_n:
        return f.copy()

    s = float(f.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return np.ones(new_n, dtype=np.float64) / float(new_n)

    fn = f / s
    if old_n == 1:
        return np.full(new_n, 1.0 / float(new_n), dtype=np.float64)

    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, fn)
    g = np.maximum(g, 0.0)
    gs = float(g.sum())
    if gs <= 1e-15:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    return g / gs


def default_sequence(n=2600):
    """Fallback smooth positive initial sequence."""
    x = np.linspace(0.0, 1.0, n)
    a = np.sin(np.pi * x) ** 2 + 1e-6
    return normalize_sequence(a)


def propose_candidate(budget_s=1800.0):
    """
    Main search entry point.  Returns the best list of non-negative heights
    found within the supplied time budget.
    """
    t_global = time.monotonic()
    budget = max(60.0, min(float(budget_s) - 70.0, 1720.0))

    base = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.ndim == 1 and h.size >= 50:
            h = np.clip(h, 0.0, 1000.0)
            if np.isfinite(h).all() and h.sum() > 1e-8:
                base = h
    except Exception:
        base = None

    if base is None:
        base = default_sequence(2600)

    best_a = normalize_sequence(base)
    best_obj = objective_direct(best_a)
    if not np.isfinite(best_obj):
        # This should not happen for valid base, but keep a safe fallback.
        best_a = default_sequence(2600)
        best_obj = objective_direct(best_a)

    print(f"Initial: n={len(best_a)}, obj={best_obj:.6f}")

    # Phase 1: quick exact local polish from the parent.
    rem = time_left(t_global, budget)
    if rem > 120:
        t1 = min(180.0, rem * 0.20)
        if t1 > 40:
            cand, obj = coordinate_polish(
                best_a, t1,
                verbose=False, seed=11,
                source_limit=20, dest_limit=8
            )
            if obj < best_obj - 1e-15:
                best_a = cand
                best_obj = obj
                print(f"Phase1 coord: obj={best_obj:.6f}, n={len(best_a)}")

    # Phase 2: top-lag mirror descent for further shape improvement.
    rem = time_left(t_global, budget)
    if rem > 160:
        t2 = min(560.0, rem * 0.35)
        if t2 > 80:
            cand, obj = toplag_mirror_descent(
                best_a, t2,
                eta_init=0.25, top_start=2, top_end=14,
                beta_weights=8.0, verbose=True, seed=101
            )
            if obj < best_obj - 1e-15:
                best_a = cand
                best_obj = obj
                print(f"Phase2 MD:    obj={best_obj:.6f}, n={len(best_a)}")

    # Phase 3: local polish after global adjustment.
    rem = time_left(t_global, budget)
    if rem > 120:
        t3 = min(420.0, rem * 0.30)
        if t3 > 60:
            cand, obj = coordinate_polish(
                best_a, t3,
                verbose=False, seed=22,
                source_limit=30, dest_limit=10
            )
            if obj < best_obj - 1e-15:
                best_a = cand
                best_obj = obj
                print(f"Phase3 coord: obj={best_obj:.6f}, n={len(best_a)}")

    # Phase 4: second mirror descent pass with a different schedule.
    rem = time_left(t_global, budget)
    if rem > 160:
        t4 = min(360.0, rem * 0.28)
        if t4 > 80:
            cand, obj = toplag_mirror_descent(
                best_a, t4,
                eta_init=0.12, top_start=5, top_end=16,
                beta_weights=10.0, verbose=False, seed=202
            )
            if obj < best_obj - 1e-15:
                best_a = cand
                best_obj = obj
                print(f"Phase4 MD:    obj={best_obj:.6f}, n={len(best_a)}")

    # Phase 5: local polish again.
    rem = time_left(t_global, budget)
    if rem > 100:
        t5 = min(360.0, rem * 0.30)
        if t5 > 60:
            cand, obj = coordinate_polish(
                best_a, t5,
                verbose=False, seed=33,
                source_limit=35, dest_limit=12
            )
            if obj < best_obj - 1e-15:
                best_a = cand
                best_obj = obj
                print(f"Phase5 coord: obj={best_obj:.6f}, n={len(best_a)}")

    # Phase 6: length sweep with short local polish.
    rem = time_left(t_global, budget)
    if rem > 220:
        n0 = len(best_a)
        length_choices = [
            n0 + 400,
            n0 - 400,
            n0 + 800,
            n0 - 800,
            n0 + 1200,
            n0 - 1200
        ]
        seen = set()
        tried = 0
        for nn in length_choices:
            rem = time_left(t_global, budget)
            if rem < 140 or tried >= 3:
                break
            nn = int(nn)
            if nn <= 300 or nn > 5000 or nn == n0 or nn in seen:
                continue
            seen.add(nn)
            tried += 1

            cand = interpolate_sequence(best_a, nn)
            talloc = min(90.0, rem * 0.35)
            if talloc < 30:
                break

            c2, o2 = coordinate_polish(
                cand, talloc,
                verbose=False, seed=303 + tried,
                source_limit=18, dest_limit=8
            )
            if o2 < best_obj - 1e-15:
                best_a = c2
                best_obj = o2
                print(f"Length {nn}: obj={best_obj:.6f}")

    # Phase 7: final global polish.
    rem = time_left(t_global, budget)
    if rem > 70:
        t6 = min(300.0, rem * 0.85)
        if t6 > 30:
            cand, obj = coordinate_polish(
                best_a, t6,
                verbose=True, seed=999,
                source_limit=35, dest_limit=12
            )
            if obj < best_obj - 1e-15:
                best_a = cand
                best_obj = obj
                print(f"Final coord: obj={best_obj:.6f}, n={len(best_a)}")

    best_a = normalize_sequence(best_a)
    direct = objective_direct(best_a)
    evaled = safe_evaluate(best_a)
    if np.isfinite(evaled) and evaled < direct:
        best_obj = evaled
    else:
        best_obj = direct

    print(f"FINAL: obj={best_obj:.6f}, n={len(best_a)}, "
          f"time={time.monotonic() - t_global:.0f}s")

    return [float(x) for x in best_a]
```
