def subgradient_at_k(a, k, n):
    """Subgradient of (a*a)[k] w.r.t. a."""
    grad = np.zeros(n)
    # (a*a)[k] = sum_j a[j]*a[k-j]
    # d/d a[j] = a[k-j] + a[k-j] (if j != k-j) or 2*a[j] (if j == k-j, i.e., k=2j)
    # Actually: d/d a[j] of sum_i a[i]*a[k-i] = a[k-j] + (if k-j != j) a[k-j]
    # Wait: d/d a[j] = sum_i (δ_{i,j} * a[k-i] + a[i] * δ_{k-i,j})
    # = a[k-j] + a[k-j] = 2*a[k-j] (if j != k-j)
    # = a[k-j] + a[k-j] = 2*a[j] (if j = k-j, i.e., k = 2j) -- same thing!
    # So d/d a[j] = 2*a[k-j] for all j where 0<=j<n and 0<=k-j<n
    # Wait no: if j = k-j, i.e., k = 2j, then the two terms are the same term a[j]*a[j],
    # and d/d a[j] of a[j]^2 = 2*a[j]. Which is the same as 2*a[k-j] = 2*a[j]. ✓
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    for j in range(lo, hi):
        grad[j] = 2.0 * a[k - j]
    return grad

def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    if size == 0:
        return None, None
    
    # Build constraint matrix T (size x n)
    # T[k_idx, j] = a_fixed[k - j] if 0 <= k-j < n, else 0
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except:
        pass
    
    return b_val, duals

    t0 = time.monotonic()
    step = 0.3 / (n ** 0.3)
    patience = 0
    max_patience = 300
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit or early_stop():
            break
        if patience > max_patience:
            break
        
        conv = fast_autocorr(a)
        M_curr = conv.max()
        
        # Active set
        tol = 0.05
        active_set = np.where(conv >= M_curr * (1 - tol))[0]
        if len(active_set) < 5:
            active_set = np.argsort(conv)[-max(5, n // 80):]
        
        # Solve LP for duals
        b, duals = lp_best_response(a, M_curr, active_set, solver)
        if b is None:
            patience += 1
            continue
        
        if duals is not None and len(duals) == len(active_set):
            weights = np.maximum(duals, 0.0)
            if weights.sum() < 1e-15:
                weights = np.ones(len(weights))
            weights = weights / weights.sum()
        else:
            weights = np.exp(-5.0 * (M_curr - conv[active_set]) / max(M_curr * 0.02, 1e-10))
            weights = weights / weights.sum()
        
        # Weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, active_set):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                grad[lo:hi] += w * 2.0 * a[k - np.arange(lo, hi)]
        
        # Project gradient to sum-zero subspace
        support = np.where(a > 0.01 * a.max())[0]
        if len(support) < 2:
            break
        grad_mean = np.mean(grad[support])
        direction = grad.copy()
        direction[support] -= grad_mean
        
        a_trial = a - step * direction
        a_trial = np.maximum(a_trial, 0.0)
        a_trial = normalize(a_trial)
        
        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.01, 2.0)
        else:
            patience += 1
            step *= 0.98
            if step < 1e-12:
                break
    
    return best_a, best_score
