def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0, restart_every=6000,
                 adaptive_beta=True, verbose=False, rng_seed=42):
    """Nesterov-accelerated mirror descent with adaptive parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if adaptive_beta and stall_count > restart_every // 2:
            cur_beta = max(cur_beta * 0.9, 2.0)
            cur_top_k = min(cur_top_k + 5, len(ac))

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 150)
            cur_beta = max(cur_beta - 1.0, 2.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}, top_k={cur_top_k}, beta={cur_beta:.1f}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=None, use_all=False):
    """LP-based Frank-Wolfe with adaptive constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    iter_count = 0

    if use_all:
        K_sel = 2 * n - 1
    elif top_K is not None:
        K_sel = min(top_K, 2 * n - 1)
    else:
        K_sel = min(300, 2 * n - 1)

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        if use_all or K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            constraint_vals = T_full @ y
            margins = M_curr - constraint_vals
            binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(len(binding_idx))

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

        # Adaptively increase K
        if not use_all and t % 20 == 0:
            K_sel = min(K_sel + 20, 2 * n - 1)

    return best_a, best_m


def peak_pair_reduction(a0, time_limit, verbose=True, rng_seed=77):
    """Targeted peak-pair reduction: identify most responsible pairs and redistribute."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200
    n_peaks = 5

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_peaks]

        improved = False
        # Build sensitivity map: for each position d, how much would increasing a[d] increase peaks?
        sens = np.zeros(n)
        for k in top_k:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            sens[idx] += a[partner]

        for k in top_k[:3]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            if contrib.max() < 1e-20:
                continue

            # Find the most responsible pair
            best_i_local = np.argmax(contrib)
            src = idx[best_i_local]
            src_partner = partner[best_i_local]

            # Choose which to reduce (reduce the one with less "global importance")
            # Global importance = how much a[j] contributes to all top peaks
            imp_src = sens[src]
            imp_partner = sens[src_partner]

            if imp_src >= imp_partner:
                reduce_idx = src
                keep_idx = src_partner
            else:
                reduce_idx = src_partner
                keep_idx = src

            if a[reduce_idx] < 1e-12:
                continue

            # Find best destination: low sensitivity
            exclude = {reduce_idx, keep_idx}
            candidates = [d for d in range(n) if d not in exclude]
            if not candidates:
                candidates = list(range(n))
            cand_arr = np.array(candidates)
            cand_sens = sens[cand_arr]
            dst = cand_arr[np.argmin(cand_sens)]

            # Try multiple fractions
            found = False
            for frac in [0.02, 0.01, 0.005, 0.002, 0.05, 0.08, 0.15]:
                amt = a[reduce_idx] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and stall > 50:
                # Random kick
                j = int(rng.integers(n))
                frac = rng.uniform(0.005, 0.03)
                amt = a[reduce_idx] * frac
                a[reduce_idx] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True

        if not improved:
            stall += 1
            # Multi-point perturbation
            n_pert = min(5, n)
            pert = rng.normal(0, 0.01, n) * a
            a_new = a + pert
            a_new = np.maximum(a_new, 0.0)
            if a_new.sum() > 1e-15:
                a_new = a_new / a_new.sum()
                M_new = fast_autocorr(a_new).max()
                if M_new < best_m:
                    a = a_new
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakPair ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakPair final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def spectral_init(n, n_modes=20, seed=0):
    """Generate a smooth non-negative sequence via Fourier synthesis."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_u_shape(n, sharpness=2.0):
    """U-shaped distribution (high at edges, low in middle)."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Aggressive coordinate descent with sensitivity-based move selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 150
    n_top = 15

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        # Build sensitivity
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:8]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved_this_round = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            dst = candidates[np.argmin(cand_sens)]

            if dst == src:
                continue

            found = False
            for frac in [0.01, 0.005, 0.02, 0.03, 0.05, 0.1]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved_this_round = True
                    found = True
                    break

            if not found and stall > 30:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved_this_round = True

        if not improved_this_round:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase optimization."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-init AMD ===
    print(f"\n=== Phase 1: Multi-init AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.25, 400.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shape(n0, sharpness=1.5))
        labels.append("u1.5")

        inits.append(make_u_shape(n0, sharpness=3.0))
        labels.append("u3.0")

        inits.append(make_bimodal(n0, width=0.2))
        labels.append("bimodal")

        inits.append(spectral_init(n0, n_modes=30, seed=42))
        labels.append("spectral")

        inits.append(spectral_init(n0, n_modes=10, seed=7))
        labels.append("spectral2")

        # Blend parent with U-shape
        if parent is not None:
            blend = 0.5 * norm_sum1(parent) + 0.5 * make_u_shape(n0, 2.0)
            inits.append(norm_sum1(blend))
            labels.append("blend_u")

        # Remove duplicates
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 50)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(20.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 15:
                break
            print(f"  AMD init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=60, beta=6.0, c0=5.0,
                                     restart_every=4000, adaptive_beta=True,
                                     verbose=False, rng_seed=42 + idx)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 40 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        # Use all constraints if feasible, else top-K
        use_all = (n_best <= 2000)
        top_K = min(2000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, top_K=top_K, use_all=use_all)
        try_update(a_out, "lp_main")

    # === Phase 3: Peak-pair reduction (new method) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak-pair reduction (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_pair_reduction(a_in, t3, verbose=True, rng_seed=77)
        try_update(a_out, "peakpair")

    # === Phase 4: AMD polish with higher top_k ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: AMD polish (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t4, top_k=100, beta=8.0, c0=3.5,
                                 restart_every=8000, adaptive_beta=True, verbose=False, rng_seed=200)
        try_update(a_out, "amd_polish")

    # === Phase 5: LP-FW again on updated best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP-FW round 2 (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 2000)
        top_K = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True, top_K=top_K, use_all=use_all)
        try_update(a_out, "lp_r2")

    # === Phase 6: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Coordinate descent (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, verbose=True, rng_seed=456)
        try_update(a_out, "coord")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")

    # Try slightly longer and shorter lengths
    trial_lengths = []
    for delta in [500, 1000, -500, 1500]:
        new_n = n_best + delta
        if 500 <= new_n <= 5000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))  # deduplicate preserving order
    trial_lengths = [l for l in trial_lengths if remaining > 80]

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.35, 200.0)
        if t7a > 30:
            a_out, _ = amd_optimize(a_long, t7a, top_k=60, beta=6.0, c0=4.0,
                                     restart_every=5000, adaptive_beta=True, verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and new_n <= 3500:
            t7b = min(remaining * 0.30, 120.0)
            if t7b > 20:
                a_in = norm_sum1(best_seq)
                use_all = (new_n <= 2000)
                top_K = min(1500, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_in, t7b, solver, verbose=True, top_K=top_K, use_all=use_all)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 8: Final LP on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    if remaining > 30 and n_best <= 4000:
        t8 = min(remaining * 0.5, 150.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            use_all = (n_best <= 2000)
            top_K = min(4000, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe(a_in, t8, solver, verbose=True, top_K=top_K, use_all=use_all)
            try_update(a_out, "lp_final")

    # === Phase 9: Final peak-pair + coord ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 9: Final polish (n={n_best}) ===")
        t9 = min(remaining * 0.5, 80.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_pair_reduction(a_in, t9 * 0.5, verbose=False, rng_seed=999)
            try_update(a_out, "peakpair_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t9b = min(remaining * 0.6, 40.0)
            if t9b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_polish(a_in, t9b, verbose=False, rng_seed=789)
                try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
