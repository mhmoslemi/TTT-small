```python
def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
    s = a.sum()

        return np.ones_like(a) / len(a)

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_val(ac, beta):
    """Compute the smoothed max of ac using log-sum-exp."""
    m = ac.max()
    if m < 1e-30:
        return 0.0
    shifted = ac - m
    max_shifted = shifted.max()
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    return (m + log_sum) / beta


def compute_smoothed_gradient(a, ac, beta, top_k=100):
    """Compute the gradient of the smoothed max at point a."""
    n = len(a)
    k_idx = np.argsort(ac)[::-1][:top_k]
    ac_top = ac[k_idx]
    
    shifted = ac_top - ac_top.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(top_k) / top_k
    else:
        w = w / w_sum

        a_idx = k - j_range
        g[j_start:j_end] += 2.0 * wi * a[a_idx]
    
    return g, k_idx, w
```
