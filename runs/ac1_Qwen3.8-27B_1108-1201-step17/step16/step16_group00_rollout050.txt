import numpy as np
import time
import cvxpy as cp

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def build_T(y, n):
    """Build the constraint matrix T where T[k,i] = y[k-i] if 0<=k-i<n."""
    T = np.zeros((2*n-1, n))
    for i in range(n):
        T[i:i+n, i] = y
    return T

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(200, nonneg=True)
            T_test = np.random.rand(400, 200)
            prob = cp.Problem(cp.Maximize(cp.sum(x)), [T_test @ x <= 1, cp.sum(x) >= 0.1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    print("  Solver: ECOS (fallback)")
    return cp.ECOS

def lp_fw_exact(a0, time_limit, solver, use_top_k=0, top_k_val=500, verbose=True, rng_seed=42):
    """
    LP Frank-Wolfe with exact line search and Nesterov acceleration.
    If use_top_k > 0, only use top_k_val most binding constraints.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        
        if M_curr < 1e-15:
            break
        
        # Build constraint matrix
        T = build_T(y, n)
        
        # Select constraints
        if use_top_k > 0:
            # Use top_k most binding constraints
            # Compute constraint values: T @ y
            T_y = T @ y
            margins = M_curr - T_y
            K = min(top_k_val, 2*n-1)
            binding = np.argsort(margins)[:K]
            T_sel = T[binding, :]
        else:
            T_sel = T
        
        # Solve LP: max sum(b) s.t. T_sel @ b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum  # direction in the simplex
        
        # Exact line search: minimize f((1-alpha)*y + alpha*g) over alpha in [0,1]
        # f(z) = max_k (T_z @ z) where T_z is built from z
        # This is expensive, so use a grid search with refinement
        best_alpha = 0.0
        best_val = fast_autocorr(y).max()
        
        # Coarse search
        alphas_coarse = np.linspace(0, 1, 20)
        for alpha in alphas_coarse[1:]:
            z = (1-alpha)*y + alpha*g
            z = np.maximum(z, 0)
            sz = z.sum()
            if sz < 1e-15:
                continue
            z = z / sz
            val = fast_autocorr(z).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine search around best
        if best_alpha > 0:
            delta = 1.0 / 20.0
            alphas_fine = np.linspace(max(0, best_alpha - delta), min(1, best_alpha + delta), 10)
            for alpha in alphas_fine:
                z = (1-alpha)*y + alpha*g
                z = np.maximum(z, 0)
                sz = z.sum()
                if sz < 1e-15:
                    continue
                z = z / sz
                val = fast_autocorr(z).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
        
        y_new = (1-best_alpha)*y + best_alpha*g
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new = y_new / sy
        
        # Nesterov
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t-1) / (t+2)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0)
            sx = x_new.sum()
            if sx > 1e-15:
                x_new = x_new / sx
            else:
                x_new = y_new.copy()
        
        y = y_new
        x = x_new
        
        # Update best
        My = fast_autocorr(y).max()
        if My < best_M:
            best_M = My
            best_a = y.copy()
        
        Mx = fast_autocorr(x).max()
        if Mx < best_M:
            best_M = Mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"  FW-exact it={t} ({elapsed:.0f}s): J={J:.6f}, alpha={best_alpha:.3f}")
    
    return best_a, best_M


def peak_shave_v2(a0, time_limit, verbose=False, rng_seed=7):
    """Improved peak shaving with adaptive step and multi-peak handling."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    alpha = 0.05
    stall = 0
    
    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 20000:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find all lags above 95% of max
        threshold = M * 0.95
        peak_mask = ac >= threshold
        peak_lags = np.where(peak_mask)[0]
        
        if len(peak_lags) == 0:
            break
        
        # Compute "pressure" at each position
        # pressure[i] = sum over peak lags k of a[k-i] (contribution to peak)
        pressure = np.zeros(n)
        for k in peak_lags:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                partners = k - idx
                pressure[idx] += a[partners]
        
        total_p = pressure.sum()
        if total_p < 1e-15:
            break
        
        # Water-filling inspired update
        # Want to reduce positions with high pressure, increase those with low
        # a_new[i] = a[i] * (avg_p / pressure[i])^beta  (soft inverse)
        beta_pw = 0.5  # exponent
        avg_p = total_p / n
        ratio = avg_p / (pressure + 1e-20)
        multiplier = np.power(np.minimum(ratio, 10.0), beta_pw)
        
        a_new = a * multiplier
        a_new = np.maximum(a_new, 0)
        # Blend with original to control step size
        a_new = (1 - alpha) * a + alpha * (a_new / a_new.sum() if a_new.sum() > 1e-15 else a)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        new_M = fast_autocorr(a_new).max()
        if new_M < M:
            a = a_new
        else:
            alpha *= 0.85
            if alpha < 1e-7:
                # Restart with perturbation
                a = best_a * (1 + 0.02 * rng.normal(0, 1, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                alpha = 0.05
                stall += 500
    
    return best_a, best_M


def coordinate_2pt(a0, time_limit, verbose=False, rng_seed=99):
    """2-point coordinate descent: pick the two most impactful coordinates and optimize jointly."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if stall > 5000:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        peak_lag = int(np.argmax(ac))
        
        # Find the pair of positions that contribute most to the peak
        i_start = max(0, peak_lag - n + 1)
        i_end = min(n - 1, peak_lag)
        idx = np.arange(i_start, i_end + 1)
        partners = peak_lag - idx
        contrib = a[idx] * a[partners]
        
        if contrib.max() < 1e-20:
            # Random pair
            i = int(rng.integers(n))
            j = int(rng.integers(n))
        else:
            top_idx = np.argsort(contrib)[::-1][:5]
            # Pick the pair with largest contribution
            i = idx[top_idx[0]]
            j = partners[top_idx[0]]
            if i == j:
                if len(top_idx) > 1:
                    i = idx[top_idx[1]]
                    j = partners[top_idx[1]]
                else:
                    j = int(rng.integers(n))
        
        if i == j:
            stall += 1
            continue
        
        # Optimize the 2-parameter subproblem
        # Move mass from i to j (or vice versa)
        # a_i -> a_i - delta, a_j -> a_j + delta (preserving sum)
        # Find optimal delta by 1D search
        ai = a[i]
        aj = a[j]
        
        best_delta = 0.0
        best_trial_M = M
        
        # Try moving from i to j and from j to i
        for direction in [1, -1]:
            # delta > 0 means move from i to j (direction=1) or from j to i (direction=-1)
            max_delta = ai if direction == 1 else aj
            if max_delta < 1e-15:
                continue
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.4, 0.7]:
                delta = frac * max_delta
                a_trial = a.copy()
                if direction == 1:
                    a_trial[i] = ai - delta
                    a_trial[j] = aj + delta
                else:
                    a_trial[i] = ai + delta
                    a_trial[j] = aj - delta
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_trial_M:
                    best_trial_M = M_trial
                    best_delta = delta * direction if direction == 1 else -delta
                    # Store the trial
                    best_trial = a_trial.copy()
        
        if best_trial_M < M - 1e-15:
            a = best_trial
            best_M = best_trial_M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Random perturbation
            if stall > 100:
                i_r = int(rng.integers(n))
                j_r = int(rng.integers(n))
                if i_r != j_r and a[i_r] > 1e-10:
                    delta = 0.01 * a[i_r]
                    a[i_r] -= delta
                    a[j_r] += delta
                    a = np.maximum(a, 0) / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
    
    return best_a, best_M


def interpolate_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = norm_sum1(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    return norm_sum1(a_interp)
