"""

import numpy as np
import cvxpy as cp
import time

# Global solver cache
_solver = None

def get_lp_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.SCS
    return _solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def smooth_subgradient(a, ac, top_k=50, beta=5.0):
    """Compute smooth subgradient with exponential weighting."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    
    # Get top-k lags
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    
    # Exponential weights
    weights = np.exp(beta * (ac[top_idx] - M))
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_max) / k_max
    else:
        weights = weights / w_sum
    
    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += w * a[j_idx]
    return g

def exact_subgradient(a, ac, threshold=0.995):
    """Compute exact subgradient from active lags."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n), 1
    
    active = np.where(ac >= threshold * M)[0]
    if len(active) == 0:
        active = np.array([int(ac.argmax())])
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += a[j_idx]
    
    # Average over active set
    g /= len(active)
    return g, len(active)

def mirror_step(a, g, eta):
    """Mirror descent step in log-space."""
    log_a = np.log(np.maximum(a, 1e-15)) - eta * g
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def backtracking_mirror_step(a, g, eta0, max_iters=20):
    """Mirror descent with backtracking line search."""
    ac_base = fast_autocorr(a)
    M_base = ac_base.max()
    
    eta = eta0
    for _ in range(max_iters):
        a_new = mirror_step(a, g, eta)
        M_new = fast_autocorr(a_new).max()
        if M_new < M_base - 1e-15 * M_base:
            return a_new, eta
        eta *= 0.5
    return a, eta

def fw_lp_direction(a, ac, top_k=50, solver=None):
    """Solve LP to find Frank-Wolfe direction."""
    if solver is None:
        solver = get_lp_solver()
    
    n = len(a)
    M = ac.max()
    
    # Get top-k lags
    top_idx = np.argsort(ac)[::-1][:min(top_k, len(ac))]
    
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    
    constraints = [cp.sum(b_var) == 1]
    
    for k in top_idx:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        # conv(a,b)[k] = sum_{i in idx} a[i] * b[k-i]
        coeff = np.zeros(n)
        for ii, j in enumerate(j_idx):
            coeff[j] += a[idx[ii]]
        constraints.append(coeff @ b_var <= t_var + ac[k_int])
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def amd_smooth(a0, time_limit, top_k=50, beta=5.0, c0=3.0, verbose=True, rng_seed=42):
    """AMD with smooth subgradient and backtracking."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 3000
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y /= y.sum()
        
        ac = fast_autocorr(y)
        g = smooth_subgradient(y, ac, top_k=top_k, beta=beta)
        
        # Adaptive step size
        eta = c0 / (np.sqrt(t) + 1.0) * np.sqrt(n)
        
        a_new = mirror_step(y, g, eta)
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart
        if stall > max_stall:
            stall = 0
            # Structured perturbation
            freq = rng.integers(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            derivs = np.ones(n)
            derivs[0] = 0.5
            derivs[-1] = 0.5
            perturb = 0.01 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            x_restarted = best_a * (1 + perturb)
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted /= x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            c0 *= 1.1
            
            if verbose:
                print(f"    AMD restart t={t}, c0={c0:.2f}")
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def fw_amd_hybrid(a0, time_limit, fw_interval=100, top_k=50, solver=None, verbose=True, rng_seed=7):
    """Hybrid: AMD steps with periodic FW-LP direction."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    c0 = 3.0
    stall = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(x)
        
        if t % fw_interval == 0:
            # FW-LP step
            b_dir = fw_lp_direction(x, ac, top_k=top_k, solver=solver)
            if b_dir is not None:
                alpha = 2.0 / (t / fw_interval + 2.0)
                x_new = (1 - alpha) * x + alpha * b_dir
                x_new = np.maximum(x_new, 0.0)
                x_new /= x_new.sum()
                x_prev = x
                x = x_new
            else:
                # Fallback to AMD
                g = smooth_subgradient(x, ac, top_k=30, beta=4.0)
                eta = c0 / (np.sqrt(t) + 1.0) * np.sqrt(n)
                x_prev = x
                x = mirror_step(x, g, eta)
        else:
            # Normal AMD step
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            y /= y.sum()
            
            ac_y = fast_autocorr(y)
            g = smooth_subgradient(y, ac_y, top_k=30, beta=4.0)
            eta = c0 / (np.sqrt(t) + 1.0) * np.sqrt(n)
            x_prev = x
            x = mirror_step(y, g, eta)
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 2000:
            stall = 0
            perturb = rng.normal(0, 0.005, n) * best_a
            x_restarted = np.maximum(best_a + perturb, 1e-10)
            x_restarted /= x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            c0 *= 1.15
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW-AMD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def influence_coord_descent(a0, time_limit, verbose=True, rng_seed=99):
    """Coordinate descent based on influence vectors."""
    n = len(a0)
    a = normalize(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top lags
        top_lags = np.argsort(ac)[::-1][:min(20, len(ac))]
        
        # Compute influence vector for each position
        # influence[p] = max over top lags of a[k-p] (contribution to peak if a[p] increases)
        influence = np.zeros(n)
        for k in top_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contrib = a[j_idx]
            influence[idx] = np.maximum(influence[idx], contrib)
        
        # Sort positions by influence (high influence = bad to increase)
        # We want to move mass from high-influence to low-influence positions
        order = np.argsort(influence)
        
        improved = False
        stall_this_round = 0
        max_attempts = 200
        
        for attempt in range(max_attempts):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick source (high influence) and destination (low influence)
            src_candidates = order[-50:]  # top 50 by influence
            dst_candidates = order[:50]   # bottom 50 by influence
            
            src = int(rng.choice(src_candidates))
            dst = int(rng.choice(dst_candidates))
            
            if src == dst or a[src] < 1e-12:
                continue
            
            # Try several transfer amounts
            for frac in [0.001, 0.003, 0.01, 0.03, 0.08]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial /= s_t
                
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    stall_this_round = 0
                    break
                else:
                    # Undo
                    a_trial[src] += amt
                    a_trial[dst] -= amt
            
            if not improved:
                stall_this_round += 1
                if stall_this_round > 30:
                    break
        
        if not improved:
            stall += 1
            # Escape: larger random perturbation
            n_pert = max(3, n // 200)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.02, 0.1)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a /= a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            break
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a /= a.sum()
        
        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Influence CD ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):

    """

    time_budget = min(float(budget_s) - 60.0, 1740.0)

    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')

        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

        if s < 0.01:
            return

def get_lp_solver():

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def smooth_subgrad(a, ac, top_k, beta):
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    weights = np.exp(beta * (ac[top_idx] - M))
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(k_max) / k_max
    else:
        weights = weights / w_sum
    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += w * a[j_idx]
    return g

def exact_subgrad(a, ac, threshold):
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n), 1
    active = np.where(ac >= threshold * M)[0]
    if len(active) == 0:
        active = np.array([int(ac.argmax())])
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += a[j_idx]
    g /= len(active)
    return g, len(active)

def mirror_step(a, g, eta):
    log_a = np.log(np.maximum(a, 1e-15)) - eta * g
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def fw_lp_direction(a, ac, top_k, solver):
    n = len(a)
    if n > 3500:
        return None
    M = ac.max()
    if M < 1e-15:
        return None
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(b_var) == 1]
    for k in top_idx:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        coeff = np.zeros(n)
        for ii in range(len(idx)):
            coeff[j_idx[ii]] += a[idx[ii]]
        constraints.append(coeff @ b_var <= t_var + ac[k_int])
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
