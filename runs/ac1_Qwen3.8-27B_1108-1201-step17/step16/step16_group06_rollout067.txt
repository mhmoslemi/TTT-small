def get_solver():

def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    f = np.asarray(f, dtype=np.float64)
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_rows(y, active_set, n):
    """Return (csr_mat, rhs) for constraints (y*b)_k <= M, k in active_set."""
    import scipy.sparse as sps
    klist = sorted(active_set)
    m = len(klist)
    rows = np.empty((m, n), dtype=object)
    lil = sps.lil_matrix((m, n), dtype=np.float64)
    for ri, k in enumerate(klist):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        pos = k - js
        vals = y[js]
        lil[ri, pos] = vals
    mat = lil.tocsr()
    return mat

def smoothed_lbgfs(a0, epsilon=0.01, max_iters=400, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    L = 2 * n - 1

    def obj_grad(z_vec):
        z_max = z_vec.max()
        ez = np.exp(z_vec - z_max)
        av = ez / ez.sum()
        c = fast_autocorr(av)
        cmax = c.max()
        expc = np.exp((c - cmax) / epsilon)
        lse = cmax + epsilon * np.log(expc.sum() + 1e-300)
        f = 2.0 * n * lse
        p = expc / (expc.sum() + 1e-300)
        # grad_a[i] = 4n * sum_m av[m] * p[i+m]  (cross-correlation)
        conv_ar = fast_conv(av, p[::-1])
        idx = L - 1 - np.arange(n)
        grad_a = 4.0 * n * conv_ar[idx]
        sga = np.dot(grad_a, av)
        grad_z = av * (grad_a - sga)
        return f, grad_z

    z = np.log(np.maximum(a, 1e-10)).astype(np.float64)
    from scipy.optimize import minimize
    res = minimize(obj_grad, z, method='L-BFGS-B', jac=True,
                   options={'maxiter': max_iters, 'maxfun': max_iters * 4, 'disp': False})
    z_best = res.x
    z_max = z_best.max()
    a_res = np.exp(z_best - z_max)
    a_res = norm_sum1(a_res)
    sc = 2.0 * n * fast_autocorr(a_res).max()
    if sc < best_score:
        best_score = sc
        best_a = a_res
    if verbose:
        print("    L-BFGS (%.1fs): score=%.6f" % (time.monotonic() - t0, best_score))
    return best_a, best_score


def mirror_descent(a0, time_limit, top_k=30, verbose=False, max_iters=1000000):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    x = a.copy()
    x_prev = a.copy()
    gamma = 4.0
    stall = 0
    max_stall = 2000
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy if sy > 1e-15 else x
        ac = fast_autocorr(y)
        kmax = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:kmax]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** 5.0
        ws = w.sum()
        w = w / ws if ws > 1e-30 else np.ones(kmax) / kmax
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i0 = max(0, ki - n + 1)
            i1 = min(n, ki + 1)
            if i0 >= i1:
                continue
            idx = np.arange(i0, i1)
            grad[idx] += 2.0 * wi * a[ki - idx]
        eta = gamma / (np.sqrt(t) + 1.0)
        loga = np.log(y) - eta * grad
        loga = loga - loga.max()
        an = np.exp(loga)
        an = np.maximum(an, 1e-30)
        sn = an.sum()
        an = an / sn if sn > 1e-15 else y
        x_prev = x
        x = an
        mn = fast_autocorr(x).max()
        if mn < best_m - 1e-14:
            best_m = mn
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        if stall > max_stall:
            stall = 0
            per = rng.normal(0, 0.02, n)
            xr = np.maximum(best_a + per * best_a, 1e-10)
            xr = xr / xr.sum()
            x = xr
            x_prev = xr
    return best_a, best_m

            continue
        ai = interpolate_sequence(best_seq, nt)
        tper_mirror = min(rem * 0.12, 30.0)
        am, _ = mirror_descent(ai, tper_mirror, top_k=40)
        upd(am, "md_n%d" % nt)
        rem = time_budget - (time.monotonic() - t_global)
        if rem < 50:
            break
        tper_lp = min(rem * 0.25, 120.0)
        al, _ = lp_fw_cutting(best_seq if len(best_seq) == nt else ai,
                              tper_lp, solver, verbose=False)
        upd(al, "cp_n%d" % nt)
        tried += 1
        if tried >= 4:
            break

    # Phase 6: final polish (L-BFGS + short LP)
    rem = time_budget - (time.monotonic() - t_global)
    if rem > 20:
        nc = len(best_seq)
        ao, _ = smoothed_lbgfs(best_seq, epsilon=0.002, max_iters=200)
        upd(ao, "lbfgs_final_n%d" % nc)
    rem = time_budget - (time.monotonic() - t_global)
    if rem > 40:
        nc = len(best_seq)
        t6 = min(rem * 0.7, 180.0)
        ao, _ = lp_fw_cutting(best_seq, t6, solver, verbose=False)
        upd(ao, "cp_final_n%d" % nc)

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print("FINAL score=%.6f n=%d t=%.0fs" %
              (best_score, len(best_seq), time.monotonic() - t_global))
    else:
        print("ERROR no sequence")
        return [1.0]
    return [float(v) for v in best_seq]
