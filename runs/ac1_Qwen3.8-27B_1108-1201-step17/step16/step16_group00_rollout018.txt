def coordinate_polish_v2(a, t_limit):
    n = len(a)
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < t_limit:
        ac = fast_autocorr(a)
        k_star = ac.argmax()
        
        # Find contributing pairs for lag k_star
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[k_star - idx]
        
        # For each contributing position i, compute the "sensitivity"
        # of moving mass from i to each position j
        # Sensitivity[j] = sum over top lags of a[lag - j] contribution
        top_lags = np.argsort(ac)[::-1][:20]
        
        # For each top lag, compute the vector of contributions per position
        sens = np.zeros(n)
        for k in top_lags:
            ki = int(k)
            for j in range(n):
                m1 = ki - j
                m2 = j - ki
                if 0 <= m1 < n:
                    sens[j] += a[m1]
                if 0 <= m2 < n:
                    sens[j] += a[m2]
        
        # Best source: highest contribution to peak
        src = idx[np.argmax(contrib)]
        # Best dest: lowest sensitivity (among positions not contributing to peak)
        in_peak = np.zeros(n, dtype=bool)
        in_peak[i_start:i_end+1] = True
        candidates = np.where(~in_peak)[0]
        if len(candidates) == 0:
            candidates = np.arange(n)
        dst = candidates[np.argmin(sens[candidates])]

        for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:

            m_try = fast_autocorr(a_try).max()
            if m_try < best_m:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_sensitivity(a, top_lags):

            a_idx = np.arange(j2_start, j2_end + 1) - k

def mirror_descent_step(a, ac, top_k, beta, eta):

    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s_new = a_new.sum()
    if s_new < 1e-15:
        a_new = a.copy()
    else:
        a_new = a_new / s_new
    return a_new

            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")
    return best_a, best_m


def coordinate_descent_v2(a0, time_limit, verbose=False, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    ac = fast_autocorr(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    n_top_lags = 30
    n_top_src = 15
    stall = 0
    max_stall = 200
    iter_count = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iter_count += 1
        if iter_count % 100 == 0:
            ac = fast_autocorr(a)
            M_curr = ac.max()
            if M_curr < best_m - 1e-15:
                best_m = M_curr
                best_a = a.copy()

        k_star = int(ac.argmax())
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        if i_start > i_end:
            stall += 1
            continue
        idx = np.arange(i_start, i_end + 1)
        src_scores = a[idx] * a[k_star - idx]
        top_src_local = np.argsort(src_scores)[::-1][:n_top_src]
        top_src = idx[top_src_local]

        top_lags = np.argsort(ac)[::-1][:n_top_lags]
        sens = compute_sensitivity(a, top_lags[:15])

        improved = False
        for src in top_src:
            src = int(src)
            if a[src] < 1e-10:
                continue
            partner = k_star - src
            mask = np.ones(n, dtype=bool)
            mask[src] = False
            if 0 <= partner < n:
                mask[partner] = False

            valid_cost = np.where(mask, sens, 1e10)
            best_dst = int(np.argmin(valid_cost))
            if not mask[best_dst]:
                continue

            a_k_src = a[k_star - src] if 0 <= k_star - src < n else 0.0
            a_k_dst = a[k_star - best_dst] if 0 <= k_star - best_dst < n else 0.0
            reduction = a_k_src - a_k_dst
            if reduction <= 1e-15:
                continue

            t_max = a[src] * 0.9
            for k in top_lags[:10]:
                k = int(k)
                if k == k_star:
                    continue
                d_dst = a[k - best_dst] if 0 <= k - best_dst < n else 0.0
                d_src = a[k - src] if 0 <= k - src < n else 0.0
                delta = d_dst - d_src
                if delta > 1e-15:
                    slack = ac[k_star] - ac[k]
                    if slack > 1e-15:
                        t_max = min(t_max, 0.7 * slack / (2.0 * delta))

            t = max(t_max, 0.05 * a[src])
            t = min(t, a[src] * 0.95)

            a[src] -= t
            a[best_dst] += t
            a = np.maximum(a, 0.0)
            a /= a.sum()

            new_ac = fast_autocorr(a)
            new_M = new_ac.max()
            if new_M < best_m - 1e-15:
                best_m = new_M
                best_a = a.copy()
                ac = new_ac
                improved = True
                stall = 0
                break
            else:
                a[src] += t
                a[best_dst] -= t
                a = np.maximum(a, 0.0)
                a /= a.sum()

        if not improved:
            stall += 1
            if stall > 30:
                n_pert = max(5, n // 50)
                pert_idx = rng.choice(n, size=n_pert, replace=False)
                for pi in pert_idx:
                    if a[pi] > 1e-8:
                        j = int(rng.integers(n))
                        if j != pi:
                            frac = rng.uniform(0.001, 0.01)
                            amt = a[pi] * frac
                            a[pi] -= amt
                            a[j] += amt
                a = np.maximum(a, 0.0)
                a /= a.sum()
                new_M = fast_autocorr(a).max()
                if new_M < best_m:
                    best_m = new_M
                    best_a = a.copy()
                    stall = 0

        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def fine_polish(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        k_star = int(ac.argmax())
        M_curr = ac[k_star]

        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        if i_start > i_end:
            break
        idx = np.arange(i_start, i_end + 1)
        src_scores = a[idx] * a[k_star - idx]

        top_lags = np.argsort(ac)[::-1][:20]
        sens = compute_sensitivity(a, top_lags[:10])

        improved = False
        n_try = min(10, len(idx))
        order = np.argsort(src_scores)[::-1][:n_try]

        for oi in order:
            src = int(idx[oi])
            if a[src] < 1e-12:
                continue
            partner = k_star - src
            mask = np.ones(n, dtype=bool)
            mask[src] = False
            if 0 <= partner < n:
                mask[partner] = False

            valid_cost = np.where(mask, sens, 1e10)
            candidates = np.argsort(valid_cost)[:5]
            for dst in candidates:
                dst = int(dst)
                if not mask[dst]:
                    continue
                a_k_src = a[k_star - src] if 0 <= k_star - src < n else 0.0
                a_k_dst = a[k_star - dst] if 0 <= k_star - dst < n else 0.0
                reduction = a_k_src - a_k_dst
                if reduction <= 1e-15:
                    continue
                for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
                    t = a[src] * frac
                    if t < 1e-15:
                        continue
                    a_try = a.copy()
                    a_try[src] -= t
                    a_try[dst] += t
                    a_try = np.maximum(a_try, 0.0)
                    a_try /= a_try.sum()
                    M_try = fast_autocorr(a_try).max()
                    if M_try < best_m - 1e-16:
                        a = a_try
                        best_m = M_try
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.0005, 0.003)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a /= a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
            else:
                break

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    FinePolish ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def make_raised_cosine(n, alpha=1.0):
    x = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x + 0.5) / n)) ** alpha
    return norm_sum1(a)


def make_rimmed(n, rim_width=0.15, rim_height=0.3, ripple_freq=3.0, ripple_amp=0.1):
    x = np.arange(n, dtype=np.float64) / n
    base = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    rim = np.exp(-((x - rim_width) ** 2) / (0.005)) + np.exp(-((x - (1 - rim_width)) ** 2) / (0.005))
    ripple = 1.0 + ripple_amp * np.cos(2.0 * np.pi * ripple_freq * x)
    a = base * ripple + rim_height * rim
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_gaussian_window(n, sigma=0.3, mod_freq=0.0, mod_amp=0.0):
    x = np.linspace(0, 1, n)
    center = 0.5
    a = np.exp(-((x - center) ** 2) / (2 * sigma ** 2))
    if mod_freq > 0:
        a *= (1.0 + mod_amp * np.cos(2 * np.pi * mod_freq * x))
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_multi_harmonic(n, harmonics=None):
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    if harmonics is None:
        harmonics = [(1, 0.15, 0.0), (2, 0.1, np.pi/3), (3, 0.05, np.pi/6)]
    for freq, amp, phase in harmonics:
        a += amp * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_raised_cosine(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Phase 1: Multi-initialization AMD
    print(f"\n=== Phase 1: Multi-init AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.20, 300.0)

    if t_phase1 > 30:
        inits = []
        labels = []
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        inits.append(make_raised_cosine(n0))
        labels.append("raised_cos")
        inits.append(make_rimmed(n0, rim_width=0.12, rim_height=0.25, ripple_freq=4.0, ripple_amp=0.08))
        labels.append("rimmed4")
        inits.append(make_rimmed(n0, rim_width=0.2, rim_height=0.2, ripple_freq=2.0, ripple_amp=0.12))
        labels.append("rimmed2")
        inits.append(make_gaussian_window(n0, sigma=0.35, mod_freq=5.0, mod_amp=0.1))
        labels.append("gauss_mod")
        inits.append(make_multi_harmonic(n0, [(1, 0.2, 0), (2, 0.12, 1.0), (3, 0.06, 0.5)]))
        labels.append("multi_h2")
        inits.append(make_multi_harmonic(n0, [(1, 0.15, 0.5), (3, 0.1, 0.2), (5, 0.05, 1.0)]))
        labels.append("multi_h5")

        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 50)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(10.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 30:
                break
            t_avail = min(t_per, time_budget - 30 - (time.monotonic() - t_global))
            if t_avail < 8:
                break
            print(f"  AMD '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=50, beta=5.0, c0=4.0,
                                     restart_every=3000, verbose=False)
            try_update(a_out, f"amd_{lbl}")

    # Phase 2: Aggressive coordinate descent (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Coordinate descent (n={n_best}) ===")
    t2 = min(remaining * 0.45, 700.0)
    if t2 > 20:
        t2a = t2 * 0.6
        t2b = t2 * 0.4
        a_in = norm_sum1(best_seq)
        print(f"  Coord2 main ({t2a:.0f}s)...")
        a_out, _ = coordinate_descent_v2(a_in, t2a, verbose=True, rng_seed=42)
        try_update(a_out, f"coord2_main_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15 and best_seq is not None:
            a_in2 = norm_sum1(best_seq)
            print(f"  Coord2 second ({t2b:.0f}s, different seed)...")
            a_out2, _ = coordinate_descent_v2(a_in2, t2b, verbose=True, rng_seed=1234)
            try_update(a_out2, f"coord2_sec_n{n_best}")

    # Phase 3: Fine polishing
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Fine polish (n={n_best}) ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = fine_polish(a_in, t3, verbose=True)
        try_update(a_out, f"fine_n{n_best}")

    # Phase 4: AMD polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: AMD polish (n={n_best}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t4, top_k=60, beta=4.0, c0=3.0,
                                 restart_every=5000, verbose=False)
        try_update(a_out, f"amd_polish_n{n_best}")

    # Phase 5: Length adjustment
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration (current n={n_best}) ===")

    length_tries = []
    for delta in [200, -200, 400, -400]:
        new_n = n_best + delta
        if 1000 <= new_n <= 6000:
            length_tries.append(new_n)

    for new_n in length_tries[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        t5 = min(remaining * 0.3, 150.0)
        if t5 > 20:
            a_out, _ = coordinate_descent_v2(a_long, t5, verbose=False, rng_seed=99)
            try_update(a_out, f"coord_n{new_n}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_out2, _ = fine_polish(np.maximum(a_out, 0.0), min(remaining * 0.3, 60.0), verbose=False)
                try_update(a_out2, f"fine_n{new_n}")

    # Phase 6: Final coordinate descent push
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final coord push (n={n_best}, remaining={remaining:.0f}s) ===")
    if remaining > 20:
        t6 = remaining * 0.7
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_v2(a_in, t6, verbose=True, rng_seed=2718)
        try_update(a_out, f"final_coord_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10 and best_seq is not None:
            a_out2, _ = fine_polish(np.maximum(best_seq, 0.0), remaining * 0.8, verbose=False)
            try_update(a_out2, f"final_fine_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_raised_cosine(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
