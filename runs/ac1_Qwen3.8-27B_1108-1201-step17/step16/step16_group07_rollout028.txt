1. Fast 1D coordinate descent with optimal step (ternary search per coordinate, O(n^2) per pass)
2. LP best-response with Nesterov acceleration for escaping local optima
3. Projected subgradient with dual-variable weighting
4. Multi-restart with spectral perturbations and multi-length exploration
Key insight: objective is scale-invariant, so we minimize max_autocorr directly then normalize.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    """Compute linear autocorrelation (convolution with self) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_to(a, n):
    """Normalize a to have sum = sqrt(2n)."""
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s


def cd_pass(a, conv, rng, n, size):
    """One pass of coordinate descent. Returns improved a and conv."""
    order = rng.permutation(n)
    improved = False

    for j in order:
        lo = j
        hi = min(size, j + n)
        if lo >= hi:
            continue

        # Max of conv outside [lo, hi)
        out_max = -1.0
        if lo > 0:
            out_max = max(out_max, conv[:lo].max())
        if hi < size:
            out_max = max(out_max, conv[hi:].max())

        conv_changed = conv[lo:hi]
        a_shifted = a[0:hi - lo]
        a_j = a[j]

        def f_of_delta(delta, cv=conv_changed, av=a_shifted, om=out_max, jj=j):
            c = cv + 2.0 * delta * av
            if jj < len(c):
                c[jj] += delta * delta
            m = c.max()
            if om > m:
                return om
            return m

        delta_lo = -a_j
        delta_hi = max(a_j * 3.0, np.mean(a) * 15.0)

        f_curr = f_of_delta(0.0)
        f_left = f_of_delta(delta_lo)
        f_right = f_of_delta(delta_hi)

        if f_left < f_curr:
            delta_hi = 0.0
        if f_right < f_curr:
            delta_lo = 0.0

        lo_d = delta_lo
        hi_d = delta_hi

        if hi_d - lo_d > 1e-14:
            for _ in range(35):
                if hi_d - lo_d < 1e-13:
                    break
                m1 = lo_d + (hi_d - lo_d) / 3.0
                m2 = hi_d - (hi_d - lo_d) / 3.0
                f1 = f_of_delta(m1)
                f2 = f_of_delta(m2)
                if f1 < f2:
                    hi_d = m2
                else:
                    lo_d = m1

            best_delta = (lo_d + hi_d) / 2.0
            new_val = a_j + best_delta
            if new_val < 0.0:
                new_val = 0.0

            if abs(new_val - a_j) > 1e-12:
                a[j] = new_val
                # Update conv in place
                delta = new_val - a_j
                conv[lo:hi] += 2.0 * delta * a_shifted
                if 2 * j < size:
                    conv[2 * j] += delta * delta
                improved = True

    return a, conv, improved


def coordinate_descent(a0, time_limit, verbose=False, seed=0, max_passes=200):
    """
    Fast 1D coordinate descent with in-place conv updates.
    Minimizes max(conv(a,a)) (scale-invariant proxy for the true objective).
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')

    size = 2 * n - 1
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 8

    for pass_num in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve >= max_no_improve:
            break

        conv = fast_autocorr(a)
        a, conv, improved = cd_pass(a, conv, rng, n, size)

        new_score = fast_autocorr(a).max()
        if new_score < best_score - 1e-13:
            best_score = new_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                a = best_a.copy()

        if verbose and pass_num % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD pass={pass_num} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def coordinate_descent_squared(a0, time_limit, verbose=False, seed=0, max_passes=100):
    """
    CD that directly minimizes max_conv / sum^2 (the true objective).
    Uses a finer search around the current value.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')

    size = 2 * n - 1
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max() / (s0 * s0)
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 10

    for pass_num in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve >= max_no_improve:
            break

        conv = fast_autocorr(a)
        cur_sum = a.sum()
        order = rng.permutation(n)
        pass_improved = False

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            lo = j
            hi = min(size, j + n)
            if lo >= hi:
                continue

            out_max = -1.0
            if lo > 0:
                out_max = max(out_max, conv[:lo].max())
            if hi < size:
                out_max = max(out_max, conv[hi:].max())

            conv_changed = conv[lo:hi]
            a_shifted = a[0:hi - lo]
            a_j = a[j]
            S_o = cur_sum - a_j

            # Try candidate values for a[j]
            candidates = [0.0, a_j * 0.5, a_j * 0.75, a_j, a_j * 1.25, a_j * 1.5, a_j * 2.0, a_j * 3.0]
            mean_a = cur_sum / n
            candidates.extend([mean_a * 0.3, mean_a, mean_a * 2.0, mean_a * 5.0])
            candidates = [max(0.0, min(c, cur_sum * 0.8)) for c in candidates]
            candidates = list(set([round(c, 10) for c in candidates]))

            best_x = a_j
            best_obj = float('inf')

            for x in candidates:
                if abs(x - a_j) < 1e-10:
                    continue
                delta = x - a_j
                new_sum = S_o + x
                if new_sum < 1e-10:
                    continue

                # Compute new max conv
                c = conv_changed + 2.0 * delta * a_shifted
                if 2 * j >= lo and 2 * j < hi:
                    pos = 2 * j - lo
                    if pos < len(c):
                        c[pos] += delta * delta

                new_max = max(c.max(), out_max) if len(c) > 0 else out_max
                if new_max < 0:
                    new_max = 0.0

                obj = 2.0 * n * new_max / (new_sum * new_sum)
                if obj < best_obj:
                    best_obj = obj
                    best_x = x

            if best_x != a_j:
                delta = best_x - a_j
                a[j] = best_x
                cur_sum = S_o + best_x
                conv[lo:hi] += 2.0 * delta * a_shifted
                if 2 * j < size:
                    conv[2 * j] += delta * delta
                pass_improved = True

        # Check improvement at end of pass
        new_score = 2.0 * n * fast_autocorr(a).max() / (a.sum() ** 2)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                a = best_a.copy()

        if verbose and pass_num % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD2 pass={pass_num} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)

    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()

    stall_count = 0
    stall_threshold = 60
    conv_size = 2 * n - 1

    for t in range(1, 10000000):

            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        if active_set is None or t % 30 == 0:
            progress = min(1.0, t / 800.0)
            tol = 0.06 * (1.0 - progress) + 0.003 * progress
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            n_top = max(5, n // 60)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            if len(active_set) > int(1.2 * n):

                keep = np.argsort(conv_at_active)[-int(1.2 * n):]
                active_set = active_set[keep]

        b, duals = lp_best_response(y, M_curr, active_set, solver)

        if b is None:
            if t % 5 == 0:
                active_set = np.argsort(conv_y)[-int(0.5 * n):]

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

            if s_a < 1e-15:

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:

        score_a = fast_autocorr(a).max()

def subgradient_polish(a0, time_limit, verbose=False, seed=0):
    """Projected subgradient with exponential peak weighting."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 2.0 / np.sqrt(n)
    patience = 0
    max_patience = 500
    beta = 5.0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Weight peaks by their height
        tol = 0.03
        active = np.where(conv >= M_curr * (1.0 - tol))[0]
        if len(active) < 3:
            active = np.argsort(conv)[-max(3, n // 50):]

        weights = np.exp(-beta * (M_curr - conv[active]) / max(M_curr * 0.01, 1e-10))
        weights = weights / weights.sum()

        # Compute subgradient: average of shifted a over active peaks
        grad = np.zeros(n)
        for w, k in zip(weights, active):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                grad[lo:hi] += w * 2.0 * a[k - np.arange(lo, hi)]

        # Project gradient onto the constraint space (sum=0 direction)
        grad -= np.mean(grad)

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        a_trial = a - step * grad * np.sqrt(n)
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.003, 3.0)
        else:
            patience += 1
            step *= 0.99
            if step < 1e-12:
                break

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}")

    return best_a, best_score


def perturb_sequence(a, scale, rng):
    """Perturb a sequence with multiplicative noise."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    perturbed = a * noise
    s = perturbed.sum()
    if s < 1e-15:
        return a.copy()
    return perturbed / s


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence."""
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False

        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)
    print(f"Starting optimization with n={n_base}, score={best_score:.8f}")

    # === Phase 1: Fast CD (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 60:
        print(f"\n=== Phase 1: Fast CD n={n_base}, budget={t1:.0f}s ===")
        a_out, _ = coordinate_descent(best_seq, t1, verbose=True, seed=0)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd1_n{n_base}", t_global)

    # === Phase 2: CD with squared objective ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        print(f"\n=== Phase 2: CD-squared n={n_cur}, budget={t2:.0f}s ===")
        a_out, _ = coordinate_descent_squared(best_seq, t2, verbose=True, seed=1)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd2_n{n_cur}", t_global)

    # === Phase 3: LP Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40:
        print(f"\n=== Phase 3: LP Nesterov n={n_cur}, budget={t3:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, seed=42)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp1_n{n_cur}", t_global)

    # === Phase 4: CD after LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 20:
        print(f"\n=== Phase 4: CD post-LP n={n_cur}, budget={t4:.0f}s ===")
        a_out, _ = coordinate_descent(best_seq, t4, verbose=True, seed=77)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd3_n{n_cur}", t_global)

    # === Phase 5: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        print(f"\n=== Phase 5: Subgradient n={n_cur}, budget={t5:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = subgradient_polish(a_norm, t5, verbose=True, seed=3)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"subg_n{n_cur}", t_global)

    # === Phase 6: Multi-restart with perturbations ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_restarts = 3
    t6_total = min(remaining * 0.25, 250.0)
    t6_each = t6_total / n_restarts

    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        t6 = min(t6_each, remaining - 30)
        if t6 < 30:
            break

        print(f"\n=== Phase 6r{r}: Perturb+CD n={n_cur}, budget={t6:.0f}s ===")
        scale = 0.02 + 0.02 * r
        perturbed = perturb_sequence(best_seq, scale, rng)
        a_out, _ = coordinate_descent(perturbed, t6 * 0.6, verbose=False, seed=100 + r)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"pert{r}_cd", t_global)

        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 60 and not best_score >= float('inf'):
            a_norm = normalize_to(best_seq, len(best_seq))
            a_out2, _ = nesterov_lp_optimize(a_norm, min(remaining2 * 0.4, 80.0), solver, verbose=False, seed=200 + r)
            best_seq, best_score, _ = try_update(best_seq, best_score, a_out2, f"pert{r}_lp", t_global)

    # === Phase 7: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 7: Multi-length exploration ===")
        n_cur = len(best_seq)
        for n_try in [n_cur - 200, n_cur + 200, n_cur - 400]:
            if n_try < 100:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t7 = min(remaining * 0.3, 80.0)
            # Interpolate parent to new length
            x_old = np.linspace(0, 1, n_cur)
            x_new = np.linspace(0, 1, n_try)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = np.maximum(a_interp, 0.0)
            s = a_interp.sum()
            if s > 1e-10:
                a_interp = a_interp / s
                a_out, _ = coordinate_descent(a_interp, t7, verbose=False, seed=300)
                best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"len{n_try}", t_global)

    # === Phase 8: Final high-precision CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final CD n={n_cur} ===")
        t8 = min(remaining * 0.6, 100.0)
        a_out, _ = coordinate_descent(best_seq, t8, verbose=False, seed=999)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "cdFinal", t_global)

    # === Phase 9: Final CD-squared polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t9 = min(remaining * 0.5, 60.0)
        a_out, _ = coordinate_descent_squared(best_seq, t9, verbose=False, seed=1234)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "cd2Final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
