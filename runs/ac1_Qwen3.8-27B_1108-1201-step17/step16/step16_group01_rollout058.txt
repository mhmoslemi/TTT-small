Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FOM (Fast Subgradient Method) with adaptive active-set and backtracking
2. Peak deflation via targeted gradient steps on the active lag
3. Vectorized batch coordinate descent

5. LP-based Frank-Wolfe polishing
6. Multi-restart strategy with progressive refinement

import numpy as np
import time
import math

def fast_autocorr(a):
    """Compute full autocorrelation (linear convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def eval_obj(a):
    """Compute the objective 2*n*max(conv)/(sum)^2 for normalized a (sum=1)."""
    ac = fast_autocorr(a)
    return 2.0 * len(a) * ac.max()

def get_subgradient_active(a, ac, threshold=0.99):
    """Compute subgradient from lags near the maximum."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    active = np.where(ac >= threshold * M)[0]
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += a[j_idx]
    return 2.0 * g

def gradient_single_lag(a, k):
    """Gradient of conv(a,a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    idx = np.arange(i_start, i_end)
    j_idx = k - idx
    g[idx] = 2.0 * a[j_idx]
    return g

def fom_optimize(a0, time_limit, c=3.0, threshold=0.99, verbose=True, seed=42,
                 restart_stall=3000, use_backtrack=True):
    """
    Fast Subgradient Method with:
    - Nesterov acceleration
    - Exact active-set subgradient
    - Adaptive step size with backtracking
    - Restart on stagnation
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_val = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    t_acc = 1.0
    stall = 0
    c_curr = c
    thresh_curr = threshold
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov extrapolation
        beta = (t_acc - 1.0) / (t_acc + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Subgradient
        ac_y = fast_autocorr(y)
        g = get_subgradient_active(y, ac_y, threshold=thresh_curr)
        
        # Step size
        eta = c_curr * math.sqrt(n) / (t_acc + 1.0)
        
        # Mirror (log-space) update
        log_y = np.log(np.maximum(y, 1e-15))
        log_new = log_y - eta * g
        max_log = log_new.max()
        a_new = np.exp(log_new - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        t_acc = (1.0 + math.sqrt(1.0 + 4.0 * t_acc * t_acc)) / 2.0
        
        # Evaluate
        M_new = fast_autocorr(x).max()
        if M_new < best_val - 1e-15:
            best_val = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart logic
        if stall > restart_stall:
            stall = 0
            # Perturb from best
            noise = rng.normal(0, 0.008, n) * best_a
            x = np.maximum(best_a + noise, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            t_acc = 1.0
            # Adapt parameters
            c_curr *= 1.3
            if c_curr > 15.0:
                c_curr = c + rng.uniform(0, 3)
            if thresh_curr > 0.97:
                thresh_curr -= 0.005
            if verbose:
                print(f"    FOM restart (it={it}, t={time.monotonic()-t0:.0f}s): c={c_curr:.2f}, thresh={thresh_curr:.4f}")
        
        if verbose and it % 10000 == 0:

def peak_deflation(a0, time_limit, verbose=True, seed=123, num_rounds=100):
    """
    Specifically target the peak lag: repeatedly compute gradient of the
    active lag and take small steps to reduce it, while checking other lags.
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_val = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    for round_idx in range(num_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))
        
        # Gradient of the peak lag
        g = gradient_single_lag(a, peak_k)
        
        # Try several step sizes
        improved = False
        for eta in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_trial = a - eta * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            ac_trial = fast_autocorr(a_trial)
            M_trial = ac_trial.max()
            if M_trial < best_val - 1e-14:
                a = a_trial
                best_val = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try moving mass from peak contributors to random positions
            peak_lags = np.argsort(ac)[::-1][:5]
            for k in peak_lags:
                k_int = int(k)
                i_start = max(0, k_int - n + 1)
                i_end = min(n, k_int + 1)
                idx = np.arange(i_start, i_end)
                j_idx = k_int - idx
                contrib = a[idx] * a[j_idx]
                
                if contrib.max() < 1e-20:
                    continue
                
                src = idx[np.argmax(contrib)]
                # Move to a random low-contribution position
                for _ in range(5):
                    dst = int(rng.integers(n))
                    if dst == src or a[dst] > a[src] * 0.5:
                        continue
                    for frac in [0.005, 0.01, 0.02]:
                        amt = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s = a_trial.sum()
                        if s < 1e-15:
                            continue
                        a_trial = a_trial / s
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_val - 1e-14:
                            a = a_trial
                            best_val = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if verbose and round_idx % 20 == 0:
            print(f"    Deflate r={round_idx} ({time.monotonic()-t0:.0f}s): obj={2*n*best_val:.6f}")
    
    return best_a, best_val

def batch_coord_descent(a0, time_limit, verbose=True, seed=777):
    """
    Vectorized batch coordinate descent:
    - Identify all "hot" positions (involved in top-10 peaks)
    - For each hot position, precompute the best "cold" destinations
    - Apply multiple small transfers in a batch
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 80:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:min(10, len(ac))]
        
        # Identify hot positions
        hot_mask = np.zeros(n, dtype=bool)
        for k in top_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            hot_mask[i_start:i_end] = True
        
        hot_idx = np.where(hot_mask)[0]
        cold_idx = np.where(~hot_mask)[0]
        
        if len(hot_idx) == 0 or len(cold_idx) == 0:
            break
        
        # For each hot position, compute its "peak sensitivity"
        peak_sens = np.zeros(n)
        for k in top_lags[:5]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            peak_sens[idx] += a[j_idx]
        
        # Sort cold positions by sensitivity (lowest first)
        cold_sorted = cold_idx[np.argsort(peak_sens[cold_idx])]
        n_cold = min(len(cold_sorted), 50)
        
        improved = False
        n_hot_check = min(len(hot_idx), 30)
        hot_sample = rng.choice(hot_idx, size=n_hot_check, replace=False) if len(hot_idx) > n_hot_check else hot_idx
        
        for src in hot_sample:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            
            # Try transfers to the lowest-sensitivity cold positions
            for dst in cold_sorted[:n_cold]:
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.005]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_val - 1e-14:
                        a = a_trial
                        best_val = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if improved:
                stall = 0
        
        if not improved:
            stall += 1
            # Escape: random perturbation
            n_pert = max(5, n // 100)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.02, 0.06)
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            if a.sum() > 1e-15:
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_val:
                    best_val = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 10 == 0:
            print(f"    Coord ({time.monotonic()-t0:.0f}s): obj={2*n*best_val:.6f}, stall={stall}")
    
    return best_a, best_val

def spectral_init(n, mode=0, seed=0):
    """Generate diverse spectral initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    
    if mode == 0:
        # Asymmetric sine
        a = np.sin(np.pi * x) ** 2 + 0.3 * np.sin(2 * np.pi * x + 0.5) ** 2
    elif mode == 1:
        # Asymmetric cosine
        a = 0.5 + 0.4 * np.cos(2 * np.pi * x - 1.0)
    elif mode == 2:
        # Two-lobe
        a = np.exp(-((x - 0.3) / 0.2) ** 2) + np.exp(-((x - 0.7) / 0.15) ** 2)
    elif mode == 3:
        # Linear ramp + noise
        a = 0.5 + 0.5 * x + rng.normal(0, 0.1, n)
    elif mode == 4:
        # Sine with random phase
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.4 * np.sin(2 * np.pi * x + phase)
    elif mode == 5:
        # Cosine with random frequency
        freq = rng.uniform(1.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * freq * x + phase)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.01)
    return a / a.sum()

def interpolate_to_length(a, new_n):
    """Resample a sequence to a new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_resampled = np.interp(x_new, x_old, a)
    return norm_sum1(a_resampled)

def lp_polish(a0, time_limit, verbose=True, max_iters=30):
    """LP-based Frank-Wolfe for final polishing."""
    try:
        import cvxpy as cp
    except ImportError:
        return a0.copy(), fast_autocorr(a0).max()
    
    n = len(a0)
    if n > 2500:
        return a0.copy(), fast_autocorr(a0).max()
    
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Find a solver
    solver = None
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob_test = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob_test.solve(solver=s, verbose=False)
            if prob_test.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:min(30, len(ac))]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(b_var) == 1]
        
        for k in top_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            coeff = np.zeros(n)
            for ii, j in zip(idx, j_idx):
                coeff[int(j)] += y[int(ii)]
            constraints.append(coeff @ b_var <= M_curr)
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_val:
            best_val = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_val:
            best_val = mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            print(f"    LP it={t} ({time.monotonic()-t0:.0f}s): obj={2*n*best_val:.6f}")
    
    return best_a, best_val

import numpy as np
import time
import math


def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def eval_obj_normalized(a):
    """Compute objective for normalized (sum=1) sequence."""
    ac = fast_autocorr(a)
    return 2.0 * len(a) * ac.max()


def get_subgradient_active(a, ac, threshold=0.99):
    """Compute subgradient from lags near the maximum (active set)."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    active = np.where(ac >= threshold * M)[0]
    if len(active) == 0:
        active = np.array([int(ac.argmax())])
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += a[j_idx]
    return 2.0 * g


def get_soft_subgradient(a, ac, top_k=30, beta=4.0):
    """Soft subgradient using top-k lags with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * wi * a[j_idx]
    return g


def gradient_single_lag(a, k):
    """Gradient of conv(a,a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    idx = np.arange(i_start, i_end)
    j_idx = k - idx
    g[idx] = 2.0 * a[j_idx]
    return g


def fom_logspace(a0, time_limit, c=3.0, threshold=0.99, verbose=True, seed=42,
                 restart_stall=3000):
    """FOM with log-space (mirror) update."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    t_acc = 1.0
    stall = 0
    c_curr = c
    thresh_curr = threshold
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        beta = (t_acc - 1.0) / (t_acc + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac_y = fast_autocorr(y)
        g = get_subgradient_active(y, ac_y, threshold=thresh_curr)

        eta = c_curr * math.sqrt(n) / (t_acc + 1.0)
        log_y = np.log(np.maximum(y, 1e-15))
        log_new = log_y - eta * g
        max_log = log_new.max()
        a_new = np.exp(log_new - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new
        t_acc = (1.0 + math.sqrt(1.0 + 4.0 * t_acc * t_acc)) / 2.0

        M_new = fast_autocorr(x).max()
        if M_new < best_val - 1e-15:
            best_val = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_stall:
            stall = 0
            noise = rng.normal(0, 0.01, n) * best_a
            x = np.maximum(best_a + noise, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            t_acc = 1.0
            c_curr *= 1.3
            if c_curr > 20.0:
                c_curr = c + rng.uniform(0, 5)
            if thresh_curr > 0.97:
                thresh_curr = max(0.97, thresh_curr - 0.005)
            if verbose:
                print(f"    FOM-log restart it={it} ({time.monotonic()-t0:.0f}s): c={c_curr:.2f}, th={thresh_curr:.4f}")

        if verbose and it % 20000 == 0:
            print(f"    FOM-log it={it} ({time.monotonic()-t0:.0f}s): obj={2.0*n*best_val:.6f}")

    return best_a, best_val

def spectral_flatten(a0, time_limit, verbose=True, seed=55, damping=0.5):
    """
    Spectral flattening: iteratively apply a filter that flattens |FFT(a)|^2.
    This pushes the autocorrelation toward being as flat (delta-like) as possible.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_len = 2 * n

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_len)
        power = np.abs(A) ** 2 + 1e-12

        # Flattening filter: reduce peaks in spectrum
        target = 1.0 / np.sqrt(power)
        # Add small random perturbation to break symmetry
        if it % 100 == 0:
            target += rng.normal(0, 0.001, len(target))
            target = np.maximum(target, 0)

        A_new = A * target
        a_new = np.fft.irfft(A_new, fft_len)[:n]

        # Project onto simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        # Damped update
        a = damping * a + (1.0 - damping) * a_new
        a = norm_sum1(a)

        M_new = fast_autocorr(a).max()
        if M_new < best_val - 1e-15:
            best_val = M_new
            best_a = a.copy()

        if verbose and it % 2000 == 0:
            print(f"    Spectral it={it} ({time.monotonic()-t0:.0f}s): obj={2.0*n*best_val:.6f}")

    return best_a, best_val


def peak_deflation(a0, time_limit, verbose=True, seed=123, max_rounds=500):
    """
    Targeted peak deflation: repeatedly apply gradient of the single peak lag,
    with multiple step sizes and mass-transfer fallbacks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    for r in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))

        # Try gradient step on the peak lag
        g = gradient_single_lag(a, peak_k)
        improved = False

        for eta in [0.005, 0.002, 0.001, 0.0005]:
            a_trial = a - eta * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_val - 1e-14:
                a = a_trial
                best_val = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            # Mass transfer from peak contributors
            top_lags = np.argsort(ac)[::-1][:5]
            for k in top_lags:
                if improved:
                    break
                k_int = int(k)
                i_start = max(0, k_int - n + 1)
                i_end = min(n, k_int + 1)
                idx = np.arange(i_start, i_end)
                j_idx = k_int - idx
                contrib = a[idx] * a[j_idx]
                if contrib.max() < 1e-20:
                    continue
                src = idx[np.argmax(contrib)]
                for _ in range(8):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    for frac in [0.005, 0.01, 0.02, 0.015]:
                        amt = a[src] * frac
                        if amt < 1e-18:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s = a_trial.sum()
                        if s < 1e-15:
                            continue
                        a_trial = a_trial / s
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_val - 1e-14:
                            a = a_trial
                            best_val = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break

        if verbose and r % 50 == 0:
            print(f"    Deflate r={r} ({time.monotonic()-t0:.0f}s): obj={2.0*n*best_val:.6f}")

    return best_a, best_val


def batch_coord_descent(a0, time_limit, verbose=True, seed=777, max_stall=100):
    """
    Vectorized batch coordinate descent with sensitivity-based targeting.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_lags = np.argsort(ac)[::-1][:min(10, len(ac))]

        # Identify hot positions
        hot_mask = np.zeros(n, dtype=bool)
        for k in top_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            hot_mask[i_start:i_end] = True

        hot_idx = np.where(hot_mask)[0]
        cold_idx = np.where(~hot_mask)[0]

        if len(hot_idx) == 0 or len(cold_idx) < 5:
            stall += 1
            continue

        # Compute peak sensitivity for each position
        peak_sens = np.zeros(n)
        for k in top_lags[:5]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            peak_sens[idx] += a[j_idx]

        # Sort cold positions by sensitivity
        cold_sorted = cold_idx[np.argsort(peak_sens[cold_idx])]
        n_cold = min(len(cold_sorted), 40)

        # Sample hot positions
        n_hot = min(len(hot_idx), 25)
        if len(hot_idx) > n_hot:
            hot_sample = rng.choice(hot_idx, size=n_hot, replace=False)
        else:
            hot_sample = hot_idx

        improved = False
        for src in hot_sample:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue

            # Target the lowest-sensitivity cold positions
            for dst in cold_sorted[:n_cold]:
                if improved:
                    break
                for frac in [0.01, 0.02, 0.005, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_val - 1e-14:
                        a = a_trial
                        best_val = M_trial
                        best_a = a.copy()
                        improved = True
                        break

        if improved:
            stall = 0
        else:
            stall += 1
            # Escape: batch random perturbation
            n_pert = max(5, n // 80)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            if a.sum() > 1e-15:
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_val:
                    best_val = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 15 == 0:
            print(f"    Coord ({time.monotonic()-t0:.0f}s): obj={2.0*n*best_val:.6f}, stall={stall}")

    return best_a, best_val


def simulated_annealing_wrap(a0, time_limit, verbose=True, seed=222,
                              T_init=0.02, T_min=0.003, cooling=0.9995):
    """
    Simulated annealing: occasionally accept worse solutions to escape local minima.
    Alternates between FOM steps and random perturbations with annealing schedule.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    T = T_init

    while time.monotonic() - t0 < time_limit and T > T_min:
        # Do a few FOM-like steps
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            ac = fast_autocorr(a)
            g = get_subgradient_active(a, ac, threshold=0.99)
            eta = 5.0 * math.sqrt(n) / math.sqrt(_ + 100.0)
            log_a = np.log(np.maximum(a, 1e-15)) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
            a = a_new

        M_curr = fast_autocorr(a).max()
        if M_curr < best_val - 1e-15:
            best_val = M_curr
            best_a = a.copy()

        # Perturbation step
        n_pert = max(3, n // 200)
        for _ in range(n_pert):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                amt = a[i] * rng.uniform(T, T * 3)
                a[i] -= amt
                a[j] += amt
        a = np.maximum(a, 1e-10)
        a = a / a.sum()

        T *= cooling

        if verbose and int((T_init - T) / (T_init - T_min)) % 100 == 0:
            print(f"    SA T={T:.5f} ({time.monotonic()-t0:.0f}s): obj={2.0*n*best_val:.6f}")

    return best_a, best_val


def interpolate_to_length(a, new_n):
    """Resample a sequence to a new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_res = np.interp(x_new, x_old, a)
    return norm_sum1(a_res)

    elif mode == 4:
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.4 * np.sin(2 * np.pi * x + phase)
    elif mode == 5:
        freq = rng.uniform(1.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * freq * x + phase)
    elif mode == 6:
        # Bump with asymmetry
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.3)
        a = np.exp(-((x - center) / width) ** 2) + 0.1
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.01)
    return a / a.sum()


def lp_polish(a0, time_limit, verbose=True, max_iters=30):
    """LP-based Frank-Wolfe for final polishing (only for smaller n)."""
    try:
        import cvxpy as cp
    except ImportError:
        return a0.copy(), fast_autocorr(a0).max()

    n = len(a0)
    if n > 2200:
        return a0.copy(), fast_autocorr(a0).max()

    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()

    solver = None
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            xv = cp.Variable(4, nonneg=True)
            pt = cp.Problem(cp.Minimize(cp.sum(xv)), [cp.sum(xv) == 1])
            pt.solve(solver=s, verbose=False)
            if pt.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS

    y = a.copy()
    x = a.copy()

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:min(40, len(ac))]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(b_var) == 1]

        for k in top_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            coeff = np.zeros(n)
            for ii, j in zip(idx, j_idx):
                coeff[int(j)] += y[int(ii)]
            constraints.append(coeff @ b_var <= M_curr)

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_val:
            best_val = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_val:
            best_val = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            print(f"    LP it={t} ({time.monotonic()-t0:.0f}s): obj={2.0*n*best_val:.6f}")

    return best_a, best_val


def evaluate_safe(a):
    """Safely evaluate the sequence using the global evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
