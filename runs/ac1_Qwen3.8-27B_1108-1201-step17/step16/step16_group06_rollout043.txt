    print("  LP solver (fallback): ECOS")
    return _global_solver

def fast_autocorr(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum 1, ensuring non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def build_conv_matrix(a):
    """Build matrix T where (T @ b)[k] = conv(a, b)[k]."""
    n = len(a)
    num_rows = 2 * n - 1
    T = np.zeros((num_rows, n), dtype=np.float64)
    a_arr = np.asarray(a, dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = a_arr[j]
    return T

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return np.asarray(f, dtype=np.float64).copy()
    s = float(np.sum(f))
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = np.asarray(f, dtype=np.float64) / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def gen_exponential(n, alpha):
    """Generate exponential decay sequence."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    return norm_sum1(a)

def gen_power_law(n, p):
    """Generate power-law decay sequence."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = i ** (-p)
    return norm_sum1(a)

def gen_asymmetric_gauss(n, center_frac, width):
    """Generate asymmetric Gaussian."""
    i = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    a = np.exp(-0.5 * ((i - center) / (width * n)) ** 2)
    return norm_sum1(a)

def gen_sine_asym(n, freq, phase):
    """Generate sine-based asymmetric sequence."""
    i = np.arange(n, dtype=np.float64) / n
    a = np.maximum(np.sin(np.pi * freq * i + phase), 0.0)
    a = a ** 2
    return norm_sum1(a)

def lp_frank_wolfe(a0, time_limit, solver_name, verbose=False, max_iters=50000):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    import cvxpy as cp
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    x_prev = a.copy()
    plateau_count = 0
    best_m_plateau = best_m

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        T = build_conv_matrix(y)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            if my < best_m_plateau - 1e-10:
                plateau_count = 0
                best_m_plateau = my
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            if mx < best_m_plateau - 1e-10:
                plateau_count = 0
                best_m_plateau = mx
            best_m = mx
            best_a = x.copy()
        else:
            plateau_count += 1

        if plateau_count > 200 and t > 100:
            break

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}")

    return best_a, best_m

def soft_subgradient_nesterov(a0, time_limit, top_k=64, beta=3.0, verbose=False, gamma=6.0):
    """Nesterov-accelerated mirror descent with top-K soft subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30) - ac.min()
        w = np.exp(beta * top_vals)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            valid = (j_idx >= 0) & (j_idx < n)
            if np.any(valid):
                grad[idx[valid]] += 2.0 * wi * y[j_idx[valid]]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 5000:
            stall_count = 0
            perturbation = rng.normal(0, 0.03, n)
            x_restarted = best_a + perturbation * best_a
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SSG t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def peak_coord_descent(a0, time_limit, verbose=False, rng_seed=123):
    """Peak-targeted coordinate descent with sensitivity analysis."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_idx = int(np.argmax(ac))

        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_idx - idx]

        if contrib.max() < 1e-20:
            stall += 1
            break

        top_peaks = np.argsort(ac)[::-1][:15]
        sensitivity = np.zeros(n, dtype=np.float64)
        for pk in top_peaks:
            pk_int = int(pk)
            for pos in range(n):
                d1 = pk_int - pos
                d2 = pos - pk_int
                if 0 <= d1 < n:
                    sensitivity[pos] += a[d1]
                if 0 <= d2 < n:
                    sensitivity[pos] += a[d2]

        improved = False
        src_ranking = np.argsort(contrib)[::-1]

        for src_rank in src_ranking[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[src_rank])
            candidates = np.argsort(sensitivity)[:max(30, n // 15)]

            for dst in candidates:
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.01, 0.03, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    if a_trial.sum() > 1e-15:
                        a_trial = a_trial / a_trial.sum()
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordPD ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-strategy approach."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1750.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2000
        best_seq = gen_exponential(n_base, 1.0 / n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # ============ Phase 1: Diversify with different initializations ============
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 1: Try diverse initializations ===")
        inits = []
        alphas = [0.5 / n0, 0.8 / n0, 1.0 / n0, 1.5 / n0, 2.0 / n0]
        for al in alphas:
            inits.append(("exp_a%.2f" % al, gen_exponential(n0, al)))
        for p_val in [0.3, 0.5, 0.7, 1.0]:
            inits.append(("pow_p%.1f" % p_val, gen_power_law(n0, p_val)))
        inits.append(("asym_g03", gen_asymmetric_gauss(n0, 0.3, 0.25)))
        inits.append(("asym_g07", gen_asymmetric_gauss(n0, 0.7, 0.25)))
        inits.append(("asym_g05w", gen_asymmetric_gauss(n0, 0.5, 0.4)))
        inits.append(("sine_1_0", gen_sine_asym(n0, 1, 0)))
        inits.append(("sine_2_0", gen_sine_asym(n0, 2, 0.5)))

        # Also try the parent and its mirror
        if parent is not None:
            inits.append(("parent", parent))
            inits.append(("parent_mirror", parent[::-1]))

        for label, init_a in inits:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            sc = evaluate_safe(init_a)
            print(f"  Init {label}: score={sc:.6f}")
            if sc < best_score:
                # Short refinement
                t_short = min(remaining * 0.02, 15.0)
                a_out, _ = lp_frank_wolfe(init_a, t_short, solver, verbose=False)
                try_update(a_out, f"lp_{label}")
                a_out, _ = soft_subgradient_nesterov(init_a, t_short, top_k=40, beta=3.0, verbose=False)
                try_update(a_out, f"ssg_{label}")
            elif sc < best_score + 0.1:
                # Close enough, quick LP to see if it can beat
                t_short = min(remaining * 0.015, 10.0)
                a_out, _ = lp_frank_wolfe(init_a, t_short, solver, verbose=False)
                try_update(a_out, f"lp_{label}")

    # ============ Phase 2: LP-FW main (40% of time) ============
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: LP-FW main n={n_cur} ===")
        t2 = remaining * 0.35
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t2, solver, verbose=True)
        try_update(a_out, f"lp_main_n{n_cur}")

    # ============ Phase 3: Soft subgradient ============
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: Soft subgradient n={n_cur} ===")
        t3 = min(remaining * 0.2, 250.0)
        a_out, _ = soft_subgradient_nesterov(norm_sum1(best_seq), t3, top_k=64, beta=3.0, verbose=True)
        try_update(a_out, f"ssg_n{n_cur}")
        a_out, _ = soft_subgradient_nesterov(norm_sum1(best_seq), min(remaining * 0.1, 100.0), top_k=32, beta=5.0, verbose=False)
        try_update(a_out, f"ssg_hi_beta_n{n_cur}")

    # ============ Phase 4: LP-FW round 2 ============
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: LP-FW round 2 n={n_cur} ===")
        t4 = min(remaining * 0.25, 300.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t4, solver, verbose=True)
        try_update(a_out, f"lp_r2_n{n_cur}")

    # ============ Phase 5: Coordinate descent polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        n_cur = len(best_seq)
        print(f"\n=== Phase 5: Coord PD n={n_cur} ===")
        t5 = min(remaining * 0.15, 150.0)
        a_out, _ = peak_coord_descent(norm_sum1(best_seq), t5, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_cur}")

    # ============ Phase 6: Multi-n exploration ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 120:
        print(f"\n=== Phase 6: Multi-n exploration ===")
        test_ns = []
        for delta in [200, 400, -200, 400, -400, 600]:
            nt = n_best + delta
            if 800 <= nt <= 3500 and nt != n_best:
                test_ns.append(nt)

        seen_ns = set()
        for nt in test_ns:
            if nt in seen_ns:
                continue
            seen_ns.add(nt)
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            print(f"  Trying n={nt}...")
            a_inter = interpolate_sequence(best_seq, nt)
            t_per = min(remaining * 0.2, 120.0)
            a_out, _ = lp_frank_wolfe(norm_sum1(a_inter), t_per, solver, verbose=False)
            try_update(a_out, f"multin_n{nt}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40:
                t_per2 = min(remaining * 0.1, 50.0)
                a_out, _ = peak_coord_descent(norm_sum1(best_seq), t_per2, verbose=False, rng_seed=nt % 10000)
                try_update(a_out, f"coord_multin_n{nt}")

    # ============ Phase 7: Final LP-FW + Coord on best ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 7: Final refinement n={n_best} ===")
        t7 = min(remaining * 0.3, 200.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t7, solver, verbose=True)
        try_update(a_out, f"lp_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_best = len(best_seq)
        t8 = min(remaining * 0.25, 150.0)
        a_out, _ = peak_coord_descent(norm_sum1(best_seq), t8, verbose=True, rng_seed=999)
        try_update(a_out, f"coord_final_n{n_best}")

    # ============ Phase 8: Last LP if time ============
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        t9 = remaining * 0.8
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t9, solver, verbose=False)
        try_update(a_out, f"lp_last_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
