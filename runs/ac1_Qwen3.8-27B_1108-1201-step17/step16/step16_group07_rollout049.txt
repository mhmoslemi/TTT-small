def smooth_gradient(z, alpha):
    """Compute gradient of smooth objective w.r.t. z."""
    n = len(z)
    a = np.exp(z - np.max(z))
    a = a / a.sum() * np.sqrt(2 * n)
    conv = fast_autocorr(a)
    
    # Softmax weights
    alpha_conv = alpha * conv
    max_ac = np.max(alpha_conv)
    weights = np.exp(alpha_conv - max_ac)
    weights = weights / weights.sum()
    
    # Gradient of f w.r.t. a
    grad_a = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-15:
            continue
        j = k
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        idx = np.arange(lo, hi)
        grad_a[lo:hi] += weights[k] * 2.0 * a[k - idx]
    
    # Gradient w.r.t. z
    mean_grad = np.dot(grad_a, a / a.sum())
    a_norm = a / a.sum()
    grad_z = a_norm * (grad_a - mean_grad)
