"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Alternating LP (primary): alternates between two LPs to find a fixed point
   where a = b and max(a*a) is minimized
2. LP Frank-Wolfe as fallback
3. Peak-targeted coordinate descent for final polish
4. Multi-length exploration
"""

import numpy as np
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    import cvxpy as cp
    for s in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_conv_matrix(fixed):
    """Build matrix C where C[k,j] = fixed[k-j] if 0<=k-j<n, else 0.
    C has shape (2n-1, n). C @ a gives (a * fixed) the convolution."""
    n = len(fixed)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j+n, j] = fixed
    return C


def alternating_lp(a0, time_limit, solver, verbose=True, max_iters=5000, tol=1e-10):
    """Alternating LP: minimize max(a*b) alternately in a and b, with a,b on simplex."""
    n = len(a0)
    a = norm_sum1(a0)
    b = a.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    prev_t = np.inf
    stale_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP for a: min t s.t. (a*b)_k <= t, a>=0, sum(a)=1
        C = build_conv_matrix(b)
        a_var = cp.Variable(n, nonneg=True)  # need to import cp
        t_var = cp.Variable()
        constraints = [C @ a_var <= t_var, cp.sum(a_var) == 1]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        prob.solve(solver=solver, verbose=False)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            stale_count += 1
            if stale_count > 10:
                break
            continue
        a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
        
        # Solve LP for b: min t s.t. (a*b)_k <= t, b>=0, sum(b)=1
        D = build_conv_matrix(a)
        b_var = cp.Variable(n, nonneg=True)
        t_var2 = cp.Variable()
        constraints2 = [D @ b_var <= t_var2, cp.sum(b_var) == 1]
        prob2 = cp.Problem(cp.Minimize(t_var2), constraints2)
        prob2.solve(solver=solver, verbose=False)
        if prob2.status not in ('optimal', 'optimal_inaccurate'):
            stale_count += 1
            if stale_count > 10:
                break
            continue
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        s = b_new.sum()
        if s < 1e-15:
            break
        b_new = b_new / s
        b = b_new
        
        t_val = prob2.value  # max_k (a*b)_k
        if t_val < prev_t - tol:
            stale_count = 0
        else:
            stale_count += 1
        prev_t = t_val
        
        if stale_count > 20:
            break
        
        # Check convergence
        gap = np.max(np.abs(a - b))
        if gap < 1e-8:
            break
        
        # Update best
        m_a = fast_autocorr(a).max()
        if m_a < best_m:
            best_m = m_a
            best_a = a.copy()
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AltLP iter={it} ({elapsed:.1f}s): J={J:.6f}, gap={gap:.2e}, t={t_val:.8f}")
    
    # At convergence, use (a+b)/2 as the final estimate
    avg = (a + b) / 2.0
    avg = np.maximum(avg, 0.0)
    avg = avg / avg.sum()
    m_avg = fast_autocorr(avg).max()
    if m_avg < best_m:
        best_m = m_avg
        best_a = avg.copy()
    
    return best_a, best_m

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Alternating LP (primary): alternates two LPs to find a stationary point
   of max(a*a) over the simplex
2. LP Frank-Wolfe with Nesterov acceleration (secondary)
3. Peak-targeted coordinate descent (final polish)
4. Multi-start and multi-length exploration
"""

import numpy as np
import time
import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_conv_matrix(fixed):
    """Build C (shape (2n-1, n)) such that C @ x = convolve(x, fixed)."""
    n = len(fixed)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j+n, j] = fixed
    return C


def alternating_lp(a0, time_limit, solver, verbose=True, max_iters=5000):
    """
    Alternating LP to minimize max(a*a) over the simplex.
    Alternates:
      a := argmin_{a in simplex} max_k (a * b)_k
      b := argmin_{b in simplex} max_k (a * b)_k
    At convergence (a ≈ b), max(a*a) is locally minimized.
    """
    n = len(a0)
    a = norm_sum1(a0)
    b = a.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    prev_t = np.inf
    stale = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # LP for a
        C = build_conv_matrix(b)
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        cons = [C @ a_var <= t_var, cp.sum(a_var) == 1]
        prob = cp.Problem(cp.Minimize(t_var), cons)
        prob.solve(solver=solver, verbose=False)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            stale += 1
            if stale > 20:
                break
            continue
        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
        
        # LP for b
        D = build_conv_matrix(a)
        b_var = cp.Variable(n, nonneg=True)
        t_var2 = cp.Variable()
        cons2 = [D @ b_var <= t_var2, cp.sum(b_var) == 1]
        prob2 = cp.Problem(cp.Minimize(t_var2), cons2)
        prob2.solve(solver=solver, verbose=False)
        if prob2.status not in ('optimal', 'optimal_inaccurate'):
            stale += 1
            if stale > 20:
                break
            continue
        b_new = np.asarray(b_var.value, dtype=np.float64)
        b_new = np.maximum(b_new, 0.0)
        s = b_new.sum()
        if s < 1e-15:
            break
        b = b_new / s
        stale = 0
        
        t_val = prob2.value
        if t_val < prev_t - 1e-12:
            pass  # still making progress
        else:
            # Check if we're at a fixed point
            gap = np.max(np.abs(a - b))
            if gap < 1e-7:
                break
            if t_val >= prev_t - 1e-10:
                stale += 1
                if stale > 30:
                    break
        prev_t = t_val
        
        # Track best
        m = fast_autocorr(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()
        
        if verbose and it % 25 == 0:
            elapsed = time.monotonic() - t0
            gap = np.max(np.abs(a - b))
            J = 2.0 * n * best_m
            print(f"    AltLP it={it} ({elapsed:.0f}s): J={J:.6f} gap={gap:.1e} t={t_val:.8f}")
    
    # Final: use average
    avg = (a + b) / 2.0
    avg = np.maximum(avg, 0.0)
    ss = avg.sum()
    if ss > 1e-15:
        avg = avg / ss
        m_avg = fast_autocorr(avg).max()
        if m_avg < best_m:
            best_m = m_avg
            best_a = avg.copy()
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    AltLP done ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_iters=50000):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    x_prev = a.copy()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Build T where T[k,j] = y[j] if j <= k <= j+n-1
        T = build_conv_matrix(y)
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

            x_new = g_dir.copy()

            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new
