"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral equalization for initialization
2. LP Frank-Wolfe with full/near-full constraints
3. Smoothed log-sum-exp subgradient with L-BFGS polish
4. Peak-targeted coordinate descent
5. Multi-length exploration
"""

import numpy as np
import cvxpy as cp
import time

def fast_autocorr(a):
    """Compute full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def eval_obj(a):
    """Compute 2*n*max_conv/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)

def spectral_equalize(a, n_iter=50, blend=0.5):
    """Flatten power spectrum via iterative DFT correction."""
    a = normalize(a).copy()
    n = len(a)
    for i in range(n_iter):
        A = np.fft.fft(a)
        P = np.abs(A) ** 2
        # Target flat spectrum
        A_target = A / np.sqrt(P + 1e-12)
        a_new = np.real(np.fft.ifft(A_target))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = (1 - blend) * a + blend * a_new
    return a

def get_solver():
    """Find the best available LP solver."""
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def solve_fw_lp(a, ac, M, top_k=None, solver=cp.CLARABEL):
    """
    Solve the FW LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0.
    Uses top_k constraints if specified, otherwise all.
    Returns normalized direction or None.
    """
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    if top_k is None:
        k_indices = range(len(ac))
    else:
        k_indices = np.argsort(ac)[::-1][:top_k]
    
    constraints = []
    for k in k_indices:
        k_int = int(k)
        # (a*b)[k] = sum_j a[j] * b[k-j] for valid j
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        a_coeff = a[k_int - j_idx]
        constraints.append(a_coeff @ b_var[j_idx] <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def lp_fw(a0, time_limit, solver, top_k=None, verbose=True):
    """
    Frank-Wolfe with LP oracle.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_obj = eval_obj(a)
    t0 = time.monotonic()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:

        ac = fast_autocorr(a)
        M = ac.max()
        
        # Solve LP
        direction = solve_fw_lp(a, ac, M, top_k=top_k, solver=solver)
        if direction is None:
            # Fallback: use the vertex that minimizes the max single constraint
            # Just pick a random direction
            break
        
        # Step size with backtracking
        alpha = 2.0 / (t + 2.0)
        
        # Try line search for better step
        best_alpha = alpha
        best_val = eval_obj((1 - alpha) * a + alpha * direction)
        
        for trial_alpha in [alpha * 0.5, alpha * 1.5, alpha * 2.0]:
            if trial_alpha > 1.0:
                continue
            a_trial = (1 - trial_alpha) * a + trial_alpha * direction
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial /= s
            val = eval_obj(a_trial)
            if val < best_val:
                best_val = val
                best_alpha = trial_alpha
        
        a = (1 - best_alpha) * a + best_alpha * direction
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a /= s
        
        cur_obj = eval_obj(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose and (t % 50 == 1 or t < 10):
                elapsed = time.monotonic() - t0
                print(f"    LP-FW t={t} ({elapsed:.1f}s): obj={best_obj:.8f}")
    
    return best_a, best_obj

            # d/d(a[m]) = a[k-m] (from j=m term) + a[k-m] (from j=k-m term, if k-m != m)
            # = a[k-m] * (1 + [k-m != m and 0<=k-m<n])
            # = 2*a[k-m] if 0<=k-m<n and k-m != m
            # = a[k-m] if k-m == m (i.e., k=2m)
            # = 0 if k-m < 0 or k-m >= n
            
            # For j_idx range, a[k-j_idx] gives the derivative contribution
            # But we need to handle the double-counting case
            
            # Simple approach: just use a[k-j] for j in valid range
            # This gives d(conv_k)/d(a[j]) = a[k-j] (single count)
            # The "true" derivative has a factor of 2 for off-diagonal terms
            # But for subgradient, using the single count is still a valid subgradient
            # (it's half the true gradient for off-diagonal, but still points in the right direction)
            
            grad[j_idx] += w_k * a[k_int - j_idx]
        
        # Projected gradient step (mirror descent)
        eta = 1.0 / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        cur_obj = eval_obj(a_new)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_new.copy()
            if verbose and t % 200 == 1:
                elapsed = time.monotonic() - t0
                print(f"    LSE t={t} ({elapsed:.1f}s): obj={best_obj:.8f}")
        else:
            a_new = a  # Don't update if worse
        
        a = a_new
    
    return best_a, best_obj

def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """
    Peak-targeted coordinate descent.
    At each step, find the peak of the convolution, and try to reduce it
    by moving mass from contributing elements to non-contributing positions.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_obj = eval_obj(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        # Get top peaks
        n_peaks = min(10, len(ac))
        top_k = np.argsort(ac)[::-1][:n_peaks]
        
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            idx = np.arange(j_start, j_end + 1)
            contrib = a[idx] * a[k_int - idx]
            
            # Find the element contributing most to this peak
            src = idx[np.argmax(contrib)]
            
            # Try moving a small fraction of mass from src to a "good" destination
            # A good destination is one that doesn't contribute to any top peak
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:5]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            
            # Try a few random candidates
            trial_count = min(50, len(candidates))
            chosen = candidates[rng.permutation(len(candidates))[:trial_count]]
            
            for dst in chosen:
                if dst == src:
                    continue
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial /= s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < M_curr - 1e-15:
                        a = a_trial
                        M_curr = M_trial
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if improved:
            cur_obj = eval_obj(a)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Perturbation to escape
            if a.any():
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a /= s_t
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Coord ({elapsed:.1f}s): obj={best_obj:.8f}")
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral equalization for initialization
2. LP Frank-Wolfe with precomputed sparse constraint matrix (full and truncated)
3. Log-sum-exp smoothed subgradient with mirror descent
4. Peak-targeted coordinate descent with exact evaluation
5. Periodic restarts to escape plateaus
6. Multi-length exploration
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csc_matrix
import time


def fast_autocorr(a):
    """Compute full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def eval_obj(a):
    """Compute 2*n*max_conv/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)


def spectral_equalize(a, n_iter=50, blend=0.5):
    """Flatten power spectrum via iterative DFT correction."""
    a = normalize(a).copy()
    n = len(a)
    for i in range(n_iter):
        A = np.fft.fft(a)
        P = np.abs(A) ** 2
        A_target = A / np.sqrt(P + 1e-12)
        a_new = np.real(np.fft.ifft(A_target))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = (1.0 - blend) * a + blend * a_new
    return a


def build_conv_indices(n, top_k=None):
    """
    Precompute row/col indices for the convolution constraint matrix.
    Returns (rows, cols) arrays where the (k,j) entry of the matrix is a[k-j].
    If top_k is None, use all 2n-1 lags. Otherwise use the first top_k.
    """
    rows = []
    cols = []
    if top_k is None:
        k_range = range(2 * n - 1)
    else:
        k_range = range(top_k)  # We'll reorder later based on ac
    for k in k_range:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j = np.arange(j_start, j_end + 1)
        rows.extend([k_int] * len(j))
        cols.extend(j.tolist())
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

    """Precompute indices for full convolution constraint (all 2n-1 lags)."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        j = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j), k, dtype=np.int32))
        cols_list.append(j.astype(np.int32))
    if not rows_list:
        return np.array([], dtype=np.int32), np.array([], dtype=np.int32)
    return np.concatenate(rows_list), np.concatenate(cols_list)


def solve_fw_lp_fast(a, ac, M, rows, cols, top_k=None, solver_kwargs=None):
    """
    Solve FW LP: max sum(b) s.t. (a*b)[k] <= M for selected k, b >= 0.
    Uses precomputed indices. Returns normalized direction or None.
    """
    n = len(a)
    
    if top_k is not None and len(ac) > top_k:
        # Select top-k lags by current ac values
        top_idx = np.argsort(ac)[::-1][:top_k]
        # Filter rows/cols to only selected lags
        mask = np.isin(rows, top_idx)
        sel_rows = rows[mask]
        sel_cols = cols[mask]
    else:
        sel_rows = rows
        sel_cols = cols
    
    if len(sel_rows) == 0:
        return np.ones(n) / n
    
    m = len(np.unique(sel_rows))
    
    # Compute values: a[sel_rows - sel_cols]
    idx = sel_rows - sel_cols
    vals = a[idx]
    
    # Build sparse matrix
    A_ub = csc_matrix((vals, (sel_rows, sel_cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    
    c = -np.ones(n)
    
    if solver_kwargs is None:
        solver_kwargs = {}
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={"presolve": True, "time_limit": 30.0})
    except Exception:
        return None
    
    if not res.success or res.x is None:
        return None
    
    b_val = np.maximum(res.x, 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def lp_fw(a0, time_limit, top_k=None, verbose=True, use_full=False, n_fniters=None):
    """
    Frank-Wolfe with LP oracle and periodic restarts.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_obj = eval_obj(a)
    t0 = time.monotonic()
    
    # Precompute indices
    if use_full:
        rows, cols = build_full_conv_indices(n)
    else:
        rows, cols = build_full_conv_indices(n)
    
    t = 0
    stall_count = 0
    max_stall = 80
    
    total_iters = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        total_iters += 1
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Determine which constraints to use
        if use_full:
            cur_top_k = None
        else:
            # Start with fewer, increase over time
            cur_top_k = min(100 + t * 2, 500, 2 * n - 1)
        
        direction = solve_fw_lp_fast(a, ac, M, rows, cols, top_k=cur_top_k)
        if direction is None:
            # Try with fewer constraints
            cur_top_k = max(50, cur_top_k // 2) if cur_top_k else 100
            direction = solve_fw_lp_fast(a, ac, M, rows, cols, top_k=cur_top_k)
            if direction is None:
                break
        
        # Step size
        alpha = 2.0 / (t + 2.0)
        
        # Try a few step sizes
        best_alpha = alpha
        a_best_trial = (1.0 - alpha) * a + alpha * direction
        a_best_trial = np.maximum(a_best_trial, 0.0)
        s_best = a_best_trial.sum()
        if s_best > 1e-15:
            a_best_trial /= s_best
        best_trial_obj = eval_obj(a_best_trial)
        
        for trial_alpha in [alpha * 0.7, alpha * 1.3, alpha * 2.0, alpha * 3.0]:
            if trial_alpha > 1.0:
                continue
            a_trial = (1.0 - trial_alpha) * a + trial_alpha * direction
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial /= s_t
            val = eval_obj(a_trial)
            if val < best_trial_obj:
                best_trial_obj = val
                best_alpha = trial_alpha
                a_best_trial = a_trial.copy()
        
        a = a_best_trial
        cur_obj = best_trial_obj
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall_count = 0
            if verbose and (t <= 5 or t % 20 == 0):
                elapsed = time.monotonic() - t0
                print(f"    LP-FW t={t} ({elapsed:.1f}s): obj={best_obj:.8f}, "
                      f"top_k={cur_top_k}, alpha={best_alpha:.4f}")
        else:
            stall_count += 1
        
        # Periodic restart
        if stall_count >= max_stall:
            a = best_a.copy()
            t = 0
            stall_count = 0
            if verbose:
                print(f"    LP-FW restart at iteration {total_iters}, "
                      f"obj={best_obj:.8f}")
        
        if n_fniters is not None and total_iters >= n_fniters:
            break
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    LP-FW done: {total_iters} iters, obj={best_obj:.8f}, "
              f"time={elapsed:.1f}s")
    
    return best_a, best_obj


def logsumexp_subgrad(a0, time_limit, n_top=100, beta=8.0, c0=2.0, verbose=True):
    """
    Mirror descent on the log-sum-exp smoothed objective.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_obj = eval_obj(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        
        # Softmax weights over top peaks
        top_idx = np.argsort(ac)[::-1][:min(n_top, len(ac))]
        top_ac = ac[top_idx]
        shifted = top_ac - top_ac.max()
        weights = np.exp(beta * shifted)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones(len(top_idx)) / len(top_idx)
        else:
            weights /= w_sum
        
        # Compute subgradient
        grad = np.zeros(n)
        for w_k, k in zip(weights, top_idx):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start > j_end:
                continue
            j_idx = np.arange(j_start, j_end + 1)
            grad[j_idx] += w_k * a[k_int - j_idx]
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        cur_obj = eval_obj(a_new)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a_new.copy()
            stall = 0
            if verbose and (t <= 5 or t % 500 == 0):
                elapsed = time.monotonic() - t0
                print(f"    LSE t={t} ({elapsed:.1f}s): obj={best_obj:.8f}")
        else:
            stall = t  # Use as stall counter
            a_new = a  # Don't accept
        
        a = a_new
        
        # Restart if stalled
        if t % 1000 == 0 and (a != best_a).all():
            pert = rng.normal(0, 0.002, n)
            x = best_a * (1.0 + pert)
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            a = x
    
    return best_a, best_obj


def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):
    """
    Peak-targeted coordinate descent.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_obj = eval_obj(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_peaks = min(8, len(ac))
        top_k = np.argsort(ac)[::-1][:n_peaks]
        
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            idx = np.arange(j_start, j_end + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find destination: positions not contributing to top peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:4]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)
            
            trial_count = min(40, len(candidates))
            chosen = candidates[rng.permutation(len(candidates))[:trial_count]]
            
            for dst in chosen:
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.01, 0.03, 0.08]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial /= s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < M_curr * (1.0 - 1e-10):
                        a = a_trial
                        M_curr = M_trial
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if improved:
            cur_obj = eval_obj(a)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
            stall = 0
        else:
            # Perturbation
            stall += 1
            if stall % 5 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a /= s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < M_curr:
                                M_curr = M_new
                                stall = 0
                                break
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Coord ({elapsed:.1f}s): obj={best_obj:.8f}")
    
    return best_a, best_obj


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase strategy."""
    t_global = time.monotonic()
    
    def elapsed():
        return time.monotonic() - t_global
    
    def remaining():
        return float(budget_s) - 60.0 - elapsed()
    
    print(f"Total budget: {budget_s:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            p_score = eval_obj(parent)
            print(f"Parent loaded: n={len(parent)}, obj={p_score:.8f}")
            best_seq = parent
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = normalize(a)
        sc = eval_obj(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: obj={sc:.8f} (n={len(a)}, t={elapsed():.0f}s)")
    
    if best_seq is None:
        n = 2400
        x = np.linspace(0, 1, n)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = normalize(best_seq)
        best_score = eval_obj(best_seq)
        print(f"Default init: n={n}, obj={best_score:.8f}")
    
    # === Phase 1: Spectral equalization ===
    if remaining() > 30:
        print(f"\n=== Phase 1: Spectral equalization ===")
        a_spec = spectral_equalize(best_seq, n_iter=60, blend=0.4)
        try_update(a_spec, "spectral")
        
        # Also try with different blend
        a_spec2 = spectral_equalize(best_seq, n_iter=40, blend=0.6)
        try_update(a_spec2, "spectral_b")
    
    # === Phase 2: LP-FW with truncated constraints (main workhorse) ===
    if remaining() > 120:
        n_best = len(best_seq)
        print(f"\n=== Phase 2: LP-FW truncated, n={n_best} ===")
        t2 = min(remaining() * 0.40, 650.0)
        a_out, _ = lp_fw(normalize(best_seq), t2, top_k=300, verbose=True, use_full=False)
        try_update(a_out, f"lpfw_trunc_n{n_best}")
    
    # === Phase 3: LP-FW with full constraints (precise) ===
    if remaining() > 120:
        n_best = len(best_seq)
        print(f"\n=== Phase 3: LP-FW full, n={n_best} ===")
        t3 = min(remaining() * 0.25, 400.0)
        a_out, _ = lp_fw(normalize(best_seq), t3, top_k=None, verbose=True, use_full=True)
        try_update(a_out, f"lpfw_full_n{n_best}")
    
    # === Phase 4: Log-sum-exp subgradient ===
    if remaining() > 80:
        n_best = len(best_seq)
        print(f"\n=== Phase 4: LSE subgrad, n={n_best} ===")
        t4 = min(remaining() * 0.12, 180.0)
        a_out, _ = logsumexp_subgrad(normalize(best_seq), t4, n_top=80, beta=6.0, c0=3.0, verbose=True)
        try_update(a_out, f"lse_n{n_best}")
    
    # === Phase 5: Coordinate polish ===
    if remaining() > 40:
        n_best = len(best_seq)
        print(f"\n=== Phase 5: Coordinate polish, n={n_best} ===")
        t5 = min(remaining() * 0.15, 200.0)
        a_out, _ = coordinate_polish(normalize(best_seq), t5, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_best}")
    
    # === Phase 6: Try alternative length ===
    if remaining() > 250:
        n_current = len(best_seq)
        n_alt = n_current + 400
        if n_alt > 3500:
            n_alt = n_current - 400
        n_alt = max(1200, min(n_alt, 3500))
        
        if n_alt != n_current:
            print(f"\n=== Phase 6: Alt length n={n_alt} ===")
            old_x = np.linspace(0, 1, n_current)
            new_x = np.linspace(0, 1, n_alt)
            a_alt = np.interp(new_x, old_x, best_seq)
            a_alt = normalize(a_alt)
            a_alt = spectral_equalize(a_alt, n_iter=30, blend=0.4)
            
            t6 = min(remaining() * 0.35, 300.0)
            if t6 > 60:
                a_out, _ = lp_fw(a_alt, t6, top_k=200, verbose=True, use_full=False)
                try_update(a_out, f"lpfw_alt_n{n_alt}")
            
            if remaining() > 60:
                t6b = min(remaining() * 0.2, 100.0)
                if t6b > 30:
                    a_out, _ = coordinate_polish(a_alt if a_alt is not None else normalize(best_seq),
                                                 t6b, verbose=False, rng_seed=99)
                    try_update(a_out, f"coord_alt_n{n_alt}")
    
    # === Phase 7: Final coordinate polish ===
    if remaining() > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Final coord polish, n={n_best} ===")
        t7 = min(remaining() * 0.6, 150.0)
        a_out, _ = coordinate_polish(normalize(best_seq), t7, verbose=True, rng_seed=1234)
        try_update(a_out, f"coord_final_n{n_best}")
    
    # === Phase 8: One more LP-FW round if time ===
    if remaining() > 80:
        n_best = len(best_seq)
        if n_best <= 3000:
            print(f"\n=== Phase 8: Final LP-FW, n={n_best} ===")
            t8 = min(remaining() * 0.7, 150.0)
            a_out, _ = lp_fw(normalize(best_seq), t8, top_k=200, verbose=False, use_full=False)
            try_update(a_out, f"lpfw_very_final_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = eval_obj(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: obj={best_score:.8f}, n={len(best_seq)}, time={elapsed():.0f}s")
        try:
            test_score = evaluate_sequence([float(x) for x in best_seq])
            print(f"Validator: score={test_score:.8f}")
            if test_score < best_score:
                best_score = test_score
        except Exception as e:
            print(f"Validation error: {e}")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
