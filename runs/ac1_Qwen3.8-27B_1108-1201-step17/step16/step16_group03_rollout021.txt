import numpy as np
import time

def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def softmax_smoothed_max(c, sigma):
    """Compute sigma * log(sum(exp(c/sigma))) as smoothed max, and gradient."""
    c = np.asarray(c, dtype=np.float64)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / sigma)
    log_sum = c_max + sigma * np.log(exp_c.sum())
    # Gradient w.r.t. c: exp_c / sum(exp_c)
    w = exp_c / exp_c.sum()
    return log_sum, w


def subgradient_of_conv(a, k):
    """Subgradient of (a*a)[k] w.r.t. a. Returns array of length n."""
    n = len(a)
    k = int(k)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    if i_start >= i_end:
        return grad
    idx = np.arange(i_start, i_end)
    j = k - idx
    valid = (j >= 0) & (j < n)
    if valid.any():
        grad[idx[valid]] = 2.0 * a[j[valid]]
    return grad


def soft_subgradient(a, c, top_k=3, beta=2.0):
    """Weighted average of subgradients at top-k peaks."""
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = np.maximum(c[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        grad += wi * subgradient_of_conv(a, int(k))
    return grad


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def interpolate_sequence(f, new_n):
    """Interpolate sequence f to new length new_n."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def smoothed_subgradient_descent(a0, time_limit, verbose=True, seed=42,
                                   sigma_start=0.01, sigma_end=0.001,
                                   top_k=3, beta=1.5):
    """
    Nesterov-accelerated subgradient descent with soft-max smoothing.
    Uses backtracking line search for step size.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 5000

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal sigma
        progress = min(t / 50000.0, 1.0)
        sigma = sigma_start * (sigma_end / sigma_start) ** progress

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute convolution
        ac = fast_conv(y)
        M_curr = ac.max()

        # Adaptive top_k: increase when stuck
        if stall > max_stall // 3:
            current_top_k = min(top_k + 2, 8)
        elif stall > max_stall * 2 // 3:
            current_top_k = min(top_k + 4, 10)
        else:
            current_top_k = top_k

        # Subgradient
        g = soft_subgradient(y, ac, top_k=current_top_k, beta=beta)

        # Backtracking line search
        eta = 0.1
        best_trial = None
        best_trial_m = M_curr
        for _ in range(10):
            trial = y - eta * g
            trial = simplex_project(trial)
            if trial.sum() < 1e-15:
                trial = y.copy()
            M_trial = fast_conv(trial).max()
            if M_trial < best_trial_m:
                best_trial_m = M_trial
                best_trial = trial
            eta *= 0.7
            if eta < 1e-6:
                break

        if best_trial is not None:
            x_prev = x
            x = best_trial
        else:
            # No improvement from subgradient step, try small random perturbation
            if stall > 100:
                perturbation = rng.normal(0, 0.01 * (1 + stall / max_stall), n)
                x_trial = x + perturbation
                x_trial = simplex_project(x_trial)
                M_trial = fast_conv(x_trial).max()
                if M_trial < best_m:
                    x = x_trial
                    x_prev = x
            else:
                x_prev = x

        # Track best
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Random restart if very stale
        if stall > max_stall:
            noise_scale = 0.02 + 0.01 * rng.random()
            x_restarted = best_a * (1 + noise_scale * rng.standard_normal(n))
            x_restarted = simplex_project(x_restarted)
            x = x_restarted
            x_prev = x_restarted
            stall = 0

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SSD t={t} ({elapsed:.1f}s): J={J:.6f}, sigma={sigma:.5f}, stall={stall}")

    return best_a, best_m


def exact_subgradient_descent(a0, time_limit, verbose=True, seed=42,

    """

    """

    t0 = time.monotonic()

        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_conv(y)
        M_curr = ac.max()

        # Adaptive top-k
        if stall < 1000:
            k_use = top_k_init
        elif stall < 3000:
            k_use = min(top_k_init + 2, top_k_max)
        else:
            k_use = min(top_k_init + 4, top_k_max)

        g = soft_subgradient(y, ac, top_k=k_use, beta=2.0)

        # Try multiple step sizes
        base_eta = 0.5 / (np.sqrt(t) * np.sqrt(n))
        etas = [base_eta * f for f in [4.0, 2.0, 1.0, 0.5, 0.25, 0.1]]

        best_trial = None
        best_trial_m = M_curr
        for eta in etas:

            trial = simplex_project(trial)
            M_trial = fast_conv(trial).max()
            if M_trial < best_trial_m:
                best_trial_m = M_trial
                best_trial = trial

        if best_trial is not None:
            x_prev = x
            x = best_trial
        else:
            x_prev = x

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Periodic restart
        if stall > 4000:
            noise = 0.03 * rng.standard_normal(n)
            x_r = x * (1.0 + noise / max(x.max(), 1e-10))
            x_r = simplex_project(x_r)
            x = x_r
            x_prev = x_r
            stall = 0

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    ESD t={t} ({elapsed:.1f}s): J={J:.6f}, k={k_use}, stall={stall}")

def coordinate_descent(a0, time_limit, verbose=True, seed=42, max_rounds=100000):
    """
    Coordinate descent: for each coordinate, find the optimal value
    that minimizes the max convolution entry.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    for iteration in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 2000:
            break

        # Get current convolution and top peaks
        ac = fast_conv(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:min(20, len(ac))]

        # Randomly permute coordinates
        perm = rng.permutation(n)
        improved = False

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break

            # Find which convolution entries are affected by a[i]
            # (a*a)[k] depends on a[i] if k-i is in [0, n-1], i.e., k in [i, i+n-1]
            k_min = i
            k_max = i + n - 1

            # Find the current max among affected entries
            # The objective we care about is max_k c(k) overall
            # When we change a[i], entries c(k) for k in [i, i+n-1] change

            # Current value of a[i]
            ai = a[i]
            if ai < 1e-15:
                # Try small positive value
                for trial_val in [1e-4, 1e-3, 5e-3]:
                    a_trial = a.copy()
                    a_trial[i] = trial_val
                    a_trial = a_trial / a_trial.sum()
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                continue

            # Try different values: decrease, increase, or set to 0
            # The key is to find a value that reduces the peak
            candidates = [0.0, ai * 0.5, ai * 0.8, ai * 1.2, ai * 1.5,
                         ai * 0.1, ai * 0.2, ai * 2.0]
            # Also try values based on the peak structure
            for k in top_peaks[:5]:
                k_int = int(k)
                j = k_int - i
                if 0 <= j < n and j != i:
                    # c(k) = ... + a[i]*a[j] + a[j]*a[i] + ... (if i != j)
                    # or c(k) = ... + a[i]^2 + ... (if i == j)
                    pass

            best_val = ai
            best_M = best_m
            for cv in candidates:
                cv = max(0.0, cv)
                a_trial = a.copy()
                a_trial[i] = cv
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M:
                    best_M = M_trial
                    best_val = cv

            if best_val != ai and best_M < best_m - 1e-14:
                a[i] = best_val
                a = a / a.sum()
                best_m = best_M
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1

        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD iter={iteration} ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def two_opt_swap(a0, time_limit, verbose=True, seed=42):

            if M_trial < best_m - 1e-14:

def diverse_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian centered
    x = np.linspace(0, 1, n)
    sigma_vals = [0.1, 0.15, 0.2, 0.25, 0.3]
    for sv in sigma_vals:
        g = np.exp(-((x - 0.5) ** 2) / (2 * sv ** 2))
        inits.append(g / g.sum())

    # Asymmetric Gaussian (shifted)
    for shift in [0.35, 0.4, 0.45, 0.55, 0.6, 0.65]:
        g = np.exp(-((x - shift) ** 2) / (2 * 0.15 ** 2))
        inits.append(g / g.sum())

    # Sum of two Gaussians
    for s1, s2, w in [(0.3, 0.7, 0.5), (0.25, 0.75, 0.5), (0.35, 0.65, 0.4)]:
        g = w * np.exp(-((x - s1) ** 2) / (2 * 0.1 ** 2)) + (1 - w) * np.exp(-((x - s2) ** 2) / (2 * 0.1 ** 2))
        inits.append(g / g.sum())

    # Sine-squared
    s = np.sin(np.pi * x) ** 2
    inits.append(s / s.sum())

    # Linear (ramp)
    for direction in [1, -1]:
        r = np.linspace(1, 2, n) if direction == 1 else np.linspace(2, 1, n)
        inits.append(r / r.sum())

    # V-shaped
    v = np.abs(x - 0.5) + 0.1
    inits.append(v / v.sum())

    # W-shaped
    w_shape = np.abs(np.sin(2 * np.pi * x)) ** 0.5
    inits.append(w_shape / w_shape.sum())

    # Random smooth
    for seed in [1, 7, 42]:
        r = np.random.default_rng(seed)
        raw = r.random(n)
        # Smooth with a kernel
        kernel_size = max(5, n // 100)
        kernel = np.ones(kernel_size) / kernel_size
        smooth = np.convolve(raw, kernel, mode='same')
        smooth = np.maximum(smooth, 1e-6)
        inits.append(smooth / smooth.sum())

    # Power law
    for p in [0.5, 1.0, 2.0, 3.0]:
        pl = x ** p + 1e-6
        inits.append(pl / pl.sum())

    return inits


def nesterov_amd_log(a0, time_limit, verbose=True, c0=3.0, top_k=1, seed=42):
    """
    Nesterov accelerated mirror descent in the log domain (exponential family).
    This is different from the previous approach: uses adaptive step size
    with a decaying schedule and peak cycling.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 3000
    current_top_k = top_k

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_conv(y)
        M_curr = ac.max()

        if stall > max_stall // 2:
            current_top_k = min(current_top_k + 1, 8)
        elif stall > max_stall:
            current_top_k = top_k

        g = soft_subgradient(y, ac, top_k=current_top_k, beta=2.0)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            x_r = best_a * (1 + 0.01 * rng.standard_normal(n))
            x_r = simplex_project(x_r)
            x = x_r
            x_prev = x_r
            stall = 0

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD-log t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Strategy:
    1. Multi-start from diverse initializations (fast)
    2. Load parent as additional start
    3. Polish best candidates with exact subgradient + coordinate descent
    4. Try nearby lengths
    5. Final two-opt polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 50.0, 1750.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-start exploration (20% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 1: Multi-start (t={t_phase1:.0f}s) ===")
    if t_phase1 > 30:
        rng_main = np.random.default_rng(12345)
        inits = diverse_initializations(n0, rng_main)
        # Add parent as first init
        inits.insert(0, norm_sum1(best_seq))

        # Quick evaluation of all inits
        init_scores = []
        for i, init in enumerate(inits):
            ac = fast_conv(init)
            m = ac.max()
            j = 2.0 * n0 * m
            init_scores.append((j, i, init))
            if i < 5:
                print(f"    Init {i}: J={j:.6f}")

        # Sort by score and polish top candidates
        init_scores.sort(key=lambda x: x[0])
        top_candidates = init_scores[:min(8, len(init_scores))]

        time_per_init = t_phase1 / len(top_candidates)
        for rank, (score, idx, init) in enumerate(top_candidates):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            t_alloc = min(time_per_init, remaining * 0.9)
            if t_alloc < 10:
                break
            print(f"    Polishing init {idx} (J={score:.6f}), t={t_alloc:.0f}s")
            a_out, _ = nesterov_amd_log(init, t_alloc, verbose=False, c0=3.0, top_k=1,
                                         seed=100 + idx)
            try_update(a_out, f"multi_{idx}")

    # === Phase 2: Exact subgradient polish (30% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase2 = min(remaining * 0.30, 450.0)
    print(f"\n=== Phase 2: Exact subgradient n={len(best_seq)} (t={t_phase2:.0f}s) ===")
    if t_phase2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = exact_subgradient_descent(a_in, t_phase2, verbose=True, seed=77,
                                              top_k_init=1, top_k_max=6)
        try_update(a_out, f"esd_n{len(best_seq)}")

    # === Phase 3: Smoothed subgradient with different params (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: Smoothed subgradient n={len(best_seq)} (t={t_phase3:.0f}s) ===")
    if t_phase3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smoothed_subgradient_descent(a_in, t_phase3, verbose=True, seed=200,
                                                 sigma_start=0.02, sigma_end=0.0005,
                                                 top_k=3, beta=1.5)
        try_update(a_out, f"ssd_n{len(best_seq)}")

    # === Phase 4: Coordinate descent (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase4 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: Coord descent n={len(best_seq)} (t={t_phase4:.0f}s) ===")
    if t_phase4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t_phase4, verbose=True, seed=42)
        try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 5: Two-opt swap (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Two-opt n={len(best_seq)} (t={t_phase5:.0f}s) ===")
    if t_phase5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = two_opt_swap(a_in, t_phase5, verbose=True, seed=999)
        try_update(a_out, f"twoopt_n{len(best_seq)}")

    # === Phase 6: Try larger n (10% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Larger n (remaining={remaining:.0f}s) ===")
    if remaining > 100:
        for delta in [200, 400]:
            n_try = n_best + delta
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            print(f"  Trying n={n_try}")
            a_long = interpolate_sequence(best_seq, n_try)
            t6 = min(remaining * 0.4, 150.0)
            if t6 > 20:
                a_out, _ = nesterov_amd_log(a_long, t6, verbose=False, c0=3.0, top_k=1, seed=500)
                try_update(a_out, f"amd_n{n_try}")
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 30:
                    a_out2, _ = two_opt_swap(norm_sum1(best_seq), min(remaining * 0.3, 80.0),
                                              verbose=False, seed=600)
                    try_update(a_out2, f"twoopt_n{n_best}")

    # === Phase 7: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_best} (remaining={remaining:.0f}s) ===")
    if remaining > 40:
        # Subgradient
        t7a = min(remaining * 0.4, 120.0)
        if t7a > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = exact_subgradient_descent(a_in, t7a, verbose=False, seed=777,
                                                  top_k_init=2, top_k_max=5)
            try_update(a_out, "esd_final")

        # Two-opt
        remaining = time_budget - (time.monotonic() - t_global)
        t7b = min(remaining * 0.4, 100.0)
        if t7b > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = two_opt_swap(a_in, t7b, verbose=False, seed=888)
            try_update(a_out, "twoopt_final")

        # Coordinate
        remaining = time_budget - (time.monotonic() - t_global)
        t7c = min(remaining * 0.4, 80.0)
        if t7c > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_descent(a_in, t7c, verbose=False, seed=999)
            try_update(a_out, "coord_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
