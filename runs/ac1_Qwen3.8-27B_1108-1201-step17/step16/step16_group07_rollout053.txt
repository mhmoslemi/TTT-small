    T = np.zeros((size, n))

        lo = max(0, k - n + 1)  # min j such that k-j is valid
        hi = min(n, k + 1)      # max j+1 such that k-j is valid

            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

def subgradient_at_k_fast(a, k, n):

def subgradient_optimize(a0, time_limit, verbose=False, seed=0):
    """
    Projected subgradient method with random peak selection and Polyak step size.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    # Lower bound: the average value of the convolution
    lower_bound = (norm ** 2) / (2 * n - 1)  # = 2n/(2n-1) ≈ 1
    t0 = time.monotonic()
    step = 1.0
    patience = 0
    max_patience = 3000

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Pick weighted random peaks (emphasize near-max)
        n_peaks = max(3, min(15, n // 60))
        # Softmax-like weights
        temp = M_curr * 0.03
        weights_raw = np.exp(-0.5 * (M_curr - conv) ** 2 / max(temp, 1e-10))
        weights_raw = weights_raw / max(weights_raw.sum(), 1e-15)
        # Sample peaks
        peak_idx = rng.choice(2 * n - 1, size=min(n_peaks, 2 * n - 1),
                              replace=False, p=weights_raw)

        # Average subgradient
        grad = np.zeros(n)
        for k in peak_idx:
            grad += subgradient_at_k_fast(a, int(k), n)
        grad /= len(peak_idx)

        # Polyak step size
        gap = M_curr - lower_bound
        if gap < 1e-10:
            break
        step = gap / max(np.dot(grad, grad), 1e-15)
        step = min(step, 2.0 / max(np.linalg.norm(grad), 1e-15))
        # Damped step
        step *= 0.5

        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
        else:
            # Try smaller step
            a_trial2 = a - 0.3 * step * grad
            a_trial2 = np.maximum(a_trial2, 0.0)
            s2 = a_trial2.sum()
            if s2 > 1e-15:
                a_trial2 = a_trial2 * norm / s2
                sc2 = fast_autocorr(a_trial2).max()
                if sc2 < best_score - 1e-14:
                    a = a_trial2
                    best_score = sc2
                    best_a = a.copy()
                    patience = 0
                    continue
            patience += 1

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        conv = fast_autocorr(a)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj_old = a[j]

            # Precompute the effect of changing a[j]
            # (a*a)[k] changes by: for k in [j, j+n-1], add 2*(new_aj - aj_old)*a[k-j]
            # But also the self-term at k=2j: add (new_aj^2 - aj_old^2) instead of 2*(new_aj-aj_old)*a[j]
            lo = j
            hi = min(size, j + n)

            # Try several candidate values
            typical = norm / n
            candidates = np.array([0.0, aj_old * 0.2, aj_old * 0.5, aj_old,
                                   aj_old * 1.5, aj_old * 2.0,
                                   typical * 0.3, typical, typical * 2.0,
                                   typical * 4.0])
            candidates = np.clip(candidates, 0.0, norm * 0.3)
            # Add golden ratio search around current best
            if aj_old > 0:
                candidates = np.append(candidates, [aj_old * 0.7, aj_old * 1.3])

            best_x = aj_old
            best_obj = conv.max() * 1.0  # current max (before any change, sum is norm)

            for x in candidates:
                if abs(x - aj_old) < 1e-12:
                    continue
                delta = x - aj_old
                new_conv = conv.copy()
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                # Self-term correction at k = 2j
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta - 2.0 * delta * a[j]

                new_sum = norm - aj_old + x
                if new_sum < 1e-15:
                    continue
                scale = norm / new_sum
                new_max = np.max(new_conv) * scale * scale

                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj_old) > 1e-10:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    improved = True

        if improved:
            new_score = fast_autocorr(a).max()
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_a = a.copy()
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def fourier_perturb(base, n, rng, strength=0.3):
    """Add random Fourier modes to create a structured perturbation."""
    a = base.copy()
    # Add 3-5 random Fourier modes
    n_modes = rng.integers(2, 6)
    for _ in range(n_modes):
        freq = rng.integers(1, n // 4)
        phase = rng.random() * 2 * np.pi
        amp = strength * rng.random() * (1.0 / np.sqrt(n))
        a += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = base.copy()
    else:
        a = a * np.sqrt(2.0 * n) / s
    return a


def random_perturb(base, n, rng, scale=0.1):
    """Add multiplicative Gaussian noise."""
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a = base * noise
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return base.copy()
    return a * np.sqrt(2.0 * n) / s


def binary_init(n, rng):
    """Create a random binary initialization."""
    a = (rng.random(n) > 0.5).astype(np.float64)
    s = a.sum()
    if s < 1:
        a = np.ones(n)
        s = n
    return a * np.sqrt(2.0 * n) / s


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False


def propose_candidate():
    """
    Main search: LP best-response (full/adaptive active set) + Nesterov,
    projected subgradient, coordinate descent, and structured restarts.
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = 1770.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base, endpoint=False)
        best_seq = 1.0 + 0.4 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x + 0.5)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.8f}")

    # === Phase 1: LP Nesterov with adaptive active set (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov (adaptive) n={n_base}, budget={t1:.0f}s ===")
        a_norm = normalize_to(best_seq, n_base)
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, seed=0, full_lp=False)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp1_n{n_base}", t_global)

    # === Phase 2: LP Nesterov with FULL active set ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP Nesterov (full LP) n={n_cur}, budget={t2:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True, seed=123, full_lp=True)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lpfull_n{n_cur}", t_global)

    # === Phase 3: Subgradient method ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 30:
        print(f"\n=== Phase 3: Subgradient n={n_cur}, budget={t3:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = subgradient_optimize(a_norm, t3, verbose=False, seed=7)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"subg_n{n_cur}", t_global)

    # === Phase 4: Multiple restarts with diverse perturbations ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_restarts = 4
    t4_total = min(remaining * 0.25, 350.0)
    t4_each = t4_total / n_restarts

    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        t4 = min(t4_each, remaining - 20)
        if t4 < 30:
            break

        # Different perturbation strategies
        if r == 0:
            perturbed = fourier_perturb(best_seq, n_cur, rng, strength=0.15)
            label = "fourier"
        elif r == 1:
            perturbed = random_perturb(best_seq, n_cur, rng, scale=0.08)
            label = "gauss"
        elif r == 2:
            perturbed = fourier_perturb(best_seq, n_cur, rng, strength=0.25)
            label = "fourier2"
        else:
            # Blend with a different structured sequence
            x = np.linspace(0.0, 1.0, n_cur, endpoint=False)
            alt = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 7 * x)
            alt = np.maximum(alt, 0.0)
            alt = normalize_to(alt, n_cur)
            blend = normalize_to(0.6 * best_seq + 0.4 * alt, n_cur)
            perturbed = blend
            label = "blend"

        print(f"\n=== Phase 4r{r}: {label} restart n={n_cur}, budget={t4:.0f}s ===")
        a_out, _ = nesterov_lp_optimize(perturbed, t4, solver, verbose=False,
                                         seed=1000 + r * 7, full_lp=False)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"{label}_n{n_cur}", t_global)

    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        print(f"\n=== Phase 5: CD polish n={n_cur}, budget={t5:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent(a_norm, t5, verbose=True, seed=55)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd_n{n_cur}", t_global)

    # === Phase 6: Final LP Nesterov (full) from best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 40:
        t6 = min(remaining * 0.7, 200.0)
        print(f"\n=== Phase 6: Final LP (full) n={n_cur}, budget={t6:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True,
                                         seed=7777, full_lp=True)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lpFinal_n{n_cur}", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
