def build_constraint_matrix(a, active_lags, n):
    """Build the constraint matrix T where T[k, i] = a[|k-i|] contribution for lag k."""
    T = np.zeros((len(active_lags), n))
    for row_idx, k in enumerate(active_lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            cols = np.arange(i_start, i_end)
            T[row_idx, cols] = a[k_int - cols]
    return T

                # The "away" step removes mass from the furthest active point
                # Condition: (g - p_i) . (x - p_i) < -gamma * ||x - p_i||^2
                x_center = np.array(active_weights) @ np.array(active_points)
                x_center = np.maximum(x_center, 0)
                sc = x_center.sum()
                if sc > 1e-15:
                    x_center = x_center / sc
                inner = np.dot(g_dir - active_points[i_ap], x_center - active_points[i_ap])
                quad = np.dot(x_center - active_points[i_ap], x_center - active_points[i_ap])
                if quad > 1e-15 and inner / quad < -gamma:
                    if inner / quad < gamma - 0.2:
                        gamma = inner / quad
                        away_idx = i_ap

        if away_idx >= 0 and gamma < -0.05:
            # Away step
            p_away = active_points[away_idx]
            # Find step sizes
            # Move from x in direction (g - p_away)
            direction = g_dir - p_away
            # alpha: max step towards g before leaving simplex
            # gamma: max step away from p_away
            min_ratio = min(r for r in [1.0])  # simplified
            alpha_fw = 2.0 / (t + 2.0)
            gamma_away = min(alpha_fw * 0.5, 0.3)

            a_new = a + alpha_fw * (g_dir - a)
            a_new = a_new - gamma_away * (p_away - a)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new

            # Update active set
            active_points.append(g_dir.copy())
            active_weights.append(alpha_fw)
            # Remove away point
            active_points.pop(away_idx)
            # Update remaining weights
            new_weights = []
            for w in active_weights:
                new_weights.append(w * (1.0 + gamma_away / (1.0 - gamma_away) if gamma_away < 1 else w))
            # Simplified: just keep the new point and best point
            active_points = [a_new.copy(), g_dir.copy()]
            active_weights = [0.5, 0.5]
        else:
            # Standard FW step
            a_new = (1.0 - alpha) * a + alpha * g_dir
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new

            # Update active set (keep max 5 points)
            active_points.append(g_dir.copy())
            if len(active_points) > 5:
                # Keep the most "spread out" points
                active_points = active_points[-5:]

        a = a_new
        m_new = fast_autocorr(a).max()

        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = score_from_maxac(best_m, n)
            print(f"    LP-AFW iter={t} ({elapsed:.1f}s): J={J:.8f}, constraints={len(constraint_lags)}")

    return best_a, best_m


# ============ Analytical Coordinate Descent ============

def coord_descent_analytical(a0, time_limit, verbose=True, rng_seed=77):
    """Coordinate descent using O(n) analytical updates (no FFT per coordinate)."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_ac = ac.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    sweeps = 0
    stall = 0
    max_stall = 8
    sweep_history = []

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        sweeps += 1
        improved = False
        sweep_best = best_m

        # Process in random order for better mixing
        order = list(range(n))
        rng.shuffle(order)

        for i in order:
            v = a[i]
            if v < 1e-20:
                # Try to increase from zero
                # The best value to set is where the largest affected lag equals current max
                # For simplicity, try a small value
                candidate_v = np.max(np.maximum(ac, 0)) / (2 * n * 0.01)
                candidate_v = min(candidate_v, 0.01)
                if candidate_v < 1e-15:
                    continue
                a[i] = candidate_v
                k_start = i
                k_end = min(i + n - 1, 2 * n - 2)
                for k in range(k_start, k_end + 1):
                    if k == 2 * i:
                        ac[k] += candidate_v * candidate_v
                    else:
                        ac[k] += 2 * a[k - i] * candidate_v
                if ac.max() < sweep_best:
                    sweep_best = ac.max()
                    improved = True
                else:
                    # Revert
                    a[i] = 0
                    for k in range(k_start, k_end + 1):
                        if k == 2 * i:
                            ac[k] -= candidate_v * candidate_v
                        else:
                            ac[k] -= 2 * a[k - i] * candidate_v
                continue

            # Find optimal v for coordinate i
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Non-affected max
            non_aff_max = 0.0
            if k_start > 0:
                non_aff_max = max(non_aff_max, np.max(ac[:k_start]))
            if k_end + 1 < len(ac):
                non_aff_max = max(non_aff_max, np.max(ac[k_end + 1:]))

            # Get top affected lags for candidate evaluation
            affected_idx = np.arange(k_start, k_end + 1)
            n_top = min(30, len(affected_idx))
            top_aff = affected_idx[np.argsort(ac[affected_idx])[::-1][:n_top]]
            if 2 * i <= 2 * n - 2:
                top_aff = np.unique(np.concatenate([top_aff, [2 * i]]))

            # Candidate values for v
            candidates = [v]
            # Try reducing towards where linear constraints meet
            for k in top_aff[:10]:
                if k == 2 * i:
                    # Quadratic: v^2 + ac[2i] - v^2 = ac[2i]... 
                    # At optimum, v^2 should equal non_aff_max - (ac[2i] - v^2)
                    # v^2 + ac[2i] - v^2 = ac[2i], so Q(v) = ac[2i] + v^2 - v^2 = ac[2i]
                    # Wait, that's just ac[2i] which is v^2. So Q(v) = v^2.
                    # At v_new, Q = v_new^2. We want v_new^2 = non_aff_max => v_new = sqrt(non_aff_max)
                    if non_aff_max > 0:
                        candidates.append(np.sqrt(non_aff_max))
                    break
                else:
                    # Linear: ac[k] + 2*a[k-i]*(v' - v)
                    # At v' where this equals non_aff_max:
                    # v' = v + (non_aff_max - ac[k]) / (2*a[k-i])
                    coeff = 2 * a[k - i]
                    if coeff > 1e-15:
                        v_cand = v + (non_aff_max - ac[k]) / coeff
                        if 0 <= v_cand <= 3 * v:
                            candidates.append(v_cand)

            # Also try 0 and some geometric interpolations
            candidates.append(0.0)
            candidates.append(v * 0.5)
            candidates.append(v * 1.5)

            # Evaluate candidates
            best_v = v
            best_trial_m = ac.max()  # current max

            for v_cand in candidates:
                v_cand = max(0.0, min(v_cand, v * 5))
                if abs(v_cand - v) < 1e-18:
                    continue

                # Compute new max efficiently
                trial_max = non_aff_max
                for k in top_aff:
                    if k == 2 * i:
                        ac_new_k = ac[k] - v * v + v_cand * v_cand
                    else:
                        ac_new_k = ac[k] + 2 * a[k - i] * (v_cand - v)
                    if ac_new_k > trial_max:
                        trial_max = ac_new_k

                if trial_max < best_trial_m - 1e-16:
                    best_trial_m = trial_max
                    best_v = v_cand

            if best_v != v and best_trial_m < sweep_best - 1e-16:
                # Apply update
                v_old = v
                v_new = best_v
                a[i] = v_new
                for k in range(k_start, k_end + 1):
                    if k == 2 * i:
                        ac[k] += v_new * v_new - v_old * v_old
                    else:
                        ac[k] += 2 * a[k - i] * (v_new - v_old)
                # Clip tiny negatives from numerical error
                np.clip(ac, 0, None, out=ac)
                sweep_best = best_trial_m
                improved = True

        # End of sweep: normalize and check
        s = a.sum()
        if s > 1e-15:
            a = a / s
            ac = fast_autocorr(a)  # Recompute to fix accumulated errors
        else:
            break

        current_m = ac.max()
        if current_m < best_m - 1e-16:
            best_m = current_m
            best_a = a.copy()
            best_ac = ac.copy()

        if improved and current_m < best_m * 1.001:
            stall = 0
        else:
            stall += 1
            if stall >= max_stall // 2:
                # Perturb
                n_pert = max(5, n // 100)
                idxs = rng.choice(n, size=n_pert, replace=False)
                factors = rng.uniform(0.8, 1.3, size=n_pert)
                a[idxs] *= factors
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                    ac = fast_autocorr(a)

        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            J = score_from_maxac(best_m, n)
            print(f"    CD sweep={sweeps} ({elapsed:.1f}s): J={J:.8f}, stall={stall}")

    return best_a, best_m


# ============ Nesterov Mirror Descent ============

def nmd_topk(a0, time_limit, initial_temp=0.01, final_temp=0.001, c0=3.0,
             top_k=50, beta=4.0, verbose=True, rng_seed=99):
    """NMD with top-k subgradient and restarts."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    max_stall = 2000
    running_avg = a.copy()
    avg_count = 1

    for t in range(1, 3000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, (t - 1) / 5000.0)
        temp = max(initial_temp * (1 - frac) + final_temp * frac, 1e-5)

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)

        # Top-k subgradient
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Restart
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(500, max_stall // 2)
            pert = rng.normal(0, 0.003, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            gamma = max(1.0, gamma * 0.7)

        # Running average
        running_avg += x
        avg_count += 1
        if avg_count % 300 == 0:
            avg = running_avg / avg_count
            avg = np.maximum(avg, 0.0)
            s_avg = avg.sum()
            if s_avg > 1e-15:
                avg = avg / s_avg
                M_avg = fast_autocorr(avg).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg.copy()

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = score_from_maxac(best_m, n)
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.8f}, temp={temp:.5f}")

    return best_a, best_m


# ============ Multi-length helper ============

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    n_base = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        x = np.linspace(0, 1, n_base)
        best_seq = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n0 = len(best_seq)

    # === Phase 1: LP-AFW (45% time) - main workhorse ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.45, 750.0)
    print(f"\n=== Phase 1: LP-AFW n={n0} (budget={t1:.0f}s) ===")
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_fw_afw(a_in, t1, solver, verbose=True, rng_seed=42)
        try_update(a_out, "LP-AFW")

    # === Phase 2: Analytical Coordinate Descent (30% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 2: CoordDescent n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_descent_analytical(a_in, t2, verbose=True, rng_seed=77)
        try_update(a_out, "CD")

    # === Phase 3: NMD with restarts (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 3: NMD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nmd_topk(a_in, t3, initial_temp=0.008, final_temp=0.001,
                            c0=3.0, top_k=40, beta=5.0, verbose=True, rng_seed=123)
        try_update(a_out, "NMD")

    # === Phase 4: Second CD pass on best so far (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: CD polish n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_descent_analytical(a_in, t4, verbose=True, rng_seed=200)
        try_update(a_out, "CD2")

    # === Phase 5: Try different length (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_alt = n0 + 200
        print(f"\n=== Phase 5: Alt length n={n_alt} (budget={remaining:.0f}s) ===")
        a_long = interpolate_sequence(best_seq, n_alt)
        a_in = norm_sum1(a_long)

        t5a = min(remaining * 0.5, 150.0)
        if t5a > 30:
            a_out, _ = lp_fw_afw(a_in, t5a, solver, verbose=False, rng_seed=55)
            try_update(a_out, f"LP_n{n_alt}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5b = min(remaining * 0.6, 100.0)
            a_in2 = norm_sum1(best_seq)
            if t5b > 20:
                a_out, _ = coord_descent_analytical(a_in2, t5b, verbose=False, rng_seed=300)
                try_update(a_out, f"CD_n{n0}_b")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
