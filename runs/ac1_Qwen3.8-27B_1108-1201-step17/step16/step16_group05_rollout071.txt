def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        rho = np.array([n - 1])
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_autocorr(a):
    """Full linear convolution a * a via FFT (matches np.convolve)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def logsumexp(x, axis=None):

    return m + np.log(np.sum(np.exp(x - m), axis=axis))

def interp_length(f, new_n):
    f = np.asarray(f, dtype=np.float64)
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    fx = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, fx)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return out / ss


def structured_inits(n):
    x = np.linspace(0.0, 1.0, n)
    inits = []
    rc = ((1 + np.cos(2 * np.pi * x)) * 0.5) ** 2          # raised cosine
    inits.append(rc / rc.sum())
    bim = np.exp(-40 * (x - 0.3) ** 2) + np.exp(-40 * (x - 0.7) ** 2)
    inits.append(bim / bim.sum())
    quad = np.maximum(1.0 - (2 * x - 1) ** 2, 0.0) ** 2     # quartic bump
    inits.append(quad / quad.sum())
    ends = 0.6 * (x ** 2) + 0.6 * ((1.0 - x) ** 2) + 0.2    # mass near ends
    inits.append(ends / ends.sum())
    inits.append(np.ones(n, dtype=np.float64) / n)
    return inits

def gs_flatten(a0, time_limit, verbose=False):

    n = len(a)

        A = np.fft.rfft(a, fft_size)
        P = (A.real * A.real + A.imag * A.imag)
        pmean = P.sum() / len(P)
        Pn = P.copy()
        for k in range(1, len(P)):
            Pn[k] = (1.0 - t) * P[k] + t * pmean
        mag = np.sqrt(np.maximum(Pn, 0.0))
        denom = np.maximum(np.abs(A), 1e-12)
        phase = A / denom
        Anew = mag * phase
        a_new = np.real(np.fft.irfft(Anew, fft_size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:

        else:
            a_new = a_new / s
        m_new = fast_autocorr(a_new).max()

def smooth_grad(a, beta, top_k):
    ac = fast_autocorr(a)
    n = len(a)
    k_eff = min(max(top_k, 1), len(ac))
    top_idx = np.argsort(ac)[::-1][:k_eff]
    top_vals = ac[top_idx]
    shifted = top_vals - top_vals[0]
    expv = np.exp(beta * shifted)
    w = expv / expv.sum()
    grad = np.zeros(n, dtype=np.float64)
    for wi, k in zip(w, top_idx):
        if wi <= 0.0:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad, float(ac.max())

def linear_fw_vertex(y, ac):
    """Cheapest-direction linear minimizer of the gradient over the simplex."""
    n = len(y)
    grad = np.zeros(n)
    top_idx = np.argsort(ac)[::-1][:12]
    for k in top_idx:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * y[j_idx]
    j = int(np.argmin(grad))
    v = np.zeros(n)
    v[j] = 1.0
    return v


def lp_fw_vertex(y, ac, M_curr, solver):
    n = len(y)
    b = cp.Variable(n, nonneg=True)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = y
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [T @ b <= M_curr])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    bv = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    bs = bv.sum()
    if bs < 1e-15:
        return None
    return bv / bs


def frank_wolfe(a0, time_limit, solver, verbose=False):
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    x_mom = y.copy()
    lp_fail = 0
    use_lp = (solver is not None)

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        v = None
        if use_lp:
            v = lp_fw_vertex(y, ac, M_curr, solver)
            if v is None:
                lp_fail += 1
                if lp_fail > 4:
                    use_lp = False
            else:
                lp_fail = 0
        if v is None:
            v = linear_fw_vertex(y, ac)
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        beta_t = (t - 1.0) / (t + 2.0)
        x_mom = y_new + beta_t * (y_new - y)
        x_mom = np.maximum(x_mom, 0.0)
        sx = x_mom.sum()
        if sx < 1e-15:
            x_mom = y_new.copy()
        else:
            x_mom = x_mom / sx
        y = y_new

        for cand in (y, x_mom):
            mc = fast_autocorr(cand).max()
            if mc < best_m - 1e-15:
                best_m = mc
                best_a = cand.copy()
        if verbose and t % 4 == 0:
            print("      FW iter=%d (%.0fs) J=%.6f lp=%s"
                  % (t, time.monotonic() - t0, 2.0 * n * best_m, use_lp))
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = max(60.0, min(float(budget_s) - 120.0, 1700.0))
    solver = get_solver()
    print("Time budget: %.0fs" % time_budget)

    state = {"best": None, "score": float('inf')}

    def remaining():
        return time_budget - (time.monotonic() - t_global)

    def try_update(a, label):
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-8:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < state["score"]:
            state["score"] = sc
            state["best"] = a.copy()
            print("  *** NEW BEST [%s]: %.6f (n=%d, t=%.0fs)"
                  % (label, sc, len(a), time.monotonic() - t_global))

    # ---- initialisation ----
    par = parent_sequence()
    if par is not None:
        a0 = norm_sum1(par)
        print("Parent: n=%d score=%.6f" % (len(a0), evaluate_safe(a0)))
    else:
        a0 = structured_inits(2400)[0]
        print("Default raised-cosine init n=2400")
    try_update(a0, "init")

    # ---- Phase 1: spectrum flattening for a diverse, well-spread start ----
    if remaining() > 40:
        t1 = min(remaining() * 0.06, 60.0)
        gs, _, _ = gs_flatten(state["best"], t1, verbose=False)
        try_update(gs, "gs_n%d" % len(gs))

    # ---- Phase 2: main APG ----
    if remaining() > 60:
        t2 = min(remaining() * 0.34, 520.0)
        out, _ = apg_smoothed(state["best"], t2,
                              beta_start=35.0, beta_end=6.0,
                              top_k=240, c0=2.4, verbose=True, seed=42)
        try_update(out, "apg_n%d" % len(out))

    # ---- Phase 3: peak shaving ----
    if remaining() > 40:
        t3 = min(remaining() * 0.14, 160.0)
        out, _ = peak_shave(state["best"], t3, verbose=True, seed=7)
        try_update(out, "shave_n%d" % len(out))

    # ---- Phase 4: Frank-Wolfe polish ----
    if remaining() > 50 and len(state["best"]) <= 3200:
        t4 = min(remaining() * 0.18, 260.0)
        out, _ = frank_wolfe(state["best"], t4, solver, verbose=True)
        try_update(out, "fw_n%d" % len(out))

    # ---- Phase 5: APG polish after FW ----
    if remaining() > 40:
        t5 = min(remaining() * 0.10, 120.0)
        out, _ = apg_smoothed(state["best"], t5,
                              beta_start=18.0, beta_end=9.0,
                              top_k=300, c0=1.6, verbose=False, seed=99)
        try_update(out, "apg_polish_n%d" % len(out))

    # ---- Phase 6: longer length exploration ----
    if remaining() > 150:
        n1 = min(len(state["best"]) + 800, 3400)
        if n1 > len(state["best"]):
            a_long = interp_length(state["best"], n1)
            t6 = min(remaining() * 0.16, 200.0)
            out, _ = apg_smoothed(a_long, t6,
                                  beta_start=25.0, beta_end=7.0,
                                  top_k=260, c0=2.2, verbose=True, seed=131)
            try_update(out, "apg_long_n%d" % n1)
            if remaining() > 50:
                t6b = min(remaining() * 0.25, 90.0)
                out2, _ = peak_shave(out, t6b, verbose=False, seed=55)
                try_update(out2, "shave_long_n%d" % n1)

    # ---- Phase 7: structured restarts on the best length ----
    if remaining() > 120:
        nb = len(state["best"])
        inits = structured_inits(nb)
        per = min(remaining() * 0.16, 90.0)
        for k in range(len(inits)):
            if remaining() < 40:
                break
            init = project_simplex(0.5 * norm_sum1(state["best"]) + 0.5 * inits[k])
            t7 = min(per, remaining() - 30.0)
            if t7 < 20:
                break
            out, _ = apg_smoothed(init, t7,
                                  beta_start=24.0, beta_end=9.0,
                                  top_k=200, c0=2.0, verbose=False, seed=200 + k)
            try_update(out, "restart%d_n%d" % (k, nb))

    # ---- Phase 8: final peak shaving ----
    if remaining() > 30:
        t8 = min(remaining() * 0.6, 140.0)
        out, _ = peak_shave(state["best"], t8, verbose=True, seed=314)
        try_update(out, "shave_final_n%d" % len(out))

    final = state["best"]
    if final is None:
        final = np.ones(2400, dtype=np.float64) / 2400
    fs = evaluate_safe(final)
    if np.isfinite(fs):
        state["score"] = fs
    print("FINAL: score=%.6f n=%d time=%.0fs"
          % (state["score"], len(final), time.monotonic() - t_global))
    return [float(x) for x in final]
