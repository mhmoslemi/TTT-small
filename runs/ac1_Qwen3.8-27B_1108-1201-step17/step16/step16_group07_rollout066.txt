"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2  (== max(conv(a,a)) when sum(a)=sqrt(2n)).

Core method (new): smoothed gradient descent on the log-sum-exp surrogate
    F_tau(a) = tau * log( sum_k exp(conv_k / tau) )
whose gradient at 'a' is a softmax-weighted sum of the per-lag subgradients,
computed in O(n log n) via FFT. Uses adaptive temperature annealing, Nesterov
momentum and adaptive step size. Refined by LP best-response (progressive
active set) and a fast coordinate descent, with multiple structured
re-starts and a small sweep over the length n.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def next_pow2(m):
    return 1 << max(0, (m - 1).bit_length())


def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    N = next_pow2(size)
    A = rfft(np.asarray(a, dtype=np.float64), N)
    return irfft(A * A, N)[:size]


def fast_grad(a, weights):
    # grad[j] = 2 * sum_m a[m] * weights[j+m]   (valid m, j+m)
    n = len(a)
    N = next_pow2(2 * n - 1)
    W = rfft(np.asarray(weights, dtype=np.float64), N)
    A = rfft(np.asarray(a, dtype=np.float64), N)
    cross = irfft(np.conj(W) * A, N)
    grad = np.zeros(n, dtype=np.float64)
    grad[0] = cross[0]
    if n > 1:
        grad[1:] = cross[N - 1 : N - n : -1]
    grad *= 2.0
    return grad


def normalize(a, n):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * np.sqrt(2.0 * n) / n
    return a * (np.sqrt(2.0 * n) / s)


def softmax_weights(conv, tau):
    mc = conv.max()
    e = np.exp((conv - mc) / tau)
    return e / e.sum()


def evaluate_safe(a):
    try:

        sc = evaluate_sequence(lst)

    except Exception:

def smoothed_gd(a0, time_limit, verbose=False, seed=42):
    """Smoothing-grad (log-sum-exp) + Nesterov + adaptive temperature/step."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = normalize(a0, n)
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()

    v = np.zeros(n)
    momentum = 0.90
    lr = 1.2 / np.sqrt(n)
    prev_score = best_score
    patience = 0

    # temperature schedule in units of the current max
    frac_start = 0.06
    frac_end = 0.006

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > 2500:
            break

        conv = fast_conv(a)
        M = conv.max()
        if M < 1e-12:
            break

        progress = min(1.0, t / 4000.0)
        frac = frac_start * (frac_end / frac_start) ** progress
        # occasional random temperature for exploration early on
        if t < 300 and rng.random() < 0.1:
            frac = frac * (0.5 + rng.random())
        tau = max(frac * M, 1e-9)

        weights = softmax_weights(conv, tau)
        grad = fast_grad(a, weights)
        grad = grad - grad.mean()  # keep the sum direction neutral

        gnorm = np.linalg.norm(grad)
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, n)

        score = fast_conv(a_new).max()

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            patience = 0
            a = a_new
            prev_score = score
            lr = min(lr * 1.03, 6.0 / np.sqrt(n))
        else:
            # no improvement: shrink step, reduce velocity, lower temperature
            a = a_new  # still move (exploration)
            lr *= 0.85
            v *= 0.5
            patience += 1
            if patience % 200 == 0:
                # nudge a random subset to escape basins
                idx = rng.choice(n, size=max(1, n // 4), replace=False)
                a[idx] *= (0.7 + 0.6 * rng.random(len(idx)))
                a = normalize(a, n)
            if lr < 1e-6 / np.sqrt(n):
                break

        if verbose and t % 500 == 0:
            el = time.monotonic() - t0
            if el < time_limit:
                print(f"    SGD t={t} ({el:.0f}s): best={best_score:.10f} tauF={frac:.4f} lr={lr:.5f}")

    return best_a, best_score


def lp_best_response(a_fixed, M, active_set, solver):
    n = len(a_fixed)
    size = 2 * n - 1
    k_indices = np.asarray(active_set, dtype=np.int64)
    kk = k_indices[k_indices < size]
    if len(kk) == 0:
        return None, None
    T = np.zeros((len(kk), n))
    for r, k in enumerate(kk):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[r, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    b_var = cp.Variable(n, nonneg=True)
    cons = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dv = cons[0].dual_value
        if dv is not None:
            duals = np.asarray(dv, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def nesterov_lp(a0, time_limit, solver, verbose=False, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm_v = np.sqrt(2.0 * n)
    a = normalize(a0, n)
    best_a = a.copy()
    best_score = fast_conv(a).max()
    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall = 0
    nit = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_conv(y)
        M = conv.max()
        if M < 1e-12:
            break
        if active_set is None or t % 30 == 0:
            progress = min(1.0, t / 800.0)
            tol = 0.08 * (1.0 - progress) + 0.004 * progress
            base = M * (1.0 - tol)
            active_set = np.where(conv >= base)[0]
            ntop = max(5, n // 80)
            topk = np.argsort(conv)[-ntop:]
            active_set = np.unique(np.concatenate([active_set, topk]))
            cap = int(1.5 * n)
            if len(active_set) > cap:
                keep = np.argsort(conv[active_set])[-cap:]
                active_set = active_set[keep]

        b, duals = lp_best_response(y, M, active_set, solver)
        if b is None:
            if t % 5 == 0:
                b, duals = lp_best_response(y, M, active_set, solver)
            if b is None:
                continue
        g = b * (norm_v / b.sum())
        nit += 1

        alpha = 2.0 / (t + 2.0)
        y_new = np.maximum((1.0 - alpha) * y + alpha * g, 0.0)
        s = y_new.sum()
        y_new = y_new * (norm_v / s) if s > 1e-15 else y.copy()
        mom = min((t - 1.0) / (t + 2.0), 0.93)
        a_new = y_new + mom * (y_new - y_prev)
        a_new = np.maximum(a_new, 0.0)
        s2 = a_new.sum()
        a_new = a_new * (norm_v / s2) if s2 > 1e-15 else y_new.copy()
        y_prev = y
        y = y_new
        a = a_new

        sy = fast_conv(y).max()
        if sy < best_score:
            best_score, best_a, stall = sy, y.copy(), 0
        else:
            stall += 1
        sa = fast_conv(a).max()
        if sa < best_score:
            best_score, best_a, stall = sa, a.copy(), 0

        if t % 50 == 0:
            d = 0.7 * best_a + 0.3 * a
            d = normalize(d, n)
            sd = fast_conv(d).max()
            if sd < best_score:
                best_score, best_a, stall = sd, d.copy(), 0

        if stall > 40:
            y = y_prev = a = best_a.copy()
            active_set = None
            stall = 0

        if verbose and nit % 300 == 0:
            el = time.monotonic() - t0
            if el < time_limit:
                print(f"    LP it={nit} ({el:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def subgradient_ify(a, k, n):
    grad = np.zeros(n)
    lo1 = max(0, k - n + 1)
    hi1 = min(n, k + 1)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        grad[lo1:hi1] += a[k - idx]
    lo2 = max(0, k)
    hi2 = min(n, k + n)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[lo2:hi2] += a[idx - k]
    return grad


def multi_subgradient(a0, time_limit, seed=1):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm_v = np.sqrt(2.0 * n)
    a = normalize(a0, n)
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    step = 1.0 / n ** 0.35
    patience = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > 600:
            break
        conv = fast_conv(a)
        M = conv.max()
        npeaks = max(3, min(10, n // 80))
        tops = np.argsort(conv)[-npeaks:]
        if t % 15 == 0:
            thr = M * (0.85 + 0.1 * rng.random())
            cand = np.where(conv >= thr)[0]
            if len(cand) > 0:
                tops = np.append(tops, cand[rng.integers(len(cand))])
        tops = tops.astype(int)
        w = np.exp(-8.0 * (M - conv[tops]) / max(M * 0.02, 1e-10))
        w = w / w.sum()
        grad = np.zeros(n)
        for wt, k in zip(w, tops):
            grad += wt * subgradient_ify(a, k, n)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        trial = a - step * grad * n ** 0.35 / gn
        trial = normalize(trial, n)
        sc = fast_conv(trial).max()
        if sc < best_score - 1e-14:
            a = trial
            best_score, best_a, patience = sc, trial.copy(), 0
            step = min(step * 1.004, 2.5)
        else:
            patience += 1
            step *= 0.985
            if step < 1e-10:
                break
        if time.monotonic() - t0 > time_limit:
            break
    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm_v = np.sqrt(2.0 * n)
    a = normalize(a0, n)
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_conv(a)
        for j in rng.permutation(n):
            if time.monotonic() - t0 > time_limit:
                break
            aj = a[j]
            lo = j
            hi = min(2 * n - 1, j + n)
            if lo >= hi:
                continue
            base = norm_v / n
            cands = {0.0, aj * 0.4, aj * 0.75, aj * 1.0, aj * 1.35, aj * 1.8,
                     base * 0.5, base, base * 1.5, base * 3.0}
            best_x = aj
            best_obj = conv.max()
            for x in cands:
                x = float(min(max(x, 0.0), 0.5 * norm_v))
                if x == aj:
                    continue
                new_conv = conv.copy()
                d = x - aj
                new_conv[lo:hi] += 2.0 * d * a[lo - j:hi - j]
                if 2 * j < (2 * n - 1):
                    new_conv[2 * j] += d * d
                obj = new_conv.max() * ((norm_v / (norm_v - aj + x)) ** 2)
                if obj < best_obj:
                    best_obj = obj
                    best_x = x
            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                a = normalize(a, n)
                sc = fast_conv(a).max()
                if sc < best_score - 1e-13:
                    best_score, best_a = sc, a.copy()
                    improved = True
                else:
                    a = best_a.copy()
                    break
        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0
    return best_a, best_score


def perturb(seq, rng, scale_max):
    n = len(seq)
    sc = 0.01 + 0.03 * rng.random()
    noise = 1.0 + sc * rng.standard_normal(n)
    noise = np.maximum(noise, 0.2)
    p = seq * noise
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-15:
        return seq.copy()
    return p / s


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    if a.sum() < 0.01:
        return best_seq, best_score, False
    a = a / a.sum()
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = np.maximum(a, 0.0).copy()
        el = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={el:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False

    if t3 > 25:
        print(f"\n=== P3 smoothed-GD r2 n={n_cur} budget={t3:.0f}s ===")
        a_out, _ = smoothed_gd(normalize(best_seq, n_cur), t3, verbose=True, seed=77)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"sgd2_n{n_cur}", t_global)

    # ---- Phase 4: multi-subgradient ----
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 20:
        print(f"\n=== P4 subgradient n={n_cur} budget={t4:.0f}s ===")
        a_out, _ = multi_subgradient(normalize(best_seq, n_cur), t4, seed=5)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"sg_n{n_cur}", t_global)

    # ---- Phase 5: coordinate descent ----
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        print(f"\n=== P5 CD n={n_cur} budget={t5:.0f}s ===")
        a_out, _ = coordinate_descent(normalize(best_seq, n_cur), t5, seed=9)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd_n{n_cur}", t_global)

    # ---- Phase 6: structured restarts on same n ----
    n_cur = len(best_seq)
    for r in range(2):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 70:
            break
        tr = min(remaining * 0.10, 200.0)
        if tr < 30:
            break
        print(f"\n=== P6 restart{r} n={n_cur} budget={tr:.0f}s ===")
        ap = perturb(best_seq, rng, 0.04)
        a_out, _ = smoothed_gd(normalize(ap, n_cur), tr, verbose=False, seed=300 + r)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"r{r}_n{n_cur}", t_global)
        a_out, _ = coordinate_descent(normalize(best_seq, n_cur), min(60.0, (time_budget - (time.monotonic() - t_global)) * 0.3), seed=r + 40)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"r{r}cd_n{n_cur}", t_global)

    # ---- Phase 7: sweep a longer length for extra attack surface ----
    for n_trial in (1700,):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        tt = min(remaining * 0.25, 300.0)
        if tt < 60:
            break
        print(f"\n=== P7 sweep n={n_trial} budget={tt:.0f}s ===")
        # band-limit the parent up to n_trial for a sane start
        xp = np.linspace(0.0, 1.0, n_trial)
        from numpy import interp
        xv = np.linspace(0.0, 1.0, len(best_seq))
        start = interp(xp, xv, best_seq)
        start = np.maximum(start, 0.0)
        start = start / start.sum()
        a_out, _ = smoothed_gd(normalize(start, n_trial), tt * 0.6, verbose=True, seed=2)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"sweep_sgd_n{n_trial}", t_global)
        a_out, _ = coordinate_descent(normalize(best_seq, n_trial), min(60.0, tt * 0.4), seed=3)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"sweep_cd_n{n_trial}", t_global)

    # ---- Final micro-polish ----
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 25:
        tf = min(remaining * 0.6, 60.0)
        a_out, _ = coordinate_descent(normalize(best_seq, n_cur), tf, seed=str(
            max(0, int(time_budget - (time.monotonic() - t_global)) % 1000)))
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"finalcd_n{n_cur}", t_global)

    if best_seq is None:
        return [1.0]

    fs = evaluate_safe(best_seq)
    if np.isfinite(fs):
        best_score = fs
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, t={time.monotonic() - t_global:.0f}s")
    out = np.maximum(best_seq, 0.0)
    return [float(x) for x in out]
