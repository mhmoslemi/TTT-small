def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    result = np.maximum(result, 0.0)
    return result


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)

    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def random_spectral_design(n, rng, target_flatness=0.7):
    """Generate a random non-negative sequence with approximately flat autocorrelation."""
    fft_size = 1 << (2 * n - 1).bit_length()
    # Random phases in frequency domain
    phase = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    # Mix flat and random magnitude to control flatness
    mag = np.ones(fft_size // 2 + 1)
    # Add some random variation
    variation = rng.normal(1.0, (1 - target_flatness) * 0.3, fft_size // 2 + 1)
    variation = np.maximum(variation, 0.1)
    mag *= variation
    
    # Construct complex spectrum with Hermitian symmetry
    A_pos = mag * np.exp(1j * phase)
    A = np.zeros(fft_size, dtype=np.complex128)
    A[:fft_size // 2 + 1] = A_pos
    # Enforce Hermitian symmetry
    A[1:fft_size // 2 + 1] = A_pos[1:fft_size // 2 + 1]
    if fft_size // 2 + 1 < fft_size:
        A[fft_size // 2 + 1:] = np.conj(A_pos[1:fft_size // 2 + 1][::-1])
    
    # Take IFFT
    a_full = np.fft.ifft(A).real
    # We need to extract a length-n sequence whose autocorrelation matches
    # Use the first n samples as our sequence
    a = np.maximum(a_full[:n], 0.0)
    a = norm_sum1(a)
    return a


def random_annealing_design(n, rng, base_seq=None):
    """Generate a perturbed version of base_seq using simulated annealing steps."""
    if base_seq is None:
        return np.ones(n) / n
    a = base_seq.copy()
    n_steps = 50
    temperature = 0.1
    current_obj = fast_autocorr(a).max()
    
    for step in range(n_steps):
        # Pick two random positions
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Move some mass from i to j or vice versa
        if a[i] > 0.01:
            frac = rng.uniform(0.05, 0.3) * min(a[i], a[j] + 0.01)
            a[i] -= frac
            a[j] += frac
        else:
            frac = rng.uniform(0.05, 0.3) * a[j]
            a[j] -= frac
            a[i] += frac
        
        a = np.maximum(a, 0.0)
        a = norm_sum1(a)
        new_obj = fast_autocorr(a).max()
        
        # Accept if better or with probability exp(-delta/T)
        if new_obj < current_obj or rng.random() < np.exp(-(new_obj - current_obj) / temperature):
            current_obj = new_obj
        else:
            # Revert
            a[i] += frac
            a[j] -= frac
            a = np.maximum(a, 0.0)
            a = norm_sum1(a)
        
        temperature *= 0.98
    
    return a


def lp_frank_wolfe_improved(a0, time_limit, solver, verbose=True, use_all_constraints=True,
                            nest_acc=True, top_K=500):
    """
    Improved LP-based Frank-Wolfe with:
    - Full or top-K constraint selection
    - Nesterov acceleration
    - Adaptive step size
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()  # Nesterov extrapolated point
    x_nest = a.copy()  # Nesterov momentum point
    prev_y = a.copy()
    iter_count = 0
    no_improve_count = 0

    n_constraints = 2 * n - 1
    # Decide how many constraints to use
    if use_all_constraints and n <= 4000:
        K_sel = n_constraints
    else:
        K_sel = min(top_K, n_constraints)

    b_var = cp.Variable(n, nonneg=True)

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute autocorrelation and current peak
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-18:
            break

        # Build constraint matrix T where (T @ b)[k] = (y * b)[k]
        # T[k, i] = y[k - i] if 0 <= k - i < n, else 0
        # For efficiency, use the top-K most binding constraints
        if K_sel < n_constraints:
            # Compute constraint values for all k
            # (y * b)[k] for b = ones/n would be proportional to row sums
            # Instead, use the current ac as a proxy for binding constraints
            # The most binding constraints are those where ac[k] is closest to M_curr
            margins = M_curr - ac
            binding_idx = np.argsort(margins)[:K_sel]
        else:
            binding_idx = np.arange(n_constraints)

        # Build T_sel matrix (sparse)
        T_sel = np.zeros((len(binding_idx), n), dtype=np.float64)
        for row, k in enumerate(binding_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[k_int - i]

        M_vec = np.full(len(binding_idx), M_curr)

        constraints = [T_sel @ b_var <= M_vec]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            no_improve_count += 1
            if no_improve_count > 20:
                break
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve_count += 1
            if no_improve_count > 20:
                break
            continue

        no_improve_count = 0
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Step size selection
        if nest_acc:
            alpha = 2.0 / (t + 2.0)
        else:
            alpha = 1.0 / (t + 1.0)

        # Update y (standard FW point)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Nesterov momentum update
        if nest_acc:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - prev_y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        else:
            x_new = y_new.copy()

        prev_y = y
        y = y_new
        x_nest = x_new
        iter_count += 1

        # Check improvement
        my = fast_autocorr(y).max()
        if my < best_m - 1e-16:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x_nest).max()
        if mx < best_m - 1e-16:
            best_m = mx
            best_a = x_nest.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, K={len(binding_idx)}")

    return best_a, best_m


def soft_subgradient_descent(a0, time_limit, top_k=60, beta=4.0, c0=5.0,
                             restart_every=3000, verbose=False):
    """Adaptive mirror descent with exponential-weighted soft subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 30000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute soft subgradient
        ac = fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        # Mirror descent step (exponential family / softmax update)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            # Diversification restart
            strategy = rng.integers(3)
            if strategy == 0:
                # Gaussian perturbation
                pert = rng.normal(0, 0.03, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                # Random spectral design blend
                spectral = random_spectral_design(n, rng)
                x_r = 0.5 * best_a + 0.5 * spectral
            else:
                # Annealing design
                x_r = random_annealing_design(n, rng, base_seq=best_a)
            
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            # Adapt parameters
            cur_top_k = min(cur_top_k + 10, 100)
            cur_beta = max(cur_beta - 0.3, 2.0)
            gamma = c0 * (1 + 0.1 * (t // restart_every))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def coordinate_descent_polish(a0, time_limit, verbose=True, rng_seed=777):
    """Coordinate descent with sensitivity-based pair moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(15, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved_this_round = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-25:
                continue

            # Find the pair that contributes most
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Compute sensitivity for all positions
            n_sens = min(8, n_top)
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:n_sens]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                in_peak[s2:e2 + 1] = True

            # Score: how much would adding mass at position d increase peaks?
            scores = np.zeros(n)
            for k2 in top_k[:n_sens]:
                k2i = int(k2)
                for d in range(n):
                    d1 = k2i - d
                    d2 = d - k2i
                    add = 0.0
                    if 0 <= d1 < n:
                        add += a[d1]
                    if 0 <= d2 < n:
                        add += a[d2]
                    scores[d] += add

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            candidate_scores = scores[candidates]
            if len(candidate_scores) > 0:
                best_dst = candidates[np.argmin(candidate_scores)]
            else:
                best_dst = int(rng.integers(n))

            dst = best_dst
            if dst == src:
                continue

            # Try multiple move sizes
            found = False
            for frac in [0.003, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-18:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved_this_round = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 30:
                # Forced random move
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved_this_round = True

        if not improved_this_round:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def make_structured_init(n, kind="gaussian"):
    """Create structured initial sequences."""
    if kind == "gaussian":
        x = np.linspace(-2, 2, n)
        a = np.exp(-0.5 * x**2 / 0.8**2)
    elif kind == "bimodal":
        x = np.linspace(0, 1, n)
        a = np.exp(-40 * (x - 0.2)**2) + np.exp(-40 * (x - 0.8)**2)
    elif kind == "cosine_taper":
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5))**2 + 0.01
    elif kind == "inverse_u":
        x = np.linspace(0.01, 0.99, n)
        a = 1.0 / (0.3 + x * (1 - x) * 4)
    else:
        a = np.ones(n)
    return norm_sum1(np.maximum(a, 0.0))

    print(f"\n=== Phase 2: LP-FW Main (n={n_best}, t={t_phase2:.0f}s) ===")
    if t_phase2 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_improved(a_in, t_phase2, solver, verbose=True,
                                            use_all_constraints=(n_best <= 3500),
                                            nest_acc=True, top_K=500)
        try_update(a_out, f"lpfw_n{n_best}")

    # === Phase 3: Spectral diversification ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t_phase3 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 3: Spectral Diversification (n={n_best}, t={t_phase3:.0f}s) ===")
    if t_phase3 > 30:
        n_trials = 5
        t_per_trial = t_phase3 / n_trials
        for trial in range(n_trials):
            elapsed_check = time.monotonic() - t_global
            if time_budget - elapsed_check < 60:
                break
            if trial % 2 == 0:
                spec_seq = random_spectral_design(n_best, rng, target_flatness=0.5 + 0.1 * trial)
                # Blend with current best
                blend_a = 0.4 * norm_sum1(best_seq) + 0.6 * spec_seq
            else:
                blend_a = random_annealing_design(n_best, rng, base_seq=best_seq)
            
            a_out, _ = soft_subgradient_descent(blend_a, t_per_trial, top_k=50, beta=4.0,
                                                 c0=5.0, restart_every=1500, verbose=False)
            try_update(a_out, f"spec_{trial}")

    # === Phase 4: Second LP-FW round on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t_phase4 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 4: LP-FW Round 2 (n={n_best}, t={t_phase4:.0f}s) ===")
    if t_phase4 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_improved(a_in, t_phase4, solver, verbose=True,
                                            use_all_constraints=(n_best <= 3500),
                                            nest_acc=True, top_K=400)
        try_update(a_out, f"lpfw2_n{n_best}")

    # === Phase 5: Length variation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t_phase5 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 5: Length Variation (t={t_phase5:.0f}s) ===")
    if t_phase5 > 60:
        candidate_lengths = [n_best + 200, n_best - 200, n_best + 400]
        candidate_lengths = [l for l in candidate_lengths if 2000 <= l <= 5000]
        t_per_len = t_phase5 / max(1, len(candidate_lengths))
        for new_n in candidate_lengths[:2]:
            elapsed_check = time.monotonic() - t_global
            if time_budget - elapsed_check < 60:
                break
            a_long = interpolate_sequence(best_seq, new_n)
            a_out, _ = soft_subgradient_descent(a_long, t_per_len, top_k=50, beta=5.0,
                                                 c0=4.0, restart_every=2000, verbose=False)
            try_update(a_out, f"len_{new_n}")

    # === Phase 6: Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t_phase6 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 6: Final Coord Polish (n={n_best}, t={t_phase6:.0f}s) ===")
    if t_phase6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_polish(a_in, t_phase6, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 7: One more LP if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40 and n_best <= 3500:
        t_phase7 = min(remaining * 0.4, 150.0)
        print(f"\n=== Phase 7: Extra LP (n={n_best}, t={t_phase7:.0f}s) ===")
        if t_phase7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_improved(a_in, t_phase7, solver, verbose=True,
                                                use_all_constraints=True,
                                                nest_acc=True, top_K=600)
            try_update(a_out, f"lpfw3_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_structured_init(n_fb, "cosine_taper")
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
