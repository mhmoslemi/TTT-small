def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fourier_project(a, alpha=0.5):
    """
    Fourier-domain 'square root' heuristic.
    Computes b = conv(a,a), then tries to find a' such that conv(a',a') ≈ b_flattened.
    Actually, we compute a_suggested = real(IFFT(sqrt(FFT(b)))) and blend.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    b = fast_conv(a)
    # Pad b to fft_size
    b_padded = np.zeros(fft_size)
    b_padded[:size] = b
    
    B = np.fft.rfft(b_padded)
    # B should be real and non-negative (it's |A|^2 for some A)
    B = np.maximum(B.real, 0)
    A = np.sqrt(B)
    a_full = np.fft.irfft(A, fft_size)
    a_suggested = a_full[:n].copy()
    
    # Project to non-negative
    a_suggested = np.maximum(a_suggested, 0)
    s = a_suggested.sum()
    if s < 1e-15:
        return a.copy()
    a_suggested = a_suggested / s
    
    # Blend
    a_new = (1 - alpha) * a + alpha * a_suggested
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    return a_new

def soft_subgradient(y, ac, top_k=5, beta=3.0):
    """Soft subgradient: weighted average of top-k peak gradients."""
    n = len(y)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        grad += wi * hard_subgradient(y, int(k))
    return grad

def hard_subgradient(y, k_star):
    """Exact subgradient of C[k_star] w.r.t. y."""
    n = len(y)
    k = int(k_star)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    for i in range(i_start, i_end):
        j = k - i
        if 0 <= j < n:
            grad[i] += 2.0 * y[j]
    return grad

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
