**Fast approach: Subgradient method with gradient computed in O(n) per active lag**

Since `∇b[k](a) = 2 * (a shifted by k)`, computing the gradient for a given k is O(n) (just a shift). For the subgradient of `max_k b[k](a)`, we use a convex combination of the gradients of the active lags.

The subgradient at `a` is: `g = sum_{k in active} λ_k * 2 * (a shifted by k)` where `λ` is a probability distribution over active lags.

If we use uniform λ over the active lags, this is: `g = (2/|active|) * sum_{k in active} (a shifted by k)`.

This can be computed in O(n * |active|) time, which for |active| ~ 20 and n = 2400 is about 48000 operations. Very fast.

import numpy as np
import cvxpy as cp
import time

def fast_autocorr(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_grad_lag(a, k):
    """Compute gradient of b[k] = sum_i a[i]*a[k-i] w.r.t. a.
    Result: grad[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    # j ranges from max(0, k-n+1) to min(k, n-1)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        # grad[j] = 2 * a[k - j] for j in [j_start, j_end]
        # k - j ranges from k - j_end to k - j_start
        idx = np.arange(j_start, j_end + 1)
        a_idx = k - idx  # indices into a
        grad[idx] = 2.0 * a[a_idx]
    return grad

def nesterov_mirror_descent_fast(a0, time_limit, top_k=15, c0=3.0, verbose=True):
    """Fast Nesterov mirror descent with top-k subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 2000
    rng = np.random.default_rng(42)
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        ac = fast_autocorr(y)
        M = ac.max()
        
        # Top-k subgradient
        if top_k >= len(ac):
            top_idx = np.arange(len(ac))
        else:
            # Get top k lags (use argpartition for speed)
            top_idx = np.argpartition(ac, -top_k)[-top_k:]
        
        # Compute subgradient as average of gradients at top-k lags
        g = np.zeros(n)
        for k in top_idx:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                g[idx] += 2.0 * y[k_int - idx]
        g /= len(top_idx)
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restart
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(500, max_stall * 0.7)
            pert = rng.normal(0, 0.01, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            c0 = max(0.5, c0 * 0.5)
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD-fast t={t} ({elapsed:.1f}s): J={J:.8f}, stall={stall_count}")
    
    return best_a, best_m

def lp_remez_step(a, max_constraints=50, r=0.1, solver=None):
    """One step of LP-based Remez: solve LP to find best perturbation."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    
    # Select active lags
    threshold = max_b * 0.999
    active = np.where(b >= threshold)[0]
    if len(active) > max_constraints:
        active = np.argsort(b)[::-1][:max_constraints]
    
    # Build LP
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [a_var >= 0, cp.sum(a_var) == 1]
    
    # Linearized constraints
    for k in active:
        k_int = int(k)
        # G_k @ a_var = sum_j a[k-j] * a_var[j] for valid j
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            # a[k-j] for j in idx
            coeffs = a[k_int - idx]
            # 2 * (G_k @ a_var) - M_var <= b[k]
            constraints.append(2.0 * cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    
    # Trust region (soft)
    constraints.append(a_var <= a * (1 + r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver:
        prob.solve(solver=solver, verbose=False)
    else:
        prob.solve(verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a, max_b
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a, max_b
    a_new = a_new / s
    
    return a_new, M_var.value

def remez_lp_method(a0, time_limit, solver=None, r=0.05, verbose=True):
    """LP-based Remez method with step size control."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_candidate, _ = lp_remez_step(a, max_constraints=30, r=r, solver=solver)
        
        # Line search for step size
        best_t = 0.5
        best_trial_m = best_m
        for t_frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_trial = (1 - t_frac) * a + t_frac * a_candidate
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m:
                best_t = t_frac
                best_trial_m = m_trial
                break
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * a_candidate
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            m_new = fast_autocorr(a).max()
            if m_new < best_m:
                best_m = m_new
                best_a = a.copy()
        
        # Adaptive r
        if m_new > best_m * 1.001:
            r = max(0.001, r * 0.5)
        elif m_new < best_m * 0.999:
            r = min(0.2, r * 1.2)
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}")
    
    return best_a, best_m

                x_r = best_a * (1 + rng.normal(0, 0.005, n))
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            c = max(0.5, c * 0.6)
        
        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.8f}, stall={stall_count}, c={c:.3f}")
    
    return best_a, best_m

def lp_remez_single(a, max_cons=30, r=0.1, solver=None):
    """Single LP Remez step. Returns candidate a'.
    Linearization: b[k](a') ≈ 2*(G_k @ a') - b[k](a)
    """
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    threshold = max_b * 0.9995
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = np.argsort(b)[::-1][:max_cons]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    
    constraints.append(a_var <= a * (1.0 + r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def remez_lp_method(a0, time_limit, solver=None, r_init=0.08, verbose=False):
    """LP-based Remez with adaptive trust region and line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_cand = lp_remez_single(a, max_cons=40, r=r, solver=solver)
        
        # Line search
        accepted = False
        for t_frac in [1.0, 0.7, 0.5, 0.3, 0.1]:
            a_trial = (1 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-18:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.3, r * 1.1)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.005, r * 0.7)
            if no_improve > 20:
                # Perturb to escape
                rng = np.random.default_rng(t)
                idx = rng.choice(n, size=min(n//5, 400), replace=False)
                a[idx] *= rng.uniform(0.7, 1.3, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent with random pair moves."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(20, n // 100)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(100, n // 3)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            
            for frac in [0.003, 0.01, 0.03, 0.08, 0.15]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2 / 0.1)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = (1.0 - (2.0 * x - 1.0) ** 4)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    n_base = 2400
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = make_init(n_base, 'triangle')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Fast NMD from parent (40% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 600.0)
    print(f"\n=== Phase 1: Fast NMD n={n0} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t1, top_k=25, c0=4.0, verbose=True)
        try_update(a_out, f"NMD_parent_n{n0}")
    
    # === Phase 2: LP Remez on best so far (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: LP Remez n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = remez_lp_method(norm_sum1(best_seq), t2, solver=solver, r_init=0.1, verbose=True)
        try_update(a_out, f"Remez_n{len(best_seq)}")
    
    # === Phase 3: NMD from different initializations (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3_total = min(remaining * 0.20, 300.0)
    if t3_total > 60:
        print(f"\n=== Phase 3: Multi-init NMD (budget={t3_total:.0f}s) ===")
        inits = ['raised_cos', 'gaussian', 'quartic']
        t_per_init = t3_total / len(inits)
        for kind in inits:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            a_init = make_init(n0, kind)
            a_out, _ = nesterov_mirror_descent(a_init, min(t_per_init, remaining - 5), 
                                               top_k=20, c0=3.0, verbose=False)
            try_update(a_out, f"NMD_{kind}_n{n0}")
    
    # === Phase 4: NMD polish on best (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: NMD polish n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t4, top_k=10, c0=1.5, verbose=True)
        try_update(a_out, f"NMD_polish_n{len(best_seq)}")
    
    # === Phase 5: LP Remez polish (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.05, 80.0)
    print(f"\n=== Phase 5: LP Remez polish (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = remez_lp_method(norm_sum1(best_seq), t5, solver=solver, r_init=0.03, verbose=False)
        try_update(a_out, f"Remez_polish_{len(best_seq)}")
    
    # === Phase 6: CD polish (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.05, 80.0)
    print(f"\n=== Phase 6: CD polish (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t6, verbose=False, rng_seed=999)
        try_update(a_out, f"CD_{len(best_seq)}")
    
    # === Phase 7: Try alternate length (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_alt = n0 + 200
        print(f"\n=== Phase 7: Alt length n={n_alt} (budget={remaining:.0f}s) ===")
        a_alt = interpolate_sequence(best_seq, n_alt)
        
        t7a = min(remaining * 0.5, 200.0)
        if t7a > 30:
            a_out, _ = nesterov_mirror_descent(a_alt, t7a, top_k=20, c0=3.0, verbose=True)
            try_update(a_out, f"NMD_n{n_alt}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t7b = min(remaining * 0.5, 100.0)
            if t7b > 15:
                a_out, _ = remez_lp_method(norm_sum1(best_seq), t7b, solver=solver, r_init=0.05, verbose=False)
                try_update(a_out, f"Remez_final_{len(best_seq)}")
    
    # === Phase 8: Final NMD + Remez on global best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final push n={n_best} (budget={remaining:.0f}s) ===")
        t8a = min(remaining * 0.5, 100.0)
        if t8a > 10:
            a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t8a, top_k=8, c0=1.0, verbose=False)
            try_update(a_out, f"NMD_final_{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out, _ = remez_lp_method(norm_sum1(best_seq), remaining - 5, solver=solver, r_init=0.02, verbose=False)
            try_update(a_out, f"Remez_last_{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
