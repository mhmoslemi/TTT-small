def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)

    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_smooth_gradient(a, beta):
    n = len(a)

    exp_bc = np.exp(bc - bc_max)
    p = exp_bc / exp_bc.sum()
    p_len = len(p)
    a_flipped = a[::-1]

    Af = np.fft.rfft(a_flipped, fft_size)
    Pf = np.fft.rfft(p, fft_size)
    conv_result = np.fft.irfft(Af * Pf, fft_size)
    g = 2.0 * conv_result[n - 1: n - 1 + n]
    return g, c, p

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
