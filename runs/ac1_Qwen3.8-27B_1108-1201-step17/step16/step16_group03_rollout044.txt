So `(a*a)[0] = sum_j a[j] * a[-j]`. For this to be non-zero, we need `-j >= 0`, i.e., `j <= 0`. So only `j=0` contributes: `(a*a)[0] = a[0]^2`.

Wait, that's not right either. Let me be more careful.

For linear convolution of two length-n sequences:
`(a*a)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]`

For `k=0`: `(a*a)[0] = sum_{j: 0<=j<n, 0<=-j<n} a[j]*a[-j] = a[0]^2` (only j=0 works).

For `k=n-1`: `(a*a)[n-1] = sum_{j: 0<=j<n, 0<=n-1-j<n} a[j]*a[n-1-j] = sum_{j=0}^{n-1} a[j]*a[n-1-j]`.

Ah, so the peak is not necessarily at k=0. The maximum of the convolution is over all k from 0 to 2n-2.

For the uniform sequence `a[j] = 1/n`:
`(a*a)[k] = (number of valid j) / n^2`
For `k < n`: number of valid j = k+1 (j=0 to k)
For `k >= n`: number of valid j = 2n-1-k (j=k-n+1 to n-1)

So the max is at k=n-1: `(a*a)[n-1] = n/n^2 = 1/n`.

And `J = 2n * (1/n) / (1)^2 = 2`.

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_s_vector(a, top_idx):
    """Compute s[j] = (1/K) * sum_{k in top_idx} a[k-j] for valid indices."""
    n = len(a)
    K = len(top_idx)
    s = np.zeros(n)
    for k in top_idx:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        indices = k - np.arange(j_start, j_end)
        s[j_start:j_end] += a[indices]
    s /= K
    return s

def mul_weights_peak_reduction(a0, time_limit, verbose=True, seed=42):
    """
    Multiplicative Weights algorithm based on KKT conditions for peak reduction.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    K_schedule = [1, 2, 3, 5, 8, 13, 21, 34, 50, 80, 120]
    eta_base = 0.15
    total_iters = 0
    global_stall = 0
    max_global_stall = 2000
    
    for phase_i, K_base in enumerate(K_schedule):
        K = min(K_base, 2 * n - 1)
        phase_start = time.monotonic()
        remaining_total = time_limit - (time.monotonic() - t0)
        if remaining_total <= 10:
            break
        
        # Time per phase: decreasing
        phase_time = min(150.0, remaining_total * 0.3)
        if phase_i == 0:
            phase_time = min(phase_time, 200.0)
        
        stall = 0
        max_stall = 400
        
        while time.monotonic() - phase_start < phase_time:
            remaining_total = time_limit - (time.monotonic() - t0)
            if remaining_total <= 5:
                break
            
            c = fast_conv(a)
            M_curr = c.max()
            
            top_idx = np.argsort(c)[::-1][:K]
            top_idx = sorted(top_idx.tolist())
            
            s_vec = compute_s_vector(a, top_idx)
            mu = np.dot(a, s_vec)
            if mu < 1e-30:
                mu = 1e-10
            
            # Adaptive eta: decrease over time within phase
            phase_iters = int(time.monotonic() - phase_start) * 10  # rough estimate
            eta = eta_base / (1.0 + 0.005 * total_iters)
            
            # Multiplicative update
            log_ratio = -eta * (s_vec - mu)
            log_ratio = np.clip(log_ratio, -3.0, 3.0)
            a_new = a * np.exp(log_ratio)
            a_new = np.maximum(a_new, 1e-15)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            
            M_new = fast_conv(a_new).max()
            
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                global_stall = 0
            elif M_new <= M_curr * (1.0 + 1e-8):
                a = a_new
                stall += 1
                global_stall += 1
            else:
                # Try half step
                log_ratio_half = -0.5 * eta * (s_vec - mu)
                log_ratio_half = np.clip(log_ratio_half, -2.0, 2.0)
                a_half = a * np.exp(log_ratio_half)
                a_half = np.maximum(a_half, 1e-15)
                a_half = a_half / a_half.sum()
                M_half = fast_conv(a_half).max()
                if M_half < M_curr:
                    a = a_half
                    stall += 1
                else:
                    stall += 1
                    global_stall += 1
            
            total_iters += 1
            
            if stall > max_stall:
                break
            
            if verbose and total_iters % 500 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      MW K={K} it={total_iters} ({elapsed:.0f}s): J={J:.6f}")
        
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MW phase K={K} done ({elapsed:.0f}s): J={J:.6f}")
        
        # If stuck globally, try a random perturbation
        if global_stall > max_global_stall:
            a_perturbed = best_a + rng.normal(0, 0.01, n) * best_a
            a_perturbed = np.maximum(a_perturbed, 1e-10)
            a_perturbed = a_perturbed / a_perturbed.sum()
            a = a_perturbed
            global_stall = 0
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    MW total ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with line search."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    iter_count = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Build constraint: (y * b)[k] <= M_curr for all k
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        solve_start = time.monotonic()
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception as e:
            if verbose:
                print(f"    LP solve error: {e}")
            continue
        solve_time = time.monotonic() - solve_start
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum
        
        # Line search
        best_alpha = 2.0 / (t + 2.0)
        best_eval = fast_conv((1 - best_alpha) * y + best_alpha * direction).max()
        
        for alpha in [0.5, 0.4, 0.3, 0.2, 0.1, 0.5 * best_alpha, 1.5 * best_alpha]:
            alpha = min(max(alpha, 0.01), 0.99)
            trial = np.maximum((1.0 - alpha) * y + alpha * direction, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            trial_m = fast_conv(trial).max()
            if trial_m < best_eval:
                best_eval = trial_m
                best_alpha = alpha
        
        y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * direction, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        y = y_new
        iter_count += 1
        
        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s, solve={solve_time:.2f}s): J={J:.6f}")
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=False, seed=123):
    """Coordinate descent with targeted mass transfers."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M_curr = c.max()
        top_peaks = np.argsort(c)[::-1][:min(20, len(c))]
        improved = False
        
        for k in top_peaks[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            idx = np.arange(j_start, j_end)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find the pair with max contribution
            max_i_local = np.argmax(contrib)
            src = idx[max_i_local]
            
            # Find destinations that minimize impact on peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:8]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            n_cand = min(len(candidates), 300)
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]
            
            best_dst = -1
            best_score = float('inf')
            for dst in chosen:
                if dst == src:
                    continue
                # Score: how much would moving to dst increase peaks?
                peak_impact = 0.0
                for k2 in top_peaks[:5]:
                    k2_int = int(k2)
                    j2 = k2_int - dst
                    if 0 <= j2 < n:
                        peak_impact += a[j2]
                if peak_impact < best_score:
                    best_score = peak_impact
                    best_dst = dst
            
            if best_dst < 0:
                continue
            
            dst = best_dst
            found = False
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if not found and stall > 200:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()
        
        if not improved:
            stall += 1
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord polish ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
