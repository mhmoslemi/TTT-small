Wait, actually, I realize that the autocorrelation of a sequence is related to the squared magnitude of its Fourier transform ( Wiener-Khinchin theorem, for the circular autocorrelation). For the linear autocorrelation, it's slightly different but related.

If we define A(ω) = sum_i a_i * e^{-iωi} (DTFT of the sequence), then the circular autocorrelation is the inverse FFT of |A(ω)|^2. The linear autocorrelation is the circular autocorrelation of the zero-padded sequence.

```python
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) / n

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
```

```python
def gradient_lag(a, k):
    n = len(a)
    grad = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start > j_end:

    idx = np.arange(j_start, j_end + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad
```

```python
def soft_gradient(a, beta, top_k):
    n = len(a)
    ac = fast_autocorr(a)
    top_idx = np.argsort(ac)[::-1][:top_k]
    top_vals = ac[top_idx]
    shift = top_vals.max()
    w = np.exp(beta * (top_vals - shift))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(top_idx)) / len(top_idx)
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        idx = np.arange(j_start, j_end + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad
```
