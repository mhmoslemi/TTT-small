def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def objective(a):
    """Compute J = 2*n*max(conv(a,a)) for normalized a."""
    n = len(a)
    ac = fast_conv(a)
    return 2.0 * n * ac.max()

def peak_subgradient(a, k):
    """Subgradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    for i in range(i_start, i_end):
        j = k - i
        if 0 <= j < n:
            grad[i] += 2.0 * a[j]
    return grad

def soft_subgradient(a, ac, temp=0.1):
    """Soft subgradient using softmax weighting over peaks."""
    n = len(a)
    # Weight peaks by softmax of (ac / temp)
    ac_clipped = np.maximum(ac, 1e-30)
    logits = ac_clipped / max(temp, 1e-10)
    logits -= logits.max()
    weights = np.exp(logits)
    wsum = weights.sum()
    if wsum < 1e-30:
        # Fall back to top-1
        k = int(np.argmax(ac))
        return peak_subgradient(a, k)
    weights = weights / wsum
    
    grad = np.zeros(n)
    # Only process significant peaks for efficiency
    threshold = weights.max() * 0.01
    for k in range(len(ac)):
        if weights[k] > threshold:
            grad += weights[k] * peak_subgradient(a, k)
    return grad

def mirror_descent_step(y, grad, eta):
    """Exponentiated gradient step (mirror descent for simplex)."""
    log_a = np.log(np.maximum(y, 1e-30)) - eta * grad
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s

    temp = temp0

            break

def coordinate_descent(a0, time_limit, seed=42, verbose=True):
    """
    Coordinate descent with smart mass redistribution.
    Focus on reducing the peak by moving mass from high-contributing
    positions to low-contributing ones.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        
        # Find the peak position
        k_peak = int(np.argmax(ac))
        
        # Get the pairs contributing to the peak
        i_start = max(0, k_peak - n + 1)
        i_end = min(n, k_peak + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[k_peak - idx]
        
        # Sort by contribution (highest first)
        order = np.argsort(contrib)[::-1]
        
        improved = False
        
        for oi in order[:5]:  # Try top 5 contributing pairs
            if time.monotonic() - t0 > time_limit:
                break
            
            src_i = int(idx[oi])
            src_j = int(k_peak - src_i)
            
            if src_i == src_j:
                continue
            
            # Find a good destination: a position that doesn't contribute
            # much to the top peaks
            top_peaks = np.argsort(ac)[::-1][:10]
            
            # Score each potential destination
            best_dst = -1
            best_score = float('inf')
            
            # Sample destinations
            candidates = rng.choice(n, size=min(500, n), replace=False)
            
            for dst in candidates:
                if dst == src_i or dst == src_j:
                    continue
                
                # Estimate the impact: how much would moving to dst
                # increase the top peaks?
                impact = 0.0
                for k2 in top_peaks[:5]:
                    k2_int = int(k2)
                    # Does dst contribute to peak k2?
                    if 0 <= k2_int - dst < n and 0 <= dst - (k2_int - dst) < n:
                        # Approximate impact
                        j2 = k2_int - dst
                        if 0 <= j2 < n:
                            impact += a[dst] + a[j2]
                
                if impact < best_score:
                    best_score = impact
                    best_dst = dst
            
            if best_dst < 0:
                continue
            dst = best_dst
            
            # Try different amounts
            found = False
            for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src_i] * frac if src_i != src_j else a[src_i] * frac * 0.5
                if amt < 1e-20:
                    continue
                
                a_trial = a.copy()
                a_trial[src_i] -= amt
                if src_i != src_j:
                    a_trial[src_j] -= amt if src_j != src_i else 0
                a_trial[dst] += 2.0 * amt if src_i != src_j else amt
                
                # Actually, let's just move mass from src_i to dst
                a_trial = a.copy()
                a_trial[src_i] -= amt
                a_trial[dst] += amt
                
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M - 1e-18:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if found:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 100 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_M

def spectral_shaping(a0, n_iters=100, beta=0.7, gamma=0.5, verbose=False):
    """
    Spectral shaping: iteratively modify the amplitude spectrum to flatten
    the time-domain convolution.
    
    Idea: In frequency domain, conv(a,a) corresponds to |A(f)|^2 (for zero-padded).
    We want to make the ifft of |A|^2 as flat as possible.
    
    Approach: 
    1. Compute A = FFT(zeropad(a))
    2. Modify |A| to reduce peaks in ifft(|A|^2)
    3. Reconstruct a from modified A (keeping phase)
    4. Clip to non-negative, renormalize
    """
    n = len(a0)
    a = normalize(a0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

import numpy as np

def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def peak_subgradient(a, k):
    """Exact subgradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    for i in range(i_start, i_end):
        j = k - i
        if 0 <= j < n:
            grad[i] += 2.0 * a[j]
    return grad


def soft_subgradient(a, ac, temp=0.05, top_k_cutoff=0.01):
    """Soft subgradient using softmax weighting over significant peaks."""
    n = len(a)
    ac_safe = np.maximum(ac, 1e-30)
    logits = ac_safe / max(temp, 1e-10)
    logits -= logits.max()
    weights = np.exp(logits)
    wsum = weights.sum()
    if wsum < 1e-30:
        k = int(np.argmax(ac))
        return peak_subgradient(a, k)
    weights = weights / wsum

    grad = np.zeros(n)
    threshold = weights.max() * top_k_cutoff
    for k in range(len(ac)):
        if weights[k] > threshold:
            grad += weights[k] * peak_subgradient(a, k)
    return grad


def topk_subgradient(a, ac, k_count=3, beta=2.0):
    """Weighted average of top-k peak subgradients."""
    n = len(a)
    k_max = min(k_count, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / ws
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        grad += wi * peak_subgradient(a, k)
    return grad


def mirror_step(y, grad, eta):
    """Exponentiated gradient (mirror) step on the simplex."""
    log_a = np.log(np.maximum(y, 1e-30)) - eta * grad
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s


def amd(a0, time_limit, mode="soft", temp0=0.05, temp_decay=0.999,

    best_M = fast_conv(a).max()

    x_prev = a.copy()
    temp = temp0

        if time.monotonic() - t0 > time_limit:

        beta_t = (t - 1.0) / (t + 1.5)

        sy = y.sum()
        y = y / sy

        ac = fast_conv(y)
        M_curr = ac.max()

def spectral_shaping(a0, n_iters=80, beta=0.75, gamma=0.4, verbose=False):
    """
    Spectral shaping: iteratively modify amplitude spectrum to flatten
    the time-domain autocorrelation.
    """
    n = len(a0)
    a = normalize(a0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(n_iters):
        a_padded = np.zeros(fft_size)
        a_padded[:n] = a
        A = np.fft.rfft(a_padded)

        # Modify power spectrum: compress → flatten time domain
        power = np.abs(A) ** 2
        power_mod = power ** beta

        # Reconstruct
        phase = A / np.maximum(np.abs(A), 1e-30)
        A_new = np.sqrt(power_mod) * phase
        a_recon = np.fft.irfft(A_new, fft_size)[:n]

        # Blend
        a_new = gamma * a + (1.0 - gamma) * a_recon
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s

        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new

        if verbose and it % 20 == 0:
            print(f"    SpecShape it={it}: J={2.0 * n * best_M:.6f}")

    return best_a, best_M


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def run_amd_and_update(a_init, time_limit, n, best_score_ref, label,
                       mode="soft", temp0=0.05, c0=3.0, seed=42, verbose=False):
    """Run AMD and return (best_a, best_score) if improved."""
    a_out, M_out = amd(a_init, time_limit, mode=mode, temp0=temp0,
                       c0=c0, seed=seed, verbose=verbose)
    score = 2.0 * n * M_out
    a_norm = normalize(a_out)
    actual_score = evaluate_safe(a_norm)
    return a_norm, actual_score


def propose_candidate(budget_s=1800.0):
    """
    Main orchestrator:
    Phase 1: Spectral shaping for initialization
    Phase 2: Primary AMD (soft subgradient)
    Phase 3: AMD with top-k subgradient (different dynamics)
    Phase 4: Coordinate descent polish
    Phase 5: Multi-length exploration
    Phase 6: Final AMD polish + coordinate descent
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 50.0, 1750.0)

    best_a = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_a = normalize(p)
            best_score = evaluate_safe(best_a)
            print(f"Parent: n={len(best_a)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_a is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_a = np.sin(np.pi * x) ** 2
        best_a = best_a / best_a.sum()
        best_score = evaluate_safe(best_a)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_a)
    print(f"Starting: n={n0}, score={best_score:.6f}, budget={time_budget:.0f}s")

    # === Phase 1: Spectral shaping ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n--- Phase 1: Spectral shaping ---")
    t1 = min(remaining * 0.04, 50.0)
    if t1 > 5:
        a_s, M_s = spectral_shaping(best_a, n_iters=60, beta=0.8, gamma=0.4, verbose=False)
        sc_s = evaluate_safe(a_s)
        if sc_s < best_score:
            best_score = sc_s
            best_a = a_s.copy()
            print(f"  SpecShape improved: {best_score:.6f}")
        else:
            print(f"  SpecShape no improvement ({2.0*n0*M_s:.6f} vs {best_score:.6f})")

    # === Phase 2: Primary AMD (soft) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_a)
    print(f"\n--- Phase 2: AMD soft n={n_curr} ---")
    t2 = min(remaining * 0.35, 550.0)
    if t2 > 30:
        a_out, M_out = amd(best_a, t2, mode="soft", temp0=0.03,
                           temp_decay=0.999, c0=3.0, seed=42, verbose=True)
        a_norm = normalize(a_out)
        sc = evaluate_safe(a_norm)
        if sc < best_score:
            best_score = sc
            best_a = a_norm.copy()
            print(f"  NEW BEST: {best_score:.6f} (n={n_curr})")

    # === Phase 3: AMD top-k ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_a)
    print(f"\n--- Phase 3: AMD topk n={n_curr} ---")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_out, M_out = amd(best_a, t3, mode="topk", temp0=0.1,
                           temp_decay=0.998, c0=5.0, seed=7, verbose=True,
                           top_k=4, beta_w=1.5)
        a_norm = normalize(a_out)
        sc = evaluate_safe(a_norm)
        if sc < best_score:
            best_score = sc
            best_a = a_norm.copy()
            print(f"  NEW BEST: {best_score:.6f} (n={n_curr})")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_a)
    print(f"\n--- Phase 4: Coord descent n={n_curr} ---")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_out, M_out = coord_polish(best_a, t4, seed=42, verbose=True)
        a_norm = normalize(a_out)
        sc = evaluate_safe(a_norm)
        if sc < best_score:
            best_score = sc
            best_a = a_norm.copy()
            print(f"  NEW BEST: {best_score:.6f} (n={n_curr})")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_a)
    print(f"\n--- Phase 5: Multi-length (base n={n_best}) ---")

    length_deltas = []
    for d in [-400, -300, -200, 200, 300, 400, -600, 600]:
        n_try = n_best + d
        if 1000 <= n_try <= 5000:
            length_deltas.append(n_try)

    for n_try in length_deltas:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break

        print(f"  Exploring n={n_try}")
        a_interp = interpolate_seq(best_a, n_try)

        # Quick AMD
        t5 = min(remaining * 0.25, 120.0)
        if t5 > 15:
            a_out, M_out = amd(a_interp, t5, mode="soft", temp0=0.05,
                               temp_decay=0.999, c0=3.0,
                               seed=500 + n_try % 500, verbose=False)
            a_norm = normalize(a_out)
            sc = evaluate_safe(a_norm)
            if sc < best_score:
                best_score = sc
                best_a = a_norm.copy()
                print(f"    NEW BEST: {best_score:.6f} (n={n_try})")

            # Quick coord
            remaining = time_budget - (time.monotonic() - t_global)
            t5b = min(remaining * 0.1, 50.0)
            if t5b > 10:
                a_out2, _ = coord_polish(a_interp, t5b, seed=n_try % 1000, verbose=False)
                a_norm2 = normalize(a_out2)
                sc2 = evaluate_safe(a_norm2)
                if sc2 < best_score:
                    best_score = sc2
                    best_a = a_norm2.copy()
                    print(f"    NEW BEST (coord): {best_score:.6f} (n={n_try})")

    # === Phase 6: Final AMD polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_a)
    print(f"\n--- Phase 6: Final polish n={n_best} ---")
    if remaining > 20:
        t6 = min(remaining * 0.45, 250.0)
        if t6 > 10:
            a_out, M_out = amd(best_a, t6, mode="soft", temp0=0.01,
                               temp_decay=0.9995, c0=2.0, seed=999, verbose=True)
            a_norm = normalize(a_out)
            sc = evaluate_safe(a_norm)
            if sc < best_score:
                best_score = sc
                best_a = a_norm.copy()
                print(f"  NEW BEST: {best_score:.6f} (n={n_best})")

    # === Phase 7: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_a)
    if remaining > 10:
        t7 = min(remaining * 0.6, 120.0)
        if t7 > 5:
            a_out, M_out = coord_polish(best_a, t7, seed=314, verbose=True)
            a_norm = normalize(a_out)
            sc = evaluate_safe(a_norm)
            if sc < best_score:
                best_score = sc
                best_a = a_norm.copy()
                print(f"  NEW BEST: {best_score:.6f} (n={n_best})")

    # Final validation
    if best_a is not None:
        fs = evaluate_safe(best_a)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_a)}, "
              f"time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_a]
