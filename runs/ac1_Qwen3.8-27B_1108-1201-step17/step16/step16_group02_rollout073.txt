```
def proj_simplex(v, s=1.0):
    v=np.asarray(v,dtype=np.float64)
    if v.size==0: return v
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)-1.0
    ind=np.arange(1,v.size+1)
    cond=u - cssv/ind > 0
    if not np.any(cond):
        return np.ones_like(v)/v.size
    # find k
    k=int(np.where(cond)[0][-1])+1
    theta=cssv[k-1]/k
    w=np.maximum(v-theta,0)
    ss=w.sum()
    if ss<1e-15: w=np.ones_like(v)/v.size
    else: w/=s
    return w
```

```
def proj_simplex(v):
   v=np.asarray(v,dtype=np.float64)
   n=v.size
   if n==0: return np.array([1.0])
   u=np.sort(v)[::-1]
   cssv=np.cumsum(u)-1.0
   ind=np.arange(1,n+1,dtype=np.float64)
   cond = u - cssv/ind > 0
   if not cond.any():
       return np.ones(n, dtype=np.float64)/n
   k = int(np.where(cond)[0][-1]) + 1
   theta = cssv[k-1] / k
   w = np.maximum(v - theta, 0.0)
   s = w.sum()
   if s < 1e-15:
       return np.ones(n)/n
   return w / s
```

```python
"""
Search for a non-negative step-function height sequence minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2.

The routine uses several stages:
  * diverse parametric initializations plus the provided parent sequence,
  * peak-weighted projected / mirror descent with FFT-based subgradients,
  * microscopic pair-mass-transfer local search,
  * final verification of saved candidates with the official evaluator.
"""

import numpy as np
import time


def next_power_of_two(m):
    m = int(max(1, m))
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    size = next_power_of_two(2 * n - 1)
    A = np.fft.rfft(a, size)
    out = np.fft.irfft(A * A, size)
    return out[:2 * n - 1]


def make_positive_normalized(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.array([1.0], dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.minimum(arr, 1000.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def local_score(a):
    a = make_positive_normalized(a)
    if a.size == 0:
        return float("inf")
    m = float(fast_autocorr(a).max())
    return 2.0 * float(a.size) * m


def evaluate_wrapper(a):
    a = make_positive_normalized(a)
    if a.size == 0:
        return float("inf")
    if float(a.sum()) < 0.01:
        return float("inf")
    try:
        seq = [float(x) for x in a]
        sc = evaluate_sequence(seq)
    except Exception:
        return float("inf")
    if sc is None or not np.isfinite(sc):
        return float("inf")
    return float(sc)


def proj_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.array([1.0], dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0.0
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    k = int(np.where(cond)[0][-1]) + 1
    theta = cssv[k - 1] / float(k)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return w / s


def random_perturb(x, rng, scale):
    x = make_positive_normalized(x)
    if x.size == 0:
        return np.array([1.0], dtype=np.float64)
    noise = rng.normal(0.0, float(scale), size=x.size)
    return make_positive_normalized(x * (1.0 + noise))


def make_peak_weights(c, top_k, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    if L == 0:
        return np.array([1.0], dtype=np.float64)
    k = int(max(1, min(int(top_k), L)))
    if k >= L:
        idx = np.arange(L)
        vals = c.copy()
    else:
        idx = np.argpartition(c, L - k)[-k:]
        vals = c[idx]
    mx = float(vals.max())
    if beta <= 0.0 or mx <= 0.0:
        w = np.ones(k, dtype=np.float64) / float(k)
    else:
        rel = np.maximum(vals / mx, 0.0)
        w = np.power(rel, float(beta))
        s = float(w.sum())
        if s < 1e-300:
            w = np.ones(k, dtype=np.float64) / float(k)
        else:
            w = w / s
    weights = np.zeros(L, dtype=np.float64)
    weights[idx] = w
    return weights


def risk_vector(a, weights):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    L = 2 * n - 1
    weights = np.asarray(weights, dtype=np.float64).ravel()
    if weights.size != L:
        w = np.zeros(L, dtype=np.float64)
        m = min(weights.size, L)
        if m > 0:
            w[:m] = weights[:m]
        weights = w
    y_rev = a[::-1].copy()
    size = next_power_of_two(L + n - 1)
    X = np.fft.rfft(weights, size)
    Y = np.fft.rfft(y_rev, size)
    z = np.fft.irfft(X * Y, size)
    d = z[n - 1:2 * n - 1]
    return np.maximum(d, 0.0)


def top_max_indices(arr, k):
    arr = np.asarray(arr, dtype=np.float64).ravel()
    n = arr.size
    k = int(max(1, min(int(k), n)))
    if k == 0:
        return np.arange(0)
    if k >= n:
        return np.arange(n)
    return np.argpartition(arr, n - k)[-k:]


def top_min_indices(arr, k):
    arr = np.asarray(arr, dtype=np.float64).ravel()
    n = arr.size
    k = int(max(1, min(int(k), n)))
    if k == 0:
        return np.arange(0)
    if k >= n:
        return np.arange(n)
    return np.argpartition(arr, k - 1)[:k]


def make_window(n, name, p1=1.0, p2=1.0, p3=0.0):

    if n == 1:
        return np.array([1.0], dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / float(n - 1)

    if name == "uniform":
        a = np.ones(n, dtype=np.float64)
    elif name == "tri":
        a = np.maximum(1.0 - np.abs(2.0 * x - 1.0), 0.0)
    elif name == "asym_tri":
        c = float(np.clip(p1, 0.05, 0.95))
        rise = x / c
        fall = (1.0 - x) / (1.0 - c)
        a = np.minimum(rise, fall)
    elif name == "hann":
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    elif name == "hann_pow":
        base = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
        a = np.power(np.maximum(base, 0.0), float(max(0.05, p1)))
    elif name == "blackman":
        a = 0.42 - 0.5 * np.cos(2.0 * np.pi * x) + 0.08 * np.cos(4.0 * np.pi * x)
    elif name == "super":
        pp = float(max(0.05, p1))
        base = np.maximum(1.0 - np.power(np.abs(2.0 * x - 1.0), pp), 0.0)
        a = np.power(base, 1.0 / pp)
    elif name == "gauss":
        sigma = float(np.clip(p1, 0.02, 0.5))
        cen = float(np.clip(p2, 0.02, 0.98))
        a = np.exp(-((x - cen) * (x - cen)) / (2.0 * sigma * sigma))
    elif name == "asym_gauss":
        sigma = float(np.clip(p1, 0.02, 0.5))
        cen = float(np.clip(p2, 0.05, 0.95))
        skew = float(np.clip(p3, -5.0, 5.0))
        z = (x - cen) / sigma
        a = np.exp(-(z * z) - skew * z)
    else:
        a = np.ones(n, dtype=np.float64)

    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def initial_candidates(n):
    res = []
    res.append(make_window(n, "uniform"))
    res.append(make_window(n, "tri"))
    res.append(make_window(n, "hann"))
    res.append(make_window(n, "blackman"))
    res.append(make_window(n, "hann_pow", 2.0))
    res.append(make_window(n, "hann_pow", 4.0))
    for sp in (1.0, 1.5, 2.0, 3.0, 4.0, 6.0):
        res.append(make_window(n, "super", sp))
    for at in (0.30, 0.40, 0.50, 0.60, 0.70):
        res.append(make_window(n, "asym_tri", at))
    for sg in (0.10, 0.15, 0.20, 0.25, 0.35):
        for cen in (0.35, 0.45, 0.50, 0.55, 0.65):
            res.append(make_window(n, "gauss", sg, cen))
            res.append(make_window(n, "asym_gauss", sg, cen, 1.0))
            res.append(make_window(n, "asym_gauss", sg, cen, -1.0))
    return res

def append_saved(saved, a, score):
    if saved is None or a is None:
        return
    a = make_positive_normalized(a)
    saved.append((a, float(score)))
    if len(saved) > 12:
        saved.pop(0)


def peak_descent(a0, time_budget,
                 top_k_start=100, top_k_end=20,
                 beta_start=0.0, beta_end=30.0,
                 eta0=0.06, max_eta=0.35,
                 use_momentum=True, use_mirror=False,
                 reset_stall=800, verbose=False):
    if time_budget <= 0.0:
        a = make_positive_normalized(a0)
        return a, float(fast_autocorr(a).max())

    x = make_positive_normalized(a0)
    x_prev = x.copy()
    best_x = x.copy()
    current_m = float(fast_autocorr(x).max())
    best_m = current_m

    t0 = time.monotonic()
    t = 0
    eta = float(eta0)
    stall = 0
    rng = np.random.default_rng(54321)
    factors = (1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04)

    while time.monotonic() - t0 < float(time_budget):
        t += 1
        if t > 10000000:
            break

        frac = min(1.0, t / 5000.0)
        top_k = int(round(top_k_start + (top_k_end - top_k_start) * frac))
        if top_k < 1:
            top_k = 1
        beta = beta_start + (beta_end - beta_start) * frac

        if use_momentum and t > 3 and stall < 100:
            mom = 0.25 + 0.35 * min(1.0, t / 5000.0)
            y = x + mom * (x - x_prev)
            y = proj_simplex(y)
        else:
            y = x.copy()

        c_y = fast_autocorr(y)
        weights = make_peak_weights(c_y, top_k, beta)
        d = risk_vector(y, weights)
        dsum = float(d.sum())

        if dsum < 1e-15:
            x = best_x.copy()
            x_prev = x.copy()
            current_m = best_m
            stall = 0
            eta = eta0
            continue

        g = d / dsum
        trial_m = current_m
        trial_x = None

        for fac in factors:
            step = eta * fac
            if use_mirror:
                log_y = np.log(np.maximum(y, 1e-300))
                z = log_y - step * g
                zmax = float(z.max())
                w = np.exp(z - zmax)
                s = float(w.sum())
                if s < 1e-15:
                    cand = y.copy()
                else:
                    cand = w / s
            else:
                cand = proj_simplex(y - step * g)

            m = float(fast_autocorr(cand).max())
            if m < trial_m - 1e-15:
                trial_m = m
                trial_x = cand

        if trial_x is not None:
            x_prev = x
            x = trial_x
            current_m = trial_m
            if current_m < best_m - 1e-15:
                best_m = current_m
                best_x = x.copy()
                stall = 0
            else:
                stall += 1
            eta = min(max_eta, eta * 1.03)
        else:
            eta = max(eta0 * 0.3, eta * 0.7)
            stall += 1

            if stall > 20 and (t % 10 == 0):
                cand = random_perturb(x, rng, max(eta * 0.5, 0.005))
                m = float(fast_autocorr(cand).max())
                if m < current_m - 1e-15:
                    x_prev = x
                    x = cand
                    current_m = m
                    best_m = m
                    best_x = x.copy()
                    stall = 0
                    eta = eta0

            if stall > reset_stall:
                cand = random_perturb(best_x, rng, 0.01)
                m = float(fast_autocorr(cand).max())
                if m < best_m - 1e-15:
                    x = cand
                    x_prev = x.copy()
                    current_m = m
                    best_m = m
                    best_x = x.copy()
                else:
                    x = best_x.copy()
                    x_prev = x.copy()
                    current_m = best_m
                stall = 50
                eta = eta0

        if verbose and (t % 1000 == 0):
            print("  peak_descent t={} local={:.8f} eta={:.5f} stall={}".format(
                t, 2.0 * x.size * current_m, eta, stall))

    return best_x, best_m


def pair_move_polish(a0, time_budget, top_k=5, beta=2.0,
                     max_stall=40, verbose=False):
    if time_budget <= 0.0:
        a = make_positive_normalized(a0)
        return a, float(fast_autocorr(a).max())

    x = make_positive_normalized(a0)
    best_x = x.copy()
    current_m = float(fast_autocorr(x).max())
    best_m = current_m
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12998)
    fractions = (1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 8e-2)

    while time.monotonic() - t0 < float(time_budget):
        c = fast_autocorr(x)
        weights = make_peak_weights(c, max(1, int(top_k)), float(beta))
        d = risk_vector(x, weights)
        if d.size != x.size:
            d = np.zeros_like(x)

        src_scores = x * np.maximum(d, 0.0)
        s_idx = top_max_indices(src_scores, 8)
        t_idx = top_min_indices(d + 0.05 * x, 8)
        rng.shuffle(s_idx)
        rng.shuffle(t_idx)

        improved = False
        for s_pos in s_idx:
            if improved:
                break
            if x[s_pos] < 1e-12:
                continue
            for t_pos in t_idx:
                if improved:
                    break
                if T_pos == s_pos:
                    continue
                if d[s_pos] <= d[t_pos] + 1e-18:
                    continue

                best_trial_m = current_m
                best_trial_x = None
                for frac in fractions:
                    amt = x[s_pos] * frac
                    if amt <= 1e-18:
                        continue
                    trial = x.copy()
                    trial[s_pos] -= amt
                    trial[t_pos] += amt
                    if trial[s_pos] < -1e-15:
                        continue
                    trial = np.maximum(trial, 0.0)
                    ss = float(trial.sum())
                    if ss < 1e-15:
                        continue
                    trial = trial / ss
                    m = float(fast_autocorr(trial).max())
                    if m < best_trial_m - 1e-15:
                        best_trial_m = m
                        best_trial_x = trial

                if best_trial_x is not None:
                    x = best_trial_x
                    current_m = best_trial_m
                    if current_m < best_m - 1e-15:
                        best_m = current_m
                        best_x = x.copy()
                    improved = True

        if improved:
            stall = 0
        else:
            stall += 1
            for _ in range(6):
                s_pos = int(rng.integers(x.size))
                t_pos = int(rng.integers(x.size))
                if s_pos == t_pos or x[s_pos] < 1e-12:
                    continue
                best_trial_m = current_m
                best_trial_x = None
                for frac in (1e-4, 1e-3, 1e-2):
                    amt = x[s_pos] * frac
                    if amt <= 1e-18:
                        continue
                    trial = x.copy()
                    trial[s_pos] -= amt
                    trial[t_pos] += amt
                    if trial[s_pos] < -1e-15:
                        continue
                    trial = np.maximum(trial, 0.0)
                    ss = float(trial.sum())
                    if ss < 1e-15:
                        continue
                    trial = trial / ss
                    m = float(fast_autocorr(trial).max())
                    if m < best_trial_m - 1e-15:
                        best_trial_m = m
                        best_trial_x = trial
                if best_trial_x is not None:
                    x = best_trial_x
                    current_m = best_trial_m
                    if current_m < best_m - 1e-15:
                        best_m = current_m
                        best_x = x.copy()
                    improved = True
                    stall = 0
                    break

            if improved:
                pass
            elif stall > max_stall:
                cand = random_perturb(best_x, rng, 0.005)
                m = float(fast_autocorr(cand).max())
                if m < best_m - 1e-15:
                    x = cand
                    current_m = m
                    best_m = m
                    best_x = x.copy()
                else:
                    x = best_x.copy()
                    current_m = best_m
                stall = 0

        if verbose and (stall % 20 == 0):
            print("  pair_polish stall={} local={:.8f}".format(
                stall, 2.0 * x.size * current_m))

    return best_x, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    requested = float(budget_s)
    total_budget = min(requested - 60.0, 1740.0)
    if total_budget < 30.0:
        total_budget = max(10.0, requested - 5.0)

    saved = []
    best_a = None
    best_score = float("inf")

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim != 1:
            p = p.ravel()
        if p.size > 0 and np.all(np.isfinite(p)) and np.sum(np.maximum(p, 0.0)) > 1e-9:
            pn = make_positive_normalized(p)
            ls = local_score(pn)
            if np.isfinite(ls):
                best_a = pn
                best_score = ls
                append_saved(saved, best_a, best_score)
    except Exception:
        pass

    if best_a is None:
        n_base = 2400
        best_a = make_window(n_base, "hann")
        best_score = local_score(best_a)
        append_saved(saved, best_a, best_score)
    else:
        n_base = int(best_a.size)

    ns = [n_base]
    if n_base + 400 <= 4000:
        ns.append(n_base + 400)
    if n_base > 2800:
        ns.append(2400)
    elif n_base < 2000:
        ns.append(2400)
    uniq = []
    for val in ns:
        val = int(val)
        if val >= 50 and val not in uniq:
            uniq.append(val)

    for n in uniq[:3]:
        try:
            cands = initial_candidates(n)
        except Exception:
            cands = []
        for cand in cands:
            try:
                c = make_positive_normalized(cand)
                if c.size < 50:
                    continue
                ls = local_score(c)
            except Exception:
                continue
            if np.isfinite(ls) and ls < best_score:
                best_score = ls
                best_a = c.copy()
                append_saved(saved, best_a, best_score)

    # Phase 1: broad peak-weighted projected descent.
    rem = total_budget - (time.monotonic() - t_global)
    if rem > 120.0 and best_a is not None:
        t1 = min(rem * 0.42, 700.0)
        t1 = max(0.0, t1 - 10.0)
        a1, _ = peak_descent(
            best_a, t1,
            top_k_start=100, top_k_end=20,
            beta_start=0.0, beta_end=30.0,
            eta0=0.06, max_eta=0.35,
            use_momentum=True, use_mirror=False,
            reset_stall=800, verbose=True)
        if a1 is not None:
            ls = local_score(a1)
            if np.isfinite(ls) and ls < best_score:
                best_score = ls
                best_a = make_positive_normalized(a1).copy()
            append_saved(saved, a1, ls)

    # Phase 2: sharper mirror descent for robustness.
    rem = total_budget - (time.monotonic() - t_global)
    if rem > 100.0 and best_a is not None:
        t2 = min(rem * 0.24, 420.0)
        t2 = max(0.0, t2 - 10.0)
        a2, _ = peak_descent(
            best_a, t2,
            top_k_start=30, top_k_end=5,
            beta_start=5.0, beta_end=80.0,
            eta0=0.03, max_eta=0.20,
            use_momentum=False, use_mirror=True,
            reset_stall=400, verbose=True)
        if a2 is not None:
            ls = local_score(a2)
            if np.isfinite(ls) and ls < best_score:
                best_score = ls
                best_a = make_positive_normalized(a2).copy()
            append_saved(saved, a2, ls)

    # Phase 3: microscopic pair transfers.
    rem = total_budget - (time.monotonic() - t_global)
    if rem > 80.0 and best_a is not None:
        t3 = min(rem * 0.28, 480.0)
        t3 = max(0.0, t3 - 10.0)
        a3, _ = pair_move_polish(best_a, t3, top_k=5, beta=3.0,
                                 max_stall=45, verbose=True)
        if a3 is not None:
            ls = local_score(a3)
            if np.isfinite(ls) and ls < best_score:
                best_score = ls
                best_a = make_positive_normalized(a3).copy()
            append_saved(saved, a3, ls)

    # Phase 4: another wider projected descent with different hyperparameters.
    rem = total_budget - (time.monotonic() - t_global)
    if rem > 100.0 and best_a is not None:
        t4 = min(rem * 0.28, 420.0)
        t4 = max(0.0, t4 - 10.0)
        a4, _ = peak_descent(
            best_a, t4,
            top_k_start=150, top_k_end=10,
            beta_start=0.0, beta_end=50.0,
            eta0=0.10, max_eta=0.50,
            use_momentum=True, use_mirror=False,
            reset_stall=500, verbose=True)
        if a4 is not None:
            ls = local_score(a4)
            if np.isfinite(ls) and ls < best_score:
                best_score = ls
                best_a = make_positive_normalized(a4).copy()
            append_saved(saved, a4, ls)

    # Phase 5: final micro polishing.
    rem = total_budget - (time.monotonic() - t_global)
    if rem > 40.0 and best_a is not None:
        t5 = min(rem * 0.88, 360.0)
        t5 = max(0.0, t5 - 15.0)
        a5, _ = pair_move_polish(best_a, t5, top_k=3, beta=4.0,
                                 max_stall=30, verbose=True)
        if a5 is not None:
            ls = local_score(a5)
            if np.isfinite(ls) and ls < best_score:
                best_score = ls
                best_a = make_positive_normalized(a5).copy()
            append_saved(saved, a5, ls)

    if best_a is None:
        best_a = make_window(2400, "hann")
        best_score = local_score(best_a)
    append_saved(saved, best_a, best_score)

    final_seq = best_a
    final_score = float("inf")

    # Verify a small set of saved candidates with the exact evaluator.
    for seq, _ in reversed(saved):
        seq = make_positive_normalized(seq)
        sc = evaluate_wrapper(seq)
        if np.isfinite(sc) and sc < final_score:
            final_score = sc
            final_seq = seq

    if final_seq is None or not np.isfinite(final_score):
        final_seq = make_positive_normalized(best_a)
        final_score = local_score(final_seq)

    print("FINAL local-ish best score: {:.8f}, n={}".format(
        final_score, int(final_seq.size)))

    return [float(v) for v in final_seq]
```
