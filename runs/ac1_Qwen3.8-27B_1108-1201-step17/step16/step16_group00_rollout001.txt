    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Get top peaks
        n_peaks = min(15, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_peaks]

        # Compute sensitivity: for each position i, how much does it contribute to peaks?
        sensitivity = np.zeros(n)
        for k in top_lags[:8]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contribution = a[idx] * a[j_idx]
            # Add to sensitivity of both idx and j_idx
            for ii, jj, cc in zip(idx, j_idx, contribution):
                sensitivity[ii] += cc
                sensitivity[jj] += cc

        # Find high-sensitivity positions with substantial mass
        high_sens = np.where((sensitivity > sensitivity.mean() + sensitivity.std()) & (a > 0.001))[0]
        low_sens = np.where((sensitivity < sensitivity.mean() - 0.5 * sensitivity.std()) | (sensitivity == 0))[0]

        if len(high_sens) == 0 or len(low_sens) == 0:
            stall += 1
            continue

        improved = False

        for _ in range(20):  # Try up to 20 moves per outer iteration
            if time.monotonic() - t0 > time_limit:
                break

            src = high_sens[rng.integers(len(high_sens))]
            dst = low_sens[rng.integers(len(low_sens))]
            if src == dst:
                continue

            src_val = a[src]
            if src_val < 1e-8:
                continue

            # Try multiple transfer fractions
            fracs = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3]
            found = False
            for frac in fracs:
                # Also try some random fractions
                amt = src_val * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break

            if found:
                stall = 0
            else:
                # Try random move as escape
                if stall > 100:
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.001, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        J_new = compute_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0

        if not improved:
            stall += 1
            # Larger random perturbation to escape
            if stall % 50 == 0:
                n_pert = max(5, n // 100)
                perturb_positions = rng.choice(n, size=n_pert, replace=False)
                for p in perturb_positions:
                    if a[p] > 1e-8:
                        frac = rng.uniform(0.01, 0.1)
                        amt = a[p] * frac
                        a[p] -= amt
                        dst = int(rng.integers(n))
                        a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRed ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    PeakRed final ({elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J


def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=False, top_K=500):
    """Improved LP Frank-Wolfe with more constraints and better step size."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    prev_y = a.copy()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        # Build constraint matrix
        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = T_full @ y
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select top-K constraints (most binding)
        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        prev_y = y
        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW2 iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def block_optimize(a0, time_limit, verbose=False, seed=200):
    """
    Block coordinate optimization: pick a block of consecutive coordinates,
    optimize them via local quadratic approximation or small LP.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    block_sizes = [4, 8, 16, 32, 64]

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:5]

        # Pick a block near a peak-contributing region
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break

            # Choose block size
            bs = block_sizes[rng.integers(len(block_sizes))]
            # Choose block start (biased toward peak regions)
            if rng.random() < 0.6 and len(top_lags) > 0:
                k = int(top_lags[0])
                center = k // 2
                start = max(0, min(n - bs, center + rng.integers(-n//4, n//4 - bs)))
            else:
                start = int(rng.integers(max(1, n - bs)))

            end = min(n, start + bs)
            if end - start < 2:
                continue

            # Try adjusting the block
            block = a[start:end].copy()
            block_sum = block.sum()
            block_sum_rest = a.sum() - block_sum

            if block_sum < 1e-10:
                continue

            # Local optimization: try to redistribute mass within block
            # to reduce peak autocorrelation
            best_block = block.copy()
            best_local_J = compute_J(a)

            for _ in range(100):
                if time.monotonic() - t0 > time_limit:
                    break
                # Random perturbation of block
                perturb = rng.normal(0, 0.1 * block_sum, end - start)
                new_block = block + perturb
                new_block = np.maximum(new_block, 0)
                nb_sum = new_block.sum()
                if nb_sum < 1e-10:
                    continue
                # Scale to preserve total sum
                new_block = new_block * (block_sum / nb_sum)

                a_trial = a.copy()
                a_trial[start:end] = new_block
                J_trial = compute_J(a_trial)
                if J_trial < best_local_J - 1e-14:
                    best_local_J = J_trial
                    best_block = new_block.copy()

            if best_local_J < best_J - 1e-14:
                a[start:end] = best_block
                best_J = best_local_J
                best_a = a.copy()

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    BlockOpt ({elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_symmetric_init(n):
    """Create a symmetric initialization (high at edges, low in middle)."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_power_law_init(n, power=0.5):
    """Power law from edges."""
    x = np.linspace(-1, 1, n)
    a = (1 - x**2) ** power + 0.001
    return norm_sum1(a)


def make_gaussian_edges(n, sigma=0.15):
    """Gaussian bumps at edges."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.05) / sigma) ** 2) + np.exp(-((x - 0.95) / sigma) ** 2) + 0.005
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with adaptive multi-method optimization.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_symmetric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Fast Nesterov SGD from multiple inits ===
    print(f"\n=== Phase 1: Nesterov SGD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.25, 400.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_symmetric_init(n0))
        labels.append("symmetric")

        inits.append(make_power_law_init(n0, power=0.7))
        labels.append("power0.7")

        inits.append(make_gaussian_edges(n0, sigma=0.12))
        labels.append("gauss_edges")

        # Blend parent with symmetric
        if parent is not None:
            blend = 0.5 * norm_sum1(parent) + 0.5 * make_symmetric_init(n0)
            inits.append(norm_sum1(blend))
            labels.append("blend_parent_sym")

        # Remove near-duplicates
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            is_dup = False
            for u in unique_inits:
                if np.sum(np.abs(init - u)) < 0.05:
                    is_dup = True
                    break
            if not is_dup:
                unique_inits.append(init)
                unique_labels.append(lbl)

        n_inits = len(unique_inits)
        t_per = max(20.0, t_phase1 / n_inits)

        for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per, remaining - 30)
            if t_avail < 15:
                break
            print(f"  SGD init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = nesterov_subgradient(init, t_avail, verbose=False,
                                            top_k=15, beta=3.0, c=3.0,
                                            exact_frac=0.4, restart_every=2000,
                                            seed=42 + idx)
            try_update(a_out, f"sgd_{lbl}")

    # === Phase 2: Peak reduction moves ===
    print(f"\n=== Phase 2: Peak reduction (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_reduction_moves(a_in, t2, verbose=True, seed=77)
        try_update(a_out, f"peakred_n{len(a_in)}")

    # === Phase 3: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW v2 (n={n_best}) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t3, solver, verbose=True, top_K=400)
        try_update(a_out, f"lpfw2_n{n_best}")

    # === Phase 4: SGD polish with higher top_k ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: SGD polish (n={n_best}) ===")
    t4 = min(remaining * 0.15, 200.0)

                                        top_k=30, beta=2.5, c=4.0,
                                        exact_frac=0.5, restart_every=5000,
                                        seed=999)
        try_update(a_out, f"sgd_polish_n{n_best}")

    # === Phase 5: Block optimization ===
    remaining = time_budget - (time.monotonic() - t_global)

    print(f"\n=== Phase 5: Block optimization (n={n_best}) ===")

        a_out, _ = block_optimize(a_in, t5, verbose=True, seed=200)

    # === Phase 6: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")
    longer_lengths = [n_best + 200, n_best + 400, n_best + 800]
    longer_lengths = [l for l in longer_lengths if l <= 5000]

    for new_n in longer_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        t6a = min(remaining * 0.25, 150.0)
        if t6a > 20:
            a_out, _ = nesterov_subgradient(a_long, t6a, verbose=False,
                                            top_k=15, beta=3.0, c=3.0,
                                            exact_frac=0.4, restart_every=2000,
                                            seed=555)
            try_update(a_out, f"sgd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_reduction_moves(a_in, min(remaining * 0.2, 80.0), verbose=False, seed=333)
            try_update(a_out, f"peakred2_n{len(a_in)}")

    # === Phase 7: Alternative shorter length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    shorter = n_best - 400
    if remaining > 60 and shorter >= 1000:
        print(f"\n  Trying shorter n={shorter}")
        a_short = interpolate_sequence(best_seq, shorter)
        t7a = min(remaining * 0.2, 100.0)
        if t7a > 20:

                                            top_k=15, beta=3.0, c=3.0,
                                            exact_frac=0.4, restart_every=2000,
                                            seed=777)
            try_update(a_out, f"sgd_n{shorter}")

    # === Phase 8: Final LP if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 3500:
        t8 = min(remaining * 0.4, 200.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_v2(a_in, t8, solver, verbose=True, top_K=500)
            try_update(a_out, f"lpfw_final_n{n_best}")

    # === Phase 9: Final peak reduction ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final peak reduction (n={n_best}) ===")
    if remaining > 20:
        t9 = min(remaining * 0.5, 150.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_reduction_moves(a_in, t9, verbose=True, seed=1234)
            try_update(a_out, f"peakred_final_n{n_best}")

    # === Phase 10: One more SGD if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 10: Final SGD (n={n_best}) ===")
        t10 = min(remaining * 0.6, 100.0)

            a_out, _ = nesterov_subgradient(a_in, t10, verbose=False,
                                            top_k=40, beta=2.0, c=5.0,
                                            exact_frac=0.6, restart_every=8000,

            try_update(a_out, f"sgd_final_n{n_best}")

    # === Final validation ===

        if np.isfinite(fs):

        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_symmetric_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
