    # Step size
    eta_t = c / sqrt(t)  (or some other rule)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Accelerated Mirror Descent (AMD) with multiplicative weight updates
2. LP-based Frank-Wolfe for global refinement
3. Peak-targeted coordinate descent for local polish
4. Multi-length exploration with interpolation
5. Adaptive step size and periodic restarts

import numpy as np
import time

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalized_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgradient(y, top_k=50, tol=0.01):
    """Compute subgradient of max_k (y*y)_k at point y.
    Returns (gradient, M) where M is the current max autocorrelation."""
    n = len(y)
    ac = fast_autocorr(y)
    M = ac.max()
    
    # Find active set: peaks within tolerance of max
    threshold = M * (1 - tol)
    active_mask = ac >= threshold
    active_idx = np.where(active_mask)[0]
    
    # Use top_k for efficiency
    if len(active_idx) > top_k:
        # Take the largest top_k
        vals = ac[active_idx]
        top_positions = np.argsort(vals)[::-1][:top_k]
        active_idx = active_idx[top_positions]
    
    # Compute subgradient: average of gradients of active components
    # grad of (y*y)_k w.r.t. y_i = 2*y_{k-i} if 0 <= k-i < n, else 0
    grad = np.zeros(n)
    for k in active_idx:
        k_int = int(k)
        # y_{k-i} for i = 0, ..., n-1: valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        # y[k - i] for i in idx
        j_idx = k_int - idx
        grad[idx] += 2.0 * y[j_idx]
    
    # Average over active set
    if len(active_idx) > 0:
        grad /= len(active_idx)
    
    return grad, M, ac

def accelerated_mirror_descent(a0, time_limit, top_k=50, tol=0.01, verbose=False, seed=42):
    """Nesterov-accelerated mirror descent with multiplicative weight updates."""
    n = len(a0)
    a = normalized_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    
    # Adaptive step size
    eta = 1.0
    eta_min = 1e-4
    eta_decay = 0.999
    stall_count = 0
    max_stall = 500
    
    for t in range(1, 1000000000):
        if time.monotonic() - t0 > time_limit:
            break
        if t % 2000 == 0 and time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 2.0) if t > 1 else 0.0
        if t == 1:
            y = x.copy()
        else:
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-30)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        
        # Compute subgradient at y
        grad, M_y, ac_y = compute_subgradient(y, top_k=top_k, tol=tol)
        
        # Mirror descent step: y * exp(-eta * grad)
        # Use log-space for stability
        log_y = np.log(np.maximum(y, 1e-30))
        scaled = log_y - eta * grad
        # Normalize (subtract max for numerical stability)
        scaled -= scaled.max()
        a_new = np.exp(scaled)
        a_new = np.maximum(a_new, 1e-30)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / sa
        
        x_prev = x
        x = a_new
        
        # Check improvement
        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive step size
        if stall_count > max_stall:
            eta *= eta_decay
            if eta < eta_min:
                # Restart with perturbation
                pert = rng.normal(0, 0.01, n)
                x_restarted = best_a + pert * np.maximum(best_a, 1e-6)
                x_restarted = np.maximum(x_restarted, 1e-10)
                x_restarted = x_restarted / x_restarted.sum()
                x = x_restarted
                x_prev = x_restarted
                eta = 1.0
                stall_count = 0
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}, M={best_M:.8f}, eta={eta:.4f}")
    
    return best_a, best_M

def get_solver():
    """Get a working LP solver."""
    import cvxpy as cp
    for s in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def lp_frank_wolfe(a0, time_limit, solver=None, verbose=False, max_iters=5000):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    import cvxpy as cp
    if solver is None:
        solver = get_solver()
    
    n = len(a0)
    a = normalized_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    x_prev = a.copy()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current max autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Build LP: max sum(b) s.t. (y*b)_k <= M_curr for all k, b >= 0
        # (y*b)_k = sum_j y_j * b_{k-j} (valid indices)
        # We build the constraint matrix A where A[k,j] = y_j if j is valid for (y*b)_k
        num_constraints = 2 * n - 1
        b_var = cp.Variable(n, nonneg=True)
        
        # Build constraint: for each k, sum_j y[j] * b[k-j] <= M_curr
        # (y * b)_k = sum_j y[j] * b[k-j] where 0 <= j < n and 0 <= k-j < n
        # = sum_j y[j] * b[k-j] = sum_i y[k-i] * b[i] where 0 <= k-i < n and 0 <= i < n
        # So the k-th row of the constraint matrix has: A[k, i] = y[k-i] if 0 <= k-i < n
        
        # For efficiency, build as a sparse matrix
        # Row k has non-zeros at positions i where 0 <= k-i < n, i.e., max(0, k-n+1) <= i <= min(n-1, k)
        rows = []
        cols = []
        data = []
        for k in range(num_constraints):
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                rows.append(k)
                cols.append(i)
                data.append(y[k - i])
        
        import scipy.sparse as sp
        A_mat = sp.csr_matrix((data, (rows, cols)), shape=(num_constraints, n))
        
        constraints = [A_mat @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov-accelerated update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_M:
            best_M = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_M:
            best_M = mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}, M={best_M:.8f}")
    
    return best_a, best_M

def coordinate_descent_polish(a0, time_limit, verbose=False, seed=123):
    """Peak-targeted coordinate descent."""
    n = len(a0)
    a = normalized_sum1(a0)

    t0 = time.monotonic()

    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_idx = int(np.argmax(ac))
        
        # Find contributing indices to the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_idx - idx]
        
        if contrib.max() < 1e-20:
            stall += 1
            continue

            i_start = max(0, pk_int - n + 1)

            for pos in range(i_start, i_end + 1):
                sensitivity[pos] += a[pk_int - pos]
        
        improved = False
        # Try moving mass from high-contrib to low-sensitivity positions
        src_ranking = np.argsort(contrib)[::-1][:10]
        low_sens = np.argsort(sensitivity)
        
        for src_rank in src_ranking:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[src_rank])
            for dst in low_sens[:30]:
                if dst == src:
                    continue
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M - 1e-14:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        # Revert
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
    
    if verbose:
        J = 2.0 * n * best_M
        print(f"    CoordPD ({time.monotonic()-t0:.1f}s): J={J:.6f}")
    
    return best_a, best_M

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in a]
        s = sum(a_list)
        if s < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
