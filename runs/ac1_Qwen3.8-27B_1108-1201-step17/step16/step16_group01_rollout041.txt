def power_weighted_subgradient(a, ac, top_k=50, beta=6.0):
    """Compute weighted subgradient from top-k lags."""
    n = len(a)
    k_max = min(top_k, len(ac))
    sorted_idx = np.argsort(ac)[::-1][:k_max]
    vals = np.maximum(ac[sorted_idx], 1e-30)
    w = vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    g = np.zeros(n)
    for wi, k in zip(w, sorted_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * wi * a[j_idx]
    return g

    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:

    return a_new / s

            x_prev = x_restarted

            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}, eta={eta:.4f}, stall={stall_count}")

def coord_descent_2d(a0, time_limit, verbose=False, rng_seed=123):
    """
    2D coordinate descent: for each pair (i,j), find optimal transfer.
    Uses the first-order gradient to determine direction, then line search.
    """
    n = len(a0)
    a = normalize(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 50
    iteration = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iteration += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top lags
        k_max = min(30, len(ac))
        top_lags = np.argsort(ac)[::-1][:k_max]
        peak_val = ac[top_lags[0]]
        
        improved = False
        
        # Identify hot positions (involved in top lags)
        hot = set()
        for k in top_lags[:15]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                hot.add(i)
                if 0 <= k_int - i < n:
                    hot.add(k_int - i)
        
        hot_list = sorted(hot)
        cold_list = [i for i in range(n) if i not in hot]
        
        if len(cold_list) == 0:
            cold_list = list(range(n))
        
        # For each hot position, try moving to cold positions
        n_hot_try = min(len(hot_list), 60)
        rng.shuffle(hot_list)
        
        for src in hot_list[:n_hot_try]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue
            
            # Compute gradient direction: which cold position to move to
            # grad[k*] contribution from a[src] is 2*a[k*-src] for each peak k*
            # We want to move mass to a position that reduces the peak
            # Moving to dst changes conv[k] by 2t*(a[k-dst] - a[k-src])
            # We want a[k-dst] < a[k-src] for peak k, so dst should have small a[k-dst] for peak k
            
            # Score each cold position by how much it would reduce the peak
            # score[dst] = sum_{k in peaks} w_k * (a[k-src] - a[k-dst])
            # Higher score = better
            
            # For efficiency, just try a subset
            n_cold_try = min(len(cold_list), 40)
            rng.shuffle(cold_list)
            
            for dst in cold_list[:n_cold_try]:
                if time.monotonic() - t0 > time_limit:
                    break
                if dst == src or a[dst] > a[src]:
                    continue
                
                # Try several transfer fractions
                for frac in [0.001, 0.003, 0.01, 0.03, 0.08, 0.15]:
                    t_move = a[src] * frac
                    if t_move < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= t_move
                    a_trial[dst] += t_move
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
            
            if improved:
                # Recompute hot/cold after improvement
                if time.monotonic() - t0 > time_limit:
                    break
                break  # restart the outer loop
        
        if not improved:
            stall += 1
            # Escape: try random perturbations
            n_esc = max(5, n // 100)
            for _ in range(n_esc):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.02, 0.10)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        if a.sum() > 1e-15:
                            a = a / a.sum()
        
        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}, iter={iteration}")
    
    return best_a, best_m

def simulated_annealing(a0, time_limit, verbose=False, rng_seed=777):
    """
    Simulated annealing: random pairwise transfers with temperature schedule.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    T_start = best_m * 0.01  # initial temperature
    T_end = best_m * 1e-6
    
    total_steps = int(time_limit * 500)  # estimate
    accepted = 0
    total = 0
    
    for step in range(max(1000, total_steps)):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Temperature schedule: exponential cooling
        progress = min(step / max(1, total_steps), 1.0)
        T = T_start * (T_end / T_start) ** progress
        
        # Pick random pair
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if a[i] < 1e-14:
            continue
        
        # Random transfer amount
        frac = rng.uniform(0.001, 0.05)
        amt = a[i] * frac
        if amt < 1e-15:
            continue
        
        a[i] -= amt
        a[j] += amt
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a[i] += amt
            a[j] -= amt
            continue
        a = a / s
        
        M_new = fast_autocorr(a).max()
        delta = M_new - best_m  # change from best
        delta_curr = M_new - (M_new if M_new < best_m else best_m)  # not quite right
        
        # Accept if better or with probability exp(-delta/T)
        if M_new < best_m:
            # Always accept improvement
            pass
        elif T > 1e-18:
            # Accept with probability based on current state change
            # Use the change from the current state
            M_old_best = best_m  # use best as reference
            delta = M_new - M_old_best
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-18)):
                pass  # accept
            else:
                # Reject: undo
                a[i] += amt
                a[j] -= amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                continue
        else:
            # At very low T, only accept improvements
            a[i] += amt
            a[j] -= amt
            a = np.maximum(a, 0.0)
            if a.sum() > 1e-15:
                a = a / a.sum()
            continue
        
        total += 1
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
            accepted += 1
        
        if verbose and step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SA step={step} ({elapsed:.1f}s): J={J:.6f}, T={T:.2e}, acc={accepted}/{total}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def generate_initialization(n, method='sinusoidal', rng=None):
    """Generate a diverse initialization."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(0, 1, n)
    
    if method == 'sinusoidal':
        a = np.sin(np.pi * x) ** 2
        # Add some structured perturbation
        freq = rng.uniform(1, 5)
        a += 0.3 * np.cos(2 * np.pi * freq * x) ** 2
    elif method == 'triangular':
        a = np.abs(np.sin(np.pi * x))
        a += 0.1 * np.random.default_rng(rng.integers(1000)).uniform(0, 1, n)
    elif method == 'cosine_bell':
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a += 0.05
    elif method == 'random_smooth':
        a = np.cumsum(rng.normal(0, 1, n))
        a = a - a.min()
        # Smooth it
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
    elif method == 'bimodal':
        a = np.exp(-10 * (x - 0.25) ** 2) + np.exp(-10 * (x - 0.75) ** 2)
        a += 0.01
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a

import numpy as np

def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the objective."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize_seq(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def power_weighted_subgradient(a, ac, top_k=50, beta=6.0):
    """Compute weighted subgradient from top-k lags with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    sorted_idx = np.argsort(ac)[::-1][:k_max]
    vals = np.maximum(ac[sorted_idx], 1e-30)
    w = vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, sorted_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * wi * a[j_idx]
    return g


def mirror_step(a, g, eta):
    """Mirror descent update in log-space."""
    log_a = np.log(np.maximum(a, 1e-15)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

            break

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute subgradient
        ac_y = fast_autocorr(y)
        g = power_weighted_subgradient(y, ac_y, top_k=top_k, beta=beta)

        # Step size with backtracking
        if use_backtrack:
            a_trial = mirror_step(y, g, eta)
            M_trial = fast_autocorr(a_trial).max()
            M_y = ac_y.max()
            attempts = 0
            while M_trial > M_y + 1e-12 and attempts < 15 and eta > 1e-12:
                eta *= 0.6
                a_trial = mirror_step(y, g, eta)
                M_trial = fast_autocorr(a_trial).max()
                attempts += 1
            if attempts == 0:
                eta = min(eta * 1.02, c0 * np.sqrt(n))
            eta = eta * 0.99999
        else:
            eta = c0 / (np.sqrt(t) + 1.0) * np.sqrt(n) * 0.1
            a_trial = mirror_step(y, g, eta)
            M_trial = fast_autocorr(a_trial).max()

        x_prev = x
        x = a_trial

        M_new = M_trial
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Restart if stuck
        if stall_count > max_stall:
            stall_count = 0
            noise_scale = rng.uniform(0.005, 0.015)
            perturbation = rng.normal(0, noise_scale, n) * best_a
            x_restarted = best_a + perturbation
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            eta = c0 * np.sqrt(n) * 0.1
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t} ({elapsed:.1f}s): J={J:.6f}, eta={eta:.6f}")

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}, eta={eta:.6f}, stall={stall_count}")

    return best_a, best_m


def coord_descent_2d(a0, time_limit, verbose=False, rng_seed=123):
    """
    2D coordinate descent: greedy pairwise transfer optimization.
    For each hot position, try moving mass to cold positions.
    """
    n = len(a0)
    a = normalize_seq(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 80
    iteration = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iteration += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Find top lags
        k_max = min(25, len(ac))
        top_lags = np.argsort(ac)[::-1][:k_max]

        # Identify hot positions
        hot = set()
        for k in top_lags[:12]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                hot.add(i)
                if 0 <= k_int - i < n:
                    hot.add(k_int - i)

        hot_list = sorted(hot)
        cold_list = [i for i in range(n) if i not in hot]
        if len(cold_list) == 0:
            cold_list = list(range(n))

        improved = False
        n_hot_try = min(len(hot_list), 50)
        rng.shuffle(hot_list)

        for src in hot_list[:n_hot_try]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue

            # Try moving to subset of cold positions
            n_cold_try = min(len(cold_list), 35)
            rng.shuffle(cold_list)

            for dst in cold_list[:n_cold_try]:
                if time.monotonic() - t0 > time_limit:
                    break
                if dst == src or a[dst] >= a[src]:
                    continue

                # Try several transfer fractions (geometric)
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.10, 0.20]:
                    t_move = a[src] * frac
                    if t_move < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= t_move
                    a_trial[dst] += t_move
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

            if improved:
                break

        if not improved:
            stall += 1
            # Escape: multiple random perturbations
            n_esc = max(8, n // 80)
            for _ in range(n_esc):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.08)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        if a.sum() > 1e-15:
                            a = a / a.sum()

        if verbose and stall % 15 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.1f}s): J={J:.6f}, stall={stall}, iter={iteration}")

    return best_a, best_m


def simulated_annealing(a0, time_limit, verbose=False, rng_seed=777):
    """Simulated annealing: random pairwise transfers with cooling schedule."""
    n = len(a0)
    a = normalize_seq(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    T_start = best_m * 0.005
    T_end = best_m * 1e-7
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        progress = min(step / max(1, int(time_limit * 300)), 1.0)
        T = T_start * (T_end / T_start) ** progress

        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j or a[i] < 1e-14:
            continue

        frac = rng.uniform(0.002, 0.04)
        amt = a[i] * frac
        if amt < 1e-15:
            continue

        a[i] -= amt
        a[j] += amt
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a[i] += amt
            a[j] -= amt
            continue
        a = a / s

        M_new = fast_autocorr(a).max()

        if M_new < best_m:
            # Always accept improvement
            best_m = M_new
            best_a = a.copy()
        else:
            # Accept with probability exp(-(M_new - best_m) / T)
            if T > 1e-18:
                delta = M_new - best_m
                if rng.random() < np.exp(-delta / T):
                    pass  # accept worse solution (exploration)
                else:
                    # Reject: undo
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
            else:
                # At very low T, reject
                a[i] += amt
                a[j] -= amt
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-15:
                    a = a / a.sum()

        if verbose and step % 15000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SA step={step} ({elapsed:.1f}s): J={J:.6f}, T={T:.2e}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def generate_initialization(n, method, rng):
    """Generate a diverse initialization."""
    x = np.linspace(0, 1, n)

    if method == 'sinusoidal':
        a = np.sin(np.pi * x) ** 2
        freq = rng.uniform(1, 5)
        a += 0.3 * np.cos(2 * np.pi * freq * x) ** 2
    elif method == 'triangular':
        a = np.abs(np.sin(np.pi * x))
        a += 0.1 * rng.uniform(0, 1, n)
    elif method == 'cosine_bell':
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a += 0.05
    elif method == 'bimodal':
        a = np.exp(-10 * (x - 0.25) ** 2) + np.exp(-10 * (x - 0.75) ** 2)
        a += 0.01
    elif method == 'raised_cosine':
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = a ** 1.5
        a += 0.01
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def update_best(state, a, label, t_global):
    """Update the best state if a is better."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < state['score']:
        state['score'] = sc
        state['seq'] = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Phases:
    1. AMD from parent (backtracking, power-weighted subgradient)
    2. 2D coordinate descent
    3. AMD with different parameters
    4. Simulated annealing
    5. Multi-length exploration
    6. Final polish cycles (AMD + coord)
    7. Diverse restarts
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    rng = np.random.default_rng(42)

    state = {'seq': None, 'score': float('inf')}

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            state['seq'] = parent.copy()
            state['score'] = p_score
    except Exception:
        pass

    if state['seq'] is None:
        n_base = 2400
        state['seq'] = generate_initialization(n_base, 'sinusoidal', rng)
        state['score'] = evaluate_safe(state['seq'])
        print(f"Default init: n={len(state['seq'])}, score={state['score']:.6f}")

    n0 = len(state['seq'])

    # === Phase 1: AMD (main workhorse, backtracking) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: AMD n={n0} ({remaining:.0f}s remaining) ===")
    t1 = min(remaining * 0.22, 300.0)
    if t1 > 30:
        a_out, _ = amd_improved(state['seq'], t1, c0=3.0, top_k=60, beta=5.0,
                                verbose=True, rng_seed=1, use_backtrack=True)
        update_best(state, a_out, "amd1", t_global)

    # === Phase 2: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state['seq'])
    print(f"\n=== Phase 2: Coord descent n={n_best} ===")
    t2 = min(remaining * 0.14, 180.0)
    if t2 > 20:
        a_out, _ = coord_descent_2d(state['seq'], t2, verbose=True, rng_seed=2)
        update_best(state, a_out, "coord1", t_global)

    # === Phase 3: AMD with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state['seq'])
    print(f"\n=== Phase 3: AMD2 n={n_best} ===")
    t3 = min(remaining * 0.18, 220.0)
    if t3 > 25:
        a_out, _ = amd_improved(state['seq'], t3, c0=2.0, top_k=80, beta=8.0,
                                verbose=True, rng_seed=3, use_backtrack=True)
        update_best(state, a_out, "amd2", t_global)

    # === Phase 4: Coordinate descent again ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state['seq'])
    print(f"\n=== Phase 4: Coord2 n={n_best} ===")
    t4 = min(remaining * 0.10, 130.0)
    if t4 > 15:
        a_out, _ = coord_descent_2d(state['seq'], t4, verbose=True, rng_seed=4)
        update_best(state, a_out, "coord2", t_global)

    # === Phase 5: Simulated annealing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state['seq'])
    print(f"\n=== Phase 5: Simulated annealing n={n_best} ===")
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 15:
        a_out, _ = simulated_annealing(state['seq'], t5, verbose=True, rng_seed=5)
        update_best(state, a_out, "sa", t_global)

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state['seq'])
    if remaining > 150:
        print(f"\n=== Phase 6: Multi-length ===")
        for delta in [300, 500, -300, -500, 800]:
            n_try = n_best + delta
            if n_try < 500 or n_try > 4000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            print(f"  Trying n={n_try}")
            a_interp = interpolate_sequence(state['seq'], n_try)
            t_try = min(remaining * 0.15, 120.0)
            if t_try > 25:
                a_out, _ = amd_improved(a_interp, t_try, c0=3.0, top_k=50, beta=6.0,
                                        verbose=False, rng_seed=6, use_backtrack=True)
                update_best(state, a_out, f"mdl_n{n_try}", t_global)
                if a_out is not None:
                    remaining = time_budget - (time.monotonic() - t_global)
                    if remaining > 40:
                        a_out2, _ = coord_descent_2d(a_out, min(remaining * 0.25, 60.0),
                                                     verbose=False, rng_seed=7)
                        update_best(state, a_out2, f"mdl_coord_n{n_try}", t_global)

    # === Phase 7: Final AMD polish (tight) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state['seq'])
    print(f"\n=== Phase 7: Final AMD polish n={n_best} ===")
    if remaining > 30:
        t7 = min(remaining * 0.25, 180.0)
        if t7 > 20:
            a_out, _ = amd_improved(state['seq'], t7, c0=1.5, top_k=100, beta=10.0,
                                    verbose=True, rng_seed=8, use_backtrack=True)
            update_best(state, a_out, "amd_polish", t_global)

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state['seq'])
    print(f"\n=== Phase 8: Final coord n={n_best} ===")
    if remaining > 20:
        t8 = min(remaining * 0.25, 100.0)
        if t8 > 10:
            a_out, _ = coord_descent_2d(state['seq'], t8, verbose=True, rng_seed=9)
            update_best(state, a_out, "coord_final", t_global)

    # === Phase 9: Diverse restarts (blend with new inits) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 9: Diverse restarts ===")
        n_best = len(state['seq'])
        methods = ['cosine_bell', 'bimodal', 'raised_cosine', 'triangular']
        for method in methods:
            if time.monotonic() - t_global > time_budget - 30:
                break
            a_init = generate_initialization(n_best, method, rng)
            # Blend with current best (50/50)
            a_blend = 0.5 * a_init + 0.5 * state['seq']
            a_blend = normalize_seq(a_blend)
            t_try = min(remaining * 0.12, 80.0)
            if t_try > 15:
                a_out, _ = amd_improved(a_blend, t_try, c0=3.0, top_k=60, beta=5.0,
                                        verbose=False, rng_seed=10, use_backtrack=True)
                update_best(state, a_out, f"div_{method}", t_global)
                if a_out is not None:
                    remaining = time_budget - (time.monotonic() - t_global)
                    if remaining > 25:
                        a_out2, _ = coord_descent_2d(a_out, min(remaining * 0.3, 40.0),
                                                     verbose=False, rng_seed=11)
                        update_best(state, a_out2, f"div_{method}_coord", t_global)

    # === Phase 10: Micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state['seq'])
    if remaining > 15:
        t10 = min(remaining * 0.50, 80.0)
        if t10 > 10:
            print(f"\n=== Phase 10: Micro-polish n={n_best} ===")
            a_out, _ = coord_descent_2d(state['seq'], t10, verbose=False, rng_seed=12)
            update_best(state, a_out, "micro", t_global)

    # Final validation
    if state['seq'] is not None:
        fs = evaluate_safe(state['seq'])
        if np.isfinite(fs):
            state['score'] = fs
        print(f"\nFINAL: score={state['score']:.6f}, n={len(state['seq'])}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in state['seq']]
