def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    return g

    """Compute gradient of smoothed max."""
    n = len(a)
    ac = fast_autocorr(a)
    num_lags = len(ac)

    weights = exp_vals / exp_vals.sum()

        for m in range(n):
            k_m = k_int - m
