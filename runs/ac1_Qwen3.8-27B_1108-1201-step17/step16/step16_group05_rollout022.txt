def fast_autocorr(a):
    """Compute linear autocorrelation (full convolution a*a) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative simplex (sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_score(a):
    """Compute the score 2*n*max_b/(sum_a)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    ac = fast_autocorr(a)
    max_b = ac.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def grad_smoothed_max(a, beta, top_k=100):
    """
    Gradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a.
    Uses top-k approximation for efficiency.
    Returns (gradient, max_b).
    """
    n = len(a)
    ac = fast_autocorr(a)
    num_lags = len(ac)
    k_eff = min(top_k, num_lags)
    top_idx = np.argsort(ac)[::-1][:k_eff]
    top_vals = ac[top_idx]
    shifted = top_vals - top_vals[0]
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros(n, dtype=np.float64)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        for i in range(n):
            idx = k_int - i
            if 0 <= idx < n:
                grad[i] += 2.0 * w * a[idx]
    return grad, ac.max()


def grad_single_peak(a, k_peak):

    n = len(a)

    for i in range(n):
        idx = k_peak - i

    return grad

def lp_frank_wolfe(a0, time_limit, solver, verbose=False):
    """
    LP-based Frank-Wolfe with Nesterov acceleration.
    At each step, solves LP to find best direction within {b>=0, (a*b)_k <= M}.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()

    for t in range(1, 50000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max() + 1e-12

        # Build LP: maximize sum(b) s.t. b>=0, T@b <= M_curr
        b_var = cp.Variable(n, nonneg=True)

        # T[k, j] = y[k-j] if 0 <= k-j < n, else 0
        # (T@b)[k] = sum_j T[k,j] * b[j] = sum_j y[k-j]*b[j] = (y*b)[k]
        # Build T efficiently using the Toeplitz structure
        T = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            T[j:j + n, j] = y

        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # FW step with Nesterov
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new

        mc = fast_autocorr(y).max()
        if mc < best_m - 1e-15:
            best_m = mc
            best_a = y.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): score={J:.6f}")

    return best_a, best_m

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
