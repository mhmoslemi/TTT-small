Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses Nesterov-accelerated LP best-response with adaptive active sets,
smoothed-max subgradient polishing, and multi-n exploration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import math

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def normalize_to(a, n):
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s

def subgradient_at_k(a, k, n):
    grad = np.zeros(n)
    # d(a*a)[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    for j in range(lo, hi):
        g = 0.0
        if 0 <= k - j and k - j < n:
            g += a[k - j]
        if 0 <= j - k and j - k < n:
            g += a[j - k]
        grad[j] = g
    return grad

def subgradient_at_k_fast(a, k, n):
    """Vectorized subgradient computation."""
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        j_idx = np.arange(lo, hi)
        # a[k-j] for valid k-j
        kj = k - j_idx
        mask1 = (kj >= 0) & (kj < n)
        grad[lo:hi] += np.where(mask1, a[kj], 0.0)
        # a[j-k] for valid j-k
        jk = j_idx - k
        mask2 = (jk >= 0) & (jk < n)
        grad[lo:hi] += np.where(mask2, a[jk], 0.0)
    return grad

def lp_best_response(a_fixed, M, active_set, solver):
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    
    # Build constraint matrix T where T[i,j] = d(a_fixed * b)[k_i]/db[j]
    # = a_fixed[k_i - j] (if valid) + a_fixed[j - k_i] (if valid)
    # This is the same as the subgradient structure
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            j_idx = np.arange(lo, hi)
            kj = k - j_idx
            mask1 = (kj >= 0) & (kj < n)
            T[idx, lo:hi] += a_fixed[np.where(mask1, kj, 0)] * mask1
            jk = j_idx - k
            mask2 = (jk >= 0) & (jk < n)
            T[idx, lo:hi] += a_fixed[np.where(mask2, jk, 0)] * mask2
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    return b_val, None

def nesterov_lp_optimize(a0, time_limit, solver, seed=0, verbose=False):
    """Nesterov-accelerated LP best-response with adaptive active-set."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    a_curr = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    n_iters = 0
    conv_size = 2 * n - 1
    fail_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set
        if active_set is None or t % 30 == 0:
            # Progressive tightening
            progress = min(1.0, t / 800.0)
            # Start with wider tolerance, tighten over time
            tol = 0.12 * (1.0 - progress) + 0.003 * progress
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            # Always include top peaks
            n_top = max(10, n // 50)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            # Cap for efficiency
            max_active = int(2 * n)
            if len(active_set) > max_active:
                conv_at_active = conv_y[active_set]
                keep = np.argsort(conv_at_active)[-max_active:]
                active_set = active_set[keep]

        b, _ = lp_best_response(y, M_curr, active_set, solver)

        if b is None:
            fail_count += 1
            if fail_count == 1:
                # Retry with all constraints
                active_set = np.arange(conv_size)
                b, _ = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                if fail_count > 10:
                    # Restart active set
                    active_set = None
                    fail_count = 0
                continue
            active_set = None  # Reset after full solve
        fail_count = 0

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            momentum = min(momentum, 0.95)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a_curr = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a_curr).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a_curr.copy()
            stall_count = 0

        # Damped combination check
        if t % 30 == 0:
            for w in [0.5, 0.7, 0.85]:
                a_damped = (1-w) * best_a + w * a_curr
                a_damped = np.maximum(a_damped, 0.0)
                s_d = a_damped.sum()
                if s_d > 1e-15:
                    a_damped = a_damped * norm / s_d
                    score_d = fast_autocorr(a_damped).max()
                    if score_d < best_score:
                        best_score = score_d
                        best_a = a_damped.copy()
                        stall_count = 0

        # Restart if stalled
        if stall_count > 80:
            y = best_a.copy()
            y_prev = best_a.copy()
            a_curr = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPN t={t} iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def smoothed_subgradient(a0, time_limit, seed=42, verbose=False):
    """
    Subgradient method using log-sum-exp smoothed max.
    The smoothed objective is: f_t(a) = t * log(sum_k exp((a*a)[k]/t))
    Its gradient is: sum_k p_k * grad_k where p_k = exp((a*a)[k]/t) / sum_j exp((a*a)[j]/t)
    This gives a smooth, well-defined gradient at all points.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 1.0 / np.sqrt(n)
    patience = 0
    max_patience = 500
    rng = np.random.default_rng(seed)
    conv_size = 2 * n - 1
    
    # Adaptive temperature
    temp = 0.05  # smoothing parameter
    min_temp = 0.005
    
    for it in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Compute smoothed weights (log-sum-exp for numerical stability)
        shifted = (conv - M_curr) / temp  # shift for stability
        log_weights = shifted - np.log(np.exp(shifted).sum())
        weights = np.exp(log_weights)

        # Compute weighted subgradient
        # Use top peaks for efficiency
        n_peaks = min(conv_size, max(20, n // 5))
        top_idx = np.argsort(weights)[-n_peaks:]
        
        grad = np.zeros(n)
        for k in top_idx:
            if weights[k] < 1e-10:
                continue
            grad += weights[k] * subgradient_at_k_fast(a, int(k), n)
        
        # Normalize gradient
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        # Projected subgradient step
        a_trial = a - step * grad * np.sqrt(n)
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            break
        a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.003, 3.0)
            # Decrease temperature
            temp = max(temp * 0.998, min_temp)
        else:
            patience += 1
            step *= 0.98
            if step < 1e-10:
                break

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothSG t={it} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}, temp={temp:.5f}")

    return best_a, best_score

def coordinate_descent_fast(a0, time_limit, seed=0, verbose=False):
    """Fast coordinate descent."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0
    max_no_improve = 6

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        conv = fast_autocorr(a)
        current_max = conv.max()

        # Focus on coordinates that most affect the max
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]

            # Try a few candidate values
            typical = norm / n
            candidates = [0.0, aj * 0.5, aj * 0.8, aj * 1.2, aj * 1.5, typical * 0.5, typical, typical * 2.0]
            candidates = sorted(set([max(0.0, c) for c in candidates]))

            best_x = aj
            best_local = current_max

            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue

                # Quick update of convolution
                delta = x - aj
                lo = max(0, j)
                hi = min(size, j + n)
                if lo >= hi:
                    continue
                
                # (a*a)[k] changes by 2*(x-aj)*a[k-j] for k where k-j is valid
                # Plus correction at k=2j
                new_conv = conv.copy()
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta

                # Also need to account for the other part of the convolution
                # (a*a)[k] = sum_i a[i]*a[k-i]
                # When a[j] changes to x, the terms involving a[j] change:
                # For term a[j]*a[k-j]: changes by (x-aj)*a[k-j] for valid k-j
                # For term a[i]*a[j] where i = k-j, i.e., a[k-j]*a[j]: changes by a[k-j]*(x-aj)
                # Wait, these are the same thing. Let me be more careful.
                
                # (a*a)[k] = sum_i a[i] * a[k-i]
                # Terms involving a[j]: when i=j, term is a[j]*a[k-j]; when k-i=j, i.e., i=k-j, term is a[k-j]*a[j]
                # If j != k-j (i.e., j != k/2), these are two separate terms, both equal to a[j]*a[k-j]
                # If j == k-j (i.e., k=2j), it's a single term a[j]^2
                # So the change when a[j] -> x is:
                #   For k != 2j: 2*(x-aj)*a[k-j] (if 0<=k-j<n)
                #   For k = 2j: (x^2 - aj^2)
                
                # Actually my update above is approximately correct for the "2*delta*a[k-j]" part
                # The correction at k=2j handles the special case.
                # But I also need to handle the case where k-j is out of bounds.
                
                # Let me redo this more carefully:
                new_conv2 = conv - 2.0 * aj * (np.zeros(size))  # start fresh conceptually
                # Actually, let's just do it the simple way:
                # conv_old[k] includes terms with a[j]
                # conv_new[k] = conv_old[k] - 2*a[j]*a[k-j] + x*a[k-j] - a[j]*x + correction
                # Hmm, this is getting complicated. Let me use a simpler approach.
                
                # Simple approach: just update the affected entries
                # Entries k where k-j is in [0, n-1]: i.e., k in [j, j+n-1]
                # For these: conv[k] += (x-aj)*a[k-j] (from the a[j]*a[k-j] term)
                #                         + a[k-j]*(x-aj) (from the a[k-j]*a[j] term, if k-j != j)
                # If k-j == j (i.e., k = 2j): only one term, so conv[k] += (x^2 - aj^2)
                # Otherwise: conv[k] += 2*(x-aj)*a[k-j]
                
                # This is what I had. Let me also handle the "a[i]*a[j]" where i = k-j:
                # Actually, in the sum sum_i a[i]*a[k-i], the term where i=j gives a[j]*a[k-j].
                # The term where k-i=j, i.e., i=k-j, gives a[k-j]*a[j].
                # These are the same term when j = k-j, i.e., k=2j.
                # So for k != 2j: two terms, total change = 2*(x-aj)*a[k-j]
                # For k = 2j: one term, change = x^2 - aj^2
                
                # My update is correct. Let me just check the bounds.
                # k ranges from j to j+n-1 (so that k-j ranges from 0 to n-1)
                # Also k must be in [0, 2n-2]
                
                new_max = new_conv.max()
                # Scale: sum changes from norm to norm - aj + x
                new_sum = norm - aj + x
                if new_sum < 1e-15:
                    continue
                scale = norm / new_sum
                scaled_max = new_max * scale * scale

                if scaled_max < best_local:
                    best_local = scaled_max
                    best_x = x

            if abs(best_x - aj) > 1e-12:
                old_a = a.copy()
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                    else:
                        a = old_a

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False

def propose_candidate():
    """
    Multi-phase optimization:
    1. Load parent or create default
    2. LP Nesterov (main optimization)
    3. Smoothed subgradient polish
    4. Coordinate descent
    5. LP round 2 with tighter settings
    6. Final CD polish
    7. Try alternative n values
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = 1770.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)
    print(f"Starting with n={n_base}, score={best_score:.8f}")

    # === Phase 1: Main LP Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
        a_norm = normalize_to(best_seq, n_base)
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, seed=0)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp1_n{n_base}", t_global)

    # === Phase 2: Smoothed subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.15, 180.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Smoothed SG n={n_cur}, budget={t2:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = smoothed_subgradient(a_norm, t2, verbose=True, seed=1)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"ssg_n{n_cur}", t_global)

    # === Phase 3: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40:
        print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur}, budget={t3:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, seed=123)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp2_n{n_cur}", t_global)

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.12, 120.0)
    if t4 > 20:
        print(f"\n=== Phase 4: CD n={n_cur}, budget={t4:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent_fast(a_norm, t4, verbose=True, seed=7)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd_n{n_cur}", t_global)

    # === Phase 5: More LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        print(f"\n=== Phase 5: LP Nesterov r3 n={n_cur}, budget={t5:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=False, seed=999)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp3_n{n_cur}", t_global)

    # === Phase 6: Final CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:

    """Compute subgradient of (a*a)[k] w.r.t. a efficiently."""
    grad = np.zeros(n)

    hi = min(n, k + 1)

        jk = j_idx - k
        mask2 = (jk >= 0) & (jk < n)

            grad[lo:hi] += np.where(mask2, a[jk], 0.0)

def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)

    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            j_idx = np.arange(lo, hi)
            kj = k - j_idx
            mask1 = (kj >= 0) & (kj < n)
            if np.any(mask1):
                T[idx, lo:hi] += a_fixed[np.where(mask1, kj, 0)] * mask1.astype(np.float64)
            jk = j_idx - k
            mask2 = (jk >= 0) & (jk < n)
            if np.any(mask2):
                T[idx, lo:hi] += a_fixed[np.where(mask2, jk, 0)] * mask2.astype(np.float64)

    constraints = [T @ b_var <= M]

    conv_size = 2 * n - 1

    # Adaptive temperature: start moderate, decrease over time
    temp = 0.08
    min_temp = 0.003

        if patience > max_patience:
            break

    """Fast coordinate descent with efficient autocorrelation updates."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

        return a, float('inf')
    a = a * norm / s0

    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0
    max_no_improve = 5

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        conv = fast_autocorr(a)
        current_max = conv.max()

        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            typical = norm / n

            # Generate candidates based on current value and typical
            candidates = [0.0, aj * 0.2, aj * 0.5, aj * 0.7, aj * 1.3, aj * 1.6,
                          typical * 0.5, typical * 0.8, typical, typical * 1.5, typical * 2.5]
            candidates = sorted(set([max(0.0, min(c, norm * 0.3)) for c in candidates]))

            best_x = aj
            best_local = current_max

            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue

                delta = x - aj
                lo = max(0, j)
                hi = min(size, j + n)
                if lo >= hi:
                    continue

                new_conv = conv.copy()
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta

                new_sum = norm - aj + x
                if new_sum < 1e-15:
                    continue
                scale = norm / new_sum
                scaled_max = new_conv.max() * scale * scale

                if scaled_max < best_local:
                    best_local = scaled_max
                    best_x = x

            if abs(best_x - aj) > 1e-12:
                old_a = a.copy()
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                    else:
                        a = old_a.copy()

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence."""
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False


def perturb_and_optimize(best_seq, time_limit, solver, rng, seed):
    """Perturb the best sequence and optimize."""
    n = len(best_seq)
    scale = 0.008 + 0.012 * rng.random()
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.3)
    perturbed = best_seq * noise
    s = perturbed.sum()
    if s < 1e-15:
        return best_seq
    perturbed = perturbed / s
    a_norm = normalize_to(perturbed, n)
    a_out, _ = nesterov_lp_optimize(a_norm, time_limit, solver, verbose=False, seed=seed)
    return a_out

    time_budget = 1770.0

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)
    print(f"Starting optimization with n={n_base}, score={best_score:.8f}")

    # === Phase 1: Main LP Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
        a_norm = normalize_to(best_seq, n_base)

        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"ssg_n{n_cur}", t_global)

    # === Phase 3: LP Nesterov round 2 (continued from improved point) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40:
        print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur}, budget={t3:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, seed=123)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp2_n{n_cur}", t_global)

    remaining = time_budget - (time.monotonic() - t_global)

        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd_n{n_cur}", t_global)

    # === Phase 6: LP round 3 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.12, 150.0)

        print(f"\n=== Phase 6: LP r3 n={n_cur}, budget={t6:.0f}s ===")

    # === Phase 8: Final CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final CD n={n_cur} ===")
        t8 = min(remaining * 0.4, 60.0)
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent_fast(a_norm, t8, verbose=False, seed=77)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cdFinal_n{n_cur}", t_global)

    # === Phase 9: Final smoothed subgradient micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 20:
        t9 = min(remaining * 0.5, 40.0)
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = smoothed_subgradient(a_norm, t9, verbose=False, seed=314)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"sgFinal_n{n_cur}", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):

        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
