"""
Search for non-negative step heights minimizing:
    2 * n * max(conv(a, a)) / (sum(a))^2

Method:
1. Start from the supplied parent construction when available, otherwise a smooth
   non-negative seed.
2. Use backtracking multiplicative mirror descent with softmax-weighted subgradients
   computed over all autocorrelation lags.
3. Use LP Frank-Wolfe with a larger active-constraint set and line search.
4. Use pressure-guided pairwise coordinate polishing with exact re-evaluation.
5. Try one or two longer lengths and finish with a final local polish.
"""

import numpy as np
import cvxpy as cp
import time


def norm1(a):
    """Clip to non-negative finite values and normalize to unit sum."""
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim == 0:
        arr = arr.reshape(1)
    arr = arr.ravel()
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(arr.size, dtype=np.float64) / arr.size
    return arr / s


def fast_conv(a):
    """Full linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    return np.maximum(out, 0.0)


def evaluate_safe(a):
    """Call the external evaluator safely."""
    try:
        lst = [float(x) for x in a]
        return float(evaluate_sequence(lst))
    except Exception:
        return float("inf")


def interp_sequence(a, m):
    """Piecewise-linear interpolation of a normalized sequence to length m."""
    a = norm1(a)
    m = int(max(1, m))
    n0 = len(a)
    if n0 == m:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, m)
    b = np.interp(x1, x0, a)
    return norm1(b)


def pick_solver():
    """Find a usable CVXPY LP solver."""
    names = ["CLARABEL", "PDLP", "ECOS", "SCS"]
    for name in names:
        solver = getattr(cp, name, None)
        if solver is None:
            continue
        try:
            x = cp.Variable(8, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver)
            if prob.status in ("optimal", "optimal_inaccurate"):
                return solver
        except Exception:
            continue
    return getattr(cp, "ECOS", None)


def softmax_weights(vals, temp):
    """Stable softmax weights over lag values."""
    vals = np.asarray(vals, dtype=np.float64)
    if vals.size == 0:
        return np.array([], dtype=np.float64)
    peak = float(vals.max())
    if peak <= 0.0 or not np.isfinite(peak):
        return np.full(vals.shape, 1.0 / vals.size, dtype=np.float64)
    t = max(float(temp), 1e-15)
    v = (vals - peak) / t
    v = np.clip(v, -50.0, 50.0)
    w = np.exp(v)
    s = float(w.sum())
    if s < 1e-300:
        w = np.zeros_like(vals, dtype=np.float64)
        w[int(np.argmax(vals))] = 1.0
        return w
    return w / s


def grad_full_from_widths(a, w):
    """
    Gradient of sum_k w[k] * conv(a,a)[k] with respect to a.

    If c[k] = sum_i a[i] a[k-i], then:
      d c[k] / d a[j] = 2 a[k-j] if 0 <= k-j < len(a), else 0.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = len(a)
    if n <= 1:
        return np.zeros(n, dtype=np.float64)
    out_len = len(w) + n - 1
    N = 1
    while N < out_len:
        N <<= 1
    W = np.fft.rfft(w, N)
    AR = np.fft.rfft(a[::-1], N)
    conv = np.fft.irfft(W * AR, N)[:out_len]
    g = 2.0 * conv[n - 1:2 * n - 1]
    return g


def mirror_descent(a0, time_limit, verbose=False,
                   rel_temp_start=0.25, rel_temp_end=0.010,
                   seed=12345):
    """
    Backtracking multiplicative mirror descent using softmax weights over lags.
    Returns (best_sequence_normalized, best_max_convolution).
    """
    a = norm1(a0)
    n = len(a)
    if n <= 2 or time_limit <= 0.0:
        c = fast_conv(a)
        return a, float(c.max())

    t0 = time.monotonic()
    x = a.copy()
    x_prev = a.copy()
    c = fast_conv(x)
    best = x.copy()
    best_m = float(c.max())
    ema = x.copy()
    best_stall = 0
    restart_limit = 800
    rng = np.random.default_rng(seed)
    targets = (1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008)

    for t in range(1, 2000000):
        if time.monotonic() - t0 >= time_limit:
            break

        if t <= 20:
            beta = 0.0
        else:
            beta = min(0.85, (t - 20.0) / float(t + 20.0))

        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-18)
        sy = float(y.sum())
        if sy < 1e-15 or not np.isfinite(sy):
            y = x.copy()
        else:
            y = y / sy

        c = fast_conv(y)
        peak = float(c.max())
        if peak < best_m:
            best_m = peak
            best = y.copy()
            best_stall = 0

        frac = min(1.0, t / 8000.0)
        rel_temp = rel_temp_start * (1.0 - frac) + rel_temp_end * frac
        w = softmax_weights(c, peak * rel_temp)
        g = grad_full_from_widths(y, w)
        gmax = float(np.max(np.abs(g)))

        last_x = None
        last_m = None
        accepted = False

        if gmax > 1e-18 and np.isfinite(gmax):
            log_y = np.log(np.maximum(y, 1e-300))
            for target in targets:
                eta = target / gmax
                shift = -eta * g
                shift = np.clip(shift, -16.0, 16.0)
                ln = log_y + shift
                ln -= ln.max()
                x_trial = np.exp(ln)
                x_trial = np.maximum(x_trial, 0.0)
                s_trial = float(x_trial.sum())
                if s_trial < 1e-15 or not np.isfinite(s_trial):
                    continue
                x_trial = x_trial / s_trial
                m_trial = float(fast_conv(x_trial).max())
                last_x = x_trial
                last_m = m_trial
                if m_trial < peak * (1.0 - 1e-10) or m_trial < best_m:
                    accepted = True
                    break

        if accepted:
            x_prev = x
            x = last_x
            m_cur = last_m
        elif last_x is not None and last_m is not None and last_m < peak * 1.05:
            x_prev = x
            x = last_x
            m_cur = last_m
        else:
            noise = 0.03 * rng.normal(0.0, 1.0, n)
            x = best * (1.0 + noise)
            x = np.maximum(x, 1e-18)
            sx = float(x.sum())
            if sx <= 0.0 or not np.isfinite(sx):
                x = best.copy()
            else:
                x = x / sx
            x_prev = x
            m_cur = float(fast_conv(x).max())

        if m_cur < best_m:
            best_m = m_cur
            best = x.copy()
            best_stall = 0
        else:
            if m_cur < peak:
                best_stall = max(0, best_stall - 1)
            else:
                best_stall += 1

        if t % 200 == 0:
            ema = 0.98 * ema + 0.02 * x
            ema = np.maximum(ema, 0.0)
            se = float(ema.sum())
            if se > 1e-15 and np.isfinite(se):
                ema = ema / se
                m_ema = float(fast_conv(ema).max())
                if m_ema < best_m:
                    best_m = m_ema
                    best = ema.copy()
                    best_stall = 0

        if best_stall > restart_limit:
            noise = 0.05 * rng.normal(0.0, 1.0, n)
            x = best * (1.0 + noise)
            x = np.maximum(x, 1e-18)
            sx = float(x.sum())
            if sx <= 0.0 or not np.isfinite(sx):
                x = best.copy()
            else:
                x = x / sx
            x_prev = x
            best_stall = 0
            restart_limit = max(300, int(restart_limit * 0.85))

        if verbose and t % 2000 == 0:
            print(f"    MD t={t} ({time.monotonic() - t0:.1f}s) "
                  f"score={2.0 * n * best_m:.8f} stall={best_stall}")

    for cand in (x, ema):
        cand = norm1(cand)
        m = float(fast_conv(cand).max())
        if m < best_m:
            best_m = m
            best = cand.copy()

    return best, best_m


def lp_frank_wolfe(a0, time_limit, solver, max_cons=500, verbose=False):
    """
    LP Frank-Wolfe with line search.
    At each step solve:
        maximize sum b
        subject to b >= 0 and conv(a, b)[k] <= peak for selected lags k.
    Returns (best_sequence_normalized, best_max_convolution).
    """
    a = norm1(a0)
    n = len(a)
    c = fast_conv(a)
    if n <= 1 or time_limit <= 0.0 or solver is None:
        return a, float(c.max())

    t0 = time.monotonic()
    best = a.copy()
    best_m = float(c.max())
    y = a.copy()
    alphas = (1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02)
    max_cons = max(50, int(max_cons))

    for t in range(1, 200000):
        if time.monotonic() - t0 >= time_limit:
            break

        c = fast_conv(y)
        peak = float(c.max())
        if peak < best_m:
            best_m = peak
            best = y.copy()

        if peak <= 0.0 or not np.isfinite(peak):
            break

        k_pool = min(max_cons, len(c))
        idx = np.argsort(c)[::-1][:k_pool]
        hi = np.where(c >= 0.97 * peak - 1e-18)[0]
        if hi.size:
            idx = np.unique(np.concatenate((idx, hi)))
            if idx.size > k_pool:
                idx = idx[np.argsort(c[idx])[::-1]][:k_pool]

        T = np.zeros((idx.size, n), dtype=np.float64)
        for r, k in enumerate(idx):
            kk = int(k)
            s = max(0, kk - n + 1)
            e = min(n, kk + 1)
            if s < e:
                cols = np.arange(s, e)
                T[r, cols] = y[kk - cols]

        bvar = cp.Variable(n, nonneg=True)
        rhs = peak * (1.0 + 1e-12) + 1e-18
        prob = cp.Problem(cp.Maximize(cp.sum(bvar)), [T @ bvar <= rhs])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ("optimal", "optimal_inaccurate"):
            continue
        if bvar.value is None:
            continue

        bval = np.asarray(bvar.value, dtype=np.float64)
        if not np.all(np.isfinite(bval)):
            continue
        bval = np.maximum(bval, 0.0)
        bs = float(bval.sum())
        if bs < 1e-12 or not np.isfinite(bs):
            continue
        bval = bval / bs

        best_line_m = peak
        best_line_y = y
        best_alpha = 0.0
        for alpha in alphas:
            cand = (1.0 - alpha) * y + alpha * bval
            cand = np.maximum(cand, 0.0)
            ss = float(cand.sum())
            if ss < 1e-15 or not np.isfinite(ss):
                continue
            cand = cand / ss
            m = float(fast_conv(cand).max())
            if m < best_line_m - 1e-15:
                best_line_m = m
                best_line_y = cand
                best_alpha = alpha

        if best_alpha > 0.0:
            y = best_line_y
            if best_line_m < best_m:
                best_m = best_line_m
                best = y.copy()

        if verbose and t % 25 == 0:
            print(f"    LPFW t={t} ({time.monotonic() - t0:.1f}s) "
                  f"score={2.0 * n * best_m:.8f} cons={idx.size} alpha={best_alpha:.3f}")

    return best, best_m


def coordinate_polish(a0, time_limit, verbose=False, seed=777):
    """
    Pressure-guided pairwise coordinate polish.
    Moves small mass from high pressure to low pressure and accepts only if
    an exact FFT convolution re-evaluation improves the best known peak.
    """
    a = norm1(a0)
    n = len(a)
    L = 2 * n - 1
    if n <= 1 or time_limit <= 0.0:
        c = fast_conv(a)
        return a, float(c.max())

    t0 = time.monotonic()
    best = a.copy()
    c = fast_conv(a)
    best_m = float(c.max())
    rng = np.random.default_rng(seed)
    fracs = (0.001, 0.002, 0.004, 0.008, 0.015, 0.03, 0.06, 0.12)
    active_count = min(60, L)
    best_stall = 0
    stall_limit = 800
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_conv(a)
        peak = float(c.max())
        if peak < best_m:
            best_m = peak
            best = a.copy()
            best_stall = 0
        else:
            best_stall += 1

        active = np.argsort(c)[::-1][:active_count]
        p = np.zeros(n, dtype=np.float64)
        for k in active:
            kk = int(k)
            s = max(0, kk - n + 1)
            e = min(n, kk + 1)
            if s < e:
                idx = np.arange(s, e)
                p[s:e] += a[kk - idx]

        if n > 20:
            hi_cut = float(np.quantile(p, 0.98))
            lo_cut = float(np.quantile(p, 0.02))
            hidx = np.where(p >= hi_cut)[0]
            lidx = np.where(p <= lo_cut)[0]
        else:
            hidx = np.arange(n)
            lidx = np.arange(n)

        if hidx.size == 0 or lidx.size == 0:
            hidx = np.arange(n)
            lidx = np.arange(n)

        for sub in range(3):
            if hidx.size and (sub == 0 or rng.random() < 0.8):
                src = int(hidx[int(rng.integers(hidx.size))])
                dst = int(lidx[int(rng.integers(lidx.size))])
            else:
                src = int(rng.integers(n))
                dst = int(rng.integers(n))

            if src == dst:
                continue

            max_p = float(a[src])
            max_n = float(a[dst])
            minmass = min(max_p, max_n)
            if minmass <= 1e-18:
                continue

            d = np.zeros(L, dtype=np.float64)
            d[src:src + n] -= 2.0 * a
            d[dst:dst + n] += 2.0 * a

            best_delta = 0.0
            best_trial = peak

            for f in fracs:
                val = float(f) * minmass
                if val <= max_p + 1e-30:
                    trial = float(np.max(c + val * d))
                    if trial < best_trial:
                        best_trial = trial
                        best_delta = val
                if val <= max_n + 1e-30:
                    trial = float(np.max(c - val * d))
                    if trial < best_trial:
                        best_trial = trial
                        best_delta = -val

            if best_delta != 0.0 and best_trial < peak - 1e-15:
                a2 = a.copy()
                if best_delta > 0.0:
                    amt = min(float(best_delta), float(a2[src]))
                    a2[src] -= amt
                    a2[dst] += amt
                else:
                    amt = min(float(-best_delta), float(a2[dst]))
                    a2[dst] -= amt
                    a2[src] += amt
                a2 = np.maximum(a2, 0.0)
                ss = float(a2.sum())
                if ss > 1e-15 and np.isfinite(ss):
                    a2 /= ss
                    m2 = float(fast_conv(a2).max())
                    if m2 < best_m - 1e-16:
                        a = a2
                        best_m = m2
                        best = a.copy()
                        c = fast_conv(a)
                        peak = float(c.max())
                        best_stall = 0
                        break

        if best_stall > stall_limit:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = float(a[i]) * float(rng.uniform(0.005, 0.02))
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                ss = float(a.sum())
                if ss > 1e-15 and np.isfinite(ss):
                    a = a / ss
            best_stall = 0
            stall_limit = max(250, int(stall_limit * 0.85))

        if verbose and it % 200 == 0:
            print(f"    CD it={it} ({time.monotonic() - t0:.1f}s) "
                  f"score={2.0 * n * best_m:.8f} stall={best_stall}")

    c = fast_conv(best)
    best_m = float(c.max())
    return best, best_m


def consider_candidate(a, best_score, best_seq, label, t0):
    """Validate a candidate with the real evaluator and update best state."""
    if a is None:
        return best_score, best_seq
    a = norm1(a)
    if len(a) == 0:
        return best_score, best_seq
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score - 1e-15:
        elapsed = time.monotonic() - t0
        print(f"  *** NEW BEST [{label}]: score={sc:.8f} n={len(a)} t={elapsed:.0f}s")
        return sc, a.copy()
    return best_score, best_seq


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator. Returns a list of non-negative heights."""
    t_global = time.monotonic()
    total = max(30.0, min(float(budget_s), 1800.0) - 60.0)

    best_score = float("inf")
    best_seq = None
    n_default = 2400

    parent = globals().get("height_sequence_1", None)
    if parent is not None:
        try:
            p = np.asarray(parent, dtype=np.float64)
            if p.ndim != 1:
                p = p.ravel()
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.clip(p, 0.0, 1000.0)
            s = float(p.sum())
            if p.size > 0 and s > 1e-10:
                cand = p / s
                sc = evaluate_safe(cand)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = cand.copy()
                    n_default = len(cand)
                    print(f"Parent loaded: n={n_default}, score={best_score:.8f}")
        except Exception:
            pass

    if best_seq is None:
        x = np.linspace(0.0, 1.0, n_default)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * x)
        best_seq = norm1(a)
        best_score = evaluate_safe(best_seq)
        print(f"Default initial: n={len(best_seq)}, score={best_score:.8f}")

    solver = pick_solver()
    print(f"Total time budget: {total:.0f}s, solver={solver}")

    # Phase 1: main mirror descent on the parent length.
    rem = total - (time.monotonic() - t_global)
    t1 = min(560.0, 0.34 * rem)
    if t1 > 30.0:
        a, _ = mirror_descent(best_seq, t1, verbose=True,
                              rel_temp_start=0.25, rel_temp_end=0.010,
                              seed=12345)
        best_score, best_seq = consider_candidate(
            a, best_score, best_seq, f"MD1_n{len(a)}", t_global)

    # Phase 2: LP Frank-Wolfe with line search.
    rem = total - (time.monotonic() - t_global)
    t2 = min(560.0, 0.36 * rem)
    if t2 > 60.0 and solver is not None:
        ncur = len(best_seq)
        maxc = 550 if ncur <= 2600 else 380
        a, _ = lp_frank_wolfe(best_seq, t2, solver, max_cons=maxc, verbose=True)
        best_score, best_seq = consider_candidate(
            a, best_score, best_seq, f"LP_n{len(a)}", t_global)

    # Phase 3: harder mirror polishing.
    rem = total - (time.monotonic() - t_global)
    t3 = min(180.0, 0.15 * rem)
    if t3 > 30.0:
        a, _ = mirror_descent(best_seq, t3, verbose=False,
                              rel_temp_start=0.10, rel_temp_end=0.005,
                              seed=54321)
        best_score, best_seq = consider_candidate(
            a, best_score, best_seq, f"MD2_n{len(a)}", t_global)

    # Phase 4: coordinate polish at current best length.
    rem = total - (time.monotonic() - t_global)
    t4 = min(300.0, 0.22 * rem)
    if t4 > 20.0:
        a, _ = coordinate_polish(best_seq, t4, verbose=True, seed=987)
        best_score, best_seq = consider_candidate(
            a, best_score, best_seq, f"CD_n{len(a)}", t_global)

    # Phase 5: try one or two longer sequences from the current shape.
    rem = total - (time.monotonic() - t_global)
    if rem > 180.0:
        n_base = len(best_seq)
        for n_alt in (n_base + 600, n_base + 1200):
            rem = total - (time.monotonic() - t_global)
            if rem < 120.0:
                break
            n_alt = int(min(n_alt, 4200))
            if n_alt == n_base or n_alt <= 0:
                continue
            a_alt = interp_sequence(best_seq, n_alt)
            best_score, best_seq = consider_candidate(
                a_alt, best_score, best_seq, f"interp_n{n_alt}", t_global)

            t_alt_md = min(90.0, 0.40 * rem)
            if t_alt_md > 20.0:
                a_alt, _ = mirror_descent(a_alt, t_alt_md, verbose=False,
                                          rel_temp_start=0.20, rel_temp_end=0.008,
                                          seed=777)
                best_score, best_seq = consider_candidate(
                    a_alt, best_score, best_seq, f"MD_alt_n{n_alt}", t_global)

            rem = total - (time.monotonic() - t_global)
            t_alt_cd = min(50.0, 0.50 * rem)
            if t_alt_cd > 15.0:
                a_alt, _ = coordinate_polish(a_alt, t_alt_cd, verbose=False, seed=313)
                best_score, best_seq = consider_candidate(
                    a_alt, best_score, best_seq, f"CD_alt_n{n_alt}", t_global)

    # Phase 6: final polish on the absolute best sequence.
    rem = total - (time.monotonic() - t_global)
    if rem > 30.0:
        a, _ = coordinate_polish(best_seq, min(180.0, rem - 10.0),
                                 verbose=True, seed=211)
        best_score, best_seq = consider_candidate(
            a, best_score, best_seq, f"CD_final_n{len(a)}", t_global)

    rem = total - (time.monotonic() - t_global)
    if rem > 30.0 and solver is not None:
        ncur = len(best_seq)
        maxc = 300 if ncur <= 3000 else 220
        a, _ = lp_frank_wolfe(best_seq, min(120.0, rem - 15.0),
                              solver, max_cons=maxc, verbose=False)
        best_score, best_seq = consider_candidate(
            a, best_score, best_seq, f"LP_final_n{len(a)}", t_global)

    # Final validation fallback.
    if best_seq is None or len(best_seq) == 0:
        best_seq = norm1(np.ones(2400, dtype=np.float64))
    final_score = evaluate_safe(best_seq)
    if not np.isfinite(final_score):
        best_seq = norm1(np.ones(2400, dtype=np.float64))
        final_score = evaluate_safe(best_seq)

    print(f"FINAL: score={final_score:.8f}, n={len(best_seq)}, "
          f"time={time.monotonic() - t_global:.0f}s")

    return [float(x) for x in best_seq]
