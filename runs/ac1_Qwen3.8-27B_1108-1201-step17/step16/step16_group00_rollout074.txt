```
def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([0.0])
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]
```

```
def hankel_product(weights, a):
    n = len(a)
    ar = a[::-1]
    conv_len = len(weights) + n - 1
    if conv_len <= 1:
       return np.asarray([weights[0]*a[0] if n==1 else 0.0])
    fft_size = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    A = np.fft.rfft(ar, fft_size)
    conv = np.fft.irfft(W * A, fft_size)
    start = n - 1
    return conv[start:start+n]
```

```
def mirror_step(x, g, lr):
    g = np.asarray(g, dtype=np.float64)
    if g.size != x.size: return x.copy()
    gm = float(np.mean(g))
    g = g - gm
    gs = float(np.max(np.abs(g)))
    if gs < 1e-30:
        return x.copy()
    z = np.log(np.maximum(x, 1e-300)) - float(lr) * (g / gs)
    zm = float(np.max(z))
    z = z - zm
    y = np.exp(z)
    y = np.maximum(y, 1e-300)
    s = float(y.sum())
    if s < 1e-30 or not np.isfinite(s):
        return x.copy()
    return y / s
```

```python
"""
Search for a non-negative sequence minimizing
    J = 2 * n * max(conv(a, a)) / sum(a)^2.

The program uses:
- FFT-based autocorrelation and a Hankel/FFT product for exact gradients
  of log-sum-exp approximations of the peak autocorrelation.
- Multiplicative mirror descent on the simplex, both full log-sum-exp
  smoothing and active near-peak subgradient modes.
- Length scaling/interpolation, random smooth anti-symmetric starts,
  light local coordinate polishing, and strict time management.

Only the global evaluate_sequence() is assumed available.
"""

import numpy as np
import time


def normalize_seq(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[:size]


def internal_score(a):
    a = normalize_seq(a)
    c = np.maximum(fast_autocorr(a), 0.0)
    return 2.0 * a.size * float(c.max())


def hankel_product(weights, a):
    """
    Returns g_i = sum_j weights[i+j] * a_j, for i=0..n-1.
    This is the gradient multiplier part of weighted autocorrelation gradients.
    """
    weights = np.asarray(weights, dtype=np.float64).ravel()
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    ar = a[::-1]
    conv_len = weights.size + n - 1
    if conv_len <= 1:
        if n == 1 and weights.size >= 1:
            return np.array([weights[0] * a[0]], dtype=np.float64)
        return np.array([0.0], dtype=np.float64)
    fft_size = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    A = np.fft.rfft(ar, fft_size)
    conv = np.fft.irfft(W * A, fft_size)
    start = n - 1
    return conv[start:start + n]


def smooth_weights(c, temp):
    c = np.asarray(c, dtype=np.float64)
    c = np.maximum(c, 0.0)
    cmax = float(c.max())
    if cmax <= 0.0 or temp <= 1e-300:
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    x = (c - cmax) / temp
    x = np.clip(x, -745.0, 0.0)
    w = np.exp(x)
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / s
    return w


def active_weights(c, max_count=50, margin=0.05, temp_factor=0.05):
    c = np.asarray(c, dtype=np.float64)
    c = np.maximum(c, 0.0)
    M = float(c.max())
    if M <= 0.0:
        w = np.ones(c.size, dtype=np.float64) / c.size
        idx = np.arange(c.size, dtype=int)
        return w, idx

    threshold = M * (1.0 - margin)
    idx = np.flatnonzero(c >= threshold)
    if idx.size > max_count:
        idx = np.argpartition(c, -max_count)[-max_count:]
    elif idx.size == 0:
        idx = np.array([int(np.argmax(c))], dtype=int)

    idx = np.asarray(idx, dtype=int)
    vals = c[idx]
    temp = max(M * temp_factor, 1e-300)
    x = (vals - float(vals.max())) / temp
    x = np.clip(x, -745.0, 0.0)
    w = np.exp(x)
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.ones(idx.size, dtype=np.float64) / idx.size
    else:
        w = w / s
    return w, idx


def full_weights_from_pairs(length, idx, vals):
    w = np.zeros(int(length), dtype=np.float64)
    idx = np.asarray(idx, dtype=int)
    vals = np.asarray(vals, dtype=np.float64)
    if idx.size == 0:
        return w
    w[idx] = vals
    return w


def mirror_step(x, g, lr):
    x = np.asarray(x, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    if x.size != g.size:
        return x.copy()
    g = g - float(np.mean(g))
    gs = float(np.max(np.abs(g)))
    if gs < 1e-30 or not np.isfinite(gs):
        return x.copy()
    z = np.log(np.maximum(x, 1e-300)) - float(lr) * (g / gs)
    zm = float(np.max(z))
    z = z - zm
    y = np.exp(z)
    y = np.maximum(y, 1e-300)
    s = float(y.sum())
    if s < 1e-30 or not np.isfinite(s):
        return x.copy()
    return y / s


def mirror_search(a0,
                  time_limit,
                  weights_mode="smooth",
                  start_temp_factor=0.25,
                  end_temp_factor=0.02,
                  active_max=80,
                  active_margin=0.02,
                  lr_base=0.25,
                  lr_min=0.01,
                  lr_max=2.5,
                  restart_stall=500,
                  seed=0,
                  label=""):
    t0 = time.monotonic()
    limit = max(0.0, float(time_limit))
    if limit < 1e-6:
        return normalize_seq(a0)

    x = normalize_seq(a0)
    n = x.size

    c = np.maximum(fast_autocorr(x), 0.0)
    best_m = float(c.max())
    best = x.copy()

    lr = float(lr_base)
    temp_factor = float(start_temp_factor)
    temp = max(best_m * max(temp_factor, float(end_temp_factor)), 1e-300)
    stall = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < limit:
        it += 1

        c = np.maximum(fast_autocorr(x), 0.0)
        M = float(c.max())

        if M < best_m - 1e-15:
            best_m = M
            best = x.copy()
            stall = 0

        if M <= 1e-30:
            break

        if M > best_m * 1.05 and stall > 50:
            x = best.copy()
            lr = max(lr * 0.5, float(lr_min))
            stall = 0
            continue

        if weights_mode == "smooth":
            temp = max(best_m * float(end_temp_factor), temp * 0.9997)
            w = smooth_weights(c, temp)
            g = 2.0 * hankel_product(w, x)
        else:
            w, idx = active_weights(c,
                                    max_count=int(active_max),
                                    margin=float(active_margin),
                                    temp_factor=0.05)
            wf = full_weights_from_pairs(c.size, idx, w)
            g = 2.0 * hankel_product(wf, x)

        if not np.all(np.isfinite(g)):
            break
        spread = float(np.max(g) - np.min(g))
        if spread < 1e-30:
            break

        new = mirror_step(x, g, lr)
        Mnew = float(np.max(np.maximum(fast_autocorr(new), 0.0)))

        if Mnew <= M + 1e-15:
            x = new
            if Mnew < M - 1e-15:
                lr = min(lr * 1.08, float(lr_max))
            else:
                lr = max(lr * 0.97, float(lr_min))
            stall = 0 if Mnew < best_m - 1e-15 else stall + 1
        else:
            accepted = False
            for frac in (0.5, 0.25, 0.1):
                new2 = mirror_step(x, g, lr * frac)
                M2 = float(np.max(np.maximum(fast_autocorr(new2), 0.0)))
                if M2 < M:
                    x = new2
                    lr = max(lr * frac * 0.8, float(lr_min))
                    accepted = True
                    stall = 0 if M2 < best_m - 1e-15 else stall + 1
                    break

            if not accepted:
                lr = max(lr * 0.5, float(lr_min))
                stall += 3

        if stall >= int(restart_stall):
            noise_scale = 0.10 + 0.20 * float(it % 5) / 4.0
            noise = rng.normal(0.0, max(noise_scale, 0.02), n)
            pert = best * np.exp(noise)
            pert = np.maximum(pert, 1e-12)
            ps = float(pert.sum())
            if ps > 0.0:
                pert = pert / ps
                x = 0.7 * best + 0.3 * pert
                x = np.maximum(x, 1e-12)
                xs = float(x.sum())
                if xs > 0.0:
                    x = x / xs
            lr = max(min(lr * 1.5, float(lr_base)), float(lr_min))
            temp_factor = float(start_temp_factor)
            temp = max(best_m * max(temp_factor, float(end_temp_factor)), 1e-300)
            stall = 0

        if label and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} time={elapsed:.0f}s J={2.0 * n * best_m:.8f}")

    return best


def coordinate_polish(a0, time_limit, seed=1234, label=""):
    t0 = time.monotonic()
    limit = max(0.0, float(time_limit))
    if limit < 5.0:
        return normalize_seq(a0)

    x = normalize_seq(a0)
    n = x.size
    c = np.maximum(fast_autocorr(x), 0.0)
    best_m = float(c.max())
    best = x.copy()

    stall = 0
    it = 0
    rng = np.random.default_rng(seed)
    top_count = min(12, n)

    while time.monotonic() - t0 < limit:
        it += 1

        c = np.maximum(fast_autocorr(x), 0.0)
        M = float(c.max())

        if M < best_m - 1e-15:
            best_m = M
            best = x.copy()
            stall = 0
        else:
            stall += 1

        if M <= 1e-30:
            break

        if M > best_m * 1.05 and stall > 50:
            x = best.copy()
            stall = 0
            continue

        if stall > 120:
            noise = rng.normal(0.0, 0.03, n)
            x = best * np.exp(noise)
            x = np.maximum(x, 1e-12)
            xs = float(x.sum())
            if xs > 0.0:
                x = x / xs
            stall = 0
            continue

        w, idx = active_weights(c, max_count=20, margin=0.08, temp_factor=0.10)
        wf = full_weights_from_pairs(c.size, idx, w)
        g = 2.0 * hankel_product(wf, x)

        score = g * (x + 1e-12) + rng.normal(0.0, 1e-9, n)
        if n > top_count:
            top = np.argpartition(score, -top_count)[-top_count:]
            bot = np.argpartition(score, top_count)[:top_count]
        else:
            order = np.argsort(score)
            bot = order[:top_count]
            top = order[-top_count:]

        improved = False
        for _ in range(6):
            if top.size == 0 or bot.size == 0:
                break
            src = int(top[rng.integers(top.size)])
            dst = int(bot[rng.integers(bot.size)])
            if src == dst:
                continue

            for frac in (0.04, 0.08, 0.15, 0.30):
                amt = x[src] * frac
                if amt <= 1e-300:
                    continue
                trial = x.copy()
                trial[src] -= amt
                trial[dst] += amt
                trial = np.maximum(trial, 0.0)
                ts = float(trial.sum())
                if ts < 1e-15:
                    continue
                trial = trial / ts
                Mt = float(np.max(np.maximum(fast_autocorr(trial), 0.0)))
                if Mt < M:
                    x = trial
                    if Mt < best_m:
                        best_m = Mt
                        best = trial.copy()
                    stall = 0
                    improved = True
                    break
                elif Mt < best_m:
                    best_m = Mt
                    best = trial.copy()

            if improved:
                break

        if not improved:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and x[i] > 0.0:
                frac = float(rng.uniform(0.002, 0.020))
                amt = x[i] * frac
                trial = x.copy()
                trial[i] -= amt
                trial[j] += amt
                trial = np.maximum(trial, 0.0)
                ts = float(trial.sum())
                if ts > 0.0:
                    trial = trial / ts
                    Mt = float(np.max(np.maximum(fast_autocorr(trial), 0.0)))
                    if Mt < M:
                        x = trial
                        if Mt < best_m:
                            best_m = Mt
                            best = trial.copy()
                        stall = 0

        if label and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} time={elapsed:.0f}s J={2.0 * n * best_m:.8f}")

    return best


def smooth_sequence(a, window=5):
    a = normalize_seq(a)
    n = a.size
    if n < window + 2:
        return a
    w = min(int(window), n)
    kernel = np.ones(w, dtype=np.float64) / float(w)
    b = np.convolve(a, kernel, mode="same")
    b = np.maximum(b, 0.0)
    return normalize_seq(b)


def interpolate_sequence(f, new_n):
    f = normalize_seq(f)
    old_n = f.size
    new_n = int(max(1, new_n))
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)

    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    g = np.interp(new_x, old_x, f)
    return normalize_seq(g)


def make_antisym_seq(n, amps):
    n = int(max(1, n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    y = x - 0.5
    g = np.zeros(n, dtype=np.float64)
    for m, amp in enumerate(amps):
        amp = float(amp)
        if abs(amp) > 1e-15:
            g += amp * np.sin((2 * m + 1) * np.pi * y)
    f = 1.0 + g
    f = np.maximum(f, 1e-6)
    return normalize_seq(f)


def make_random_antisym(n, seed):
    rng = np.random.default_rng(seed)
    terms = 8
    amps = rng.normal(0.0, 0.7, terms)
    amps = amps / np.sqrt(np.arange(1, terms + 1, dtype=np.float64))
    return make_antisym_seq(n, amps)


def choose_start_seq(base_seq, n, seed=0):
    n = int(max(1, n))
    candidates = []

    if base_seq is not None and base_seq.size > 0:
        candidates.append(interpolate_sequence(base_seq, n))

    candidates.append(np.ones(n, dtype=np.float64) / n)
    candidates.append(make_antisym_seq(n, [1.0, -0.40, 0.20]))
    candidates.append(make_antisym_seq(n, [0.80, 0.30, -0.20, 0.10]))
    candidates.append(make_antisym_seq(n, [0.55, -0.35, 0.25, -0.15, 0.10]))

    for s in range(5):
        candidates.append(make_random_antisym(n, seed + 100 + s))

    best = None
    best_sc = float("inf")
    for cand in candidates:
        cand = normalize_seq(cand)
        sc = internal_score(cand)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = cand

    if best is None:
        best = np.ones(n, dtype=np.float64) / n
    return best


def validate_sequence(state, seq, label):
    a = normalize_seq(seq)
    try:
        vals = [float(v) for v in a]
        sc = float(evaluate_sequence(vals))
    except Exception:
        sc = float("inf")

    state["last_eval_time"] = time.monotonic()

    if state["true_seq"] is None:
        if np.isfinite(sc):
            state["true_seq"] = a.copy()
            state["true_score"] = sc
            print(f"  validated {label}: score={sc:.8f} n={a.size}")
    elif np.isfinite(sc) and sc < state["true_score"] - 1e-12:
        state["true_seq"] = a.copy()
        state["true_score"] = sc
        print(f"  validated {label}: score={sc:.8f} n={a.size}")


def try_update_state(state, seq, label, force=False):
    if seq is None:
        return

    a = normalize_seq(seq)
    c = np.maximum(fast_autocorr(a), 0.0)
    internal = 2.0 * a.size * float(c.max())
    if not np.isfinite(internal):
        return

    now = time.monotonic()
    if state["tent_seq"] is None or internal < state["tent_internal"] - 1e-9:
        state["tent_seq"] = a.copy()
        state["tent_internal"] = float(internal)

        n = a.size
        if n <= 4096:
            threshold = 1e-7
            interval = 3.0
        elif n <= 6144:
            threshold = 2e-6
            interval = 8.0
        else:
            threshold = 5e-6
            interval = 15.0

        should_eval = (
            force or
            state["true_seq"] is None or
            internal < state["true_score"] - threshold or
            (now - state["last_eval_time"] > interval)
        )
        if should_eval:
            validate_sequence(state, a, label)
    elif state["true_seq"] is None:
        validate_sequence(state, a, label)


def run_on_length(state, n, allocated, hard_deadline):
    n = int(max(1, n))
    start = time.monotonic()
    t_avail = min(float(allocated), max(0.0, hard_deadline - start - 10.0))
    if t_avail < 45.0:
        return

    base = state["tent_seq"] if state["tent_seq"] is not None else None

    if n <= 4096:
        a0 = choose_start_seq(base, n, seed=17 + (n % 97))
        current = a0
    else:
        if base is not None:
            a0 = interpolate_sequence(base, n)
        else:
            a0 = np.ones(n, dtype=np.float64) / n
        current = a0

    try_update_state(state, current, f"init_n{n}")

    remaining = hard_deadline - time.monotonic() - 10.0
    if remaining < 30.0:
        return
    t1 = min(remaining * 0.45, 300.0)
    if t1 > 20.0:
        a1 = mirror_search(current,
                           t1,
                           weights_mode="smooth",
                           start_temp_factor=0.35,
                           end_temp_factor=0.03,
                           lr_base=0.30,
                           lr_min=0.01,
                           lr_max=2.5,
                           restart_stall=700,
                           seed=1000 + n,
                           label=f"smooth_n{n}")
        try_update_state(state, a1, f"smooth_n{n}")
        try_update_state(state, smooth_sequence(a1, 5), f"smooth_smn_n{n}")
        current = a1 if a1.size == n else interpolate_sequence(a1, n)

    remaining = hard_deadline - time.monotonic() - 10.0
    if remaining < 30.0:
        return
    t2 = min(remaining * 0.35, 260.0)
    if t2 > 20.0:
        a2 = mirror_search(current,
                           t2,
                           weights_mode="active",
                           active_max=80,
                           active_margin=0.02,
                           lr_base=0.20,
                           lr_min=0.01,
                           lr_max=2.0,
                           restart_stall=500,
                           seed=2000 + n,
                           label=f"active_n{n}")
        try_update_state(state, a2, f"active_n{n}")
        current = a2 if a2.size == n else interpolate_sequence(a2, n)

    remaining = hard_deadline - time.monotonic() - 10.0
    if remaining < 30.0:
        return
    t3 = min(remaining * 0.25, 220.0)
    if t3 > 20.0:
        a3 = mirror_search(current,
                           t3,
                           weights_mode="smooth",
                           start_temp_factor=0.08,
                           end_temp_factor=0.008,
                           lr_base=0.12,
                           lr_min=0.005,
                           lr_max=1.2,
                           restart_stall=900,
                           seed=3000 + n,
                           label=f"polish_smooth_n{n}")
        try_update_state(state, a3, f"polish_smooth_n{n}")
        try_update_state(state, smooth_sequence(a3, 3), f"polish_smn_n{n}")


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    hard_deadline = t_global + min(float(budget_s) - 75.0, 1725.0)

    state = {
        "true_seq": None,
        "true_score": float("inf"),
        "tent_seq": None,
        "tent_internal": float("inf"),
        "last_eval_time": 0.0,
    }

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.ndim == 1 and p.size > 0 and float(p.sum()) > 1e-12:
            parent = normalize_seq(p)
            try_update_state(state, parent, "parent", force=True)
    except Exception:
        parent = None

    if state["tent_seq"] is None:
        n_init = 2048
        a_init = choose_start_seq(None, n_init, seed=42)
        try_update_state(state, a_init, "init_default", force=True)

        remaining = hard_deadline - time.monotonic() - 10.0
        if remaining > 90.0:
            a_tmp = mirror_search(a_init, min(remaining * 0.5, 120.0),
                                  weights_mode="smooth",
                                  start_temp_factor=0.30,
                                  end_temp_factor=0.03,
                                  lr_base=0.30,
                                  seed=7,
                                  label="default_smooth")
            try_update_state(state, a_tmp, "default_smooth")

    if state["tent_seq"] is None:
        state["tent_seq"] = np.ones(2048, dtype=np.float64) / 2048.0
        state["tent_internal"] = internal_score(state["tent_seq"])

    print(f"Initial: true_score={state['true_score']:.8f}, "
          f"tent_internal={state['tent_internal']:.8f}, "
          f"tent_n={state['tent_seq'].size}")

    current_n = state["tent_seq"].size
    candidates = [2048, 3200, 4096, 6144, 8192]
    if current_n not in candidates:
        candidates.append(current_n)
    candidates = sorted(set(candidates))

    order = [x for x in candidates if x >= current_n]
    order += [x for x in candidates if x < current_n]

    previous = -1
    for n in order:
        if n == previous:
            continue
        previous = n

        remaining = hard_deadline - time.monotonic() - 20.0
        if remaining < 80.0:
            break

        if n <= 2048:
            alloc = 300.0
        elif n <= 3200:
            alloc = 380.0
        elif n <= 4096:
            alloc = 340.0
        elif n <= 6144:
            alloc = 300.0
        else:
            alloc = 260.0

        print(f"\n=== Stage n={n}, budget approx={min(alloc, remaining):.0f}s ===")
        run_on_length(state, n, alloc, hard_deadline)

    # Final polish on the best internal sequence.
    if state["tent_seq"] is not None:
        a_best = state["tent_seq"]
        n_best = a_best.size

        remaining = hard_deadline - time.monotonic() - 20.0
        if remaining > 120.0:
            tfin = min(remaining * 0.42, 240.0)
            if tfin > 30.0:
                a_fin = mirror_search(a_best,
                                      tfin,
                                      weights_mode="active",
                                      active_max=120,
                                      active_margin=0.01,
                                      lr_base=0.10,
                                      lr_min=0.005,
                                      lr_max=1.0,
                                      restart_stall=1200,
                                      seed=4242,
                                      label="final_active")
                try_update_state(state, a_fin, "final_active")
                try_update_state(state, smooth_sequence(a_fin, 3), "final_active_sm")

        remaining = hard_deadline - time.monotonic() - 20.0
        if remaining > 70.0 and n_best <= 6000:
            tcoord = min(remaining * 0.55, 160.0)
            if tcoord > 30.0:
                a_coord = coordinate_polish(state["tent_seq"], tcoord, seed=88, label="final_coord")
                try_update_state(state, a_coord, "final_coord")
                try_update_state(state, smooth_sequence(a_coord, 3), "final_coord_sm")

    # Force final validation of the best tentative sequence.
    if state["tent_seq"] is not None:
        try_update_state(state, state["tent_seq"], "final_tent", force=True)

    if state["true_seq"] is None and state["tent_seq"] is not None:
        try_update_state(state, state["tent_seq"], "final_fallback", force=True)

    final_seq = state["true_seq"]
    if final_seq is None:
        final_seq = state["tent_seq"]

    if final_seq is None:
        final_seq = np.ones(2048, dtype=np.float64) / 2048.0

    final_seq = normalize_seq(final_seq)

    try:
        final_score = evaluate_sequence([float(v) for v in final_seq])
        if np.isfinite(final_score):
            print(f"FINAL returned score={final_score:.8f}, n={final_seq.size}")
        else:
            print("FINAL evaluation returned non-finite; returning valid normalized sequence.")
    except Exception:
        print("FINAL validation could not be executed.")

    return [float(v) for v in final_seq]
```
