        active_peaks = np.argmax(ac)
        if np.isscalar(active_peaks):
            active_peaks = np.array([active_peaks])
    # Sort by value, take top n_active
    order = np.argsort(ac[active_peaks])[::-1]
    top = active_peaks[order[:min(n_active, len(active_peaks))]]
    # Weight by value
    vals = ac[top]
    w = vals ** 3  # Cubic weighting to emphasize the largest
    w = w / w.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, top):
        grad += wi * hard_subgradient(y, int(k))
    return grad
