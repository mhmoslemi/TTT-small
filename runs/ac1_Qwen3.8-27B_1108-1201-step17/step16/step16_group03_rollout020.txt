def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_to_length(a, new_n):
    """Interpolate sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss

    a = norm_nonneg(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = restart_interval
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute convolution and find top peaks
        ac = fast_conv(y)
        M_curr = ac.max()
        
        # Multi-peak subgradient
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta_power
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                j = k_int - i
                if 0 <= j < n:
                    grad[i] += 2.0 * wi * y[j]
        
        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Track best
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Periodic restart
        if stall > max_stall:
            restart = best_a * (1.0 + rng.normal(0, noise_scale, n))
            restart = norm_nonneg(restart)
            x = restart
            x_prev = restart
            stall = 0
            
            # Adaptive: increase noise if stuck for too long
            if stall == 0 and gamma > 1.0:
                gamma *= 0.9
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      MD t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={top_k}, stall={stall}")
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit=100.0, solver=None, verbose=False):
    """LP-based Frank-Wolfe with exact line search."""
    if not HAS_CVXPY or solver is None:
        return a0, fast_conv(a0).max()
    
    n = len(a0)
    a = norm_nonneg(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Build T matrix: (T @ b)[k] = (y * b)[k]
        # T has shape (2n-1, n), T[k, j] = y[k-j] if 0<=k-j<n else 0
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y
        
        b_var = cp.Variable(n, nonneg=True)
        try:
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M_curr])
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            break
        direction = b_val / b_sum
        
        # Line search
        alphas = [2.0 / (t + 2.0), 0.5, 0.3, 0.2, 0.1, 0.05]
        best_alpha = alphas[0]
        best_eval = fast_conv((1 - best_alpha) * y + best_alpha * direction).max()
        
        for alpha in alphas[1:]:
            trial = (1.0 - alpha) * y + alpha * direction
            trial = norm_nonneg(trial)
            trial_m = fast_conv(trial).max()
            if trial_m < best_eval:
                best_eval = trial_m
                best_alpha = alpha
        
        y = (1.0 - best_alpha) * y + best_alpha * direction
        y = norm_nonneg(y)
        
        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def coordinate_polish(a0, time_limit=60.0, seed=42, verbose=False):
    """Coordinate descent with mass transfer moves."""
    n = len(a0)
    a = norm_nonneg(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:min(5, len(ac))]
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            contrib_max = contrib.max()
            if contrib_max < 1e-20:
                continue
            
            src = int(idx[np.argmax(contrib)])
            
            # Find best destination
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:3]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            n_cand = min(len(candidates), 100)
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]
            
            best_dst = -1
            best_score = float('inf')
            for dst in chosen:
                if dst == src:
                    continue
                if 0 <= k_int - dst < n:
                    sens = a[dst] + a[k_int - dst]
                else:
                    sens = a[dst]
                if sens < best_score:
                    best_score = sens
                    best_dst = int(dst)
            
            if best_dst < 0:
                continue
            
            found = False
            for frac in [0.001, 0.003, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = norm_nonneg(a_trial)
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if not found and stall > 200:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = norm_nonneg(a)
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a = best_a.copy()
        
        if not improved:
            stall += 1
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      Coord ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m

def simulated_annealing(a0, time_limit=200.0, seed=42, verbose=False):
    """Simulated annealing with structured moves."""
    n = len(a0)
    a = norm_nonneg(a0)
    best_a = a.copy()
    best_score = objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    T_init = best_score * 0.05
    T_min = best_score * 0.001
    current_score = best_score
    
    n_iter = 0
    while time.monotonic() - t0 < time_limit:
        n_iter += 1
        # Temperature schedule
        frac = min(1.0, n_iter / 100000.0)
        T = T_init * (T_min / T_init) ** frac
        
        # Generate a move
        move_type = rng.integers(3)
        a_new = a.copy()
        
        if move_type == 0:
            # Kick: transfer mass between two random positions
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                amt = a[i] * rng.uniform(0.01, 0.1)
                a_new[i] -= amt
                a_new[j] += amt
        elif move_type == 1:
            # Block shift: shift a contiguous block
            block_len = int(rng.integers(max(2, n // 50), max(3, n // 10)))
            start = int(rng.integers(0, max(1, n - block_len)))
            shift = int(rng.integers(-block_len, block_len + 1))
            if shift != 0:
                block = a_new[start:start + block_len].copy()
                a_new[start:start + block_len] = 0
                new_start = max(0, min(n - block_len, start + shift))
                a_new[new_start:new_start + block_len] += block
        else:
            # Peak-targeted squeeze: reduce mass at peak contributors, spread to random positions
            ac = fast_conv(a_new)
            k_max = int(np.argmax(ac))
            i_start = max(0, k_max - n + 1)
            i_end = min(n, k_max + 1)
            idx = np.arange(i_start, i_end)
            contrib = a_new[idx] * a_new[k_max - idx]
            # Find the pair with max contribution
            pair_idx = np.argmax(contrib)
            i = int(idx[pair_idx])
            j = int(k_max - idx[pair_idx])
            # Reduce both and spread
            amt = min(a_new[i], a_new[j]) * rng.uniform(0.05, 0.2)
            a_new[i] -= amt / 2
            a_new[j] -= amt / 2
            # Spread to random positions
            spread = rng.integers(3, 10)
            targets = rng.integers(0, n, spread)
            a_new[targets] += amt / spread
        
        a_new = norm_nonneg(a_new)
        new_score = objective(a_new)
        
        # Metropolis acceptance
        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            a = a_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        
        if verbose and n_iter % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SA iter={n_iter} ({elapsed:.1f}s): best={best_score:.6f}, T={T:.6f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start initialization (parent, cosine, geometric, two-lobe, random)
2. Fast Nesterov mirror descent with multi-peak subgradient
3. LP-based Frank-Wolfe for exploitation
4. Simulated annealing with structured moves for global exploration
5. Coordinate descent for final refinement
6. Adaptive multi-length search
"""

import numpy as np
import time

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def objective(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    c = fast_conv(a)
    return 2.0 * len(a) * c.max()


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def get_solver():
    if not HAS_CVXPY:
        return None
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS


def interpolate_to_length(a, new_n):
    """Interpolate sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss


def init_cosine(n):
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    return norm_nonneg(a)


def init_geometric(n, r=0.97):
    a = r ** np.arange(n, dtype=np.float64)
    return norm_nonneg(a)


def init_two_lobe(n, frac=0.35):
    a = np.zeros(n)
    w = max(2, int(n * frac))
    x1 = np.linspace(0, 1, w)
    a[:w] = np.sin(np.pi * x1) ** 2
    a[-w:] = np.sin(np.pi * x1) ** 2
    return norm_nonneg(a)


def init_random(n, seed=0):
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    return norm_nonneg(a)


def init_triangle(n):
    """Triangle (Bartlett) window."""
    x = np.linspace(-1, 1, n)
    a = np.maximum(0, 1 - np.abs(x))
    return norm_nonneg(a)


def mirror_descent(a0, time_limit=100.0, c0=5.0, top_k=3, beta_power=2.0,
                   restart_interval=3000, noise_scale=0.01, seed=42, verbose=False):
    """Nesterov accelerated mirror descent on the simplex with multi-peak subgradient."""
    n = len(a0)
    a = norm_nonneg(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)

            y = y / sy

        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta_power
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                j = k_int - i
                if 0 <= j < n:
                    grad[i] += 2.0 * wi * y[j]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_interval:
            restart = best_a * (1.0 + rng.normal(0, noise_scale, n))
            restart = norm_nonneg(restart)
            x = restart
            x_prev = restart
            stall = 0
            gamma = max(gamma * 0.85, 1.0)
            noise_scale = min(noise_scale * 1.3, 0.1)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0

def lp_frank_wolfe(a0, time_limit=100.0, solver=None, verbose=False):
    """LP-based Frank-Wolfe with adaptive line search."""
    if not HAS_CVXPY or solver is None:
        return a0, fast_conv(a0).max()

    n = len(a0)
    a = norm_nonneg(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    staliters = 0
    max_stal_iters = 50

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Build T matrix
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        b_var = cp.Variable(n, nonneg=True)
        try:
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M_curr])
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            break
        direction = b_val / b_sum

        # Check if direction is same as current (converged)
        cos_sim = np.dot(y, direction) / (np.linalg.norm(y) * np.linalg.norm(direction) + 1e-15)
        if cos_sim > 1 - 1e-6:
            staliters += 1
        else:
            staliters = 0

        if staliters > max_stal_iters:
            break

        # Line search with golden section
        best_alpha = 0.5
        best_eval = fast_conv((1 - best_alpha) * y + best_alpha * direction).max()

        for alpha in [0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.9]:
            trial = (1.0 - alpha) * y + alpha * direction
            trial = norm_nonneg(trial)
            trial_m = fast_conv(trial).max()
            if trial_m < best_eval:
                best_eval = trial_m
                best_alpha = alpha

        y = (1.0 - best_alpha) * y + best_alpha * direction
        y = norm_nonneg(y)

        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP iter={t} ({elapsed:.1f}s): J={J:.6f}, alpha={best_alpha:.3f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit=60.0, seed=42, verbose=False):
    """Coordinate descent with mass transfer moves targeting top peaks."""
    n = len(a0)
    a = norm_nonneg(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        top_peaks = np.argsort(ac)[::-1][:min(8, len(ac))]
        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])

            # Build exclusion zone around all top peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            n_cand = min(len(candidates), 150)
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]

            best_dst = -1
            best_score = float('inf')
            for dst in chosen:
                dst = int(dst)
                if dst == src:
                    continue
                # Estimate how much this destination affects
                impact = a[dst]
                if 0 <= k_int - dst < n:
                    impact += a[k_int - dst] * 0.5
                if impact < best_score:
                    best_score = impact
                    best_dst = dst

            if best_dst < 0:
                continue

            found = False
            for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = norm_nonneg(a_trial)
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found:
                # Try random perturbation if stuck
                if stall > max_stall // 2:
                    for _ in range(10):
                        i = int(rng.integers(n))
                        j = int(rng.integers(n))
                        if i == j or a[i] < 1e-10:
                            continue
                        frac = rng.uniform(0.002, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = norm_nonneg(a)
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            found = True
                            break
                        else:
                            a = best_a.copy()
                break

        if not improved:
            stall += 1

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"      Coord ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def simulated_annealing(a0, time_limit=200.0, seed=42, verbose=False):
    """Simulated annealing with structured moves for global exploration."""
    n = len(a0)
    a = norm_nonneg(a0)
    best_a = a.copy()
    best_score = objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    T_init = best_score * 0.03
    T_min = best_score * 0.0005
    current_score = best_score
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        # Batch of 10 evaluations
        batch = 10
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            total_iters += 1
            frac = min(1.0, total_iters / 200000.0)
            T = T_init * (T_min / T_init) ** frac

            move_type = rng.integers(4)
            a_new = a.copy()

            if move_type == 0:
                # Kick: transfer mass between two positions
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j:
                    amt = a[i] * rng.uniform(0.005, 0.08)
                    a_new[i] -= amt
                    a_new[j] += amt
            elif move_type == 1:
                # Block shift
                block_len = int(rng.integers(max(2, n // 100), max(3, n // 20)))
                start = int(rng.integers(0, max(1, n - block_len)))
                shift = int(rng.integers(-block_len, block_len + 1))
                if shift != 0:
                    block = a_new[start:start + block_len].copy()
                    a_new[start:start + block_len] = 0.0
                    new_start = max(0, min(n - block_len, start + shift))
                    a_new[new_start:new_start + block_len] += block
            elif move_type == 2:
                # Peak-targeted squeeze
                ac = fast_conv(a)
                top_k = np.argsort(ac)[::-1][:3]
                k_picked = top_k[int(rng.integers(len(top_k)))]
                k_int = int(k_picked)
                i_start = max(0, k_int - n + 1)
                i_end = min(n, k_int + 1)
                idx = np.arange(i_start, i_end)
                contrib = a[idx] * a[k_int - idx]
                pair_idx = np.argmax(contrib)
                i = int(idx[pair_idx])
                j = int(k_int - idx[pair_idx])
                amt = min(a[i], a[j]) * rng.uniform(0.03, 0.15)
                a_new[i] -= amt / 2
                a_new[j] -= amt / 2
                spread = int(rng.integers(5, 20))
                targets = rng.integers(0, n, spread)
                a_new[targets] += amt / spread
            else:
                # Gaussian perturbation on a random window
                w = int(rng.integers(n // 20, n // 5))
                start = int(rng.integers(0, max(1, n - w)))
                noise = rng.normal(0, 0.01, w) * a[start:start + w]
                a_new[start:start + w] += noise

            a_new = norm_nonneg(a_new)
            new_score = objective(a_new)

            delta = new_score - current_score
            if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
                a = a_new
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()

        if verbose and total_iters % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SA iter={total_iters} ({elapsed:.1f}s): best={best_score:.6f}")

    return best_a, best_score


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with adaptive time allocation."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)
    solver = get_solver() if HAS_CVXPY else None
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    n_base = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            n_base = len(p)
            parent = norm_nonneg(p)
            p_score = objective(parent)
            print(f"Parent: n={n_base}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = init_cosine(n_base)
        best_seq = parent.copy()
        best_score = objective(parent)
        print(f"No parent found, using cosine init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_nonneg(a)
        sc = objective(a)
        if sc < best_score - 1e-10:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n0 = len(best_seq)

    # === Phase 1: Multi-start mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Multi-start MD (t={time.monotonic()-t_global:.0f}s) ===")
    inits = [
        ("parent", best_seq),
        ("cosine", init_cosine(n0)),
        ("geometric", init_geometric(n0, 0.97)),
        ("two_lobe", init_two_lobe(n0, 0.35)),
        ("triangle", init_triangle(n0)),
    ]
    t_per = min(remaining * 0.25 / len(inits), 60.0)
    for name, init_a in inits:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t1 = min(remaining * 0.15, t_per)
        if t1 < 15:
            break
        print(f"  MD from {name}...")
        a_out, _ = mirror_descent(init_a, t1, c0=5.0, top_k=3, seed=hash(name) % 10000, verbose=False)
        try_update(a_out, f"md_{name}_n{n0}")

    # === Phase 2: LP Frank-Wolfe on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP FW n={n_cur} (t={time.monotonic()-t_global:.0f}s) ===")
    if remaining > 100 and solver is not None:
        t2 = min(remaining * 0.30, 400.0)
        print(f"  Allocating {t2:.0f}s for LP")
        a_out, _ = lp_frank_wolfe(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp1_n{n_cur}")
    elif remaining > 60:
        # Fallback: longer MD
        t2 = min(remaining * 0.30, 400.0)
        a_out, _ = mirror_descent(best_seq, t2, c0=6.0, top_k=5, beta_power=3.0, seed=77, verbose=True)
        try_update(a_out, f"md_long_n{n_cur}")

    # === Phase 3: Simulated annealing for exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: SA exploration n={n_cur} (t={time.monotonic()-t_global:.0f}s) ===")
    if remaining > 100:
        t3 = min(remaining * 0.25, 400.0)
        print(f"  Allocating {t3:.0f}s for SA")
        a_out, _ = simulated_annealing(best_seq, t3, seed=123, verbose=True)
        try_update(a_out, f"sa_n{n_cur}")

    # === Phase 4: LP Frank-Wolfe again on improved best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP FW round 2 n={n_cur} (t={time.monotonic()-t_global:.0f}s) ===")
    if remaining > 80 and solver is not None:
        t4 = min(remaining * 0.25, 350.0)
        print(f"  Allocating {t4:.0f}s for LP")
        a_out, _ = lp_frank_wolfe(best_seq, t4, solver, verbose=True)
        try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Multi-length (t={time.monotonic()-t_global:.0f}s) ===")
    if remaining > 150:
        for delta in [-200, 200, -400, 400]:
            n_try = n_cur + delta
            if n_try < 1500 or n_try > 3500:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            print(f"  Trying n={n_try}...")
            a_interp = interpolate_to_length(best_seq, n_try)
            t5 = min(remaining * 0.20, 120.0)
            if t5 < 20:
                break
            a_out, _ = mirror_descent(a_interp, t5, c0=4.0, top_k=3, seed=55 + n_try % 100, verbose=False)
            try_update(a_out, f"md_n{n_try}")
            # Quick coord polish
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t5b = min(remaining * 0.08, 40.0)
                if t5b > 10:
                    a_out2, _ = coordinate_polish(a_out, t5b, seed=n_try % 100, verbose=False)
                    try_update(a_out2, f"coord_n{n_try}")

    # === Phase 6: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Final LP n={n_cur} (t={time.monotonic()-t_global:.0f}s) ===")
    if remaining > 60 and solver is not None and n_cur <= 3000:
        t6 = min(remaining * 0.40, 300.0)
        print(f"  Allocating {t6:.0f}s for final LP")
        a_out, _ = lp_frank_wolfe(best_seq, t6, solver, verbose=True)
        try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final coord n={n_cur} (t={time.monotonic()-t_global:.0f}s) ===")
    if remaining > 30:
        t7 = min(remaining * 0.70, 120.0)
        if t7 > 10:
            a_out, _ = coordinate_polish(best_seq, t7, seed=314, verbose=True)
            try_update(a_out, f"coord_final_n{n_cur}")

    # === Phase 8: One more MD refinement if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        t8 = min(remaining * 0.80, 100.0)
        if t8 > 15:
            print(f"\n=== Phase 8: Final MD n={n_cur} ===")
            a_out, _ = mirror_descent(best_seq, t8, c0=8.0, top_k=1, beta_power=1.0,
                                       noise_scale=0.003, seed=999, verbose=False)
            try_update(a_out, f"md_final_n{n_cur}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
