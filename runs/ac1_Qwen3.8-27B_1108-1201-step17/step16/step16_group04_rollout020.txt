        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.4 * a + 0.6 * a_new
    return normalize(a)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def solve_lp_direction(y, M, active_lags, solver):
    """
    Solve LP: max sum(b) s.t. (y*b)[k] <= M for k in active_lags, b >= 0.
    Returns normalized b/sum(b) or None.
    """
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)

    constraints = []
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_idx = np.arange(j_start, j_end + 1)
        y_coeff = y[k_int - j_idx]
        constraints.append(y_coeff @ b_var[j_idx] <= M)

    if len(constraints) == 0:
        return np.ones(n) / n

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def compute_objective(a):
    """Compute 2*n*max_conv/(sum)^2 for non-negative a."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)


def lp_fw_enhanced(a0, time_limit, solver, top_k=400, verbose=True):
    """
    Enhanced LP Frank-Wolfe with:
    - Top-K constraint selection (larger K)
    - Backtracking line search
    - Nesterov acceleration
    - Periodic constraint set refresh
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_obj = compute_objective(a)
    best_m = fast_autocorr(a).max() / a.sum()**2
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    ac_count = 0
    improve_count = 0
    no_improve = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        # Compute autocorrelation
        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Track best
        curr_obj = 2.0 * n * M_curr / (y.sum()**2)
        if curr_obj < best_obj - 1e-14:
            best_obj = curr_obj
            best_a = y.copy()
            best_m = M_curr / y.sum()**2
            no_improve = 0
        else:
            no_improve += 1

        # Restart if stalled
        if no_improve > 5000:
            no_improve = 0
            rng = np.random.default_rng(t)
            pert = rng.normal(0, 0.003, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-12)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new
            y = x.copy()
            continue

        # Select active constraints: top-K by value
        active_lags = np.argsort(ac)[::-1][:top_k]

        # Solve LP
        direction = solve_lp_direction(y, M_curr, active_lags, solver)
        if direction is None:
            continue

        # Backtracking line search
        alpha = 0.5
        best_alpha = 0.0
        best_trial_m = float('inf')
        for _ in range(10):
            x_trial = (1.0 - alpha) * y + alpha * direction
            x_trial = np.maximum(x_trial, 0.0)
            sx = x_trial.sum()
            if sx < 1e-15:
                alpha /= 2
                continue
            x_trial = x_trial / sx
            ac_trial = fast_autocorr(x_trial)
            m_trial = ac_trial.max()
            if m_trial < M_curr * (1.0 - 0.01 * alpha):
                best_alpha = alpha
                best_trial_m = m_trial
                break
            best_trial_m = min(best_trial_m, m_trial)
            best_alpha = max(best_alpha, alpha * 0.5)
            alpha /= 2

        if best_alpha < 1e-6:
            # Use standard FW step size
            best_alpha = 2.0 / (t + 2.0)

        x_new = (1.0 - best_alpha) * y + best_alpha * direction
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx

        x_prev = y
        x = x_new

        # Check improvement at x
        mx = fast_autocorr(x).max()
        obj_x = 2.0 * n * mx
        if obj_x < best_obj:
            best_obj = obj_x
            best_a = x.copy()
            best_m = mx
            no_improve = 0
        else:
            no_improve += 1

        ac_count += 1
        if verbose and ac_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW t={t} ({elapsed:.1f}s): obj={best_obj:.6f}, M={best_m:.8f}")

    return best_a, best_obj


def pair_coordinate_descent(a0, time_limit, verbose=True, rng_seed=42):
    """
    Coordinate descent with pair moves:
    - Identify peak lags
    - For each peak, find the contributing pair (i, j) with i+j=k
    - Move mass from the larger contributor to a better position
    - Also try pair scaling moves
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_obj = compute_objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500
    move_count = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        # Top peak lags
        top_k = np.argsort(ac)[::-1][:min(30, len(ac))]

        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            k_int = int(k)
            # Find contributing pairs
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            pairs = idx * a[k_int - idx]
            total = pairs.sum()
            if total < 1e-20:
                continue

            # Focus on top contributing pairs
            pair_order = np.argsort(pairs)[::-1][:min(10, len(pairs))]
            for pi in pair_order:
                i = idx[pi]
                j = k_int - i
                if i == j:
                    # Self-pair: reduce a[i] by moving to elsewhere
                    src = i
                else:
                    # Move from the larger of a[i], a[j]
                    src = i if a[i] >= a[j] else j

                if a[src] < 1e-12:
                    continue

                # Find good destination: a position that doesn't contribute much to top peaks
                dest_scores = np.zeros(n)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    # Contribution of position d to lag k2: a[d]*a[k2-d] + a[k2-d]*a[d-k2]
                    c1 = k2i - np.arange(n)
                    valid1 = (c1 >= 0) & (c1 < n)
                    contrib1 = np.zeros(n)
                    contrib1[valid1] = a[c1[valid1]]
                    # Also the other direction
                    c2 = np.arange(n) - k2i
                    valid2 = (c2 >= 0) & (c2 < n)
                    contrib2 = np.zeros(n)
                    contrib2[valid2] = a[c2[valid2]]
                    dest_scores += contrib1 + contrib2

                # Exclude positions already in peak contributing sets
                for k2 in top_k[:5]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    dest_scores[s2:e2+1] += 0.5

                # Pick destination with lowest score
                dest_scores[src] = np.inf
                candidates = np.argsort(dest_scores)[:min(50, n)]
                candidates = [c for c in candidates if c != src]
                if not candidates:
                    continue
                dst = candidates[0]

                # Try different move sizes
                found = False
                for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < M_curr * (1 - 1e-10):
                        a = a_trial
                        M_curr = M_trial
                        improved = True
                        found = True
                        move_count += 1
                        break

                if found:
                    break  # Move to next peak lag

            if improved:
                obj = 2.0 * n * M_curr
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                stall = 0
                break  # Restart peak scan

        if not improved:
            stall += 1
            # Perturbation to escape
            if stall % 50 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.01)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / max(a.sum(), 1e-15)
                        M_new = fast_autocorr(a).max()
                        obj_new = 2.0 * n * M_new
                        if obj_new < best_obj:
                            best_obj = obj_new
                            best_a = a.copy()
                            stall = 0
                            break
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / max(a.sum(), 1e-15)

        if verbose and move_count % 100 == 0 and move_count > 0:
            elapsed = time.monotonic() - t0
            print(f"    Pair-CD moves={move_count} ({elapsed:.1f}s): obj={best_obj:.6f}")

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Pair-CD done ({elapsed:.1f}s): obj={best_obj:.6f}, moves={move_count}")

    return best_a, best_obj


def peak_shaving(a0, time_limit, verbose=True, rng_seed=99):
    """
    Direct peak-shaving: repeatedly target the single worst lag and
    perform targeted mass redistribution to reduce it.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_obj = compute_objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        peak_lag = int(np.argmax(ac))
        M_curr = ac[peak_lag]

        # Identify top contributors to this peak
        js = max(0, peak_lag - n + 1)
        je = min(n - 1, peak_lag)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_lag - idx]
        top_contrib_order = np.argsort(contrib)[::-1][:min(5, len(idx))]

        improved = False
        for ci in top_contrib_order:
            i = idx[ci]
            j = peak_lag - i

            # Try moving mass from the larger element to a random low-impact position
            src = i if a[i] >= a[j] else j
            if a[src] < 1e-12:
                continue

            # Score all positions by their impact on top 5 peaks
            top_peaks = np.argsort(ac)[::-1][:5]
            impact = np.zeros(n)
            for kp in top_peaks:
                kpi = int(kp)
                for pos in range(n):
                    other1 = kpi - pos
                    if 0 <= other1 < n:
                        impact[pos] += a[other1]
                    other2 = pos - kpi
                    if 0 <= other2 < n and other2 != other1:
                        impact[pos] += a[other2]

            impact[src] = np.inf
            # Sample from low-impact positions
            low_impact = np.argsort(impact)[:min(100, n)]
            low_impact = [p for p in low_impact if p != src]
            if not low_impact:
                continue
            dst = low_impact[rng.integers(len(low_impact))]

            # Try move
            for frac in [0.002, 0.005, 0.01, 0.02]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < M_curr - 1e-15:
                    a = a_trial
                    improved = True
                    break

            if improved:
                obj = 2.0 * n * fast_autocorr(a).max()
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                stall = 0
                break

        if not improved:
            stall += 1
            # Random perturbation
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.001, 0.005)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / max(a.sum(), 1e-15)
                    if 2.0 * n * fast_autocorr(a).max() < best_obj:
                        best_obj = 2.0 * n * fast_autocorr(a).max()
                        best_a = a.copy()
                        stall = 0
                        break
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0)
                        a = a / max(a.sum(), 1e-15)

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave stall={stall} ({elapsed:.1f}s): obj={best_obj:.6f}")

    return best_a, best_obj


def nesterov_multiplicative(a0, time_limit, top_k=60, beta=4.0, c0=3.0, verbose=True):
    """
    Nesterov-accelerated multiplicative update (mirror descent) approach.
    Uses the gradient of the smooth max approximation (top-K weighted).
    """
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_obj = compute_objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)

        # Compute weighted subgradient from top-K peaks
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = y
        x = a_new

        M_new = fast_autocorr(x).max()
        obj_new = 2.0 * n * M_new
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 2000:
            stall_count = 0
            pert = rng.normal(0, 0.004, n)
            x_new = best_a * (1.0 + pert)
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            x_prev = x_new

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSG t={t} ({elapsed:.1f}s): obj={best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize(parent)
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 0: Spectral equalization ===
    elapsed = time.monotonic() - t_global
    print(f"\n=== Phase 0: Spectral Equalization (t={elapsed:.0f}s) ===")
    a_spec = spectral_equalize(best_seq, n_iter=40)
    try_update(a_spec, "spectral")

    # === Phase 1: Enhanced LP-FW on base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 1: LP-FW n={n_cur}, budget={remaining:.0f}s ===")
    t1 = min(remaining * 0.35, 600.0)
    if t1 > 60:
        a_in = normalize(best_seq)
        a_out, _ = lp_fw_enhanced(a_in, t1, solver, top_k=400, verbose=True)
        try_update(a_out, f"lpfw1_n{n_cur}")

    # === Phase 2: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Peak Shaving n={n_cur}, budget={remaining:.0f}s ===")
    t2 = min(remaining * 0.12, 200.0)
    if t2 > 30:
        a_in = normalize(best_seq)
        a_out, _ = peak_shaving(a_in, t2, verbose=True, rng_seed=99)
        try_update(a_out, f"peakshave_n{n_cur}")

    # === Phase 3: Pair coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Pair CD n={n_cur}, budget={remaining:.0f}s ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_in = normalize(best_seq)
        a_out, _ = pair_coordinate_descent(a_in, t3, verbose=True, rng_seed=42)
        try_update(a_out, f"paircd_n{n_cur}")

    # === Phase 4: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    n_long = min(n_best + 600, 4000)
    if n_long != n_best and remaining > 200:
        print(f"\n=== Phase 4: Long n={n_long}, budget={remaining:.0f}s ===")
        a_long = interpolate_sequence(best_seq, n_long)
        a_long = spectral_equalize(a_long, n_iter=20)
        try_update(a_long, f"interp_n{n_long}")

        t4a = min(remaining * 0.40, 400.0)
        if t4a > 60:
            a_out, _ = lp_fw_enhanced(a_long, t4a, solver, top_k=300, verbose=True)
            try_update(a_out, f"lpfw_long_n{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t4b = min(remaining * 0.3, 120.0)
            if t4b > 20:
                a_out, _ = pair_coordinate_descent(normalize(best_seq), t4b, verbose=False, rng_seed=123)
                try_update(a_out, f"paircd_long_n{len(best_seq)}")

    # === Phase 5: LP-FW again on best (refined) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: LP-FW refined n={n_cur}, budget={remaining:.0f}s ===")
    t5 = min(remaining * 0.25, 300.0)
    if t5 > 40:
        a_in = normalize(best_seq)
        a_out, _ = lp_fw_enhanced(a_in, t5, solver, top_k=500, verbose=True)
        try_update(a_out, f"lpfw2_n{n_cur}")

    # === Phase 6: Nesterov multiplicative ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: NSG n={n_cur}, budget={remaining:.0f}s ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 30:
        a_in = normalize(best_seq)
        a_out, _ = nesterov_multiplicative(a_in, t6, top_k=80, beta=5.0, c0=4.0, verbose=True)
        try_update(a_out, f"nsg_n{n_cur}")

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final CD n={n_cur}, budget={remaining:.0f}s ===")
    t7 = min(remaining * 0.30, 150.0)
    if t7 > 20:
        a_in = normalize(best_seq)
        a_out, _ = pair_coordinate_descent(a_in, t7, verbose=False, rng_seed=777)
        try_update(a_out, f"final_cd_n{n_cur}")

    # === Phase 8: Last peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 8: Final PeakShave n={n_cur} ===")
        t8 = min(remaining * 0.7, 80.0)
        if t8 > 20:
            a_in = normalize(best_seq)
            a_out, _ = peak_shaving(a_in, t8, verbose=False, rng_seed=555)
            try_update(a_out, f"peakshave_final_n{n_cur}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
