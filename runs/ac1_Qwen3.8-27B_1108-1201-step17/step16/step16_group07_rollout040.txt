def spectral_factorization(C_target, fft_size, n, phase_ref=None):
    """
    Given a target convolution C_target (length >= 2n-1), find a (length n) such that conv(a,a) ≈ C_target.
    Uses the relationship: DFT(a*a) = DFT(a)^2.
    """
    C_padded = np.zeros(fft_size)
    C_padded[:len(C_target)] = C_target
    G = np.fft.rfft(C_padded)
    
    # Take square root
    # G[k] = F[k]^2 where F = DFT(a)
    # F[k] = sqrt(G[k]) with appropriate branch
    F = np.sqrt(G + 0j)  # complex square root, principal branch
    
    # Ensure F[0] is positive real
    if F[0].real < 0:
        F = -F
    
    if phase_ref is not None:
        # Use phase from reference to resolve ambiguity
        # F_new = |F| * exp(i * phase(phase_ref))
        magnitudes = np.abs(F)
        phases = np.angle(phase_ref)
        F = magnitudes * np.exp(1j * phases)
        # Ensure DC is positive
        if F[0].real < 0:
            F[0] = np.abs(F[0])
    
    a_new = np.fft.irfft(F, fft_size)[:n]
    return a_new

def flatten_convolution(C, mode='cap', alpha=0.9):
    """
    Modify convolution C to be flatter.
    Returns modified C with the same sum.

    C = C.copy()
    total_sum = C.sum()
    
    if mode == 'cap':
        threshold = alpha * C.max()
        excess = np.maximum(C - threshold, 0)
        C[C > threshold] = threshold
        deficit_total = total_sum - C.sum()
        mask = C < threshold
        if mask.sum() > 0:
            C[mask] += deficit_total * (threshold - C[mask]) / (threshold - C[mask]).sum()
    
    elif mode == 'soft':
        # Apply soft ceiling
        peak = C.max()
        if peak < 1e-15:
            return C
        ratio = C / peak
        penalty = 1.0 / (1.0 + 10.0 * (ratio - alpha)**2)
        C = C * penalty
        C = C * total_sum / C.sum()
    
    elif mode == 'blend':
        flat = np.ones_like(C) * total_sum / len(C)
        C = (1 - alpha) * C + alpha * flat
    
    elif mode == 'power':
        # Apply power compression
        peak = C.max()
        if peak < 1e-15:
            return C
        C = peak * (C / peak) ** alpha
        C = C * total_sum / C.sum()
    
    return np.maximum(C, 0) * total_sum / np.maximum(C.sum(), 1e-15)

def fast_convolution(a):

    n = len(a)

def spectral_factor(C_target, fft_size, n, phase_ref=None):
    """Given target convolution, find a such that conv(a,a) ~ C_target via root of DFT."""
    C_pad = np.zeros(fft_size)
    C_pad[:len(C_target)] = C_target
    G = np.fft.rfft(C_pad)
    F = np.sqrt(G + 0j)
    if F[0].real < 0:
        F = -F
    if phase_ref is not None:
        mags = np.abs(F)

        if F[0].real < 0:
            F[0] = np.abs(F[0])
        if fft_size % 2 == 0:
            F[-1] = np.abs(F[-1])
    a_new = np.fft.irfft(F, fft_size)[:n]
    return a_new


def flatten_convolution(C, mode='cap', alpha=0.92, n_peaks=3):
    """Modify convolution to be flatter, preserving sum."""
    C = C.copy()
    total = C.sum()
    if total < 1e-30:
        return C
    if mode == 'cap':
        thresh = alpha * C.max()
        excess = np.maximum(C - thresh, 0).sum()
        C = np.minimum(C, thresh)
        deficit_mask = C < thresh
        if deficit_mask.sum() > 0 and total > 0:
            deficit_weights = (thresh - C[deficit_mask])
            if deficit_weights.sum() > 0:
                C[deficit_mask] += excess * deficit_weights / deficit_weights.sum()
    elif mode == 'power':
        peak = C.max()
        if peak > 1e-30:
            C = peak * (C / peak) ** alpha
    elif mode == 'blend':
        flat = np.ones_like(C) * total / len(C)
        C = (1.0 - alpha) * C + alpha * flat

def coordinate_descent(a0, time_limit, verbose=False, seed=0, n_candidates=16):
    """Fast coordinate descent with multi-candidate 1D optimization."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    conv = fast_convolution(a)
    best_a = a.copy()
    best_score = conv.max()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0
    max_no_improve = 3

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        any_improve = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            a_j = a[j]
            S_others = norm - a_j

            # Determine affected range
            k_start = j
            k_end = min(j + n, 2 * n - 1)
            length = k_end - k_start
            if length <= 0:
                continue

            # Coefficients: a[k-j] for k in [k_start, k_end)
            coeff = a[0:length]
            conv_base = conv[k_start:k_end].copy()

            # Max outside the affected range
            if k_start > 0:
                max_left = np.max(conv[:k_start]) if k_start > 0 else 0.0
            else:
                max_left = 0.0
            if k_end < 2 * n - 1:
                max_right = np.max(conv[k_end:]) if k_end < 2 * n - 1 else 0.0
            else:
                max_right = 0.0
            max_outside = max(max_left, max_right)

            # Generate candidate x values
            typical = norm / n
            cands = [0.0, a_j]
            for f in [0.1, 0.3, 0.5, 0.7, 1.3, 1.7, 2.5]:
                cands.append(a_j * f)
            for f in [0.2, 0.5, 1.0, 2.0, 3.5]:
                cands.append(typical * f)
            # Add a smart candidate: x that would reduce the peak at position k_start+j
            # (the peak most likely affected)
            peak_pos = np.argmax(conv)
            if k_start <= peak_pos < k_end:
                # conv[peak_pos] + 2*(x-a_j)*a[peak_pos-j] = max_outside
                # x = a_j + (max_outside - conv[peak_pos]) / (2*a[peak_pos-j])
                apj = a[peak_pos - j]
                if apj > 1e-15:
                    x_smart = a_j + (max_outside - conv[peak_pos]) / (2.0 * apj)
                    if x_smart > 0:
                        cands.append(x_smart)
            # Candidate from quadratic at 2j
            if 2 * j >= k_start and 2 * j < k_end:
                # conv[2j] + (x-a_j)^2 + 2*(x-a_j)*a[j] ... actually x^2 - a_j^2
                # conv[2j] + x^2 - a_j^2 = max_outside
                # x^2 = max_outside - conv[2j] + a_j^2
                rhs = max_outside - conv[2 * j] + a_j * a_j
                if rhs > 0:
                    cands.append(np.sqrt(rhs))

            cands = [max(0.0, c) for c in cands]
            cands = list(set(cands))
            if len(cands) > n_candidates:
                cands = sorted(cands)[:n_candidates]
            xs = np.array(cands)

            if len(xs) == 0:
                continue

            # Compute new conv max for each candidate (vectorized)
            deltas = 2.0 * (xs[:, None] - a_j) * coeff[None, :]
            if 2 * j >= k_start and 2 * j < k_end:
                idx_2j = 2 * j - k_start
                deltas[:, idx_2j] += (xs - a_j) ** 2

            conv_new_range = conv_base[None, :] + deltas
            max_new_range = np.max(conv_new_range, axis=1)
            max_full = np.maximum(max_new_range, max_outside)
            objectives = 2.0 * n * max_full / (S_others + xs) ** 2

            best_idx = np.argmin(objectives)
            best_x = xs[best_idx]

            if abs(best_x - a_j) > 1e-12:
                # Update convolution incrementally
                delta = best_x - a_j
                conv[k_start:k_end] += 2.0 * delta * coeff
                if 2 * j >= k_start and 2 * j < k_end:
                    conv[2 * j] += delta * delta
                a[j] = best_x
                any_improve = True

                # Check score
                curr_max = conv.max()
                if curr_max < best_score - 1e-13:
                    best_score = curr_max
                    best_a = a.copy()
                else:
                    # Revert
                    conv[k_start:k_end] -= 2.0 * delta * coeff
                    if 2 * j >= k_start and 2 * j < k_end:
                        conv[2 * j] -= delta * delta
                    a[j] = a_j

        if not any_improve:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)

    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

        if b is None:
            if t % 5 == 0:
                active_set = np.arange(2 * n - 1)
                b = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.9)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        score_y = fast_convolution(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        if stall > 60:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall = 0

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPN iter={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Using LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver fallback: ECOS")
    return cp.ECOS


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False


def perturb_and_reoptimize(best_seq, t_budget, solver, rng):
    """Perturb and run a short GS+CD cycle."""
    n = len(best_seq)
    scale = 0.02 + 0.02 * rng.random()
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.3)
    perturbed = best_seq * noise
    s = perturbed.sum()
    if s < 1e-15:
        return best_seq
    perturbed = perturbed / s
    norm = np.sqrt(2.0 * n)
    perturbed = perturbed * norm / perturbed.sum()

    remaining = t_budget
    t_gs = remaining * 0.4
    t_cd = remaining * 0.5
    a_out, _ = spectral_gs(perturbed, t_gs, verbose=False)
    remaining2 = remaining - (time.monotonic() - (time.monotonic() - t_gs))
    a_out, _ = coordinate_descent(a_out, t_cd, verbose=False)
    return a_out
