    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)

        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:

        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass

The problem is: minimize max_k (a*a)[k] s.t. a >= 0, sum(a) = c.

This is equivalent to: find the minimum M such that there exists a >= 0 with sum(a) = c and (a*a)[k] <= M for all k.

- For a given M, check if the feasible set is non-empty
- The check is: minimize 0 s.t. a >= 0, sum(a) = c, (a*a)[k] <= M for all k

1. Start with some M (e.g., current max autocorrelation)
2. Check if M is achievable via LP
3. If not achievable, increase M
4. If achievable, decrease M
5. The b from the LP gives us the next iterate for a

This is exactly what the Nesterov LP does, but without the bisection. The Nesterov approach implicitly searches for the right M through the iteration.

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)

def normalize_to(a, n):
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return np.maximum(a, 0.0) * norm / s


def subgradient_at_k(a, k, n):
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        idx = np.arange(lo, hi)
        grad[lo:hi] += a[k - idx]
    return grad

    if b_val.sum() < 1e-15:

    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    conv_size = 2 * n - 1

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 35

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        if active_set is None or t % 25 == 0:
            progress = min(1.0, t / 600.0)
            tol = 0.05 * (1.0 - progress) + 0.001 * progress
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            n_top = max(8, n // 50)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            max_size = int(1.5 * n)
            if len(active_set) > max_size:
                conv_at_active = conv_y[active_set]
                keep = np.argsort(conv_at_active)[-max_size:]
                active_set = active_set[keep]

        b = _lp_solve(y, M_curr, active_set, solver)
        if b is None:
            if t % 4 == 0:
                fallback_set = np.argsort(conv_y)[-min(3 * n, conv_size):]
                b = _lp_solve(y, M_curr, fallback_set, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.94)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        if t % 40 == 0:
            blend = 0.6 * best_a + 0.4 * a
            blend = np.maximum(blend, 0.0)
            sb = blend.sum()
            if sb > 1e-15:
                blend = blend * norm / sb
                score_b = fast_autocorr(blend).max()
                if score_b < best_score:
                    best_score = score_b
                    best_a = blend.copy()
                    stall_count = 0

        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and t % 400 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def smooth_max_opt(a0, time_limit, verbose=False, seed=7):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    alpha = 8.0
    step = 0.8 / n ** 0.3
    patience = 0
    rng = np.random.default_rng(seed)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > 400:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        shifted = alpha * (conv - M_curr)
        exp_vals = np.exp(shifted)
        weights = exp_vals / exp_vals.sum()

        threshold_frac = 0.85
        significant = np.where(conv >= M_curr * threshold_frac)[0]
        if len(significant) > 80:
            top_idx = np.argsort(conv[significant])[-80:]
            significant = significant[top_idx]

        w_sig = weights[significant]
        w_sum = w_sig.sum()
        if w_sum < 1e-15:
            w_sig = np.ones(len(significant)) / len(significant)
        else:
            w_sig = w_sig / w_sum

        grad = np.zeros(n)
        for w, k in zip(w_sig, significant):
            grad += w * subgradient_at_k(a, int(k), n)

        grad -= np.mean(grad)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        update = m_hat / (np.sqrt(v_hat) + 1e-8)

        a_trial = a - step * update
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            patience += 1
            step *= 0.95
            continue
        a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-15:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.005, 5.0)
            alpha = min(alpha * 1.015, 50.0)
        else:
            patience += 1
            step *= 0.97

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Smooth t={t} ({elapsed:.0f}s): best={best_score:.10f}, alpha={alpha:.1f}, step={step:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    patience = 0
    step_scale = 0.4

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > 300:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()
        median_conv = np.median(conv)

        n_peaks = max(3, min(8, n // 150))
        peak_indices = np.argsort(conv)[-n_peaks:]

        total_grad = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            excess = conv[k] - median_conv
            if excess > 0:
                total_grad += excess * subgradient_at_k(a, k, n)

        gnorm = np.linalg.norm(total_grad)
        if gnorm < 1e-15:
            break

        total_grad -= np.mean(total_grad)
        gnorm2 = np.linalg.norm(total_grad)
        if gnorm2 < 1e-15:
            break

        step = step_scale * best_score / (n * gnorm2 + 1e-10)
        a_trial = a - step * total_grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            patience += 1
            continue
        a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score - 1e-15:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step_scale = min(step_scale * 1.02, 1.0)
        else:
            patience += 1
            step_scale *= 0.9

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coord_polish(a0, time_limit, verbose=False, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    max_no_improve = 5

    while time.monotonic() - t0 < time_limit:
        if no_improve >= max_no_improve:
            break
        no_improve = 0
        improved = False
        conv = fast_autocorr(a)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            aj = a[j]
            s_others = norm - aj
            lo = max(0, j)
            hi = min(size, j + n)
            if lo >= hi:
                continue

            typical = norm / n
            spread = (np.max(a) - np.min(a)) * 0.15
            candidates = [0.0, aj * 0.4, aj * 0.7, aj * 1.3, aj * 1.7,
                          typical * 0.6, typical * 1.4,
                          max(0.0, aj + spread * rng.standard_normal()),
                          max(0.0, aj - spread * rng.standard_normal())]
            candidates = sorted(set([max(0.0, min(c, norm * 0.4)) for c in candidates]))

            best_x = aj
            best_obj = conv.max()

            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                delta = x - aj
                new_conv = conv.copy()
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta
                new_sum = s_others + x
                if new_sum < 1e-15:
                    continue
                scale = norm / new_sum
                new_max = new_conv.max() * scale * scale
                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        conv = fast_autocorr(a)
                        improved = True
                    else:
                        a = best_a.copy()
                        conv = fast_autocorr(a)

        if not improved:
            no_improve += 1

        if verbose and no_improve == 0 and (t0 > 0):
            elapsed = time.monotonic() - t0
            print(f"    Coord ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def _try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def _resize_sequence(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    resized = np.interp(new_x, old_x, a)
    resized = np.maximum(resized, 0.0)
    s = resized.sum()
    if s < 1e-15:
        resized = np.ones(new_n) / new_n
    else:
        resized = resized / s
    return resized
