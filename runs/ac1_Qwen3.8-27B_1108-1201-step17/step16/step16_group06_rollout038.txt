def fast_conv(a, b):
    """Compute full linear convolution using FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def fast_autocorr(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

def build_conv_matrix_sparse(a):
    """Build sparse matrix S where (S @ b)[k] = conv(a, b)[k].
    S[k,j] = a[k-j] if 0 <= k-j < n, else 0.
    Column j has non-zeros at rows j..j+n-1 with values a[0..n-1].
    """
    n = len(a)
    num_rows = 2 * n - 1
    col_indices = np.repeat(np.arange(n), n)
    row_indices = np.concatenate([np.arange(j, j + n) for j in range(n)])
    values = np.tile(a, n)
    S = sp.csr_matrix((values, (row_indices, col_indices)), shape=(num_rows, n))
    return S


def lp_best_response(a, solver_name):
    """Solve LP: min t s.t. conv(a,b)[k] <= t for all k, sum(b)=1, b>=0.
    Returns (b, t_opt) or (None, None) on failure."""
    import cvxpy as cp
    n = len(a)
    S = build_conv_matrix_sparse(a)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [
        cp.sum(b_var) == 1,
        S @ b_var <= t_var
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver_name, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None, None
    b_val = b_val / b_sum
    t_val = float(t_var.value)
    return b_val, t_val

def mirror_descent_step(a, k_peak, eta):
    """Multiplicative weights update targeting peak reduction.
    Reduces a[j] proportionally to its contribution to the peak."""
    n = len(a)
    weights = np.zeros(n)
    for j in range(n):
        d1 = k_peak - j
        d2 = j - k_peak
        if 0 <= d1 < n:
            weights[j] += a[d1]
        if 0 <= d2 < n:
            weights[j] += a[d2]
    log_a = np.log(np.maximum(a, 1e-30)) - eta * weights
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2000
        rng = np.random.default_rng(0)
        best_seq = rng.random(n_base)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting n={n0}, score={best_score:.6f}")

    # === Phase 1: Fast mirror descent warmup (20s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 1: Mirror descent warmup n={n0} ===")
        t1 = min(remaining * 0.03, 20.0)
        a_out, _ = mirror_descent_search(norm_sum1(best_seq), t1, verbose=False)
        try_update(a_out, "md_warmup")

    # === Phase 2: Subgradient FW (30s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 2: Subgradient FW n={n0} ===")
        t2 = min(remaining * 0.05, 30.0)
        a_out, _ = subgradient_frank_wolfe(norm_sum1(best_seq), t2, verbose=False)
        try_update(a_out, "subgd_n0")

    # === Phase 3: CORRECTED LP-FW (main workhorse, ~45% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3: Corrected LP-FW n={n_cur} (MAIN) ===")
        t3 = min(remaining * 0.45, 700.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t3, solver, verbose=True)
        try_update(a_out, f"lp_correct_n{n_cur}")

    # === Phase 4: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        print(f"\n=== Phase 4: Coord polish n={n_cur} ===")
        t4 = min(remaining * 0.08, 60.0)
        a_out, _ = peak_targeted_coordinate_descent(norm_sum1(best_seq), t4, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_cur}")

    # === Phase 5: More LP-FW after polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        print(f"\n=== Phase 5: LP-FW round 2 n={n_cur} ===")
        t5 = min(remaining * 0.25, 350.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t5, solver, verbose=True)
        try_update(a_out, f"lp_r2_n{n_cur}")

    # === Phase 6: Try slightly larger n ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_target = n_cur + 300
    if n_target > n_cur and remaining > 100:
        print(f"\n=== Phase 6: Expand to n={n_target} ===")
        a_long = interpolate_sequence(best_seq, n_target)
        t6a = min(remaining * 0.05, 30.0)
        a_out, _ = subgradient_frank_wolfe(norm_sum1(a_long), t6a, verbose=False)
        try_update(a_out, f"subgd_n{n_target}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t6b = min(remaining * 0.2, 200.0)
            a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t6b, solver, verbose=True)
            try_update(a_out, f"lp_n{n_target}")

    # === Phase 7: Multi-n exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 7: Multi-n exploration ===")
        n_best = len(best_seq)
        test_ns = []
        for delta in [200, 400, -200, 600]:
            nt = n_best + delta
            if 1000 <= nt <= 4000 and nt != n_best:
                test_ns.append(nt)
        for nt in test_ns[:3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 50:
                break
            print(f"  Trying n={nt}...")
            a_inter = interpolate_sequence(best_seq, nt)
            t_per = min(remaining * 0.2, 80.0)
            a_out, _ = lp_frank_wolfe(norm_sum1(a_inter), t_per, solver, verbose=False)
            try_update(a_out, f"multin_n{nt}")

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Final coord polish n={n_best} ===")
        t8 = min(remaining * 0.3, 80.0)
        a_out, _ = peak_targeted_coordinate_descent(norm_sum1(best_seq), t8, verbose=True, rng_seed=777)
        try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 9: Final LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 9: Final LP n={n_best} ===")
        t9 = min(remaining * 0.7, 150.0)
        a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t9, solver, verbose=True)
        try_update(a_out, f"lp_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
