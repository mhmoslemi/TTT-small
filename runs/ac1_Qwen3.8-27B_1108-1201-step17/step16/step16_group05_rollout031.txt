2. Direct optimization with a solver
3. Iterative methods (like the previous program)

So `conv(x,x)[k] = sum_{i: 0<=i<n, 0<=k-i<n} x_i * x_{k-i}`.

This can be written as `x^T Q_k x` where `Q_k` is a symmetric "anti-diagonal" matrix (with 1s on the anti-diagonal corresponding to `i+j=k`).

So the problem is: minimize `t` subject to `x^T Q_k x <= t` for all `k`, `x >= 0`, `sum(x) = 1`.

This is a problem with quadratic constraints. The feasible set is non-convex in general (each constraint `x^T Q_k x <= t` defines a non-convex set since `Q_k` is indefinite in general... actually `Q_k` is a permutation-like matrix, so it's symmetric with eigenvalues ±1, making `x^T Q_k x` indefinite).

For solving this, we could use:
1. **Sequential Quadratic Programming (SQP)**: Linearize the constraints iteratively.
2. **Penalty method**: Add penalty terms for constraint violations.
3. **ADMM**: Split the problem.
4. **Projected gradient with constraint projection**: Hard to project onto the feasible set.
5. **Bisection + semidefinite relaxation**: Relax `x^T Q_k x <= t` to a SDP constraint. SDP relaxation: `[[t, x^T], [x, I]] >= 0`... this doesn't quite work because `x^T Q_k x` involves a specific quadratic form.

def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(y, top_k=None):
    """
    Build the (2n-1) x n matrix T such that T @ b = conv(y, b).
    T[k, j] = y[k-j] if 0 <= k-j < n, else 0.
    Only keep top_k rows if specified.
    """
    n = len(y)
    num_rows = 2 * n - 1
    if top_k is not None and top_k < num_rows:
        # Select most important constraints (largest current conv values)
        ac_y = fast_autocorr(y)
        idx = np.argsort(ac_y)[::-1][:top_k]
        T = np.zeros((top_k, n))
        for new_i, k in enumerate(idx):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            for j in range(j_start, j_end):
                T[new_i, j] = y[k_int - j]
        return T, idx
    else:
        T = np.zeros((num_rows, n))
        for j in range(n):
            T[j:j + n, j] = y
        return T, np.arange(num_rows)


def lp_frank_wolfe(a0, time_limit, solver, top_k_start=200, top_k_max=800,
                    verbose=True, momentum_decay=0.98):
    """
    LP-based Frank-Wolfe with Nesterov acceleration and adaptive active set.
    At each step, solve: max 1^T b s.t. T @ b <= M, b >= 0
    where T is the convolution matrix for current y, M = max conv(y,y).
    """
    n = len(a0)
    y = norm_sum1(a0).copy()
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    x_mom = y.copy()
    current_top_k = top_k_start
    stall = 0
    max_stall = 50

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Adaptive active set: increase top_k as we stall
        if stall > max_stall // 3:
            current_top_k = min(current_top_k + 50, top_k_max, 2 * n - 1)

        T, active_idx = build_conv_matrix(y, top_k=current_top_k)
        T_rows = T.shape[0]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            stall += 1
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            stall += 1
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            stall += 1
            continue
        g_dir = b_val / b_sum

        # Nesterov FW update

        y_new = (1.0 - alpha) * y + alpha * g_dir

        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        x_mom = y_new + beta_t * (y_new - y)
        x_mom = np.maximum(x_mom, 0.0)
        sx = x_mom.sum()
        if sx < 1e-15:
            x_mom = y_new.copy()
        else:
            x_mom = x_mom / sx

        y = y_new

        improved = False
        for candidate in [y, x_mom]:
            mc = fast_autocorr(candidate).max()
            if mc < best_m - 1e-12:

                improved = True

def smoothed_gradient_step(a, beta, top_k=200):
    """Compute gradient of smoothed max autocorrelation."""
    n = len(a)
    ac = fast_autocorr(a)
    k_eff = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_eff]
    top_vals = ac[top_idx]
    shifted = top_vals - top_vals[0]
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / exp_vals.sum()

    grad = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def accelerated_gradient(a0, time_limit, beta_start=30.0, beta_end=5.0,
                         top_k=200, c0=2.0, verbose=False, seed=42):
    """Accelerated projected gradient on simplex with smoothed max objective."""
    n = len(a0)
    x = norm_sum1(a0).copy()
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 2000

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / 15000.0)
        cur_beta = beta_start + (beta_end - beta_start) * frac

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y)

        g = smoothed_gradient_step(y, cur_beta, top_k=top_k)
        eta = c0 / (np.sqrt(t) + 1.0)
        x_new = project_simplex(y - eta * g)

        x_prev = x
        x = x_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            max_stall = int(max_stall * 1.4)
            noise = rng.normal(0, 0.003, n) * best_a
            x = np.maximum(best_a + noise, 1e-15)
            x = x / x.sum()
            x_prev = x.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      AG t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, seed=42):
    """
    Coordinate descent: for each coordinate i, find optimal a_i given others fixed.
    The objective is max_k conv[k], where conv depends quadratically on a_i.
    For fixed a_{-i}, conv[k] is quadratic in a_i (for k involving i).
    We do a 1D minimax: minimize max_k (c_k + d_k * a_i + e_k * a_i^2)
    subject to a_i >= 0, and rescale to maintain sum=1.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200

    orders = np.arange(n)
    np.random.seed(seed)
    np.random.shuffle(orders)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved_iter = False

        for i in orders:
            if time.monotonic() - t0 > time_limit:
                break

            if a[i] < 1e-12:
                # Try adding a small amount
                for frac in [0.001, 0.005, 0.01]:
                    a_trial = a.copy()
                    # Take from a random other coordinate
                    j = int(rng.integers(n))
                    if j == i or a[j] < 1e-10:
                        continue
                    a_trial[j] -= a[j] * frac
                    a_trial[i] += a[j] * frac
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial /= s_t
                    M_t = fast_autocorr(a_trial).max()
                    if M_t < best_m - 1e-15:
                        a = a_trial
                        best_m = M_t
                        best_a = a.copy()
                        improved_iter = True
                        break
                continue

            # Current value is a[i]. Try different values.
            # The constraint sum=1 means if we change a[i] to v, we need to
            # redistribute (a[i]-v) among other coordinates.
            # Simple approach: scale all others proportionally.
            others = a.copy()
            others[i] = 0
            s_others = others.sum()
            if s_others < 1e-15:
                continue

            # Try a few values of a[i]
            for frac in [0.5, 0.7, 0.8, 1.0, 1.2, 1.3, 1.5]:
                v = a[i] * frac
                if v < 0 or v >= 1.0:
                    continue
                remaining = 1.0 - v
                if remaining < 1e-10:
                    continue
                a_trial = others * (remaining / s_others)
                a_trial[i] = v
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if abs(s_t - 1) > 1e-6:
                    a_trial /= s_t
                M_t = fast_autocorr(a_trial).max()
                if M_t < best_m - 1e-15:
                    a = a_trial
                    best_m = M_t
                    best_a = a.copy()
                    improved_iter = True
                    break

        if improved_iter:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.001, 0.005)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a /= a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0)
                    a /= a.sum()

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        J = 2.0 * n * best_m
        print(f"      CD done: J={J:.6f}")
    return best_a, best_m


def get_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Ramp (linear increase)
    ramp = (x + 1)
    inits.append(ramp / ramp.sum())

    # Quadratic
    quad = (x + 1) ** 2
    inits.append(quad / quad.sum())

    # Inverse ramp
    inv_ramp = (n - x)
    inits.append(inv_ramp / inv_ramp.sum())

    # Sine
    sine = np.sin(np.pi * (x + 0.5) / n) ** 2
    inits.append(sine / sine.sum())

    # Two-hump (asymmetric)
    hump = np.exp(-20 * (x / n - 0.3) ** 2) + 0.7 * np.exp(-15 * (x / n - 0.8) ** 2)
    inits.append(hump / hump.sum())

    # Power law (heavy tail)
    plaw = (x + 1) ** 0.5
    inits.append(plaw / plaw.sum())

    # Oscillating with growth
    osc = (x + 1) * (1 + 0.3 * np.sin(2 * np.pi * x / 8))
    osc = np.maximum(osc, 0.01)
    inits.append(osc / osc.sum())

    # Random with smoothing
    r = rng.random(n)
    kernel = np.ones(5) / 5
    r_smooth = np.convolve(r, kernel, mode='same')
    r_smooth = np.maximum(r_smooth, 0.01)
    inits.append(r_smooth / r_smooth.sum())

    return inits


def get_parent_sequence():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p.copy()
    except Exception:
        pass
    return None


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    1. Generate multiple initializations (ramp, parent, parametric)

    5. Return best found

        return time_budget - (time.monotonic() - t_global)

    def try_update(a, label):

        if a is None:
            return

        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    # === Step 1: Generate initializations ===
    parent = get_parent_sequence()
