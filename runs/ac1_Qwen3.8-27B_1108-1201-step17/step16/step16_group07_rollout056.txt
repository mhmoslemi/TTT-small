def lse_gradient_fast(a, beta, n_top=None):
    """Faster version using only top peaks."""
    n = len(a)
    conv = fast_autocorr(a)
    
    if n_top is not None and n_top < 2*n-1:
        top_idx = np.argsort(conv)[-n_top:]
        top_idx = np.sort(top_idx)
    else:
        top_idx = np.arange(2*n-1)
    
    # Softmax weights on top peaks only (approximation)
    shifted = beta * (conv[top_idx] - conv.max())
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        if w < 1e-15:
            continue
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        js = np.arange(j_lo, j_hi + 1)
        ks = k - js
        grad[js] += w * a[ks]
        mask = js != ks
        if mask.any():
            grad[ks[mask]] += w * a[js[mask]]
    
    return grad, conv

def nesterov_lse_optimize(a0, time_limit, verbose=False, seed=0):

    else:
        a = a * norm / s0

    y = a.copy()  # Nesterov momentum variable

    stall_count = 0
    n_iters = 0

        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient at y (Nesterov look-ahead)

        if gn < 1e-15:

        grad = grad / gn
        
        # Update velocity (momentum)
        v = momentum * v - lr * grad
        y = y + v
        y = np.maximum(y, 0.0)
        
        # Normalize
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y * norm / sy
        
        # Update a (combine with previous)
        a = y.copy()
        n_iters += 1
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive beta: increase as we converge
        if t % 20 == 0:
            beta = min(beta * beta_growth, 200.0)
        
        # Adaptive learning rate: decrease if stalling
        if stall_count > 50:
            lr *= 0.9
            beta = max(beta * 0.8, 1.0)  # Reduce smoothing to be more precise

                v = np.zeros(n)

        if verbose and n_iters % 500 == 0:

    return best_a, best_score

        # Scale to prevent overflow

        if max_grad < 1e-15:
            break
        scaled_grad = grad_centered / max_grad
        
        a = a * np.exp(-eta * scaled_grad)
        a = np.maximum(a, 1e-10)
        
        # Normalize
        s = a.sum()
        a = a * norm / s
        
        # Track best
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            patience = 0
            eta = min(eta * 1.01, 3.0)
        else:
            patience += 1
            eta *= 0.99
            if eta < 1e-6:
                break
        
        # Adapt beta
        if t % 50 == 0:
            beta = min(beta * 1.01, 100.0)
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW iter={t} ({elapsed:.0f}s): best={best_score:.10f}, eta={eta:.6f}")
    
    return best_a, best_score


def analytical_cd(a0, time_limit, verbose=False, seed=0):
    """
    Coordinate descent where for each coordinate we solve the 1D subproblem.
    For coordinate j, the objective as a function of a[j] (with sum fixed) is:
    max_k (quadratic in a[j])
    We minimize this by checking breakpoints.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    no_improve_passes = 0
    max_no_improve = 3
    n_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        conv = fast_autocorr(a)
        
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            aj = a[j]
            if aj < 1e-12:
                continue
            
            # If we change a[j] to x and renormalize (divide by sum),
            # the new sum is (norm - aj + x), so scale = norm / (norm - aj + x)
            # New a'[j] = x * scale, a'[i] = a[i] * scale for i != j
            
            # New conv[k] = conv[k] + 2*(x-aj)*a[k-j]*scale^2 + ... 
            # This is getting complicated. Let's use a simpler approach:
            # Try a few candidate values and evaluate.
            
            s_others = norm - aj
            # Try: reduce, keep, increase
            candidates = []
            if aj > 0:
                candidates = [aj * 0.1, aj * 0.3, aj * 0.5, aj * 0.7, 
                             aj, aj * 1.3, aj * 1.7, aj * 2.5, aj * 4.0, 0.0]
            # Add typical value
            typical = norm / n
            candidates.extend([typical * 0.3, typical, typical * 2, typical * 5])
            candidates = [max(0, min(c, norm * 0.4)) for c in candidates]
            candidates = list(set([round(c, 10) for c in candidates]))
            
            best_x = aj
            best_obj = conv.max()
            
            for x in candidates:
                if abs(x - aj) < 1e-10:
                    continue
                new_sum = s_others + x
                if new_sum < 1e-12:
                    continue
                scale = norm / new_sum
                
                # Update conv
                # conv_new[k] = scale^2 * (conv[k] + 2*(x - aj)*a_at_kj + (x^2 - aj^2) * delta_{k,2j})
                # where a_at_kj = a[k-j] if 0<=k-j<n, and also a[j+k] if 0<=j+k<n... 
                # Actually: conv[k] = sum_i a[i]*a[k-i]
                # After changing a[j] to x: 
                #   conv_new[k] = conv[k] - a[j]*a[k-j] - a[j]*a[k-j] + a[j]^2*delta... 
                # No, let's just do it directly:
                # The terms involving a[j] in conv[k] are:
                #   a[j]*a[k-j] (when i=j) + a[k-j]*a[j] (when k-i=j, i.e., i=k-j)
                #   = 2*a[j]*a[k-j] (if j != k-j) or a[j]^2 (if j = k-j)
                
                # Simpler: just compute the change
                # conv_new[k] = conv[k] + (x-a[j])*(a[k-j] + a[k-j]) + ... 
                # This is error-prone. Let's just use the FFT for each candidate.
                
                new_a = a.copy()
                new_a[j] = x
                new_a = new_a * scale
                new_conv = fast_autocorr(new_a)
                new_max = new_conv.max()
                
                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x
            
            if best_x != aj:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-12:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-12:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
        
        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= max_no_improve:
                break
        else:
            no_improve_passes = 0
        
        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score


def spectral_flatten(a0, time_limit, verbose=False):
    """
    Try to flatten the spectrum |A|^2 while maintaining non-negativity.
    A flat spectrum corresponds to flat autocorrelation (except at lag 0).
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2*n-2).bit_length()
    
    lr = 0.01
    patience = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > 200:
            break
        
        A = rfft(a, fft_size)
        power = np.abs(A)**2
        power_mean = power.mean()
        
        # Gradient of (max autocorrelation) w.r.t. a can be approximated by
        # the "error" in the spectrum: (|A|^2 - target) mapped back
        # target: we want |A|^2 to be as flat as possible
        # The autocorrelation conv[k] = IDFT(|A|^2)[k] (approximately, with padding)
        # To reduce max autocorrelation, we want to reduce the peaks in the spectrum
        
        # Compute "spectral error"
        spectral_error = power - power_mean
        # Map back to time domain
        time_signal = irfft(spectral_error, fft_size)[:n]
        
        # Project to non-negative: reduce where time_signal is positive (increases peaks)
        # This is inexact but provides a direction
        a_trial = a - lr * time_signal
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-12:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial
        
        score = fast_autocorr(a_trial).max()
        if score < best_score:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            patience = 0
            lr = min(lr * 1.02, 1.0)
        else:
            patience += 1
            lr *= 0.95
            if lr < 1e-8:
                break
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral iter={t} ({elapsed:.0f}s): best={best_score:.10f}, lr={lr:.6f}")
    
    return best_a, best_score


def peak_reduction(a0, time_limit, verbose=False, seed=0):
    """
    Direct peak reduction: identify peaks and their contributing pairs,
    then reduce the contributing elements.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    alpha = 0.1  # Reduction strength
    patience = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > 500:
            break
        
        conv = fast_autocorr(a)
        M = conv.max()
        
        # Find top peaks
        n_peaks = max(3, min(10, n // 100))
        peak_indices = np.argsort(conv)[-n_peaks:]
        
        # Compute reduction direction
        reduction = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            # conv[k] = sum_j a[j]*a[k-j]
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                kj = k - j
                contrib = a[j] * a[kj]
                # Reduce both j and kj proportionally
                reduction[j] += contrib
                if kj != j:
                    reduction[kj] += contrib
        
        if reduction.max() < 1e-15:
            break
        
        # Apply reduction
        a_trial = a - alpha * reduction / reduction.max()
        
        # Add to elements that don't contribute to peaks
        # Find "safe" elements
        peak_contribution = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                peak_contribution[j] += a[j] * a[k - j]
        
        safe_mask = peak_contribution < 0.1 * peak_contribution.max()
        if safe_mask.sum() > 0:
            total_reduction = (a - a_trial).sum()
            if total_reduction > 0:
                safe_sums = a[safe_mask].sum()
                if safe_sums > 1e-12:
                    a_trial[safe_mask] += total_reduction * a[safe_mask] / safe_sums
        
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-12:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial
        
        score = fast_autocorr(a_trial).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            patience = 0
            alpha = min(alpha * 1.01, 0.5)
        else:
            patience += 1
            alpha *= 0.99
            if alpha < 1e-6:
                break
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRed iter={t} ({elapsed:.0f}s): best={best_score:.10f}, alpha={alpha:.6f}")
    
    return best_a, best_score

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_to(a, n):
    """Normalize a to have sum = sqrt(2n)."""
    norm = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s


def lse_gradient_fast(a, beta, n_top=None):
    """
    Compute gradient of log-sum-exp smoothed max autocorrelation.
    Uses only top peaks for efficiency.
    """
    n = len(a)
    conv = fast_autocorr(a)
    conv_size = 2 * n - 1

    if n_top is not None and n_top < conv_size:
        top_idx = np.argsort(conv)[-n_top:]
        top_idx = np.sort(top_idx)
    else:
        top_idx = np.arange(conv_size)

    shifted = beta * (conv[top_idx] - conv.max())
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n), conv
    weights /= wsum

    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        if w < 1e-12:
            continue
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        js = np.arange(j_lo, j_hi + 1)
        ks = k - js
        grad[js] += w * a[ks]
        mask = js != ks
        if mask.any():
            grad[ks[mask]] += w * a[js[mask]]

    return grad, conv


def nesterov_lse_optimize(a0, time_limit, verbose=False, seed=0):
    """
    Nesterov-accelerated log-sum-exp gradient descent with adaptive beta and step size.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    beta = 8.0
    beta_growth = 1.003
    lr = 1.0 / np.sqrt(n)
    momentum = 0.85
    conv_size = 2 * n - 1

    y = a.copy()
    v = np.zeros(n)
    stall_count = 0
    n_iters = 0
    best_beta = beta
    best_lr = lr

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute gradient at y
        grad, conv_y = lse_gradient_fast(y, beta, n_top=min(25, conv_size))

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        # Nesterov momentum update
        v = momentum * v - lr * grad
        y = y + v
        y = np.maximum(y, 0.0)

        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y * norm / sy

        a = y.copy()
        n_iters += 1

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
            best_beta = beta
            best_lr = lr
        else:
            stall_count += 1

        # Adaptive parameters
        if t % 30 == 0:
            beta = min(beta * beta_growth, 300.0)

        if stall_count > 100:
            lr *= 0.85
            beta = max(beta * 0.75, 2.0)
            if stall_count > 400:
                # Soft restart with best
                y = best_a.copy()
                v = np.zeros(n)
                stall_count = 0
                lr = best_lr * 0.5
                beta = best_beta * 0.5

        if verbose and n_iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}, beta={beta:.1f}, lr={lr:.6f}")

    return best_a, best_score


def multiplicative_weights_optimize(a0, time_limit, verbose=False, seed=0):
    """
    Mirror descent with negative entropy (multiplicative weights / FTRL).
    Naturally maintains non-negativity.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-8)
    s0 = a.sum()
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    eta = 2.0 / np.sqrt(n)
    beta = 15.0
    patience = 0
    max_patience = 400
    conv_size = 2 * n - 1

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        grad, conv = lse_gradient_fast(a, beta, n_top=min(20, conv_size))

        # Center gradient
        g_mean = grad.mean()
        grad_centered = grad - g_mean

        gmax = np.abs(grad_centered).max()
        if gmax < 1e-15:
            break
        scaled_grad = grad_centered / gmax

        # Multiplicative update
        a = a * np.exp(-eta * scaled_grad)
        a = np.maximum(a, 1e-10)

        s = a.sum()
        if s < 1e-12:
            a = best_a.copy()
            break
        a = a * norm / s

        score = fast_autocorr(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            patience = 0
            eta = min(eta * 1.008, 5.0)
        else:
            patience += 1
            eta *= 0.995
            if eta < 1e-5:
                break

        if t % 100 == 0:
            beta = min(beta * 1.005, 150.0)

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW iter={t} ({elapsed:.0f}s): best={best_score:.10f}, eta={eta:.6f}, beta={beta:.1f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=False, seed=0):
    """
    Coordinate descent with FFT-based evaluation of 1D subproblems.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    no_improve_passes = 0
    max_no_improve = 4
    n_passes = 0
    typical = norm / n

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        conv = fast_autocorr(a)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            s_others = norm - aj

            candidates = [0.0, aj * 0.2, aj * 0.5, aj * 0.8]
            if aj > 0:
                candidates += [aj, aj * 1.5, aj * 2.0, aj * 3.0]
            candidates += [typical * 0.3, typical * 0.7, typical, typical * 2.0, typical * 4.0]
            candidates = list(set([max(0.0, min(c, norm * 0.4)) for c in candidates]))

            best_x = aj
            best_obj = conv.max()

            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                new_sum = s_others + x
                if new_sum < 1e-12:
                    continue
                new_a = a.copy()
                new_a[j] = x
                new_a = new_a * (norm / new_sum)
                new_conv = fast_autocorr(new_a)
                new_max = new_conv.max()
                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-12:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                    else:
                        a = best_a.copy()

        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= max_no_improve:
                break
        else:
            no_improve_passes = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def peak_reduction(a0, time_limit, verbose=False, seed=0):
    """
    Direct peak reduction: identify peaks, reduce contributing elements,
    redistribute to safe elements.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    alpha = 0.08
    patience = 0
    max_patience = 600
    conv_size = 2 * n - 1

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M = conv.max()

        n_peaks = max(5, min(15, n // 80))
        peak_indices = np.argsort(conv)[-n_peaks:]

        # Compute reduction vector
        reduction = np.zeros(n)
        peak_contribution = np.zeros(n)

        for k in peak_indices:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                kj = k - j
                contrib = a[j] * a[kj]
                reduction[j] += contrib
                if kj != j:
                    reduction[kj] += contrib
                peak_contribution[j] += contrib
                if kj != j:
                    peak_contribution[kj] += contrib

        if reduction.max() < 1e-15:
            break

        # Apply reduction
        a_trial = a - alpha * reduction / reduction.max()

        # Redistribute to safe elements
        safe_thresh = 0.05 * peak_contribution.max() if peak_contribution.max() > 0 else 0
        safe_mask = peak_contribution < safe_thresh
        if safe_mask.sum() > 0:
            total_reduction = max(0, (a - a_trial).sum())
            safe_sum = a[safe_mask].sum()
            if safe_sum > 1e-12 and total_reduction > 0:
                a_trial[safe_mask] += total_reduction * (a[safe_mask] / safe_sum)

        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-12:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score = fast_autocorr(a_trial).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            patience = 0
            alpha = min(alpha * 1.005, 0.4)
        else:
            patience += 1
            alpha *= 0.997
            if alpha < 1e-5:
                break

        # Periodic random perturbation to escape local minima
        if patience > 200 and patience % 50 == 0:
            idx = rng.integers(n)
            a[idx] *= (0.7 + 0.6 * rng.random())
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-12:
                a = a * norm / s

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRed iter={t} ({elapsed:.0f}s): best={best_score:.10f}, alpha={alpha:.6f}")

    return best_a, best_score


def spectral_optimize(a0, time_limit, verbose=False, seed=0):
    """
    Optimize in the spectral domain: try to flatten |A|^2 while maintaining non-negativity.
    Uses iterative projection between time and frequency domains.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-8)
    s0 = a.sum()
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    lr = 0.05
    patience = 0
    max_patience = 300

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        # Compute spectrum
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        power_mean = power.mean()

        # We want power to be flat. The "error" signal in time domain tells us
        # which elements to adjust.
        # conv[k] = IDFT(|A|^2)[k], we want to reduce max conv.
        # Gradient direction: reduce a[i] where it contributes to peaks.
        # Approximate via spectral error mapped back.
        spectral_error = power - power_mean
        # Only use the positive part (elements that increase variance)
        spectral_error_pos = np.maximum(spectral_error, 0)
        time_signal = irfft(spectral_error_pos, fft_size)[:n]

        if time_signal.max() < 1e-15:
            break

        # Update: reduce where time_signal is large
        a_trial = a - lr * time_signal / time_signal.max()
        a_trial = np.maximum(a_trial, 1e-8)
        s_trial = a_trial.sum()
        if s_trial < 1e-12:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score = fast_autocorr(a_trial).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            patience = 0
            lr = min(lr * 1.01, 1.0)
        else:
            patience += 1
            lr *= 0.98
            if lr < 1e-7:
                break

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral iter={t} ({elapsed:.0f}s): best={best_score:.10f}, lr={lr:.6f}")

    return best_a, best_score


def subgradient_at_k(a, k, n):
    """Compute subgradient of (a*a)[k] w.r.t. a."""
    grad = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    js = np.arange(j_lo, j_hi + 1)
    ks = k - js
    grad[js] += a[ks]
    mask = js != ks
    if mask.any():
        grad[ks[mask]] += a[js[mask]]
    return grad


def weighted_subgradient_optimize(a0, time_limit, verbose=False, seed=0):
    """
    Weighted multi-peak subgradient with Nesterov acceleration.
    Uses exponential weighting based on peak heights.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    conv_size = 2 * n - 1

    lr = 1.5 / np.sqrt(n)
    momentum = 0.88
    y = a.copy()
    y_prev = a.copy()
    patience = 0
    max_patience = 500

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(y)
        M = conv.max()

        # Select peaks with exponential weighting
        n_peaks = max(5, min(12, n // 70))
        top_idx = np.argsort(conv)[-n_peaks:]

        # Add some random near-max for exploration
        if t % 20 == 0:
            thresh = M * (0.8 + 0.15 * rng.random())
            candidates = np.where(conv >= thresh)[0]
            if len(candidates) > 0:
                k_rand = candidates[rng.integers(len(candidates))]
                top_idx = np.append(top_idx, k_rand)

        top_idx = np.unique(top_idx.astype(int))

        # Exponential weights based on peak height
        peak_vals = M - conv[top_idx]
        scale = max(M * 0.03, 1e-10)
        weights = np.exp(-5.0 * peak_vals / scale)
        weights /= weights.sum()

        # Weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            grad += w * subgradient_at_k(y, int(k), n)

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * (y - lr * grad)
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / sy

        # Momentum
        if t > 1:
            mom = min((t - 1.0) / (t + 2.0), 0.9)
            a_new = y_new + mom * (y_new - y_prev)
        else:
            a_new = y_new
        a_new = np.maximum(a_new, 0.0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / sa

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        for candidate in [y, a]:
            sc = fast_autocorr(candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
                patience = 0

        if patience > 0:
            patience += 1

        # Adaptive step size
        if patience > 100:
            lr *= 0.95
        elif patience == 0:
            lr = min(lr * 1.003, 3.0)

        if lr < 1e-8:
            break

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    WGSubg iter={t} ({elapsed:.0f}s): best={best_score:.10f}, lr={lr:.6f}")

    return best_a, best_score


def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set)
    size = len(k_indices)

    if size == 0:
        return None, None

    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[idx, lo:hi] = a_fixed[k - np.arange(lo, hi)]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    """Nesterov-accelerated LP best-response with progressive active-set tightening."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 60
    n_iters = 0
    conv_size = 2 * n - 1

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        if active_set is None or t % 50 == 0:
            progress = min(1.0, t / 800.0)
            tol = 0.1 * (1.0 - progress) + 0.008 * progress
            base_threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y >= base_threshold)[0]
            n_top = max(10, n // 60)
            top_k = np.argsort(conv_y)[-n_top:]
            active_set = np.unique(np.concatenate([active_set, top_k]))
            if len(active_set) > int(2.0 * n):
                conv_at_active = conv_y[active_set]
                keep = np.argsort(conv_at_active)[-int(2.0 * n):]
                active_set = active_set[keep]

        b, duals = lp_best_response(y, M_curr, active_set, solver)

        if b is None:
            if t % 10 == 0:
                active_set = np.arange(conv_size)
                b, duals = lp_best_response(y, M_curr, active_set, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.92)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Periodic blended check
        if t % 80 == 0:
            a_blend = 0.6 * best_a + 0.4 * a
            a_blend = np.maximum(a_blend, 0.0)
            s_b = a_blend.sum()
            if s_b > 1e-15:
                a_blend = a_blend * norm / s_b
                sc_b = fast_autocorr(a_blend).max()
                if sc_b < best_score:
                    best_score = sc_b
                    best_a = a_blend.copy()
                    stall_count = 0

        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence."""
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False


def perturb_optimize(best_seq, time_limit, solver, rng, seed, verbose=False):
    """Perturb the best sequence and optimize."""
    n = len(best_seq)
    scale = 0.01 + 0.02 * rng.random()
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.3)
    perturbed = best_seq * noise
    s = perturbed.sum()
    if s < 1e-12:
        return best_seq, float('inf')
    perturbed = perturbed / s
    a_norm = normalize_to(perturbed, n)
    a_out, sc = nesterov_lp_optimize(a_norm, time_limit, solver, verbose=verbose, seed=seed)
    return a_out, sc

    # === Phase 1: LP Nesterov (strong baseline) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.22, 350.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
        a_norm = normalize_to(best_seq, n_base)
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, seed=0)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # === Phase 2: LSE Gradient (new method) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LSE Gradient n={len(best_seq)}, budget={t2:.0f}s ===")
        a_norm = normalize_to(best_seq, len(best_seq))
        a_out, _ = nesterov_lse_optimize(a_norm, t2, verbose=True, seed=1)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "lse1", t_global)

    # === Phase 3: Weighted Subgradient with Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Weighted Subgradient n={len(best_seq)}, budget={t3:.0f}s ===")
        a_norm = normalize_to(best_seq, len(best_seq))
        a_out, _ = weighted_subgradient_optimize(a_norm, t3, verbose=True, seed=3)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "wsubg", t_global)

    # === Phase 4: Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.13, 220.0)
    if t4 > 20:
        print(f"\n=== Phase 4: CD n={len(best_seq)}, budget={t4:.0f}s ===")
        a_norm = normalize_to(best_seq, len(best_seq))
        a_out, _ = coordinate_descent(a_norm, t4, verbose=True, seed=5)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "cd1", t_global)

    # === Phase 5: Peak Reduction (new method) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 20:
        print(f"\n=== Phase 5: Peak Reduction n={len(best_seq)}, budget={t5:.0f}s ===")
        a_norm = normalize_to(best_seq, len(best_seq))
        a_out, _ = peak_reduction(a_norm, t5, verbose=True, seed=7)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "peaked1", t_global)

    # === Phase 6: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 250.0)
    if t6 > 30:
        print(f"\n=== Phase 6: LP Nesterov r2 n={len(best_seq)}, budget={t6:.0f}s ===")
        a_norm = normalize_to(best_seq, len(best_seq))
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True, seed=999)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "lp2", t_global)

    # === Phase 7: Multiplicative Weights ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.08, 120.0)
    if t7 > 15:
        print(f"\n=== Phase 7: Multiplicative Weights n={len(best_seq)}, budget={t7:.0f}s ===")
        a_norm = normalize_to(best_seq, len(best_seq))
        a_out, _ = multiplicative_weights_optimize(a_norm, t7, verbose=False, seed=11)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "mw1", t_global)

    # === Phase 8: Perturbation restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_restarts = 2
    t8_total = min(remaining * 0.25, 300.0)
    t8_each = t8_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        t8 = min(t8_each, remaining - 20)
        if t8 < 20:
            break
        print(f"\n=== Phase 8r{r}: Perturbation n={len(best_seq)}, budget={t8:.0f}s ===")
        a_out, _ = perturb_optimize(best_seq, t8, solver, rng, seed=500 + r, verbose=False)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"pert{r}", t_global)

    # === Phase 9: Final CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 9: Final CD n={len(best_seq)} ===")
        t9 = min(remaining * 0.5, 80.0)
        a_norm = normalize_to(best_seq, len(best_seq))
        a_out, _ = coordinate_descent(a_norm, t9, verbose=False, seed=77)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "cdFinal", t_global)

    # === Phase 10: Final LSE micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t10 = min(remaining * 0.5, 40.0)
        a_norm = normalize_to(best_seq, len(best_seq))
        a_out, _ = nesterov_lse_optimize(a_norm, t10, verbose=False, seed=314)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, "lseFinal", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
