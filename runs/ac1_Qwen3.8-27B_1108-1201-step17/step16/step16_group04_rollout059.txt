    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

    if s < 1e-15:

    return 2.0 * len(a) * fast_autocorr(a).max()

def build_constraint_rows(y, lags):
    """Build constraint rows for LP: row i encodes (y*b)[lags[i]]."""
    n = len(y)
    K = len(lags)

def spectral_equalize(a, n_iter=20, eps=1e-8):
    """Flatten power spectrum via iterative DFT correction."""
    a = normalize_arr(a).copy()
    n = len(a)
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = A.real ** 2 + A.imag ** 2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.5 * a + 0.5 * a_new
    return a

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def nesterov_lp_fw(a0, t0, timeout_s, base_k=200, max_k=600, verbose=True):

    best_obj = obj_value(a)

    x_prev = a.copy()
    current_k = base_k
    stall_count = 0

    for t in range(1, 100000):
        elapsed = time.monotonic() - t_start
        if elapsed > timeout_s:
            break

        # Nesterov extrapolation
        if t == 1:
            y = x.copy()
        else:
            beta_t = (t - 1.0) / (t + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        # Compute autocorrelation

        M_curr = ac.max()
        curr_obj = 2.0 * n * M_curr

            best_obj = curr_obj
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Restart if stalled

            stall_count = 0
            pert = rng.normal(0, 0.003, n)
            x_new = best_a * (1.0 + pert)

            x_new = x_new / x_new.sum()
            x = x_new

        x_new = normalize_arr((1.0 - alpha) * y + alpha * direction)
        x_new_obj = obj_value(x_new)

            best_obj = x_new_obj

        x = x_new

    a = normalize_arr(a0)
    a = np.maximum(a, 1e-15)
    w = np.log(a)
    best_a = a.copy()
    best_obj = obj_value(a)
    t_start = time.monotonic()
    rng = np.random.default_rng(999)

    for t in range(1, 500000):
        elapsed = time.monotonic() - t_start
        if elapsed > timeout_s:
            break

        ac = fast_autocorr(a)
        k_star = int(np.argmax(ac))
        peak_val = ac[k_star]

        # Gradient in log-space: grad[j] = 2*a[j]*(a[k* - j] - peak_val) with boundary handling
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        js_arr = np.arange(js, je + 1)
        partner_idx = k_star - js_arr
        grad[js_arr] = 2.0 * a[js_arr] * (a[partner_idx] - peak_val)

        # Adaptive step size
        eta = 0.5 / (np.sqrt(t + 1.0) + 1.0) * n * 0.1
        # Try a few step sizes
        best_eta = eta
        best_trial_obj = obj_value(a)
        for eta_try in [eta * 0.25, eta * 0.5, eta, eta * 2.0, eta * 4.0]:
            w_trial = w - eta_try * grad
            max_w = w_trial.max()
            a_trial = np.exp(w_trial - max_w)
            a_trial = a_trial / a_trial.sum()
            o = obj_value(a_trial)
            if o < best_trial_obj:
                best_trial_obj = o
                best_eta = eta_try

        w = w - best_eta * grad
        max_w = w.max()
        a = np.exp(w - max_w)
        a = a / a.sum()

        if obj_value(a) < best_obj - 1e-15:
            best_obj = obj_value(a)
            best_a = a.copy()

        # Periodic perturbation
        if t % 2000 == 0:
            pert = rng.normal(0, 0.002, n)
            w = np.log(np.maximum(a, 1e-15)) + pert
            max_w = w.max()
            a = np.exp(w - max_w)
            a = a / a.sum()

        if verbose and t % 5000 == 0:
            print(f"    LogGD t={t} ({elapsed:.1f}s): obj={best_obj:.6f}, eta={best_eta:.4f}")

    return best_a, best_obj

def coordinate_polish(a0, t0, timeout_s, verbose=True, rng_seed=42):
    """
    Peak-targeted coordinate descent: move mass from peak-contributing positions
    to low-impact positions.
    """
    n = len(a0)
    a = normalize_arr(a0)
    best_a = a.copy()
    best_obj = obj_value(a)
    t_start = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t_start < timeout_s and stall < 200:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        # Focus on top peaks
        top_peaks = np.argsort(ac)[::-1][:min(10, len(ac))]
        improved = False

        for k in top_peaks:
            if time.monotonic() - t_start > timeout_s:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find destination with minimal impact on peaks
            # Score: sum of partner values at peak lags
            peak_lags = top_peaks[:5]
            best_dst = -1
            best_score = float('inf')

            # Sample candidate destinations
            n_candidates = min(300, n)
            candidates = rng.choice(n, size=n_candidates, replace=False)

            for dst in candidates:
                dst_int = int(dst)
                if dst_int == src:
                    continue
                score = 0.0
                for k2 in peak_lags:
                    k2i = int(k2)
                    d1 = k2i - dst_int
                    if 0 <= d1 < n:
                        score += a[d1]
                if score < best_score:
                    best_score = score
                    best_dst = dst_int

            if best_dst < 0:
                continue

            # Try different fractions
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                o = obj_value(a_trial)
                if o < best_obj - 1e-15:
                    a = a_trial
                    best_obj = o
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Global perturbation to escape local minimum
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.005)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    if obj_value(a) < best_obj:
                        best_obj = obj_value(a)
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = np.maximum(a, 0.0) / max(a.sum(), 1e-15)

    if verbose:
        print(f"    Coord ({time.monotonic() - t_start:.1f}s): obj={best_obj:.6f}")
    return best_a, best_obj

def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with multi-phase strategy:
    Phase 1: LP-FW with Nesterov + line search (main workhorse)
    Phase 2: Log-gradient fine-tuning
    Phase 3: Coordinate polish
    Phase 4: Alternative length exploration
    Phase 5: Final refinement
    """
    t_global = time.monotonic()
    total_budget = min(float(budget_s) - 30.0, 1770.0)

    best_a = None
    best_score = float('inf')

    def try_update(candidate, label):
        nonlocal best_a, best_score
        if candidate is None:
            return
        candidate = np.maximum(np.asarray(candidate, dtype=np.float64), 0.0)
        s = candidate.sum()
        if s < 1e-10:
            return
        candidate = candidate / s
        sc = eval_safe(candidate)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = candidate.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: score={sc:.6f}, n={len(candidate)}, t={elapsed:.0f}s")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize_arr(p)
            p_score = eval_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_a = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_a is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_a = np.sin(np.pi * x) ** 2
        best_a = normalize_arr(best_a)
        best_score = eval_safe(best_a)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_a)

    # Quick spectral equalization
    remaining = total_budget - (time.monotonic() - t_global)
    a_spec = spectral_equalize(best_a, n_iter=20)
    try_update(a_spec, "spectral")

    # === Phase 1: Main LP-FW (large K, line search) ===
    remaining = total_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.55, 900.0)
    print(f"\n=== Phase 1: LP-FW (n={len(best_a)}, budget={t1:.0f}s) ===")
    if t1 > 60:
        a_in = normalize_arr(best_a)
        a_out, _ = nesterov_lp_fw(a_in, time.monotonic(), t1, base_k=200, max_k=600, verbose=True)
        try_update(a_out, f"lpfw1_n{len(a_out)}")

    # === Phase 2: Log-gradient fine-tuning ===
    remaining = total_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: LogGD (n={len(best_a)}, budget={t2:.0f}s) ===")
    if t2 > 30:
        a_in = normalize_arr(best_a)
        a_out, _ = log_gradient_descent(a_in, time.monotonic(), t2, verbose=True)
        try_update(a_out, f"loggd_n{len(a_out)}")

    # === Phase 3: Coordinate polish ===
    remaining = total_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: Coord (n={len(best_a)}, budget={t3:.0f}s) ===")
    if t3 > 20:
        a_in = normalize_arr(best_a)
        a_out, _ = coordinate_polish(a_in, time.monotonic(), t3, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{len(a_out)}")

    # === Phase 4: Alternative length ===
    remaining = total_budget - (time.monotonic() - t_global)
    n_best = len(best_a)
    if remaining > 100:
        # Try a slightly different length
        n_alt = n_best + 200
        if n_alt > 3500:
            n_alt = n_best - 200
        n_alt = max(1200, min(n_alt, 3500))
        if n_alt != n_best:
            print(f"\n=== Phase 4: Alt length n={n_alt}, budget={remaining:.0f}s ===")
            a_alt = interpolate_sequence(best_a, n_alt)
            a_alt = spectral_equalize(a_alt, n_iter=15)
            try_update(a_alt, f"interp_n{n_alt}")

            t4a = min(remaining * 0.5, 250.0)
            if t4a > 30:
                a_out, _ = nesterov_lp_fw(a_alt, time.monotonic(), t4a, base_k=150, max_k=400, verbose=True)
                try_update(a_out, f"lpfw_alt_n{n_alt}")

    # === Phase 5: Second LP-FW on best ===
    remaining = total_budget - (time.monotonic() - t_global)
    n_best = len(best_a)
    if remaining > 80:
        print(f"\n=== Phase 5: LP-FW refinement (n={n_best}, budget={remaining:.0f}s) ===")
        t5 = min(remaining * 0.5, 300.0)
        if t5 > 30:
            a_in = normalize_arr(best_a)
            a_out, _ = nesterov_lp_fw(a_in, time.monotonic(), t5, base_k=300, max_k=700, verbose=True)
            try_update(a_out, f"lpfw2_n{n_best}")

    # === Phase 6: Final coordinate polish ===
    remaining = total_budget - (time.monotonic() - t_global)
    n_best = len(best_a)
    if remaining > 30:
        print(f"\n=== Phase 6: Final Coord (n={n_best}) ===")
        t6 = min(remaining * 0.6, 120.0)
        if t6 > 15:
            a_in = normalize_arr(best_a)
            a_out, _ = coordinate_polish(a_in, time.monotonic(), t6, verbose=True, rng_seed=777)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 7: Final log-gradient ===
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_a)
        print(f"\n=== Phase 7: Final LogGD (n={n_best}) ===")
        t7 = min(remaining * 0.7, 60.0)
        if t7 > 15:
            a_in = normalize_arr(best_a)
            a_out, _ = log_gradient_descent(a_in, time.monotonic(), t7, verbose=False)
            try_update(a_out, f"loggd_final_n{n_best}")

    # Final validation
    if best_a is not None:
        fs = eval_safe(best_a)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_a)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_a]
