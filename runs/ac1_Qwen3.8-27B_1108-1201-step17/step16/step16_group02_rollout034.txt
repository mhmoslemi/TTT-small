def fast_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def max_conv(a):
    """Maximum of full convolution."""
    return fast_conv(a).max()

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def lp_frank_wolfe(a0, time_limit, solver, max_constraints=200, verbose=True):
    """LP Frank-Wolfe with backtracking line search."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = max_conv(a)
    t0 = time.monotonic()
    
    y = a.copy()
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(y)
        M_curr = ac.max()
        
        # Select active constraints
        threshold = M_curr * 0.97
        active_lags = np.where(ac >= threshold)[0]
        if len(active_lags) > max_constraints:
            active_lags = np.argsort(ac)[::-1][:max_constraints]
        if len(active_lags) == 0:
            active_lags = np.array([int(np.argmax(ac))])
        
        # Build LP: maximize sum(b) subject to (y*b)[m] <= M_curr for active m, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = []
        for m in active_lags:
            m_int = int(m)
            j_min = max(0, m_int - n + 1)
            j_max = min(n - 1, m_int)
            if j_min <= j_max:
                row = np.zeros(n)
                row[j_min:j_max + 1] = y[m_int - (j_min:j_max + 1)]
                constraints.append(row @ b_var <= M_curr)
        
        constraints.append(cp.sum(b_var) <= 1.0)
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Backtracking line search
        alpha = 1.0
        f_curr = M_curr
        for _ in range(50):
            y_new = (1.0 - alpha) * y + alpha * g_dir
            y_new = np.maximum(y_new, 0.0)

            if sy < 1e-15:

def gradient_descent_momentum(a0, time_limit, verbose=True, lr=0.01, momentum=0.9):

    t0 = time.monotonic()

            break
        
        # Determine beta for softmax
        beta = 5.0
        for thresh, b_val in beta_schedule:
            if t >= thresh:
                beta = b_val
        if t > 40000:
            beta = min(200.0, 100.0 + (t - 40000) * 0.01)
        
        ac = fast_conv(a)
        M = ac.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()

        ac_clipped = np.maximum(ac, 1e-30)
        log_w = beta * (ac_clipped - M)
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(ac)
            w[np.argmax(ac)] = 1.0
            w_sum = 1.0

        # Gradient of softmax-weighted sum
        grad = np.zeros(n)
        for k in range(len(ac)):
            wi = w[k]
            if wi < 1e-10:

            k_int = int(k)
            j_min = max(0, k_int - n + 1)

                grad[j_min:j_max + 1] += 2.0 * wi * a[k_int - (j_min:j_max + 1)]
        
        # Project gradient to sum-zero subspace
        grad -= np.mean(grad)
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v

            if time.monotonic() - t0 > time_limit:

            best_trial_m = best_m
            best_move = None

                    a_trip[dst] += amt
                    if a_trip[i] < -1e-14:

                a[dst] += amt

    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = max_conv(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(100, n // 3)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            if min(a[i], a[j]) < 1e-14:
                continue
            
            for src, dst in [(i, j), (j, i)]:
                if a[src] < 1e-14:
                    continue
                for frac in [0.001, 0.005, 0.01, 0.03, 0.05]:
                    amt = a[src] * frac
                    a_trip = a.copy()
                    a_trip[src] -= amt
                    a_trip[dst] += amt
                    a_trip = np.maximum(a_trip, 0.0)
                    s_t = a_trip.sum()
                    if s_t < 1e-15:
                        continue
                    a_trip = a_trip / s_t
                    m_trip = max_conv(a_trip)
                    if m_trip < best_m - 1e-15:
                        a = a_trip
                        best_m = m_trip
                        best_a = a.copy()
                        improved = True
                        break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent sequence
    n_base = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        # Use a smooth bump function as initial guess
        # Optimal shape is roughly a raised cosine or similar
        best_seq = 0.5 * (1 + np.cos(2 * np.pi * x)) ** 0.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Gauss-Newton Active Set (40% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 600.0)
    print(f"\n=== Phase 1: Gauss-Newton n={len(best_seq)} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = gauss_newton_active_set(a_in, t1, verbose=True, max_active=40)
        try_update(a_out, f"GN_n{len(a_out)}")
    
    # === Phase 2: Gradient Descent with Momentum (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 2: GD-Momentum n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = gradient_descent_momentum(a_in, t2, verbose=True, lr=0.005, momentum=0.92)
        try_update(a_out, f"GD_n{len(a_out)}")
    
    # === Phase 3: Gauss-Newton again (polish after GD) (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 3: GN Polish n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = gauss_newton_active_set(a_in, t3, verbose=True, max_active=30)
        try_update(a_out, f"GN2_n{len(a_out)}")
    
    # === Phase 4: LP Frank-Wolfe (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: LP-FW n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, max_constraints=200, verbose=True)
        try_update(a_out, f"LP_n{len(a_out)}")
    
    # === Phase 5: Coordinate Descent Polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: CD n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_polish(a_in, t5, verbose=True, rng_seed=789)
        try_update(a_out, f"CD_n{len(a_out)}")
    
    # === Phase 6: Paired Swap (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.05, 60.0)
    print(f"\n=== Phase 6: PairSwap n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = paired_swap(a_in, t6, verbose=True, rng_seed=321)
        try_update(a_out, f"PS_n{len(a_out)}")
    
    # === Phase 7: Alt length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        # Try slightly larger n
        n_alt = n0 + 300
        print(f"\n=== Phase 7: Alt length n={n_alt} (budget={remaining:.0f}s) ===")
        a_long = interpolate_sequence(best_seq, n_alt)
        a_in = norm_sum1(a_long)
        t7 = min(remaining * 0.5, 200.0)
        if t7 > 30:
            a_out, _ = gauss_newton_active_set(a_in, t7, verbose=True, max_active=40)
            try_update(a_out, f"GN_n{n_alt}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t7b = min(remaining * 0.4, 100.0)
            if t7b > 15:
                a_in = norm_sum1(best_seq if len(best_seq) == n0 else a_out)
                a_out2, _ = coordinate_descent_polish(a_in, t7b, verbose=False, rng_seed=444)
                try_update(a_out2, f"CD_alt")
    
    # === Phase 8: Final GN polish at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final GN n={n_best} (budget={remaining:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = gauss_newton_active_set(a_in, remaining * 0.7, verbose=False, max_active=25)
        try_update(a_out, f"GN_final_n{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = paired_swap(a_in, remaining * 0.8, verbose=False, rng_seed=999)
            try_update(a_out, f"PS_final")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
