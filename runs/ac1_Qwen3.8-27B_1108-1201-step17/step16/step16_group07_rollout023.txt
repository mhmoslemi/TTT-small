    """Compute gradient of smoothed objective.
    a: sequence of length n
    p: softmax weights of length 2n-1
    N: FFT size (power of 2, >= 3n-2)
    Returns: gradient of length n
    """
    n = len(a)
    a_ext = np.zeros(2 * n - 1)
    a_ext[:n] = a
    A = rfft(a_ext, N)
    P = rfft(p, N)
    c = irfft(A * np.conj(P), N)
    grad = np.zeros(n)
    grad[0] = 2.0 * c[0]
    if n > 1:
        idx = (N - np.arange(1, n)) % N
        grad[1:] = 2.0 * c[idx]
    return grad

def project_simplex(v, c):
    """Project v onto {x >= 0, sum(x) = c}.
    Uses bisection to find the threshold.
    """
    n = len(v)
    # Binary search for tau such that sum(max(v - tau, 0)) = c
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(100):
        tau = (lo + hi) / 2
        s = np.sum(np.maximum(v - tau, 0))
        if s > c:
            lo = tau
        else:
            hi = tau
    proj = np.maximum(v - tau, 0)
    # Correct for numerical error
    diff = c - proj.sum()
    if abs(diff) > 1e-10 and proj.sum() > 0:
        # Distribute the difference among non-zero entries
        support = proj > 0
        proj[support] += diff / support.sum()
        proj = np.maximum(proj, 0)
    return proj

def fista_smoothed(a_init, n, time_limit, seed=0, verbose=False):
    """FISTA with log-sum-exp smoothing for the max-of-quadratics objective."""
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    N = 1 << (3 * n - 2 - 1).bit_length()
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()  # momentum variable
    t_mom = 1.0  # momentum parameter
    
    t0 = time.monotonic()
    L = 1.0  # Lipschitz estimate (for backtracking)
    step = 1.0 / L
    eps = best_score * 0.05  # smoothing parameter
    patience = 0
    max_patience = 300
    
    for t in range(1, 5_000_000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break
        
        # Compute convex and softmax weights
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        
        # Update smoothing parameter (decrease over time)
        eps = max(M_curr * 0.01, best_score * 0.005)
        
        # Softmax weights (log-sum-exp trick)
        shifted = conv_y - M_curr
        logsumexp = np.log(np.sum(np.exp(shifted / eps))) + M_curr / eps
        p = np.exp((conv_y - M_curr) / eps)
        p = p / p.sum()
        
        # Compute gradient at y
        grad_y = smooth_gradient(y, p, N)
        
        # Project gradient to maintain sum constraint
        # We want the step to be in the "tangential" direction
        grad_mean = np.mean(grad_y)
        grad_tan = grad_y - grad_mean
        
        # FISTA update: x_new = project(y - step * grad)
        a_trial = y - step * grad_tan
        a_trial = project_simplex(a_trial, norm)
        
        # Check improvement
        score_trial = fast_autocorr(a_trial).max()
        
        # Backtracking: if no improvement, reduce step
        if score_trial < best_score:
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            # Increase step slightly
            step = min(step * 1.005, 2.0 * norm / n ** 0.8)
        else:
            patience += 1
            step *= 0.95
            if step < 1e-14:
                break
            # Use the momentum update anyway (FISTA doesn't require improvement at each step)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        beta = (t_mom - 1.0) / t_new
        y_new = a_trial + beta * (a_trial - a)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = a_trial.copy()
        else:
            y_new = project_simplex(y_new, norm)
        
        t_mom = t_new
        a = a_trial
        y = y_new
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.6f}, eps={eps:.6f}")
    
    return best_a, best_score

def coordinate_descent(a_init, time_limit, seed=0, verbose=False):
    """Coordinate descent with efficient line search."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    norm = np.sqrt(2.0 * n)
    
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    no_improve_passes = 0
    max_no_improve = 5
    
    while time.monotonic() - t0 < time_limit:
        if no_improve_passes >= max_no_improve:
            break
        
        conv = fast_autocorr(a)
        M_curr = conv.max()
        improved = False
        
        # Process coordinates in an order based on their contribution to peaks
        # Prioritize coordinates that affect the maximum peaks
        peak_indices = np.argsort(conv)[-max(10, n // 50):]
        
        # Build a priority order: coordinates that appear in peak computations first
        coord_priority = np.zeros(n)
        for k in peak_indices:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            coord_priority[lo:hi] += 1.0
            # Also the "mirrored" contribution
            lo2 = max(0, k - n + 1)
            hi2 = min(n, k + 1)
            coord_priority[lo2:hi2] += 1.0
        
        # Mix priority with random order
        random_scores = rng.random(n)
        order = np.argsort(-coord_priority + 0.5 * random_scores)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            aj = a[j]
            if aj < 1e-15:
                # Try turning on this coordinate
                candidates = [norm / (10 * n), norm / (5 * n), norm / (2 * n)]
            else:
                # Try different values
                typical = norm / n
                candidates = [0.0, aj * 0.5, aj * 0.7, aj * 1.3, aj * 1.7, typical * 0.5, typical, typical * 2]
                candidates = [x for x in candidates if 0 <= x <= norm * 0.3]
            
            best_x = aj
            best_obj = M_curr
            
            for x in candidates:
                if abs(x - aj) < 1e-10:
                    continue
                
                delta = x - aj
                # Update convolution
                new_conv = conv.copy()
                lo = max(0, j)
                hi = min(size, j + n)
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta
                
                # Renormalize
                new_sum = a.sum() - aj + x
                if new_sum < 1e-15:
                    continue
                scale = norm / new_sum
                new_max = new_conv.max() * scale * scale
                
                if new_max < best_obj:
                    best_obj = new_max
                    best_x = x
            
            if abs(best_x - aj) > 1e-10:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        conv = new_conv * (norm / s_new) ** 2
                        improved = True
                    else:
                        a = best_a.copy()
                        conv = fast_autocorr(a)
        
        if improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD pass complete ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def lp_refine(a_init, time_limit, solver, verbose=False):
    """LP-based refinement using best response."""
    n = len(a_init)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Modified LP best response (simplified from before)
    y = a.copy()
    stall = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 50:
            break
        
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        
        # Active set: peaks near the maximum
        tol = 0.05
        active_set = np.where(conv_y >= M_curr * (1 - tol))[0]
        n_active = min(len(active_set), int(1.5 * n))
        if len(active_set) > n_active:
            active_set = active_set[np.argsort(conv_y[active_set])[-n_active:]]
        
        # Build constraint matrix
        size = len(active_set)
        T = np.zeros((size, n))
        for idx, k in enumerate(active_set):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                T[idx, lo:hi] = y[k - np.arange(lo, hi)]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            stall += 5

            continue
