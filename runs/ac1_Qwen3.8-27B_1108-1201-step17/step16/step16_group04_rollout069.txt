def fast_autocorr(a):
    n=len(a)
    if n==0: return np.array([],float)
    size=1
    target=2*n-1
    while size < target: size <<=1
    F=np.fft.rfft(a, size)
    c=np.fft.irfft(F*F, size)[:target]
    return c
```

```python
"""
Search for non-negative step-function heights minimizing
    2 * n * max(convolve(a, a)) / (sum(a))**2

Method overview:
- Generate a broad library of smooth asymmetric seeds (beta-power, exponential,
  random non-negative Fourier shapes, and interpolated parent).
- Polish strong seeds with mirror descent on a log-sum-exp smoothed version of
  the maximum autocorrelation.  The mirror update keeps the simplex {a >= 0,
  sum(a)=1} naturally.
- Use L-BFGS-B on the log-density representation when SciPy is available.
- Finish with a small targeted transfer routine that moves mass to reduce the
  currently largest autocorrelation coefficients.
"""

import numpy as np
import time
import math
import operator

try:
    import scipy.optimize as _scipy_optimize
    _HAS_SCIPY = True
except Exception:
    _scipy_optimize = None
    _HAS_SCIPY = False

_LBFGS_BETA = 1.0
_LBFGS_SCALE = 1.0


def _next_power_of_two(m):
    s = 1
    while s < int(max(1, m)):
        s <<= 1
    return s


def _normalize_sequence(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _softmax_from_logz(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([1.0], dtype=np.float64)
    zmax = float(np.max(z))
    if not np.isfinite(zmax):
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    e = np.exp(z - zmax)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        out = np.zeros(z.size, dtype=np.float64)
        out[int(np.argmax(z))] = 1.0
        return out
    return e / s


def _softmax_theta(theta):
    return _softmax_from_logz(theta)


def _fast_autocorr(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    target = 2 * n - 1
    size = _next_power_of_two(target)
    F = np.fft.rfft(np.asarray(a, dtype=np.float64), n=size)
    c = np.fft.irfft(F * F, n=size)[:target]
    return c


def _score_fast(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        return float("inf")
    p = np.clip(np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0), 0.0, 1000.0)
    s = float(np.sum(p))
    if s <= 1e-15:
        return float("inf")
    p = p / s
    c = _fast_autocorr(p)
    if c.size == 0:
        return float("inf")
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return float("inf")
    return float(2.0 * len(p) * mx)


def _correlate_weights(p, w):
    """
    Return y_i = sum_{r=0}^{n-1} p_r * w_{i+r}, where w has length 2n-1.
    This is the part of the d/da_i of the linear autocorrelation.
    """
    n = len(p)
    if n == 0:
        return np.array([], dtype=np.float64)
    target = len(w) + n - 1
    size = _next_power_of_two(target)
    pw = np.zeros(size, dtype=np.float64)
    pw[:n] = p[::-1]
    wf = np.fft.rfft(np.asarray(w, dtype=np.float64), n=size)
    pf = np.fft.rfft(pw, n=size)
    conv = np.fft.irfft(wf * pf, n=size)
    return conv[n - 1:n - 1 + n]


def _smooth_value_p(p, B):
    p = np.asarray(p, dtype=np.float64)
    n = len(p)
    if n == 0:
        return float("inf")
    c = _fast_autocorr(p)
    c = np.maximum(c, 0.0)
    if c.size == 0:
        return float("inf")
    if B <= 0.0 or not np.isfinite(B):
        return float(2.0 * n * float(np.max(c)))
    z = B * c
    zmax = float(np.max(z))
    if not np.isfinite(zmax):
        return float(2.0 * n * float(np.max(c)))
    e = np.exp(z - zmax)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return float(2.0 * n * float(np.max(c)))
    return float(2.0 * n * (zmax - math.log(s)) / B)


def _smooth_value_grad_p(p, B):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = len(p)
    if n == 0:
        return float("inf"), np.array([], dtype=np.float64)
    c = _fast_autocorr(p)
    c = np.maximum(c, 0.0)
    if c.size == 0:
        return float("inf"), np.zeros(n, dtype=np.float64)
    if B <= 0.0 or not np.isfinite(B):
        mx = int(np.argmax(c))
        w = np.zeros(c.size, dtype=np.float64)
        w[mx] = 1.0
        f = float(2.0 * n * float(c[mx]))
    else:
        z = B * c
        zmax = float(np.max(z))
        if not np.isfinite(zmax):
            f = float(2.0 * n * float(np.max(c)))
            mx = int(np.argmax(c))
            w = np.zeros(c.size, dtype=np.float64)
            w[mx] = 1.0
        else:
            e = np.exp(z - zmax)
            s = float(np.sum(e))
            if s <= 0.0 or not np.isfinite(s):
                f = float(2.0 * n * float(np.max(c)))
                mx = int(np.argmax(c))
                w = np.zeros(c.size, dtype=np.float64)
                w[mx] = 1.0
            else:
                f = float(2.0 * n * (zmax - math.log(s)) / B)
                w = e / s
    y = _correlate_weights(p, w)
    g = 4.0 * float(n) * y
    return f, g


def _interpolate_sequence(f, new_n):
    f = np.asarray(f, dtype=np.float64).ravel()
    if f.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    f = np.clip(f, 0.0, None)
    s = float(np.sum(f))
    if s <= 1e-15:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    f = f / s
    old_n = f.size
    new_n = int(max(1, new_n))
    if old_n == new_n:
        return f.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f)
    g = np.maximum(g, 0.0)
    ss = float(np.sum(g))
    if ss <= 1e-15:
        return np.ones(new_n, dtype=np.float64)
    return g / ss


def _make_beta_seed(n, p_left, p_right):
    n = int(max(1, n))
    idx = np.arange(1, n + 1, dtype=np.float64)
    left = np.power(idx, float(p_left))
    right = np.power((float(n) + 1.0) - idx, float(p_right))
    a = left * right
    a = np.maximum(a, 0.0)
    return _normalize_sequence(a)


def _make_exp_seed(n, alpha):
    n = int(max(1, n))
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(float(alpha) * t)
    return _normalize_sequence(a)


def _make_random_fourier_seed(n, seed):
    n = int(max(1, n))
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    rng = np.random.default_rng(seed)
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    m = int(rng.integers(2, 5))
    for _ in range(m):
        freq = float(rng.uniform(0.25, 3.25))
        amp = float(rng.uniform(0.35, 1.0))
        phi = float(rng.uniform(0.0, 2.0 * math.pi))
        a = a + amp * np.sin(2.0 * math.pi * freq * t + phi)
    a = a * a
    a = np.maximum(a, 0.0) + 1e-8
    return _normalize_sequence(a)


def _spectral_equalize(a, iterations=8, eps=1e-12):
    a = _normalize_sequence(a)
    n = len(a)
    if n < 2:
        return a
    for _ in range(int(max(0, iterations))):
        A = np.fft.fft(a)
        P = A.real * A.real + A.imag * A.imag
        C = 1.0 / np.sqrt(P + eps)
        A2 = A * C
        a2 = np.real(np.fft.ifft(A2))
        a2 = np.maximum(a2, 0.0)
        s2 = float(np.sum(a2))
        if s2 <= 1e-15:
            break
        a2 = a2 / s2
        a = 0.60 * a + 0.40 * a2
    return _normalize_sequence(a)


def _generate_seeds_for_length(n, parent):
    n = int(max(1, n))
    seeds = []

    if parent is not None:
        seeds.append((n, _interpolate_sequence(parent, n)))

    powers = [
        -3.0, -2.0, -1.5, -1.0, -0.75, -0.5,
        -0.25, 0.0, 0.25, 0.5, 0.75, 1.0,
        1.5, 2.0, 3.0
    ]
    for pl in powers:
        for pr in powers:
            seeds.append((n, _make_beta_seed(n, pl, pr)))

    for alpha in (-12.0, -6.0, -3.0, -1.5, -0.75, 0.75, 1.5, 3.0, 6.0, 12.0):
        seeds.append((n, _make_exp_seed(n, alpha)))

    for s in range(12):
        seeds.append((n, _make_random_fourier_seed(n, 90000 + 17 * n + s)))

    return seeds


def _select_candidates(all_seeds, max_candidates=6):
    scored = []
    for n, p in all_seeds:
        if p is None:
            continue
        sc = _score_fast(p)
        if np.isfinite(sc):
            scored.append((sc, p, n))

    if not scored:
        return []

    best_by_n = {}
    for sc, p, n in scored:
        old = best_by_n.get(n)
        if old is None or sc < old[0]:
            best_by_n[n] = (sc, p, n)

    candidates = []
    added_n = set()

    by_n = sorted(best_by_n.values(), key=operator.itemgetter(0))
    for item in by_n:
        n = item[2]
        if n not in added_n:
            candidates.append(item)
            added_n.add(n)

    top = sorted(scored, key=operator.itemgetter(0))
    for item in top:
        if len(candidates) >= max_candidates:
            break
        n = item[2]
        if n not in added_n:
            candidates.append(item)
            added_n.add(n)

    candidates = sorted(candidates[: max_candidates], key=operator.itemgetter(0))
    return candidates


def _mirror_smooth_phase(p0, B, time_limit, rng):
    p = _normalize_sequence(p0)
    n = len(p)
    if n == 0:
        return p, float("inf")
    if not np.isfinite(B) or B <= 0.0:
        return p.copy(), _score_fast(p)
    if time_limit <= 0.0:
        return p.copy(), _score_fast(p)

    best_p = p.copy()
    best_score = _score_fast(p)
    smooth_best = _smooth_value_p(p, B)
    eta = 0.5
    stall = 0
    start = time.monotonic()
    accepted = 0
    exact_check_every = 8

    while time.monotonic() - start < float(max(0.0, time_limit)):
        f0, g = _smooth_value_grad_p(p, B)
        if not np.isfinite(f0):
            p = best_p.copy()
            eta = 0.5
            smooth_best = _smooth_value_p(p, B)
            continue

        if not np.all(np.isfinite(g)):
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)

        if np.isfinite(smooth_best) and f0 < smooth_best:
            smooth_best = f0

        # A constant added to g cancels in the multiplicative update.
        g = g - float(np.mean(g))

        tol = 1e-13 * max(1.0, abs(float(f0)))
        success = False
        factor_used = 0.0

        for factor in (1.0, 0.5, 0.25, 0.1):
            step = eta * factor
            if step <= 0.0:
                continue
            z = np.log(np.maximum(p, 1e-300)) - step * g
            zmax = float(np.max(z))
            if not np.isfinite(zmax):
                continue
            e = np.exp(z - zmax)
            s = float(np.sum(e))
            if s <= 0.0 or not np.isfinite(s):
                continue
            q = e / s
            f1 = _smooth_value_p(q, B)
            if not np.isfinite(f1):
                continue
            if f1 < float(f0) - tol or f1 < smooth_best - tol:
                p = q
                smooth_best = min(smooth_best, f1)
                success = True
                factor_used = factor
                break

        if success:
            accepted += 1
            stall = 0
            if factor_used >= 0.999:
                eta = min(eta * 1.05, 2.0)
            elif factor_used >= 0.499:
                eta = min(max(eta * 1.0, 0.5), 2.0)
            else:
                eta = max(eta * 0.8, 1e-3)

            if accepted % exact_check_every == 0:
                sc = _score_fast(p)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best_score = sc
                    best_p = p.copy()
        else:
            stall += 1
            if eta > 1e-4:
                eta = max(eta * 0.5, 1e-4)
            else:
                if stall % 100 == 0:
                    eta = 0.5

            if stall > 250:
                # Restart from the best exact point with a multiplicative noise.
                p = best_p.copy()
                noise = rng.normal(0.0, 0.05, n)
                p = p * (1.0 + noise)
                p = _normalize_sequence(p)
                eta = 0.5
                stall = 0
                smooth_best = _smooth_value_p(p, B)

    sc = _score_fast(p)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score


def _polish_candidate(p0, time_limit, rng, label="candidate"):
    p = _normalize_sequence(p0)
    n = len(p)
    if n == 0 or time_limit <= 10.0:
        return p, _score_fast(p)

    # Cheap structural preprocessing.
    pe = _spectral_equalize(p, iterations=6)
    sce = _score_fast(pe)
    if np.isfinite(sce) and sce < _score_fast(p):
        p = pe

    uniform = np.ones(n, dtype=np.float64) / float(n)
    for mix in (0.2, 0.4, 0.6):
        pm = (1.0 - mix) * p + mix * uniform
        pm = _normalize_sequence(pm)
        scm = _score_fast(pm)
        if np.isfinite(scm) and scm < _score_fast(p):
            p = pm

    best_p = p.copy()
    best_score = _score_fast(p)
    start = time.monotonic()

    targets = (5.0, 10.0, 20.0, 40.0, 80.0, 160.0)
    for tgt in targets:
        elapsed = time.monotonic() - start
        remaining = float(max(0.0, time_limit - elapsed))
        if remaining <= 15.0:
            break

        max_c = best_score / (2.0 * float(n))
        if not np.isfinite(max_c) or max_c <= 1e-300:
            break

        B = min(1.0e9, float(tgt) / max_c)
        phase_time = min(remaining * 0.22, 70.0)
        if phase_time < 3.0:
            if remaining > 10.0:
                phase_time = min(remaining, 30.0)
            else:
                break

        p_out, sc_out = _mirror_smooth_phase(p, B, phase_time, rng)
        p_out = _normalize_sequence(p_out)
        sc_out = _score_fast(p_out)
        if np.isfinite(sc_out) and sc_out < best_score - 1e-12:
            best_score = sc_out
            best_p = p_out.copy()

        if np.isfinite(_score_fast(p_out)) and _score_fast(p_out) < best_score * 1.02 + 1e-9:
            p = p_out
        else:
            p = best_p

    remaining = float(max(0.0, time_limit - (time.monotonic() - start)))
    if remaining > 25.0:
        max_c = best_score / (2.0 * float(n))
        if np.isfinite(max_c) and max_c > 1e-300:
            B = min(1.0e9, 256.0 / max_c)
            p_out, sc_out = _mirror_smooth_phase(best_p, B, min(remaining, 80.0), rng)
            p_out = _normalize_sequence(p_out)
            sc_out = _score_fast(p_out)
            if np.isfinite(sc_out) and sc_out < best_score - 1e-12:
                best_score = sc_out
                best_p = p_out.copy()

    final_sc = _score_fast(best_p)
    if np.isfinite(final_sc):
        best_score = min(best_score, final_sc)
    return best_p, best_score


def _lbfgs_fun(theta):
    n = len(theta)
    if n == 0:
        return 1e100
    p = _softmax_theta(theta)
    c = _fast_autocorr(p)
    c = np.maximum(c, 0.0)
    if c.size == 0:
        return 1e100
    B = _LBFGS_BETA
    if B <= 0.0 or not np.isfinite(B):
        return float(_LBFGS_SCALE * 2.0 * n * float(np.max(c)))
    z = B * c
    zmax = float(np.max(z))
    if not np.isfinite(zmax):
        return float(_LBFGS_SCALE * 2.0 * n * float(np.max(c)))
    e = np.exp(z - zmax)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return 1e100
    return float(_LBFGS_SCALE * 2.0 * n * (zmax - math.log(s)) / B)


def _lbfgs_grad(theta):
    n = len(theta)
    if n == 0:
        return np.array([], dtype=np.float64)
    p = _softmax_theta(theta)
    c = _fast_autocorr(p)
    c = np.maximum(c, 0.0)
    if c.size == 0:
        return np.zeros(n, dtype=np.float64)
    B = _LBFGS_BETA
    if B <= 0.0 or not np.isfinite(B):
        mx = int(np.argmax(c))
        w = np.zeros(c.size, dtype=np.float64)
        w[mx] = 1.0
    else:
        z = B * c
        zmax = float(np.max(z))
        e = np.exp(z - zmax)
        s = float(np.sum(e))
        if s <= 0.0 or not np.isfinite(s):
            mx = int(np.argmax(c))
            w = np.zeros(c.size, dtype=np.float64)
            w[mx] = 1.0
        else:
            w = e / s
    y = _correlate_weights(p, w)
    gp = 4.0 * float(n) * y
    D = float(np.dot(p, gp))
    grad_theta = p * (gp - D)
    return float(_LBFGS_SCALE) * grad_theta


def _lbfgs_polish(p0, time_limit, rng):
    if not _HAS_SCIPY:
        return p0, _score_fast(p0)

    p = _normalize_sequence(p0)
    n = len(p)
    if n == 0 or time_limit <= 5.0:
        return p, _score_fast(p)

    best_p = p.copy()
    best_score = _score_fast(p)
    start = time.monotonic()

    for tgt in (20.0, 60.0, 120.0):
        elapsed = time.monotonic() - start
        if elapsed > float(max(0.0, time_limit)) * 0.75:
            break

        max_c = best_score / (2.0 * float(n))
        if not np.isfinite(max_c) or max_c <= 1e-300:
            break

        global _LBFGS_BETA, _LBFGS_SCALE
        _LBFGS_BETA = min(1.0e9, float(tgt) / max_c)
        _LBFGS_SCALE = float(n)

        theta = np.log(np.maximum(p, 1e-20))
        theta = theta - float(np.mean(theta))
        if not np.all(np.isfinite(theta)):
            theta = np.zeros(n, dtype=np.float64)

        maxiter = 60 if n > 3000 else 110
        try:
            res = _scipy_optimize.minimize(
                _lbfgs_fun,
                theta,
                method="L-BFGS-B",
                jac=_lbfgs_grad,
                options={
                    "maxiter": int(maxiter),
                    "ftol": 1e-10,
                    "gtol": 1e-5,
                    "maxls": 20,
                    "maxcor": 30
                }
            )
            if hasattr(res, "x") and res.x is not None:
                tr = np.asarray(res.x, dtype=np.float64)
                if tr.shape == (n,) and np.all(np.isfinite(tr)):
                    pr = _softmax_theta(tr)
                    sc = _score_fast(pr)
                    if np.isfinite(sc) and sc < best_score - 1e-12:
                        best_score = sc
                        best_p = pr.copy()
                        p = pr.copy()
        except Exception:
            pass

    final_sc = _score_fast(best_p)
    if np.isfinite(final_sc):
        best_score = min(best_score, final_sc)
    return best_p, best_score


def _targeted_transfer_refine(p0, time_limit, rng):
    p = _normalize_sequence(p0)
    n = len(p)
    if n < 2 or time_limit <= 5.0:
        return p, _score_fast(p)

    best_p = p.copy()
    best_score = _score_fast(p)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < float(max(0.0, time_limit)):
        c = _fast_autocorr(p)
        c = np.maximum(c, 0.0)
        if c.size == 0:
            break
        top_idx = np.argsort(c)[::-1][:min(8, c.size)]
        top4 = top_idx[:4].astype(int)
        moved = False

        for k in top4:
            k = int(k)
            i0 = max(0, k - n + 1)
            i1 = min(n, k + 1)
            if i0 >= i1:
                continue
            idx = np.arange(i0, i1)
            coef = np.maximum(p[k - idx], 0.0)

            src_order = np.argsort(coef)[::-1][:min(4, len(idx))]
            srcs = idx[src_order]

            cands = list(rng.integers(0, n, 50).astype(int))
            low_order = np.argsort(coef)[:min(20, len(idx))]
            cands.extend(idx[low_order].astype(int))
            if idx.size:
                cands.extend(idx[np.argsort(coef)[::-1][:min(5, idx.size)]].astype(int))

            try:
                cands = list(set(cands))
            except Exception:
                cands = list(dict.fromkeys(cands))

            for src in srcs:
                src = int(src)
                best_dst = -1
                best_sens = float("inf")
                for dst in cands:
                    dst = int(dst)
                    if dst == src or dst < 0 or dst >= n:
                        continue
                    sens = 0.0
                    for k2 in top4:
                        d = int(k2) - dst
                        if 0 <= d < n:
                            sens += float(p[d])
                    if sens < best_sens:
                        best_sens = sens
                        best_dst = dst

                if best_dst < 0:
                    continue

                for frac in (0.02, 0.08, 0.20):
                    q = p.copy()
                    amt = q[src] * frac
                    if amt <= 1e-300:
                        continue
                    q[src] = max(0.0, q[src] - amt)
                    q[best_dst] += amt
                    q = _normalize_sequence(q)
                    sc = _score_fast(q)
                    if np.isfinite(sc) and sc < best_score - 1e-12:
                        p = q
                        best_score = sc
                        best_p = q.copy()
                        moved = True
                        break
                if moved:
                    break
            if moved:
                break

        if moved:
            stall = 0
        else:
            stall += 1
            if stall > 35:
                p = best_p.copy()
                noise = rng.normal(0.0, 0.05, n)
                p = p * (1.0 + noise)
                p = _normalize_sequence(p)
                sc = _score_fast(p)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_p = p.copy()
                stall = 0

    final_sc = _score_fast(best_p)
    if np.isfinite(final_sc):
        best_score = min(best_score, final_sc)
    return best_p, best_score


def propose_candidate(budget_s=1800.0):
    start = time.monotonic()
    time_budget = float(max(60.0, min(float(budget_s) - 120.0, 1740.0)))

    parent = None
    parent_score = float("inf")
    try:
        parr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parr.size > 0:
            parent = _normalize_sequence(parr)
            parent_score = _score_fast(parent)
            if np.isfinite(parent_score):
                print(f"Loaded parent: n={len(parent)}, fast_score={parent_score:.8f}")
    except Exception:
        parent = None

    lengths = []
    if parent is not None and 500 <= len(parent) <= 5000:
        lengths.append(len(parent))
    for n in (1600, 2400, 3200, 4000):
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(n for n in lengths if 500 <= n <= 5000))[:5]
    if not lengths:
        lengths = [2400]

    all_seeds = []
    for n in lengths:
        seeds = _generate_seeds_for_length(n, parent)
        all_seeds.extend([(n, p) for (n2, p) in seeds if n2 == n])

    candidates = _select_candidates(all_seeds, max_candidates=6)

    if not candidates:
        n_default = 2400
        best_p = np.ones(n_default, dtype=np.float64) / float(n_default)
        best_score = _score_fast(best_p)
    else:
        best_p, best_score, _ = candidates[0]
        best_p = _normalize_sequence(best_p)
        best_score = _score_fast(best_p)

    print(
        "Initial candidate list: "
        + ", ".join(f"{item[2]}:{item[0]:.6f}" for item in candidates)
    )
    print(f"Best initial: fast_score={best_score:.8f}, n={len(best_p)}")

    rng = np.random.default_rng(20240607)

    # Candidate polishing round.
    max_cands = min(6, len(candidates))
    for i in range(max_cands):
        elapsed = time.monotonic() - start
        remaining = float(time_budget - elapsed)
        if remaining <= 60.0:
            break

        sc, p, n, _label = candidates[i]

        # Skip hopeless unrelated candidates if the best is already strong.
        if i > 0 and np.isfinite(best_score) and sc > best_score * 1.35:
            continue

        if i == 0:
            cand_time = min(remaining * 0.25, 300.0)
        else:
            cand_time = min(remaining * 0.18, 220.0)

        if cand_time <= 15.0:
            cand_time = min(remaining, 30.0)
        if cand_time < 15.0:
            break

        print(f"Polishing candidate {i}: n={n}, seed={sc:.6f}, budget={cand_time:.1f}s")
        p_out, sc_out = _polish_candidate(p, cand_time, rng, f"cand{i}")
        p_out = _normalize_sequence(p_out)
        sc_out = _score_fast(p_out)
        if np.isfinite(sc_out) and sc_out < best_score - 1e-12:
            best_score = sc_out
            best_p = p_out.copy()
            print(f"  NEW BEST: {best_score:.8f}, n={len(best_p)}")

    # L-BFGS polish of current best.
    elapsed = time.monotonic() - start
    remaining = float(time_budget - elapsed)
    if _HAS_SCIPY and remaining > 90.0 and best_p is not None:
        lb_time = min(remaining * 0.40, 300.0)
        print(f"L-BFGS polish: n={len(best_p)}, current={best_score:.8f}, budget={lb_time:.1f}s")
        p_out, sc_out = _lbfgs_polish(best_p, lb_time, rng)
        p_out = _normalize_sequence(p_out)
        sc_out = _score_fast(p_out)
        if np.isfinite(sc_out) and sc_out < best_score - 1e-12:
            best_score = sc_out
            best_p = p_out.copy()
            print(f"  NEW BEST after L-BFGS: {best_score:.8f}, n={len(best_p)}")

    # Targeted transfer refinement.
    elapsed = time.monotonic() - start
    remaining = float(time_budget - elapsed)
    if remaining > 70.0 and best_p is not None:
        tr_time = min(remaining * 0.25, 180.0)
        print(f"Targeted transfer refine: n={len(best_p)}, current={best_score:.8f}, budget={tr_time:.1f}s")
        p_out, sc_out = _targeted_transfer_refine(best_p, tr_time, rng)
        p_out = _normalize_sequence(p_out)
        sc_out = _score_fast(p_out)
        if np.isfinite(sc_out) and sc_out < best_score - 1e-12:
            best_score = sc_out
            best_p = p_out.copy()
            print(f"  NEW BEST after transfer: {best_score:.8f}, n={len(best_p)}")

    # Final long mirror polish.
    elapsed = time.monotonic() - start
    remaining = float(time_budget - elapsed)
    if remaining > 35.0 and best_p is not None:
        final_time = min(remaining * 0.90, 600.0)
        print(f"Final mirror polish: n={len(best_p)}, current={best_score:.8f}, budget={final_time:.1f}s")
        p_out, sc_out = _polish_candidate(best_p, final_time, rng, "final")
        p_out = _normalize_sequence(p_out)
        sc_out = _score_fast(p_out)
        if np.isfinite(sc_out) and sc_out < best_score - 1e-12:
            best_score = sc_out
            best_p = p_out.copy()
            print(f"  NEW BEST final: {best_score:.8f}, n={len(best_p)}")

    if best_p is None:
        best_p = np.ones(2400, dtype=np.float64) / 2400.0
        best_score = _score_fast(best_p)

    # Final safety normalization and validation.
    final_p = np.clip(np.asarray(best_p, dtype=np.float64).ravel(), 0.0, 1000.0)
    s = float(np.sum(final_p))
    if s <= 0.01:
        final_p = np.ones(2400, dtype=np.float64) / 2400.0
        s = 1.0
    final_p = final_p / s

    final_list = [float(x) for x in final_p]
    try:
        official_score = float(evaluate_sequence(final_list))
    except Exception:
        official_score = float("inf")

    if not np.isfinite(official_score):
        # Last-resort valid fallback.
        final_p = np.ones(2400, dtype=np.float64) / 2400.0
        final_list = [float(x) for x in final_p]
        official_score = 2.0

    print(f"FINAL: fast_score={_score_fast(final_p):.8f}, official_score={official_score:.8f}, n={len(final_list)}, time={time.monotonic()-start:.1f}s")
    return final_list
