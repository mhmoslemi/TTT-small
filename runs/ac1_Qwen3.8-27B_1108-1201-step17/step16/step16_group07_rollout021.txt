def fast_autocorr(a):

def normalize_to(a, n):

    s = a.sum()

def subgradient_at_k(a, k, n):
    """Subgradient of conv(a,a)[k] w.r.t. a."""
    grad = np.zeros(n)
    # d/d a[m] of sum_j a[j]*a[k-j] = a[k-m] (if valid) + a[m-k] (if valid)
    for m in range(n):
        val = 0.0
        km = k - m
        if 0 <= km < n:
            val += a[km]
        mk = m - k
        if 0 <= mk < n:
            val += a[mk]
        grad[m] = val
    return grad


def subgradient_at_k_fast(a, k, n):
    """Vectorized subgradient of conv(a,a)[k] w.r.t. a."""
    grad = np.zeros(n)
    km = k
    for m in range(n):
        val = 0.0
        idx1 = k - m
        if 0 <= idx1 < n:
            val += a[idx1]
        idx2 = m - k
        if 0 <= idx2 < n:
            val += a[idx2]
        grad[m] = val
    return grad


def subgradient_max(a, tol_frac=0.01, n_top=10):
    """Compute weighted subgradient at the peaks of conv(a,a)."""
    n = len(a)
    conv = fast_autocorr(a)
    M = conv.max()
    if M < 1e-15:
        return np.zeros(n), M
    threshold = M * (1.0 - tol_frac)
    active = np.where(conv >= threshold)[0]
    if len(active) < n_top:
        active = np.argsort(conv)[-n_top:]
    # Weight by proximity to max
    weights = (conv[active] - threshold) / max(M - threshold, 1e-15)
    weights = weights / max(weights.sum(), 1e-15)
    grad = np.zeros(n)
    for w, k in zip(weights, active):
        grad += w * subgradient_at_k_fast(a, int(k), n)
    return grad, M


def lp_best_response(a_fixed, M, active_set, solver):
    """Solve LP: max sum(b) s.t. (a_fixed * b)[k] <= M for k in active_set, b >= 0."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    if size == 0:
        return None, None

    # Build constraint matrix (sparse banded structure)
    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        for m in range(n):
            val = 0.0
            km = k - m
            if 0 <= km < n:
                val += a_fixed[km]
            mk = m - k
            if 0 <= mk < n:
                val += a_fixed[mk]
            T[idx, m] = val

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def lp_best_response_vec(a_fixed, M, active_set, solver):
    """Faster LP using vectorized constraint construction."""
    n = len(a_fixed)
    k_indices = np.asarray(active_set, dtype=int)
    size = len(k_indices)
    if size == 0 or size > 3 * n:
        return None, None

    T = np.zeros((size, n))
    for idx in range(size):
        k = int(k_indices[idx])
        # For each m, T[idx,m] = a[k-m]*1{0<=k-m<n} + a[m-k]*1{0<=m-k<n}
        # First term: a[k-m] for m in [max(0,k-n+1), min(n,k+1)]
        lo1 = max(0, k - n + 1)
        hi1 = min(n, k + 1)
        if lo1 < hi1:
            ms = np.arange(lo1, hi1)
            T[idx, ms] += a[k - ms]
        # Second term: a[m-k] for m in [max(0,k), min(n, k+n)]
        lo2 = max(0, k)
        hi2 = min(n, k + n)
        if lo2 < hi2:
            ms = np.arange(lo2, hi2)
            T[idx, ms] += a[ms - k]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

def coordinate_descent(a0, time_limit, verbose=False, seed=0):
    """Peak-focused coordinate descent."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 6

    while time.monotonic() - t0 < time_limit:
        if no_improve >= max_no_improve:
            break
        no_improve += 1

        conv = fast_autocorr(a)
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            typical = norm / n
            candidates = [0.0, aj * 0.5, aj * 1.0, aj * 1.5, typical, typical * 2.0]
            if aj > 0:
                candidates.append(aj * 0.1)
            candidates = sorted(set([max(0.0, min(c, norm * 0.3)) for c in candidates]))

            best_x = aj
            best_obj = best_score

            for x in candidates:
                if abs(x - aj) < 1e-14:
                    continue
                a[j] = x
                s_new = a.sum()
                if s_new < 1e-15:
                    a[j] = aj
                    continue
                a_scaled = a * norm / s_new
                sc = fast_autocorr(a_scaled).max()
                if sc < best_obj:
                    best_obj = sc
                    best_x = x
            a[j] = best_x

            if best_x != aj:
                a = a * norm / a.sum()
                actual = fast_autocorr(a).max()
                if actual < best_score - 1e-14:
                    best_score = actual
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                else:
                    a = best_a.copy()

        if verbose and no_improve % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD no_improve={no_improve} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def gen_antisymmetric_init(n, seed=0):
    """Generate anti-symmetric perturbation of uniform."""
    rng = np.random.default_rng(seed)
    c = 1.0
    d = rng.standard_normal(n)
    # Make anti-symmetric
    for i in range(n // 2):
        d[i] = rng.standard_normal()
        d[n - 1 - i] = -d[i]
    if n % 2 == 1:
        d[n // 2] = 0.0
    # Scale d so that c/2 + d[i] >= 0
    max_d = np.abs(d).max()
    if max_d > 0:
        d = d * (0.4 * c / max_d)
    a = c / 2 + d
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-15:
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def gen_smooth_random(n, seed=0, bandwidth=0.1):
    """Generate a smooth random sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    # Sum of random Gaussians
    a = np.zeros(n)
    for _ in range(5):
        center = rng.random()
        width = bandwidth * (0.5 + rng.random())
        a += np.exp(-0.5 * ((x - center) / width) ** 2) * (0.5 + rng.random())
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def gen_msequence(n, seed=0):
    """Generate a sequence inspired by m-sequence structure."""
    rng = np.random.default_rng(seed)
    # Start with binary m-seq-like pattern, then smooth
    m = n
    # Use a pseudo-random construction
    a = np.zeros(m)
    # Place "1"s in a pattern with good spread
    step = max(1, m // 8)

    # Add some intermediate values

    a_smooth = np.convolve(a, kernel, mode='same')

    a_smooth = a_smooth / a_smooth.sum()
    return a_smooth


def gen_frequency_init(n, seed=0):
    """Generate sequence by maximizing spectral flatness of FFT(a)^2."""
    rng = np.random.default_rng(seed)
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.maximum(a, 0.1)
    a = a / a.sum()
    # A few iterations to flatten FFT(a)^2
    for _ in range(10):
        A = rfft(a, 2 * n - 1)
        A2 = A * A
        # Want ifft(A2) to be flat -> want |A2| roughly constant in freq
        # Gradient: increase a where it reduces variance of ifft(A2)
        conv = irfft(A2, 2 * n - 1)[:2 * n - 1]
        target = conv.mean()
        # Push towards flat
        error = conv - target
        # Gradient of w.r.t. a: approximately (convolve_error_with_a)
        a_new = a - 0.1 * np.convolve(error, a, mode='same')[:n]
        a_new = np.maximum(a_new, 0.01)
        a_new = a_new / a_new.sum()
        a = a_new
    return a

    perturbed = perturbed / s
    a_norm = normalize_to(perturbed, n)
    a_out, _ = nesterov_lp_optimize(a_norm, time_limit, solver, verbose=False, seed=seed)
    return a_out


def propose_candidate():
    """
    Main search: diverse init -> LP Nesterov -> subgradient -> coordinate descent -> restarts.
    """
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = 1770.0

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.8f}")

    # === Phase 1: Diverse initialization (150s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 140.0)
    if t1 > 30:
        print(f"\n=== Phase 1: Diverse init, budget={t1:.0f}s ===")
        inits = []
        inits.append(("parent", best_seq.copy()))
        inits.append(("antisym", gen_antisymmetric_init(n_base, 1)))
        inits.append(("smooth", gen_smooth_random(n_base, 2, 0.08)))
        inits.append(("mseq", gen_msequence(n_base, 3)))
        inits.append(("freq", gen_frequency_init(n_base, 4)))
        inits.append(("sine", np.maximum(1.0 + 0.2 * np.cos(2 * np.pi * 7 / n_base * np.arange(n_base)), 0.1)))
        inits[-1] = ("sine", inits[-1][1] / inits[-1][1].sum())
        inits.append(("antisym2", gen_antisymmetric_init(n_base, 99)))
        inits.append(("smooth2", gen_smooth_random(n_base, 77, 0.15)))

        # Quick evaluate each
        init_scores = []
        for name, seq in inits:
            sc = evaluate_safe(seq)
            init_scores.append((sc, name, seq))
            print(f"    {name}: {sc:.8f}")
        init_scores.sort()
        for sc, name, seq in init_scores[:4]:
            print(f"    Best init: {name} = {sc:.8f}")

        # Run short LP on top 3 inits (excluding parent which is already optimized)
        t_per_init = t1 / 4
        for i, (sc, name, seq) in enumerate(init_scores[:4]):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_alloc = min(t_per_init, remaining - 20)
            if t_alloc < 20:
                break
            print(f"    Optimizing init '{name}' (budget={t_alloc:.0f}s)")
            a_norm = normalize_to(seq, n_base)
            a_out, _ = nesterov_lp_optimize(a_norm, t_alloc, solver, verbose=False, seed=100 + i)
            best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"init_{name}", t_global)

    # === Phase 2: Main LP Nesterov (500s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP Nesterov n={n_cur}, budget={t2:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True, seed=42)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp_main_n{n_cur}", t_global)

    # === Phase 3: Subgradient refinement (150s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 30:
        print(f"\n=== Phase 3: Subgradient n={n_cur}, budget={t3:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = subgradient_optimize(a_norm, t3, verbose=True, seed=123)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"subg_n{n_cur}", t_global)

    # === Phase 4: LP Nesterov round 2 (300s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 60:
        print(f"\n=== Phase 4: LP Nesterov r2 n={n_cur}, budget={t4:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True, seed=777)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"lp_r2_n{n_cur}", t_global)

    # === Phase 5: Coordinate descent (150s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20:
        print(f"\n=== Phase 5: CD n={n_cur}, budget={t5:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent(a_norm, t5, verbose=True, seed=55)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cd_n{n_cur}", t_global)

    # === Phase 6: Subgradient round 2 (100s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 20:
        print(f"\n=== Phase 6: Subgrad r2 n={n_cur}, budget={t6:.0f}s ===")
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = subgradient_optimize(a_norm, t6, verbose=False, seed=314)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"subg_r2_n{n_cur}", t_global)

    # === Phase 7: Perturbation restarts (200s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_restarts = 3
    t7_total = min(remaining * 0.18, 200.0)
    t7_each = t7_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        t7 = min(t7_each, remaining - 15)
        if t7 < 15:
            break
        print(f"\n=== Phase 7r{r}: Perturb+LP n={n_cur}, budget={t7:.0f}s ===")
        a_out = perturb_and_optimize(best_seq, t7, solver, seed=500 + r * 137)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"pert{r}_n{n_cur}", t_global)

    # === Phase 8: Final CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 25:
        t8 = min(remaining * 0.5, 60.0)
        print(f"\n=== Phase 8: Final CD n={len(best_seq)}, budget={t8:.0f}s ===")
        n_cur = len(best_seq)
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = coordinate_descent(a_norm, t8, verbose=False, seed=999)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"cdFinal_n{n_cur}", t_global)

    # === Phase 9: Final subgradient micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t9 = min(remaining * 0.6, 40.0)
        n_cur = len(best_seq)
        a_norm = normalize_to(best_seq, n_cur)
        a_out, _ = subgradient_optimize(a_norm, t9, verbose=False, seed=271)
        best_seq, best_score, _ = try_update(best_seq, best_score, a_out, f"subgFinal_n{n_cur}", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
