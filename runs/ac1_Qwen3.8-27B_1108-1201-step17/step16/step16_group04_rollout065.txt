"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for a non-negative sequence a.

Strategy (differently tuned from the prior LP-FW baseline):
  1. FFT-based linear autocorrelation and an analytically-verified O(n log n)
     gradient of the weighted-autocorrelation objective.
  2. Nesterov-accelerated smoothed (log-sum-exp / soft-max) gradient descent
     on the simplex, with adaptive temperature and adaptive, self-normalizing
     step length, plus periodic perturbed restarts.
  3. Peak-targeted coordinate refinement (move mass off peak-contributing
     coordinates onto low-sensitivity coordinates) with exact line search.
  4. Multi-length exploration via spectral interpolation so we do not get stuck
     in a single local minimum length.

Only numpy is required.  evaluate_sequence is assumed to be pre-imported.
"""

import numpy as np
import time


# ---------------------------------------------------------------------------
# Core numerical helpers
# ---------------------------------------------------------------------------

def fast_autocorr(a):
    """Linear autocorrelation via FFT.  O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)
    return out[:size]


def compute_grad(a, w):
    """
    Gradient of sum_k w[k] * ac[k] with respect to a, where ac = a * a is the
    linear autocorrelation.  Verified identity:
        grad[i] = 2 * (a_rev * w)[n-1 + i]
    computed with zero-padded linear convolution (FFT).
    """
    n = len(a)
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = np.fft.rfft(a[::-1], fft_size)
    W = np.fft.rfft(w, fft_size)
    result = np.fft.irfft(A_rev * W, fft_size)
    return 2.0 * result[n - 1:n - 1 + n]


def normalize(a):
    """Project onto the simplex {a >= 0, sum(a) = 1} via clip + rescale."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length, keeping sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    if f.sum() < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / s


def spectral_equalize(a, n_iter=15):
    """Iterative DFT power flattening (real, non-negative, sum=1)."""
    a = normalize(a).copy()
    n = len(a)
    eps = 1e-8
    for _ in range(n_iter):
        A = np.fft.fft(a)
        P = A.real ** 2 + A.imag ** 2
        C = 1.0 / np.sqrt(P + eps)
        A_new = A * C
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = 0.5 * a + 0.5 * a_new
    return normalize(a)

# Optimizer 1: Nesterov smoothed gradient descent on the simplex
# ---------------------------------------------------------------------------

def smooth_grad_desc(a0, time_limit, beta_start=400.0, beta_end=25000.0,
                     c0=0.6, c_scale=4000, restart_after=600,
                     max_t=40000000, verbose=True):
    """
    Minimizes the soft-max surrogate (1/beta)*log(sum(exp(beta*ac))) which
    converges to max(ac) as beta grows.  Nesterov acceleration, adaptive
    self-normalizing step, and perturbed restarts on stall.
    Returns (best_a, best_max_ac).
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    stall = 0

    for t in range(1, max_t):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            mu = (t - 1.0) / (t + 1.5)
            y = x + mu * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m:
            best_m = M_curr
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive temperature (anneal from smooth to sharp)
        prog = min(1.0, t / 30000.0)
        beta = beta_start + (beta_end - beta_start) * prog

        # Stable softmax weights over the lags
        ac_s = ac - ac.max()
        w = np.exp(beta * ac_s)

        if ws < 1e-300:

        else:
            w = w / ws

        gmax = float(np.max(np.abs(grad)))

        x_new = y - eta * (grad / gmax)
        x_new = normalize(x_new)

        x_prev = y
        x = x_new

        # Restart on long stall

            J = 2.0 * n * best_m
            print("    SGD t=%d (%.1fs): J=%.6f beta=%.0f stall=%d" %
                  (t, elapsed, J, beta, stall))

    return best_a, best_m

def _peak_sensitivity(a, top_k):
    """Two-sided per-coordinate sensitivity to the given peak lags."""
    n = len(a)
    sens = np.zeros(n)
    for k2 in top_k:
        k2 = int(k2)
        idx = np.arange(n)
        d = k2 - idx
        v = (d >= 0) & (d < n)
        sens[v] += a[d[v]]
        d2 = idx - k2
        v2 = (d2 >= 0) & (d2 < n)
        sens[v2] += a[d2[v2]]
    return sens

        state[1] = a.copy()
        print("  *** NEW BEST [%s]: score=%.6f n=%d t=%.0fs" %
