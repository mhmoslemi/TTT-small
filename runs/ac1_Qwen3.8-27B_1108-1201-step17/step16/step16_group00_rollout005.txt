def line_search_fw(a, b, ac_a, ac_ab, ac_b, n):
    """Find best tau in [0,1] for FW step."""
    def J_of_tau(tau):
        C = (1-tau)**2 * ac_a + 2*(1-tau)*tau * ac_ab + tau**2 * ac_b
        return 2 * n * C.max()
    
    # Golden section search
    golden_ratio = (np.sqrt(5) - 1) / 2
    a_lo, a_hi = 0.0, 1.0
    b_pt = a_hi - golden_ratio * (a_hi - a_lo)
    c_pt = a_lo + golden_ratio * (a_hi - a_lo)
    f_b = J_of_tau(b_pt)
    f_c = J_of_tau(c_pt)
    
    for _ in range(50):
        if f_b < f_c:
            a_hi = c_pt
            c_pt = b_pt
            f_c = f_b
            b_pt = a_hi - golden_ratio * (a_hi - a_lo)
            f_b = J_of_tau(b_pt)
        else:
            a_lo = b_pt
            b_pt = c_pt
            f_b = f_c
            c_pt = a_lo + golden_ratio * (a_hi - a_lo)
            f_c = J_of_tau(c_pt)
    
    tau = (a_lo + a_hi) / 2
    return tau

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe with full constraint set and golden section line search
2. Multi-length exploration (3200, 3600, 4000, 4800)
3. Accelerated Mirror Descent (AMD) for polishing
4. Local random search for final improvement

import numpy as np
from scipy import sparse
import cvxpy as cp
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def build_lp_indices(n):
    """Build row/col index arrays for the (2n-1) x n constraint matrix."""
    n_rows = 2 * n - 1
    # For row k, nonzeros at cols i where max(0, k-n+1) <= i <= min(k, n-1)
    # Number of nonzeros in row k = min(k, n-1) - max(0, k-n+1) + 1
    row_lengths = np.minimum(np.arange(n_rows), n - 1) - np.maximum(np.arange(n_rows) - n + 1, 0) + 1
    total_nz = int(row_lengths.sum())
    
    rows = np.empty(total_nz, dtype=np.int32)
    cols = np.empty(total_nz, dtype=np.int32)
    
    pos = 0
    for k in range(n_rows):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        length = i_end - i_start + 1
        rows[pos:pos+length] = k
        cols[pos:pos+length] = np.arange(i_start, i_end + 1)
        pos += length
    
    return rows, cols

def solve_fw_lp(a, M, rows, cols, solver):
    """Solve the FW LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Compute values: A[k, i] = a[k-i]
    vals = a[rows.astype(np.intp) - cols.astype(np.intp)]
    
    A_mat = sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [A_mat @ b_var <= M])
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def golden_section_J(a, b, ac_a, ac_ab, ac_b, n):
    """Golden section search for best tau in [0,1].
    J(tau) = 2n * max_k [C_k(tau)] where C_k(tau) is the conv of (1-tau)*a+tau*b.
    """
    def J_val(tau):
        C = (1.0-tau)**2 * ac_a + 2.0*(1.0-tau)*tau * ac_ab + tau**2 * ac_b
        return C.max()
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = J_val(x1)
    f2 = J_val(x2)
    
    for _ in range(60):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - gr * (hi - lo)
            f1 = J_val(x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = J_val(x2)
    
    return (lo + hi) / 2.0

        # And conv(b,a) = reverse(conv(a,b)), so for the max, conv(a,b) and conv(b,a) have the same max

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Evaluate
        J_new = 2.0 * n * fast_autocorr(a_new).max()
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        
        a = a_new
        
        elapsed = time.monotonic() - t0
        if verbose and iteration % 10 == 0:
            step_time = time.monotonic() - step_start
            print(f"    FW iter {iteration} ({elapsed:.0f}s, {step_time:.1f}s/iter): J={best_J:.6f}")
    
    return best_a, best_J
