```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for nonnegative a.

Strategy:
- Work with the probability simplex (sum(a)=1, a>=0).
- Use a multiplicative/log-domain Adam-style optimizer with the exact
  softmax-gradient of a smoothed max-convolution objective.
- Use exact FFT-based convolutions and gradients.
- Polish with L-BFGS-B on the same smoothed objective.
- Use a short coordinate polish at the end.
- Multi-resolution refinement: current length, ~3072, ~4096.
"""

import numpy as np
import time

try:
    from scipy import optimize as _scipy_optimize
    HAVE_SCIPY = True
except Exception:
    _scipy_optimize = None
    HAVE_SCIPY = False

try:
    np.seterr(under="ignore", over="ignore")
except Exception:
    pass

_N1_CACHE = {}
_N2_CACHE = {}
_H_info = {"t_start": 0.0, "time_limit": 0.0, "best_score": float("inf"), "best_a": None}


def next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def get_N1(n):
    v = _N1_CACHE.get(n)
    if v is None:
        v = next_pow2(2 * int(n) - 1)
        _N1_CACHE[n] = v
    return v


def get_N2(n):
    v = _N2_CACHE.get(n)
    if v is None:
        v = next_pow2(3 * int(n) - 2)
        _N2_CACHE[n] = v
    return v


def normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def softmax(x):
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    if n == 0:
        return np.ones(0, dtype=np.float64)
    x = x - np.mean(x)
    m = x.max()
    if not np.isfinite(m):
        return np.ones(n, dtype=np.float64) / n
    e = np.exp(x - m)
    s = e.sum()
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / n
    return e / s


def fast_conv(a, N=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.ones(0, dtype=np.float64)
    if N is None:
        N = get_N1(n)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)
    return out[:2 * n - 1]


def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    s = a.sum()
    if s <= 1e-300 or not np.isfinite(s):
        return float("inf")
    if abs(s - 1.0) > 1e-12:
        a = a / s
    a = np.maximum(a, 0.0)
    n = a.size
    c = fast_conv(a)
    if c.size == 0:
        return float("inf")
    M = float(c.max())
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * n * M)


def compute_c_h(a, beta, use_factor=False, N=None):
    """
    Computes:
      c = linear convolution a*a
      M = max(c)
      h = gradient of smoothed max wrt a, using weights on c
      F = smoothed max objective (if beta is positive)
      w = weights over c_lags
    If use_factor is True, 'beta' is interpreted as desired zmax = beta / M.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    K = 2 * n - 1
    if N is None:
        N = get_N2(n)

    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:K]
    M = float(c.max())

    if M <= 0.0 or not np.isfinite(M):
        w = np.ones(K, dtype=np.float64) / K
        h = a + (1.0 / K if K > 0 else 0.0)
        F = M
        return c, M, h, F, w

    if use_factor:
        beta = float(beta) / M
    beta = float(beta)

    if beta <= 0.0 or not np.isfinite(beta):
        w = np.ones(K, dtype=np.float64) / K
        F = M
    else:
        z = beta * (c - M)
        ez = np.exp(z)
        s = ez.sum()
        if s <= 1e-300 or not np.isfinite(s):
            w = np.ones(K, dtype=np.float64) / K
            F = M
        else:
            w = ez / s
            F = M + (np.log(s) - np.log(float(K))) / beta

    w_rev = w[::-1]
    W = np.fft.rfft(w_rev, N)
    corr = np.fft.irfft(W * A, N)
    d = corr[n - 1:2 * n - 1][::-1]
    h = a + d

    if not np.all(np.isfinite(h)):
        h = np.ones(n, dtype=np.float64) / n

    return c, M, h, F, w


def soft_F_grad_a(a, beta, N=None):
    a = np.asarray(a, dtype=np.float64)
    c, M, h, F, w = compute_c_h(a, beta, use_factor=False, N=N)
    return F, h


def spectral_equalize(a, n_iter=20, eps=1e-12):
    a = normalize(a).copy()
    n = a.size
    if n == 0:
        return a
    for _ in range(int(n_iter)):
        A = np.fft.fft(a)
        P = A.real * A.real + A.imag * A.imag
        C = 1.0 / np.sqrt(P + eps)
        A2 = A * C
        a2 = np.real(np.fft.ifft(A2))
        a2 = np.maximum(a2, 0.0)
        s = a2.sum()
        if s <= 1e-300:
            break
        a2 = a2 / s
        a = 0.7 * a + 0.3 * a2
    return normalize(a)


def interpolate_sequence(f, new_n):
    f = normalize(f)
    old_n = f.size
    new_n = int(new_n)
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        return np.ones(new_n, dtype=np.float64) / new_n
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, f)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s <= 1e-300:
        return np.ones(new_n, dtype=np.float64) / new_n
    return out / s


def default_initial(n=2400):
    x = np.linspace(0.0, 1.0, int(n))
    a = 0.5 + 0.5 * np.sin(np.pi * x)
    a = np.maximum(a, 1e-6)
    return normalize(a)


def random_smooth_start(n, rng, base=None, sigma=0.8):
    n = int(n)
    x = np.linspace(0.0, 1.0, n)
    window = 0.7 + 0.3 * np.sin(np.pi * x)
    a = window * np.exp(sigma * rng.standard_normal(n))
    if base is not None:
        b = interpolate_sequence(base, n)
        a = 0.5 * a + 0.5 * b
    return normalize(a)


def mult_log_descent(a0, time_limit, rng,
                     beta_factor_start=4.0, beta_factor_end=30.0,
                     lr0=0.015, lr_min=0.0005,
                     beta1=0.9, beta2=0.999,
                     verbose=True):
    a0 = normalize(a0)
    n = a0.size
    if n == 0 or time_limit <= 0:
        return a0, score_fast(a0)

    a = np.maximum(a0, 1e-300)
    s = a.sum()
    a = a / s

    x = np.log(a) - np.mean(np.log(a))
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    best_a = a.copy()
    best_x = x.copy()
    best_score = score_fast(a)

    N = get_N2(n)
    eps = 1e-8
    lr = float(lr0)
    stall = 0
    t = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < float(time_limit):
        t += 1

        if t % 1000 == 0:
            lr = max(lr_min, lr * 0.985)

        frac = min(1.0, t / 5000.0)
        bf = beta_factor_start + (beta_factor_end - beta_factor_start) * frac

        c, M, h, F, w = compute_c_h(a, bf, use_factor=True, N=N)
        cur_score = 2.0 * n * M

        if np.isfinite(cur_score) and cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            best_x = x.copy()
            stall = 0
            if verbose and t % 1000 == 0:
                print(f"    [mult t={t}] best={best_score:.6f} bf={bf:.1f}")
        else:
            stall += 1

        if verbose and t % 5000 == 0:
            print(f"    [mult t={t}] cur={cur_score:.6f} best={best_score:.6f} lr={lr:.5f} bf={bf:.1f}")

        if stall >= 400:
            noise_scale = 0.01 if n < 3000 else 0.007
            x = best_x + noise_scale * rng.standard_normal(n)
            x -= np.mean(x)
            a = softmax(x)
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            lr = max(lr_min, lr * 0.75)
            stall = 0
            if verbose:
                print(f"    [mult reset t={t}] lr={lr:.5f}")
            continue

        h_dot_a = np.dot(h, a)
        u = a * (h - h_dot_a)

        m = beta1 * m + (1.0 - beta1) * u
        v = beta2 * v + (1.0 - beta2) * (u * u)

        if t <= 100:
            f1 = 1.0 - beta1 ** t
            f2 = 1.0 - beta2 ** t
        else:
            f1 = 1.0
            f2 = 1.0

        step = lr * (m / f1) / (np.sqrt(v / f2) + eps)
        max_step = 0.15
        np.clip(step, -max_step, max_step, out=step)

        x = x - step
        x -= np.mean(x)
        a = softmax(x)

    cur = score_fast(a)
    if cur < best_score:
        best_score = cur
        best_a = a.copy()

    if verbose:
        print(f"    [mult done t={t}] best={best_score:.6f}")

    return best_a, best_score


def lbfgs_fun(x, beta, scale):
    try:
        a = softmax(x)
        F, h = soft_F_grad_a(a, beta)
        val = scale * F
        if not np.isfinite(val):
            return 1e100
        return float(val)
    except Exception:
        return 1e100


def lbfgs_jac(x, beta, scale):
    try:
        a = softmax(x)
        F, h = soft_F_grad_a(a, beta)
        g = a * (h - np.dot(h, a))
        if not np.all(np.isfinite(g)):
            return np.zeros_like(x, dtype=np.float64)
        return scale * g
    except Exception:
        return np.zeros_like(np.asarray(x, dtype=np.float64), dtype=np.float64)


def _lbfgs_callback(xk):
    info = _H_info
    try:
        if time.monotonic() - info["t_start"] > info["time_limit"]:
            return True
        a = softmax(xk)
        sc = score_fast(a)
        if np.isfinite(sc) and sc < info["best_score"]:
            info["best_score"] = sc
            info["best_a"] = a.copy()
    except Exception:
        pass
    return False


def lbfgs_polish(a0, time_limit, beta_factor=8.0, maxiter=300, verbose=True):
    if not HAVE_SCIPY or time_limit <= 0:
        return None

    a0 = normalize(a0)
    n = a0.size
    if n == 0:
        return a0

    M0 = float(fast_conv(a0).max())
    if M0 <= 0 or not np.isfinite(M0):
        return a0

    beta = float(beta_factor) / M0
    scale = 10.0 * (float(n) ** 1.5)

    a_pos = np.maximum(a0, 1e-300)
    x = np.log(a_pos) - np.mean(np.log(a_pos))

    info = _H_info
    info["t_start"] = time.monotonic()
    info["time_limit"] = float(time_limit)
    info["best_score"] = score_fast(a0)
    info["best_a"] = a0.copy()

    try:
        res = _scipy_optimize.minimize(
            lbfgs_fun,
            x,
            args=(beta, scale),
            jac=lbfgs_jac,
            method="L-BFGS-B",
            options={
                "maxiter": int(maxiter),
                "maxls": 100,
                "ftol": 1e-12,
                "gtol": 1e-8,
                "disp": False,
            },
            callback=_lbfgs_callback,
        )
    except Exception as e:
        if verbose:
            print(f"    [lbfgs error] {e}")
        return info["best_a"]

    best_a = info["best_a"]
    best_sc = info["best_score"]

    x2 = None
    if res is not None:
        x2 = getattr(res, "x", None)

    if x2 is None:
        x2 = x

    try:
        a2 = softmax(np.asarray(x2, dtype=np.float64))
        sc2 = score_fast(a2)
        if np.isfinite(sc2) and sc2 < best_sc:
            best_a = a2.copy()
            best_sc = sc2
    except Exception:
        pass

    info["best_a"] = best_a
    info["best_score"] = best_sc

    if verbose:
        print(f"    [lbfgs n={n}] best={best_sc:.6f}")

    return best_a


def coordinate_polish(a0, time_limit, rng, verbose=True):
    a = normalize(a0).copy()
    n = a.size
    if n == 0 or time_limit <= 0:
        return a, score_fast(a)

    best_a = a.copy()
    best_sc = score_fast(a)

    N1 = get_N1(n)
    N2 = get_N2(n)

    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < float(time_limit):
        c, M, h, F, w = compute_c_h(a, 30.0, use_factor=True, N=N2)

        if not np.all(np.isfinite(h)):
            break

        src_order = np.argsort(h)[::-1][:8]
        dst_order = np.argsort(h)[:12]

        src_candidates = []
        for i in src_order:
            i = int(i)
            if a[i] > 1e-10:
                src_candidates.append(i)
            if len(src_candidates) >= 4:
                break

        dst_candidates = []
        bad = set(src_candidates)
        for i in dst_order:
            i = int(i)
            if i not in bad:
                dst_candidates.append(i)
            if len(dst_candidates) >= 8:
                break

        if len(dst_candidates) > 1:
            perm = rng.permutation(len(dst_candidates))
            dst_candidates = [dst_candidates[int(i)] for i in perm[:8]]

        improved = False
        for si in src_candidates:
            base_amt = a[si]
            if base_amt <= 1e-12:
                continue
            for di in dst_candidates:
                for frac in (0.003, 0.001, 0.01, 0.005):
                    amt = base_amt * frac
                    if amt <= 1e-15:
                        continue
                    trial = a.copy()
                    trial[si] -= amt
                    trial[di] += amt
                    if trial[si] < -1e-15:
                        continue
                    if trial[si] < 0.0:
                        trial[si] = 0.0
                    sc = 2.0 * n * float(np.max(fast_conv(trial, N1)))
                    if np.isfinite(sc) and sc < best_sc - 1e-14:
                        best_sc = sc
                        best_a = trial.copy()
                        a = trial
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            si = int(rng.integers(n))
            di = int(rng.integers(n))
            if si != di and a[si] > 1e-12:
                for frac in (0.001, 0.0005, 0.0002):
                    amt = a[si] * frac
                    trial = a.copy()
                    trial[si] -= amt
                    trial[di] += amt
                    if trial[si] < 0.0:
                        continue
                    sc = 2.0 * n * float(np.max(fast_conv(trial, N1)))
                    if np.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_a = trial.copy()
                        a = trial
                        improved = True
                        stall = 0
                        break

        if not improved:
            stall += 1
            if stall > 80:
                a = best_a.copy()
                for _ in range(3):
                    si = int(rng.integers(n))
                    di = int(rng.integers(n))
                    if si != di and a[si] > 1e-12:
                        amt = a[si] * 0.001
                        a[si] -= amt
                        a[di] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-300:
                    a = a / s
                stall = 0
            if stall > 400:
                break

    if verbose:
        print(f"    [coord] best={best_sc:.6f}")

    return best_a, best_sc


def consider_candidate(best_a, best_score, cand, label, verbose=True):
    if cand is None:
        return best_a, best_score
    cand = normalize(cand)
    if cand.size == 0:
        return best_a, best_score
    sc = score_fast(cand)
    if np.isfinite(sc) and sc < best_score:
        if verbose:
            print(f"*** improve [{label}]: score={sc:.6f} n={cand.size}")
        return cand.copy(), sc
    return best_a, best_score


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    limit = max(60.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(20240501)

    best_a = None
    best_score = float("inf")

    try:
        par = np.asarray(height_sequence_1, dtype=np.float64)
        if par.ndim != 1:
            par = par.ravel()
        par = np.clip(par, 0.0, 1000.0)
        if par.size > 0 and np.all(np.isfinite(par)) and par.sum() > 1e-12:
            par = par / par.sum()
            best_a = par.copy()
            best_score = score_fast(par)
            print(f"Loaded parent: n={best_a.size}, score={best_score:.6f}")
    except Exception:
        pass

    if best_a is None:
        best_a = default_initial(2400)
        best_score = score_fast(best_a)
        print(f"Default initial: n={best_a.size}, score={best_score:.6f}")

    if best_a.size > 6000:
        best_a = interpolate_sequence(best_a, 4096)
        best_score = score_fast(best_a)
        print(f"Downsampled long input to n={best_a.size}, score={best_score:.6f}")

    # Quick spectral preprocessing on the current sequence.
    rem = limit - (time.monotonic() - t_start)
    if rem > 30 and best_a.size <= 5000:
        a_spec = spectral_equalize(best_a, 20)
        best_a, best_score = consider_candidate(best_a, best_score, a_spec, "spectral0", verbose=True)

    # Phase 1: optimize current length.
    rem = limit - (time.monotonic() - t_start)
    if rem > 80 and best_a.size <= 5000:
        t_mult = min(rem * 0.12, 180.0)
        if t_mult > 20:
            a2, _ = mult_log_descent(
                best_a, t_mult, rng,
                beta_factor_start=4.0, beta_factor_end=20.0,
                lr0=0.02, lr_min=0.001, verbose=True
            )
            best_a, best_score = consider_candidate(best_a, best_score, a2, "mult0", verbose=True)

        rem = limit - (time.monotonic() - t_start)
        if rem > 40:
            t_lb = min(rem * 0.06, 80.0)
            a2 = lbfgs_polish(best_a, t_lb, beta_factor=8.0, maxiter=300, verbose=True)
            best_a, best_score = consider_candidate(best_a, best_score, a2, "lbfgs0", verbose=True)

    # Targets for larger search.
    targets = []
    if best_a.size < 3072:
        targets.append(3072)
    if best_a.size < 4096:
        targets.append(4096)

    for n_tgt in targets:
        rem = limit - (time.monotonic() - t_start)
        if rem <= 120:
            break

        a_tgt = interpolate_sequence(best_a, n_tgt)

        if rem > 60:
            a_spec = spectral_equalize(a_tgt, 12)
            a_tgt, _ = consider_candidate(None, float("inf"), a_spec, "", verbose=False)
            if score_fast(a_spec) < score_fast(a_tgt):
                a_tgt = a_spec

        if n_tgt == 3072:
            t_mult = min(rem * 0.25, 400.0)
            bf0, bf1, lr0, lr_min = 3.0, 18.0, 0.015, 0.0008
        else:
            t_mult = min(rem * 0.60, 700.0)
            bf0, bf1, lr0, lr_min = 3.0, 28.0, 0.011, 0.0005

        if t_mult > 30:
            a2, _ = mult_log_descent(
                a_tgt, t_mult, rng,
                beta_factor_start=bf0, beta_factor_end=bf1,
                lr0=lr0, lr_min=lr_min, verbose=True
            )
            best_a, best_score = consider_candidate(best_a, best_score, a2, f"mult_n{n_tgt}", verbose=True)

        rem = limit - (time.monotonic() - t_start)
        if rem > 50:
            if n_tgt == 3072:
                t_lb = min(rem * 0.10, 130.0)
            else:
                t_lb = min(rem * 0.50, 400.0)
            a2 = lbfgs_polish(best_a, t_lb, beta_factor=10.0, maxiter=500, verbose=True)
            best_a, best_score = consider_candidate(best_a, best_score, a2, f"lbfgs_n{n_tgt}", verbose=True)

    # Final coordinate polish.
    rem = limit - (time.monotonic() - t_start)
    if rem > 40 and best_a.size <= 5000:
        t_coord = min(rem * 0.5, 100.0)
        a2, _ = coordinate_polish(best_a, t_coord, rng, verbose=True)
        best_a, best_score = consider_candidate(best_a, best_score, a2, "coord", verbose=True)

    # Final L-BFGS polish.
    rem = limit - (time.monotonic() - t_start)
    if rem > 30 and best_a.size <= 5000:
        t_lb = min(rem * 0.8, 200.0)
        a2 = lbfgs_polish(best_a, t_lb, beta_factor=6.0, maxiter=400, verbose=True)
        best_a, best_score = consider_candidate(best_a, best_score, a2, "lbfgs_final", verbose=True)

    # Tiny final multiplicative run with sharp peaks if time remains.
    rem = limit - (time.monotonic() - t_start)
    if rem > 25 and best_a.size <= 5000:
        t_mult = min(rem * 0.7, 80.0)
        a2, _ = mult_log_descent(
            best_a, t_mult, rng,
            beta_factor_start=10.0, beta_factor_end=35.0,
            lr0=0.004, lr_min=0.0002, verbose=False
        )
        best_a, best_score = consider_candidate(best_a, best_score, a2, "mult_final", verbose=True)

    # Keep returned length moderate so final direct convolution is safe.
    if best_a.size > 6000:
        best_a = interpolate_sequence(best_a, 4096)
        best_score = score_fast(best_a)

    out = [float(x) for x in best_a]

    try:
        exact = evaluate_sequence(out)
        if np.isfinite(exact):
            print(f"FINAL approximate score={best_score:.6f}, exact score={exact:.6f}, n={len(out)}, time={time.monotonic()-t_start:.1f}s")
        else:
            print(f"FINAL approximate score={best_score:.6f}, exact invalid, n={len(out)}, time={time.monotonic()-t_start:.1f}s")
    except Exception as e:
        print(f"FINAL approximate score={best_score:.6f}, exact eval error={e}, n={len(out)}, time={time.monotonic()-t_start:.1f}s")

    return out
```
