2. Set temperature `beta` (start moderate, increase over time)
3. Nesterov loop:
   a. Compute `conv = fast_autocorr(y)` where `y` is the Nesterov lookahead point
   b. Compute `p = softmax(beta * conv)`
   c. Compute gradient `g = 2 * (p-correlated-with-a)`
   d. Update Nesterov state
   e. Project onto feasible set (`a >= 0`, `sum(a) = sqrt(2n)`)
   f. Track best
4. Periodically try LP best-response for comparison
5. Final polish with coordinate descent

The projection onto `{a >= 0, sum(a) = c}`:
- This is the Euclidean projection onto the intersection of the non-negative orthant and a hyperplane.
- Can be solved via bisection on the Lagrange multiplier.
- `proj(a)[i] = max(0, a[i] - t)` where `t` is chosen so that `sum_i max(0, a[i]-t) = c`.

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def project_nonneg_sum(a, c):

        return np.zeros(n)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with full constraint set (all 2n-1 constraints) and line-search blending
2. Nesterov-accelerated log-sum-exp smooth gradient method
3. Exact 1D coordinate descent with FFT-based updates
4. Multi-restart with diverse perturbations
Key insight: use full constraint LP for precise best responses, line search for optimal blending,
and FFT-accelerated gradient for fast polishing between LP calls.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.2])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_autocorr(a)
    return 2.0 * n * conv.max() / (s * s)

def project_nonneg_sum(a, c):
    n = len(a)
    if c <= 1e-15:
        return np.zeros(n)
    lo = a.min() - 1.0
    hi = a.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.maximum(a - mid, 0.0).sum()
        if s > c:
            lo = mid
        else:
            hi = mid
    return np.maximum(a - 0.5 * (lo + hi), 0.0)

def normalize(a, n):
    target = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * target / n
    return a * target / s

def build_T(a, n):
    """Build constraint matrix T[k,j] = a[k-j] if 0<=k-j<n else 0."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for k in range(size):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            T[k, lo:hi] = a[k - np.arange(lo, hi)]
    return T

def lp_best_response(a_fixed, M, n, T, solver):
    """Solve: max sum(b) s.t. T@b <= M, b >= 0. Returns (b, duals)."""
    b_var = cp.Variable(n, nonneg=True)
    constraint = T @ b_var <= M
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [constraint])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dv = constraint.dual_value
        if dv is not None:
            duals = np.asarray(dv, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals

def line_search_blend(a, b, target_sum):
    """Find optimal blending parameter t for a_new = (1-t)*a + t*b."""
    best_t = 0.0
    best_sc = score_of(a)
    for t in [0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
        trial = (1.0 - t) * a + t * b
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-15:
            continue
        trial = trial * target_sum / s
        sc = score_of(trial)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    # Refine
    if best_t > 0.001 and best_t < 0.999:
        for dt in [-0.01, -0.005, -0.002, 0.002, 0.005, 0.01]:
            t = best_t + dt
            if 0.001 < t < 0.999:
                trial = (1.0 - t) * a + t * b
                trial = np.maximum(trial, 0.0)
                s = trial.sum()
                if s < 1e-15:
                    continue
                trial = trial * target_sum / s
                sc = score_of(trial)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t
    return best_t, best_sc

def smooth_gradient(a, beta, conv):
    """Gradient of log-sum-exp approximation to max."""
    n = len(a)
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    vals = np.exp(shifted)
    p = vals / vals.sum()
    # grad[j] = 2 * sum_i p[j+i] * a[i] via FFT cross-correlation
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)
    C = irfft(np.conjugate(P) * A, fft_size)[:n]
    return 2.0 * C
