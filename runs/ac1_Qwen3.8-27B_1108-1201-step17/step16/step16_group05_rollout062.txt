   from scipy.optimize import minimize
   import time
   
   def softmax(x):
       x -= x.max()
       e = np.exp(x)
       return e / e.sum()
   
   def full_conv(a):
       """Full linear convolution of a with itself."""
       n = len(a)
       size = 2 * n - 1
       fft_size = 1 << (size - 1).bit_length()
       A = np.fft.rfft(a, fft_size)
       return np.fft.irfft(A * A, fft_size)[:size]
   
   def lse_objective_and_grad(x, beta):
       n = len(x)
       a = softmax(x)
       conv = full_conv(a)
       
       # LSE computation
       beta_conv = beta * conv
       max_bc = beta_conv.max()
       shifted = beta_conv - max_bc
       exp_sum = np.sum(np.exp(shifted))
       obj = (max_bc + np.log(exp_sum)) / beta
       
       # Weights
       weights = np.exp(shifted) / exp_sum
       
       # Gradient of conv w.r.t. a
       grad_a = 2.0 * np.correlate(weights, a, mode='valid')
       
       # Gradient w.r.t. x (softmax chain rule)
       grad_x = a * (grad_a - np.dot(grad_a, a))
       
       return obj, grad_x

import numpy as np

def softmax(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def full_convolution(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def lse_objective_and_grad(x, beta):
    """Compute LSE objective and gradient w.r.t. log-domain parameter x."""
    n = len(x)
    a = softmax(x)
    conv = full_convolution(a)
    
    beta_conv = beta * conv
    max_bc = beta_conv.max()
    shifted = beta_conv - max_bc
    exp_vals = np.exp(shifted)
    exp_sum = exp_vals.sum()
    
    obj = (max_bc + np.log(exp_sum)) / beta
    
    # Weights for the softmax-like distribution over lags
    weights = exp_vals / exp_sum
    
    # Gradient of weighted sum of conv w.r.t. a
    # grad_a[j] = 2 * sum_{m=0}^{n-1} weights[j+m] * a[m]
    grad_a = 2.0 * np.correlate(weights, a, mode='valid')
    
    # Chain rule for softmax: grad_x = a * (grad_a - dot(grad_a, a))
    grad_x = a * (grad_a - np.dot(grad_a, a))
    
    return obj, grad_x

def interpolate_to_length(f, new_n):

        return f.copy()
    s = f.sum()
    if s < 1e-15:

    old_x = np.linspace(0.0, 1.0, old_n)

    f_interp = np.interp(new_x, old_x, f_norm)

    return np.fft.irfft(A * A, fft_size)[:size]


def _lse_obj_grad(x, beta):
    """Compute LSE objective and gradient w.r.t. log-domain parameter x."""
    n = len(x)
    a = _softmax(x)
    conv = _full_conv(a)
    bc = beta * conv
    mbc = bc.max()
    shifted = bc - mbc
    ev = np.exp(shifted)
    es = ev.sum()
    obj = (mbc + np.log(es)) / beta
    w = ev / es
    # grad_a[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    # Softmax chain rule
    dot_ga_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot_ga_a)
    return obj, grad_x


def _scipy_callback(x):
    """Top-level callback for scipy minimize (uses global _g_beta)."""
    return _lse_obj_grad(x, _g_beta)


def _run_lbfgs(x0, beta, max_iter, gtol, max_ls):
    """Run L-BFGS-B with LSE objective. Returns (x_result, info_dict)."""
    global _g_beta
    _g_beta = beta
    t0 = time.monotonic()
    try:
        res = minimize(
            _scipy_callback, x0, jac=True, method='L-BFGS-B',
            options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                     'gtol': gtol, 'maxls': max_ls, 'iprint': -1}
        )
        elapsed = time.monotonic() - t0
        a = _softmax(res.x)
        conv = _full_conv(a)
        n = len(a)
        score = 2.0 * n * conv.max()
        info = {'x': res.x, 'niter': res.nit, 'elapsed': elapsed,
                'max_conv': conv.max(), 'score': score, 'success': res.success}
        return res.x, info
    except Exception as e:
        print(f"      L-BFGS-B exception: {e}")
        return x0, {'x': x0, 'niter': 0, 'elapsed': 0, 'max_conv': 0,
                    'score': float('inf'), 'success': False}


def _eval_safe(a):
    try:
        al = [float(x) for x in a]
        sc = evaluate_sequence(al)
        return float(sc)
    except Exception:
        return float('inf')


def _exact_score(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = _full_conv(a)
    return 2.0 * len(a) * conv.max() / (s * s)


def _cd_exact(a0, time_limit, verbose, seed):
    """Coordinate descent on exact max-conv objective (sum normalized to 1)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_m = _full_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = _full_conv(a)
        M = conv.max()
        thresh = M * 0.995
        active = np.where(conv >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])

        improved = False
        # Compute contribution to active peaks
        contrib = np.zeros(n)
        for k in active[:40]:
            ki = int(k)
            i_lo = max(0, ki - n + 1)
            i_hi = min(n, ki + 1)
            idx = np.arange(i_lo, i_hi)
            jdx = ki - idx
            contrib[idx] += a[idx] * a[jdx]

        sources = np.argsort(contrib)[::-1][:80]
        for src in sources:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue

            best_dst = -1
            best_m_new = M
            best_amt = 0.0

            n_try = min(n, 800)
            if n > n_try:
                cands = rng.choice(n, n_try, replace=False)
            else:
                cands = np.arange(n)

            for dst in cands:
                if dst == src:
                    continue
                # Quick impact estimate
                imp = 0.0
                for k in active[:12]:
                    ki = int(k)
                    d = ki - dst
                    if 0 <= d < n:
                        imp += a[d]
                    d2 = dst - ki + n - 1
                    if 0 <= d2 < n:
                        imp += a[d2]
                if imp > 0.5 * best_m_new:
                    continue

                for frac in (0.03, 0.08, 0.15):
                    amt = a[src] * frac
                    if amt < 1e-14:
                        continue
                    a_try = a.copy()
                    a_try[src] -= amt
                    a_try[dst] += amt
                    a_try = np.maximum(a_try, 0.0)
                    st = a_try.sum()
                    if st < 1e-15:
                        continue
                    a_try = a_try / st
                    m_try = _full_conv(a_try).max()
                    if m_try < best_m_new:
                        best_m_new = m_try
                        best_dst = dst
                        best_amt = amt

            if best_dst >= 0 and best_m_new < M - 1e-14:
                a[src] -= best_amt
                a[best_dst] += best_amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M = best_m_new
                improved = True

        if improved:
            stall = 0
            if M < best_m - 1e-15:
                best_m = M
                best_a = a.copy()
        else:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.02, 0.08)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                mn = _full_conv(a).max()
                if mn < best_m:
                    best_m = mn
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()

        if verbose and (stall % 60 == 0 or improved):
            elapsed = time.monotonic() - t0
            print(f"        CD ({elapsed:.0f}s): M={best_m:.8f}, "
                  f"score={2.0*n*best_m:.6f}, stall={stall}")

    return best_a


def _get_parent():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p.copy()
    except NameError:
        pass
    except Exception:
        pass
    return None


def _interp_len(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def _to_logdomain(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-30)
    x = np.log(a)
    x -= x.mean()
    return x


def _run_multi_beta_lbfgs(x_init, n, time_budget, verbose=True):
    """Run L-BFGS-B with increasing beta. Returns best x found."""
    global _g_beta
    best_x = x_init.copy()
    best_score = float('inf')
    t0 = time.monotonic()

    betas = [12, 25, 50, 100, 180, 300, 500]
    x_cur = x_init.copy()

    for bi, beta in enumerate(betas):
        rem = time_budget - (time.monotonic() - t0)
        if rem < 20:
            break
        # Allocate time: later betas get less
        if bi < 2:
            t_alloc = min(rem * 0.25, 300.0)
        elif bi < 4:
            t_alloc = min(rem * 0.22, 250.0)
        else:
            t_alloc = min(rem * 0.18, 180.0)
        if t_alloc < 10:
            break

        # Estimate max iterations (~3-8ms per iter for n~2400)
        max_iter = max(100, int(t_alloc / 0.006))
        gtol = 1e-9 if beta >= 50 else 1e-8

        if verbose:
            print(f"      L-BFGS-B β={beta}, max_iter={max_iter}, "
                  f"t_alloc={t_alloc:.0f}s")

        x_new, info = _run_lbfgs(x_cur, beta, max_iter, gtol, 20)
        if info['success'] or info['score'] < best_score:
            a_try = _softmax(x_new)
            sc = _exact_score(a_try)
            if sc < best_score:
                best_score = sc
                best_x = x_new.copy()
                if verbose:
                    print(f"        → improved: score={sc:.6f} "
                          f"({info['niter']} iters, {info['elapsed']:.1f}s)")
        x_cur = x_new

    elapsed = time.monotonic() - t0
    if verbose:
        print(f"      Multi-beta done in {elapsed:.0f}s, best score={best_score:.6f}")
    return best_x, best_score


def _try_length(a_best, n_try, time_budget, verbose=True):
    """Interpolate to new length and optimize."""
    a_interp = _interp_len(a_best, n_try)
    x_init = _to_logdomain(a_interp)
    x_best, sc = _run_multi_beta_lbfgs(x_init, n_try, time_budget, verbose=False)
    a_out = _softmax(x_best)
    return a_out


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)

    def time_left():
        return time_budget - (time.monotonic() - t_global)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = _get_parent()
    if parent is not None:
        p = np.maximum(parent, 0.0)
        p = p / p.sum()
        p_score = _eval_safe(p)
        print(f"Parent: n={len(p)}, score={p_score:.6f}")
        best_seq = p.copy()
        best_score = p_score
    else:
        n_base = 2400
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = _eval_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    def try_upd(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = _eval_safe(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            el = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={el:.0f}s)")

    # === Phase 1: Main L-BFGS-B with beta annealing ===
    print(f"\n=== Phase 1: L-BFGS-B annealing (n={n0}) ===")
    x_init = _to_logdomain(best_seq)
    t1 = min(time_left() * 0.45, 600.0)
    if t1 > 30:
        x_best, _ = _run_multi_beta_lbfgs(x_init, n0, t1, verbose=True)
        a_out = _softmax(x_best)
        try_upd(a_out, "lbfgs_main")

    # === Phase 2: Coordinate descent with exact max ===
    rem = time_left()
    print(f"\n=== Phase 2: CD exact (n={len(best_seq)}) ===")
    t2 = min(rem * 0.20, 250.0)
    if t2 > 20:
        a_out = _cd_exact(best_seq, t2, verbose=True, seed=42)
        try_upd(a_out, "cd_p2")

    # === Phase 3: Another L-BFGS-B round (higher betas from improved start) ===
    rem = time_left()
    print(f"\n=== Phase 3: L-BFGS-B round 2 (n={len(best_seq)}) ===")
    t3 = min(rem * 0.20, 250.0)
    if t3 > 30:
        x_init2 = _to_logdomain(best_seq)
        # Start from higher beta since we're closer to optimum
        global _g_beta
        x_cur = x_init2.copy()
        for beta in [80, 150, 300, 600, 1000]:
            rem2 = time_left()
            t3_rem = t3 - (time.monotonic() - t_global + (time_budget - rem2))
            # Simpler: just allocate fraction of t3 remaining
            elapsed_p3 = time.monotonic() - t_global
            frac = 1.0 - max(0, t3 - (rem2 - 50)) / max(t3, 1)
            t_this = min(rem2 * 0.18, 150.0)
            if t_this < 10:
                break
            max_iter = max(100, int(t_this / 0.006))
            x_new, info = _run_lbfgs(x_cur, beta, max_iter, 1e-10, 20)
            a_try = _softmax(x_new)
            sc = _eval_safe(a_try)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_try.copy()
                print(f"        β={beta}: score={sc:.6f}")
            x_cur = x_new

    # === Phase 4: CD again ===
    rem = time_left()
    print(f"\n=== Phase 4: CD polish (n={len(best_seq)}) ===")
    t4 = min(rem * 0.15, 150.0)
    if t4 > 15:
        a_out = _cd_exact(best_seq, t4, verbose=True, seed=77)
        try_upd(a_out, "cd_p4")

    # === Phase 5: Try different lengths ===
    rem = time_left()
    print(f"\n=== Phase 5: Multi-length exploration ===")
    for n_try in [n0 - 300, n0 + 300, n0 + 600]:
        rem = time_left()
        if rem < 80 or n_try < 500:
            continue
        t_len = min(rem * 0.18, 200.0)
        if t_len < 20:
            continue
        print(f"      Trying n={n_try} (t={t_len:.0f}s)")
        a_out = _try_length(best_seq, n_try, t_len, verbose=False)
        try_upd(a_out, f"len_{n_try}")

    # === Phase 6: Final CD ===
    rem = time_left()
    print(f"\n=== Phase 6: Final CD (n={len(best_seq)}) ===")
    t6 = min(rem * 0.5, 300.0)
    if t6 > 15:
        a_out = _cd_exact(best_seq, t6, verbose=True, seed=314)
        try_upd(a_out, "cd_final")

    # === Phase 7: One more L-BFGS at very high beta ===
    rem = time_left()
    if rem > 30:
        print(f"\n=== Phase 7: Ultra-high beta L-BFGS ===")
        x_init7 = _to_logdomain(best_seq)
        for beta in [500, 1000, 2000]:
            rem = time_left()
            if rem < 15:
                break
            max_iter = max(100, int(min(rem * 0.3, 100.0) / 0.006))
            x_new, info = _run_lbfgs(x_init7, beta, max_iter, 1e-12, 10)
            a_try = _softmax(x_new)
            sc = _eval_safe(a_try)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_try.copy()
                x_init7 = x_new.copy()
                print(f"        β={beta}: score={sc:.6f}")

    # Final validation
    if best_seq is not None:
        fs = _eval_safe(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
