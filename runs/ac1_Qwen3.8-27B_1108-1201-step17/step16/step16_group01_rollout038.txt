"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact coordinate descent
with 1D bisection on the peak value, multiple diverse initializations,
incremental convolution updates, and adaptive perturbation for escaping local minima.
Key improvement over subgradient methods: exact per-coordinate optimization guarantees
monotone improvement and finds precise coordinate-wise local minima.
"""

import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def incremental_conv_update(conv, a, j, old_val, new_val, n):
    """Update convolution incrementally when a[j] changes from old_val to new_val."""
    delta = new_val - old_val
    if abs(delta) < 1e-18:
        return
    # For k in [j, j+n-1], k != 2j: conv[k] += 2*delta*a[k-j]
    k_start = j
    k_end = j + n - 1
    k_start = max(0, k_start)
    k_end = min(2 * n - 2, k_end)
    if k_start <= k_end:
        ks = np.arange(k_start, k_end + 1)
        js = ks - j
        # Skip k = 2j for the linear part
        mask = ks != 2 * j
        conv[ks[mask]] += 2.0 * delta * a[js[mask]]
    # For k = 2j: conv[2j] += 2*delta*old_val + delta^2
    kq = 2 * j
    if 0 <= kq < 2 * n - 1:
        conv[kq] += 2.0 * delta * old_val + delta * delta


def optimize_coord_exact(a, j, conv, n, top_k=80):
    """
    Find optimal a[j] >= 0 minimizing max(conv) using bisection on T.
    Uses only top_k lags for efficiency.
    Returns the new optimal value for a[j].
    """
    x = a[j]
    size = 2 * n - 1

    # Compute d and c for the top lags
    # d_k = 2*a[k-j] (if valid, k != 2j), 0 otherwise
    # c_k = conv[k] - d_k * x (for linear), conv[2j] - x^2 (for quadratic)

    # Identify top lags
    if top_k < size:
        peak_idx = np.argsort(conv)[::-1][:top_k]
    else:
        peak_idx = np.arange(size)

    # Build d and c arrays for only these lags
    d_lag = np.zeros(len(peak_idx))
    c_lag = np.zeros(len(peak_idx))
    is_quad = np.zeros(len(peak_idx), dtype=bool)

    for idx, k in enumerate(peak_idx):
        ki = int(k)
        if ki == 2 * j:
            # Quadratic case
            is_quad[idx] = True
            c_lag[idx] = conv[ki] - x * x
            d_lag[idx] = 0.0
        else:
            # Linear case: d = 2*a[k-j] if 0 <= k-j < n
            diff = ki - j
            if 0 <= diff < n:
                d_lag[idx] = 2.0 * a[diff]
            else:
                d_lag[idx] = 0.0
            c_lag[idx] = conv[ki] - d_lag[idx] * x

    # Feasibility check function (inlined)
    # For given T: find x_new >= 0 such that for all lags:
    #   Linear (d>0): c + d*x <= T => x <= (T-c)/d
    #   Linear (d<0): c + d*x <= T => x >= (T-c)/d
    #   Linear (d=0): c <= T
    #   Quadratic: c + x^2 <= T => x <= sqrt(T-c)

    # Precompute masks
    pos_mask = d_lag > 1e-18
    neg_mask = d_lag < -1e-18
    zero_mask = (d_lag >= -1e-18) & (d_lag <= 1e-18) & ~is_quad
    quad_mask = is_quad

    # Precompute coefficients for linear constraints
    # For d>0: x <= (T-c)/d = T/d - c/d
    pos_inv_d = np.zeros(np.sum(pos_mask))
    pos_c_over_d = np.zeros(np.sum(pos_mask))
    np.divide(1.0, d_lag[pos_mask], out=pos_inv_d)
    np.divide(c_lag[pos_mask], d_lag[pos_mask], out=pos_c_over_d)

    # For d<0: x >= (T-c)/d. Since d<0, (T-c)/d = T/d - c/d (note: dividing by negative)
    neg_inv_d = np.zeros(np.sum(neg_mask))
    neg_c_over_d = np.zeros(np.sum(neg_mask))
    np.divide(1.0, d_lag[neg_mask], out=neg_inv_d)
    np.divide(c_lag[neg_mask], d_lag[neg_mask], out=neg_c_over_d)

    # Bisection on T
    T_lo = 0.0
    T_hi = float(conv.max())

    best_x = x
    for _ in range(50):
        T_mid = 0.5 * (T_lo + T_hi)

        # Check zero-d constraints
        feasible = True
        if np.any(zero_mask):
            if np.max(c_lag[zero_mask]) > T_mid + 1e-15:
                feasible = False

        # Quadratic constraint
        quad_upper = np.inf
        if np.any(quad_mask):
            if T_mid < np.max(c_lag[quad_mask]) - 1e-15:
                feasible = False
            else:
                quad_upper = np.sqrt(max(0.0, T_mid - np.max(c_lag[quad_mask])))

        if feasible:
            # Compute bounds
            upper = np.inf
            lower = 0.0

            if np.any(pos_mask):
                # x <= T/d - c/d for each pos constraint
                ub = T_mid * pos_inv_d - pos_c_over_d
                upper = min(upper, float(np.min(ub)))

            if np.any(neg_mask):
                # x >= T/d - c/d for each neg constraint (note d<0 so this is a lower bound)
                lb = T_mid * neg_inv_d - neg_c_over_d
                lower = max(lower, float(np.max(lb)))

            upper = min(upper, quad_upper)

            if lower <= upper + 1e-15:
                best_x = max(0.0, 0.5 * (lower + upper))
                T_hi = T_mid
            else:
                T_lo = T_mid
        else:
            T_lo = T_mid

    return max(0.0, best_x)

                new_total = total * best_scale
                a[j1] = best_t * new_total
                a[j2] = (1 - best_t) * new_total
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                best_m = best_pair_m
                best_a = a.copy()
                conv = fast_autocorr(a)
                improved = True

        if not improved:
            # Also try single coordinate updates
            for j in hot_list[:30]:
                if time.monotonic() - t0 > time_limit:
                    break
                if a[j] < 1e-12:
                    continue
                # Try scaling a[j]
                best_m_val = best_m
                best_scale = 1.0
                for scale in [0.5, 0.7, 0.85, 1.15, 1.3]:
                    a_trial = a.copy()
                    a_trial[j] *= scale
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m_val:
                        best_m_val = float(m_trial)
                        best_scale = scale

                if best_scale != 1.0:
                    a[j] *= best_scale
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                    best_m = best_m_val
                    best_a = a.copy()
                    conv = fast_autocorr(a)
                    improved = True

        if not improved:
            stall += 1
            # Random perturbation
            n_pert = max(3, n // 100)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                jj = int(rng.integers(n))
                if i != jj and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.02, 0.06)
                    a[i] -= amt
                    a[jj] += amt
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            conv = fast_autocorr(a)

        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Pair: stall={stall} ({elapsed:.0f}s), J={J:.6f}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search: multi-start exact coordinate descent with diverse initializations,
    followed by AMD polish and pair descent for final refinement.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Total time budget: {time_budget:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_safe(parent):.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Determine target length
    if parent is not None:
        n_target = len(parent)
    else:
        n_target = 2400

    # Generate initializations
    inits = make_initializations(parent, n_target)
    print(f"Generated {len(inits)} initializations with n={n_target}")

    # Phase 1: Run coordinate descent on each initialization
    n_inits = len(inits)
    # Allocate time: more time for parent, less for others
    per_init_time = []
    if parent is not None:
        per_init_time.append(time_budget * 0.30)  # 30% for parent
    remaining_frac = 0.50  # 50% for other inits
    for i in range(len(per_init_time), n_inits):
        per_init_time.append(remaining_frac / (n_inits - len(per_init_time)) * time_budget)

    for i, init in enumerate(inits):
        elapsed = time.monotonic() - t_global
        remaining = time_budget - elapsed
        t_alloc = min(per_init_time[i], remaining * 0.6)
        if t_alloc < 30:
            break

        label = f"CD_init{i}"
        if i == 0 and parent is not None:
            label = "CD_parent"

        print(f"\n  Running {label} (n={n_target}, alloc={t_alloc:.0f}s)")
        a_out, m_out = coord_descent(
            init, t_alloc, top_k=100, verbose=True,
            rng_seed=100 + i, max_stall_sweeps=120
        )
        try_update(a_out, label)

    # Phase 2: AMD polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is not None and remaining > 60:
        print(f"\n  Phase 2: AMD polish (remaining={remaining:.0f}s)")
        t2 = min(remaining * 0.25, 200.0)
        if t2 > 30:
            a_out, _ = amd_polish(best_seq, t2, c0=2.5, verbose=False, rng_seed=200)
            try_update(a_out, "AMD_polish")

    # Phase 3: Pair descent
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is not None and remaining > 40:
        print(f"\n  Phase 3: Pair descent (remaining={remaining:.0f}s)")
        t3 = min(remaining * 0.25, 150.0)
        if t3 > 20:
            a_out, _ = pair_descent(best_seq, t3, verbose=True, rng_seed=300)
            try_update(a_out, "pair_descent")

    # Phase 4: Another round of coordinate descent on best
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is not None and remaining > 60:
        print(f"\n  Phase 4: CD final polish (remaining={remaining:.0f}s)")
        t4 = min(remaining * 0.40, 300.0)
        if t4 > 30:
            a_out, _ = coord_descent(
                best_seq, t4, top_k=120, verbose=True,
                rng_seed=400, max_stall_sweeps=80
            )
            try_update(a_out, "CD_final")

    # Phase 5: Try different n values
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is not None and remaining > 90:
        print(f"\n  Phase 5: Multi-n exploration (remaining={remaining:.0f}s)")
        for dn in [300, -300, 600]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            n_try = len(best_seq) + dn
            if n_try < 1000 or n_try > 3500:
                continue
            a_try = interpolate_seq(best_seq, n_try)
            t5 = min(remaining * 0.3, 120.0)
            if t5 < 30:
                break
            print(f"    Trying n={n_try} (alloc={t5:.0f}s)")
            a_out, _ = coord_descent(
                a_try, t5, top_k=100, verbose=False,
                rng_seed=500 + dn, max_stall_sweeps=80
            )
            try_update(a_out, f"CD_n{n_try}")

    # Phase 6: Final AMD sub-polish
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is not None and remaining > 30:
        t6 = min(remaining * 0.5, 80.0)
        if t6 > 15:
            a_out, _ = amd_polish(best_seq, t6, c0=1.0, verbose=False, rng_seed=600)
            try_update(a_out, "AMD_subpolish")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        # Fallback
        n_fb = 2400
        x = np.linspace(0, 1, n_fb)
        fb = (1 - np.sin(np.pi * x)) ** 2 + 0.1
        fb = fb / fb.sum()
        best_score = evaluate_safe(fb)
        best_seq = fb

    return [float(x) for x in best_seq]
