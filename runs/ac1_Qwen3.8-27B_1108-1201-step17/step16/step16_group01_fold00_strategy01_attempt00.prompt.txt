def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def norm_sum1(a):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def exact_subgradient(a, ac, threshold_frac=0.995):
    """Compute subgradient using only lags near the maximum (active set)."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n), 1
    
    active = np.where(ac >= threshold_frac * M)[0]
    if len(active) == 0:
        active = np.array([int(ac.argmax())])
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        g[idx] += 2.0 * a[j_idx]
    
    # Normalize to unit l2 norm for consistent step size interpretation
    gn = np.linalg.norm(g)
    if gn > 1e-15:
        g = g / gn
    return g, len(active)


def soft_subgradient(a, ac, top_k=30, beta=4.0):
    """Soft subgradient using top-k lags with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad_contrib = 2.0 * wi * a[j_idx]
        g[idx] += grad_contrib
    return g


def mirror_update(y, g, eta):
    """Mirror descent update in log-space."""
    log_a = np.log(np.maximum(y, 1e-15)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s


def amd_exact(a0, time_limit, c0=3.0, threshold_frac=0.995, verbose=True, rng_seed=42):
    """
    Accelerated Mirror Descent with EXACT active-set subgradient.
    Key improvement: uses exact subgradient from lags within threshold_frac of max.
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 4000
    current_threshold = threshold_frac
    c_current = c0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute autocorrelation and exact subgradient
        ac = fast_autocorr(y)
        g, n_active = exact_subgradient(y, ac, threshold_frac=current_threshold)

        # Step size: adaptive with scale
        eta = c_current / (np.sqrt(t) + 1.0) * np.sqrt(n)

        # Mirror update
        a_new = mirror_update(y, g, eta)

        x_prev = x
        x = a_new

        # Evaluate
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive: adjust threshold and restart if stuck
        if stall_count > max_stall:
            stall_count = 0
            # Perturb from best
            noise_scale = 0.005 + 0.01 * np.sqrt(stall_count / max_stall)
            perturbation = rng.normal(0, noise_scale, n) * best_a
            x_restarted = best_a + perturbation
            x_restarted = np.maximum(x_restarted, 1e-10)
            x_restarted = x_restarted / x_restarted.sum()
            x = x_restarted
            x_prev = x_restarted
            # Adapt parameters
            if current_threshold > 0.98:
                current_threshold = max(0.98, current_threshold - 0.005)
            c_current = c_current * 1.2
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    AMD restart t={t} ({elapsed:.1f}s): thresh={current_threshold:.4f}, c={c_current:.2f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.1f}s): J={J:.6f}, active={n_active}, stall={stall_count}")

    return best_a, best_m

def coordinate_descent(a0, time_limit, verbose=True, rng_seed=123):

    stall = 0
    max_stall = 100

    peak_k = np.argsort(ac)[::-1][:20]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        improved = False

        for k in peak_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break

            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            if len(idx) == 0:
                continue
            j_idx = k_int - idx
            # Contribution of each pair
            contrib = a[idx] * a[j_idx]

            if contrib.max() < 1e-20:
                continue

            # Find the pair with maximum contribution
            max_pair_local = np.argmax(contrib)
            src_i = idx[max_pair_local]
            dst_j = j_idx[max_pair_local]

            # Try moving mass from the larger of (a[src_i], a[dst_j]) to a random low-position
            if a[src_i] >= a[dst_j]:
                src = src_i

                src = dst_j

            # Compute "peak sensitivity" of each position: how much it contributes to peaks
            peak_sens = np.zeros(n)
            for k2 in peak_k[:10]:

            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            # conv[k] = sum_{i in idx} y[i] * b[j_idx[i-i_start]]
            # This is a linear form in b: sum coeff_j * b[j]
            coeff = np.zeros(n)
            for ii, j in zip(idx, j_idx):
                coeff[j] += y[ii]
            constraints.append(coeff @ b_var <= M_curr)

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m
