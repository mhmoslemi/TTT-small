def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def fast_corr_valid(w, a):
    """c[i] = sum_j w[i+j]*a[j] for i=0,...,len(a)-1, w has len >= len(a)."""
    n = len(a)
    m = len(w)
    fft_size = 1 << (m + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    full = np.fft.irfft(W * np.conj(A), fft_size)
    return full[:m - n + 1]

def smooth_grad(a, beta):
    """Compute LSE-smoothed max of conv(a,a) and its gradient."""
    ac = fast_conv(a)
    x_max = ac.max()
    shifted = beta * (ac - x_max)
    e = np.exp(shifted)
    s_e = e.sum()
    M_soft = x_max + np.log(s_e) / beta
    w = e / s_e
    grad = 2.0 * fast_corr_valid(w, a)
    return M_soft, grad

def hard_grad(a, top_k=1):
    """Compute subgradient using top-k peaks."""
    ac = fast_conv(a)
    n = len(a)
    top_idx = np.argsort(ac)[::-1][:top_k]
    grad = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            j = k_int - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    if top_k > 1:
        grad /= top_k
    return ac.max(), grad

def mirror_step(a, grad, eta):
    """Multiplicative weights update (mirror descent step for simplex)."""
    log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def nesterov_smooth(a0, time_limit, beta_start, beta_end, c0, verbose=True, seed=42):
    """Nesterov accelerated mirror descent with smoothed objective."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 2000
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta (exponential ramp)
        progress = min(t / 10000.0, 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Nesterov extrapolation
        momentum = (t - 1.0) / (t + 1.5)
        y = x + momentum * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute smoothed gradient
        M_soft, grad = smooth_grad(y, beta)
        
        # Step size
        eta = c0 / (np.sqrt(t) + 1.0)
        
        # Mirror descent step
        x_new = mirror_step(y, grad, eta)
        
        x_prev = x
        x = x_new
        
        # Track best (using hard max)
        M_hard = fast_conv(x).max()
        if M_hard < best_m - 1e-15:
            best_m = M_hard
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

            pert = rng.normal(0, 0.01, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            stall = 0
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothND t={t} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}, stall={stall}")
    
    return best_a, best_m

def lp_direction(a, M, solver=None):
    """Solve LP to find best direction: max sum(b) s.t. b>=0, (a*b)[k]<=M."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    
    try:
        import cvxpy as cp
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        prob.solve(solver=solver or cp.CLARABEL, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
            b_sum = b_val.sum()
            if b_sum > 1e-15:
                return b_val / b_sum
    except Exception:
        pass
    return None

def frank_wolfe_lp(a0, time_limit, solver=None, verbose=True):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(y)
        M_curr = ac.max()
        
        direction = lp_direction(y, M_curr, solver)
        if direction is None:
            break
        
        # Line search
        best_alpha = 2.0 / (t + 2.0)
        best_eval = fast_conv((1 - best_alpha) * y + best_alpha * direction).max()
        for alpha in [0.5, 0.3, 0.2, 0.1, 0.05]:
            trial = np.maximum((1.0 - alpha) * y + alpha * direction, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            M_trial = fast_conv(trial).max()
            if M_trial < best_eval:
                best_eval = M_trial
                best_alpha = alpha
        
        y_new = (1.0 - best_alpha) * y + best_alpha * direction
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        y = y_new
        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW-LP iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind, seed=0):

    else:
        a = np.sin(np.pi * x) ** 2
    return norm_sum1(a)

def coordinate_polish(a0, time_limit, verbose=False, seed=123):

    n = len(a0)

                    prob.solve(solver=s, verbose=False)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        solver = s
                        break
                except Exception:
                    continue
            if solver:
                print(f"    Using solver: {solver}")
                a_out, _ = frank_wolfe_lp(norm_sum1(best_seq), t4, solver, verbose=True)
                try_update(a_out, f"lp_fw_n{len(best_seq)}")
        except Exception as e:
            print(f"    LP phase failed: {e}")
    
    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 200:
        n_best = len(best_seq)
        for n_try in [n_best - 400, n_best - 200, n_best + 200, n_best + 400, 1600, 3200]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 100:
                break
