Actually, for linear convolution, if we pad $a$ to length $L \geq 2n-1$ with zeros and do circular convolution, we get the linear convolution. So $(a*a)_k = \frac{1}{L} \sum_{\omega} |\hat{a}_L(\omega)|^2 e^{i\omega k}$ where $\hat{a}_L$ is the DFT of the padded sequence.

The max of $(a*a)_k$ over $k$ is the max of the inverse DTFT of $|\hat{a}_L|^2$. This is the "peak" of the periodic autocorrelation function.

This doesn't immediately lead to a simpler formulation, but it suggests that we want $|\hat{a}_L(\omega)|^2$ to be as "flat" as possible (white spectrum) to minimize the peak.

**For a flat spectrum**: We want $|\hat{a}_L(\omega)|^2 \approx \text{const}$ for all $\omega$. This means $\hat{a}_L(\omega)$ has approximately constant magnitude. In the time domain, this corresponds to a sequence with "random-like" structure. But we're constrained to $a \geq 0$!

For non-negative sequences, the spectrum cannot be perfectly flat (that would require $a_j$ to take both positive and negative values, like a Barker sequence). So the optimal is some compromise.

def _fast_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def _normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _interp(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def _subgradient(y, k):
    """Subgradient of (y*y)_k w.r.t. y."""
    n = len(y)
    g = np.zeros(n)
    i_lo = max(0, k - n + 1)
    i_hi = min(n, k + 1)
    for i in range(i_lo, i_hi):
        j = k - i
        if 0 <= j < n:
            g[i] += 2.0 * y[j]
    return g


def _subgradient_fast(y, k):
    """Subgradient of (y*y)_k w.r.t. y using vectorized operations."""
    n = len(y)
    i_lo = max(0, k - n + 1)
    i_hi = min(n, k + 1)
    idx = np.arange(i_lo, i_hi)
    j_idx = k - idx
    valid = (j_idx >= 0) & (j_idx < n)
    g = np.zeros(n)
    g[idx[valid]] += 2.0 * y[j_idx[valid]]
    return g


def _nesterov_mirror(a0, time_limit, top_k=1, c0=5.0, seed=42, verbose=True):
    """Nesterov accelerated mirror descent (exponential family)."""
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_m = _fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 5000
    current_top_k = top_k
    current_beta = 2.0

    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Convolution and peaks
        ac = _fast_conv(y)
        M_curr = ac.max()

        # Adaptive subgradient
        if stall > max_stall // 2:
            current_top_k = min(current_top_k + 1, 6)
            current_beta = max(current_beta * 0.85, 0.5)
        elif stall > max_stall * 2:
            current_top_k = top_k
            current_beta = 2.0

        k_max = min(current_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** current_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            g += wi * _subgradient_fast(y, int(k))

        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = _fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Periodic restart
        if stall > max_stall:
            x_r = best_a * (1.0 + rng.normal(0, 0.003, n))
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            stall = 0
            current_top_k = min(current_top_k + 1, 8)

        # Large periodic jump
        if t % 20000 == 0 and t > 0:
            x_j = best_a * (1.0 + rng.normal(0, 0.01, n))
            x_j = np.maximum(x_j, 1e-10)
            x_j = x_j / x_j.sum()
            M_j = _fast_conv(x_j).max()
            if M_j < best_m:
                best_m = M_j
                best_a = x_j.copy()
                x = x_j
                x_prev = x_j

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NM t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={current_top_k}, stall={stall}")

    return best_a, best_m


def _lp_fw(a0, time_limit, solver, verbose=True, max_iters=200000):
    """Accelerated Frank-Wolfe with LP oracle."""
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_m = _fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_mom = a.copy()

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        ac = _fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Build LP
        # T[k, j] = y[k-j] if 0 <= k-j < n
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum

        # Exact line search
        best_alpha = 2.0 / (t + 2.0)
        best_m_trial = _fast_conv((1 - best_alpha) * y + best_alpha * direction).max()

        for alpha in [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.05, 0.02]:
            trial = (1.0 - alpha) * y + alpha * direction
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            m_trial = _fast_conv(trial).max()
            if m_trial < best_m_trial:
                best_m_trial = m_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * direction
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Nesterov momentum
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x_mom = x_new

        my = _fast_conv(y_new).max()
        if my < best_m:
            best_m = my
            best_a = y_new.copy()
        mx = _fast_conv(x_mom).max()
        if mx < best_m:
            best_m = mx
            best_a = x_mom.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW iter={t} ({elapsed:.1f}s): J={J:.6f}, alpha={best_alpha:.3f}")

    return best_a, best_m

"""

1. Analytical family initialization (Gaussian, cosine, modulated)
2. Nesterov accelerated mirror descent with adaptive multi-peak subgradient
3. Accelerated Frank-Wolfe with sparse LP oracle and golden-section line search

_solver = None

def _get_solver():

        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _solver


def _fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _eval(a):
    try:
        a_list = [float(x) for x in a]

def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _interp(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def _subgrad_fast(y, k):
    n = len(y)
    i_lo = max(0, k - n + 1)
    i_hi = min(n, k + 1)
    g = np.zeros(n)
    idx = np.arange(i_lo, i_hi)
    j_idx = k - idx
    valid = (j_idx >= 0) & (j_idx < n)
    g[idx[valid]] += 2.0 * y[j_idx[valid]]
    return g


def _make_initial(n, family):
    j = np.arange(n, dtype=np.float64)
    if family == 'uniform':
        a = np.ones(n)
    elif family == 'gauss':
        a = np.exp(-2.0 * ((j - n / 2.0) / n) ** 2)
    elif family == 'cosine':
        a = np.cos(np.pi * (j - n / 2.0) / n) ** 2
    elif family == 'triangle':
        a = 1.0 - np.abs(2.0 * j / n - 1.0)
    elif family == 'modulated':
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 2.0 * j / n)
    elif family == 'modulated3':
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 3.0 * j / n)
    elif family == 'edge':
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * j / n)
    elif family == 'exponential_edge':
        a = np.exp(0.5 * np.cos(2.0 * np.pi * j / n))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _nesterov_mirror(a0, time_limit, top_k=1, c0=5.0, seed=42, verbose=True,

    a = _normalize(a0)
    best_a = a.copy()
    best_m = _fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    max_stall = 5000
    current_top_k = top_k
    current_beta = 2.0
    peak_cycle = 0

    for t in range(1, 500000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = _fast_conv(y)

        # Adaptive peak selection
        if use_multi_peak and stall > plateau_threshold:
            # Use a random sub-peak to escape
            sorted_idx = np.argsort(ac)[::-1]
            if current_top_k == 1 and stall > plateau_threshold * 2:
                # Pick a random peak from top 5
                pick = sorted_idx[rng.integers(0, min(5, len(ac)))]
                k_use = int(pick)
                g = _subgrad_fast(y, k_use)
            else:
                k_max = min(current_top_k, len(ac))
                top_idx = sorted_idx[:k_max]
                top_vals = np.maximum(ac[top_idx], 1e-30)
                w = top_vals ** current_beta
                w_sum = w.sum()
                if w_sum < 1e-30:
                    w = np.ones(k_max) / k_max
                else:
                    w = w / w_sum
                g = np.zeros(n)
                for wi, ki in zip(w, top_idx):
                    g += wi * _subgrad_fast(y, int(ki))
        else:
            k_max = min(current_top_k, len(ac))
            top_idx = np.argsort(ac)[::-1][:k_max]
            top_vals = np.maximum(ac[top_idx], 1e-30)
            w = top_vals ** current_beta
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(k_max) / k_max
            else:
                w = w / w_sum
            g = np.zeros(n)
            for wi, ki in zip(w, top_idx):
                g += wi * _subgrad_fast(y, int(ki))

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:

        x_prev = x
        x = a_new

        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
            if current_top_k > top_k:
                current_top_k = max(top_k, current_top_k - 1)
                current_beta = min(current_beta * 1.1, 2.0)
        else:
            stall += 1

        # Adaptive parameters
        if stall > max_stall // 3:
            current_top_k = min(current_top_k + 1, 4)
            current_beta = max(current_beta * 0.9, 0.5)
        elif stall > max_stall // 2:
            current_top_k = min(current_top_k + 1, 6)
            current_beta = max(current_beta * 0.85, 0.3)

        # Restart on long stall
        if stall > max_stall:
            x_r = best_a * (1.0 + rng.normal(0, 0.005, n))
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            stall = 0
            current_top_k = min(current_top_k + 2, 8)

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0

            print(f"    NM t={t} ({elapsed:.1f}s): J={J:.6f}, tk={current_top_k}, stall={stall}")

def _build_conv_matrix(y, n):
    """Build sparse convolution matrix T where (T@b)_k = (y*b)_k."""
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for j in range(n):
        k_lo = j
        k_hi = j + n
        for k in range(k_lo, k_hi):
            rows[count] = k
            cols[count] = j
            vals[count] = y[k - j]
            count += 1
    T = sparse.csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(2 * n - 1, n))
    return T


def _lp_fw(a0, time_limit, solver, verbose=True):
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_m = _fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_mom = a.copy()
    prev_b = None

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = _fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Build sparse T
        rows_list = []
        cols_list = []
        vals_list = []
        for j in range(n):
            k_start = j
            k_end = j + n
            ks = np.arange(k_start, k_end)
            rows_list.append(ks)
            cols_list.append(np.full(n, j, dtype=np.int32))
            vals_list.append(y[:n])
        rows = np.concatenate(rows_list).astype(np.int32)
        cols = np.concatenate(cols_list)
        vals = np.concatenate(vals_list)
        T = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum

        # Golden section line search
        lo, hi = 0.0, 1.0
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)

        def _line_eval(alpha):
            trial = (1.0 - alpha) * y + alpha * direction
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                return 1e10
            trial = trial / st
            return _fast_conv(trial).max()

        f_c = _line_eval(c_pt)
        f_d = _line_eval(d_pt)
        for _ in range(8):
            if f_c < f_d:
                hi = d_pt
                d_pt = c_pt
                f_d = f_c
                c_pt = hi - gr * (hi - lo)
                f_c = _line_eval(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                f_c = f_d
                d_pt = lo + gr * (hi - lo)
                f_d = _line_eval(d_pt)
        best_alpha = (lo + hi) / 2.0

        y_new = (1.0 - best_alpha) * y + best_alpha * direction
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Nesterov momentum
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x_mom = x_new

        my = _fast_conv(y_new).max()
        if my < best_m:
            best_m = my
            best_a = y_new.copy()
        mx = _fast_conv(x_mom).max()
        if mx < best_m:
            best_m = mx
            best_a = x_mom.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW iter={t} ({elapsed:.1f}s): J={J:.6f}, alpha={best_alpha:.4f}")

    return best_a, best_m


def _coord_descent(a0, time_limit, seed=42, verbose=True):
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_m = _fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 500
    fracs = [0.001, 0.002, 0.003, 0.005, 0.008, 0.01, 0.015, 0.02, 0.03, 0.05]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = _fast_conv(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n, k_int + 1)
            idx = np.arange(i_lo, i_hi)
            j_idx = k_int - idx
            valid = (j_idx >= 0) & (j_idx < n)
            contrib = a[idx[valid]] * a[j_idx[valid]]

            if contrib.max() < 1e-20:
                continue

            src = idx[valid][np.argmax(contrib)]

            # Find good destination (low contribution to top peaks)
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:8]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            n_cand = min(len(candidates), 500)
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]

            # Score: how much does adding mass at dst affect top peaks?
            scores = np.zeros(len(chosen))
            for ci, dst in enumerate(chosen):
                s = 0.0
                for k2 in top_peaks[:5]:
                    k2i = int(k2)
                    j2 = k2i - dst
                    if 0 <= j2 < n:
                        s += a[j2]
                scores[ci] = s

            good_dsts = chosen[scores <= np.median(scores)]
            if len(good_dsts) == 0:
                good_dsts = chosen[:max(1, len(chosen) // 4)]

            for dst in good_dsts[:20]:
                if dst == src:
                    continue
                found = False
                for frac in fracs:
                    amt = a[src] * frac
                    if amt < 1e-18 or a[src] - amt < -1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = _fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 50 == 0:
                n_perturb = min(n // 10, 100)
                srcs = rng.integers(0, n, n_perturb)
                dsts = rng.integers(0, n, n_perturb)
                for si, di in zip(srcs, dsts):
                    if si == di or a[si] < 1e-10:
                        continue
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[si] * frac
                    a[si] -= amt
                    a[di] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = _fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD stall={stall} ({elapsed:.1f}s): J={J:.6f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CD done ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def _symmetrize(a):
    """Enforce symmetry: a_j = a_{n-1-j}."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_sym / s

    n_best = len(best_seq)
    t4 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 4: Coord descent n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = _coord_descent(_normalize(best_seq), t4, seed=42, verbose=True)
        _try_update(a_out, f"cd_n{n_best}")
        a_out_sym = _symmetrize(a_out)
        _try_update(a_out_sym, f"cd_sym_n{n_best}")

    # Phase 5: Try nearby lengths - 10% time
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length n_best={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 100:
        for delta in [200, -200, 400, -400]:
            n_try = n_best + delta
            if n_try < 800 or n_try > 5000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            t5 = min(remaining * 0.25, 150.0)
            if t5 < 20:
                continue
            print(f"  Trying n={n_try}...")
            a_interp = _interp(best_seq, n_try)
            a_interp = _symmetrize(a_interp)
            a_out, _ = _nesterov_mirror(a_interp, t5, top_k=1, c0=4.0,
                                         seed=100 + n_try % 50, verbose=False)
            _try_update(a_out, f"nm_n{n_try}")
            a_out_sym = _symmetrize(a_out)
            _try_update(a_out_sym, f"nm_sym_n{n_try}")

    # Phase 6: Final LP-FW if time - up to 5%
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best <= 2500 and remaining > 60:
        t6 = min(remaining * 0.5, 150.0)
        if t6 > 30:
            print(f"\n=== Phase 6: Final LP-FW n={n_best} (budget={t6:.0f}s) ===")
            a_out, _ = _lp_fw(_normalize(best_seq), t6, solver, verbose=True)
            _try_update(a_out, f"fw_final_n{n_best}")
            a_out_sym = _symmetrize(a_out)
            _try_update(a_out_sym, f"fw_final_sym_n{n_best}")

    # Phase 7: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t7 = min(remaining * 0.7, 100.0)
        if t7 > 10:
            print(f"\n=== Phase 7: Final CD n={n_best} (budget={t7:.0f}s) ===")
            a_out, _ = _coord_descent(_normalize(best_seq), t7, seed=314, verbose=True)
            _try_update(a_out, f"cd_final_n{n_best}")
            a_out_sym = _symmetrize(a_out)
            _try_update(a_out_sym, f"cd_final_sym_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = _eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
