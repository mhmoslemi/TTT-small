Now, let me design the full algorithm:

**Algorithm: Smoothed Max Descent (SMD)**

1. Start with `a` (normalized to sum=1, non-negative)
2. For iteration `t = 1, 2, ...`:
   a. Compute `f = conv(a, a)` (full linear convolution, length 2n-1)
   b. Compute weights: `w[k] = exp(p_t * f[k] / max(f)) / Z` (softmax with temperature)
      - Or simpler: `w[k] = (f[k] / sum(f))^beta` for some `beta > 1`
      - Or: top-k weighted: `w[k] = f[k]^beta` for top k peaks, 0 elsewhere, normalized
   c. Compute gradient: `g[i] = 2 * sum_j a[j] * w[i+j]`
   d. Take mirror descent step:
      - `log_a_new = log(a) - eta * g`
      - `a_new = exp(log_a_new - max(log_a_new))`
      - Normalize: `a_new = a_new / sum(a_new)`
   e. Update `p_t` (increase over time for annealing)
   f. Track best

3. Periodically, apply LP Frank-Wolfe or coordinate descent for polishing.

**Key improvements over previous approach:**
1. The log-sum-exp / softmax weighting gives a smoother gradient that considers ALL peaks simultaneously (not just the single worst peak).
2. Annealing the temperature `p` from low to high gives a "coarse-to-fine" optimization.
3. The gradient computation is efficient (O(n log n) via FFT or O(n^2) via direct correlation, both fine for n=2400).

**Temperature schedule:**
- Start with `p = 1` (uniform-ish weighting, smooth landscape)
- Gradually increase to `p = 50` or `100` (sharper, focuses on peaks)
- Use exponential schedule: `p_t = p_0 * exp(alpha * t)` or linear: `p_t = p_0 + t * delta_p`

**Step size:**
- Use backtracking line search or fixed step with decay
- `eta_t = c / (t + 1)` or adapt based on improvement

**Multiple restarts with different initializations:**
- Try different initial shapes: uniform, sin^2, cosine, rectangular with corners cut, etc.
- Try different `n` values

Let me also think about what initial shapes are good.

For the uniform sequence, `f = conv(a,a)` is triangular (for the linear convolution). The peak is at the center.

To make `f` flatter, we want `a` to have a shape that "spreads out" the energy. Intuitively:
- If `a` is concentrated in the middle, the convolution will have a sharp peak in the center.
- If `a` is concentrated at the edges, the convolution will have peaks at the edges (positions 0 and 2n-2).
- A good shape might be something like a "raised cosine" or "Hann window" shape.

Actually, the optimal shape is related to the "Bownik sequence" or "Achlioptas sequence" in the literature. These are related to the problem of finding a non-negative sequence with flat autocorrelation.

```python
import time

def fast_conv(a):
    n = len(a)
    size = 1 << (2*n-2).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]

    """grad[i] = 2 * sum_j a[j] * w[i+j]"""
    n = len(a)
    # Using FFT
    size = 1 << (n + len(w) - 1 - 1).bit_length()
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w, size)
    corr = np.fft.irfft(np.conj(A) * W, size)

    s = a.sum()

        a = np.ones(n) / n
    else:
        a = a / s

    p_max = 100.0

    # Step size
    c0 = 2.0
    eta = c0
    
    # Top-k for weighting
    top_k = 20
    
    stall = 0
    max_stall = 500
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        f = fast_conv(a)
        M_max = f.max()
        
        # Compute weights
        if top_k < len(f):
            # Top-k weighted
            top_idx = np.argpartition(f, -top_k)[-top_k:]
            w = np.zeros(len(f))
            w[top_idx] = f[top_idx]
            w = w ** p
            w_sum = w.sum()
            if w_sum > 0:
                w = w / w_sum
            else:
                w = np.ones(len(f)) / len(f)
        else:
            w = f ** p
            w_sum = w.sum()
            if w_sum > 0:
                w = w / w_sum
            else:
                w = np.ones(len(f)) / len(f)

        g = compute_grad(a, w)

            stall = 0
            # Decrease eta slightly on success
            # eta *= 0.999
        else:
            stall += 1

            # For now, just accept and continue

        # Anneal temperature
```

```python
import numpy as np
import cvxpy as cp
import time

def fast_conv(a):
    n = len(a)
    size = 1 << (2*n-2).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]

def compute_grad_smoothed(a, w):
    """Compute grad[i] = 2 * sum_j a[j] * w[i+j] using FFT."""
    n = len(a)
    size = 1 << (n + len(w) - 1 - 1).bit_length()
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w, size)
    corr = np.fft.irfft(np.conj(A) * W, size)
    return 2.0 * corr[:n]

def softmax_weights(f, p, top_k=50):
    """Compute softmax weights for the peaks."""
    if top_k < len(f):
        top_idx = np.argpartition(f, -top_k)[-top_k:]
        w = np.zeros(len(f))
        vals = f[top_idx]
        # Scale to avoid overflow
        vals_scaled = p * vals / max(vals.max(), 1e-30)
        w[top_idx] = np.exp(vals_scaled - vals_scaled.max())
    else:
        vals = p * f / max(f.max(), 1e-30)
        w = np.exp(vals - vals.max())
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(f)) / len(f)
    else:
        w = w / w_sum
    return w

def smoothed_mirror_descent(a0, time_limit, seed=42, c0=3.0, p_init=2.0, p_rate=1.002, top_k_init=20, verbose=True):
    """
    Smoothed max descent using mirror descent on the simplex.
    Uses softmax-weighted subgradient for smoothness.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    p = p_init
    c = c0
    top_k = top_k_init
    stall = 0
    max_stall = 1000
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        f = fast_conv(a)
        M_curr = f.max()
        
        # Compute smoothed weights
        w = softmax_weights(f, p, top_k)
        
        # Compute smoothed gradient
        g = compute_grad_smoothed(a, w)
        
        # Line search: try a few step sizes
        eta_t = c / (np.sqrt(t) + 1.0)
        best_eta = eta_t
        best_step_m = M_curr
        
        for scale in [0.25, 0.5, 0.75, 1.0, 1.5, 2.0]:
            eta_try = eta_t * scale
            log_a_try = np.log(np.maximum(a, 1e-30)) - eta_try * g
            log_a_try -= log_a_try.max()
            a_try = np.exp(log_a_try)
            a_try = np.maximum(a_try, 1e-30)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            m_try = fast_conv(a_try).max()
            if m_try < best_step_m:
                best_step_m = m_try
                best_eta = eta_try
        
        a = a  # (will be updated below)
        log_a_new = np.log(np.maximum(a, 1e-30)) - best_eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        
        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Anneal
        p = min(p * p_rate, 50.0)
        
        # Adaptive top_k
        if stall > max_stall // 2:
            top_k = min(top_k + 10, 100)
        
        # Restart
        if stall > max_stall:
            perturbation = 0.01 + 0.01 * (stall // max_stall)
            a = best_a * (1 + rng.normal(0, perturbation, n))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            stall = 0
            p = p_init
            top_k = top_k_init
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SMD t={t} ({elapsed:.1f}s): J={J:.6f}, p={p:.1f}, top_k={top_k}, stall={stall}")
    
    return best_a, best_m
```

The LP is:
- Variables: `b[0], ..., b[n-1] >= 0`
- Maximize: `sum(b)`
- Subject to: `(y * b)[k] <= M` for `k = 0, ..., 2n-2`

where `(y * b)[k] = sum_j y[j] * b[k-j]` (with appropriate bounds).

The constraint matrix `T` has `T[k, j] = y[k-j]` if `0 <= k-j < n`, else 0.

For `n = 2400`, this is a `4799 x 2400` matrix. That's about 11.5M entries, most of which are non-zero (each column has `n` non-zero entries, so total non-zero is `n^2 = 5.76M`).

Solving this LP is the bottleneck. A few ideas to speed it up:
1. Use only the top `K` most binding constraints (reduced LP)
2. Use warm-starting

def fast_conv(a):
    """Compute full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def compute_grad_smoothed(a, w):
    """Compute grad[i] = 2 * sum_j a[j] * w[i+j] using FFT-based cross-correlation."""
    n = len(a)
    total_len = n + len(w) - 1
    size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w, size)
    R = np.fft.irfft(np.conj(A) * W, size)
    return 2.0 * R[:n]


def softmax_weights(f, p, top_k=50):
    """Compute softmax weights emphasizing top peaks."""
    f_max = f.max()
    if f_max < 1e-30:
        return np.ones(len(f)) / len(f)
    if top_k < len(f) and top_k > 0:
        top_idx = np.argpartition(f, -top_k)[-top_k:]
        w = np.zeros(len(f))
        vals = p * f[top_idx] / f_max
        vals = vals - vals.max()
        w[top_idx] = np.exp(vals)
    else:
        vals = p * f / f_max
        vals = vals - vals.max()
        w = np.exp(vals)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(f)) / len(f)
    else:
        w = w / w_sum
    return w


def smoothed_mirror_descent(a0, time_limit, seed=42, c0=3.0, p_init=2.0,

    f = fast_conv(a)
    best_m = f.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    p = p_init
    c = c0
    top_k = top_k_init
    stall = 0
    max_stall = 800

    x = a.copy()
    x_prev = a.copy()

        if time.monotonic() - t0 > time_limit:
            break

        f = fast_conv(x)
        M_curr = f.max()

        w = softmax_weights(f, p, top_k)
        g = compute_grad_smoothed(x, w)

        eta_t = c / (np.sqrt(t) + 1.0)

        best_eta = eta_t
        best_step_m = M_curr

        for scale in [0.5, 0.75, 1.0, 1.5, 2.0, 3.0]:
            eta_try = eta_t * scale
            log_a_try = np.log(np.maximum(x, 1e-30)) - eta_try * g
            log_a_try -= log_a_try.max()
            a_try = np.exp(log_a_try)
            a_try = np.maximum(a_try, 1e-30)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            m_try = fast_conv(a_try).max()
            if m_try < best_step_m:
                best_step_m = m_try
                best_eta = eta_try

        log_a_new = np.log(np.maximum(x, 1e-30)) - best_eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = x.copy()

        if use_nesterov and t > 1:
            beta_t = (t - 1.0) / (t + 1.5)
            x_new = a_new + beta_t * (a_new - x)
            x_new = np.maximum(x_new, 1e-15)
            sx = x_new.sum()
            if sx > 1e-15:
                x_new = x_new / sx
            x_prev = x
            x = x_new
        else:
            x_prev = x
            x = a_new

        M_x = fast_conv(x).max()
        M_a = fast_conv(a_new).max()
        M_best_iter = min(M_x, M_a)

            if M_x < M_a:
                best_m = M_x
                best_a = x.copy()
            else:
                best_m = M_a
                best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        p = min(p * p_rate, 80.0)

        if stall > max_stall // 2:
            top_k = min(top_k + 10, 100)

def lp_frank_wolfe_reduced(a0, time_limit, solver, verbose=True, n_constraints=200):
    """LP-based Frank-Wolfe using only top-K binding constraints for speed."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    iter_count = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        top_idx = np.argsort(ac)[::-1][:n_constraints]

        T_sub = np.zeros((len(top_idx), n))
        for row_i, k in enumerate(top_idx):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            T_sub[row_i, j_start:j_end] = y[k_int - np.arange(j_start, j_end)]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sub @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        direction = b_val / b_sum

        alphas = [2.0 / (t + 2.0), 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]
        best_alpha = alphas[0]
        best_eval = fast_conv((1 - best_alpha) * y + best_alpha * direction).max()

        for alpha in alphas[1:]:
            trial = (1.0 - alpha) * y + alpha * direction
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            trial_m = fast_conv(trial).max()
            if trial_m < best_eval:
                best_eval = trial_m
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * direction
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        y = y_new
        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        iter_count += 1

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, seed=123):
    """Coordinate descent with peak-targeted mass transfer."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:min(15, len(ac))]
        improved = False

        for k in top_peaks[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_int - idx]
            cmax = contrib.max()
            if cmax < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:6]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n, int(k2) + 1)
                in_peak[s2:e2] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            n_cand = min(len(candidates), 300)
            chosen = candidates[rng.permutation(len(candidates))[:n_cand]]

            best_dst = -1
            best_score = float('inf')
            for dst in chosen:
                dst_int = int(dst)
                if dst_int == int(src):
                    continue
                if 0 <= k_int - dst_int < n:
                    sens = a[dst_int] * a[k_int - dst_int]
                else:
                    sens = 0.0
                if sens < best_score:
                    best_score = sens
                    best_dst = dst_int

            if best_dst < 0:
                continue
            dst = best_dst

            found = False
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 100:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a / a.sum()

        if not improved:
            stall += 1

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord polish ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
