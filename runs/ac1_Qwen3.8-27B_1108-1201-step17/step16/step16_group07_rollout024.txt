    if warm_start is not None:
        try:
            b_var.value = warm_start
        except Exception:
            pass
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    duals = None
    try:
        dual_val = constraints[0].dual_value
        if dual_val is not None:
            duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass
    return b_val, duals


def smooth_penalty_gradient(a, M_target, mu=1.0):
    n = len(a)
    conv = fast_autocorr(a)
    excess = np.maximum(0.0, conv - M_target)
    total_excess = excess.sum()
    if total_excess < 1e-15:
        return np.zeros(n), 0.0
    N = 1 << (4 * n).bit_length()
    B = rfft(excess, N)
    A = rfft(a, N)
    c1 = irfft(B * np.conj(A), N)[:n]
    c2 = irfft(A * B, N)[:n]
    grad = 2.0 * mu * (c1 + c2)
    return grad, total_excess


def smooth_penalty_optimize(a0, time_limit, verbose=False, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 1.0
    patience = 0
    max_patience = 200
    mu = 1.0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        if patience > max_patience:
            break

        conv = fast_autocorr(a)
        M_curr = conv.max()

        # Target slightly below current max for aggressiveness
        M_target = M_curr * (1.0 - 0.02)
        grad, excess_sum = smooth_penalty_gradient(a, M_target, mu)

        if np.linalg.norm(grad) < 1e-15:
            break

        # Backtracking line search
        lr = step
        a_trial = a - lr * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_trial

        score_trial = fast_autocorr(a_trial).max()
        n_backtracks = 0
        while score_trial > M_curr and n_backtracks < 20:
            lr *= 0.5
            a_trial = a - lr * grad
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                a_trial = a.copy()
            else:
                a_trial = a_trial * norm / s_trial
            score_trial = fast_autocorr(a_trial).max()
            n_backtracks += 1

        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a_trial.copy()
            patience = 0
            step = min(step * 1.01, 5.0)
            mu = min(mu * 1.005, 10.0)
        else:
            patience += 1
            step *= 0.9
            if step < 1e-12:
                break

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothPen t={t} ({elapsed:.0f}s): best={best_score:.10f}, step={step:.4f}, mu={mu:.2f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0, active_max=None):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 80
    n_iters = 0
    conv_size = 2 * n - 1
    prev_b = None
    max_active = active_max if active_max else int(2.0 * n)

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(a)
        M_curr = conv_y.max()
        if M_curr < 1e-12:
            break

        # Adaptive active set
        if active_set is None or t % 30 == 0:
            progress = min(1.0, t / 800.0)
            # Start with few constraints, end with more
            n_constraints = int(max(50, n * (0.3 + 0.7 * progress)))
            n_constraints = min(n_constraints, max_active, conv_size)
            # Use top peaks
            top_idx = np.argsort(conv_y)[-n_constraints:]
            # Also include peaks above a threshold
            thresh = M_curr * (1.0 - 0.15 * (1.0 - progress))
            above_thresh = np.where(conv_y >= thresh)[0]
            active_set = np.unique(np.concatenate([top_idx, above_thresh]))
            if len(active_set) > max_active:
                order = np.argsort(conv_y[active_set])[-max_active:]
                active_set = active_set[order]

        b, duals = lp_best_response(a, M_curr, active_set, solver, warm_start=prev_b)
        if b is None:
            if t % 10 == 0:
                active_set = np.argsort(conv_y)[-int(0.5 * n):]
                b, duals = lp_best_response(a, M_curr, active_set, solver, warm_start=prev_b)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        prev_b = b.copy()
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update with restart
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * a + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = a.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0) ** 0.5, 0.95)
            a_new = y_new + momentum * (y_new - a)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        a = a_new

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Nesterov restart on stall
        if stall_count > stall_threshold:
            stall_count = int(stall_count / 2)
            if stall_count > 40:
                a = best_a.copy()
                active_set = None
                stall_count = 0
                prev_b = None

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, verbose=False, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve_passes = 0
    max_no_improve = 6

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_autocorr(a)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            aj = a[j]
            lo = max(0, j)
            hi = min(size, j + n)
            if lo >= hi:
                continue

            candidates = [0.0, aj * 0.2, aj * 0.5, aj * 1.5, aj * 2.0]
            typical = norm / n
            candidates += [typical * 0.3, typical, typical * 2.0, typical * 4.0]
            candidates = sorted(set([max(0.0, min(c, norm * 0.4)) for c in candidates]))

            best_x = aj
            best_obj = conv.max()

            for x in candidates:
                if abs(x - aj) < 1e-12:
                    continue
                delta = x - aj
                new_conv = conv.copy()
                new_conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
                if 2 * j < size:
                    new_conv[2 * j] += delta * delta
                scale = norm / (a.sum() - aj + x)
                new_max = new_conv.max() * scale * scale
                if new_max < best_obj - 1e-13:
                    best_obj = new_max
                    best_x = x

            if abs(best_x - aj) > 1e-12:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_score = fast_autocorr(a).max()
                    if new_score < best_score - 1e-13:
                        best_score = new_score
                        best_a = a.copy()
                        conv = fast_autocorr(a)
                        improved = True
                    else:
                        a = best_a.copy()
                        conv = fast_autocorr(a)

        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= max_no_improve:
                break
        else:
            no_improve_passes = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def perturb_and_optimize(best_seq, time_limit, solver, rng, seed):
    n = len(best_seq)
    # Structured perturbation: add a cosine at random frequency
    freq = rng.uniform(1.0, n / 4.0)
    phase = rng.uniform(0, 2 * np.pi)
    t = np.arange(n)
    cos_perturb = 0.1 * np.cos(2 * np.pi * freq * t / n + phase)
    scale = 0.02 + 0.03 * rng.random()
    noise = 1.0 + scale * rng.standard_normal(n) + cos_perturb
    noise = np.maximum(noise, 0.1)
    perturbed = best_seq * noise
    s = perturbed.sum()
    if s < 1e-15:
        return best_seq
    perturbed = perturbed / s
    a_norm = normalize_to(perturbed, n)
    a_out, _ = nesterov_lp_optimize(a_norm, time_limit, solver, verbose=False, seed=seed)
    return a_out


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq, best_score, True
    return best_seq, best_score, False
