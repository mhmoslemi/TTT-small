"""
Minimize  2 * n * max(convolve(a,a)) / sum(a)**2
for a non-negative step sequence a (length n).

The objective is scale-invariant, so we work with sum(a)=1 and minimize
2*n*max(conv(a,a)).  Key structural facts exploited here:

  * max of the full autocorrelation can sit at the EDGES (conv[0]=a_0^2,
    conv[2n-2]=a_{n-1}^2), so tapering the boundaries is valuable.
  * A first-order peak-reducing direction is obtained from the LP
        min  t
        s.t. 2*conv(a,b)[k] + conv(a,a)[k] <= t   (top-K active lags k)
             b >= -a,  sum(b) = 0
    followed by a line search on alpha (a -> a + alpha*b).
  * A Nesterov subgradient method (active-set subgradient of the max)
    supplies cheap in-fill iterations.
  * A peak-targeted coordinate descent polishes the final sequence.
  * Multi-length exploration via interpolation avoids n-specific traps.

All helpers are top-level; no lambdas, no nested function definitions,
no filesystem/network IO.  `evaluate_sequence` is assumed available.
"""

import time
import numpy as np
import scipy.sparse as sp
import cvxpy as cp


# ----------------------------------------------------------------- helpers

def eval_safe(a):
    """Robust wrapper around evaluate_sequence."""
    try:
        lst = [float(x) for x in a]
        v = float(evaluate_sequence(lst))
        return v
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Clip to >=0 and normalise to unit sum."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / max(1, len(a))
    return a / s


def fast_autocorr(a):
    """Full autocorrelation via FFT, length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def peak_of(a):
    return float(fast_autocorr(a).max())


def simple_score(a):
    """2*n*max_conv / sum^2 computed directly from a (matches evaluator when sum>=0.01)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = a.size
    m = float(fast_autocorr(a).max())
    return 2.0 * n * m / (s * s)


def project_simplex(v, s=1.0):
    """Project v onto {x : x>=0, sum(x)=s} in O(n log n)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cand = cssv - s
    valid = u * np.arange(1, n + 1) > cand
    if not np.any(valid):
        return np.full(n, s / n)
    rho = int(np.nonzero(valid)[0][-1])
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def apply_taper(a, edge_frac=0.10):
    """Multiply by a smooth raised-cosine taper to suppress edge peaks."""
    n = a.size
    w = np.ones(n)
    half = max(1, int(round(edge_frac * n)))
    x = np.linspace(0.0, 1.0, half + 2)
    taper = 0.5 * (1.0 - np.cos(np.pi * x))
    w[:half + 1] = taper
    w[-(half + 1):] = taper[::-1]
    a = a * w
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.asarray(a, dtype=np.float64).copy()
    else:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Resample a unit-sum sequence to a new length, staying unit-sum & >=0."""
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    ox = np.linspace(0.0, 1.0, len(fn))
    nx = np.linspace(0.0, 1.0, new_n)
    gi = np.interp(nx, ox, fn)
    gi = np.maximum(gi, 0.0)
    gs = gi.sum()
    if gs < 1e-15:
        return np.ones(new_n) / new_n
    return gi / gs


def get_solver():
    for sname in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=sname)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return sname
        except Exception:
            continue
    return cp.ECOS


# --------------------------------------------------------------- directional LP

def lp_peak_direction(a, C, solver_name, K=150, verbose=False):
    """
    Solve  min t  s.t.  2*conv(a,b)[k]+C[k] <= t  (top-K active lags),
              b >= -a,  sum(b)=0.
    Returns (b_val, t_val) or None.  b_val is the peak-reducing direction.
    """
    n = a.size
    C = np.asarray(C, dtype=np.float64)
    K = min(K, 2 * n - 1)
    top = np.argsort(C)[::-1][:K]

    # Build sparse A (K x n): A[krow, j] = 2*a[j] if j in [k-n+1, k] ∩ [0,n-1]
    rows = []
    cols = []
    for r, k in enumerate(top):
        k = int(k)
        js = range(max(0, k - n + 1), min(n - 1, k) + 1)
        rows.extend([r] * len(js))
        cols.extend(list(js))
    rows = np.asarray(rows, dtype=np.int32)
    cols = np.asarray(cols, dtype=np.int32)
    vals = 2.0 * a[cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(K, n))

    b = cp.Variable(n)
    t = cp.Variable()
    rhs = -C[top]
    constraints = [A @ b - t <= rhs, b >= -a, cp.sum(b) == 0]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=solver_name, verbose=False)
    except Exception:
        if verbose:
            print("    LP solve exception")
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        if verbose:
            print("    LP status:", prob.status)
        return None
    bval = np.asarray(b.value, dtype=np.float64)
    bval = np.maximum(bval, -a)
    bval = bval - bval.mean()
    return bval, float(t.value if t.value is not None else np.nan)


# ----------------------------------------------------------------- line search

def line_search(a, direction, max_alpha=1.0, n_trials=8):
    """Choose alpha in [0, max_alpha] minimising peak(a+alpha*direction)."""
    n = a.size
    best_a = a.copy()
    best_peak = peak_of(a)
    neg = direction < -1e-18
    if neg.any():
        max_fea = float(np.min(-a[neg] / direction[neg]))
    else:
        max_fea = max_alpha
    max_alpha = min(max_alpha, max_fea * 0.98)
    if max_alpha <= 0.0:
        return a.copy(), best_peak
    for alpha in np.linspace(0.0, max_alpha, n_trials):
        tr = a + alpha * direction
        tr = np.maximum(tr, 0.0)
        s = tr.sum()
        if s < 1e-15:
            continue
        tr = tr / s
        pk = peak_of(tr)
        if pk < best_peak:
            best_peak = pk
            best_a = tr.copy()
    if best_a is not a:
        lo = max(0.0, (max_alpha * (n_trials - 2) / n_trials))
        hi = max_alpha
        for alpha in np.linspace(lo, hi, 8):
            tr = a + alpha * direction
            tr = np.maximum(tr, 0.0)
            s = tr.sum()
            if s < 1e-15:
                continue
            tr = tr / s
            pk = peak_of(tr)
            if pk < best_peak:
                best_peak = pk
                best_a = tr.copy()
    return best_a, best_peak


# ------------------------------------------------------------- subgradient step

def subgrad_of_max(a, C, top_k=6, sharpness=8.0):
    """Smoothed subgradient of max_k C[k] using top-k lags."""
    n = a.size
    order = np.argsort(C)[::-1][:top_k]
    mmax = C.max()
    w = np.exp(-sharpness * (mmax - C[order]) / max(1e-15, mmax))
    w = w / max(1e-30, w.sum())
    g = np.zeros(n)
    for wi, k in zip(w, order):
        k = int(k)
        j0 = max(0, k - n + 1)
        j1 = min(n, k + 1)
        idx = np.arange(j0, j1)
        g[idx] += (2.0 * wi) * a[k - idx]
    return g


def subgradient_descent(a0, time_limit, top_k=6, gamma=3.0, verbose=False):
    """Nesterov subgradient descent on the simplex."""
    a = norm_sum1(a0)
    n = a.size
    best_a = a.copy()
    best_m = peak_of(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(7)
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    step = 0
    while time.monotonic() - t0 < time_limit:
        t = step + 1.0
        beta = (t - 1.0) / (t + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy if sy > 1e-15 else x.copy()
        C = fast_autocorr(y)
        g = subgrad_of_max(y, C, top_k=top_k)
        eta = gamma / (np.sqrt(t) + 1.0)
        y_new = y - eta * g
        a_new = project_simplex(y_new, 1.0)
        x_prev = x
        x = a_new
        step += 1
        pk = peak_of(x)
        if pk < best_m - 1e-14:
            best_m = pk
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        if stall > 4000:
            pert = rng.normal(0.0, 0.02, n)
            xr = best_a + pert * best_a
            xr = np.maximum(xr, 1e-10)
            xr = xr / xr.sum()
            x = xr
            x_prev = xr
            stall = 0
        if verbose and step % 5000 == 0:
            print("    SGD step %d: J=%.6f" % (step, 2.0 * n * best_m))
    return best_a, best_m

                        best_m = pk

                        stall = 0
                        break
                if improved:
                    break
            if improved:

            i = int(rng.integers(n))

                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                pk = peak_of(a)
                if pk < best_m:
                    best_m = pk
                    best_a = a.copy()
                    stall = 0
    if verbose:
        print("    CoordDescend: J=%.6f" % (2.0 * n * best_m))
    return best_a, best_m


# ----------------------------------------------------------------- orchestrator

def update_best(state, a, label):
    """Update state['seq']/['score'] if a improves it.  (top-level, no closure)."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    an = a / s
    sc = simple_score(an)
    if sc < state['score']:
        state['score'] = sc
        state['seq'] = an.copy()
        el = time.monotonic() - state['t0']
        print("  *** NEW BEST [%s]: %.6f (n=%d, t=%.0fs)" % (label, sc, an.size, el))


def propose_candidate(budget_s=1800.0):
    t0 = time.monotonic()
    state = {'seq': None, 'score': float('inf'), 't0': t0}
    time_budget = min(float(budget_s) - 90.0, 1700.0)
    solver = get_solver()
    print("budget=%.0fs solver=%s" % (time_budget, solver))

    # --- initialisation: parent if available, else tapered base ---
    a0 = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size >= 50 and np.isfinite(p).all() and p.sum() > 0.01:
            a0 = norm_sum1(p)
            print("parent n=%d score=%.6f" % (a0.size, simple_score(a0)))
    except Exception as e:
        print("no parent: %s" % e)
    if a0 is None:
        n0 = 2000
        x = np.linspace(0.0, 1.0, n0)
        a0 = np.sin(np.pi * x) ** 2
        a0 = norm_sum1(a0)

    state['seq'] = a0.copy()
    state['score'] = simple_score(a0)
    print("start n=%d score=%.6f" % (a0.size, state['score']))

    def remaining():
        return time_budget - (time.monotonic() - t0)

    def lp_phase(a_in, tlim, K, tag):
        """Iterate: LP peak direction + line search."""
        a = norm_sum1(a_in)
        for it in range(20000):
            if (time.monotonic() - t0) > time_budget - tlim:
                break
            C = fast_autocorr(a)
            res = lp_peak_direction(a, C, solver, K=K, verbose=False)
            if res is None:
                break
            bval, _ = res
            a, _ = line_search(a, bval, max_alpha=1.0, n_trials=7)
            if it % 5 == 0:
                print("    %s it=%d J=%.6f (%.0fs)" %
                      (tag, it, 2.0 * a.size * peak_of(a), time.monotonic() - t0))
            update_best(state, a, "%s_%d" % (tag, it))
        update_best(state, a, tag)

    # === Phase 1: taper the boundaries ===
    if remaining() > 30:
        at = norm_sum1(apply_taper(state['seq'], 0.10))
        update_best(state, at, "taper")

    # === Phase 2: main LP peak reduction at parent length ===
    if remaining() > 90:
        t2 = min(remaining() * 0.45, 650.0)
        lp_phase(state['seq'], t2, K=150, tag="lp2")

    # === Phase 3: subgradient in-fill ===
    if remaining() > 60:
        t3 = min(remaining() * 0.15, 180.0)
        oa, _ = subgradient_descent(state['seq'], t3, top_k=6, gamma=3.0, verbose=False)
        update_best(state, oa, "sgd3")

    # === Phase 4: coordinate polish ===
    if remaining() > 40:
        t4 = min(remaining() * 0.10, 120.0)
        oa, _ = coord_descend_peak(state['seq'], t4, seed=11, verbose=False)
        update_best(state, oa, "cd4")

    # === Phase 5: second LP round (larger K) ===
    if remaining() > 90:
        t5 = min(remaining() * 0.25, 450.0)
        lp_phase(state['seq'], t5, K=250, tag="lp5")

    # === Phase 6: multi-length exploration ===
    if remaining() > 120:
        nb = state['seq'].size
        for delta in (300, -300, 600):
            nt = nb + delta
            if nt < 400 or nt > 4000:
                continue
            if remaining() < 60:
                break
            inter = interpolate_sequence(state['seq'], nt)
            print("  explore n=%d" % nt)
            lp_phase(inter, min(remaining() * 0.20, 130.0), K=120, tag="exp%d" % nt)
            oa, _ = coord_descend_peak(state['seq'], min(remaining() * 0.04, 40.0),
                                       seed=3, verbose=False)
            update_best(state, oa, "cd_exp%d" % nt)

    # === Phase 7: final coord polish ===
    if remaining() > 30:
        t7 = min(remaining() * 0.35, 150.0)
        oa, _ = coord_descend_peak(state['seq'], t7, seed=777, verbose=False)
        update_best(state, oa, "cd7")

    # === Phase 8: final LP if time remains ===
    if remaining() > 40:
        lp_phase(state['seq'], min(remaining() * 0.7, 250.0), K=300, tag="lp8")

    # --- final validation against the real evaluator ---
    if state['seq'] is not None:
        fins = eval_safe(state['seq'])
        if np.isfinite(fins):
            state['score'] = fins
        print("FINAL score=%.6f n=%d time=%.0fs" %
              (state['score'], state['seq'].size, time.monotonic() - t0))
        out = [float(v) for v in np.clip(state['seq'], 0.0, 1000.0)]
    else:
        out = [1.0]
    return out
