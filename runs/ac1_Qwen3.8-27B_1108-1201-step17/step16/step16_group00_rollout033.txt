def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_subgradient(a, ac, active_lags):
    """Compute exact subgradient from a set of active lags."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_lags:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * a[j_idx]
    return grad / len(active_lags)

def mirror_descent(a0, time_limit, verbose=True, c0=3.0, top_k=30, restart_every=5000):

    vel = np.zeros(n)

        M_curr = ac.max()

                    if 0 <= d2 < n:

                    scores[d] += add

            candidates = np.where(scores < scores[src])[0]
            if len(candidates) == 0:

            best_dst = candidates[np.argmin(scores[candidates])]
            dst = best_dst
            if dst == src:
                continue

            found = False
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue

                a_trial[dst] += amt

                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()

                    found = True
                    stall = 0
                    break

        if not improved_this_round:
            stall += 1

            j = int(rng.integers(n))

                a[i] -= amt
                a[j] += amt

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_conv_matrix(y, n):
    """Build the convolution constraint matrix T where T[k,i] = y[k-i] if valid."""
    total = 2 * n - 1
    T = np.zeros((total, n))
    for j in range(n):
        T[j:j + n, j] = y
    return T


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=400, use_all=False,
                   restart_every=150, max_active=1500):
    """LP-based Frank-Wolfe with diverse constraint selection, Nesterov acceleration, and restarts."""
    n = len(a0)
    total_constraints = 2 * n - 1
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    prev_y = a.copy()
    iter_count = 0
    stall = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        T_full = build_conv_matrix(y, n)
        constraint_vals = T_full @ y
        margins = M_curr - constraint_vals

        if use_all and total_constraints <= 3000:
            active_list = np.arange(total_constraints)
        else:
            # Diverse constraint selection
            n_active = min(max_active, total_constraints)
            sorted_idx = np.argsort(margins)
            # Half most binding, quarter random from middle, quarter from far
            n_bind = n_active // 2
            n_mid = n_active // 4
            n_far = n_active - n_bind - n_mid
            bind_sel = sorted_idx[:n_bind]
            mid_start = n_bind
            mid_end = min(n_bind + (total_constraints - n_bind) // 2, total_constraints)
            if mid_end > mid_start:
                mid_sel = sorted_idx[mid_start:mid_end]
                if len(mid_sel) > n_mid:
                    mid_sel = mid_sel[np.random.choice(len(mid_sel), n_mid, replace=False)]
            else:
                mid_sel = np.array([])
            far_start = mid_end if mid_end > mid_start else n_bind
            if total_constraints > far_start:
                far_sel = sorted_idx[far_start:far_start + n_far]
            else:
                far_sel = np.array([])
            active_list = np.unique(np.concatenate([bind_sel, mid_sel, far_sel]))[:n_active]

        T_sel = T_full[active_list, :]
        # Normalize RHS for numerical stability
        M_norm = M_curr
        rhs = np.ones(len(active_list))

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_norm * rhs]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-12:
            continue
        g_dir = b_val / b_sum

        # Safety check: verify direction doesn't violate too many constraints
        all_vals = T_full @ g_dir
        n_violated = np.sum(all_vals > M_curr + 1e-6)
        if n_violated > max(5, 0.01 * total_constraints):
            # Too many violations; try with more constraints next iter
            continue

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        my = fast_autocorr(y_new).max()
        mx = fast_autocorr(x_new).max()

        improved = False
        if mx < my:
            if mx < best_m - 1e-15:
                best_m = mx
                best_a = x_new.copy()
                improved = True
        else:
            x_new = y_new.copy()
            if my < best_m - 1e-15:
                best_m = my
                best_a = y_new.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1

        # Restart if stalled
        if stall > restart_every:
            stall = 0
            rng = np.random.default_rng(t)
            pert = rng.normal(0, 0.015, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            y = x_r
            x = x_r
            prev_y = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    LP-FW restart {restart_every} ({elapsed:.0f}s): J={J:.6f}")
            continue

        prev_y = y
        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}, stall={stall}")

    return best_a, best_m


def projected_subgradient(a0, time_limit, verbose=True, c0=2.0, top_k=20, restart_every=3000):
    """Projected subgradient descent with exact subgradients from active lags."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Active lags: top k
        k_max = min(top_k, len(ac))
        active_lags = np.argsort(ac)[::-1][:k_max]

        # Subgradient: average of gradients from active lags
        grad = np.zeros(n)
        for k in active_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * a[j_idx]
        grad /= k_max

        # Step size
        eta = c0 / (np.sqrt(t) + 1.0)

        # Gradient step
        a_try = a - eta * grad

        # Project onto simplex {a >= 0, sum(a) = 1}
        # Use bisection on tau
        a_try = np.sort(a_try)[::-1]
        cumsum = np.cumsum(a_try)
        rho = np.nonzero(a_try * np.arange(1, n + 1) > (cumsum - 1))[0][-1]
        theta = (cumsum[rho] - 1) / (rho + 1.0)
        a_new = np.maximum(a - eta * grad - theta, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new

        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r
            top_k = min(top_k + 5, 50)

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PSub t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Phase 1: LP-FW on parent
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-FW main (n={n0}) ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t1, solver, verbose=True, top_K=400, restart_every=120)
        try_update(a_out, f"lpfw_p1_n{n0}")

    # Phase 2: LP-FW with different params (more constraints)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW diverse (n={n_best}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30 and n_best <= 3500:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, top_K=600,
                                   max_active=2000, restart_every=100)
        try_update(a_out, f"lpfw_p2_n{n_best}")

    # Phase 3: Projected subgradient
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Projected subgradient (n={n_best}) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = projected_subgradient(a_in, t3, verbose=True, c0=2.0, top_k=20)
        try_update(a_out, f"psub_n{n_best}")

    # Phase 4: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate descent (n={n_best}) ===")
    t4 = min(remaining * 0.10, 130.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_p4_n{n_best}")

    # Phase 5: Try different lengths
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length exploration ===")
    test_lengths = []
    for delta in [200, 400, -200, 600]:
        ln = n_best + delta
        if 2400 <= ln <= 4500:
            test_lengths.append(ln)

    for new_n in test_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n  --- Trying n={new_n} ---")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick LP-FW
        t5a = min(remaining * 0.35, 180.0)
        if t5a > 25 and new_n <= 3800:
            a_out, _ = lp_frank_wolfe(a_long, t5a, solver, verbose=False, top_K=300, restart_every=100)
            try_update(a_out, f"lpfw_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            # Quick coord
            t5b = min(remaining * 0.20, 80.0)
            if t5b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_polish(a_in, t5b, verbose=False, rng_seed=new_n)
                try_update(a_out, f"coord_n{new_n}")

    # Phase 6: Final LP-FW on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP-FW (n={n_best}) ===")
    if remaining > 40 and n_best <= 3800:
        t6 = min(remaining * 0.35, 250.0)
        if t6 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True, top_K=500,
                                       max_active=2000, restart_every=80)
            try_update(a_out, f"lpfw_final_n{n_best}")

    # Phase 7: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord (n={n_best}) ===")
    if remaining > 15:
        t7 = min(remaining * 0.5, 130.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t7, verbose=True, rng_seed=777)
            try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = (np.sin(np.pi * x)) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
