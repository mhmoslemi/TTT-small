2. Semi-smooth Newton (Nesterov) acceleration with backtracking line search

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:

def build_lp_constraint(y, k, n):
    """Build coefficient vector for constraint conv(y,b)[k] <= M."""
    coeff = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    for i in range(i_start, i_end):
        j = k - i
        if 0 <= j < n:
            coeff[j] += y[i]
    return coeff

def solve_lp_with_cutting_plane(y, M, initial_lags, solver, max_extra_constraints=100, max_iters=10, time_limit=30.0):
    """
    Solve the LP with cutting-plane: start with initial_lags, add violated constraints.
    Returns the optimal direction g (normalized to sum=1) or None.
    """
    n = len(y)
    t_start = time.monotonic()
    
    active_lags = list(initial_lags)
    
    for iter in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            break
        
        b = cp.Variable(n, nonneg=True)
        constraints = []
        for k in active_lags:
            coeff = build_lp_constraint(y, k, n)
            if coeff.max() > 1e-30:
                constraints.append(coeff @ b <= M)
        
        if not constraints:
            return np.ones(n) / n
        
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        
        b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
        s = b_val.sum()
        if s < 1e-15:
            return None
        
        # Check for violated constraints (cutting plane)
        g = b_val / s
        ac_cross = fast_autocorr(y)  # This is conv(y,y), not what we want
        # We need conv(y, g) for all lags
        # conv(y, g)[k] = sum_i y[i]*g[k-i]
        # Use FFT for efficiency
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Y = np.fft.rfft(y, fft_size)
        G = np.fft.rfft(g, fft_size)
        conv_yg = np.fft.irfft(Y * G, fft_size)[:size]
        
        violated = []
        for k in range(size):
            if k not in active_lags and conv_yg[k] > M * 1.001:
                violated.append(k)
        
        if len(violated) == 0:
            break
        
        # Add most violated constraints
        violated_sorted = sorted(violated, key=lambda k: -conv_yg[k])
        n_add = min(len(violated_sorted), 20)
        for k in violated_sorted[:n_add]:
            active_lags.append(k)
        
        if len(active_lags) - len(initial_lags) >= max_extra_constraints:
            # Re-solve with all accumulated constraints
            break
    
    # Final solve with all constraints
    b = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active_lags:
        coeff = build_lp_constraint(y, k, n)
        if coeff.max() > 1e-30:
            constraints.append(coeff @ b <= M)
    
    if not constraints:
        return np.ones(n) / n
    
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def lp_frank_wolfe_accelerated(a0, time_limit, solver, verbose=True,
                                initial_lags=15, max_lags_grow=100,
                                grow_period=30, nesterov=True, rng_seed=42):
    """

    """

    x = y.copy() if nesterov else None

    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    n_lags = initial_lags
    stall = 0
    max_stall = 25
    prev方向 = None
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M = ac.max()
        
        # Get active lags
        lag_indices = np.argsort(ac)[::-1]
        active_lags = lag_indices[:n_lags]
        
        # Solve LP
        g_dir = solve_lp_with_cutting_plane(y, M, active_lags, solver,
                                             max_extra_constraints=50,
                                             max_iters=5, time_limit=20.0)
        
        if g_dir is None:
            g_dir = np.ones(n) / n
            stall += 5
        else:
            # Check direction quality
            if prev_direction is not None and np.linalg.norm(g_dir - prev_direction) < 1e-6:
                stall += 3
            else:
                stall = 0
            prev_direction = g_dir.copy()
        
        # Backtracking line search
        base_alpha = 2.0 / (t + 2.0)
        alphas = [base_alpha, base_alpha * 0.5, base_alpha * 2.0, base_alpha * 0.25, base_alpha * 4.0]
        best_alpha = base_alpha
        best_trial_m = best_m
        
        for alpha in alphas:
            alpha = float(np.clip(alpha, 0.005, 0.95))
            y_trial = (1.0 - alpha) * y + alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            s = y_trial.sum()
            if s < 1e-15:
                continue
            y_trial = y_trial / s
            M_trial = fast_autocorr(y_trial).max()
            if M_trial < best_trial_m:
                best_trial_m = M_trial
                best_alpha = alpha
        
        alpha = best_alpha
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s
        
        # Evaluate and track best
        M_y = fast_autocorr(y_new).max()
        if M_y < best_m - 1e-15:
            best_m = M_y
            best_a = y_new.copy()
        
        # Nesterov extrapolation
        if nesterov:
            if t == 1:
                x = y_new.copy()
            else:
                beta_t = (t - 1.0) / (t + 2.0)
                x_new = y_new + beta_t * (y_new - y)
                x_new = np.maximum(x_new, 0.0)
                sx = x_new.sum()
                if sx < 1e-15:
                    x_new = y_new.copy()
                else:
                    x_new = x_new / sx
                x = x_new
            M_x = fast_autocorr(x).max()
            if M_x < best_m - 1e-15:
                best_m = M_x
                best_a = x.copy()
        
        y = y_new
        
        # Grow active set
        if t % grow_period == 0 and n_lags < max_lags_grow:
            n_lags = min(n_lags + 5, max_lags_grow)

            y = y / y.sum()

            if verbose:

                print(f"    LW restart t={t} ({elapsed:.1f}s): J={J:.6f}, lags={n_lags}")

            J = 2.0 * n * best_m
            print(f"    LW t={t} ({elapsed:.1f}s): J={J:.6f}, M={M:.2e}, lags={n_lags}, alpha={alpha:.4f}")
    
    return best_a, best_m

def coordinate_descent_structured(a0, time_limit, verbose=True, rng_seed=123):
    """
    Structure-aware coordinate descent: for each coordinate, try multiple
    values and keep the best. Exploits the bilinear structure of conv(a,a).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 50
    order = np.arange(n)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        np.random.default_rng(rng_seed + int((time.monotonic() - t0) * 10)).shuffle(order)
        
        improved = False
        ac = fast_autocorr(a)
        M_peak = ac.max()
        peak_lags = np.argsort(ac)[::-1][:10]
        
        # Focus on coordinates involved in peak lags
        hot_coords = set()
        for k in peak_lags:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                hot_coords.add(i)
                j = k_int - i
                if 0 <= j < n:
                    hot_coords.add(j)
        
        # Process hot coords first, then random others
        hot_list = list(hot_coords)
        rng.shuffle(hot_list)
        cold_list = [i for i in order if i not in hot_coords][:n // 4]
        process_list = hot_list + cold_list
        
        for i in process_list:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue
            
            # Try multiple candidate values for a[i]
            # Redistribute the difference to a random other coordinate
            candidates = [a[i] * 0.0, a[i] * 0.3, a[i] * 0.7, a[i] * 1.3, a[i] * 2.0]
            # Also try moving to specific targets
            j = int(rng.integers(n))
            if j == i:
                j = (j + 1) % n
            
            best_trial_a = a.copy()
            best_trial_m = best_m
            
            for new_val in candidates:
                delta = new_val - a[i]
                a_trial = a.copy()
                a_trial[i] = max(0.0, new_val)
                a_trial[j] = max(0.0, a[j] + delta)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_t = fast_autocorr(a_trial).max()
                if M_t < best_trial_m - 1e-15:
                    best_trial_m = M_t
                    best_trial_a = a_trial.copy()
            
            if best_trial_m < best_m - 1e-15:
                a = best_trial_a
                best_m = best_trial_m
                best_a = a.copy()
                improved = True
        
        if not improved:
            stall += 1
            # Escape: random perturbation
            n_perturb = max(5, n // 50)
            for _ in range(n_perturb):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.05, 0.2)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            improved = True
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            if a.sum() > 1e-15:
                                a = a / a.sum()
        
        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def amd_subgradient(a0, time_limit, c0=2.0, threshold=0.999, verbose=True, rng_seed=77):
    """Mirror descent with exact subgradient for final sub-polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 2000
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        M = ac.max()
        
        # Exact subgradient from active set
        active = np.where(ac >= threshold * M)[0]
        if len(active) == 0:
            active = np.array([ac.argmax()])
        
        g = np.zeros(n)
        for k in active:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            g[idx] += 2.0 * y[j_idx]
        
        gn = np.linalg.norm(g)
        if gn > 1e-15:
            g = g / gn
        
        eta = c0 / (np.sqrt(t) + 1.0) * np.sqrt(n)
        log_a = np.log(np.maximum(y, 1e-15)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

            stall = 0
            noise = rng.normal(0, 0.005, n) * best_a
            x = np.maximum(best_a + noise, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def generate_initialization(n, seed=0):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    
    inits = []

    inits.append(np.sin(np.pi * x) ** 2)
    # Raised cosine with different params
    inits.append(0.5 * (1 + np.cos(2 * np.pi * x)))
    # Gaussian-like
    center = n // 2
    sigma = n / 4.0
    g = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(g)
    # Triangular
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1
    inits.append(tri)
    # Sin^2 with small oscillation
    inits.append(np.sin(np.pi * x) ** 2 + 0.05 * np.sin(4 * np.pi * x) * np.sin(np.pi * x))
    # Power-law taper
    inits.append(np.sin(np.pi * x) ** 1.5)
    
    result = []
    for init in inits:
        init = np.maximum(init, 0)
        s = init.sum()
        if s > 1e-10:
            result.append(init / s)
    
    return result

                norm_sum1(best_seq), t4, solver,
                verbose=True, initial_lags=40, max_lags_grow=150,
                grow_period=15, nesterov=True, rng_seed=4)
            try_update(a_out, f"LP3_n{n_best}")
    
    # === Phase 5: Coordinate descent again ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 5: CD round 2 (n={n_best}) ===")
        t5 = min(remaining * 0.10, 120.0)
        if t5 > 15:
            a_out, _ = coordinate_descent_structured(norm_sum1(best_seq), t5, verbose=True, rng_seed=5)
            try_update(a_out, f"CD2_n{n_best}")
    
    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 6: Multi-length exploration ===")
        # Try larger n
        n_large = min(n_best + 300, 3000)
        if n_large > n_best:
            print(f"  Trying n={n_large}")
            a_long = interpolate_sequence(best_seq, n_large)
            t6a = min(remaining * 0.25, 150.0)
            if t6a > 30:
                a_out, _ = lp_frank_wolfe_accelerated(
                    norm_sum1(a_long), t6a, solver,
                    verbose=True, initial_lags=20, max_lags_grow=80,
                    grow_period=30, nesterov=True, rng_seed=6)
                try_update(a_out, f"LP_long_n{n_large}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        n_best = len(best_seq)
        
        # Try smaller n
        n_small = max(n_best - 300, 1200)
        if remaining > 60 and n_small < len(best_seq):
            print(f"  Trying n={n_small}")
            a_short = interpolate_sequence(best_seq, n_small)
            t6b = min(remaining * 0.20, 100.0)
            if t6b > 20:
                a_out, _ = lp_frank_wolfe_accelerated(
                    norm_sum1(a_short), t6b, solver,
                    verbose=True, initial_lags=20, max_lags_grow=80,
                    grow_period=30, nesterov=True, rng_seed=7)
                try_update(a_out, f"LP_short_n{n_small}")
    
    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Final LP polish (n={n_best}) ===")
        t7 = min(remaining * 0.30, 150.0)
        if t7 > 15:
            a_out, _ = lp_frank_wolfe_accelerated(
                norm_sum1(best_seq), t7, solver,
                verbose=True, initial_lags=30, max_lags_grow=100,
                grow_period=15, nesterov=True, rng_seed=8)
            try_update(a_out, f"LP7_n{n_best}")
    
    # === Phase 8: AMD sub-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 8: AMD sub-polish (n={n_best}) ===")
        t8 = min(remaining * 0.30, 100.0)
        if t8 > 10:
            a_out, _ = amd_subgradient(norm_sum1(best_seq), t8, c0=1.5,
                                       threshold=0.9995, verbose=True, rng_seed=9)
            try_update(a_out, f"AMD_n{n_best}")
    
    # === Phase 9: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 9: Final CD (n={n_best}) ===")
        t9 = min(remaining * 0.40, 80.0)
        if t9 > 10:
            a_out, _ = coordinate_descent_structured(norm_sum1(best_seq), t9, verbose=True, rng_seed=10)
            try_update(a_out, f"CD_final_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
