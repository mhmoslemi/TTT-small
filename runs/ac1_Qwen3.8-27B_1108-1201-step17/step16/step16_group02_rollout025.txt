def nesterov_mirror_descent_v2(a0, time_limit, initial_temp=0.01, final_temp=0.001,

    rng = np.random.default_rng(777)

    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    max_stall = 500
    momentum = 0.0
    momentum_target = 0.9

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive temperature (exponential decay)
        frac = min(1.0, t / 3000.0)
        temp = initial_temp * ((final_temp / initial_temp) ** frac)
        temp = max(temp, 1e-8)

        # Momentum (Nesterov)
        beta_t = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = momentum_target * (1.0 - 1.0 / beta_t)
        y = x + momentum * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        half = (len(ac) + 1) // 2
        ac_half = ac[:half]

        # Soft-max subgradient
        log_ac = np.log(np.maximum(ac_half, 1e-30) / temp)
        log_ac -= log_ac.max()
        weights = np.exp(log_ac)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            top_idx = np.argsort(ac_half)[::-1][:min(20, len(ac_half))]
            weights = np.zeros_like(ac_half)
            weights[top_idx] = 1.0
            w_sum = weights.sum()
        weights /= w_sum

        # Compute subgradient
        grad = np.zeros(n)
        significant = np.where(weights > 1e-10)[0]
        for k_int in significant:
            w = weights[k_int]
            k_val = int(k_int)
            for i in range(n):
                j = k_val - i
                if 0 <= j < n:
                    grad[i] += 2.0 * w * y[j]

        # Multiplicative update (mirror descent)
        eta = gamma / (math.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new, _, _ = compute_conv_max_and_lag(x)
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Restart with perturbation
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(100, max_stall // 2)
            pert = rng.normal(0, 0.003, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            gamma = max(0.5, gamma * 0.7)

        # Ergodic iterate check
        if t % 2000 == 0:
            avg = (x + best_a) / 2.0
            avg = np.maximum(avg, 0.0)
            s_avg = avg.sum()
            if s_avg > 1e-15:
                avg = avg / s_avg
                M_avg, _, _ = compute_conv_max_and_lag(avg)
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.6f}, temp={temp:.6f}, stall={stall_count}")

    return best_a, best_m

def gradient_guided_cd(a0, time_limit, verbose=True, rng_seed=42):
    """Grouped coordinate descent using gradient information to select moves."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m, _, _ = compute_conv_max_and_lag(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 10

    group_size = max(2, n // 500)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False

        # Compute gradient at current max lag
        M_curr, k_star, ac_curr = compute_conv_max_and_lag(a)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]

        # Identify "bad" coordinates (high gradient * value) and "good" coordinates
        contributions = a * np.maximum(grad, 0)
        n_bad = max(1, n // 10)
        n_good = max(1, n // 10)
        bad_idx = np.argsort(contributions)[::-1][:n_bad]
        good_idx = np.argsort(contributions)[:n_good]

        for i in bad_idx:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-10:
                continue

            for j in good_idx:
                if j == i:
                    continue
                # Try moving mass from i to j
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial, _, _ = compute_conv_max_and_lag(a_trial)
                    if m_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break

        # Also try random pairs
        for _ in range(min(20, n // 50)):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-10:
                continue
            for frac in [0.01, 0.03, 0.08]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial, _, _ = compute_conv_max_and_lag(a_trial)
                if m_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            # Escap
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                m_new, _, _ = compute_conv_max_and_lag(a)
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()

        if verbose and (stall % 5 == 0):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    GGCD ({elapsed:.1f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent sequence
    n_base = 2400
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        # Raise-cosine window initialization
        n_init = 2400
        x = np.linspace(0, 1, n_init)
        best_seq = 0.5 * (1 + np.cos(2 * np.pi * x))
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n0 = len(best_seq)

    # === Phase 1: FW with full LP (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.45, 700.0)
    print(f"\n=== Phase 1: FW-full n={n0} (budget={t1:.0f}s) ===")
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = frank_wolfe_full(a_in, t1, solver, verbose=True, use_away=True)
        try_update(a_out, f"FW_n{n0}")

    # === Phase 2: NMD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 300.0)
    print(f"\n=== Phase 2: NMD n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_mirror_descent_v2(a_in, t2, initial_temp=0.008, final_temp=0.001,
                                              c0=3.0, verbose=True)
        try_update(a_out, f"NMD_n{len(best_seq)}")

    # === Phase 3: Run FW again from NMD result ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: FW round2 n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = frank_wolfe_full(a_in, t3, solver, verbose=True, use_away=True)
        try_update(a_out, f"FW2_n{len(best_seq)}")

    # === Phase 4: Gradient-guided CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 4: GGCD n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = gradient_guided_cd(a_in, t4, verbose=True, rng_seed=123)
        try_update(a_out, f"GGCD_{len(best_seq)}")

    # === Phase 5: NMD with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 100.0)
    print(f"\n=== Phase 5: NMD-alt n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_mirror_descent_v2(a_in, t5, initial_temp=0.003, final_temp=0.0005,
                                              c0=5.0, verbose=False)
        try_update(a_out, f"NMD_alt_{len(best_seq)}")

    # === Phase 6: Try different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        n_alt = n0 + 400
        print(f"\n=== Phase 6: Alt length n={n_alt} (budget={remaining:.0f}s) ===")
        a_long = interpolate_sequence(best_seq, n_alt)
        a_in = norm_sum1(a_long)
        t6 = min(remaining * 0.5, 200.0)
        if t6 > 30:
            a_out, _ = frank_wolfe_full(a_in, t6, solver, verbose=True, use_away=True)
            try_update(a_out, f"FW_n{n_alt}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t6b = min(remaining * 0.5, 120.0)
            if t6b > 20:
                a_in2 = norm_sum1(best_seq)
                a_out, _ = nesterov_mirror_descent_v2(a_in2, t6b, initial_temp=0.005,
                                                      final_temp=0.001, c0=3.0, verbose=False)
                try_update(a_out, f"NMD_n{n_alt}")

    # === Phase 7: Final FW polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 7: Final FW n={n_best} (budget={remaining:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        t7 = min(remaining * 0.7, 100.0)
        if t7 > 20:
            a_out, _ = frank_wolfe_full(a_in, t7, solver, verbose=False, use_away=True)
            try_update(a_out, f"FW_final_{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.6, 50.0)
            if t7b > 10:
                a_in2 = norm_sum1(best_seq)
                a_out, _ = gradient_guided_cd(a_in2, t7b, verbose=False, rng_seed=999)
                try_update(a_out, f"GGCD_final_{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
