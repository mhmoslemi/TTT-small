def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def fast_conv(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_ab(a, b):
    """Compute conv(a, b) using FFT. Linear in b."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def solve_lp_cutting(a, M, K=80, max_rounds=5, time_limit=5.0):
    """
    Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0
    using cutting-plane method with K initial constraints.
    Returns the optimal b (or best found within time).
    """
    n = len(a)
    t0 = time.monotonic()
    
    # Compute conv(a, a) to identify active constraints
    ac = fast_conv(a)
    
    # Select top-K constraints (those closest to M)
    sorted_idx = np.argsort(ac)[::-1]
    active_set = sorted_idx[:K].copy()
    active_set.sort()
    
    best_b = None
    best_obj = -1
    
    for round in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Build constraint matrix for active set
        # T[k, i] = a[i] if 0 <= k-i < n, else 0
        # For each active constraint k: sum_{i} a[i] * b[k-i] <= M
        # Which means: sum_i T[k,i] * b[i] <= M where T[k,i] = a[k-i] if 0<=k-i<n else 0
        m = len(active_set)
        T = np.zeros((m, n))
        for row, k in enumerate(active_set):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T[row, i] = a[k_int - i]
        
        # Solve LP: max sum(b) s.t. T @ b <= M, b >= 0
        c = -np.ones(n)
        try:
            res = linprog(c, A_ub=T, b_ub=np.full(m, M), bounds=[(0, None)]*n, method='highs')
            if res.status != 0:
                continue
            b_sol = np.maximum(res.x, 0)
        except Exception:
            continue
        
        obj = b_sol.sum()
        if obj <= best_obj + 1e-15:
            # Check for violations
            conv_ab = fast_conv_ab(a, b_sol)
            violations = conv_ab - M
            if violations.max() < 1e-10 * max(1, M):
                best_b = b_sol
                best_obj = obj
                break
            # Add violated constraints
            viol_idx = np.where(violations > 1e-12 * max(1, M))[0]
            if len(viol_idx) > 0:
                new_set = np.union1d(active_set, viol_idx[:min(20, len(viol_idx))])
                active_set = new_set[:min(K + round * 20, 2*n-1)]
            continue
        
        best_b = b_sol
        best_obj = obj
        
        # Check for violations
        conv_ab = fast_conv_ab(a, b_sol)
        violations = conv_ab - M
        if violations.max() < 1e-10 * max(1, M):
            break
        # Add violated constraints
        viol_idx = np.where(violations > 1e-12 * max(1, M))[0]
        if len(viol_idx) > 0:
            new_set = np.union1d(active_set, viol_idx[:min(20, len(viol_idx))])
            active_set = new_set[:min(K + (round+1) * 20, 2*n-1)]
    
    return best_b


def lp_frank_wolfe_cutting(a0, time_limit, verbose=True, K=80, max_iters=50000):
    """LP-based Frank-Wolfe with cutting planes and Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()  # Nesterov lookahead
    x = a.copy()  # current point
    x_prev = a.copy()
    iter_count = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov lookahead
        if t == 1:
            y = x.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            y = x + beta * (x - x_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        
        # Compute M at y
        ac_y = fast_conv(y)
        M_curr = ac_y.max()
        if M_curr < 1e-14:
            break
        
        # Solve LP at y
        b_sol = solve_lp_cutting(y, M_curr, K=K, max_rounds=3, time_limit=0.1)
        if b_sol is None or b_sol.sum() < 1e-15:
            # Fallback: use y itself
            g_dir = y.copy()
        else:
            g_dir = b_sol / b_sol.sum()
        
        # FW update
        alpha = 2.0 / (t + 2.0)
        x_prev = x.copy()
        x_new = (1.0 - alpha) * x + alpha * g_dir
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / sx
        x = x_new
        
        iter_count += 1
        
        # Track best
        M_x = fast_conv(x).max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x.copy()
        
        M_y = fast_conv(y).max()
        if M_y < best_m - 1e-15:
            best_m = M_y
            best_a = y.copy()
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}")
    
    return best_a, best_m, iter_count

def coordinate_descent_peak(a0, time_limit, verbose=True, rng_seed=123):
    """
    Coordinate descent targeting the peak of the autocorrelation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        peak_idx = int(np.argmax(ac))
        M_curr = ac[peak_idx]
        
        # Find which (i, j) pairs contribute most to the peak
        # conv(a,a)[k] = sum_i a[i]*a[k-i]
        # At k = peak_idx: sum_i a[i]*a[peak_idx-i]
        i_start = max(0, peak_idx - n + 1)
        i_end = min(n, peak_idx + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_idx - idx]
        
        if contrib.max() < 1e-20:
            stall += 1
            break
        
        # Compute sensitivity: for each position j, how much does it contribute to top peaks?
        top_peaks = np.argsort(ac)[::-1][:8]
        sensitivity = np.zeros(n)
        for pk in top_peaks:
            pk_int = int(pk)
            for j in range(n):
                kj = pk_int - j
                if 0 <= kj < n:
                    sensitivity[j] += a[kj]
        
        # Try to move mass from high-contributing to low-sensitivity positions
        src_ranking = np.argsort(contrib)[::-1][:10]
        dst_candidates = np.argsort(sensitivity)[:max(30, n // 20)]
        
        improved = False
        for src_rank in src_ranking:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[src_rank])
            if a[src] < 1e-15:
                continue
            
            for dst in dst_candidates:
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.01, 0.03, 0.1]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

import numpy as np
import time
from scipy.optimize import linprog

_lp_method = None


def get_lp_method():

    if _lp_method is not None:
        return _lp_method
    try:

        if res.status == 0:
            _lp_method = 'highs'
            print("  LP method: HiGHS (scipy)")
            return _lp_method
    except Exception:

    _lp_method = 'revised'

def fast_conv(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_ab(a, b):
    """Compute conv(a, b) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def evaluate_safe(a):

        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_T_rows(a, active_idx, n):
    """Build constraint matrix rows for given active lags."""
    m = len(active_idx)
    T = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        js = np.arange(i_start, i_end)
        T[row, js] = a[k_int - js]
    return T


def solve_lp_cutting(a, M, K=80, max_rounds=4, time_limit=0.15):
    """
    Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0
    using cutting-plane method.
    """
    n = len(a)
    t0 = time.monotonic()

    ac = fast_conv(a)
    sorted_idx = np.argsort(ac)[::-1]
    active_set = np.sort(sorted_idx[:K])

    best_b = None
    best_obj = -1.0

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        m = len(active_set)
        T = build_T_rows(a, active_set, n)

        c = -np.ones(n)
        b_ub = np.full(m, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs')
            if res.status != 0:
                continue
            b_sol = np.maximum(res.x, 0.0)
        except Exception:
            continue

        obj = b_sol.sum()
        if obj < 1e-15:
            continue

        conv_ab = fast_conv_ab(a, b_sol)
        violations = conv_ab - M
        max_viol = violations.max()

        if max_viol < 1e-9 * max(1.0, M):
            if obj > best_obj:
                best_b = b_sol
                best_obj = obj
            break

        if obj > best_obj:
            best_b = b_sol
            best_obj = obj

        viol_mask = violations > 1e-10 * max(1.0, M)
        viol_count = int(viol_mask.sum())
        if viol_count == 0:
            break

        viol_idx = np.where(viol_mask)[0]
        n_add = min(30, len(viol_idx))
        new_active = np.union1d(active_set, viol_idx[:n_add])
        max_constraints = min(2 * n - 1, K + (rnd + 1) * 30)
        active_set = np.sort(new_active)[:max_constraints]

    return best_b


def lp_frank_wolfe_cutting(a0, time_limit, K=80, verbose=True, allow_restart=True):
    """LP-based Frank-Wolfe with cutting planes and Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    iter_count = 0
    stall = 0
    rng = np.random.default_rng(7)

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov lookahead
        if t <= 2:
            y = x.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            y = x + beta * (x - x_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        ac_y = fast_conv(y)
        M_curr = float(ac_y.max())
        if M_curr < 1e-15:
            break

        b_sol = solve_lp_cutting(y, M_curr, K=K, max_rounds=3, time_limit=0.12)
        if b_sol is None or b_sol.sum() < 1e-15:
            g_dir = y.copy()
        else:
            g_dir = b_sol / b_sol.sum()

        # FW step
        alpha = 2.0 / (t + 2.0)
        x_prev = x.copy()
        x_new = (1.0 - alpha) * x + alpha * g_dir
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / sx
        x = x_new
        iter_count += 1

        M_x = fast_conv(x).max()
        improved = False
        if M_x < best_m - 1e-14:
            best_m = float(M_x)
            best_a = x.copy()
            improved = True

        M_y = fast_conv(y).max()
        if M_y < best_m - 1e-14:
            best_m = float(M_y)
            best_a = y.copy()
            improved = True

        if improved:
            stall = 0
        else:
            stall += 1

        if allow_restart and stall > 300:
            stall = 0
            x = best_a + rng.normal(0, 0.01, n) * best_a
            x = np.maximum(x, 0.0)
            x = x / x.sum()
            x_prev = x.copy()

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.8f}")

    return best_a, best_m, iter_count


def subgradient_polish(a0, time_limit, top_k=25, beta=3.0, verbose=True):
    """Nesterov-accelerated multiplicative subgradient descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_conv(a).max())
    t0 = time.monotonic()
    x = a.copy()
    x_prev = a.copy()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        ac = fast_conv(y)
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            js = np.arange(j_start, j_end)
            grad[js] += 2.0 * wi * y[k_int - js]

        eta = 5.0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = float(fast_conv(x).max())
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SubGrad t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_descent_peak(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent targeting peak reduction with symmetrization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_conv(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        peak_idx = int(np.argmax(ac))
        M_curr = ac[peak_idx]

        i_start = max(0, peak_idx - n + 1)
        i_end = min(n, peak_idx + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_idx - idx]

        if contrib.max() < 1e-20:
            break

        # Sensitivity: contribution to top peaks
        top_peaks = np.argsort(ac)[::-1][:10]
        sensitivity = np.zeros(n)
        for pk in top_peaks:
            pk_int = int(pk)
            j_start = max(0, pk_int - n + 1)
            j_end = min(n, pk_int + 1)
            js = np.arange(j_start, j_end)
            sensitivity[js] += a[pk_int - js]

        src_ranking = np.argsort(contrib)[::-1][:8]
        dst_candidates = np.argsort(sensitivity)[:max(40, n // 15)]

        improved = False
        for src_rank in src_ranking:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[src_rank])
            if a[src] < 1e-15:
                continue
            for dst in dst_candidates:
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = float(fast_conv(a_trial).max())
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Try symmetrization
            a_sym = (a + a[::-1]) / 2.0
            a_sym = np.maximum(a_sym, 0.0)
            s_sym = a_sym.sum()
            if s_sym > 1e-15:
                a_sym = a_sym / s_sym
                M_sym = float(fast_conv(a_sym).max())
                if M_sym < best_m - 1e-16:
                    a = a_sym
                    best_m = M_sym
                    best_a = a.copy()
                    stall = 0
                    continue
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.005)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                    M_new = float(fast_conv(a).max())
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = a / a.sum()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordDesc ({elapsed:.1f}s): J={J:.6f}")
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update_state(state, a, label):
    """Update state dict with new best if a yields lower score."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < state['best_score']:
        state['best_score'] = sc
        state['best_seq'] = a_norm.copy()
        elapsed = time.monotonic() - state['t_global']
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
