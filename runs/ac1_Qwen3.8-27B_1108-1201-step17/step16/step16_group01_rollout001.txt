import numpy as np

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def objective(a):
    """Compute the objective value (scale-invariant)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def mirror_descent_backtrack(a0, time_limit, c_init=5.0, max_iter=200000, verbose=True, rng_seed=42):
    """
    Mirror descent with backtracking line search.
    Uses exact subgradient from the active set (peaks of autocorrelation).
    """
    n = len(a0)
    a = normalize(a0.copy())
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_obj = objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 5000
    c = c_init
    
    for t in range(1, max_iter + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute subgradient
        ac_y = fast_autocorr(y)
        M = ac_y.max()

        g = np.zeros(n)
        for k in active:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            g[idx] += 2.0 * y[j_idx]
        
        gn = np.linalg.norm(g)
        if gn < 1e-30:
            break
        g = g / gn
        
        # Backtracking line search in log-space
        eta = c / (np.sqrt(t) + 1.0) * np.sqrt(n)
        
        # Try the step, if it doesn't improve, shrink
        for shrink in range(20):
            log_a = np.log(np.maximum(y, 1e-15)) - eta * g
            max_log = log_a.max()
            a_candidate = np.exp(log_a - max_log)
            a_candidate = np.maximum(a_candidate, 1e-30)
            a_candidate = a_candidate / a_candidate.sum()
            
            obj_candidate = objective(a_candidate)
            if obj_candidate < objective(y) or shrink == 19:
                break
            eta *= 0.7
        
        x_prev = x
        x = a_candidate
        
        if x.max() > 1e10:  # overflow protection
            x = np.zeros(n)
            x[rng.integers(n)] = 1.0
            x_prev = x.copy()
            continue
        
        obj_x = objective(x)
        if obj_x < best_obj - 1e-12:
            best_obj = obj_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

            stall = 0

            x_prev = x.copy()
            c *= 1.3
            max_stall = int(max_stall * 1.1)
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MD t={t} ({elapsed:.1f}s): obj={best_obj:.6f}, stall={stall}")
    
    return best_a, best_obj

def pair_swap_cd(a0, time_limit, verbose=True, rng_seed=42):

        ac = fast_autocorr(a)
        M = ac.max()
        if M < 1e-20:
            break

        # Compute "hotness" of each position

            j_idx = ki - idx
            hotness[idx] += a[j_idx]

            if time.monotonic() - t0 > time_limit:

                    if obj_t < best_obj - 1e-12:
                        a = a_trial
                        best_obj = obj_t

                    a[i] -= amt
                    a[j] += amt

                        a = a / a.sum()
                        obj_new = objective(a)

                        else:

                            a = np.maximum(a, 0.0)

                                a = a / a.sum()

            elapsed = time.monotonic() - t0
            print(f"  CD ({elapsed:.1f}s): obj={best_obj:.6f}, stall={stall}")
    
    return best_a, best_obj

def spectral_shake(a, rng, strength=0.05):
    """Add structured frequency-domain perturbation."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    # Add random phase perturbation to reduce peaks
    A_pert = A * (1.0 + strength * (rng.standard_normal(A.shape) + 1j * rng.standard_normal(A.shape)))
    a_new = np.fft.irfft(A_pert, fft_size)[:n]
    return np.maximum(a_new, 0.0)

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

**Another idea: Use the gradient of the SUM of top-k peaks (smoothed max)**

Instead of `max_k ac[k]`, use `sum_k ac[k]^p / n` for large p (approximates max) or use the "soft max" `log(sum exp(ac[k]/tau)) * tau`. The gradient of the soft max is smooth and gives a better descent direction.

Let me implement this:

```python
def soft_max_gradient(a, tau=0.01, n_top=None):
    """
    Compute gradient of soft-max of autocorrelation.
    soft_max = tau * log(sum(exp(ac[k]/tau)))
    gradient wrt a[i] = sum_k softmax_weight[k] * d(ac[k])/d(a[i])
    """
    n = len(a)
    ac = fast_autocorr(a)
    if n_top is not None:
        # Only use top n_top lags
        idx = np.argsort(ac)[::-1][:n_top]
        ac_top = ac[idx]
    else:
        idx = np.arange(len(ac))
        ac_top = ac
    
    # Softmax weights
    ac_shifted = ac_top - ac_top.max()
    w = np.exp(ac_shifted / tau)
    w = w / w.sum()
    
    # Gradient
    g = np.zeros(n)
    for wi, k in zip(w, idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx_local = np.arange(i_start, i_end)
        j_idx = ki - idx_local
        g[idx_local] += 2.0 * wi * a[j_idx]
    
    return g, ac.max()
```

This is actually a much better approach! The soft-max gradient gives a smooth descent direction that accounts for ALL near-peaks simultaneously. When tau is small, it approximates the exact max subgradient. When tau is larger, it's smoother and converges faster (at the cost of converging to a slightly suboptimal point).

We can use an **annealing schedule** for tau: start large (smooth, fast convergence) and decrease (sharper, fine-tuning).

Let me redesign the algorithm around this:

**New algorithm: Annealed soft-max mirror descent**

1. Start with large tau (e.g., 0.1 * max_ac)
2. Do mirror descent steps with soft-max gradient
3. Periodically decrease tau
4. When tau is very small, switch to exact subgradient
5. Finish with coordinate descent

This should give better convergence properties than the hard subgradient approach.

Let me also think about the step size. For mirror descent on the simplex with a smooth function, the optimal step size is related to the smoothness constant. The soft-max is L-smooth where L depends on tau. So the step size should scale with tau.

Actually, for mirror descent with a non-smooth function (hard subgradient), the step size is O(1/sqrt(t)). For a smooth function (soft-max), we can use O(1/t) step sizes which converge faster.

Let me write the final implementation:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Annealed soft-max mirror descent (smooth gradient that anneals to exact)
2. Pair-swap coordinate descent with peak-aware mass transfer
3. Multi-length exploration with spectral initialization
4. Adaptive restarts with structured perturbation
"""

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def obj_value(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * fast_autocorr(a).max() / (s * s)

def soft_max_grad(a, tau, n_top=None):
    """
    Gradient of soft-max(tau) of autocorrelation peaks.
    Returns (gradient, max_ac).
    """
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n), M
    
    if n_top is not None:
        top_idx = np.argsort(ac)[::-1][:min(n_top, len(ac))]
    else:
        top_idx = np.arange(len(ac))
    
    # Softmax weights over the selected lags
    vals = ac[top_idx]
    vals_shift = vals - vals.max()
    w = np.exp(vals_shift / tau)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(w)) / len(w)
    else:
        w = w / w_sum
    
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        g[idx] += 2.0 * wi * a[j_idx]
    
    return g, M

def exact_subgrad(a, ac=None, thresh=0.99):
    """Exact subgradient from active set."""
    n = len(a)
    if ac is None:
        ac = fast_autocorr(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n), M
    active = np.where(ac >= thresh * M)[0]
    g = np.zeros(n)
    for k in active:
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = ki - idx
        g[idx] += 2.0 * a[j_idx]
    return g, M

def annealed_mirror_descent(a0, time_limit, verbose=True, rng_seed=42):
    """
    Mirror descent with annealed soft-max gradient.
    Phase 1: large tau (smooth, fast)
    Phase 2: decreasing tau (finer)
    Phase 3: exact subgradient (polish)
    """
    n = len(a0)
    a = normalize(a0.copy())
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_obj = 2.0 * n * ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 3000
    
    # Annealing schedule
    # Phase 1: tau = 0.1*M for first 30% of time
    # Phase 2: tau decreasing from 0.05*M to 0.01*M for next 40%
    # Phase 3: exact subgradient for last 30%
    
    M0 = ac.max()
    total_iters = 200000  # max, we'll stop by time
    phase1_end = int(0.30 * total_iters)
    phase2_end = int(0.70 * total_iters)
    
    c = 5.0  # step size constant
    
    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Determine gradient type based on phase
        if t <= phase1_end:
            # Smooth phase: large tau
            tau = 0.1 * M0
            n_top = 50
            g, M_curr = soft_max_grad(y, tau, n_top=n_top)
        elif t <= phase2_end:
            # Annealing phase
            frac = (t - phase1_end) / (phase2_end - phase1_end)
            tau = 0.05 * M0 * (1 - frac) + 0.005 * M0 * frac
            n_top = 30
            g, M_curr = soft_max_grad(y, tau, n_top=n_top)
        else:
            # Exact phase
            ac_y = fast_autocorr(y)
            g, M_curr = exact_subgrad(y, ac_y, thresh=0.995)
        
        gn = np.linalg.norm(g)
        if gn < 1e-30:
            break
        g = g / gn
        
        # Step size: 1/sqrt(t) for non-smooth, could use 1/t for smooth
        if t <= phase2_end:
            eta = c / (t + 1.0) * np.sqrt(n) * 2.0  # faster for smooth
        else:
            eta = c / (np.sqrt(t) + 1.0) * np.sqrt(n)
        
        # Mirror update
        log_a = np.log(np.maximum(y, 1e-15)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        # Evaluate
        M_new = fast_autocorr(x).max()
        obj_new = 2.0 * n * M_new
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart
        if stall > max_stall:
            stall = 0
            # Structured perturbation
            noise = rng.normal(0, 0.005, n)
            x = normalize(best_a * (1 + noise))
            x_prev = x.copy()
            c = min(c * 1.2, 20.0)
            M0 = fast_autocorr(best_a).max()
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            phase = "smooth" if t <= phase1_end else ("anneal" if t <= phase2_end else "exact")
            print(f"    AMD t={t} ({elapsed:.1f}s, {phase}): obj={best_obj:.6f}, stall={stall}")
    
    return best_a, best_obj

def pair_swap_cd(a0, time_limit, verbose=True, rng_seed=42):
    """
    Aggressive coordinate descent: move mass from peak-contributing
    positions to low-contributing positions.
    """
    n = len(a0)
    a = normalize(a0.copy())
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_obj = 2.0 * n * ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 80
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Peak lags
        peak_lags = np.argsort(ac)[::-1][:min(20, len(ac))]
        
        # Compute position hotness (contribution to peaks)
        hotness = np.zeros(n)
        for k in peak_lags:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            hotness[idx] += a[j_idx]
        
        # Sort positions by hotness
        hot_sorted = np.argsort(-hotness)
        cold_sorted = np.argsort(hotness)
        
        improved = False
        n_hot = min(len(hot_sorted), 40)
        n_cold = min(len(cold_sorted), 40)
        
        for src in hot_sorted[:n_hot]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue
            
            for dst in cold_sorted[:n_cold]:
                if dst == src:
                    continue
                if a[dst] >= a[src] * 1.5:  # don't move to much larger positions
                    continue
                
                for frac in [0.003, 0.008, 0.015, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < M - 1e-15:
                        a = a_trial
                        M = M_trial
                        best_obj = 2.0 * n * M
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Escape: try multiple random transfers
            n_attempts = max(10, n // 100)
            for _ in range(n_attempts):
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.005, 0.04)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t < 1e-15:
                    break
                a = a / s_t
                M_new = fast_autocorr(a).max()
                if M_new < M - 1e-15:
                    M = M_new
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    stall = 0
                    improved = True
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = np.maximum(a, 0.0)
                    if a.sum() > 1e-15:
                        a = a / a.sum()
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.1f}s): obj={best_obj:.6f}, stall={stall}")
    
    return best_a, best_obj

def interpolate_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss
