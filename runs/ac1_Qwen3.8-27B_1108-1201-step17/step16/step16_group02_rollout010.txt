def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def compute_smooth_gradient(a, ac, T, top_k=200):
    """Compute gradient of log-sum-exp(ac/T) w.r.t. a, using top-k lags."""
    n = len(a)
    # Compute softmax weights
    ac_clipped = np.maximum(ac, 1e-30)
    log_w = ac_clipped / T
    log_w -= log_w.max()
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        # Fallback: uniform
        w = np.ones(len(ac)) / len(ac)
    else:
        w /= w_sum
    
    # Use top-k lags for efficiency
    if top_k < len(ac):
        top_idx = np.argsort(ac)[::-1][:top_k]
        w_top = w[top_idx]
        w_top_sum = w_top.sum()
        if w_top_sum < 1e-30:
            top_idx = np.arange(len(ac))
            w_top = w
        else:
            w_top /= w_top_sum
    else:
        top_idx = np.arange(len(ac))
        w_top = w
    
    # Compute gradient
    grad = np.zeros(n)
    for wi, k in zip(w_top, top_idx):
        if wi < 1e-12:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    
    return grad

def exact_subgradient_polish(a_init, time_limit, verbose=True, rng_seed=123):
    """Polish using exact subgradient (top lags) with adaptive step size."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a.copy(), fast_autocorr(a).max()
    a = a / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    eta = 1.0 / n  # Initial step size
    stall = 0
    max_stall = 500
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top lags
        top_k = min(50, len(ac))
        top_idx = np.argsort(ac)[::-1][:top_k]
        
        # Compute subgradient from top lags (equal weight)
        grad = np.zeros(n)
        for k in top_idx:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                j_idx = k_int - idx
                grad[idx] += 2.0 * a[j_idx]
        grad /= len(top_idx)
        
        # Projected gradient step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.1, 3.0 / n)  # Increase step on success
        else:
            a = a_new  # Keep the step anyway (might help explore)
            stall += 1
            eta *= 0.95  # Decrease step on failure
        
        # Occasionally add noise
        if stall > 100:
            noise = rng.normal(0, eta * 0.1, n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            stall = 0
        
        if stall > max_stall:
            break
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    ExactSG t={t} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.6f}")
    
    return best_a, best_m

def diverse_init(name, n, parent=None):
    """Create diverse initializations."""
    x = np.linspace(0.0, 1.0, n)
    if name == "parent" and parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        old_x = np.linspace(0.0, 1.0, len(p))
        a = np.interp(x, old_x, p / p.sum())
    elif name == "uniform":
        a = np.ones(n)
    elif name == "triangle":
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif name == "gaussian":
        a = np.exp(-50.0 * (x - 0.5) ** 2)
    elif name == "raised_cosine":
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x - 0.5)))
    elif name == "sinc":
        t = (x - 0.5) * n * 0.1
        a = np.sinc(t) ** 2
        a = np.maximum(a, 0.0)
    elif name == "random":
        rng = np.random.default_rng(hash(name) % 2**32)
        a = rng.random(n) ** 0.5
    elif name == "bimodal":
        a = np.exp(-200.0 * (x - 0.3) ** 2) + np.exp(-200.0 * (x - 0.7) ** 2)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a

def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

import numpy as np
import time
import math

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def compute_gradient_a(a, ac, T, top_k=300):
    """Compute gradient of log-sum-exp(ac/T) w.r.t. a using top-k active lags."""
    n = len(a)
    ac_safe = np.maximum(ac, 1e-30)
    log_w = ac_safe / T
    log_w -= log_w.max()
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        top_idx = np.argsort(ac)[::-1][:min(top_k, len(ac))]
        w = np.zeros(len(ac))
        w[top_idx] = 1.0 / len(top_idx)
    else:
        w /= w_sum

    if top_k < len(ac):
        top_idx = np.argsort(ac)[::-1][:top_k]
        w_top = w[top_idx].copy()
        w_top_sum = w_top.sum()
        if w_top_sum < 1e-30:
            w_top = np.ones(len(top_idx)) / len(top_idx)
        else:
            w_top /= w_top_sum
    else:
        top_idx = np.arange(len(ac))
        w_top = w

    grad = np.zeros(n)
    for wi, k in zip(w_top, top_idx):
        if wi < 1e-14:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]

    return grad

def adam_smooth_optimize(a_init, time_limit, T_start=0.04, T_end=0.001,
                         lr_start=0.01, lr_end=0.0005,
                         beta1=0.9, beta2=0.999, eps=1e-8,
                         top_k=300, verbose=True, verbose_every=5000,
                         rng_seed=42, restart_interval=3000):
    """Adam optimizer on softmax(z) parameterization for smooth max autocorrelation."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()
    z = np.log(a).copy()

    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    stall = 0
    max_stall = restart_interval
    rng = np.random.default_rng(rng_seed)

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / 30000.0)
        T = T_start * 0.5 * (1.0 + math.cos(math.pi * frac)) + T_end * 0.5 * (1.0 - math.cos(math.pi * frac))
        T = max(T, T_end)

        lr = lr_start * (1.0 - frac) + lr_end * frac

        ac = fast_autocorr(a)
        g = compute_gradient_a(a, ac, T, top_k=top_k)

        dot_ag = np.dot(a, g)
        grad_z = a * (g - dot_ag)

        t_count += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t_count)
        v_hat = v / (1.0 - beta2 ** t_count)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_fn(z)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new

        if stall > max_stall:
            stall = 0
            max_stall = max(800, max_stall // 2)
            noise_scale = 0.005 + 0.02 * (1.0 - frac)
            pert = rng.normal(0, noise_scale, n)
            a_r = best_a * (1.0 + pert)
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            z = np.log(a_r)
            a = a_r
            m = np.zeros(n)
            v = np.zeros(n)
            t_count = 0

        if verbose and t % verbose_every == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Adam t={t} ({elapsed:.0f}s): J={J:.6f}, T={T:.5f}, lr={lr:.6f}, stall={stall}")

    return best_a, best_m

def nesterov_smooth_optimize(a_init, time_limit, T_start=0.03, T_end=0.001,
                             c0=3.0, top_k=200, verbose=False, verbose_every=5000,
                             rng_seed=77):
    """Nesterov accelerated mirror descent on the simplex with smooth max objective."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    max_stall = 3000

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / 25000.0)
        T = max(T_start * (1 - frac) + T_end * frac, 1e-5)

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        ac = fast_autocorr(y)
        g = compute_gradient_a(y, ac, T, top_k=top_k)

        eta = c0 / (np.sqrt(t) + 1.0)
        log_new = np.log(y) - eta * g
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            max_stall = max(1000, max_stall // 2)
            pert = rng.normal(0, 0.008, n)
            x_r = best_a * (1.0 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            c0 = max(1.0, c0 * 0.7)

        if verbose and t % verbose_every == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.0f}s): J={J:.6f}, T={T:.5f}")

    return best_a, best_m

def subgradient_polish(a_init, time_limit, verbose=False, rng_seed=123):
    """Polish with exact subgradient on active lags and adaptive step size."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a.copy(), fast_autocorr(a).max()
    a = a / s

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    eta = 2.0 / n
    stall = 0
    max_stall = 800

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        threshold = M_curr * 0.995
        active = np.where(ac >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(ac)])
        if len(active) > 80:
            active = np.argsort(ac)[::-1][:80]

        grad = np.zeros(n)
        for k in active:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                j_idx = k_int - idx
                grad[idx] += 2.0 * a[j_idx]
        grad /= len(active)

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.05, 5.0 / n)
        else:
            a = a_new
            stall += 1
            eta *= 0.97

        if stall > 200:
            noise = rng.normal(0, eta * 0.05, n)
            a = a + noise
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new > 1e-15:
                a = a / s_new
            stall = 0

        if stall > max_stall:
            break

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SG t={t} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.6f}")

    return best_a, best_m

def diverse_init(name, n, parent=None):
    """Create diverse initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    if name == "parent" and parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        old_x = np.linspace(0.0, 1.0, len(p))
        a = np.interp(x, old_x, p / p.sum())
    elif name == "uniform":
        a = np.ones(n)
    elif name == "triangle":
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif name == "gaussian":
        a = np.exp(-40.0 * (x - 0.5) ** 2)
    elif name == "raised_cosine":
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x - 0.5)))
    elif name == "sinc_sq":
        t = (x - 0.5) * n * 0.15
        a = np.sinc(t) ** 2
        a = np.maximum(a, 0.0)
    elif name == "bimodal":
        a = np.exp(-150.0 * (x - 0.3) ** 2) + np.exp(-150.0 * (x - 0.7) ** 2)
    elif name == "random":
        rng = np.random.default_rng(12345)
        a = rng.random(n) ** 0.5
    elif name == "random_smooth":
        rng = np.random.default_rng(54321)
        a = rng.random(n)
        from scipy.ndimage import uniform_filter
        a = uniform_filter(a, size=max(5, n // 100))
        a = np.maximum(a, 0.0)
    elif name == "comb":
        period = max(3, n // 200)
        a = np.ones(n)
        a[::period] = 2.0
        a[1::period] = 0.5
        from scipy.ndimage import uniform_filter
        a = uniform_filter(a, size=period)
        a = np.maximum(a, 0.0)
    else:
        a = np.ones(n)

    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a / s
    return a

def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

            a_in = best_seq.copy()
            a_out, _ = nesterov_smooth_optimize(a_in, t5, T_start=0.015, T_end=0.0008,
                                                 c0=2.5, top_k=150, verbose=True, rng_seed=999)
            try_update(a_out, f"NMDFin_{n_final}")

    # === Phase 6: Final Adam polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_final = len(best_seq)
        print(f"\n=== Phase 6: Final Adam n={n_final} (budget={remaining:.0f}s) ===")
        t6 = min(remaining * 0.6, 150.0)
        if t6 > 20:
            a_in = best_seq.copy()
            a_out, _ = adam_smooth_optimize(a_in, t6, T_start=0.01, T_end=0.0005,
                                             lr_start=0.003, lr_end=0.0001,
                                             top_k=150, verbose=False, rng_seed=777)
            try_update(a_out, f"AdamFin_{n_final}")

    # === Phase 7: Final subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_final = len(best_seq)
        print(f"\n=== Phase 7: Final SG polish n={n_final} (budget={remaining:.0f}s) ===")
        t7 = min(remaining * 0.8, 100.0)
        if t7 > 15:
            a_pol, _ = subgradient_polish(best_seq, t7, verbose=True, rng_seed=444)
            try_update(a_pol, "SGFinal")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
