import numpy as np
import time

def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def proj_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - (cssv - 1) / rho > 0
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val-1] - 1) / rho_val
    return np.maximum(v - theta, 0)

def softmax_subgradient(a, ac, T):
    """Compute softmax-weighted subgradient of max_k ac[k]."""
    n = len(a)

    w = np.exp(ac_shifted)

    if w_sum < 1e-300:

        w = np.zeros_like(ac)
