"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 on nonnegative sequences.

Strategy:
- Work under sum(a)=1, so the score is 2*n*max(autocorr(a)).
- Use FFT-based autocorrelation for fast objective and gradient evaluation.
- Run Nesterov accelerated mirror descent with a smoothed/active-set max
  autocorrelation subgradient, restarting when stalled.
- Polish with an LP Frank-Wolfe heuristic that searches a feasible
  direction under the current lag constraints.
- Finish with a lightweight local pair-move polish.
- Try several diverse smooth initializations and, if time remains,
  resampled longer sequences.
"""

import numpy as np
import time

try:
    import cvxpy as cp
    HAVE_CVXPY = True
except Exception:
    HAVE_CVXPY = False

_global_solver = None


def get_lp_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    if not HAVE_CVXPY:
        return None
    candidates = ["CLARABEL", "PDLP", "ECOS", "SCS", "MOSEK", "SCIP", "GLOP", "GLPK", "CBC", "CVXOPT"]
    for name in candidates:
        try:
            x = cp.Variable(6, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=name, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                _global_solver = name
                print(f"  LP solver selected: {name}")
                return name
        except Exception:
            pass
    _global_solver = None
    print("  LP solver: none available")
    return None


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.asarray([], dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    ac = np.fft.irfft(A * A, fft_size)[:size]
    return np.maximum(ac, 0.0)


def internal_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    ac = fast_autocorr(a)
    if ac.size == 0:
        return float("inf")
    return float(2.0 * len(a) * float(ac.max()) / (s * s))


def evaluate_safe(a):
    try:
        out = evaluate_sequence([float(x) for x in a])
        return float(out)
    except Exception:
        return float("inf")


def norm_sum1(a):
    a = np.array(a, dtype=np.float64, copy=True)
    if a.size == 0:
        return a
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones_like(a) / float(len(a))
    return a / s


def make_initialization(n, kind, parent=None):
    if n <= 0:
        return np.asarray([], dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    if kind == "parent" and parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim != 1 or p.size == 0 or not np.isfinite(p).all() or float(p.sum()) < 1e-12:
            kind = "triangle"
        else:
            if p.size == n:
                return norm_sum1(p)
            return resample_sequence(p, n)
    if kind == "uniform":
        v = np.ones(n, dtype=np.float64)
    elif kind == "triangle":
        v = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == "raised":
        v = 0.5 * (1.0 + np.cos(np.pi * (2.0 * x - 1.0)))
    elif kind == "gaussian":
        v = np.exp(-((x - 0.5) ** 2) / (2.0 * 0.13 ** 2))
    elif kind == "exp":
        v = np.exp(-3.0 * np.abs(x - 0.5))
    elif kind == "parabola":
        v = np.maximum(1.0 - 4.0 * (x - 0.5) ** 2, 0.0)
    else:
        v = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    return norm_sum1(v)


def resample_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    old_n = len(a)
    if old_n == 0 or new_n == 0:
        return np.zeros(new_n, dtype=np.float64)
    if old_n == new_n:
        return a.copy()
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    a = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f = np.interp(new_x, old_x, a)
    f = np.maximum(f, 0.0)
    sf = float(f.sum())
    if sf < 1e-15:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    return f / sf


def active_lags(ac, frac=0.97, cap=60):
    ac = np.asarray(ac, dtype=np.float64)
    if ac.size == 0:
        return np.asarray([], dtype=int)
    m = float(ac.max())
    if m <= 0.0:
        return np.asarray([], dtype=int)
    thresh = frac * m
    inds = np.where(ac >= thresh)[0]
    if inds.size == 0:
        inds = np.array([int(np.argmax(ac))], dtype=int)
    if inds.size > cap:
        order = np.argsort(ac)[::-1][:cap]
        return order.astype(int, copy=False)
    order = np.argsort(ac[inds])[::-1]
    return inds[order].astype(int, copy=False)


def gradient_smoothed(a, ac, beta, top_k=90):
    n = len(a)
    ac = np.asarray(ac, dtype=np.float64).copy()
    if ac.size == 0:
        return np.zeros(n, dtype=np.float64)
    if top_k is not None and top_k < ac.size and top_k > 0:
        idx = np.argsort(ac)[::-1][:top_k]
        ac2 = np.zeros_like(ac)
        ac2[idx] = ac[idx]
        ac = ac2
    bx = beta * ac
    bmax = float(bx.max())
    w = np.exp(bx - bmax)
    sw = float(w.sum())
    if sw < 1e-300:
        k = int(np.argmax(ac))
        g = np.zeros(n, dtype=np.float64)
        istart = max(0, k - n + 1)
        iend = min(n, k + 1)
        if istart < iend:
            idx = np.arange(istart, iend)
            g[idx] = 2.0 * a[k - idx]
        return g
    w = w / sw
    nz = np.where(w > 1e-13)[0]
    g = np.zeros(n, dtype=np.float64)
    for k in nz:
        wk = float(w[k])
        ki = int(k)
        istart = max(0, ki - n + 1)
        iend = min(n, ki + 1)
        if istart >= iend:
            continue
        idx = np.arange(istart, iend)
        g[idx] += 2.0 * wk * a[ki - idx]
    return g


def gradient_active(a, ac, active_lag_array):
    n = len(a)
    g = np.zeros(n, dtype=np.float64)
    m = len(active_lag_array)
    if m == 0:
        return g
    scale = 1.0 / float(m)
    for k in active_lag_array:
        ki = int(k)
        istart = max(0, ki - n + 1)
        iend = min(n, ki + 1)
        if istart >= iend:
            continue
        idx = np.arange(istart, iend)
        g[idx] += 2.0 * scale * a[ki - idx]
    return g


def nesterov_mirror_descent(a0, time_limit, beta_start=20.0, beta_end=900.0,
                            c0=2.2, verbose=False, seed=123, top_k=90):
    a0 = np.asarray(a0, dtype=np.float64)
    if time_limit <= 0:
        a = norm_sum1(a0)
        return a.copy(), float(fast_autocorr(a).max())

    n = len(a0)
    a = norm_sum1(a0)
    ac0 = fast_autocorr(a)
    best_a = a.copy()
    best_m = float(ac0.max())

    x = a.copy()
    x_prev = a.copy()
    gamma = float(c0)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 2500
    running_avg = best_a.copy()
    avg_count = 1
    last_print = 0.0

    max_iter = 10_000_000
    for t in range(1, max_iter + 1):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, float(t) / 4500.0)
        if beta_end <= beta_start:
            beta = beta_start
        else:
            beta = beta_start * (beta_end / beta_start) ** frac

        theta = float(t - 1) / float(t + 1.5)
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = float(y.sum())
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)

        if beta <= 220.0:
            use_topk = max(60, min(170, int(top_k) + 30))
            g = gradient_smoothed(y, ac, beta, use_topk)
        else:
            act = active_lags(ac, frac=0.985, cap=45)
            if act.size == 0:
                act = np.array([int(np.argmax(ac))], dtype=int)
            g = gradient_active(y, ac, act)

        eta = gamma / (np.sqrt(float(t)) + 1.0)
        log_a = np.log(y) - eta * g
        lm = float(log_a.max())
        a_new = np.exp(log_a - lm)
        a_new = np.maximum(a_new, 1e-30)
        sn = float(a_new.sum())
        if sn < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / sn

        x_prev = x
        x = a_new

        mx = float(fast_autocorr(x).max())
        if mx < best_m - 1e-15:
            best_m = mx
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        running_avg += x
        avg_count += 1
        if t % 200 == 0:
            avg = running_avg / float(avg_count)
            avg = norm_sum1(avg)
            ma = float(fast_autocorr(avg).max())
            if ma < best_m:
                best_m = ma
                best_a = avg.copy()

        if stall > max_stall:
            stall = 0
            max_stall = max(600, max_stall // 2)
            pert = rng.normal(0.0, 0.02, size=n)
            xr = best_a * (1.0 + pert)
            xr = np.maximum(xr, 1e-12)
            sxr = float(xr.sum())
            if sxr < 1e-15:
                xr = best_a.copy()
            else:
                xr = xr / sxr
            x = xr
            x_prev = xr.copy()
            gamma = max(0.4, gamma * 0.65)
            running_avg = best_a.copy()
            avg_count = 1

        if verbose:
            now = time.monotonic()
            if now - last_print >= 3.0:
                last_print = now
                print(f"    NMD t={t} beta={beta:.1f} score={2.0 * n * best_m:.8f} stall={stall} time={now - t0:.1f}s")

    avg = running_avg / float(avg_count)
    avg = norm_sum1(avg)
    ma = float(fast_autocorr(avg).max())
    if ma < best_m:
        best_m = ma
        best_a = avg.copy()

    return best_a, best_m


def lp_frank_wolfe_polish(a0, time_limit, solver=None, max_constraints=80, verbose=False):
    a0 = np.asarray(a0, dtype=np.float64)
    a = norm_sum1(a0)

            break

        for r, k in enumerate(act):
            ki = int(k)
            istart = max(0, ki - n + 1)
            iend = min(n, ki + 1)
            if istart < iend:

        b_var = cp.Variable(n, nonneg=True)
        cons = [
            T @ b_var <= M + 1e-12,
            b_var <= 5.0,
            cp.sum(b_var) <= 100.0,
        ]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), cons)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            fails += 1
            if fails > 25:

            continue

        if prob.status not in ("optimal", "optimal_inaccurate"):

            if fails > 25:
                break
            continue

        fails = 0
        bval = np.asarray(b_var.value, dtype=np.float64)

        bs = float(bval.sum())
        if bs < 1e-12:
            continue
        g_dir = bval / bs

        alpha = 2.0 / float(t + 2)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = float(y_new.sum())

            y_new = y.copy()

        y = y_new
        x = x_new

        my = float(fast_autocorr(y).max())

            best_a = y.copy()
        mx = float(fast_autocorr(x).max())
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 20 == 0:
            print(f"    LPFW iter={t} score={2.0 * n * best_m:.8f} time={time.monotonic() - t0:.1f}s")

def local_pair_polish(a0, time_limit, seed=99, verbose=False):
    a0 = np.asarray(a0, dtype=np.float64)
    if time_limit <= 0:
        a = norm_sum1(a0)
        return a.copy(), float(fast_autocorr(a).max())

    n = len(a0)
    a = norm_sum1(a0)
    ac0 = fast_autocorr(a)
    best_a = a.copy()
    best_m = float(ac0.max())

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 35

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = float(ac.max())
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()

        act = active_lags(ac, frac=0.995, cap=30)
        if act.size == 0:
            act = np.array([int(np.argmax(ac))], dtype=int)
        g = gradient_active(a, ac, act)

        pos = np.where(a > 1e-12)[0]
        if pos.size == 0:
            break

        high_pos = pos[np.argsort(g[pos])[::-1][:25]]
        low_all = np.argsort(g)[:25]

        improved = False
        for _ in range(70):
            i = int(high_pos[rng.integers(len(high_pos))])
            j = int(low_all[rng.integers(len(low_all))])
            if i == j or a[i] <= 1e-13:
                continue

            best_trial = None
            best_trial_m = best_m
            for frac in (0.02, 0.05, 0.10, 0.18, 0.35, 0.60):
                amt = a[i] * frac
                trial = a.copy()
                trial[i] -= amt
                trial[j] += amt
                if trial[i] < -1e-15:
                    continue
                trial = np.maximum(trial, 0.0)
                st = float(trial.sum())
                if st < 1e-15:
                    continue
                trial = trial / st
                mt = float(fast_autocorr(trial).max())
                if mt < best_trial_m - 1e-16:
                    best_trial_m = mt
                    best_trial = trial

            if best_trial is not None:
                a = best_trial
                best_m = best_trial_m
                best_a = a.copy()
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            pert = rng.normal(0.0, 0.006, size=n)
            trial = best_a * (1.0 + pert)
            trial = np.maximum(trial, 1e-12)
            st = float(trial.sum())
            trial = trial / st
            mt = float(fast_autocorr(trial).max())
            if mt <= best_m * 1.004 + 1e-13:
                a = trial
            else:
                a = best_a.copy()

        if verbose and stall % 10 == 0:
            print(f"    local polish score={2.0 * n * best_m:.8f} stall={stall} time={time.monotonic() - t0:.1f}s")

    return best_a, best_m

        a_tmp, _ = nesterov_mirror_descent(
            norm_sum1(best_a),
            t3,
            beta_start=250.0,
            beta_end=1800.0,
            c0=1.1,
            verbose=True,
            seed=77,
            top_k=80
        )
        sc = evaluate_safe(a_tmp)
        if np.isfinite(sc) and sc < best_score:
            best_a = norm_sum1(a_tmp)
            best_score = sc
            print(f"  Focused polish improved: {best_score:.8f}")

    # Local pair polish.
    rem = budget - (time.monotonic() - t_start)
    t4 = min(0.10 * rem, 120.0)
    if t4 > 30 and len(best_a) <= 6000:
        print(f"\nLocal pair polish budget={t4:.1f}s")
        a_tmp, _ = local_pair_polish(norm_sum1(best_a), t4, seed=2024, verbose=False)
        sc = evaluate_safe(a_tmp)
        if np.isfinite(sc) and sc < best_score:
            best_a = norm_sum1(a_tmp)
            best_score = sc
            print(f"  Local polish improved: {best_score:.8f}")

    # Try a longer resampled sequence if there is substantial time left.
    rem = budget - (time.monotonic() - t_start)
    if rem > 320:
        base_n = len(best_a)
        n_alt = int(min(4500, max(base_n + 500, int(base_n * 1.25))))
        if rem < 800:
            n_alt = min(n_alt, 3200)
        print(f"\nAlt length n={n_alt}")
        a_alt = resample_sequence(best_a, n_alt)

        t6a = min(140.0, rem * 0.35)
        if t6a > 50:
            a_tmp, _ = nesterov_mirror_descent(
                a_alt,
                t6a,
                beta_start=18.0,
                beta_end=900.0,
                c0=2.3,
                verbose=True,
                seed=555,
                top_k=90
            )
            sc = evaluate_safe(a_tmp)
            if np.isfinite(sc) and sc < best_score:
                best_a = norm_sum1(a_tmp)
                best_score = sc
                print(f"  Alt length NMD improved: {best_score:.8f}")

        rem = budget - (time.monotonic() - t_start)
        if rem > 140 and solver is not None and len(best_a) <= 5000:
            t6b = min(rem * 0.40, 180.0)
            a_tmp, _ = lp_frank_wolfe_polish(
                norm_sum1(best_a),
                t6b,
                solver=solver,
                max_constraints=70,
                verbose=False
            )
            sc = evaluate_safe(a_tmp)
            if np.isfinite(sc) and sc < best_score:
                best_a = norm_sum1(a_tmp)
                best_score = sc
                print(f"  Alt/final LP improved: {best_score:.8f}")

        rem = budget - (time.monotonic() - t_start)
        if rem > 40 and len(best_a) <= 5000:
            t6c = min(rem * 0.55, 70.0)
            a_tmp, _ = local_pair_polish(norm_sum1(best_a), t6c, seed=314, verbose=False)
            sc = evaluate_safe(a_tmp)
            if np.isfinite(sc) and sc < best_score:
                best_a = norm_sum1(a_tmp)
                best_score = sc
                print(f"  Alt/final local improved: {best_score:.8f}")

    # Final short polish.
    rem = budget - (time.monotonic() - t_start)
    if rem > 90 and solver is not None and len(best_a) <= 5000:
        t7 = min(0.45 * rem, 120.0)
        a_tmp, _ = lp_frank_wolfe_polish(
            norm_sum1(best_a),
            t7,
            solver=solver,
            max_constraints=60,
            verbose=False
        )
        sc = evaluate_safe(a_tmp)
        if np.isfinite(sc) and sc < best_score:
            best_a = norm_sum1(a_tmp)
            best_score = sc
            print(f"  Final LP improved: {best_score:.8f}")

    rem = budget - (time.monotonic() - t_start)
    if rem > 25 and len(best_a) <= 5000:
        t8 = min(0.85 * rem, 50.0)
        a_tmp, _ = local_pair_polish(norm_sum1(best_a), t8, seed=999, verbose=False)
        sc = evaluate_safe(a_tmp)
        if np.isfinite(sc) and sc < best_score:
            best_a = norm_sum1(a_tmp)
            best_score = sc
            print(f"  Final local improved: {best_score:.8f}")

    # Final validation.
    final_score = evaluate_safe(best_a)
    if not np.isfinite(final_score):
        final_score = internal_score(best_a)
    if not np.isfinite(final_score):
        print("Fallback: constant sequence")
        return [1.0]

    print(f"\nReturned n={len(best_a)}, final score={final_score:.10f}, elapsed={time.monotonic() - t_start:.1f}s")
    return [float(v) for v in best_a]
