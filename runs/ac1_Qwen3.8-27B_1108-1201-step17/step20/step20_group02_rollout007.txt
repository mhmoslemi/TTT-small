def fast_convolve(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_A(y, active_set, n):
    """Build constraint matrix for LP: A[k,i] = y[k-i] for valid indices."""
    K = len(active_set)
    cols = np.zeros((K, n))
    for jj in range(K):
        k_int = int(active_set[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            cols[jj, idx] = y[k_int - idx]
    return cols


def lp_fw_optimize(a0, time_limit, solver, initial_K=600, max_K=4000, verbose=True):

    n = len(a0)

    best_a = y.copy()

    best_m = ac_y.max()

    max_no_improve = 600

    stagnation_restarts = 0

        iter_count += 1

        # = sum_k w[k] * y[n-1+j-k-n] = sum_k w[k]*y[j-1-k+1]... getting confused.
        
        # Let me just do it the clean way:
        # For the convolution c = a * a, c[k] = sum_i a[i]*a[k-i]
        # d(c[k])/d(a[j]) = a[k-j] (if 0<=k-j<n)
        # So d(sum_k w[k]*c[k])/d(a[j]) = sum_k w[k]*a[k-j] (valid range)
        # = sum_{k=j}^{j+n-1} w[k]*a[k-j] (with w[k]=0 for k>=2n-1)
        # = sum_{m=0}^{n-1} w[j+m]*a[m]
        
        # This is exactly what we compute as a "sliding dot product" of w and a
        # For all j simultaneously, this is cross-correlation:
        # grad = correlate(w_full, a) where w_full has length 2n-1
        # Using FFT:
        w_full = np.zeros(2 * n - 1)
        w_full[:n] = w  # Actually w has length 2n-1 from fast_convolve... no, w is the softmax of conv
        # Wait, w = softmax(beta * ac) where ac = fast_convolve(y) has length 2n-1
        # So w has length 2n-1
        # grad[j] = sum_{k=0}^{2n-2} w[k] * y[k-j] (if 0<=k-j<n)
        # = sum_{k=j}^{j+n-1} w[k] * y[k-j]
        # Let m = k-j: sum_{m=0}^{n-1} w[j+m] * y[m]
        # This is the valid cross-correlation of w (length 2n-1) and y (length n) at lag j
        
        # FFT approach:
        # P = 1 << (3*n).bit_length()  (power of 2 >= 3n)
        # F_w = fft(w, P), F_y = fft(y, P)
        # grad_circ = ifft(F_w * conj(F_y))
        # grad[j] = grad_circ[j] (for 0<=j<n, accounting for the "head" of w)
        
        P = 1 << (2 * n).bit_length()
        F_w = np.fft.rfft(w, P)
        F_y = np.fft.rfft(y, P)
        grad_circ = np.fft.irfft(F_w * np.conj(F_y), P)
        # grad_circ[m] = sum_k w[k]*y[(k+m) % P] (circular cross-correlation)
        # For m in [0, n-1] and w concentrated in [0, 2n-2]:
        # grad_circ[m] = sum_{k=0}^{2n-2} w[k]*y[(k+m) mod P]
        # For m < n and k+m < 2n-1 < P: = sum_k w[k]*y[k+m]
        # But we want sum_k w[k]*y[k-j] = sum_k w[k+j]*y[k] (sub m'=k+j)
        # Hmm, the indices are tricky. Let me just define:
        # We want G[j] = sum_{m=0}^{n-1} w[j+m]*y[m] for j in [0, n-1]
        # = (w cross-correlated with y) at "lag" j (w shifted by j, dot with y)
        # In signal processing: x(n)*h(n) where * is cross-correlation
        # corr[j] = sum_n x[n]*h[n-j] (lag j means x delayed by j)
        # We want: sum_m w[j+m]*y[m] = sum_m w'[m]*y[j+m] where w' = w[j:]
        # This is correlation of w and y at lag (-j) or equivalently lag (n-1-j) of reversed...
        
        # I'll just use the direct formula with FFT:
        # Define v[k] = w[k] for k in [0, 2n-2], else 0
        # Define u[m] = y[m] for m in [0, n-1], else 0
        # G[j] = sum_m v[j+m] * u[m]
        # = sum_m v[m+m2] * u[m] where m2 = j
        # This is the valid part of conv(v, reverse(u))[j + n - 1]
        # = conv(v, u_rev)[j + n - 1] where u_rev = u[n-1:0:-1]
        
        u_rev = y[::-1]
        conv_vu = np.fft.irfft(np.fft.rfft(w, fft_size) * np.fft.rfft(u_rev, fft_size), fft_size)
        # conv_vu[k] = sum_p w[p] * u_rev[k-p] = sum_p w[p] * y[n-1-(k-p)]
        # For k = j + n - 1: sum_p w[p] * y[n-1-j-n+1+p] = sum_p w[p]*y[p-j]
        # = sum_{p=j}^{n-1+j} w[p]*y[p-j] (valid)
        # = sum_{m=0}^{n-1} w[j+m]*y[m] = G[j] ✓
        
        grad = conv_vu[n - 1 + np.arange(n)]
        
        # Mirror descent update
        eta = 0.08 / (np.sqrt(t) + 1.0)
        log_w = np.log(np.maximum(y, 1e-30))
        new_log = log_w - eta * grad
        new_log -= new_log.max()
        y_new = np.exp(new_log)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_new
        y_new = np.maximum(y_new, 0)
        s_new = y_new.sum()
        if s_new > 1e-15:
            y_new = y_new / s_new
        y = y_new

        M_new = fast_convolve(y).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = y.copy()
            stalls = 0
        else:
            stalls += 1
            if stalls % 2000 == 0 and stalls < max_stalls:
                # Perturb
                noise = rng.normal(0, 0.03 * max(1.0/(1+stalls/2000.0), 0.1), n) * best_a
                y_p = best_a + noise
                y_p = np.maximum(y_p, 1e-10)
                sp = y_p.sum()
                if sp > 1e-12:
                    y = y_p / sp

        if verbose and (t % 50000 == 0 or stalls % 10000 == 0 and stalls > 0):
            elapsed = time.monotonic() - t0
            print("    MirrorDescent t=" + str(t) + " (" + str(round(elapsed, 1)) +
                  "s): J=" + repr(2.0 * n * best_m) + ", stalls=" + str(stalls))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    MirrorDescent done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))
    return best_a, best_m
