import numpy as np

import cvxpy as cp

def fast_autocorr(a):
    """Compute full autocorrelation (convolution with self) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(z):
    """Project z onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0)

def get_subgrad(a, ac, K=50):
    """Compute weighted subgradient using top K peaks."""
    n = len(a)
    size = len(ac)
    top_k = np.argsort(ac)[::-1][:K]
    # Weight by peak height
    w = ac[top_k]
    w = w / w.sum()
    
    g = np.zeros(n)
    for wi, k in zip(w, top_k):
        ki = int(k)
        # grad_k(a)[j] = 2*a[k-j] for valid j
        # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        js = max(0, ki - n + 1)
        je = min(n - 1, ki)
        if js <= je:
            g[js:je+1] += 2.0 * wi * a[ki - np.arange(js, je + 1)]
    return g

    eta = eta0

    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    return _MODULE_SOLVER

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def simplex_project(z):
    n = len(z)
    u = np.sort(z)[::-1].copy()
    cssv = np.cumsum(u)
    ar = np.arange(1, n + 1)
    cond = u * ar > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def get_subgrad(a, ac, K=80):
    n = len(a)
    kmax = min(K, len(ac))
    top_k = np.argsort(ac)[::-1][:kmax]
    w = ac[top_k].astype(np.float64)
    w = w / w.sum()
    g = np.zeros(n, dtype=np.float64)
    for wi, k in zip(w, top_k):
        ki = int(k)
        js = max(0, ki - n + 1)
        je = min(n - 1, ki)
        if js <= je:
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * wi * a[ki - idx]
    return g

def nesterov_proj_grad(a0, time_limit, K=80, verbose=True):
    n = len(a0)
    a = normalize(a0)
    a_prev = a.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    eta = 0.005
    fail = 0
    stall = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        if fail > 8000:
            break

        beta = (t - 1.0) / (t + 2.0)
        y = a + beta * (a - a_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        ac = fast_autocorr(y)
        M = ac.max()

        if M < best_m - 1e-14:
            best_m = M
            best_a = y.copy()
            fail = 0
            stall = 0
        else:
            fail += 1
            stall += 1

        g = get_subgrad(y, ac, K=K)

        # Try multiple step sizes (backtracking)
        accepted = False
        for trial_eta in [eta, eta * 0.5, eta * 0.25, eta * 0.1]:
            z = y - trial_eta * g
            a_new = simplex_project(z)
            ac_new = fast_autocorr(a_new)
            if ac_new.max() < M + 1e-13:
                a = a_new
                eta = trial_eta
                accepted = True
                break

        if not accepted:
            a = y  # keep momentum point
            eta *= 0.5

        a_prev = a if accepted else a_prev

        # Periodic restart
        if stall > 3000 and stall % 3000 == 0:
            noise = 0.02 * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            a = x_r
            a_prev = x_r.copy()
            eta = 0.005
            stall = 0

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print("    NPG t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(2 * n * best_m) + " eta=" + repr(round(eta, 6)))

    return best_a, best_m

def lp_fw_best(a0, time_limit, solver, init_K=3000, max_K=8000, verbose=True):
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    ac = fast_autocorr(y)
    active = set(int(k) for k in np.argsort(ac)[::-1][:init_K])
    no_improve = 0

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 800:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Prune active set
        if len(active) > max_K:
            margins = np.array([M_curr - ac[k] for k in active])
            keep = np.argsort(margins)[:max_K]
            active = set(int(list(active)[i]) for i in keep)

        K = len(active)
        if K == 0:
            break

        clist = sorted(active)

        # Build constraint matrix
        coeffs = np.zeros((K, n), dtype=np.float64)
        for jj, k in enumerate(clist):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx = np.arange(js, je + 1)
                coeffs[jj, idx] = y[k - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        lp_ok = False
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                lp_ok = True
        except Exception:
            pass

        if not lp_ok:
            if len(clist) > 500:
                active = set(int(k) for k in clist[:1000])
                continue
            if no_improve > 400:
                break
            continue

        g_dir = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = g_dir.sum()
        if gs < 1e-15:
            continue
        g_dir = g_dir / gs

        # Check violations, add to active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                active.add(int(k))

        # Backtracking line search
        alpha = 1.0
        best_trial_m = fast_autocorr(y).max()
        for _ in range(25):
            y_trial = (1.0 - alpha) * y + alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            s = y_trial.sum()
            if s < 1e-15:
                break
            y_trial = y_trial / s
            trial_m = fast_autocorr(y_trial).max()
            if trial_m < best_trial_m:
                best_trial_m = trial_m
            else:
                alpha *= 0.5
                continue
            alpha *= 1.0  # accept
            break
        else:
            # Use the last alpha that worked
            alpha *= 1.0

        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s > 1e-15:
            y = y_new / s
        else:
            y = g_dir.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(2 * n * best_m) + " |S|=" + str(len(active)))

    return best_a, best_m

def coord_polish(a0, time_limit, seed=42, verbose=True):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 2000:
        ac = fast_autocorr(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            top_srcs = idx[np.argsort(contrib)[::-1][:3]]
            for src in top_srcs:
                src = int(src)
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Score destinations by impact on top peaks
                scores = np.zeros(n)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                        scores[k2i - idx2] += a[idx2]
                scores[src] = 1e10

                candidates = np.argsort(scores)[:25]
                found = False
                for dst in candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.003, 0.008, 0.015, 0.03]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-15:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            found = True
                            break
                    if found:
                        break

        if not improved:
            stall += 1
            if stall > 100:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 500 == 0:
            print("    Coord (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2 * n * best_m) + " stall=" + str(stall))

    return best_a, best_m

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def diverse_init(n, parent_seq):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = normalize(parent_seq) if parent_seq is not None else normalize(np.sin(np.pi * x) ** 2)
    best_val = obj_val(best_a)

    candidates = []
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        candidates.append(normalize((i_arr + 1.0) ** p))
        candidates.append(normalize((n - i_arr) ** p))
    for c in [0.25, 0.35, 0.5, 0.65, 0.75]:
        for sg in [0.08, 0.12, 0.18, 0.25]:
            candidates.append(normalize(np.exp(-((x - c) ** 2) / (2 * sg * sg))))
    for f in [1, 2, 3, 5]:
        candidates.append(normalize(np.maximum(1.0 + np.cos(2 * np.pi * f * x), 0.01)))
    for p in [0.5, 1.0, 1.5, 2.0]:
        candidates.append(normalize(np.maximum(x ** p * (1 - x) ** p, 1e-10)))
    candidates.append(normalize(np.sin(np.pi * x) ** 2))
    candidates.append(normalize(np.sin(np.pi * x) ** 4))

    if parent_seq is not None:
        bi = normalize(parent_seq)
        for w in [0.3, 0.5, 0.7]:
            candidates.append(normalize(w * bi + (1 - w) * np.sin(np.pi * x) ** 2))

    for c in candidates:
        v = obj_val(c)
        if v < best_val:
            best_val = v
            best_a = c.copy()

    return best_a, best_val
