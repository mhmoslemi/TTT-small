def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_gradient(a, b, beta):
    """Compute gradient of log-sum-exp smoothed max autocorrelation.
    grad_i = 2 * sum_k p_k * a[k-i] where p_k = exp(beta*b[k]) / Z
    Computed via FFT cross-correlation.
    """
    n = len(a)
    size = 2 * n - 1
    # Compute softmax weights (numerically stable)
    b_shifted = beta * (b - b.max())
    exp_vals = np.exp(b_shifted)
    Z = exp_vals.sum()
    p = exp_vals / Z  # length 2n-1
    
    # Cross-correlation: g[i] = sum_j p[i+j] * a[j] for i=0,...,n-1
    L = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, L)
    A = np.fft.rfft(a, L)
    corr = np.fft.irfft(P * np.conj(A), L)
    grad = 2.0 * corr[:n]
    return grad


def fista_optimize(a0, time_limit, beta_start=10.0, beta_end=2000.0,
                   beta_growth=1.003, verbose=True):
    """FISTA (Nesterov accelerated projected gradient) on the simplex."""
    n = len(a0)
    x = norm_sum1(a0).copy()
    y = x.copy()  # Nesterov momentum variable
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t = 0
    t_k = 1.0  # Nesterov parameter
    rng = np.random.default_rng(12345)
    stall = 0
    L = 1 << (3 * n - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(y)
        m_curr = b.max()
        
        if m_curr < best_m - 1e-14:
            best_m = m_curr
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Compute gradient at y
        grad = compute_gradient(y, b, beta)
        
        # Lipschitz estimate: use power method approximation
        # For our problem, L ≈ 2 * max(a) * (2n-1) roughly, but let's be adaptive
        # Use a simple bound: L = 4 * sum(a)^2 * n (conservative)
        L_est = 4.0 * n * (beta + 1) / max(beta, 1)
        L_est = max(L_est, 1.0)
        
        # Actually, let's use a more practical step size
        # The gradient magnitude gives us a scale
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        step = 1.0 / (L_est + 1e-10)
        step = min(step, 0.5 / g_norm)
        
        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y_new = simplex_project(y - step * grad)
        
        # Nesterov extrapolation
        alpha = (t_k - 1.0) / t_new
        z = y_new + alpha * (y_new - x)
        
        x = y_new
        y = z
        t_k = t_new
        
        # Adaptive beta
        if t % 100 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        # Restart on stall
        if stall > 2000:
            stall = 0
            t_k = 1.0
            noise_scale = 0.01 / (1.0 + 0.5 * (t // 2000))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-14)
            x_r = x_r / x_r.sum()
            x = x_r
            y = x_r.copy()
            beta = max(beta * 0.5, 10.0)

            margins = np.array([M_curr - b[k] for k in active_set])
            keep_idx = np.argsort(margins)[:2000]
            active_list = sorted(active_set)
            active_set = set(int(active_list[i]) for i in keep_idx)
        
        if len(active_set) == 0:
            break
        
        constraint_list = sorted(active_set)
        K = len(constraint_list)
        
        # Build constraint matrix: for each k, (a*g)[k] = sum_i a[k-i]*g[i] <= M
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                coeffs[jj, idx] = a[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1]
        constraints.append(coeffs @ g_var <= M_var)
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check for violated constraints (a * g_dir > M_curr)
        fft_size = 1 << (2 * n - 1).bit_length()
        Af = np.fft.rfft(a, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ag = np.fft.irfft(Af * Gf, fft_size)[:2 * n - 1]
        
        violations = conv_ag > M_curr + 1e-5
        if np.any(violations):
            viol_k = np.where(violations)[0]
            viol_amt = conv_ag[violations] - M_curr
            top_v = viol_k[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                if len(active_set) < 4000:
                    active_set.add(int(k))
        
        # Line search
        default_alpha = 2.0 / (t + 2.0)
        best_alpha = default_alpha
        best_trial_M = fast_autocorr(a).max()
        
        for alpha in [default_alpha, 0.5 * default_alpha, 1.0 / (t + 1), 0.3 * default_alpha]:
            alpha = float(np.clip(alpha, 0.01, 0.95))
            trial = (1.0 - alpha) * a + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        a_new = (1.0 - best_alpha) * a + best_alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        st = a_new.sum()
        if st > 1e-15:
            a_new = a_new / st
        a = a_new
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " |S|=" + str(len(active_set)) + " alpha=" + repr(round(best_alpha, 3)))
    
    return best_a, best_m


def parametric_init(n):
    """Generate good parametric initializations and return the best."""
    x = np.linspace(0.0, 1.0, n)
    best_a = None
    best_m = float('inf')
    
    # Asymmetric power laws
    for alpha in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]:
        for beta_p in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]:
            a_raw = (x + 0.005) ** alpha * (1.0 - x + 0.005) ** beta_p
            a = norm_sum1(a_raw)
            m = fast_autocorr(a).max()
            if m < best_m:
                best_m = m
                best_a = a.copy()
    
    # Cosine-cutoff
    for c1 in [0.2, 0.3, 0.4, 0.5]:
        for c2 in [0.5, 0.6, 0.7, 0.8]:
            a_raw = np.cos(np.pi * (x - c1) / (2 * (1 - c1))) ** 2 * np.maximum(x - (c1 - 0.1), 0)
            a_raw += np.cos(np.pi * (c2 - x) / (2 * c2)) ** 2 * np.maximum((c2 + 0.1) - x, 0)
            a = norm_sum1(a_raw)
            m = fast_autocorr(a).max()
            if m < best_m:
                best_m = m
                best_a = a.copy()
    
    # Random starts
    rng = np.random.default_rng(42)
    for _ in range(20):
        a = norm_sum1(rng.random(n) ** rng.uniform(0.5, 3.0))
        m = fast_autocorr(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()
    
    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search: corrected FISTA + improved peak shaving + LP Frank-Wolfe + multi-n."""
    t_global = time.monotonic()

    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s, solver: " + str(solver))
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None
    
    if best_seq is None:
        n_base = 7000
        a_init, _ = parametric_init(n_base)
        best_seq = a_init
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))
    
    n0 = len(best_seq)
    
    # === Phase 1: FISTA with corrected gradient ===
    remaining = time_budget - (time.monotonic() - t0) if False else time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: FISTA v2 (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.20, 350.0)
    if t1 > 60:
        a_out, _ = fista_v2(best_seq, t1, beta_start=5.0, beta_end=3000.0,

    remaining = time_budget - (time.monotonic() - t_global)
