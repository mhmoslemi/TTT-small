def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.01, old_n)

    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:

    return f_interp / ss

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np

import scipy.optimize

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_smoothed_grad_and_val(a, beta, top_k=None):
    n = len(a)
    c = fast_autocorr(a)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    val = c_max + (1.0/beta) * np.log(total)
    fft_size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A_rev = np.fft.rfft(a[::-1], fft_size)
    Conv = np.fft.irfft(P * A_rev, fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return val, grad, p, c

def lbfgs_continuation(a0, time_limit, beta_start=15.0, beta_end=5000.0,
                        beta_growth=1.003, top_k_ratio=0.02,
                        max_iter_per_beta=200, verbose=True, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    beta = beta_start
    top_k = max(10, int(n * top_k_ratio))
    
    a_safe = np.maximum(a, 1e-15)
    b = np.log(a_safe).copy()
    
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        # Define objective and gradient for current beta
        def obj_and_grad(b_val):
            a_val = np.exp(b_val - b_val.max())
            a_val = a_val / a_val.sum()
            val, grad_a, _, _ = compute_smoothed_grad_and_val(a_val, beta, top_k)
            dot = np.dot(grad_a, a_val)
            grad_b = a_val * (grad_a - dot)
            return 2 * n * val, 2 * n * grad_b
        
        stopped = [False]
        def callback(b_val):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return True
            return None
        
        try:
            result = scipy.optimize.minimize(
                obj_and_grad, b, method='L-BFGS-B', jac=True,
                options={'maxiter': max_iter_per_beta, 'ftol': 1e-14,
                         'gtol': 1e-12, 'maxcor': 50, 'callback': callback}
            )
            b = result.x
            total_iters += result.nit
        except Exception:
            pass
        
        a_new = np.exp(b - b.max())
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        
        M = fast_autocorr(a_new).max()
        if M < best_M:
            best_M = M
            best_a = a_new.copy()
        
        # Check for time
        if time.monotonic() - t0 > time_limit:
            break
        
        # Check if we should increase beta
        # Heuristic: if the smoothed value is close to the true max, increase beta
        c = fast_autocorr(a_new)
        c_max = c.max()
        # Smoothed value approximates c_max; if beta is large enough, they're close
        if beta < beta_end and (stopped[0] or result.nit >= max_iter_per_beta):
            beta = min(beta * beta_growth, beta_end)
            # Adjust top_k as beta increases
            top_k = max(5, int(n * top_k_ratio / (1 + beta/100)))
        
        if verbose and total_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    LBFGS: " + str(total_iters) + " iters, beta=" + str(round(beta,1)) +
                  ", J=" + repr(round(2*n*best_M, 6)) + ", t=" + str(round(elapsed,1)) + "s")
    
    if verbose:
        elapsed = time.monotonic() - t0
        print("    LBFGS done: " + str(total_iters) + " iters, " + str(round(elapsed,1)) +
              "s, J=" + repr(round(2*n*best_M, 6)))
    
    return best_a, best_M

def spectral_flatten(a0, alpha=0.85, n_iter=5000, time_limit=60.0, verbose=False):
    """IFTA-style iteration to flatten the power spectrum."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2*n-1).bit_length()
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        magnitude = np.abs(A)
        phase = np.angle(A)
        
        # Compress dynamic range
        mag_new = np.power(magnitude + 1e-30, alpha)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        M = fast_autocorr(a_new).max()
        if M < best_M - 1e-15:
            best_M = M
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_M
