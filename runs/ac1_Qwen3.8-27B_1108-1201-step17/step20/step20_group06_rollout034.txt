def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_grad_softmax(y, beta, n_peaks=None):
    """Compute gradient of smoothed max using softmax weighting over peaks."""
    n = len(y)
    b = fast_autocorr(y)
    size = 2 * n - 1

    if n_peaks is not None and n_peaks < size:
        idx = np.argsort(b)[::-1][:n_peaks]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
    else:
        b_use = b

    b_max = b.max()
    if b_max <= 0:
        return np.zeros(n), b

    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s

    a_rev = y[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, b

                # Spectral randomization
                fft_size = 1 << (2 * n - 2).bit_length()
                A = np.fft.rfft(best_a, fft_size)
                A[1:] *= np.exp(1j * 0.03 * rng.standard_normal(len(A) - 1))
                A[0] = A[0].real
                x_r = np.fft.irfft(A, fft_size)[:n]
                x_r = np.maximum(x_r, 0.0)

            x_r = np.maximum(x_r, 1e-12)
            s_r = x_r.sum()
            if s_r < 1e-15:
                x_r = best_a.copy()
            else:
                x_r = x_r / s_r
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 10.0)
            eta = eta_init

            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Peel restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Peel t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) +
                  " beta=" + str(int(beta)) + " nk=" + str(n_peaks) + " stall=" + str(stall))

    return best_a, best_max_b


def lp_fw_smart(a0, time_limit, solver, initial_K=400, max_K=3000,
                add_per_iter=30, verbose=True):
    """
    LP-Frank-Wolfe with smart active set management:
    - Start with top-K peaks
    - Add violations each iteration
    - Periodically prune to most binding constraints
    - Adaptive line search
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 400
    lp_calls = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: no improvement for " + str(no_improve) + " iters, stopping")
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Prune active set if too large
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
            lp_calls += 1
        except Exception:
            # Shrink active set on failure
            active_set = set(int(k) for k in np.argsort(ac)[::-1][:max(100, initial_K // 3)])
            no_improve += 10
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 5
            if no_improve > max_no_improve:
                break
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            g_val = np.ones(n) / n
        g_dir = g_val / g_val.sum()

        # Check violations and add to active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-6)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Adaptive line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        alphas = [best_alpha, best_alpha * 0.5, best_alpha * 2.0,
                  0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
        for alpha in alphas:
            alpha = max(0.01, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW it=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" +
                  repr(round(J, 6)) + " |S|=" + str(len(active_set)) + " LPs=" + str(lp_calls))

    return best_a, best_m


def spectral_shape_init(n, mode=0):
    """Generate diverse initial shapes."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        f = np.sin(np.pi * x) ** 2
    elif mode == 1:
        f = (1.0 + np.cos(2.0 * np.pi * x)) ** 2
    elif mode == 2:
        f = np.sin(np.pi * x) ** 4
    elif mode == 3:
        f = np.sin(2.0 * np.pi * x * 3.0 / 2.0) ** 2 * np.sin(np.pi * x) ** 2
    elif mode == 4:
        f = np.exp(-4.0 * (x - 0.5) ** 2)
    else:
        f = np.sin(np.pi * x) ** 2 * (1.0 + 0.3 * np.cos(2.0 * np.pi * x))
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return f / s


def spectral_refine(a0, time_limit, verbose=False, seed=1):
    """Spectral domain refinement: flatten peaks in autocorrelation via frequency shaping."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()

    alpha = 0.03
    for it in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Target: reduce spectral peaks that cause autocorrelation peaks
        target = np.median(power[1:]) if len(power) > 2 else power[0]
        if target < 1e-30:
            target = power.max() * 0.1

        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.5:
                red = 1.0 - alpha * (ratio - 1.5)
                A_new[k] *= max(0.2, np.sqrt(max(red, 0.1)))
            elif ratio < 0.3:
                inc = 1.0 + alpha * 0.2 * (0.3 - ratio)
                A_new[k] *= min(3.0, np.sqrt(inc))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 500 == 0:
            # Random phase perturbation
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A) - 1) * 0.05
            A[1:] *= np.exp(1j * phase_pert)
            A[0] = A[0].real
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial

        if it % 1000 == 0:
            alpha = max(0.005, alpha * 0.95)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def coordinate_descent_polish(a0, time_limit, verbose=False, seed=77):
    """
    Coordinate descent: for each coordinate, find the optimal value
    that minimizes the max autocorrelation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        # Process coordinates in random order
        order = rng.permutation(n)
        improved_this_pass = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Try several step sizes for coordinate i
            best_val = a[i]
            best_local_m = best_m

            if a[i] > 1e-12:
                # Try reducing
                for frac in [0.8, 0.6, 0.4, 0.2]:
                    a_trial = a.copy()
                    a_trial[i] *= frac
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_t = fast_autocorr(a_trial).max()
                    if m_t < best_local_m:
                        best_local_m = m_t
                        best_val = a_trial[i]
            # Try increasing (taking from others)
            j = int(rng.integers(n))
            if j != i and a[j] > 1e-12:
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    amt = a[j] * frac
                    a_trial = a.copy()
                    a_trial[i] += amt
                    a_trial[j] -= amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_t = fast_autocorr(a_trial).max()
                    if m_t < best_local_m:
                        best_local_m = m_t
                        best_val = a_trial

            if best_local_m < best_m - 1e-16:
                a = norm_sum1(a)
                a[i] = best_val
                a = norm_sum1(a)
                best_m = fast_autocorr(a).max()
                best_a = a.copy()
                improved_this_pass = True

            # Early exit if time is running out
            if time.monotonic() - t0 > time_limit - 5:
                break

        if improved_this_pass:
            stall = 0
        else:
            stall += 1
            # Random restart when stalled
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[i] * frac if a[i] > a[j] else a[j] * frac
                    src = i if a[i] > a[j] else j
                    dst = j if a[i] > a[j] else i
                    a[src] -= amt
                    a[dst] += amt
                    a = np.maximum(a, 0.0)
                    a = norm_sum1(a)
                    m_new = fast_autocorr(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        stall = 0
                        improved_this_pass = True

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    CoordDesc (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Multi-strategy optimization for minimizing 2*n*max(conv)/(sum^2):
    - Diverse spectral initialization
    - Peak-peeling Nesterov (main optimizer)
    - LP-Frank-Wolfe (global moves)
    - Coordinate descent (local polish)
    - Multi-length homotopy
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(round(p_score, 6)))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        best_seq = spectral_shape_init(n_base, 0)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(round(best_score, 6)))

    n0 = len(best_seq)
    print("Start: n=" + str(n0) + ", J=" + repr(round(best_score, 6)))

    # Phase 0: Quick spectral refinement for initialization diversity
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P0: Spectral init (n=" + str(n0) + ") ===")
    t0_time = min(remaining * 0.03, 30.0)
    if t0_time > 10:
        a_out, _ = spectral_refine(norm_sum1(best_seq), t0_time, verbose=False, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec0", t_global)

    # Phase 1: Peak-peeling Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P1: Peak-peel Nesterov (n=" + str(n_best) + ") ===")
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 60:
        a_out, _ = nesterov_peel(norm_sum1(best_seq), t1,
                                  beta_start=50.0, beta_end=5000.0,
                                  eta_init=0.8, n_peaks_start=80, n_peaks_end=5,
                                  verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peel1", t_global)

    # Phase 2: LP-Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P2: LP-FW (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.18, 250.0)
    if t2 > 40 and n_best <= 12000:
        a_out, _ = lp_fw_smart(norm_sum1(best_seq), t2, solver,
                                initial_K=350, max_K=2500,
                                add_per_iter=25, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 3: Second Nesterov round (high beta, focused)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P3: Nesterov r2 (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.18, 250.0)
    if t3 > 40:
        a_out, _ = nesterov_peel(norm_sum1(best_seq), t3,
                                  beta_start=200.0, beta_end=10000.0,
                                  eta_init=0.5, n_peaks_start=20, n_peaks_end=3,
                                  verbose=True, seed=123)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peel2", t_global)

    # Phase 4: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P4: Multi-length (base n=" + str(n_best) + ") ===")

    # Try a few different lengths
    candidate_ns = []
    for delta in [-1000, -500, 500, 1000, 2000, 3000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 25000 and new_n != n_best:
            candidate_ns.append(new_n)

    # Deduplicate and sort by proximity to best
    candidate_ns = sorted(set(candidate_ns), key=lambda x: abs(x - n_best))
    candidate_ns = candidate_ns[:4]  # Limit to 4 candidates

    for new_n in candidate_ns:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print("  -> n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick Nesterov
        t4a = min(remaining * 0.30, 150.0)
        if t4a > 30:
            a_out, _ = nesterov_peel(a_long, t4a,
                                      beta_start=80.0, beta_end=3000.0,
                                      eta_init=0.6, n_peaks_start=50, n_peaks_end=5,
                                      verbose=False, seed=new_n % 7919)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peel_n" + str(new_n), t_global)

        # Quick LP-FW if n is manageable
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 10000:
            t4b = min(remaining * 0.35, 120.0)
            if t4b > 20:
                a_long2 = interpolate_sequence(best_seq, new_n)
                a_out, _ = lp_fw_smart(norm_sum1(a_long2), t4b, solver,
                                        initial_K=300, max_K=1500,
                                        add_per_iter=20, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(new_n), t_global)

    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P5: CoordDescent (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.10, 100.0)
    if t5 > 15:
        a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t5, verbose=True, seed=77)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "cd1", t_global)

    # Phase 6: Final intensive LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P6: Final LP-FW (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.25, 200.0)
    if t6 > 30 and n_best <= 12000:
        a_out, _ = lp_fw_smart(norm_sum1(best_seq), t6, solver,
                                initial_K=500, max_K=3000,
                                add_per_iter=35, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Phase 7: Final Nesterov polish (very high beta)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P7: Final Nesterov (n=" + str(n_best) + ") ===")
    t7 = min(remaining * 0.30, 150.0)
    if t7 > 20:
        a_out, _ = nesterov_peel(norm_sum1(best_seq), t7,
                                  beta_start=500.0, beta_end=20000.0,
                                  eta_init=0.3, n_peaks_start=10, n_peaks_end=2,
                                  verbose=True, seed=999)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peel_final", t_global)

    # Phase 8: Final quick coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), min(remaining * 0.5, 60.0), verbose=False, seed=333)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "cd_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: J=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        best_seq = spectral_shape_init(n_fb, 0)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
