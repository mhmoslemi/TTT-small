def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    """Compute gradient of smoothed max of autocorrelation w.r.t. a."""
    n = len(y)
    c = fast_autocorr(y)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    fft_size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    a_rev = y[::-1]
    Conv = np.fft.irfft(P * np.fft.rfft(a_rev, fft_size), fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad, p, c

        y = np.maximum(y, 1e-15)

            noise_scale = 0.04 / (1.0 + 0.1 * (t // restart_every))

def sensitivity_matrix(a, peak_lags, n_dest_samples=None):
    """
    Compute sensitivity S(k, p, q) = a[k-q] + a[q-k] - a[k-p] - a[p-k]
    for peak lags and all source positions p.
    Returns array of shape (len(peak_lags), n, n) or computed on the fly.
    For efficiency, we compute per-peak.
    """
    n = len(a)
    results = []
    for k in peak_lags:
        k_int = int(k)
        # For each source p, sensitivity to moving to destination q:
        # S(k, p, q) = a[k-q] + a[q-k] - a[k-p] - a[p-k]
        # We want S < 0 (reduces peak at k)
        # And we want the move to not increase other peaks much
        # Precompute the "source penalty" for each p: a[k-p] + a[p-k]
        source_pen = np.zeros(n)
        for p in range(n):
            kp = k_int - p
            pk = p - k_int  # always negative, so a[pk] = 0 unless we handle differently
            # a[x] = 0 for x outside [0, n-1]
            val = 0.0
            if 0 <= kp < n:
                val += a[kp]
            if 0 <= pk < n:
                val += a[pk]
            source_pen[p] = val
        # For each destination q, the "dest gain": a[k-q] + a[q-k]
        dest_gain = np.zeros(n)
        for q in range(n):
            kq = k_int - q
            qk = q - k_int
            val = 0.0
            if 0 <= kq < n:
                val += a[kq]
            if 0 <= qk < n:
                val += a[qk]
            dest_gain[q] = val
        # S(k, p, q) = dest_gain[q] - source_pen[p]
        # We want to minimize this (most negative is best for reducing peak k)
        results.append((source_pen, dest_gain))
    return results

                    weights = 1.0 / (pressure + 0.01)
                    weights[reduce_i] = 0
                    if weights.sum() < 1e-15:
                        weights = np.ones(n)
                        weights[reduce_i] = 0
                    probs = weights / weights.sum()
                    candidates = rng.choice(n, size=n_cand, replace=False, p=probs)
                else:
                    candidates = np.arange(n)
                    candidates = candidates[candidates != reduce_i]

                for q in candidates:
                    q_int = int(q)
                    if q_int == reduce_i:
                        continue
                    dest_val = 0.0
                    kq = k_int - q_int
                    if 0 <= kq < n:
                        dest_val += a[kq]
                    qk = q_int - k_int
                    if 0 <= qk < n:
                        dest_val += a[qk]

                    # Sensitivity for this peak: negative is good
                    sens_k = dest_val - src_val
                    if sens_k >= 0:
                        continue  # Would increase this peak

                    # Estimate impact on other top peaks
                    max_other_impact = 0.0
                    for k2 in top_idx[:5]:
                        if k2 == k:
                            continue
                        k2i = int(k2)
                        # Impact on peak k2 from moving reduce_i to q:
                        # Δc[k2] ≈ ε * (a[k2-q] + a[q-k2] - a[k2-reduce_i] - a[reduce_i-k2])
                        d_val = 0.0
                        k2q = k2i - q_int
                        if 0 <= k2q < n:
                            d_val += a[k2q]
                        qk2 = q_int - k2i
                        if 0 <= qk2 < n:
                            d_val += a[qk2]
                        k2r = k2i - reduce_i
                        if 0 <= k2r < n:
                            d_val -= a[k2r]
                        rk2 = reduce_i - k2i
                        if 0 <= rk2 < n:
                            d_val -= a[rk2]
                        if d_val > max_other_impact:
                            max_other_impact = d_val

                    # Score: want negative sens_k (reduces peak) and small max_other_impact
                    score = sens_k + 0.5 * max(0, max_other_impact)
                    if score < best_score:
                        best_score = score
                        best_dst = q_int

                if best_dst < 0:
                    continue

                # Try different transfer amounts
                for frac in [0.002, 0.005, 0.01, 0.02, 0.03]:
                    amt = a[reduce_i] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_i] -= amt
                    a_trial[best_dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found_move = True
                        break
                if found_move:
                    break

        if not improved:
            # Random exploration when stuck
            if no_improve > 200:
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                break
                if not improved and no_improve > 500:
                    # Bigger perturbation
                    pert = rng.normal(0, 0.02 * best_m, n)
                    a_try = a + pert * a
                    a_try = np.maximum(a_try, 0)
                    s_t = a_try.sum()
                    if s_t > 1e-15:
                        a_try = a_try / s_t
                        M_try = fast_autocorr(a_try).max()
                        if M_try < best_m:
                            a = a_try
                            best_m = M_try
                            best_a = a.copy()
                            no_improve = 0

        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    APS (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve))

    return best_a, best_m


def simulated_annealing_peaks(a0, time_limit, verbose=True, seed=99):
    """
    Simulated annealing with peak-directed mutations.
    Uses sensitivity analysis to propose moves that target specific peaks.
    Accepts worse moves with probability exp(-ΔJ/T) to escape local minima.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Temperature schedule
    T_start = best_m * 0.1
    T_end = best_m * 0.001
    n_expected = 3000  # expected number of iterations

    improved_count = 0
    total_count = 0

    for it in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        # Linear cooling
        frac = min(1.0, it / float(n_expected))
        T = T_start * (T_end / T_start) ** frac

        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-16:
            best_m = c_max
            best_a = a.copy()
            improved_count += 1

        # Pick a peak to target (weighted by height above median)
        c_sorted = np.sort(c)[::-1]
        n_peaks = min(10, len(c))
        top_idx = np.argsort(c)[::-1][:n_peaks]

        # Weight peaks by how much they exceed the rest
        peak_weights = np.zeros(n_peaks)
        for i in range(n_peaks):
            peak_weights[i] = max(0, c[top_idx[i]] - c_max * 0.9)
        peak_weights += 0.1  # Avoid zero weights
        peak_weights /= peak_weights.sum()
        k_int = int(top_idx[rng.choice(n_peaks, p=peak_weights)])

        # Find contributing pairs for this peak
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        products = a[idx] * a[k_int - idx]

        if products.max() < 1e-20:
            continue

        # Pick source (weighted by product contribution)
        prod_weights = products / max(products.sum(), 1e-20)
        src_pos = int(idx[rng.choice(len(idx), p=prod_weights)])
        partner = k_int - src_pos
        if partner < 0 or partner >= n or a[src_pos] < 1e-14:
            continue

        # Choose which to reduce
        if a[src_pos] >= (a[partner] if 0 <= partner < n else 0):
            reduce_i = src_pos
        else:
            reduce_i = partner
            if reduce_i < 0 or reduce_i >= n or a[reduce_i] < 1e-14:
                continue

        # Find destination using sensitivity
        src_val = 0.0
        kp = k_int - reduce_i
        if 0 <= kp < n:
            src_val += a[kp]
        rk = reduce_i - k_int
        if 0 <= rk < n:
            src_val += a[rk]

        # Sample some destinations
        n_cand = min(50, n)
        weights = 1.0 / (1.0 + np.arange(n) * 0.001)  # Slight preference for early positions
        weights[reduce_i] = 0
        # Prefer positions far from the peak contributors
        for p in [src_pos, partner]:
            if 0 <= p < n:
                weights[p] *= 0.1
        probs = weights / weights.sum()
        candidates = rng.choice(n, size=n_cand, replace=False, p=probs)

        best_dst = -1
        best_sens = 0.0
        for q in candidates:
            q_int = int(q)
            if q_int == reduce_i:
                continue
            dest_val = 0.0
            kq = k_int - q_int
            if 0 <= kq < n:
                dest_val += a[kq]
            qk = q_int - k_int
            if 0 <= qk < n:
                dest_val += a[qk]
            sens = dest_val - src_val
            if sens < best_sens:
                best_sens = sens
                best_dst = q_int

        if best_dst < 0:
            continue

        # Choose transfer amount (scaled by temperature for exploration)
        max_amt = a[reduce_i] * 0.1
        frac = rng.uniform(0.002, 0.05) * (1.0 + T / T_start)
        amt = min(max_amt, a[reduce_i] * frac)

        a_trial = a.copy()
        a_trial[reduce_i] -= amt
        a_trial[best_dst] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t

        M_trial = fast_autocorr(a_trial).max()
        total_count += 1

        # Acceptance criterion
        delta = M_trial - c_max
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            a = a_trial

        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SA (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) +
                  " T=" + repr(round(T, 8)) + " acc=" + str(improved_count) + "/" + str(total_count))

    return best_a, best_m


def coordinate_descent_log(a0, time_limit, verbose=True, seed=200):
    """
    Coordinate descent in log-parameterization.
    a[j] = exp(x[j]) / sum(exp(x))
    Minimize max autocorrelation by adjusting one x at a time.
    """
    n = len(a0)
    a = norm_sum1(a0)
    # Initialize x from a
    x = np.log(np.maximum(a, 1e-15))
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 300

    order = rng.permutation(n)
    pos = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        j = int(order[pos % n])
        pos += 1

        # Try adjusting x[j] by various amounts
        # Current a[j] = exp(x[j]) / sum(exp(x))
        current_a_j = a[j]
        if current_a_j < 1e-15:
            stall += 1
            continue

        best_delta = 0
        best_new_m = best_m

        for delta_frac in [-0.2, -0.1, -0.05, 0.05, 0.1, 0.2]:
            new_val = current_a_j * (1 + delta_frac)
            if new_val < 0:
                continue
            # Move (new_val - current_a_j) from j to random other positions (or just rescale)
            # Actually in log-param: x[j] += delta, then renormalize
            # Simpler: just try different values for a[j] and distribute the difference
            diff = new_val - current_a_j
            if abs(diff) < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[j] = new_val
            # Distribute the difference to a random subset
            if diff > 0:
                # Need to take from elsewhere
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                others = np.where(mask)[0]
                # Take proportionally
                total_others = a[others].sum()
                if total_others < 1e-15:
                    continue
                a_trial[others] -= diff * (a[others] / total_others)
            else:
                # Give to a few random positions
                targets = rng.choice(n - 1, size=min(5, n - 1), replace=False)
                targets = targets[targets != j] if j not in targets else targets
                per_target = -diff / len(targets)
                for t_idx in targets:
                    a_trial[t_idx] += per_target
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_new_m:
                best_new_m = m_trial
                best_delta = delta_frac
                a_best = a_trial.copy()

        if best_delta != 0:
            a = a_best
            x[j] += np.log(1 + best_delta)  # Update log param
            if best_new_m < best_m - 1e-16:
                best_m = best_new_m
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    CDlog (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main search combining Nesterov, analytical peak surgery, simulated annealing,
    and log-parameterized coordinate descent. Uses multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(round(p_score, 6)))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        x = np.linspace(0, 1, n_base)
        a_init = (1.0 - x) ** 1.5 + 0.1
        best_seq = norm_sum1(a_init)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(n_base) + ", score=" + repr(round(best_score, 6)))

    n0 = len(best_seq)
    print("Starting: n=" + str(n0) + ", score=" + repr(round(best_score, 6)))

    # Phase 1: Nesterov with high initial beta
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Nesterov high-beta (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 60:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t1,
                                beta_start=50.0, beta_end=3000.0,
                                beta_growth=1.008, eta_base=0.5,
                                top_k_ratio=0.015, restart_every=1500,
                                verbose=True, seed=1)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 2: Analytical Peak Surgery
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Analytical Peak Surgery (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 40:
        a_out, _ = analytical_peak_surgery(norm_sum1(best_seq), t2, verbose=True, seed=11)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "aps1", t_global)

    # Phase 3: Simulated Annealing
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Simulated Annealing (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 40:
        a_out, _ = simulated_annealing_peaks(norm_sum1(best_seq), t3, verbose=True, seed=99)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sa1", t_global)

    # Phase 4: Nesterov refine
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Nesterov refine (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 60:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t4,
                                beta_start=100.0, beta_end=5000.0,
                                beta_growth=1.01, eta_base=0.4,
                                top_k_ratio=0.01, restart_every=1200,
                                verbose=True, seed=2)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-length ===")
    lengths_to_try = [n_best + 800, n_best + 2000, max(4000, n_best - 1500)]
    lengths_to_try = sorted(set([l for l in lengths_to_try if 3000 <= l <= 12000]))

    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("  Trying n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.30, 150.0)
        if t5a > 40:
            a_out, _ = nesterov_opt(a_long, t5a,
                                    beta_start=40.0, beta_end=2500.0,
                                    beta_growth=1.007, eta_base=0.5,
                                    top_k_ratio=0.015, restart_every=1500,
                                    verbose=True, seed=3)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ml_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t5b = min(remaining * 0.25, 100.0)
            if t5b > 30:
                a_try = interpolate_sequence(best_seq, new_n)
                a_out, _ = analytical_peak_surgery(a_try, t5b, verbose=False, seed=33)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "mlaps_n" + str(new_n), t_global)

    # Phase 6: Simulated Annealing again on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Simulated Annealing 2 (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 30:
        a_out, _ = simulated_annealing_peaks(norm_sum1(best_seq), t6, verbose=True, seed=111)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sa2", t_global)

    # Phase 7: Analytical Peak Surgery final
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: APS Final (n=" + str(n_best) + ") ===")
    t7 = min(remaining * 0.12, 150.0)
    if t7 > 30:
        a_out, _ = analytical_peak_surgery(norm_sum1(best_seq), t7, verbose=True, seed=44)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "aps_final", t_global)

    # Phase 8: Log coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Log CD (n=" + str(n_best) + ") ===")
    t8 = min(remaining * 0.10, 120.0)
    if t8 > 20:
        a_out, _ = coordinate_descent_log(norm_sum1(best_seq), t8, verbose=True, seed=200)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "cdlog1", t_global)

    # Phase 9: Final Nesterov burst with very high beta
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 9: Final Nesterov (n=" + str(n_best) + ") ===")
    if remaining > 50:
        t9 = min(remaining * 0.30, 180.0)
        if t9 > 30:
            a_out, _ = nesterov_opt(norm_sum1(best_seq), t9,
                                    beta_start=300.0, beta_end=10000.0,
                                    beta_growth=1.015, eta_base=0.25,
                                    top_k_ratio=0.005, restart_every=800,
                                    verbose=True, seed=5)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

    # Phase 10: Final APS + SA
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print("\n=== Phase 10: Final APS + SA (n=" + str(n_best) + ") ===")
        t10a = min(remaining * 0.40, 100.0)
        if t10a > 20:
            a_out, _ = analytical_peak_surgery(norm_sum1(best_seq), t10a, verbose=False, seed=55)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "aps_ult", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t10b = min(remaining * 0.50, 80.0)
            if t10b > 20:
                a_out, _ = simulated_annealing_peaks(norm_sum1(best_seq), t10b, verbose=False, seed=66)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "sa_ult", t_global)

    # Phase 11: One more Nesterov if time allows
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_best = len(best_seq)
        print("\n=== Phase 11: Last Nesterov (n=" + str(n_best) + ") ===")
        t11 = min(remaining * 0.70, 80.0)
        if t11 > 20:
            a_out, _ = nesterov_opt(norm_sum1(best_seq), t11,
                                    beta_start=500.0, beta_end=15000.0,
                                    beta_growth=1.02, eta_base=0.2,
                                    top_k_ratio=0.003, restart_every=600,
                                    verbose=True, seed=7)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_last", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1((1.0 - x) ** 1.5 + 0.1)
        best_score = evaluate_safe(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)

    return [float(x) for x in best_seq]
