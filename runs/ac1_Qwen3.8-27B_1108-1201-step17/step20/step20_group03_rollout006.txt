def simplex_project(z):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Multi-start diverse initialization with coarse-to-fine scaling
2. FISTA with log-sum-exp smoothing and adaptive beta/restarts
3. LP-Frank-Wolfe with adaptive active set management
4. Gradient-guided coordinate descent and 2-exchange
5. Frequency-domain spectral peak reduction
6. Cyclic shift optimization and power-transform diversification

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None


def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    return _MODULE_SOLVER


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def lse_smooth_grad(y, beta, top_k=None):
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and top_k < len(b) and top_k > 0:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -400, 400)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b_use)
        mask = (b_use == b_max)
        cnt = max(1, int(mask.sum()))
        p[mask] = 1.0 / cnt
    else:
        p = exp_vals / p_total
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] * 2.0
    return grad, p, b


def fista_advanced(a0, time_limit, beta_start=20.0, beta_end=1500.0,
                   beta_growth=1.004, top_k=None, restart_every=3000,
                   verbose=True, seed=42):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    s_prev = None
    y_prev = None
    step = 1.0 / max(1, int(np.sqrt(n)))
    restart_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, b = lse_smooth_grad(y, beta, top_k)

        if s_prev is not None and y_prev is not None:
            sy = y - y_prev
            gy = grad - s_prev
            sy_gy = sy.dot(gy)
            gy_gy = gy.dot(gy)
            if gy_gy > 1e-20:
                bb = sy_gy / gy_gy
                if 1e-8 < bb < 10.0:
                    step = max(1e-6, min(0.5 * step + 0.5 * bb, 3.0 / beta))
                else:
                    step = max(step * 0.85, 1e-6)
            s_prev = grad.copy()
            y_prev = y.copy()
        else:
            s_prev = grad.copy()
            y_prev = y.copy()

        eta = min(step, 0.15)
        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 80 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            restart_count += 1
            t_nest = 1.0
            s_prev = None
            y_prev = None
            step = 1.0 / max(1, int(np.sqrt(n)))
            noise_scale = 0.02 / (1.0 + 0.1 * restart_count)

            pert_type = restart_count % 5
            if pert_type == 0:
                pert = rng.normal(0, noise_scale, n) * best_a
                x_r = best_a + pert
            elif pert_type == 1:
                shift_amt = int(rng.integers(1, max(2, n // 50)))
                x_r = np.roll(best_a, shift_amt)
                x_r = np.maximum(x_r, 0.0)
            elif pert_type == 2:
                p_power = 1.0 + rng.uniform(-0.3, 0.5)
                x_r = best_a ** p_power
            elif pert_type == 3:
                x_r = best_a.copy()
                mid = n // 2
                spread = n // 10
                wing = x_r[spread:mid].copy()
                x_r[spread:mid] = x_r[mid-n//2:mid-spread]
                x_r[mid-n//2:mid-spread] = wing
            else:
                x_r = best_a + rng.normal(0, noise_scale, n) * best_a

            x_r = np.maximum(x_r, 1e-12)
            s_r = x_r.sum()
            if s_r < 1e-15:
                x_r = best_a.copy()
            else:
                x_r = x_r / s_r
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 10.0)

            if verbose and t % 2000 < 80:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    FISTA restart t=%d (%.1fs): J=%s beta=%.1f" % (t, elapsed, repr(round(J, 6)), beta))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    FISTA t=%d (%.1fs): J=%s beta=%.1f stall=%d" % (t, elapsed, repr(round(J, 6)), beta, stall))

    return best_a, best_max_b


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2000, add_per_iter=20, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 600

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size_ac = 2 * n - 1
        fft_s = 1 << (size_ac - 1).bit_length()
        Yf = np.fft.rfft(y, fft_s)
        Gf = np.fft.rfft(g_dir, fft_s)
        conv_ys = np.fft.irfft(Yf * Gf, fft_s)[:size_ac]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5),
                  max(0.02, default_alpha * 1.5), 1.0 / (t + 1)]
        alphas = [max(0.01, min(1.0, a)) for a in alphas]
        alphas = list(set(alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t <= 2:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW it=%d (%.1fs): J=%s |S|=%d" % (t, elapsed, repr(round(J, 6)), len(active_set)))

    return best_a, best_m


def gradient_guided_local_search(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for outer in range(20000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()

        top_peaks = np.argsort(ac)[::-1][:20]
        improved = False

        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(top_peaks[rng.integers(len(top_peaks))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            top_contrib = np.argsort(contrib)[::-1][:5]
            src = int(idx[top_contrib[0]])

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)

            best_dst = -1
            best_trial_m = M_curr
            for _try in range(8):
                dst = int(candidates[rng.integers(len(candidates))])
                if dst == src:
                    continue
                if a[src] < 1e-12:
                    break
                frac = float(rng.choice([0.003, 0.008, 0.02, 0.05]))
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_trial_m:
                    best_trial_m = M_trial
                    best_dst = dst
                    best_amt = amt

            if best_dst >= 0 and best_trial_m < best_m - 1e-14:
                a[src] -= best_amt
                a[best_dst] += best_amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                best_m = best_trial_m
                best_a = a.copy()
                improved = True

        if not improved:
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.002, 0.02))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            improved = True
                            break

        if verbose and outer % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    GGLS it=%d (%.1fs): J=%s" % (outer, elapsed, repr(round(J, 6))))

    return best_a, best_m


def spectral_peak_reduction(a0, time_limit, n_iters=100, verbose=True):
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        A = np.fft.rfft(a, fft_size)
        b = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
        b_max = b.max()
        if b_max < 1e-15:
            break
        b_norm = b / b_max
        freqs = np.fft.rfftfreq(fft_size, d=1.0)
        decay = 1.0 - 0.4 * b_norm[np.minimum(np.abs(np.arange(len(A))) % n, n - 1)]
        decay = np.maximum(decay, 0.3)
        A_new = A * decay
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        a = a_new
        if verbose and it % 30 == 0:
            J = 2.0 * n * best_m
            print("    Spectral it=%d: J=%s" % (it, repr(round(J, 6))))

    return best_a, best_m


def cyclic_shift_best(a0):
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a0, fft_size)
    for s in range(1, n):
        phase = np.exp(-2j * np.pi * s * np.arange(len(A)) / fft_size)
        AS = A * phase
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def diverse_initializer(n, rng):
    x = np.linspace(0, 1, n)
    init_type = rng.integers(0, 8)
    if init_type == 0:
        f = np.sin(np.pi * x) ** 2
    elif init_type == 1:
        f = np.exp(-8.0 * (x - 0.5) ** 2)
    elif init_type == 2:
        f = (1.0 + np.cos(2 * np.pi * x)) ** 1.5
    elif init_type == 3:
        f = np.abs(np.sin(np.pi * x * (1.0 + 0.5 * rng.random()))) ** 2
    elif init_type == 4:
        f = np.exp(-4.0 * np.abs(x - 0.5)) * np.sin(np.pi * x) ** 2
    elif init_type == 5:
        f = np.maximum(0.0, 1.0 - 2.0 * np.abs(x - 0.5)) ** 3
    elif init_type == 6:
        f = np.exp(-6.0 * (x - 0.5) ** 2) * (1.0 + 0.3 * np.cos(4 * np.pi * x))
    else:
        f = np.ones(n) / n
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    return f


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [%s]: %s (n=%d, t=%.1fs)" % (label, repr(sc), len(a), elapsed))
    return best_seq, best_score

    # Phase 2: Main FISTA on best
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: FISTA main (n=%d) ===" % len(best_seq))
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 60:
        a_out, _ = fista_advanced(norm_sum1(best_seq), t2,
                                  beta_start=15.0, beta_end=800.0,
                                  beta_growth=1.003, top_k=15,
                                  restart_every=3000, verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_main", t_global)

    # Phase 3: Cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: Cyclic shift ===")
    if remaining > 30:
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic1", t_global)

    # Phase 4: LP-Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW (n=%d) ===" % n_best)
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 40 and n_best <= 8000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t4, solver,
                         initial_K=400, max_K=2000, add_per_iter=25, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1", t_global)

    # Phase 5: Gradient-guided local search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: GGLS (n=%d) ===" % n_best)
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 30:
        a_out, _ = gradient_guided_local_search(norm_sum1(best_seq), t5, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "gglss1", t_global)

    # Phase 6: Spectral peak reduction
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 6: Spectral peak reduction ===")
    t6 = min(remaining * 0.05, 60.0)
    if t6 > 15:
        a_out, _ = spectral_peak_reduction(norm_sum1(best_seq), t6, n_iters=50, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spectral1", t_global)

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Multi-length exploration ===")
    target_lengths = []
    for dl in [500, 1000, 2000, 3000]:
        ln = n_best + dl
        if ln <= 12000 and ln > n_best:
            target_lengths.append(ln)
    for ln in [n_best - 1000, n_best - 2000]:
        if ln >= 1000:
            target_lengths.append(ln)
    target_lengths = sorted(set(target_lengths), key=lambda x: abs(x - n_best))

    for new_n in target_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("  --- n=%d ---" % new_n)
        a_long = interpolate_sequence(best_seq, new_n)

        t7 = min(remaining * 0.15, 120.0)
        if t7 > 30:
            a_out, _ = fista_advanced(a_long, t7,
                                      beta_start=20.0, beta_end=600.0,
                                      beta_growth=1.004, top_k=12,
                                      restart_every=2500, verbose=False, seed=77)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_n%d" % new_n, t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
            best_seq, best_score = try_update(best_seq, best_score, a_out if a_out is not None else best_seq, "shift_n%d" % new_n, t_global)

    # Phase 8: Second FISTA pass with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: FISTA v2 (n=%d) ===" % n_best)
    t8 = min(remaining * 0.15, 200.0)
    if t8 > 40:
        a_out, _ = fista_advanced(norm_sum1(best_seq), t8,
                                  beta_start=50.0, beta_end=2000.0,
                                  beta_growth=1.006, top_k=8,
                                  restart_every=2000, verbose=True, seed=555)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista2", t_global)

    # Phase 9: Final LP burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 9: LP burst (n=%d) ===" % n_best)
    if remaining > 60 and n_best <= 8000:
        t9 = min(remaining * 0.15, 150.0)
        if t9 > 30:
            a_out, _ = lp_fw(norm_sum1(best_seq), t9, solver,
                             initial_K=500, max_K=2500, add_per_iter=30, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2", t_global)

    # Phase 10: Final local search polish
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 10: Final GGLS polish ===")
    if remaining > 40:
        t10 = min(remaining * 0.25, 100.0)
        if t10 > 20:
            a_out, _ = gradient_guided_local_search(norm_sum1(best_seq), t10, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "gglss2", t_global)

    # Final: cyclic shift + micro FISTA
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print("\n=== Final: Cyclic shift ===")
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_final", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== Final: FISTA micro ===")
        tf = min(remaining * 0.6, 80.0)
        if tf > 20:
            a_out, _ = fista_advanced(norm_sum1(best_seq), tf,
                                      beta_start=100.0, beta_end=3000.0,
                                      beta_growth=1.01, top_k=6,
                                      restart_every=1500, verbose=False, seed=314)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_final", t_global)

    # Final GGLS micro
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        tg = min(remaining * 0.7, 60.0)
        if tg > 15:
            a_out, _ = gradient_guided_local_search(norm_sum1(best_seq), tg, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "gglss_final", t_global)

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=%s, n=%d, time=%.1fs" % (repr(best_score), len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
