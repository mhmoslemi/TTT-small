def lp_direction(a, M, top_k=50):
    """Solve reduced LP to find best direction."""
    n = len(a)
    c = fast_autocorr(a)
    # Find top_k peaks
    peak_indices = np.argsort(c)[::-1][:top_k]
    
    g = cp.Variable(n, nonneg=True)
    obj = cp.Maximize(cp.sum(g))
    constraints = [cp.sum(g) == 1]  # Normalize
    
    for k in peak_indices:
        # (a*g)[k] = sum_j a[k-j]*g[j] (valid j)
        # Build the coefficient vector
        coeff = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            coeff[j] = a[k - j]
        constraints.append(coeff @ g <= M)
    
    prob = cp.Problem(obj, constraints)
    prob.solve()
    
    if g.value is not None:
        g_val = np.maximum(g.value, 0)
        s = g_val.sum()
        if s > 0:
            return g_val / s
    return None

import numpy as np

import cvxpy as cp

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_range = np.arange(j_start, j_end + 1)
        coeff_vals = a[k_int - j_range]
        if coeff_vals.max() < 1e-18:
            continue
        sub_g = g[j_start:j_end + 1]
        constraints.append(coeff_vals @ sub_g <= z)

    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None, None

    if g.value is not None:
        g_val = np.maximum(g.value, 0.0)
        s = g_val.sum()
        if s > 1e-15:
            return g_val / s, z.value
    return None, None

def solve_bounded_lp(a, peak_indices, M, solver, verbose=False):
    """Solve: max sum(g) s.t. c_ag[k] <= M for k in peak_indices, g>=0.
    Returns normalized g."""
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    obj = cp.Maximize(cp.sum(g))
    constraints = []

    for k in peak_indices:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        j_range = np.arange(j_start, j_end + 1)
        coeff_vals = a[k_int - j_range]
        if coeff_vals.max() < 1e-18:
            continue
        sub_g = g[j_start:j_end + 1]
        constraints.append(coeff_vals @ sub_g <= M)

    if len(constraints) == 0:
        return None

    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None

    if g.value is not None:
        g_val = np.maximum(g.value, 0.0)
        s = g_val.sum()
        if s > 1e-10:
            return g_val / s
    return None

def line_search_blend(c_aa, c_ag, c_gg, n, n_coarse=30, n_fine=30):
    """Find best t minimizing max((1-t)^2*c_aa + 2t(1-t)*c_ag + t^2*c_gg)."""
    best_t = 0.0
    best_val = float('inf')

    ts = np.linspace(0.0, 1.0, n_coarse)
    for t in ts:
        one_minus_t = 1.0 - t
        c_blend = one_minus_t * one_minus_t * c_aa + 2.0 * t * one_minus_t * c_ag + t * t * c_gg
        val = 2.0 * n * c_blend.max()
        if val < best_val:
            best_val = val
            best_t = t

    if best_t > 0.01 and best_t < 0.99:
        lo = max(0.0, best_t - 1.0 / n_coarse)
        hi = min(1.0, best_t + 1.0 / n_coarse)
        ts_fine = np.linspace(lo, hi, n_fine)
        for t in ts_fine:
            one_minus_t = 1.0 - t
            c_blend = one_minus_t * one_minus_t * c_aa + 2.0 * t * one_minus_t * c_ag + t * t * c_gg
            val = 2.0 * n * c_blend.max()
            if val < best_val:
                best_val = val
                best_t = t

    return best_t, best_val

def lp_optimize(a0, time_limit, verbose=True, seed=42):
    """Main LP-directed optimization loop with both minimax and bounded LP formulations."""
    solver = detect_solver()
    n = len(a0)
    a = norm_sum1(a0)
    c_aa = fast_autocorr(a)
    best_a = a.copy()
    best_score = 2.0 * n * c_aa.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 150
    iteration = 0

    top_k_schedule = [30, 50, 80, 120, 200, 400]
    k_idx = 1
    use_bounded = False
    bounded_count = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iteration += 1
        current_score = 2.0 * n * c_aa.max()

        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            stall = 0

        M = c_aa.max()
        top_k = top_k_schedule[min(k_idx, len(top_k_schedule) - 1)]
        peak_indices = np.argsort(c_aa)[::-1][:top_k]

        g = None
        if not use_bounded:
            g, z_val = solve_minimax_lp(a, peak_indices, solver)
        else:
            g = solve_bounded_lp(a, peak_indices, M * 1.0, solver)

        if g is None:
            stall += 2
            use_bounded = not use_bounded
            continue

        c_ag = fast_cross_corr(a, g)
        c_gg = fast_autocorr(g)

        best_t, best_val = line_search_blend(c_aa, c_ag, c_gg, n)

        if best_t > 0.002 and best_val < current_score - 1e-12:
            a = (1.0 - best_t) * a + best_t * g
            a = norm_sum1(a)
            c_aa = fast_autocorr(a)
            stall = 0
        else:
            stall += 1
            k_idx = (k_idx + 1) % len(top_k_schedule)
            if stall % 30 == 0:
                use_bounded = not use_bounded
                bounded_count += 1
            if stall % 60 == 0:
                noise_scale = 0.02 * np.exp(-stall / 200.0)
                noise = rng.normal(0, noise_scale, n) * a
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0.0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a_pert = a_pert / s_pert
                    c_pert = fast_autocorr(a_pert)
                    if c_pert.max() < c_aa.max():
                        a = a_pert
                        c_aa = c_pert
                        stall = 0
                k_idx = int(rng.integers(len(top_k_schedule)))

        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - t0
            print("    LP opt t=" + str(iteration) + " (" + str(round(elapsed, 1)) + "s): "
                  + "score=" + repr(round(2.0 * n * c_aa.max(), 8)) +
                  " best=" + repr(round(best_score, 8)) +
                  " stall=" + str(stall) +
                  " k=" + str(top_k) +
                  " mode=" + ("bounded" if use_bounded else "minimax"))

    return best_a, best_score / (2.0 * n)

def compute_smoothed_grad(y, beta, top_k):
    n = len(y)
    c = fast_autocorr(y)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total

    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(y, fft_size)
    P = np.fft.rfft(p, fft_size)
    a_rev = y[::-1]
    Conv = np.fft.irfft(P * np.fft.rfft(a_rev, fft_size), fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad, p, c

def nesterov_opt(a0, time_limit, beta_start=50.0, beta_end=5000.0,
                 beta_growth=1.008, eta_base=0.5, top_k_ratio=0.015,
                 restart_every=1500, verbose=True, seed=42):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, c = compute_smoothed_grad(y, beta, top_k)

        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.03 / (1.0 + 0.05 * (t // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + ", beta=" + str(round(beta, 1)))

    return best_a, best_max_b

def peak_surgery(a0, time_limit, verbose=True, seed=77):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 5000

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_top = min(50, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]

        improved = False
        found_move = False

        for k in top_idx[:20]:
            if found_move:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            if products.max() < 1e-20:
                continue

            src_i = int(idx[np.argmax(products)])
            partner = k_int - src_i
            if partner < 0 or partner >= n:
                continue

            src_val = a[src_i]
            par_val = a[partner]

            if src_val >= par_val:
                reduce_i = src_i
            else:
                reduce_i = partner

            if a[reduce_i] < 1e-14:
                continue

            pressure = np.zeros(n)
            for k2 in top_idx[:15]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_idx[:15]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 100:
                candidates = np.arange(n)

            mask = candidates != reduce_i
            if not mask.any():
                continue
            candidates = candidates[mask]

            cand_pressure = pressure[candidates]
            n_try = min(8, len(candidates))
            try_dsts = candidates[np.argsort(cand_pressure)[:n_try]]

            for dst in try_dsts:
                dst_int = int(dst)
                if dst_int == reduce_i:
                    continue
                for frac in [0.003, 0.005, 0.01, 0.015, 0.02, 0.03]:
                    amt = a[reduce_i] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_i] -= amt
                    a_trial[dst_int] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found_move = True
                        break
                if found_move:
                    break

        if not improved:
            if no_improve > 500:
                for _ in range(15):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.04)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                break

    return best_a, best_m

def pairwise_cd(a0, time_limit, verbose=True, seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 800:
        c = fast_autocorr(a)
        threshold = np.percentile(c, 95)
        peak_positions = np.where(c > threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.argsort(c)[::-1][:20]

        improved = False

        for k in peak_positions[:25]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]

            top_pairs = np.argsort(products)[::-1][:3]
            for pi in top_pairs:
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n or partner == src_i:
                    continue
                if a[src_i] < 1e-14 and a[partner] < 1e-14:
                    continue

                larger_i = src_i if a[src_i] >= a[partner] else partner
                if a[larger_i] < 1e-14:
                    continue

                for frac in [0.001, 0.002, 0.003, 0.005]:
                    amt = a[larger_i] * frac
                    if amt < 1e-18:
                        continue
                    for _ in range(4):
                        dst = int(rng.integers(n))
                        if dst == larger_i:
                            continue
                        a_trial = a.copy()
                        a_trial[larger_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall % 150 == 0:
                anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
                eps = 0.008 / (1.0 + stall / 100.0)
                a_try = a + eps * anti * a
                a_try = np.maximum(a_try, 0.0)
                s_t = a_try.sum()
                if s_t > 1e-15:
                    a_try = a_try / s_t
                    M_try = fast_autocorr(a_try).max()
                    if M_try < best_m:
                        a = a_try
                        best_m = M_try
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PairCD (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 8)) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

    t6 = min(remaining * 0.18, 250.0)
    if t6 > 30:
        a_out, _ = pairwise_cd(norm_sum1(best_seq), t6, verbose=True, seed=55)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd1", t_global)

    # Phase 7: Final LP burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final LP (n=" + str(n_best) + ") ===")
    if remaining > 80:
        t7 = min(remaining * 0.30, 200.0)
        if t7 > 40:
            a_out, _ = lp_optimize(norm_sum1(best_seq), t7, verbose=True, seed=99)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Phase 8: Final Nesterov + Pairwise
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final polish (n=" + str(n_best) + ") ===")
    if remaining > 80:
        t8a = min(remaining * 0.35, 150.0)
        if t8a > 30:
            a_out, _ = nesterov_opt(norm_sum1(best_seq), t8a,
                                    beta_start=200.0, beta_end=8000.0,
                                    beta_growth=1.012, eta_base=0.3,
                                    top_k_ratio=0.005, restart_every=1000,
                                    verbose=False, seed=44)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t8b = min(remaining * 0.50, 120.0)
            if t8b > 20:
                a_out, _ = pairwise_cd(norm_sum1(best_seq), t8b, verbose=False, seed=66)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1((1.0 - x) ** 1.5 + 0.1)
        best_score = evaluate_safe(best_seq)

    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)

    return [float(x) for x in best_seq]
