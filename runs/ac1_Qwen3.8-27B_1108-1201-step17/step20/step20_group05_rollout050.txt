Hybrid optimizer for minimizing 2*n*max(autocorrelation)/sum^2.
Uses L-BFGS smoothing, accelerated Frank-Wolfe, and coordinate descent
with multi-start and adaptive parameters.
"""

import numpy as np
import time
import math

# === Core utilities ===

def next_pow2(x):
    x = int(max(1, x))
    return 1 << (x - 1).bit_length()

def norm_sum1(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    np.maximum(a, 0.0, out=a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(a.size) / max(1, a.size)
    return a / s

def fft_conv(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return np.where(c > 0, c, 0.0)

def fft_conv2(a, b):
    n = a.size
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    return np.where(c > 0, c, 0.0)

def cross_conv_p_a(p, a):
    """Compute the 'pressure' vector: for each i, sum over active peaks k of p_k * a_{k-i}"""
    n = a.size
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    # p has length 2n-1, we need cross-correlation
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    out = conv[n - 1 : 2 * n - 1][::-1]
    return np.where(np.isfinite(out), out, 0.0)

def lse_value_grad(a, beta, top_k=None):
    """Log-sum-exp smoothed objective and its gradient."""
    n = a.size
    c = fft_conv(a)
    size = 2 * n - 1
    
    if top_k is not None and top_k < size:
        idx = np.argsort(c)[::-1][:top_k]
        cv = c[idx]
    else:
        cv = c
    
    if cv.size == 0:
        return 0.0, np.zeros(n)
    
    m = cv.max()
    shifted = np.clip(beta * (cv - m), -500, 500)
    e = np.exp(shifted)

    if se < 1e-300:
        return m, np.zeros(n)
    p = e / se
    
    # Build full weight vector

        pf = np.zeros(size)
        pf[idx] = p
    else:
        pf = p

    grad = 2.0 * conv[n - 1 : 2 * n - 1][::-1]

def softmax_params(u):
    """Convert unnormalized params to probability vector."""
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return np.ones(1)
    u = u - u.max()
    e = np.exp(u)
    s = e.sum()
    if s < 1e-300:
        return np.ones(u.size) / u.size
    return e / s

def to_params(a, ref):
    """Convert a to (v, ref) where a = softmax([v_0,...,v_{ref-1}, 0, v_{ref+1},...])"""
    n = a.size
    if n <= 1:
        return np.array([], dtype=np.float64), 0
    ref = int(ref)
    a_safe = np.maximum(a, 1e-300)
    u = np.log(a_safe) - math.log(a_safe[ref])
    if ref == 0:
        v = u[1:]
    elif ref == n - 1:
        v = u[:-1]
    else:
        v = np.concatenate([u[:ref], u[ref+1:]])
    return v, ref

def from_params(v, ref):
    """Convert (v, ref) back to probability vector."""
    n = v.size + 1
    u = np.zeros(n)
    ref = int(ref)
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref+1:] = v[ref:]
    return softmax_params(u)

def obj_grad_params(v, beta, ref, top_k=None):
    """Objective and gradient in parameter space."""
    n = v.size + 1
    a = from_params(v, ref)

    gu = a * (ga - mean_g)
    gv = np.empty(v.size)
    if ref > 0:

    if ref < n - 1:

def fw_game_direction(f, active_idx, n_iter=30, seed=0):
    """Solve the FW subproblem approximately using a zero-sum game."""
    n = f.size
    c = fft_conv(f)
    K = active_idx.size
    if K == 0:
        return np.ones(n) / n
    
    rng = np.random.default_rng(seed)
    logits = np.zeros(K)
    # Initialize with current peak values
    logits = np.log(np.maximum(c[active_idx], 1e-300))
    logits -= logits.max()
    logits += rng.normal(0, 0.1, K)
    
    for _ in range(n_iter):
        # Player 1: choose g on simplex to minimize max_k f*g at active k
        lam = softmax_params(logits)
        pf = np.zeros(2 * n - 1)
        pf[active_idx] = lam
        
        pressure = cross_conv_p_a(pf, f)
        pressure = np.maximum(pressure, 1e-300)
        
        # g is proportional to exp(-gamma * pressure)
        pmax = pressure.max()
        pmin = pressure.min()
        if pmax - pmin < 1e-30:
            g = np.ones(n) / n
        else:
            scaled = (pressure - pmin) / (pmax - pmin)
            g = np.exp(-6.0 * scaled)
            g = np.maximum(g, 1e-300)
            g = g / g.sum()
            # Add small uniform component
            g = 0.98 * g + 0.02 * (np.ones(n) / n)
            g = g / g.sum()

        if vrange > 1e-30:
            eta = 10.0 / vrange
            logits += eta * (val - val.mean())
        logits = np.where(np.isfinite(logits), logits, 0.0)
    
    return g

    best_val = c.max()
    for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
        new_c = (1-alpha)**2 * c + 2*alpha*(1-alpha) * fg + alpha**2 * cg
        val = new_c.max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha == 0:
        return a, c, 0.0
    
    new_a = (1 - best_alpha) * a + best_alpha * g
    new_a = np.maximum(new_a, 0)
    new_a = new_a / new_a.sum()
    new_c = fft_conv(new_a)
    return new_a, new_c, best_alpha

def accelerated_fw(a0, budget, max_active=2000, verbose=True):
    """Nesterov-accelerated Frank-Wolfe."""
    t0 = time.monotonic()
    a = a0.copy()
    y = a0.copy()  # Nesterov y
    c = fft_conv(a)
    M = c.max()
    best_a = a.copy()
    best_M = M
    n = a.size
    
    stall = 0
    t = 0
    active = np.argsort(c)[::-1][:min(max_active, 2*n-1)]
    
    while time.monotonic() - t0 < budget - 5 and stall < 100:
        t += 1
        # Nesterov update
        t_nest = 1 + 2 * math.sqrt(t)
        y = (t * a + a0) / t_nest  # Wait, this is for convex minimization
        # Actually for FW, the standard update is:
        # y_t = (1 - alpha_t) * x_{t-1} + alpha_t * direction
        # With Nesterov: y_t = (1 - 1/t) * x_{t-1} + ... 
        # Let me just use standard FW with adaptive step
        
        alpha = 2.0 / (t + 2.0)
        
        g = fw_game_direction(y, active, game_iters=20, seed=1000+t)
        if g is None:
            break
        
        # Line search from a to g
        cg = fft_conv(g)
        ag = fft_conv2(a, g)
        
        best_alpha_ls = 0.0
        best_M_ls = M
        for test_alpha in [alpha, 2*alpha, 0.5*alpha, 3*alpha]:
            test_alpha = min(1.0, max(0.01, test_alpha))
            new_c = (1-test_alpha)**2 * c + 2*test_alpha*(1-test_alpha) * ag + test_alpha**2 * cg
            val = new_c.max()
            if val < best_M_ls:
                best_M_ls = val
                best_alpha_ls = test_alpha
        
        if best_alpha_ls > 0:
            a = (1 - best_alpha_ls) * a + best_alpha_ls * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            c = fft_conv(a)
            M = c.max()
            
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            
        # Update y (Nesterov interpolation)
        y = (t / (t+1)) * a + (1 / (t+1)) * a0
        
        # Update active set
        if t % 5 == 0:
            active = np.argsort(c)[::-1][:min(max_active, 2*n-1)]
        
        if stall > 20:
            # Perturb and restart
            rng = np.random.default_rng(42 + t)
            a = a + rng.normal(0, 0.02, n) * a
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            c = fft_conv(a)
            M = c.max()
            stall = 0
            active = np.argsort(c)[::-1][:min(max_active, 2*n-1)]
        
        if verbose and t % 20 == 0:
            print(f"  FW t={t} M={M:.8f} best_M={best_M:.8f} alpha={best_alpha_ls:.4f}")
    
    return best_a

def coord_polish(a0, budget, verbose=True):
    """Coordinate descent polish with single and pair moves."""
    t0 = time.monotonic()
    a = a0.copy()
    n = a.size
    c = fft_conv(a)
    M = c.max()
    best_a = a.copy()
    best_M = M
    rng = np.random.default_rng(123)
    stall = 0
    
    def apply_move(c, a, s, d, dt):
        """Apply a transfer of dt from position s to position d."""
        c_new = c.copy()
        # The convolution changes as:
        # (a*a)_k gains: 2*dt*(a_{k-s} + a_{k-d}) - 2*dt*a_{s+d-s}... 
        # Actually: new_a = a - dt*e_s + dt*e_d
        # new_c = c - 2*dt*(a shifted by s, reversed) + 2*dt*(a shifted by d) - dt^2*(delta_{2s} - delta_{2d} - delta_{s+d})
        # More precisely:
        # (new_a * new_a)_k = (a*a)_k - 2*dt*sum_j a_j*delta(k-j-s) + 2*dt*sum_j a_j*delta(k-j+d)
        #   - dt^2*delta(k-2s) + dt^2*delta(k-2d) + 2*dt^2*delta(k-(s+d))
        # Wait, let me be more careful.
        # new_a[i] = a[i] - dt if i==s, a[i] + dt if i==d, a[i] otherwise
        # (new_a * new_a)[k] = sum_i new_a[i] * new_a[k-i]
        # = sum_i (a[i] + dt*(delta_{i,d} - delta_{i,s})) * (a[k-i] + dt*(delta_{k-i,d} - delta_{k-i,s}))
        # = (a*a)[k] + dt*sum_i a[i]*(delta_{k-i,d} - delta_{k-i,s}) + dt*sum_i (delta_{i,d} - delta_{i,s})*a[k-i]
        #   + dt^2*(delta_{d,d}-delta_{d,s})*(delta_{k-d,d}-delta_{k-d,s})  -- wait this is getting complicated
        
        # Let me just use the rank-1 update:
        # new_a = a + dt*(e_d - e_s)
        # new_c = c + 2*dt*(a convolved with (e_d - e_s)) + dt^2*(e_d - e_s)*(e_d - e_s)
        # (a convolved with e_d)[k] = a[k-d] (if 0 <= k-d < n)
        # (a convolved with e_s)[k] = a[k-s] (if 0 <= k-s < n)
        
        # Contribution from 2*dt*(a*e_d):
        if 0 <= d:
            start = d
            end = min(d + n, 2*n - 1)
            c_new[start:end] += 2 * dt * a[:end-start]
        # Contribution from -2*dt*(a*e_s):
        if 0 <= s:
            start = s
            end = min(s + n, 2*n - 1)
            c_new[start:end] -= 2 * dt * a[:end-start]
        # dt^2 * e_d * e_d = dt^2 at position 2*d
        if 2*d < 2*n-1:
            c_new[2*d] += dt*dt
        # -dt^2 * e_s * e_s
        if 2*s < 2*n-1:
            c_new[2*s] -= dt*dt
        # -dt^2 * e_d * e_s at position d+s... wait
        # (e_d - e_s)*(e_d - e_s) = e_d*e_d - e_d*e_s - e_s*e_d + e_s*e_s
        # The cross terms: -2*dt * (a*e_s)[k] already handles a*s part
        # But we also need the dt^2 cross terms:
        # dt^2 * (e_d - e_s) * (e_d - e_s) = dt^2 * (delta_{2d} - delta_{d+s} - delta_{d+s} + delta_{2s})
        # = dt^2 * (delta_{2d} - 2*delta_{d+s} + delta_{2s})
        # We already added dt^2 at 2d and subtracted dt^2 at 2s
        # Now add the cross terms:
        if s + d < 2*n-1:
            c_new[s+d] -= 2 * dt * dt
        
        return np.where(c_new > 0, c_new, 0.0)
    
    while time.monotonic() - t0 < budget - 3 and stall < 200:
        # Find top peaks
        top_k = min(30, 2*n-1)
        top_idx = np.argsort(c)[::-1][:top_k]
        
        # Compute pressure
        p_top = np.zeros(2*n-1)
        p_top[top_idx] = 1.0 / top_k
        pressure = cross_conv_p_a(p_top, a)
        
        # Try single moves: move from high-pressure to low-pressure
        best_move = None
        best_val = M
        
        # Source: high pressure + high value
        src_score = pressure * a
        # Dest: low pressure
        dst_score = -pressure
        
        src_candidates = np.argpartition(src_score, min(10, n-1))[::-1][:10]
        dst_candidates = np.argpartition(dst_score, min(15, n-1))[:15]
        
        for s in src_candidates:
            s = int(s)
            if a[s] < 1e-12:
                continue
            for d in dst_candidates:
                d = int(d)
                if d == s:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    dt = a[s] * frac
                    if dt < 1e-15:
                        continue
                    c_new = apply_move(c, a, s, d, dt)
                    val = c_new.max()
                    if val < best_val - 1e-18:
                        best_val = val
                        best_move = (s, d, dt)
        
        if best_move is not None:
            s, d, dt = best_move
            a[s] -= dt
            a[d] += dt
            a = np.maximum(a, 0)
            a = a / a.sum()
            c = fft_conv(a)
            M = c.max()
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
        else:
            # Random exploration
            improved = False
            for _ in range(50):
                s = int(rng.integers(n))
                d = int(rng.integers(n))
                if s == d or a[s] < 1e-12:
                    continue
                frac = float(rng.uniform(0.001, 0.05))
                dt = a[s] * frac
                c_new = apply_move(c, a, s, d, dt)
                val = c_new.max()
                if val < M - 1e-18:
                    a[s] -= dt
                    a[d] += dt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    c = fft_conv(a)
                    M = c.max()
                    if M < best_M:
                        best_M = M
                        best_a = a.copy()
                    improved = True
                    break
            if improved:
                stall = 0
            else:
                stall += 1
    
    if verbose:
        print(f"  Coord polish done: best_M={best_M:.8f}")
    return best_a
