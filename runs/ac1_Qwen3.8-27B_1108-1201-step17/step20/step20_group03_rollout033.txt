            vals.append(y[k - j])

def _norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def _smoothed_max_grad(y, beta, top_k=None):
    n = len(y)
    b = _fast_autocorr(y)
    b_max = b.max()
    if top_k is not None and 0 < top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
    else:
        b_use = b
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -400, 400)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b_use)
        mask = b_use >= b_max - 1e-12
        p[mask] = 1.0 / mask.sum()
    else:
        p = exp_vals / p_total
    # Gradient of smooth max of conv(y,y) w.r.t. y
    # d/dy[j] of conv(y,y)[k] = 2*y[k-j] (when valid), so gradient = 2 * conv(p, y_reversed) / ... 
    # Actually: grad[j] = sum_k p[k] * d(conv(y,y)[k])/dy[j] = sum_k p[k] * 2*y[k-j] (valid)
    # This equals 2 * conv(p, y)[j] ... let me think
    # conv(p, y)[j] = sum_i p[i] * y[j-i] (with bounds)
    # We want: sum_k p[k] * y[k-j] for valid k,j (0<=k-j<n)
    # Let m = k-j, then k = j+m, sum_m p[j+m] * y[m] = ...
    # This is the cross-correlation, which equals conv(p, y_flipped) shifted.
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_sz = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_sz)
    Pr = np.fft.rfft(p, fft_sz)
    conv = np.fft.irfft(Ar * Pr, fft_sz)
    # conv[i] = sum_m a_rev[m] * p[i-m] = sum_m y[n-1-m] * p[i-m]
    # We want grad[j] = 2 * sum_{k: 0<=k-j<n} p[k] * y[k-j]
    # Let's just compute it directly using the convolution
    # sum_k p[k]*y[k-j] for 0<=k-j<=n-1 means k from j to j+n-1
    # This is the cross-correlation of p and y at lag -j
    # = conv(p_padded, y_reversed)[j + n - 1] ... let me just do it with FFT
    # Actually the simplest: grad = 2 * ifft(fft(p) * conj(fft(y))) ... no
    # Let's use: grad[j] = 2 * sum_k p[k] * y[k-j] = 2 * (p cross-correlated with y) at lag j
    # cross_corr_p_y[j] = sum_k p[k] * y[k-j]
    # This equals conv(p, y_flipped)[j + n - 1] where y_flipped = y[::-1]
    # conv(p, y_flipped)[m] = sum_i p[i] * y_flipped[m-i] = sum_i p[i] * y[n-1-(m-i)]
    # For m = j + n - 1: = sum_i p[i] * y[n-1-j-n+1+i] = sum_i p[i] * y[i-j]
    # Hmm, that's sum_i p[i]*y[i-j], not sum_k p[k]*y[k-j]. These are the same by substitution i=k.
    # So cross_corr[j] = conv(p, y_flipped)[j+n-1]
    # But we need to handle bounds. Let's just use the full convolution.
    pf = np.fft.rfft(p, fft_sz)
    yf = np.fft.rfft(y, fft_sz)
    # cross-correlation: sum_k p[k]*y[k-j] = ifft(fft(p) * conj(fft(y))) ... 
    # Actually: ifft(FFT(p) * conj(FFT(y)))[j] = sum_k p[k] * conj(y)[(k-j) mod N]
    # For real y: = sum_k p[k] * y[(k-j) mod N]
    # With N = fft_sz >> 2n, this equals sum_k p[k] * y[k-j] (no wraparound) for valid j
    cross = np.fft.irfft(pf * np.conj(yf), fft_sz)
    grad = 2.0 * cross[:n]
    return grad, b


def _fista_optimize(a0, time_limit, beta_start=30.0, beta_end=1500.0,
                    beta_growth=1.004, top_k=20, restart_every=3000, verbose=True):
    n = len(a0)
    x = _norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_mb = _fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(42)
    bb_step = 1.0 / np.sqrt(n)
    s_prev = None
    y_prev = None

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, b = _smoothed_max_grad(y, beta, top_k)

        # BB step size
        if s_prev is not None and y_prev is not None:
            sy = y - y_prev
            gy = grad - s_prev
            sy_gy = sy.dot(gy)
            gy_gy = gy.dot(gy)
            if gy_gy > 1e-20:
                bb_cand = sy_gy / gy_gy
                if 1e-8 < bb_cand < 5.0 / beta:
                    bb_step = 0.7 * bb_step + 0.3 * bb_cand
                    bb_step = np.clip(bb_step, 1e-6, 5.0 / beta)
                else:
                    bb_step = max(bb_step * 0.85, 1e-6)
        s_prev = grad.copy()
        y_prev = y.copy()

        eta = min(bb_step, 0.15)
        z = y - eta * grad
        x_new = _simplex_project(z)

        b_new = _fast_autocorr(x_new)
        max_new = b_new.max()

        if max_new < best_mb - 1e-15:
            best_mb = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.02 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 10.0)
            s_prev = None
            y_prev = None
            bb_step = 1.0 / np.sqrt(n)

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_mb
            print(f"      FISTA t={t} ({elapsed:.0f}s): J={J:.6f} beta={beta:.0f}")

    return best_a, best_mb


def _lp_fw_optimize(a0, time_limit, solver, K_max=4000, verbose=True):
    n = len(a0)
    y = _norm_sum1(a0)
    best_a = y.copy()
    best_m = _fast_autocorr(y).max()
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 500

    # Precompute structure
    n_rows = 2 * n - 1
    row_list = []
    col_list = []
    for k in range(n_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            row_list.extend([k] * (j_end - j_start + 1))
            col_list.extend(range(j_start, j_end + 1))
    row_arr = np.array(row_list, dtype=np.int32)
    col_arr = np.array(col_list, dtype=np.int32)
    nnz = len(row_arr)

    # Initial active set: top K peaks
    ac = _fast_autocorr(y)
    active = set(int(k) for k in np.argsort(ac)[::-1][:min(K_max, n_rows)])

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        M_curr = _fast_autocorr(y).max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Build constraint matrix for active set
        active_sorted = sorted(active)
        K = len(active_sorted)
        if K == 0:
            break

        # Build sparse rows for active constraints
        r_idx = []
        c_idx = []
        vals = []
        y_arr = y
        for k in active_sorted:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                r_idx.extend([len(r_idx) // (j_end - j_start + 1)] * (j_end - j_start + 1))
                c_idx.extend(range(j_start, j_end + 1))
                vals.extend(y_arr[k - j] for j in range(j_start, j_end + 1))

        if len(vals) == 0:
            break

        A = sp.coo_matrix((np.array(vals), (np.array(r_idx), np.array(c_idx))),
                          shape=(K, n)).tocsr()

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # Shrink active set
            ac_now = _fast_autocorr(y)
            active = set(int(k) for k in np.argsort(ac_now)[::-1][:max(500, K // 2)])
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64).ravel(), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check for new violations
        size = 2 * n - 1
        fft_sz = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_sz)
        Gf = np.fft.rfft(g_dir, fft_sz)
        conv_ys = np.fft.irfft(Yf * Gf, fft_sz)[:size]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                active.add(int(k))

        # Cap active set size
        if len(active) > K_max:
            ac_now = _fast_autocorr(y)
            # Keep the K_max most "important" (closest to peak)
            active = set(int(k) for k in np.argsort(ac_now)[::-1][:K_max])

        # Backtracking step size
        best_alpha = 2.0 / (t + 2.0)
        best_trial_m = _fast_autocorr((1 - best_alpha) * y + best_alpha * g_dir).max()
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            m_t = _fast_autocorr(trial).max()
            if m_t < best_trial_m:
                best_trial_m = m_t
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f} |S|={len(active)}")

    return best_a, best_m


def _local_search(a0, time_limit, verbose=True):
    n = len(a0)
    a = _norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    max_stall = 100

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = _fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:20]
        improved = False

        for _ in range(80):
            if time.monotonic() - t0 > time_limit:
                break
            k = int(top_peaks[rng.integers(len(top_peaks))])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])

            # Find a quiet region to move mass to
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src or a[src] < 1e-12:
                continue

            frac = float(rng.choice([0.002, 0.005, 0.01, 0.02, 0.05]))
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_t = _fast_autocorr(a_trial).max()
            if M_t < best_m - 1e-14:
                a = a_trial
                best_m = M_t
                best_a = a.copy()
                improved = True
                ac = _fast_autocorr(a)
                top_peaks = np.argsort(ac)[::-1][:20]

        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = rng.integers(1, 5)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.002, 0.02))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = _fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LocalSearch ({elapsed:.0f}s): J={J:.6f} stall={stall}")

    return best_a, best_m


def _cyclic_shift_best(a0):
    n = len(a0)
    best_m = _fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def _interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _init_shape(n, shape_type=0):
    x = np.linspace(0, 1, n)
    if shape_type == 0:
        f = np.sin(np.pi * x) ** 2
    elif shape_type == 1:
        f = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif shape_type == 2:
        f = np.exp(-4 * (x - 0.5) ** 2)
    elif shape_type == 3:
        f = 1.0 / (0.1 + x * (1 - x))
    elif shape_type == 4:
        f = np.sin(np.pi * x) ** 4
    else:
        f = np.ones(n)
    return _norm_sum1(f)


def _try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = _evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.1f}s)")
    return best_seq, best_score


def _build_sparse_for_active(y, active_set, n):
    """Build sparse constraint matrix for given active set."""
    K = len(active_set)
    if K == 0:
        return None
    rows = []
    cols = []
    vals = []
    for ii, k in enumerate(sorted(active_set)):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            rows.extend([ii] * (j_end - j_start + 1))
            cols.extend(range(j_start, j_end + 1))
            for j in range(j_start, j_end + 1):
                vals.append(y[k - j])
    if len(vals) == 0:
        return None
    A = sp.coo_matrix((np.array(vals, dtype=np.float64),
                       (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))),
                      shape=(K, n)).tocsr()
    return A


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1755.0)
    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = _evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = _norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6000
        best_seq = _init_shape(n_base, 0)
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # Determine candidate lengths
    n_parent = len(best_seq) if best_seq is not None else 6700
    candidate_lengths = set()
    candidate_lengths.add(n_parent)
    for delta in [-1000, -500, 500, 1000, 2000, 3000]:
        l = n_parent + delta
        if 3000 <= l <= 12000:
            candidate_lengths.add(l)
    # Always include some standard lengths
    for std_n in [5000, 6700, 8000]:
        if 3000 <= std_n <= 12000:
            candidate_lengths.add(std_n)
    candidate_lengths = sorted(candidate_lengths)

    # Limit to at most 4 lengths to have enough time per length
    if len(candidate_lengths) > 4:
        # Keep the ones closest to n_parent
        def dist(n):
            return abs(n - n_parent)
        candidate_lengths.sort(key=dist)
        candidate_lengths = candidate_lengths[:4]
        candidate_lengths = sorted(candidate_lengths)

    print(f"Candidate lengths: {candidate_lengths}")

    for n in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break

        print(f"\n=== Optimizing n={n} (remaining={remaining:.0f}s) ===")

        # Initialize
        if parent is not None and len(parent) == n:
            a = _norm_sum1(parent)
        elif best_seq is not None:
            a = _interpolate_seq(best_seq, n)
        else:
            a = _init_shape(n, 0)

        # Quick FISTA to establish good starting point
        t_fista = min(remaining * 0.15, 200.0)
        if t_fista > 30:
            a_out, _ = _fista_optimize(a, t_fista, beta_start=20, beta_end=500,
                                       top_k=15, restart_every=2000, verbose=False)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, f"fista_n{n}", t_global)

        # Main LP-FW optimization
        remaining = time_budget - (time.monotonic() - t_global)
        t_lp = min(remaining * 0.55, 600.0)
        if t_lp > 60 and n <= 10000:
            print(f"    LP-FW: {t_lp:.0f}s budget")
            a_out, _ = _lp_fw_optimize(_norm_sum1(best_seq if best_seq is not None else a),
                                       t_lp, solver, K_max=3000, verbose=True)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, f"lpfw_n{n}", t_global)

        # Cyclic shift
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            a_shifted, _ = _cyclic_shift_best(_norm_sum1(best_seq))
            best_seq, best_score = _try_update(best_seq, best_score, a_shifted, f"shift_n{n}", t_global)

        # FISTA polish with high beta
        remaining = time_budget - (time.monotonic() - t_global)
        t_f2 = min(remaining * 0.25, 250.0)
        if t_f2 > 40:
            a_out, _ = _fista_optimize(_norm_sum1(best_seq), t_f2, beta_start=80, beta_end=2000,
                                       top_k=10, restart_every=2000, verbose=False)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, f"fista2_n{n}", t_global)

        # Local search polish
        remaining = time_budget - (time.monotonic() - t_global)
        t_ls = min(remaining * 0.35, 150.0)
        if t_ls > 30:
            a_out, _ = _local_search(_norm_sum1(best_seq), t_ls, verbose=False)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, f"ls_n{n}", t_global)

    # Final: try a different shape to escape local optima
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 300 and best_seq is not None:
        print(f"\n=== Final: Different shape exploration (remaining={remaining:.0f}s) ===")
        n_use = len(best_seq)
        for shape in [1, 2, 3, 4]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            a_alt = _init_shape(n_use, shape)
            # Blend with current best
            blend = 0.5 * a_alt + 0.5 * best_seq
            blend = _norm_sum1(blend)
            t_blend = min(remaining * 0.3, 150.0)
            if t_blend > 20:
                a_out, _ = _fista_optimize(blend, t_blend, beta_start=30, beta_end=800,
                                           top_k=15, restart_every=2000, verbose=False)
                best_seq, best_score = _try_update(best_seq, best_score, a_out, f"blend{shape}", t_global)

    # Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and best_seq is not None:
        t_final = min(remaining * 0.6, 200.0)
        if t_final > 30:
            a_out, _ = _fista_optimize(_norm_sum1(best_seq), t_final, beta_start=150, beta_end=3000,
                                       top_k=8, restart_every=1500, verbose=True)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "final_fista", t_global)

    # Final cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        a_shifted, _ = _cyclic_shift_best(_norm_sum1(best_seq))
        best_seq, best_score = _try_update(best_seq, best_score, a_shifted, "final_shift", t_global)

    # Verify and return
    if best_seq is not None:
        final_score = _evaluate_safe(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:
        n_fb = 5000
        best_seq = _init_shape(n_fb, 0)
        best_score = _evaluate_safe(best_seq)
        print(f"\nFALLBACK: score={best_score:.6f}, n={n_fb}")

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
