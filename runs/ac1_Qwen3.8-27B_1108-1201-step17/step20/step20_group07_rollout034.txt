def fast_full_conv(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_cross_conv(a, b):
    """Full cross-convolution of a and b via FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def cross_corr_peaks(a, p):
    """
    Compute C[j] = sum_{m=0}^{n-1} p[j+m] * a[m] for j=0..n-1.
    Uses FFT: C = full_convolve(a[::-1], p)[n-1 : 2*n-1]
    """
    n = len(a)
    a_rev = a[::-1]
    conv = fast_cross_conv(a_rev, p)
    return conv[n - 1: 2 * n - 1]

def compute_gradient(a, beta):
    """
    Exact gradient of f(a) = (1/beta)*log(sum_k exp(beta*c[k])) where c = full_conv(a,a).
    Returns (grad, f_value, c, p).
    """
    n = len(a)
    c = fast_full_conv(a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    f_val = c_max + np.log(exp_vals.sum()) / beta

    # S[j] = sum_{k=j}^{j+n-1} p[k], j=0..n-1
    p_cum = np.concatenate([[0.0], np.cumsum(p)])
    S = p_cum[n:n + n] - p_cum[0:n]

    # C[j] = sum_{m=0}^{n-1} p[j+m] * a[m]
    C = cross_corr_peaks(a, p)

    grad = (C + a * S) / beta
    return grad, f_val, c, p

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def spectral_carving(a0, time_limit, n_iter=200, verbose=True):
    """
    Spectral domain peak reduction:
    Identify "constructive" frequencies for top peaks and reduce their magnitude.
    """

    best_a = a.copy()
    best_m = fast_full_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)

    N = 1 << (2 * n - 1).bit_length()  # FFT size >= 2n-1

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_full_conv(a)
        c_max = c.max()
        if c_max < best_m - 1e-14:
            best_m = c_max
            best_a = a.copy()

        # Get top peaks
        top_k = 5
        peak_indices = np.argsort(c)[::-1][:top_k]

        # Compute FFT
        F = np.fft.rfft(a, N)
        abs_F = np.abs(F).copy()
        phase_F = np.angle(F)

        # For each peak, identify constructive frequencies and slightly reduce them
        freq_weights = np.ones(N // 2 + 1)
        for ki in peak_indices:
            ki = int(ki)
            # c[k] = (1/N) * sum_m |F[m]|^2 * exp(2*pi*i*m*k/N)
            # Constructive when m*k mod N is close to 0 or N
            m_vals = np.arange(N // 2 + 1)
            phases = 2.0 * np.pi * m_vals * ki / N
            # Weight: reduce where cos(phase) > 0 (constructive)
            constructiveness = np.cos(phases)
            # Only reduce strongly constructive ones
            mask = constructiveness > 0.7
            freq_weights[mask] *= 0.98  # Reduce by 2%

        # Apply spectral modification
        abs_F_new = abs_F * freq_weights
        F_new = abs_F_new * np.exp(1j * phase_F)
        a_new = np.fft.irfft(F_new, N)[:n]

        # Project to simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        # Check improvement
        c_new = fast_full_conv(a_new)
        if c_new.max() < c_max:
            a = a_new
        else:
            # Blend
            for alpha in [0.3, 0.5, 0.7]:
                a_blend = (1 - alpha) * a + alpha * a_new

                if sb < 1e-15:

                a_blend = a_blend / sb
                c_blend = fast_full_conv(a_blend)
                if c_blend.max() < c_max:
                    a = a_blend
                    break

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("      [SPEC] it={} elapsed={:.1f}s J={:.6f}".format(it, elapsed, J))

    return best_a, best_m

def parametric_init(n, rng=None):

    x = np.linspace(0.0, 1.0, n)

    candidates = []

    # Power laws
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        for beta_p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            a_raw = (x + 0.01) ** alpha * (1.0 - x + 0.01) ** beta_p
            candidates.append(norm_sum1(a_raw))

    # Gaussian mixtures
    for c1 in [0.3, 0.4, 0.5]:
        for c2 in [0.6, 0.7, 0.8]:
            for s1 in [0.08, 0.12, 0.18]:
                a_raw = (np.exp(-((x - c1) ** 2) / (2 * s1 ** 2)) +
                         np.exp(-((x - c2) ** 2) / (2 * s1 ** 2)))
                candidates.append(norm_sum1(a_raw))

    # Multi-peak
    for npeaks in [3, 4, 5]:
        centers = np.linspace(0.2, 0.8, npeaks)
        a_raw = np.zeros(n)
        for c in centers:
            a_raw += np.exp(-((x - c) ** 2) / (2 * 0.08 ** 2))
        candidates.append(norm_sum1(a_raw))

    # Random smooth
    for _ in range(10):
        a_raw = np.abs(rng.normal(1.0, 0.5, n))
        # Smooth it
        kernel = np.ones(20) / 20
        a_raw = np.convolve(a_raw, kernel, 'same')
        candidates.append(norm_sum1(a_raw))

    return candidates

    if t7 > 30:
        a_out, _ = gradient_optimize(best_seq, t7, beta_start=100.0, beta_end=800.0, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "final_grad", t_global)

    # === Phase 8: Final peak search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.3, 100.0)
    if t8 > 20:
        a_out, _ = peak_local_search(best_seq, t8, top_k=8, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "final_peak", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.6f}, n={}, time={:.1f}s".format(best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 7000
        best_seq = parametric_init(n_fb, rng)[0]
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
