import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def simplex_project(z):
    """Project z onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_peak_grad(a, k):
    """Compute gradient of b[k] = sum_i a[i]*a[i+k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    k = int(k)
    if k == 0:
        g[:] = 2.0 * a
    else:
        # b[k] = sum_{i=0}^{n-1-k} a[i]*a[i+k]
        # d/db/d a[j] = a[j+k] (if j+k < n) + a[j-k] (if j-k >= 0)
        g[:n - k] += a[k:]
        g[k:] += a[:n - k]
    return g


def maybe_add_peak_grad(g, a, k, weight=1.0):
    """Add weighted gradient of b[k] to g in-place."""
    n = len(a)
    k = int(k)
    if k == 0:
        g[:] += weight * 2.0 * a
    else:
        g[:n - k] += weight * a[k:]
        g[k:] += weight * a[:n - k]


def spectral_init(n, seed=42):
    """Create an initial sequence with approximately flat power spectrum."""
    rng = np.random.default_rng(seed)
    # Start with a smooth "tent" shape
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    a = a + 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    a = normalize(a)
    return a


def spectral_refine(a, n_iters=200, alpha=0.05, seed=0):
    """Iteratively flatten the power spectrum in frequency domain."""
    rng = np.random.default_rng(seed)
    n = len(a)
    a = normalize(a.copy())
    fft_size = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    for it in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Target: mean power excluding DC
        if len(power) > 1:
            target = power[1:].mean()
        else:
            target = power[0]

        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.0:
                red = 1.0 - alpha * min(ratio - 1.0, 2.0)
                A_new[k] *= max(0.3, np.sqrt(max(red, 0.1)))
            elif ratio < 0.3:
                inc = 1.0 + alpha * 0.5 * (0.3 - ratio)
                A_new[k] *= min(2.0, np.sqrt(max(inc, 0.1)))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 50 == 0:
            # Random phase perturbation
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.05
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial

        if it % 100 == 0:
            alpha = max(0.01, alpha * 0.9)

    return best_a, best_m


def deflation_optimize(a0, time_limit, top_k_init=20, top_k_min=3,
                       verbose=True, seed=42):
    """
    Core optimizer: iteratively reduce top peaks of autocorrelation
    using exact subgradients with adaptive step size.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = 0.05
    eta_min = 1e-8
    top_k = top_k_init
    stall = 0
    spectral_counter = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        size = len(b)

        if max_b < best_m - 1e-15:
            best_m = max_b
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Find top K peaks
        if top_k >= size:
            idx = np.arange(size)
        else:
            idx = np.argsort(b)[::-1][:top_k]

        # Compute combined subgradient
        g = np.zeros(n)
        weights = b[idx]
        w_sum = weights.sum()
        if w_sum > 0:
            w_norm = weights / w_sum
        else:
            w_norm = np.ones(len(idx)) / len(idx)

        for i, k in enumerate(idx):
            maybe_add_peak_grad(g, a, int(k), float(w_norm[i]))

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            # Degenerate - add noise
            a = simplex_project(a + 0.01 * rng.standard_normal(n))
            continue

        # Backtracking line search
        best_trial_m = max_b
        best_eta = 0
        found = False
        eta_trial = eta

        for attempt in range(8):
            if eta_trial < eta_min:
                break
            a_try = simplex_project(a - eta_trial * g / gnorm * gnorm)
            # Actually: a - eta * g (g already has the right scale)
            a_try = simplex_project(a - eta_trial * g)
            m_try = fast_autocorr(a_try).max()
            if m_try < best_trial_m - 1e-15:
                best_trial_m = m_try
                best_eta = eta_trial
                found = True
                break
            eta_trial *= 0.5

        if found:
            a = simplex_project(a - best_eta * g)
            eta = min(eta * 1.05, 0.5)
        else:
            # No improvement - adapt
            if stall > 200:
                # Reset with perturbation
                strategy = rng.integers(4)
                if strategy == 0:
                    a = simplex_project(best_a + 0.02 * rng.standard_normal(n))
                elif strategy == 1:
                    # Scale up peaks, down valleys
                    b_mean = b.mean()
                    pert = 0.03 * (a - b_mean / max(fast_autocorr(a).max(), 1e-10))
                    a = simplex_project(a + pert)
                elif strategy == 2:
                    # Random subset scaling
                    mask = rng.random(n) > 0.5
                    a_new = a.copy()
                    a_new[mask] *= 0.8
                    a_new[~mask] *= 1.2
                    a = simplex_project(a_new)
                else:
                    # Spectral jump
                    a_spec, _ = spectral_refine(a, n_iters=20, alpha=0.1, seed=step % 9999)
                    a = a_spec

                eta = 0.05
                stall = 0
                top_k = min(top_k_init, top_k + 5)
            else:
                eta *= 0.7
                if eta < eta_min:
                    # Very stuck - small random step
                    a = simplex_project(a + 0.005 * rng.standard_normal(n))
                    eta = 0.02

        # Periodically adjust top_k
        if step % 500 == 0:
            top_k = max(top_k_min, min(top_k - 1, top_k_init))

        # Periodic spectral refinement
        spectral_counter += 1
        if spectral_counter >= 300 and time.monotonic() - t0 < time_limit - 10:
            spectral_counter = 0
            a_spec, m_spec = spectral_refine(a, n_iters=30, alpha=0.03, seed=step % 7777)
            m_a = fast_autocorr(a).max()
            if m_spec < m_a:
                a = a_spec
                if m_spec < best_m:
                    best_m = m_spec
                    best_a = a.copy()

        step += 1

        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Deflate step={} ({}s): J={} max_b={} top_k={}".format(
                step, round(elapsed, 1), repr(round(J, 6)),
                repr(round(best_m, 10)), top_k))

    return best_a, best_m

                    a_try = a.copy()
                    a_try[i_max] -= amt
                    for t_idx in targets:
                        a_try[t_idx] += amt / 5
                    a_try = np.maximum(a_try, 0)
                    s_t = a_try.sum()
                    if s_t > 1e-15:
                        a_try = a_try / s_t
                        if fast_autocorr(a_try).max() < max_b:
                            a = a_try
                            improved = True
                            break
            else:
                # Find the pair (i, i+k_star) with largest product
                n_terms = n - k_star
                products = a[:n_terms] * a[k_star:]
                pair_idx = int(np.argmax(products))
                # Reduce one of the pair
                for which in [0, 1]:
                    if which == 0:
                        src = pair_idx
                    else:
                        src = pair_idx + k_star
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    for frac in [0.01, 0.005]:
                        amt = a[src] * frac
                        a_try = a.copy()
                        a_try[src] -= amt
                        a_try[dst] += amt
                        a_try = np.maximum(a_try, 0)
                        s_t = a_try.sum()
                        if s_t > 1e-15:
                            a_try = a_try / s_t
                            if fast_autocorr(a_try).max() < max_b:
                                a = a_try
                                improved = True
                                break
                    if improved:
                        break

        if not improved:
            # Random exploration
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.01)
                a[i] -= a[i] * frac
                a[j] += a[i] * frac  # approximate
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s

        if verbose and step % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    CoordPolish ({}s): J={} stall={}".format(
                round(elapsed, 1), repr(round(J, 6)), stall))

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best if a is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {} (n={}, t={:.1f}s)".format(
            label, repr(round(sc, 6)), len(a), elapsed))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Load/create initial sequence
    2. Spectral initialization
    3. Deflation optimization (main workhorse)
    4. Spectral refinement
    5. Coordinate polish
    6. Multi-length exploration
    7. Final polish rounds
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = normalize(parent)
                best_score = p_score
                print("Parent: n={}, score={}".format(len(parent), repr(round(p_score, 6))))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        best_seq = spectral_init(n_base, seed=42)
        best_score = evaluate_safe(best_seq)
        print("Default init: n={}, score={}".format(len(best_seq), repr(round(best_score, 6))))

    n0 = len(best_seq)
    print("Starting: n={}, J={}".format(n0, repr(round(best_score, 6))))

    # Phase 1: Spectral refinement for good starting point
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Spectral init (n={}) ===".format(n0))
    t1 = min(remaining * 0.04, 60.0)
    if t1 > 10:
        a_out, _ = spectral_refine(best_seq, n_iters=1000, alpha=0.08, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec_init", t_global)

    # Phase 2: Main deflation optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P2: Deflation (n={}) ===".format(n_best))
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 50:
        a_out, _ = deflation_optimize(best_seq, t2, top_k_init=30, top_k_min=5,
                                      verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "defl1", t_global)

    # Phase 3: Spectral refinement on best
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P3: Spectral refine (n={}) ===".format(len(best_seq)))
    t3 = min(remaining * 0.08, 100.0)
    if t3 > 15:
        a_out, _ = spectral_refine(best_seq, n_iters=500, alpha=0.04, seed=777)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec2", t_global)

    # Phase 4: More deflation with different params
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P4: Deflation r2 (n={}) ===".format(len(best_seq)))
    t4 = min(remaining * 0.25, 350.0)
    if t4 > 50:
        a_out, _ = deflation_optimize(best_seq, t4, top_k_init=15, top_k_min=3,
                                      verbose=True, seed=123)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "defl2", t_global)

    # Phase 5: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P5: Coord polish (n={}) ===".format(len(best_seq)))
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 20:
        a_out, _ = coordinate_polish(best_seq, t5, verbose=True, seed=200)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord1", t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P6: Multi-length (base n={}) ===".format(n_best))

    candidate_lengths = []
    if n_best < 6000:
        candidate_lengths = [n_best + 500, n_best + 1000, n_best + 2000]
    else:
        candidate_lengths = [n_best + 500, n_best + 1500]
    # Also try slightly smaller
    if n_best > 4000:
        candidate_lengths.append(max(3000, n_best - 1000))
    candidate_lengths = [l for l in candidate_lengths if 2000 <= l <= 15000]
    # Remove duplicates and sort
    candidate_lengths = sorted(set(candidate_lengths))

    for new_n in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print("  -> n={}".format(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.30, 200.0)
        if t6a > 30:
            a_out, _ = deflation_optimize(a_long, t6a, top_k_init=25, top_k_min=4,
                                          verbose=False, seed=new_n % 9973)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "defl_n{}".format(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t6b = min(remaining * 0.25, 100.0)
            if t6b > 20:
                a_out, _ = spectral_refine(best_seq, n_iters=min(300, int(t6b * 10)),
                                           alpha=0.05, seed=new_n % 3333)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "spec_n{}".format(new_n), t_global)

    # Phase 7: Final deflation polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print("\n=== P7: Final deflation (n={}) ===".format(n_final))
    t7 = min(remaining * 0.25, 250.0)
    if t7 > 30:
        a_out, _ = deflation_optimize(best_seq, t7, top_k_init=10, top_k_min=2,
                                      verbose=True, seed=555)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "defl_final", t_global)

    # Phase 8: Final spectral + coordinate
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t8a = min(remaining * 0.3, 80.0)
        if t8a > 10:
            a_out, _ = spectral_refine(best_seq, n_iters=200, alpha=0.03, seed=888)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "spec_final", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8b = min(remaining * 0.4, 80.0)
            if t8b > 10:
                a_out, _ = coordinate_polish(best_seq, t8b, verbose=False, seed=600)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Phase 9: One more deflation with very focused peaks
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t9 = min(remaining * 0.5, 100.0)
        if t9 > 10:
            a_out, _ = deflation_optimize(best_seq, t9, top_k_init=5, top_k_min=1,
                                          verbose=True, seed=999)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "defl_ultra", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: J={}, n={}, time={:.1f}s".format(
            repr(round(best_score, 6)), len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        best_seq = spectral_init(n_fb, seed=0)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
