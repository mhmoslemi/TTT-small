def simplex_project(z):
    n = len(z)
    if n == 0:

    rho = rho[-1]

        else:
            step = max(step * 0.5, 1e-10 / n)
        
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = x_new.copy()
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print("    FISTA-exact it={}: J={} (t={:.1f}s)".format(it, repr(round(J, 6)), elapsed))
    
    return best_a, best_M

def peak_directed_search(a0, time_limit, verbose=True):
    """Local search that specifically targets the peaks in the convolution."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        b = fast_conv(a)
        b_max = b.max()
        # Find top peaks
        top_peaks = np.argsort(b)[::-1][:20]
        improved = False
        
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick a peak to reduce
            k = int(top_peaks[rng.integers(len(top_peaks))])
            # Positions that contribute to this peak: i and k-i for valid i
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            
            if i_min > i_max:
                continue
            
            # Find the position with the largest contribution
            js = np.arange(i_min, i_max + 1)
            contribs = a[js] * a[k - js]
            src = int(js[np.argmax(contribs)])
            
            if a[src] < 1e-12:
                continue
            
            # Find a destination that doesn't contribute to any top peak
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2+1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src:
                continue
            
            # Try moving a small fraction
            for frac in [0.001, 0.003, 0.01, 0.03, 0.1]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_M - 1e-14:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    b = fast_conv(a)
                    top_peaks = np.argsort(b)[::-1][:20]
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.001, 0.05))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_conv(a).max()
                        if M_new < best_M:
                            best_M = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print("    Peak-directed ({:.1f}s): J={} stall={}".format(elapsed, repr(round(J, 6)), stall))
    
    return best_a, best_M

def freq_phase_optimize(a0, time_limit, verbose=True):
    """Optimize phase in frequency domain to reduce convolution peaks."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute FFT
        A = np.fft.rfft(a, fft_size)
        # Power spectrum
        power = np.abs(A) ** 2
        # Time domain convolution = IFFT(power)
        b = np.fft.irfft(power, fft_size)[:2*n-1]
        b_max = b.max()
        
        # Try random phase perturbations
        best_phase = None
        best_M_trial = best_M
        
        for _ in range(20):
            # Random smooth phase perturbation
            phase = np.angle(A)
            # Add a random smooth perturbation
            freqs = np.arange(len(phase))
            noise = rng.normal(0, 1, len(phase))
            # Low-pass filter the noise for smoothness
            from scipy.ndimage import gaussian_filter1d
            noise_smooth = gaussian_filter1d(noise, sigma=5)
            amp_scale = float(rng.uniform(0.01, 0.2))
            phase_new = phase + amp_scale * noise_smooth
            
            A_new = np.abs(A) * np.exp(1j * phase_new)
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            M_new = fast_conv(a_new).max()
            if M_new < best_M_trial:
                best_M_trial = M_new
                best_phase_a = a_new.copy()
        
        if best_phase_a is not None:
            a = best_phase_a
            if best_M_trial < best_M:
                best_M = best_M_trial
                best_a = a.copy()
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print("    Freq-phase it={}: J={} (t={:.1f}s)".format(it, repr(round(J, 6)), elapsed))
    
    return best_a, best_M

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2000, verbose=True):
    """LP-Frank-Wolfe method."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    active_set = set(int(k) for k in np.argsort(fast_conv(y))[::-1][:initial_K])
    no_improve = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 500:
            break
        
        ac = fast_conv(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = set(int(k) for k in np.argsort(fast_conv(y))[::-1][:initial_K // 2])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Line search
        default_alpha = 2.0 / (t + 2.0)
        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in [0.01, 0.02, 0.05, 0.1, default_alpha, 0.5, 1.0]:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_conv(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y = y_new / s_y
        
        # Update active set
        conv_ys = fast_conv(y)
        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        for k in violations[:30]:
            active_set.add(int(k))
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter={}: J={} |S|={} (t={:.1f}s)".format(t, repr(round(J, 6)), len(active_set), elapsed))
    
    return best_a, best_m

def cyclic_shift_best(a0):
    n = len(a0)
    best_m = fast_conv(a0).max()
    best_shift = 0
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        m = fast_conv(a_shift).max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {} (n={}, t={:.1f}s)".format(label, repr(sc), len(a), elapsed))
    return best_seq, best_score
