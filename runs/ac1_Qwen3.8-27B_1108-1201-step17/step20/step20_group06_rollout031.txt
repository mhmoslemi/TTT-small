    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def simplex_project(z):
    """Project onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_subgradient(a, active_indices):
    """Compute exact subgradient of max(conv(a,a)) at given active indices."""
    n = len(a)
    size = 2 * n - 1
    w = np.zeros(size)
    w[active_indices] = 1.0 / len(active_indices)
    
    fft_size = 1 << (2 * size - 2).bit_length()
    a_rev = a[::-1]
    Ar = np.fft.rfft(a_rev, fft_size)
    Wr = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(Ar * Wr, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]

    """Compute smoothed subgradient using log-sum-exp or top-k/margin selection."""
    n = len(a)
    b = fast_autocorr(a)
    size = 2 * n - 1
    b_max = b.max()
    
    if margin is not None:

    elif top_k is not None and top_k < size:
        active = np.argsort(b)[::-1][:top_k]
    else:
        active = np.arange(size)

        return np.zeros(n), b
    
    w = np.zeros(size)

    b_sel = b[active] - b_max
    b_sel = np.clip(b_sel * beta, -500, 0)
    exp_vals = np.exp(b_sel)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(len(active)) / len(active)
    else:
        p = exp_vals / s
