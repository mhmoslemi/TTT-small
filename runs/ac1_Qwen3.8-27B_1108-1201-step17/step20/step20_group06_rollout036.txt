def fast_autocorr(a):
    """Compute full self-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(z):
    """Project onto the probability simplex (non-negative, sum=1)."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    A = np.fft.rfft(a, fft_size)
    power = np.abs(A) ** 2
    
    # Flatten top peaks (excluding DC)
    if len(power) > n_peaks_to_flatten + 1:
        idx = np.argsort(power)[::-1][:n_peaks_to_flatten]
        idx = idx[idx > 0]  # exclude DC
        if len(idx) > 0:
            mean_power = np.median(power[1:])
            for k in idx:
                target = np.sqrt(max(mean_power / max(power[k], 1e-30), 0.1))
                A[k] *= np.sqrt(alpha + (1 - alpha) * target)
    
    a_new = np.fft.irfft(A, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def nesterov_opt(a0, time_limit, beta_start=30.0, beta_end=1000.0,
                 beta_growth=1.004, eta_init=0.5, top_k_ratio=0.02,
                 restart_every=2000, verbose=True, seed=42,
                 spectral_snap_every=800):
    """
    Nesterov-accelerated gradient descent with:
    - Top-k focused smooth max
    - Backtracking line search
    - Periodic spectral snap
    - Diverse restarts
    """
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    eta = eta_init

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, b = compute_smoothed_grad(y, beta, top_k)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            x_new = x.copy()
            max_new = best_max_b
        else:
            # Backtracking line search
            eta_trial = eta
            found = False
            y_max = fast_autocorr(y).max()
            for _ in range(8):
                z = y - eta_trial * grad
                x_new = simplex_project(z)
                b_new = fast_autocorr(x_new)
                max_new = b_new.max()
                if max_new <= y_max + 1e-14:
                    found = True
                    break
                eta_trial *= 0.5
            if not found:
                x_new = x.copy()
                max_new = best_max_b
                eta = max(eta_trial, 0.01)

        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.05, 3.0)
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 40 == 0:
            beta = min(beta * beta_growth, beta_end)

        # Periodic spectral snap
        if spectral_snap_every > 0 and t % spectral_snap_every == 0 and stall > 100:
            x_snapped = spectral_snap(x, alpha=0.2)
            m_snapped = fast_autocorr(x_snapped).max()
            if m_snapped < best_max_b:
                best_max_b = m_snapped
                best_a = x_snapped.copy()
                x = x_snapped.copy()
                x_prev = x_snapped.copy()
                t_nest = 1.0
                stall = 0

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = rng.integers(6)
            if strategy == 0:
                noise_level = 0.01 + 0.02 * rng.random()
                pert = rng.normal(0, noise_level, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                # Blend with structured random
                r = rng.dirichlet(np.ones(n) * 0.2)
                x_r = 0.7 * best_a + 0.3 * r
            elif strategy == 2:
                # Multiplicative perturbation
                x_r = best_a * np.exp(0.04 * rng.standard_normal(n))
            elif strategy == 3:
                # Flatten top peaks
                ac = fast_autocorr(best_a)
                top = np.argsort(ac)[::-1][:10]
                x_r = best_a.copy()
                for k in top:
                    k_int = int(k)
                    js = max(0, k_int - n + 1)
                    je = min(n - 1, k_int)
                    if js <= je:
                        x_r[js:je+1] *= 0.9
            elif strategy == 4:
                # Shift
                shift = int(rng.integers(1, max(2, n // 20)))
                x_r = np.roll(best_a, shift)
            else:
                # Spectral snap + noise
                x_r = spectral_snap(best_a, alpha=0.3)
                x_r += 0.01 * rng.standard_normal(n) * best_a
            
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 15.0)
            eta = eta_init

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("      Nest t={} ({}s): J={:.6f} b={}".format(t, round(elapsed,1), J, int(beta)))

    return best_a, best_max_b


def block_cd(a0, time_limit, block_size=12, max_rounds=50, verbose=True):
    """
    Block coordinate descent: for each block of coordinates,
    optimize using SLSQP with the local objective.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            B = end - start
            
            # Get current contribution
            # The objective is max_k (a*a)[k], which is complex to express per-block.
            # Instead, use the smooth max as a surrogate.
            # For the block, we optimize the sum of top autocorrelation values.
            
            ac_full = fast_autocorr(a)
            top_k_idx = np.argsort(ac_full)[::-1][:20]
            
            # Define local objective: sum of top autocorrelation values
            def local_obj(block_vals):
                trial = a.copy()
                trial[start:end] = np.maximum(block_vals, 0)
                s = trial.sum()
                if s < 1e-15:
                    return 1e10
                trial = trial / s
                ac = fast_autocorr(trial)
                return ac[top_k_idx].sum()
            
            # Initial block values
            x0 = a[start:end].copy()
            
            # Bounds: each value in [0, 1] (roughly), sum constraint handled by normalization
            bounds = [(0, 1.0)] * B
            # Add constraint: sum of block + sum of rest = 1
            rest_sum = a.sum() - a[start:end].sum()
            constraints = [{'type': 'eq', 'fun': lambda x, rs=rest_sum: np.sum(x) + rs - 1.0}]
            
            result = scipy_minimize(local_obj, x0, method='SLSQP', bounds=bounds,
                                    constraints=constraints, options={'maxiter': 30, 'ftol': 1e-10})
            
            if result.success and result.fun < local_obj(x0) - 1e-12:
                a_new = a.copy()
                a_new[start:end] = np.maximum(result.x, 0)
                s = a_new.sum()
                if s > 1e-15:
                    a_new = a_new / s
                    m_new = fast_autocorr(a_new).max()
                    if m_new < best_m - 1e-15:
                        a = a_new.copy()
                        best_m = m_new
                        best_a = a_new.copy()
                        improved = True
        
        if not improved and rnd > 0:
            break
        
        if verbose and rnd % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("      BlockCD rnd={} ({}s): J={:.6f}".format(rnd, round(elapsed,1), J))

    return best_a, best_m


def peak_targeting(a0, time_limit, verbose=True, seed=789):
    """
    Directly target the top peaks: for each peak at lag k,
    identify contributing pairs and move mass to reduce the peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        size = 2 * n - 1
        top_peaks = np.argsort(ac)[::-1][:8]
        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            # Get contributions to this peak
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k_int - idx]
            total_contrib = contribs.sum()
            if total_contrib < 1e-15:
                continue

            # Find the pair with maximum contribution
            max_local = np.argmax(contribs)
            src = int(idx[max_local])
            partner = k_int - src
            
            if not (0 <= partner < n) or src == partner:
                continue

            # Try moving mass from the larger of src/partner to a random position
            # that doesn't contribute to this peak
            if a[src] >= a[partner]:
                donor = src
            else:
                donor = partner
            
            # Find positions that don't contribute to this peak
            non_contrib = [i for i in range(n) if i != donor and (i < js or i > je or (k_int - i) < 0 or (k_int - i) >= n)]
            if not non_contrib:
                non_contrib = [i for i in range(n) if i != donor]
            
            # Try several amounts and destinations
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                amt = a[donor] * frac
                if amt < 1e-15:
                    continue
                
                best_dest = None
                best_trial_m = best_m
                
                for _ in range(min(20, len(non_contrib))):
                    dst = non_contrib[int(rng.integers(len(non_contrib)))]
                    if dst == donor:
                        continue
                    a_trial = a.copy()
                    a_trial[donor] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_trial_m:
                        best_trial_m = m_trial
                        best_dest = dst
                
                if best_dest is not None:
                    a[donor] -= amt
                    a[best_dest] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    best_m = best_trial_m
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation when stalled
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    m_new = fast_autocorr(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("      PeakTarget ({}s): J={:.6f} stall={}".format(round(elapsed,1), J, stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence if candidate is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {:.6f} (n={}, t={:.1f}s)".format(label, sc, len(a), elapsed))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Multi-strategy optimization for minimizing autocorrelation peak.
    """
    t_global = time.monotonic()

    solver = get_solver()
    print("Time budget: {:.0f}s".format(time_budget))

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n={}, score={:.6f}".format(len(parent), p_score))
    except Exception:
        parent = None

    if best_seq is None:
        # Try several initializations
        inits = []
        for n_try in [2000, 4000, 6000, 8000]:
            x = np.linspace(0, 1, n_try)
            seq = np.sin(np.pi * x) ** 2
            inits.append((norm_sum1(seq), n_try))
            seq2 = np.exp(-4 * (x - 0.5) ** 2)
            inits.append((norm_sum1(seq2), n_try))
        
        for seq, n_try in inits:
            sc = evaluate_safe(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
        
        print("Default init: n={}, score={:.6f}".format(len(best_seq), best_score))

    n0 = len(best_seq)
    print("Start: n={}, J={:.6f}".format(n0, best_score))

    # Phase 1: Quick Nesterov to establish baseline (120s)
    remaining = time_budget - (time.monotonic() - t_global)

        if 500 <= nn <= 15000:
            n_candidates.append(nn)
    
    # Remove duplicates and sort by proximity to n_best
    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - n_best))
    
    for new_n in n_candidates:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        if new_n == n_best:
            continue
        print("  -> n={}".format(new_n))

                                    beta_start=25.0, beta_end=400.0,

            best_seq, best_score = try_update(best_seq, best_score, a_out, "n={}".format(new_n), t_global)

    # Phase 3: Main Nesterov at best n (400s)

                                    verbose=True, seed=555)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)

    # Phase 7: Peak targeting round 2 (150s)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P7: Peak Targeting r2 (n={}) ===".format(len(best_seq)))

                                    spectral_snap_every=400,
                                    verbose=True, seed=777)
