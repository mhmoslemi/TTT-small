def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_exact_gradient(y, beta, top_k=None):
    """Compute gradient of smoothed max of autocorrelation."""
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and 0 < top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    if b_max < 1e-15:
        return np.zeros(n), b, b_max
    shifted = beta * (b_use / b_max - 1.0)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b_use)
        p[b_max == b_use] = 1.0 / max(1, (b_max == b_use).sum())
    else:
        p = exp_vals / p_total
    # grad_m = sum_k p_k * 2*y[k-m] (for valid k-m)
    # This is: 2 * convolve(p, y_reversed)[n-1 : 2n-1]
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    # Add diagonal correction for k=2m terms
    for m in range(n):
        k = 2 * m
        if k < len(p):
            grad[m] += p[k] * y[m]
    return grad, p, b, b_max


def nesterov_opt(a0, time_limit, beta_start=10.0, beta_end=1500.0,
                 beta_schedule='exp', top_k_start=10, top_k_max=50,
                 restart_every=3000, verbose=True, seed=42):
    """Nesterov-accelerated proximal gradient with adaptive parameters."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    L = 2.0 * n  # Lipschitz constant upper bound for the smoothed function
    step = 1.0 / L
    fail_count = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Adaptive top_k
        frac = t / restart_every
        top_k = min(int(top_k_start + frac * (top_k_max - top_k_start)), top_k_max)

        grad, p, b, b_max = compute_exact_gradient(y, beta, top_k)

        # Backtracking line search
        local_step = step
        trial = y - local_step * grad
        trial = simplex_project(trial)
        trial_max = fast_autocorr(trial).max()

        # Simple adaptive: if it didn't improve, reduce step
        if trial_max < best_max_b:
            fail_count = 0
        else:
            fail_count += 1

        step = max(step * 0.5, 1e-10) if trial_max > best_max_b * 1.001 else \
               min(step * 1.5, 5.0 / beta)

        z = y - step * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        # Adaptive beta
        if beta_schedule == 'exp':
            if t % 50 == 0:
                beta = min(beta * 1.004, beta_end)
        elif beta_schedule == 'sqrt':
            beta = beta_start + (beta_end - beta_start) * np.sqrt(t / max(1, restart_every))
            beta = min(beta, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0

            if s_r > 1e-15:

            else:

            x = x_r
            x_prev = x_r.copy()

    return best_a, best_max_b


def q_peak_equalize(a0, time_limit, solver, n_top=30, verbose=True):
    """
    QP-based peak equalization: linearize the quadratic constraints
    and solve a QP to equalize the top peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    ac = fast_autocorr(a)

    for it in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()

        # Identify top peaks
        top_idx = np.argsort(ac)[::-1][:n_top]
        target = M_curr * 0.95  # Try to reduce peaks by 5%

        # Linearize: c[k] ≈ ac_k + grad_k^T (a - a_current)
        # Constraint: ac_k + grad_k^T (a - a_current) <= target
        # => grad_k^T a <= target - ac_k + grad_k^T a_current

        a_var = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(a_var) == 1]

        feasible = True
        for k in top_idx:
            ki = int(k)
            # grad of c[ki] w.r.t. a at current point
            grad_k = np.zeros(n)
            for m in range(n):
                j2 = ki - m
                if 0 <= j2 < n:
                    if ki == 2 * m:
                        grad_k[m] += a[j2]
                    else:
                        grad_k[m] += 2.0 * a[j2]

            rhs = target - ac[ki] + grad_k.dot(a)
            constraints.append(grad_k.dot(a_var) <= rhs)

        obj = cp.Minimize(cp.sum_squares(a_var - a))
        prob = cp.Problem(obj, constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Infeasible, try more relaxed target
            target = M_curr * 0.99
            del constraints[1:]
            for k in top_idx:
                ki = int(k)
                grad_k = np.zeros(n)
                for m in range(n):
                    j2 = ki - m
                    if 0 <= j2 < n:
                        if ki == 2 * m:
                            grad_k[m] += a[j2]
                        else:
                            grad_k[m] += 2.0 * a[j2]
                rhs = target - ac[ki] + grad_k.dot(a)
                constraints.append(grad_k.dot(a_var) <= rhs)
            prob = cp.Problem(obj, constraints)
            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new > M_curr * 1.01:
            # Solution worsened, stop
            break

        a = a_new

        if verbose and it % 10 == 0:

            J = 2.0 * n * best_m
            print("    QP-eq it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " M=" + repr(round(M_curr, 10)))

    return best_a, best_m


def spectral_equalize(a0, n_iters=100, decay=0.5, verbose=True):
    """
    Frequency-domain equalization: reduce DFT magnitude at frequencies
    corresponding to autocorrelation peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()

    for it in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        b_circ = np.fft.irfft(power, fft_size)[:2 * n - 1]
        b_max = b_circ.max()
        if b_max < 1e-15:
            break

        # Identify which frequency bins correspond to the peak
        # The autocorrelation peak at lag k corresponds to all frequencies
        # So we reduce frequency components that contribute to the peak
        # Use a soft threshold on the power spectrum
        b_norm = b_circ / b_max

        # Create a mask in frequency domain: reduce amplitudes where
        # the circular autocorrelation is high
        freq_idx = np.arange(len(A))
        # The contribution of frequency f to autocorrelation at lag k is
        # |A(f)|^2 * exp(-i*2*pi*f*k/N). We want to reduce components
        # that make b_circ[k] large at the peak locations.

        # Simple approach: reduce amplitudes proportional to local power
        # at frequencies where the phase aligns with the peak
        peak_lag = int(np.argmax(b_circ[:n]))  # Main peak in first half
        phases = np.angle(A)
        alignments = np.cos(2.0 * np.pi * freq_idx * peak_lag / fft_size - phases)
        # Reduce where alignment is positive (constructive at peak)
        reduction = np.maximum(0, alignments) * 0.3
        A_new = A * (1.0 - reduction * (power / max(power.max(), 1e-15)))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        m_new = fast_autocorr(a_new).max()

        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()

        # Blend with previous to avoid oscillation
        a = 0.7 * a + 0.3 * a_new
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a > 1e-15:
            a = a / s_a

        if verbose and it % 20 == 0:
            J = 2.0 * n * best_m
            print("    Spectral it=" + str(it) + ": J=" + repr(round(J, 6)))

    return best_a, best_m


def cyclic_shift_best(a0):
    """Find the best cyclic shift of the sequence."""
    n = len(a0)
    if n <= 1:
        return a0.copy(), fast_autocorr(a0).max()
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    # Only check a subset of shifts for efficiency
    if n > 2000:
        shifts = list(range(0, n, max(1, n // 200)))
        shifts.append(0)
    else:
        shifts = list(range(n))
    for s in shifts:
        a_shift = np.roll(a0, -s)
        m = fast_autocorr(a_shift).max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def diversify(a0, rng, strength=0.1):
    """Create a diversified version of a sequence."""
    n = len(a0)
    # Random multiplicative perturbation
    noise = rng normal(1.0, strength, n) if False else rng.normal(1.0, strength, n)
    a_new = a0 * noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def propose_candidate(budget_s=1800.0):
    """
    Main optimization: combines Nesterov gradient, QP peak equalization,
    spectral equalization, multi-scale exploration, and diverse restarts.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')
    rng = np.random.default_rng(2024)

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p.copy())
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        pass

    if best_seq is None:
        # Default initialization: sinusoidal
        n_base = 8000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print("  *** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")

    # === Phase 1: Nesterov at base size ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Nesterov (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 60:
        a_out, _ = nesterov_opt(best_seq, t1,
                                beta_start=10.0, beta_end=1000.0,
                                beta_schedule='exp', top_k_start=8,
                                top_k_max=40, restart_every=2500,
                                verbose=True, seed=1)
        try_update(a_out, "nesterov1")

    # === Phase 2: Cyclic shift ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: Cyclic shift ===")
    if remaining > 30:
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        try_update(a_shifted, "cyclic1")

    # === Phase 3: QP Peak Equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: QP Peak Equalization (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 30 and n_best <= 12000:
        a_out, _ = q_peak_equalize(norm_sum1(best_seq), t3, solver,
                                   n_top=20, verbose=True)
        try_update(a_out, "qp1")

    # === Phase 4: Spectral Equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 4: Spectral Equalization ===")
    if remaining > 60:
        a_out, _ = spectral_equalize(norm_sum1(best_seq), n_iters=80,
                                     decay=0.4, verbose=True)
        try_update(a_out, "spectral1")

    # === Phase 5: Nesterov v2 (different params) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Nesterov v2 (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 50:
        a_out, _ = nesterov_opt(best_seq, t5,
                                beta_start=50.0, beta_end=2000.0,
                                beta_schedule='sqrt', top_k_start=5,
                                top_k_max=25, restart_every=2000,
                                verbose=True, seed=99)
        try_update(a_out, "nesterov2")

    # === Phase 6: Multi-scale exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-scale exploration ===")
    scales = [n_best + 500, n_best + 1500, n_best + 3000]
    scales = [s for s in scales if s <= 15000 and s > n_best]

    for new_n in scales:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("  Scaling to n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        # Nesterov at new scale
        t6a = min(remaining * 0.25, 250.0)
        if t6a > 50:
            a_out, _ = nesterov_opt(a_long, t6a,
                                    beta_start=15.0, beta_end=800.0,
                                    beta_schedule='exp', top_k_start=8,
                                    top_k_max=30, restart_every=2000,
                                    verbose=True, seed=new_n % 100)
            try_update(a_out, "nesterov_n" + str(new_n))

        # Quick spectral at new scale
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            a_out, _ = spectral_equalize(norm_sum1(best_seq), n_iters=40,
                                         decay=0.3, verbose=False)
            a_out_interp = interpolate_sequence(a_out, new_n)
            try_update(a_out_interp, "spectral_n" + str(new_n))

        # Cyclic shift at new scale
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
            a_shifted_interp = interpolate_sequence(a_shifted, new_n)
            try_update(a_shifted_interp, "cyclic_n" + str(new_n))

    # === Phase 7: Diversified restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Diversified restarts ===")
    n_restarts = 3
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        diversified = diversify(norm_sum1(best_seq), rng, strength=0.05 + 0.03 * r)
        t7r = min(remaining * 0.20, 200.0)
        if t7r > 40:
            a_out, _ = nesterov_opt(diversified, t7r,
                                    beta_start=20.0, beta_end=800.0,
                                    beta_schedule='exp', top_k_start=10,
                                    top_k_max=35, restart_every=1500,
                                    verbose=False, seed=100 + r * 7)
            try_update(a_out, "diversify" + str(r))

    # === Phase 8: Final QP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final QP polish (n=" + str(n_best) + ") ===")
    if remaining > 60 and n_best <= 12000:
        t8 = min(remaining * 0.30, 200.0)
        if t8 > 30:
            a_out, _ = q_peak_equalize(norm_sum1(best_seq), t8, solver,
                                       n_top=15, verbose=True)
            try_update(a_out, "qp_final")

    # === Phase 9: Final Nesterov micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 9: Final micro-polish ===")
    if remaining > 40:
        t9 = min(remaining * 0.50, 150.0)
        if t9 > 20:
            a_out, _ = nesterov_opt(norm_sum1(best_seq), t9,
                                    beta_start=200.0, beta_end=3000.0,
                                    beta_schedule='exp', top_k_start=5,
                                    top_k_max=15, restart_every=1500,
                                    verbose=True, seed=777)
            try_update(a_out, "polish")

    # === Final: Cyclic shift ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        try_update(a_shifted, "cyclic_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 8000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
