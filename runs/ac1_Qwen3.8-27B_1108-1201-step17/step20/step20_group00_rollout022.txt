    # We want h[j] = sum_k indicator[k] * a[k-j] (for 0 <= k-j < n)
    # This is the cross-correlation of indicator and a.
    # crosscorr[m] = sum_j indicator[j+m] * a[j] = sum_j indicator[j] * a[j-m]
    # So h[j] = crosscorr of indicator and a at lag j... 
    # Actually: h[j] = sum_k indicator[k] * a[k-j]
    # Let m = k-j, then k = m+j: h[j] = sum_m indicator[m+j] * a[m]
    # This is the cross-correlation of a and indicator at lag j.
    # crosscorr(a, indicator)[j] = sum_m a[m] * indicator[m+j]
    # = sum_k a[k-j] * indicator[k]  (substituting m = k-j)
    # = h[j] ✓
    
    # To compute cross-correlation via FFT:
    # crosscorr(x, y) = ifft(conj(FFT(x)) * FFT(y))
    A = np.fft.rfft(a, n=fft_size)
    I = np.fft.rfft(indicator, n=fft_size)
    h = np.fft.irfft(np.conj(A) * I, n=fft_size)
    
    # h[j] for j = 0, ..., n-1 is our gradient (times 2/|S|)
    g = 2.0 * h[:n] / len(active_lags)
    
    # Add correction for k=2j terms (small)
    # For each k in active_lags, if k is even, j=k/2 gets an extra -a[j]/|S|
    for k in active_lags:
        if k % 2 == 0 and k // 2 < n:
            j = k // 2
            g[j] -= a[j] / len(active_lags)
    
    return g

def proj_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w
