def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)

def simplex_project(z):
    """Project z onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_to(f, new_n):
    """Interpolate sequence f to length new_n, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_grad_smoothed(y, beta, c=None, top_k=None):
    """
    Compute gradient of the smoothed max: (1/beta)*logsumexp(beta*conv)
    Gradient w.r.t. y.
    Returns (grad, p, c) where p is the softmax weights.
    """
    n = len(y)
    if c is None:
        c = fast_autocorr(y)
    
    if top_k is not None and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_top = np.zeros(len(c))
        c_top[idx] = c[idx]
        c_max = c_top.max()
    else:
        c_top = c
        c_max = c.max()
    
    if beta * c_max > 300:
        c_use = c_top - c_max
    else:
        c_use = c_top
    
    shifted = beta * c_use
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    
    # Gradient: (2/beta) * sum_k p[k] * a[k-j] (with a=0 outside)
    # = (2/beta) * (p * a_rev)[n-1+j] via FFT
    fft_size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A_rev = np.fft.rfft(y[::-1], fft_size)
    conv_pa = np.fft.irfft(P * A_rev, fft_size)
    grad = (2.0 / beta) * conv_pa[n - 1:2 * n - 1]
    
    return grad, p, c

                   beta_growth=1.005, eta_base=0.5, top_k_int=50,
                   restart_every=1500, verbose=True, seed=42, label=""):

    n = len(a0)
    x = np.maximum(a0, 0.0)
    s = x.sum()

    x = x / s
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        y = y / s_y
        
        c = fast_autocorr(y)
        grad, p, _ = compute_grad_smoothed(y, beta, c=c, top_k=top_k_int)
        
        # Adaptive step size
        grad_norm = np.maximum(np.abs(grad).max(), 1e-10)
        eta = eta_base / (beta * grad_norm + 1e-10)
        eta = min(eta, 0.5 / (beta * c.max() + 1e-10))
        eta = max(eta, 1e-8)
        
        z = y - eta * grad
        x_new = simplex_project(z)
        
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        if t % 30 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            # Structured restart
            scale = 0.02 / (1.0 + 0.1 * (t // restart_every))
            # Anti-symmetric perturbation
            anti = np.sin(2 * np.pi * np.arange(n) / n)
            # Random perturbation
            rand = rng.normal(0, 1, n)
            pert = scale * (0.7 * anti + 0.3 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b

    return best_a, best_max_b
