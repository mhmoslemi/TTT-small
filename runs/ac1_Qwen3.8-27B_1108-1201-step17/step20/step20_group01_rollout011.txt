Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated subgradient descent with softmax-smoothed max
2. Targeted peak-shaving via fine coordinate moves
3. Mirror descent with adaptive step sizes and momentum
4. Smoothing operators to reduce local oscillations
5. Multi-K exploration and periodic restarts

import numpy as np
import time
import math

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        for v in a_list:
            if v < 0 or not math.isfinite(v):
                return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:

def compute_grad_topk(x, top_lags, ac, K=None):
    """Compute gradient of sum of weighted top-K peaks."""
    n = len(x)
    K = K if K is not None else len(top_lags)
    top_k = top_lags[:K]
    grad = np.zeros(n, dtype=np.float64)
    for k in top_k:
        ki = int(k)
        w_val = ac[ki]
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            grad[idx] += 2.0 * w_val * x[ki - idx]
    return grad

def subgrad_descent(a0, time_limit, K=30, verbose=True):
    """Mirror descent with top-K peak gradient and adaptive step size."""
    n = len(a0)
    x = normalize(a0).copy()
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    no_improve = 0
    stall_limit = 3000
    momentum = np.zeros(n, dtype=np.float64)
    mu = 0.7

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break

        ac = fast_autocorr(x)
        cur_max = ac.max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val - 1e-13:
            best_val = cur_val
            best_a = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_k = np.argsort(ac)[::-1]
        k_use = min(K, len(top_k))

        # Temperature-smoothed peaks for gradient
        top_vals = ac[top_k[:k_use]]
        temp = 5.0
        w = np.exp(-temp * (top_vals[0] - top_vals))
        w = w / w.sum()

        grad = np.zeros(n, dtype=np.float64)
        for ii in range(k_use):
            ki = int(top_k[ii])
            w_val = w[ii]
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                grad[idx] += 2.0 * w_val * x[ki - idx]

        # Nesterov-like momentum
        momentum = mu * momentum + (1 - mu) * grad
        lr = 0.03 / (1.0 + 0.002 * t)
        log_x = np.log(x + 1e-30) - lr * momentum
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        x_new = np.maximum(x_new, 1e-25)
        x_new /= x_new.sum()

        new_val = 2.0 * n * fast_autocorr(x_new).max()
        if new_val > cur_val + 1e-10:
            # Backtrack
            for shrink in [0.5, 0.25, 0.1, 0.05]:
                log_x2 = np.log(x + 1e-30) - lr * shrink * momentum
                log_x2 -= log_x2.max()
                x2 = np.exp(log_x2)
                x2 = np.maximum(x2, 1e-25)
                x2 /= x2.sum()
                v2 = 2.0 * n * fast_autocorr(x2).max()
                if v2 < cur_val:
                    x_new = x2
                    break
            else:
                x_new = x.copy()
        x = x_new

        if verbose and t % 5000 == 0:
            print("  SubGrad t={} ({:.1f}s) J={:.8f} no_imp={}".format(
                t, time.monotonic() - t0, best_val, no_improve))

    return best_a, best_val

def coord_fine_tune(a0, time_limit, n_targets=8, verbose=True):
    """Fine-grained coordinate moves: shift tiny mass from peak contributors to quiet spots."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_val = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    fracs = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]

    while time.monotonic() - t0 < time_limit and stall < 8000:
        ac = fast_autocorr(a)
        cur_max = ac.max()
        if cur_max < best_m - 1e-15:
            best_m = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        n_top = min(n_targets, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            if j_start > j_end:
                continue
            idx = np.arange(j_start, j_end + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-28:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-18:
                continue

            # Find quiet positions (low pressure)
            pressure = np.zeros(n, dtype=np.float64)
            for kk in top_lags[:5]:
                kki = int(kk)
                ps = max(0, kki - n + 1)
                pe = min(n - 1, kki)
                if ps <= pe:
                    ii = np.arange(ps, pe + 1)
                    pressure[ii] += a[ii] * a[kki - ii]
            pressure[src] = 1e10

            n_quiet = min(20, n)
            quiet_pos = np.argsort(pressure)[:n_quiet]

            for dst in quiet_pos:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-20:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t /= st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-16:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

            if not improved and stall > 200:
                # Random move to escape
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-12:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a /= st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
                            improved = True
                if not improved:
                    break

        if not improved:
            # Try a random pair move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.001, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a /= st
                    Mn = fast_autocorr(a).max()
                    if Mn < best_m:
                        best_m = Mn
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 2000 == 0 and stall > 0:
            print("  Coord ({:.1f}s) J={:.8f} stall={}".format(
                time.monotonic() - t0, 2.0 * n * best_m, stall))

    return best_a, 2.0 * n * best_m

def smooth_step(a, alpha=0.1, width=3):
    """Apply mild Gaussian-like smoothing."""
    n = len(a)
    kernel = np.ones(width, dtype=np.float64) / width
    a_smooth = np.convolve(a, kernel, mode='same')
    a_smooth = np.maximum(a_smooth, 0.0)
    s = a_smooth.sum()
    if s < 1e-15:
        return a.copy()
    a_smooth /= s
    return (1 - alpha) * a + alpha * a_smooth

def random_perturb_search(a0, time_limit, verbose=True):

    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent: n={} score={}".format(len(p), repr(best_score)))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_init = 10000
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)
        print("Default init n={} score={}".format(n_init, repr(best_score)))

    n0 = len(best_seq)

    # Phase 1: Subgradient descent with K=30
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: SubGrad K=30 (n={}) ===".format(n0))
    t1 = min(remaining * 0.20, 350.0)
    if t1 > 30:
        a_out, v_out = subgrad_descent(best_seq, t1, K=30, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P1: {:.8f}".format(best_score))

    # Phase 2: Fine coordinate tuning
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P2: CoordFine (n={}) ===".format(len(best_seq)))
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 30:
        a_out, v_out = coord_fine_tune(best_seq, t2, n_targets=8, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P2: {:.8f}".format(best_score))

    # Phase 3: Multi-K sweep
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P3: MultiK (n={}) ===".format(len(best_seq)))
    t3 = min(remaining * 0.20, 340.0)
    if t3 > 30:
        a_out, v_out = multi_k_sweep(best_seq, t3, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P3: {:.8f}".format(best_score))

    # Phase 4: Smooth + re-optimize
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P4: Smooth+SubGrad (n={}) ===".format(len(best_seq)))
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_smooth = smooth_step(best_seq, alpha=0.05, width=5)
        a_out, v_out = subgrad_descent(a_smooth, t4 * 0.6, K=50, verbose=False)
        sc1 = eval_safe(a_out)
        a_out2, v_out2 = coord_fine_tune(a_out, t4 * 0.4, n_targets=10, verbose=False)
        sc2 = eval_safe(a_out2)
        if np.isfinite(min(sc1, sc2)) and min(sc1, sc2) < best_score:
            best_score = min(sc1, sc2)
            best_seq = a_out if sc1 < sc2 else a_out2
            print("  P4: {:.8f}".format(best_score))

    # Phase 5: Another round of subgradient with different K
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P5: SubGrad K=100 (n={}) ===".format(len(best_seq)))
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 20:
        a_out, v_out = subgrad_descent(best_seq, t5, K=100, verbose=False)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P5: {:.8f}".format(best_score))

    # Phase 6: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P6: Final Coord (n={}) ===".format(len(best_seq)))
    t6 = min(remaining * 0.15, 250.0)
    if t6 > 20:
        a_out, v_out = coord_fine_tune(best_seq, t6, n_targets=12, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P6: {:.8f}".format(best_score))

    # Phase 7: Final subgradient polish
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P7: Final SubGrad (n={}) ===".format(len(best_seq)))
    t7 = min(remaining * 0.15, 250.0)
    if t7 > 15:
        a_out, v_out = subgrad_descent(best_seq, t7, K=20, verbose=False)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P7: {:.8f}".format(best_score))

    # Phase 8: Try slightly different length
    remaining = time_budget - (time.monotonic() - t_global)
    n_try = len(best_seq) + 500
    if remaining > 60 and n_try <= 20000:
        print("\n=== P8: Try n={} ===".format(n_try))
        t8 = min(remaining * 0.3, 300.0)
        if t8 > 30:
            a_long = interpolate_seq(best_seq, n_try)
            a_out, v_out = subgrad_descent(a_long, t8 * 0.5, K=30, verbose=False)
            a_out2, v_out2 = coord_fine_tune(a_out, t8 * 0.5, n_targets=8, verbose=False)
            sc_long = eval_safe(a_out)
            sc_long2 = eval_safe(a_out2)
            sc_short = best_score
            if np.isfinite(min(sc_long, sc_long2)) and min(sc_long, sc_long2) < best_score:
                best_score = min(sc_long, sc_long2)
                best_seq = a_out if sc_long < sc_long2 else a_out2
                print("  P8 (longer): {:.8f}".format(best_score))
            # Also try shorter
            n_short = len(best_seq) - 300
            if n_short >= 5000:
                a_short = interpolate_seq(best_seq, n_short)
                a_out3, v_out3 = subgrad_descent(a_short, 100.0, K=30, verbose=False)
                sc_short = eval_safe(a_out3)
                if np.isfinite(sc_short) and sc_short < best_score:
                    best_score = sc_short
                    best_seq = a_out3
                    print("  P8 (shorter): {:.8f}".format(best_score))

    # Final: one more coordinate pass with remaining time
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print("\n=== P9: Final polish (n={}) ===".format(len(best_seq)))
        a_out, v_out = coord_fine_tune(best_seq, remaining * 0.8, n_targets=10, verbose=False)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P9: {:.8f}".format(best_score))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        # Ensure all values are valid
        best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
        sa = best_seq.sum()
        if sa < 0.01:
            best_seq = best_seq * (1.0 / sa) if sa > 0 else np.ones(len(best_seq)) / len(best_seq)
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={} n={} t={:.1f}s".format(
            repr(best_score), len(best_seq), elapsed))
    else:
        print("\nERROR: using fallback")
        n_fb = 10000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)

    return [float(x) for x in best_seq]
