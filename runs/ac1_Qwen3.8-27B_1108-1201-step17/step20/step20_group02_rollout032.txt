Let me also add a key novel component: **the "spectral shaping" step**.

The idea is:
1. Compute the DFT of a: `A(ω) = DFT(a)`
2. The power spectrum is `P(ω) = |A(ω)|^2`
3. The autocorrelation is the inverse DFT of P: `c = IDFT(P)`
4. We want `c` to be flat, which means `P` should be... wait, that's circular.

Actually, the relationship is: `c_k` is the inverse DFT of `|A(ω)|^2` (up to scaling). So to make `c` flat, we want `|A(ω)|^2` to be as close to constant as possible (i.e., a "flat" power spectrum).

But `a >= 0` constrains what power spectra are achievable.

**Spectral shaping algorithm**:
1. Compute `A(ω) = DFT(a)`
2. Target power spectrum: `P_target(ω) = mean(|A(ω)|^2)` (flat)
3. Adjust `a` to move its power spectrum closer to the target:
   - In the frequency domain: `A_new(ω) = A(ω) * sqrt(P_target(ω) / |A(ω)|^2)` (normalize the magnitude)
   - But this might make `a` have negative entries.
   - So: `a_new = IDFT(A_new)`, then `a_new = max(a_new, 0)`, then normalize.
4. Repeat.

4. Reduce the magnitude at high-excess frequencies and increase at low-excess frequencies.

def norm_sum1(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_constraint_matrix_full(y, n):
    """Build the full constraint matrix for all lags."""
    K = 2 * n - 1
    from scipy.sparse import lil_matrix
    C = lil_matrix((K, n), dtype=np.float64)
    for k in range(K):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            C[k, idx] = y[k - idx]
    return C.tocsr()


def build_constraint_matrix_active(y, active_set, n):
    """Build constraint matrix for active set of lags."""
    K = len(active_set)
    from scipy.sparse import lil_matrix
    C = lil_matrix((K, n), dtype=np.float64)
    for jj in range(K):
        k_int = int(active_set[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            C[jj, idx] = y[k_int - idx]
    return C.tocsr()

def spectral_shaping(a0, time_limit, n_iters=2000, verbose=False):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_convolve(y).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute power spectrum
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        P = np.abs(Yf) ** 2

        # Target: flatten power spectrum
        P_mean = P.mean()
        # Soft flatten: reduce peaks, boost valleys
        # Use a "soft clip" approach
        P_ratio = P / (P_mean + 1e-30)
        # Damp frequencies above target, boost below
        alpha_flat = 0.3
        P_adjusted = P * (1.0 + alpha_flat * (1.0 - P_ratio))
        P_adjusted = np.maximum(P_adjusted, 1e-30)

        # Adjust magnitude of Yf
        mag_new = np.sqrt(P_adjusted)
        phase = np.angle(Yf)
        Yf_new = mag_new * np.exp(1j * phase)

        # Invert
        y_new = np.fft.irfft(Yf_new, fft_size)[:n]
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_new

        M_new = fast_convolve(y_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = y_new.copy()

        y = y_new

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    Spectral iter=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    Spectral done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))
    return best_a, best_m


def soft_peak_descent(a0, time_limit, beta_start=5.0, beta_end=50.0, verbose=False):
    """Mirror descent on log-sum-exp approximation of max autocorrelation."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_convolve(y).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    total_iters = 50000
    stall = 0

    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 3000:
            break

        beta = beta_start + (beta_end - beta_start) * min(1.0, t / total_iters)
        eta = 0.05 / (1.0 + t / 1000.0)

        ac = fast_convolve(y)
        ac_max = ac.max()

        # Soft weights: exp(beta * (c_k - max)) to focus on peaks
        weights = np.exp(beta * (ac - ac_max))
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights = np.zeros_like(ac)
            weights[np.argmax(ac)] = 1.0
            w_sum = 1.0
        weights = weights / w_sum

        # Gradient: d/d a_i of sum_k w_k * c_k = sum_k w_k * (a_{k-i} + a_{i-k+n-1})
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if weights[k] < 1e-10:
                continue
            wk = weights[k]
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            grad[idx] += wk * y[k - idx]

        # Small random perturbation for exploration
        grad += 0.005 * rng.standard_normal(n) * y

        # Mirror descent (multiplicative update)
        log_a = np.log(np.maximum(y, 1e-30)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_convolve(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        y = a_new

        # Restart if stalled
        if stall > 500 and stall % 500 == 0:
            noise = rng.normal(0, 0.05, n) * best_a
            y_p = best_a + noise
            y_p = np.maximum(y_p, 1e-10)
            sp = y_p.sum()
            if sp > 1e-12:
                y = y_p / sp
            stall = 0

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print("    SoftPeak t=" + str(t) + " beta=" + str(round(beta, 1)) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    SoftPeak done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=77, verbose=False):
    """Peak-aware coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score all positions by their contribution to top peaks
            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 5:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])

            if dst == src:
                continue

            found = False
            for frac in [0.003, 0.008, 0.015, 0.03, 0.06]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and rng.random() < 0.3:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.003, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def get_initial_sequences(t_global):
    """Get multiple starting sequences."""
    sequences = []

    # Try parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
                sequences.append((p_norm, sc, "parent"))
    except Exception:
        pass

    # Sine-based
    n = 5000
    x = np.linspace(0, 1, n)
    for p in [2, 3, 4]:
        seq = norm_sum1(np.sin(np.pi * x) ** p)
        sc = evaluate_safe(seq)
        sequences.append((seq, sc, "sin_p" + str(p)))

    # Cosine modulated
    seq = norm_sum1(0.5 + 0.4 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x))
    sc = evaluate_safe(seq)
    sequences.append((seq, sc, "cos_mod"))

    # Power law edges
    seq = norm_sum1(x * (1 - x) ** 0.5)
    sc = evaluate_safe(seq)
    sequences.append((seq, sc, "power_edge"))

    # Broad hat
    seq = norm_sum1(1.0 - np.abs(2 * x - 1) ** 3)
    sc = evaluate_safe(seq)
    sequences.append((seq, sc, "hat"))

    # Random smooth
    rng = np.random.default_rng(123)
    raw = rng.standard_normal(n)
    # Smooth with a wide kernel
    kernel = np.exp(-0.5 * (np.arange(-200, 201) / 50.0) ** 2)
    kernel = kernel / kernel.sum()
    smooth = np.convolve(raw, kernel, mode='same')
    seq = norm_sum1(smooth)
    sc = evaluate_safe(seq)
    sequences.append((seq, sc, "random_smooth"))

    # Sort by score
    sequences.sort(key=lambda x: x[1])
    print("Best initial: " + sequences[0][2] + " score=" + repr(sequences[0][1]))
    return sequences


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = get_solver()

    # Phase 0: Get initial sequences
    print("\n=== Phase 0: Initialization ===")
    sequences = get_initial_sequences(t_global)
    best_seq, best_score = sequences[0][0].copy(), sequences[0][1]
    n_best = len(best_seq)

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # Phase 1: LP-FW on best initial (long)
    print("\n=== Phase 1: LP-FW initial (n=" + str(n_best) + ") ===")
    t1 = min(remaining_time() * 0.20, 320.0)
    if t1 > 50 and n_best <= 10000:
        a_out, _ = lp_fw_full(norm_sum1(best_seq), t1, solver,
                              max_active=min(2 * n_best - 1, 6000), verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1_n" + str(n_best), t_global)

    # Phase 2: Spectral shaping
    print("\n=== Phase 2: Spectral shaping (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining_time() * 0.08, 120.0)
    if t2 > 20:
        a_out, _ = spectral_shaping(norm_sum1(best_seq), t2, n_iters=3000, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spectral_n" + str(len(best_seq)), t_global)

    # Phase 3: Soft-peak descent
    print("\n=== Phase 3: Soft-peak descent (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining_time() * 0.10, 160.0)
    if t3 > 30:
        a_out, _ = soft_peak_descent(norm_sum1(best_seq), t3, beta_start=5.0, beta_end=60.0, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "softpeak_n" + str(len(best_seq)), t_global)

    # Phase 4: LP-FW second pass (with spectral + softpeak result)
    print("\n=== Phase 4: LP-FW second (n=" + str(len(best_seq)) + ") ===")
    t4 = min(remaining_time() * 0.20, 320.0)
    if t4 > 50 and len(best_seq) <= 10000:
        a_out, _ = lp_fw_full(norm_sum1(best_seq), t4, solver,
                              max_active=min(2 * len(best_seq) - 1, 8000), verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2_n" + str(len(best_seq)), t_global)

    # Phase 5: Multi-n exploration
    print("\n=== Phase 5: Multi-n exploration ===")
    n_cur = len(best_seq)
    # Try nearby n values
    candidates_n = [n_cur - 500, n_cur + 500, n_cur - 1000, n_cur + 1000,
                    int(n_cur * 0.8), int(n_cur * 1.2)]
    candidates_n = sorted(set([c for c in candidates_n if 3000 <= c <= 12000]))
    candidates_n.sort(key=lambda c: abs(c - n_cur))

    for new_n in candidates_n[:4]:
        rt = remaining_time()
        if rt < 100:
            break
        print("\n  Multi-n: n=" + str(new_n) + " (remaining=" + str(round(rt, 1)) + "s)")
        a_long = interpolate_seq(best_seq, new_n)

        t5 = min(rt * 0.15, 120.0)
        if t5 > 30 and new_n <= 10000:
            a_out, _ = lp_fw_full(a_long, t5, solver,
                                  max_active=min(2 * new_n - 1, 5000), verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

        rt = remaining_time()
        if rt > 40 and len(best_seq) <= 10000:
            t5b = min(rt * 0.08, 60.0)
            if t5b > 15:
                a_out, _ = soft_peak_descent(norm_sum1(best_seq), t5b, beta_start=10.0, beta_end=40.0, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "softpeak_n" + str(new_n), t_global)

    # Phase 6: LP-FW third pass on best (longest)
    print("\n=== Phase 6: LP-FW third (n=" + str(len(best_seq)) + ") ===")
    t6 = min(remaining_time() * 0.18, 250.0)
    if t6 > 40 and len(best_seq) <= 10000:
        # Use the best from initial exploration + all phases
        a_out, _ = lp_fw_full(norm_sum1(best_seq), t6, solver,
                              max_active=min(2 * len(best_seq) - 1, 10000), verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw3_n" + str(len(best_seq)), t_global)

    # Phase 7: Spectral shaping round 2
    print("\n=== Phase 7: Spectral round 2 (n=" + str(len(best_seq)) + ") ===")
    t7 = min(remaining_time() * 0.06, 80.0)
    if t7 > 15:
        a_out, _ = spectral_shaping(norm_sum1(best_seq), t7, n_iters=2000, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spectral2_n" + str(len(best_seq)), t_global)

    # Phase 8: Coordinate polish
    print("\n=== Phase 8: Coordinate polish (n=" + str(len(best_seq)) + ") ===")
    t8 = min(remaining_time() * 0.10, 120.0)
    if t8 > 20:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t8, rng_seed=42, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n" + str(len(best_seq)), t_global)

    # Phase 9: Final LP-FW
    print("\n=== Phase 9: Final LP-FW (n=" + str(len(best_seq)) + ") ===")
    t9 = min(remaining_time() * 0.15, 150.0)
    if t9 > 30 and len(best_seq) <= 10000:
        a_out, _ = lp_fw_full(norm_sum1(best_seq), t9, solver,
                              max_active=min(2 * len(best_seq) - 1, 12000), verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # Final coordinate polish
    rt = remaining_time()
    if rt > 15:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), min(rt * 0.6, 60.0), rng_seed=2024, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final soft-peak
    rt = remaining_time()
    if rt > 15:
        a_out, _ = soft_peak_descent(norm_sum1(best_seq), min(rt * 0.5, 40.0), beta_start=20.0, beta_end=80.0, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "softpeak_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
