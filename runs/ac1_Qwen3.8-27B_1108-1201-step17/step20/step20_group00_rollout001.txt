import numpy as np

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def build_lp_matrix(a, lags, n):
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs

def lp_direction(a, active_lags, solver):
    n = len(a)
    coeffs = build_lp_matrix(a, active_lags, n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [cp.sum(g) == 1, coeffs @ g <= M]
    prob = cp.Problem(cp.Minimize(M), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
            gs = g_val.sum()
            if gs > 1e-15:
                return g_val / gs, M.value
    except Exception:
        pass
    return None, None

def smoothed_gradient(a, R, top_B=20, beta=50.0):
    """Compute smoothed gradient of max autocorrelation."""
    n = len(a)
    g = np.zeros(n, dtype=np.float64)
    top_vals = np.sort(R)[::-1][:top_B]
    w = np.exp(beta * (top_vals - top_vals[0]))
    w = w / w.sum()
    top_idx = np.argsort(R)[::-1][:top_B]
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i_s = max(0, ki - n + 1)
        i_e = min(n, ki + 1)
        idx = np.arange(i_s, i_e)
        g[idx] += 2.0 * wi * a[ki - idx]
    return g

def nesterov_smoothed(a0, time_limit, top_B=20, beta_init=30.0, beta_max=200.0,
                        c0=2.0, mu_init=0.5, mu_end=0.9, verbose=True):
    """Smoothed gradient descent with Nesterov momentum."""
    n = len(a0)
    f = normalize(a0)
    best_f = f.copy()
    best_val = obj_val(f)
    t0 = time.monotonic()
    v = np.zeros(n, dtype=np.float64)
    y = f.copy()  # Nesterov extrapolation point
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = min(beta_init + t * (beta_max - beta_init) / 5000.0, beta_max)
        mu_t = min(mu_init + t * (mu_end - mu_init) / 2000.0, mu_end)
        
        # Nesterov extrapolation
        if t == 1:
            y = f.copy()
        else:
            mu_prev = min(mu_init + (t-2) * (mu_end - mu_init) / 2000.0, mu_end)
            y = f + (mu_t - 1.0) / (mu_prev - 1.0) * (f - y)
            # Project y
            y = np.maximum(y, 0.0)
            ys = y.sum()
            if ys > 1e-15:
                y = y / ys
        
        R = fast_autocorr(y)
        g = smoothed_gradient(y, R, top_B=top_B, beta=beta_t)
        
        eta_t = c0 / (np.sqrt(t) + 1.0)
        
        # Backtracking line search
        alpha = 1.0
        f_curr_val = obj_val(y)
        for _ in range(20):
            f_trial = np.maximum(y - eta_t * alpha * g, 0.0)
            fs = f_trial.sum()
            if fs < 1e-15:
                alpha *= 0.5
                continue
            f_trial = f_trial / fs
            val_trial = obj_val(f_trial)
            if val_trial < f_curr_val - 1e-15:
                break
            alpha *= 0.5
        
        f_new = np.maximum(y - eta_t * alpha * g, 0.0)
        fs = f_new.sum()
        if fs > 1e-15:
            f_new = f_new / fs
        else:
            f_new = y.copy()
        
        f = f_new
        val = obj_val(f)
        if val < best_val - 1e-13:
            best_val = val
            best_f = f.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print("  NestSmoothed t={} ({}s) J={}".format(t, round(elapsed,1), repr(best_val)))
        
        if best_val < 1.500 and t > 1000:
            break
    
    return best_f, best_val

                    dest_scores[idx2] += a[k2i - idx2]

        if time.monotonic() - t0 > time_limit:
            break
        
        F = np.fft.rfft(f, fft_size)
        P = np.abs(F)**2  # Periodogram
        R = np.fft.irfft(F * np.conj(F), fft_size)[:size]  # = P autocorrelation... wait
        
        # Actually R = IFFT(|F|^2) which is what we want
        # We want to reduce max R while keeping F[0] = 1
        
        R = fast_autocorr(f)
        Rmax = R.max()
        
        # Try to whiten: scale down components of F that contribute to peaks
        # R[k] = IFFT(P)[k] where P = |F|^2
        # dR/d|F[m]|^2 = (1/fft_size) * exp(-2πikm/fft_size)
        # So to reduce R at lag k, reduce |F[m]|^2 where exp(2πikm/fft_size) is large
        
        # Simple approach: multiply F by 1/sqrt(1+alpha*P)
        # This reduces high-P components more
        alpha = 0.1
        F_new = F / np.sqrt(1.0 + alpha * P)
        # Preserve DC
        F_new[0] = 1.0  # renormalize later
        F_new = np.real_if_close(F_new)
        
        f_new = np.fft.irfft(F_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new / s
        
        val = obj_val(f_new)
        if val < best_val - 1e-14:
            best_val = val
            best_f = f_new.copy()
            f = f_new
        else:
            # Try smaller alpha
            alpha *= 0.5
            F_new = F / np.sqrt(1.0 + alpha * P)
            F_new[0] = 1.0
            F_new = np.real_if_close(F_new)
            f_new = np.fft.irfft(F_new, fft_size)[:n]
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s > 1e-15:
                f_new = f_new / s
                val = obj_val(f_new)
                if val < best_val - 1e-14:
                    best_val = val
                    best_f = f_new.copy()
                    f = f_new
        
        if verbose and it % 10 == 0:
            print("  FreqWhit it={} ({}s) J={}".format(it, round(time.monotonic()-t0,1), repr(best_val)))
    
    return best_f, best_val

def parametric_init(n):
    """Generate diverse parametric initializations."""
    x = np.linspace(0.0, 1.0, n)
    i = np.arange(n, dtype=np.float64)
    candidates = []
    
    # Raised cosine
    candidates.append(normalize(1.0 - np.cos(2 * np.pi * x)))
    # Sine squared
    candidates.append(normalize(np.sin(np.pi * x) ** 2))
    # Power-law
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(normalize((i + 1.0) ** p))
        candidates.append(normalize((n - i) ** p))
    # Gaussian in center
    for sg in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5]:
        candidates.append(normalize(np.exp(-((x - 0.5) ** 2) / (2 * sg ** 2))))
    # Semicircle (Wigner)
    sc = np.sqrt(np.maximum(4 * x * (1 - x), 0))
    candidates.append(normalize(sc))
    # 1/sqrt(4x(1-x))
    inv = 1.0 / np.sqrt(np.maximum(4 * x * (1 - x), 1e-10))
    candidates.append(normalize(inv))
    # Double bump
    for c1, c2, sg in [(0.25, 0.75, 0.1), (0.25, 0.75, 0.15), (0.3, 0.7, 0.1), (0.3, 0.7, 0.2)]:
        db = np.exp(-((x-c1)**2)/(2*sg**2)) + np.exp(-((x-c2)**2)/(2*sg**2))
        candidates.append(normalize(db))
    # Tukey
    for alpha in [0.1, 0.2, 0.3, 0.5]:
        tk = np.ones(n)
        m = max(1, int(alpha * n))
        taper = 0.5 * (1 + np.cos(np.pi * np.arange(m) / m))
        tk[:m] = taper
        tk[-m:] = taper[::-1]
        candidates.append(normalize(tk))
    
    best_a = candidates[0]
    best_v = obj_val(best_a)
    for c in candidates:
        v = obj_val(c)
        if v < best_v:
            best_v = v
            best_a = c.copy()
    return best_a, best_v

def lp_fw_refine(a0, time_limit, solver, init_K=500, max_K=3000, verbose=True):
    """LP-based Frank-Wolfe refinement."""
    n = len(a0)
    y = normalize(a0)
    best_y = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[-init_K:])
    no_improve = 0
    stall_limit = 2000
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_y = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)
        
        if len(clist) == 0:
            break
        
        g_dir, M_lp = lp_direction(y, clist, solver)
        if g_dir is None:
            if len(clist) > 100:
                active_set = set(int(k) for k in clist[:200])
                continue
            break
        
        # Find violations
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:40]]
            for kk in top_v:
                active_set.add(int(kk))
        
        # Line search
        best_alpha = 1.0
        best_trial = M_curr
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            trial = np.maximum((1 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            Mt = fast_autocorr(trial).max()
            if Mt < best_trial:
                best_trial = Mt
                best_alpha = alpha
        
        y_new = np.maximum((1 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy
        else:
            y = g_dir.copy()
        
        if verbose and t % 500 == 0:
            print("  LP-FW t={} ({}s) J={} |S|={}".format(t, round(time.monotonic()-t0,1), repr(2*n*best_m), len(active_set)))
    
    return best_y, 2.0 * n * best_m
