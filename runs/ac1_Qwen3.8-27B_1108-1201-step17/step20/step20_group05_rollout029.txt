    """Solve the LP subproblem for Frank-Wolfe using scipy's linprog with HiGHS."""
    n = a.size
    K = active_idx.size
    
    if K == 0 or n == 0:
        return None
    
    # Build the constraint matrix A_ub where A_ub @ [g, t] <= b_ub
    # For each k in active_idx: sum_i a[i]*g[k-i] - t <= 0
    # The coefficient of t is -1
    
    rows = []
    cols = []
    vals = []
    
    for r, k in enumerate(active_idx):
        # (f*g)[k] = sum_{i: 0<=k-i<n} f[i]*g[k-i] = sum_{i: max(0,k-n+1)<=i<=min(n-1,k)} f[i]*g[k-i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            cols_i = np.arange(lo, hi + 1)
            rows.extend([r] * (hi - lo + 1))
            cols.extend(cols_i.tolist())
            vals.extend(a[hi - lo:k - lo + 1][::-1].tolist())
        # Coefficient of t variable (column n) is -1
        rows.append(r)
        cols.append(n)
        vals.append(-1.0)
    
    if not rows:
        return None
    
    from scipy.sparse import csr_matrix
    A_ub = csr_matrix((vals, (rows, cols)), shape=(K, n + 1))
    b_ub = np.zeros(K)
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize t (variable at index n)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                                     A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                                     method='HiGHS', 
                                     options={'disp': False, 'time_limit': time_limit})
        if res.success:
            g = res.x[:n]
            g = np.maximum(g, 0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return np.maximum(c, 0.0)


def fft_cross(a, b):

    b = np.asarray(b, dtype=np.float64)

    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    return np.maximum(c, 0.0)

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x) = target_sum, x >= 0}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    # Standard projection onto simplex
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fw_lp_direction(a, active_idx, time_limit=5.0):
    """Solve the LP subproblem for Frank-Wolfe using HiGHS."""
    if not SCIPY_OK:
        return None
    n = a.size
    K = active_idx.size
    if K == 0 or n <= 1:
        return None
    
    rows = []
    cols = []
    vals = []
    
    for r, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            rows.extend([r] * (hi - lo + 1))
            cols.extend(idx.tolist())
            # a[k - i] for i in [lo, hi]
            rev_idx = k - idx
            vals.extend(a[rev_idx].tolist())
        rows.append(r)
        cols.append(n)
        vals.append(-1.0)
    
    if not rows:
        return None
    
    A_ub = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n + 1))
    b_ub = np.zeros(K)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    try:
        res = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                                     A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                                     method='HiGHS',
                                     options={'disp': False, 'time_limit': time_limit})
        if res.success:
            g = res.x[:n]
            g = np.maximum(g, 0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def fw_game_direction(a, active_idx, iters=30):
    """Solve the LP subproblem approximately using multiplicative weights."""
    n = a.size
    K = active_idx.size
    if K == 0 or n <= 1:
        return np.ones(n) / n
    
    # Start with uniform distribution over active constraints
    lam = np.ones(K) / K
    
    for _ in range(iters):
        # Compute pressure: (lam-weighted sum of (f*g)[k]) projected back
        # pf[k] = lam[k] for k in active_idx
        pf = np.zeros(2 * n - 1)
        pf[active_idx] = lam
        
        # pressure[i] = sum_k pf[k] * a[k-i] = (pf * a_reversed)[i] ... 
        # Actually: pressure[i] = sum_k pf[k] * a[k-i] where 0<=k-i<n
        # This is the cross-correlation of pf with a.
        # pressure = ifft(fft(pf) * conj(fft(a)))... 
        # Or more simply: pressure[i] = sum_j pf[i+j] * a[j] (valid range)
        # Let me just use fft_cross
        pressure = fft_cross(a, pf)  # This gives (a * pf)[k] for k=0..2n-2
        # But I want for each i: sum_k pf[k] * a[k-i]
        # That's the cross-correlation, which is: ifft(conj(fft(a)) * fft(pf))
        # Or equivalently: reverse pf, cross-correlate with a
        
        # Actually, let me define it clearly:
        # For the LP: minimize max_k (f*g)[k] s.t. g >= 0, sum g = 1
        # The dual is: minimize t s.t. sum_i f[i]*g[k-i] <= t for all k, sum g = 1, g >= 0
        # Dual of dual: maximize min_g <pressure, g> where pressure = weighted sum of constraint gradients
        # 
        # The gradient of constraint k w.r.t. g is: grad_k[i] = f[k-i] (if 0<=k-i<n)
        # So pressure[i] = sum_r lam[r] * f[active_idx[r] - i] (if valid)
        
        # Let me compute this directly:
        pressure = np.zeros(n)
        for r in range(K):
            k = active_idx[r]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                pressure[lo:hi+1] += lam[r] * a[k - hi:k - lo + 1][::-1]
        
        # g ∝ exp(-eta * pressure) (to minimize <pressure, g>)
        pmin = pressure.min()
        pmax = pressure.max()
        ptp = pmax - pmin
        if ptp < 1e-30:
            g = np.ones(n) / n
        else:
            scaled = (pressure - pmin) / ptp
            w = np.exp(-10.0 * scaled)
            w = np.maximum(w, 1e-300)
            g = w / w.sum()
        
        # Update lam: move mass to constraints where (f*g) is largest
        fg = fft_cross(a, g)
        fg_active = fg[active_idx]
        fg_max = fg_active.max()
        fg_min = fg_active.min()
        fg_range = fg_max - fg_min
        if fg_range > 1e-30:
            eta = 15.0 / fg_range
            lam *= np.exp(eta * fg_active)
            lam /= lam.sum()
        else:
            lam = np.ones(K) / K
    
    return g


def line_search_fw(a, g, c_a, mask=None, max_iters=20):
    """Line search for FW step: minimize f((1-alpha)*a + alpha*g)."""
    n = a.size
    cg = fft_conv(g)
    fg = fft_cross(a, g)
    
    # f(alpha) = max_k [(1-a)^2 * c_a[k] + 2*a*(1-a)*fg[k] + a^2*cg[k]]
    # This is the max of quadratic functions in alpha, so it's convex.
    
    best_alpha = 0.0
    best_val = float(c_a.max()) if mask is None else float(c_a[mask].max())
    
    # Try a grid of alphas
    alphas = np.linspace(0, 1, 21)
    for alpha in alphas:
        a1 = 1.0 - alpha
        val = a1 * a1 * c_a + 2.0 * alpha * a1 * fg + alpha * alpha * cg
        if mask is None:
            m = val.max()
        else:
            m = val[mask].max()
        if m < best_val - 1e-15:
            best_val = m
            best_alpha = alpha
    
    # Refine with ternary search around best_alpha
    if 0 < best_alpha < 1:
        lo = best_alpha - 0.05
        hi = best_alpha + 0.05
        lo = max(0.0, lo)
        hi = min(1.0, hi)
        gr = (math.sqrt(5) - 1) / 2
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        
        def eval_alpha(al):
            a1 = 1.0 - al
            val = a1 * a1 * c_a + 2.0 * al * a1 * fg + al * al * cg
            if mask is None:
                return val.max()
            return val[mask].max()
        
        f1 = eval_alpha(c1)
        f2 = eval_alpha(c2)
        for _ in range(20):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = eval_alpha(c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = eval_alpha(c2)
        
        opt = (lo + hi) / 2
        f_opt = eval_alpha(opt)
        if f_opt < best_val:
            best_alpha = opt
            best_val = f_opt
    
    return best_alpha, best_val


def frank_wolfe_exact(a0, budget, active_max=500, lp_every=3, lp_time=4.0, verbose=True):
    """Frank-Wolfe with exact LP directions (when available)."""
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = float(c.max())
    best_a = a.copy()
    best_M = M
    active = np.argsort(c)[::-1][:active_max]
    
    no_improve = 0
    t = 0
    
    while time.monotonic() - t0 < time_limit - 5 and no_improve < 60:
        t += 1
        if time_limit - (time.monotonic() - t0) < 30:
            break
        
        # Get direction
        g = None
        if lp_every > 0 and (t % lp_every == 0):
            g = fw_lp_direction(a, active, time_limit=lp_time)
        
        if g is None:
            g = fw_game_direction(a, active, iters=30)
        
        if g is None:
            break
        
        # Line search
        alpha, new_M = line_search_fw(a, g, c)
        
        if alpha > 0 and new_M < M - 1e-16:
            one = 1.0 - alpha
            a = one * a + alpha * g
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s
            c = fft_conv(a)
            M = float(c.max())
            no_improve = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
                if verbose and (t <= 3 or t % 10 == 0):
                    print(f"    FW t={t} score={2*n*best_M:.8f} alpha={alpha:.4f}")
            active = np.argsort(c)[::-1][:active_max]
        else:
            no_improve += 1
            if no_improve % 10 == 9:
                # Perturb to escape
                rng = np.random.default_rng(12345 + t)
                a = best_a + rng.normal(0, 0.01, n) * best_a
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                c = fft_conv(a)
                M = float(c.max())
                active = np.argsort(c)[::-1][:active_max]
                no_improve = 0
        
        if verbose and t == 1:
            print(f"    FW t=1 score={2*n*M:.8f} active={active_max}")
    
    return best_a


def pairwise_optimize(a, i, j, c_a, active_idx):
    """
    Optimize the split between a[i] and a[j].
    Returns the new a and new c_a, or None if no improvement.
    """
    n = a.size
    c_total = a[i] + a[j]
    if c_total <= 1e-15:
        return None, None, None
    
    # Precompute the "rest" convolution and shifted arrays
    b = a.copy()
    b[i] = 0.0
    b[j] = 0.0
    conv_b = fft_conv(b)
    
    # bi[k] = b[k-i] if 0 <= k-i < n, else 0
    bi = np.zeros(2 * n - 1)
    lo_i = max(0, i)
    hi_i = min(2 * n - 2, i + n - 1)
    if lo_i <= hi_i:
        bi[lo_i:hi_i+1] = b[0:(hi_i - lo_i + 1)]
    
    # bj[k] = b[k-j] if 0 <= k-j < n, else 0
    bj = np.zeros(2 * n - 1)
    lo_j = max(0, j)
    hi_j = min(2 * n - 2, j + n - 1)
    if lo_j <= hi_j:
        bj[lo_j:hi_j+1] = b[0:(hi_j - lo_j + 1)]
    
    current_M = float(c_a[active_idx].max())
    
    def eval_at_x(x):
        cx = c_total - x
        # conv[k] = conv_b[k] + 2*x*bi[k] + 2*cx*bj[k]
        # + x*cx*[k==i+j] + x^2*[k==2i] + cx^2*[k==2j]
        vals = conv_b[active_idx] + 2.0 * x * bi[active_idx] + 2.0 * cx * bj[active_idx]
        # Add special terms
        m = float(vals.max())
        # Check special positions
        if (i + j) < (2 * n - 1):
            pos = np.where(active_idx == (i + j))[0]
            if len(pos) > 0:
                # conv[i+j] at this x: conv_b[i+j] + 2*x*bi[i+j] + 2*cx*bj[i+j] + x*cx
                # bi[i+j] = b[j] = 0, bj[i+j] = b[i] = 0
                # So: conv_b[i+j] + x*cx
                v = conv_b[i+j] + x * cx
                m = max(m, v)
        if (2 * i) < (2 * n - 1):
            pos = np.where(active_idx == (2 * i))[0]
            if len(pos) > 0:
                # conv[2i] = conv_b[2i] + 2*x*bi[2i] + 2*cx*bj[2i] + x^2
                # bi[2i] = b[i] = 0
                v = conv_b[2*i] + 2.0*cx*bj[2*i] + x * x
                m = max(m, v)
        if (2 * j) < (2 * n - 1):
            pos = np.where(active_idx == (2 * j))[0]
            if len(pos) > 0:
                # conv[2j] = conv_b[2j] + 2*x*bi[2j] + 2*cx*bj[2j] + cx^2
                # bj[2j] = b[j] = 0
                v = conv_b[2*j] + 2.0*x*bi[2*j] + cx * cx
                m = max(m, v)
        return m
    
    # Golden section search
    gr = (math.sqrt(5) - 1) / 2
    lo = 0.0
    hi = c_total
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = eval_at_x(c1)
    f2 = eval_at_x(c2)
    
    for _ in range(30):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = eval_at_x(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = eval_at_x(c2)
    
    x_opt = (lo + hi) / 2
    f_opt = eval_at_x(x_opt)
    
    if f_opt < current_M - 1e-18:
        a[i] = x_opt
        a[j] = c_total - x_opt
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return None, None, None
        a = a / s
        c_new = fft_conv(a)
        return a, c_new, f_opt
    
    return None, None, None


def coordinate_search(a, budget, active_max=200, verbose=True):
    """Efficient coordinate search using pairwise optimization."""
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a)
    n = a.size
    c = fft_conv(a)
    M = float(c.max())
    best_a = a.copy()
    best_M = M
    active = np.arg
