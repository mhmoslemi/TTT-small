"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Multi-scale FISTA with adaptive smoothing and Nesterov acceleration
2. Mirror descent (exponentiated gradient) for simplex-structured refinement
3. Peak shaving with exact subgradient and backtracking
4. Smart local search (pairwise mass transfers targeting peak lags)
5. Cyclic shift enumeration
6. Multi-length exploration with interpolation

import numpy as np
import time

def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum to 1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    """Project onto the probability simplex."""
    n = len(z)
    if n <= 1:
        return np.maximum(z, 0.0)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_smoothed_grad(a, beta, top_k=None):
    """Compute gradient of smoothed max of autocorrelation."""
    n = len(a)
    b = fast_autocorr(a)
    if top_k is not None and 0 < top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b_use.max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b_use)
        mask = (b_use == b_max)
        p[mask] = 1.0 / max(1, mask.sum())
    else:
        p = exp_vals / p_total
    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j] (with bounds)
    # This is 2 * cross_correlation(p, a) at the right lag
    # Use FFT for efficiency
    fft_size = 1 << (n + len(p) - 1).bit_length()
    # cross_corr(p, a)[j] = sum_m p[j+m] * a[m] for m=0..n-1
    # = sum_m a[m] * p[j+m]
    # This is the (linear) cross-correlation of a and p evaluated at... 
    # Let's think: we want g[j] = sum_m a[m] * p[j+m] for j=0..n-1
    # Let q = p reversed: q[i] = p[len(p)-1-i]
    # Then g[j] = sum_m a[m] * p[j+m]
    # Let n_p = len(p) = 2n-1
    # g[j] = sum_{m=0}^{n-1} a[m] * p[j+m]
    # This is (a * p_flipped)[j + n_p - 1 - (n-1)] ... 
    # Actually, let's just use the convolution directly.
    # g[j] = sum_m a[m] p[j+m] = (a * p_rev)[(n-1) + j + (n_p - 1) - (n-1)]
    # Hmm, let me just use a direct FFT approach.
    # g = cross_correlation where g[j] = <a, shift_j(p)>
    # = sum_m a[m] p[m+j]
    # This is the cross-correlation of a and p at lag j (p shifted right by j).
    # Equivalently: g = ifft(fft(a) * conj(fft(p_reversed)))... 
    # Let me just compute it as: full_convolution(a, p)[n-1 : 2n-1]
    # Wait: (a * p)[k] = sum_i a[i] p[k-i]
    # We want sum_m a[m] p[j+m] = sum_i a[i] p[j+i] = (a * p_flipped)[(len(p)-1-j) + ...]
    # 
    # Simpler: use the fact that g[j] = sum_m a[m] * p[m+j]
    # This is the "cross-correlation" which can be computed as:
    # g = ifft(fft(a, N) * conj(fft(p, N)))[:n] where N >= n + len(p) - 1
    # Wait, cross-correlation: (a ⋆ p)[j] = sum_i a[i+j] conj(p[i])
    # That's not quite what we want either.
    #
    # Let me just be explicit. We want:
    # g[j] = sum_{m=0}^{n-1} a[m] * p[m+j] for j = 0, 1, ..., n-1
    #
    # Define p_ext of length n + n - 1 = 2n-1 (same as p since p has length 2n-1)
    # Actually p already has length 2n-1.
    # g[j] = sum_{m=0}^{n-1} a[m] * p[m+j]
    # Let's define q[m] = a[m] for m=0..n-1, q[m]=0 otherwise.
    # Then g[j] = sum_m q[m] * p[m+j] = (q ⋆ p)[j] where ⋆ is cross-correlation
    # Cross-correlation: (q ⋆ p)[j] = sum_m q[m] * p[m+j]
    # This can be computed as: ifft(conj(fft(q)) * fft(p)) but with appropriate sizing
    # Actually the standard way: (q ⋆ p)[j] = (q * p_flipped)[j + len(p) - 1]
    # where * is linear convolution and p_flipped = p[::-1]
    #
    # So: conv = convolve(q, p_flipped), g[j] = conv[j + len(p) - 1] = conv[j + 2n - 2]
    # conv has length n + (2n-1) - 1 = 3n - 2
    # g[j] = conv[j + 2n - 2] for j = 0, ..., n-1
    #
    # Alternatively: g = correlate(q, p, mode='valid')... but that might not be available.
    #
    # Let me just use FFT directly:
    # g[j] = sum_m a[m] * p[m+j]
    # FFT approach: Let A = FFT(a, N), P = FFT(p, N) where N >= n + 2n-1 - 1 = 3n-2
    # Then g = IFFT(A * P)[?]... 
    # 
    # Actually the cleanest way:
    # g[j] = sum_{m=0}^{n-1} a[m] p[m+j]
    # = sum_{m=0}^{n-1} a[m] p[j+m]
    # 
    # Define r[k] = p[k] for k=0..2n-2
    # g[j] = sum_m a[m] r[j+m]
    # This is the "sliding dot product" of a with r, at offset j.
    # This is exactly (a * r_flipped)[j + (2n-2)] where r_flipped = r[::-1]
    # Let me verify: (a * r_flipped)[k] = sum_i a[i] * r_flipped[k-i] = sum_i a[i] * r[2n-2-(k-i)]
    # Setting k = j + 2n-2: = sum_i a[i] * r[2n-2-j-2n+2+i] = sum_i a[i] * r[i-j]
    # Hmm, that gives r[i-j] not r[j+i].
    #
    # Let me try: (r * a_flipped)[k] where a_flipped = a[::-1]
    # = sum_i r[i] * a_flipped[k-i] = sum_i r[i] * a[n-1-(k-i)] = sum_i r[i] * a[n-1-k+i]
    # Setting k = n-1-j: = sum_i r[i] * a[j+i]
    # But we want sum_m a[m] r[j+m] = sum_m r[j+m] a[m]
    # Let i = j+m: sum_i r[i] a[i-j] ... that's not quite right either.
    #
    # OK let me just be very explicit and use a simple approach:
    # g[j] = sum_{m=0}^{n-1} a[m] * p[m+j]
    # 
    # Let's compute this using FFT:
    # Pad a to length N and p to length N where N >= n + (2n-1) - 1 = 3n-2
    # Actually, we can think of it as:
    # g = [dot(a, p[0:n]), dot(a, p[1:n+1]), ..., dot(a, p[n-1:2n-1])]
    # = [a·p[0:n], a·p[1:n+1], ..., a·p[n-1:2n-1]]
    #
    # This is a "cross-correlation" in the signal processing sense.
    # In numpy: np.correlate(p, a, mode='valid') gives exactly this!
    # np.correlate(p, a, mode='valid')[j] = sum_m p[m+j] * a[m] for m=0..n-1
    # Yes! That's exactly g[j].
    #
    # But np.correlate is O(n^2) for large arrays. Let's use FFT.
    # np.correlate(p, a) = convolve(p, a[::-1])
    # So g = convolve(p, a[::-1], mode='valid')
    # = irfft(fft(p) * fft(a[::-1]))[?]
    
    # Let me just use the FFT approach directly
    N = 1 << (n + len(p) - 1).bit_length()
    A = np.fft.rfft(a, N)
    # We need: g[j] = sum_m a[m] * p[m+j]
    # = sum_m a[m] * p[m+j]
    # Let's define b[m] = a[m], c[k] = p[k]
    # g[j] = sum_m b[m] c[m+j]
    # This is the cross-correlation (b ⋆ c)[j]
    # FFT formula: (b ⋆ c)[j] = ifft(FFT(b) * conj(FFT(c)))[j] ... but this is circular
    # For linear: we need N >= len(b) + len(c) - 1
    C = np.fft.rfft(p, N)
    # Circular cross-correlation: ifft(B * conj(C))
    # But we need linear. With zero-padding to N, the first (N - len(c) + 1) values of the circular
    # cross-correlation equal the linear cross-correlation.
    # (b ⋆ c)[j] for j = 0, ..., len(b)-1 = ifft(B * conj(C))[j] when N >= len(b)+len(c)-1
    # Wait, I need to be careful about the definition.
    # Standard: (x ⋆ y)[n] = sum_k x[k] y[k+n] (some define it differently)
    # In terms of FFT: ifft(FFT(x) * conj(FFT(y))) gives sum_k x[k] y[k-n] (negative lag)
    # Or: ifft(conj(FFT(x)) * FFT(y)) gives sum_k x[k] y[k+n] (positive lag)
    
    # Let me verify with a small example mentally:
    # a = [a0, a1], p = [p0, p1, p2]
    # g[0] = a0*p0 + a1*p1
    # g[1] = a0*p1 + a1*p2
    # 
    # FFT approach: N >= 2+3-1 = 4, use N=4
    # A = [a0+a1, a0-a1, a0+a1 (wait, for N=4: FFT(a_padded) where a_padded = [a0,a1,0,0])
    # A[0] = a0+a1, A[1] = a0-a1 (for N=4, the 4-point DFT of [a0,a1,0,0]:
    #   A[k] = a0 + a1*e^{-2πik/4} = a0 + a1*i^{-k}
    #   A[0] = a0+a1, A[1] = a0-a1*i... this is getting complicated.
    
    # Let me just use the convolution identity:
    # sum_m a[m] p[m+j] = (a * p_rev)[(len(p)-1) + j] where p_rev = p[::-1]
    # Because (a * p_rev)[k] = sum_i a[i] p_rev[k-i] = sum_i a[i] p[len(p)-1-k+i]
    # With k = len(p)-1+j: = sum_i a[i] p[len(p)-1-(len(p)-1-j)+i] = sum_i a[i] p[j+i] ✓
    
    p_rev = p[::-1]
    fft_size2 = 1 << (n + len(p_rev) - 1).bit_length()
    A2 = np.fft.rfft(a, fft_size2)
    Pr = np.fft.rfft(p_rev, fft_size2)
    conv = np.fft.irfft(A2 * Pr, fft_size2)
    offset = len(p) - 1  # = 2n-2
    grad = 2.0 * conv[offset:offset + n]
    return grad, p, b

def fista_multiscale(a0, time_limit, verbose=True):
    """Multi-scale FISTA with adaptive parameters."""
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    # Multi-scale schedule: pairs of (beta_start, beta_end, num_iters)
    scales = [
        (5.0, 30.0, 8000),
        (30.0, 100.0, 8000),
        (100.0, 400.0, 8000),
        (400.0, 1500.0, 6000),
    ]
    
    restart_every = 3000
    
    for si, (beta_start, beta_end, max_iters) in enumerate(scales):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta = beta_start
        x_prev = x.copy()
        t_nest = 1.0
        stall = 0
        s_prev = None
        y_prev = None
        bb_step = 1.0 / beta
        
        for t in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
            theta = (t_nest - 1.0) / t_new
            y = x + theta * (x - x_prev)
            y = np.maximum(y, 1e-15)
            s_y = y.sum()
            if s_y < 1e-15:
                y = x.copy()
            else:
                y = y / s_y
            
            # Adaptive top-k: start with 20%, reduce to 5%
            frac = max(0.05, 0.20 - 0.15 * si / max(1, len(scales) - 1))
            top_k = max(5, int(frac * (2 * n - 1)))
            
            grad, p, b = compute_smoothed_grad(y, beta, top_k)
            
            # BB step size
            if s_prev is not None and y_prev is not None:
                sy = y - y_prev
                gy = grad - s_prev
                sy_gy = sy.dot(gy)
                gy_gy = gy.dot(gy)
                if gy_gy > 1e-20:
                    bb_cand = sy_gy / gy_gy
                    if bb_cand > 1e-8:
                        bb_step = max(1e-8, min(bb_cand, 5.0 / beta))
            s_prev = grad.copy()
            y_prev = y.copy()
            
            eta = min(bb_step, 0.5)
            z = y - eta * grad
            x_new = simplex_project(z)
            
            b_new = fast_autocorr(x_new)
            max_new = b_new.max()
            
            if max_new < best_max_b - 1e-15:
                best_max_b = max_new
                best_a = x_new.copy()
                stall = 0
            else:
                stall += 1
            
            x_prev = x
            x = x_new
            t_nest = t_new
            
            # Grow beta
            if t % 50 == 0:
                beta = min(beta * 1.005, beta_end)
            
            # Restart
            if stall > restart_every:
                stall = 0
                t_nest = 1.0
                noise = 0.02 / (1.0 + 0.1 * si)
                pert = rng.normal(0, noise, n) * best_a
                x_r = best_a + pert
                x_r = np.maximum(x_r, 1e-10)
                x_r = x_r / x_r.sum()
                x = x_r
                x_prev = x_r.copy()
                beta = max(beta * 0.7, beta_start)
                s_prev = None
                y_prev = None
                bb_step = 1.0 / beta
        
        if verbose:
            J = 2.0 * n * best_max_b
            print(f"    FISTA scale {si+1} done: J={J:.6f} (t={time.monotonic()-t0:.1f}s)")
    
    return best_a, best_max_b


def mirror_descent(a0, time_limit, beta=50.0, top_k=None, verbose=True):
    """Exponentiated gradient / mirror descent on the simplex."""
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    
    # Adaptive eta
    eta = 1.0 / np.sqrt(n)
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        if top_k is None:
            top_k = max(10, int(0.1 * (2 * n - 1)))
        
        grad, p, b = compute_smoothed_grad(x, beta, top_k)
        
        # Mirror descent update: x_new = x * exp(-eta * grad) / Z
        log_updates = np.log(x + 1e-30) - eta * grad
        log_updates -= log_updates.max()  # numerical stability
        weights = np.exp(log_updates)
        z = weights.sum()
        if z < 1e-30:
            break
        x_new = weights / z
        
        # Ensure non-negativity and sum to 1
        x_new = np.maximum(x_new, 1e-30)
        x_new = x_new / x_new.sum()
        
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
        
        # Decay eta
        eta *= 0.9999
        
        x = x_new
        
        if verbose and t % 5000 == 0:
            J = 2.0 * n * best_max_b
            print(f"    Mirror t={t} ({time.monotonic()-t0:.1f}s): J={J:.6f} eta={eta:.6f}")
    
    return best_a, best_max_b


def peak_shaving(a0, time_limit, verbose=True):
    """Direct subgradient method on the exact max autocorrelation."""
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    
    eta0 = 0.1
    eta = eta0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(x)
        k_star = int(np.argmax(b))
        current_max = b[k_star]
        
        # Subgradient of max autocorrelation at x:
        # grad[j] = 2 * a[k_star - j] (with bounds, a[m]=0 if m not in [0,n-1])
        js = np.arange(n)
        idx = k_star - js
        valid = (idx >= 0) & (idx < n)
        grad = np.zeros(n)
        grad[valid] = 2.0 * x[idx[valid]]
        # Handle the case where k_star - j = j (i.e., j = k_star/2): already counted once above
        # Actually the formula grad[j] = 2*a[k-j] already handles this correctly as we derived.
        
        # Normalized step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-30:
            break
        
        # Backtracking line search
        best_trial = None
        best_trial_val = current_max
        for back in range(10):
            trial = x - eta * (grad / grad_norm)
            trial = simplex_project(trial)
            b_trial = fast_autocorr(trial)
            max_trial = b_trial.max()
            if max_trial < best_trial_val:
                best_trial_val = max_trial
                best_trial = trial.copy()
            if max_trial < current_max * (1 - 0.01):
                break
            eta *= 0.5
        
        if best_trial is not None:
            x = best_trial
            if best_trial_val < best_max_b - 1e-15:
                best_max_b = best_trial_val
                best_a = x.copy()
        else:
            # No improvement, try a small random perturbation
            x = norm_sum1(x + 0.001 * np.random.default_rng(t).normal(0, 1, n))
            eta *= 0.9
        
        if verbose and t % 2000 == 0:
            J = 2.0 * n * best_max_b
            print(f"    PeakShave t={t} ({time.monotonic()-t0:.1f}s): J={J:.6f}")
    
    return best_a, best_max_b


def local_search(a0, time_limit, verbose=True):
    """Local search with smart pairwise moves."""
    n = len(a0)
    x = norm_sum1(a0).copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        b = fast_autocorr(x)
        k_star = int(np.argmax(b))
        
        # Find positions that contribute most to the peak at k_star
        js = np.arange(n)
        idx = k_star - js
        valid = (idx >= 0) & (idx < n)
        contrib = np.zeros(n)
        contrib[valid] = x[idx[valid]] * x[js[valid]]
        
        # Get top contributing positions
        top_contrib = np.argsort(contrib)[::-1][:20]
        
        improved = False
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Source: one of the top contributing positions
            src = int(top_contrib[rng.integers(len(top_contrib))])
            if x[src] < 1e-10:
                continue
            
            # Destination: a position that doesn't contribute to the main peak
            # and has low "involvement" in the autocorrelation
            in_peak = np.zeros(n, dtype=bool)
            top_peaks = np.argsort(b)[::-1][:10]
            for kp in top_peaks:
                s = max(0, int(kp) - n + 1)
                e = min(n - 1, int(kp))
                in_peak[s:e+1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src:
                continue
            
            # Try different fractions

def cyclic_shift_best(a0):
    """Try all cyclic shifts and return the best."""
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for s in range(n):
        if s == 0:
            continue
        a_shift = np.roll(a0, -s)
        A = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best if a is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s

        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.1f}s)")
    return best_seq, best_score

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n <= 1:
        return np.maximum(z, 0.0)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_smoothed_grad(a, beta, top_k=None):
    n = len(a)
    b = fast_autocorr(a)
    if top_k is not None and 0 < top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b_use[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b_use)
        mask = (b_use == b_max)
        p[mask] = 1.0 / max(1, mask.sum())
    else:
        p = exp_vals / p_total
    # grad[j] = 2 * sum_m a[m] * p[m+j] for j=0..n-1
    # = 2 * conv(a_rev, p)[n-1 : 2n-1]
    a_rev = a[::-1]
    N = 1 << (n + len(p) - 1).bit_length()
    A = np.fft.rfft(a_rev, N)
    P = np.fft.rfft(p, N)
    conv = np.fft.irfft(A * P, N)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, p, b


def fista_multiscale(a0, time_limit, verbose=True):
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    scales = [
        (5.0, 40.0, 10000),
        (40.0, 150.0, 10000),
        (150.0, 500.0, 8000),
        (500.0, 2000.0, 6000),
    ]

    restart_every = 3000

    for si, (beta_start, beta_end, max_iters) in enumerate(scales):
        if time.monotonic() - t0 > time_limit:
            break

        beta = beta_start
        x_prev = x.copy()
        t_nest = 1.0
        stall = 0
        s_prev = None
        y_prev = None
        bb_step = 2.0 / beta

        for t in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
            theta = (t_nest - 1.0) / t_new
            y = x + theta * (x - x_prev)
            y = np.maximum(y, 1e-15)
            s_y = y.sum()
            if s_y < 1e-15:
                y = x.copy()
            else:
                y = y / s_y

            frac = max(0.03, 0.25 - 0.20 * si / max(1, len(scales) - 1))
            top_k = max(5, int(frac * (2 * n - 1)))

            grad, p, b = compute_smoothed_grad(y, beta, top_k)

            if s_prev is not None and y_prev is not None:
                sy = y - y_prev
                gy = grad - s_prev
                sy_gy = sy.dot(gy)
                gy_gy = gy.dot(gy)
                if gy_gy > 1e-20:
                    bb_cand = sy_gy / gy_gy
                    if bb_cand > 1e-8 and bb_cand < 20.0 / beta:
                        bb_step = max(1e-8, min(bb_cand, 10.0 / beta))
                    else:
                        bb_step = max(bb_step * 0.9, 1e-8)
            s_prev = grad.copy()
            y_prev = y.copy()

            eta = min(bb_step, 2.0 / beta)
            z = y - eta * grad
            x_new = simplex_project(z)

            b_new = fast_autocorr(x_new)
            max_new = b_new.max()

            if max_new < best_max_b - 1e-15:
                best_max_b = max_new
                best_a = x_new.copy()
                stall = 0
            else:
                stall += 1

            x_prev = x
            x = x_new
            t_nest = t_new

            if t % 50 == 0:
                beta = min(beta * 1.004, beta_end)

            if stall > restart_every:
                stall = 0
                t_nest = 1.0
                noise = 0.02 / (1.0 + 0.1 * si)
                pert = rng.normal(0, noise, n) * best_a
                x_r = best_a + pert
                x_r = np.maximum(x_r, 1e-10)
                x_r = x_r / x_r.sum()
                x = x_r
                x_prev = x_r.copy()
                beta = max(beta * 0.75, beta_start)
                s_prev = None
                y_prev = None
                bb_step = 2.0 / beta

        if verbose:
            J = 2.0 * n * best_max_b
            print(f"    FISTA scale {si+1}/4 done: J={J:.6f} (t={time.monotonic()-t0:.1f}s)")

    return best_a, best_max_b

def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2000, add_per_iter=20, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 600

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5),
                  max(0.02, default_alpha * 2.0), 1.0 / (t + 1)]
        alphas = [max(0.01, min(1.0, al)) for al in alphas]
        alphas = list(set(alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f} |S|={len(active_set)}")

    return best_a, best_m


def mirror_descent(a0, time_limit, beta=80.0, verbose=True):
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    eta = 0.5 / np.sqrt(n)

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        top_k = max(10, int(0.1 * (2 * n - 1)))
        grad, p, b = compute_smoothed_grad(x, beta, top_k)

        log_updates = np.log(x + 1e-30) - eta * grad
        log_updates -= log_updates.max()
        weights = np.exp(log_updates)
        z = weights.sum()
        if z < 1e-30:
            break
        x_new = weights / z
        x_new = np.maximum(x_new, 1e-30)
        x_new = x_new / x_new.sum()

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()

        eta *= 0.9995
        x = x_new

        if verbose and t % 5000 == 0:
            J = 2.0 * n * best_max_b
            print(f"    Mirror t={t} ({time.monotonic()-t0:.1f}s): J={J:.6f}")

    return best_a, best_max_b


def peak_shaving(a0, time_limit, verbose=True):
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(55)
    eta = 0.05

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(x)
        k_star = int(np.argmax(b))
        current_max = b[k_star]

        js = np.arange(n)
        idx = k_star - js
        valid = (idx >= 0) & (idx < n)
        grad = np.zeros(n)
        grad[valid] = 2.0 * x[idx[valid]]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-30:
            break

        best_trial = None
        best_trial_val = current_max
        for back in range(8):
            trial = x - eta * (grad / grad_norm)
            trial = simplex_project(trial)
            b_trial = fast_autocorr(trial)
            max_trial = b_trial.max()
            if max_trial < best_trial_val:
                best_trial_val = max_trial
                best_trial = trial.copy()
            if max_trial < current_max * 0.999:
                break
            eta *= 0.5

        if best_trial is not None:
            x = best_trial
            if best_trial_val < best_max_b - 1e-15:
                best_max_b = best_trial_val
                best_a = x.copy()
        else:
            x = norm_sum1(x + 0.002 * rng.normal(0, 1, n) * x)
            eta = max(eta * 0.9, 0.001)

        if verbose and t % 3000 == 0:
            J = 2.0 * n * best_max_b
            print(f"    PeakShave t={t} ({time.monotonic()-t0:.1f}s): J={J:.6f}")

    return best_a, best_max_b


def spectral_shape(a0, time_limit, alpha=0.3, n_iters=200, verbose=True):
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(x, fft_size)
        power = np.abs(A) ** 2
        p_max = power.max()
        if p_max < 1e-30:
            break
        p_norm = power / p_max
        A_new = A * np.power(p_norm + 1e-10, -alpha / 2.0)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        m_new = fast_autocorr(a_new).max()
        if m_new < best_max_b:
            best_max_b = m_new
            best_a = a_new.copy()
        x = a_new

    if verbose:
        J = 2.0 * n * best_max_b
        print(f"    Spectral done: J={J:.6f} (t={time.monotonic()-t0:.1f}s)")

    return best_a, best_max_b
