    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def solve_lp_full(y, n, solver):
    """Solve the full LP: min M s.t. sum(g)=1, g>=0, (y*g)[k]<=M for all k."""
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # Build constraint matrix
    # (y*g)[k] = sum_i y[k-i] * g[i] for valid i
    # We need C[k, i] = y[k-i] if 0 <= k-i < n, else 0
    size = 2 * n - 1
    C = np.zeros((size, n))
    for k in range(size):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            C[k, idx] = y[k - idx]
    
    constraints = [cp.sum(g) == 1, C @ g <= M]
    prob = cp.Problem(cp.Minimize(M), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    return np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)

Actually, I just realized something important. Let me re-examine the LP subproblem.

The FW method for minimizing f(y) = max_k (y*y)[k] subject to y in P (simplex):

At each iteration, given current point y:
1. Find a subgradient: if k* = argmax_k (y*y)[k], then sg = 2 * y_shifted[k*] where y_shifted[k*][i] = y[k*-i] (with zero padding).
2. Solve: min_g <sg, g> s.t. g in P. Solution: g = e_{argmin_i sg[i]}.
3. Take step: y_new = (1-t)*y + t*g.

But this "vanilla" FW is slow. The "accelerated" or "conservative" variants solve a different subproblem:

Conservative CW (Cutting-Workspace): Maintain a set of "cuts" (subgradients from previous iterations) and solve:
min_g max_{sg in S} <sg, g> s.t. g in P.

This is: min M s.t. <sg_j, g> <= M for all j in S, g in P.

This is exactly what the previous code does! The "active set" S contains the subgradients from the most important lags. Each subgradient sg_k corresponds to a lag k, and <sg_k, g> = 2*(y*g)[k] (approximately).

So the previous code is implementing the conservative CW variant of FW. Good.

For the away-step FW:
1. Solve the subproblem to get g (the "best" vertex).
2. Also find the "worst" vertex h (the one that maximizes <sg, v> for v in P, i.e., the vertex farthest from the optimum in some sense).
3. Try moving in direction g - h (if it reduces the objective).

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def convolve_pair(a, b):
    """Linear convolution of two arrays using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    Af = np.fft.rfft(a, fft_size)
    Bf = np.fft.rfft(b, fft_size)
    return np.fft.irfft(Af * Bf, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_sparse_matrix(y, active_set, n):
    """Build sparse constraint matrix C where C[j,i] = y[active_set[j]-i] if valid."""
    K = len(active_set)
    rows = []
    cols = []
    vals = []
    for jj, k in enumerate(active_set):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            rows.extend([jj] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(y[k - idx].tolist())
    if len(rows) == 0:
        return sp.csr_matrix((K, n))
    C = sp.csr_matrix((vals, (rows, cols)), shape=(K, n))
    return C

def lp_fw(y0, time_limit, solver, max_K=4000, initial_K=1500, verbose=True, label=""):

    Key improvements:
    - Sparse constraint matrix

            rng_local = np.random.default_rng(stagnation_restarts * 1000)
            keep = list(rng_local.choice(len(active_set), size=len(active_set) - n_pert, replace=False))
            new_adds = np.argsort(ac_curr)[::-1][:n_pert * 3]

def coordinate_polish(a0, time_limit, verbose=True, label=""):
    """
    Intensive pairwise coordinate descent.
    Moves mass from high-contributing positions to low-impact positions.
    Uses multi-scale step sizes and random restarts.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    stall = 0
    max_stall = 3000
    step_scales = [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        # Focus on top peaks
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score destinations: how much would adding mass at j affect top peaks?
            dst_scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2_i = int(k2)
                s2 = max(0, k2_i - n + 1)
                e2 = min(n - 1, k2_i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    dst_scores[idx2] += a[k2_i - idx2]

            # Prefer destinations with low scores
            candidates = np.argsort(dst_scores)[:30]

            found = False
            for dst in candidates:
                if dst == src or found:
                    continue
                for frac in step_scales:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_convolve(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

            if not found and rng.random() < 0.2:
                # Random exploration
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.001, 0.03)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            # Periodic random perturbation to escape local minima
            if stall % 300 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"    [{label}] Coord ({elapsed:.1f}s): J={2*n*best_m:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    [{label}] Coord done ({elapsed:.1f}s): J={2*n*best_m:.6f}")

def try_update(best_seq, best_score, a, label, t_global):

def propose_candidate(budget_s=1800.0):
    """
    Main search function. Multi-phase optimization with time-aware scheduling.
    Phases:
    1. LP-FW on parent (primary workhorse)
    2. Multi-n exploration
    3. LP-FW second pass on best
    4. Coordinate polish
    5. LP-FW third pass (larger active set)
    6. Final coordinate polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Total budget: {time_budget:.1f}s")

    solver = get_solver()
    best_seq, best_score = get_initial_sequence(t_global)
    n0 = len(best_seq)

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: LP-FW on parent ===
    print(f"\n=== Phase 1: LP-FW on parent (n={n0}) ===")
    t1 = min(remaining_time() * 0.20, 300.0)
    if t1 > 60:
        max_K_1 = min(5000, 4 * n0)
        a_out, _ = lp_fw(norm_sum1(best_seq), t1, solver,
                         max_K=max_K_1, initial_K=min(2000, 2 * n0),
                         verbose=True, label="P1")
        best_seq, best_score = try_update(best_seq, best_score, a_out, "P1_lp_fw", t_global)

    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n exploration ===")
    n_best = len(best_seq)
    # Try n values around the current best
    candidate_ns = sorted(set([
        n_best, n_best - 500, n_best + 500,
        n_best - 2000, n_best + 2000,
        max(3000, n_best // 2), min(12000, n_best * 2)
    ]))
    candidate_ns = [l for l in candidate_ns if 2000 <= l <= 12000]
    candidate_ns.sort(key=lambda l: abs(l - n_best))

    n_tried = 0
    for new_n in candidate_ns[:5]:
        rt = remaining_time()
        if rt < 100 or n_tried >= 3:
            break
        if new_n == len(best_seq):
            continue
        print(f"\n  Multi-n: n={new_n} (remaining={rt:.1f}s)")
        a_long = interpolate_seq(best_seq, new_n)

        # Also try a parametric initialization
        a_param = generate_initial(new_n, mode=n_tried % 5, seed=42 + new_n)

        # Use whichever has better initial score
        sc_interp = evaluate_safe(a_long)
        sc_param = evaluate_safe(a_param)
        if sc_param < sc_interp:
            a_start = a_param
            print(f"    Using parametric init (score={sc_param:.6f})")
        else:
            a_start = a_long
            print(f"    Using interpolated init (score={sc_interp:.6f})")

        t_n = min(rt * 0.15, 180.0)
        if t_n > 40:
            max_K_n = min(4000, 4 * new_n)
            a_out, _ = lp_fw(a_start, t_n, solver,
                             max_K=max_K_n, initial_K=min(1500, 2 * new_n),
                             verbose=False, label=f"P2_n{new_n}")
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"P2_n{new_n}", t_global)
        n_tried += 1

    # === Phase 3: LP-FW second pass with larger active set ===
    print(f"\n=== Phase 3: LP-FW pass 2 (n={len(best_seq)}) ===")
    t3 = min(remaining_time() * 0.20, 280.0)
    if t3 > 60:
        max_K_3 = min(6000, 5 * len(best_seq))
        a_out, _ = lp_fw(norm_sum1(best_seq), t3, solver,
                         max_K=max_K_3, initial_K=min(3000, 3 * len(best_seq)),
                         verbose=True, label="P3")
        best_seq, best_score = try_update(best_seq, best_score, a_out, "P3_lp_fw", t_global)

    # === Phase 4: Coordinate polish ===
    print(f"\n=== Phase 4: Coordinate polish ===")
    t4 = min(remaining_time() * 0.12, 150.0)
    if t4 > 30:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t4, verbose=True, label="P4")
        best_seq, best_score = try_update(best_seq, best_score, a_out, "P4_coord", t_global)

    # === Phase 5: LP-FW third pass (most aggressive) ===
    print(f"\n=== Phase 5: LP-FW final (n={len(best_seq)}) ===")
    t5 = min(remaining_time() * 0.22, 300.0)
    if t5 > 60:
        max_K_5 = min(8000, 6 * len(best_seq))
        a_out, _ = lp_fw(norm_sum1(best_seq), t5, solver,
                         max_K=max_K_5, initial_K=min(4000, 4 * len(best_seq)),
                         verbose=True, label="P5")
        best_seq, best_score = try_update(best_seq, best_score, a_out, "P5_lp_fw", t_global)

    # === Phase 6: Final coordinate polish ===
    rt = remaining_time()
    if rt > 30:
        print(f"\n=== Phase 6: Final polish ===")
        t6 = min(rt * 0.5, 150.0)
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t6, verbose=True, label="P6")
        best_seq, best_score = try_update(best_seq, best_score, a_out, "P6_coord", t_global)

    if rt > 80 and len(best_seq) <= 10000:
        print(f"\n=== Phase 7: Ultimate LP-FW ===")

        a_out, _ = lp_fw(norm_sum1(best_seq), t7, solver,
                         max_K=max_K_7, initial_K=min(5000, 5 * len(best_seq)),
                         verbose=True, label="P7")
        best_seq, best_score = try_update(best_seq, best_score, a_out, "P7_final", t_global)
