    if s < 1e-15:

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

                    a = a_pert

        if it % 500 == 0:
            alpha = max(0.01, alpha * 0.95)
            clamp_factor = max(1.02, clamp_factor * 0.98)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SpectralClamp it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " clamped=" + str(n_clamped))

    return best_a, best_m


def lp_fw_fast(a0, time_limit, solver, initial_K=200, max_K=3000,
               add_per_iter=30, verbose=True, warm_start_set=None):
    """
    Fast LP-Frank-Wolfe with aggressive active set management.
    Uses margin-based constraint selection and adaptive line search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    y = a.copy()
    ac_y = fast_autocorr(y)

    if warm_start_set is not None:
        active_set = set(int(k) for k in warm_start_set)
    else:
        active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 800
    lp_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-18:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update active set: keep constraints with smallest margins
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        # Add most violated new constraints
        if len(constraint_list) < max_K:
            # Find violated constraints not in active set
            in_active = np.zeros(size, dtype=bool)
            in_active[list(active_set)] = True
            violated = np.where((ac > M_curr - 1e-10) & ~in_active)[0]
            if len(violated) > 0:
                # Add the ones closest to the peak
                top_v = violated[np.argsort(ac[violated])[::-1][:add_per_iter]]
                for k in top_v:
                    active_set.add(int(k))

        constraint_list = sorted(active_set)
        K = len(constraint_list)
        if K < 3:
            # Add more initial constraints
            top_k = np.argsort(ac)[::-1][:initial_K]
            for k in top_k:
                active_set.add(int(k))
            constraint_list = sorted(active_set)
            K = len(constraint_list)
            if K < 3:
                continue

        # Build constraint matrix
        # conv(y, g)[k] = sum_i y[i] * g[k-i] = sum_i g[i] * y[k-i]
        # For constraint k: sum_{i=max(0,k-n+1)}^{min(n-1,k)} g[i] * y[k-i] <= M
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP: min M s.t. coeffs @ g <= M, sum(g)=1, g>=0
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        t_lp_start = time.monotonic()
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K:
                active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
            continue

        t_lp = time.monotonic() - t_lp_start
        lp_count += 1

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Add violated constraints from new direction
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_yg > M_curr + 1e-8)[0]
        if len(violations) > 0:
            top_v = violations[np.argsort(conv_yg[violations])[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Adaptive line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        alphas = [best_alpha, best_alpha * 0.5, best_alpha * 1.5,
                  1.0 / (t + 1), 0.05, 0.1, 0.15, 0.2, 0.3]
        for alpha in alphas:
            alpha = max(0.01, min(0.99, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new
        ac_y = fast_autocorr(y)

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW it=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" +
                  repr(round(J, 6)) + " |S|=" + str(len(active_set)) +
                  " lp=" + str(round(t_lp, 3)) + "s")

    return best_a, best_m, active_set


def peak_coordinate_descent(a0, time_limit, verbose=True, seed=42):
    """
    Targeted coordinate descent: directly reduce the peak autocorrelation
    by adjusting the elements that contribute most to it.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    size = 2 * n - 1

    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        peak_idx = int(np.argmax(ac))
        peak_val = ac[peak_idx]

        if peak_val < best_m * 0.999:
            best_m = peak_val
            best_a = a.copy()

        # Find contributing pairs for the peak
        k = peak_idx
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue

        idx = np.arange(i_start, i_end + 1)
        partners = k - idx
        contrib = a[idx] * a[partners]
        total = contrib.sum()
        if total < 1e-15:
            continue

        # Find the top contributors
        top_contrib = np.argsort(contrib)[::-1]
        improved = False

        for tc in top_contrib[:5]:
            i_pos = int(idx[tc])
            j_pos = int(partners[tc])
            if i_pos == j_pos:
                continue

            # Try reducing the contribution by moving mass
            # Option 1: reduce a[i_pos] and add to a random position
            frac_options = [0.005, 0.01, 0.02, 0.05]
            for frac in frac_options:
                amt = a[i_pos] * frac
                if amt < 1e-15:
                    continue

                # Try several destinations that don't contribute to the peak
                for _ in range(3):
                    dst = int(rng.integers(n))
                    if dst == i_pos or dst == j_pos:
                        continue
                    # Check if dst doesn't contribute to this peak
                    if max(0, k - n + 1) <= dst <= min(n - 1, k):
                        # dst is a partner position, skip
                        continue

                    a_trial = a.copy()
                    a_trial[i_pos] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < peak_val * (1 - 1e-10):
                        a = a_trial
                        peak_val = m_trial
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try swapping mass between two peak contributors
            if len(top_contrib) >= 2:
                i1 = int(idx[top_contrib[0]])
                j1 = int(partners[top_contrib[0]])
                i2 = int(idx[top_contrib[1]])
                j2 = int(partners[top_contrib[1]])
                if i1 != i2:
                    amt = min(a[i1], a[i2]) * 0.05
                    if amt > 1e-15:
                        a_trial = a.copy()
                        a_trial[i1] -= amt
                        a_trial[i2] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial / s_t
                            m_trial = fast_autocorr(a_trial).max()
                            if m_trial < peak_val:
                                a = a_trial
                                peak_val = m_trial
                                improved = True

        if not improved and it % 50 == 0:
            # Random perturbation for escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    m_new = fast_autocorr(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakCD it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def smoothed_nesterov(a0, time_limit, beta_start=30.0, beta_end=1000.0,
                      beta_growth=1.004, eta_init=0.4, top_k_ratio=0.02,
                      restart_every=1000, verbose=True, seed=42):
    """Nesterov with smoothed max and diverse restarts."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    eta = eta_init
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        b = fast_autocorr(y)
        b_max = b.max()

        if top_k < size:
            idx_top = np.argsort(b)[::-1][:top_k]
            b_use = np.zeros(size)
            b_use[idx_top] = b[idx_top]
            b_max_use = b[idx_top].max()
        else:
            b_use = b
            b_max_use = b_max

        if b_max_use <= 0:
            grad = np.zeros(n)
        else:
            shifted = beta * (b_use - b_max_use)
            shifted = np.clip(shifted, -500, 0)
            exp_vals = np.exp(shifted)
            s_exp = exp_vals.sum()
            if s_exp < 1e-300:
                p = np.ones(size) / size
            else:
                p = exp_vals / s_exp

            # Gradient: d/dy_k [sum_l p_l * conv(y,y)[l]]
            # conv(y,y)[l] = sum_i y[i]*y[l-i]
            # d/dy_k = sum_l p_l * (y[l-k] + y[k-l]) (when valid)
            # = sum_l p_l * y[|l-k|] (adjusted for boundaries)
            # More precisely: sum_l p_l * (y[l-k] + y[k-l]) where y[j]=0 for j<0 or j>=n
            # = sum_l p_l * y[l-k] + sum_l p_l * y[k-l]
            # Let m = l-k: first term = sum_m p[m+k] * y[m]
            # Let m = k-l: second term = sum_m p[k-m] * y[m]
            # So grad[k] = sum_m y[m] * (p[m+k] + p[k-m])
            # This is a convolution: grad = convolve(y, p_relevant)
            # Specifically: grad[k] = sum_m y[m] * (p[m+k] + p[k-m])
            # = (y * p_flip)[k] + (y * p)[k] where p_flip[m] = p[-m]... 
            # Actually let's just compute it directly via FFT
            y_rev = y[::-1]
            conv_len = n + size - 1
            cs_fft_size = 1 << (conv_len - 1).bit_length()
            Ar = np.fft.rfft(y_rev, cs_fft_size)
            Pr = np.fft.rfft(p, cs_fft_size)
            conv_result = np.fft.irfft(Ar * Pr, cs_fft_size)
            grad = conv_result[n - 1:2 * n - 1] / beta

        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            x_new = x.copy()
            max_new = fast_autocorr(x_new).max()
        else:
            eta_trial = eta
            found = False
            for _ in range(8):
                z = y - eta_trial * grad
                x_new = simplex_project(z)
                max_new = fast_autocorr(x_new).max()
                if max_new <= b_max + 1e-12:
                    found = True
                    break
                eta_trial *= 0.5
            if not found:
                x_new = x
                max_new = b_max
                eta = max(eta_trial, 0.01)

        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.05, 1.5)
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 40 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = rng.integers(6)
            if strategy == 0:
                pert = rng.normal(0, 0.015, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                x_r = 0.85 * best_a + 0.15 * rng.dirichlet(np.ones(n) * 0.2)
            elif strategy == 2:
                x_r = best_a * np.exp(0.02 * rng.standard_normal(n))
            elif strategy == 3:
                mask = rng.random(n) > 0.6
                x_r = best_a.copy()
                x_r[mask] *= 0.8
                x_r[~mask] *= 1.2
            elif strategy == 4:
                shift = int(rng.integers(1, max(2, n // 20)))
                x_r = np.roll(best_a, shift)
            else:
                # Blend with spectral solution
                A = np.fft.rfft(best_a, fft_size)
                power = np.abs(A) ** 2
                mean_p = power[1:].mean()
                # Flatten
                A_flat = A.copy()
                for k in range(1, len(A_flat)):
                    if power[k] > 2 * mean_p:
                        A_flat[k] *= np.sqrt(2 * mean_p / power[k])
                a_spec = np.fft.irfft(A_flat, fft_size)[:n]
                a_spec = np.maximum(a_spec, 0)
                s_s = a_spec.sum()
                if s_s > 1e-15:
                    a_spec = a_spec / s_s
                    x_r = 0.5 * best_a + 0.5 * a_spec
                else:
                    x_r = best_a.copy()

            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 15.0)
            eta = eta_init
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nesterov restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nesterov t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " b=" + str(int(beta)))

    return best_a, best_max_b


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Multi-strategy optimization combining:
    - Spectral peak clamping
    - LP-Frank-Wolfe (primary workhorse)
    - Peak-targeted coordinate descent
    - Smoothed Nesterov
    - Multi-length exploration
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(round(p_score, 6)))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(round(best_score, 6)))

    n0 = len(best_seq)
    print("Start: n=" + str(n0) + ", J=" + repr(round(best_score, 6)))

    # === Phase 1: Quick spectral clamping ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Spectral Clamping (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.05, 70.0)
    if t1 > 15:
        a_out, _ = spectral_clamp(norm_sum1(best_seq), t1, verbose=True, seed=123)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec1", t_global)

    # === Phase 2: LP-FW (primary) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P2: LP-FW (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.22, 380.0)
    active_set = None
    if t2 > 30 and n_best <= 12000:
        a_out, _, active_set = lp_fw_fast(norm_sum1(best_seq), t2, solver,
                                          initial_K=250, max_K=3000,
                                          add_per_iter=30, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # === Phase 3: Peak coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P3: Peak CD (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 15:
        a_out, _ = peak_coordinate_descent(norm_sum1(best_seq), t3, verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd1", t_global)

    # === Phase 4: Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P4: Nesterov (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        a_out, _ = smoothed_nesterov(norm_sum1(best_seq), t4,
                                     beta_start=30.0, beta_end=800.0,
                                     beta_growth=1.004, eta_init=0.4,
                                     top_k_ratio=0.02, restart_every=800,
                                     verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # === Phase 5: LP-FW round 2 (warmer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P5: LP-FW r2 (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30 and n_best <= 12000:
        a_out, _, active_set = lp_fw_fast(norm_sum1(best_seq), t5, solver,
                                          initial_K=300, max_K=4000,
                                          add_per_iter=40, verbose=True,
                                          warm_start_set=active_set)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp2", t_global)

    # === Phase 6: Peak CD round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P6: Peak CD r2 (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.06, 80.0)
    if t6 > 10:
        a_out, _ = peak_coordinate_descent(norm_sum1(best_seq), t6, verbose=True, seed=789)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd2", t_global)

    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P7: Multi-length (base n=" + str(n_best) + ") ===")

    # Try a few different lengths
    test_lengths = []
    if n_best < 8000:
        test_lengths.append(n_best + 500)
    if n_best > 4000:
        test_lengths.append(n_best - 500)
    if n_best < 10000:
        test_lengths.append(n_best + 1000)
    test_lengths = [l for l in test_lengths if 2000 <= l <= 12000]

    for new_n in test_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print("  -> n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick Nesterov on new length
        t7a = min(remaining * 0.30, 150.0)
        if t7a > 20:
            a_out, _ = smoothed_nesterov(a_long, t7a,
                                         beta_start=40.0, beta_end=800.0,
                                         beta_growth=1.004, eta_init=0.3,
                                         top_k_ratio=0.015, restart_every=800,
                                         verbose=False, seed=new_n % 7919)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(new_n), t_global)

        # Quick LP-FW on new length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 10000:
            t7b = min(remaining * 0.40, 150.0)
            if t7b > 20:
                a_long2 = interpolate_sequence(best_seq, new_n)
                a_out, _, _ = lp_fw_fast(a_long2, t7b, solver,
                                         initial_K=250, max_K=2500,
                                         add_per_iter=30, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(new_n), t_global)

    # === Phase 8: Final polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P8: Final polish (n=" + str(n_best) + ") ===")

    # Quick spectral clamp
    if remaining > 20:
        t8a = min(remaining * 0.15, 60.0)
        if t8a > 10:
            a_out, _ = spectral_clamp(norm_sum1(best_seq), t8a, verbose=False, seed=555)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "spec_final", t_global)

    # Quick peak CD
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t8b = min(remaining * 0.20, 60.0)
        if t8b > 8:
            a_out, _ = peak_coordinate_descent(norm_sum1(best_seq), t8b, verbose=False, seed=999)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd_final", t_global)

    # Final LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and n_best <= 12000:
        t8c = min(remaining * 0.60, 200.0)
        if t8c > 15:
            a_out, _, _ = lp_fw_fast(norm_sum1(best_seq), t8c, solver,
                                     initial_K=400, max_K=5000,
                                     add_per_iter=50, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: J=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
