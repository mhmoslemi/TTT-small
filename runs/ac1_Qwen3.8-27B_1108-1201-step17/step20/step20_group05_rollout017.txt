def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)))
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)

def apply_single_move(c, a, s, d, tmove):
    """Fast rank-1 update of convolution when moving mass from s to d."""
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = a.size
    c_new = c.copy()
    # Moving tmove from position s to position d:
    # a_new[s] = a[s] - tmove, a_new[d] = a[d] + tmove
    # b_new = b + 2*tmove * shift(a, d) - 2*tmove * shift(a, s) + tmove^2 * (delta_d - delta_s)^2
    # where shift(a, k) means the convolution contribution from a at offset k
    
    # b_k = sum_i a_i * a_{k-i}
    # After move: a[s] -> a[s]-t, a[d] -> a[d]+t
    # Delta b_k = 2t * (a_{k-d} - a_{k-s}) + t^2 * (delta_{k,2d} - 2*delta_{k,s+d} + delta_{k,2s})
    # where a_{k-d} is valid only if 0 <= k-d < n
    
    # The cross terms: 2t * a_{k-d} (for the +t at d) and -2t * a_{k-s} (for the -t at s)
    # a_{k-d} is valid when 0 <= k-d < n, i.e., d <= k < d+n
    # a_{k-s} is valid when 0 <= k-s < n, i.e., s <= k < s+n
    
    c_new[d:d+n] += 2.0 * tmove * a
    c_new[s:s+n] -= 2.0 * tmove * a
    
    t2 = tmove * tmove
    if 2 * d < c_new.size:
        c_new[2 * d] += t2
    if 2 * s < c_new.size:
        c_new[2 * s] += t2
    if s + d < c_new.size:
        c_new[s + d] -= 2.0 * t2
    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
    return c_new

                    if i == j:

                    else:

        # The "effective gradient" for the max at position k is:
        # grad_i = sum over active k of a_{k-i} (for i where k-i is in [0,n))
        # This tells us which coordinates contribute to the peak.
        # We want to REDUCE contributions from high a_i to the peak,
        # i.e., move mass from positions i where a_{k-i} is large (for active k).

        # High pressure at position i means a[i] contributes a lot to the peaks.
        
        pressure = np.zeros(n, dtype=np.float64)

            s = max(0, k - n + 1)
            e = min(n, k + 1)

                pressure[i] += a[j]

        pmax = float(np.max(pressure)) if np.max(pressure) > 0 else 1.0
        pressure = pressure / pmax

        # Destination: low pressure AND low mass → receiving it hurts least
        src_score = pressure * a
        dst_score = pressure * (1.0 - a)  # Low pressure regions

        n_src = min(20, n)
        n_dst = min(30, n)
        src_candidates = np.argpartition(src_score, n_src - 1)[:n_src] if n > n_src else np.arange(n)
        dst_candidates = np.argpartition(-dst_score, n_dst - 1)[:n_dst] if n > n_dst else np.arange(n)

        best_new_M = M

            s = int(s)
            if a[s] < 1e-14:

                d = int(d)

                    m_new = float(np.max(c_new))
                    if m_new < best_new_M:
                        best_new_M = m_new
                        best_move = (s, d, tmove)
        
        if best_move is not None:
            s, d, tmove = best_move
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            ss = float(a.sum())

            c = fft_conv(a)
            M = float(np.max(c))
            stall = 0
            if M < best_M - 1e-14:

                best_a = a.copy()
            if verbose and (iteration <= 5 or iteration % 50 == 0):

                      " M=" + repr(best_M) +
                      " score=" + repr(2*n*best_M) +
                      " move=(" + str(s) + "," + str(d) + "," + repr(tmove) + ")")
        else:

                idx = rng.integers(0, n, n_perturb)

                a = np.maximum(a, 0.0)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = float(np.max(c))
                stall = 0
    
    return best_a

Wait, I realize the pressure computation is O(n * top_k) which for n=5000 and top_k=50 is 250,000 operations per iteration. That's fine.

But actually, I can compute the pressure more efficiently using FFT. The pressure at position `i` from peak at position `k` is `a[k-i]` (if valid). This is essentially a cross-correlation of `a` with the indicator of the active peaks.

Let me use FFT for this:

```python
def compute_pressure(a, top_indices):
    """Compute pressure field using FFT."""
    n = a.size
    size = 2 * n - 1
    total = 3 * n - 2
    N = next_pow2(total)
    
    # Create indicator of active peaks
    p = np.zeros(size, dtype=np.float64)
    for k in top_indices:
        p[int(k)] = 1.0
    
    # cross_convolve: pressure[i] = sum_k p[k] * a[k-i] (where valid)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    pressure = conv[n-1:2*n-1][::-1]
    pressure = np.where(np.isfinite(pressure), pressure, 0.0)
    return pressure
```

This is O(n log n) instead of O(n * top_k).

**Direct convex optimization reformulation:**

The problem `min max_k b_k s.t. sum a = 1, a >= 0` can be written as:
`min t s.t. b_k <= t, sum a = 1, a >= 0`

The constraint `b_k <= t` is `sum_i a_i * a_{k-i} <= t`, which is a quadratic constraint (not linear). So this is a non-convex QP (quadratically constrained program).

However, we can use the following trick: introduce variables `x_{i,k} = a_i * a_{k-i}` and relax to `x_{i,k} >= 0, sum_i x_{i,k} <= t`. The relaxation is valid but weak. The LP relaxation gives a lower bound.

Actually, a better approach: the problem is a "quadratic programming with a max constraint". We can use a Dinkelbach-type parametrization:

For a given `t`, check if there exists `a >= 0, sum a = 1` with all `b_k <= t`. This is a feasibility problem with quadratic constraints, which is still non-convex.

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n) / n
    return arr / s

def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c

def fft_conv2(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if b.size != n:
        b = np.resize(b, n)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c

def simulated_annealing_coord(a0, budget, top_k=30, T_init=0.01, T_min=0.001, verbose=True):
    """Coordinate descent with simulated annealing acceptance."""
    t0 = time.monotonic()
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    rng = np.random.default_rng(999)
    T = T_init
    n_iter = 0
    total_iters_estimate = int(budget / 0.002)  # rough estimate
    
    while time.monotonic() - t0 < budget - 3:
        n_iter += 1
        # Cool down
        progress = min(1.0, n_iter / max(1, total_iters_estimate))
        T = T_init * (T_min / T_init) ** progress
        
        top_indices = np.argsort(c)[::-1][:top_k]
        pressure = compute_pressure(a, top_indices)
        
        # Pick source and destination
        # Weight by pressure for source, inverse pressure for destination
        src_weights = np.maximum(pressure * a, 1e-30)
        dst_weights = np.maximum(0.01 + 1.0 - pressure, 1e-30)
        total_src = float(src_weights.sum())
        total_dst = float(dst_weights.sum())
        src_prob = src_weights / total_src
        dst_prob = dst_weights / total_dst
        
        s = int(rng.choice(n, p=src_prob))
        d = int(rng.choice(n, p=dst_prob))
        
        if s == d or a[s] < 1e-12:
            # Retry
            s = int(rng.integers(n))
            d = int(rng.integers(n))
            if s == d or a[s] < 1e-12:
                continue
        
        frac = float(rng.uniform(0.01, 0.15))
        tmove = a[s] * frac
        tmove = min(tmove, a[s] * 0.8)
        if tmove < 1e-16:
            continue
        
        c_new = apply_single_move(c, a, s, d, tmove)
        m_new = float(np.max(c_new))
        delta = m_new - M
        
        if delta < 0 or (T > T_min and rng.random() < math.exp(-delta / T)):
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            a = a / ss
            c = c_new  # Use the pre-computed convolution (approximately correct after renormalization)
            # Actually renormalization changes things slightly, so recompute occasionally
            if n_iter % 10 == 0:
                c = fft_conv(a)
            M = float(np.max(c))
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
        
        if verbose and n_iter % 1000 == 0:
            elapsed = time.monotonic() - t0
            print("  SA iter=" + str(n_iter) +
                  " M=" + repr(best_M) +
                  " score=" + repr(2*n*best_M) +
                  " T=" + repr(T) +
                  " elapsed=" + repr(elapsed))

def lsp_smoothed(a0, beta, maxfun=200):
    """L-BFGS on log-sum-exp smoothed objective."""
    if not SCIPY_OK:
        return a0
    a = norm_sum1(a0)
    n = a.size
    size = 2 * n - 1
    
    def obj_grad(v):
        # v is the log-space parameterization
        a_local = np.exp(v - np.max(v))
        a_local = a_local / a_local.sum()
        c = fft_conv(a_local)
        m = float(np.max(c))
        shifted = beta * (c - m)
        shifted = np.clip(shifted, -500, 500)
        e = np.exp(shifted)
        se = float(e.sum())
        if se < 1e-30:
            f = 1e100
            return f, np.zeros(n)
        p = e / se
        f = m + math.log(se) / beta
        
        # Gradient w.r.t. a
        total = 3 * n - 2
        N = next_pow2(total)
        A = np.fft.rfft(a_local, N)
        Pr = np.fft.rfft(p[::-1], N)
        conv = np.fft.irfft(A * Pr, N)[:total]
        ga = 2.0 * conv[n-1:2*n-1][::-1]
        ga = np.where(np.isfinite(ga), ga, 0.0)
        
        # Chain rule through softmax-like normalization
        mean = float(np.dot(a_local, ga))
        # v -> a: a_i = exp(v_i) / sum(exp(v_j))
        # da_i/dv_j = a_i * (delta_ij - a_j)
        # So grad_v = a * (ga - dot(a, ga)) = a * (ga - mean)
        gv = a_local * (ga - mean)
        gv = np.where(np.isfinite(gv), gv, 0.0)
        
        # Add small regularization to avoid flat directions
        gv = gv + 1e-10 * v
        f += 0.5e-10 * float(np.dot(v, v))
        
        if not math.isfinite(f):
            f = 1e100
            gv = np.zeros(n)
        return f, gv
    
    v0 = np.log(np.maximum(a, 1e-300))
    v0 = v0 - np.max(v0)
    
    try:
        res = scipy.optimize.minimize(
            obj_grad, v0, method='L-BFGS-B', jac=True,
            options={'maxfun': maxfun, 'maxiter': maxfun + 50,
                     'ftol': 1e-12, 'gtol': 1e-7, 'disp': False}
        )
        if res is not None and hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            a_out = np.exp(res.x - np.max(res.x))
            a_out = a_out / a_out.sum()
            a_out = np.where(np.isfinite(a_out), a_out, 0.0)
            return a_out
    except:
        pass
    return a0

def coarse_to_fine(a0, budget, factor=4, verbose=True):
    """Multi-scale refinement: optimize coarse version, then refine."""
    t0 = time.monotonic()
    a = norm_sum1(a0)
    n = a.size
    best_a = a.copy()
    
    # Coarsen
    coarse_n = max(100, n // factor)
    a_coarse = interpolate_sequence(a, coarse_n)
    
    # Optimize coarse version quickly
    time_left = budget - (time.monotonic() - t0)
    coarse_budget = max(5.0, time_left * 0.3)
    a_coarse_opt = pressure_descent(a_coarse, coarse_budget, top_k=20, verbose=False)
    
    # Interpolate back
    a_interp = interpolate_sequence(a_coarse_opt, n)
    
    # Refine at full resolution
    time_left = budget - (time.monotonic() - t0)
    fine_budget = max(5.0, time_left * 0.7)
    a_fine = pressure_descent(a_interp, fine_budget, top_k=30, verbose=False)
    
    # Also try the original (coarse-to-fine might not help)
    time_left = budget - (time.monotonic() - t0)
    if time_left > 5:
        a_orig = pressure_descent(a, min(10.0, time_left), top_k=30, verbose=False)
        # Compare
        M1 = float(np.max(fft_conv(a_fine)))
        M2 = float(np.max(fft_conv(a_orig)))
        if M1 < M2:
            return a_fine
        else:
            return a_orig
    
    return a_fine

def diversity_restart(a0, n_restarts=5, budget_per=20.0, verbose=True):
    """Try diverse perturbations and keep the best."""
    a = norm_sum1(a0)
    n = a.size
    best_a = a.copy()
    best_M = float(np.max(fft_conv(a)))
    rng = np.random.default_rng(777)
    
    for r in range(n_restarts):
        # Different perturbation strategies
        if r == 0:
            # Gaussian noise
            a_pert = a + rng.normal(0, 0.05, n) * a
        elif r == 1:
            # Shift (circular)
            shift = rng.integers(1, max(2, n//10))
            a_pert = np.roll(a, int(shift))
        elif r == 2:
            # Scale inversion (emphasis on tails)
            x = np.linspace(0, 1, n)
            a_pert = a * (x**2 + 0.1)
        elif r == 3:
            # Block swap
            bsize = max(5, n // 20)
            i = rng.integers(0, n - 2*bsize)
            i = int(i)
            a_pert = a.copy()
            a_pert[i:i+bsize], a_pert[i+2*bsize:i+3*bsize] = a_pert[i+2*bsize:i+3*bsize].copy(), a_pert[i:i+bsize].copy()
        else:
            # Sparse randomization
            mask = rng.random(n) < 0.1
            a_pert = a.copy()
            a_pert[mask] = rng.random(int(mask.sum()) + 1)
        
        a_pert = norm_sum1(a_pert)
        a_opt = pressure_descent(a_pert, budget_per, top_k=25, verbose=False)
        M = float(np.max(fft_conv(a_opt)))
        if M < best_M:
            best_M = M
            best_a = a_opt.copy()
    
    return best_a
