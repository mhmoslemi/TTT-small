def _next_pow2(x):
    x = int(max(1, x))

def _norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    arr[~np.isfinite(arr)] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s

def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = _next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c[~np.isfinite(c)] = 0.0
    np.maximum(c, 0.0, out=c)
    return c

def _fft_conv2(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)

    return c

def _interpolate_seq(a, new_n):

def _lse_objective(v, beta, mask, ref):
    """LSE-smoothed objective in softmax reference coordinates."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    if n <= 1:
        return 0.0, np.array([], dtype=np.float64)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1

    u = np.empty(n, dtype=np.float64)
    u[ref] = 0.0
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]

    um = float(np.max(u))
    if not math.isfinite(um):
        return 1e100, np.zeros(v.size, dtype=np.float64)
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return 1e100, np.zeros(v.size, dtype=np.float64)
    a = e / s

    size = 2 * n - 1
    total = 3 * n - 2
    N = _next_pow2(total)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c[~np.isfinite(c)] = 0.0
    np.maximum(c, 0.0, out=c)

    if mask is None:
        cv = c
    else:
        cv = c[mask]
    if cv.size == 0:
        f = 0.0
        grad = np.zeros(n, dtype=np.float64)
    else:
        m = float(np.max(cv))
        shifted = beta * (cv - m)
        shifted = np.clip(shifted, -700.0, 700.0)
        expv = np.exp(shifted)
        se = float(expv.sum())
        if se <= 0 or not math.isfinite(se):
            expv = np.zeros(cv.size)
            expv[0] = 1.0
            se = 1.0
        p = expv / se

        pf = np.zeros(size, dtype=np.float64)
        if mask is None:
            pf[:] = p
        else:
            pf[mask] = p

        Pr = np.fft.rfft(pf[::-1], N)
        conv = np.fft.irfft(A * Pr, N)[:total]
        ga = 2.0 * conv[n - 1:2 * n - 1][::-1]
        ga[~np.isfinite(ga)] = 0.0
        f = m + math.log(se) / beta

    mean = float(np.dot(a, ga))

    gv[~np.isfinite(gv)] = 0.0

                a_out = _norm_sum1(a_out)
                if verbose:
                    print("    L-BFGS beta={} n={} maxfun={} success={}".format(
                        beta, n, maxfun, getattr(res, 'success', False)))
                return a_out
    except Exception as e:
        if verbose:
            print("    L-BFGS error: {}".format(e))
    return a

def _lbfegs_curriculum(a0, budget, mask):
    """Multi-scale L-BFGS with increasing beta."""
    a = _norm_sum1(a0)
    n = a.size
    c = _fft_conv(a)
    best_a = a.copy()
    best_M = float(np.max(c)) if c.size > 0 else 0.0

    schedules = [
        (5.0, 0),
        (15.0, 0),
        (40.0, n // 5),
        (100.0, n // 8),
        (300.0, n // 12),
        (800.0, n // 20),
        (2000.0, n // 30),
        (5000.0, n // 50),
    ]

    t0 = time.monotonic()
    # Estimate cost
    v0, ref0 = _make_v(a)
    if v0.size > 0:
        t1 = time.monotonic()
        _lse_objective(v0, 30.0, None, ref0)
        cost = max(0.01, time.monotonic() - t1)
    else:
        cost = 0.1

    total_evals = int(budget / max(2.0 * cost, 0.05))
    n_schedules = len(schedules)
    per_run = int(max(50, min(500, total_evals // n_schedules)))

    for beta, top_k in schedules:
        if time.monotonic() - t0 > budget - 5:
            break
        if top_k > 0:
            c_cur = _fft_conv(a)
            msk = _top_k_mask(c_cur, mask, top_k)
        else:
            msk = mask
        a_new = _run_lbfegs(a, beta, msk, per_run, verbose=False)
        c_new = _fft_conv(a_new)
        M_new = float(np.max(c_new)) if c_new.size > 0 else 1e100
        if math.isfinite(M_new):
            if M_new <= best_M * (1.0 + 1e-4) + 1e-18:
                a = a_new
            if M_new < best_M - 1e-15:
                best_M = M_new
                best_a = a_new.copy()
                print("    Curriculum beta={}: M={:.10f} score={:.8f}".format(
                    beta, best_M, 2.0 * n * best_M))

    return best_a

def _spectral_perturb(a, sigma, rng):
    """Apply a random spectral perturbation and project back to valid set."""
    n = a.size
    size = 2 * n - 1
    N = _next_pow2(size)
    A = np.fft.rfft(a, N)
    n_bins = A.size

    # Pick a random contiguous band of frequency bins
    band_width = max(2, n_bins // 20)
    start = int(rng.integers(0, max(1, n_bins - band_width)))
    band = slice(start, min(start + band_width, n_bins))

    # Random amplitude and phase perturbation
    amp_pert = 1.0 + sigma * rng.standard_normal(n_bins)
    amp_pert[band] = 1.0 + 3.0 * sigma * rng.standard_normal(band_width)
    amp_pert[~np.isfinite(amp_pert)] = 1.0
    amp_pert = np.maximum(amp_pert, 0.1)

    new_A = A * amp_pert
    new_a_full = np.fft.irfft(new_A, N)[:n]
    new_a_full[~np.isfinite(new_a_full)] = 0.0
    np.maximum(new_a_full, 0.0, out=new_a_full)
    s = float(new_a_full.sum())
    if s < 1e-15:
        return a.copy()
    return new_a_full / s

def _spectral_exploration(a0, budget, num_restarts=5, verbose=True):
    """Simulated annealing with spectral perturbations."""
    a = _norm_sum1(a0)
    n = a.size
    c = _fft_conv(a)
    M = float(np.max(c)) if c.size > 0 else 1e100
    best_a = a.copy()
    best_M = M
    t_ref = 2.0 * n * best_M

    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    # Temperature schedule: start high, decay
    T_start = 0.05
    T_end = 0.001

    no_improve = 0
    iter_count = 0
    sigma_base = 0.02

    while time.monotonic() - t0 < budget - 3:
        iter_count += 1
        frac = min(1.0, iter_count / max(1, int(budget * 3)))
        T = T_start * (T_end / T_start) ** frac
        sigma = sigma_base * (1.0 - 0.5 * frac)

        # Randomly choose: gradient step vs spectral move
        r = rng.random()
        if r < 0.4:
            # Gradient step (small L-BFGS run)
            v, ref = _make_v(a)
            if v.size > 0:
                a_new = _run_lbfegs(a, 500.0, None, 30, verbose=False)
            else:
                a_new = a.copy()
        elif r < 0.7:
            # Single-peak gradient step
            c_cur = _fft_conv(a)
            peak_idx = int(np.argmax(c_cur))
            # Compute gradient for this peak
            n_local = a.size
            grad = np.zeros(n_local, dtype=np.float64)
            k = peak_idx
            for i in range(n_local):
                j = k - i
                if 0 <= j < n_local:
                    grad[i] += a[j]
                    if i != j:
                        grad[i] += a[j]
            # Move in negative gradient direction
            step = 0.05 / max(1.0, float(np.max(np.abs(grad))))
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = float(a_new.sum())
            if s > 1e-15:
                a_new = a_new / s
            else:
                a_new = a.copy()
        else:
            # Spectral perturbation
            a_new = _spectral_perturb(a, sigma, rng)

        c_new = _fft_conv(a_new)
        M_new = float(np.max(c_new)) if c_new.size > 0 else 1e100
        delta = M_new - M

        # Accept/reject based on temperature
        if delta < 0:
            accept = True
        else:
            accept = rng.random() < math.exp(-delta / max(T, 1e-10))

        if accept:
            a = a_new
            c = c_new
            M = M_new
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
                no_improve = 0
                if verbose and iter_count % 50 == 0:
                    print("    Spectral t={}: score={:.8f} T={:.6f}".format(
                        iter_count, 2.0 * n * best_M, T))
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Periodic restart if stuck
        if no_improve > 200:
            if verbose:
                print("    Spectral restart at iter {}, best_score={:.8f}".format(
                    iter_count, 2.0 * n * best_M))
            a = _spectral_perturb(best_a, 0.15, rng)
            c = _fft_conv(a)
            M = float(np.max(c))
            no_improve = 0

    return best_a

def _coordinate_polish(a0, budget, verbose=False):
    """Exact coordinate descent with rank-1 convolution updates."""
    a = _norm_sum1(a0)
    n = a.size
    c = _fft_conv(a)
    M = float(np.max(c)) if c.size > 0 else 1e100
    best_a = a.copy()
    best_M = M
    n_per_2 = 2 * n - 1

    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    rnd = 0

    def apply_move(c_arr, a_arr, s, d, tmove):
        """Apply a single transfer move using rank-1 update."""
        c_new = c_arr.copy()
        # When we move tmove from s to d:
        # a_new[s] = a[s] - tmove, a_new[d] = a[d] + tmove
        # Delta in convolution: -tmove*a + tmove*a shifted
        # c_new[k] = c[k] - tmove*(a shifted to lag s) - tmove*(a shifted to lag s)
        #            + tmove*(a shifted to lag d) + tmove*(a shifted to lag d)
        # Plus the second-order term: -tmove^2*(delta at 2s - 2*delta at s+d + delta at 2d)
        t2 = tmove * tmove
        # c_new[k] += 2*tmove*(delta_d - delta_s) * a[k - offset]
        # More precisely:
        # (a_new * a_new)[k] = (a*a)[k]
        #   + 2*tmove * (a shifted by d - a shifted by s) convolved with a
        #   + tmove^2 * (delta_2d - 2*delta_{s+d} + delta_{2s})
        for i in range(n):
            ks = s + i
            kd = d + i
            if ks < n_per_2:
                c_new[ks] -= 2.0 * tmove * a_arr[i]
            if kd < n_per_2:
                c_new[kd] += 2.0 * tmove * a_arr[i]
        if 2 * s < n_per_2:
            c_new[2 * s] += t2
        if 2 * d < n_per_2:
            c_new[2 * d] += t2
        if s + d < n_per_2:
            c_new[s + d] -= 2.0 * t2
        c_new[~np.isfinite(c_new)] = 0.0
        np.maximum(c_new, 0.0, out=c_new)
        return c_new

    while time.monotonic() - t0 < budget - 2 and stall < 200:
        rnd += 1
        if n <= 1:
            break

        # Identify top peaks for targeting
        top_k = min(30, n_per_2)
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        top_set = set(int(x) for x in top_idx)

        # Compute pressure at each position
        pressure = np.zeros(n, dtype=np.float64)
        for k in top_idx:
            k = int(k)
            s_start = max(0, k - n + 1)
            e_end = min(n, k + 1)
            for i in range(s_start, e_end):
                pressure[i] += 2.0 * a[k - i] if k - i != i else a[k - i]

        # Find good source (high pressure) and destination (low pressure) candidates
        src_candidates = np.argsort(pressure * (1.0 + n * a))[-min(10, n):]
        dst_candidates = np.argsort(pressure + 0.01 * a)[:min(15, n)]

        best_move = None
        best_M_move = M

        for s in src_candidates:
            s = int(s)
            if a[s] < 1e-14:
                continue
            for d in dst_candidates:
                d = int(d)
                if d == s:
                    continue
                for frac in (0.01, 0.02, 0.05, 0.1):
                    tmove = a[s] * frac
                    tmove = min(tmove, a[s] * 0.99)
                    if tmove <= 1e-16:
                        continue
                    c_new = apply_move(c, a, s, d, tmove)
                    m_new = float(np.max(c_new))
                    if m_new < best_M_move - 1e-18:
                        best_M_move = m_new
                        best_move = (s, d, tmove, m_new)

        # Also try random moves
        for _ in range(20):
            s = int(rng.integers(n))
            d = int(rng.integers(n))
            if s == d or a[s] < 1e-14:
                continue
            frac = float(rng.uniform(0.005, 0.08))
            tmove = min(a[s] * frac, a[s] * 0.99)
            if tmove <= 1e-16:
                continue
            c_new = apply_move(c, a, s, d, tmove)
            m_new = float(np.max(c_new))
            if m_new < best_M_move - 1e-18:
                best_M_move = m_new
                best_move = (s, d, tmove, m_new)

        if best_move is not None:
            s, d, tmove, m_new = best_move
            a[s] -= tmove
            a[d] += tmove
            np.maximum(a, 0.0, out=a)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = _fft_conv(a)
            M = float(np.max(c))
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
                stall = 0
                if verbose and rnd % 50 == 0:
                    print("    Coord rnd={}: score={:.8f}".format(rnd, 2.0 * n * best_M))
            else:
                stall = 1
        else:
            stall += 1
            if stall % 20 == 0:
                # Perturb and retry
                a = best_a.copy()
                pert = rng.normal(0, 0.01, n) * a
                a = np.maximum(a + pert, 1e-14)
                a = a / a.sum()
                c = _fft_conv(a)
                M = float(np.max(c))
                stall = 5

    return best_a

def _frank_wolfe_step(a, active_idx, mw_iters=20, seed=0):
    """Solve the FW subproblem approximately via a multiplicative-weights game."""
    f = _norm_sum1(a)
    n = f.size
    size = 2 * n - 1
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K <= 0:
        return np.ones(n) / float(n)

    rng = np.random.default_rng(seed)
    logits = rng.normal(0, 0.1, K)

    def norm_logits(loc):
        loc = np.where(np.isfinite(loc), loc, -1000.0)
        loc = loc - float(np.max(loc))
        w = np.exp(loc)
        s = float(w.sum())
        if s <= 0 or not math.isfinite(s):
            return np.ones(K) / float(K)
        return w / s

    lambda_w = norm_logits(logits)
    g = np.ones(n) / float(n)

    for _ in range(max(1, mw_iters)):
        pf = np.zeros(size, dtype=np.float64)
        pf[active_idx] = lambda_w

        # pressure_i = sum_k pf[k] * f[|k-i|] ... use cross-conv trick
        total = 3 * n - 2
        N = _next_pow2(total)
        F = np.fft.rfft(f, N)
        Pr = np.fft.rfft(pf[::-1], N)
        conv = np.fft.irfft(F * Pr, N)[:total]
        pressure = conv[n - 1:2 * n - 1][::-1]
        pressure = np.where(np.isfinite(pressure) & (pressure > 0), pressure, 0.0)

        pmx = float(np.max(pressure)) if pressure.size > 0 else 0.0
        pmn = float(np.min(pressure)) if pressure.size > 0 else 0.0
        if pmx - pmn > 1e-30:
            scaled = (pressure - pmn) / (pmx - pmn)
            w = np.exp(-10.0 * scaled)
            w = np.where(np.isfinite(w), w, 0.0)
            w += 1e-12
            g = w / float(w.sum())
            g = 0.95 * g + 0.05 * (np.ones(n) / float(n))
            g = g / float(g.sum())
        else:
            g = np.ones(n) / float(n)

        # Update lambda
        h = _fft_conv2(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        if ha.size > 0:
            hmac_ = float(np.max(ha))
            hmin = float(np.min(ha))
            if hmac_ - hmin > 1e-30:
                eta = 15.0 / (hmac_ - hmin)
                logits += eta * (ha - float(np.mean(ha)))
        lambda_w = norm_logits(logits)

    return g

def _fw_search(a0, budget, verbose=False):
    """Frank-Wolfe with adaptive active set."""
    a = _norm_sum1(a0)
    n = a.size
    c = _fft_conv(a)
    M = float(np.max(c)) if c.size > 0 else 1e100
    best_a = a.copy()
    best_M = M
    active_max = min(2000, 2 * n - 1)
    active = np.argsort(c)[::-1][:active_max]

    t0 = time.monotonic()
    t = 0
    no_improve = 0

    while time.monotonic() - t0 < budget - 3 and no_improve < 100:
        t += 1
        time_left = budget - (time.monotonic() - t0)
        if time_left < 10:
            break

        g = _frank_wolfe_step(a, active, mw_iters=18, seed=1000 + t)
        if g is None:
            break

        cg = _fft_conv(g)
        fg = _fft_conv2(a, g)

        # Line search
        best_alpha = 0.0
        best_m = M
        for alpha in [0.5 / (1 + t * 0.1), 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            one = 1.0 - alpha
            val = (one * one) * c + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
            m = float(np.max(val))
            if math.isfinite(m) and m < best_m:
                best_m = m
                best_alpha = alpha

        if best_alpha > 0 and best_m < M - 1e-16:
            a = (1.0 - best_alpha) * a + best_alpha * g
            np.maximum(a, 0.0, out=a)
            s = float(a.sum())
            if s < 1e-15:
                break
            a /= s
            c = _fft_conv(a)
            M = float(np.max(c))
            no_improve = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
                if verbose and (t <= 3 or t % 20 == 0):
                    print("    FW t={}: score={:.8f} alpha={:.3f}".format(t, 2*n*best_M, best_alpha))
            active = np.argsort(c)[::-1][:active_max]
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                rng = np.random.default_rng(999 + t)
                a = best_a + rng.normal(0, 0.02, n) * best_a
                a = np.maximum(a, 1e-12)
                a /= a.sum()
                c = _fft_conv(a)
                M = float(np.max(c))
                active = np.argsort(c)[::-1][:active_max]
                no_improve = 0

    return best_a

def _default_sequence(n=5000):
    x = np.linspace(0.0, 1.0, int(n))
    return _norm_sum1(np.sin(np.pi * x) ** 2)

def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    total_budget = max(60.0, min(float(budget_s) - 90.0, 1710.0))
    print("Starting: budget={:.0f}s".format(total_budget))

    # Initialization
    try:
        parent_list = height_sequence_1
    except Exception:
        parent_list = None

    seq_best = None
    if parent_list is not None:
        try:
            p = np.asarray(parent_list, dtype=np.float64)
            p[~np.isfinite(p)] = 0.0
            np.clip(p, 0.0, 1000.0, out=p)
            if p.ndim != 1:
                p = p.ravel()
            if p.size > 0 and float(p.sum()) > 1e-10:
                seq_best = _norm_sum1(p)
        except Exception:
            pass

    if seq_best is None:
        seq_best = _default_sequence(5000)

    # Work size: compress if too long
    max_work = 8000
    if seq_best.size > max_work:
        work_seq = _interpolate_seq(seq_best, max_work)
    else:
        work_seq = seq_best.copy()

    best_score = _safe_eval(seq_best)
    if not math.isfinite(best_score):
        best_score = 2.0 * seq_best.size * float(np.max(_fft_conv(seq_best)))
    print("Initial: n={} score={:.8f}".format(seq_best.size, best_score))

    seq_best, best_score = _try_update(seq_best, best_score, work_seq, "init", t_start)

    # --- Phase 1: L-BFGS Curriculum ---
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 400:
        phase_budget = min(time_left * 0.35, 650.0)
        print("\n=== Phase 1: L-BFGS Curriculum (budget={:.0f}s) ===".format(phase_budget))
        a_p1 = _lbfegs_curriculum(work_seq, phase_budget, None)
        seq_best, best_score = _try_update(seq_best, best_score, a_p1, "lbfgs_curve", t_start)
        if seq_best.size > max_work:
            work_seq = _interpolate_seq(seq_best, max_work)
        else:
            work_seq = _norm_sum1(seq_best)

    # --- Phase 2: Frank-Wolfe ---
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 350:
        phase_budget = min(time_left * 0.25, 450.0)
        print("\n=== Phase 2: Frank-Wolfe (budget={:.0f}s) ===".format(phase_budget))
        a_p2 = _fw_search(work_seq, phase_budget, verbose=True)
        seq_best, best_score = _try_update(seq_best, best_score, a_p2, "fw", t_start)
        work_seq = _norm_sum1(seq_best) if seq_best.size <= max_work else _interpolate_seq(seq_best, max_work)

    # --- Phase 3: Spectral Exploration ---
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 250:
        phase_budget = min(time_left * 0.25, 400.0)
        print("\n=== Phase 3: Spectral Exploration (budget={:.0f}s) ===".format(phase_budget))
        a_p3 = _spectral_exploration(work_seq, phase_budget, num_restarts=3, verbose=True)
        seq_best, best_score = _try_update(seq_best, best_score, a_p3, "spectral", t_start)
        work_seq = _norm_sum1(seq_best) if seq_best.size <= max_work else _interpolate_seq(seq_best, max_work)

    # --- Phase 4: Frank-Wolfe refinement (second pass) ---
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 200:
        phase_budget = min(time_left * 0.20, 300.0)
        print("\n=== Phase 4: FW Refinement (budget={:.0f}s) ===".format(phase_budget))
        a_p4 = _fw_search(work_seq, phase_budget, verbose=True)
        seq_best, best_score = _try_update(seq_best, best_score, a_p4, "fw2", t_start)
        work_seq = _norm_sum1(seq_best) if seq_best.size <= max_work else _interpolate_seq(seq_best, max_work)

    # --- Phase 5: Coordinate Polish ---
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 150:
        phase_budget = min(time_left * 0.20, 250.0)
        print("\n=== Phase 5: Coordinate Polish (budget={:.0f}s) ===".format(phase_budget))
        a_p5 = _coordinate_polish(work_seq, phase_budget, verbose=True)
        seq_best, best_score = _try_update(seq_best, best_score, a_p5, "coord", t_start)
        work_seq = _norm_sum1(seq_best) if seq_best.size <= max_work else _interpolate_seq(seq_best, max_work)

    # --- Phase 6: L-BFGS final sharp polish with very high beta ---
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 100 and SCIPY_OK:
        phase_budget = min(time_left * 0.25, 200.0)
        print("\n=== Phase 6: Final Sharp L-BFGS (budget={:.0f}s) ===".format(phase_budget))
        c_cur = _fft_conv(work_seq)
        n_w = work_seq.size
        top_k = max(20, n_w // 25)
        msk = _top_k_mask(c_cur, None, top_k)
        a_p6 = _run_lbfegs(work_seq, 5000.0, msk, int(max(100, phase_budget // 0.5)), verbose=True)
        seq_best, best_score = _try_update(seq_best, best_score, a_p6, "sharp_lb", t_start)
        work_seq = _norm_sum1(seq_best) if seq_best.size <= max_work else _interpolate_seq(seq_best, max_work)

    # --- Phase 7: Final coordinate polish ---
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 60:
        phase_budget = min(time_left * 0.30, 120.0)
        print("\n=== Phase 7: Final Coord (budget={:.0f}s) ===".format(phase_budget))
        a_p7 = _coordinate_polish(work_seq, phase_budget, verbose=False)
        seq_best, best_score = _try_update(seq_best, best_score, a_p7, "final_coord", t_start)

    # --- Phase 8: Length probes ---
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 200:
        print("\n=== Phase 8: Length Probes ===")
        base_len = seq_best.size if seq_best.size <= max_work else work_seq.size
        for delta in [300, -300, 600, -600, 1000, -1000]:
            time_left = total_budget - (time.monotonic() - t_start)
            if time_left < 120:
                break
            target = int(max(2000, min(max_work, base_len + delta)))
            if target == base_len:
                continue
            print("  Trying n={}".format(target))
            a0 = _interpolate_seq(seq_best if seq_best.size <= max_work else work_seq, target)
            if time_left > 150:
                a_new = _run_lbfegs(a0, 200.0, None, int(max(50, min(200, time_left // 1.0))), verbose=False)
                seq_best, best_score = _try_update(seq_best, best_score, a_new, "len_{}".format(target), t_start)
            time_left = total_budget - (time.monotonic() - t_start)
            if time_left > 80:
                a_fw = _fw_search(a0 if seq_best.size != target else _norm_sum1(seq_best),
                                  min(time_left * 0.5, 60.0), verbose=False)
                seq_best, best_score = _try_update(seq_best, best_score, a_fw, "lenfw_{}".format(target), t_start)

    # Final validation
    final_val = _safe_eval(seq_best)
    if math.isfinite(final_val) and final_val < best_score:
        best_score = final_val

    arr = np.asarray(seq_best, dtype=np.float64)
    arr[~np.isfinite(arr)] = 0.0
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if arr.size == 0 or s < 0.01:
        arr = _default_sequence(5000)
    else:
        arr = arr / s

    elapsed = time.monotonic() - t_start
    print("\nFINAL: score={:.8f} n={} elapsed={:.1f}s".format(best_score, arr.size, elapsed))
    return [float(x) for x in arr]
