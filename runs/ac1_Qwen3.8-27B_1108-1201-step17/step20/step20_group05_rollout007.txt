def _next_pow2(x):

def _norm1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(max(1, arr.size), dtype=np.float64) / max(1, arr.size)
    return arr / s


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = _next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return np.where(np.isfinite(c) & (c > 0.0), c, 0.0)


def _fft_conv2(a, b):

def _interp_seq(a, new_n):

def _lse_obj(a, beta, reg=0.0):
    """Log-sum-exp smoothed objective and gradient.
    Returns (f, grad) where f approximates max(conv(a,a)) and grad is its gradient.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 0.0, np.array([], dtype=np.float64)
    size = 2 * n - 1
    total = 3 * n - 2
    N = _next_pow2(total)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)

    m = float(np.max(c))
    shifted = np.clip(beta * (c - m), -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0 or not math.isfinite(se):
        e = np.zeros(size, dtype=np.float64)
        e[0] = 1.0
        se = 1.0
    p = e / se  # soft probabilities over convolution indices

    # Gradient: grad_i = 2 * sum_k p_k * (cross-conv properly)
    # conv(a,a)_k = sum_j a_j * a_{k-j}
    # d/d a_i of conv(a,a)_k = a_{k-i} + a_{k-i} = 2*a_{k-i} (if valid)
    # So grad_i = 2 * sum_k p_k * a_{k-i}
    # This is equivalent to 2 * (p * a)[n-1-i] ... let me use the FFT approach.
    
    # Using FFT: grad = 2 * ifft(fft(a) * fft(reversed_p)) [indexed appropriately]
    Pr = np.fft.rfft(p[::-1], N)
    conv_grad = np.fft.irfft(A * Pr, N)[:total]
    grad = 2.0 * conv_grad[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)

    f = m + math.log(se) / beta

    if reg > 0.0:
        f += 0.5 * reg * float(np.dot(a, a))
        grad = grad + reg * a

    if not math.isfinite(f):
        f = 1e100
        grad = np.zeros(n, dtype=np.float64)
    return f, grad


def _make_v(a):
    """Convert probability vector to log-softmax coordinates with one reference fixed."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 1:
        return np.array([], dtype=np.float64), 0
    if np.any(a > 1e-300):
        ref = int(np.argmax(a))
    else:
        ref = 0
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.where(np.isfinite(v), v, -700.0)
    return v, ref


def _from_v(v, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]

    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):

    return e / s
