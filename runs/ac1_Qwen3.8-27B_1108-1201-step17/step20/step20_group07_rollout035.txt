import numpy as np
import time

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum to 1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    """Project onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.searchsorted(cssv, 1.0, side='left'))
    # More careful:
    cond = (u * np.arange(1, n + 1) - (cssv - 1.0)) > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

        b_use[idx] = b[idx]
    else:
        b_use = b

    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 500)

import numpy as np

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum to 1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto the probability simplex {x >= 0, sum(x) = 1} in O(n) time."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    ar = np.arange(1, n + 1)
    cond = (u * ar) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

            print("      Nest t={} ({:.1f}s): J={:.7f} beta={:.1f}".format(it, elapsed, J, beta))
    
    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=True):
    """
    Stochastic coordinate descent: pick random coordinates, try to optimize them.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute current autocorrelation
        b = fast_autocorr(a)
        m_curr = b.max()
        
        if m_curr < best_m - 1e-12:
            best_m = m_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 500:
            # Random restart perturbation
            idx = rng.integers(n, size=10)
            amt = best_a[idx] * rng.uniform(0.01, 0.05, 10)
            a = best_a.copy()
            a[idx] -= amt
            dst = rng.integers(n, size=10)
            a[dst] += amt
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            stall = 0
            continue
        
        # Pick a random coordinate to modify
        i = int(rng.integers(n))
        if a[i] < 1e-14:
            continue
        
        # Try moving some mass from i to a random other coordinate
        j = int(rng.integers(n))
        if i == j:
            continue
        
        # Try several fractions
        improved = False
        for frac in [0.01, 0.05, 0.1, 0.2, 0.4]:
            amt = a[i] * frac
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m - 1e-13:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                stall = 0
                improved = True
                break
        
        if not improved:
            stall += 1
        
        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("      CoorD stall={} ({:.1f}s): J={:.7f}".format(stall, elapsed, J))
    
    return best_a, best_m


def peak_shaving_v2(a0, time_limit, n_top=15, verbose=True):
    """
    Aggressive peak shaving: directly reduce the top autocorrelation peaks
    by redistributing mass from contributing positions to low-pressure positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(4242)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        m_curr = b.max()
        
        if m_curr < best_m - 1e-12:
            best_m = m_curr
            best_a = a.copy()
            stall = 0
        
        # Get top peaks
        peak_indices = np.argsort(b)[::-1][:n_top]
        
        # Compute pressure map
        pressure = np.zeros(n)
        for k_i in peak_indices[:8]:
            k_int = int(k_i)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                pressure[idx] += a[k_int - idx]
                pressure[k_int - idx] += a[idx]
        
        # Try to reduce each top peak
        improved = False
        for k_i in peak_indices[:6]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k_i)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find the pair with max contribution
            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            if partner < 0 or partner >= n:
                continue
            
            # Determine which to reduce (reduce the larger one)
            if a[src] >= a[partner]:
                reduce_idx = src
            else:
                reduce_idx = partner
            
            if a[reduce_idx] < 1e-14:
                continue
            
            # Find low-pressure destination
            low_pressure = np.argsort(pressure)[:n // 4]
            low_pressure = low_pressure[low_pressure != reduce_idx]
            if len(low_pressure) == 0:
                continue
            
            dst = int(low_pressure[rng.integers(len(low_pressure))])
            
            # Try to move mass
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.35]:
                amt = a[reduce_idx] * frac
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-13:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            if stall > 100:
                # Perturb
                noise = min(0.05, 0.01 * stall / 100)
                pert = rng.normal(0, noise, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-12)
                a_r = a_r / a_r.sum()
                m_r = fast_autocorr(a_r).max()
                if m_r < best_m:
                    best_m = m_r
                    best_a = a_r.copy()
                a = a_r if m_r < best_m + 0.005 else best_a.copy()
                stall = 0
        
        if verbose and stall % 200 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("      PeakShave stall={} ({:.1f}s): J={:.7f}".format(stall, elapsed, J))
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving the shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n, seed=0):
    """Generate diverse parametric initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    candidates = []
    
    # Beta distributions
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        for beta_p in [0.5, 1.0, 1.5, 2.0, 3.0]:
            a = x ** (alpha - 1) * (1 - x) ** (beta_p - 1)
            a = np.maximum(a, 1e-10)
            candidates.append(a / a.sum())
    
    # Gaussians
    for c in [0.3, 0.5, 0.7]:
        for s in [0.1, 0.2, 0.3]:
            a = np.exp(-0.5 * ((x - c) / s) ** 2)
            candidates.append(a / a.sum())
    
    # Cosine tapers
    for p in [1, 2, 3]:
        a = np.cos(np.pi * x / 2) ** p
        a = np.maximum(a, 1e-10)
        candidates.append(a / a.sum())
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    return candidates


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence/score."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {:.7f} (n={}, t={:.1f}s)".format(label, sc, len(a), elapsed))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main search function. Returns the best sequence found within the time budget.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    print("Time budget: {:.1f}s".format(time_budget))
    
    best_seq = None
    best_score = float('inf')
    rng = np.random.default_rng(2024)
    
    # --- Load parent ---
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = norm_sum1(p)
            best_score = evaluate_safe(best_seq)
            print("Parent: n={}, score={:.7f}".format(len(p), best_score))
    except Exception:
        best_seq = None
    
    if best_seq is None:
        n_base = 7000
        x = np.linspace(0, 1, n_base)
        a_raw = x ** 1.5 * (1 - x + 0.01)
        a_raw = np.maximum(a_raw, 1e-10)
        best_seq = a_raw / a_raw.sum()
        best_score = evaluate_safe(best_seq)
        print("Default init: n={}, score={:.7f}".format(n_base, best_score))
    
    n0 = len(best_seq)
    
    # --- Phase 1: Try parametric initializations at parent length ---
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.02, 30.0)
    if t1 > 5:
        print("\n=== Phase 1: Parametric init ===")
        cands = parametric_init(n0, seed=42)
        for i, c in enumerate(cands):
            if time.monotonic() - t_global > time_budget:
                break
            m = fast_autocorr(c).max()
            sc = 2.0 * n0 * m
            if sc < best_score:
                best_score = sc
                best_seq = c.copy()
                print("  Parametric init {}: J={:.7f}".format(i, sc))
    
    # --- Phase 2: Nesterov smoothed descent (main workhorse) ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 60:
        print("\n=== Phase 2: Nesterov descent (n={}) ===".format(n_best))
        a_out, _ = nesterov_smoothed_descent(
            best_seq, t2, beta_start=25.0, beta_end=600.0,
            n_top_frac=0.03, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nesterov", t_global)
    
    # --- Phase 3: Peak shaving ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        print("\n=== Phase 3: Peak shaving (n={}) ===".format(n_best))
        a_out, _ = peak_shaving_v2(best_seq, t3, n_top=12, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakshave", t_global)
    
    # --- Phase 4: Nesterov descent round 2 (with higher beta) ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.2, 300.0)
    if t4 > 60:
        print("\n=== Phase 4: Nesterov round 2 (n={}) ===".format(n_best))
        a_out, _ = nesterov_smoothed_descent(
            best_seq, t4, beta_start=80.0, beta_end=1500.0,
            n_top_frac=0.02, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nesterov2", t_global)
    
    # --- Phase 5: Coordinate descent fine-tuning ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.1, 120.0)
    if t5 > 20:
        print("\n=== Phase 5: Coordinate descent (n={}) ===".format(n_best))
        a_out, _ = coordinate_descent(best_seq, t5, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "cooord", t_global)
    
    # --- Phase 6: Peak shaving round 2 ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 15:
        print("\n=== Phase 6: Peak shaving 2 (n={}) ===".format(n_best))
        a_out, _ = peak_shaving_v2(best_seq, t6, n_top=8, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakshave2", t_global)
    
    # --- Phase 7: Multi-length exploration ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        print("\n=== Phase 7: Multi-length exploration ===")
        for dn in [200, -200, 400, -400]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            new_n = n_best + dn
            if new_n < 3000 or new_n > 10000:
                continue
            print("  Trying n={}...".format(new_n))
            a_new = interpolate_sequence(best_seq, new_n)
            
            # Quick Nesterov
            t7 = min(remaining * 0.3, 120.0)
            if t7 > 20:
                a_out, _ = nesterov_smoothed_descent(
                    a_new, t7, beta_start=30.0, beta_end=400.0,
                    n_top_frac=0.03, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "ml_n{}".format(new_n), t_global)
            
            # Quick peak shave
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_out, _ = peak_shaving_v2(
                    best_seq if len(best_seq) == new_n else a_new,
                    min(remaining * 0.3, 40.0), n_top=10, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "ps_n{}".format(new_n), t_global)
    
    # --- Phase 8: Final high-precision Nesterov ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        print("\n=== Phase 8: Final polish (n={}) ===".format(n_best))
        t8 = min(remaining * 0.5, 200.0)
        if t8 > 30:
            a_out, _ = nesterov_smoothed_descent(
                best_seq, t8, beta_start=200.0, beta_end=3000.0,
                n_top_frac=0.01, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_nest", t_global)
    
    # --- Phase 9: Final peak shave + coord desc ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t9a = min(remaining * 0.4, 60.0)
        if t9a > 10:
            a_out, _ = peak_shaving_v2(best_seq, t9a, n_top=6, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_ps", t_global)
        
        remaining = time_budget - (time.monotonic() - t_global)
        t9b = min(remaining * 0.4, 40.0)
        if t9b > 10:
            a_out, _ = coordinate_descent(best_seq, t9b, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_cd", t_global)
    
    # --- Final validation ---
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.7f}, n={}, time={:.1f}s".format(best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 7000
        x = np.linspace(0, 1, n_fb)
        a_raw = np.ones(n_fb)
        best_seq = a_raw / a_raw.sum()
        best_score = evaluate_safe(best_seq)
    
    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
