def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    """Project z onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_gradient_smoothed(a, beta, top_k=None):
    """
    Compute the gradient of (1/beta)*log(sum_k exp(beta * b[k])) 
    where b = a * a (linear convolution).
    We only use top_k peaks for efficiency.
    """
    n = len(a)
    b = fast_autocorr(a)
    
    if top_k is not None and top_k < len(b):

        b_use = np.zeros_like(b)
        b_use[idx] = b[idx]
    
    b_max = b_use.max()

    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    
    # Gradient of b[k] w.r.t. a[i]:
    # b[k] = sum_j a[j]*a[k-j]
    # db[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i] (if 0<=k-i<n, i.e., same condition)

    # db[k]/da[i] = sum_j [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)]
    #             = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n, from the second term where j=k-i)
    #             Wait, let me be more careful.
    # b[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
    # db[k]/da[i] = sum_j [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)] * [validity]
    # First term: j=i, needs 0<=k-i<n, contributes a[k-i]
    # Second term: k-j=i, j=k-i, needs 0<=k-i<n and 0<=j<n, contributes a[k-i]
    # So db[k]/da[i] = 2*a[k-i] if 0<=k-i<n, else 0
    # (The factor of 2 is because both terms give the same contribution when i != k-i,
    #  and when i = k-i, i.e., 2i=k, it's just a[i] * d(a[i])/da[i] = a[i], 
    #  but actually the derivative of a[i]^2 is 2*a[i], so the factor of 2 is correct.)
    
    # So grad[i] = sum_k p[k] * 2*a[k-i] (where 0<=k-i<n)
    #            = 2 * sum_k p[k] * a[k-i] (where valid)
    # This is 2 * (p * a_rev)[i] where a_rev[i] = a[n-1-i]
    # Wait, let me think again.
    # sum_k p[k] * a[k-i] where 0<=k-i<n means k-i is in [0,n-1], so k is in [i, i+n-1].
    # Let j = k-i, then j is in [0, n-1], k = i+j.
    # sum_j p[i+j] * a[j] (where i+j < 2n-1)
    # This is a cross-correlation of p and a.
    
    # Using FFT:
    fft_size = 1 << (2 * n - 1 - 1).bit_length()  # need at least 2n-1
    fft_size = 1 << (2 * (len(p) + n - 1) - 1).bit_length()  # safer
    Actually, let me just compute it as a convolution.
    # grad[i] = 2 * sum_{j=0}^{n-1} a[j] * p[i+j] (where i+j < len(p))
    # This is 2 * (reversed a convolved with p)[i]... 
    # Actually: conv(p, a_rev)[i] where a_rev = a[::-1]
    # Let me verify: conv(p, a_rev)[m] = sum_j p[j] * a_rev[m-j] = sum_j p[j] * a[n-1-(m-j)]
    # That's not quite what I want.
    
    # Let me just use the direct formula with FFT:
    # grad[i] = 2 * sum_k p[k] * a[k-i] for valid k-i
    # = 2 * sum_j a[j] * p[i+j] for j=0..n-1 (where i+j < 2n-1)
    
    # Define q[m] = p[m] for m=0..2n-2
    # Then grad[i] = 2 * sum_j a[j] * q[i+j]
    # This is the "cross-correlation" of a and q, at lag i.
    # crosscorr(a, q)[i] = sum_j a[j] * q[i+j]
    # = conv(a_rev, q)[i + n - 1]... let me just compute it directly.
    
    # Simplest: use FFT
    # crosscorr(a, q)[i] = sum_j a[j] * conj(q[i+j]) (for real, just q[i+j])
    # This is the same as conv(a_rev, q)[n-1+i]
    
    a_rev = a[::-1].copy()
    conv_len = n + len(p) - 1

    P_rfft = np.fft.rfft(p, fft_size)

    # crosscorr[i] = conv[n-1+i] for i=0..n-1
    grad = 2.0 * conv[n-1:2*n-1]
    
    return grad / beta  # gradient of the smoothed objective

def fw_plus_optimize(a0, time_limit, beta_start=10.0, beta_end=500.0,

            best_m = m_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        grad = compute_gradient_smoothed(a, beta, top_k)

        v_fw[int(np.argmin(grad))] = 1.0

        v_away = None
        if len(support) > 10:
            # Find the element in support with maximum gradient
            support_arr = np.array(list(support))
            grad_support = grad[support_arr]

        best_trial_m = m_curr
        best_trial_a = a.copy()

                if s < 1e-15:

                m_trial = fast_autocorr(trial).max()

def spectral_equalization(a0, time_limit, n_iters=200, verbose=True):
    """
    Iterative spectral equalization:
    1. FFT of current sequence
    2. Flatten the power spectrum
    3. IFFT
    4. Project to non-negative simplex
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute power spectrum
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        P = A * np.conj(A)  # Power spectrum
        P = np.maximum(P, 1e-30)
        
        # Target: make |A|^2 as flat as possible
        # The flat target is mean(P)
        # But we need sum(a) = 1, which means A[0] = 1 (DC component)
        # So we should keep DC fixed and flatten the rest
        
        # Soft flattening: P_new = P^alpha for alpha < 1
        alpha = 0.5  # Compress dynamic range
        P_flat = np.power(P, alpha)
        
        # Preserve DC
        P_flat[0] = P[0]
        
        # New magnitude: sqrt(P_flat), phase from original
        phase = np.angle(A)
        A_new = np.sqrt(P_flat) * np.exp(1j * phase)
        
        # IFFT
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        # Check improvement
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and (it + 1) % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SpecEq it={it+1} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def targeted_peak_reduction(a0, time_limit, verbose=True):
    """
    More aggressive peak reduction:
    - For each top peak, compute per-index sensitivity
    - Move mass from high-sensitivity to low-sensitivity positions
    - Use adaptive step sizes
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 300
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_autocorr(a)
        m_curr = b.max()
        
        if m_curr < best_m - 1e-15:
            best_m = m_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Get top peaks
        top_k = min(30, len(b))
        peak_idx = np.argsort(b)[::-1][:top_k]
        
        # Compute sensitivity map: for each position i, total contribution to top peaks
        sensitivity = np.zeros(n)
        for k_i in peak_idx[:15]:
            k_int = int(k_i)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                # Contribution of a[i] to b[k] is a[k-i]
                sensitivity[idx] += a[k_int - idx]
        
        # For each top peak, try to reduce it
        improved = False
        for k_i in peak_idx[:10]:
            k_int = int(k_i)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find the position with maximum contribution
            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])
            src_val = a[src]
            
            if src_val < 1e-14:
                continue
            
            # Find destination: low sensitivity, not contributing to any top peak
            # Score each position by its sensitivity (lower is better for destination)
            dst_score = sensitivity.copy()
            dst_score[src] = np.inf
            dst_score[src + 1:n] = np.maximum(dst_score[src + 1:n], sensitivity[src + 1:n])  # Avoid neighbors
            dst_score[src - 1] = np.inf if src > 0 else dst_score[src - 1]
            
            # Also avoid positions that contribute to other top peaks
            for k2 in peak_idx[:8]:
                k2_int = int(k2)
                partner = k2_int - src
                if 0 <= partner < n and partner != src:
                    dst_score[partner] = np.max(dst_score) + 1
            
            dst = int(np.argmin(dst_score))
            if dst == src:
                continue
            
            # Try to move mass
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
                amt = src_val * frac
                if amt < 1e-16:
                    continue
                trial = a.copy()
                trial[src] -= amt
                trial[dst] += amt
                trial = np.maximum(trial, 0)
                s_t = trial.sum()
                if s_t < 1e-15:
                    continue
                trial = trial / s_t
                m_trial = fast_autocorr(trial).max()
                if m_trial < best_m - 1e-15:
                    a = trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            # Random exploration
            if stall % 30 == 0:
                n_pert = min(200, n // 10)
                pert_idx = rng.choice(n, n_pert, replace=False)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)


def interpautocorr(a, p):
