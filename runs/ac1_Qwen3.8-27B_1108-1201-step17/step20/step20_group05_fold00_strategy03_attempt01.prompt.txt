```python
import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))
```

```python
def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def fft_conv2(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if b.size != n:
        b = np.resize(b, n)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def cross_convolve_p_a(p, a, factor=1.0):
    """
    Return grad-like vector g_i = factor * sum_k p_k a_{k-i}, valid ranges only.
    p has length 2n-1, a has length n.
    """
    p = np.asarray(p, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    total = 3 * n - 2
    if p.size != 2 * n - 1:
        p = np.resize(p, 2 * n - 1)
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    out = conv[n - 1:2 * n - 1][::-1]
    out = np.where(np.isfinite(out), out, 0.0)
    return factor * out


def proxy_max(c, mask=None):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0
    if mask is None:
        vals = c
    else:
        vals = c[mask]
    if vals.size == 0:
        return 0.0
    vals = vals[np.isfinite(vals)]
    if vals.size == 0:
        return 0.0
    return float(np.max(vals))


def active_top(c, mask, max_k):
    c = np.asarray(c, dtype=np.float64)
    size = c.size
    max_k = int(min(max_k, size))
    if max_k <= 0:
        return np.array([], dtype=int)
    if mask is None:
        return np.argsort(c)[::-1][:max_k]
    vals = c.copy()
    vals[~mask] = -np.inf
    return np.argsort(vals)[::-1][:max_k]

def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)


def choose_mask(a, actual_score):
    """
    Infer which subset of convolution coefficients is being maximized by the
    evaluator, mostly to make internal proxies safer. The likely actual metric
    is the full convolution maximum.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0 or not math.isfinite(actual_score):
        return None
    c = fft_conv(a)
    size = 2 * n - 1
    full_score = 2.0 * n * proxy_max(c, None)
    best_mask = None
    best_diff = abs(full_score - actual_score)

    masks = []
    if n > 1:
        m = np.ones(size, dtype=bool)
        nc = m.copy()
        nc[n - 1] = False
        masks.append(nc)

        ne = m.copy()
        ne[0] = False
        ne[-1] = False
        masks.append(ne)

        nce = nc.copy()
        nce[0] = False
        nce[-1] = False
        masks.append(nce)

        left = np.zeros(size, dtype=bool)
        left[:n] = True
        masks.append(left)

        right = np.zeros(size, dtype=bool)
        right[n:] = True
        masks.append(right)

    for mm in masks:
        if mm is None or not np.any(mm):
            continue
        sc = 2.0 * n * proxy_max(c, mm)
        df = abs(sc - actual_score)
        if df < best_diff:
            best_diff = df
            best_mask = mm

    # If no alternative is materially closer, use full.
    if best_mask is not None and best_diff > 0.25 * max(1.0, abs(actual_score)):
        return None
    return best_mask


def lse_a(a, beta, mask=None, reg_a=0.0):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 0.0, np.array([], dtype=np.float64)
    size = 2 * n - 1
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)

    if mask is None:
        cv = c
    else:
        cv = c[mask]

    if cv.size == 0:
        f = 0.0
        grad = np.zeros(n, dtype=np.float64)
    else:
        m = float(np.max(cv))
        shifted = beta * (cv - m)
        shifted = np.clip(shifted, -700.0, 700.0)
        e = np.exp(shifted)
        se = float(e.sum())
        if se <= 0.0 or not math.isfinite(se):
            e = np.zeros(cv.size, dtype=np.float64)
            e[0] = 1.0
            se = 1.0
        p = e / se

        pf = np.zeros(size, dtype=np.float64)
        if mask is None:
            pf[:] = p
        else:
            pf[mask] = p

        Pr = np.fft.rfft(pf[::-1], N)
        conv = np.fft.irfft(A * Pr, N)[:total]
        grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
        grad = np.where(np.isfinite(grad), grad, 0.0)
        f = m + math.log(se) / beta

    if reg_a > 0.0:
        f += 0.5 * reg_a * float(np.dot(a, a))
        grad = grad + reg_a * a

    if not math.isfinite(f):
        f = 1e100
    return f, grad


def softmax_from_u(u):
    u = np.asarray(u, dtype=np.float64)
    n = u.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(u)):
        return np.ones(n, dtype=np.float64) / float(n)
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def make_v(a, ref=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 1:
        return np.array([], dtype=np.float64), 0
    if ref is None:
        if np.any(a > 0.0):
            ref = int(np.argmax(a))
        else:
            ref = 0
    ref = int(ref)
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.where(np.isfinite(v), v, -700.0)
    return v, ref


def from_v(v, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    return softmax_from_u(u)


def objgrad_v(v, beta, mask=None, reg_a=0.0, reg_u=0.0, ref=0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    if n <= 0:
        return 1e100, np.array([], dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return 1e100, np.zeros(v.size, dtype=np.float64)

    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1

    u = np.empty(n, dtype=np.float64)
    u[ref] = 0.0
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]

    a = softmax_from_u(u)
    f, ga = lse_a(a, beta, mask, reg_a)
    if not math.isfinite(f):
        f = 1e100
        ga = np.zeros(n, dtype=np.float64)

    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)

    gv = np.empty(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]

    if reg_u > 0.0:
        f += 0.5 * reg_u * float(np.dot(v, v))
        gv = gv + reg_u * v

    if not np.all(np.isfinite(gv)):
        gv = np.zeros(v.size, dtype=np.float64)
    return f, gv


def estimate_lbfegs_cost(a, mask=None, beta=30.0):

def run_lbfegs(a0, beta, mask, maxfun=500, reg_a=0.0, reg_u=1e-12, verbose=True):
    a = norm_sum1(a0)
    n = a.size
    if n <= 1:
        return a, np.array([], dtype=np.float64), 0

    v0, ref = make_v(a)
    if not SCIPY_OK or maxfun <= 0:
        return a, v0, ref

    t0 = time.monotonic()
    options = {
        "maxfun": int(maxfun),
        "maxiter": int(maxfun + 30),
        "ftol": 1e-12,
        "gtol": 1e-7,
        "disp": False
    }
    res = None
    try:
        res = scipy.optimize.minimize(
            objgrad_v,
            v0,
            args=(beta, mask, reg_a, reg_u, ref),
            method="L-BFGS-B",
            jac=True,
            options=options
        )
    except Exception as e:
        print("    L-BFGS exception: " + repr(e))
        res = None

    if res is not None and hasattr(res, "x"):
        vx = np.asarray(res.x, dtype=np.float64)
        if vx.size == v0.size and np.all(np.isfinite(vx)):
            a_out = from_v(vx, ref)
            a_out = norm_sum1(a_out)
            v_out, ref_out = make_v(a_out)
            if verbose:
                print("    L-BFGS beta=" + repr(beta) +
                      " n=" + str(n) +
                      " eval<= " + str(int(maxfun)) +
                      " time=" + repr(time.monotonic() - t0) +
                      " success=" + str(getattr(res, "success", False)))
            return a_out, v_out, ref_out

    if verbose:
        print("    L-BFGS beta=" + repr(beta) +
              " n=" + str(n) +
              " time=" + repr(time.monotonic() - t0) +
              " (fallback)")
    return a, v0, ref


def lbfegs_phase(a_init, base_mask, budget, cost):
    a = norm_sum1(a_init)
    n = a.size
    size = 2 * n - 1
    best_a = a.copy()
    best_M = proxy_max(fft_conv(a), base_mask)
    current_a = a.copy()

    t1 = int(min(1000, max(50, size // 10)))
    t2 = int(min(600, max(40, size // 20)))
    t3 = int(min(400, max(30, size // 30)))
    t4 = int(max(12, size // 100))
    t5 = int(max(8, size // 160))

    schedules = [
        (10.0, None),
        (30.0, None),
        (80.0, t1),
        (120.0, t2),
        (250.0, t3),
        (600.0, t4),
        (1000.0, t5)
    ]

    total_eval_budget = int(budget / max(2.5 * cost, 0.03))
    per_run = int(max(80, min(700, max(1, total_eval_budget // max(1, len(schedules))))))

    t0 = time.monotonic()
    for beta, top_count in schedules:
        if time.monotonic() - t0 > budget - 5:
            break
        if top_count is None:
            msk = None
        else:
            msk = make_top_mask(current_a, top_count, base_mask)

        a_new, v_new, ref_new = run_lbfegs(
            current_a,
            beta,
            msk,
            maxfun=per_run,
            reg_a=0.0,
            reg_u=1e-12,
            verbose=True
        )
        c_new = fft_conv(a_new)
        m_new = proxy_max(c_new, base_mask)

        if math.isfinite(m_new):
            # Allow small temporary increases to explore basins, but not huge excursions.
            if m_new <= best_M * (1.0 + 1e-4) + 1e-18:
                current_a = a_new
        else:
            current_a = best_a

        if math.isfinite(m_new) and m_new < best_M - 1e-15:
            best_M = m_new
            best_a = a_new.copy()

    return best_a


def line_search_mix(a, g, cf, cg, fg, mask, current_M, t):
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    n = a.size

    base_alphas = [
        max(0.01, 1.0 / (t + 1.0)),
        max(0.02, 2.0 / (t + 2.0)),
        0.04, 0.08, 0.15, 0.25, 0.40, 0.60, 0.80, 1.0
    ]
    alphas = []
    seen = set()
    for al in base_alphas:
        al = float(al)
        if not math.isfinite(al) or al <= 0.0 or al > 1.0:
            continue
        key = int(round(al * 1000.0))
        if key in seen:
            continue
        seen.add(key)
        alphas.append(al)
    if len(alphas) == 0:
        alphas = [0.1]

    best_alpha = 0.0
    best_m = current_M
    for alpha in alphas:
        one = 1.0 - alpha
        val = (one * one) * cf + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
        if mask is None:
            m = float(np.max(val))
        else:
            m = float(np.max(val[mask]))
        if math.isfinite(m) and m < best_m:
            best_m = m
            best_alpha = alpha

    if best_alpha <= 0.0:
        return a.copy(), best_m, 0.0

    new_a = (1.0 - best_alpha) * a + best_alpha * g
    new_a = np.maximum(new_a, 0.0)
    s = float(new_a.sum())
    if s < 1e-15:
        return a.copy(), best_m, 0.0
    new_a = new_a / s
    return new_a, best_m, best_alpha


def game_direction(f, active_idx, mw_iters=25, seed=0, init_logits=None):
    f = norm_sum1(f)
    n = f.size
    size = 2 * n - 1
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K <= 0:
        return np.ones(n, dtype=np.float64) / float(n)

    rng = np.random.default_rng(seed)
    if init_logits is None:
        logits = np.zeros(K, dtype=np.float64)
    else:
        logits = np.asarray(init_logits, dtype=np.float64).copy()
        if logits.size != K:
            logits = np.tile(logits[:K] if logits.size > 0 else np.zeros(K), (max(1, K // max(1, logits.size)),))[:K]
        logits = np.where(np.isfinite(logits), logits, 0.0)
        logmx = float(np.max(logits))
        if math.isfinite(logmx):
            logits = logits - logmx
    logits += rng.normal(0.0, 0.05, K)

    def normalize_logits_to_lambda(loc):
        loc = np.where(np.isfinite(loc), loc, -1000.0)
        loc = loc - float(np.max(loc))
        w = np.exp(loc)
        s = float(w.sum())
        if s <= 0.0:
            w = np.ones(K, dtype=np.float64)
            s = float(K)
        return w / s

    lambda_w = normalize_logits_to_lambda(logits)
    g = None
    for _ in range(max(1, int(mw_iters))):
        pf = np.zeros(size, dtype=np.float64)
        pf[active_idx] = lambda_w

        pressure = cross_convolve_p_a(pf, f, 1.0)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)

        pmx = float(np.max(pressure))
        pmn = float(np.min(pressure))
        ptp = pmx - pmn
        if ptp <= 1e-30:
            g = np.ones(n, dtype=np.float64) / float(n)
        else:
            scaled = (pressure - pmn) / ptp
            w = np.exp(-8.0 * scaled)
            w = np.where(np.isfinite(w), w, 0.0)
            w += 1e-12
            g = w / float(w.sum())
            # small diffuse component avoids overly brittle directions
            g = 0.97 * g + 0.03 * (np.ones(n, dtype=np.float64) / float(n))
            g = g / float(g.sum())

        h = fft_conv2(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmac_ = float(np.max(ha))
        hmin = float(np.min(ha))
        hptp = hmac_ - hmin
        if hptp > 1e-30:
            eta = 12.0 / hptp
            logits += eta * (ha - float(np.mean(ha)))
        lambda_w = normalize_logits_to_lambda(logits)

    return g


def lp_direction_hihts(f, active_idx, time_limit=8.0, max_k=1000):
    if not SCIPY_OK:
        return None
    if not hihts_available():
        return None

    f = norm_sum1(f)
    n = f.size
    active_idx = np.asarray(active_idx, dtype=int)
    if active_idx.size == 0:
        return None
    if active_idx.size > max_k:
        active_idx = active_idx[:max_k]
    K = active_idx.size

    row_parts = []
    col_parts = []
    val_parts = []
    for r, k in enumerate(active_idx):
        s = max(0, int(k) - n + 1)
        e = min(n, int(k) + 1)
        if s >= e:
            continue
        cols = np.arange(s, e, dtype=np.int32)
        row_parts.append(np.full(cols.size, r, dtype=np.int32))
        col_parts.append(cols)
        val_parts.append(f[cols.shape[0] + s - 1 - cols[::1]].copy() if False else f[int(k) - cols].astype(np.float64))

    if not row_parts:
        return None

    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0

    neg_one = scipy.sparse.csc_matrix(-np.ones((K, 1)), dtype=np.float64)
    A_ub = scipy.sparse.hstack((A, neg_one), format="csr")
    A_eq = np.ones((1, n + 1), dtype=np.float64)
    A_eq[0, n] = 0.0
    b_eq = np.ones(1, dtype=np.float64)
    b_ub = np.zeros(K, dtype=np.float64)

    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = scipy.optimize.linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="HiGHS",
            options={"disp": False, "time_limit": float(time_limit)}
        )
    except Exception:
        return None

    if res is None or not getattr(res, "success", False):
        return None

    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.where(np.isfinite(g) & (g > 0.0), g, 0.0)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs


def frank_wolfe_search(a0, budget, mask, active_max=2500, game_iters=22,
                       exact_every=7, exact_time=8.0, verbose=True):
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = proxy_max(c, mask)
    best_a = a.copy()
    best_M = M
    active = active_top(c, mask, active_max)

    no_improve = 0
    t = 0
    while time.monotonic() - t0 < time_limit - 5 and no_improve < 90:
        t += 1
        time_left = time_limit - (time.monotonic() - t0)
        if time_left < 25:
            break

        g = None
        # Exact LP direction occasionally, only when active set is manageable.
        if hihts_available() and exact_every > 0 and (t % exact_every == 0) and time_left > 100:
            g = lp_direction_hihts(a, active, exact_time, max_k=1000)

        if g is None:
            init_log = None
            if active.size > 0:
                av = c[active]
                av = np.where(np.isfinite(av) & (av > 0.0), av, 1e-300)
                init_log = np.log(av)
                init_log -= float(np.max(init_log))
            g = game_direction(a, active, mw_iters=game_iters, seed=1000 + t, init_logits=init_log)

        if g is None:
            break

        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        new_a, new_M, alpha = line_search_mix(a, g, c, cg, fg, mask, M, t)

        if new_M < M - 1e-16:
            if alpha > 0:
                one = 1.0 - alpha
                c = (one * one) * c + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
                c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
            a = new_a
            M = new_M
            no_improve = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            active = active_top(c, mask, active_max)
            if verbose and (t == 1 or t % 10 == 0):
                print("    FW t=" + str(t) +
                      " score_proxy=" + repr(2.0 * n * best_M) +
                      " alpha=" + repr(alpha) +
                      " active=" + str(active.size))
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                rng = np.random.default_rng(9973 + t)
                pert = rng.normal(0.0, 0.015, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-12)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = proxy_max(c, mask)
                active = active_top(c, mask, active_max)
                no_improve = 0
            if verbose and (t == 1 or t % 15 == 0):
                print("    FW stall t=" + str(t) +
                      " best_score_proxy=" + repr(2.0 * n * best_M))

    return best_a


def apply_single_move(c, a, s, d, tmove):
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = a.size
    c_new = c.copy()
    c_new[d:d + n] += 2.0 * tmove * a
    c_new[s:s + n] -= 2.0 * tmove * a

    t2 = tmove * tmove
    if 2 * d < c_new.size:
        c_new[2 * d] += t2
    if 2 * s < c_new.size:
        c_new[2 * s] += t2
    if s + d < c_new.size:
        c_new[s + d] -= 2.0 * t2
    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
    return c_new


def random_transfer_improve(a, c, M, mask, count, rng):
    n = a.size
    improved = False
    for _ in range(int(count)):
        if int(a.size) <= 1:
            break
        s = int(rng.integers(n))
        d = int(rng.integers(n))
        if s == d or a[s] < 1e-12:
            continue
        frac = float(rng.uniform(0.002, 0.05))
        tmove = a[s] * frac
        tmove = min(tmove, a[s] * 0.99)
        if tmove <= 1e-16:
            continue
        c_new = apply_single_move(c, a, s, d, tmove)
        m_new = proxy_max(c_new, mask)
        if m_new < M - 1e-17:
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = fft_conv(a)
            M = proxy_max(c, mask)
            improved = True
            break
    return a, c, M, improved


def coordinate_polish(a0, budget, mask, verbose=True):
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    c = fft_conv(a)
    M = proxy_max(c, mask)
    best_a = a.copy()
    best_M = M
    n = a.size
    stall = 0
    rnd = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit - 3 and stall < 120:
        rnd += 1
        if n <= 1:
            break

        top_k = min(40, 2 * n - 1)
        if top_k <= 0:
            break
        top_idx = active_top(c, mask, top_k)
        if top_idx.size == 0:
            break

        p_top = np.zeros(2 * n - 1, dtype=np.float64)
        p_top[top_idx] = 1.0 / float(top_idx.size)
        pressure = cross_convolve_p_a(p_top, a, 2.0)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)

        src_count = min(8, n)
        dst_count = min(12, n)

        src_score = pressure * (1.0 + float(n) * a)
        src_candidates = np.argpartition(src_score, min(src_count - 1, n - 1))[:src_count]
        dst_score = pressure + 1e-5 * a
        dst_candidates = np.argpartition(dst_score, min(dst_count - 1, n - 1))[:dst_count]

        improved_any = False
        best_cand = None

        for srr in src_candidates:
            s = int(srr)
            if a[s] < 1e-12:
                continue
            for drr in dst_candidates:
                d = int(drr)
                if d == s:
                    continue
                for frac in (0.005, 0.01, 0.02, 0.05):
                    tmove = a[s] * frac
                    if tmove <= 1e-16:
                        continue
                    tmove = min(tmove, a[s] * 0.99)
                    c_new = apply_single_move(c, a, s, d, tmove)
                    m_new = proxy_max(c_new, mask)
                    if m_new < M - 1e-18:
                        cand = (m_new, s, d, tmove)
                        if best_cand is None or cand[0] < best_cand[0]:
                            best_cand = cand
                            improved_any = True

        if not improved_any:
            # A few exploratory small transfers.
            for _ in range(25):
                s = int(rng.integers(n))
                d = int(rng.integers(n))
                if s == d or a[s] < 1e-12:
                    continue
                frac = float(rng.uniform(0.002, 0.03))
                tmove = a[s] * frac
                tmove = min(tmove, a[s] * 0.99)
                if tmove <= 1e-16:
                    continue
                c_new = apply_single_move(c, a, s, d, tmove)
                m_new = proxy_max(c_new, mask)
                if m_new < M - 1e-18:
                    cand = (m_new, s, d, tmove)
                    if best_cand is None or cand[0] < best_cand[0]:
                        best_cand = cand
                        improved_any = True

        if best_cand is not None:
            m_new, s, d, tmove = best_cand
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = fft_conv(a)
            M = proxy_max(c, mask)
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and (rnd <= 3 or rnd % 25 == 0):
                print("    Coord rnd=" + str(rnd) +
                      " score_proxy=" + repr(2.0 * n * best_M) +
                      " move s=" + str(s) + " d=" + str(d))
        else:
            stall += 1
            if stall % 12 == 11:
                a, c, M, did = random_transfer_improve(a, c, M, mask, 120, rng)
                if did:
                    stall = 0
                    if M < best_M:
                        best_M = M
                        best_a = a.copy()
                else:
                    stall += 7

    return best_a


def default_sequence(n=5000):
    x = np.linspace(0.0, 1.0, int(n))
    return norm_sum1(np.sin(np.pi * x) ** 2)
```
