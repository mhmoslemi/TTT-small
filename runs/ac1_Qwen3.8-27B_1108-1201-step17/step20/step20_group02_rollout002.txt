def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs

def lp_fw(a0, time_limit, solver, initial_K=400, max_K=3000, add_per_iter=50):
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac_y = fast_convolve(y)
    best_m = ac_y.max()
    t0 = time.monotonic()

    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 600
    iter_count = 0
    stagnation_count = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if stagnation_count > 5:
                break
            # Restart with perturbed active set
            ac_curr = fast_convolve(y)
            n_pert = max(50, len(active_set) // 4)
            rng_r = np.random.default_rng(42 + stagnation_count)
            keep = list(rng_r.choice(len(active_set), size=len(active_set) - n_pert, replace=False))
            new_adds = np.argsort(ac_curr)[::-1][:n_pert * 2]
            new_set = set([active_set[i] for i in keep])
            for k in new_adds:
                new_set.add(int(k))
            active_set = sorted(new_set)
            no_improve = 0
            stagnation_count += 1
            continue

        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            stagnation_count = 0
        else:
            no_improve += 1

        # Trim active set
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep_idx = np.argsort(margins)[:int(max_K * 0.6)]
            active_set_base = [active_set[i] for i in keep_idx]
            violations = np.where(ac > M_curr - 1e-8)[0]
            for v in violations[:80]:
                if int(v) not in active_set_base:
                    active_set_base.append(int(v))
            active_set = sorted(set(active_set_base))

        K = len(active_set)
        if K == 0:
            break

        coeffs = build_constraint_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(100, K // 3)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]

        all_viol = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(all_viol) > 0:
            viol_amt = conv_ys[all_viol] - M_curr
            top_v = all_viol[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        # Line search
        best_alpha = 1.0 / (iter_count + 2.0)
        best_trial_M = fast_convolve(y).max()

        alphas_to_try = [1.0, 0.8, 0.6, 0.4, 0.25,
                         max(0.1, 3.0 / (iter_count + 2.0)),
                         max(0.05, 2.0 / (iter_count + 2.0)),
                         max(0.02, 1.0 / (iter_count + 1.0))]

        for alpha in alphas_to_try:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 2:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        if iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", |S|=" + str(len(active_set)))

    elapsed = time.monotonic() - t0
    print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", iters=" + str(iter_count))
    return best_a, best_m

# Spectral smoothing: use frequency domain to identify and reduce peaks
def spectral_smooth(a0, time_limit, num_iter=2000):
    """Use frequency domain analysis to identify and reduce autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(7)
    
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    for it in range(num_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        # Power spectrum
        P = np.abs(A) ** 2
        # Autocorrelation in frequency
        ac = fast_convolve(a)
        M = ac.max()
        peak_idx = np.argsort(ac)[::-1][:20]
        
        # Identify which frequency components contribute to peaks
        # For peak at lag k, the contribution from frequency f is P[f] * exp(2j*pi*f*k/fft_size)
        # We want to reduce P at frequencies that align at peak positions
        
        # Strategy: reduce amplitude at frequencies that constructively interfere at peaks
        # Compute the "peak contribution" for each frequency
        phase_at_peaks = np.zeros(len(A))
        for k in peak_idx[:10]:
            ki = int(k)
            phase_at_peaks += np.cos(2.0 * np.pi * np.arange(len(A)) * ki / fft_size)
        
        # Reduce amplitude where phase aligns with peaks
        eta = 0.02 / (1.0 + it / 100.0)
        reduction = eta * np.maximum(phase_at_peaks, 0) * P / (P.max() + 1e-20)
        
        A_new = A.copy()
        A_new[1:] *= (1.0 - reduction)  # Don't change DC
        # Preserve real constraint (a is real)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        M_new = fast_convolve(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try small random perturbation in frequency domain
            noise = rng.normal(0, 0.01, len(A))
            A_pert = A + noise
            A_pert[0] = A[0]
            a_pert = np.fft.irfft(A_pert, fft_size)[:n]
            a_pert = np.maximum(a_pert, 0.0)
            sp = a_pert.sum()
            if sp > 1e-15:
                a_pert = a_pert / sp
                M_pert = fast_convolve(a_pert).max()
                if M_pert < M:
                    a = a_pert

    return best_a, best_m

def batch_coordinate_polish(a0, time_limit, batch_size=5, rng_seed=77):
    """Batch coordinate descent: transfer mass from peak-contributing to non-peak positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 2000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for _ in range(batch_size * 10):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick a peak to reduce
            k = int(top_k[rng.integers(len(top_k))])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-20:
                continue

            # Source: position with highest contribution
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Destination: position with lowest contribution to any peak
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    in_peak[s2:e2 + 1] = True

            # Batch: transfer from top-batch sources to bottom-batch destinations
            batch_srcs = np.where(~in_peak)[0]
            # Actually we want to move FROM peak-contributing TO non-peak
            # src is already in peak region, find dst outside
            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 10:
                candidates = np.arange(n)

            # Pick batch destinations
            n_dst = min(batch_size, len(candidates))
            dst_candidates = candidates[rng.choice(len(candidates), size=min(n_dst * 3, len(candidates)), replace=False)]
            dst_candidates = dst_candidates[np.argsort(scores[dst_candidates])[:n_dst]]

            for dst in dst_candidates:
                dst = int(dst)
                if dst == src:
                    continue

                found = False
                for frac in [0.01, 0.005, 0.002, 0.001]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_convolve(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break

            if found:
                break

        if not improved:
            stall += 1
            if stall % 100 == 0:
                # Random perturbation
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.001, 0.01)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

    return best_a, best_m

def generate_initializations(n):
    """Generate diverse initial sequences."""
    x = np.linspace(0, 1, n)
    inits = []
    
    # Sine squared
    inits.append(norm_sum1(np.sin(np.pi * x) ** 2))
    
    # Raised cosine
    inits.append(norm_sum1(0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))))
    
    # Hamming window
    inits.append(norm_sum1(0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1))))
    
    # Blackman
    a1, a2, a3 = 0.42, 0.5, 0.08
    inits.append(norm_sum1(a1 - a2 * np.cos(2 * np.pi * x / (n-1)) + a3 * np.cos(4 * np.pi * x / (n-1))))
    
    # Flat-top approximation
    inits.append(norm_sum1(np.exp(-(x - 0.5)**2 * 2.0)))
    
    # Double bump
    inits.append(norm_sum1(np.exp(-((x-0.3)**2)*30) + np.exp(-((x-0.7)**2)*30)))
    
    # Power law (concentrated in middle)
    inits.append(norm_sum1((0.5 - np.abs(x - 0.5)) ** 0.5 + 1e-10))
    
    # Random (fixed seed)
    rng = np.random.default_rng(123)
    inits.append(norm_sum1(rng.uniform(0, 1, n)))
    
    # Von Mises-like
    inits.append(norm_sum1(np.exp(3.0 * np.cos(2 * np.pi * (x - 0.5)))))
    
    # Asymmetric
    inits.append(norm_sum1(np.exp(-((x - 0.4)**2) * 8) + 0.3 * np.exp(-((x - 0.75)**2) * 15)))
    
    # Triangle
    inits.append(norm_sum1(1 - 2*np.abs(x - 0.5)))
    
    # Multi-peak
    inits.append(norm_sum1(np.exp(-((x-0.2)**2)*40) + np.exp(-((x-0.5)**2)*40) + np.exp(-((x-0.8)**2)*40)))
    
    # Kaiser-like (beta=8)
    try:
        from scipy.special import i0
        r = (x - 0.5) * 2  # -1 to 1
        inits.append(norm_sum1(i0(8.0 * np.sqrt(1 - r**2)) / i0(8.0)))
    except:
        pass
    
    # Array (uniform)
    inits.append(np.ones(n) / n)
    
    return inits

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()

        try:
            x = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])

            if prob.status in ('optimal', 'optimal_inaccurate'):
