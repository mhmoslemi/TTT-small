```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using soft-max smoothed multiplicative
weights (mirror descent) with adaptive temperature and multi-length exploration.
"""

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_grad(a, ac, T, K):
    """Compute gradient of soft-max approximation to max autocorrelation."""
    n = len(a)
    # Select top K lags
    k_idx = np.argsort(ac)[::-1][:K]
    k_vals = ac[k_idx]
    # Exponential weights
    w = np.exp(T * (k_vals - k_vals[0]))
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(k_vals)
        w[0] = 1.0
    else:
        w /= ws
    # Build full weight vector
    w_full = np.zeros(2 * n - 1)
    w_full[k_idx] = w
    # Gradient = 2 * conv(w_full, a_reversed)[:n]
    a_rev = a[::-1]
    req_size = (2 * n - 1) + n - 1
    fft_size = 1 << (req_size - 1).bit_length()
    W = np.fft.rfft(w_full, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    grad = 2.0 * np.fft.irfft(W * AR, fft_size)[:n]
    return grad

def obj(a):
    """Objective value for normalized sequence."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def interpolate_to(f, new_n):
    """Interpolate sequence to new length, maintaining normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def mirror_descent_optimize(a0, time_limit, T0=30.0, K0=100, Kmax=3000, eta0=0.05, verbose=True):
    """Main optimization: soft-max mirror descent with adaptive parameters."""
    a = normalize(a0)
    n = len(a)
    best_a = a.copy()
    best_val = obj(a)
    t0 = time.monotonic()
    
    T = T0
    K = K0
    eta = eta0
    stall = 0
    rng = np.random.default_rng(123)
    
    # Running average (Polyak)
    polyak_a = a.copy()
    polyak_count = 0
    
    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 50000:
            break
        
        ac = fast_autocorr(a)
        max_ac = ac.max()
        cur_val = 2.0 * n * max_ac
        
        if cur_val < best_val - 1e-13:
            best_val = cur_val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Compute gradient
        grad = compute_grad(a, ac, T, K)
        
        # Mirror descent update
        log_a = np.log(a + 1e-30) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-25)
        a_new = a_new / a_new.sum()
        
        # Check if improvement
        new_ac = fast_autocorr(a_new)
        new_val = 2.0 * n * new_ac.max()
        
        if new_val < cur_val:
            a = a_new
            # Polyak averaging
            polyak_count += 1
            polyak_a = (polyak_a * (polyak_count - 1) + a_new) / polyak_count
        else:
            # Reject, shrink step size
            eta *= 0.8
        
        # Adapt T (increase to sharpen)
        T = min(T * 1.0005, 500.0)
        # Adapt K (increase to capture more lags)
        K = min(K + 2, Kmax)
        # Gradually decay eta
        eta = max(eta * 0.9999, 0.001)
        
        # Periodic restart with noise if stalled
        if stall > 10000 and stall % 5000 == 0:
            noise_scale = 0.01 * stall / 10000.0
            a_r = best_a + noise_scale * rng.standard_normal(n)
            a_r = np.maximum(a_r, 0.0)
            sr = a_r.sum()
            if sr > 1e-10:
                a = a_r / sr
            eta = eta0 * 0.5
            stall = 0
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print("    MD iter={} ({:.1f}s) best={:.8f} cur={:.8f} T={:.1f} K={} eta={:.6f} stall={}".format(
                it, elapsed, best_val, cur_val, T, K, eta, stall))
    
    # Compare Polyak average with best
    p_val = obj(polyak_a)
    if p_val < best_val:
        best_a = polyak_a
        best_val = p_val
    
    return best_a, best_val

def coord_polish(a0, time_limit, verbose=True):
    """Coordinate descent polish: move mass from high-pressure to low-pressure coords."""
    a = normalize(a0)
    n = len(a)
    best_a = a.copy()
    best_val = obj(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 5000:
        ac = fast_autocorr(a)
        max_ac = ac.max()
        top_k = np.argsort(ac)[::-1][:20]
        
        # Compute pressure
        pressure = np.zeros(n)
        for ki in top_k:
            ki = int(ki)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                pressure[idx] += ac[ki] * a[ki - idx]
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Symmetrized soft-max smoothed multiplicative weights (mirror descent)
2. Adaptive temperature and top-K lag selection
3. Polyak averaging for convergence

Key insight: optimal sequence can be assumed symmetric (a[i] = a[n-1-i])

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def symmetrize(a):
    return (a + a[::-1]) * 0.5

def compute_grad_fft(a, ac, T, K):
    n = len(a)
    k_idx = np.argsort(ac)[::-1][:K]
    k_vals = ac[k_idx]
    w = np.exp(T * (k_vals - k_vals[0]))
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(k_vals)
        w[0] = 1.0
    else:
        w /= ws
    w_full = np.zeros(2 * n - 1)
    w_full[k_idx] = w
    a_rev = a[::-1]
    req_size = (2 * n - 1) + n - 1
    fft_size = 1 << (req_size - 1).bit_length()
    W = np.fft.rfft(w_full, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    grad = 2.0 * np.fft.irfft(W * AR, fft_size)[:n]
    return grad

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def interpolate_to(f, new_n):
    old_n = len(f)
    if old_n == new_n:

    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        a_new = a_new / a_new.sum()
        if use_sym:
            a_new = symmetrize(a_new)
            a_new = a_new / a_new.sum()

        # FISTA extrapolation
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista))
        theta = (t_fista - 1.0) / t_next
        y = a_new + theta * (a_new - a_prev)
        if use_sym:
            y = symmetrize(y)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = a_new.copy()
        else:
            y = y / sy

        a_prev = a.copy()
        a = a_new
        t_fista = t_next

        # Adapt
        T = min(T * 1.0004, 600.0)
        K = min(K + 1, Kmax)
        eta = max(eta * 0.99998, 0.001)

        # Damped FISTA if objective doesn't decrease
        new_cur = 2.0 * n * fast_autocorr(a).max()
        if new_cur > cur_val + 1e-10:
            t_fista = 1.0
            eta = max(eta * 0.8, 0.001)

        if stall > 15000 and stall % 5000 == 0:
            noise_scale = 0.02
            a_r = best_a + noise_scale * rng.standard_normal(n) * best_a.max()
            a_r = np.maximum(a_r, 0.0)
            if use_sym:
                a_r = symmetrize(a_r)
            sr = a_r.sum()
            if sr > 1e-10:
                a = a_r / sr
                y = a.copy()
                a_prev = a.copy()
            t_fista = 1.0
            eta = eta0 * 0.5
            stall = 0

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print("    AMD it={:7d} ({:6.1f}s) best={:.9f} T={:6.1f} K={:5d} t_f={:.1f} stall={}".format(
                it, elapsed, best_val, T, K, t_fista, stall))

    return best_a, best_val

def coord_polish(a0, time_limit, verbose=True):
    a = normalize(a0)
    a = symmetrize(a)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(321)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 8000:
        ac = fast_autocorr(a)
        max_ac = ac.max()
        top_k = np.argsort(ac)[::-1][:15]

        # Compute pressure (weight by ac value)
        pressure = np.zeros(n)
        for ki in top_k:
            ki = int(ki)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                pressure[idx] += ac[ki] / max_ac * a[ki - idx]

        # Symmetrize pressure for symmetric search
        pressure = symmetrize(pressure)

        src = int(np.argmax(pressure))
        if pressure[src] < 1e-18:
            break

        # Find low-pressure destination (also consider symmetric partner)
        dst_candidates = np.argsort(pressure)[:80]
        improved = False

        for dst in dst_candidates:
            dst = int(dst)
            if dst == src or dst == n - 1 - src:
                continue
            for frac in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-18:
                    break
                a_t = a.copy()
                a_t[src] -= amt
                a_t[n - 1 - src] -= amt
                a_t[dst] += amt
                a_t[n - 1 - dst] += amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                Mt = fast_autocorr(a_t).max()
                if Mt < max_ac - 1e-15:
                    a = a_t
                    max_ac = Mt
                    best_val = 2.0 * n * Mt
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 500 and stall % 100 == 0:
                # Symmetric random perturbation
                i = int(rng.integers(n // 2))
                j = int(rng.integers(n // 2))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[n - 1 - i] -= amt
                    a[j] += amt
                    a[n - 1 - j] += amt
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                        Mt = fast_autocorr(a).max()
                        if Mt < max_ac:
                            best_val = 2.0 * n * Mt
                            best_a = a.copy()
                            stall = 0
        else:
            stall = 0

        if verbose and stall % 3000 == 0 and stall > 0:
            print("    Coord ({:.1f}s) best={:.9f} stall={}".format(
                time.monotonic() - t0, best_val, stall))

    return best_a, best_val

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            bs = symmetrize(best_seq)
            best_seq = bs / bs.sum()
            best_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print("Parent (symmetrized): n={} score={:.9f}".format(len(p), best_score))
    except Exception as e:
        print("Parent load failed: {}".format(e))
        best_seq = None

    if best_seq is None:
        n_init = 11700
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_seq = symmetrize(best_seq) / symmetrize(best_seq).sum()
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))
        print("Default init n={} score={:.9f}".format(n_init, best_score))

    n0 = len(best_seq)
    print("Start: n={} score={:.9f}".format(n0, best_score))

    # Phase 1: Standard mirror descent (longer run, moderate T)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 60:
        print("\n=== P1: Mirror Descent (n={}, T0=40, K0=100) ===".format(n0))
        a_out, v_out = mirror_descent(best_seq, t1, T0=40.0, K0=100, Kmax=2000, eta0=0.05, use_sym=True)
        sc = float(evaluate_sequence([float(x) for x in a_out]))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P1 improved: {:.9f}".format(best_score))
        else:
            print("  P1 no improvement: {:.9f}".format(sc))

    # Phase 2: Accelerated MD (FISTA-style)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 40:
        print("\n=== P2: Accelerated MD (n={}) ===".format(n_cur))
        a_out, v_out = accelerated_md(best_seq, t2, T0=50.0, K0=200, Kmax=3000, eta0=0.03, use_sym=True)
        sc = float(evaluate_sequence([float(x) for x in a_out]))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P2 improved: {:.9f}".format(best_score))
        else:
            print("  P2 no improvement: {:.9f}".format(sc))

    # Phase 3: Length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 200:
        print("\n=== P3: Length exploration ===")
        n_cur = len(best_seq)
        lengths = [n_cur + 1000, n_cur + 2000, n_cur - 1000, n_cur + 3000, n_cur - 2000]
        lengths = [l for l in lengths if 5000 <= l <= 20000]
        # Deduplicate
        seen = set()
        lengths_unique = []
        for l in lengths:
            if l not in seen:
                seen.add(l)
                lengths_unique.append(l)
        t_per = (remaining * 0.6) / max(1, len(lengths_unique))
        for new_n in lengths_unique:
            if time.monotonic() - t_global > time_budget:
                break
            t_sub = min(t_per, 250.0)
            if t_sub < 30:
                continue
            print("  Trying n={}".format(new_n))
            a_interp = interpolate_to(best_seq, new_n)
            a_interp = symmetrize(a_interp) / symmetrize(a_interp).sum()
            a_out, v_out = mirror_descent(a_interp, t_sub * 0.65, T0=30.0, K0=100, Kmax=1500, eta0=0.04, use_sym=True, verbose=False)
            a_out2, v_out2 = coord_polish(a_out, t_sub * 0.35, verbose=False)
            sc1 = float(evaluate_sequence([float(x) for x in a_out]))
            sc2 = float(evaluate_sequence([float(x) for x in a_out2]))
            if np.isfinite(sc1) and sc1 < best_score:
                best_score = sc1
                best_seq = a_out
                print("  n={} MD improved: {:.9f}".format(new_n, best_score))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_out2
                print("  n={} Coord improved: {:.9f}".format(new_n, best_score))
            else:
                print("  n={} best: {:.9f}".format(new_n, min(sc1, sc2) if np.isfinite(min(sc1, sc2)) else float('inf')))

    # Phase 4: High-T mirror descent (sharp focus on active lags)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        print("\n=== P4: High-T MD (n={}, T0=100) ===".format(n_cur))
        a_out, v_out = mirror_descent(best_seq, t4, T0=100.0, K0=500, Kmax=5000, eta0=0.015, use_sym=True)
        sc = float(evaluate_sequence([float(x) for x in a_out]))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P4 improved: {:.9f}".format(best_score))
        else:
            print("  P4 no improvement: {:.9f}".format(sc))

    # Phase 5: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 20:
        print("\n=== P5: Coord Polish (n={}) ===".format(n_cur))
        a_out, v_out = coord_polish(best_seq, t5, verbose=True)
        sc = float(evaluate_sequence([float(x) for x in a_out]))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P5 improved: {:.9f}".format(best_score))
        else:
            print("  P5 no improvement: {:.9f}".format(sc))

    # Phase 6: Final short MD burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 20:
        print("\n=== P6: Final MD burst (n={}) ===".format(n_cur))
        a_out, v_out = mirror_descent(best_seq, t6, T0=200.0, K0=500, Kmax=5000, eta0=0.008, use_sym=True, verbose=False)
        sc = float(evaluate_sequence([float(x) for x in a_out]))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P6 improved: {:.9f}".format(best_score))
        else:
            print("  P6 no improvement: {:.9f}".format(sc))

    # Final validation
    if best_seq is not None:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.9f} n={} t={:.1f}s".format(best_score, len(best_seq), elapsed))
    else:
        n_fb = 10000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))
        print("\nFALLBACK: score={:.9f}".format(best_score))

    return [float(x) for x in best_seq]
```
