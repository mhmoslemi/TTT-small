import numpy as np
import time

def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def spectral_flatten(a, n_iters, alpha=0.85, L_factor=4, verbose=False):
    """
    Gerchberg-Saxton style spectral flattening.
    
    Parameters:
    - a: input sequence (non-negative, normalized to sum 1)
    - n_iters: number of iterations
    - alpha: flattening strength (0=full flatten, 1=no change). 
              Lower = more aggressive flattening.
    - L_factor: zero-pad factor (L = L_factor * n, rounded to power of 2)
    - verbose: print progress
    """
    n = len(a)
    x = a.copy().astype(np.float64)
    best_a = x.copy()
    best_obj = obj_val(x)
    
    L = 1 << (int(L_factor * n) - 1).bit_length()
    
    for it in range(n_iters):
        # Zero-pad and FFT
        x_padded = np.zeros(L)
        x_padded[:n] = x
        X = np.fft.rfft(x_padded)
        power = X * np.conj(X)  # = |X|^2
        
        # Flatten towards target
        mean_power = power.mean()
        # target_power = alpha * power + (1-alpha) * mean_power
        # New magnitude = sqrt(target_power)
        target_power = alpha * power + (1.0 - alpha) * mean_power
        # Ensure non-negative
        target_power = np.maximum(target_power, 0.0)
        new_mag = np.sqrt(target_power)
        
        # Keep original phase
        phase = np.angle(X)
        X_new = new_mag * np.exp(1j * phase)
        
        # Inverse FFT
        x_new = np.fft.irfft(X_new, L)
        x = x_new[:n]
        
        # Project to non-negative and normalize
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            x = best_a.copy()
            break
        x = x / s
        
        # Track best
        cur_obj = obj_val(x)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = x.copy()
        
        if verbose and it % 100 == 0:
            print("    SpecFlat it={} obj={:.7f}".format(it, cur_obj))
    
    return best_a, best_obj

    """
    Compute gradient of sum_k w[k]*conv(a,a)[k] with respect to a.
    grad[m] = 2 * sum_k w[k] * a[k-m] (with a[i]=0 for i out of [0,n-1]).
    """
    n = len(a)
    Kw = len(w)
    L = 1 << (Kw + n - 2).bit_length()
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    cross = np.fft.irfft(W * np.conj(A), L)
    return 2.0 * cross[:n]

def spectral_flatten(a0, time_limit, alpha_start=0.8, alpha_end=0.99,
                     L_factor=4, verbose=True):
    """
    Gerchberg-Saxton style spectral flattening to reduce autocorrelation peak.
    Progressively reduces flattening aggressiveness for convergence.
    """
    n = len(a0)
    x = a0.copy().astype(np.float64)
    best_a = x.copy()
    best_obj = obj_val(x)
    t0 = time.monotonic()
    L = 1 << (int(L_factor * n) - 1).bit_length()
    it = 0

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        x_padded = np.zeros(L)
        x_padded[:n] = x
        X = np.fft.rfft(x_padded)
        power = np.abs(X) ** 2

        mean_power = power.mean()
        target_power = alpha * power + (1.0 - alpha) * mean_power
        target_power = np.maximum(target_power, 1e-30)
        new_mag = np.sqrt(target_power)

        phase = np.angle(X)
        X_new = new_mag * np.exp(1j * phase)
        x_new = np.fft.irfft(X_new, L)
        x = x_new[:n]

        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            x = best_a.copy()
            break
        x = x / s

        cur_obj = obj_val(x)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = x.copy()

        it += 1
        if verbose and it % 200 == 0:
            print("    SpecFlat it={} ({:.1f}s) obj={:.7f} alpha={:.4f}".format(
                it, time.monotonic() - t0, best_obj, alpha))

    if verbose:
        print("    SpecFlat done ({:.1f}s) obj={:.7f}".format(
            time.monotonic() - t0, best_obj))
    return best_a, best_obj

def spectral_flatten_phase_random(a0, time_limit, n_particles=5, L_factor=4, verbose=True):
    """
    Multi-particle spectral flattening with random phase perturbations.
    Tries multiple phase perturbations and keeps the best result.
    """
    n = len(a0)
    best_a = a0.copy().astype(np.float64)
    best_obj = obj_val(best_a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    L = 1 << (int(L_factor * n) - 1).bit_length()
    time_per_particle = time_limit / n_particles

    for p in range(n_particles):
        if time.monotonic() - t0 >= time_limit:
            break
        particle_time = min(time_per_particle, time_limit - (time.monotonic() - t0))

        x = best_a.copy()
        for it in range(5000):
            if time.monotonic() - t0 >= time_limit:
                break

            x_padded = np.zeros(L)
            x_padded[:n] = x
            X = np.fft.rfft(x_padded)
            power = np.abs(X) ** 2
            mean_power = power.mean()

            alpha = 0.85 + 0.1 * (it / 500.0)
            target_power = alpha * power + (1.0 - alpha) * mean_power
            target_power = np.maximum(target_power, 1e-30)
            new_mag = np.sqrt(target_power)

            # Add random phase perturbation
            if it < 50:
                phase_noise = rng.standard_normal(len(X)) * 0.1
                phase = np.angle(X) + phase_noise
            else:
                phase = np.angle(X)

            X_new = new_mag * np.exp(1j * phase)
            x_new = np.fft.irfft(X_new, L)
            x = x_new[:n]
            x = np.maximum(x, 0.0)
            s = x.sum()
            if s < 1e-15:
                x = best_a.copy()
                break
            x = x / s

            cur_obj = obj_val(x)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = x.copy()

        if verbose:
            print("    SpecFlatPhase particle={} ({:.1f}s) obj={:.7f}".format(
                p, time.monotonic() - t0, best_obj))

    return best_a, best_obj

def grad_peak_descent(a0, time_limit, n_top=100, lr0=0.08, verbose=True):
    """
    Gradient descent on the sum of top-K autocorrelation peaks.
    Uses mirror descent (log-space) for non-negativity.
    """
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_obj = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 8000:
            break

        ac = fast_autocorr(x)
        M = ac.max()
        cur_obj = 2.0 * n * M

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        k_sel = min(n_top, len(ac))
        idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = ac[idx]
        top_max = top_vals[0] + 1e-15

        temp = 20.0 + 10.0 * (t / 1000.0)
        w = np.exp(temp * (top_vals - top_max) / top_max)
        w = w / w.sum()

        w_full = np.zeros(len(ac))
        w_full[idx] = w
        grad = compute_autocorr_gradient(x, w_full)

        grad_centered = grad - grad.sum() / n
        lr = lr0 / (1.0 + 0.005 * t)

        log_x = np.log(x + 1e-30) - lr * grad_centered / (x + 1e-30)
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        x_new = np.maximum(x_new, 1e-25)
        x = x_new / x_new.sum()

        if stall > 3000 and stall % 1000 == 0:
            noise_scale = 0.01 * (stall / 3000.0)
            noise = noise_scale * rng.standard_normal(n) * best_a
            x = best_a + noise
            x = np.maximum(x, 1e-12)
            x = x / x.sum()
            stall = 0

        if verbose and t % 300 == 0:
            print("    GradDescent t={} ({:.1f}s) obj={:.7f} stall={}".format(
                t, time.monotonic() - t0, best_obj, stall))

    return best_a, best_obj

def coord_search_peaks(a0, time_limit, n_targets=20, verbose=True):
    """
    For each top autocorrelation peak, find the coordinate contributing most
    to it and try reducing that coordinate's value.
    """
    n = len(a0)
    x = a0.copy().astype(np.float64)
    sx = x.sum()
    best_a = x.copy()
    best_obj = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 3000:
        ac = fast_autocorr(x)
        cur_obj = 2.0 * n * ac.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        top_k = np.argsort(ac)[::-1][:min(n_targets, len(ac))]
        improved = False

        for k in top_k[:10]:
            ki = int(k)
            ws = max(0, ki - n + 1)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            contrib = x[idx] * x[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])

            # Find quiet positions (low total pressure from top peaks)
            pressure = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                ws2 = max(0, k2i - n + 1)
                we2 = min(n - 1, k2i)
                if ws2 <= we2:
                    idx2 = np.arange(ws2, we2 + 1)
                    pressure[idx2] += x[ki - idx2] * 0 + x[idx2]
            pressure[src] = 1e10
            quiet = np.argsort(pressure)[:50]

            for dst in quiet:
                dst = int(dst)
                if dst == src:
                    continue

                found = False
                for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.2, 0.35]:
                    amt = x[src] * frac
                    if amt < 1e-18:
                        break
                    x_t = x.copy()
                    x_t[src] -= amt
                    x_t[dst] += amt
                    x_t = np.maximum(x_t, 0.0)
                    st = x_t.sum()
                    if st < 1e-15:
                        continue
                    Mt = fast_autocorr(x_t).max()
                    if Mt < ac.max() - 1e-15:
                        x = x_t
                        improved = True
                        found = True
                        break
                if found:
                    break

        if not improved:
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and x[i] > 1e-10:
                frac = rng.uniform(0.01, 0.08)
                amt = x[i] * frac
                x[i] -= amt
                x[j] += amt
                x = np.maximum(x, 0.0)
                Mn = fast_autocorr(x).max()
                if Mn < ac.max():
                    stall = 0
                else:
                    x = best_a.copy()

        if verbose and stall % 500 == 0 and stall > 0:
            print("    CoordSearch ({:.1f}s) obj={:.7f} stall={}".format(
                time.monotonic() - t0, best_obj, stall))

    return best_a, best_obj

def try_shorter_length(a0, target_n, time_limit, verbose=True):
    """Try to find a good sequence at a shorter length by truncation/optimization."""
    n = len(a0)
    if target_n >= n:
        return a0, obj_val(a0)

    # Take the middle portion or use interpolation
    x = np.interp(
        np.linspace(0, 1, target_n),
        np.linspace(0, 1, n),
        a0
    )
    x = np.maximum(x, 0.0)
    x = x / x.sum()

    # Quick spectral flatten
    best_a, best_obj = spectral_flatten(x, min(time_limit * 0.4, 60), verbose=False)

    # Quick gradient descent
    best_a2, best_obj2 = grad_peak_descent(best_a, min(time_limit * 0.4, 60),
                                            n_top=50, verbose=False)
    if best_obj2 < best_obj:
        best_a, best_obj = best_a2, best_obj2

    return best_a, best_obj

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent loaded: n={} score={:.7f}".format(len(p), best_score))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_init = 10000
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)
        print("Default init: n={} score={:.7f}".format(n_init, best_score))

    remaining = time_budget - (time.monotonic() - t_global)
    n0 = len(best_seq)

    # Phase 1: Spectral flattening (the main novel contribution)
    print("\n=== P1: Spectral Flattening (n={}) ===".format(n0))
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        a_out, v_out = spectral_flatten(normalize(best_seq), t1,
                                         alpha_start=0.75, alpha_end=0.98,
                                         L_factor=4, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P1 improved: {:.7f}".format(best_score))

    # Phase 2: Multi-particle spectral flattening with phase randomization
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P2: SpecFlat Phase Random (n={}) ===".format(len(best_seq)))
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 30:
        a_out, v_out = spectral_flatten_phase_random(normalize(best_seq), t2,
                                                      n_particles=4, L_factor=4, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P2 improved: {:.7f}".format(best_score))

    # Phase 3: Gradient-based peak descent
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P3: Gradient Peak Descent (n={}) ===".format(len(best_seq)))
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30:
        a_out, v_out = grad_peak_descent(normalize(best_seq), t3,
                                          n_top=150, lr0=0.06, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P3 improved: {:.7f}".format(best_score))

    # Phase 4: Coordinate search on peaks
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P4: Coord Search (n={}) ===".format(len(best_seq)))
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_out, v_out = coord_search_peaks(normalize(best_seq), t4,
                                           n_targets=20, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P4 improved: {:.7f}".format(best_score))

    # Phase 5: Second round of spectral + gradient (from improved point)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print("\n=== P5: Second round SpecFlat+Grad (n={}) ===".format(len(best_seq)))
        t5a = min(remaining * 0.3, 200.0)
        t5b = min(remaining * 0.3, 200.0)
        if t5a > 20:
            a_out, _ = spectral_flatten(normalize(best_seq), t5a,
                                         alpha_start=0.9, alpha_end=0.995,
                                         L_factor=6, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  P5a improved: {:.7f}".format(best_score))
        remaining = time_budget - (time.monotonic() - t_global)
        if t5b > 20 and remaining > 20:
            a_out, _ = grad_peak_descent(normalize(best_seq), t5b,
                                          n_top=200, lr0=0.04, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  P5b improved: {:.7f}".format(best_score))

    # Phase 6: Try different length (interpolation)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_alt = min(len(best_seq) + 2000, 20000)
        if n_alt != len(best_seq):
            print("\n=== P6: Alt length n={} ===".format(n_alt))
            t6 = min(remaining * 0.2, 200.0)
            if t6 > 20:
                x_alt = np.interp(
                    np.linspace(0, 1, n_alt),
                    np.linspace(0, 1, len(best_seq)),
                    best_seq
                )
                x_alt = np.maximum(x_alt, 0.0)
                x_alt = x_alt / x_alt.sum()
                a_out, _ = spectral_flatten(x_alt, t6 * 0.5,
                                             alpha_start=0.8, alpha_end=0.99,
                                             verbose=False)
                a_out2, _ = grad_peak_descent(a_out, t6 * 0.4,
                                               n_top=100, lr0=0.05, verbose=False)
                sc1 = eval_safe(a_out)
                sc2 = eval_safe(a_out2)
                best_sc = min(sc1, sc2)
                if np.isfinite(best_sc) and best_sc < best_score:
                    best_score = best_sc
                    best_seq = a_out if sc1 < sc2 else a_out2
                    print("  P6 improved: {:.7f}".format(best_score))

    # Phase 7: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== P7: Final Coord Polish (n={}) ===".format(len(best_seq)))
        t7 = min(remaining * 0.3, 200.0)
        if t7 > 15:
            a_out, _ = coord_search_peaks(normalize(best_seq), t7,
                                            n_targets=30, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  P7 improved: {:.7f}".format(best_score))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.7f} n={} t={:.1f}s".format(
            best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: using fallback")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)

    return [float(x) for x in best_seq]
