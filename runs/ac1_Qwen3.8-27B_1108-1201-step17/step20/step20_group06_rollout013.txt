    return _MODULE_SOLVER


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute 2*n*max_b/sum_a^2 without calling evaluate_sequence."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    return float(2.0 * n * max_b / (s * s))


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_grad_smoothed(y, beta, top_k=None):
    """Smoothed gradient of max autocorrelation."""
    n = len(y)
    b = fast_autocorr(y)
    size = 2 * n - 1
    
    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    
    if b_max <= 0:
        return np.zeros(n), b
    
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s
    
    # grad[j] = sum_k p[k] * y[k-j] (for valid k-j)
    # This is a cross-correlation of p with y
    a_rev = y[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, b


def gradient_descent_v4(a0, time_limit, beta_start=50.0, beta_end=3000.0,
                        beta_growth=1.004, eta_init=0.3, top_k_ratio=0.02,
                        restart_every=800, verbose=True, seed=42):
    """
    Nesterov gradient descent with:
    - Adaptive beta based on peak gap
    - Backtracking line search
    - Multiple restart strategies
    """
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    eta = eta_init
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        grad, b = compute_grad_smoothed(y, beta, top_k)
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-15:
            x_new = x
            max_new = fast_autocorr(x_new).max()
        else:
            # Backtracking
            eta_trial = eta
            found = False
            y_max = fast_autocorr(y).max()
            for _ in range(8):
                z = y - eta_trial * grad
                x_new = simplex_project(z)
                max_new = fast_autocorr(x_new).max()
                if max_new <= y_max + 1e-15:
                    found = True
                    break
                eta_trial *= 0.5
            if not found:
                x_new = x
                max_new = fast_autocorr(x_new).max()
                eta = max(eta_trial, 0.01)
        
        if max_new < best_m - 1e-16:
            best_m = max_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.05, 1.5)
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        # Adaptive beta: increase faster when peaks are close
        if t % 30 == 0:
            b_curr = fast_autocorr(x)
            sorted_b = np.sort(b_curr)[::-1]
            gap = sorted_b[0] - sorted_b[1] if len(sorted_b) > 1 else 1.0
            # If gap is small, increase beta to focus more on the max
            if gap < 0.01 * sorted_b[0]:
                beta = min(beta * 1.01, beta_end)
            else:
                beta = min(beta * beta_growth, beta_end)
        
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strat = int(rng.integers(7))
            if strat == 0:
                noise = 0.01 + 0.02 * rng.random()
                x_r = best_a + noise * rng.standard_normal(n) * best_a
            elif strat == 1:
                # Target the peak: increase values at positions that DON'T contribute to peak
                b_curr = fast_autocorr(best_a)
                peak_k = np.argmax(b_curr)
                # Positions contributing to peak
                js = max(0, peak_k - n + 1)
                je = min(n, peak_k + 1)
                hot = np.zeros(n)
                for i in range(js, je):
                    partner = peak_k - i
                    if 0 <= partner < n:
                        hot[i] = best_a[partner]
                cold = 1.0 - hot / (hot.max() + 1e-15)
                x_r = best_a * (1 + 0.05 * cold * rng.standard_normal(n))
            elif strat == 2:
                # Power perturbation
                x_r = best_a * np.exp(0.04 * rng.standard_normal(n))
            elif strat == 3:
                # Block swap
                blk = max(10, n // 10)
                x_r = best_a.copy()
                for _ in range(3):
                    i = int(rng.integers(0, n - blk))
                    j = int(rng.integers(0, n - blk))
                    if abs(i - j) < blk:
                        continue
                    x_r[i:i+blk], x_r[j:j+blk] = x_r[j:j+blk].copy(), x_r[i:i+blk].copy()
            elif strat == 4:
                # Scale hot/cold regions
                b_curr = fast_autocorr(best_a)
                above_mean = (b_curr > b_curr.mean()).astype(float)
                # Map back to position importance
                pos_importance = np.zeros(n)
                for k in range(len(b_curr)):
                    if b_curr[k] > b_curr.mean():
                        js = max(0, k - n + 1)
                        je = min(n, k + 1)
                        pos_importance[js:je] += 1
                x_r = best_a.copy()
                x_r[pos_importance > 0] *= 0.95
                x_r[pos_importance == 0] *= 1.05
            elif strat == 5:
                # Frequency domain perturbation
                fft_size = 1 << (2 * n - 2).bit_length()
                A = np.fft.rfft(best_a, fft_size)
                phase_pert = rng.standard_normal(len(A)) * 0.01
                A[1:] *= np.exp(1j * phase_pert[1:])
                x_r = np.fft.irfft(A, fft_size)[:n]
            else:
                # Blend with smooth reference
                xx = np.linspace(0, 1, n)
                ref = np.sin(np.pi * xx) ** 1.5
                ref = norm_sum1(ref)
                x_r = 0.7 * best_a + 0.3 * ref
            
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 10.0)
            eta = eta_init
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("  GD4 restart t=" + str(t) + " (" + str(round(elapsed,1)) + "s) J=" + str(round(J,6)))
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  GD4 t=" + str(t) + " (" + str(round(elapsed,1)) + "s) J=" + str(round(J,6)) + " b=int" + str(int(beta)))
    
    return best_a, best_m


def lp_direction(y, M, active_set, n, solver):
    """Solve LP to find direction that maintains constraint."""
    K = len(active_set)
    if K == 0:
        return None
    
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(active_set):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    
    g_var = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(g_var) == 1]
    if K <= 5000:
        constraints.append(coeffs @ g_var <= M)
    
    prob = cp.Problem(cp.Maximize(cp.min(g_var)), constraints)
    # Actually, we want to maximize sum while keeping constraints
    # But that's always 1. Let's minimize the max violation instead.
    # Or: find g that minimizes max_k (y*g)[k] subject to sum(g)=1, g>=0
    
    M_var = cp.Variable()
    constraints2 = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints2)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return None, None
    return g_val / g_sum, M_var.value


def lp_fw_v3(a0, time_limit, solver, initial_K=200, max_K=1500, verbose=True):
    """LP-Frank-Wolfe with improved active set management."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = list(np.argsort(ac_y)[::-1][:initial_K])
    
    no_improve = 0
    max_no_improve = 400
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Trim active set
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep = np.argsort(margins)[:max_K]
            active_set = [int(active_set[i]) for i in keep]
        
        if len(active_set) == 0:
            break
        
        # Solve LP
        g_dir, lp_obj = lp_direction(y, M_curr, active_set, n, solver)
        if g_dir is None:
            no_improve += 10
            continue
        
        # Add violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                if int(k) not in active_set:
                    active_set.append(int(k))
        
        # Line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        for alpha in [best_alpha, best_alpha*0.5, best_alpha*2.0, 0.05, 0.1, 0.15, 0.25]:
            alpha = max(0.01, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        y = (1.0 - best_alpha) * y + best_alpha * g_dir
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y / s_y
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  LP3 t=" + str(t) + " (" + str(round(elapsed,1)) + "s) J=" + str(round(J,6)) + " |S|=" + str(len(active_set)))
    
    return best_a, best_m


def peak_shave(a0, time_limit, verbose=True, seed=99):
    """
    Targeted peak shaving: identify the top contributing pairs to the peak
    and systematically redistribute mass.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if stall > 3000:
            break
        
        ac = fast_autocorr(a)
        size = 2 * n - 1
        peak_k = int(np.argmax(ac))
        M = ac[peak_k]
        
        # Find top contributing pairs to the peak
        js = max(0, peak_k - n + 1)
        je = min(n, peak_k + 1)
        idx = np.arange(js, je)
        partners = peak_k - idx
        contrib = a[idx] * a[partners]
        
        # Top contributors
        top_idx = idx[np.argsort(contrib)[::-1][:min(20, len(idx))]]
        top_contrib = np.sort(contrib)[::-1][:min(20, len(idx))]
        
        improved = False
        for src_i in top_idx:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src_i)
            partner = int(peak_k - src_i)
            if not (0 <= partner < n):
                continue
            
            # Try moving mass from src_i to a random other position
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                amt = a[src_i] * frac
                if amt < 1e-15:
                    continue
                for _ in range(8):
                    dst = int(rng.integers(n))
                    if dst == src_i or dst == partner:
                        continue
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-18:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation when stuck
            if stall % 50 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  PeakShave (" + str(round(elapsed,1)) + "s) J=" + str(round(J,6)) + " stall=" + str(stall))
    
    return best_a, best_m


def spectral_flat_v2(a0, time_limit, verbose=True, seed=555):
    """
    Spectral flattening: work in frequency domain to equalize power spectrum.
    Uses iterative amplitude/phase splitting.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    
    alpha = 0.05
    for it in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # Target: geometric mean of power spectrum (excluding DC)
        if len(power) > 1:
            log_p = np.log(power[1:] + 1e-30)
            target_log = np.mean(log_p)
            target = np.exp(target_log)
        else:
            target = 1.0
        
        A_new = A.copy()
        for k in range(1, len(A)):
            p_k = power[k]
            if p_k > target * 1.01:
                # Reduce
                scale = (target / p_k) ** (alpha / 2.0)
                A_new[k] *= scale
            elif p_k < target * 0.3:
                # Small boost
                scale = min(1.5, (target * 0.5 / p_k) ** (alpha / 4.0))
                A_new[k] *= scale
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Sometimes accept non-improving moves (simulated annealing style)
            if rng.random() < 0.1 and it < 10000:
                a = a_new
        
        if it % 500 == 0:
            alpha = max(0.01, alpha * 0.97)
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  SpecFlat2 it=" + str(it) + " (" + str(round(elapsed,1)) + "s) J=" + str(round(J,6)))
    
    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed,1)) + "s)")
    return best_seq, best_score
