        # Shape the spectrum to reduce peaks in time domain
        # Reduce amplitude where autocorrelation peaks are
        # This is a heuristic: reduce the "energy" at frequencies that contribute to peaks
        
        # Alternative: iterative amplitude adjustment
        # Compute the contribution of each frequency to the autocorrelation
        # and adjust amplitudes to flatten
        
        # Simple approach: reduce the overall amplitude at frequencies
        # where the phase alignment causes peaks
        # Use a "water-filling" approach on the amplitude spectrum
        
        power = np.abs(A) ** 2
        total_power = power.sum()
        
        # Target: flat power spectrum (which would give flat autocorrelation in principle)
        # But we can't achieve perfectly flat due to the non-negativity constraint
        # Instead, iteratively reduce contrast in the power spectrum
        
        target = total_power / len(power)
        alpha = 0.1  # Blending parameter
        new_power = alpha * target + (1 - alpha) * power
        new_amplitude = np.sqrt(new_power)
        new_phase = np.angle(A)
        
        A_new = new_amplitude * np.exp(1j * new_phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 100 == 0:
            J = 2.0 * n * best_m
            print("    Spectral it={}: J={:.6f}".format(it, J))
    
    return best_a, best_m

        return float(evaluate_sequence(a_list))
