    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    Returns (obj_value, grad_a) where grad_a is the gradient of 2n*logsumexp(beta*c)/beta w.r.t. a.

    c = fast_convolve(a)
    # Soft-max
    beta_c = beta * c

    exp_bc = np.exp(beta_c - max_bc)

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_softmax_gradient(a, beta):
    """Compute gradient of 2n*logsumexp(beta*c)/beta w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    beta_c = beta * c
    max_bc = beta_c.max()
    exp_bc = np.exp(beta_c - max_bc)
    sum_exp = exp_bc.sum()
    w = exp_bc / sum_exp  # softmax weights, length 2n-1
    # grad_a[j] = 4n * sum_{m=0}^{n-1} w[j+m] * a[m]
    # = 4n * (w * a_rev)[j + n - 1] where a_rev = a[::-1]
    N = 1 << (3 * n - 2).bit_length()
    w_p = np.zeros(N)
    w_p[:len(w)] = w
    a_rev = a[::-1]
    a_rev_p = np.zeros(N)
    a_rev_p[:n] = a_rev
    conv = np.fft.irfft(np.fft.rfft(w_p) * np.fft.rfft(a_rev_p), N)
    grad_a = 4.0 * n * conv[n - 1: 2 * n - 1]
    return grad_a, c

def mirror_descent(a0, time_limit, verbose=False, label=""):
    """Mirror descent (exponential weight) on the simplex with top-K subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    eta0 = 0.02
    t = 0
    stall = 0
    max_stall = 3000
    top_K = 20

    while t < 80000 and time.monotonic() - t0 < time_limit and stall < max_stall:
        t += 1
        eta = eta0 / (np.sqrt(t) + 1.0)

        # Compute autocorrelation
        c = fast_convolve(a)
        M = c.max()

        # Top-K subgradient (weighted combination)
        top_idx = np.argsort(c)[::-1][:top_K]
        weights = np.zeros(2 * n - 1)
        for k in top_idx:
            weights[k] = 1.0
        # Normalize weights
        weights = weights / weights.sum()

        # Compute weighted subgradient
        N = 1 << (3 * n - 2).bit_length()
        w_p = np.zeros(N)
        w_p[:len(weights)] = weights
        a_rev = a[::-1]
        a_rev_p = np.zeros(N)
        a_rev_p[:n] = a_rev
        conv = np.fft.irfft(np.fft.rfft(w_p) * np.fft.rfft(a_rev_p), N)
        grad = 2.0 * n * conv[n - 1: 2 * n - 1]

        # Mirror descent update: a_new[i] proportional to a[i] * exp(-eta * grad[i])
        log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        a_new = a_new / s

        # Evaluate
        val_new = 2.0 * n * fast_convolve(a_new).max()
        if val_new < best_val - 1e-14:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            # Stay at current a (don't accept)
            if stall % 500 == 0:
                eta0 *= 0.8
            if stall > 1000 and stall % 1000 == 0:
                # Perturb
                noise = rng.normal(0, 0.02, n) * best_a
                a_p = best_a + noise
                a = norm_sum1(a_p)
                eta0 *= 1.3

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print("    " + label + " MD t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): "
                  + "best=" + repr(best_val) + ", stall=" + str(stall))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    " + label + " MD done (" + str(round(elapsed, 1)) + "s): best=" + repr(best_val))
    return best_a, best_val

                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    danger[idx2] += a[k2i - idx2]

            # Transfer to a low-danger position
            candidates = np.where(danger < np.median(danger))[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src:
                continue

            # Try several transfer fractions
            found = False
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_val - 1e-15:
                    a = a_trial
                    best_val = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found:
                # Random small transfer
                j = int(rng.integers(n))
                if j != src:
                    frac = rng.uniform(0.001, 0.005)
                    amt = src_val * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_val - 1e-15:
                            best_val = M_new
                            best_a = a.copy()
                            improved = True
                            stall = 0

        if not improved:
            stall += 1
            if stall % 200 == 0 and stall < max_stall:
                # Random perturbation
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.01)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_val - 1e-15:
                                best_val = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    " + label + " Coord (" + str(round(elapsed, 1)) + "s): "
                  + "J=" + repr(2.0 * n * best_val) + ", stall=" + str(stall))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    " + label + " Coord done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_val))
    return best_a, best_val


def bump_optimize(a0, time_limit, n_bumps=8, seed=42, verbose=False, label=""):
    """Optimize Gaussian bump parameters using a simple coordinate descent + random search."""
    n = len(a0)
    x = np.linspace(0.0, 1.0, n)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Initialize: fit bumps to a0
    best_a = norm_sum1(a0)
    best_val = 2.0 * n * fast_convolve(best_a).max()

    # Initial parameters
    params = np.zeros(3 * n_bumps)
    centers = np.linspace(0.15, 0.85, n_bumps)
    for j in range(n_bumps):
        params[3 * j] = 1.0
        params[3 * j + 1] = centers[j]
        params[3 * j + 2] = 0.08

    def bumps_to_seq(p):
        a = np.zeros(n)
        for j in range(n_bumps):
            w = max(0.0, p[3 * j])
            c = np.clip(p[3 * j + 1], 0.02, 0.98)
            s = max(0.015, min(0.5, p[3 * j + 2]))
            a += w * np.exp(-((x - c) ** 2) / (2.0 * s * s))
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-10:
            return np.ones(n) / n
        return a / s_a

    def eval_params(p):
        a = bumps_to_seq(p)
        return 2.0 * n * fast_convolve(a).max()

    # Optimization: iterative perturbation with acceptance
    step_size = 0.1
    for it in range(20000):
        if time.monotonic() - t0 > time_limit:
            break
        # Random coordinate perturbation
        dim = 3 * n_bumps
        d = int(rng.integers(dim))
        delta = rng.uniform(-step_size, step_size)
        p_new = params.copy()
        p_new[d] += delta
        # Clip
        for j in range(n_bumps):
            p_new[3 * j] = max(0.0, min(5.0, p_new[3 * j]))
            p_new[3 * j + 1] = np.clip(p_new[3 * j + 1], 0.02, 0.98)
            p_new[3 * j + 2] = np.clip(p_new[3 * j + 2], 0.015, 0.5)

        val_new = eval_params(p_new)
        if val_new < best_val:
            best_val = val_new
            best_a = bumps_to_seq(p_new).copy()
            params = p_new
            step_size = min(step_size * 1.05, 0.3)
        else:
            step_size *= 0.995
            if step_size < 0.001:
                step_size = 0.01

        if it % 2000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print("    " + label + " Bump it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): "
                  + "J=" + repr(2.0 * n * best_val))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    " + label + " Bump done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_val))
    return best_a, best_val


def get_initializations(n, t_global):
    """Generate diverse initial sequences."""
    inits = []
    names = []

    # 1. Parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = interpolate_seq(p / p.sum(), n)
            inits.append(p_norm)
            names.append("parent")
    except Exception:
        pass

    # 2. Uniform
    inits.append(np.ones(n) / n)
    names.append("uniform")

    # 3. Raised cosine (Hann-like)
    x = np.linspace(0.0, 1.0, n)
    rc = np.sin(np.pi * x) ** 2
    inits.append(rc / rc.sum())
    names.append("raised_cos")

    # 4. Arch (bowl - higher at edges)
    arch = 1.0 + 0.3 * np.cos(2.0 * np.pi * x)
    arch = np.maximum(arch, 0.1)
    inits.append(arch / arch.sum())
    names.append("arch")

    # 5. Inverse arch (peak in middle)
    iarch = 1.0 - 0.3 * np.cos(2.0 * np.pi * x)
    iarch = np.maximum(iarch, 0.1)
    inits.append(iarch / iarch.sum())
    names.append("inv_arch")

    # 6. W-shape (two peaks at 1/3 and 2/3)
    w_shape = np.exp(-((x - 0.3) ** 2) / (2 * 0.15 ** 2)) + np.exp(-((x - 0.7) ** 2) / (2 * 0.15 ** 2)) + 0.3
    inits.append(w_shape / w_shape.sum())
    names.append("w_shape")

    # 7. M-shape (peaks at edges and center)
    m_shape = np.exp(-((x - 0.1) ** 2) / (2 * 0.1 ** 2)) + np.exp(-((x - 0.5) ** 2) / (2 * 0.1 ** 2)) + np.exp(-((x - 0.9) ** 2) / (2 * 0.1 ** 2)) + 0.2
    inits.append(m_shape / m_shape.sum())
    names.append("m_shape")

    # 8. Smooth step (high on left, low on right)
    step = 1.0 / (1.0 + np.exp(8.0 * (x - 0.5)))
    inits.append(step / step.sum())
    names.append("smooth_step")

    return inits, names


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total budget: " + repr(time_budget) + "s")

    def remaining():
        return time_budget - (time.monotonic() - t_global)

    # Determine n from parent or default
    n0 = 5000
    parent_init = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            n0 = len(p)
            parent_init = p / p.sum()
            print("Parent: n=" + str(n0) + ", score=" + repr(evaluate_safe(parent_init)))
    except Exception:
        pass

    # Choose n
    n = n0
    if n > 12000:
        n = 12000
    if n < 3000:
        n = 5000

    # Get initializations
    inits, names = get_initializations(n, t_global)
    print("Generated " + str(len(inits)) + " initializations for n=" + str(n))

    best_seq = None
    best_score = float('inf')

    # === Phase 1: Quick evaluation of all inits ===
    print("\n=== Phase 1: Evaluate initializations ===")
    for name, init in zip(names, inits):
        sc = evaluate_safe(init)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = init.copy()
            print("  [" + name + "]: score=" + repr(sc))
    print("  Best init: " + repr(best_score))

    # === Phase 2: Soft-max optimization on top inits ===
    print("\n=== Phase 2: Soft-max optimization ===")
    # Sort inits by score
    init_scores = []
    for name, init in zip(names, inits):
        sc = evaluate_safe(init)
        init_scores.append((sc, name, init))
    init_scores.sort(key=lambda x: x[0])

    # Optimize top 4 initializations
    n_optim = min(4, len(init_scores))
    per_init_time = min(remaining() * 0.30 / n_optim, 180.0)

    for i in range(n_optim):
        rt = remaining()
        if rt < 30:
            break
        sc, name, init = init_scores[i]
        t_alloc = min(per_init_time, rt * 0.25)
        if t_alloc < 15:
            break
        print("  Optimizing [" + name + "] (init score=" + repr(sc) + ", time=" + str(round(t_alloc, 1)) + "s)")
        a_out, _ = softmax_optimize(init, t_alloc, beta_start=30.0, beta_end=250.0,
                                    lr=0.008, verbose=False, label=name)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sm_" + name, t_global)

    # === Phase 3: Mirror descent on best ===
    print("\n=== Phase 3: Mirror descent ===")
    t3 = min(remaining() * 0.15, 200.0)
    if t3 > 30 and best_seq is not None:
        a_out, _ = mirror_descent(best_seq, t3, verbose=False, label="best")
        best_seq, best_score = try_update(best_seq, best_score, a_out, "MD_best", t_global)

    # === Phase 4: Bump optimization (global) ===
    print("\n=== Phase 4: Bump optimization ===")
    t4 = min(remaining() * 0.12, 150.0)
    if t4 > 25:
        # Try different bump configurations
        for nb in [6, 10]:
            rt = remaining()
            if rt < 20:
                break
            t4b = min(t4 / 2, rt * 0.4)
            if t4b < 15:
                break
            a_out, _ = bump_optimize(best_seq, t4b, n_bumps=nb, seed=42 + nb,
                                     verbose=False, label="bump" + str(nb))
            best_seq, best_score = try_update(best_seq, best_score, a_out, "bump" + str(nb), t_global)

    # === Phase 5: Soft-max with different hyperparams on best ===
    print("\n=== Phase 5: Second round soft-max ===")
    t5 = min(remaining() * 0.18, 250.0)
    if t5 > 30 and best_seq is not None:
        # Lower beta, smaller lr for finer optimization
        a_out, _ = softmax_optimize(best_seq, t5, beta_start=80.0, beta_end=400.0,
                                    lr=0.004, verbose=True, label="sm2")
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sm2", t_global)

    # === Phase 6: Multi-n exploration ===
    print("\n=== Phase 6: Multi-n ===")
    n_best = len(best_seq) if best_seq is not None else n
    try_ns = sorted(set([n_best - 500, n_best + 500, n_best + 1000, n_best - 1000]))
    try_ns = [l for l in try_ns if 3000 <= l <= 12000]
    try_ns.sort(key=lambda l: abs(l - n_best))

    for new_n in try_ns[:3]:
        rt = remaining()
        if rt < 60:
            break
        print("  Multi-n: n=" + str(new_n) + " (remaining=" + str(round(rt, 1)) + "s)")
        a_long = interpolate_seq(best_seq, new_n)

        t6 = min(rt * 0.15, 150.0)
        if t6 > 20:
            a_out, _ = softmax_optimize(a_long, t6, beta_start=50.0, beta_end=300.0,
                                        lr=0.006, verbose=False, label="n" + str(new_n))
            best_seq, best_score = try_update(best_seq, best_score, a_out, "sm_n" + str(new_n), t_global)

    # === Phase 7: Mirror descent on best ===
    print("\n=== Phase 7: Mirror descent round 2 ===")
    t7 = min(remaining() * 0.12, 150.0)
    if t7 > 25 and best_seq is not None:
        a_out, _ = mirror_descent(best_seq, t7, verbose=False, label="md2")
        best_seq, best_score = try_update(best_seq, best_score, a_out, "MD2", t_global)

    # === Phase 8: Coordinate polish ===
    print("\n=== Phase 8: Coordinate polish ===")
    t8 = min(remaining() * 0.15, 200.0)
    if t8 > 20 and best_seq is not None:
        a_out, _ = coordinate_polish(best_seq, t8, verbose=False, label="coord")
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord", t_global)

    # === Phase 9: Final soft-max with very high beta ===
    print("\n=== Phase 9: Final refinement ===")
    t9 = min(remaining() * 0.12, 150.0)
    if t9 > 20 and best_seq is not None:
        a_out, _ = softmax_optimize(best_seq, t9, beta_start=150.0, beta_end=500.0,
                                    lr=0.003, verbose=False, label="final")
        best_seq, best_score = try_update(best_seq, best_score, a_out, "final_sm", t_global)

    # === Phase 10: Final coordinate polish ===
    rt = remaining()
    if rt > 20 and best_seq is not None:
        t10 = min(rt * 0.7, 100.0)
        if t10 > 15:
            a_out, _ = coordinate_polish(best_seq, t10, verbose=False, label="coord_final")
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
