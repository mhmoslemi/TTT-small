def fast_convolve(a):
    """Compute convolve(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(z):
    """Project onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_vals = np.nonzero(cond)[0]
    if len(rho_vals) == 0:
        return np.ones(n) / n
    rho_val = rho_vals[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def subgradient_max_conv(a, top_m=1):

Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FISTA with exact subgradient of the max (no smoothing bias) + Nesterov
2. Peak-targeted local search (identifying critical pairs per peak)
3. Frequency-domain spectral flattening
4. L2-objective FISTA (minimize sum b_k^2 for flatness)
5. Multi-scale combined objective
6. Adaptive restarts with diverse initializations
Key difference from previous: exact subgradient (not softmax-smoothed),

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_vals = np.nonzero(cond)[0]
    if len(rho_vals) == 0:
        return np.ones(n) / n
    rho_val = rho_vals[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgrad_exponent(y, b, top_m, beta):
    """Compute smoothed subgradient using softmax over top_m peaks."""
    n = len(y)
    if top_m >= len(b):
        idx = np.arange(len(b))
        weights = np.ones(len(idx)) / len(idx)
    else:
        idx = np.argsort(b)[::-1][:top_m]
        b_vals = b[idx]
        shifts = (b_vals - b.max()) * beta
        shifts = np.clip(shifts, -500, 0)
        weights = np.exp(shifts)
        wsum = weights.sum()
        if wsum < 1e-300:
            weights = np.ones(len(idx)) / len(idx)
        else:
            weights = weights / wsum
    
    grad = np.zeros(n)
    for k, w in zip(idx, weights):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += w * 2.0 * y[k - js]
    return grad, b.max()

        theta_t = (t_nest - 1.0) / t_new

        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        b_max = b.max()

        a_prev = a
        a = a_new
        t_nest = t_new
        
        if stall > 4000:
            stall = 0
            t_nest = 1.0
            eta = 1.0 / (n ** 0.4)
        
        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_score
            print(f"    FISTA-L2 t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_score

def peak_targeted_search(a0, time_limit, verbose=True):
    """Target peaks by moving mass from critical positions to safe ones."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_convolve(a)
        b_max = b.max()
        # Get top peaks
        n_peaks = min(20, len(b))
        peak_idxs = np.argsort(b)[::-1][:n_peaks]
        
        # Compute "criticality" map: for each position j, how much it contributes to peaks
        crit = np.zeros(n)
        safe_mask = np.ones(n, dtype=bool)
        for k in peak_idxs[:10]:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                js = np.arange(j_start, j_end + 1)
                contrib = a[js] * a[k - js]
                crit[j_start:j_end + 1] += contrib
                # Mark as not-safe if it's a big contributor
                for j in js:
                    if a[j] * a[k - j] > b_max * 0.1 / max(1, j_end - j_start + 1):
                        safe_mask[int(j)] = False
        
        # Find good source and destination positions
        improved = False
        safe_positions = np.where(safe_mask & (a < a.mean() * 2))[0]
        high_crit = np.where(crit > crit.mean() + crit.std())[0]
        
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            if len(high_crit) == 0 or len(safe_positions) == 0:
                break
            
            src = int(high_crit[rng.integers(len(high_crit))])
            dst = int(safe_positions[rng.integers(len(safe_positions))])
            if src == dst or a[src] < 1e-12:
                continue
            
            for frac in [0.01, 0.03, 0.05, 0.1]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_convolve(a_trial).max()
                if m_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Perturb to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    m_new = fast_convolve(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakSearch ({elapsed:.1f}s): J={J:.6f} stall={stall}")
    
    return best_a, best_m

def spectral_flatten(a0, time_limit, n_iters=500, verbose=True):
    """Flatten the convolution spectrum via frequency-domain manipulation."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        b = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
        b_max = b.max()
        
        if b_max < 1e-30:
            break
        
        # Identify peaks and create a "mask" in frequency domain
        # We want to reduce the contribution to time-domain peaks
        b_norm = b / b_max
        # Create a soft mask: 1 where b is close to max, 0 elsewhere
        thresh = 0.5
        mask_time = (b_norm > thresh).astype(np.float64)
        
        # Transform mask to frequency domain
        Mask_f = np.fft.rfft(mask_time, fft_size)
        
        # Modify A to reduce peaks: A_new = A * (1 - alpha * Mask_f / (n+1))
        alpha = 0.1
        A_new = A * (1.0 - alpha * Mask_f / (2 * n))
        
        # Inverse FFT, enforce non-negativity
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        m_new = fast_convolve(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        
        # Blend for stability
        a = 0.7 * a + 0.3 * a_new
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        if verbose and it % 50 == 0:
            J = 2.0 * n * best_m
            print(f"    Spectral it={it}: J={J:.6f}")
    
    return best_a, best_m

def multi_exchange_search(a0, time_limit, verbose=True):
    """More aggressive multi-point exchange search."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    max_stall = 150
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_convolve(a)
        peak_idxs = np.argsort(b)[::-1][:15]
        improved = False
        
        for _ in range(80):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick a peak to target
            k = int(peak_idxs[rng.integers(len(peak_idxs))])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue
            
            # Find the most contributing pair
            js = np.arange(j_start, j_end + 1)
            contrib = a[js] * a[k - js]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue
            src = int(js[np.argmax(contrib)])
            
            # Find a safe destination (low contribution to all top peaks)
            dest_crit = np.zeros(n)
            for k2 in peak_idxs[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    js2 = np.arange(s2, e2 + 1)
                    dest_crit[s2:e2 + 1] += a[js2] * a[k2i - js2]
            
            # Find positions with low dest_crit
            candidates = np.where(dest_crit < dest_crit.mean())[0]
            candidates = candidates[np.isin(candidates, [src], invert=True)]
            if len(candidates) < 5:
                candidates = np.arange(n)
            
            dst = int(candidates[rng.integers(len(candidates))])
            
            # Try multiple transfer fractions
            best_trial = None
            best_trial_m = best_m
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt > a[src]:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_convolve(a_trial).max()
                if m_trial < best_trial_m:
                    best_trial_m = m_trial
                    best_trial = a_trial.copy()
            
            if best_trial is not None and best_trial_m < best_m - 1e-14:
                a = best_trial
                best_m = best_trial_m
                best_a = a.copy()
                improved = True
                b = fast_convolve(a)
                peak_idxs = np.argsort(b)[::-1][:15]
        
        if not improved:
            stall += 1
            # Random perturbation
            idx = rng.integers(n, size=10)
            for i in idx:
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.005, 0.03)
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0)
            a = a / a.sum()
            m_new = fast_convolve(a).max()
            if m_new < best_m:
                best_m = m_new
                best_a = a.copy()
                stall = 0
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MultiExch ({elapsed:.1f}s): J={J:.6f} stall={stall}")
    
    return best_a, best_m

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_sequence([float(x) for x in a_norm])
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
    return best_seq, best_score

def cyclic_shift_best(a0):
    """Find best cyclic shift."""
    n = len(a0)
    if n < 2:
        return a0.copy(), fast_convolve(a0).max()
    best_m = fast_convolve(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    for s in range(1, n):
        a_s = np.roll(a0, -s)
        A = np.fft.rfft(a_s, fft_size)
        b = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m
