def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(arr.size, dtype=np.float64) / arr.size
    return arr / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return np.maximum(c, 0.0)


def fft_conv2(a, b):

def spectral_flatten(a, iters=30, verbose=False):
    """Iteratively flatten the spectrum of a to reduce peak autocorrelation."""
    a = norm_sum1(a.astype(np.float64))

        if target_mag < 1e-15:
            break
        phases = A / np.where(mags > 1e-30, mags, 1.0)
        A_new = target_mag * phases
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = float(a_new.sum())
        if s < 1e-15:

        m = float(np.max(c))
        a = a_new

def lse_objective(a, beta, active_idx=None):

        return 0.0, np.zeros(a.size)
    m = float(np.max(cv))
    shifted = beta * (cv - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    e = np.exp(shifted)

    if se < 1e-300:
        return m, np.zeros(a.size)
    p = e / se
    # Gradient: p_k * 2 * conv(a, e_k) for each k

    n = a.size
    size = 2 * n - 1

def obj_v(v, beta, ref, active_idx=None):
    """Objective in log-coordinates v (with reference index ref)."""
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    u[ref] = 0.0
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref+1:] = v[ref:]
    # Softmax
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s < 1e-300:
        return 1e100, np.zeros(v.size)

        gv[ref:] = gu[ref+1:]

    return f, gv

def subgradient_descent(a0, iters=200, verbose=False):
    """Subgradient descent on the exact max of convolution."""
    a = norm_sum1(a0.astype(np.float64))
    n = a.size
    size = 2 * n - 1
    
    # Initialize step size
    c0 = fft_conv(a)
    M0 = float(np.max(c0))
    best_a = a.copy()
    best_M = M0
    avg_M = M0
    
    for t in range(1, iters + 1):
        c = fft_conv(a)
        M = float(np.max(c))
        
        # Exponential averaging of M
        avg_M = 0.99 * avg_M + 0.01 * M
        eta = 0.5 / (t + 10)
        # Scale eta by the ratio of actual to average
        if avg_M > 1e-15:
            eta *= M0 / avg_M
        
        # Subgradient: 2 * a shifted by peak position
        peak = int(np.argmax(c))
        g = np.zeros(n, dtype=np.float64)

        # Project onto simplex {a >= 0, sum(a) = 1}

        a = a_new
        c = fft_conv(a)
        M = float(np.max(c))
        if M < best_M - 1e-12:

            best_a = a.copy()

    return best_a


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""

Multi-strategy optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses spectral flattening, multi-temperature L-BFGS on log-sum-exp smoothed

import numpy as np
import time
import math

try:
    import scipy.optimize

except Exception:
    SCIPY_OK = False


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(arr.size, dtype=np.float64) / max(1, arr.size)
    return arr / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return np.maximum(c, 0.0)

def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    new_n = int(max(500, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)


def make_v(a, ref=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if ref is None:
        ref = int(np.argmax(a))
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.where(np.isfinite(v), v, -700.0)
    return v, ref


def v_to_a(v, ref):
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s < 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return e / s


def lse_obj_grad(a, beta, active_idx=None):
    c = fft_conv(a)
    n = a.size
    size = 2 * n - 1
    if active_idx is not None and len(active_idx) > 0:
        cv = c[active_idx]
    else:
        cv = c
    if cv.size == 0:
        return 0.0, np.zeros(n)
    m = float(np.max(cv))
    shifted = beta * (cv - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se < 1e-300:
        return m, np.zeros(n)
    p = e / se
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    pf_full = np.zeros(size, dtype=np.float64)
    if active_idx is not None and len(active_idx) > 0:
        pf_full[active_idx] = p
    else:
        pf_full = p
    Pr = np.fft.rfft(pf_full[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    f = m + math.log(se) / beta
    return f, grad


def obj_v(v, beta, ref, active_idx=None):
    n = v.size + 1
    a = v_to_a(v, ref)
    f, ga = lse_obj_grad(a, beta, active_idx)
    if not math.isfinite(f):
        return 1e100, np.zeros(v.size)
    mean_g = float(np.dot(a, ga))
    gu = a * (ga - mean_g)
    gv = np.empty(v.size, dtype=np.float64)
    ref = int(ref)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]
    if not np.all(np.isfinite(gv)):
        gv = np.zeros(v.size)
    return f, gv

        x = np.linspace(0.0, 1.0, n_default)
        parent = norm_sum1(np.sin(np.pi * x) ** 2)

    n_parent = parent.size
    print(f"Parent: n={n_parent}")

    # Try several lengths
    n_candidates = sorted(set([
        n_parent,
        n_parent - 500,
        n_parent + 500,
        n_parent - 1000,
        n_parent + 1000,
        int(n_parent * 0.8),
        int(n_parent * 1.2),
    ]))
    n_candidates = [n for n in n_candidates if 2000 <= n <= 12000]
    if not n_candidates:
        n_candidates = [5000]

    # Allocate time per length
    n_lens = len(n_candidates)
    per_len_budget = (total_budget * 0.75) / n_lens
    # Give extra time to the parent's own length
    time_alloc = []
    for n_c in n_candidates:
        if n_c == n_parent:
            time_alloc.append(per_len_budget * 2.0)
        else:
            time_alloc.append(per_len_budget)

    best_overall = None
    best_score_overall = float('inf')

    for idx, n_c in enumerate(n_candidates):
        time_left = total_budget - (time.monotonic() - t_start)
        if time_left < 60:
            break
        alloc = min(time_alloc[idx], time_left * 0.45)
        if alloc < 30:
            alloc = 30

        print(f"\n  === Length {n_c} (budget={alloc:.0f}s) ===")
        a_init = interpolate_sequence(parent, n_c)

        # Add small perturbation for non-parent lengths to explore different basins
        if n_c != n_parent and idx > 0:
            rng = np.random.default_rng(idx * 7919)
            pert = rng.normal(0, 0.01, n_c)
            a_init = a_init + pert * a_init
            a_init = norm_sum1(a_init)

        a_result, score = main_search(a_init, alloc, verbose=True)

        # Validate with actual evaluator
        actual_score = safe_eval(a_result)
        if math.isfinite(actual_score) and actual_score < best_score_overall:
            best_score_overall = actual_score
            best_overall = a_result.copy()
            print(f"  ** NEW BEST: score={actual_score:.6f} n={n_c}")

    # Use remaining time for intensive polish on the best found
    time_left = total_budget - (time.monotonic() - t_start)
    if best_overall is not None and time_left > 60:
        print(f"\n  === Final intensive polish (budget={time_left*.8:.0f}s) ===")
        n_best = best_overall.size
        # Multi-temperature L-BFGS with tighter tolerances
        if SCIPY_OK and time_left > 120:
            a_p = lbfegs_run(best_overall, [500.0, 1000.0, 2000.0, 4000.0],
                             maxfun_per_beta=400, verbose=True)
            sc = safe_eval(a_p)
            if math.isfinite(sc) and sc < best_score_overall:
                best_score_overall = sc
                best_overall = a_p.copy()
                print(f"  ** NEW BEST after final L-BFGS: {best_score_overall:.6f}")
            # Coordinate polish
            time_left2 = total_budget - (time.monotonic() - t_start)
            if time_left2 > 30:
                a_cp = coordinate_polish(best_overall, time_budget=min(60.0, time_left2 * 0.5),
                                         verbose=False)
                sc2 = safe_eval(a_cp)
                if math.isfinite(sc2) and sc2 < best_score_overall:
                    best_score_overall = sc2
                    best_overall = a_cp.copy()
                    print(f"  ** NEW BEST after final coord: {best_score_overall:.6f}")

    # Validate final answer
    if best_overall is None:
        best_overall = parent

    final_score = safe_eval(best_overall)
    if not math.isfinite(final_score):
        best_overall = parent
        final_score = safe_eval(best_overall)

    arr = np.asarray(best_overall, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        arr = norm_sum1(parent)
    else:
        arr = arr / s

    elapsed = time.monotonic() - t_start
    final_check = safe_eval(arr)
    print(f"\nFINAL: score={final_check:.6f} n={arr.size} elapsed={elapsed:.1f}s")
    return [float(x) for x in arr]
