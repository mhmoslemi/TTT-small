```python
def proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if v.sum() <= 1 and (v >= 0).all():
        return v.copy()
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

```python
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_vp(a, tau):
    n = len(a)
    ac = fast_autocorr(a)
    size = len(ac)
    ac_max = ac.max()
    shifted = tau * (ac - ac_max)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(size)
        w[np.argmax(ac)] = 1.0
    else:
        w = w / w_sum
    val = ac_max + np.log(w_sum + 1e-300) / tau
    return val, w

def grad_from_weights(a, w):
    n = len(a)
    size = 2 * n - 1
    a_pad = np.zeros(size, dtype=np.float64)
    a_pad[:n] = a
    w_flipped = w[::-1]
    req = 2 * size - 1
    ft = 1 << (req - 1).bit_length()
    Wf = np.fft.rfft(w_flipped, ft)
    Af = np.fft.rfft(a_pad, ft)
    conv = np.fft.irfft(Wf * Af, ft)
    idx = size - 1 - np.arange(n)
    return 2.0 * conv[idx]

def fista_anneal(a0, time_limit, tau_seq=(40.0, 150.0, 400.0, 800.0), verbose=True):
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_v = obj_val(a)
    t0 = time.monotonic()
    for tau in tau_seq:
        if time.monotonic() - t0 > time_limit:
            break
        sub_budget = min((time_limit - (time.monotonic() - t0)) * 0.7, time_limit)
        if sub_budget < 5:
            break
        a_out, v_out = _fista_single(a, sub_budget, tau, verbose)
        sc = obj_val(a_out)
        if sc < best_v:
            best_v = sc
            best_a = a_out.copy()
        a = a_out
    if verbose:
        print("    FISTA-anneal done ({:.1f}s) J={:.7f}".format(time.monotonic() - t0, best_v))
    return best_a, best_v

def _fista_single(a0, time_limit, tau, verbose=False):
    n = len(a0)
    a = normalize(a0).copy()
    y = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_v = obj_val(a)
    t0 = time.monotonic()
    L = tau * 6.0
    step = 1.0 / L
    backtr_count = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        val_y, w_y = build_vp(y, tau)
        grad_y = grad_from_weights(y, w_y)
        a_new = y + t_mom * (a - y) - step * grad_y
        a_new = proj_simplex(a_new)
        cur_v = obj_val(a_new)
        if cur_v < best_v - 1e-14:
            best_v = cur_v
            best_a = a_new.copy()
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = a.copy()
        a = a_new
        t_mom = t_new
        if it % 10000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print("      FISTA tau={:.0f} it={:6d} ({:.1f}s) J={:.7f}".format(tau, it, elapsed, best_v))
    return best_a, best_v

def lp_fw_dual(a0, time_limit, solver, init_K=150, max_K=4000, verbose=True):
    """LP-based Frank-Wolfe with dual-variable-guided active set."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    ac_y = fast_autocorr(y)
    active_list = list(np.argsort(ac_y)[::-1][:init_K])
    active_set = set(int(k) for k in active_list)
    no_improve = 0
    stall_limit = 3000
    t_iter = 0

    while time.monotonic() - t0 < time_limit and no_improve < stall_limit:
        t_iter += 1
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        if t_iter % 40 == 0 or t_iter == 1:
            top_lags = np.argsort(ac)[::-1]
            target_K = min(max_K, init_K + t_iter * 3)
            for k in top_lags[:target_K]:
                active_set.add(int(k))
            if len(active_set) > max_K:
                clist = sorted(active_set)
                margins = np.array([M_curr - ac[k] for k in clist])
                keep = np.argsort(margins)[:max_K]
                active_set = set(int(clist[i]) for i in keep)
        active_set.add(int(np.argmax(ac)))
        clist = sorted(active_set)
        K = len(clist)
        if K == 0:
            break
        try:
            coeffs = np.zeros((K, n), dtype=np.float64)
            for jj in range(K):
                ki = int(clist[jj])
                i_start = max(0, ki - n + 1)
                i_end = min(n, ki + 1)
                if i_start < i_end:
                    idx = np.arange(i_start, i_end)
                    coeffs[jj, idx] = y[ki - idx]
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            prob.solve(solver=solver, verbose=False)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                raise Exception("LP failed: " + str(prob.status))
            # Extract dual variables for active set management
            found_dual = False
            dual_vals = None
            try:
                for ii in range(len(constraints)):
                    dval = constraints[ii].dual_value
                    if dval is not None:
                        if isinstance(dval, np.ndarray) and dval.size == K:
                            dual_vals = dval
                            found_dual = True
                            break
            except Exception:
                pass
        except Exception:
            if len(active_set) > init_K // 2:
                clist2 = sorted(active_set)
                active_set = set(int(clist2[i]) for i in np.argsort([M_curr - ac[k] for k in clist2])[:init_K])
            continue
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        # Check violations using FFT
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        viol_mask = conv_ys > M_curr + 1e-10
        if np.any(viol_mask):
            viol_idx = np.where(viol_mask)[0]
            viol_amt = conv_ys[viol_idx] - M_curr
            top_v = viol_idx[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                active_set.add(int(k))
        # Dual-guided active set cleanup
        if found_dual and dual_vals is not None and len(dual_vals) == K:
            sort_idx = np.argsort(dual_vals)[::-1]
            keep_count = min(max_K, max(len(active_set) // 2, init_K))
            keep_lags = [clist[sort_idx[i]] for i in range(min(keep_count, K))]
            viol_lags = list(set(int(k) for k in np.where(viol_mask)[0]) - set(keep_lags))
            new_active = set(keep_lags + viol_lags[:max(0, max_K - len(keep_lags))])
            new_active.add(int(np.argmax(ac)))
            active_set = new_active
        # Line search
        best_alpha = 2.0 / (t_iter + 2.0)
        best_trial = obj_val(np.maximum((1 - best_alpha) * y + best_alpha * g_dir, 0.0))
        for alpha in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            tv = obj_val(trial)
            if tv < best_trial - 1e-14:
                best_trial = tv
                best_alpha = alpha
        lo_a = max(0.0, best_alpha - 0.12)
        hi_a = min(1.0, best_alpha + 0.12)
        for _ in range(5):
            m1 = lo_a + (hi_a - lo_a) / 3.0
            m2 = hi_a - (hi_a - lo_a) / 3.0
            v1 = obj_val(np.maximum((1 - m1) * y + m1 * g_dir, 0.0))
            v2 = obj_val(np.maximum((1 - m2) * y + m2 * g_dir, 0.0))
            if v1 < v2:
                hi_a = m2
            else:
                lo_a = m1
        best_alpha = (lo_a + hi_a) / 2.0
        if t_iter <= 2:
            y = g_dir.copy()
        else:
            y = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
        if verbose and t_iter % 300 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW-Dual t={:5d} ({:.1f}s) J={:.7f} |S|={}".format(t_iter, elapsed, 2 * n * best_m, len(active_set)))
    elapsed = time.monotonic() - t0
    if verbose:
        print("    LP-FW-Dual done ({:.1f}s) J={:.7f} |S|={}".format(elapsed, 2 * n * best_m, len(active_set)))
    return best_a, 2.0 * n * best_m
```
