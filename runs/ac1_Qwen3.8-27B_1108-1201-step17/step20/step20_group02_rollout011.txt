def fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def eval_quick(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    conv = fft_convolve(a)
    return float(2.0 * len(a) * conv.max())

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def lp_fw_away(a0, time_limit, solver, initial_K=600, max_K=4000, verbose=True):

    best_m = ac.max()

    active_set = sorted(np.argsort(ac)[::-1][:initial_K].tolist())

    max_no_improve = 500
    iter_count = 0
    stagnation_count = 0
    dual_gap_history = []

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if stagnation_count > 5:
                if verbose:
                    print("    LP-FW-AFW: persistent stagnation at iter " + str(iter_count))
                break
            ac_curr = fft_convolve(y)
            n_pert = max(80, len(active_set) // 4)
            rng = np.random.default_rng(stagnation_count * 7919)
            keep = list(rng.choice(len(active_set), size=len(active_set) - n_pert, replace=False))
            new_adds = np.argsort(ac_curr)[::-1]

            for k in new_adds[:n_pert * 3]:
                new_set.add(int(k))

            stagnation_count += 1
            continue

        iter_count += 1
        ac = fft_convolve(y)

        else:
            no_improve += 1

        if len(active_set) > max_K:

        dual_gap = abs(M_curr - M_lp)
        dual_gap_history.append(dual_gap)

        away_idx = int(np.argmax(y))

            trial = np.maximum(trial, 0.0)

            if s_t < 1e-15:

            else:

                if s_y < 1e-15:

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0

# --- Smoothed L-BFGS on softmax parameterization ---
def smooth_lbfgs(a0, time_limit, verbose=True):
    n = len(a0)
    a_init = norm_sum1(a0)
    best_a = a_init.copy()
    best_val = 2.0 * n * fft_convolve(best_a).max()
    t0 = time.monotonic()

    # theta parameterization: a = softmax(theta)
    # Initialize theta from a_init
    theta = np.log(np.maximum(a_init, 1e-30))
    theta -= theta.mean()  # centering for numerical stability

    eps_schedule = [200.0, 100.0, 50.0, 25.0, 15.0, 10.0, 7.0, 5.0]
    eps_idx = 0

    for eps in eps_schedule:
        if time.monotonic() - t0 > time_limit:
            break

        def objective_and_grad(theta_vec):
            a_tmp = np.exp(theta_vec - theta_vec.max())
            a_tmp = a_tmp / a_tmp.sum()
            conv = fft_convolve(a_tmp)
            m = conv.max()
            scaled = eps * (conv - m)
            p = np.exp(scaled)
            p_sum = p.sum()
            if p_sum < 1e-300:
                p = np.zeros(len(conv))
                p[int(np.argmax(conv))] = 1.0
            else:
                p = p / p_sum

            val = m + np.log(p_sum) / eps

            # Gradient: grad_a[j] = 2 * sum_k p[k] * a[k-j] (if 0<=k-j<n)
            # = 2 * (a_reversed * p_reversed)[n-1+j]
            b = a_tmp[::-1]
            p_r = p[::-1]
            full_conv = np.convolve(b, p_r)
            grad_a = 2.0 * full_conv[n - 1:n - 1 + n]

            # Convert to theta gradient
            grad_theta = grad_a - a_tmp * grad_a.sum()
            return val, grad_theta

        time_per_eps = (time_limit - (time.monotonic() - t0)) / max(1, len(eps_schedule) - eps_idx)

        result = sp_minimize(objective_and_grad, theta, method='L-BFGS-B',
                              options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10,
                                       'maxfun': 500, 'disp': False},
                              )

        if result.success or hasattr(result, 'fun'):
            a_new = np.exp(result.x - result.x.max())
            a_new = a_new / a_new.sum()
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                val_new = 2.0 * n * fft_convolve(a_new).max()
                if val_new < best_val:
                    best_val = val_new
                    best_a = a_new.copy()
                    if verbose:
                        elapsed = time.monotonic() - t0
                        print("    Smooth-LBFGS new best: J=" + repr(val_new) + " (eps=" + repr(eps) + ", t=" + str(round(elapsed, 1)) + "s)")

        # Check time
        elapsed = time.monotonic() - t0
        if verbose and eps > 50:
            print("    Smooth-LBFGS eps=" + repr(eps) + ": J=" + repr(2.0 * n * best_val) + " (" + str(round(elapsed, 1)) + "s)")

    return best_a, best_val

            if time.monotonic() - t0 > time_limit:
                break

            # Pick source: weighted by value
            weights = a + 1e-10
            src = int(rng.choice(n, p=weights / weights.sum()))

            # Pick destination: prefer positions that reduce peak contributions
            # Score each position by how much it contributes to top peaks
            scores = np.zeros(n)
            for k in top_k[:15]:
                ki = int(k)
                s2 = max(0, ki - n + 1)
                e2 = min(n, ki + 1)
                if s2 < e2:
                    idx2 = np.arange(s2, e2)
                    scores[idx2] += a[ki - idx2]

            # Prefer low-score destinations
            dst_weights = 1.0 / (scores + 1e-10)
            dst_weights = dst_weights / dst_weights.sum()
            dst = int(rng.choice(n, p=dst_weights))

            if dst == src or a[src] < 1e-12:
                continue

            frac = rng.uniform(0.001, 0.02)
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t

                best_m = M_trial
                best_a = a.copy()

                            if M_new < best_m:

        if verbose and iter_count % 50 == 0:

                    best_m = M_trial
                    best_a = a.copy()
                    found = True
                    break
            if not found and iter_count > 100:
                # Try random exploration
                for _ in range(10):
                    i = int(np.random.default_rng(iter_count * 101 + _).integers(n))
                    j = int(np.random.default_rng(iter_count * 101 + _ + 50).integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = 0.02
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fft_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                found = True
                                break
            if not found and iter_count > 200:
                if verbose:
                    print("    Coord refine: no improvement after " + str(iter_count) + " iters")
                break

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    Coord iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))

    return best_a, best_m

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = eval_quick(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = get_solver()
    best_seq, best_score = get_initial_sequence()
    n0 = len(best_seq)

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: LP-FW initial (aggressive) ===
    print("\n=== Phase 1: LP-FW-AFW initial (n=" + str(n0) + ") ===")
    t1 = min(remaining_time() * 0.20, 300.0)
    if t1 > 60 and n0 <= 12000:
        a_out, _ = lp_fw_away(norm_sum1(best_seq), t1, solver,
                              initial_K=800, max_K=5000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1_n" + str(n0), t_global)

    # === Phase 2: Smoothed L-BFGS ===
    print("\n=== Phase 2: Smoothed L-BFGS (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining_time() * 0.12, 180.0)
    if t2 > 40:
        a_out, _ = smooth_lbfgs(norm_sum1(best_seq), t2, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "smooth_n" + str(len(best_seq)), t_global)

    # === Phase 3: LP-FW second pass ===
    print("\n=== Phase 3: LP-FW-AFW second (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining_time() * 0.20, 300.0)
    if t3 > 60 and len(best_seq) <= 12000:
        a_out, _ = lp_fw_away(norm_sum1(best_seq), t3, solver,
                              initial_K=1000, max_K=6000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2_n" + str(len(best_seq)), t_global)

    # === Phase 4: Multi-n exploration ===
    print("\n=== Phase 4: Multi-n exploration ===")
    n_best = len(best_seq)
    offsets = [0, -500, 500, -1000, 1000, -2000, 2000, -3000, 3000]
    try_ns = sorted(set([n_best + off for off in offsets if 2000 <= n_best + off <= 14000]))
    # Sort by proximity to best
    try_ns.sort(key=lambda x: abs(x - n_best))

    for new_n in try_ns[:4]:
        rt = remaining_time()
        if rt < 150:
            break
        if new_n == n_best:
            continue
        print("\n  Multi-n: n=" + str(new_n) + " (remaining=" + str(round(rt, 1)) + "s)")
        a_long = interpolate_seq(best_seq, new_n)

        t5 = min(rt * 0.18, 180.0)
        if t5 > 40 and new_n <= 12000:
            a_out, _ = lp_fw_away(a_long, t5, solver,
                                  initial_K=600, max_K=4000, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

        rt = remaining_time()
        if rt > 60:
            t5b = min(rt * 0.10, 80.0)
            if t5b > 30:
                a_out, _ = smooth_lbfgs(norm_sum1(best_seq), t5b, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "smooth_n" + str(len(best_seq)), t_global)

    # === Phase 5: LP-FW third pass on best ===
    print("\n=== Phase 5: LP-FW-AFW third (n=" + str(len(best_seq)) + ") ===")
    t6 = min(remaining_time() * 0.15, 200.0)
    if t6 > 50 and len(best_seq) <= 12000:
        a_out, _ = lp_fw_away(norm_sum1(best_seq), t6, solver,
                              initial_K=1200, max_K=7000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw3_n" + str(len(best_seq)), t_global)

    # === Phase 6: Transfer search ===
    print("\n=== Phase 6: Transfer search (n=" + str(len(best_seq)) + ") ===")
    t7 = min(remaining_time() * 0.12, 150.0)
    if t7 > 30:
        a_out, _ = transfer_search(norm_sum1(best_seq), t7, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "transfer_n" + str(len(best_seq)), t_global)

    # === Phase 7: Coordinate refinement ===
    print("\n=== Phase 7: Coord refine (n=" + str(len(best_seq)) + ") ===")
    t8 = min(remaining_time() * 0.08, 100.0)
    if t8 > 20:
        a_out, _ = coord_refine(norm_sum1(best_seq), t8, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n" + str(len(best_seq)), t_global)

    # === Phase 8: Final LP-FW ===
    print("\n=== Phase 8: Final LP-FW-AFW (n=" + str(len(best_seq)) + ") ===")
    t9 = min(remaining_time() * 0.25, 250.0)
    if t9 > 40 and len(best_seq) <= 12000:
        a_out, _ = lp_fw_away(norm_sum1(best_seq), t9, solver,
                              initial_K=1500, max_K=8000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # === Final polish: transfer + smooth ===
    rt = remaining_time()
    if rt > 60:
        t10 = min(rt * 0.4, 80.0)
        a_out, _ = transfer_search(norm_sum1(best_seq), t10, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "transfer_final", t_global)

    rt = remaining_time()
    if rt > 40:
        t11 = min(rt * 0.4, 60.0)
        a_out, _ = smooth_lbfgs(norm_sum1(best_seq), t11, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "smooth_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = eval_quick(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 6000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = eval_quick(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in np.asarray(best_seq)]
