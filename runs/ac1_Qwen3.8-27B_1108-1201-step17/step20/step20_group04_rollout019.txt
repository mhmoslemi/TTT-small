import numpy as np

def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_adaptive_top_k(c):
    """Determine effective number of significant peaks."""
    c_max = c.max()
    if c_max < 1e-30:
        return 1
    # Count peaks above 95% of max
    n_sig = int(np.sum(c > 0.95 * c_max))
    # Also count peaks above 90% of max (weighted)
    n_90 = int(np.sum(c > 0.90 * c_max))
    # Use a blend
    top_k = max(2, min(int(0.7 * n_sig + 0.3 * n_90), len(c) // 4))
    return top_k

def nesterov_opt(a0, time_limit, beta_start=30.0, beta_end=20000.0,
                 beta_growth=1.006, eta_base=0.6, adaptive_topk=True,
                 top_k_ratio=0.02, restart_every=2000, verbose=True, seed=42):

    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Compute gradient with adaptive top-k
        c = fast_autocorr(y)
        if adaptive_topk:
            top_k = compute_adaptive_top_k(c)
        c_use = np.zeros(len(c))
        idx = np.argsort(c)[::-1][:top_k]
        c_use[idx] = c[idx]
        c_max = c.max()
        shifted = beta * (c_use - c_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        total = exp_vals.sum()
        if total < 1e-300:
            p = np.zeros_like(c_use)
            p[np.argmax(c)] = 1.0
        else:
            p = exp_vals / total

        # Compute gradient via FFT
        fft_size = 1 << (2 * n - 1).bit_length()
        P = np.fft.rfft(p, fft_size)
        a_rev = y[::-1]
        A_rev = np.fft.rfft(a_rev, fft_size)
        Conv = np.fft.irfft(P * A_rev, fft_size)
        grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]

        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            # Structured restart
            noise_scale = 0.03 / (1.0 + 0.05 * (t // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 2000 == 0:

            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + ", beta=" + str(round(beta, 1)) + ", topk=" + str(top_k))

    return best_a, best_max_b

                    reduce_i = src_i
                else:
                    reduce_i = partner

                if a[reduce_i] < 1e-14:
                    continue

                # Compute pressure at each position
                pressure = np.zeros(n)
                for k2 in top_idx[:12]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        pressure[idx2] += a[k2i - idx2]
                        pressure[k2i - idx2] += a[idx2]

                # Find best destination
                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_idx[:12]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2 + 1] = True

                candidates = np.where(~in_peak)[0]
                if len(candidates) < 100:
                    candidates = np.arange(n)

                mask = candidates != reduce_i
                candidates = candidates[mask]
                if len(candidates) == 0:
                    continue

                cand_pressure = pressure[candidates]
                n_try = min(8, len(candidates))
                try_dsts = candidates[np.argsort(cand_pressure)[:n_try]]

                for dst in try_dsts:
                    if found_move:
                        break
                    dst_int = int(dst)
                    if dst_int == reduce_i:
                        continue
                    # Try different fractions with quadratic-inspired selection
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08]:
                        amt = a[reduce_i] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[reduce_i] -= amt
                        a_trial[dst_int] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            found_move = True
                            break
                if found_move:
                    break

        if not improved:
            # Random exploration when stuck
            if no_improve > 300:
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.08)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                break

        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSurgeryV2 (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve))

    return best_a, best_m


def pairwise_cd_v2(a0, time_limit, verbose=True, seed=123):
    """Enhanced pairwise coordinate descent with more candidates."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 800:
        c = fast_autocorr(a)
        threshold = np.percentile(c, 93)
        peak_positions = np.where(c > threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.argsort(c)[::-1][:20]

        improved = False

        for k in peak_positions[:25]:
            if improved and (time.monotonic() - t0 > time_limit * 0.9):
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]

            top_pairs = np.argsort(products)[::-1][:5]
            for pi in top_pairs:
                if improved:
                    break
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n or partner == src_i:
                    continue
                if a[src_i] < 1e-14 and a[partner] < 1e-14:
                    continue

                larger_i = src_i if a[src_i] >= a[partner] else partner
                smaller_i = partner if a[src_i] >= a[partner] else src_i
                if a[larger_i] < 1e-14:
                    continue

                for frac in [0.001, 0.002, 0.005, 0.01, 0.02]:
                    amt = a[larger_i] * frac
                    if amt < 1e-18:
                        continue
                    # Try more random destinations
                    for _ in range(5):
                        dst = int(rng.integers(n))
                        if dst == larger_i:
                            continue
                        a_trial = a.copy()
                        a_trial[larger_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break

        if not improved:
            stall += 1
            if stall % 200 == 0:
                anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
                eps = 0.01 / (1.0 + stall / 200.0)
                a_try = a + eps * anti * a
                a_try = np.maximum(a_try, 0.0)
                s_t = a_try.sum()
                if s_t > 1e-15:
                    a_try = a_try / s_t
                    M_try = fast_autocorr(a_try).max()
                    if M_try < best_m:
                        a = a_try
                        best_m = M_try
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 400 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PairCDV2 (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def peak_tracking(a0, time_limit, verbose=True, seed=99):
    """Explicitly track the argmax and optimize to reduce it."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    argmax_k = int(fast_autocorr(a).argmax())
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 2000
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        current_max = c.max()
        argmax_k = int(c.argmax())
        
        if current_max < best_m - 1e-15:
            best_m = current_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Focus on the tracked peak
        k = argmax_k
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            stall += 1
            continue
        
        idx = np.arange(i_start, i_end + 1)
        products = a[idx] * a[k - idx]
        total_prod = products.sum()
        
        if total_prod < 1e-20:
            stall += 1
            continue
        
        # The peak value is total_prod. We want to reduce it.
        # Each pair (i, k-i) contributes products[j] to the peak.
        # Moving mass from position p to q changes the peak by:
        # Δc_k ≈ 2δ*(a[k-q] - a[k-p])
        # To reduce, we want a[k-q] < a[k-p]
        
        # Find the position contributing most to the peak
        # and try to move some of its mass to a position where
        # the partner value is lower
        
        top_contrib = np.argsort(products)[::-1][:3]
        improved = False
        
        for ci in top_contrib:
            if improved:
                break
            p = int(idx[ci])
            
            # Find where to move: we want a[k-dst] to be as small as possible
            # but also the destination shouldn't create new peaks
            # Approximate: pick dst where a[dst] is small and a[k-dst] is small
            dst_scores = a * a[abs(k - np.arange(n))[0:n]]  # a[i] * |a[k-i]|
            # Actually we want to minimize a[k-dst] (the partner at the new position)
            # but also not increase other peaks
            valid_dsts = np.arange(n)
            valid_dsts[valid_dsts == p] = -1
            # Score: we want low a[abs(k-dst)] (partner value)
            partner_vals = np.zeros(n)
            for d in range(n):
                if 0 <= k - d < n:
                    partner_vals[d] = a[k - d]
            
            # Blend: low partner value, low a[d] (so we don't create new peaks)
            score = partner_vals + 0.5 * a
            # Exclude p
            score[p] = np.inf
            
            # Try top 5 lowest score destinations
            top_dsts = np.argsort(score)[:5]
            
            for dst in top_dsts:
                if improved:
                    break
                dst = int(dst)
                if dst == p:
                    continue
                
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[p] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[p] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
        
        if not improved:
            # Try random moves
            for _ in range(10):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            break
        
        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakTrack (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall) + " k=" + str(k))

    return best_a, best_m


def basin_hopping(a0, time_limit, verbose=True, seed=200):
    """Basin hopping: make large perturbations then locally optimize."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Pool of candidates
    pool = [(best_m, best_a.copy())]
    
    sigma = 0.05  # Initial perturbation scale
    n_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        # Pick a candidate to perturb (best with 70% probability, random with 30%)
        if rng.random() < 0.7:
            m, a_base = pool[0]  # Best
        else:
            idx = int(rng.integers(len(pool)))
            m, a_base = pool[idx]
        
        # Generate perturbation
        pert_type = rng.random()
        if pert_type < 0.3:
            # Random Gaussian
            noise = rng.normal(0, sigma, n)
            a_new = a_base + noise * a_base
        elif pert_type < 0.5:
            # Circular shift
            shift = int(rng.integers(1, min(20, n // 10)))
            a_new = np.roll(a_base, shift)
            a_new = np.maximum(a_new, 0)
        elif pert_type < 0.7:
            # Mirror + noise
            a_new = a_base[::-1] + rng.normal(0, sigma * 0.5, n) * a_base
        elif pert_type < 0.85:
            # Twin peaks: create a mixture
            w = rng.uniform(0.3, 0.7)
            a_new = w * a_base + (1 - w) * a_base[::-1]
            a_new += rng.normal(0, sigma * 0.3, n) * a_new
        else:
            # Block swap
            block = int(rng.integers(10, max(20, n // 20)))
            start = int(rng.integers(0, max(1, n - block)))
            a_new = a_base.copy()
            a_new[start:start+block] = a_base[start:start+block] * 0.5
            # Add to random other location
            dst_start = int(rng.integers(0, max(1, n - block)))
            a_new[dst_start:dst_start+block] += a_base[start:start+block] * 0.5
        
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        
        # Quick evaluation
        m_new = fast_autocorr(a_new).max()
        
        # Do a few steps of local optimization (mini Nesterov)
        # Just do 100 steps of simple gradient descent
        beta = 100.0
        for _ in range(100):
            c = fast_autocorr(a_new)
            c_max = c.max()
            shifted = beta * (c - c_max)
            shifted = np.clip(shifted, -300, 300)
            exp_vals = np.exp(shifted)
            total = exp_vals.sum()
            if total < 1e-300:
                p = np.zeros_like(c)
                p[np.argmax(c)] = 1.0
            else:
                p = exp_vals / total
            
            fft_size = 1 << (2 * n - 1).bit_length()
            P = np.fft.rfft(p, fft_size)
            a_rev = a_new[::-1]
            A_rev = np.fft.rfft(a_rev, fft_size)
            Conv = np.fft.irfft(P * A_rev, fft_size)
            grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
            
            eta = min(0.5 / max(1.0, beta * 2.0), 0.05)
            z = a_new - eta * grad
            a_new = simplex_project(z)
        
        m_final = fast_autocorr(a_new).max()
        
        if m_final < best_m - 1e-14:
            best_m = m_final
            best_a = a_new.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    BasinHop (" + str(round(elapsed, 1)) + "s): NEW BEST J=" + repr(round(J, 6)))
        
        # Add to pool
        pool.append((m_final, a_new.copy()))
        # Keep only top 5
        pool.sort(key=lambda x: x[0])
        pool = pool[:5]
        
        # Adapt sigma
        n_iters += 1
        if n_iters % 20 == 0:
            # Decrease sigma slowly
            sigma = max(0.01, sigma * 0.95)
        
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    BasinHop (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " sigma=" + str(round(sigma, 4)) + " iter=" + str(n_iters))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score
