def spectral_pocs(a0, time_limit, verbose=True, seed=42):
    """POCS-style spectral optimization."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # We want |FFT_padded(a, fft_size)[k]|^2 to be flat for k > 0
    # But the constraint is a >= 0, sum(a) = 1
    
    gamma = 0.5  # Mixing parameter for the spectrum projection
    
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Time domain: ensure a is in the simplex
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        
        # Compute padded FFT
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # Target: flat power for k > 0
        # The DC component power[0] = (sum a)^2 = 1 (fixed)
        # Target for others: average of current non-DC power
        if fft_size // 2 > 1:
            avg_power = power[1:].mean()
            target_power = np.ones(len(A))
            target_power[0] = power[0]  # Keep DC
            target_power[1:] = avg_power
        
        # Project spectrum to target: adjust magnitudes
        A_new = A.copy()
        for k in range(1, len(A)):
            if power[k] > 0:
                scale = np.sqrt(target_power[k] / power[k])
                # Blend: keep some of the original
                scale = 1.0 - gamma + gamma * scale
                # Clip to avoid extreme changes
                scale = np.clip(scale, 0.3, 3.0)
                A_new[k] *= scale
        
        # Back to time domain
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        a_new = np.maximum(a_new, 0)

def fast_autocorr(a):

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def pocs_spectral(a0, time_limit, verbose=True, seed=42):
    """POCS-style spectral optimization: alternate between target spectrum and simplex."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 2).bit_length()

    gamma = 0.4
    for it in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        if len(A) > 1:
            non_dc_power = power[1:].sum()
            target_flat = non_dc_power / max(1, len(A) - 1)
            A_new = A.copy()
            for k in range(1, len(A_new)):
                if power[k] > 1e-30:
                    ratio = target_flat / power[k]
                    scale = 1.0 - gamma + gamma * np.sqrt(ratio)
                    scale = np.clip(scale, 0.2, 5.0)
                    A_new[k] *= scale
                else:
                    A_new[k] = np.sqrt(target_flat) * np.exp(1j * rng.uniform(0, 2 * np.pi))

            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
        else:
            continue

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new

                    a = a_p

        if it % 200 == 0:

def multi_peak_nesterov(a0, time_limit, beta_start=30.0, beta_end=1200.0,
                        beta_growth=1.004, eta_init=0.4, top_k_ratio=0.04,
                        restart_every=2000, verbose=True, seed=42):
    """Nesterov with multi-peak softmax subgradient and diverse restarts."""
    n = len(a0)
    x = normalize(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    eta = eta_init
    fft_size = 1 << (2 * n - 2).bit_length()

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Multi-peak softmax subgradient
        b = fast_autocorr(y)
        size = 2 * n - 1
        idx = np.argsort(b)[::-1][:top_k]
        b_top = b[idx]
        b_max = b_top[0]

        shifted = (b_top - b_max) / beta
        shifted = np.clip(shifted, -500, 0)
        exp_vals = np.exp(shifted)
        p_top = exp_vals / exp_vals.sum()

        # Compute gradient: weighted sum of constraint gradients
        grad = np.zeros(n)
        for i, k in enumerate(idx):
            k_int = int(k)
            if p_top[i] < 1e-10:
                continue
            js = max(0, k_int - n + 1)
            je = min(n, k_int + 1)
            idx_range = np.arange(js, je)
            # c[k] = sum_j y[j] * y[k-j], gradient w.r.t. y is y[k_j] + y[j] (symmetric)
            partners = k_int - idx_range
            valid = (partners >= 0) & (partners < n)
            idx_range = idx_range[valid]
            partners = partners[valid]
            grad[idx_range] += p_top[i] * y[partners]
            grad[partners] += p_top[i] * y[idx_range]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            x_new = x
            b_new = fast_autocorr(x_new)
            max_new = b_new.max()
        else:
            eta_trial = eta
            found = False
            y_max = b.max()
            for _ in range(8):
                z = y - eta_trial * grad
                x_new = simplex_project(z)
                b_new = fast_autocorr(x_new)
                max_new = b_new.max()
                if max_new <= y_max + 1e-14:
                    found = True
                    break
                eta_trial *= 0.5
            if not found:
                x_new = x
                max_new = best_max_b
                eta = eta_trial

        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.05, 2.0)
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 80 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = rng.integers(6)
            if strategy == 0:
                noise_level = 0.01 + 0.02 * rng.random()
                pert = rng.normal(0, noise_level, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                x_r = 0.7 * best_a + 0.3 * rng.dirichlet(np.ones(n) * 0.2)
            elif strategy == 2:
                x_r = best_a * np.exp(0.04 * rng.standard_normal(n))
            elif strategy == 3:
                mask = rng.random(n) > 0.5
                x_r = best_a.copy()
                x_r[mask] *= 0.6
                x_r[~mask] *= 1.4
            elif strategy == 4:
                shift = int(rng.integers(1, max(2, n // 15)))
                x_r = np.roll(best_a, shift)
            else:
                # Invert and blend
                x_inv = 1.0 - best_a / best_a.max()
                x_r = 0.5 * best_a + 0.5 * x_inv

            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 15.0)
            eta = eta_init
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("  Nest reStart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("  Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)) + " b=" + str(int(beta)))

    return best_a, best_max_b


def lp_fw_improved(a0, time_limit, solver, initial_K=400, max_K=3000,
                   add_per_iter=30, verbose=True, seed=99):
    """LP-Frank-Wolfe with dual-guided active set and adaptive restarts."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    y = a.copy()
    ac_y = fast_autocorr(y)
    # Initial active set: top K peaks
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 800
    stall_count = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            # Restart with new active set
            ac = fast_autocorr(y)
            active_set = set(int(k) for k in np.argsort(ac)[::-1][:min(initial_K, 2 * len(active_set))])
            no_improve = 0
            y = best_a.copy()
            stall_count += 1
            if stall_count > 10:
                break
            continue

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations and grow active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Also add near-peak constraints (within 2% of max) that aren't already active
        near_peaks = np.where(ac > 0.98 * M_curr)[0]
        for k in near_peaks:
            if int(k) not in active_set and len(active_set) < max_K:
                active_set.add(int(k))

        # Line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        alphas = [best_alpha, best_alpha * 0.5, best_alpha * 2.0,
                  1.0 / (t + 1), 0.03, 0.08, 0.15, 0.3]
        for alpha_trial in alphas:
            alpha_trial = max(0.01, min(1.0, alpha_trial))
            trial = (1.0 - alpha_trial) * y + alpha_trial * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha_trial

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and t % 150 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  LP-FW it=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def peak_peeling(a0, time_limit, verbose=True, seed=77):
    """Targeted peak elimination: focus on reducing the single worst peak."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 3000:
        ac = fast_autocorr(a)
        size = 2 * n - 1
        k_peak = int(np.argmax(ac))
        peak_val = ac[k_peak]

        # Find the most contributing pair to this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break

        # Try to reduce contribution from the largest pair
        src = int(idx[np.argmax(contrib)])
        partner = k_peak - src

        if not (0 <= partner < n) or src == partner:
            # Try random pair
            fracs = np.random.default_rng(rng.integers(10000)).dirichlet(np.ones(len(idx)))
            src = int(idx[np.argmax(fracs)])
            partner = k_peak - src
            if not (0 <= partner < n):
                stall += 1
                a[i] = np.maximum(a, 0)
                a = a / a.sum()
                continue

        # Move mass from src to a position that doesn't contribute to this peak
        for frac in [0.005, 0.01, 0.02, 0.05]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue

            found_improve = False
            # Try multiple random destinations
            for _ in range(8):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                # Check if dst would contribute to the peak
                if abs(dst - k_peak) <= n - 1 and (k_peak - dst) < n and (k_peak - dst) >= 0:
                    if (k_peak - dst) != src and (k_peak - dst) == dst:
                        continue

                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-17:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    stall = 0
                    found_improve = True
                    break

            if found_improve:
                break

        if not found_improve:
            stall += 1
            # Escalate: try broader random moves
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Multi-strategy optimization for minimizing 2*n*max(conv)/(sum)^2.
    Uses POCS spectral, multi-peak Nesterov, LP-FW with dynamic active set,
    peak peeling, and multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = _get_solver()
    print("Time budget: " + str(round(time_budget, 1)) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = normalize(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + str(round(p_score, 6)))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6000
        x = np.linspace(0, 1, n_base)
        # Try a few initializations
        candidates = [
            np.sin(np.pi * x) ** 2,
            np.sin(np.pi * x) ** 4,
            np.exp(-4 * (x - 0.5) ** 2),
            np.ones(n_base),
        ]
        for c in candidates:
            c_norm = normalize(c)
            sc = evaluate_safe(c_norm)
            if sc < best_score:
                best_score = sc
                best_seq = c_norm
        print("Default init: n=" + str(len(best_seq)) + ", score=" + str(round(best_score, 6)))

    n0 = len(best_seq)
    print("Start: n=" + str(n0) + ", J=" + str(round(best_score, 6)))

    # Phase 1: POCS spectral (short)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: POCS Spectral (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 15:
        a_out, _ = pocs_spectral(normalize(best_seq), t1, verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pocs1", t_global)

    # Phase 2: Multi-peak Nesterov (primary)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P2: Multi-peak Nesterov (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 50:
        a_out, _ = multi_peak_nesterov(normalize(best_seq), t2,
                                       beta_start=30.0, beta_end=1000.0,
                                       beta_growth=1.004, eta_init=0.4,
                                       top_k_ratio=0.04, restart_every=2000,
                                       verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 3: LP-FW (key method)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P3: LP-FW (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 30 and n_best <= 9000:
        a_out, _ = lp_fw_improved(normalize(best_seq), t3, solver,
                                  initial_K=400, max_K=3000,
                                  add_per_iter=35, verbose=True, seed=99)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 4: Peak peeling
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P4: Peak Peeling (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.07, 90.0)
    if t4 > 15:
        a_out, _ = peak_peeling(normalize(best_seq), t4, verbose=True, seed=77)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peel1", t_global)

    # Phase 5: Second Nesterov with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P5: Nesterov r2 (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 30:
        a_out, _ = multi_peak_nesterov(normalize(best_seq), t5,
                                       beta_start=60.0, beta_end=2000.0,
                                       beta_growth=1.005, eta_init=0.25,
                                       top_k_ratio=0.02, restart_every=1500,
                                       verbose=True, seed=999)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P6: Multi-length (base n=" + str(n_best) + ") ===")

    # Try a range of lengths
    length_candidates = []
    for delta in [-1000, -500, 0, 500, 1000, 2000, 3000]:
        ln = n_best + delta
        if 3000 <= ln <= 12000:
            length_candidates.append(ln)

    # Deduplicate and sort
    length_candidates = sorted(set(length_candidates))

    for new_n in length_candidates:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        print("  -> n=" + str(new_n))

        a_long = interpolate_sequence(best_seq, new_n)

        # Nesterov on new length
        t6a = min(remaining * 0.30, 150.0)
        if t6a > 20:
            a_out, _ = multi_peak_nesterov(a_long, t6a,
                                           beta_start=30.0, beta_end=800.0,
                                           beta_growth=1.004, eta_init=0.35,
                                           top_k_ratio=0.03, restart_every=1500,
                                           verbose=False, seed=new_n % 9973)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(new_n), t_global)

        # LP-FW on new length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 9000:
            t6b = min(remaining * 0.40, 150.0)
            if t6b > 20:
                a_long2 = interpolate_sequence(best_seq, new_n)
                a_out, _ = lp_fw_improved(normalize(a_long2), t6b, solver,
                                          initial_K=300, max_K=2000,
                                          add_per_iter=25, verbose=False, seed=new_n % 7919)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(new_n), t_global)

    # Phase 7: Final polish - LP-FW with large active set
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P7: Final LP polish (n=" + str(n_best) + ") ===")
    if remaining > 30 and n_best <= 9000:
        t7 = min(remaining * 0.35, 200.0)
        if t7 > 20:
            a_out, _ = lp_fw_improved(normalize(best_seq), t7, solver,
                                      initial_K=500, max_K=4000,
                                      add_per_iter=40, verbose=True, seed=1234)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Phase 8: Final Nesterov polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P8: Final Nesterov (n=" + str(n_best) + ") ===")
    if remaining > 20:
        t8 = min(remaining * 0.40, 150.0)
        if t8 > 15:
            a_out, _ = multi_peak_nesterov(normalize(best_seq), t8,
                                           beta_start=100.0, beta_end=3000.0,
                                           beta_growth=1.006, eta_init=0.15,
                                           top_k_ratio=0.01, restart_every=800,
                                           verbose=True, seed=5555)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

    # Phase 9: Peak peeling final
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P9: Final Peak Peel (n=" + str(n_best) + ") ===")
    if remaining > 15:
        t9 = min(remaining * 0.50, 100.0)
        if t9 > 10:
            a_out, _ = peak_peeling(normalize(best_seq), t9, verbose=True, seed=888)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peel_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: J=" + str(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
