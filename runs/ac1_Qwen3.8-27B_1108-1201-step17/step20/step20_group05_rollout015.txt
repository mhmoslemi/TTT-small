```
def clean_norm(x):
    arr=np.asarray(x, dtype=np.float64).ravel()
    if arr.size==0: return np.ones(1)
    arr=np.where(np.isfinite(arr), arr, 0.0)
    np.maximum(arr,0.0,out=arr)
    s=float(arr.sum())
    if s<1e-15: return np.ones(arr.size)/arr.size
    return arr/s
```

```python
"""
Hybrid search for non-negative sequences minimizing the evaluation score.

The evaluator is scale invariant, so all local optimization is done on
probability vectors.  The program:
  1. Starts from the supplied parent, deterministic sinusoidal patterns,
     sparse comb/Sidon-style patterns, and transformed parent variants.
  2. Infers a useful active convolution mask from a sample actual score.
  3. Optimizes with:
       - L-BFGS on a log-sum-exp smoothed active-set objective,
       - Frank-Wolfe directions from a multiplicative-weights game,
         with occasional exact small LP directions,
       - fast one-dimensional transfer moves with rank-1 convolution updates.
  4. Performs length probes and final polishing, always returning a clean
     finite non-negative list.
"""

import numpy as np
import time
import math

try:
    import scipy.optimize
    import scipy.sparse
    SCIPY_OK = True
except Exception:
    SCIPY_OK = False


def time_rem(deadline):
    return max(0.0, float(deadline) - time.monotonic())


def next_pow2(x):
    x = int(max(1, x))
    return 1 << (x - 1).bit_length()


def clean_norm(x):
    arr = np.asarray(x, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones_like(arr, dtype=np.float64) / float(arr.size)
    return arr / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def fft_conv2(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if b.size != n:
        b = np.resize(b, n)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def cross_convolve_p_a(p, a):
    """Return grad-like vector g_i = sum_k p_k a_{k-i}, valid coordinates."""
    p = np.asarray(p, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if p.size != 2 * n - 1:
        p = np.resize(p, 2 * n - 1)
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    out = conv[n - 1:2 * n - 1][::-1]
    out = np.where(np.isfinite(out), out, 0.0)
    return out


def proxy_max(c, mask):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0
    if mask is None:
        vals = c
    else:
        if np.size(mask) != c.size:
            vals = c
        else:
            vals = c[mask]
            if vals.size == 0:
                return 0.0
    vals = vals[np.isfinite(vals)]
    if vals.size == 0:
        return 0.0
    return float(np.max(vals))


def evaluate_proxy(a, mask):
    c = fft_conv(a)
    return float(2.0 * a.size * proxy_max(c, mask))


def active_top(c, mask, k):
    c = np.asarray(c, dtype=np.float64)
    size = c.size
    k = int(min(max(0, k), size))
    if k <= 0:
        return np.array([], dtype=int)
    if mask is None or np.size(mask) != size:
        vals = c
    else:
        vals = c.copy()
        vals[~np.asarray(mask, dtype=bool)] = -np.inf
    if k >= size:
        return np.arange(size, dtype=int)
    idx = np.argpartition(vals, -k)[-k:]
    return idx[np.argsort(vals[idx])[::-1]].astype(int)


def make_top_mask(a, top_count, base_mask):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    if top_count is None or int(top_count) <= 0:
        return base_mask
    top_count = int(min(int(top_count), size))
    c = fft_conv(a)
    top = active_top(c, base_mask, top_count)
    if top.size == 0:
        return base_mask
    m = np.zeros(size, dtype=bool)
    m[top] = True
    if base_mask is not None and np.size(base_mask) == size:
        m = m & np.asarray(base_mask, dtype=bool)
    if not np.any(m):
        return base_mask
    return m


def make_simple_mask(n, kind):
    n = int(n)
    if n <= 0:
        return None
    size = 2 * n - 1
    kind = str(kind)
    if kind == "full":
        return None
    m = np.ones(size, dtype=bool)
    c = n - 1
    if kind == "off":
        if c < size:
            m[c] = False
    elif kind == "nce":
        if c < size:
            m[c] = False
        m[0] = False
        m[-1] = False
    elif kind == "ne":
        m[0] = False
        m[-1] = False
    elif kind == "left_off":
        m = np.zeros(size, dtype=bool)
        if n - 1 > 0:
            m[:n - 1] = True
    elif kind == "right_off":
        m = np.zeros(size, dtype=bool)
        if n < size:
            m[n:] = True
    else:
        return None
    return m


def choose_mask_by_score(a, actual_score):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 0 or not math.isfinite(actual_score):
        return make_simple_mask(max(1, n), "full"), "full"
    c = fft_conv(a)
    kinds = ["full", "off", "nce", "ne", "left_off", "right_off"]
    best_kind = None
    best_diff = float("inf")
    for kind in kinds:
        m = make_simple_mask(n, kind)
        pred = 2.0 * n * proxy_max(c, m)
        diff = abs(pred - actual_score)
        if diff < best_diff:
            best_diff = diff
            best_kind = kind
    tol = max(0.06, 0.08 * max(1.0, abs(actual_score)))
    if best_kind is None or best_diff > tol:
        if actual_score < 1.95:
            best_kind = "off"
        elif actual_score < 2.20:
            best_kind = "full"
        else:
            best_kind = "full"
    return make_simple_mask(n, best_kind), best_kind


def safe_eval_arr(arr):
    """Call the real evaluator on a normalized clean vector."""
    try:
        arr = np.asarray(arr, dtype=np.float64)
        if arr.size == 0:
            return float("inf")
        arr = np.where(np.isfinite(arr), arr, 0.0)
        np.maximum(arr, 0.0, out=arr)
        s = float(arr.sum())
        if s < 1e-15:
            return float("inf")
        arr = arr / s
        if arr.size > 16000:
            return float("inf")
        lst = [float(x) for x in arr]
        try:
            val = evaluate_sequence(lst)
        except NameError:
            return evaluate_proxy(arr, None)
        if val is None:
            return float("inf")
        val = float(val)

        return val
    except Exception:

def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return clean_norm(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return clean_norm(y)


def softmax_from_u(u):
    u = np.asarray(u, dtype=np.float64)
    n = u.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(u)):
        return np.ones(n, dtype=np.float64) / float(n)
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def softmax_logits(loc):
    loc = np.asarray(loc, dtype=np.float64)
    if loc.size == 0:
        return np.array([], dtype=np.float64)
    loc = np.where(np.isfinite(loc), loc, -1000.0)
    loc = loc - float(np.max(loc))
    w = np.exp(loc)
    s = float(w.sum())
    if s <= 0.0 or not math.isfinite(s):
        w = np.ones_like(w, dtype=np.float64)
        s = float(w.size)
    return w / s


def make_v(a, ref=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 1:
        return np.array([], dtype=np.float64), 0
    if ref is None:
        pos = np.where(a > 0.0)[0]
        if pos.size:
            ref = int(pos[np.argmax(a[pos])])
        else:
            ref = 0
    ref = int(ref)
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.where(np.isfinite(v), v, -700.0)
    return v.astype(np.float64), ref


def from_v(v, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    ref = int(ref)
    if ref < 0:
        ref = 0
    if ref >= n:
        ref = n - 1
    u = np.zeros(n, dtype=np.float64)
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    return softmax_from_u(u)


def lse_a(a, beta, mask):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 0.0, np.array([], dtype=np.float64)
    size = 2 * n - 1
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    if mask is None or np.size(mask) != size:
        cv = c
    else:
        cv = c[np.asarray(mask, dtype=bool)]
    if cv.size == 0:
        return 0.0, np.zeros(n, dtype=np.float64)
    m = float(np.max(cv))
    shifted = beta * (cv - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0 or not math.isfinite(se):
        e = np.zeros(cv.size, dtype=np.float64)
        e[0] = 1.0
        se = 1.0
    p = e / se
    pf = np.zeros(size, dtype=np.float64)
    if mask is None or np.size(mask) != size:
        pf[:] = p
    else:
        pf[mask] = p
    Pr = np.fft.rfft(pf[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    f = m + math.log(se) / float(beta)
    if not math.isfinite(f):
        f = 1e100
    return f, grad


def objgrad_v(v, beta, mask, reg_u, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    if n <= 1:
        return 0.0, np.array([], dtype=np.float64)
    ref = int(ref)
    if ref < 0:
        ref = 0
    if ref >= n:
        ref = n - 1
    u = np.zeros(n, dtype=np.float64)
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    a = softmax_from_u(u)
    f, ga = lse_a(a, beta, mask)
    if not math.isfinite(f):
        return 1e100, np.zeros(v.size, dtype=np.float64)
    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)
    gv = np.empty(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]
    if reg_u > 0.0:
        f += 0.5 * reg_u * float(np.dot(v, v))
        gv = gv + reg_u * v
    gv = np.where(np.isfinite(gv), gv, 0.0)
    return f, gv


def estimate_lbfgs_cost(a, mask, beta=30.0):
    if not SCIPY_OK:
        return 0.25
    v, ref = make_v(a)
    try:
        t0 = time.monotonic()
        objgrad_v(v, beta, mask, 0.0, ref)
        dt = time.monotonic() - t0
        return max(0.02, float(dt))
    except Exception:
        return 0.25


def run_lbfgs_once(a0, beta, mask, maxfun, deadline):
    a = clean_norm(a0)
    n = a.size
    if n < 2 or maxfun <= 0 or not SCIPY_OK:
        return a
    v0, ref = make_v(a)
    options = {
        "maxfun": int(maxfun),
        "maxiter": int(maxfun) + 80,
        "ftol": 1e-12,
        "gtol": 1e-7,
        "disp": False,
    }
    try:
        res = scipy.optimize.minimize(
            objgrad_v,
            v0,
            args=(beta, mask, 1e-12, ref),
            method="L-BFGS-B",
            jac=True,
            options=options,
        )
    except Exception:
        return a
    if res is not None and hasattr(res, "x"):
        vx = np.asarray(res.x, dtype=np.float64)
        if vx.size == v0.size and np.all(np.isfinite(vx)):
            a = from_v(vx, ref)
            a = clean_norm(a)
    return a


def lbfgs_phase(a0, mask, budget, cost, deadline):
    a = clean_norm(a0)
    n = a.size
    if n < 2 or budget < 20:
        return a
    size = 2 * n - 1
    current = a.copy()
    best = a.copy()
    best_M = evaluate_proxy(a, mask)
    t1 = int(min(1400, max(80, size // 8)))
    t2 = int(min(900, max(50, size // 16)))
    t3 = int(min(600, max(35, size // 32)))
    t4 = int(max(16, size // 80))
    schedules = [
        (20.0, None),
        (60.0, None),
        (150.0, t1),
        (300.0, t2),
        (700.0, t3),
        (1500.0, t4),
    ]
    total_eval_budget = max(60, int(budget / max(3.0 * cost, 0.05)))
    per_run = int(max(50, min(650, total_eval_budget // max(1, len(schedules)))))
    t0 = time.monotonic()
    for beta, top_count in schedules:
        if time_rem(deadline) < 20:
            break
        if time.monotonic() - t0 > budget - 8:
            break
        msk = make_top_mask(current, top_count, mask)
        a_new = run_lbfgs_once(current, beta, msk, per_run, deadline)
        Mn = evaluate_proxy(a_new, mask)
        if math.isfinite(Mn) and Mn <= best_M * (1.0 + 1e-4) + 1e-18:
            current = a_new
        if math.isfinite(Mn) and Mn < best_M - 1e-15:
            best_M = Mn
            best = a_new.copy()
    return best


def perturb_sequence(a, seed, scale=0.02):
    a = clean_norm(a)
    n = a.size
    if n <= 1:
        return a
    rng = np.random.default_rng(seed)
    b = a + rng.normal(0.0, scale, n) * a
    b = np.where(np.isfinite(b) & (b > 0.0), b, 0.0)
    s = float(b.sum())
    if s < 1e-15:
        return a
    return b / s


def game_direction(f, active, c, mw_iters=18, seed=0):
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    active = np.asarray(active, dtype=int)
    K = active.size
    if K <= 0:
        return np.ones(n, dtype=np.float64) / float(n)
    size = 2 * n - 1
    rng = np.random.default_rng(seed)
    if c is not None and np.size(c) == size:
        av = c[active]
        av = np.where(np.isfinite(av) & (av > 1e-300), av, 1e-300)
        logits = np.log(av)
        logits -= float(np.max(logits))
        logits *= 0.6
    else:
        logits = np.zeros(K, dtype=np.float64)
    logits += rng.normal(0.0, 0.08, K)
    lambda_w = softmax_logits(logits)
    uniform_g = np.ones(n, dtype=np.float64) / float(n)
    g = uniform_g.copy()
    for _ in range(max(1, int(mw_iters))):
        pf = np.zeros(size, dtype=np.float64)
        pf[active] = lambda_w
        pressure = cross_convolve_p_a(pf, f, 1.0)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)
        pmx = float(np.max(pressure))
        pmn = float(np.min(pressure))
        ptp = pmx - pmn
        if ptp <= 1e-30:
            g = uniform_g.copy()
        else:
            scaled = (pressure - pmn) / ptp
            w = np.exp(-7.0 * scaled)
            w = np.where(np.isfinite(w), w, 0.0)
            w += 1e-12
            g = w / float(w.sum())
            g = 0.96 * g + 0.04 * uniform_g
            g = g / float(g.sum())
        h = fft_conv2(f, g)
        ha = h[active]
        ha = np.where(np.isfinite(ha) & (ha > 0.0), ha, 0.0)
        hmac = float(np.max(ha))
        hmin = float(np.min(ha))
        hspan = hmac - hmin
        if hspan > 1e-30:
            logits += (10.0 / hspan) * (ha - float(np.mean(ha)))
        lambda_w = softmax_logits(logits)
    return g


def lp_direction_minmax(f, active, time_limit=5.0):
    if not SCIPY_OK:
        return None
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    active = np.asarray(active, dtype=int)
    K = active.size
    if n < 2 or K == 0:
        return None
    try:
        row_parts = []
        col_parts = []
        val_parts = []
        rows0 = 0
        for r, k in enumerate(active):
            k = int(k)
            s = max(0, k - n + 1)
            e = min(n, k + 1)
            if s >= e:
                continue
            cols = np.arange(s, e, dtype=np.int32)
            row_parts.append(np.full(cols.size, r, dtype=np.int32))
            col_parts.append(cols)
            val_parts.append((f[k - cols]).astype(np.float64))
        if not row_parts:
            return None
        rows = np.concatenate(row_parts)
        cols = np.concatenate(col_parts)
        vals = np.concatenate(val_parts)
        A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))
        neg_t = scipy.sparse.csc_matrix(-np.ones((K, 1), dtype=np.float64))
        A_ub = scipy.sparse.hstack((A, neg_t), format="csr")
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0], dtype=np.float64)
        c_obj = np.zeros(n + 1, dtype=np.float64)
        c_obj[n] = 1.0
        bounds = [(0.0, None)] * n + [(None, None)]
        res = scipy.optimize.linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=np.zeros(K, dtype=np.float64),
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="HiGHS",
            options={"disp": False, "time_limit": float(time_limit)},
        )
        if res is not None and getattr(res, "success", False):
            g = np.asarray(res.x[:n], dtype=np.float64)
            g = np.where(np.isfinite(g) & (g > 0.0), g, 0.0)
            s = float(g.sum())
            if s > 1e-15:
                return g / s
    except Exception:
        return None
    return None


def line_search_mix(a, g, cf, cg, fg, mask, current_M):
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    alphas = [0.03, 0.06, 0.10, 0.16, 0.25, 0.36, 0.50, 0.68, 0.84, 1.00]
    best_alpha = 0.0
    best_m = current_M
    for alpha in alphas:
        one = 1.0 - alpha
        val = (one * one) * cf + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
        if mask is None:
            vals = val
        else:
            vals = val[np.asarray(mask, dtype=bool)]
        if vals.size == 0:
            m = 0.0
        else:
            vals = vals[np.isfinite(vals)]
            m = float(np.max(vals)) if vals.size else 0.0
        if math.isfinite(m) and m < best_m - 1e-18:
            best_m = m
            best_alpha = alpha
    if best_alpha <= 0.0:
        return a.copy(), best_m, 0.0
    new_a = (1.0 - best_alpha) * a + best_alpha * g
    new_a = np.where(np.isfinite(new_a) & (new_a > 0.0), new_a, 0.0)
    s = float(new_a.sum())
    if s < 1e-15:
        return a.copy(), best_m, 0.0
    return new_a / s, best_m, best_alpha


def fw_search(a0, mask, budget, active_max=1500, game_iters=16, deadline):
    a = clean_norm(a0)
    n = a.size
    if n < 2 or budget < 20:
        return a
    c = fft_conv(a)
    M = proxy_max(c, mask)
    best = a.copy()
    best_M = M
    active = active_top(c, mask, active_max)
    t0 = time.monotonic()
    t = 0
    stall = 0
    while time_rem(deadline) > 25 and time.monotonic() - t0 < budget:
        t += 1
        g = None
        if active.size >= 1 and t % 12 == 0:
            small = active[: min(120, active.size)]
            g = lp_direction_minmax(a, small, 4.0)
        if g is None:
            g = game_direction(a, active, c, mw_iters=game_iters, seed=5000 + t)
        g = clean_norm(g)
        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        new_a, new_M, alpha = line_search_mix(a, g, c, cg, fg, mask, M)
        if new_M < M - 1e-14 and alpha > 0.0:
            one = 1.0 - alpha
            c = (one * one) * c + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
            c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
            a = new_a
            M = new_M
            stall = 0
            if M < best_M - 1e-15:
                best_M = M
                best = a.copy()
            if t % 8 == 0:
                active = active_top(c, mask, active_max)
        else:
            stall += 1
            if stall % 8 == 0:
                a = perturb_sequence(best, 9100 + t, 0.018)
                c = fft_conv(a)
                M = proxy_max(c, mask)
                active = active_top(c, mask, active_max)
                stall = 0
        if stall > 80:
            break
    return best


def apply_single_move(c, a, s, d, t):
    c = np.asarray(c, dtype=np.float64).copy()
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = c.size
    c[d:d + n] += 2.0 * t * a
    c[s:s + n] -= 2.0 * t * a
    t2 = float(t) * float(t)
    if 2 * d < size:
        c[2 * d] += t2
    if 2 * s < size:
        c[2 * s] += t2
    if s + d < size:
        c[s + d] -= 2.0 * t2
    c = np.where(np.isfinite(c), c, 0.0)
    return c


def pair_polish(a0, mask, budget, active_max=3500, deadline):
    a = clean_norm(a0)
    n = a.size
    if n < 2 or budget < 20:
        return a
    c = fft_conv(a)
    M = proxy_max(c, mask)
    best = a.copy()
    best_M = M
    size = 2 * n - 1
    K = int(min(active_max, size))
    active = active_top(c, mask, K)
    if active.size == 0:
        return best
    active_sorted = np.sort(active)
    base_vals = c[active]
    t0 = time.monotonic()
    rnd = 0
    stall = 0
    rng = np.random.default_rng(24680)
    fracs = (0.002, 0.005, 0.01, 0.02, 0.05, 0.10, 0.20, 0.35)
    while time_rem(deadline) > 25 and time.monotonic() - t0 < budget and stall < 70:
        rnd += 1
        vk = c[active]
        vmax = float(np.max(vk)) if vk.size else 0.0
        if vmax <= 1e-300:
            w = np.ones(vk.size, dtype=np.float64)
            w /= float(max(1, vk.size))
        else:
            tau = max(1e-300, 0.04 * vmax)
            e = np.exp((vk - vmax) / tau)
            e = np.where(np.isfinite(e), e, 0.0)
            se = float(e.sum())
            if se <= 0.0:
                w = np.ones(vk.size, dtype=np.float64)
                w /= float(max(1, vk.size))
            else:
                w = e / se
        p = np.zeros(size, dtype=np.float64)
        p[active] = w
        grad = 2.0 * cross_convolve_p_a(p, a, 1.0)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        src_scores = grad * a
        dst_scores = grad + 1e-8 * a

        n_src = int(min(10, n))
        n_dst = int(min(14, n))
        if n_src <= 0 or n_dst <= 0:
            break
        if n > 1:
            src_idx = np.argpartition(src_scores, n_src - 1)[:n_src]
            dst_idx = np.argpartition(dst_scores, n_dst - 1)[:n_dst]
        else:
            src_idx = np.array([0], dtype=int)
            dst_idx = np.array([0], dtype=int)

        cands = []
        for ss in src_idx:
            s0 = int(ss)
            for dd in dst_idx:
                d0 = int(dd)
                if s0 != d0:
                    cands.append((s0, d0))
                if len(cands) >= 80:
                    break
            if len(cands) >= 80:
                break
        nrand = int(min(40, max(5, n)))
        while nrand > 0 and n > 1 and len(cands) < 130:
            s0 = int(rng.integers(n))
            d0 = int(rng.integers(n))
            if s0 != d0:
                cands.append((s0, d0))
            nrand -= 1

        best_move = None
        best_move_m = M
        base_vals = c[active]
        for s, d in cands:
            for source, dest in ((s, d), (d, s)):
                if source == dest:
                    continue
                src_val = float(a[source])
                if src_val < 1e-15:
                    continue
                Klen = active.size
                u1 = np.zeros(Klen, dtype=np.float64)
                idx1 = active - dest
                valid1 = (idx1 >= 0) & (idx1 < n)
                if np.any(valid1):
                    u1[valid1] += a[idx1[valid1]]
                u2 = np.zeros(Klen, dtype=np.float64)
                idx2 = active - source
                valid2 = (idx2 >= 0) & (idx2 < n)
                if np.any(valid2):
                    u2[valid2] -= a[idx2[valid2]]
                u = u1 + u2
                wq = np.zeros(Klen, dtype=np.float64)
                for pt, cv in ((2 * dest, 1.0), (2 * source, 1.0), (source + dest, -2.0)):
                    if 0 <= pt < size:
                        pos = int(np.searchsorted(active_sorted, pt))
                        if pos < Klen and active_sorted[pos] == pt:
                            wq[pos] += cv
                for f in fracs:
                    tmove = src_val * float(f)
                    if tmove <= 1e-18:
                        continue
                    vals = base_vals + 2.0 * tmove * u + (tmove * tmove) * wq
                    vals = vals[np.isfinite(vals)]
                    if vals.size == 0:
                        continue
                    mn = float(np.max(vals))
                    if math.isfinite(mn) and mn < best_move_m - 1e-15:
                        best_move_m = mn
                        best_move = (source, dest, tmove)
        if best_move is not None:
            source, dest, tmove = best_move
            c = apply_single_move(c, a, source, dest, tmove)
            a[source] = max(0.0, a[source] - tmove)
            a[dest] = a[dest] + tmove
            a = np.where(np.isfinite(a) & (a > 0.0), a, 0.0)
            M = best_move_m
            if M < best_M - 1e-15:
                best_M = M
                best = a.copy()
            if rnd % 4 == 0:
                active = active_top(c, mask, K)
                if active.size == 0:
                    break
                active_sorted = np.sort(active)
            stall = 0
        else:
            stall += 1
            if stall % 7 == 0:
                a = perturb_sequence(best, 4242 + rnd, 0.018)
                c = fft_conv(a)
                M = proxy_max(c, mask)
                active = active_top(c, mask, K)
                if active.size == 0:
                    break
                active_sorted = np.sort(active)
                stall = 0
    return best


def make_sym_sequence(a):
    a = clean_norm(a)
    b = 0.5 * (a + a[::-1])
    return clean_norm(b)


def make_sine_candidates(n, seed=11):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    out = []
    patterns = [
        np.sin(np.pi * x) ** 2,
        np.sin(2.0 * np.pi * x) ** 2 + 1e-6,
        (1.0 + np.cos(2.0 * np.pi * x)) ** 2,
        np.exp(-((x - 0.5) ** 2) / 0.035),
        0.2 + 0.8 * ((np.sin(4.0 * np.pi * x) + 1.0) / 2.0),
        np.ones(n, dtype=np.float64),
    ]
    for p in patterns:
        out.append(clean_norm(p))
    rng = np.random.default_rng(seed)
    for _ in range(3):
        out.append(clean_norm(rng.random(n)))
    return out


def greedy_sidon_sequence(n, k, seed):
    n = int(max(2, n))
    k = int(min(max(2, k), n))
    rng = np.random.default_rng(seed)
    S = []
    Sset = set()
    counts = {}
    first = int(rng.integers(1, n - 1))
    S.append(first)
    Sset.add(first)
    for _ in range(k - 1):
        best_pos = -1
        best_score = 10 ** 9
        base = list(S)
        for _trial in range(70):
            cand = int(rng.integers(1, n - 1))
            if cand in Sset:
                continue
            maxc = 0
            for s0 in base:
                d = abs(cand - s0)
                cc = counts.get(d, 0) + 1
                if cc > maxc:
                    maxc = cc
            if maxc < best_score:
                best_score = maxc
                best_pos = cand
                if best_score <= 1:
                    break
        if best_pos < 0:
            continue
        S.append(best_pos)
        Sset.add(best_pos)
        base = list(S)
        for s0 in base[:-1]:
            d = abs(best_pos - s0)
            counts[d] = counts.get(d, 0) + 1
    if not S:
        return None
    a = np.zeros(n, dtype=np.float64)
    a[S] = 1.0
    return clean_norm(a)


def make_sparse_candidates(n, seed=7):
    n = int(max(2, n))
    out = []
    k_target = int(max(2, 0.95 * math.sqrt(2.0 * n)))
    k_target = min(n, k_target)
    # single impulse
    a = np.zeros(n, dtype=np.float64)
    a[n // 2] = 1.0
    out.append(clean_norm(a))
    # random supports
    for k in (max(2, k_target // 2), max(2, int(0.85 * k_target)), k_target):
        rng = np.random.default_rng(seed + k)
        idx = rng.choice(n, size=min(k, n), replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[idx] = 1.0
        out.append(clean_norm(a))
    # greedy Sidon-like
    for sd in (seed + 101, seed + 202):
        a = greedy_sidon_sequence(n, k_target, sd)
        if a is not None:
            out.append(a)
    # periodic combs
    for period in (3, 5, 7, 11, 17, 29):
        if period < 2 or period > n:
            continue
        a = np.zeros(n, dtype=np.float64)
        a[::period] = 1.0
        out.append(clean_norm(a))
    # zero endpoints for best sparse-ish candidates
    for a in out[:8]:
        b = a.copy()
        b[0] = 0.0
        b[-1] = 0.0
        out.append(clean_norm(b))
    return out


def make_parent_variants(a):
    a = clean_norm(a)
    out = [make_sym_sequence(a)]
    for seed in range(4):
        out.append(clean_norm(perturb_sequence(a, seed + 17, 0.015)))
    return out


def consider_candidate(best_seq, best_score, seq, label, deadline, verbose=True):
    if seq is None:
        return best_seq, best_score
    arr = clean_norm(seq)
    if arr.size == 0:
        return best_seq, best_score
    if arr.size > 12000:
        arr = interpolate_sequence(arr, 8000)
    sc = safe_eval_arr(arr)
    if math.isfinite(sc):
        if (best_score is None) or (not math.isfinite(best_score)) or (sc < float(best_score) - 1e-12):
            if verbose:
                print("  NEW BEST [%s]: " % label + repr(sc) + " n=" + str(arr.size))
            return arr.copy(), float(sc)
    elif best_seq is None:
        # If the evaluator is unavailable, safe_eval_arr falls back to a proxy,
        # but keep a structural fallback if everything is invalid.
        best_seq = arr.copy()
        best_score = 2.0 * arr.size * proxy_max(fft_conv(arr), None)
    return best_seq, best_score


def optimize_mask(a0, mask, budget, deadline, verbose=False):
    a = clean_norm(a0)
    n = a.size
    if n < 2 or budget < 25:
        return a
    current = a.copy()
    current_score = safe_eval_arr(current)
    best_local = current.copy()
    best_local_score = current_score

    cost = estimate_lbfgs_cost(current, mask)

    if SCIPY_OK and time_rem(deadline) > 260:
        b1 = min(budget * 0.38, time_rem(deadline) - 160)
        a1 = lbfgs_phase(current, mask, b1, cost, deadline)
        sc1 = safe_eval_arr(a1)
        if math.isfinite(sc1) and ((not math.isfinite(best_local_score)) or sc1 < best_local_score - 1e-12):
            best_local_score = sc1
            best_local = a1.copy()
            current = a1.copy()
            if verbose:
                print("    mask-LBFGS " + repr(best_local_score))

    if time_rem(deadline) > 200:
        b2 = min(budget * 0.35, time_rem(deadline) - 130)
        amax = int(min(2400, max(500, 2 * n // 7)))
        a2 = fw_search(current, mask, b2, active_max=amax, game_iters=18, deadline=deadline)
        sc2 = safe_eval_arr(a2)
        if math.isfinite(sc2) and ((not math.isfinite(best_local_score)) or sc2 < best_local_score - 1e-12):
            best_local_score = sc2
            best_local = a2.copy()
            current = a2.copy()
            if verbose:
                print("    mask-FW " + repr(best_local_score))

    if time_rem(deadline) > 100:
        b3 = min(budget * 0.28, time_rem(deadline) - 60)
        a3 = pair_polish(current, mask, b3, active_max=min(4096, 2 * n), deadline=deadline)
        sc3 = safe_eval_arr(a3)
        if math.isfinite(sc3) and ((not math.isfinite(best_local_score)) or sc3 < best_local_score - 1e-12):
            best_local_score = sc3
            best_local = a3.copy()
            current = a3.copy()
            if verbose:
                print("    mask-PAIR " + repr(best_local_score))

    if time_rem(deadline) > 60:
        a4 = pair_polish(current, mask, min(80.0, time_rem(deadline) - 30),
                         active_max=min(2500, 2 * n), deadline=deadline)
        sc4 = safe_eval_arr(a4)
        if math.isfinite(sc4) and ((not math.isfinite(best_local_score)) or sc4 < best_local_score - 1e-12):
            best_local_score = sc4
            best_local = a4.copy()

    return best_local


def propose_candidate():
    t_start = time.monotonic()
    deadline = t_start + max(60.0, 1800.0 - 45.0)
    print("propose_candidate start, budget=" + repr(time_rem(deadline)))

    best_seq = None
    best_score = float("inf")

    # Use supplied parent if present.
    try:
        parent_list = height_sequence_1
    except Exception:
        parent_list = None

    parent_arr = None
    if parent_list is not None:
        try:
            p = np.asarray(parent_list, dtype=np.float64).ravel()
            p = np.where(np.isfinite(p), p, 0.0)
            np.maximum(p, 0.0, out=p)
            if p.size > 0 and float(p.sum()) > 1e-15:
                parent_arr = clean_norm(p)
        except Exception:
            parent_arr = None

    if parent_arr is not None:
        if parent_arr.size <= 12000:
            best_seq, best_score = consider_candidate(best_seq, best_score, parent_arr, "parent", deadline)
        else:
            best_seq, best_score = consider_candidate(
                best_seq, best_score, interpolate_sequence(parent_arr, 6700), "parent_interp", deadline)

    # Choose a working length.
    if best_seq is None:
        work_len = 5000
    else:
        work_len = int(best_seq.size)
    if work_len < 1200 or work_len > 9000:
        work_len = 5500

    # Candidate families.
    candidates = []
    if parent_arr is not None:
        pwork = interpolate_sequence(parent_arr, work_len) if parent_arr.size != work_len else parent_arr
        candidates.extend(make_parent_variants(pwork))
        candidates.append(pwork)

    candidates.extend(make_sine_candidates(work_len, seed=11))
    candidates.extend(make_sparse_candidates(work_len, seed=555))

    for idx, cand in enumerate(candidates):
        if time_rem(deadline) < 20:
            break
        best_seq, best_score = consider_candidate(
            best_seq, best_score, cand, "cand_%d" % idx, deadline, verbose=(idx % 8 == 0))

    if best_seq is None:
        best_seq = clean_norm(np.ones(5000, dtype=np.float64))
        best_score = safe_eval_arr(best_seq)
        if not math.isfinite(best_score):
            best_score = 2.0

    # If already at a known constructive bound, stop immediately.
    if math.isfinite(best_score) and best_score <= 1.0002:
        print("  early return, score=" + repr(best_score))
        arr = clean_norm(best_seq)
        return [float(x) for x in arr]

    # Infer active mask from actual score.
    mask, mask_kind = choose_mask_by_score(best_seq, best_score)
    print("  inferred mask kind=" + str(mask_kind) + " score=" + repr(best_score))

    # Build a small list of masks to optimize.
    mask_kinds = []
    if mask_kind is not None and mask_kind not in mask_kinds:
        mask_kinds.append(mask_kind)
    if best_score < 1.95:
        for kk in ("off", "nce"):
            if kk not in mask_kinds:
                mask_kinds.append(kk)
    elif best_score < 2.15:
        for kk in ("full", "off"):
            if kk not in mask_kinds:
                mask_kinds.append(kk)
    else:
        for kk in ("full", "nce"):
            if kk not in mask_kinds:
                mask_kinds.append(kk)
    mask_kinds = mask_kinds[:3]

    for mki, kind in enumerate(mask_kinds):
        if time_rem(deadline) < 120:
            break
        m = make_simple_mask(best_seq.size, kind)
        phase_budget = min(620.0, max(120.0, time_rem(deadline) / float(max(1, len(mask_kinds) - mki))))
        print("  optimizing mask kind=" + kind + " budget=" + repr(phase_budget))
        a_out = optimize_mask(best_seq, m, phase_budget, deadline, verbose=True)
        best_seq, best_score = consider_candidate(best_seq, best_score, a_out, "opt_" + str(kind), deadline)
        if math.isfinite(best_score) and best_score <= 1.0002:
            break

    # Length probes.
    if time_rem(deadline) > 320:
        base = best_seq
        base_n = int(base.size)
        targets = []
        for delta in (1200, -1200, 600, -600):
            tt = base_n + delta
            if 1600 <= tt <= 9000 and tt not in targets:
                targets.append(tt)
        for target in targets[:3]:
            if time_rem(deadline) < 170:
                break
            print("  length probe n=" + str(target))
            a0 = interpolate_sequence(base, target)
            best_seq, best_score = consider_candidate(best_seq, best_score, a0, "len_" + str(target), deadline)
            if time_rem(deadline) > 260:
                kind_use = mask_kind if mask_kind in mask_kinds else "full"
                m = make_simple_mask(target, kind_use)
                a_out = optimize_mask(a0, m, min(200.0, time_rem(deadline) * 0.35), deadline, verbose=False)
                best_seq, best_score = consider_candidate(best_seq, best_score, a_out, "lenpolish_" + str(target), deadline)
            if math.isfinite(best_score) and best_score <= 1.0002:
                break

    # Final quick polish on the actual best.
    if time_rem(deadline) > 70 and best_seq is not None:
        mfin = make_simple_mask(best_seq.size, mask_kind if mask_kind in mask_kinds else "full")
        a_fin = pair_polish(best_seq, mfin, min(150.0, time_rem(deadline) - 30),
                            active_max=min(3500, 2 * int(best_seq.size)), deadline=deadline)
        best_seq, best_score = consider_candidate(best_seq, best_score, a_fin, "final_pair", deadline)

    # Final validation.
    if best_seq is None:
        best_seq = clean_norm(np.ones(5000, dtype=np.float64))
    final_arr = clean_norm(best_seq)
    final_score = safe_eval_arr(final_arr)
    if math.isfinite(final_score) and (not math.isfinite(best_score) or final_score < best_score):
        best_score = final_score

    # If the real evaluator could not be used at all, return a valid normalized sequence.
    if not math.isfinite(best_score):
        best_score = evaluate_proxy(final_arr, None)

    print("  FINAL score=" + repr(best_score) + " n=" + str(final_arr.size) +
          " elapsed=" + repr(time.monotonic() - t_start))
    return [float(x) for x in final_arr]
```
