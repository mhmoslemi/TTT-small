        # Adaptive active set management
        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)

        # Grow active set: add near-binding constraints
        if len(clist) < max_K:
            near = np.argsort(ac)[::-1][:max_K]
            for k in near:
                k = int(k)
                if ac[k] > M_curr - 0.02 * M_curr:
                    active_set.add(k)
            clist = sorted(active_set)

        K = len(clist)
        if K == 0:
            break

        g_dir, M_lp = lp_solve_direction(y, clist, n, solver, warm_start=warm_g)
        if g_dir is None:
            if len(clist) > init_K:
                active_set = set(int(clist[:init_K]))
                continue
            break
        warm_g = g_dir.copy()

        # Check violations and add to active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                active_set.add(int(k))

        # Golden section line search
        best_alpha = golden_line_search(y, g_dir, n)
        y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy
        else:
            y = g_dir.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t=%d (%.1fs) J=%.6f |S|=%d" % (t, elapsed, 2 * n * best_m, len(active_set)))

    if verbose:
        print("    LP-FW done (%.1fs) J=%.6f" % (time.monotonic() - t0, 2 * n * best_m))
    return best_a, 2.0 * n * best_m

def softmax_gradient_descent(a0, time_limit, tau0=0.5, tau_decay=0.9995, lr0=2.0, verbose=True):
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = 2.0 * n * fast_autocorr(x).max()
    t0 = time.monotonic()
    v = np.zeros(n, dtype=np.float64)
    beta = 0.85
    tau = tau0
    stall = 0
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 20000:
            break

        ac = fast_autocorr(x)
        c_max = ac.max()

        # Adaptive temperature
        tau = max(0.01, tau0 * (tau_decay ** t))

        # Softmax weights focusing on top peaks
        c_scaled = (ac - c_max) / (tau + 1e-10)
        # Clip to avoid overflow
        c_scaled = np.maximum(c_scaled, -50.0)
        w = np.exp(c_scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(ac)
            w[int(np.argmax(ac))] = 1.0
        else:
            w = w / w_sum

        # Use top K weights for efficiency
        K_eff = min(100, len(w))
        top_idx = np.argsort(w)[::-1][:K_eff]
        w_top = w[top_idx]
        w_top = w_top / w_top.sum()

        # Compute gradient: weighted sum of shifted copies of x
        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w_top, top_idx):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            if i_s < i_e:
                idx = np.arange(i_s, i_e)
                grad[idx] += 2.0 * wi * x[ki - idx]

        # AdaGrad-style adaptive learning rate
        lr = lr0 / (np.sqrt(t + 1.0) + 1.0)
        v = beta * v - lr * grad
        x_new = x + 0.5 * v
        x_new = np.maximum(x_new, 1e-15)
        x_new = x_new / x_new.sum()

        val = 2.0 * n * fast_autocorr(x_new).max()
        if val < best_val - 1e-12:
            best_val = val
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x = x_new

        if verbose and t % 2000 == 0:
            print("    SoftmaxGD t=%d (%.1fs) J=%.6f tau=%.4f" % (t, time.monotonic() - t0, best_val, tau))

    return best_a, best_val

def mult_gradient_descent(a0, time_limit, top_k=50, temp=30.0, c0=2.5, verbose=True):
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = 2.0 * n * fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    x_prev = x.copy()
    stall = 0
    cur_temp = temp

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 8000:
            break

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]
        cur_temp = temp * max(0.2, 1.0 - t / 80000.0)
        w = np.exp(cur_temp * (top_vals - top_vals[0]))
        w = w / w.sum()

        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            idx_arr = np.arange(i_s, i_e)
            grad[idx_arr] += 2.0 * wi * y[ki - idx_arr]

        eta_t = c0 / (np.sqrt(max(t, 1)) + 1.0)
        log_a = np.log(y) - eta_t * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        val = 2.0 * n * fast_autocorr(x).max()
        if val < best_val - 1e-12:
            best_val = val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000 and stall % 3000 == 0:
            noise_scale = 0.005 + 0.01 * (stall / 10000.0)
            noise = noise_scale * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            stall = 0

        if verbose and t % 2000 == 0:
            print("    MultGD t=%d (%.1fs) J=%.6f" % (t, time.monotonic() - t0, best_val))

    return best_a, best_val

def peak_deflation(a0, time_limit, n_targets=10, verbose=True):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for rnd in range(2000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        targets = np.argsort(ac)[::-1][:n_targets]
        improved = False

        for k in targets[:6]:
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score destinations
            scores = np.zeros(n)
            for k2 in targets[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
            scores[src] = 1e10

            candidate_dsts = np.argsort(scores)[:40]
            best_m_trial = best_m
            best_frac = None
            best_dst = None

            for dst in candidate_dsts:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in [0.003, 0.007, 0.015, 0.03]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m_trial - 1e-15:
                        best_m_trial = Mt
                        best_frac = frac
                        best_dst = dst

            if best_frac is not None:
                amt = a[src] * best_frac
                a[src] -= amt
                a[best_dst] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    best_m = fast_autocorr(a).max()
                    best_a = a.copy()
                    improved = True

        if not improved:
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.006)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            improved = True
                            break

        if verbose and rnd % 200 == 0:
            print("    Deflate rnd=%d J=%.6f" % (rnd, 2 * n * best_m))

    return best_a, 2.0 * n * best_m

def coord_polish(a0, time_limit, seed=42, verbose=True):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 1000:
        ac = fast_autocorr(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            top_srcs = idx[np.argsort(contrib)[::-1][:3]]
            for src in top_srcs:
                src = int(src)
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                scores = np.zeros(n)
                for k2 in top_k[:5]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                scores[src] = 1e10

                candidates = np.argsort(scores)[:25]
                found = False
                for dst in candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-15:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            found = True
                            break
                    if found:
                        break

        if not improved:
            stall += 1
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.008)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 200 == 0:
            print("    Coord (%.1fs) J=%.6f stall=%d" % (time.monotonic() - t0, 2 * n * best_m, stall))

    return best_a, 2.0 * n * best_m

def freq_domain_perturb(a0, n_trials=20, verbose=False):
    """Perturb the magnitude spectrum and project back."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_val = obj_val(a)
    rng = np.random.default_rng(123)

    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_flat = mag.copy().flatten()
    mag_mean = mag_flat.mean()

    for trial in range(n_trials):
        # Create a perturbation that flattens the spectrum
        alpha = rng.uniform(0.1, 0.5)
        mag_new = (1 - alpha) * mag + alpha * mag_mean

        # Random frequency-selective modification
        freqs = rng.choice(len(mag), size=n, replace=False)
        for f in freqs[:n // 4]:
            mag_new[f] *= rng.uniform(0.7, 1.3)

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        val = obj_val(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            if verbose:
                print("    FreqP: trial %d J=%.6f" % (trial, best_val))

    return best_a, best_val

def diverse_init(n, parent_seq):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)

    candidates = []

    # Sine squared
    candidates.append(normalize(np.sin(np.pi * x) ** 2))
    # Raised cosine
    candidates.append(normalize(0.5 * (1 + np.cos(2 * np.pi * x - np.pi))))
    # Tukey windows
    for alpha in [0.1, 0.2, 0.3, 0.5]:
        cands = np.ones(n)
        m = int(alpha * n)
        if m > 0:
            taper = 0.5 * (1 + np.cos(np.pi * np.arange(m) / m))
            cands[:m] = taper
            cands[-m:] = taper[::-1]
        candidates.append(normalize(cands))

    # Gaussian
    for c in [0.3, 0.4, 0.5]:
        for sg in [0.1, 0.15, 0.2, 0.25]:
            candidates.append(normalize(np.exp(-((x - c) ** 2) / (2 * sg * sg))))

    # Double bump
    for c1, c2 in [(0.25, 0.75), (0.2, 0.8), (0.3, 0.7)]:
        for sg in [0.08, 0.12, 0.18]:
            cands = np.exp(-((x - c1) ** 2) / (2 * sg * sg)) + np.exp(-((x - c2) ** 2) / (2 * sg * sg))
            candidates.append(normalize(cands))

    # Power law
    for p in [0.3, 0.5, 0.7, 1.0, 1.5]:
        candidates.append(normalize(np.maximum(x ** p, 1e-10)))
        candidates.append(normalize(np.maximum((1 - x) ** p, 1e-10)))

    # Sine power
    for f in [1, 2, 3]:
        candidates.append(normalize(np.sin(np.pi * x) ** (2 * f)))

    # Parent blends
    if parent_seq is not None:
        p = normalize(parent_seq)
        for w in [0.5, 0.7, 0.9]:
            candidates.append(normalize(w * p + (1 - w) * normalize(np.sin(np.pi * x) ** 2)))
            candidates.append(normalize(w * p + (1 - w) * candidates[2]))

    best_a = candidates[0]
    best_val = obj_val(best_a)
    for c in candidates:
        v = obj_val(c)
        if v < best_val:
            best_val = v
            best_a = c.copy()

    return best_a, best_val

    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20:
        a_out, v_out = peak_deflation(normalize(best_seq), t5, n_targets=10, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  Deflation: %.6f" % sc)

    # Phase 6: Frequency domain perturbations
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print("\n=== P6: Freq Perturb ===")
        a_out, v_out = freq_domain_perturb(normalize(best_seq), n_trials=30, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  FreqP: %.6f" % sc)

    # Phase 7: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P7: Coord Polish (n=%d) ===" % n_cur)
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 15:
        a_out, v_out = coord_polish(normalize(best_seq), t7, seed=42, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  Polish: %.6f" % sc)

    # Phase 8: Final short LP burst
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        n_fin = min(n_cur, 10000)
        a_fin = interpolate_seq(best_seq, n_fin) if n_fin < n_cur else best_seq.copy()
        print("\n=== P8: Final Sprint (n=%d) ===" % n_fin)
        t8 = min(remaining * 0.6, 200.0)
        if t8 > 20:
            a_out, v_out = lp_frank_wolfe(normalize(a_fin), t8, solver,
                                           init_K=500, max_K=5000, verbose=False)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out if n_fin == n_cur else interpolate_seq(a_out, n_cur)
                print("  Final Sprint: %.6f" % sc)

    # Final coord polish if time
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15 and best_seq is not None:
        a_out, v_out = coord_polish(normalize(best_seq), remaining - 10, seed=99, verbose=False)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  Final polish: %.6f" % sc)

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=%.6f n=%d t=%.1fs" % (best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: fallback")
        n_fb = 8000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in np.asarray(best_seq, dtype=np.float64)]
    return result
