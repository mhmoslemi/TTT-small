    # (a*a)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
    # 
    # d/d a[i] (a*a)[k] = a[k-i] (from j=i term) + a[k-i] (from k-j=i term, i.e., j=k-i)
    #                    = 2*a[k-i] if i ≠ k-i
    #                    = a[i] if i = k-i (i.e., 2i=k)
    #                    = 0 if k-i not in [0,n-1]
    
    # So for the smoothed objective: F(a) = sum_k p_k * (a*a)[k]
    # dF/d a[i] = sum_k p_k * (2*a[k-i] - a[k-i]*delta_{2i,k})
    # ≈ 2 * sum_k p_k * a[k-i] (ignoring the small diagonal correction)
    
    # Now: sum_k p_k * a[k-i] where 0<=k-i<n means i <= k < i+n
    # = sum_{j=0}^{n-1} p[i+j] * a[j] (substituting j=k-i)
    # = (p_shifted_i * a) where p_shifted_i[j] = p[i+j]
    
    # This is a cross-correlation! Specifically:
    # sum_j p[i+j] * a[j] = (a * p_reversed)[i] ... let me think again
    # 
    # Actually, it's the cross-correlation of a and p:
    # (a ⊛ p)[i] = sum_j a[j] * conj(p[i+j]) for real signals = sum_j a[j] * p[i+j]
    # 
    # So dF/d a[i] ≈ 2 * (a ⊛ p_full)[i]
    
    # Cross-correlation can be computed via FFT:
    # a ⊛ p = ifft(conj(fft(a)) * fft(p))
    # But we need the "toeplitz" version (not linear), or we can use:
    # (a ⊛ p)[i] = ifft(fft(p, 2n) * conj(fft(a, 2n))) ... hmm
    
    # Actually, let me use a different formulation:
    # sum_j a[j] * p[i+j] = sum_j a[j] * p[i+j]
    # Let's define b[j] = a[-j] (reversed a, with b[j] = a[-j] for j in -n+1..0)
    # Then sum_j a[j] * p[i+j] = sum_j b[-j] * p[i+j] ... this is getting complicated
    
    # Let me just use the approach from the previous code which works:
    # conv = irfft(rfft(a_rev, fft_size) * rfft(p_full, fft_size), fft_size)
    # grad = conv[n-1 : 2*n-1]
    
    # Let me verify with a small example:
    # a = [a0, a1, a2], n=3
    # p_full = [p0, p1, p2, p3, p4] (size 5)
    # a_rev = [a2, a1, a0]
    # 
    # conv[m] = sum_j p_full[j] * a_rev[m-j] (linear convolution)
    # conv[2] = sum_j p[j]*a_rev[2-j] = p[0]*a[2] + p[1]*a[1] + p[2]*a[0]
    # 
    # grad[0] should be sum_k p[k] * 2*a[k] (since k-0=k, need 0<=k<3)
    # = 2*(p[0]*a[0] + p[1]*a[1] + p[2]*a[2])
    # 
    # conv[2] = p[0]*a[2] + p[1]*a[1] + p[2]*a[0]
    # That's not the same as what we want for grad[0].
    
    # Let me try conv[0]:
    # conv[0] = p[0]*a[2]
    # That's not right either.
    
    # OK let me think about this differently.
    # We want: grad[i] = 2 * sum_k p[k] * a[k-i] for i=0..n-1
    # = 2 * sum_j p[i+j] * a[j] for j=0..n-1 (substituting j=k-i)
    
    # This is: 2 * cross_corr(a, p)[i] where cross_corr(a,p)[i] = sum_j a[j]*p[i+j]
    
    # Using FFT: cross_corr(a,p)[i] = ifft(fft(p, 2n) * conj(fft(a, 2n)))[1:n+1]... 
    # Actually, the standard formula is:
    # cross_corr(x,y)[i] = sum_j x[j+y-i] * y[j] ... conventions vary
    
    # Let me use numpy directly for clarity (for moderate sizes):
    # grad[i] = 2 * sum_{j=0}^{n-1} a[j] * p[i+j] (with p[k]=0 for k outside [0,2n-2])
    
    # This is a "sliding dot product" which can be computed as:
    # Create an array of length 2n-1: q[k] = p[k]
    # grad[i] = 2 * dot(a, q[i:i+n])
    
    # For this, we can use: np.correlate or FFT
    # np.correlate(a, p_full, mode='valid') gives sum_j a[j]*p_full[i+j] for valid i
    # But the lengths need to match...
    
    # Actually: np.correlate(p_full, a, mode='full')[n-1:n-1+n] 
    # Hmm, let me just verify the previous code's approach works and use it.
    
    # The previous code's approach:
    # Ar = rfft(a_rev)
    # Pr = rfft(p_full)  
    # conv = irfft(Ar * Pr)
    # grad = conv[n-1:2*n-1]
    
    # Let me verify: conv = a_rev * p_full (linear convolution)
    # conv[m] = sum_j a_rev[j] * p_full[m-j]
    # a_rev[j] = a[n-1-j]
    # conv[m] = sum_j a[n-1-j] * p_full[m-j]
    # Let i = n-1-j, so j = n-1-i:
    # conv[m] = sum_i a[i] * p_full[m-(n-1-i)] = sum_i a[i] * p_full[m-n+1+i]
    # 
    # For m in range [n-1, 2n-2]:
    # Let m = n-1+i0 where i0 in [0, n-1]:
    # conv[n-1+i0] = sum_i a[i] * p_full[i0+i]
    # = sum_i a[i] * p[i0+i]
    # 
    # And we want: grad[i0] = 2 * sum_i a[i] * p[i0+i]
    
    # So grad = 2 * conv[n-1:2*n-1] ✓
    
    # Great, the formula is correct!
    
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, b, b_max


def nesterov_opt(a0, time_limit, beta_start=30.0, beta_end=1000.0,
                 beta_growth=1.004, eta_init=0.5, top_k_ratio=0.02,
                 restart_every=1000, verbose=True, seed=42):
    """Nesterov with adaptive parameters and diverse restarts."""
    n = len(a0)
    x = norm1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(20, int(n * top_k_ratio))
    eta = eta_init
    size = 2 * n - 1
    fft_size = 1 << (n + size - 2).bit_length()

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        y = y / s_y

        # Compute gradient
        grad, b, b_max = compute_subgradient(y, top_k, beta)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            x_new = x
            b_new = fast_autocorr(x_new)
            max_new = b_new.max()
        else:
            # Backtracking line search
            y_max = b.max()  # Use full max for acceptance
            found = False
            eta_trial = eta
            for _ in range(8):
                z = y - eta_trial * grad
                x_new = simplex_project(z)
                b_new = fast_autocorr(x_new)
                max_new = b_new.max()
                if max_new <= y_max + 1e-14:
                    found = True
                    break
                eta_trial *= 0.5
            if not found:
                x_new = x
                max_new = b.max()
                eta = max(eta_trial, 0.01)

        if max_new < best_m - 1e-16:
            best_m = max_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.05, 3.0)
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        # Adaptive beta
        if t % 40 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        # Adaptive top_k: increase when many peaks are close
        if t % 200 == 0:
            b_check = fast_autocorr(x)
            b_sorted = np.sort(b_check)[::-1]
            if len(b_sorted) > 10:
                # Count peaks within 10% of max
                n_close = int(np.sum(b_sorted > 0.9 * b_sorted[0]))
                top_k = max(20, min(size // 10, n_close + 10))

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = int(rng.integers(8))
            if strategy == 0:
                noise = 0.01 + 0.02 * rng.random()
                x_r = best_a + noise * rng.standard_normal(n) * (best_a.max() + 1e-10)
            elif strategy == 1:
                x_r = 0.85 * best_a + 0.15 * rng.dirichlet(np.ones(n) * 0.2)
            elif strategy == 2:
                x_r = best_a * np.exp(0.05 * rng.standard_normal(n))
            elif strategy == 3:
                mask = rng.random(n) > 0.5
                x_r = best_a.copy()
                x_r[mask] *= (0.8 + 0.1 * rng.random(int(mask.sum())))
                x_r[~mask] *= (1.1 + 0.1 * rng.random(int((~mask).sum())))
            elif strategy == 4:
                shift = int(rng.integers(1, max(2, n // 15)))
                x_r = np.roll(best_a, shift)
            elif strategy == 5:
                # Flip
                x_r = best_a[::-1].copy()
            elif strategy == 6:
                # Gaussian perturbation centered on peaks
                b_curr = fast_autocorr(best_a)
                peak_locs = np.argsort(b_curr)[::-1][:5]
                x_r = best_a.copy()
                for pl in peak_locs:
                    center = pl // 2
                    width = max(1, n // 50)
                    lo = max(0, center - width)
                    hi = min(n, center + width)
                    x_r[lo:hi] *= (1.0 - 0.02 * rng.random())
            else:
                # Mix with smooth curve
                x_smooth = np.sin(np.pi * np.linspace(0, 1, n)) ** 2
                x_smooth = norm1(x_smooth)
                x_r = 0.9 * best_a + 0.1 * x_smooth

            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 10.0)
            eta = eta_init
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    Nest restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " b=" + str(int(beta)) + " tk=" + str(top_k))

    return best_a, best_m


def peak_targeted_lsearch(a0, time_limit, verbose=True, seed=321):
    """Direct peak-targeted local search."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    size = 2 * n - 1

    while time.monotonic() - t0 < time_limit and stall < 3000:
        ac = fast_autocorr(a)
        
        # Get top peaks
        n_peaks = min(15, size)
        top_idx = np.argpartition(ac, -n_peaks)[-n_peaks:]
        top_idx = top_idx[np.argsort(ac[top_idx])[::-1]]
        
        improved = False
        
        for k in top_idx[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            
            # Find contributing pairs
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue
            
            # Target the largest contributors
            top_pairs = idx[np.argsort(contrib)[::-1][:5]]
            
            for src in top_pairs:
                src = int(src)
                partner = ki - src
                if not (0 <= partner < n):
                    continue
                
                # Strategy 1: Reduce a[src] and redistribute
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    
                    # Try moving mass to positions that don't create new peaks
                    # Good destinations: positions where a is already high but not
                    # contributing to this peak
                    for _ in range(8):
                        dst = int(rng.integers(n))
                        if dst == src or dst == partner:
                            continue
                        # Check if dst would create a problem
                        # dst + (ki - dst) = ki... that's the same peak
                        # Check other peaks
                        a_trial = a.copy()
                        a_trial[src] = max(0, a_trial[src] - amt)
                        a_trial[dst] += amt
                        a_trial = a_trial / a_trial.sum()
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            
            # Strategy 2: Reduce both a[src] and a[partner]
            if not improved:
                amt1 = a[src] * 0.02
                amt2 = a[partner] * 0.02
                if amt1 > 1e-15 and amt2 > 1e-15:
                    for _ in range(4):
                        dst1 = int(rng.integers(n))
                        dst2 = int(rng.integers(n))
                        if dst1 == src or dst1 == partner or dst2 == src or dst2 == partner:
                            continue
                        a_trial = a.copy()
                        a_trial[src] = max(0, a_trial[src] - amt1)
                        a_trial[partner] = max(0, a_trial[partner] - amt2)
                        a_trial[dst1] += amt1
                        a_trial[dst2] += amt2
                        a_trial = np.maximum(a_trial, 0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 50:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSearch (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2000, verbose=True):
    """LP-Frank-Wolfe method."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 500
    size = 2 * n - 1
    fft_size = 1 << (n + size - 2).bit_length()

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[ki - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:20]]
            for k in top_v:
                active_set.add(int(k))

        # Line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        for alpha in [best_alpha, best_alpha * 0.5, best_alpha * 2.0, 0.05, 0.1, 0.15, 0.2]:
            alpha = max(0.01, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW it=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def spectral_shape(a0, time_limit, verbose=True, seed=777):
    """Spectral domain shaping to flatten autocorrelation."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()

    alpha = 0.03
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Target flat spectrum (excluding DC)
        if len(power) > 1:
            target = np.median(power[1:])
        else:
            target = power[0]

        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.05:
                red = 1.0 - alpha * min(ratio - 1.0, 3.0)
                A_new[k] *= max(0.2, np.sqrt(max(0.1, red)))
            elif ratio < 0.7:
                inc = 1.0 + alpha * 0.5 * (0.7 - ratio)
                A_new[k] *= min(2.5, np.sqrt(max(0.1, inc)))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 500 == 0:
            # Phase perturbation
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.03
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial

        if it % 300 == 0:
            alpha = max(0.003, alpha * 0.97)

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.
