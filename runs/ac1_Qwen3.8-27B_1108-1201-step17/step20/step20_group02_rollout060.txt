def _get_solver():

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    if s < 1e-15:

def smooth_gradient(a, tau):
    """Compute smooth gradient of logsumexp(conv(a,a)/tau) w.r.t. a.
    Returns gradient vector of length n."""
    n = len(a)
    C = fast_convolve(a)
    C_shifted = C - C.max()
    p = np.exp(C_shifted / tau)
    p = p / p.sum()
    # grad_j = 2 * sum_{i=0}^{n-1} p_{j+i} * a_i
    # Using FFT: reverse p, convolve with a
    L = 1 << (4 * n - 1).bit_length()
    pr = p[::-1]
    PR = np.fft.rfft(pr, L)
    A = np.fft.rfft(a, L)
    corr = np.fft.irfft(PR * A, L)
    # (p * a)_j = corr[2n-2-j]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * corr[idx]
    return grad, p

def build_lp_matrix(y, active_set, n):
    K = len(active_set)
    coeffs = np.zeros((K, n))
    for jj in range(K):

        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs

        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]
        all_viol = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(all_viol) > 0:
            viol_amt = conv_ys[all_viol] - M_curr
            top_v = all_viol[np.argsort(viol_amt)[::-1][:60]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        # Backtracking line search
        current_M = fast_convolve(y).max()
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_M = current_M

        alphas = [1.0, 0.7, 0.5, 0.3, 0.15,
                  max(0.08, 3.0 / (iter_count + 2)),
                  max(0.03, 1.0 / (iter_count + 1))]
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 3:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("  LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + " |S|=" + str(len(active_set)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("  LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + " iters=" + str(iter_count))
    return best_a, best_m


def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def coord_polish(a0, time_limit, verbose=False):
    """Coordinate descent: transfer mass from peak contributors to non-peak regions."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        k_argmax = int(np.argmax(ac))
        # Find which pairs contribute to the peak
        i_start = max(0, k_argmax - n + 1)
        i_end = min(n, k_argmax + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[k_argmax - idx]
        total_contrib = contrib.sum()
        if total_contrib < 1e-20:
            break

        # Source: element with largest contribution
        src_offset = int(np.argmax(contrib))
        src = int(idx[src_offset])

        # Find destination: minimize impact on top peaks
        top_k = np.argsort(ac)[::-1][:20]
        scores = np.zeros(n)
        for k2 in top_k[:10]:
            k2i = int(k2)
            s2 = max(0, k2i - n + 1)
            e2 = min(n - 1, k2i)
            if s2 <= e2:
                idx2 = np.arange(s2, e2 + 1)
                scores[idx2] += a[k2i - idx2]

        in_peak = np.zeros(n, dtype=bool)
        for k2 in top_k[:10]:
            k2i = int(k2)
            s2 = max(0, k2i - n + 1)
            e2 = min(n - 1, k2i)
            if s2 <= e2:
                in_peak[s2:e2 + 1] = True

        candidates = np.where(~in_peak)[0]
        if len(candidates) < 10:
            candidates = np.arange(n)

        # Pick a few candidate destinations
        cand_idx = np.argsort(scores[candidates])[::-1]  # highest impact
        # Actually we want lowest impact
        cand_idx_low = np.argsort(scores[candidates])[:10]

        found = False
        for ci in cand_idx_low[:5]:
            dst = int(candidates[ci])
            if dst == src:
                continue
            fracs = [0.01, 0.03, 0.05, 0.1, 0.2]
            for frac in fracs:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    stall = 0
                    found = True
                    break
            if found:
                break

        if not found:
            # Try random moves
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            found = True
                            break
            if not found:
                stall += 1

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print("  Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + " stall=" + str(stall))

    return best_a, best_m


def get_initial(t_global):
    """Get best initial sequence."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = eval_safe(p_norm)
            if np.isfinite(sc):
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
                return p_norm, sc
    except Exception:
        pass

    # Fallback: use a designed initialization
    n = 4000
    x = np.linspace(0.0, 1.0, n)
    # Raised cosine shape (zero at endpoints, smooth)
    f = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    f = np.maximum(f, 1e-6)
    f_norm = f / f.sum()
    sc = eval_safe(f_norm)
    print("Default init: n=" + str(n) + ", score=" + repr(sc))
    return f_norm, sc


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total budget: " + repr(time_budget) + "s")

    def remaining():
        return time_budget - (time.monotonic() - t_global)

    def try_update(best_seq, best_score, a, label):
        if a is None:
            return best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return best_seq, best_score
        a_norm = a / s
        sc = eval_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
        return best_seq, best_score

    solver = _get_solver()
    best_seq, best_score = get_initial(t_global)

    # === Phase 1: Smooth gradient descent ===
    rt = remaining()
    t1 = min(rt * 0.35, 550.0)
    if t1 > 30:
        print("\n=== Phase 1: Smooth GD (n=" + str(len(best_seq)) + ", t=" + str(round(t1, 0)) + "s) ===")
        a_out, _ = smooth_gd(norm1(best_seq), t1, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "smooth_gd")

    # === Phase 2: First LP-FW pass ===
    rt = remaining()
    t2 = min(rt * 0.25, 400.0)
    if t2 > 30 and len(best_seq) <= 12000:
        print("\n=== Phase 2: LP-FW (n=" + str(len(best_seq)) + ", t=" + str(round(t2, 0)) + "s) ===")
        a_out, _ = lp_fw(norm1(best_seq), t2, solver, verbose=True, initial_K=500, max_K=4000)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1")

    # === Phase 3: Second smooth GD round (from LP-FW result) ===
    rt = remaining()
    t3 = min(rt * 0.12, 180.0)
    if t3 > 20:
        print("\n=== Phase 3: Smooth GD round 2 (t=" + str(round(t3, 0)) + "s) ===")
        a_out, _ = smooth_gd(norm1(best_seq), t3, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "smooth_gd2")

    # === Phase 4: Second LP-FW pass ===
    rt = remaining()
    t4 = min(rt * 0.20, 300.0)
    if t4 > 30 and len(best_seq) <= 12000:
        print("\n=== Phase 4: LP-FW round 2 (n=" + str(len(best_seq)) + ", t=" + str(round(t4, 0)) + "s) ===")
        a_out, _ = lp_fw(norm1(best_seq), t4, solver, verbose=True, initial_K=800, max_K=5000)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2")

    # === Phase 5: Multi-n exploration ===
    n_best = len(best_seq)
    rt = remaining()
    if rt > 200:
        print("\n=== Phase 5: Multi-n exploration ===")
        # Try a few different n values
        candidate_ns = []
        for dn in [-1000, -500, 500, 1000, 2000]:
            nn = n_best + dn
            if 2000 <= nn <= 12000 and nn != n_best:
                candidate_ns.append(nn)
        # Also try specific values
        for nn in [3000, 4000, 5000, 6000, 8000, 10000]:
            if nn not in candidate_ns and 2000 <= nn <= 12000:
                candidate_ns.append(nn)

        candidate_ns = sorted(set(candidate_ns), key=lambda x: abs(x - n_best))[:4]

        for new_n in candidate_ns:
            rt = remaining()
            if rt < 100:
                break
            print("\n  Multi-n: n=" + str(new_n) + " (remaining=" + str(round(rt, 1)) + "s)")
            a_long = interp_seq(best_seq, new_n)

            # Quick smooth GD
            t5a = min(rt * 0.15, 120.0)
            if t5a > 15:
                a_out, _ = smooth_gd(a_long, t5a, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "smooth_n" + str(new_n))

            # Quick LP-FW
            rt = remaining()
            t5b = min(rt * 0.20, 150.0)
            if t5b > 20 and new_n <= 10000:
                a_out, _ = lp_fw(norm1(best_seq), t5b, solver, verbose=False, initial_K=400, max_K=3000)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n))

    # === Phase 6: Coordinate polish ===
    rt = remaining()
    t6 = min(rt * 0.10, 120.0)
    if t6 > 15:
        print("\n=== Phase 6: Coordinate polish (n=" + str(len(best_seq)) + ") ===")
        a_out, _ = coord_polish(norm1(best_seq), t6, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord")

    # === Phase 7: Final LP-FW ===
    rt = remaining()
    t7 = min(rt * 0.25, 250.0)
    if t7 > 20 and len(best_seq) <= 12000:
        print("\n=== Phase 7: Final LP-FW (n=" + str(len(best_seq)) + ", t=" + str(round(t7, 0)) + "s) ===")
        a_out, _ = lp_fw(norm1(best_seq), t7, solver, verbose=True, initial_K=1000, max_K=6000)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final")

    # === Final coordinate polish ===
    rt = remaining()
    if rt > 15:
        a_out, _ = coord_polish(norm1(best_seq), min(rt * 0.5, 60.0), verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final")

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm1((1.0 + np.cos(2.0 * np.pi * x)) / 2.0)
        best_score = eval_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
