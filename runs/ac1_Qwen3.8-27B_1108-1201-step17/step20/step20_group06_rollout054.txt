def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:

def norm_sum1(a):
    """Normalize to sum 1 and clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def simplex_project(z):
    """Project onto the probability simplex."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def iaa_optimize(a0, time_limit, alpha_start=0.1, alpha_end=0.5, 
                 verbose=True, seed=42):
    """
    IAA-style optimization: iteratively compress the power spectrum
    to flatten the autocorrelation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    fft_size = 1 << (2 * n - 2).bit_length()
    alpha = alpha_start
    no_improve = 0
    max_no_improve = 500
    
    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        # Compute spectrum
        A = np.fft.rfft(a, fft_size)
        
        # Apply power compression
        # |A_new(f)| = |A(f)|^alpha (for f ≠ 0)
        # Keep DC component unchanged
        A_new = A.copy()
        mags = np.abs(A_new)
        mags_new = mags.copy()
        mags_new[1:] = mags[1:] ** alpha
        # Preserve phases
        phases = A_new / np.maximum(mags, 1e-30)
        A_new = mags_new * phases
        A_new[0] = A[0]  # Keep DC
        
        # Inverse FFT
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            no_improve += 1
            alpha = max(0.7, alpha * 1.2)  # Be less aggressive
            continue
        a_new = a_new / s_new
        
        m_new = fast_autocorr(a_new).max()
        
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Increase compression
            alpha = min(alpha * 1.05, alpha_end)
        else:
            no_improve += 1
            # Try less compression
            alpha = max(0.05, alpha * 0.95)
            
            # If stuck, try a random phase perturbation
            if no_improve % 100 == 0:
                A = np.fft.rfft(a, fft_size)
                phase_pert = rng.standard_normal(len(A)) * 0.1
                A[1:] *= np.exp(1j * phase_pert[1:])
                a_trial = np.fft.irfft(A, fft_size)[:n]
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    m_t = fast_autocorr(a_trial).max()
                    if m_t < best_m:
                        best_m = m_t
                        best_a = a_trial.copy()
                        a = a_trial
                        no_improve = 0
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    IAA it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + 
                  repr(round(J, 6)) + " alpha=" + str(round(alpha, 3)))
    
    return best_a, best_m

        # The gradient is: grad_a[i] = sum_k p[k] * (a[k-i] + a[i-k]) for 0<=i<n
        # = sum_k p[k]*a[k-i] + sum_k p[k]*a[i-k]
        
        # First sum: let j = k-i, so k = j+i: sum_j p[j+i]*a[j] for j=0..n-1-i
        # Second sum: let j = i-k, so k = i-j: sum_j p[i-j]*a[j] for j=0..i (or j=0..n-1)
        
        # Combined: sum_j a[j] * (p[j+i] + p[i-j]) where p is defined as 0 outside [0, 2n-2]
        
        # Since p is symmetric (p[k] = p[2n-2-k]):
        # p[j+i] + p[i-j] = p[j+i] + p[2n-2-i+j]
        # This doesn't simplify nicely in general.
        
        # Let me just compute it numerically using FFT:
        # Define an array c of length 2n-1: c[k] = p[k]
        # Define an array d of length 2n-1: d[k] = a[k] for k<n, 0 otherwise
        # Then the convolution (c * d) has length 4n-3, and:
        # (c*d)[i + (n-1)] = sum_k c[k] * d[i+n-1-k] = sum_k p[k] * a[i+n-1-k]
        # For 0 <= i < n: this is sum_{k: 0<=i+n-1-k<n} p[k]*a[i+n-1-k]
        # = sum_{k: n-1 <= k <= n-1+i} p[k]*a[i+n-1-k]
        # = sum_{j=0}^{i} p[n-1+j]*a[i-j] (letting j = n-1-k... wait this is getting confusing)
        
        # I'll just use the straightforward FFT-based gradient computation:
        # The key insight is that b = conv(a,a), and the gradient of a scalar function f(b) w.r.t. a
        # is: grad_a = conv(db/da, ...) 
        # Specifically: d f(b) / d a[i] = sum_k (f'(b[k]) * db[k]/da[i])
        # = sum_k p[k] * (a[k-i] + a[i-k])  (for valid indices)
        
        # This can be computed as:
        # grad_a = convodecorrelate(p, a) + convodecorrelate(p, a)  (two terms)
        # where the convolution is linear and only over valid indices.
        
        # In practice, for the zero-padded FFT:
        # Let P = rfft(p, 2*fft_size) and A = rfft(a, 2*fft_size) (zero-padded)
        # Then IFFT(P * conj(A)) gives the cross-correlation... 
        
        # Actually, the cleanest way:
        # grad_a[i] = sum_k p[k]*a[k-i] + sum_k p[k]*a[i-k]
        # The first term is the cross-correlation (p ⊛ a)[i] 
        # The second term is (a ⊛ p)[i] = (p ⊛ a)[i] (by symmetry of cross-correlation for real signals)
        # Wait: (p ⊛ a)[i] = sum_k p[k+i]*a[k] (one definition)
        #     (a ⊛ p)[i] = sum_k a[k+i]*p[k]
        # These are the same: (p ⊛ a)[i] = sum_k p[k+i]*a[k] = sum_j p[j]*a[j-i] (let j=k+i)
        # And (a ⊛ p)[i] = sum_k a[k+i]*p[k] = sum_j a[j]*p[j-i] (let j=k+i)
        # So (p ⊛ a)[i] = sum_j p[j]*a[j-i] and (a ⊛ p)[i] = sum_j a[j]*p[j-i]
        # These are NOT the same in general!
        
        # grad_a[i] = sum_j a[j]*p[j+i] + sum_j a[j]*p[j-i]  (reindexing the original)
        # = sum_j a[j] * (p[j+i] + p[j-i])
        
        # Since p is symmetric about n-1: p[j] = p[2(n-1)-j]
        # p[j+i] + p[j-i] = p[j+i] + p[2n-2-j+i]
        
        # OK this is taking too long. Let me just compute the gradient numerically
        # using the FFT of the "reversed" version. The correct formula is:
        
        # grad_a = 2 * real(ifft(rfft(p, fft_size) * rfft(a, fft_size)))[:n]
        # NO. That's the convolution p * a, not what we want.
        
        # The correct formula:
        # sum_k p[k] * a[k-i] = (p_linear_conv_a_reversed)[i] ... 
        # Actually for linear convolution: (p * a_reversed)[k] = sum_j p[j]*a_reversed[k-j] = sum_j p[j]*a[n-1-k+j]
        # This is getting too complicated analytically.
        
        # Let me just compute it directly for each i (O(n^2) but n might be manageable)
        # Or better: use the fact that it's a convolution and use FFT properly.
        
        # FINAL APPROACH: 
        # grad_a[i] = sum_k p[k]*a[k-i] + sum_k p[k]*a[i-k]
        # Term 1: sum_k p[k]*a[k-i], valid when 0<=k<n and 0<=k-i<n
        #         = sum_{max(0,i)}^{n-1} p[k]*a[k-i]
        #         Let j = k-i: sum_{j=0}^{n-1-i} p[j+i]*a[j]
        # Term 2: sum_k p[k]*a[i-k], valid when 0<=k<n and 0<=i-k<n
        #         = sum_{max(0,i-n+1)}^{min(n-1,i)} p[k]*a[i-k]
        #         Let j = i-k: sum_{j=0}^{min(i,n-1)} p[i-j]*a[j]
        
        # So: grad_a[i] = sum_j a[j]*(p[j+i] + p[i-j]) for valid j
        # where p is defined on [0, 2n-2] and is 0 outside.
        
        # This is a VARIATION (not standard) convolution. But I can compute it using:
        # Create a padded array: P_pad of length 4n-3 (or next power of 2)
        # P_pad[k] = p[k] for k in [0, 2n-2], 0 otherwise
        # Then: sum_j a[j]*p[j+i] = (a * P_pad)[i] (standard linear convolution, but shifted)
        # And: sum_j a[j]*p[i-j] = (a * P_pad_reversed)[i] ... 
        
        # You know what, I'll just compute it with a matrix-free approach using FFT:
        # Create arrays of length L >= 4n:
        # A_pad[j] = a[j] for 0<=j<n, 0 otherwise
        # P_pad[k] = p[k] for 0<=k<2n-1, 0 otherwise
        # Then:
        # Term1[i] = sum_j A_pad[j] * P_pad[j+i] = (A_pad * P_pad_reversed)[i+2n-2] ... 
        # This is the cross-correlation!
        
        # Cross-correlation via FFT:
        # (f ⊛ g)[k] = sum_j f[j]*g[j+k] = IFFT(conj(FFT(f)) * FFT(g))
        # (for circular) or with zero-padding for linear.
        
        # So: Term1 = cross_correlation(a, p) evaluated at lags 0, 1, ..., n-1
        #     Term2 = cross_correlation(a, p) evaluated at lags 0, -1, ..., -(n-1)
        #     = cross_correlation(a, p) at negative lags = cross_correlation(p, a) at positive lags
        
        # Since p is symmetric: p[k] = p[2n-2-k]
        # cross_correlation(a, p) at lag i = sum_j a[j]*p[j+i]
        # cross_correlation(p, a) at lag i = sum_j p[j]*a[j+i] = sum_k p[k-i]*a[k] (let k=j+i)
        #                                    = sum_k p[k-i]*a[k]
        
        # Hmm, let me just use the FFT cross-correlation formula:
        # Define the "extended cross-correlation": 
        # R[m] = sum_j a[j]*p[j+m] for m = -(2n-3), ..., 2n-3
        # Then: grad_a[i] = R[i] + R[-i] for i = 0, ..., n-1
        # (where R[0] is counted once, but since both terms give R[0], it's 2*R[0]... wait)
        
        # Actually: Term1[i] = R[i] and Term2[i] = R[-i]
        # So grad_a[i] = R[i] + R[-i]
        
        # And R[m] = cross_correlation(a, p) at lag m, which can be computed via FFT:
        # R = IFFT(conj(FFT(a_padded)) * FFT(p_padded)) (with appropriate padding)
        
        # Let me implement this:
        L = 1 << (4 * n - 3).bit_length()  # Ensure no circular wraparound
        A_pad = np.zeros(L)
        A_pad[:n] = a
        P_pad = np.zeros(L)
        P_pad[:2*n-1] = p
        
        A_f = np.fft.rfft(A_pad)
        P_f = np.fft.rfft(P_pad)
        
        # Cross-correlation: R[m] = sum_j a[j]*p[j+m]
        # = IFFT(conj(A_f) * P_f)[m] (for the circular case, with enough padding it's linear)
        R_full = np.fft.irfft(np.conj(A_f) * P_f, L)
        
        # R[m] for m = 0, 1, ..., L-1 (circular indices)
        # We need R[i] and R[-i] = R[L-i] for i = 0, ..., n-1
        grad_a = R_full[:n] + R_full[L-n+1:]  # R[0..n-1] + R[L-n..L-1] = R[i] + R[-i]
        # Wait: R[-i] in circular = R[L-i]. For i=0: R[0]+R[0] = 2*R[0] ✓
        # For i>0: R[i] + R[L-i] ✓
        
        # This should be correct!
        
        # Now, gradient w.r.t. x (log-parameterization):
        # a = softmax(x), so da[i]/dx[j] = a[i]*(delta[ij] - a[j])
        # grad_x[j] = sum_i grad_a[i] * da[i]/dx[j]
        #           = sum_i grad_a[i] * a[i] * (delta[ij] - a[j])
        #           = grad_a[j]*a[j] - a[j]*sum_i grad_a[i]*a[i]
        
        ax = grad_a * a
        grad_x = ax - a * ax.sum()
        
        # Adam update
        t_adam += 1
        b1, b2, eps = 0.9, 0.999, 1e-8
        m_v = b1 * m_v + (1 - b1) * grad_x
        v_v = b2 * v_v + (1 - b2) * grad_x ** 2
        m_hat = m_v / (1 - b1 ** t_adam)
        v_hat = v_v / (1 - b2 ** t_adam)
        
        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project x to keep a numerically stable
        # (x can be any real vector; softmax handles the rest)
        # But for numerical stability, re-center:
        x_new = x_new - np.mean(x_new)
        
        x = x_new
        
        # Check improvement
        a_current = np.exp(x - x.max())
        a_current = a_current / a_current.sum()
        m_current = fast_autocorr(a_current).max()
        
        if m_current < best_m - 1e-16:
            best_m = m_current
            best_a = a_current.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive beta
        if it % 100 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        # Adaptive learning rate
        if no_improve % 200 == 0:
            lr *= 0.9
        if no_improve > 500:
            lr *= 0.5
            no_improve = 0
            # Perturb
            rng_local = np.random.default_rng(it)
            x += 0.01 * rng_local.standard_normal(n)
            x -= np.mean(x)
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LogGrad it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" +
                  repr(round(J, 6)) + " beta=" + str(int(beta)) + " lr=" + str(round(lr, 5)))
    
    return best_a, best_m

def coordinate_descent(a0, time_limit, verbose=True, seed=777):
    """
    Directed coordinate descent: for each coordinate, try to reduce the peak

    """

        max_val = b[max_idx]

        k = int(max_idx)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        
        if js > je:
            stall += 1
            break

        contrib = np.zeros(n)
        idx = np.arange(js, je + 1)
        contrib[idx] = a[idx] * a[k - idx]

        order = np.argsort(contrib)[::-1]

                break

                    break

                a_trial[i] -= amt
