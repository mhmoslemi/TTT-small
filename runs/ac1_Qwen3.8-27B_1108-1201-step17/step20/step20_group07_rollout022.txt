    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def build_conv_matrix_row(f, k, n):
    """Build row k of the convolution matrix: C[k,j] = f[k-j] if 0<=k-j<n"""
    row = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        row[j_start:j_end+1] = f[k - np.arange(j_start, j_end+1)]
    return row

def build_conv_matrix(f, active_lags, n):
    """Build the K x n convolution matrix for given active lags."""
    K = len(active_lags)
    C = np.zeros((K, n))
    for i, k in enumerate(active_lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            C[i, j_start:j_end+1] = f[k - np.arange(j_start, j_end+1)]
    return C

def lp_frank_wolfe(a, time_limit, K_active=1000, max_iter=500, verbose=True):
    """LP-Frank-Wolfe: at each step, solve LP to find best direction."""
    n = len(a)
    a = norm_sum1(a).copy()
    best_a = a.copy()
    b = fast_autocorr(a)
    best_m = b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        M = b.max()
        
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
        
        # Select active set: top K lags
        sorted_lags = np.argsort(b)[::-1][:K_active]
        
        # Build constraint matrix
        C = build_conv_matrix(a, sorted_lags, n)
        
        # Solve LP: max sum(g) s.t. C @ g <= M, g >= 0
        g = cp.Variable(n, nonneg=True)
        constraints = [C @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        
        solver = get_solver()
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Try with smaller active set
            K_smaller = max(100, K_active // 3)
            sorted_lags_sm = np.argsort(b)[::-1][:K_smaller]
            C_sm = build_conv_matrix(a, sorted_lags_sm, n)
            g2 = cp.Variable(n, nonneg=True)
            constraints2 = [C_sm @ g2 <= M]
            prob2 = cp.Problem(cp.Maximize(cp.sum(g2)), constraints2)
            try:
                prob2.solve(solver=solver, verbose=False)
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(np.asarray(g2.value, dtype=np.float64), 0.0)
                else:
                    continue
            except Exception:
                continue
        else:
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                continue
            g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        
        g_sum = g_val.sum()
        if g_sum < 1e-10:
            continue
        
        g_dir = g_val / g_sum
        
        # Line search
        best_alpha = 0.0
        best_trial_m = best_m
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            trial = (1 - alpha) * a + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s < 1e-15:
                continue
            trial = trial / s
            m_trial = fast_autocorr(trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha
        
        if best_alpha == 0.0:
            # Try smaller steps
            for alpha in [0.02, 0.05, 0.08]:
                trial = (1 - alpha) * a + alpha * g_dir
                trial = np.maximum(trial, 0.0)
                s = trial.sum()
                if s < 1e-15:
                    continue
                trial = trial / s
                m_trial = fast_autocorr(trial).max()
                if m_trial < best_trial_m:
                    best_trial_m = m_trial
                    best_alpha = alpha
        
        if best_alpha > 0:
            a = (1 - best_alpha) * a + best_alpha * g_dir
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        else:
            # Stuck: add small random perturbation
            pert = rng.normal(0, 0.01, n) * a
            a_pert = a + pert
            a_pert = np.maximum(a_pert, 1e-12)
            a_pert = a_pert / a_pert.sum()
            m_pert = fast_autocorr(a_pert).max()
            if m_pert < best_m:
                best_m = m_pert
                best_a = a_pert.copy()
            a = a_pert
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  LP-FW it={it} ({elapsed:.1f}s): J={J:.6f}, alpha={best_alpha:.3f}, g_sum={g_sum:.4f}")
    
    return best_a, best_m
