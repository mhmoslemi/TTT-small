Main ingredients:

  3. Linear-programming Frank-Wolfe-style directions at moderate length.
  4. Periodic asymmetry breaking, symmetry checks, and final polishing.

def next_pow2(m):
    m = max(1, int(m))
    return 1 << (m - 1).bit_length()


def normalize_nonneg(a, target_sum=1.0):

    if a.size == 0:
        return np.zeros(0, dtype=np.float64)

    if s <= 1e-300 or not np.isfinite(s):
        if a.size == 0:
            return a
        return np.full(a.size, float(target_sum) / a.size, dtype=np.float64)
    return a * (float(target_sum) / s)


def full_convolution(a, fsize=None):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    L = 2 * n - 1
    if fsize is None:
        fsize = next_pow2(L)
    F = np.fft.rfft(a, fsize)
    c = np.fft.irfft(F * F, fsize)[:L]
    return np.maximum(c, 0.0)


def fft_and_conv(a, fsize):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    L = 2 * n - 1
    F = np.fft.rfft(a, fsize)
    c = np.fft.irfft(F * F, fsize)[:L]
    return F, np.maximum(c, 0.0)


def score_normalized(a):
    a = normalize_nonneg(a, 1.0)
    if len(a) == 0:
        return float("inf")
    b = full_convolution(a)
    return float(2.0 * len(a) * np.max(b))


def interpolate_seq(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n_old = len(a)
    n_new = max(1, int(n_new))
    if n_old == n_new:
        return normalize_nonneg(a, 1.0)
    if n_old == 0:
        return np.ones(n_new, dtype=np.float64) / n_new
    if n_old == 1:
        return np.full(n_new, float(a[0]), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n_old, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n_new, dtype=np.float64)
    y = np.interp(x_new, x_old, a)
    return normalize_nonneg(y, 1.0)


def parametric_init(n, parent=None):
    n = max(8, int(n))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    rng = np.random.default_rng(20240701)
    cands = []

    cands.append(np.ones(n, dtype=np.float64))
    for p in (0.25, 0.5, 0.8, 1.2, 2.0, 4.0):
        cands.append(np.power(x, p))
        cands.append(np.power(1.0 - x, p))
    for p in (0.3, 0.5, 1.0, 2.0, 4.0):
        cands.append(np.maximum(np.power(x, p) * np.power(1.0 - x, p), 1e-8))
        cands.append(np.maximum(1.0 - np.power(np.abs(2.0 * x - 1.0), p), 1e-6))
    for f in (0.5, 1.0, 2.0, 3.0, 4.0, 6.0):
        cands.append(1.0 + 0.7 * np.cos(2.0 * np.pi * f * x))
        cands.append(1.0 + 0.4 * np.cos(2.0 * np.pi * f * x) * np.sin(2.0 * np.pi * f * x))

    for c in (0.25, 0.35, 0.5, 0.65, 0.75):
        for sg in (0.04, 0.08, 0.12, 0.2):
            cands.append(np.exp(-((x - c) ** 2) / (2.0 * sg * sg)))
    for c1, c2, sg in ((0.25, 0.75, 0.08), (0.3, 0.7, 0.12),
                        (0.2, 0.5, 0.2, 0.08), (0.5, 0.8, 0.12)):
        if c2 < 2.0:
            cands.append(np.exp(-((x - c1) ** 2) / (2.0 * sg * sg)) +
                         np.exp(-((x - c2) ** 2) / (2.0 * sg * sg)))

    for _ in range(25):
        v = np.ones(n, dtype=np.float64)
        for m in range(1, 9):
            amp = rng.uniform(0.02, 0.14)
            ph = rng.uniform(0.0, 2.0 * np.pi)
            v += amp * np.cos(2.0 * np.pi * m * x + ph)
        cands.append(np.exp(v))

    if parent is not None:
        p = interpolate_seq(parent, n)
        cands.append(p)
        cands.append(0.7 * p + 0.3 * np.ones(n))
        cands.append(0.5 * p + 0.5 * np.ones(n))
        cands.append(0.3 * p + 0.7 * np.ones(n))

    best_a = np.ones(n) / n
    best_v = float("inf")
    for c in cands:
        c = normalize_nonneg(c, 1.0)
        try:
            v = score_normalized(c)
        except Exception:
            continue
        if np.isfinite(v) and v < best_v:
            best_v = v
            best_a = c.copy()
    return best_a, best_v


def mgd_optimize(a0, time_limit, temp=10.0, eta=0.1, max_eta=8.0,
                 verbose=True, verbose_every=30):
    n = int(np.size(a0))
    if n <= 2:
        a = np.ones(max(1, n), dtype=np.float64)
        if a.size:
            a /= a.sum()
        return a, 2.0

    a = normalize_nonneg(a0, 1.0)
    fsize = next_pow2(max(1, 3 * n - 2))
    F, B = fft_and_conv(a, fsize)
    cur_score = float(2.0 * n * B.max())
    best_a = a.copy()
    best_score = cur_score

    # If the input is exactly symmetric, try one tiny asymmetry.
    if n >= 50:
        asym = float(np.linalg.norm(a - a[::-1])) / max(1e-300, float(np.linalg.norm(a)))
        if asym < 1e-7:
            tri = np.linspace(0.0, 1.0, n, dtype=np.float64) - 0.5
            pert = normalize_nonneg(a * (1.0 + 0.01 * tri), 1.0)
            v = float(2.0 * n * full_convolution(pert).max())
            if np.isfinite(v) and v < cur_score:
                a = pert
                F, B = fft_and_conv(a, fsize)
                cur_score = v
                best_a = a.copy()
                best_score = v

    rng = np.random.default_rng(987654321 ^ n)
    t0 = time.monotonic()
    last_print = -100.0
    T = float(temp)
    eta_cur = float(eta)
    min_eta = 1e-5
    max_eta = float(max_eta)
    fail = 0
    no_best = 0
    L = 2 * n - 1

    while time.monotonic() - t0 < float(max(time_limit, 0.0)):
        elapsed = time.monotonic() - t0
        if verbose and elapsed - last_print > 10.0:
            print("    MGD n=%d t=%.1fs score=%.9f T=%.2f eta=%.5f fail=%d nobest=%d"
                  % (n, elapsed, best_score, T, eta_cur, fail, no_best))
            last_print = elapsed

        M = float(B.max())
        if M <= 1e-300 or not np.isfinite(M):
            break

        rel = (B - M) * (T / M)
        rel = np.clip(rel, -40.0, 0.0)
        w = np.exp(rel)
        ws = float(w.sum())
        if not np.isfinite(ws) or ws <= 1e-300:
            w = np.zeros(L, dtype=np.float64)
            w[int(np.argmax(B))] = 1.0
            ws = 1.0
        else:
            w = w / ws

        W = np.fft.rfft(w, fsize)
        cw = np.fft.irfft(F * W, fsize)
        wr = w[::-1]
        Wr = np.fft.rfft(wr, fsize)
        cwr = np.fft.irfft(F * Wr, fsize)

        # Gradient of sum_k w_k * (a*a)_k:
        # conv(w, a)[0:n] + reverse segment of conv(w[::-1], a).
        grad = cw[:n] + cwr[n - 1:2 * n - 1][::-1]
        grad = np.maximum(grad, 0.0)

        g = grad / M
        g = g - float(g.mean())
        rms = float(np.sqrt(np.mean(g * g)))

        if not np.isfinite(rms) or rms <= 1e-15:
            fail += 1
            if fail > 5:
                T = max(0.5, T * 0.7)
        else:
            g = g / rms
            g = np.clip(g, -20.0, 20.0)
            logfac = -eta_cur * g
            logfac = np.clip(logfac, -25.0, 25.0)
            a_try = a * np.exp(logfac)
            a_try = np.maximum(a_try, 1e-300)
            ss = float(a_try.sum())
            if not np.isfinite(ss) or ss <= 1e-300:
                a_try = a.copy()
            else:
                a_try = a_try / ss

            Ftry, Btry = fft_and_conv(a_try, fsize)
            score_try = float(2.0 * n * Btry.max())

            if not np.isfinite(score_try):
                fail += 1
                eta_cur = max(min_eta, eta_cur * 0.5)
            else:
                accept = False
                kind = 0
                # Strict monotone descent in the true max objective.
                if score_try < cur_score - max(1e-12, 1e-12 * max(1.0, cur_score)):
                    accept = True
                    kind = 1
                else:
                    wdot_old = float(np.dot(w, B))
                    wdot_try = float(np.dot(w, Btry))
                    smooth_tol = max(1e-15, 1e-9 * M)
                    if (wdot_try < wdot_old - smooth_tol and
                            score_try < cur_score + max(1e-9, 1e-7 * cur_score) and
                            score_try < best_score + 1e-6):
                        accept = True
                        kind = 2

                if accept:
                    a = a_try
                    F = Ftry
                    B = Btry
                    cur_score = score_try
                    if cur_score < best_score - 1e-13:
                        best_score = cur_score
                        best_a = a.copy()
                        no_best = 0
                    else:
                        no_best += 1

                    if kind == 1:
                        eta_cur = min(max_eta, eta_cur * 1.10)
                    else:
                        eta_cur = min(max_eta, eta_cur * 1.03)
                    fail = 0

                    w2 = float(np.dot(w, w))
                    keff = 1.0 / w2 if w2 > 1e-300 else 1e12
                    if keff < 35.0:
                        T = max(0.5, T * 0.7)
                    elif keff > 900.0 and T < 80.0:
                        T = min(120.0, T * 1.05)
                else:
                    fail += 1
                    no_best += 1
                    eta_cur = max(min_eta, eta_cur * 0.5)

        if fail > 35:
            if fail > 100:
                amp = 0.01 + 0.05 * min(1.0, no_best / 5000.0)
                a = best_a * (1.0 + amp * rng.standard_normal(n))
                a = np.maximum(a, 0.0)
                ss = float(a.sum())
                if ss > 1e-300 and np.isfinite(ss):
                    a = a / ss
                    F, B = fft_and_conv(a, fsize)
                    cur_score = float(2.0 * n * B.max())
                else:
                    a = best_a.copy()
                    F, B = fft_and_conv(a, fsize)
                    cur_score = best_score
                fail = 0
                eta_cur = max(0.02, 0.5 * float(eta))
                T = float(min(40.0, max(2.0, temp * 0.7)))
            else:
                T = float(min(120.0, max(0.5, T * 0.7)))
                fail = 0

        if no_best > 4000:
            amp = 0.02 + 0.08 * min(1.0, no_best / 30000.0)
            a = best_a * (1.0 + amp * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss > 1e-300 and np.isfinite(ss):
                a = a / ss
                F, B = fft_and_conv(a, fsize)
                cur_score = float(2.0 * n * B.max())
            else:
                a = best_a.copy()
                F, B = fft_and_conv(a, fsize)
                cur_score = best_score
            no_best = 0
            fail = 0
            eta_cur = 0.05
            T = float(max(2.0, temp * 0.8))

    # Final symmetry check.
    if n >= 2:
        sym = 0.5 * (best_a + best_a[::-1])
        sym = normalize_nonneg(sym, 1.0)
        v_sym = float(2.0 * n * full_convolution(sym).max())
        if np.isfinite(v_sym) and v_sym < best_score - 1e-12:
            best_a = sym
            best_score = v_sym

    return best_a, best_score


def lp_fw_optimize(a0, time_limit, K=120, verbose=True):
    if not _HAVE_SCIPY:
        a = normalize_nonneg(a0, 1.0)
        return a, score_normalized(a)

    a = normalize_nonneg(a0, 1.0)
    n = len(a)
    if n < 8:
        return a, score_normalized(a)

    fsize = next_pow2(max(1, 2 * n - 1))
    F, B = fft_and_conv(a, fsize)
    cur_score = float(2.0 * n * B.max())
    best_a = a.copy()
    best_score = cur_score
    best_B = B.copy()

    rng = np.random.default_rng(424242)
    t0 = time.monotonic()
    last_print = -100.0
    stall = 0
    solves = 0

    while time.monotonic() - t0 < float(max(time_limit, 0.0)):
        elapsed = time.monotonic() - t0
        if verbose and elapsed - last_print > 8.0:
            print("    LP n=%d t=%.1fs score=%.9f stall=%d solves=%d"
                  % (n, elapsed, best_score, stall, solves))
            last_print = elapsed
            if stall > 18:
                break

        M = float(B.max())
        if M <= 1e-300:
            break

        k_used = max(30, min(int(K), max(30, len(B) // 20), n))
        top = np.argsort(B)[-k_used:]
        active = sorted(int(k) for k in top)

        # Add a few safe boundary constraints so fast moving mass is controlled.
        for ex in (0, n - 1, 2 * n - 2):
            if 0 <= ex < len(B) and ex not in active:
                active.append(ex)

        Kact = len(active)
        if Kact <= 0:
            break

        rows = []
        cols = []
        data = []
        for r, k in enumerate(active):
            s = max(0, k - n + 1)
            e = min(n, k + 1)
            m = e - s
            if m <= 0:
                continue
            idx = np.arange(s, e, dtype=np.intp)
            rows.append(np.full(m, r, dtype=np.intp))
            cols.append(idx)
            data.append(np.asarray(a[k - idx], dtype=np.float64))

        rows.append(np.arange(Kact, dtype=np.intp))
        cols.append(np.full(Kact, n, dtype=np.intp))
        data.append(np.full(Kact, -1.0, dtype=np.float64))

        if not rows:
            break

        try:
            A = sp.coo_matrix((np.concatenate(data),
                               (np.concatenate(rows), np.concatenate(cols))),
                              shape=(Kact, n + 1)).tocsr()
            c_obj = np.zeros(n + 1, dtype=np.float64)
            c_obj[-1] = 1.0
            aeq_row = np.ones(n + 1, dtype=np.float64)
            aeq_row[-1] = 0.0
            Aeq = sp.csr_matrix(aeq_row.reshape(1, -1))
            b_ub = np.zeros(Kact, dtype=np.float64)
            b_eq = np.array([1.0], dtype=np.float64)
            bounds = [(0.0, None)] * n + [(None, None)]

            res = sciopt.linprog(
                c_obj,
                A_ub=A,
                b_ub=b_ub,
                A_eq=Aeq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options={"presolve": "high"}
            )
        except Exception:
            stall += 3
            if stall > 12:
                break
            k_used = max(20, k_used // 2)
            continue

        solves += 1
        if res is None or not getattr(res, "success", False):
            stall += 2
            if stall > 12:
                break
            continue

        g = np.asarray(res.x[:n], dtype=np.float64)
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        gs = float(g.sum())
        if not np.isfinite(gs) or gs <= 1e-15:
            stall += 1
            continue
        g = g / gs

        # Cheap 1D line search.
        best_alpha = 0.0
        best_val = cur_score
        best_mix = None
        for alpha in (0.0, 0.05, 0.1, 0.18, 0.3, 0.45, 0.6, 0.8, 1.0):
            if alpha == 0.0:
                val = cur_score
                mix = None
            else:
                mix = (1.0 - alpha) * a + alpha * g
                sm = float(mix.sum())
                if not np.isfinite(sm) or sm <= 1e-300:
                    continue
                mix = mix / sm
                val = float(2.0 * n * full_convolution(mix).max())
            if np.isfinite(val) and val < best_val:
                best_val = val
                best_alpha = float(alpha)
                best_mix = mix

        if best_mix is not None and best_alpha > 0.0 and best_val < cur_score - max(1e-12, 1e-10 * cur_score):
            a = normalize_nonneg(best_mix, 1.0)
            F, B = fft_and_conv(a, fsize)
            cur_score = float(2.0 * n * B.max())
            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best_a = a.copy()
                best_B = B.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            # Occasionally accept a noisy mix to escape LP stall.
            if stall > 4 and rng.random() < 0.3:
                mix = 0.9 * a + 0.1 * g
                mix = normalize_nonneg(mix, 1.0)
                val = float(2.0 * n * full_convolution(mix).max())
                if np.isfinite(val) and val < cur_score:
                    a = mix
                    F, B = fft_and_conv(a, fsize)
                    cur_score = val
                else:
                    # revert by doing nothing; just impair stall counter
                    stall += 1

    if n >= 2:
        sym = normalize_nonneg(0.5 * (best_a + best_a[::-1]), 1.0)
        v_sym = score_normalized(sym)
        if np.isfinite(v_sym) and v_sym < best_score - 1e-12:
            best_a = sym
            best_score = v_sym

    return best_a, best_score


def evaluate_safe_internal(a, scale=1.0):
    a = np.asarray(a, dtype=np.float64)
    arr = a * float(scale)
    arr = np.clip(arr, 0.0, 1000.0)
    lst = [float(x) for x in arr]
    try:
        return float(evaluate_sequence(lst))
    except Exception:
        return float("inf")


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    total_budget = min(max(60.0, float(budget_s) - 60.0), 1740.0)

    best_a = None
    best_score = float("inf")

    print("Total budget: %.1fs" % total_budget)
    print("Scipy available: %s" % str(_HAVE_SCIPY))

    # Load parent if present.
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.clip(p, 0.0, 1000.0)
        if p.size >= 8 and p.sum() > 1e-10:
            best_a = p / p.sum()
            best_score = score_normalized(best_a)
            print("Loaded parent: n=%d score=%.9f" % (len(best_a), best_score))
    except Exception:
        best_a = None
        best_score = float("inf")

    # If parent missing or poor, build a parametric start.
    if (best_a is None or not np.isfinite(best_score) or best_score > 1.8):
        n_init = 6000
        a_par, v_par = parametric_init(n_init, best_a)
        if np.isfinite(v_par) and v_par < best_score:
            best_a = a_par
            best_score = v_par
            print("Parametric init improved: n=%d score=%.9f" % (len(best_a), best_score))

    if best_a is None or not np.isfinite(best_score):
        n_fb = 10000
        best_a = np.ones(n_fb, dtype=np.float64) / n_fb
        best_score = 2.0
        print("Fallback uniform: n=%d score=%.9f" % (n_fb, best_score))

    # Quick symmetry acceptance.
    if len(best_a) >= 2:
        sym = normalize_nonneg(0.5 * (best_a + best_a[::-1]), 1.0)
        v_sym = score_normalized(sym)
        if np.isfinite(v_sym) and v_sym < best_score - 1e-12:
            best_a = sym
            best_score = v_sym
            print("Initial symmetrization improved: score=%.9f" % best_score)

    # Main working length.
    n_main = int(len(best_a))
    if n_main < 8000:
        n_main = 8000
    elif n_main > 18000:
        n_main = 18000

    # =====================================================================
    # Phase 1: small-resolution refinement / shape escape
    # =====================================================================
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 450.0:
        print("\n=== Phase 1: small-resolution MGD/LP ===")
        t_small = min(remaining * 0.10, 150.0)
        n_small = max(1200, min(3600, n_main // 3))
        a_s = interpolate_seq(best_a, n_small)
        a_s, v_s = mgd_optimize(a_s, t_small * 0.55, temp=8.0, eta=0.08,
                                max_eta=5.0, verbose=False)
        if _HAVE_SCIPY and t_small * 0.45 > 25.0:
            a_s2, v_s2 = lp_fw_optimize(a_s, t_small * 0.45, K=90, verbose=False)
            if np.isfinite(v_s2) and v_s2 <= v_s:
                a_s = a_s2
                v_s = v_s2

        up = interpolate_seq(a_s, n_main)
        v_up = score_normalized(up)
        if np.isfinite(v_up) and v_up < best_score - 1e-12:
            best_a = up
            best_score = v_up
            print("Small-phase improved: n=%d score=%.9f" % (len(best_a), best_score))
        else:
            if np.isfinite(v_s) and v_s < best_score - 1e-12:
                best_a = a_s
                best_score = v_s
                print("Small-phase direct improved: n=%d score=%.9f" % (len(best_a), best_score))

    # =====================================================================
    # Phase 2: main MGD at moderate/large length
    # =====================================================================
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 120.0:
        print("\n=== Phase 2: main MGD, n=%d ===" % n_main)
        t_main = min(remaining * 0.45, 700.0)
        a_main = interpolate_seq(best_a, n_main)
        a_out, v_out = mgd_optimize(a_main, t_main, temp=12.0, eta=0.09,
                                    max_eta=8.0, verbose=True)
        if np.isfinite(v_out) and v_out < best_score - 1e-12:
            best_a = a_out
            best_score = v_out
            print("Main MGD improved: n=%d score=%.9f" % (len(best_a), best_score))

    # =====================================================================
    # Phase 3: LP-FW at moderate length
    # =====================================================================
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 140.0 and _HAVE_SCIPY:
        print("\n=== Phase 3: LP-FW moderate ===")
        n_lp = min(n_main, 6000)
        t_lp = min(remaining * 0.20, 240.0)
        a_lp = interpolate_seq(best_a, n_lp)
        a_out_lp, v_out_lp = lp_fw_optimize(a_lp, t_lp, K=120, verbose=True)
        if np.isfinite(v_out_lp) and v_out_lp < best_score - 1e-12:
            best_a = a_out_lp
            best_score = v_out_lp
            print("LP direct improved: n=%d score=%.9f" % (len(best_a), best_score))
        # Also try upsampling the LP result.
        if n_lp != len(best_a):
            up_lp = interpolate_seq(a_out_lp, len(best_a))
            v_up_lp = score_normalized(up_lp)
            if np.isfinite(v_up_lp) and v_up_lp < best_score - 1e-12:
                best_a = up_lp
                best_score = v_up_lp
                print("LP upsample improved: n=%d score=%.9f" % (len(best_a), best_score))

    # =====================================================================
    # Phase 4: long-sequence MGD
    # =====================================================================
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 220.0:
        print("\n=== Phase 4: long MGD ===")
        n_big = max(n_main + 1500, 15500)
        n_big = min(n_big, 18000)
        a_big = interpolate_seq(best_a, n_big)
        t_big = min(remaining * 0.38, 480.0)
        a_out, v_out = mgd_optimize(a_big, t_big, temp=16.0, eta=0.07,
                                    max_eta=6.5, verbose=True)
        if np.isfinite(v_out) and v_out < best_score - 1e-12:
            best_a = a_out
            best_score = v_out
            print("Long MGD improved: n=%d score=%.9f" % (len(best_a), best_score))

    # =====================================================================
    # Phase 5: symmetry-breaking / stochastic restart
    # =====================================================================
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 220.0:
        print("\n=== Phase 5: stochastic escape ===")
        rng = np.random.default_rng(777777)
        for seed_amp in (0.02, 0.05, 0.09):
            remaining = total_budget - (time.monotonic() - t_global)
            if remaining < 80.0:
                break
            curved = best_a * (1.0 + seed_amp * rng.standard_normal(len(best_a)))
            curved = normalize_nonneg(curved, 1.0)
            t_esc = min(remaining * 0.35, 180.0)
            a_out, v_out = mgd_optimize(curved, t_esc, temp=9.0, eta=0.08,
                                        max_eta=5.0, verbose=True)
            if np.isfinite(v_out) and v_out < best_score - 1e-12:
                best_a = a_out
                best_score = v_out
                print("Stochastic escape improved: n=%d score=%.9f amp=%.2f"
                      % (len(best_a), best_score, seed_amp))
                break

    # =====================================================================
    # Phase 6: final fast polish
    # =====================================================================
    remaining = total_budget - (time.monotonic() - t_global)
    if remaining > 80.0:
        print("\n=== Phase 6: final polish ===")
        t_fin = remaining - 45.0
        a_fin = best_a
        if len(a_fin) < 8000:
            a_fin = interpolate_seq(a_fin, 8000)
        a_out, v_out = mgd_optimize(a_fin, t_fin, temp=6.0, eta=0.05,
                                    max_eta=3.0, verbose=True)
        if np.isfinite(v_out) and v_out < best_score - 1e-12:
            best_a = a_out
            best_score = v_out
            print("Final polish improved: n=%d score=%.9f" % (len(best_a), best_score))

    # Final symmetry acceptance.
    if len(best_a) >= 2:
        sym = normalize_nonneg(0.5 * (best_a + best_a[::-1]), 1.0)
        v_sym = score_normalized(sym)
        if np.isfinite(v_sym) and v_sym < best_score - 1e-12:
            best_a = sym
            best_score = v_sym
            print("Final symmetrization improved: score=%.9f" % best_score)

    best_a = normalize_nonneg(best_a, 1.0)

    # Scale to a comfortable magnitude for the external evaluator.
    n_out = len(best_a)
    scale = math.sqrt(2.0 * n_out) if n_out > 0 else 1.0
    raw = best_a * scale
    raw = np.clip(raw, 0.0, 999.0)
    ss = float(raw.sum())
    if not np.isfinite(ss) or ss <= 0.01:
        raw = best_a * max(1.0, scale)

    result = [float(x) for x in raw]

    # Validate with the provided evaluator if possible.
    ext_score = evaluate_safe_internal(raw, 1.0)
    if not np.isfinite(ext_score) or ext_score > 2.5:
        n_fb = max(1000, len(best_a))
        uni = np.ones(n_fb, dtype=np.float64) / n_fb
        scale_fb = math.sqrt(2.0 * n_fb)
        result = [float(x * scale_fb) for x in uni]
        ext_score = evaluate_safe_internal(uni, scale_fb)
        print("External validation failed, using uniform fallback.")
        print("Fallback external score: %.9f" % ext_score)

    elapsed = time.monotonic() - t_global
    print("\nFINAL internal score: %.9f, n=%d, elapsed=%.1fs" % (best_score, n_out, elapsed))
    print("FINAL external score: %.9f" % ext_score)
    return result
