    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)

                res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs',
                             options={'presolve': True, 'time_limit': 3.0,
                                      'primal feasibility tolerance': 1e-7,
                                      'dual feasibility tolerance': 1e-7})
                lp_time = time.monotonic() - lp_time_start
            except Exception:
                active_set = active_set[:max(50, K // 3)]
                iters_since_lp = 999
                continue

            if res.status not in (0, 3):
                active_set = active_set[:max(50, K // 2)]
                iters_since_lp = 999
                continue

            g_val = np.maximum(np.asarray(res.x[:n], dtype=np.float64), 0.0)
            g_sum = g_val.sum()
            if g_sum < 1e-15:
                continue
            last_lp_dir = g_val / g_sum
            iters_since_lp = 0
        else:
            last_lp_dir = last_lp_dir  # reuse

        # Line search
        best_alpha = 1.0 / (iter_count + 1.5)
        best_trial_M = M_curr

        if iter_count <= 3:
            alphas = [1.0, 0.8, 0.6, 0.4]
        else:
            alphas = [0.5, 0.3, 0.2, max(0.1, 2.0 / (iter_count + 1.0)),
                      max(0.05, 1.0 / (iter_count + 1.0))]

        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * last_lp_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 2:
            y = last_lp_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * last_lp_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y / s_y
            else:
                y = best_a.copy()

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={iter_count} ({elapsed:.1f}s): J={2.0*n*best_m:.6f}, |S|={K}")

    return best_a, best_m


def random_walk_polish(a0, time_limit, n_steps_max=50000, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    temperature = 0.002
    step_size = 0.005
    accept_count = 0
    total_count = 0
    stall = 0

    for step in range(n_steps_max):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 3000:
            # Restart with larger perturbation
            noise = rng.normal(0, 0.02, n) * best_a
            a = best_a + noise
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            temperature = 0.003
            step_size = 0.008
            stall = 0
            continue

        # Pick source (high value) and destination (low value)
        # Weight towards positions that contribute to peaks
        ac = fast_convolve(a)
        peak_k = int(np.argmax(ac))

        # Find positions contributing to the peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]
        contrib_norm = contrib / (contrib.sum() + 1e-30)

        # Source: weighted by contribution to peak
        src = int(idx[rng.choice(len(idx), p=contrib_norm)])

        # Destination: prefer positions with low total contribution
        total_contrib = np.zeros(n)
        for kk in np.argsort(ac)[::-1][:5]:
            kk = int(kk)
            s2 = max(0, kk - n + 1)
            e2 = min(n, kk + 1)
            if s2 < e2:
                idx2 = np.arange(s2, e2)
                total_contrib[idx2] += a[kk - idx2]

        total_contrib[src] = np.inf
        dest_weights = 1.0 / (total_contrib + 0.01)
        dest_weights /= dest_weights.sum()
        dst = int(rng.choice(n, p=dest_weights))

        if src == dst:
            continue

        # Move mass
        amt = a[src] * step_size * rng.uniform(0.5, 1.5)
        if amt < 1e-15:
            continue

        a_trial = a.copy()
        a_trial[src] -= amt
        a_trial[dst] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t

        M_trial = fast_convolve(a_trial).max()
        delta = M_trial - best_m

        total_count += 1
        if delta < 0 or (rng.random() < np.exp(-delta / (temperature + 1e-15))):
            a = a_trial
            accept_count += 1
            if M_trial < best_m:
                best_m = M_trial
                best_a = a.copy()
                stall = 0
                temperature = max(0.0005, temperature * 0.999)
            else:
                stall += 1
        else:
            stall += 1

        # Cool down
        temperature = max(0.0003, temperature * 0.9995)
        step_size = max(0.001, step_size * 0.9999)

        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    RW step={step} ({elapsed:.1f}s): J={2.0*n*best_m:.6f}, T={temperature:.5f}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        top_ks = np.argsort(ac)[::-1][:20]
        improved = False

        for k in top_ks[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination: minimize peak contribution
            scores = np.zeros(n)
            for k2 in top_ks[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            scores[src] = np.inf
            cand = np.where(scores < scores.max())[0]
            if len(cand) == 0:
                cand = np.arange(n)
            dst = int(cand[np.argmin(scores[cand])])
            if dst == src:
                continue

            for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    break
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Random kick
            if stall % 200 == 0:
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.005, 0.02)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            break

    return best_a, best_m


def get_initial_sequence():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                print(f"Parent: n={len(p)}, score={sc:.6f}")
                return p_norm, sc
    except Exception:
        pass

    n = 5000
    x = np.linspace(0, 1, n)
    best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
    best_score = evaluate_safe(best_seq)
    print(f"Default init: n={n}, score={best_score:.6f}")
    return best_seq, best_score


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"*** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.1f}s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Total budget: {time_budget:.1f}s")

    def remaining():
        return time_budget - (time.monotonic() - t_global)

    best_seq, best_score = get_initial_sequence()
    rng = np.random.default_rng(2024)

    # === Phase 1: Multi-start with diverse initializations (25% time) ===
    print("\n=== Phase 1: Multi-start (diverse initializations) ===")
    t1_budget = time_budget * 0.25
    t1_start = time.monotonic()

    n_base = len(best_seq)
    init_kinds = ['raised_cos', 'raised_cos4', 'blackman', 'two_bumps',
                  'three_bumps', 'spectral', 'ramp']

    candidates = [(best_seq, best_score, 'parent')]

    for kind in init_kinds:
        if time.monotonic() - t1_start > t1_budget:
            break
        for n_try in [n_base]:
            a_init = make_initialization(n_try, kind, rng)
            sc = evaluate_safe(a_init)
            if np.isfinite(sc) and sc < best_score + 0.5:
                candidates.append((a_init, sc, f'{kind}_n{n_try}'))
                print(f"  Init {kind} n={n_try}: score={sc:.6f}")

    # Run short LP-FW on top candidates
    candidates.sort(key=lambda x: x[1])
    n_candidates = min(len(candidates), 5)
    t_per_cand = (t1_budget - (time.monotonic() - t1_start)) / max(1, n_candidates)

    for i in range(n_candidates):
        if time.monotonic() - t1_start > t1_budget:
            break
        a_cand, sc_cand, label = candidates[i]
        t_avail = min(t_per_cand, remaining() * 0.3)
        if t_avail < 20:
            break
        print(f"  LP-FW on {label} (n={len(a_cand)}, budget={t_avail:.0f}s): current={sc_cand:.6f}")
        a_out, _ = lp_fw(a_cand, t_avail, initial_K=300, max_K=2000, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f'ps1_{label}', t_global)

    # === Phase 2: Long LP-FW (35% time) ===
    print(f"\n=== Phase 2: Long LP-FW (n={len(best_seq)}) ===")
    t2_budget = time_budget * 0.35
    t2_start = time.monotonic()
    if len(best_seq) <= 10000:
        a_out, _ = lp_fw(best_seq, t2_budget, initial_K=500, max_K=3500,
                         add_per_iter=40, lazy_interval=2, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, 'ps2_longlpfw', t_global)

    # Random walk polish
    rt = remaining()
    if rt > 60:
        print(f"\n  Random walk polish (budget={min(rt*0.3, 120):.0f}s)")
        t_rw = min(rt * 0.3, 120.0)
        a_out, _ = random_walk_polish(best_seq, t_rw, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, 'ps2_rw', t_global)

    # === Phase 3: Random restarts for global exploration (20% time) ===
    print(f"\n=== Phase 3: Random restarts (n={len(best_seq)}) ===")
    t3_budget = time_budget * 0.20
    t3_start = time.monotonic()
    rng3 = np.random.default_rng(777)

    n_restarts = 6
    t_per_restart = t3_budget / n_restarts

    for r in range(n_restarts):
        if time.monotonic() - t3_start > t3_budget:
            break
        t_avail = min(t_per_restart, remaining() * 0.4)
        if t_avail < 15:
            break

        # Create perturbed version
        perturbation_type = r % 3
        if perturbation_type == 0:
            # Gaussian noise
            noise = rng3.normal(0, 0.05, len(best_seq)) * best_seq
            a_pert = best_seq + noise
        elif perturbation_type == 1:
            # Frequency domain perturbation
            A = np.fft.rfft(best_seq)
            A *= (1.0 + 0.1 * rng3.standard_normal(len(A)) + 0.1j * rng3.standard_normal(len(A)))
            a_pert = np.fft.irfft(A, len(best_seq))
        else:
            # Morph with a new random initialization
            a_new = make_initialization(len(best_seq), init_kinds[r % len(init_kinds)], rng3)
            a_pert = 0.6 * best_seq + 0.4 * a_new

        a_pert = np.maximum(a_pert, 0.0)
        s_p = a_pert.sum()
        if s_p > 1e-15:
            a_pert = a_pert / s_p

        a_out, _ = lp_fw(a_pert, t_avail, initial_K=300, max_K=2500, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f'ps3_restart{r}', t_global)

    # === Phase 4: Multi-n exploration (10% time) ===
    print(f"\n=== Phase 4: Multi-n (n={len(best_seq)}) ===")
    t4_budget = time_budget * 0.10
    t4_start = time.monotonic()

    n_best = len(best_seq)
    try_ns = sorted(set([max(2000, n_best - 1000), min(12000, n_best + 1000),
                         max(2000, n_best - 2000), min(12000, n_best + 2000),
                         max(2000, n_best // 2), min(12000, n_best * 2)]))
    try_ns = [nn for nn in try_ns if 2000 <= nn <= 12000]

    for new_n in try_ns[:3]:
        rt = remaining()
        if rt < 40:
            break
        print(f"  Multi-n: n={new_n} (remaining={rt:.0f}s)")
        a_long = interpolate_seq(best_seq, new_n)
        sc_long = evaluate_safe(a_long)
        print(f"    Interpolated score: {sc_long:.6f}")

        t_avail = min((t4_budget - (time.monotonic() - t4_start)) * 0.9, 120.0)
        if t_avail < 20 or new_n > 10000:
            # Just evaluate the interpolated version
            best_seq, best_score = try_update(best_seq, best_score, a_long, f'ps4_interp_n{new_n}', t_global)
            continue

        a_out, _ = lp_fw(a_long, t_avail, initial_K=300, max_K=2000, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, f'ps4_lpfw_n{new_n}', t_global)

    # === Phase 5: Final optimization (10% time) ===
    print(f"\n=== Phase 5: Final optimization (n={len(best_seq)}) ===")
    rt = remaining()
    if rt > 30:
        # Final LP-FW
        t5a = min(rt * 0.5, 150.0)
        if len(best_seq) <= 10000:
            a_out, _ = lp_fw(best_seq, t5a, initial_K=600, max_K=4000,
                             add_per_iter=50, lazy_interval=1, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, 'ps5_final_lpfw', t_global)

        # Final coordinate polish
        rt2 = remaining()
        if rt2 > 20:
            a_out, _ = coordinate_polish(best_seq, min(rt2 * 0.6, 60.0), verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, 'ps5_coord', t_global)

        # Final random walk
        rt3 = remaining()
        if rt3 > 15:
            a_out, _ = random_walk_polish(best_seq, min(rt3 * 0.7, 40.0), n_steps_max=20000, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, 'ps5_rw', t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
