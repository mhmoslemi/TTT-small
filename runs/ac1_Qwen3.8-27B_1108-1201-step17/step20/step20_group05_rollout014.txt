def norm_sum1(a):

def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)

    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c

def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)

    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0 or not math.isfinite(se):
        e = np.zeros(size, dtype=np.float64)
        e[0] = 1.0
        se = 1.0
    p = e / se

    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    f = m + math.log(se) / beta
    return f, grad


def make_v(a, ref=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 1:
        return np.array([], dtype=np.float64), 0
    if ref is None:
        if np.any(a > 0.0):
            ref = int(np.argmax(a))
        else:
            ref = 0
    ref = int(ref)
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.where(np.isfinite(v), v, -700.0)
    return v, ref


def from_v(v, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def objgrad_v(v, beta, ref=0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    if n <= 0:
        return 1e100, np.array([], dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return 1e100, np.zeros(v.size, dtype=np.float64)

    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1

    u = np.empty(n, dtype=np.float64)
    u[ref] = 0.0
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]

    a = from_v(v, ref)
    f, ga = lse_objective(a, beta)
    if not math.isfinite(f):
        f = 1e100
        ga = np.zeros(n, dtype=np.float64)

    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)

    gv = np.empty(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]

    if not np.all(np.isfinite(gv)):
        gv = np.zeros(v.size, dtype=np.float64)
    return f, gv


def run_lbfegs(a0, beta, maxfun=400, reg_u=1e-12):
    a = norm_sum1(a0)
    n = a.size
    if n <= 1:
        return a
    v0, ref = make_v(a)
    if not SCIPY_OK or maxfun <= 0:
        return a
    try:
        options = {
            "maxfun": int(maxfun),
            "maxiter": int(maxfun + 30),
            "ftol": 1e-13,
            "gtol": 1e-9,
            "disp": False
        }
        res = scipy.optimize.minimize(
            objgrad_v, v0, args=(beta, ref),
            method="L-BFGS-B", jac=True, options=options
        )
        if res is not None and hasattr(res, "x"):
            vx = np.asarray(res.x, dtype=np.float64)
            if vx.size == v0.size and np.all(np.isfinite(vx)):
                a_out = from_v(vx, ref)
                return norm_sum1(a_out)
    except Exception as e:
        print("    L-BFGS exception: " + repr(e))
    return a


def apply_single_move(c, a, s, d, tmove):
    c_new = c.copy()
    n = a.size
    c_new[d:d + n] += 2.0 * tmove * a
    c_new[s:s + n] -= 2.0 * tmove * a
    t2 = tmove * tmove
    if 2 * d < c_new.size:
        c_new[2 * d] += t2
    if 2 * s < c_new.size:
        c_new[2 * s] += t2
    if s + d < c_new.size:
        c_new[s + d] -= 2.0 * t2
    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
    return c_new


def mirror_descent_phase(a0, budget, active_k=2000):
    """
    Accelerated mirror descent with adaptive active set.
    Uses multiplicative weight updates in log-space with Nesterov momentum.
    """
    t0 = time.monotonic()
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    size = 2 * n - 1

    # Active set indices (top-k convolution coefficients)
    active = np.argsort(c)[::-1][:min(active_k, size)]
    active = np.sort(active)

    # Momentum buffers
    v_prev = a.copy()
    momentum = np.zeros(n, dtype=np.float64)
    beta_mom = 0.9

    # Adaptive learning rate
    eta = 1.0 / math.sqrt(n)
    eta_min = 1e-8
    eta_decay = 0.995
    patience = 0
    max_patience = 80

    # Backtracking parameters
    ftol = 1e-12
    backtrack_factor = 0.5

    it = 0
    while time.monotonic() - t0 < budget - 3:
        it += 1
        time_left = budget - (time.monotonic() - t0)
        if time_left < 10:
            break

        # Recompute active set every 20 iterations or when M changes significantly
        if it % 20 == 0 or it == 1:
            c = fft_conv(a)
            M = float(np.max(c))
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
                patience = 0
            else:
                patience += 1

            # Update active set: top-k plus some margin
            margin = 1.0 + 0.02 * (it // 100)
            threshold = best_M * margin
            active = np.where(c >= threshold * 0.95)[0]
            if active.size < 10:
                active = np.argsort(c)[::-1][:min(active_k, size)]
            elif active.size > active_k:
                c_idx = c[active]
                top_in = np.argsort(c_idx)[::-1][:active_k]
                active = active[top_in]
            active = np.sort(active)

            if patience >= max_patience:
                # Perturb to escape local minimum
                rng = np.random.default_rng(42 + it)
                scale = 0.02 * (1.0 + patience * 0.1)
                pert = rng.normal(0.0, scale * 0.1, n) * a
                a = a + pert
                a = np.maximum(a, 1e-14)
                a = a / a.sum()
                v_prev = a.copy()
                momentum = np.zeros(n, dtype=np.float64)
                patience = 0

        # Compute subgradient: uniform over active set
        p_active = np.zeros(size, dtype=np.float64)
        p_active[active] = 1.0 / float(active.size)
        grad = cross_convolve_p_a(p_active, a) * 2.0
        grad = np.where(np.isfinite(grad), grad, 0.0)

        # Subtract mean for simplicity (doesn't change the update on the simplex)
        grad_mean = float(np.mean(grad))
        grad_centered = grad - grad_mean

        # Nesterov momentum in log-space
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad_centered

        # Backtracking line search for mirror descent step
        best_alpha = eta
        best_new_M = M
        for bt in range(12):
            alpha = eta * (backtrack_factor ** bt)
            if alpha < eta_min:
                break
            # Update: a_new_i = a_i * exp(-alpha * momentum_i) / Z
            log_shift = -alpha * momentum
            log_shift -= float(np.max(log_shift))
            w = np.exp(log_shift)
            a_new = a * w
            s_new = float(a_new.sum())
            if s_new < 1e-14 or not math.isfinite(s_new):
                break
            a_new = a_new / s_new

            # Evaluate new max using the rank-2 update approximation
            # For accuracy, compute full convolution periodically
            if bt == 0 and it % 5 == 0:
                c_new = fft_conv(a_new)
                new_M = float(np.max(c_new))
            else:
                # Approximate using cross-convolution
                fg = fft_conv2(a, a_new)
                c_approx = (1.0 - 1.0) * c + 2.0 * 1.0 * fg * 0.5
                # Actually just use the active set values for speed
                na_new = (a + a_new) * 0.5  # rough
                # Better: use the exact formula for the new active values
                # (a_new * a_new)_k = ((a + alpha_dir) * (a + alpha_dir))_k
                # For small alpha, approx = c_k + 2*alpha*(grad contribution)
                active_vals = 2.0 * fg[active] - c[active]  # this is (a_new*a_new)[active] approx
                # Actually fg = conv(a, a_new), so (a_new*a_new) isn't directly fg
                # Let's just do full conv for safety on active indices
                c_new_full = fft_conv(a_new)
                new_M = float(np.max(c_new_full[active]))
                # Check if new peaks appeared outside active set
                new_global_M = float(np.max(c_new_full))
                if new_global_M > new_M:
                    new_M = new_global_M

            if new_M < M - ftol and new_M < best_new_M:
                best_new_M = new_M
                best_alpha = alpha
            else:
                break

        if best_alpha >= eta_min:
            log_shift = -best_alpha * momentum
            log_shift -= float(np.max(log_shift))
            w = np.exp(log_shift)
            a = a * w
            s = float(a.sum())
            if s < 1e-14:
                a = best_a.copy()
                break
            a = a / s
            v_prev = a.copy()
            eta = eta * eta_decay
        else:
            patience += 5
            eta *= 0.5

        # Periodic full evaluation
        if it % 50 == 0:
            c = fft_conv(a)
            M = float(np.max(c))
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
                print("    MD it=" + str(it) + " M=" + repr(best_M) +
                      " score_proxy=" + repr(2.0 * n * best_M))

    # Final evaluation
    c = fft_conv(best_a)
    M = float(np.max(c))
    return best_a


def frank_wolfe_exact(a0, budget, active_k=3000):
    """
    Frank-Wolfe with exact subgradient (uniform over tight constraints).
    Solves the convex subproblem: min_{g in simplex} g^T (sum_k p_k T_k) g
    approximately via mirror descent on the subproblem.
    """
    t0 = time.monotonic()
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    size = 2 * n - 1
    t = 0
    no_improve = 0

    while time.monotonic() - t0 < budget - 5 and no_improve < 200:
        t += 1
        time_left = budget - (time.monotonic() - t0)
        if time_left < 15:
            break

        # Find tight constraints (within 1% of max)
        threshold = M * 0.99
        tight = np.where(c >= threshold)[0]
        if tight.size == 0:
            tight = np.argsort(c)[::-1][:min(10, size)]
        tight = np.sort(tight[:active_k])

        # Subgradient: uniform over tight set
        p_tight = np.zeros(size, dtype=np.float64)
        p_tight[tight] = 1.0 / float(tight.size)
        grad = cross_convolve_p_a(p_tight, a) * 2.0
        grad = np.where(np.isfinite(grad), grad, 0.0)

        # Solve subproblem: find vertex of simplex closest to -grad
        # The FW direction is the vertex e_i where grad_i is most negative
        # But for better results, use a "soft" vertex
        grad_normalized = grad - float(np.mean(grad))
        # Soft vertex: softmax of -grad with temperature
        temp = 1.0 / (float(np.std(grad_normalized)) + 1e-10)
        logits = -grad_normalized * temp
        logits -= float(np.max(logits))
        w = np.exp(logits)
        w = np.where(np.isfinite(w), w, 0.0)
        w += 1e-14
        g = w / w.sum()

        # Line search for mixing
        cg = fft_conv(g)
        fg = fft_conv2(a, g)

        best_alpha = 0.0
        best_m = M
        # Try geometric sequence of alphas
        for alpha in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            val = (1.0 - alpha) ** 2 * c + 2.0 * alpha * (1.0 - alpha) * fg + alpha ** 2 * cg
            m = float(np.max(val))
            if math.isfinite(m) and m < best_m:
                best_m = m
                best_alpha = alpha

        if best_alpha > 0 and best_m < M - 1e-15:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            a = a / ss
            c = fft_conv(a)
            M = float(np.max(c))
            no_improve = 0
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
                if t == 1 or t % 20 == 0:
                    print("    FW exact t=" + str(t) + " M=" + repr(best_M) +
                          " alpha=" + repr(best_alpha))
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                rng = np.random.default_rng(12345 + t)
                pert = rng.normal(0.0, 0.01, n) * a
                a = a + pert
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                c = fft_conv(a)
                M = float(np.max(c))
                no_improve = 0

    return best_a


def coordinate_polish(a0, budget):
    """
    Fast coordinate descent using exact rank-1 convolution updates.
    Focuses on high-pressure coordinates (those contributing to the max).
    """
    t0 = time.monotonic()
    a = norm_sum1(a0)
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    n = a.size
    size = 2 * n - 1
    rnd = 0

    while time.monotonic() - t0 < budget - 3:
        rnd += 1
        if n <= 1:
            break

        # Find top convolution indices
        top_k = min(80, size)
        top_idx = np.argsort(c)[::-1][:top_k]

        # Compute pressure (gradient contribution from top peaks)
        p_top = np.zeros(size, dtype=np.float64)
        p_top[top_idx] = 1.0 / float(top_k)
        pressure = cross_convolve_p_a(p_top, a) * 2.0
        pressure = np.where(np.isfinite(pressure), pressure, 0.0)

        # Identify source and destination candidates
        # Source: high value, high pressure (reduce these)
        # Dest: low value, low pressure (add here)
        src_score = pressure * (1.0 + float(n) * a)
        dst_score = pressure + 1e-6 * a

        n_src = min(15, n)
        n_dst = min(20, n)
        src_candidates = np.argpartition(src_score, n_src - 1)[:n_src]
        dst_candidates = np.argpartition(dst_score, n_dst - 1)[:n_dst]

        improved = False
        best_cand = None
        best_cand_M = M

        for s_idx in src_candidates:
            s = int(s_idx)
            if a[s] < 1e-13:
                continue
            for d_idx in dst_candidates:
                d = int(d_idx)
                if d == s:
                    continue
                for frac in (0.003, 0.007, 0.015, 0.03, 0.06):
                    tmove = a[s] * frac
                    if tmove <= 1e-16:
                        continue
                    tmove = min(tmove, a[s] * 0.95)
                    c_new = apply_single_move(c, a, s, d, tmove)
                    m_new = float(np.max(c_new))
                    if m_new < best_cand_M - 1e-17:
                        best_cand = (m_new, s, d, tmove)
                        best_cand_M = m_new
                        improved = True

        if not improved:
            # Try random transfers
            rng = np.random.default_rng(rnd * 7919)
            for _ in range(40):
                s = int(rng.integers(n))
                d = int(rng.integers(n))
                if s == d or a[s] < 1e-13:
                    continue
                frac = float(rng.uniform(0.001, 0.02))
                tmove = a[s] * frac
                tmove = min(tmove, a[s] * 0.9)
                if tmove <= 1e-16:
                    continue
                c_new = apply_single_move(c, a, s, d, tmove)
                m_new = float(np.max(c_new))
                if m_new < best_cand_M - 1e-17:
                    best_cand = (m_new, s, d, tmove)
                    best_cand_M = m_new
                    improved = True

        if best_cand is not None:
            m_new, s, d, tmove = best_cand
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-14:
                break
            a = a / ss
            c = fft_conv(a)
            M = float(np.max(c))
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
                if rnd <= 5 or rnd % 100 == 0:
                    print("    Coord rnd=" + str(rnd) + " M=" + repr(best_M))

            # Also try the reverse move (push back a tiny amount)
            # This helps escape narrow valleys
            if rnd % 10 == 0:
                tmove2 = min(a[d] * 0.002, tmove * 0.1)
                if tmove2 > 1e-16:
                    c_new = apply_single_move(c, a, d, s, tmove2)
                    m_new2 = float(np.max(c_new))
                    if m_new2 < M:
                        a[d] -= tmove2
                        a[s] += tmove2
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        c = fft_conv(a)
                        M = float(np.max(c))
        else:
            # Stuck: try larger perturbation
            if rnd % 50 == 0:
                rng = np.random.default_rng(99991 + rnd)
                n_pert = min(200, n // 5)
                idx = rng.choice(n, size=n_pert, replace=False)
                a[idx] *= (1.0 + rng.normal(0, 0.05, n_pert))
                a = np.maximum(a, 1e-14)
                a = a / a.sum()
                c = fft_conv(a)
                M = float(np.max(c))

    return best_a


def default_sequence(n=5000):
    x = np.linspace(0.0, 1.0, int(n))
    # Raised cosine with slight asymmetry for better initial peaks
    base = np.sin(np.pi * x) ** 2
    # Add a gentle harmonic to break symmetry
    base *= (1.0 + 0.05 * np.sin(2.0 * np.pi * x))
    return norm_sum1(base)

            b5b = min(time_left * 0.40, 150.0)
            a_fw2 = frank_wolfe_exact(work_seq, b5b, active_k=2000)
            seq_best, best_score = try_update(seq_best, best_score, a_fw2, "fw2", t_start)
            work_seq = seq_best.copy() if seq_best.size <= 8000 else interpolate_sequence(seq_best, 7000)

    # === Phase 6: Final coordinate polish ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 80:
        budget6 = min(time_left * 0.50, 120.0)
        print("\n=== Phase 6: Final polish (" + repr(budget6) + "s) ===")
        a_cp2 = coordinate_polish(work_seq, budget6)
        seq_best, best_score = try_update(seq_best, best_score, a_cp2, "final_coord", t_start)

    # === Phase 7: Length probes ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 200:
        print("\n=== Phase 7: Length probes ===")
        base_n = seq_best.size if seq_best.size <= 8000 else 7000
        for delta in [300, -300, 600, -600, 1000]:
            time_left = total_budget - (time.monotonic() - t_start)
            if time_left < 150:
                break
            target_n = int(max(2000, min(9000, base_n + delta)))
            print("  Probe n=" + str(target_n))
            a_probe = interpolate_sequence(seq_best, target_n)

            if SCIPY_OK and time_left > 200:
                a_probe = run_lbfegs(a_probe, 80.0, maxfun=150)
            seq_best, best_score = try_update(seq_best, best_score, a_probe, "probe_" + str(target_n), t_start)

            time_left = total_budget - (time.monotonic() - t_start)
            if time_left > 120:
                a_probe2 = frank_wolfe_exact(a_probe, min(time_left * 0.5, 80.0), active_k=1500)
                seq_best, best_score = try_update(seq_best, best_score, a_probe2, "probe_" + str(target_n) + "_fw", t_start)

    # Final validation
    final_val = safe_eval(seq_best)
    if math.isfinite(final_val) and final_val < best_score:
        best_score = final_val

    arr = np.asarray(seq_best, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if arr.size == 0 or s < 0.01:
        arr = default_sequence(5000)
    else:
        arr = arr / s

    elapsed = time.monotonic() - t_start
    print("\nFINAL: score=" + repr(best_score) + " n=" + str(arr.size) + " elapsed=" + repr(elapsed))

    return [float(x) for x in arr]
