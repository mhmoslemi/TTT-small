3. Structured pattern initialization (sinusoidal, alternating, power-law)

def smooth_obj_grad(a, tau):
    """Compute smoothed objective and gradient using log-sum-exp."""
    n = len(a)
    b = fast_convolve(a)
    b_s = b / tau
    b_max = b_s.max()
    exp_b = np.exp(b_s - b_max)
    log_sum_exp = b_max + np.log(exp_b.sum())
    obj = 2.0 * n * tau * log_sum_exp
    p = exp_b / exp_b.sum()
    # grad_i = sum_j p[j] * a[j-i] (for valid j-i)

    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(P * np.conj(A), fft_size)
    grad = corr[:n]
    return obj, grad

def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    mu = np.zeros(n)
    sv = np.sort(v)[::-1]
    cv = np.cumsum(sv)
    rho = np.nonzero(sv * np.arange(1, n+1) > (cv - 1))[0][-1]
    mu = (sv[rho] - 1) / (rho + 1)
    return np.maximum(v - mu, 0)

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def actual_obj(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (a.sum() ** 2)


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    sv = np.sort(v)[::-1]
    cv = np.cumsum(sv)
    cond = sv * np.arange(1, n + 1) > (cv - 1)
    rho_vals = np.nonzero(cond)[0]
    if len(rho_vals) == 0:
        return np.ones(n) / n
    rho = rho_vals[-1]
    mu = (sv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - mu, 0.0)


def smooth_obj_grad(a, tau):
    n = len(a)
    b = fast_convolve(a)
    b_s = b / tau
    b_max = b_s.max()
    exp_b = np.exp(b_s - b_max)
    log_sum_exp = b_max + np.log(exp_b.sum())
    obj = 2.0 * n * tau * log_sum_exp
    p = exp_b / exp_b.sum()
    fft_size = 1 << (len(b) + n - 1 - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(P * np.conj(A), fft_size)
    grad = corr[:n]
    return obj, grad


def smooth_grad_desc(a0, time_limit, tau_init=None, eta_init=0.05, momentum=0.85, verbose=True, seed=0):
    n = len(a0)
    x = norm_sum1(a0)
    best_x = x.copy()
    best_actual = actual_obj(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    if tau_init is None:
        b_init = fast_convolve(x)
        tau = b_init.max() / n * 5.0
    else:
        tau = tau_init

    eta = eta_init
    vel = np.zeros(n)
    stall = 0
    max_stall = 2000
    iter_count = 0
    tau_floor = b_init.max() / n * 0.3 if tau_init is None else tau * 0.05

    while time.monotonic() - t0 < time_limit:
        if stall > max_stall:
            break

        iter_count += 1
        x_look = x + momentum * vel
        x_look = simplex_project(x_look)

        obj_s, grad = smooth_obj_grad(x_look, tau)

        x_new = simplex_project(x_look - eta * grad)

        vel = momentum * vel + (x_new - x)
        x = x_new

        actual = actual_obj(x)
        if actual < best_actual - 1e-12:
            best_actual = actual
            best_x = x.copy()
            stall = 0
        else:
            stall += 1

        tau = max(tau * 0.9995, tau_floor)
        eta = max(eta * 0.99995, 1e-6)

        if stall > 500 and stall % 500 == 0:
            noise = rng.normal(0, 0.01, n) * best_x
            x_p = best_x + noise
            x_p = simplex_project(x_p)
            x = x_p
            vel = np.zeros(n)
            tau = max(tau * 1.5, tau_floor)
            eta = max(eta * 2.0, 1e-4)

        if verbose and iter_count % 2000 == 0:
            elapsed = time.monotonic() - t0
            print("    SGD iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): obj=" + repr(best_actual) + " tau=" + repr(tau))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    SGD done (" + str(round(elapsed, 1)) + "s): obj=" + repr(best_actual) + " iters=" + str(iter_count))
    return best_x, best_actual


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw(a0, time_limit, solver, initial_K=500, max_K=4000, add_per_iter=60, verbose=True):
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac_y = fast_convolve(y)
    best_m = ac_y.max()
    t0 = time.monotonic()

    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 600
    iter_count = 0
    stagnation_count = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if stagnation_count > 5:
                if verbose:
                    print("    LP-FW: persistent stagnation, stopping")
                break
            ac_curr = fast_convolve(y)
            n_pert = max(50, len(active_set) // 5)
            rng_s = np.random.default_rng(stagnation_count * 137)
            keep = list(rng_s.choice(len(active_set), size=len(active_set) - n_pert, replace=False))
            new_adds = np.argsort(ac_curr)[::-1][:n_pert * 3]
            new_set = set([active_set[i] for i in keep])
            for k in new_adds:
                new_set.add(int(k))
            active_set = sorted(new_set)
            no_improve = 0
            stagnation_count += 1
            continue

        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            stagnation_count = 0
        else:
            no_improve += 1

        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep_idx = np.argsort(margins)[:int(max_K * 0.7)]
            active_set_base = [active_set[i] for i in keep_idx]
            violations = np.where(ac > M_curr - 1e-8)[0]
            for v in violations[:150]:
                if int(v) not in active_set_base:
                    active_set_base.append(int(v))
            active_set = sorted(set(active_set_base))

        K = len(active_set)
        if K == 0:
            break
        if K > max_K:
            active_set = active_set[:max_K]
            K = max_K

        coeffs = build_constraint_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(100, K // 3)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]

        all_viol = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(all_viol) > 0:
            viol_amt = conv_ys[all_viol] - M_curr
            top_v = all_viol[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        best_alpha = 1.0 / (iter_count + 2.0)
        best_trial_M = fast_convolve(y).max()

        alphas_to_try = list(reversed([
            1.0, 0.8, 0.6, 0.4, 0.25, 0.1,
            max(0.05, 2.0 / (iter_count + 2.0)),
            max(0.02, 1.0 / (iter_count + 1.0)),
        ]))

        for alpha in alphas_to_try:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 2:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + " |S|=" + str(len(active_set)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + " iters=" + str(iter_count))
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=77, verbose=False):

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(40, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])

                a_trial = a_trial / s_t

                if M_trial < best_m - 1e-15:

                    a[src] -= amt

                    if s_t > 1e-15:

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def generate_structured_inits(n, parent=None):
    inits = []
    x = np.linspace(0, 1, n)

    inits.append(norm_sum1(np.ones(n)))

    for freq in [1, 2, 3, 5]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            pattern = np.ones(n) + amp * np.sin(2 * np.pi * freq * x)
            inits.append(norm_sum1(pattern))

    for amp in [0.2, 0.4, 0.6]:
        pattern = np.ones(n) + amp * np.cos(np.pi * np.arange(n))
        inits.append(norm_sum1(pattern))

    for p in [0.5, 1.0, 2.0, 3.0]:
        pattern = np.ones(n) + 0.5 * np.abs(2 * x - 1) ** p
        inits.append(norm_sum1(pattern))

    sigma_range = [0.1, 0.2, 0.3, 0.4]
    for sigma in sigma_range:
        pattern = np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2))
        inits.append(norm_sum1(pattern))

    if parent is not None and len(parent) > 0:
        inits.append(norm_sum1(interpolate_seq(parent, n)))

    return inits


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def get_initial_sequence(t_global):
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
                return p_norm, sc
    except Exception:
        pass

    n = 5000
    x = np.linspace(0, 1, n)
    best_seq = norm_sum1(np.ones(n) + 0.5 * np.sin(2 * np.pi * 2 * x))
    best_score = evaluate_safe(best_seq)
    print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = get_solver()
    best_seq, best_score = get_initial_sequence(t_global)
    n0 = len(best_seq)

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: Smoothed gradient descent on parent ===
    print("\n=== Phase 1: Smoothed GD (n=" + str(n0) + ") ===")
    t1 = min(remaining_time() * 0.15, 200.0)
    if t1 > 30:
        a_out, _ = smooth_grad_desc(best_seq, t1, eta_init=0.08, momentum=0.82, verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sgd1_n" + str(n0), t_global)

    # === Phase 2: LP-FW on parent ===
    print("\n=== Phase 2: LP-FW (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining_time() * 0.15, 200.0)
    if t2 > 40 and len(best_seq) <= 10000:
        a_out, _ = lp_fw(best_seq, t2, solver, initial_K=500, max_K=3500, add_per_iter=60, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1_n" + str(len(best_seq)), t_global)

    # === Phase 3: Multi-init exploration with smoothed GD ===
    print("\n=== Phase 3: Multi-init structured exploration ===")
    n_explore = min(len(best_seq), 7000)
    if remaining_time() > 200:
        inits = generate_structured_inits(n_explore, parent=best_seq)
        n_inits = min(len(inits), 12)
        time_per_init = min(remaining_time() * 0.15 / n_inits, 30.0)

        for ii in range(n_inits):
            rt = remaining_time()
            if rt < 60:
                break
            init_seq = inits[ii]
            sc = actual_obj(init_seq)
            label = "init" + str(ii) + "_n" + str(n_explore)
            if sc < best_score:
                a_out, _ = smooth_grad_desc(init_seq, time_per_init, eta_init=0.06, momentum=0.8, verbose=False, seed=100 + ii)
                best_seq, best_score = try_update(best_seq, best_score, a_out, label + "_sgd", t_global)
            else:
                a_out, _ = smooth_grad_desc(init_seq, time_per_init * 0.7, eta_init=0.06, momentum=0.8, verbose=False, seed=100 + ii)
                best_seq, best_score = try_update(best_seq, best_score, a_out, label + "_sgd", t_global)

    # === Phase 4: LP-FW second pass ===
    print("\n=== Phase 4: LP-FW second (n=" + str(len(best_seq)) + ") ===")
    t4 = min(remaining_time() * 0.15, 200.0)
    if t4 > 40 and len(best_seq) <= 10000:
        a_out, _ = lp_fw(best_seq, t4, solver, initial_K=600, max_K=4500, add_per_iter=80, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2_n" + str(len(best_seq)), t_global)

    # === Phase 5: Smoothed GD second pass (lower tau) ===
    print("\n=== Phase 5: Smoothed GD fine (n=" + str(len(best_seq)) + ") ===")
    t5 = min(remaining_time() * 0.10, 120.0)
    if t5 > 25:
        b_best = fast_convolve(best_seq)
        tau_fine = b_best.max() / len(best_seq) * 2.0
        a_out, _ = smooth_grad_desc(best_seq, t5, tau_init=tau_fine, eta_init=0.03, momentum=0.88, verbose=True, seed=200)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sgd_fine_n" + str(len(best_seq)), t_global)

    # === Phase 6: Multi-n exploration ===
    print("\n=== Phase 6: Multi-n ===")
    n_best = len(best_seq)
    try_ns = sorted(set([n_best - 500, n_best + 500, n_best + 1000, n_best - 1000]))
    try_ns = [l for l in try_ns if 3000 <= l <= 10000]
    try_ns.sort(key=lambda l: abs(l - n_best))

    for new_n in try_ns[:3]:
        rt = remaining_time()
        if rt < 150:
            break
        print("\n  Multi-n: n=" + str(new_n) + " (remaining=" + str(round(rt, 1)) + "s)")
        a_long = interpolate_seq(best_seq, new_n)

        t6 = min(rt * 0.20, 150.0)
        if t6 > 30:
            a_out, _ = smooth_grad_desc(a_long, t6 * 0.5, eta_init=0.05, momentum=0.85, verbose=False, seed=300 + new_n)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "sgd_n" + str(new_n), t_global)

        rt = remaining_time()
        if rt > 60 and new_n <= 10000:
            t6b = min(rt * 0.25, 120.0)
            if t6b > 30:
                a_out, _ = lp_fw(best_seq, t6b, solver, initial_K=400, max_K=3000, add_per_iter=50, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

    # === Phase 7: LP-FW third pass (aggressive) ===
    print("\n=== Phase 7: LP-FW aggressive (n=" + str(len(best_seq)) + ") ===")
    t7 = min(remaining_time() * 0.15, 200.0)
    if t7 > 40 and len(best_seq) <= 10000:
        a_out, _ = lp_fw(best_seq, t7, solver, initial_K=800, max_K=5000, add_per_iter=100, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw3_n" + str(len(best_seq)), t_global)

    # === Phase 8: Coordinate polish ===
    print("\n=== Phase 8: Coordinate polish (n=" + str(len(best_seq)) + ") ===")
    t8 = min(remaining_time() * 0.10, 120.0)
    if t8 > 20:
        a_out, _ = coordinate_polish(best_seq, t8, rng_seed=99, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n" + str(len(best_seq)), t_global)

    # === Phase 9: Final smoothed GD with very low tau ===
    print("\n=== Phase 9: Final fine SGD (n=" + str(len(best_seq)) + ") ===")
    t9 = min(remaining_time() * 0.08, 80.0)
    if t9 > 15:
        b_best = fast_convolve(best_seq)
        tau_final = b_best.max() / len(best_seq) * 1.0
        a_out, _ = smooth_grad_desc(best_seq, t9, tau_init=tau_final, eta_init=0.02, momentum=0.9, verbose=False, seed=500)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sgd_final_n" + str(len(best_seq)), t_global)

    # === Final coordinate polish ===
    rt = remaining_time()
    if rt > 15:
        a_out, _ = coordinate_polish(best_seq, min(rt * 0.6, 60.0), rng_seed=2024, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        best_seq = norm_sum1(np.ones(n_fb))
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
