def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(z):
    """Project onto the probability simplex."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

        return np.ones(n) / n
    rho = rho[-1]

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_grad_smoothed(y, beta, top_k=None):
    n = len(y)
    b = fast_autocorr(y)
    size = 2 * n - 1

    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()

    if b_max <= 0:
        return np.zeros(n), b

    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s

    a_rev = y[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1]
    return grad, b

def peak_directed_cd(a0, time_limit, verbose=True, seed=55):
    """
    Peak-directed coordinate descent:
    Directly target the top autocorrelation peaks by moving mass
    from positions that contribute most to the peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    peak_count = min(5, 2 * n - 1)
    step_fracs = [0.005, 0.01, 0.02, 0.05]

    while time.monotonic() - t0 < time_limit and stall < 1500:
        b = fast_autocorr(a)
        size = 2 * n - 1
        top_lags = np.argsort(b)[::-1][:peak_count]

        improved = False
        for lag in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            lag_int = int(lag)

            # Compute which pairs (i, j) with i+j=lag contribute to this peak
            i_start = max(0, lag_int - n + 1)
            i_end = min(n - 1, lag_int)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            partners = lag_int - idx
            contrib = a[idx] * a[partners]

            if contrib.max() < 1e-15:
                continue

            # Find the pair with maximum contribution
            max_ci = int(np.argmax(contrib))
            src_i = int(idx[max_ci])
            src_j = int(partners[max_ci])

            for frac in step_fracs:
                amt = a[src_i] * frac
                if amt < 1e-15:
                    continue

                # Try moving mass from src_i to several candidate destinations
                # Prefer destinations that don't contribute to the peak
                candidates = []
                # Random destinations
                for _ in range(8):
                    d = int(rng.integers(n))
                    candidates.append(d)
                # Destinations far from the peak contribution
                for d in [0, n - 1, int(n // 2)]:
                    if d not in candidates:
                        candidates.append(d)

                for dst in candidates:
                    if dst == src_i or dst == src_j:
                        continue
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 50:
                # Informed random perturbation
                k = peak_count
                for _ in range(k):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = rng.uniform(0.003, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 300 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakCD ({elapsed:.1f}s): J={J:.6f} stall={stall}")

    return best_a, best_m


def spectral_equalize(a0, time_limit, verbose=True, seed=777):
    """
    Spectral equalization: iteratively flatten the power spectrum
    to reduce autocorrelation peaks. Uses adaptive gain control.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()

    alpha = 0.03
    for it in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Target: geometric mean of non-DC power (flatter spectrum)
        non_dc = power[1:]
        if non_dc.size == 0:
            break
        target = np.exp(np.mean(np.log(np.maximum(non_dc, 1e-30))))

        A_new = A.copy()
        gains = np.ones(len(A))
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.0:
                gains[k] = 1.0 - alpha * min(ratio - 1.0, 5.0)
            elif ratio < 0.3:
                gains[k] = 1.0 + alpha * 0.5 * (0.3 - ratio)
            gains[k] = max(0.2, min(2.0, gains[k]))

        A_new[1:] *= np.sqrt(gains[1:])
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 200 == 0:
            # Random phase perturbation to escape local minima
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.03
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial

        if it % 300 == 0:
            alpha = max(0.005, alpha * 0.97)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral it={it} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.1f}s)")
    return best_seq, best_score

        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Start: n={n0}, J={best_score:.6f}")

    # Phase 1: Quick Nesterov warm-up
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== P1: Nesterov warm-up (n={n0}) ===")
    t1 = min(remaining * 0.08, 100.0)
    if t1 > 20:
        a_out, _ = nesterov_v4(norm_sum1(best_seq), t1,
                               beta_start=30.0, beta_end=500.0,
                               beta_growth=1.005, eta_init=0.6,
                               top_k_ratio=0.02, restart_every=400,
                               verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_warmup", t_global)

    # Phase 2: Main Nesterov (primary workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== P2: Nesterov main (n={n_best}) ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 50:
        a_out, _ = nesterov_v4(norm_sum1(best_seq), t2,
                               beta_start=50.0, beta_end=3000.0,
                               beta_growth=1.006, eta_init=0.5,
                               top_k_ratio=0.01, restart_every=500,
                               verbose=True, seed=123)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_main", t_global)

    # Phase 3: Peak-directed coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== P3: Peak-directed CD (n={n_best}) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        a_out, _ = peak_directed_cd(norm_sum1(best_seq), t3, verbose=True, seed=55)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakcd", t_global)

    # Phase 4: Spectral equalization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== P4: Spectral (n={n_best}) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        a_out, _ = spectral_equalize(norm_sum1(best_seq), t4, verbose=True, seed=777)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec1", t_global)

    # Phase 5: Mirror descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== P5: Mirror descent (n={n_best}) ===")
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 15:
        a_out, _ = mirror_descent_v2(norm_sum1(best_seq), t5,
                                     tau_start=0.05, tau_decay=0.9995,
                                     verbose=True, seed=200)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "mirror", t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== P6: Multi-length (base n={n_best}) ===")

    target_lengths = []
    for delta in [-500, 300, 700, 1200, 2000]:
        nl = n_best + delta
        if 2000 <= nl <= 12000:
            target_lengths.append(nl)
    # Remove duplicates and sort
    target_lengths = sorted(set(target_lengths))

    for new_n in target_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  -> n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.35, 150.0)
        if t6a > 25:
            a_out, _ = nesterov_v4(a_long, t6a,
                                    beta_start=40.0, beta_end=2000.0,
                                    beta_growth=1.005, eta_init=0.5,
                                    top_k_ratio=0.015, restart_every=500,
                                    verbose=True, seed=new_n % 7919)
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"nest_n{new_n}", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.30, 80.0)
            if t6b > 15:
                a_long2 = interpolate_sequence(best_seq, new_n)
                a_out, _ = peak_directed_cd(a_long2, t6b, verbose=False, seed=new_n % 3331)
                best_seq, best_score = try_update(best_seq, best_score, a_out, f"peakcd_n{new_n}", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6c = min(remaining * 0.25, 60.0)
            if t6c > 10:
                a_out, _ = spectral_equalize(norm_sum1(best_seq), t6c, verbose=False, seed=new_n % 4441)
                best_seq, best_score = try_update(best_seq, best_score, a_out, f"spec_n{new_n}", t_global)

    # Phase 7: Second Nesterov round with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== P7: Nesterov r2 (n={n_best}) ===")
    if remaining > 60:
        t7 = min(remaining * 0.20, 200.0)
        if t7 > 30:
            a_out, _ = nesterov_v4(norm_sum1(best_seq), t7,
                                   beta_start=100.0, beta_end=5000.0,
                                   beta_growth=1.008, eta_init=0.3,
                                   top_k_ratio=0.008, restart_every=400,
                                   verbose=True, seed=999)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)

    # Phase 8: Final polishing
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== P8: Final polish (n={n_best}) ===")
    if remaining > 30:
        t8a = min(remaining * 0.30, 100.0)
        if t8a > 15:
            a_out, _ = nesterov_v4(norm_sum1(best_seq), t8a,
                                   beta_start=200.0, beta_end=8000.0,
                                   beta_growth=1.01, eta_init=0.2,
                                   top_k_ratio=0.005, restart_every=300,
                                   verbose=True, seed=555)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8b = min(remaining * 0.30, 60.0)
            if t8b > 10:
                a_out, _ = peak_directed_cd(norm_sum1(best_seq), t8b, verbose=True, seed=654)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "peakcd_final", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t8c = min(remaining * 0.30, 50.0)
            if t8c > 10:
                a_out, _ = spectral_equalize(norm_sum1(best_seq), t8c, verbose=True, seed=888)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "spec_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: J={best_score:.6f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
