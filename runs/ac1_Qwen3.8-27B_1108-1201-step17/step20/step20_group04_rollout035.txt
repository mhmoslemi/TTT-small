    """Find direction d that reduces all active peaks."""
    n = len(a)
    # Variables: d[0], ..., d[n-1], t
    # Minimize: t (we want t to be as negative as possible)
    # Subject to:
    #   For each active peak k: sum_j A[k,j] * d[j] - t <= 0  i.e., A[k,:] @ d <= t
    #   sum d = 0
    #   d[j] >= -a[j] (for step size alpha=1, a+d >= 0)
    #   |d[j]| <= 1 (box constraint for sanity)
    
    # Actually, let's normalize differently.
    # We want to minimize the maximum increase in active peaks.
    # minimize t
    # s.t. sum_j a[k-j] * d[j] <= t for each active k
    #      sum d = 0
    #      -a[j] <= d[j] <= a[j] (symmetric box, ensures a±d >= 0)
    
    # c vector: minimize t => c = [0,...,0, 1] (t is the last variable)
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active peak k_i: sum_j a[k_i-j]*d[j] - t <= 0
    # Matrix row: [a[k_i-0], a[k_i-1], ..., a[k_i-(n-1)], -1]
    A_ub = np.zeros((len(active_peaks), n + 1))
    for row, k in enumerate(active_peaks):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[row, j] = a[idx]
        A_ub[row, n] = -1.0
    b_ub = np.zeros(len(active_peaks))
    
    # Equality constraint: sum d = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds: -a[j] <= d[j] <= 0 (we can only decrease, or use symmetric)
    # Actually, let's use -a[j] <= d[j] <= a[j] for flexibility
    bounds = [(-a[j], a[j]) for j in range(n)]
    bounds.append((-1.0, np.inf))  # t is unbounded above (we minimize it)
    
    # Solve LP
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:

def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def objective_value(a):
    """Compute 2*n*max_b for normalized sequence (sum=1)."""
    n = len(a)
    c = fast_autocorr(a)
    return 2.0 * n * c.max()


def solve_direction_lp(a, active_peaks, max_bound=None):
    """
    Solve LP to find direction d that minimizes worst-case first-order increase
    in active peaks. Returns (d, t) where t is the optimal worst-case change.
    """
    n = len(a)
    K = len(active_peaks)
    if K == 0:
        return None, 0.0

    active = np.array(active_peaks, dtype=np.int64)

    # Build A_ub: K x (n+1) matrix
    # Row i: a[k_i - j] for j=0..n-1 (0 if out of range), last col = -1
    A_ub = np.zeros((K, n + 1))
    for i in range(K):
        k = active[i]
        # Valid j: max(0, k-n+1) <= j <= min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            idx = k - np.arange(j_start, j_end + 1)
            A_ub[i, j_start:j_end + 1] = a[idx]
        A_ub[i, n] = -1.0

    b_ub = np.zeros(K)

    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # Equality: sum d = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    # Bounds: -a[j] <= d[j] <= a[j] for j=0..n-1; t in [-10, 10]
    lb = np.maximum(-a, -1.0)
    ub = np.minimum(a, 1.0)
    bounds = list(zip(lb, ub))
    bounds.append((-10.0, 10.0))

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                                  'primal_feasibility_tolerance': 1e-9})
        if result.success:
            return result.x[:n], float(result.x[n])
    except Exception:
        pass
    return None, 0.0


def solve_single_peak_lp(a, peak_k):
    """Solve LP to minimize first-order change in a single peak."""
    n = len(a)
    k = int(peak_k)
    j_start = max(0, k - n + 1)
    j_end = min(k, n - 1)

    c_obj = np.zeros(n)
    if j_start <= j_end:
        idx = k - np.arange(j_start, j_end + 1)
        c_obj[j_start:j_end + 1] = a[idx]

    # Maximize reduction: minimize -c_obj @ d
    c_obj = -c_obj

    A_eq = np.ones((1, n))
    b_eq = np.array([0.0])

    lb = np.maximum(-a, -1.0)
    ub = np.minimum(a, 1.0)
    bounds = list(zip(lb, ub))

    try:
        result = linprog(c_obj, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if result.success:
            return result.x, float(-result.fun)  # return positive reduction amount
    except Exception:
        pass
    return None, 0.0


def line_search(a, d, current_max, n_evals=12):
    """Find best alpha in [0, alpha_max] that minimizes max autocorrelation peak."""
    n = len(a)
    # alpha_max: ensure a + alpha*d >= 0
    neg_mask = d < -1e-15
    if neg_mask.any():
        alpha_max = np.min(a[neg_mask] / (-d[neg_mask]))
        alpha_max = min(alpha_max, 10.0)
    else:
        alpha_max = 10.0

    if alpha_max <= 0:
        alpha_max = 1.0

    best_alpha = 0.0
    best_val = current_max

    # Try a set of alpha values (log-spaced + some linear)
    alphas = np.unique(np.concatenate([
        np.linspace(0.01, min(alpha_max, 1.0), 8),
        [0.5, 1.0, 1.5, 2.0]
    ]))
    alphas = alphas[alphas <= alpha_max]

    for alpha in alphas:
        a_trial = a + alpha * d
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s
        val = fast_autocorr(a_trial).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    return best_alpha, best_val, a + best_alpha * d if best_alpha > 0 else a


def lp_optimize(a0, time_limit, verbose=True, seed=42):
    """Main LP-based active set optimization."""
    global _t0_global
    if _t0_global is None:
        _t0_global = time.monotonic()
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = objective_value(a)
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    stall = 0
    max_stall = 500
    K_active = 30
    strategy_cycle = 0

    while time.monotonic() - t_start < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        M = c.max()
        current_obj = 2.0 * n * M

        # Identify active peaks
        top_idx = np.argsort(c)[::-1]
        # Include peaks within 2% of max
        threshold = M * 0.98
        active_mask = c >= threshold
        active_peaks = np.where(active_mask)[0]
        if len(active_peaks) > K_active:
            active_peaks = top_idx[:K_active]
        elif len(active_peaks) < 5:
            active_peaks = top_idx[:min(10, len(top_idx))]

        # Solve LP for direction
        d, t_opt = solve_direction_lp(a, active_peaks)

        if d is not None and t_opt < -1e-10:
            # Good direction: do line search
            alpha, val_new, a_new = line_search(a, d, M)
            if val_new < M - 1e-12:
                a = norm_sum1(a_new)
                new_obj = objective_value(a)
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            # Stuck: try alternative strategies
            stall += 1
            strategy_cycle += 1

            if stall % 50 == 1:
                # Strategy 1: Try single peak optimization
                k_max = top_idx[0]
                d1, red = solve_single_peak_lp(a, k_max)
                if d1 is not None and red > 1e-10:
                    alpha, val_new, a_new = line_search(a, d1, M, n_evals=8)
                    if val_new < M - 1e-12:
                        a = norm_sum1(a_new)
                        new_obj = objective_value(a)
                        if new_obj < best_obj:
                            best_obj = new_obj
                            best_a = a.copy()
                            stall = 0
                            continue

            elif stall % 50 == 2:
                # Strategy 2: Structured perturbation (anti-symmetric)
                anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
                anti -= anti.mean()
                eps = 0.02 / (1.0 + 0.1 * (stall // 200))
                a_try = a + eps * anti * a
                a_try = np.maximum(a_try, 0.0)
                a_try = a_try / a_try.sum()
                M_try = fast_autocorr(a_try).max()
                if M_try < M:
                    a = a_try
                    new_obj = objective_value(a)
                    if new_obj < best_obj:
                        best_obj = new_obj
                        best_a = a.copy()
                    stall = 0
                    continue

            elif stall % 50 == 3:
                # Strategy 3: Random perturbation
                noise = rng.normal(0, 1, n)
                eps = 0.01 / (1.0 + 0.05 * (stall // 200))
                a_try = a + eps * noise * a
                a_try = np.maximum(a_try, 0.0)
                a_try = a_try / a_try.sum()
                M_try = fast_autocorr(a_try).max()
                if M_try < M * 1.05:  # Accept if not too bad
                    a = a_try
                    new_obj = objective_value(a)
                    if new_obj < best_obj:
                        best_obj = new_obj
                        best_a = a.copy()
                            stall = 0
                    else:
                        stall = 0  # Reset to allow exploration
                    continue

            elif stall % 50 == 4:
                # Strategy 4: SOTA-like LP (find g maximizing sum(g) s.t. a*g <= M)
                # Simplified: find direction that reduces a weighted combination
                # Use top peaks with exponential weighting
                weights = np.exp(-0.1 * np.arange(len(active_peaks)))
                # Build combined objective
                k_comb = active_peaks[0]  # Just use the top peak for simplicity
                d_combo, _ = solve_single_peak_lp(a, k_comb)
                if d_combo is not None:
                    alpha, val_new, a_new = line_search(a, d_combo, M, n_evals=6)
                    if val_new < M:
                        a = norm_sum1(a_new)
                        new_obj = objective_value(a)
                        if new_obj < best_obj:
                            best_obj = new_obj
                            best_a = a.copy()
                        stall = 0
                        continue

            else:
                # Strategy 5: Larger perturbation (restart-like)
                noise = rng.normal(0, 1, n)
                eps = 0.05
                a_try = best_a + eps * noise * best_a
                a_try = np.maximum(a_try, 1e-15)
                a_try = a_try / a_try.sum()
                a = a_try
                stall = 0

        if verbose and (stall == 0 or stall % 100 == 0):
            elapsed = time.monotonic() - t_start
            print("    LP t=" + str(stall) + " (" + str(round(elapsed, 1)) + "s): obj=" +
                  repr(round(best_obj, 6)) + ", M=" + repr(round(M, 8)))

    return best_a, best_obj / (2.0 * n)  # return (seq, max_b)


def nesterov_opt(a0, time_limit, beta_start=50.0, beta_end=5000.0,
                 beta_growth=1.008, eta_base=0.5, top_k_ratio=0.015,
                 restart_every=1500, verbose=True, seed=1):
    """Nesterov accelerated projected gradient with adaptive beta."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        c = fast_autocorr(y)
        if top_k < len(c):
            idx = np.argsort(c)[::-1][:top_k]
            c_use = np.zeros(len(c))
            c_use[idx] = c[idx]
            c_max = c[idx].max()
        else:
            c_use = c
            c_max = c.max()
        shifted = beta * (c_use - c_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        total = exp_vals.sum()
        if total < 1e-300:
            p = np.zeros_like(c_use)
            p[np.argmax(c)] = 1.0
        else:
            p = exp_vals / total

        # Gradient computation via FFT
        fft_size = 1 << (2 * n - 1).bit_length()
        A_fft = np.fft.rfft(y, fft_size)
        P_fft = np.fft.rfft(p, fft_size)
        a_rev = y[::-1]
        Conv = np.fft.irfft(P_fft * np.fft.rfft(a_rev, fft_size), fft_size)
        grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]

        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.03 / (1.0 + 0.05 * (t // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_max_b


def peak_surgery(a0, time_limit, verbose=True, seed=77):
    """Targeted local search: reduce top peaks via mass redistribution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 2000

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_idx = np.argsort(c)[::-1][:20]
        found_move = False

        for k in top_idx[:10]:
            if found_move:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            if products.max() < 1e-20:
                continue
            src_i = int(idx[np.argmax(products)])
            partner = k_int - src_i
            if partner < 0 or partner >= n:
                continue
            if src_i == partner:
                continue
            src_val = a[src_i]
            par_val = a[partner]
            reduce_i = src_i if src_val >= par_val else partner
            if a[reduce_i] < 1e-14:
                continue

            # Compute pressure
            pressure = np.zeros(n)
            for k2 in top_idx[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_idx[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)
            mask = candidates != reduce_i
            candidates = candidates[mask]
            if len(candidates) == 0:
                continue
            cand_pressure = pressure[candidates]
            n_try = min(5, len(candidates))
            try_dsts = candidates[np.argsort(cand_pressure)[:n_try]]

            for dst in try_dsts:
                dst_int = int(dst)
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                    amt = a[reduce_i] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_i] -= amt
                    a_trial[dst_int] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        no_improve = 0
                        found_move = True
                        break
                if found_move:
                    break

        if not found_move and no_improve > 200:
            # Random exploration
            for _ in range(20):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            no_improve = 0
                            break

        if verbose and no_improve % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSurgery (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve))

    return best_a, best_m


def pairwise_cd(a0, time_limit, verbose=True, seed=123):
    """Fine-grained pairwise coordinate descent for final polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        c = fast_autocorr(a)
        threshold = np.percentile(c, 95)
        peak_positions = np.where(c > threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.argsort(c)[::-1][:20]

        improved = False
        for k in peak_positions[:15]:
            if improved:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            top_pairs = np.argsort(products)[::-1][:3]
            for pi in top_pairs:
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n or partner == src_i:
                    continue
                larger_i = src_i if a[src_i] >= a[partner] else partner
                if a[larger_i] < 1e-14:
                    continue
                for frac in [0.001, 0.002, 0.003, 0.005]:
                    amt = a[larger_i] * frac
                    if amt < 1e-18:
                        continue
                    for _ in range(3):
                        dst = int(rng.integers(n))
                        if dst == larger_i:
                            continue
                        a_trial = a.copy()
                        a_trial[larger_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            stall += 1

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PairCD (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):

    n0 = len(best_seq)

    remaining = time_budget - (time.monotonic() - t_global)

    t1 = min(remaining * 0.30, 450.0)
    if t1 > 60:
        a_out, _ = lp_optimize(norm_sum1(best_seq), t1, verbose=True, seed=1)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 2: Nesterov refine
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 60:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t2,
                                beta_start=100.0, beta_end=5000.0,
                                beta_growth=1.01, eta_base=0.4,
                                top_k_ratio=0.01, restart_every=1200,
                                verbose=True, seed=2)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 3: LP again (on improved point)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP Round 2 (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 60:
        a_out, _ = lp_optimize(norm_sum1(best_seq), t3, verbose=True, seed=3)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp2", t_global)

    # Phase 4: Peak surgery
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Peak Surgery (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 30:
        a_out, _ = peak_surgery(norm_sum1(best_seq), t4, verbose=True, seed=11)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps1", t_global)

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-length ===")
    lengths_to_try = [n_best + 500, n_best + 1000, n_best + 2000]
    lengths_to_try = [l for l in lengths_to_try if l <= 10000]

    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print("  Trying n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.30, 180.0)
        if t5a > 40:
            a_out, _ = lp_optimize(a_long, t5a, verbose=False, seed=5)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_ml" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t5b = min(remaining * 0.25, 120.0)
            if t5b > 30:
                a_try2 = interpolate_sequence(best_seq, new_n)
                a_out, _ = nesterov_opt(a_try2, t5b,
                                        beta_start=50.0, beta_end=3000.0,
                                        beta_growth=1.008, eta_base=0.5,
                                        top_k_ratio=0.015, restart_every=1500,
                                        verbose=False, seed=6)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_ml" + str(new_n), t_global)

    # Phase 6: Pairwise CD for final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Pairwise CD (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        a_out, _ = pairwise_cd(norm_sum1(best_seq), t6, verbose=True, seed=55)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd1", t_global)

    # Phase 7: Final LP burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final LP (n=" + str(n_best) + ") ===")
    if remaining > 60:
        t7 = min(remaining * 0.25, 180.0)
        if t7 > 30:
            a_out, _ = lp_optimize(norm_sum1(best_seq), t7, verbose=True, seed=7)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Phase 8: Final Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final Nesterov (n=" + str(n_best) + ") ===")
    if remaining > 40:
        t8 = min(remaining * 0.30, 120.0)
        if t8 > 20:
            a_out, _ = nesterov_opt(norm_sum1(best_seq), t8,
                                    beta_start=300.0, beta_end=10000.0,
                                    beta_growth=1.015, eta_base=0.25,
                                    top_k_ratio=0.005, restart_every=800,
                                    verbose=False, seed=8)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

    # Phase 9: Final pairwise CD
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        t9 = min(remaining * 0.5, 80.0)
        if t9 > 15:
            a_out, _ = pairwise_cd(norm_sum1(best_seq), t9, verbose=False, seed=66)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd_final", t_global)

    # Phase 10: One final LP if time allows
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        t10 = min(remaining * 0.6, 60.0)
        if t10 > 20:
            a_out, _ = lp_optimize(norm_sum1(best_seq), t10, verbose=False, seed=9)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_ultimate", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1((1.0 - x) ** 1.5 + 0.1)
        best_score = evaluate_safe(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)

    return [float(x) for x in best_seq]
