def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c

def lse_topk_objective(a, beta, top_k):
    """
    Log-sum-exp of the top-k convolution values.
    Returns (value, gradient) with respect to a.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 0.0, np.array([], dtype=np.float64)
    size = 2 * n - 1
    c = fft_conv(a)
    # Find top-k indices
    if top_k is not None and top_k < size:
        top_idx = np.argsort(c)[::-1][:top_k]
        cv = c[top_idx]
    else:
        top_idx = None
        cv = c

    if cv.size == 0:
        return 0.0, np.zeros(n, dtype=np.float64)

    m = float(np.max(cv))
    shifted = beta * (cv - m)
    shifted = np.clip(shifted, -800.0, 800.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0 or not math.isfinite(se):
        return 0.0, np.zeros(n, dtype=np.float64)
    p = e / se  # probability over active peaks

    # Reconstruct full pressure vector
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    pf = np.zeros(size, dtype=np.float64)
    if top_idx is not None:
        pf[top_idx] = p
    else:
        pf[:] = p

    Pr = np.fft.rfft(pf[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    f = m + math.log(se) / beta
    if not math.isfinite(f):
        f = 1e100
    return f, grad


def objgrad_v(v, beta, top_k, ref, reg_u=1e-14):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    if n <= 0:
        return 1e100, np.array([], dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return 1e100, np.zeros(v.size, dtype=np.float64)

    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1

    u = np.empty(n, dtype=np.float64)
    u[ref] = 0.0
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]

    a = softmax_from_u(u)
    f, ga = lse_topk_objective(a, beta, top_k)
    if not math.isfinite(f):
        f = 1e100
        ga = np.zeros(n, dtype=np.float64)

    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)

    gv = np.empty(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]

    if reg_u > 0.0:
        f += 0.5 * reg_u * float(np.dot(v, v))
        gv = gv + reg_u * v

    if not np.all(np.isfinite(gv)):
        gv = np.zeros(v.size, dtype=np.float64)
    return f, gv


def run_lbfegs(a0, beta, top_k, maxfun=400, verbose=True):
    a = norm_sum1(a0)
    n = a.size
    if n <= 1:
        return a
    v0, ref = make_v(a)
    if not SCIPY_OK or maxfun <= 0:
        return a

    t0 = time.monotonic()
    options = {
        "maxfun": int(maxfun),
        "maxiter": int(maxfun + 20),
        "ftol": 1e-13,
        "gtol": 1e-8,
        "disp": False
    }
    try:
        res = scipy.optimize.minimize(
            objgrad_v, v0,
            args=(beta, top_k, ref, 1e-14),
            method="L-BFGS-B", jac=True, options=options
        )
    except Exception as e:
        print("    L-BFGS exc: " + repr(e))
        return a

    if res is not None and hasattr(res, "x"):
        vx = np.asarray(res.x, dtype=np.float64)
        if vx.size == v0.size and np.all(np.isfinite(vx)):
            a_out = from_v(vx, ref)
            a_out = norm_sum1(a_out)
            if verbose:
                print("    L-BFGS beta=" + repr(beta) +
                      " topk=" + str(top_k) +
                      " n=" + str(n) +
                      " maxfun=" + str(maxfun) +
                      " t=" + repr(time.monotonic() - t0))
            return a_out
    return a


def apply_single_move(c, a, s, d, tmove):
    """Rank-1 update of self-convolution when moving tmove from s to d."""
    c_new = c.copy()
    n = a.size
    # Linear terms
    c_new[d:d + n] += 2.0 * tmove * a
    c_new[s:s + n] -= 2.0 * tmove * a
    # Quadratic terms
    t2 = tmove * tmove
    if 2 * d < c_new.size:
        c_new[2 * d] += t2
    if 2 * s < c_new.size:
        c_new[2 * s] += t2
    if s + d < c_new.size:
        c_new[s + d] -= 2.0 * t2
    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
    return c_new


def ultrafine_coordinate_polish(a0, budget, top_k_track=5):
    """
    Very fine coordinate polish using rank-1 convolution updates.
    Tries many small transfers and accepts any that reduce the max.
    Uses pressure-based candidate selection for efficiency.
    """
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = proxy_max(c, None)
    best_a = a.copy()
    best_M = M
    stall = 0
    rnd = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit - 2:
        rnd += 1
        if n <= 1:
            break
        time_left = time_limit - (time.monotonic() - t0)
        if time_left < 5:
            break

        # Find top peaks
        top_idx = active_top(c, None, top_k_track)
        if top_idx.size == 0:
            break

        # Compute pressure: for each position i, how much does it contribute to the top peaks?
        # pressure[i] = sum_{k in top} a[k-i] (convolution contribution)
        p_top = np.zeros(2 * n - 1, dtype=np.float64)
        p_top[top_idx] = 1.0 / float(top_idx.size)

        # Cross-convolve p_top with a to get pressure at each position
        total = 3 * n - 2
        Np = next_pow2(total)
        A_f = np.fft.rfft(a, Np)
        Pr_f = np.fft.rfft(p_top[::-1], Np)
        conv_p = np.fft.irfft(A_f * Pr_f, Np)[:total]
        pressure = 2.0 * conv_p[n - 1:2 * n - 1][::-1]
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)

        # Source candidates: high pressure, currently non-zero
        src_score = pressure * (1.0 + n * a)
        src_count = min(20, n)
        src_cands = np.argpartition(src_score, min(src_count - 1, n - 1))[:src_count]

        # Destination candidates: low pressure
        dst_score = pressure
        dst_count = min(30, n)
        dst_cands = np.argpartition(dst_score, min(dst_count - 1, n - 1))[:dst_count]

        improved_any = False
        best_cand = None
        best_cand_M = M

        # Try transfers from sources to destinations
        for srr in src_cands:
            s = int(srr)
            if a[s] < 1e-14:
                continue
            for drr in dst_cands:
                d = int(drr)
                if d == s:
                    continue
                # Quick check: does this direction reduce the top peak?
                k_star = top_idx[0]
                lin_change = 2.0 * (a[int(k_star) - d] if 0 <= int(k_star) - d < n else 0.0) - \
                             2.0 * (a[int(k_star) - s] if 0 <= int(k_star) - s < n else 0.0)
                if lin_change >= 0:
                    # Won't help the primary peak; skip
                    continue
                for frac in (0.001, 0.003, 0.005, 0.01, 0.02):
                    tmove = a[s] * frac
                    if tmove <= 1e-16:
                        continue
                    tmove = min(tmove, a[s] * 0.99)
                    c_new = apply_single_move(c, a, s, d, tmove)
                    m_new = proxy_max(c_new, None)
                    if m_new < best_cand_M - 1e-18:
                        best_cand = (m_new, s, d, tmove)
                        best_cand_M = m_new
                        improved_any = True

        # Also try random small moves
        n_random = min(50, n)
        for _ in range(n_random):
            s = int(rng.integers(n))
            d = int(rng.integers(n))
            if s == d or a[s] < 1e-14:
                continue
            frac = float(rng.uniform(0.0005, 0.008))
            tmove = a[s] * frac
            tmove = min(tmove, a[s] * 0.99)
            if tmove <= 1e-16:
                continue
            c_new = apply_single_move(c, a, s, d, tmove)
            m_new = proxy_max(c_new, None)
            if m_new < best_cand_M - 1e-18:
                best_cand = (m_new, s, d, tmove)
                best_cand_M = m_new
                improved_any = True

        if best_cand is not None:
            m_new, s, d, tmove = best_cand
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = fft_conv(a)
            M = proxy_max(c, None)
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
            if rnd <= 5 or rnd % 50 == 0:
                print("    UF round=" + str(rnd) +
                      " proxy=" + repr(2.0 * n * best_M) +
                      " M=" + repr(M) +
                      " s=" + str(s) + " d=" + str(d) +
                      " t=" + repr(tmove))
        else:
            stall += 1
            if stall >= 60:
                # Perturb and reset
                pert = rng.normal(0.0, 0.002, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-12)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = proxy_max(c, None)
                stall = 0
                if rnd % 100 == 0:
                    print("    UF perturb round=" + str(rnd) +
                          " proxy=" + repr(2.0 * n * 2.0 * n * M / (1.0)**2))
            elif stall >= 120:
                break

    return best_a


def frank_wolfe_simple(a0, budget, active_max=2000):
    """Simple Frank-Wolfe for min-max(self-conv) on the simplex."""
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = proxy_max(c, None)
    best_a = a.copy()
    best_M = M
    active = active_top(c, None, active_max)
    no_improve = 0
    t = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit - 3 and no_improve < 60:
        t += 1
        if time_limit - (time.monotonic() - t0) < 15:
            break

        # Solve: min_g sum_k lambda_k * (f*g)[k] approximately
        # Using iterative mirror descent on lambda
        K = active.size
        if K == 0:
            break

        # Initialize lambda proportional to current peak values
        av = c[active]
        av = np.where(av > 0, av, 1e-300)
        lam = av / float(av.sum())

        # Iterate MW updates
        logits = np.log(av) - np.max(np.log(av))
        for _ in range(15):
            lam = np.exp(logits - np.max(logits))
            lam = lam / float(lam.sum())

            # Build pressure
            pf = np.zeros(2 * n - 1, dtype=np.float64)
            pf[active] = lam
            total = 3 * n - 2
            Np = next_pow2(total)
            A_f = np.fft.rfft(a, Np)
            Pr_f = np.fft.rfft(pf[::-1], Np)
            conv_p = np.fft.irfft(A_f * Pr_f, Np)[:total]
            pressure = 2.0 * conv_p[n - 1:2 * n - 1][::-1]
            pressure = np.where(np.isfinite(pressure) & (pressure > 0), pressure, 0.0)

            # best response: minimize pressure -> g = softmin
            pmin = float(np.min(pressure))
            ptp = float(np.max(pressure)) - pmin
            if ptp < 1e-30:
                g = np.ones(n) / n
            else:
                scaled = (pressure - pmin) / ptp
                w = np.exp(-10.0 * scaled)
                w = np.where(np.isfinite(w), w, 0.0)
                w += 1e-10
                g = w / w.sum()
                g = 0.95 * g + 0.05 * (np.ones(n) / n)
                g = g / g.sum()

            # Update logits
            h = fft_conv2(a, g)
            ha = h[active]
            ha = np.where(np.isfinite(ha), ha, 0.0)
            hmax = float(np.max(ha))
            hmean = float(np.mean(ha))
            htp = hmax - float(np.min(ha))
            if htp > 1e-30:
                eta = 8.0 / htp
                logits += eta * (ha - hmean)

        # Line search
        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        best_alpha = 0.0
        best_mit = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            one = 1.0 - alpha
            val = (one * one) * c + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
            m = float(np.max(val))
            if m < best_mit:
                best_mit = m
                best_alpha = alpha

        if best_alpha > 0 and best_mit < M - 1e-16:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            a = a / float(a.sum())
            c = fft_conv(a)
            M = proxy_max(c, None)
            no_improve = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            active = active_top(c, None, active_max)
            if t <= 3 or t % 10 == 0:
                print("    FW t=" + str(t) + " proxy=" + repr(2.0 * n * best_M))
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                pert = rng.normal(0.0, 0.005, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-12)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = proxy_max(c, None)
                active = active_top(c, None, active_max)
                no_improve = 0

    return best_a


def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    new_n = int(max(2, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)
