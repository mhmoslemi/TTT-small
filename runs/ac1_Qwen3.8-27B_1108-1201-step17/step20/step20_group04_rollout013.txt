    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_grad_at_peak(y, k):
    """Compute gradient of c_k = sum_i a_i * a[k-i] at point y."""
    n = len(y)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    for i in range(i_start, i_end + 1):
        j = k - i
        if 0 <= j < n:
            grad[i] += y[j]
            if j != i:
                grad[j] += y[i]
    return grad

def compute_grad_top_k(y, k_list):
    """Compute average gradient of top-k peaks."""
    n = len(y)
    grad = np.zeros(n)
    for k in k_list:
        grad += compute_grad_at_peak(y, int(k))
    grad /= len(k_list)
    return grad

def nesterov_opt_v2(a0, time_limit, beta_start=30.0, beta_end=5000.0,
                     beta_growth=1.005, eta_base=0.5, top_k_ratio=0.02,
                     restart_every=1500, verbose=True, seed=42):
    """Nesterov accelerated projected gradient with adaptive beta (improved)."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(5, int(n * top_k_ratio))
    restart_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        c = fast_autocorr(y)
        # Use top-k for gradient
        if top_k < len(c):
            idx = np.argsort(c)[::-1][:top_k]
            c_use = np.zeros(len(c))
            c_use[idx] = c[idx]
            c_max = c[idx].max()
        else:
            c_use = c
            c_max = c.max()

        shifted = beta * (c_use - c_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        total = exp_vals.sum()
        if total < 1e-300:
            p = np.zeros_like(c_use)
            p[np.argmax(c)] = 1.0
        else:
            p = exp_vals / total

        # Gradient via FFT
        fft_size = 1 << (2 * n - 1).bit_length()
        P = np.fft.rfft(p, fft_size)
        a_rev = y[::-1]
        A_rev = np.fft.rfft(a_rev, fft_size)
        Conv = np.fft.irfft(P * A_rev, fft_size)
        grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]

        # Adaptive step size
        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.05)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 30 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            restart_count += 1
            # More diverse restarts
            noise_scale = 0.05 / (1.0 + 0.1 * restart_count)
            if restart_count % 3 == 0:
                # Anti-symmetric
                anti = (np.arange(n) - (n - 1) / 2.0) / n
                anti -= anti.mean()
                pert = noise_scale * anti * best_a
            elif restart_count % 3 == 1:
                # Random
                rand = rng.normal(0, 1, n)
                rand -= rand.mean()
                pert = noise_scale * rand * best_a
            else:
                # Block-wise perturbation
                block_size = max(1, n // 20)
                pert = np.zeros(n)
                for b in range(0, n, block_size):
                    e = min(b + block_size, n)
                    scale = rng.uniform(-1, 1) * noise_scale
                    pert[b:e] = scale * best_a[b:e]
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    NestV2 restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 8)))

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    NestV2 t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 8)) + ", beta=" + str(round(beta, 1)))

    return best_a, best_max_b

def lp_direction(a, time_limit, n_constraints=30, verbose=True):
    """
    LP-based direction finding from the SOTA approach.
    Solve: max sum(g) s.t. g >= 0, (f*g)_k <= M for top K peaks.
    """
    n = len(a)
    a = norm_sum1(a)
    M = fast_autocorr(a).max()
    
    # Find top peaks
    c = fast_autocorr(a)
    top_k = np.argsort(c)[::-1][:n_constraints]
    
    # Build constraint matrix: A_lp @ g <= M
    # (f*g)_k = sum_i f_i * g_{k-i} for valid i
    # For each peak k, the constraint is: sum_{i} a[i] * g[k-i] <= M
    # where k-i must be in [0, n-1]
    
    # This is a matrix A where A[k_idx, j] = a[k-j] if k-j is valid, else 0
    # We want A @ g <= M (vector of M's)
    
    # Build A efficiently
    A_rows = []
    for k in top_k:
        k_int = int(k)
        row = np.zeros(n)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        for i in range(i_start, i_end + 1):
            j = k_int - i
            if 0 <= j < n:
                row[j] = a[i]
        A_rows.append(row)
    
    A_mat = np.array(A_rows)
    b_vec = np.full(n_constraints, M)
    
    # Solve: max 1^T g s.t. A_mat @ g <= b_vec, g >= 0
    # Convert to min -1^T g
    try:
        from scipy.optimize import linprog
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_vec, bounds=[(0, None)] * n, method='highs')
        if res.success:
            g = res.x
            s = g.sum()
            if s > 1e-10:
                g_norm = g / s
                return g_norm
        except Exception:
            pass
    
    return None

def peak_balancing(a0, time_limit, verbose=True, seed=99):
    """
    Iterative peak balancing: try to make top peaks equal by redistributing mass.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 2000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-16:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find top 3 peaks
        top_idx = np.argsort(c)[::-1][:3]
        
        improved = False
        for k in top_idx:
            if improved:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            
            if products.max() < 1e-20:
                continue
            
            # Top contributing pairs
            top_pairs = np.argsort(products)[::-1][:5]
            
            for pi in top_pairs:
                if improved:
                    break
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n:
                    continue
                
                # Reduce the larger of the pair
                if a[src_i] >= a[partner]:
                    reduce_i = src_i
                else:
                    reduce_i = partner
                
                if a[reduce_i] < 1e-14:
                    continue
                
                # Find safe destinations: positions that don't contribute to top peaks
                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_idx:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2+1] = True
                
                candidates = np.where(~in_peak)[0]
                if len(candidates) < 10:
                    candidates = np.arange(n)
                
                # Remove reduce_i from candidates
                candidates = candidates[candidates != reduce_i]
                if len(candidates) == 0:
                    continue
                
                # Try the least "pressured" candidates
                # Pressure: how much does adding mass at position j increase the top peaks?
                pressure = np.zeros(len(candidates))
                for ci, cand in enumerate(candidates):
                    for k2 in top_idx:
                        k2i = int(k2)
                        j = int(cand)
                        partner2 = k2i - j
                        if 0 <= partner2 < n:
                            pressure[ci] += a[partner2]
                
                n_try = min(5, len(candidates))
                try_dsts = candidates[np.argsort(pressure)[:n_try]]
                
                for dst in try_dsts:
                    if improved:
                        break
                    dst_int = int(dst)
                    for frac in [0.001, 0.002, 0.005, 0.01, 0.02]:
                        amt = a[reduce_i] * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[reduce_i] -= amt
                        a_trial[dst_int] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
        
        if not improved:
            # Random kicks when stuck
            if no_improve > 200:
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                break
        
        if verbose and no_improve % 500 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakBal (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 8)) + " stall=" + str(no_improve))

    return best_a, best_m

def randomized_local_search(a0, time_limit, verbose=True, seed=200):
    """
    Randomized local search with diverse move types.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 3000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-16:
            best_m = c_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        top_idx = np.argsort(c)[::-1][:min(20, len(c))]
        
        # Move type 1: directed transfer from peak contributors
        if rng.random() < 0.5:
            k = int(top_idx[rng.integers(min(5, len(top_idx)))])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k - idx]
            if products.max() > 1e-20:
                src_i = int(idx[np.argmax(products)])
                partner = k - src_i
                if 0 <= partner < n:
                    reduce_i = src_i if a[src_i] >= a[partner] else partner
                    if a[reduce_i] > 1e-12:
                        dst = int(rng.integers(n))
                        if dst != reduce_i:
                            for frac in [0.001, 0.003, 0.01, 0.03]:
                                amt = a[reduce_i] * frac
                                a_trial = a.copy()
                                a_trial[reduce_i] -= amt
                                a_trial[dst] += amt
                                a_trial = np.maximum(a_trial, 0.0)
                                s_t = a_trial.sum()
                                if s_t > 1e-15:
                                    a_trial = a_trial / s_t
                                    M_t = fast_autocorr(a_trial).max()
                                    if M_t < best_m - 1e-17:
                                        a = a_trial
                                        best_m = M_t
                                        best_a = a.copy()
                                        stall = 0
                                        break
        
        # Move type 2: random pair transfer
        elif rng.random() < 0.8:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.0005, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial
