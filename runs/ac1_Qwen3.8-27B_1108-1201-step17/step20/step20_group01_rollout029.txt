Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe for coarse optimization
2. Exact pairwise coordinate descent with bisection for fine polishing
3. Multi-length exploration with interpolation
Key innovation: pairwise CD uses the exact quadratic form of the autocorrelation

import numpy as np
import time
import cvxpy as cp

_module_solver = None

def get_solver():
    global _module_solver
    if _module_solver is not None:
        return _module_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _module_solver = s
                return s
        except Exception:
            continue
    _module_solver = cp.ECOS
    return _module_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_matrix(a, active_list, n):
    K = len(active_list)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj in range(K):
        ki = int(active_list[jj])
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[ki - idx]
    return coeffs

def lp_fw(a0, time_limit, solver, init_K=200, max_K=4000, verbose=True):
    """LP-based Frank-Wolfe with adaptive active set."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:init_K])
    no_improve = 0
    stall_limit = 3000

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t % 20 == 0 or t == 1:
            top_lags = np.argsort(ac)[::-1]
            target_size = min(max_K, init_K + t * 5)
            for k in top_lags[:target_size]:
                active_set.add(int(k))
            if len(active_set) > max_K:
                clist = sorted(active_set)
                margins = np.array([M_curr - ac[k] for k in clist])
                keep = np.argsort(margins)[:max_K]
                active_set = set(int(clist[i]) for i in keep)
        
        active_set.add(int(np.argmax(ac)))
        clist = sorted(active_set)
        K = len(clist)
        if K == 0:
            break

        try:
            coeffs = build_lp_matrix(y, clist, n)
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            prob.solve(solver=solver, verbose=False)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                raise Exception("LP failed")
        except Exception:
            if len(active_set) > init_K // 2:
                clist = sorted(active_set)
                active_set = set(int(clist[i]) for i in np.argsort([M_curr - ac[k] for k in clist])[:init_K])
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                active_set.add(int(k))

        # Line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial = 1e10
        for alpha in [best_alpha, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            Mt = fast_autocorr(trial / ts).max()
            if Mt < best_trial:
                best_trial = Mt
                best_alpha = alpha

        # Bisection refinement
        lo = max(0.0, best_alpha - 0.2)
        hi = min(1.0, best_alpha + 0.2)
        for _ in range(8):
            mid1 = lo + (hi - lo) / 3.0
            mid2 = hi - (hi - lo) / 3.0
            t1 = np.maximum((1 - mid1) * y + mid1 * g_dir, 0.0)
            t2 = np.maximum((1 - mid2) * y + mid2 * g_dir, 0.0)
            s1, s2 = t1.sum(), t2.sum()
            m1 = fast_autocorr(t1 / s1).max() if s1 > 1e-15 else 1e10
            m2 = fast_autocorr(t2 / s2).max() if s2 > 1e-15 else 1e10
            if m1 < m2:
                hi = mid2
            else:
                lo = mid1
        
        best_alpha = (lo + hi) / 2.0

        if t <= 2:
            y = g_dir.copy()
        else:
            y = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy

        if verbose and t % 100 == 0:
            print("  LP-FW t={} ({:.1f}s) J={:.7f} |S|={}".format(t, time.monotonic() - t0, 2 * n * best_m, len(active_set)))

    return best_a, 2.0 * n * best_m

def pairwise_cd(a0, time_limit, verbose=True):
    """Exact pairwise coordinate descent with incremental autocorrelation updates."""
    n = len(a0)
    a = normalize(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    size = 2 * n - 1
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        # Identify source and target coordinates
        # Sources: coordinates with high values (contribute to peaks)
        # Targets: coordinates with low values
        
        # Weight sources by their contribution to the peak lags
        peak_lags = np.argsort(ac)[::-1][:20]
        
        # Compute "pressure" on each coordinate
        pressure = np.zeros(n, dtype=np.float64)
        for k in peak_lags:
            ki = int(k)
            ws = max(0, ki - n + 1)
            we = min(n - 1, ki)
            if ws <= we:
                idx = np.arange(ws, we + 1)
                # pressure[idx] += a[idx] * a[ki - idx]
                pressure[idx] += a[ki - idx]  # marginal contribution
        
        # Sources: high a[i] * pressure[i]
        source_score = a * pressure
        top_sources = np.argsort(source_score)[::-1][:50]
        
        # Targets: low a[i] (and low pressure)
        target_score = -a - 0.1 * pressure
        top_targets = np.argsort(target_score)[::-1][:50]
        
        improved = False
        best_delta_this_round = 0.0
        
        for si in top_sources:
            if time.monotonic() - t0 > time_limit:
                break
            si = int(si)
            if a[si] < 1e-12:
                continue
            
            for ti in top_targets:
                if time.monotonic() - t0 > time_limit:
                    break
                ti = int(ti)
                if ti == si:
                    continue
                if a[ti] > a[si]:  # only move from source to target if target has less
                    continue
                
                # Compute d[k] = a[k-si] - a[k-ti] (with zero-padding)
                d = np.zeros(size, dtype=np.float64)
                # a[k-si] is valid when 0 <= k-si < n, i.e., si <= k < si+n
                k_range_si = slice(si, si + n)
                d[k_range_si] += a[:n]  # a[k-si] for k in [si, si+n-1]
                # a[k-ti] is valid when 0 <= k-ti < n, i.e., ti <= k < ti+n
                k_range_ti = slice(ti, ti + n)
                d[k_range_ti] -= a[:n]  # a[k-ti] for k in [ti, ti+n-1]
                
                # q[k] = delta_{k,2si} - 2*delta_{k,si+ti} + delta_{k,2ti}
                q = np.zeros(size, dtype=np.float64)
                if 2 * si < size:
                    q[2 * si] += 1.0
                if si + ti < size:
                    q[si + ti] -= 2.0
                if 2 * ti < size:
                    q[2 * ti] += 1.0
                
                # Now find delta that minimizes max_k (ac[k] + 2*delta*d[k] + delta^2*q[k])
                # Subject to -a[si] += delta >= 0 and a[ti] - delta >= 0
                # i.e., -a[si] <= delta <= a[ti]
                lo = -a[si]
                hi = a[ti]
                
                if hi - lo < 1e-15:
                    continue
                
                # Bisection / ternary search
                # The function is approximately convex (max of quadratics that are mostly linear)
                # Use ternary search
                f_lo = ac.max()  # at delta=0
                for _ in range(50):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    val1 = np.max(ac + 2.0 * m1 * d + m1 * m1 * q)
                    val2 = np.max(ac + 2.0 * m2 * d + m2 * m2 * q)
                    if val1 < val2:
                        hi = m2
                    else:
                        lo = m1
                
                best_delta = (lo + hi) / 2.0
                best_val = np.max(ac + 2.0 * best_delta * d + best_delta * best_delta * q)
                
                if best_val < best_m - 1e-14:
                    # Apply the change
                    a[si] += best_delta
                    a[ti] -= best_delta
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s < 1e-15:
                        break
                    a = a / s  # re-normalize (should be ~1 already)
                    ac = fast_autocorr(a)  # recompute to be safe
                    best_m = ac.max()
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break  # restart the target loop with updated a
                    
            if improved:
                break  # restart with updated pressure
        
        if not improved:
            stall += 1
            if stall > 500:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    ac = fast_autocorr(a)
                    if ac.max() < best_m:
                        best_m = ac.max()
                        best_a = a.copy()
                        stall = 0
            elif stall > 100 and stall % 100 == 0:
                if verbose:
                    print("  PairCD stall={} ({:.1f}s) J={:.7f}".format(stall, time.monotonic() - t0, 2*n*best_m))
        
        if verbose and improved and stall == 0:
            pass  # could log here
    
    return best_a, 2.0 * n * best_m

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def eval_safe(a):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_matrix(a, active_list, n):
    K = len(active_list)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj in range(K):
        ki = int(active_list[jj])
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[ki - idx]
    return coeffs

def lp_fw(a0, time_limit, solver, init_K=200, max_K=4000, verbose=True):

        clist = sorted(active_set)
        K = len(clist)

        g_sum = g_val.sum()
        if g_sum < 1e-15:

        g_dir = g_val / g_sum

            if Mt < best_trial:
