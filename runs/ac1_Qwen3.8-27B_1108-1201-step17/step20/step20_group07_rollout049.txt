        if i_start <= i_end:

def greedy_fw_direction(a, b, top_k, max_iters=50):
    """Solve the Frank-Wolfe subproblem greedily:
    min_g max_k (A@g)[k] where A[k,j] = a[k-j] (linearized at current a)
    Returns a direction g on the simplex.
    """

    idx = np.argsort(b)[::-1][:top_k]

    g = np.ones(n) / n  # start uniform

        gs = g.sum()
        if gs > 1e-15:

    except Exception:
        return float('inf')


def main_optimizer(a0, time_limit, verbose=True):
    """Main projected gradient optimizer with multiple strategies."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    beta = 8.0
    momentum = 0.75
    eta = 0.15
    top_k = 40
    v = np.zeros(n)
    stall = 0
    t = 0
    fw_counter = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(a)
        m_curr = b.max()

        if m_curr < best_m - 1e-16:
            best_m = m_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Compute smoothed gradient
        grad = compute_gradient_topk(a, b, top_k, beta)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            break
        grad = grad / grad_norm

        # Sometimes use FW direction instead of gradient
        use_fw = (t % 30 == 0 and stall > 20)
        if use_fw:
            fw_counter += 1
            direction = greedy_fw_direction(a, b, top_k, max_iters=30)
            direction = direction - a
            dnorm = np.linalg.norm(direction)
            if dnorm > 1e-15:
                direction = direction / dnorm
                # Try FW direction
                best_trial = None
                best_trial_m = m_curr
                for step in [eta, eta * 0.5, eta * 0.25]:
                    a_trial = a + step * direction
                    a_trial = simplex_project(a_trial)
                    b_trial = fast_autocorr(a_trial)
                    mt = b_trial.max()
                    if mt < best_trial_m - 1e-16:
                        best_trial = a_trial
                        best_trial_m = mt
                if best_trial is not None:
                    a = best_trial
                    continue

        # Momentum update
        v = momentum * v - eta * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-18:
            v = -grad
        else:
            v = v / vnorm

        # Backtracking line search
        best_trial = None
        best_trial_m = m_curr
        for step_mult in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            step = eta * step_mult
            a_trial = a + step * v
            a_trial = simplex_project(a_trial)
            b_trial = fast_autocorr(a_trial)
            mt = b_trial.max()
            if mt < best_trial_m - 1e-16:
                best_trial = a_trial
                best_trial_m = mt

        if best_trial is not None:
            a = best_trial
        else:
            # Stalled - try perturbations
            if stall > 50:
                # Graduated restart
                if stall > 200:
                    scale = 0.05
                else:
                    scale = 0.02
                noise = rng.normal(0, scale, n) * best_a
                a_r = best_a + noise
                a_r = np.maximum(a_r, 1e-12)
                a_r = a_r / a_r.sum()
                m_r = fast_autocorr(a_r).max()
                if m_r < best_m:
                    a = a_r
                    best_m = m_r
                    best_a = a_r.copy()
                    stall = 0
                else:
                    a = best_a.copy()
                    v = np.zeros(n)
                    eta *= 0.9
                    stall = 0
            elif stall > 20:
                # Small random kick
                j1 = int(rng.integers(n))
                j2 = int(rng.integers(n))
                if j1 != j2 and a[j1] > 1e-10:
                    amt = a[j1] * 0.01
                    a[j1] -= amt
                    a[j2] += amt
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s

        # Adaptive parameters
        if t % 30 == 0:
            beta = min(beta * 1.03, 300.0)
        if t % 50 == 0:
            momentum = min(0.85, 0.7 + 0.1 * min(1.0, t / 500.0))
        if stall > 0 and stall % 30 == 0:
            eta = max(eta * 0.95, 0.005)
        if stall == 0:
            eta = min(eta * 1.02, 0.3)

        # Increase top_k as beta increases (more peaks matter)
        top_k = min(100, 30 + t // 100)

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  Opt t={t} ({elapsed:.1f}s): J={J:.6f} beta={beta:.1f} eta={eta:.4f} stall={stall}")

    return best_a, best_m


def peak_shaving_fast(a0, time_limit, verbose=True):
    """Directly target top peaks and redistribute mass."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        b = fast_autocorr(a)
        n_peaks = min(15, len(b))
        peak_indices = np.argsort(b)[::-1][:n_peaks]

        improved = False
        for k_int in peak_indices[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_i = int(k_int)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_i - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            partner = k_i - src
            if partner < 0 or partner >= n:
                continue

            # Reduce the larger of src, partner
            if a[src] >= a[partner]:
                reduce_idx = src
            else:
                reduce_idx = partner

            if a[reduce_idx] < 1e-14:
                continue

            # Find safe destination: minimize contribution to all top peaks
            pressure = np.zeros(n)
            for k2 in peak_indices[:6]:
                k2i = int(k2)
                p = k2i - reduce_idx  # not used, computing for dst
            # For potential destinations, compute how much they'd contribute
            # Use a quick heuristic: prefer positions with low a value not in peak regions
            dangerous = set()
            for k2 in peak_indices[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                dangerous.update(range(s2, e2 + 1))

            safe_candidates = [i for i in range(n) if i not in dangerous and i != reduce_idx]
            if not safe_candidates:
                safe_candidates = [i for i in range(n) if i != reduce_idx]

            # Pick destination with lowest a value among safe candidates
            dst = min(safe_candidates, key=lambda i: a[i])

            for frac in [0.01, 0.03, 0.06, 0.1, 0.15]:
                amt = a[reduce_idx] * frac
                if amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall % 30 == 0:
                scale = min(0.04, 0.005 * stall / 30)
                pert = rng.normal(0, scale, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-12)
                a_r = a_r / a_r.sum()
                m_r = fast_autocorr(a_r).max()
                if m_r < best_m:
                    best_m = m_r
                    best_a = a_r.copy()
                a = a_r if m_r < best_m + 0.005 else best_a.copy()

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  PeakShave ({elapsed:.1f}s): J={J:.6f} stall={stall}")

    return best_a, best_m


def diverse_initialization(n, rng):
    """Generate diverse initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    init_type = rng.integers(8)

    if init_type == 0:
        # Asymmetric power law
        alpha = rng.uniform(0.5, 3.0)
        beta_p = rng.uniform(0.5, 3.0)
        a_raw = (x + 0.01) ** alpha * (1.0 - x + 0.01) ** beta_p
    elif init_type == 1:
        # Gaussian mixture
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        s1 = rng.uniform(0.05, 0.2)
        s2 = rng.uniform(0.05, 0.2)
        a_raw = np.exp(-((x - c1) ** 2) / (2 * s1 ** 2)) + rng.uniform(0.5, 1.5) * np.exp(-((x - c2) ** 2) / (2 * s2 ** 2))
    elif init_type == 2:
        # Modulated
        freq = rng.uniform(2, 10)
        a_raw = (1.0 + np.cos(2 * np.pi * freq * x)) * (x + 0.01) ** 0.5 * (1.0 - x + 0.01) ** 0.5
    elif init_type == 3:
        # Sigmoid-like
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.1, 0.3)
        a_raw = 1.0 / (1.0 + np.exp((x - center) / width))
    elif init_type == 4:
        # Random smooth
        a_raw = rng.random(n)
        # Smooth with a kernel
        kernel_size = max(5, n // 100)
        kernel = np.ones(kernel_size) / kernel_size
        a_raw = np.convolve(a_raw, kernel, mode='same')
    elif init_type == 5:
        # Bimodal with gap
        c1 = rng.uniform(0.1, 0.3)
        c2 = rng.uniform(0.7, 0.9)
        s = rng.uniform(0.05, 0.1)
        a_raw = np.exp(-((x - c1) ** 2) / (2 * s ** 2)) + np.exp(-((x - c2) ** 2) / (2 * s ** 2))
    elif init_type == 6:
        # Power law with oscillation
        alpha = rng.uniform(0.3, 2.0)
        freq = rng.uniform(1, 5)
        a_raw = (x + 0.01) ** alpha * (1.0 + 0.3 * np.sin(2 * np.pi * freq * x))
    else:
        # Pure random
        a_raw = rng.random(n) ** rng.uniform(0.5, 3.0)

    a_raw = np.maximum(a_raw, 0.0)
    return norm_sum1(a_raw)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global, verbose=True):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        if verbose:
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.1f}s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search function with multi-phase diverse optimization."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    best_seq = None
    best_score = float('inf')
    rng = np.random.default_rng(2024)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    n0 = len(best_seq) if best_seq is not None else 6700

    # Phase 0: Quick diverse initialization to find better starting point
    print("\n=== Phase 0: Diverse initialization ===")
    t_phase = time.monotonic()
    time_limit_0 = min(30.0, time_budget * 0.02)
    for i in range(20):
        if time.monotonic() - t_phase > time_limit_0:
            break
        a_init = diverse_initialization(n0, rng)
        m = fast_autocorr(a_init).max()
        J = 2.0 * n0 * m
        if np.isfinite(J) and J < best_score:
            best_score = J
            best_seq = a_init.copy()
            print(f"  Init {i}: J={J:.6f}")

    # Phase 1: Peak shaving on current best
    print(f"\n=== Phase 1: Peak shaving (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    if t1 > 20:
        a_out, _ = peak_shaving_fast(best_seq, t1, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "PS1", t_global)

    # Phase 2: Main gradient optimization
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Gradient optimization (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 60:
        a_out, _ = main_optimizer(best_seq, t2, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "Grad1", t_global)

    # Phase 3: Peak shaving again
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak shaving 2 (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 20:
        a_out, _ = peak_shaving_fast(best_seq, t3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "PS2", t_global)

    # Phase 4: Second gradient pass with different params
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Gradient optimization 2 (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 60:
        a_out, _ = main_optimizer(best_seq, t4, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "Grad2", t_global)

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    nearby_lengths = []
    for dn in [-800, -400, -200, 200, 400, 800, 1200]:
        nl = n_best + dn
        if 4000 < nl <= 10000 and nl not in nearby_lengths:
            nearby_lengths.append(nl)

    print(f"\n=== Phase 5: Multi-length exploration ===")
    for new_n in nearby_lengths[:5]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Exploring n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.2, 80.0)
        if t5a > 15:
            a_out, _ = peak_shaving_fast(a_long, min(t5a * 0.5, 40.0), verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"PS_n{new_n}", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        t5b = min(remaining * 0.3, 150.0)
        if t5b > 30:
            a_out, _ = main_optimizer(best_seq, t5b, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"Grad_n{new_n}", t_global)

    # Phase 6: Diverse restarts at best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Diverse restarts (n={n_best}) ===")
    for i in range(5):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        # Perturb the best solution significantly
        noise = rng.normal(0, 0.05, n_best) * best_seq
        a_restart = best_seq + noise
        a_restart = np.maximum(a_restart, 1e-12)
        a_restart = a_restart / a_restart.sum()

        t6 = min(remaining * 0.25, 120.0)
        if t6 > 30:
            a_out, _ = main_optimizer(a_restart, t6, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"Restart{i}", t_global)

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}) ===")
    if remaining > 40:
        t7a = min(remaining * 0.35, 120.0)
        if t7a > 20:
            a_out, _ = peak_shaving_fast(best_seq, t7a, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "FinalPS", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        t7b = min(remaining * 0.5, 150.0)
        if t7b > 30:
            a_out, _ = main_optimizer(best_seq, t7b, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "FinalGrad", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:
        print("\nERROR: No valid sequence, using fallback")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        a_raw = (x + 0.01) ** 1.0 * (1.0 - x + 0.01) ** 1.0
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
