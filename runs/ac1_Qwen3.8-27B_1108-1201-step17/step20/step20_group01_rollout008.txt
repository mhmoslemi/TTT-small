The LP in the FW step is:
minimize M
subject to:
  sum(g) = 1
  g >= 0
  C @ g <= M

where C is the active-set matrix (K x n).

This can be rewritten as:
minimize M
subject to:
  sum(g) = 1
  g - M * e >= -C @ g ... no, this doesn't simplify nicely.

Actually, I can eliminate M:
M >= C@g and M >= 0
So minimize max(0, C@g)
subject to sum(g) = 1, g >= 0

This is a min-max problem. Can be solved by bisection on M:
For a given M, check if there exists g with sum(g)=1, g>=0, C@g <= M.
This is a feasibility LP: sum(g)=1, g>=0, C@g <= M.

The feasibility LP has n+K+1 constraints and n variables. With scipy's linprog, this should be fast for moderate n and K.

Actually, there's an even simpler approach. The FW direction for our problem:
- We want to find g (sum=1, g>=0) that minimizes the increase in max(c)
- First-order: minimize sum_k alpha_k * (c_a * g)[k] where alpha_k are the active weights
- (c_a * g)[k] = sum_i a[k-i] * g[i]
- So the objective is sum_k alpha_k * sum_i a[k-i] * g[i] = sum_i g[i] * sum_k alpha_k * a[k-i]
- This is linear in g!
- So g = e_{argmin_i} (sum_k alpha_k * a[k-i])

The FW direction is just the coordinate with minimum "pressure"! That's much simpler.

But this ignores the constraint that we shouldn't increase the inactive lags. The LP version adds those constraints.

For a simpler approach, we can:
1. Compute pressure p[i] = sum_{k in top_lags} w_k * a[k-i]
2. FW direction: g = e_{argmin p} (or smooth version)
3. Update: y <- (1-t)*y + t*g_dir, normalize
4. Check if inactive lags are violated; if so, add them to the set

This is the "vanilla" FW step without LP. It's fast but might not be as effective.

For better performance, we can use a "soft" FW direction:
g_dir[i] = exp(-p[i] / tau) / Z
This spreads the direction over multiple low-pressure coordinates.

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def pressure_vec(a, top_lags, weights, n):
    """Compute pressure vector: p[i] = sum_k w_k * a[k-i] for valid indices."""
    p = np.zeros(n, dtype=np.float64)
    for idx, k in enumerate(top_lags):
        ki = int(k)
        w = weights[idx]
        ws = max(0, ki - (n - 1))
        we = min(n - 1, ki)
        if ws > we:
            continue
        i_range = np.arange(ws, we + 1)
        a_range = a[ki - i_range]
        p[i_range] += w * a_range
    return p

def softmax_subgrad(a0, time_limit, K=100, tau0=50.0, tau_decay=0.999, lr0=0.1, seed=123):
    """FFT-accelerated softmax subgradient descent with multiplicative updates."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    best_val = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    tau = tau0
    lr = lr0
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for t in range(1, 5000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        if stall > 5000:
            break
        
        ac = fast_autocorr(x)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr
        
        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_m = M_curr
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        k_sel = min(K, len(ac))
        idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = ac[idx]
        
        # Softmax weights with temperature
        w = np.exp((top_vals - top_vals[0]) / tau)
        w = w / w.sum()
        
        # Compute pressure vector
        p = pressure_vec(x, idx, w, n)
        
        # Multiplicative update (mirror descent on simplex)
        lr_adj = lr / (1.0 + 0.001 * t)
        log_x = np.log(x + 1e-30) - lr_adj * p / (p.max() + 1e-30)
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        x_new = np.maximum(x_new, 1e-25)
        x = x_new / x_new.sum()
        
        tau *= tau_decay
        tau = max(tau, 1.0)
        
        # Periodic restart
        if stall > 2000 and stall % 500 == 0:
            noise = 0.01 * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x = x_r / x_r.sum()
            stall = 0
            tau = tau0 * 0.5
        
        if t % 2000 == 0:
            print("    SoftSubgrad t={} ({:.1f}s) J={:.10f} stall={} tau={:.1f}".format(
                t, elapsed, best_val, stall, tau))
    
    return best_a, best_val

def soft_fw(a0, time_limit, K_init=100, K_max=2000, lr0=0.5, seed=42):
    """Soft Frank-Wolfe with adaptive active set."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    best_val = 2.0 * n * best_m

    rng = np.random.default_rng(seed)
    stall = 0
    size = 2 * n - 1

    for t in range(1, 3000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

            break
        
        ac = fast_autocorr(y)

        cur_val = 2.0 * n * M_curr

            stall = 0

        k_sel = min(K_init + t, K_max, len(ac))
        idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = ac[idx]
        
        # Uniform weights for simplicity
        w = np.ones(len(idx)) / len(idx)
        
        # Pressure
        p = pressure_vec(y, idx, w, n)
        
        # Soft FW direction: g[i] proportional to exp(-p[i]/tau)
        tau_fw = max(0.01, 1.0 / (t + 1))
        g_dir = np.exp(-p / (tau_fw + 1e-10))
        g_dir = np.maximum(g_dir, 0)
        gs = g_dir.sum()
        if gs < 1e-15:
            g_dir = np.ones(n) / n
        else:
            g_dir = g_dir / gs
        
        # Line search (simple)
        alphas = [0.5, 0.3, 1.0/(t+1), 0.7, 1.0]
        best_alpha = 0.5

        for alpha in alphas:

            if ts < 1e-15:
                continue

            if Mt < best_trial:
                best_trial = Mt
                best_alpha = alpha
        
        y = (1 - best_alpha) * y + best_alpha * g_dir
        y = np.maximum(y, 0)

        if sy > 1e-15:
            y = y / sy
        
        # Check violations
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        if t % 50 == 0:
            violations = np.where(conv_ys > M_curr + 1e-10)[0]
            # Add to active set by just increasing k_sel
        
        if t % 2000 == 0:
            print("    SoftFW t={} ({:.1f}s) J={:.10f} stall={}".format(
                t, elapsed, best_val, stall))
    
    return best_a, best_val

def fine_coord_desc(a0, time_limit, n_targets=8, n_candidates=50, seed=77):
    """Fine coordinate descent: move mass from high-pressure to low-pressure positions."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_val = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    fracs = [0.0001, 0.0003, 0.001, 0.003, 0.01, 0.03]
    
    while time.monotonic() - t0 < time_limit:
        if stall > 5000:
            break
        
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_targets]
        improved = False
        
        # Compute pressure from top lags
        w = np.ones(len(top_k)) / len(top_k)
        p = pressure_vec(a, top_k, w, n)
        
        # Find top sources (high pressure) and candidates for destinations
        src_candidates = np.argsort(p)[::-1][:n_candidates]
        dst_pool = np.argsort(p)[:n_candidates * 3]
        
        for si in range(min(n_candidates, len(src_candidates))):
            src = int(src_candidates[si])
            if src < 0 or src >= n or a[src] < 1e-15:
                continue
            
            # Find best destination
            best_move = None
            best_move_val = best_m
            
            for di in range(min(len(dst_pool), n_candidates * 3)):
                dst = int(dst_pool[di])
                if dst == src:
                    continue
                
                for frac in fracs:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        break
                    
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_move_val - 1e-16:
                        best_move_val = Mt
                        best_move = (src, dst, frac, a_t)
                # Early exit if we found a good move
                if best_move is not None and best_move_val < best_m - 0.01 * best_m:
                    break
            
            if best_move is not None:
                src, dst, frac, a_t = best_move
                a = a_t.copy()
                if Mt < best_m - 1e-16:
                    best_m = Mt
                    best_a = a.copy()
                    best_val = 2.0 * n * best_m
                    improved = True
                    stall = 0
        
        if not improved:
            stall += 1
            # Random perturbation when stuck
            if stall > 500:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            best_val = 2.0 * n * Mn
                            stall = 0
        
        if stall % 1000 == 0 and stall > 0:
            print("    CoordDesc ({:.1f}s) J={:.10f} stall={}".format(
                time.monotonic() - t0, best_val, stall))
    
    return best_a, best_val

def sinusoidal_perturb(a0, time_limit, max_freq=200, eps_range=(1e-5, 1e-3), seed=55):
    """Add small sinusoidal perturbations and keep improvements."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_val = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    freqs = list(range(1, max_freq + 1))
    rng.shuffle(freqs)
    
    for freq in freqs:
        if time.monotonic() - t0 > time_limit:
            break
        
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n) / n
        pert = np.sin(2 * np.pi * freq * x + phase)
        
        for eps in eps_range:
            for sign in [1, -1]:
                a_t = a + sign * eps * pert * a.max()
                a_t = np.maximum(a_t, 0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                Mt = fast_autocorr(a_t).max()
                if Mt < best_m - 1e-16:
                    best_m = Mt
                    best_a = a_t.copy()
                    best_val = 2.0 * n * best_m
                    a = a_t.copy()
                    break
            if a is not a_t:  # updated
                break
    
    return best_a, best_val

import numpy as np
import time
import math


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def compute_pressure(a, top_lags, weights, n):
    p = np.zeros(n, dtype=np.float64)
    for idx in range(len(top_lags)):
        ki = int(top_lags[idx])
        w = float(weights[idx])
        ws = max(0, ki - (n - 1))
        we = min(n - 1, ki)
        if ws > we:
            continue
        i_range = np.arange(ws, we + 1)
        p[i_range] += w * a[ki - i_range]
    return p


def mirror_descent(a0, time_limit, K=100, tau0=30.0, tau_decay=0.9995, lr0=0.05, seed=123):
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    tau = tau0
    lr = lr0

    for t in range(1, 5000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit or stall > 5000:
            break

        ac = fast_autocorr(x)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr

        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        k_sel = min(K, len(ac))
        idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = ac[idx]

        w = np.exp((top_vals - top_vals[0]) / tau)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_sel) / k_sel
        else:
            w = w / w_sum

        p = compute_pressure(x, idx, w, n)
        p_max = p.max()
        if p_max < 1e-30:
            break

        lr_adj = lr / (1.0 + 0.0005 * t)
        log_x = np.log(x + 1e-30) - lr_adj * p / (p_max + 1e-30)
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        x_new = np.maximum(x_new, 1e-25)
        x = x_new / x_new.sum()

        tau *= tau_decay
        tau = max(tau, 0.5)

        if stall > 1500 and stall % 300 == 0:
            noise = 0.005 * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x = x_r / x_r.sum()
            stall = 0
            tau = max(tau0 * 0.5, 5.0)

        if t % 3000 == 0:
            print("    MirrorDesc t={:5d} ({:.1f}s) J={:.10f} stall={}".format(
                t, elapsed, best_val, stall))

    return best_a, best_val

                continue
            Mt = fast_autocorr(trial / ts).max()
            if Mt < best_trial:
                best_trial = Mt
                best_alpha = alpha

        y = (1.0 - best_alpha) * y + best_alpha * g_dir
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        if t % 3000 == 0:
            print("    SoftFW t={:5d} ({:.1f}s) J={:.10f} stall={}".format(
                t, elapsed, best_val, stall))

    return best_a, best_val


def coord_desc(a0, time_limit, n_targets=10, n_src=30, n_dst=60, seed=77):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_val = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    fracs = [0.0001, 0.0003, 0.001, 0.003, 0.01, 0.03, 0.05]

    while time.monotonic() - t0 < time_limit and stall < 3000:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_targets]
        w = np.ones(len(top_k)) / len(top_k)
        p = compute_pressure(a, top_k, w, n)

        src_sorted = np.argsort(p)[::-1][:n_src]
        dst_sorted = np.argsort(p)[:n_dst]

        improved = False

        for si in range(len(src_sorted)):
            src = int(src_sorted[si])
            if a[src] < 1e-15:
                continue

            found = False
            for di in range(len(dst_sorted)):
                dst = int(dst_sorted[di])
                if dst == src:
                    continue

                for frac in fracs:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-16:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        best_val = 2.0 * n * best_m
                        improved = True
                        found = True
                        break
                if found:
                    break

        if not improved:
            stall += 1
            if stall > 200:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            best_val = 2.0 * n * Mn
                            stall = 0
                            improved = True

        if stall % 500 == 0 and stall > 0:
            print("    CoordDesc ({:.1f}s) J={:.10f} stall={}".format(
                time.monotonic() - t0, best_val, stall))

    return best_a, best_val


def freq_perturb(a0, time_limit, max_freq=150, seed=55):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_val = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64) / n
    eps_vals = [5e-5, 1e-4, 2e-4, 5e-4, 1e-3, 2e-3]

    freqs = list(range(1, max_freq + 1))
    rng.shuffle(freqs)

    for freq in freqs:
        if time.monotonic() - t0 > time_limit:
            break

        phase = rng.uniform(0, 2 * np.pi)
        pert = np.sin(2 * np.pi * freq * x + phase)
        amp = a.max()

        updated = False
        for eps in eps_vals:
            for sign in [1.0, -1.0]:
                a_t = a + sign * eps * pert * amp
                a_t = np.maximum(a_t, 0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t / st
                Mt = fast_autocorr(a_t).max()
                if Mt < best_m - 1e-16:
                    best_m = Mt
                    best_a = a_t.copy()
                    best_val = 2.0 * n * best_m
                    a = a_t.copy()
                    updated = True
                    break
            if updated:
                break

    return best_a, best_val


def micro_refine(a0, time_limit, n_top=5, seed=999):
    """Very fine precision refinement: tiny coordinate adjustments."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_val = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    micro_fracs = [1e-5, 3e-5, 1e-4, 3e-4, 1e-3]

    while time.monotonic() - t0 < time_limit and stall < 5000:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]
        w = np.ones(len(top_k)) / len(top_k)
        p = compute_pressure(a, top_k, w, n)

        src_sorted = np.argsort(p)[::-1][:20]
        dst_sorted = np.argsort(p)[:40]

        improved = False
        for si in range(len(src_sorted)):
            src = int(src_sorted[si])
            if a[src] < 1e-15:
                continue
            for di in range(len(dst_sorted)):
                dst = int(dst_sorted[di])
                if dst == src:
                    continue
                for frac in micro_fracs:
                    amt = a[src] * frac
                    if amt < 1e-20:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0)
                    a_t = a_t / a_t.sum()
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-17:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        best_val = 2.0 * n * best_m
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 500 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(1e-5, 1e-3)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    Mn = fast_autocorr(a).max()
                    if Mn < best_m:
                        best_m = Mn
                        best_a = a.copy()
                        best_val = 2.0 * n * Mn
                        stall = 0

        if stall % 1000 == 0 and stall > 0:
            print("    MicroRefine ({:.1f}s) J={:.10f} stall={}".format(
                time.monotonic() - t0, best_val, stall))

    return best_a, best_val
