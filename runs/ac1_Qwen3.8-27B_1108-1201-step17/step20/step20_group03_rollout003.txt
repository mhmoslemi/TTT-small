- Actually, a[i] appears in b[k] for k = i (with a[0]), k = i+1 (with a[1]), ..., up to k = i+n-1 (with a[n-1]), and also from the "other side": k = 2n-2-i (with a[n-1]), etc.

Wait, let me be precise. b[k] = sum_j a[j] * a[k-j] (with valid indices).
a[i] appears in b[k] when j=i (then we need k-i to be a valid index: 0 <= k-i < n, so i <= k <= i+n-1) OR when k-j=i (j=k-i, need 0 <= k-i < n: i-n+1 <= k <= i... wait this is the same condition).

Actually: b[k] = sum_{j: max(0,k-n+1) <= j <= min(k,n-1)} a[j] * a[k-j].
a[i] appears when j=i: requires max(0,k-n+1) <= i <= min(k,n-1), i.e., max(0, k-n+1) <= i AND i <= k AND i <= n-1 (latter always true). So k >= i AND k <= i+n-1.
OR when k-j=i, j=k-i: requires max(0,k-n+1) <= k-i <= min(k,n-1). k-i >= 0 means k >= i. k-i <= n-1 means k <= i+n-1. Also k-i >= 0 (same as k>=i) and k-i <= k (always true if i>=0). And k-i <= n-1 means k <= i+n-1. Also need k-i >= 0 which is k >= i.

So a[i] appears in b[k] for k in [i, i+n-1] (from the j=i term) AND for k in [i, i+n-1] (from the k-j=i term, which is the same range but different pairs).

Wait, I think I'm double-counting. Let me just note: a[i] * a[k-j] appears in b[k] for specific j. If j=i, the term is a[i]*a[k-i] (valid when 0<=k-i<n). If k-j=i, i.e., j=k-i, the term is a[k-i]*a[i] (valid when 0<=k-i<n). These are the same term! (a[i]*a[k-i] = a[k-i]*a[i]). So a[i] appears in b[k] for k in [i, i+n-1] with value a[i]*a[k-i] (counted once, or twice if i = k-i, i.e., k=2i).

This is getting complicated. For coordinate descent, I'd need to compute the change in max(b) when a single element changes, which is expensive.

A simpler local search: swap small amounts of mass between two positions and check if max(b) decreases. This is what the 2-exchange does.

"""

        except Exception:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_grad_fista(a, beta, top_p=50):
    """Compute gradient of smoothed max of autocorrelation using top-p softmax."""
    n = len(a)
    b = fast_autocorr(a)
    # Use top_p largest values for the gradient
    if top_p < len(b):
        idx = np.argpartition(b, -top_p)[-top_p:]
        top_vals = b[idx]
        top_max = top_vals.max()
        # Softmax weights on top values
        shifted = beta * (top_vals - top_max)
        shifted = np.clip(shifted, -50, 0)
        w = np.exp(shifted)
        w /= w.sum()
        # Build sparse weight vector for b
        b_w = np.zeros(len(b))
        b_w[idx] = w * beta / top_p  # Scale to approximate max gradient
    else:
        b_max = b.max()
        shifted = beta * (b - b_max)
        shifted = np.clip(shifted, -50, 0)
        w = np.exp(shifted)
        w /= w.sum()
        b_w = w * beta
    
    # Gradient: da/d(b_k) involves a at specific positions
    # b_k = sum_i a[i]*a[k-i], so db_k/da[j] = a[k-j] + (k-2j >= 0 and k-2j < n ? a[k-j] : 0)
    # Actually db_k/da[j] = a[k-j] if 0<=k-j<n, plus a[k-j] if k-j != j (i.e., counted twice)
    # Wait: b_k = sum_i a[i]*a[k-i]. db_k/da[j] = a[k-j] (from i=j term) + a[k-j] (from k-i=j, i.e., i=k-j term)
    # But if k-j = j (i.e., k=2j), it's only counted once: db_k/da[j] = a[j]
    # So db_k/da[j] = 2*a[k-j] if k != 2j, else a[j]
    # More precisely: db_k/da[j] = a[k-j] + (a[k-j] if k-j != j else 0) = 2*a[k-j] - a[j]*delta(k,2j)
    
    # For efficiency, compute gradient via FFT:
    # grad[j] = sum_k b_w[k] * (2*a[k-j] - a[j]*delta(k,2j)) for valid k-j
    # The "2*a[k-j]" part is a convolution of b_w with a (reversed)
    # The "-a[j]*delta(k,2j)" part: for each j, the term is -a[j]*b_w[2*j] (if 2j < len(b))
    
    conv_size = len(b_w) + n - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    
    # Part 1: 2 * (b_w * a_reversed) [cross-correlation]
    a_rev = a[::-1]
    Bw = np.fft.rfft(b_w, fft_size)
    Ar = np.fft.rfft(a_rev, fft_size)
    conv1 = np.fft.irfft(Bw * Ar, fft_size)[:conv_size]
    
    # The convolution b_w * a_rev gives: sum_k b_w[k] * a_rev[k-? ]
    # Actually: (b_w * a_rev)[m] = sum_k b_w[k] * a_rev[m-k] = sum_k b_w[k] * a[n-1-(m-k)]
    # We want: sum_k b_w[k] * a[k-j] for j = 0,...,n-1
    # a[k-j] = a_rev[n-1-(k-j)] = a_rev[n-1-k+j]
    # So sum_k b_w[k] * a_rev[n-1-k+j] = (b_w * a_rev)[j + n - 1] (reversed convolution)
    # (b_w * a_rev)[m] = sum_k b_w[k] * a_rev[m-k]
    # Set m-k = n-1-k+j => m = n-1+j
    # So the part we want is conv1[n-1 : 2*n-1]
    
    grad = 2.0 * conv1[n-1:2*n-1]
    
    # Part 2: subtract a[j]*b_w[2*j]
    diag_idx = 2 * np.arange(n)
    diag_idx = diag_idx[diag_idx < len(b_w)]
    if len(diag_idx) > 0:
        diag_j = diag_idx // 2
        grad[diag_j] -= a[diag_j] * b_w[diag_idx]
    
    return grad, b


def fista_adaptive(a0, time_limit, beta_start=20, beta_end=2000, 
                   beta_growth=1.004, top_p=30, restart_every=3000,
                   verbose=True):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(42)
    
    # Backtracking parameters
    eta = 1.0 / (n * 0.1)  # Initial step size
    L = 1.0  # Lipschitz estimate
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        grad, b = compute_grad_fista(y, beta, top_p)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-20:
            break
        
        # Backtracking line search
        for _bt in range(20):
            z = y - eta * grad
            x_trial = simplex_project(z)
            b_trial = fast_autocorr(x_trial)
            m_trial = b_trial.max()
            # Check descent: m_trial < m_current - 0.5*eta*||grad||^2 (approx)
            m_curr = b.max()
            if m_trial < m_curr + 1e-14 or _bt == 19:
                break
            eta *= 0.5
            L *= 2.0
        
        x_new = x_trial
        b_new = fast_autocorr(x_new)
        m_new = b_new.max()
        
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Increase eta if we used full step
        if _bt == 0 or m_trial < m_curr - 1e-14:
            eta *= 1.5
            L = max(L * 0.7, 1e-10)
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            # Asymmetric restart
            noise_level = 0.02 / (1.0 + 0.1 * (t // restart_every))
            shift_amount = rng.uniform(0.01, 0.05)
            x_r = best_x.copy()
            # Shift mass from left to right or vice versa
            half = n // 2
            if rng.random() < 0.5:
                x_r[:half] *= (1 - shift_amount)
                x_r[half:] *= (1 + shift_amount * half / (n-half))
            else:
                x_r[half:] *= (1 - shift_amount)
                x_r[:half] *= (1 + shift_amount * (n-half) / half)
            pert = rng.normal(0, noise_level, n) * best_x
            x_r = x_r + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 10.0)
            eta = 1.0 / (n * 0.1)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    FISTA restart t={t} ({elapsed:.1f}s): J={J:.6f}")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA t={t} ({elapsed:.1f}s): J={J:.6f} beta={beta:.1f} eta={eta:.2e}")
    
    J = 2.0 * n * best_m
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    FISTA done ({elapsed:.1f}s): J={J:.6f}")
    return best_x, best_m


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2000, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 500
    
    for t in range(1, 3000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-6)[0]
        if len(violations) > 0:
            for k in violations[:20]:
                active_set.add(int(k))
        
        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.05, default_alpha * 0.5),
                  max(0.05, default_alpha * 2.0), 0.3, 0.5, 1.0]
        alphas = [max(0.01, min(1.0, a)) for a in set(alphas)]
        
        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        if t <= 3:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y > 1e-15:
                y_new = y_new / s_y
        
        y = y_new
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f} |S|={len(active_set)}")
    
    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=True):
    """Directly target and reduce the top peaks in the convolution."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_autocorr(a)
        # Get top 20 peaks
        top_idx = np.argsort(b)[::-1][:20]
        improved = False
        
        for k in top_idx:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            total = contrib.sum()
            if total < 1e-20:
                continue
            
            # Find the top contributing pair
            top_contrib = np.argsort(contrib)[::-1][:10]
            
            for ci in top_contrib:
                src = int(idx[ci])
                partner = k_int - src
                if partner < 0 or partner >= n or partner == src:
                    continue
                if a[src] < 1e-10:
                    continue
                
                # Try moving mass from src to a random other position
                for frac in [0.001, 0.005, 0.01, 0.02]:
                    amt = a[src] * frac
                    # Find a good destination: one that doesn't create new peaks
                    # Sample a few candidates
                    for _ in range(5):
                        dst = int(rng.integers(n))
                        if dst == src or dst == partner:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        m_trial = fast_autocorr(a_trial).max()
                        if m_trial < best_m - 1e-14:
                            a = a_trial
                            best_m = m_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = max(1, n // 100)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        m_new = fast_autocorr(a).max()
                        if m_new < best_m:
                            best_m = m_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave ({elapsed:.1f}s): J={J:.6f} stall={stall}")
    
    return best_a, best_m


def cyclic_shift_best(a0):
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a0, fft_size)
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent: optimize one element at a time."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    
    b = fast_autocorr(a)
    
    for outer in range(20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved_any = False
        # Process elements in random order
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                    break
            
            if a[i] < 1e-12:
                continue
            
            # What's the optimal value for a[i]?
            # b[k] for k in [i, i+n-1] depends on a[i]:
            # b[k] = a[i]*a[k-i] + (other terms not involving a[i])
            # We want to minimize max(b) by adjusting a[i]
            # But a[i] also appears in b[k] for k in [2n-2-i, ...] from the "other side"
            # Actually for linear conv: b[k] includes a[i]*a[k-i] when valid
            # a[i] appears in b[k] for k = i,...,i+n-1 (as a[i]*a[k-i])
            # AND for k = 2n-2-i,...,2n-2 (as a[k-(n-1-i)]*a[n-1-i]... )
            # This is getting complicated. Let's just do a local search.
            
            # Try a few candidate values for a[i]
            orig = a[i]
            candidates = [orig * 0.5, orig * 0.75, orig * 1.25, orig * 1.5,
                         orig + a.mean() * 0.1, orig - a.mean() * 0.1]
            candidates = [max(0, c) for c in candidates]
            
            best_c = orig
            best_local_m = b.max()
            
            for c in candidates:
                if c == orig:
                    continue
                delta = c - orig
                a[i] = c
                # Renormalize
                s = a.sum()
                if s < 1e-15:
                    a[i] = orig
                    continue
                a /= s
                m_local = fast_autocorr(a).max()
                if m_local < best_local_m:
                    best_local_m = m_local
                    best_c = c
                a = a.sum() * a  # undo normalization temporarily
                a[i] = orig  # restore
