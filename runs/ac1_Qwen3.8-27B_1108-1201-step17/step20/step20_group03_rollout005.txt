def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))

def norm_sum1(a):
    """Normalize to sum 1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    """Project onto probability simplex."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

import numpy as np

import math

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_grad_smooth(y, beta, top_k):
    """Compute smoothed gradient of max autocorrelation."""
    n = len(y)
    b = fast_autocorr(y)
    b_max = b.max()
    if top_k is not None and top_k < len(b) and top_k > 0:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
    else:
        b_use = b.copy()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        mask = (b_use == b_max)
        p = np.zeros_like(b_use)
        p[mask] = 1.0 / max(1, mask.sum())
    else:
        p = exp_vals / p_total
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, b, b_max

def fista_enhanced(a0, time_limit, beta_start=10.0, beta_end=5000.0,
                   top_k=15, restart_every=3000, verbose=True, seed=42):
    """Enhanced FISTA with Polyak step size and staircase beta."""
    n = len(a0)
    x = norm_sum1(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    best_val = 2.0 * n * best_max_b
    t_start = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    step_size = 1.0 / (n ** 0.5)
    val_smooth = best_val
    nesterov_avg = x.copy()
    avg_w = 0.0

    for t in range(1, 50000000):
        if time.monotonic() - t_start > time_limit:
            break

        # Nesterov extrapolation
        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        grad, b, b_max = compute_grad_smooth(y, beta, top_k)

        # Polyak step size
        val_smooth = 2.0 * n * b_max
        gap = val_smooth - best_val
        gnorm_sq = np.dot(grad, grad)
        if gnorm_sq > 1e-20:
            polyak_step = gap / (2.0 * gnorm_sq)
            step_size = max(1e-6, min(polyak_step, 1.0 / beta * 2.0))
        else:
            step_size = 1.0 / (beta * 0.5)

        # Gradient step + projection
        z = y - step_size * grad
        x_new = simplex_project(z)

        # Evaluate
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        val_new = 2.0 * n * max_new

        # Nesterov averaging
        avg_w = (avg_w + 1.0) / (avg_w + 1.0)
        nesterov_avg = nesterov_avg * avg_w + x_new * (1.0 - avg_w)

        if val_new < best_val - 1e-12:
            best_val = val_new
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        # Staircase beta: hold for 200 iters, then jump
        if t % 200 == 0:
            beta = min(beta * 1.15, beta_end)

        # Restart with structural perturbation
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            step_size = 1.0 / (n ** 0.5)
            r = stall // restart_every
            noise_scale = 0.01 / (1.0 + 0.1 * r)
            if r % 3 == 0:
                # Additive noise
                pert = rng.normal(0, noise_scale, n) * best_a
                x_r = best_a + pert
            elif r % 3 == 1:
                # Swap random blocks
                x_r = best_a.copy()
                bsz = max(1, n // 20)
                for _ in range(3):
                    i = rng.integers(0, n - bsz)
                    j = rng.integers(0, n - bsz)
                    if abs(i - j) > bsz:
                        x_r[i:i+bsz], x_r[j:j+bsz] = x_r[j:j+bsz].copy(), x_r[i:i+bsz].copy()
            else:
                # Shape perturbation: multiply by random smooth function
                x_r = best_a.copy()
                warp = 1.0 + noise_scale * 5.0 * np.sin(2 * np.pi * rng.uniform(1, 8) * np.linspace(0, 1, n) + rng.uniform(0, 2*np.pi))
                x_r = x_r * warp
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            # Also try the nesterov average as restart point
            if rng.random() < 0.3:
                x_r = np.maximum(nesterov_avg, 1e-10)
                x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, beta_start)
            if verbose and t % (restart_every * 2) == 0:
                elapsed = time.monotonic() - t_start
                print("    FISTA-E restart t=%d (%.1fs): J=%.6f beta=%.0f" % (t, elapsed, best_val, beta))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t_start
            print("    FISTA-E t=%d (%.1fs): J=%.6f beta=%.0f step=%.4f" % (t, elapsed, best_val, beta, step_size))

    # Also evaluate nesterov average
    J_avg = 2.0 * n * fast_autocorr(nesterov_avg).max()
    if J_avg < best_val:
        best_a = nesterov_avg.copy()
        best_val = J_avg

    return best_a, best_val

def peak_chipping(a0, time_limit, verbose=True):
    """Directly target and reduce the top autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()

    best_val = 2.0 * n * best_m
    t_start = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    max_stall = 500

    while time.monotonic() - t_start < time_limit and stall < max_stall:
        b = fast_autocorr(a)
        b_max = b.max()
        top_peaks = np.argsort(b)[::-1][:20]

        improved = False
        for _ in range(100):
            if time.monotonic() - t_start > time_limit:
                break

            k_idx = int(top_peaks[rng.integers(min(10, len(top_peaks)))])
            # Find contributing pairs to conv[k_idx]
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])

            # Find target: positions NOT contributing much to any top peak
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True
            in_peak[src] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
                candidates = candidates[candidates != src]
            if len(candidates) == 0:
                continue

            # Choose destination to minimize increase in max
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src:
                continue
            if a[src] < 1e-12:
                continue

            # Try multiple fractions
            best_frac = None
            best_trial_val = best_val
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                val_trial = 2.0 * n * M_trial
                if val_trial < best_trial_val:
                    best_trial_val = val_trial
                    best_frac = frac

            if best_frac is not None and best_trial_val < best_val - 1e-12:
                amt = a[src] * best_frac
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_val = best_trial_val
                best_m = best_trial_val / (2.0 * n)
                best_a = a.copy()
                improved = True
                stall = 0
                b = fast_autocorr(a)
                top_peaks = np.argsort(b)[::-1][:20]

        if not improved:
            stall += 1
            # Random kicks to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.002, 0.02))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    val_new = 2.0 * n * fast_autocorr(a).max()
                    if val_new < best_val:
                        best_val = val_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t_start
            print("    PeakChip (%.1fs): J=%.6f stall=%d" % (elapsed, best_val, stall))

    return best_a, best_val

def cyclic_shift_best(a0, verbose=False):
    """Find the best cyclic shift."""
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        A = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m

def cyclic_shift_sampled(a0, num_samples=500):
    """Sampled cyclic shift search for very long sequences."""
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    rng = np.random.default_rng(12345)
    candidates = [0] + list(rng.choice(range(1, n), size=min(num_samples, n-1), replace=False))
    for s in candidates:
        a_shift = np.roll(a0, -int(s))
        A = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = int(s)
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [%s]: %.6f (n=%d, t=%.1fs)" % (label, sc, len(a), elapsed))
    return best_seq, best_score

def shape_optimize(a0, time_limit, verbose=True):
    """Optimize shape via spline control point perturbation."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_val = 2.0 * n * fast_autocorr(a).max()
    t_start = time.monotonic()
    rng = np.random.default_rng(555)

    # Work with a coarse control grid
    n_ctrl = max(20, n // 200)
    ctrl = a.copy()
    ctrl_idx = np.linspace(0, n - 1, n_ctrl, dtype=int)
    ctrl_vals = ctrl[ctrl_idx].copy()

    for it in range(5000):
        if time.monotonic() - t_start > time_limit:
            break

        # Perturb a random control point
        ci = int(rng.integers(n_ctrl))
        delta = rng.normal(0, 0.01) * ctrl_vals.max()
        ctrl_vals[ci] = max(0, ctrl_vals[ci] + delta)

        # Interpolate to full resolution
        a_trial = np.interp(np.arange(n), ctrl_idx, ctrl_vals)
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t
        val_trial = 2.0 * n * fast_autocorr(a_trial).max()

        if val_trial < best_val - 1e-12:
            best_val = val_trial
            best_a = a_trial.copy()
            a = a_trial.copy()
            # Sync control points
            ctrl_vals = a[ctrl_idx].copy()
        else:
            # Revert
            ctrl_vals[ci] = a[ctrl_idx[ci]] if ci < len(ctrl_idx) else ctrl_vals[ci]

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t_start
            print("    ShapeOpt it=%d (%.1fs): J=%.6f" % (it, elapsed, best_val))

    return best_a, best_val

    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        a_out, _ = shape_optimize(norm_sum1(best_seq), t6, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "shape", t_global)

    # === Phase 7: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 7: Final Polish (n=%d) ===" % n_cur)
    t7a = min(remaining * 0.30, 150.0)
    if t7a > 20:
        a_out, _ = fista_enhanced(norm_sum1(best_seq), t7a,
                                  beta_start=200.0, beta_end=10000.0,
                                  top_k=6, restart_every=1500,
                                  verbose=True, seed=2024)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_final", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    t7b = min(remaining * 0.30, 120.0)
    if t7b > 20:
        a_out, _ = peak_chipping(norm_sum1(best_seq), t7b, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "chip_final", t_global)

    # === Phase 8: Final cyclic + micro FISTA ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_shifted, _ = cyclic_shift_sampled(norm_sum1(best_seq), 500)
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyc_final", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_out, _ = fista_enhanced(norm_sum1(best_seq), min(remaining * 0.5, 60.0),
                                  beta_start=500.0, beta_end=15000.0,
                                  top_k=5, restart_every=1000,
                                  verbose=False, seed=9999)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "micro", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=%.6f, n=%d, time=%.1fs" % (best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence! Using fallback.")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x)**2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
