def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def spectral_flatten(a0, time_limit, verbose=True, seed=42, alpha_init=0.3, alpha_decay=0.9995):
    """
    Gerchberg-Saxton style spectral flattening.
    Iteratively equalizes the power spectrum to flatten the convolution.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    alpha = alpha_init
    
    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        mags = np.abs(A)
        
        # Target: median of non-DC magnitudes (robust to outliers)
        if len(mags) > 2:
            target = np.median(mags[1:])
        else:
            target = mags[0] / np.sqrt(2)
        
        # Blend magnitudes toward target
        new_mags = np.empty_like(mags)
        new_mags[0] = mags[0]  # Keep DC
        for m in range(1, len(mags)):
            new_mags[m] = (1 - alpha) * mags[m] + alpha * target
        
        # Preserve phase
        phases = np.ones_like(A, dtype=np.complex128)
        nonzero = mags > 1e-30
        phases[nonzero] = A[nonzero] / mags[nonzero]
        
        A_new = new_mags * phases
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        m_new = fast_autocorr(a_new).max()
        
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 500 == 0:
            # Phase perturbation to escape local minima
            A_curr = np.fft.rfft(a, fft_size)
            pert = rng.standard_normal(len(A_curr)) * 0.05
            A_curr[1:] *= np.exp(1j * pert[1:])
            a_trial = np.fft.irfft(A_curr, fft_size)[:n]

            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial
        
        alpha *= alpha_decay
        if alpha < 0.01:
            alpha = 0.01 + 0.02 * rng.random()  # Reheat

    return best_a, best_m
