def fast_autocorr(a):

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return a_interp / ss

# Spectral shaping (Gerchberg-Saxton)
def spectral_shape(a0, time_limit, n_iters=10000, verbose=True):
    """Iteratively shape the spectrum to flatten autocorrelation."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    L = 2 * n - 1
    fft_size = 1 << (L - 1).bit_length()
    
    for t in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # FFT
        A = np.fft.fft(a, n=fft_size)
        
        # Amplitude constraint: flatten |A|^2
        amp = np.abs(A)
        target = amp.mean()
        # Soft flattening: reduce peaks, boost valleys
        alpha = 0.5  # mixing parameter
        new_amp = amp * (1 - alpha) + target * alpha
        
        # Preserve phase
        phase = np.angle(A)
        A_new = new_amp * np.exp(1j * phase)
        
        # IFFT
        a_new = np.fft.ifft(A_new).real[:n]
        
        # Support constraint: non-negative
        a_new = np.maximum(a_new, 0.0)
        
        # Normalize
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        a = a_new
        
        # Track best
        val = obj_val(a)

            best_a = a.copy()

            print("    Spectral t=" + str(t) + " (" + str(round(time.monotonic()-t0,1)) + "s) J=" + repr(best_val))
    
    return best_a, best_val
