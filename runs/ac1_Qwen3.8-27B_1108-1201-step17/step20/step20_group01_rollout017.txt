8. Project to non-negative: a_new = max(0, a_new)
9. Normalize
10. Repeat

    A = np.fft.rfft(a.astype(np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val_normalized(a):
    """Objective assuming a is normalized (sum=1). Returns 2*n*max(autocorr)."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def spectral_shape(a, alpha, n_iters=50, verbose=False):
    """
    Haffmeyer-style spectral shaping to reduce peak autocorrelation.
    Iteratively flattens the power spectrum in frequency domain.
    """
    a = normalize(a)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    for it in range(n_iters):
        A = np.fft.rfft(a, n=fft_size)
        P = np.abs(A) ** 2
        # Flatten power spectrum
        P_new = P ** alpha
        # Reconstruct with original phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try blending
            for t in [0.3, 0.5, 0.7]:
                blend = (1 - t) * best_a + t * a_new
                blend = np.maximum(blend, 0.0)
                sb = blend.sum()
                if sb < 1e-15:
                    continue
                blend = blend / sb
                mb = fast_autocorr(blend).max()
                if mb < best_m - 1e-14:
                    best_m = mb
                    best_a = blend.copy()
                    a = a_new
                    break
    
    if verbose:
        print("    Spectral: M={:.7f}".format(fast_autocorr(best_a).max()))
    return best_a

def spectral_shape_multi(a, time_limit, verbose=True):
    """Try multiple alpha values for spectral shaping."""
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for alpha in [0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 0.5, 0.4]:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        n_iters = max(20, int(remaining * 30))
        a_try = spectral_shape(normalize(best_a), alpha, n_iters=min(n_iters, 200), verbose=False)
        m_try = fast_autocorr(a_try).max()
        if m_try < best_m - 1e-12:
            best_m = m_try
            best_a = a_try
            if verbose:
                print("    Spectral(a={:.1f}): M={:.7f}".format(alpha, best_m))
    
    return best_a

def build_lp_matrix(a, active_list, n):
    K = len(active_list)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj in range(K):
        ki = int(active_list[jj])
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[ki - idx]
    return coeffs

def fast_fw_step(a, ac, n_top=50, temperature=5.0):
    """
    Fast Frank-Wolfe: find optimal vertex by weighted subgradient.
    Returns the one-hot direction (indexed position).
    """
    n = len(a)
    # Select top lags with temperature weighting
    k_sel = min(n_top, len(ac))
    idx = np.argsort(ac)[::-1][:k_sel]
    top_vals = ac[idx]
    w = np.exp(temperature * (top_vals - top_vals[0]))
    w = w / (w.sum() + 1e-30)
    
    # Compute weighted gradient: sum_k w_k * a[shifted by k]
    grad = np.zeros(n, dtype=np.float64)
    for ii in range(len(idx)):
        ki = int(idx[ii])
        ws = max(0, ki - n + 1)
        we = min(n, ki + 1)
        if ws < we:
            idx_range = np.arange(ws, we)
            grad[idx_range] += w[ii] * a[ki - idx_range]
    
    # FW direction: coordinate with minimum gradient value
    j = int(np.argmin(grad))
    return j, grad

def fw_fast(a0, time_limit, n_top=50, verbose=True):
    """
    Fast Frank-Wolfe with greedy weighted subgradient steps.
    No LP solve - just picks best coordinate to move toward.
    """
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    stall = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 5000:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Get FW direction
        adj_temp = 3.0 + 2.0 * np.log1p(t / 100.0)
        j, grad = fast_fw_step(y, ac, n_top=n_top, temperature=adj_temp)
        
        # e_j direction (move toward coordinate j)
        e_j = np.zeros(n, dtype=np.float64)
        e_j[j] = 1.0
        
        # Line search: golden section
        best_alpha = 2.0 / (t + 2.0)
        best_trial_m = 1e10
        
        # Quick check at a few points
        for alpha in [best_alpha, 0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
            trial = (1 - alpha) * y + alpha * e_j
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            Mt = fast_autocorr(trial / st).max()
            if Mt < best_trial_m:
                best_trial_m = Mt
                best_alpha = alpha
        
        # Refine with bisection
        lo = max(0.0, best_alpha - 0.2)
        hi = min(1.0, best_alpha + 0.2)
        for _ in range(5):
            mid1 = lo + (hi - lo) / 3.0
            mid2 = hi - (hi - lo) / 3.0
            t1 = np.maximum((1 - mid1) * y + mid1 * e_j, 0.0)
            t2 = np.maximum((1 - mid2) * y + mid2 * e_j, 0.0)
            s1, s2 = t1.sum(), t2.sum()
            m1 = fast_autocorr(t1 / s1).max() if s1 > 1e-15 else 1e10
            m2 = fast_autocorr(t2 / s2).max() if s2 > 1e-15 else 1e10
            if m1 < m2:
                hi = mid2
            else:
                lo = mid1
        best_alpha = (lo + hi) / 2.0
        
        y = np.maximum((1.0 - best_alpha) * y + best_alpha * e_j, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        if verbose and t % 500 == 0:
            print("    FW t={:5d} ({:.1f}s) J={:.7f} stall={}".format(
                t, time.monotonic() - t0, 2 * n * best_m, stall))
    
    return best_a, 2.0 * n * best_m

def lp_fw(a0, time_limit, solver, init_K=200, max_K=4000, verbose=True):
    """
    LP-based Frank-Wolfe with adaptive active set.
    """
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:init_K])
    no_improve = 0
    stall_limit = 3000
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Refresh active set periodically
        if t % 20 == 0 or t == 1:
            top_lags = np.argsort(ac)[::-1]
            target_size = min(max_K, init_K + t * 3)
            if len(active_set) < target_size:
                for k in top_lags[:target_size]:
                    active_set.add(int(k))
            if len(active_set) > max_K:
                clist = sorted(active_set)
                margins = np.array([M_curr - ac[k] for k in clist])
                keep = np.argsort(margins)[:max_K]
                active_set = set(int(clist[i]) for i in keep)
        active_set.add(int(np.argmax(ac)))
        
        clist = sorted(active_set)
        K = len(clist)
        if K == 0:
            break
        
        try:
            coeffs = build_lp_matrix(y, clist, n)
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            prob.solve(solver=solver, verbose=False)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                raise Exception("LP: " + str(prob.status))
        except Exception:
            if len(active_set) > init_K // 2:
                clist = sorted(active_set)
                active_set = set(int(clist[i]) for i in np.argsort(
                    [M_curr - ac[k] for k in clist])[:init_K])
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check violations to grow active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                active_set.add(int(k))
        
        # Golden section line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_m = 1e10
        
        for alpha in [best_alpha, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            trial = np.maximum((1 - alpha) * y + alpha * g_dir, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            Mt = fast_autocorr(trial / st).max()
            if Mt < best_trial_m:
                best_trial_m = Mt
                best_alpha = alpha
        
        # Refine
        lo = max(0.0, best_alpha - 0.15)
        hi = min(1.0, best_alpha + 0.15)
        for _ in range(7):
            mid1 = lo + (hi - lo) / 3.0
            mid2 = hi - (hi - lo) / 3.0
            t1 = np.maximum((1 - mid1) * y + mid1 * g_dir, 0.0)
            t2 = np.maximum((1 - mid2) * y + mid2 * g_dir, 0.0)
            s1, s2 = t1.sum(), t2.sum()
            m1 = fast_autocorr(t1 / s1).max() if s1 > 1e-15 else 1e10
            m2 = fast_autocorr(t2 / s2).max() if s2 > 1e-15 else 1e10
            if m1 < m2:
                hi = mid2
            else:
                lo = mid1
        best_alpha = (lo + hi) / 2.0
        
        y = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        if verbose and t % 100 == 0:
            print("    LP-FW t={:5d} ({:.1f}s) J={:.7f} |S|={}".format(
                t, time.monotonic() - t0, 2 * n * best_m, len(active_set)))
    
    return best_a, 2.0 * n * best_m

def spectral_lp_hybrid(a0, time_limit, solver, verbose=True):
    """
    Alternate between spectral shaping and LP-FW steps.
    Spectral steps help escape local optima.
    """
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    cycle = 0
    while time.monotonic() - t0 < time_limit:
        # Phase A: Spectral shaping (short burst)
        t_spec = min(30.0, time_limit - (time.monotonic() - t0))
        if t_spec > 5:
            alphas = [0.4 + 0.1 * (cycle % 5), 0.5, 0.6]
            for alpha in alphas:
                if time.monotonic() - t0 >= time_limit:
                    break
                y_try = spectral_shape(y, alpha, n_iters=30, verbose=False)
                m_try = fast_autocorr(y_try).max()
                if m_try < best_m - 1e-12:
                    best_m = m_try
                    best_a = y_try.copy()
                    y = y_try
        
        # Phase B: A few LP-FW iterations
        t_lp = min(120.0, time_limit - (time.monotonic() - t0))
        if t_lp > 10:
            ac = fast_autocorr(y)
            init_K = 150 + 50 * cycle
            max_K = min(3000, 1500 + 200 * cycle)
            
            active_set = set(int(k) for k in np.argsort(ac)[::-1][:init_K])
            
            n_lp_iters = max(5, int(t_lp / 3))
            for it in range(n_lp_iters):
                if time.monotonic() - t0 >= time_limit:
                    break
                
                ac = fast_autocorr(y)
                M_curr = ac.max()
                
                if M_curr < best_m - 1e-14:
                    best_m = M_curr
                    best_a = y.copy()
                
                active_set.add(int(np.argmax(ac)))
                
                if len(active_set) > max_K:
                    clist = sorted(active_set)
                    margins = np.array([M_curr - ac[k] for k in clist])
                    keep = np.argsort(margins)[:max_K]
                    active_set = set(int(clist[i]) for i in keep)
                
                clist = sorted(active_set)
                try:
                    coeffs = build_lp_matrix(y, clist, n)
                    g_var = cp.Variable(n, nonneg=True)
                    M_var = cp.Variable()
                    constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
                    prob = cp.Problem(cp.Minimize(M_var), constraints)
                    prob.solve(solver=solver, verbose=False)
                    if prob.status not in ('optimal', 'optimal_inaccurate'):
                        continue
                except Exception:
                    continue
                
                g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
                g_sum = g_val.sum()
                if g_sum < 1e-15:
                    continue
                g_dir = g_val / g_sum
                
                # Line search
                best_alpha = 0.5
                best_trial_m = 1e10
                for alpha in [0.1, 0.3, 0.5, 0.7, 0.9]:
                    trial = np.maximum((1 - alpha) * y + alpha * g_dir, 0.0)
                    st = trial.sum()
                    if st < 1e-15:
                        continue
                    Mt = fast_autocorr(trial / st).max()
                    if Mt < best_trial_m:
                        best_trial_m = Mt
                        best_alpha = alpha
                
                y = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
            
            m_final = fast_autocorr(y).max()
            if m_final < best_m:
                best_m = m_final
                best_a = y.copy()
        
        cycle += 1
        if verbose and cycle % 2 == 0:
            print("    SpecLP cycle={} ({:.1f}s) J={:.7f}".format(
                cycle, time.monotonic() - t0, 2 * n * best_m))
    
    return best_a, 2.0 * n * best_m

def coord_refine(a0, time_limit, n_targets=15, seed=42, verbose=True):
    """
    Coordinate refinement: move mass from high-pressure positions to low-pressure ones.
    Uses FFT-based pressure computation.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 3000
    fracs = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
        stall += 1
        
        n_top = min(n_targets, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            ws = max(0, ki)
            we = min(n - 1, ki + n - 1)
            # Valid indices for a[i] * a[ki - i]
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-25:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-15:
                continue
            
            # Find quiet positions (low contribution to top lags)
            pressure = np.zeros(n, dtype=np.float64)
            for kk in top_k[:6]:
                kkk = int(kk)
                ps = max(0, kkk - n + 1)
                pe = min(n, kkk + 1)
                if ps < pe:
                    pidx = np.arange(ps, pe)
                    pressure[pidx] += a[kkk - pidx]
            pressure[src] = 1e10
            quiet = np.argsort(pressure)[:20]
            
            for dst in quiet:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-18:
                        break
                    a_t = a.copy()
                    a_t[src] = a_t[src] - amt
                    a_t[dst] = a_t[dst] + amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-15:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                if improved:
                    break
            
            if stall > 200 and not improved:
                # Random perturbation to escape
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
                break
        
        if not improved:
            if stall > 500:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
        
        if verbose and stall % 1000 == 0 and stall > 0:
            print("    CoordRef ({:.1f}s) J={:.7f} stall={}".format(
                time.monotonic() - t0, 2 * n * best_m, stall))
    
    return best_a, 2.0 * n * best_m

def propose_candidate(budget_s=1800.0):
    """
    Main search function. Uses spectral shaping, LP-FW, and coordinate refinement.
    """
    t_global = time.monotonic()

    solver = get_solver()
    
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent loaded: n={} score={}".format(len(p), repr(best_score)))
    except Exception:
        best_seq = None
    
    if best_seq is None:
        n_init = 10000
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(np.sin(np.pi * x) ** 2 + 0.1)
        best_score = eval_safe(best_seq)
        print("Default init: n={} score={}".format(n_init, repr(best_score)))
    
    n0 = len(best_seq)
    
    # === Phase 1: Spectral shaping for basin exploration (120s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Spectral shaping (n={}) ===".format(n0))
    t1 = min(remaining * 0.07, 120.0)
    if t1 > 20:
        a_out = spectral_shape_multi(normalize(best_seq), t1, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P1 improved: {}".format(repr(best_score)))
    
    # === Phase 2: Fast FW for quick progress (200s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P2: Fast FW (n={}) ===".format(len(best_seq)))
    t2 = min(remaining * 0.12, 200.0)
    if t2 > 30:
        a_out, v_out = fw_fast(normalize(best_seq), t2, n_top=60, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P2 improved: {}".format(repr(best_score)))
    
    # === Phase 3: Main LP-FW (600s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P3: LP-FW main (n={}) ===".format(n_cur))
    t3 = min(remaining * 0.35, 600.0)
    if t3 > 60:
        a_out, v_out = lp_fw(normalize(best_seq), t3, solver, init_K=200, max_K=3500, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P3 improved: {}".format(repr(best_score)))
    
    # === Phase 4: Spectral-LP hybrid (250s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P4: SpecLP hybrid (n={}) ===".format(n_cur))
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        a_out, v_out = spectral_lp_hybrid(normalize(best_seq), t4, solver, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P4 improved: {}".format(repr(best_score)))
    
    # === Phase 5: Coordinate refinement (200s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P5: CoordRefine (n={}) ===".format(n_cur))
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 20:
        a_out, v_out = coord_refine(normalize(best_seq), t5, n_targets=15, seed=42, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P5 improved: {}".format(repr(best_score)))
    
    # === Phase 6: Try slightly longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 1000
    if n_long <= 14000 and remaining > 100:
        print("\n=== P6: Longer n={} ===".format(n_long))
        t6 = min(remaining * 0.12, 180.0)
        if t6 > 30:
            a_long = interpolate_seq(best_seq, n_long)
            a_out1, _ = fw_fast(a_long, t6 * 0.4, n_top=50, verbose=False)
            a_out2, _ = coord_refine(a_out1, t6 * 0.4, n_targets=10, seed=77, verbose=False)
            a_out3, _ = lp_fw(a_out2, t6 * 0.3, solver, init_K=100, max_K=1500, verbose=False)
            sc1 = eval_safe(a_out1)
            sc2 = eval_safe(a_out2)
            sc3 = eval_safe(a_out3)
            best_sc = min(sc1, sc2, sc3)
            if np.isfinite(best_sc) and best_sc < best_score:
                best_score = best_sc
                best_seq = a_out1 if sc1 == best_sc else (a_out2 if sc2 == best_sc else a_out3)
                print("  P6 improved: {}".format(repr(best_score)))
    
    # === Phase 7: Final LP-FW polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P7: Final LP-FW (n={}) ===".format(n_cur))
    t7 = min(remaining * 0.15, 250.0)
    if t7 > 30:
        a_out, v_out = lp_fw(normalize(best_seq), t7, solver, init_K=300, max_K=4000, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P7 improved: {}".format(repr(best_score)))
    
    # === Phase 8: Final coord + spectral polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P8: Final polish ===")
    t8 = min(remaining * 0.15, 200.0)
    if t8 > 20:
        # Quick spectral
        t8a = t8 * 0.3
        a_sp = spectral_shape_multi(normalize(best_seq), t8a, verbose=False)
        sc_sp = eval_safe(a_sp)
        if np.isfinite(sc_sp) and sc_sp < best_score:
            best_score = sc_sp
            best_seq = a_sp
        
        # Final coord
        t8b = min(t8 * 0.6, time_budget - (time.monotonic() - t_global))
        if t8b > 10:
            a_out, _ = coord_refine(normalize(best_seq), t8b, n_targets=20, seed=999, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  P8 improved: {}".format(repr(best_score)))
    
    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={} n={} t={:.1f}s".format(repr(best_score), len(best_seq), elapsed))
    else:
        print("\nERROR: using fallback")
        n_fb = 10000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2 + 0.1)
        best_score = eval_safe(best_seq)
    
    return [float(x) for x in best_seq]
