def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def fft_conv2(a, b):

    b = np.asarray(b, dtype=np.float64)

    size = 2 * n - 1

    return c

def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    new_n = int(max(2, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)


def lse_objective_1d(u, beta):
    """Compute LSE-smoothed objective for sequence a = softmax(u) normalized."""
    n = u.size
    if n <= 1:
        return 0.0, np.zeros(n, dtype=np.float64)
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return 1e100, np.zeros(n, dtype=np.float64)
    a = e / s

    size = 2 * n - 1
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)

    m = float(np.max(c))
    shifted = beta * (c - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    expv = np.exp(shifted)
    se = float(expv.sum())
    if se <= 0.0 or not math.isfinite(se):
        expv = np.zeros(size, dtype=np.float64)
        expv[0] = 1.0
        se = 1.0
    p = expv / se

    pf = p
    Pr = np.fft.rfft(pf[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    ga = 2.0 * conv[n - 1:2 * n - 1][::-1]
    ga = np.where(np.isfinite(ga), ga, 0.0)

    f = m + math.log(se) / beta

    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)
    gu = np.where(np.isfinite(gu), gu, 0.0)

    if not math.isfinite(f):
        f = 1e100
        gu = np.zeros(n, dtype=np.float64)
    return f, gu


def run_lbfegs(a0, beta, maxfun=400):
    """Run L-BFGS-B on the LSE-smoothed objective in free coordinates."""
    a = norm_sum1(a0)
    n = a.size
    if n <= 1 or not SCIPY_OK or maxfun <= 0:
        return a

    # Use log coordinates: u_i = log(a_i) - log(a_ref), with a_ref fixed
    # Actually, use a simpler approach: optimize in log-space directly
    # a_i = exp(v_i) / sum(exp(v_j))
    v0 = np.log(np.maximum(a, 1e-300))

    def obj_and_grad(v):
        return lse_objective_1d(v, beta)

    t0 = time.monotonic()
    try:
        res = scipy.optimize.minimize(
            obj_and_grad, v0, method="L-BFGS-B", jac=True,
            options={"maxiter": int(maxfun), "maxfun": int(maxfun),
                     "ftol": 1e-14, "gtol": 1e-8, "disp": False}
        )
        if res is not None and hasattr(res, "x"):
            vx = np.asarray(res.x, dtype=np.float64)
            if vx.size == n and np.all(np.isfinite(vx)):
                a_out = norm_sum1(np.exp(vx - np.max(vx)))
                return a_out
    except Exception:
        pass
    return a


def apply_rank1_move(c, a, s, d, tmove):
    """Apply rank-1 update for moving mass t from s to d."""
    n = a.size
    c_new = c.copy()
    # Add contribution from new position d (positive)
    d_start = d
    d_end = min(d + n, c_new.size)
    a_part = a[:d_end - d_start]
    c_new[d_start:d_end] += 2.0 * tmove * a_part
    # Remove contribution from position s (negative)
    s_start = s
    s_end = min(s + n, c_new.size)
    a_part2 = a[:s_end - s_start]
    c_new[s_start:s_end] -= 2.0 * tmove * a_part2
    # t^2 corrections
    t2 = tmove * tmove
    if 2 * d < c_new.size:
        c_new[2 * d] += t2
    if 2 * s < c_new.size:
        c_new[2 * s] += t2
    if s + d < c_new.size:
        c_new[s + d] -= 2.0 * t2
    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
    np.maximum(c_new, 0.0, out=c_new)
    return c_new

        if time_left < 1.0:
            break

        # Find top peaks
        top_k = min(20, 2 * n - 1)

        # Weight peaks by how much they exceed the median
        med = float(np.median(c))
        weights = np.maximum(c[top_idx] - med, 1e-10)

        pressure = cross_pressure(p_top, a)

        # Try structured candidates

        n_dst = min(30, n)

        dst_cands = np.argpartition(dst_score, n_dst - 1)[:n_dst] if n > n_dst else np.argsort(dst_score)

            if s == d or a[s] < 1e-14:

            frac = float(rng.uniform(0.002, 0.08))
            tmove = a[s] * frac
            tmove = min(tmove, a[s] * 0.99)
            if tmove <= 1e-16:

            c_new = apply_rank1_move(c, a, s, d, tmove)
            m_new = float(np.max(c_new))
            if m_new < best_m_new:
                best_m_new = m_new
                best_move = (s, d, tmove)

        if best_move is not None:
            s, d, tmove = best_move
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = fft_conv(a)
            M = float(np.max(c))
            if M < best_M - 1e-16:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and (it <= 5 or it % 50 == 0):
                print("    CD it={} M={:.10f} score_proxy={:.8f}".format(
                    it, best_M, 2.0 * n * best_M))
        else:
            stall += 1
            if stall >= 50:
                # Perturb and restart
                pert = rng.normal(0.0, 0.005, n) * a

def game_direction_v2(f, active_idx, mw_iters=30, seed=0):
    """Min-max game to find Frank-Wolfe direction."""
    f = norm_sum1(f)

    K = active_idx.size

        return np.ones(n, dtype=np.float64) / float(n)

    rng = np.random.default_rng(seed)
    logits = np.zeros(K, dtype=np.float64) + rng.normal(0, 0.1, K)

    for _ in range(mw_iters):
        # Update g (column player): minimize max pressure
        pf = np.zeros(size, dtype=np.float64)
        loc = logits - np.max(logits)
        w = np.exp(loc)
        sw = float(w.sum())
        if sw <= 0:
            w = np.ones(K) / K
        else:
            w = w / sw
        pf[active_idx] = w

        pressure = cross_pressure(pf, f)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0), pressure, 0.0)
        pmx = float(np.max(pressure))
        pmn = float(np.min(pressure))
        if pmx - pmn > 1e-30:
            scaled = (pressure - pmn) / (pmx - pmn)
            gw = np.exp(-6.0 * scaled)
        else:
            gw = np.ones(n, dtype=np.float64)
        gw = np.where(np.isfinite(gw), gw, 0.0)
        gw += 1e-10
        g = gw / float(gw.sum())

        # Update lambda (row player): concentrate on highest f*g
        h = fft_conv2(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmx = float(np.max(ha))
        hmn = float(np.min(ha))
        if hmx - hmn > 1e-30:
            eta = 10.0 / (hmx - hmn)
            logits += eta * (ha - float(np.mean(ha)))

    return g


def frank_wolfe_v2(a0, budget, verbose=True):
    """Frank-Wolfe with game-based directions."""
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    no_improve = 0
    t = 0

    while time.monotonic() - t0 < time_limit - 3:
        t += 1
        time_left = time_limit - (time.monotonic() - t0)
        if time_left < 5.0:
            break

        # Get active set
        active_k = min(500, 2 * n - 1)
        active = np.argsort(c)[::-1][:active_k]

        g = game_direction_v2(a, active, mw_iters=20, seed=1000 + t)
        if g is None:
            break

        cg = fft_conv(g)
        fg = fft_conv2(a, g)

        # Line search
        best_alpha = 0.0
        best_m = M
        for alpha in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0):
            one = 1.0 - alpha
            val = (one * one) * c + 2.0 * alpha * one * fg + alpha * alpha * cg
            m = float(np.max(val))
            if m < best_m:
                best_m = m
                best_alpha = alpha

        if best_alpha > 0 and best_m < M - 1e-16:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            s = float(a.sum())
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
                c = fft_conv(a)
                M = float(np.max(c))
            no_improve = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
                if verbose and (t <= 3 or t % 20 == 0):
                    print("    FW t={} score_proxy={:.8f}".format(t, 2.0 * n * best_M))
        else:
            no_improve += 1
            if no_improve >= 30:
                break

    return best_a


def peak_reduction_move(a, c, M):
    """Try to reduce the maximum peak by moving mass from its contributors."""
    n = a.size
    size = 2 * n - 1
    max_idx = int(np.argmax(c))
    # The convolution at max_idx: c[max_idx] = sum_i a[i]*a[max_idx - i]
    # Find which pair (i, j) with i+j=max_idx contributes most
    s = max(0, max_idx - n + 1)
    e = min(n, max_idx + 1)
    if s >= e:
        return None

    pairs = []
    for i in range(s, e):
        j = max_idx - i
        if 0 <= j < n:
            contrib = a[i] * a[j]
            pairs.append((contrib, i, j))

    if not pairs:
        return None

    # Sort by contribution (highest first)
    pairs.sort(reverse=True)

    # Try moving mass from the highest contributor to a random low-pressure position
    rng = np.random.default_rng(12345 + max_idx)
    best_move = None
    best_m = M

    for contrib, i, j in pairs[:5]:
        # Try moving from i to somewhere else
        for _ in range(5):
            d = int(rng.integers(n))
            if d == i or d == j:
                continue
            frac = float(rng.uniform(0.01, 0.15))
            tmove = a[i] * frac
            tmove = min(tmove, a[i] * 0.95)
            if tmove <= 1e-16:
                continue
            c_new = apply_rank1_move(c, a, i, d, tmove)
            m_new = float(np.max(c_new))
            if m_new < best_m:
                best_m = m_new
                best_move = (i, d, tmove)

        # Also try moving from j
        for _ in range(5):
            d = int(rng.integers(n))
            if d == i or d == j:
                continue
            frac = float(rng.uniform(0.01, 0.15))
            tmove = a[j] * frac
            tmove = min(tmove, a[j] * 0.95)
            if tmove <= 1e-16:
                continue
            c_new = apply_rank1_move(c, a, j, d, tmove)
            m_new = float(np.max(c_new))
            if m_new < best_m:
                best_m = m_new
                best_move = (j, d, tmove)

    if best_move is not None:
        s_idx, d_idx, tmove = best_move
        a[s_idx] -= tmove
        a[d_idx] += tmove
        a = np.maximum(a, 0.0)
        ss = float(a.sum())
        if ss > 1e-15:
            a = a / ss
        return a, best_move
    return None, None


def propose_candidate(budget_s=1800.0):
    """Main optimization driver with multi-phase strategy."""
    t_start = time.monotonic()
    total_budget = max(120.0, min(float(budget_s) - 60.0, 1740.0))

    print("Starting optimization, budget={:.1f}s".format(total_budget))

    # Get parent sequence
    seq_best = None
    try:
        parent_list = height_sequence_1
        p = np.asarray(parent_list, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        np.clip(p, 0.0, 1000.0, out=p)
        if p.ndim != 1:
            p = p.ravel()
        if p.size > 0 and float(p.sum()) > 1e-10:
            seq_best = norm_sum1(p)
    except Exception:
        pass

    if seq_best is None:
        n0 = 5000
        x = np.linspace(0.0, 1.0, n0)
        seq_best = norm_sum1(np.sin(np.pi * x) ** 2)

    best_score = safe_eval(seq_best)
    if not math.isfinite(best_score):
        best_score = 2.0 * seq_best.size * float(np.max(fft_conv(seq_best)))
    print("Initial: n={} score={:.10f}".format(seq_best.size, best_score))

    # Phase 1: Multi-length exploration with L-BFGS
    print("\n=== Phase 1: Multi-length L-BFGS ===")
    t0 = time.monotonic()
    phase1_budget = min(total_budget * 0.35, 550.0)

    base_n = seq_best.size
    # Try a range of lengths around the base
    lengths_to_try = []
    if base_n > 0:
        for offset in (0, 500, -500, 1000, -1000, 2000, -2000, 3000):
            ln = int(base_n + offset)
            if 1000 <= ln <= 15000:
                lengths_to_try.append(ln)
    # Also try some round numbers
    for ln in (2000, 3000, 4000, 5000, 6000, 7000, 8000, 10000, 12000):
        if ln not in lengths_to_try:
            lengths_to_try.append(ln)

    # Sort by proximity to base_n first, then by size
    lengths_to_try = sorted(set(lengths_to_try), key=lambda x: (abs(x - base_n), x))

    for ln in lengths_to_try[:8]:  # Try at most 8 lengths
        time_left = total_budget - (time.monotonic() - t_start)
        if time_left < 60:
            break
        per_budget = min(time_left * 0.4, 80.0)
        print("  Trying n={} (budget {:.1f}s)".format(ln, per_budget))

        a_init = interpolate_sequence(seq_best, ln)

        # Run L-BFGS with increasing beta
        for beta in (50.0, 200.0, 800.0, 3000.0):
            time_left = total_budget - (time.monotonic() - t_start)
            if time_left < 30:
                break
            maxfun = int(max(50, min(300, per_budget / 4.0)))
            a_new = run_lbfegs(a_init, beta, maxfun=maxfun)
            seq_best, best_score = try_update(seq_best, best_score, a_new, "lb_n{}_b{}".format(ln, int(beta)), t_start)
            if a_new is not None and a_new.size == ln:
                a_init = a_new

    refresh_best()

    # Phase 2: Frank-Wolfe on best
    print("\n=== Phase 2: Frank-Wolfe ===")
    t0 = time.monotonic()
    phase2_budget = min(total_budget * 0.15, 200.0)
    time_left = total_budget - (time.monotonic() - t_start)
    if time_left > 50:
        a_fw = frank_wolfe_v2(seq_best, phase2_budget, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_fw, "fw", t_start)

    # Phase 3: Intensive coordinate descent (main workhorse)
    print("\n=== Phase 3: Coordinate Descent ===")
    phase3_budget = min(total_budget * 0.35, 550.0)
    time_left = total_budget - (time.monotonic() - t_start)
    if time_left > 60:
        cd_budget = min(phase3_budget, time_left * 0.7)
        a_cd = coordinate_descent(seq_best, cd_budget, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_cd, "cd", t_start)

    # Phase 4: Peak reduction + coordinate descent again
    print("\n=== Phase 4: Peak Reduction + CD ===")
    time_left = total_budget - (time.monotonic() - t_start)
    if time_left > 60:
        a_cur = norm_sum1(seq_best)
        c_cur = fft_conv(a_cur)
        M_cur = float(np.max(c_cur))
        improved = True
        peak_budget = time_left * 0.5
        t_peak = time.monotonic()
        while improved and time.monotonic() - t_peak < peak_budget:
            improved = False
            a_cur, mv = peak_reduction_move(a_cur, c_cur, M_cur)
            if a_cur is not None and mv is not None:
                c_cur = fft_conv(a_cur)
                M_cur = float(np.max(c_cur))
                if M_cur < best_score / (2.0 * a_cur.size) - 1e-12:
                    improved = True
        seq_best, best_score = try_update(seq_best, best_score, a_cur, "peak", t_start)

        # Another CD round
        time_left = total_budget - (time.monotonic() - t_start)
        if time_left > 40:
            a_cd2 = coordinate_descent(seq_best, min(time_left * 0.5, 100.0), verbose=False)
            seq_best, best_score = try_update(seq_best, best_score, a_cd2, "cd2", t_start)

    # Phase 5: Final L-BFGS polish with very high beta
    print("\n=== Phase 5: Final High-Beta Polish ===")
    time_left = total_budget - (time.monotonic() - t_start)
    if SCIPY_OK and time_left > 40:
        for beta in (2000.0, 8000.0):
            time_left = total_budget - (time.monotonic() - t_start)
            if time_left < 20:
                break
            maxfun = int(max(30, min(150, time_left * 0.6)))
            a_polish = run_lbfegs(seq_best, beta, maxfun=maxfun)
            seq_best, best_score = try_update(seq_best, best_score, a_polish, "polish_b{}".format(int(beta)), t_start)

    # Phase 6: Final coordinate descent
    print("\n=== Phase 6: Final CD ===")
    time_left = total_budget - (time.monotonic() - t_start)
    if time_left > 30:
        a_final = coordinate_descent(seq_best, min(time_left * 0.8, 80.0), verbose=False)
        seq_best, best_score = try_update(seq_best, best_score, a_final, "final_cd", t_start)

    # Final validation
    final_val = safe_eval(seq_best)
    if math.isfinite(final_val):
        if final_val < best_score:
            best_score = final_val
    else:
        x = np.linspace(0.0, 1.0, 5000)
        seq_best = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = safe_eval(seq_best)

    arr = np.asarray(seq_best, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if arr.size == 0 or s < 0.01:
        x = np.linspace(0.0, 1.0, 5000)
        arr = norm_sum1(np.sin(np.pi * x) ** 2)
    else:
        arr = arr / s

    elapsed = time.monotonic() - t_start
    print("\nFINAL: score={:.10f} n={} elapsed={:.1f}s".format(best_score, arr.size, elapsed))
    return [float(x) for x in arr]
