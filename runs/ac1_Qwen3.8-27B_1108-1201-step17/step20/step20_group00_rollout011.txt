def interpolate_seq(f, new_n):

def build_sparse_constraint(a, lags, n):

            break

            active_set = set(int(k) for k in np.argsort(ac)[-min(init_K, 2 * init_K):])
            no_improve = 200

        max_keep = max_K

                    return 1e10

                return fast_autocorr(trial).max()

            f1 = trial_obj_fn(c1)
            f2 = trial_obj_fn(c2)
            for _ in range(16):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_phi * (hi - lo)
                    f1 = trial_obj_fn(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = hi - inv_phi * (hi - lo)
                    f2 = trial_obj_fn(c2)

            best_alpha = (lo + hi) / 2.0

        y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:

        else:
            y = g_dir.copy()

        if verbose and t % 80 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t={} ({}s) obj={:.6f} |S|={}".format(

    if verbose:

def parametric_init(n, parent_seq, parent_score):
    """Generate diverse parametric candidates and return best."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)

    # Power-law decreasing
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        cands = (1.0 - i_arr / n + 1e-6) ** p
        candidates.append(normalize(cands))
        cands2 = (i_arr / n + 1e-6) ** p
        candidates.append(normalize(cands2))

        candidates.append(normalize(r ** (n - 1 - i_arr)))

    # Gaussian bumps (asymmetric)
    for c in [0.15, 0.25, 0.35, 0.45, 0.55, 0.65, 0.75, 0.85]:
        for sg in [0.08, 0.12, 0.18, 0.25, 0.35]:
            candidates.append(normalize(np.exp(-((x - c) ** 2) / (2 * sg * sg))))

    # Double bump asymmetric
    for c1 in [0.2, 0.3, 0.4]:
        for c2 in [0.6, 0.7, 0.8]:
            for sg in [0.08, 0.12, 0.2]:
                cands = np.exp(-((x - c1) ** 2) / (2 * sg * sg)) + 0.5 * np.exp(-((x - c2) ** 2) / (2 * sg * sg))
                candidates.append(normalize(cands))

    # Tukey windows
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7]:
        cands = np.ones(n)
        m = int(alpha * n)
        if m > 0:
            taper = 0.5 * (1 + np.cos(np.pi * np.arange(m) / m))
            cands[:m] = taper
            cands[-m:] = taper[::-1]
        candidates.append(normalize(cands))

    # Sine/cosine windows
    for f in [1, 2, 3, 5]:
        candidates.append(normalize(np.maximum(np.sin(np.pi * x), 1e-6) ** (f + 1)))
        candidates.append(normalize(np.maximum(1.0 + 0.8 * np.cos(2 * np.pi * f * x), 0.01)))

    # Blend with parent
    if parent_seq is not None:
        bi = normalize(parent_seq)
        for w in [0.3, 0.5, 0.7, 0.9]:
            for other in [candidates[0], candidates[len(candidates)//2]]:
                candidates.append(normalize(w * bi + (1 - w) * other))

    best_a = normalize(parent_seq) if parent_seq is not None else candidates[0].copy()
    best_val = obj_val(best_a)

    for c in candidates:
        v = obj_val(c)
        if v < best_val:
            best_val = v
            best_a = c.copy()

    return best_a, best_val

def multi_length_explore(best_seq, budget_s, solver):
    """Try a few nearby lengths and keep the best."""
    n0 = len(best_seq)
    lengths = []
    for delta in [-3000, -2000, -1000, 0, 1000, 2000, 3000, 5000]:
        nl = n0 + delta
        if 5000 <= nl <= 20000:
            lengths.append(nl)
    # Remove duplicates and keep at most 5
    lengths = list(dict.fromkeys(lengths))[:5]

    best_overall = best_seq.copy()
    best_score = obj_val(best_seq)
    t_start = time.monotonic()
    time_per = (budget_s - 5) / max(len(lengths), 1)

    for nl in lengths:
        if time.monotonic() - t_start > budget_s:
            break
        if nl == n0:
            a_trial = best_seq.copy()
        else:
            a_trial = interpolate_seq(best_seq, nl)

        v = obj_val(a_trial)
        if v < best_score:
            best_score = v
            best_overall = a_trial.copy()
            print("    Length {} immediate: obj={:.6f}".format(nl, v))

        t_rem = min(time_per, budget_s - (time.monotonic() - t_start))
        if t_rem > 20:
            a_out, v_out = lp_fw_refine(a_trial, t_rem, solver,
                                         init_K=200, max_K=2000, verbose=False)
            if v_out < best_score:
                best_score = v_out
                best_overall = a_out.copy()
                print("    Length {} LP: obj={:.6f}".format(nl, v_out))

    return best_overall, best_score

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)
    solver = get_solver()
    print("Total budget: {:.1f}s".format(time_budget))

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent: n={} score={:.6f}".format(len(p), best_score))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_init = 10000
        x = np.linspace(0, 1, n_init)
        best_seq = normalize((1.0 - x + 1e-6) ** 1.0)
        best_score = eval_safe(best_seq)
        print("Default init n={} score={:.6f}".format(n_init, best_score))

    # === Phase 1: Parametric init ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    t1 = min(remaining * 0.05, 60.0)
    print("\n=== P1: Parametric Init ===")
    if t1 > 10:
        n_init = len(best_seq)
        a_p, v_p = parametric_init(n_init, best_seq, best_score)
        if v_p < best_score:
            best_score = v_p
            best_seq = a_p
            print("  Init improved: {:.6f}".format(best_score))

    # === Phase 2: LP-FW main ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_lp = min(len(best_seq), 12000)
    a_lp = interpolate_seq(best_seq, n_lp) if n_lp != len(best_seq) else best_seq.copy()
    t2 = min(remaining * 0.35, 500.0)
    print("\n=== P2: LP-FW main (n={}) ===".format(n_lp))
    if t2 > 60:
        a_out, v_out = lp_fw_refine(a_lp, t2, solver, init_K=400, max_K=3500, verbose=True)
        if v_out < best_score:
            sc = eval_safe(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = interpolate_seq(a_out, len(best_seq)) if len(best_seq) != n_lp else a_out
                print("  LP-FW improved: {:.6f}".format(best_score))

    # === Phase 3: Subgradient escape ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 180.0)
    print("\n=== P3: Subgradient (n={}) ===".format(len(best_seq)))
    if t3 > 20:
        a_sub, v_sub = subgradient_step(best_seq, t3, n_top=25)
        sc = eval_safe(a_sub)
        if sc < best_score:
            best_score = sc
            best_seq = a_sub
            print("  SubGrad improved: {:.6f}".format(best_score))

    # === Phase 4: LP-FW second round ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_lp2 = min(len(best_seq), 12000)
    a_lp2 = interpolate_seq(best_seq, n_lp2) if n_lp2 != len(best_seq) else best_seq.copy()
    t4 = min(remaining * 0.25, 400.0)
    print("\n=== P4: LP-FW round 2 (n={}) ===".format(n_lp2))
    if t4 > 60:
        a_out, v_out = lp_fw_refine(a_lp2, t4, solver, init_K=500, max_K=4000, verbose=True)
        if v_out < best_score:
            sc = eval_safe(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = interpolate_seq(a_out, len(best_seq)) if len(best_seq) != n_lp2 else a_out
                print("  LP-FW2 improved: {:.6f}".format(best_score))

    # === Phase 5: Coord descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cd = min(len(best_seq), 12000)
    a_cd = interpolate_seq(best_seq, n_cd) if n_cd != len(best_seq) else best_seq.copy()
    t5 = min(remaining * 0.15, 220.0)
    print("\n=== P5: Coord Descent (n={}) ===".format(n_cd))
    if t5 > 20:
        a_out, v_out = coord_descent(a_cd, t5, top_lags=12)
        sc = eval_safe(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = interpolate_seq(a_out, len(best_seq)) if len(best_seq) != n_cd else a_out
            print("  Coord improved: {:.6f}".format(best_score))

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 200.0)
    print("\n=== P6: Length Exploration ===")
    if t6 > 30:
        a_best_len, sc_best_len = multi_length_explore(best_seq, t6, solver)
        if sc_best_len < best_score:
            best_score = sc_best_len
            best_seq = a_best_len
            print("  Length explore improved: {:.6f}".format(best_score))

    # === Phase 7: Final LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_fin = min(len(best_seq), 10000)
    a_fin = interpolate_seq(best_seq, n_fin) if n_fin != len(best_seq) else best_seq.copy()
    t7 = min(remaining * 0.2, 250.0)
    print("\n=== P7: Final LP-FW (n={}) ===".format(n_fin))
    if t7 > 30:
        a_out, v_out = lp_fw_refine(a_fin, t7, solver, init_K=600, max_K=4000, verbose=True)
        if v_out < best_score:
            sc = eval_safe(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = interpolate_seq(a_out, len(best_seq)) if len(best_seq) != n_fin else a_out
                print("  Final LP improved: {:.6f}".format(best_score))

    # === Phase 8: Final coord polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15 and best_seq is not None:
        n_pol = min(len(best_seq), 10000)
        a_pol = interpolate_seq(best_seq, n_pol) if n_pol != len(best_seq) else best_seq.copy()
        t8 = remaining - 10
        print("\n=== P8: Final Polish ===")
        if t8 > 10:
            a_out, v_out = coord_descent(a_pol, t8, top_lags=10)
            sc = eval_safe(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = interpolate_seq(a_out, len(best_seq)) if len(best_seq) != n_pol else a_out
                print("  Final polish: {:.6f}".format(best_score))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.6f} n={} t={:.1f}s".format(best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: fallback")
        n_fb = 10000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize((1.0 - x + 1e-6) ** 1.0)
        best_score = eval_safe(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in np.asarray(best_seq, dtype=np.float64)]
    return result
