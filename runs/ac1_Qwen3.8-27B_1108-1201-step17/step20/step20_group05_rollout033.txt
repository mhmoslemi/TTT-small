    v = np.where(np.isfinite(v), v, 0.0)
    # Bisection for tau
    lo = float(np.min(v)) - 1.0 / n - 1e-10
    hi = float(np.max(v))
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    tau = (lo + hi) / 2.0
    a = np.maximum(v - tau, 0.0)
    a = a / float(a.sum())
    return a


def compute_subgrad(a, active_idx):
    """
    Compute subgradient of max_k (a*a)_k using active set.
    grad[i] = 2 * sum_{k in active} p[k] * a[k-i]
    """
    n = a.size
    size = 2 * n - 1
    K = len(active_idx)
    if K == 0:
        return np.zeros(n)
    p = np.zeros(size)
    p[active_idx] = 1.0 / K
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    grad = 2.0 * conv[2 * n - 2 : n - 2 : -1]
    return np.where(np.isfinite(grad), grad, 0.0)


def nesterov_subgrad(a_init, budget, top_k=200, eta0=0.02):
    """
    Nesterov-accelerated subgradient method on max(conv(a,a)).
    Returns best sequence found.
    """
    a = norm_sum1(a_init)
    n = a.size
    c = fft_conv(a)
    M = max_conv(a, c)
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()

    # Nesterov state
    x = a.copy()
    y = a.copy()
    t_val = 1.0
    active = np.argsort(c)[::-1][:top_k]

    for it in range(100000):
        if time.monotonic() - t0 > budget - 2:
            break
        # Compute subgradient at y
        g = compute_subgrad(y, active)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        # Step size: eta = eta0 / sqrt(it+1)
        eta = eta0 / math.sqrt(it + 1)
        # Cap the step to avoid wild moves
        step = eta * g / g_norm
        step = np.minimum(step, 0.1 * y + 0.01)

        x_new = y - step
        x_new = project_simplex(x_new)

        # Nesterov update
        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        beta = (t_val - 1.0) / t_new
        y_new = x_new + beta * (x_new - x)
        y_new = project_simplex(y_new)

        # Evaluate
        c_new = fft_conv(y_new)
        M_new = max_conv(y_new, c_new)

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = y_new.copy()
            if it < 10 or it % 200 == 0:
                print("    Nesterov it={} M={:.8f} best_score={:.6f}".format(
                    it, M_new, 2.0 * n * best_M))

        # Update active set every 50 iterations
        if it % 50 == 0 and it > 0:
            active = np.argsort(c_new)[::-1][:top_k]

        x = x_new
        y = y_new
        c = c_new
        M = M_new
        t_val = t_new

    return best_a


def lse_objective(a, beta, mask=None):
    """Log-sum-exp smoothed objective and gradient."""
    n = a.size
    c = fft_conv(a)
    if mask is not None:
        cv = c[mask]
    else:
        cv = c
    m = float(np.max(cv))
    shifted = beta * (cv - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    f = m + math.log(se) / beta

    # Gradient via cross-correlation
    if mask is not None:
        pf = np.zeros(2 * n - 1)
        pf[mask] = e / se
    else:
        pf = e / se
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(pf[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    grad = 2.0 * conv[2 * n - 2 : n - 2 : -1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return f, grad


def lse_objective_simplex(u, beta, mask=None, ref=0):
    """Objective in softmax parameterization for L-BFGS."""
    n = u.size + 1
    a = np.zeros(n)
    if ref > 0:
        a[:ref] = np.exp(u[:ref])
    a[ref] = 1.0
    if ref < n - 1:
        a[ref + 1:] = np.exp(u[ref + 1:])
    s = float(a.sum())
    if s < 1e-15:
        return 1e100, np.zeros(u.size)
    a = a / s
    f, ga = lse_objective(a, beta, mask)
    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)
    gu = np.where(np.isfinite(gu), gu, 0.0)
    gv = np.zeros(u.size)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]
    return f, gv


def run_lbfgs(a_init, beta_list, mask, maxfun_each=300, verbose=False):
    """Run L-BFGS with a sequence of beta values (increasing smoothness)."""
    a = norm_sum1(a_init)
    n = a.size
    if n <= 1:
        return a
    ref = int(np.argmax(a))
    v = np.zeros(n - 1)
    for i in range(n):
        if i == ref:
            continue
        idx = i if i < ref else i - 1
        v[idx] = math.log(max(float(a[i]), 1e-300)) - math.log(max(float(a[ref]), 1e-300))

    for beta in beta_list:
        t0 = time.monotonic()
        try:
            res = scipy.optimize.minimize(
                lse_objective_simplex, v,
                args=(beta, mask, ref),
                method='L-BFGS-B', jac=True,
                options={'maxfun': maxfun_each, 'ftol': 1e-12, 'gtol': 1e-7, 'disp': False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                v = res.x
        except Exception:
            pass
        # Reconstruct a
        a = np.zeros(n)
        a[ref] = 1.0
        if ref > 0:
            a[:ref] = np.exp(v[:ref])
        if ref < n - 1:
            a[ref + 1:] = np.exp(v[ref:])
        a = norm_sum1(a)
        if verbose:
            c = fft_conv(a)
            print("    L-BFGS beta={} M={:.8f} score={:.6f} t={:.1f}s".format(
                beta, max_conv(a, c), 2.0 * n * max_conv(a, c), time.monotonic() - t0))
    return a


def coordinate_polish(a_init, budget, top_k=30, verbose=False):
    """Focused coordinate descent: transfer mass from high-pressure to low-pressure positions."""
    a = norm_sum1(a_init)
    n = a.size
    c = fft_conv(a)
    M = max_conv(a, c)
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0

    while time.monotonic() - t0 < budget - 2 and stall < 200:
        # Get active peaks
        active = np.argsort(c)[::-1][:top_k]
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / len(active)

        # Compute pressure (subgradient)
        g = compute_subgrad(a, active)

        # Find good source (high a, high pressure) and destination (low a, low pressure)
        src_score = a * g
        dst_score = -a * g + 0.01 * a

        src_candidates = np.argsort(src_score)[::-1][:8]
        dst_candidates = np.argsort(dst_score)[:12]

        improved = False
        best_move = None
        best_m = M

        for s_idx in src_candidates:
            s = int(s_idx)
            if a[s] < 1e-10:
                continue
            for d_idx in dst_candidates:
                d = int(d_idx)
                if d == s:
                    continue
                for frac in (0.005, 0.01, 0.02, 0.04):
                    tmove = a[s] * frac
                    if tmove < 1e-14:
                        continue
                    tmove = min(tmove, a[s] * 0.95)
                    # Rank-1 update of convolution
                    c_new = c.copy()
                    c_new[d:d + n] += 2.0 * tmove * a[s - s + np.arange(n)] if False else 2.0 * tmove * a
                    c_new[s:s + n] -= 2.0 * tmove * a
                    # Correction for the a_d^2 and a_s^2 terms
                    # Actually the rank-1 update: (a-e_s*t+e_d*t)*(a-e_s*t+e_d*t) - a*a
                    # = -2t*a shifted by s + 2t*a shifted by d + t^2*(e_s-e_d)(e_s-e_d)
                    # Let me just recompute properly for small moves
                    a_temp = a.copy()
                    a_temp[s] -= tmove
                    a_temp[d] += tmove
                    a_temp = np.maximum(a_temp, 0)
                    a_temp = a_temp / a_temp.sum()
                    c_check = fft_conv(a_temp)
                    m_check = max_conv(a_temp, c_check)
                    if m_check < best_m - 1e-15:
                        best_m = m_check
                        best_move = (s, d, tmove)
                        improved = True

        if best_move is not None:
            s, d, tmove = best_move
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0)
            a = a / a.sum()
            c = fft_conv(a)
            M = max_conv(a, c)
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
                if verbose and stall == 0:
                    pass
            stall = 0
        else:
            stall += 1
            if stall % 20 == 0:
                # Perturb
                pert = rng.normal(0, 0.02 * best_M, n)
                a_pert = best_a + pert
                a_pert = np.maximum(a_pert, 1e-12)
                a_pert = a_pert / a_pert.sum()
                c = fft_conv(a_pert)
                M = max_conv(a_pert, c)
                a = a_pert

    return best_a


def interpolate_seq(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)


def make_spectral_init(n, mode=1, alpha=0.5, seed=0):
    """Create a spectral-pattern initialization."""
    x = np.linspace(0.0, 1.0, n)
    if mode == 0:
        # Uniform
        a = np.ones(n)
    elif mode == 1:
        # 1 + alpha*cos(2*pi*x)
        a = 1.0 + alpha * np.cos(2.0 * np.pi * x)
    elif mode == 2:
        # 1 + alpha*cos(4*pi*x)
        a = 1.0 + alpha * np.cos(4.0 * np.pi * x)
    elif mode == 3:
        # Gaussian mixture
        c = np.random.default_rng(seed)
        centers = c.uniform(0.1, 0.9, 2)
        a = np.zeros(n)
        for ci in centers:
            a += np.exp(-50.0 * (x - ci) ** 2)
    elif mode == 4:
        # Sine power
        a = np.abs(np.sin(np.pi * x)) ** 2.0
    elif mode == 5:
        # 1 + alpha*cos(2*pi*k*x) for random k
        k = np.random.default_rng(seed).integers(2, max(3, n // 500))
        a = 1.0 + alpha * np.cos(2.0 * np.pi * k * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def run_single_search(a_init, budget, n, verbose=False):
    """Run a single optimization search starting from a_init."""
    a = norm_sum1(a_init)
    current_n = a.size
    best_a = a.copy()
    best_score = 2.0 * current_n * max_conv(a)
    t0 = time.monotonic()

    # Phase 1: Nesterov subgradient (30% of budget)
    if budget > 30:
        nb = 0.3 * budget
        if verbose:
            print("  [Nesterov] budget={:.1f}s".format(nb))
        a1 = nesterov_subgrad(a, nb, top_k=min(200, 2 * current_n - 1), eta0=0.015)
        s1 = 2.0 * current_n * max_conv(a1)
        if s1 < best_score:
            best_score = s1
            best_a = a1
            a = a1

    # Phase 2: L-BFGS with increasing smoothness (40% of budget)
    t_remaining = budget - (time.monotonic() - t0)
    if t_remaining > 20 and SCIPY_OK:
        lb_budget = 0.4 * t_remaining
        if verbose:
            print("  [L-BFGS] budget={:.1f}s".format(lb_budget))
        per_beta = max(100, int(lb_budget / 5.0 / 0.05))
        beta_list = [20.0, 60.0, 200.0, 600.0, 1500.0]
        # Determine top mask for sharper betas
        c = fft_conv(a)
        top_k_mask = max(50, (2 * current_n - 1) // 15)
        mask = np.zeros(2 * current_n - 1, dtype=bool)
        top_idx = np.argsort(c)[::-1][:top_k_mask]
        mask[top_idx] = True

        a2 = run_lbfgs(a, beta_list, mask, maxfun_each=per_beta, verbose=verbose)
        s2 = 2.0 * current_n * max_conv(a2)
        if s2 < best_score:
            best_score = s2
            best_a = a2
            a = a2

    # Phase 3: Coordinate polish (30% of budget)
    t_remaining = budget - (time.monotonic() - t0)
    if t_remaining > 15:
        cp_budget = 0.3 * t_remaining
        if verbose:
            print("  [Coord] budget={:.1f}s".format(cp_budget))
        a3 = coordinate_polish(a, cp_budget, top_k=25, verbose=verbose)
        s3 = 2.0 * current_n * max_conv(a3)
        if s3 < best_score:
            best_score = s3
            best_a = a3
            a = a3

    # Final L-BFGS polish if time allows
    t_remaining = budget - (time.monotonic() - t0)
    if t_remaining > 20 and SCIPY_OK:
        c = fft_conv(a)
        top_k_mask = max(30, (2 * current_n - 1) // 25)
        mask_final = np.zeros(2 * current_n - 1, dtype=bool)
        mask_final[np.argsort(c)[::-1][:top_k_mask]] = True
        a4 = run_lbfgs(a, [2000.0, 5000.0], mask_final, maxfun_each=150, verbose=False)
        s4 = 2.0 * current_n * max_conv(a4)
        if s4 < best_score:
            best_score = s4
            best_a = a4

    return best_a, best_score


def propose_candidate(budget_s=1800.0):
    """Main entry point: multi-restart search for optimal sequence."""
    t_start = time.monotonic()
    total_budget = max(100.0, min(float(budget_s) - 60.0, 1740.0))

    print("Starting multi-restart search, total_budget={:.1f}s".format(total_budget))

    # Get parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        np.clip(p, 0.0, 1000.0, out=p)
        if p.ndim != 1:
            p = p.ravel()
        if p.size > 0 and p.sum() > 1e-10:
            parent = norm_sum1(p)
    except Exception:
        pass

    if parent is not None:
        print("Parent: n={} score={:.6f}".format(parent.size, 2.0 * parent.size * max_conv(parent)))
    else:
        parent = make_spectral_init(5000, mode=0)
        print("No parent, using uniform n=5000")

    # Initialize best
    best_seq = parent.copy()
    best_score = 2.0 * parent.size * max_conv(parent)

    # Generate diverse initializations
    n_base = parent.size if parent.size <= 9000 else 5000
    inits = []

    # 1: Parent itself
    if parent is not None and parent.size <= 9000:
        inits.append(("parent", parent.copy()))
    elif parent is not None:
        inits.append(("parent_interp", interpolate_seq(parent, n_base)))

    # 2: Parent + noise
    if parent is not None:
        rng = np.random.default_rng(123)
        noisy = parent.copy()
        noisy += rng.normal(0, 0.3, noisy.size)
        inits.append(("parent_noise", norm_sum1(noisy)))

    # 3: Spectral patterns at base length
    spec_patterns = [
        ("spec_cos1", make_spectral_init(n_base, mode=1, alpha=0.7)),
        ("spec_cos2", make_spectral_init(n_base, mode=5, alpha=0.6, seed=7)),
        ("spec_gauss", make_spectral_init(n_base, mode=3, seed=42)),
        ("spec_sine", make_spectral_init(n_base, mode=4)),
    ]
    inits.extend(spec_patterns)

    # Assign time budgets: give more time to parent-based inits
    # Strategy: top 3 get full search, remaining get shorter
    n_inits = len(inits)
    # Sort by initial quality (proxy score)
    scored = []
    for name, a in inits:
        sc = 2.0 * a.size * max_conv(a)
        scored.append((sc, name, a))
    scored.sort(key=lambda x: x[0])

    # Allocate time: best initial gets 40%, next two get 25% each, rest get 5% each (capped)
    if n_inits == 1:
        budgets = [total_budget * 0.9]
    else:
        weights = [0.35, 0.25, 0.20] + [max(0.03, 0.2 / max(1, n_inits - 3))] * max(0, n_inits - 3)
        total_w = sum(weights)
        budgets = [total_budget * 0.85 * w / total_w for w in weights[:n_inits]]
        # Cap individual budgets
        budgets = [min(b, total_budget * 0.6) for b in budgets]

    # Ensure total doesn't exceed budget
    total_alloc = sum(budgets)
    if total_alloc > total_budget * 0.9:
        scale = total_budget * 0.9 / total_alloc
        budgets = [b * scale for b in budgets]

    # Run searches
    for i, ((sc, name, a_init), b) in enumerate(zip(scored, budgets)):
        t_left = total_budget - (time.monotonic() - t_start)
        if t_left < 30:
            break
        b = min(b, t_left - 10)
        if b < 20:
            continue
        print("\n--- Search {} ({}): init_score={:.6f} budget={:.1f}s t_remaining={:.1f}s".format(
            i + 1, name, sc, b, t_left))

        a_out, sc_out = run_single_search(a_init, b, a_init.size, verbose=(i < 2))

        if sc_out < best_score:
            best_score = sc_out
            best_seq = a_out.copy()
            print("  *** NEW BEST: {:.8f} (n={})".format(sc_out, best_seq.size))

    # Final: try a different length on the best found
    t_left = total_budget - (time.monotonic() - t_start)
    if t_left > 120:
        print("\n--- Length exploration (t_left={:.1f}s)".format(t_left))
        for n_try in [best_seq.size + 500, best_seq.size - 500, int(best_seq.size * 1.3), int(best_seq.size * 0.7)]:
            t_left = total_budget - (time.monotonic() - t_start)
            if t_left < 60:
                break
            n_try = int(max(1000, min(12000, n_try)))
            if abs(n_try - best_seq.size) < 100:
                continue
            a_interp = interpolate_seq(best_seq, n_try)
            b = min(80.0, t_left - 20)
            a_out, sc_out = run_single_search(a_interp, b, n_try, verbose=False)
            if sc_out < best_score:
                best_score = sc_out
                best_seq = a_out.copy()
                print("  *** NEW BEST (len probe n={}): {:.8f}".format(n_try, sc_out))

    # Final validation
    final_val = safe_eval(best_seq)
    if math.isfinite(final_val) and final_val < best_score:
        best_score = final_val

    arr = np.asarray(best_seq, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if arr.size == 0 or s < 0.01:
        arr = make_spectral_init(5000, mode=0)
        best_score = 2.0 * arr.size * max_conv(arr)
    else:
        arr = arr / s

    elapsed = time.monotonic() - t_start
    print("\nFINAL: score={:.8f} n={} elapsed={:.1f}s".format(best_score, arr.size, elapsed))
    return [float(x) for x in arr]
