Multi-scale optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS on log-sum-exp smoothed objectives, Frank-Wolfe game directions,
and multi-scale coordinate polishing.
"""

import numpy as np
import time
import math

try:
    import scipy.optimize
    SCIPY_OK = True
except:
    SCIPY_OK = False

def next_pow2(x):
    x = int(x)
    if x <= 1: return 1
    return 1 << (x-1).bit_length()

def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1: arr = arr.ravel()
    n = arr.size
    if n == 0: return np.ones(1)
    arr[~np.isfinite(arr)] = 0.0
    np.maximum(arr, 0, out=arr)
    s = arr.sum()
    if s < 1e-15: return np.ones(n)/n
    return arr/s

def fft_conv(a):
    n = a.size
    if n == 0: return np.array([])
    size = 2*n-1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A*A, N)[:size]
    return np.where(np.isfinite(c) & (c > 0), c, 0.0)

def fft_cross(a, b):
    n = a.size
    if n == 0: return np.array([])
    if b.size != n: b = np.resize(b, n)
    size = 2*n-1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A*B, N)[:size]
    return np.where(np.isfinite(c) & (c > 0), c, 0.0)

def lse_objective(a, beta, top_k=None):
    """Log-sum-exp smoothed max of convolution."""
    n = a.size
    size = 2*n-1
    A = np.fft.rfft(a, next_pow2(3*n-2))
    c = np.fft.irfft(A*A, next_pow2(3*n-2))[:size]
    c = np.where(np.isfinite(c) & (c > 0), c, 0.0)
    
    if top_k is not None and top_k < size:
        idx = np.argpartition(c, -top_k)[-top_k:]
        cv = c[idx]
    else:
        cv = c
    
    m = cv.max()
    shifted = np.clip(beta*(cv - m), -700, 700)
    e = np.exp(shifted)
    se = e.sum()
    if se <= 0 or not np.isfinite(se):
        return m, c, np.zeros(n)
    p = e/se
    
    # Gradient: 2 * cross_conv(p_full, a)
    if top_k is not None and top_k < size:
        p_full = np.zeros(size)
        p_full[idx] = p
    else:
        p_full = p
    
    N2 = next_pow2(3*n-2)
    A2 = np.fft.rfft(a, N2)
    Pf = np.fft.rfft(p_full[::-1], N2)
    conv = np.fft.irfft(A2*Pf, N2)[:3*n-2]
    grad = 2.0 * conv[n-1:2*n-1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    
    f = m + math.log(se)/beta
    return f, c, grad

    n = u.size
    if n == 0: return np.ones(1)
    if not np.all(np.isfinite(u)): return np.ones(n)/n
    um = u.max()
    e = np.exp(u - um)
    s = e.sum()
    if s <= 0 or not np.isfinite(s): return np.ones(n)/n
    return e/s

def log_param(a, ref=None):
    """Convert a to log-softmax parameters (dropping one coordinate)."""
    n = a.size
    if n <= 1: return np.array([]), 0
    if ref is None: ref = int(np.argmax(a))
    ref = int(ref)
    a_ref = max(a[ref], 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0: v = loga[1:]
    elif ref == n-1: v = loga[:-1]
    else: v = np.concatenate([loga[:ref], loga[ref+1:]])
    v = np.where(np.isfinite(v), v, -700.0)
    return v, ref

def from_log_param(v, ref):
    n = v.size + 1
    u = np.zeros(n)
    ref = int(ref)
    if not (0 <= ref < n): ref = n-1
    if ref > 0: u[:ref] = v[:ref]
    if ref < n-1: u[ref+1:] = v[ref:]
    return softmax(u)

def lse_obj_v(v, beta, ref, top_k=None):
    """Objective in log-parameter space."""
    n = v.size + 1
    ref = int(ref)
    u = np.zeros(n)
    if ref > 0: u[:ref] = v[:ref]
    if ref < n-1: u[ref+1:] = v[ref:]
    a = softmax(u)
    f, c, ga = lse_objective(a, beta, top_k)
    mean = np.dot(a, ga)
    gu = a * (ga - mean)
    gv = np.zeros(v.size)
    if ref > 0: gv[:ref] = gu[:ref]
    if ref < n-1: gv[ref:] = gu[ref+1:]
    if not np.all(np.isfinite(gv)): gv = np.zeros(v.size)
    return f, gv

def run_lbfgs(a0, beta, maxfun=400, top_k=None):
    n = a0.size
    if n <= 1 or not SCIPY_OK or maxfun <= 0:
        return a0
    v0, ref = log_param(a0)
    try:
        res = scipy.optimize.minimize(
            lse_obj_v, v0,
            args=(beta, ref, top_k),
            method='L-BFGS-B', jac=True,
            options={'maxfun': maxfun, 'maxiter': maxfun+50, 'ftol': 1e-14, 'gtol': 1e-9, 'disp': False}
        )
        if hasattr(res, 'x') and res.x.size == v0.size and np.all(np.isfinite(res.x)):
            return from_log_param(res.x, ref)
    except:
        pass
    return a0

def game_direction(f, active_idx, n_iters=30, seed=0):
    """Solve the zero-sum game for FW direction."""
    f = norm_sum1(f)
    n = f.size
    size = 2*n-1
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K <= 0: return np.ones(n)/n
    
    rng = np.random.default_rng(seed)
    logits = np.zeros(K) + rng.normal(0, 0.05, K)
    
    for _ in range(n_iters):
        # Player 1: choose distribution over active lag positions
        loc = np.where(np.isfinite(logits), logits, -1000)
        loc -= loc.max()
        w = np.exp(loc)
        sw = w.sum()
        if sw <= 0: w = np.ones(K); sw = K
        lam = w/sw
        
        pf = np.zeros(size)
        pf[active_idx] = lam
        
        # Pressure on a
        N2 = next_pow2(3*n-2)
        A = np.fft.rfft(f, N2)
        Pf = np.fft.rfft(pf[::-1], N2)
        conv = np.fft.irfft(A*Pf, N2)[:3*n-2]
        pressure = 2.0 * conv[n-1:2*n-1][::-1]
        pressure = np.where(np.isfinite(pressure) & (pressure > 0), pressure, 0.0)
        
        # Player 2: choose distribution over positions
        pmx = pressure.max()
        pmn = pressure.min()
        if pmx - pmn < 1e-30:
            g = np.ones(n)/n
        else:
            scaled = (pressure - pmn)/(pmx - pmn)
            g = np.exp(-6.0 * scaled)
            g = np.where(np.isfinite(g), g, 0.0) + 1e-12
            g = g / g.sum()
            g = 0.95*g + 0.05*(np.ones(n)/n)
            g = g / g.sum()
        
        # Update logits
        h = fft_cross(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmac = ha.max()
        hmin = ha.min()
        if hmac - hmin > 1e-30:
            eta = 8.0/(hmac - hmin)
            logits += eta*(ha - ha.mean())
    
    return g

def coord_move(c, a, s, d, t):
    """Apply a single coordinate transfer and return new convolution."""
    n = a.size
    c_new = c.copy()
    # Moving t from s to d changes a[s]-=t, a[d]+=t
    # conv[k] changes by: -t*a[k-s] - t*a[k-d] + t*a[d-s] + t*a[s-d] (for valid indices)
    # Plus second order: +t^2 at positions 2s, 2d, s+d (with appropriate signs)
    # Actually let's just use the formula directly
    # delta_a = -t at s, +t at d
    # delta_conv[k] = sum_i delta_a[i]*a[k-i] + a[i]*delta_a[k-i]
    # = -t*a[k-s] + t*a[k-d] (from first term, if indices valid)
    #   + (-t)*a[k-s] + t*a[k-d] (from second term, if indices valid)
    # Wait this is getting complicated. Let me just use the direct formula.
    
    # Actually the simplest: delta_conv[k] = conv(a+delta, a+delta)[k] - conv(a,a)[k]
    # = 2*sum_i delta[i]*a[k-i] + sum_i delta[i]*delta[k-i]
    
    # For single move: delta[s] = -t, delta[d] = +t
    # First order: 2*(-t)*a[k-s] + 2*(+t)*a[k-d] = 2t*(a[k-d] - a[k-s]) for valid k-s, k-d in [0,n)
    # Second order: (-t)*(-t)*1[s+... ] ... this is 0 except at k=2s (contribution t^2), k=2d (contribution t^2), k=s+d (contribution -2t^2)
    
    # First order
    k_range = max(0, s-n+1)  # smallest k such that k-s >= 0
    k_end = min(2*n-1, s+n-1)  # largest k such that k-s < n
    ks = np.arange(max(0, s), min(2*n-1, s+n))  # k such that k-s in [0, n-1]
    if ks.size > 0:
        c_new[ks] -= 2.0*t*a[ks-s]
    
    kd = np.arange(max(0, d), min(2*n-1, d+n))  # k such that k-d in [0, n-1]
    if kd.size > 0:
        c_new[kd] += 2.0*t*a[kd-d]
    
    # Second order
    if 2*s < 2*n-1: c_new[2*s] += t*t
    if 2*d < 2*n-1: c_new[2*d] += t*t
    if s+d < 2*n-1: c_new[s+d] -= 2.0*t*t
    
    return np.where(np.isfinite(c_new), c_new, 0.0)

def coordinate_search(a, budget, verbose=False):
    """Local search using single-coordinate transfers."""
    t0 = time.monotonic()
    a = norm_sum1(a)
    n = a.size
    c = fft_conv(a)
    M = c.max()
    best_a = a.copy()
    best_M = M
    stall = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < budget - 2 and stall < 200:
        # Find top active peaks
        top_k = min(30, 2*n-1)
        top_idx = np.argpartition(-c, top_k-1)[:top_k] if top_k < 2*n-1 else np.arange(2*n-1)
        
        # Compute pressure
        p = np.zeros(2*n-1)
        p[top_idx] = 1.0/top_idx.size
        N2 = next_pow2(3*n-2)
        A = np.fft.rfft(a, N2)
        Pf = np.fft.rfft(p[::-1], N2)
        conv = np.fft.irfft(A*Pf, N2)[:3*n-2]
        pressure = 2.0 * conv[n-1:2*n-1][::-1]
        pressure = np.where(np.isfinite(pressure) & (pressure > 0), pressure, 0.0)
        
        # Try moves from high-pressure to low-pressure positions
        best_move = None
        best_new_M = M
        
        # Source candidates: high pressure + high value
        src_score = pressure * (1.0 + a*n)
        src_cands = np.argpartition(-src_score, min(10, n-1))[:10]
        
        # Destination candidates: low pressure
        dst_score = pressure + 1e-8*a
        dst_cands = np.argpartition(dst_score, min(15, n-1))[:15]
        
        for s in src_cands:
            s = int(s)
            if a[s] < 1e-12: continue
            for d in dst_cands:
                d = int(d)
                if d == s: continue
                for frac in (0.003, 0.01, 0.03):
                    t = a[s]*frac
                    if t < 1e-14: continue
                    t = min(t, a[s]*0.9)
                    c_new = coord_move(c, a, s, d, t)
                    m_new = c_new.max()
                    if m_new < best_new_M:
                        best_new_M = m_new
                        best_move = (s, d, t)
        
        # Also try random moves
        for _ in range(20):
            s = int(rng.integers(n))
            d = int(rng.integers(n))
            if s == d or a[s] < 1e-12: continue
            frac = rng.uniform(0.001, 0.02)
            t = a[s]*frac
            if t < 1e-14: continue
            c_new = coord_move(c, a, s, d, t)
            m_new = c_new.max()
            if m_new < best_new_M:
                best_new_M = m_new
                best_move = (s, d, t)
        
        if best_move is not None:
            s, d, t = best_move
            a[s] -= t
            a[d] += t
            np.maximum(a, 0, out=a)
            a /= a.sum()
            c = fft_conv(a)
            M = c.max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and (stall == 0):
                print(f"  Coord: M={M:.8f} score={2*n*M:.6f}")
        else:
            stall += 1
    
    return best_a

def fw_search(a0, budget, active_max=2000, verbose=False):
    """Frank-Wolfe with game-theoretic direction."""
    t0 = time.monotonic()
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = c.max()
    best_a = a.copy()
    best_M = M
    
    no_improve = 0
    t_iter = 0
    while time.monotonic() - t0 < budget - 5 and no_improve < 60:
        t_iter += 1
        time_left = budget - (time.monotonic() - t0)
        if time_left < 20: break
        
        # Update active set
        top_k = min(active_max, 2*n-1)
        active = np.argpartition(-c, top_k-1)[:top_k] if top_k < 2*n-1 else np.arange(2*n-1)
        
        # Get direction
        g = game_direction(a, active, n_iters=20, seed=1000+t_iter)
        
        # Line search
        cg = fft_conv(g)
        fg = fft_cross(a, g)
        
        best_alpha = 0
        best_m = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]:
            one = 1-alpha
            val = one*one*c + 2*alpha*one*fg + alpha*alpha*cg
            m = val.max()
            if m < best_m:
                best_m = m
                best_alpha = alpha
        
        if best_alpha > 0:
            a = (1-best_alpha)*a + best_alpha*g
            a = np.maximum(a, 0)
            a = a/a.sum()
            c = fft_conv(a)
            M = c.max()
            no_improve = 0
            if M < best_M:
                best_M = M
                best_a = a.copy()
                if verbose and (t_iter <= 5 or t_iter % 20 == 0):
                    print(f"  FW t={t_iter} M={M:.8f} score={2*n*M:.6f} alpha={best_alpha:.3f}")
        else:
            no_improve += 1
            if no_improve % 10 == 9:
                # Perturb
                rng = np.random.default_rng(9999+t_iter)
                a = best_a + rng.normal(0, 0.01, n)*best_a
                a = np.maximum(a, 1e-12)
                a = a/a.sum()
                c = fft_conv(a)
                M = c.max()
                no_improve = 0
    
    return best_a

def default_seq(n=5000):
    x = np.linspace(0, 1, n)
    return norm_sum1(np.sin(np.pi*x)**2)

def interpolate(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == new_n: return norm_sum1(a)
    old_x = np.linspace(0, 1, a.size)
    new_x = np.linspace(0, 1, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0, out=y)
    return norm_sum1(y)

"""

Combines log-sum-exp L-BFGS with adaptive active sets, Frank-Wolfe game
directions, bump-shifting coordinate moves, and multi-scale refinement.

import numpy as np
import time
import math

try:
    import scipy.optimize
    SCIPY_OK = True
except Exception:
    SCIPY_OK = False


def _next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _norm(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    arr[~np.isfinite(arr)] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def _fft_conv(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2 * n - 1]
    return np.where(np.isfinite(c) & (c > 0.0), c, 0.0)


def _fft_cross(a, b):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if b.size != n:
        b = np.resize(b, n)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:2 * n - 1]
    return np.where(np.isfinite(c) & (c > 0.0), c, 0.0)


def _cross_p_a(p, a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    total = 3 * n - 2
    N = _next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    out = conv[n - 1:2 * n - 1][::-1]
    return 2.0 * np.where(np.isfinite(out), out, 0.0)


def _lse_obj(a, beta, top_idx=None):
    n = a.size
    size = 2 * n - 1
    N = _next_pow2(3 * n - 2)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    if top_idx is not None and top_idx.size > 0 and top_idx.size < size:
        cv = c[top_idx]
    else:
        cv = c
    m = float(cv.max())
    if cv.size == 0:
        return m, c, np.zeros(n)
    shifted = np.clip(beta * (cv - m), -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0 or not np.isfinite(se):
        return m, c, np.zeros(n)
    p = e / se
    if top_idx is not None and top_idx.size > 0 and top_idx.size < size:
        p_full = np.zeros(size, dtype=np.float64)
        p_full[top_idx] = p
    else:
        p_full = p
    grad = _cross_p_a(p_full, a)
    f = m + math.log(se) / beta
    if not np.isfinite(f):
        f = 1e100
    return f, c, grad


def _softmax(u):
    n = u.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(u)):
        return np.ones(n, dtype=np.float64) / float(n)
    um = float(u.max())
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def _log_param(a, ref=None):
    n = a.size
    if n <= 1:
        return np.array([], dtype=np.float64), 0
    if ref is None:
        ref = int(np.argmax(a))
    ref = int(ref)
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.where(np.isfinite(v), v, -700.0)
    return v, ref


def _from_log(v, ref):
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    return _softmax(u)


def _obj_v(v, beta, ref, top_idx):
    n = v.size + 1
    ref = int(ref)
    u = np.zeros(n, dtype=np.float64)
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    a = _softmax(u)
    f, c, ga = _lse_obj(a, beta, top_idx)
    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)
    gv = np.zeros(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]
    if not np.all(np.isfinite(gv)):
        gv = np.zeros(v.size, dtype=np.float64)
    return f, gv


def _lbfgs(a0, beta, maxfun, top_idx=None):
    n = a0.size
    if n <= 1 or not SCIPY_OK or maxfun <= 0:
        return a0
    v0, ref = _log_param(a0)
    try:
        res = scipy.optimize.minimize(
            _obj_v, v0, args=(beta, ref, top_idx),
            method='L-BFGS-B', jac=True,
            options={'maxfun': int(maxfun), 'maxiter': int(maxfun) + 50,
                     'ftol': 1e-14, 'gtol': 1e-9, 'disp': False}
        )
        if hasattr(res, 'x') and res.x.size == v0.size and np.all(np.isfinite(res.x)):
            out = _from_log(res.x, ref)
            return _norm(out)
    except Exception:
        pass
    return a0

def _game_dir(f, active, iters=25, seed=0):
    f = _norm(f)
    n = f.size
    size = 2 * n - 1
    active = np.asarray(active, dtype=int)
    K = active.size
    if K <= 0:
        return np.ones(n, dtype=np.float64) / float(n)
    rng = np.random.default_rng(seed)
    logits = rng.normal(0.0, 0.05, K)
    g = np.ones(n, dtype=np.float64) / float(n)
    for _ in range(max(1, iters)):
        loc = np.where(np.isfinite(logits), logits, -1000.0)
        loc -= float(loc.max())
        w = np.exp(loc)
        sw = float(w.sum())
        if sw <= 0.0:
            w = np.ones(K, dtype=np.float64)
            sw = float(K)
        lam = w / sw
        pf = np.zeros(size, dtype=np.float64)
        pf[active] = lam
        pressure = _cross_p_a(pf, f)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)
        pmx = float(pressure.max())
        pmn = float(pressure.min())
        if pmx - pmn < 1e-30:
            g = np.ones(n, dtype=np.float64) / float(n)
        else:
            scaled = (pressure - pmn) / (pmx - pmn)
            g = np.exp(-7.0 * scaled)
            g = np.where(np.isfinite(g), g, 0.0) + 1e-12
            g /= float(g.sum())
            g = 0.95 * g + 0.05 * (np.ones(n, dtype=np.float64) / float(n))
            g /= float(g.sum())
        h = _fft_cross(f, g)
        ha = h[active]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmac = float(ha.max())
        hmin = float(ha.min())
        if hmac - hmin > 1e-30:
            eta = 10.0 / (hmac - hmin)
            logits += eta * (ha - float(ha.mean()))
    return g


def _coord_move(c, a, s, d, t):
    n = a.size
    c_new = c.copy()
    ks = np.arange(max(0, s), min(2 * n - 1, s + n))
    if ks.size > 0:
        c_new[ks] -= 2.0 * t * a[ks - s]
    kd = np.arange(max(0, d), min(2 * n - 1, d + n))
    if kd.size > 0:
        c_new[kd] += 2.0 * t * a[kd - d]
    if 2 * s < 2 * n - 1:
        c_new[2 * s] += t * t
    if 2 * d < 2 * n - 1:
        c_new[2 * d] += t * t
    if s + d < 2 * n - 1:
        c_new[s + d] -= 2.0 * t * t
    return np.where(np.isfinite(c_new), c_new, 0.0)


def _bump_move(c, a, s0, s1, d0, d1, t):
    n = a.size
    c_new = c.copy()
    for s in range(s0, s1 + 1):
        if s < 0 or s >= n:
            continue
        for d in range(d0, d1 + 1):
            if d < 0 or d >= n:
                continue
            if s == d:
                continue
            c_new = _coord_move(c_new, a, s, d, t)
    return c_new

def _fw_search(a0, budget, active_max=2000, verbose=False):

    t_iter = 0
    while time.monotonic() - t0 < budget - 5 and no_improve < 80:
        t_iter += 1
        time_left = budget - (time.monotonic() - t0)
        if time_left < 20:
            break
        top_k = min(active_max, 2 * n - 1)
        if top_k < 2 * n - 1:
            active = np.argpartition(-c, top_k - 1)[:top_k]

        g = _game_dir(a, active, iters=22, seed=5000 + t_iter)
        cg = _fft_conv(g)
        fg = _fft_cross(a, g)
        best_alpha = 0.0
        best_m = M
        for alpha in (0.03, 0.06, 0.1, 0.15, 0.2, 0.3, 0.4, 0.6):
            one = 1.0 - alpha
            val = one * one * c + 2.0 * alpha * one * fg + alpha * alpha * cg

            if m < best_m:

                best_alpha = alpha
        if best_alpha > 0.0:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            s = float(a.sum())
            if s < 1e-15:
                break
            a /= s
            c = _fft_conv(a)
            M = float(c.max())
            no_improve = 0
            if M < best_M:
                best_M = M
                best_a = a.copy()
                if verbose and (t_iter <= 3 or t_iter % 15 == 0):
                    print(f"    FW it={t_iter} M={M:.8f} score={2*n*best_M:.6f} a={best_alpha:.3f}")
        else:
            no_improve += 1
            if no_improve % 12 == 11:
                rng = np.random.default_rng(31337 + t_iter)
                freq = float(rng.uniform(0.5, 10.0))
                phase = float(rng.uniform(0.0, 2.0 * math.pi))
                x = np.arange(n)
                bump = 0.03 * np.sin(2.0 * math.pi * freq * x / n + phase)
                a = best_a + bump * best_a
                a = np.maximum(a, 1e-12)
                a /= float(a.sum())
                c = _fft_conv(a)
                M = float(c.max())
                no_improve = 0
    return best_a


def _default_seq(n=5000):
    x = np.linspace(0.0, 1.0, int(n))
    return _norm(np.sin(np.pi * x) ** 2)


def _interp(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return _norm(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return _norm(y)


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    try:
        total = max(60.0, min(float(budget_s) - 80.0, 1720.0))
    except Exception:
        total = 1720.0

    print(f"Optimization start, budget={total:.0f}s")

    # Get starting sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.where(np.isfinite(parent), parent, 0.0)
        np.clip(parent, 0.0, 1000.0, out=parent)
        if parent.ndim != 1:
            parent = parent.ravel()
        if parent.size > 0 and float(parent.sum()) > 1e-10:
            a = _norm(parent)
        else:
            a = _default_seq(5000)
    except Exception:
        a = _default_seq(5000)

    # Cap working size
    if a.size > 8000:
        a = _interp(a, 7000)

    n = a.size
    c = _fft_conv(a)
    M = float(c.max())
    best_a = a.copy()
    best_score = _safe_eval(a)
    if not np.isfinite(best_score):
        best_score = 2.0 * n * M
    print(f"  Initial: n={n}, proxy_score={2.0*n*M:.6f}, actual={best_score:.6f}")

    def _try_update(candidate, label):
        nonlocal best_a, best_score
        if candidate is None:
            return
        sc = _safe_eval(candidate)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = float(sc)
            best_a = _norm(candidate)
            print(f"  *** NEW BEST [{label}]: score={best_score:.6f}, n={best_a.size}")

    # Phase 1: L-BFGS with increasing beta (40% of budget)
    t0 = time.monotonic()
    b1 = min(total * 0.38, 600.0)
    print(f"\n=== Phase 1: L-BFGS multi-beta ({b1:.0f}s) ===")
    if SCIPY_OK:
        beta_schedule = [8.0, 20.0, 50.0, 120.0, 300.0, 700.0, 1500.0, 3000.0]
        per_time = b1 / len(beta_schedule)
        for beta in beta_schedule:
            elapsed_total = time.monotonic() - t_start
            if elapsed_total > total - 150:
                break
            remain = b1 - (time.monotonic() - t0)
            if remain < 30:
                break
            # Determine active set
            c = _fft_conv(a)
            if beta < 50:
                top_idx = None
            else:
                tk = min(max(30, (2 * n - 1) // 15), 2 * n - 1)
                top_idx = np.argpartition(-c, tk - 1)[:tk]
            maxfun = int(max(80, min(600, remain / max(0.03, 1.0 / max(10, n // 100)))))
            a_new = _lbfgs(a, beta, maxfun=maxfun, top_idx=top_idx)
            c_new = _fft_conv(a_new)
            M_new = float(c_new.max())
            # Accept if not too much worse
            if M_new <= M * 1.002 + 1e-15:
                a = a_new
                c = c_new
                M = M_new
            _try_update(a, f"lb_{beta:.0f}")
            if time.monotonic() - t0 > remain:
                break

    # Phase 2: Frank-Wolfe (25% of budget)
    t0 = time.monotonic()
    b2 = min(total * 0.25, 420.0)
    print(f"\n=== Phase 2: Frank-Wolfe ({b2:.0f}s) ===")
    a_fw = _fw_search(a, b2, active_max=2500, verbose=True)
    _try_update(a_fw, "fw")
    c_fw = _fft_conv(a_fw)
    if float(c_fw.max()) < M:
        a = a_fw
        c = c_fw
        M = float(c_fw.max())

    # Phase 3: High-beta L-BFGS with tight active set (18% of budget)
    t0 = time.monotonic()
    b3 = min(total * 0.18, 300.0)
    print(f"\n=== Phase 3: High-beta L-BFGS ({b3:.0f}s) ===")
    if SCIPY_OK:
        for beta in [2000.0, 5000.0, 10000.0]:
            elapsed_total = time.monotonic() - t_start
            if elapsed_total > total - 100:
                break
            remain = b3 - (time.monotonic() - t0)
            if remain < 25:
                break
            c = _fft_conv(a)
            tk = min(80, 2 * n - 1)
            top_idx = np.argpartition(-c, tk - 1)[:tk]
            maxfun = int(max(60, min(400, remain / 0.05)))
            a_new = _lbfgs(a, beta, maxfun=maxfun, top_idx=top_idx)
            c_new = _fft_conv(a_new)
            M_new = float(c_new.max())
            if M_new <= M * 1.001 + 1e-15:
                a = a_new
                c = c_new
                M = M_new
            _try_update(a, f"hi_{beta:.0f}")

    # Phase 4: Coordinate search (12% of budget)
    t0 = time.monotonic()
    b4 = min(total * 0.12, 180.0)
    print(f"\n=== Phase 4: Coordinate search ({b4:.0f}s) ===")
    a_coord = _coord_search(a, b4, verbose=True)
    _try_update(a_coord, "coord")
    c_coord = _fft_conv(a_coord)
    if float(c_coord.max()) < M:
        a = a_coord
        c = c_coord
        M = float(c_coord.max())

    # Phase 5: Final combined polish (remaining time)
    t0 = time.monotonic()
    time_left = total - (time.monotonic() - t_start)
    b5 = min(time_left * 0.5, 200.0)
    print(f"\n=== Phase 5: Final polish ({b5:.0f}s) ===")
    if SCIPY_OK and b5 > 30:
        c = _fft_conv(a)
        tk = min(50, 2 * n - 1)
        top_idx = np.argpartition(-c, tk - 1)[:tk]
        maxfun = int(max(40, min(250, b5 / 0.06)))
        a_new = _lbfgs(a, 5000.0, maxfun=maxfun, top_idx=top_idx)
        _try_update(a_new, "final_lb")
        c_new = _fft_conv(a_new)
        if float(c_new.max()) < M:
            a = a_new
            M = float(c_new.max())
        # Quick coord polish
        b5r = b5 - (time.monotonic() - t0)
        if b5r > 20:
            a_co = _coord_search(a, b5r, verbose=False)
            _try_update(a_co, "final_coord")

    # Phase 6: Length exploration
    t0 = time.monotonic()
    time_left = total - (time.monotonic() - t_start)
    if time_left > 80:
        print(f"\n=== Phase 6: Length probes ===")
        base_n = best_a.size
        for delta in (200, -200, 400, -400, 600, -600):
            if time.monotonic() - t_start > total - 30:
                break
            target = int(base_n + delta)
            target = max(1500, min(9000, target))
            a_probe = _interp(best_a, target)
            if SCIPY_OK and time.monotonic() - t_start < total - 60:
                a_probe = _lbfgs(a_probe, 300.0, maxfun=80)
            _try_update(a_probe, f"n={target}")

    # Final validation
    final_val = _safe_eval(best_a)
    if np.isfinite(final_val):
        best_score = min(best_score, final_val)
    else:
        best_a = _default_seq(5000)
        best_score = _safe_eval(best_a)

    arr = np.asarray(best_a, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        arr = _default_seq(5000)
    else:
        arr = arr / s

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: score={best_score:.6f}, n={arr.size}, elapsed={elapsed:.0f}s")
    return [float(x) for x in arr]
