def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def fft_conv(a):
    """Self-convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def fft_cross(a, b):
    """Cross-convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if b.size != n:
        b = np.resize(b, n)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c = np.where(np.isfinite(c), c, 0.0)
    return c


def safe_eval(a):
    """Safely evaluate a sequence."""
    try:
        arr = np.asarray(a, dtype=np.float64)
        if arr.size == 0:
            return float("inf")
        arr = np.clip(arr, 0.0, 1000.0)
        if not np.all(np.isfinite(arr)):
            return float("inf")
        s = float(arr.sum())
        if s < 0.01:
            return float("inf")
        lst = [float(x) for x in arr]
        val = evaluate_sequence(lst)
        if val is None:
            return float("inf")
        val = float(val)
        if not math.isfinite(val):
            return float("inf")
        return val
    except Exception:
        return float("inf")


def score_proxy(a):
    """Fast score computation (proxy for evaluate_sequence)."""
    arr = np.asarray(a, dtype=np.float64)
    n = arr.size
    s = float(arr.sum())
    if s < 0.01:
        return float("inf")
    arr = arr / s
    c = fft_conv(arr)

def lp_fw_direction(a, active_idx):
    """
    Solve the Frank-Wolfe subproblem:
    min t s.t. (a*g)[k] <= t for k in active_idx, sum(g)=1, g>=0
    
    Returns the optimal g vector.
    """
    n = a.size
    K = active_idx.size
    if K == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    
    if not SCIPY_OK:
        # Fallback: uniform on active set positions
        return np.ones(n, dtype=np.float64) / float(n)
    
    active_idx = np.asarray(active_idx, dtype=int)
    
    # Build sparse constraint matrix
    # (a*g)[k] = sum_j a[k-j] * g[j] for valid j
    # Constraint: sum_j A[k,j] * g[j] - t <= 0
    # where A[k,j] = a[k-j] if 0<=k-j<n, else 0
    
    rows = []
    cols = []
    vals = []
    
    for r, k in enumerate(active_idx):
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        js = np.arange(j_min, j_max + 1)
        a_k_minus_j = a[k - js]
        rows.extend([r] * len(js))
        cols.extend(js.tolist())
        vals.extend(a_k_minus_j.tolist())
    
    if not rows:
        return np.ones(n, dtype=np.float64) / float(n)
    
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    
    # A_spp is K x n (convolution constraints)
    A_spp = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))
    
    # Full problem:
    # Variables: (g[0], ..., g[n-1], t) -> n+1 variables
    # Minimize: t (last variable)
    # Constraints:
    #   A_spp @ g - t <= 0  (K inequality constraints)
    #   sum(g) = 1          (1 equality constraint)
    #   g >= 0              (bound constraints)
    #   t >= 0              (bound constraint, optional)
    
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0  # minimize t
    
    # Inequality: A_spp @ g - t*1 <= 0
    # A_ub = [A_spp, -1] (K x (n+1))
    neg_col = np.full(K, -1.0, dtype=np.float64)
    A_ub = scipy.sparse.hstack([A_spp, scipy.sparse.csc_matrix(neg_col.reshape(K, 1))], format='csr')
    b_ub = np.zeros(K, dtype=np.float64)
    
    # Equality: sum(g) = 1, t is free in this constraint
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1, dtype=np.float64)
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = scipy.optimize.linprog(
            c_obj, A_ub=A_ub, b_ub=b_ub,
            A_eq=A_eq, b_eq=b_eq,
            bounds=bounds, method='HiGHS',
            options={'disp': False, 'time_limit': 10.0}
        )
    except Exception:
        return np.ones(n, dtype=np.float64) / float(n)
    
    if not res.success:
        return np.ones(n, dtype=np.float64) / float(n)
    
    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.where(np.isfinite(g) & (g > 0.0), g, 0.0)
    gs = float(g.sum())
    if gs < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return g / gs


def line_search(a, g, c_a, c_g, c_ag, active_mask=None):
    """
    Find best alpha in [0,1] to minimize max_k of:
    (1-alpha)^2 * c_a[k] + 2*alpha*(1-alpha) * c_ag[k] + alpha^2 * c_g[k]
    """
    n = a.size
    alphas = np.array([0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0])
    
    best_alpha = 0.0
    best_val = float(np.max(c_a)) if active_mask is None else float(np.max(c_a[active_mask]))
    
    for alpha in alphas:
        one = 1.0 - alpha
        val = one * one * c_a + 2.0 * alpha * one * c_ag + alpha * alpha * c_g
        if active_mask is not None:
            m = float(np.max(val[active_mask]))
        else:
            m = float(np.max(val))
        if math.isfinite(m) and m < best_val:
            best_val = m
            best_alpha = alpha
    
    if best_alpha <= 0:
        return 0.0, best_val
    
    # Refine around best_alpha
    lo = max(0.0, best_alpha - 0.05)
    hi = min(1.0, best_alpha + 0.05)
    for _ in range(20):
        mid = (lo + hi) / 2.0
        one = 1.0 - mid
        val = one * one * c_a + 2.0 * mid * one * c_ag + mid * mid * c_g
        if active_mask is not None:
            m = float(np.max(val[active_mask]))
        else:
            m = float(np.max(val))
        if m < best_val:
            best_val = m
            best_alpha = mid
            lo = mid
        else:
            hi = mid
    
    return best_alpha, best_val


def coordinate_polish(a, c, mask, budget, rng, verbose=False):
    """
    Fast coordinate polish using rank-1 convolution updates.
    """
    t0 = time.monotonic()
    a = a.copy()
    c = c.copy()
    n = a.size
    M = float(np.max(c)) if mask is None else float(np.max(c[mask]))
    best_a = a.copy()
    best_M = M
    stall = 0
    
    while time.monotonic() - t0 < budget - 1 and stall < 50:
        # Compute pressure: gradient of max c at current point
        # Simplified: use top few peaks
        top_k = min(20, 2 * n - 1)
        if mask is not None:
            top_idx = np.argsort(c[mask])[::-1][:top_k]
            # Convert back to full indices
            mask_idx = np.where(mask)[0]
            top_idx = mask_idx[top_idx]
        else:
            top_idx = np.argsort(c)[::-1][:top_k]
        
        # Compute pressure vector: sum of top peaks' gradients
        # Gradient of c[k] w.r.t. a[i] is 2*a[k-i] (for valid k-i)
        pressure = np.zeros(n, dtype=np.float64)
        for k in top_idx:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            pressure[i_min:i_max+1] += 2.0 * a[k - (np.arange(i_min, i_max+1))]
        
        # Find high-pressure sources and low-pressure destinations
        # Move mass from high-pressure positions to low-pressure positions
        n_src = min(10, n)
        n_dst = min(15, n)
        
        src_candidates = np.argsort(pressure)[-n_src:]  # highest pressure
        dst_candidates = np.argsort(pressure)[:n_dst]   # lowest pressure
        
        improved = False
        best_move = None
        best_move_M = M
        
        for s in src_candidates:
            s = int(s)
            if a[s] < 1e-14:
                continue
            for d in dst_candidates:
                d = int(d)
                if d == s:
                    continue
                for frac in (0.01, 0.02, 0.05, 0.1):
                    t = a[s] * frac
                    if t < 1e-16:
                        continue
                    # Apply move
                    c_new = c.copy()
                    # Add at d: c_new[d:d+n] += 2*t*a (but careful with boundaries)
                    d_start = d
                    d_end = min(d + n, 2*n - 1)
                    a_slice = a[:d_end - d]  # a[0], a[1], ...
                    c_new[d_start:d_end] += 2.0 * t * a_slice
                    
                    # Remove from s: c_new[s:s+n] -= 2*t*a
                    s_start = s
                    s_end = min(s + n, 2*n - 1)
                    a_slice2 = a[:s_end - s]
                    c_new[s_start:s_end] -= 2.0 * t * a_slice2
                    
                    # O(1) corrections for t^2
                    if 2*d < 2*n - 1:
                        c_new[2*d] += t * t
                    if 2*s < 2*n - 1:
                        c_new[2*s] += t * t
                    if s + d < 2*n - 1:
                        c_new[s + d] -= 2.0 * t * t
                    
                    c_new = np.maximum(c_new, 0.0)
                    m_new = float(np.max(c_new)) if mask is None else float(np.max(c_new[mask]))
                    
                    if m_new < best_move_M - 1e-18:
                        best_move_M = m_new
                        best_move = (s, d, t)
                        improved = True
        
        if best_move is not None:
            s, d, t = best_move
            a[s] -= t
            a[d] += t
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = fft_conv(a)
            M = float(np.max(c)) if mask is None else float(np.max(c[mask]))
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Try random moves
            found = False
            for _ in range(50):
                s = int(rng.integers(n))
                d = int(rng.integers(n))
                if s == d or a[s] < 1e-14:
                    continue
                frac = float(rng.uniform(0.005, 0.05))
                t = a[s] * frac
                c_new = c.copy()
                d_end = min(d + n, 2*n - 1)
                c_new[d:d_end] += 2.0 * t * a[:d_end-d]
                s_end = min(s + n, 2*n - 1)
                c_new[s:s_end] -= 2.0 * t * a[:s_end-s]
                if 2*d < 2*n-1: c_new[2*d] += t*t
                if 2*s < 2*n-1: c_new[2*s] += t*t
                if s+d < 2*n-1: c_new[s+d] -= 2.0*t*t
                c_new = np.maximum(c_new, 0.0)
                m_new = float(np.max(c_new)) if mask is None else float(np.max(c_new[mask]))
                if m_new < M - 1e-18:
                    a[s] -= t
                    a[d] += t
                    a = np.maximum(a, 0.0)
                    a = a / float(a.sum())
                    c = fft_conv(a)
                    M = m_new
                    if M < best_M - 1e-15:
                        best_M = M
                        best_a = a.copy()
                    stall = 0
                    found = True
                    break
            if not found:
                stall += 1
    
    return best_a


def fw_iterate(a, top_k=100, verbose=False):
    """
    One iteration of LP-based Frank-Wolfe.
    Returns updated a and the new max convolution value.
    """
    n = a.size
    c = fft_conv(a)
    M = float(np.max(c))
    
    # Find active set
    if top_k >= 2 * n - 1:
        active_idx = np.arange(2 * n - 1)
    else:
        active_idx = np.argsort(c)[::-1][:top_k]
    
    # Solve LP for FW direction
    g = lp_fw_direction(a, active_idx)
    
    # Line search
    c_g = fft_conv(g)
    c_ag = fft_cross(a, g)
    
    # Create active mask for line search (use all peaks for safety)
    alpha, new_M = line_search(a, g, c, c_g, c_ag)
    
    if alpha <= 0:
        return a, M
    
    one = 1.0 - alpha
    a_new = one * a + alpha * g
    a_new = np.maximum(a_new, 0.0)
    s = float(a_new.sum())
    if s < 1e-15:
        return a, M
    a_new = a_new / s
    
    return a_new, new_M


def default_sequence(n=5000, pattern='sine'):
    """Generate a default initial sequence."""
    x = np.linspace(0.0, 1.0, int(n))
    if pattern == 'sine':
        a = np.sin(np.pi * x) ** 2
    elif pattern == 'cos':
        a = (1.0 + np.cos(2 * np.pi * 3 * x)) ** 2
    elif pattern == 'comb':
        # Comb pattern with 7 teeth
        a = np.zeros(n)
        period = n // 7
        for i in range(7):
            center = i * period + period // 2
            width = max(1, period // 10)
            start = max(0, center - width)
            end = min(n, center + width)
            a[start:end] = 1.0
    elif pattern == 'gaussian':
        a = np.exp(-2.0 * (x - 0.5) ** 2)
    else:
        a = np.sin(np.pi * x) ** 2
    return norm_sum1(a)


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    a = np.asarray(a, dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)

When we move mass `t` from position `s` to position `d`:
- New a: a[s] -= t, a[d] += t
- Change in convolution:
  - The self-convolution c[k] = sum_i a[i]*a[k-i] changes by:

def _next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = _next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def _fft_cross(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if b.size != n:
        b = np.resize(b, n)
    size = 2 * n - 1
    N = _next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c = np.where(np.isfinite(c), c, 0.0)
    return c

def _lp_fw_direction(a, active_idx, time_limit=8.0):
    n = a.size
    K = active_idx.size
    if K == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if not SCIPY_OK:
        return None
    active_idx = np.asarray(active_idx, dtype=int)
    rows = []
    cols = []
    vals = []
    for r in range(K):
        k = int(active_idx[r])
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        js = np.arange(j_min, j_max + 1)
        a_km = a[k - js]
        rows.extend([r] * len(js))
        cols.extend(js.tolist())
        vals.extend(a_km.tolist())
    if not rows:
        return None
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    A_spp = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0
    neg_col = np.full(K, -1.0, dtype=np.float64)
    A_ub = scipy.sparse.hstack([A_spp, scipy.sparse.csc_matrix(neg_col.reshape(K, 1))], format='csr')
    b_ub = np.zeros(K, dtype=np.float64)
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1, dtype=np.float64)
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                     bounds=bounds, method='HiGHS',
                                     options={'disp': False, 'time_limit': float(time_limit)})
    except Exception:
        return None
    if not res.success:
        return None
    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.where(np.isfinite(g) & (g > 0.0), g, 0.0)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs


def _pressure_direction(a, active_idx, scale=6.0):
    n = a.size
    if active_idx.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    pressure = np.zeros(n, dtype=np.float64)
    for k in active_idx:
        k = int(k)
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            pressure[i_min:i_max + 1] += a[k - np.arange(i_min, i_max + 1)]
    pmax = float(np.max(pressure))
    pmin = float(np.min(pressure))
    ptp = pmax - pmin
    if ptp < 1e-30:
        return np.ones(n, dtype=np.float64) / float(n)
    scaled = (pressure - pmin) / ptp
    w = np.exp(-scale * scaled)
    w = np.where(np.isfinite(w), w, 0.0)
    w += 1e-12
    w = 0.95 * w / float(w.sum()) + 0.05 * (np.ones(n) / float(n))
    return w / float(w.sum())
