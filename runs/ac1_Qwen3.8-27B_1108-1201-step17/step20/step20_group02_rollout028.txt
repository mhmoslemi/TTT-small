def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_vec(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()

    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def norm_sum1(a):

    s = a.sum()

    return a / s


def build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend((a[k - idx]).tolist())
    return sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)

        if g_sum < 1e-15:

    temp_holder = [0.05]

    # Actually, let me just do projected gradient with the simplex constraint directly
    # This avoids the softmax parameterization complexity
    
    a_cur = norm_sum1(a0)
    eta = 0.01
    for temp in temps:
        if time.monotonic() - t0 > time_limit:
            break
        max_iter = max(50, int(time_limit * 10 / len(temps)))
        for it in range(max_iter):
            if time.monotonic() - t0 > time_limit:
                break
            # Compute gradient of soft-max-conv
            conv = fast_convolve(a_cur)
            scaled = conv / temp
            max_s = scaled.max()
            p = np.exp(scaled - max_s)
            p = p / p.sum()
            
            # grad_a[i] = (2/temp) * sum_j a_cur[j] * p[i+j]
            # Use FFT
            a_rev = a_cur[::-1]
            q = np.zeros(3 * n - 1)
            q[:2 * n - 1] = p
            fft_sz = 1 << (4 * n - 2).bit_length()
            AR = np.fft.rfft(a_rev, fft_sz)
            Q = np.fft.rfft(q, fft_sz)
            conv_result = np.fft.irfft(AR * Q, fft_sz)
            grad_a = (2.0 / temp) * conv_result[n - 1 : 2 * n - 1]
            
            # Projected gradient step on simplex
            a_trial = a_cur - eta * grad_a
            a_trial = simplex_project(a_trial)
            a_trial = a_trial / a_trial.sum()
            
            val = fast_convolve(a_trial).max()
            val_obj = 2.0 * n * val
            if val_obj < best_val - 1e-15:
                best_val = val_obj
                best_a = a_trial.copy()
                a_cur = a_trial
                eta = min(eta * 1.1, 0.5)
            else:
                eta = eta * 0.9
                if eta < 0.001:
                    break
    
    if verbose:
        elapsed = time.monotonic() - t0
        print("    Smooth refine done (" + str(round(elapsed, 1)) + "s): J=" + repr(best_val))
    return best_a, best_val


def coordinate_polish(a0, time_limit, rng_seed=77, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(50, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-impact destination
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 5:
                candidates = np.arange(n)

            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src:
                continue

            for frac in [0.05, 0.02, 0.01, 0.005]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_t = fast_convolve(a_trial).max()
                if M_t < best_m - 1e-15:
                    a = a_trial
                    best_m = M_t
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
        else:
            stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + " stall=" + str(stall))

    return best_a, best_m


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_vec(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def get_initial_sequence():
    """Get the best starting sequence from parent or parametric families."""
    candidates = []
    scores = []

    # Try parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_vec(p_norm)
            if np.isfinite(sc):
                candidates.append(p_norm)
                scores.append(sc)
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
    except Exception:
        pass

    # Parametric families
    for n in [4000, 5000, 6000, 7000]:
        x = np.linspace(0, 1, n)
        for name, shape in [("sin2", np.sin(np.pi * x) ** 2),
                            ("raised_cos", 0.5 * (1 + np.cos(2 * np.pi * x))),
                            ("tri", np.maximum(0, 1 - 2 * np.abs(x - 0.5))),
                            ("gauss", np.exp(-((x - 0.5) ** 2) / 0.05))]:
            shape = np.maximum(shape, 1e-10)
            shape = shape / shape.sum()
            sc = evaluate_vec(shape)
            if np.isfinite(sc):
                candidates.append(shape)
                scores.append(sc)

    if not candidates:
        n = 5000
        x = np.linspace(0, 1, n)
        s = np.sin(np.pi * x) ** 2
        s = s / s.sum()
        return s, evaluate_vec(s)

    best_idx = np.argmin(scores)
    return candidates[best_idx], scores[best_idx]


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = get_solver()
    best_seq, best_score = get_initial_sequence()
    print("Initial best: " + repr(best_score) + " (n=" + str(len(best_seq)) + ")")

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: LP-FW on initial best ===
    n0 = len(best_seq)
    print("\n=== Phase 1: LP-FW (n=" + str(n0) + ") ===")
    t1 = min(remaining_time() * 0.25, 400.0)
    if t1 > 60:
        if n0 <= 5000:
            a_out, _ = lp_fw_full(norm_sum1(best_seq), t1, solver, verbose=True)
        else:
            a_out, _ = lp_fw_active(norm_sum1(best_seq), t1, solver, initial_K=400, max_K=4000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1_n" + str(n0), t_global)

    # === Phase 2: Smooth refinement ===
    print("\n=== Phase 2: Smooth refine (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining_time() * 0.10, 150.0)
    if t2 > 30:
        a_out, _ = smooth_refine(norm_sum1(best_seq), t2, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "smooth_n" + str(len(best_seq)), t_global)

    # === Phase 3: LP-FW second pass ===
    print("\n=== Phase 3: LP-FW second (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining_time() * 0.20, 300.0)
    if t3 > 60 and len(best_seq) <= 7000:
        if len(best_seq) <= 5000:
            a_out, _ = lp_fw_full(norm_sum1(best_seq), t3, solver, verbose=True)
        else:
            a_out, _ = lp_fw_active(norm_sum1(best_seq), t3, solver, initial_K=500, max_K=5000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2_n" + str(len(best_seq)), t_global)

    # === Phase 4: Multi-n exploration ===
    print("\n=== Phase 4: Multi-n exploration ===")
    n_best = len(best_seq)
    try_ns = sorted(set([n_best - 1000, n_best + 1000, n_best + 2000, n_best - 2000,
                         4000, 5000, 6000, 7000, 8000]))
    try_ns = [l for l in try_ns if 3000 <= l <= 10000]
    try_ns.sort(key=lambda l: abs(l - n_best))

    for new_n in try_ns[:4]:
        rt = remaining_time()
        if rt < 100:
            break
        print("\n  Multi-n: n=" + str(new_n) + " (remaining=" + str(round(rt, 1)) + "s)")
        a_long = interpolate_seq(best_seq, new_n)

        t5 = min(rt * 0.15, 150.0)
        if t5 > 40:
            if new_n <= 5000:
                a_out, _ = lp_fw_full(a_long, t5, solver, verbose=False)
            else:
                a_out, _ = lp_fw_active(a_long, t5, solver, initial_K=300, max_K=3500, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

        rt = remaining_time()
        if rt > 50:
            t5b = min(rt * 0.10, 80.0)
            if t5b > 20:
                a_out, _ = smooth_refine(norm_sum1(best_seq), t5b, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "smooth_n" + str(new_n), t_global)

    # === Phase 5: LP-FW third pass on best ===
    print("\n=== Phase 5: LP-FW third (n=" + str(len(best_seq)) + ") ===")
    t6 = min(remaining_time() * 0.15, 250.0)
    if t6 > 60 and len(best_seq) <= 7000:
        if len(best_seq) <= 5000:
            a_out, _ = lp_fw_full(norm_sum1(best_seq), t6, solver, verbose=True)
        else:
            a_out, _ = lp_fw_active(norm_sum1(best_seq), t6, solver, initial_K=600, max_K=5000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw3_n" + str(len(best_seq)), t_global)

    # === Phase 6: Smooth refine again ===
    print("\n=== Phase 6: Smooth refine ===")
    t7 = min(remaining_time() * 0.08, 100.0)
    if t7 > 20:
        a_out, _ = smooth_refine(norm_sum1(best_seq), t7, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "smooth2_n" + str(len(best_seq)), t_global)

    # === Phase 7: Coordinate polish ===
    print("\n=== Phase 7: Coordinate polish ===")
    t8 = min(remaining_time() * 0.10, 120.0)
    if t8 > 20:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t8, rng_seed=2024, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n" + str(len(best_seq)), t_global)

    # === Phase 8: Final LP-FW ===
    print("\n=== Phase 8: Final LP-FW ===")
    t9 = min(remaining_time() * 0.20, 200.0)
    if t9 > 40 and len(best_seq) <= 7000:
        if len(best_seq) <= 5000:
            a_out, _ = lp_fw_full(norm_sum1(best_seq), t9, solver, verbose=True)
        else:
            a_out, _ = lp_fw_active(norm_sum1(best_seq), t9, solver, initial_K=800, max_K=5000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # === Phase 9: Final coordinate polish ===
    rt = remaining_time()
    if rt > 20:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), min(rt * 0.5, 60.0), rng_seed=999, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        best_seq = np.maximum(best_seq, 0.0)
        s = best_seq.sum()
        if s > 1e-10:
            best_seq = best_seq / s
        fs = evaluate_vec(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_vec(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
