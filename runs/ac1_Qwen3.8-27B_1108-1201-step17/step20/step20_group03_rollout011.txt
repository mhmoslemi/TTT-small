def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z.copy()
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n, dtype=np.float64) / n
    rho = int(rho_idx[-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_grad(y, beta, top_k):
    n = len(y)
    b = fast_autocorr(y)
    b_max = b.max()
    if top_k is not None and 0 < top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b), dtype=np.float64)
        b_use[idx] = b[idx]
    else:
        b_use = b
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b_use)
        mask = (b_use == b_max)
        cnt = max(1, int(mask.sum()))
        p[mask] = 1.0 / cnt
    else:
        p = exp_vals / p_total
    a_rev = y[::-1].copy()
    conv_len = n + len(p) - 1
    fft_s = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_s)
    Pr = np.fft.rfft(p, fft_s)
    conv = np.fft.irfft(Ar * Pr, fft_s)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad


def fista_opt(a0, time_limit, beta_start=15.0, beta_end=800.0,
              beta_growth=1.003, top_k=15, restart_every=3000, verbose=False):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(42)
    s_prev = None
    y_prev = None
    bb_step = 1.0 / (n ** 0.6)
    n_restart = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad = compute_grad(y, beta, top_k)

        if s_prev is not None and y_prev is not None:
            sy = y - y_prev
            gy = grad - s_prev
            sy_gy = sy.dot(gy)
            gy_gy = gy.dot(gy)
            if gy_gy > 1e-20:
                bb_cand = sy_gy / gy_gy
                if 1e-8 < bb_cand < 10.0:
                    bb_step = max(1e-6, min(bb_cand, 5.0 / max(beta, 1.0)))
                else:
                    bb_step = max(bb_step * 0.9, 1e-6)
            s_prev = grad.copy()
            y_prev = y.copy()

        eta = min(bb_step, 0.15)
        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x.copy()
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            n_restart += 1
            noise_scale = 0.02 / (1.0 + 0.1 * n_restart)
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 8.0)
            s_prev = None
            y_prev = None
            bb_step = 1.0 / (n ** 0.6)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    FISTA t={} ({}s): J={} beta={}".format(t, round(elapsed, 1), round(J, 6), round(beta, 1)))

    return best_a, best_max_b


def lp_fw(a0, time_limit, initial_K=300, max_K=2000, add_per_iter=20, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 500

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep_idx = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep_idx)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n), dtype=np.float64)
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=find_solver(), verbose=False)
        except Exception:
            if len(active_set) > 100:
                active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:100])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size_c = 2 * n - 1
        fft_size_c = 1 << (size_c - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size_c)
        Gf = np.fft.rfft(g_dir, fft_size_c)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size_c)[:size_c]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5),
                  max(0.02, default_alpha * 1.5), 1.0 / (t + 1)]
        alphas = [max(0.01, min(1.0, a)) for a in alphas]
        alphas = list(set(alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter={} ({}s): J={} |S|={}".format(t, round(elapsed, 1), round(J, 6), len(active_set)))

    return best_a, best_m


def two_exchange(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    max_stall = 150

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:25]
        improved = False

        for _ in range(40):
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(top_peaks[rng.integers(len(top_peaks))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            cmax = contrib.max()
            if cmax < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:12]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src:
                continue
            if a[src] < 1e-12:
                continue

            frac = float(rng.choice([0.003, 0.005, 0.01, 0.02, 0.05]))
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-14:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                ac = fast_autocorr(a)
                top_peaks = np.argsort(ac)[::-1][:25]

        if not improved:
            stall += 1
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.002, 0.02))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

    return best_a, best_m


def spectral_reshape(a0, time_limit, verbose=False):
    """Frequency-domain reshaping inspired by Gerchberg-Saxton."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    n_iters = 300
    rng = np.random.default_rng(7)

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        c = np.fft.irfft(P, fft_size)[:2 * n - 1]
        c_max = c.max()

        if c_max < 1e-15:
            break

        # Flatten: reduce peaks in autocorrelation
        alpha = 0.05 + 0.05 * (it / n_iters)
        c_median = np.median(c)
        c_soft = c - alpha * np.maximum(c - c_median, 0)
        c_soft = np.maximum(c_soft, 0)

        # Compute new spectrum
        P_new = np.fft.rfft(c_soft, fft_size)
        # Check validity
        neg_frac = float(np.sum(P_new < -1e-6)) / len(P_new)
        if neg_frac > 0.1:
            # Too aggressive; use amplitude damping instead
            A_mag = np.abs(A)
            A_phase = A / np.maximum(A_mag, 1e-30)
            # Damp high-power frequencies
            A_mag_new = A_mag * (1.0 - alpha * A_mag / np.maximum(A_mag.max(), 1e-30))
            A_new = A_mag_new * A_phase
        else:
            P_new_clipped = np.maximum(P_new, 0)
            A_mag = np.abs(A)
            A_phase = A / np.maximum(A_mag, 1e-30)
            ratio = np.sqrt(P_new_clipped / np.maximum(P, 1e-30))
            A_new = A_mag * ratio * A_phase

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new

        # Occasionally add noise for exploration
        if it % 30 == 0 and it > 0:
            noise = rng.normal(0, 0.005, n)
            a_noisy = a + noise
            a_noisy = np.maximum(a_noisy, 0.0)
            a_noisy = a_noisy / a_noisy.sum()
            m_noisy = fast_autocorr(a_noisy).max()
            if m_noisy < best_m:
                best_m = m_noisy
                best_a = a_noisy.copy()
                a = a_noisy

        if verbose and it % 50 == 0:
            J = 2.0 * n * best_m
            print("    Spectral it={} ({}s): J={}".format(it, round(time.monotonic() - t0, 1), round(J, 6)))

    return best_a, best_m


def cyclic_shift_best(a0):
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m - 1e-15:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {} (n={}, t={}s)".format(label, round(sc, 6), len(a), round(elapsed, 1)))
    return best_seq, best_score


def make_initial(n, shape='sincos', seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if shape == 'sincos':
        a = np.sin(np.pi * x) ** 2
    elif shape == 'raised_cos':
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    elif shape == 'triangular':
        a = np.minimum(x * 2, (1 - x) * 2)
    elif shape == 'gaussian':
        a = np.exp(-8 * (x - 0.5) ** 2)
    elif shape == 'bicubic':
        a = np.sin(np.pi * x) ** 2 * (1 - (2 * x - 1) ** 4)
    else:
        a = np.sin(np.pi * x) ** 2
    a = a + 0.01 * rng.random(n)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1760.0)
    print("Total time budget: {}s".format(round(time_budget, 1)))

    # Try to use parent
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n={}, score={}".format(len(parent), round(p_score, 6)))
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        best_seq = make_initial(n_base, 'sincos', 0)
        best_score = evaluate_safe(best_seq)
        print("Default init: n={}, score={}".format(n_base, round(best_score, 6)))

    n0 = len(best_seq)

    # === Phase 1: FISTA main (largest allocation) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: FISTA main (n={}) ===".format(n0))
    t1 = min(remaining * 0.30, 480.0)
    if t1 > 60:
        a_out, _ = fista_opt(norm_sum1(best_seq), t1,
                             beta_start=12.0, beta_end=700.0,
                             beta_growth=1.003, top_k=15,
                             restart_every=3000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista1", t_global)

    # === Phase 2: Cyclic shift ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: Cyclic shift ===")
    if remaining > 30:
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyc1", t_global)

    # === Phase 3: LP-Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 3: LP-FW (n={}) ===".format(n_cur))
    t3 = min(remaining * 0.18, 260.0)
    if t3 > 40 and n_cur <= 9000 and _HAS_CP:
        a_out, _ = lp_fw(norm_sum1(best_seq), t3,
                         initial_K=400, max_K=2500, add_per_iter=25, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # === Phase 4: Spectral reshaping (new) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 4: Spectral reshape (n={}) ===".format(n_cur))
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 30:
        a_out, _ = spectral_reshape(norm_sum1(best_seq), t4, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec1", t_global)

    # === Phase 5: FISTA round 2 with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 5: FISTA v2 (n={}) ===".format(n_cur))
    t5 = min(remaining * 0.15, 220.0)
    if t5 > 40:
        a_out, _ = fista_opt(norm_sum1(best_seq), t5,
                             beta_start=50.0, beta_end=1500.0,
                             beta_growth=1.005, top_k=10,
                             restart_every=2500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista2", t_global)

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 6: Multi-length (n={}) ===".format(n_cur))

    for new_n in [n_cur + 800, n_cur + 2000]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        if new_n > 12000:
            continue
        print("  Trying n={}".format(new_n))
        a_long = interpolate_seq(best_seq, new_n)

        t_ma = min(remaining * 0.25, 180.0)
        if t_ma > 40:
            a_out, _ = fista_opt(a_long, t_ma,
                                 beta_start=20.0, beta_end=600.0,
                                 beta_growth=1.004, top_k=12,
                                 restart_every=2500, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_n{}".format(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
            best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyc_n{}".format(new_n), t_global)

    # === Phase 7: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 7: LP-FW v2 (n={}) ===".format(n_cur))
    t7 = min(remaining * 0.15, 180.0)
    if t7 > 30 and n_cur <= 9000 and _HAS_CP:
        a_out, _ = lp_fw(norm_sum1(best_seq), t7,
                         initial_K=500, max_K=3000, add_per_iter=30, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp2", t_global)

    # === Phase 8: 2-exchange polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 8: 2-exchange (n={}) ===".format(n_cur))
    t8 = min(remaining * 0.12, 120.0)
    if t8 > 20:
        a_out, _ = two_exchange(norm_sum1(best_seq), t8, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "2ex1", t_global)

    # === Phase 9: Final FISTA micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== Phase 9: Final FISTA (n={}) ===".format(n_cur))
    t9 = min(remaining * 0.20, 150.0)
    if t9 > 20:
        a_out, _ = fista_opt(norm_sum1(best_seq), t9,
                             beta_start=100.0, beta_end=2500.0,
                             beta_growth=1.01, top_k=8,
                             restart_every=2000, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_final", t_global)

    # === Phase 10: Final cyclic + 2-exchange ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyc_final", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        a_out, _ = two_exchange(norm_sum1(best_seq), remaining * 0.6, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "2ex_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={}, n={}, time={}s".format(round(best_score, 6), len(best_seq), round(elapsed, 1)))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        best_seq = make_initial(n_fb, 'sincos', 0)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
