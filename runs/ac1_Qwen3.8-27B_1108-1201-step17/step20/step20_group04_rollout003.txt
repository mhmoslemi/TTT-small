```python
def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
```

```python
def softmax_grad(x, beta, top_k):
    """Compute gradient of smoothed max autocorrelation w.r.t. x (softmax params)."""
    a = np.exp(x - x.max())
    a = a / a.sum()
    n = len(a)
    
    c = fast_autocorr(a)
    if top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_max = c[idx].max()
        shifted = beta * (c[idx] - c_max)
        shifted = np.clip(shifted, -300, 0)
        exp_vals = np.exp(shifted)
        p_sparse = exp_vals / exp_vals.sum()
    else:
        c_max = c.max()
        shifted = beta * (c - c_max)
        shifted = np.clip(shifted, -300, 0)
        exp_vals = np.exp(shifted)
        p_sparse = exp_vals / exp_vals.sum()
    
    # grad_a = (2/beta) * (p * a_rev) [appropriate slice]
    fft_size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p_sparse, fft_size)  # p_sparse has length 2n-1
    A_rev = np.fft.rfft(a[::-1], fft_size)
    Conv = np.fft.irfft(P * A_rev, fft_size)
    grad_a = (2.0 / beta) * Conv[n-1:2*n-1]
    
    # Softmax gradient
    mean_grad = np.sum(grad_a * a)
    grad_x = grad_a - mean_grad * a
    
    # Value for the objective
    obj = (1.0 / beta) * (np.log(np.sum(exp_vals)) + c_max)
    return obj, grad_x
```
