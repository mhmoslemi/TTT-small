def next_pow2(x):

def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)

    if n == 0:

    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return np.where(np.isfinite(c) & (c > 0.0), c, 0.0)

    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    return np.where(np.isfinite(c) & (c > 0.0), c, 0.0)


def cross_convolve_p_a(p, a):
    p = np.asarray(p, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    out = conv[n - 1:2 * n - 1][::-1]
    return np.where(np.isfinite(out), out, 0.0)


def proxy_max(c):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0
    vals = c[np.isfinite(c)]
    if vals.size == 0:
        return 0.0
    return float(np.max(vals))


def active_top(c, max_k):
    c = np.asarray(c, dtype=np.float64)
    size = c.size
    max_k = int(min(max_k, size))
    if max_k <= 0:
        return np.array([], dtype=int)
    return np.argsort(c)[::-1][:max_k]


def safe_eval(a):
    try:
        arr = np.asarray(a, dtype=np.float64)
        if arr.size == 0:
            return float("inf")
        arr = np.clip(arr, 0.0, 1000.0)
        if not np.all(np.isfinite(arr)):
            return float("inf")
        s = float(arr.sum())
        if s < 0.01:
            return float("inf")
        lst = [float(x) for x in arr]
        val = evaluate_sequence(lst)
        if val is None:
            return float("inf")
        val = float(val)
        if not math.isfinite(val):
            return float("inf")
        return val
    except Exception:
        return float("inf")


def try_update(best_seq, best_score, a, label, t0):
    if a is None:
        return best_seq, best_score
    try:
        arr = np.asarray(a, dtype=np.float64)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        np.maximum(arr, 0.0, out=arr)
        s = float(arr.sum())
        if arr.size == 0 or s < 0.01:
            return best_seq, best_score
        arr = arr / s
        sc = safe_eval(arr)
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_seq = arr.copy()
            best_score = float(sc)
            print("  NEW BEST [" + label + "]: " + repr(sc) +
                  " n=" + str(arr.size) +
                  " t=" + repr(time.monotonic() - t0))
    except Exception:
        pass
    return best_seq, best_score


def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)


def lse_objective(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 0.0, np.array([], dtype=np.float64)
    size = 2 * n - 1
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    m = float(np.max(c))
    shifted = beta * (c - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0 or not math.isfinite(se):
        return m + 700.0 / beta, np.zeros(n, dtype=np.float64)
    p = e / se
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    f = m + math.log(se) / beta
    if not math.isfinite(f):
        f = 1e100
    return f, grad


def softmax_from_u(u):
    u = np.asarray(u, dtype=np.float64)
    n = u.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(u)):
        return np.ones(n, dtype=np.float64) / float(n)
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def make_v(a, ref=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 1:
        return np.array([], dtype=np.float64), 0
    if ref is None:
        ref = int(np.argmax(a))
    ref = int(ref)
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.where(np.isfinite(v), v, -700.0)
    return v, ref


def from_v(v, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    return softmax_from_u(u)


def objgrad_v(v, beta, ref=0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    if n <= 0:
        return 1e100, np.array([], dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return 1e100, np.zeros(v.size, dtype=np.float64)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    u = np.empty(n, dtype=np.float64)
    u[ref] = 0.0
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    a = softmax_from_u(u)
    f, ga = lse_objective(a, beta)
    if not math.isfinite(f):
        return 1e100, np.zeros(v.size, dtype=np.float64)
    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)
    gv = np.empty(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]
    if not np.all(np.isfinite(gv)):
        gv = np.zeros(v.size, dtype=np.float64)
    return f, gv


def run_lbfegs(a0, beta, maxfun=400, verbose=True):
    a = norm_sum1(a0)
    n = a.size
    if n <= 1:
        return a
    v0, ref = make_v(a)
    if not SCIPY_OK or maxfun <= 0:
        return a
    t0 = time.monotonic()
    options = {
        "maxfun": int(maxfun),
        "maxiter": int(maxfun + 30),
        "ftol": 1e-13,
        "gtol": 1e-8,
        "disp": False
    }
    try:
        res = scipy.optimize.minimize(
            objgrad_v, v0, args=(beta, ref),
            method="L-BFGS-B", jac=True, options=options
        )
    except Exception:
        return a
    if res is not None and hasattr(res, "x"):
        vx = np.asarray(res.x, dtype=np.float64)
        if vx.size == v0.size and np.all(np.isfinite(vx)):
            a_out = from_v(vx, ref)
            return norm_sum1(a_out)
    return a


def lbfegs_schedule(a_init, budget):
    a = norm_sum1(a_init)
    n = a.size
    size = 2 * n - 1
    best_a = a.copy()
    best_M = proxy_max(fft_conv(a))
    t0 = time.monotonic()

    schedules = [
        (5.0, 60),
        (15.0, 80),
        (40.0, 100),
        (100.0, 120),
        (200.0, 140),
        (500.0, 160),
        (1000.0, 180),
        (2000.0, 200),
        (4000.0, 220),
    ]

    for beta, maxfun in schedules:
        if time.monotonic() - t0 > budget - 10:
            break
        a_new = run_lbfegs(a, beta, maxfun=maxfun, verbose=False)
        c_new = fft_conv(a_new)
        m_new = proxy_max(c_new)
        if math.isfinite(m_new):
            if m_new < best_M - 1e-15:
                best_M = m_new
                best_a = a_new.copy()
                a = a_new
            else:
                a = a_new if m_new < best_M * 1.01 else best_a
    return best_a


def game_direction(f, active_idx, mw_iters=40, seed=0, init_logits=None):
    f = norm_sum1(f)
    n = f.size
    size = 2 * n - 1
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K <= 0:
        return np.ones(n, dtype=np.float64) / float(n)

    rng = np.random.default_rng(seed)
    if init_logits is None:
        logits = np.zeros(K, dtype=np.float64)
    else:
        logits = np.asarray(init_logits, dtype=np.float64).copy()
        if logits.size != K:
            logits = np.zeros(K, dtype=np.float64)
        logits = np.where(np.isfinite(logits), logits, 0.0)
        logmx = float(np.max(logits))
        if math.isfinite(logmx):
            logits = logits - logmx
    logits += rng.normal(0.0, 0.1, K)

    def norm_logits(loc):
        loc = np.where(np.isfinite(loc), loc, -1000.0)
        loc = loc - float(np.max(loc))
        w = np.exp(loc)
        s = float(w.sum())
        if s <= 0.0:
            w = np.ones(K, dtype=np.float64)
            s = float(K)
        return w / s

    lambda_w = norm_logits(logits)
    for it in range(max(1, int(mw_iters))):
        pf = np.zeros(size, dtype=np.float64)
        pf[active_idx] = lambda_w
        pressure = cross_convolve_p_a(pf, f)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)
        pmx = float(np.max(pressure))
        pmn = float(np.min(pressure))
        ptp = pmx - pmn
        if ptp <= 1e-30:
            g = np.ones(n, dtype=np.float64) / float(n)
        else:
            temp = 6.0 + 6.0 * (it / max(1, mw_iters))
            scaled = (pressure - pmn) / ptp
            w = np.exp(-temp * scaled)
            w = np.where(np.isfinite(w), w, 0.0)
            w += 1e-12
            g = w / float(w.sum())
            g = 0.95 * g + 0.05 * (np.ones(n, dtype=np.float64) / float(n))
            g = g / float(g.sum())

        h = fft_conv2(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmac_ = float(np.max(ha))
        hmin = float(np.min(ha))
        hptp = hmac_ - hmin
        if hptp > 1e-30:
            eta = 10.0 / hptp
            logits += eta * (ha - float(np.mean(ha)))
        lambda_w = norm_logits(logits)

    return g


def lp_direction(f, active_idx, time_limit=6.0):
    if not SCIPY_OK or not highs_check():
        return None
    f = norm_sum1(f)
    n = f.size
    active_idx = np.asarray(active_idx, dtype=int)
    if active_idx.size == 0 or active_idx.size > 800:
        return None
    K = active_idx.size
    row_parts = []
    col_parts = []
    val_parts = []
    for r, k in enumerate(active_idx):
        s = max(0, int(k) - n + 1)
        e = min(n, int(k) + 1)
        if s >= e:
            continue
        cols = np.arange(s, e, dtype=np.int32)
        row_parts.append(np.full(cols.size, r, dtype=np.int32))
        col_parts.append(cols)
        val_parts.append(f[int(k) - cols].astype(np.float64))
    if not row_parts:
        return None
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0
    neg_one = scipy.sparse.csc_matrix(-np.ones((K, 1)), dtype=np.float64)
    A_ub = scipy.sparse.hstack((A, neg_one), format="csr")
    A_eq = np.ones((1, n + 1), dtype=np.float64)
    A_eq[0, n] = 0.0
    b_eq = np.ones(1, dtype=np.float64)
    b_ub = np.zeros(K, dtype=np.float64)
    bounds = [(0.0, None)] * n + [(None, None)]
    try:
        res = scipy.optimize.linprog(
            c_obj, A_ub=A_ub, b_ub=b_ub,
            A_eq=A_eq, b_eq=b_eq, bounds=bounds,
            method="HiGHS", options={"disp": False, "time_limit": float(time_limit)}
        )
    except Exception:
        return None
    if res is None or not getattr(res, "success", False):
        return None
    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.where(np.isfinite(g) & (g > 0.0), g, 0.0)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs


def line_search_mix(a, g, cf, cg, fg, current_M, t):
    n = a.size
    base_alphas = [
        max(0.01, 1.0 / (t + 1.0)),
        max(0.02, 2.0 / (t + 2.0)),
        0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.50, 0.70, 1.0
    ]
    alphas = []
    seen = set()
    for al in base_alphas:
        al = float(al)
        if not math.isfinite(al) or al <= 0.0 or al > 1.0:
            continue
        key = int(round(al * 1000.0))
        if key in seen:
            continue
        seen.add(key)
        alphas.append(al)
    if not alphas:
        alphas = [0.1]
    best_alpha = 0.0
    best_m = current_M
    for alpha in alphas:
        one = 1.0 - alpha
        val = (one * one) * cf + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
        m = float(np.max(val))
        if math.isfinite(m) and m < best_m:
            best_m = m
            best_alpha = alpha
    if best_alpha <= 0.0:
        return a.copy(), best_m, 0.0
    new_a = (1.0 - best_alpha) * a + best_alpha * g
    new_a = np.maximum(new_a, 0.0)
    s = float(new_a.sum())
    if s < 1e-15:
        return a.copy(), best_m, 0.0
    new_a = new_a / s
    return new_a, best_m, best_alpha


def frank_wolfe_search(a0, budget, active_max=3000, game_iters=35,
                       exact_every=5, exact_time=5.0, verbose=True):
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = proxy_max(c)
    best_a = a.copy()
    best_M = M
    active = active_top(c, active_max)
    no_improve = 0
    t = 0
    while time.monotonic() - t0 < time_limit - 3 and no_improve < 120:
        t += 1
        time_left = time_limit - (time.monotonic() - t0)
        if time_left < 20:
            break
        g = None
        if (highs_check() and exact_every > 0 and
                (t % exact_every == 0) and time_left > 80 and active.size <= 800):
            g = lp_direction(a, active, exact_time)
        if g is None:
            init_log = None
            if active.size > 0:
                av = c[active]
                av = np.where(np.isfinite(av) & (av > 0.0), av, 1e-300)
                init_log = np.log(av)
                init_log -= float(np.max(init_log))
            g = game_direction(a, active, mw_iters=game_iters,
                              seed=1000 + t, init_logits=init_log)
        if g is None:
            break
        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        new_a, new_M, alpha = line_search_mix(a, g, c, cg, fg, M, t)
        if new_M < M - 1e-16:
            if alpha > 0:
                one = 1.0 - alpha
                c = (one * one) * c + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
                c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
            a = new_a
            M = new_M
            no_improve = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            active = active_top(c, active_max)
            if verbose and (t == 1 or t % 15 == 0):
                print("    FW t=" + str(t) +
                      " proxy=" + repr(2.0 * n * best_M) +
                      " alpha=" + repr(alpha))
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                rng = np.random.default_rng(9973 + t)
                sigma = 0.02 + 0.01 * (no_improve // 20)
                pert = rng.normal(0.0, sigma, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-12)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = proxy_max(c)
                active = active_top(c, active_max)
                no_improve = 0
    return best_a


def apply_single_move(c, a, s, d, tmove):
    n = a.size
    c_new = c.copy()
    c_new[d:d + n] += 2.0 * tmove * a
    c_new[s:s + n] -= 2.0 * tmove * a
    t2 = tmove * tmove
    if 2 * d < c_new.size:
        c_new[2 * d] += t2
    if 2 * s < c_new.size:
        c_new[2 * s] += t2
    if s + d < c_new.size:
        c_new[s + d] -= 2.0 * t2
    return np.where(np.isfinite(c_new), c_new, 0.0)


def coordinate_polish(a0, budget, verbose=True):
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    c = fft_conv(a)
    M = proxy_max(c)
    best_a = a.copy()
    best_M = M
    n = a.size
    stall = 0
    rnd = 0
    rng = np.random.default_rng(42)
    temperature = 0.0

    while time.monotonic() - t0 < time_limit - 2 and stall < 200:
        rnd += 1
        if n <= 1:
            break
        top_k = min(50, 2 * n - 1)
        top_idx = active_top(c, top_k)
        if top_idx.size == 0:
            break
        p_top = np.zeros(2 * n - 1, dtype=np.float64)
        p_top[top_idx] = 1.0 / float(top_idx.size)
        pressure = cross_convolve_p_a(p_top, a)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)

        src_count = min(12, n)
        dst_count = min(16, n)
        src_score = pressure * (1.0 + float(n) * a)
        src_candidates = np.argpartition(src_score, min(src_count - 1, n - 1))[:src_count]
        dst_score = pressure + 1e-5 * a
        dst_candidates = np.argpartition(dst_score, min(dst_count - 1, n - 1))[:dst_count]

        improved_any = False
        best_cand = None
        best_cand_M = M - 1e-18

        for srr in src_candidates:
            s = int(srr)
            if a[s] < 1e-13:
                continue
            for drr in dst_candidates:
                d = int(drr)
                if d == s:
                    continue
                for frac in (0.003, 0.008, 0.015, 0.03, 0.06):
                    tmove = a[s] * frac
                    if tmove <= 1e-16:
                        continue
                    tmove = min(tmove, a[s] * 0.99)
                    c_new = apply_single_move(c, a, s, d, tmove)
                    m_new = proxy_max(c_new)
                    if m_new < best_cand_M:
                        best_cand = (m_new, s, d, tmove)
                        best_cand_M = m_new
                        improved_any = True

        if not improved_any:
            # Simulated annealing: try random moves, accept if slightly worse
            temperature = 0.001 * max(0, 1.0 - rnd / 50.0)
            for _ in range(40):
                s = int(rng.integers(n))
                d = int(rng.integers(n))
                if s == d or a[s] < 1e-12:
                    continue
                frac = float(rng.uniform(0.002, 0.04))
                tmove = a[s] * frac
                tmove = min(tmove, a[s] * 0.99)
                if tmove <= 1e-16:
                    continue
                c_new = apply_single_move(c, a, s, d, tmove)
                m_new = proxy_max(c_new)
                if m_new < best_cand_M:
                    best_cand = (m_new, s, d, tmove)
                    best_cand_M = m_new
                    improved_any = True
                elif temperature > 0 and m_new < M + temperature:
                    # Accept worse move with probability based on temperature
                    if rng.random() < math.exp(-(m_new - M) / max(temperature, 1e-10)):
                        best_cand = (m_new, s, d, tmove)
                        best_cand_M = m_new
                        improved_any = True

        if best_cand is not None:
            m_new, s, d, tmove = best_cand
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = fft_conv(a)
            M = proxy_max(c)
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
                stall = 0
            else:
                if m_new >= best_M:
                    stall += 1
                    # Revert if annealing move made things worse
                    if m_new > M + 1e-16:
                        a = best_a.copy()
                        c = fft_conv(a)
                        M = best_M
                else:
                    stall += 1
            if verbose and (rnd <= 3 or rnd % 50 == 0):
                print("    Coord rnd=" + str(rnd) +
                      " proxy=" + repr(2.0 * n * best_M))
        else:
            stall += 1
            if stall % 15 == 14:
                # Random restart
                a = best_a + rng.normal(0.0, 0.01, n) * best_a
                a = np.maximum(a, 1e-12)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = proxy_max(c)
                stall = 0

    return best_a


def default_sequence(n=5000):
    x = np.linspace(0.0, 1.0, int(n))
    return norm_sum1(np.sin(np.pi * x) ** 2)


def cosine_sequence(n=5000):
    x = np.linspace(0.0, 1.0, int(n))
    return norm_sum1(0.5 + 0.5 * np.cos(2 * np.pi * x))


def raised_cosine_seq(n=5000):
    x = np.linspace(0.0, 1.0, int(n))
    return norm_sum1(np.cos(np.pi * (x - 0.5)) ** 2)


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    try:
        total_budget = max(60.0, min(float(budget_s) - 60.0, 1740.0))
    except Exception:
        total_budget = 1740.0

    print("Start: budget=" + repr(total_budget))

    # Get parent sequence
    seq_best = None
    try:
        parent_list = height_sequence_1
    except Exception:
        parent_list = None

    if parent_list is not None:
        try:
            p = np.asarray(parent_list, dtype=np.float64)
            p = np.where(np.isfinite(p), p, 0.0)
            np.clip(p, 0.0, 1000.0, out=p)
            if p.ndim != 1:
                p = p.ravel()
            if p.size > 0 and float(p.sum()) > 1e-10:
                seq_best = norm_sum1(p)
        except Exception:
            seq_best = None

    if seq_best is None:
        seq_best = default_sequence(5000)

    best_score = safe_eval(seq_best)
    if not math.isfinite(best_score):
        best_score = 2.0 * seq_best.size * proxy_max(fft_conv(seq_best))

    print("Initial: n=" + str(seq_best.size) + " score=" + repr(best_score))

    # Compress if too long
    max_work_len = 9000
    if seq_best.size > max_work_len:
        work_seq = interpolate_sequence(seq_best, max_work_len)
    else:
        work_seq = seq_best.copy()

    seq_best, best_score = try_update(seq_best, best_score, work_seq, "init", t_start)

    # Phase 1: L-BFGS from multiple starts
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 300:
        phase_budget = min(time_left * 0.38, 650.0)
        print("\n=== Phase 1: L-BFGS multi-start (budget=" + repr(phase_budget) + "s) ===")

        # Start 1: parent
        a1 = lbfegs_schedule(work_seq, phase_budget * 0.55)
        seq_best, best_score = try_update(seq_best, best_score, a1, "lb_parent", t_start)

        time_left = total_budget - (time.monotonic() - t_start)
        if time_left > 100:
            # Start 2: raised cosine
            n_w = work_seq.size
            a2_init = raised_cosine_seq(n_w)
            a2 = lbfegs_schedule(a2_init, min(phase_budget * 0.25, time_left * 0.3))
            seq_best, best_score = try_update(seq_best, best_score, a2, "lb_cosine", t_start)

        time_left = total_budget - (time.monotonic() - t_start)
        if time_left > 80:
            # Start 3: perturbed parent
            rng = np.random.default_rng(123)
            a3_init = work_seq + rng.normal(0.0, 0.05, work_seq.size) * work_seq
            a3_init = np.maximum(a3_init, 0.0)
            a3_init = a3_init / float(a3_init.sum())
            a3 = lbfegs_schedule(a3_init, min(phase_budget * 0.2, time_left * 0.3))
            seq_best, best_score = try_update(seq_best, best_score, a3, "lb_perturb", t_start)

        # Update work sequence from best
        if seq_best.size <= max_work_len:
            work_seq = seq_best.copy()
        else:
            work_seq = interpolate_sequence(seq_best, max_work_len)

    # Phase 2: Frank-Wolfe
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 200:
        phase_budget = min(time_left * 0.28, 420.0)
        print("\n=== Phase 2: Frank-Wolfe (budget=" + repr(phase_budget) + "s) ===")
        a_fw = frank_wolfe_search(work_seq, phase_budget,
                                  active_max=3000, game_iters=40,
                                  exact_every=5, exact_time=5.0, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_fw, "fw", t_start)

        if seq_best.size <= max_work_len:
            work_seq = seq_best.copy()
        else:
            work_seq = interpolate_sequence(seq_best, max_work_len)

    # Phase 3: Coordinate polish
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 150:
        phase_budget = min(time_left * 0.22, 300.0)
        print("\n=== Phase 3: Coordinate polish (budget=" + repr(phase_budget) + "s) ===")
        a_cp = coordinate_polish(work_seq, phase_budget, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_cp, "coord", t_start)

        if seq_best.size <= max_work_len:
            work_seq = seq_best.copy()
        else:
            work_seq = interpolate_sequence(seq_best, max_work_len)

    # Phase 4: Second round of FW + Coord (refinement)
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 180:
        fw_bud = min(time_left * 0.30, 200.0)
        cp_bud = min(time_left * 0.30, 200.0)
        print("\n=== Phase 4: Refinement FW+Coord ===")
        a_fw2 = frank_wolfe_search(work_seq, fw_bud,
                                   active_max=2500, game_iters=30,
                                   exact_every=7, exact_time=4.0, verbose=False)
        seq_best, best_score = try_update(seq_best, best_score, a_fw2, "fw2", t_start)
        if seq_best.size <= max_work_len:
            work_seq = seq_best.copy()
        else:
            work_seq = interpolate_sequence(seq_best, max_work_len)
        time_left = total_budget - (time.monotonic() - t_start)
        if time_left > 40:
            a_cp2 = coordinate_polish(work_seq, min(cp_bud, time_left * 0.5), verbose=False)
            seq_best, best_score = try_update(seq_best, best_score, a_cp2, "coord2", t_start)
        if seq_best.size <= max_work_len:
            work_seq = seq_best.copy()
        else:
            work_seq = interpolate_sequence(seq_best, max_work_len)

    # Phase 5: Length exploration
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 200:
        print("\n=== Phase 5: Length exploration ===")
        base_len = work_seq.size
        length_tries = [base_len + 200, base_len + 400, base_len - 300, base_len + 600]
        tried = set()
        for target in length_tries:
            time_left = total_budget - (time.monotonic() - t_start)
            if time_left < 100:
                break
            target = int(max(1500, min(10000, target)))
            if target in tried:
                continue
            tried.add(target)
            print("  Trying n=" + str(target))
            a0 = interpolate_sequence(work_seq, target)
            if SCIPY_OK and time_left > 150:
                a_len = run_lbfegs(a0, 200.0, maxfun=120, verbose=False)
                seq_best, best_score = try_update(seq_best, best_score, a_len,
                                                  "len_lb_" + str(target), t_start)
            time_left = total_budget - (time.monotonic() - t_start)
            if time_left > 60:
                a_fw_len = frank_wolfe_search(a0, min(time_left * 0.35, 60.0),
                                              active_max=1500, game_iters=20,
                                              exact_every=8, exact_time=3.0,
                                              verbose=False)
                seq_best, best_score = try_update(seq_best, best_score, a_fw_len,
                                                  "len_fw_" + str(target), t_start)
            if seq_best.size <= max_work_len:
                work_seq = seq_best.copy()
            else:
                work_seq = interpolate_sequence(seq_best, max_work_len)

    # Phase 6: Final intensive coordinate polish
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 60:
        phase_budget = min(time_left * 0.5, 120.0)
        print("\n=== Phase 6: Final coordinate polish (budget=" + repr(phase_budget) + "s) ===")
        a_final = coordinate_polish(work_seq, phase_budget, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_final, "final_coord", t_start)

    # Phase 7: Final L-BFGS with very high beta
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if SCIPY_OK and time_left > 40:
        print("\n=== Phase 7: Final L-BFGS high beta ===")
        if seq_best.size <= max_work_len:
            work_seq = seq_best.copy()
        else:
            work_seq = interpolate_sequence(seq_best, max_work_len)
        a_hb = run_lbfegs(work_seq, 5000.0, maxfun=200, verbose=False)
        seq_best, best_score = try_update(seq_best, best_score, a_hb, "final_lb", t_start)

    # Final validation
    final_val = safe_eval(seq_best)
    if math.isfinite(final_val):
        if final_val < best_score:
            best_score = final_val
    else:
        seq_best = default_sequence(5000)
        best_score = safe_eval(seq_best)

    arr = np.asarray(seq_best, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if arr.size == 0 or s < 0.01:
        arr = default_sequence(5000)
    else:
        arr = arr / s

    elapsed = time.monotonic() - t_start
    print("\nFINAL: score=" + repr(best_score) +
          " n=" + str(arr.size) +
          " elapsed=" + repr(elapsed))

    return [float(x) for x in arr]
