def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgradient(a, b=None):
    """Compute the exact subgradient of max_k (a*a)_k at point a.
    Returns the gradient of the active term (the one achieving the max)."""
    if b is None:
        b = fast_autocorr(a)
    k_max = int(np.argmax(b))
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k_max - n + 1)
    je = min(n - 1, k_max)
    if js <= je:
        idx = np.arange(js, je + 1)
        grad[idx] = a[k_max - idx]
    return grad, k_max

def mirror_descent(a0, time_limit, num_iters=5000, verbose=True):
    """Mirror descent with entropy regularization (multiplicative weights).
    a_i <- a_i * exp(-eta * grad_i) / Z
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = 1.0 / np.sqrt(n)
    
    for t in range(1, num_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        b_max = b.max()
        k_max = int(np.argmax(b))
        
        if b_max < best_m - 1e-15:
            best_m = b_max
            best_a = a.copy()
        
        # Compute subgradient
        grad = np.zeros(n)
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        if js <= je:
            idx = np.arange(js, je + 1)
            grad[idx] = a[k_max - idx]
        
        # Multiplicative update
        # a_i <- a_i * exp(-eta * grad_i)
        # Use log-domain for numerical stability
        log_a = np.log(np.maximum(a, 1e-300))
        log_a -= eta * grad
        # Normalize
        max_log = log_a.max()
        log_a -= max_log
        weights = np.exp(log_a)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            break
        a = weights / w_sum
        
        # Anneal eta
        eta = max(0.01 / np.sqrt(n), eta * 0.9995)
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  Mirror t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " eta=" + repr(round(eta, 6)))
    
    return best_a, best_m


def multi_peak_mirror_descent(a0, time_limit, top_k=5, verbose=True):
    """Mirror descent targeting multiple peaks simultaneously."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    eta = 2.0 / np.sqrt(n)
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        b_max = b.max()
        
        if b_max < best_m - 1e-15:
            best_m = b_max
            best_a = a.copy()
        
        # Compute weighted gradient from top peaks
        top_indices = np.argsort(b)[::-1][:top_k]
        # Weight by how close to max
        weights = b[top_indices] - b[top_indices[-1]] + 1.0
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js <= je:
                idx = np.arange(js, je + 1)
                grad[idx] += w * a[k_int - idx]
        
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-300))
        log_a -= eta * grad
        max_log = log_a.max()
        log_a -= max_log
        weights_arr = np.exp(log_a)
        w_sum = weights_arr.sum()
        if w_sum < 1e-300:
            break
        a = weights_arr / w_sum
        
        eta = max(0.02 / np.sqrt(n), eta * 0.999)
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  MultiPeak t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
    
    return best_a, best_m


def cyclic_shift_best(a0):
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def local_search(a0, time_limit, verbose=True):

        iterations += 1
        ac = fast_autocorr(a)

            continue

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_autocorr(a):

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def lp_frank_wolfe_aggro(a0, time_limit, solver, initial_K=200, max_K=2500,
                         verbose=True):
    """Aggressive LP-Frank-Wolfe with backtracking line search and adaptive active set."""
    n = len(a0)
    y = norm_sum1(a0).copy()
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 600

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive active set: add violations, prune large slack
        if t % 30 == 0:
            viol = np.where(ac >= M_curr - 1e-8)[0]
            for k in viol[:30]:
                active_set.add(int(k))
            if len(active_set) > max_K:
                slacks = np.array([M_curr - ac[k] for k in active_set])
                sorted_indices = np.argsort(slacks)
                # Remove the ones with largest slack
                to_remove = sorted_indices[slacks[sorted_indices] > 0][max(0, len(slacks) - max_K):]
                active_set = set(int(k) for k in active_set if active_set and True)
                # Simpler: just keep the K most violated
                active_list = list(active_set)
                slacks_arr = np.array([M_curr - ac[k] for k in active_list])
                keep_idx = np.argsort(slacks_arr)[:max_K]
                active_set = set(int(active_list[i]) for i in keep_idx)

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Backtracking line search
        best_t = 0.0
        best_M = M_curr
        alphas = [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]
        fw_alpha = 2.0 / (t + 2.0)
        alphas.append(fw_alpha)
        alphas.append(max(0.01, fw_alpha * 2))

        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_M - 1e-15:
                best_M = M_trial
                best_t = alpha

        # If no improvement, try very small steps
        if best_t == 0.0:
            for alpha in [0.001, 0.002, 0.005]:
                trial = (1.0 - alpha) * y + alpha * g_dir
                trial = np.maximum(trial, 0.0)
                s_t = trial.sum()
                if s_t < 1e-15:
                    continue
                trial = trial / s_t
                M_trial = fast_autocorr(trial).max()
                if M_trial < best_M - 1e-15:
                    best_M = M_trial
                    best_t = alpha

        # Add newly violated constraints
        if best_t > 0:
            y_trial = (1.0 - best_t) * y + best_t * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            s_yt = y_trial.sum()
            if s_yt > 1e-15:
                y_trial = y_trial / s_yt
                ac_trial = fast_autocorr(y_trial)
                new_viol = np.where(ac_trial > M_curr + 1e-7)[0]
                for k in new_viol[:20]:
                    active_set.add(int(k))
                y = y_trial
            else:
                y = best_a.copy()
        else:
            # Stuck - try a random perturbation
            if no_improve > 100:
                rng_local = np.random.default_rng(t)
                y = best_a + rng_local.normal(0, 0.005, n) * best_a
                y = np.maximum(y, 0.0)
                s_y = y.sum()
                if s_y > 1e-15:
                    y = y / s_y
                active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K])

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  LP-FW aggro iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def fista_smoothed(a0, time_limit, beta_start=15.0, beta_end=1000.0,
                   beta_growth=1.003, top_k=15, restart_every=3000, verbose=True):
    """FISTA with softmax-smoothed max and BB step size."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(12345)

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        b = fast_autocorr(y)
        if top_k is not None and 0 < top_k < len(b):
            idx_top = np.argsort(b)[::-1][:top_k]
            b_use = np.zeros(len(b))
            b_use[idx_top] = b[idx_top]
            b_max = b[idx_top].max()
        else:
            b_use = b
            b_max = b.max()

        shifted = beta * (b_use - b_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        p_total = exp_vals.sum()
        if p_total < 1e-300:
            p = np.zeros_like(b_use)
            mask = b_use == b_max
            p[mask] = 1.0 / max(1, mask.sum())
        else:
            p = exp_vals / p_total

        # Gradient via convolution
        a_rev = y[::-1]
        conv_len = n + len(p) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        Ar = np.fft.rfft(a_rev, fft_size)
        Pr = np.fft.rfft(p, fft_size)
        conv = np.fft.irfft(Ar * Pr, fft_size)
        grad = conv[n - 1:2 * n - 1] / beta

        # Step size
        eta = min(1.0 / (2.0 * beta), 0.15)
        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 100 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.01 / (1.0 + 0.03 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 5.0)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("  FISTA t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_max_b


def peak_shaving_search(a0, time_limit, verbose=True):
    """Intelligent peak-shaving: redistribute mass from peak-contributing positions to quiet zones."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:25]
        improved_this_round = False

        for _ in range(80):
            if time.monotonic() - t0 > time_limit:
                break

            k_int = int(top_peaks[rng.integers(min(10, len(top_peaks)))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            if a[src] < 1e-12:
                continue

            # Find quiet destinations (positions not contributing to top peaks)
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:12]:

                in_peak[s2:e2 + 1] = True

            quiet = np.where(~in_peak)[0]
            if len(quiet) < 10:
                quiet = np.arange(n)

            dst = int(quiet[rng.integers(len(quiet))])
            if dst == src:
                continue

            # Try multiple fractions
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-20:

                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)

                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t

                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved_this_round = True
                    ac = fast_autocorr(a)
                    top_peaks = np.argsort(ac)[::-1][:25]
                    break

        if improved_this_round:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.002, 0.02))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()

def cyclic_shift_best(a0):
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    for s in range(1, n):
        AS = np.fft.rfft(np.roll(a0, -s), fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_interp(f, new_n):
    """Spectral interpolation: upsample via FFT."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_fft = np.fft.rfft(f)
    # Zero-pad in frequency domain
    new_fft = np.zeros(new_n // 2 + 1, dtype=np.complex128)
    copy_len = min(len(old_fft), len(new_fft))
    new_fft[:copy_len] = old_fft[:copy_len] * (new_n / old_n)
    f_new = np.fft.irfft(new_fft, new_n)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / ss

        n_base = 8000
        x = np.linspace(0, 1, n_base)
        state[0] = norm_sum1(np.sin(np.pi * x) ** 2)
        state[1] = evaluate_safe(state[0])
        print("Default init: n=" + str(len(state[0])) + ", score=" + repr(state[1]))

    n0 = len(state[0])

    # === Phase 1: FISTA (main optimizer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: FISTA (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.22, 350.0)
    if t1 > 60:
        a_out, _ = fista_smoothed(norm_sum1(state[0]), t1,
                                  beta_start=12.0, beta_end=800.0,
                                  beta_growth=1.003, top_k=12,
                                  restart_every=2500, verbose=True)
        gp_update_score(a_out, "fista1", t_global, state)

    # === Phase 2: LP-Frank-Wolfe (aggressive) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: LP-FW Aggro (n=" + str(len(state[0])) + ") ===")
    t2 = min(remaining * 0.25, 430.0)
    if t2 > 40 and len(state[0]) <= 10000:
        a_out, _ = lp_frank_wolfe_aggro(norm_sum1(state[0]), t2, solver,
                                         initial_K=300, max_K=2000, verbose=True)
        gp_update_score(a_out, "lp1", t_global, state)

    # === Phase 3: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: Peak Shaving (n=" + str(len(state[0])) + ") ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        a_out, _ = peak_shaving_search(norm_sum1(state[0]), t3, verbose=True)
        gp_update_score(a_out, "peak1", t_global, state)

    # === Phase 4: Cyclic shift ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== Phase 4: Cyclic Shift ===")
        a_shifted, _ = cyclic_shift_best(norm_sum1(state[0]))
        gp_update_score(a_shifted, "cyclic1", t_global, state)

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state[0])
    longer_lengths = [n_best + 500, n_best + 1500, n_best + 3000]
    longer_lengths = [l for l in longer_lengths if l <= 12000 and l > n_best]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("\n=== Phase 5: Scale to n=" + str(new_n) + " ===")
        a_long = interpolate_sequence(state[0], new_n)

        t5a = min(remaining * 0.20, 150.0)
        if t5a > 30:
            a_out, _ = fista_smoothed(a_long, t5a,
                                       beta_start=15.0, beta_end=600.0,
                                       beta_growth=1.003, top_k=10,
                                       restart_every=2000, verbose=True)
            gp_update_score(a_out, "fista_n" + str(new_n), t_global, state)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 9000:
            t5b = min(remaining * 0.20, 120.0)
            if t5b > 20:
                a_out, _ = lp_frank_wolfe_aggro(norm_sum1(state[0]), t5b, solver,
                                                 initial_K=200, max_K=1500, verbose=True)
                gp_update_score(a_out, "lp_n" + str(new_n), t_global, state)

    # === Phase 6: Second LP-FW round on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state[0])
    print("\n=== Phase 6: LP-FW Round 2 (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.20, 250.0)
    if t6 > 40 and n_best <= 10000:
        a_out, _ = lp_frank_wolfe_aggro(norm_sum1(state[0]), t6, solver,
                                         initial_K=500, max_K=3000, verbose=True)
        gp_update_score(a_out, "lp2", t_global, state)

    # === Phase 7: Peak shaving round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.08, 100.0)
    if t7 > 20:
        print("\n=== Phase 7: Peak Shaving 2 ===")
        a_out, _ = peak_shaving_search(norm_sum1(state[0]), t7, verbose=True)
        gp_update_score(a_out, "peak2", t_global, state)

    # === Phase 8: FISTA with high beta (fine-tuning) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 8: FISTA Fine (n=" + str(len(state[0])) + ") ===")
    t8 = min(remaining * 0.10, 120.0)
    if t8 > 20:
        a_out, _ = fista_smoothed(norm_sum1(state[0]), t8,
                                   beta_start=80.0, beta_end=2000.0,
                                   beta_growth=1.008, top_k=6,
                                   restart_every=1500, verbose=True)
        gp_update_score(a_out, "fista_fine", t_global, state)

    # === Final: Cyclic + Local ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== Final: Cyclic ===")
        a_shifted, _ = cyclic_shift_best(norm_sum1(state[0]))
        gp_update_score(a_shifted, "cyclic_final", t_global, state)

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 25:
        t_f = min(remaining * 0.4, 80.0)
        if t_f > 15:
            print("\n=== Final: Peak Shaving ===")
            a_out, _ = peak_shaving_search(norm_sum1(state[0]), t_f, verbose=True)
            gp_update_score(a_out, "peak_final", t_global, state)

    # Final FISTA micro-polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t_fp = min(remaining * 0.5, 80.0)
        if t_fp > 15:
            print("\n=== Final: FISTA Micro ===")
            a_out, _ = fista_smoothed(norm_sum1(state[0]), t_fp,
                                       beta_start=200.0, beta_end=3000.0,
                                       beta_growth=1.01, top_k=5,
                                       restart_every=1000, verbose=True)
            gp_update_score(a_out, "fista_micro", t_global, state)

    if state[0] is not None:
        fs = evaluate_safe(state[0])
        if np.isfinite(fs):
            state[1] = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(state[1]) + ", n=" + str(len(state[0])) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 8000
        x = np.linspace(0, 1, n_fb)
        state[0] = norm_sum1(np.sin(np.pi * x) ** 2)
        state[1] = evaluate_safe(state[0])

    return [float(x) for x in np.asarray(state[0], dtype=np.float64)]
