def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Compute a * b (convolution) using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_constraint_rows(y, constraint_list, n):
    """Build rows of the constraint matrix for LP.
    For each lag k, the row has y[k-i] at position i (for valid i)."""
    K = len(constraint_list)
    # Use sparse representation for efficiency
    rows = []
    cols = []
    vals = []
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            rows.append(jj)
            cols.append(i)
            vals.append(y[k_int - i])
    return rows, cols, vals


def lp_step(y, active_set, n, solver):
    """Solve the LP: min M s.t. sum(g)=1, g>=0, (g*y)[k]<=M for k in active_set."""
    K = len(active_set)
    if K == 0:
        return None, None
    
    rows, cols, vals = build_constraint_rows(y, active_set, n)
    
    if len(rows) == 0:
        return None, None
    
    import scipy.sparse as sp
    A_sparse = sp.csr_matrix((vals, (rows, cols)), shape=(K, n))
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1, A_sparse @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, M_var.value if M_var.value is not None else None
    
    g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return None, None
    
    return g_val / g_sum, M_var.value

def smooth_grad_descent(a0, time_limit, num_learns=1000, verbose=False):
    """Gradient descent with log-sum-exp smoothing of the max."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_convolve(y).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    # Temperature for smoothing
    T = 0.01 * n  # scale with n
    
    x = y.copy()
    v = np.zeros(n)  # momentum
    beta = 0.9
    lr = 0.01
    lr_decay = 0.999
    
    for step in range(num_learns):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute smoothed "peak" using log-sum-exp
        ac = fast_convolve(x)
        # Use only the top portion to be efficient
        top_k = min(200, len(ac))
        top_indices = np.argsort(ac)[::-1][:top_k]
        ac_top = ac[top_indices]
        
        # Log-sum-exp smoothing
        ac_shifted = ac_top - ac_top.max()
        log_sum_exp = ac_top.max() + np.log(np.sum(np.exp(ac_shifted / T)))
        smooth_val = log_sum_exp
        
        # Gradient of smooth_val w.r.t. x
        # d/dx_i of (x*x)[k] = x[k-i] + x[i-k+n-1] (symmetric part)
        # For the log-sum-exp: grad = sum_k exp((ac[k] - max)/T) * d(ac[k])/dx / sum_k exp(...)
        
        weights = np.exp(ac_shifted / T)
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for jj in range(top_k):
            k = int(top_indices[jj])
            w = weights[jj]
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            grad[idx] += w * x[k - idx]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-30:
            grad = grad / grad_norm
        
        # Update with momentum
        v = beta * v - lr * grad
        x = x + v
        x = np.maximum(x, 0.0)
        s_x = x.sum()
        if s_x < 1e-15:
            x = best_a.copy()
        else:
            x = x / s_x
        
        # Track best
        M_curr = fast_convolve(x).max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = x.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print("    Smooth GD new best (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))
        
        # Decay learning rate
        lr *= lr_decay
        
        # Anneal temperature
        if step % 200 == 0 and step > 0:

"""
Enhanced LP-Frank-Wolfe optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: adaptive active-set LP, away-step FW, diverse initializations (FFT-magnitude,
dithered), multi-n exploration, and coordinate polish.
"""

import numpy as np
import time
import scipy.sparse as sp_sparse
import cvxpy as cp

_MODULE_SOLVER = None


def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(200, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_sparse_constraint(y, constraint_list, n):
    K = len(constraint_list)
    rows = []
    cols = []
    vals = []
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        rows.extend([(jj, i) for i in idx])
        vals.extend([float(y[k_int - i]) for i in idx])
    if not rows:
        return None
    r = np.array([x[0] for x in rows], dtype=np.int32)
    c = np.array([x[1] for x in rows], dtype=np.int32)
    v = np.array(vals, dtype=np.float64)
    return sp_sparse.csr_matrix((v, (r, c)), shape=(K, n))


def solve_lp(y, active_set, n, solver):
    K = len(active_set)
    if K == 0:
        return None
    A_sparse = build_sparse_constraint(y, active_set, n)
    if A_sparse is None:
        return None
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1, A_sparse @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    g_val = np.asarray(g_var.value, dtype=np.float64)
    if g_val is None:
        return None
    g_val = np.maximum(g_val, 0.0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return None
    return g_val / g_sum

    stagnation_count = 0

        if no_improve > max_no_improve:

                    print("    LP-FW: persistent stagnation, stopping")
                break
            ac_curr = fast_convolve(y)
            n_pert = max(30, len(active_set) // 5)
            keep = list(rng.choice(len(active_set), size=len(active_set) - n_pert, replace=False))
            new_adds = np.argsort(ac_curr)[::-1][:n_pert * 2]
            new_set = set([active_set[i] for i in keep])
            for k in new_adds:
                new_set.add(int(k))
            active_set = sorted(new_set)
            no_improve = 0
            stagnation_count += 1
            continue

        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            stagnation_count = 0
        else:
            no_improve += 1

        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep_count = int(max_K * 0.5)
            keep_idx = np.argsort(margins)[:keep_count]
            active_set_base = [active_set[i] for i in keep_idx]
            top_peaks = np.argsort(ac)[::-1][:100]
            for v in top_peaks:
                if int(v) not in active_set_base:
                    active_set_base.append(int(v))
            active_set = sorted(set(active_set_base))

        K = len(active_set)
        g_dir = solve_lp(y, active_set, n, solver)

        if g_dir is None:
            active_set = sorted(active_set[:max(50, K // 4)])
            continue

        cross_conv = fast_cross_convolve(g_dir, y)
        all_viol = np.where(cross_conv > M_curr + 1e-10)[0]
        if len(all_viol) > 0:
            viol_amt = cross_conv[all_viol] - M_curr
            top_v = all_viol[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        best_alpha = 1.0 / (iter_count + 2.0)
        best_trial_M = ac.max()

        alphas = [1.0, 0.7, 0.5, 0.3, 0.15,
                  1.0 / (iter_count + 1.0),
                  2.0 / (iter_count + 2.0),
                  0.1, 0.5 / (iter_count + 1.0)]
        alphas = list(set(max(0.01, min(1.0, a)) for a in alphas))
        alphas.sort(reverse=True)

        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
                if M_trial < M_curr * 0.9:
                    break

        if no_improve > 80:
            i_max = int(np.argmax(y))
            for away_frac in [0.02, 0.05, 0.1, 0.15]:
                trial = y.copy()
                trial[i_max] *= (1.0 - away_frac)
                trial = np.maximum(trial, 0.0)
                s_t = trial.sum()
                if s_t < 1e-15:
                    continue
                trial = trial / s_t
                M_trial = fast_convolve(trial).max()
                if M_trial < best_trial_M * 0.99:
                    y = trial
                    break
            else:
                y = (1.0 - best_alpha) * y + best_alpha * g_dir
                y = np.maximum(y, 0.0)
                s_y = y.sum()
                if s_y < 1e-15:
                    y = best_a.copy()
                else:
                    y = y / s_y
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", |S|=" + str(len(active_set)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", iters=" + str(iter_count))
    return best_a, best_m


def fft_magnitude_init(n, seed=42):
    rng = np.random.default_rng(seed)
    phases = rng.uniform(0, 2 * np.pi, n)
    signal = np.exp(1j * phases)
    a = np.abs(np.fft.irfft(signal * n))[:n]
    return norm_sum1(a)


def dithered_init(n, seed=123):
    rng = np.random.default_rng(seed)
    base = np.ones(n)
    r = rng.uniform(-1, 1, n)
    a = base + 0.5 * r
    return norm_sum1(a)


def coordinate_polish(a0, time_limit, rng_seed=77, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 600
    iter_count = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iter_count += 1
        ac = fast_convolve(a)
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            stress = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    stress[idx2] += a[k2i - idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 10:
                candidates = np.arange(n)
            cand_stress = stress[candidates]
            dst = int(candidates[np.argmin(cand_stress)])

            if dst == src:
                continue

            found = False
            for frac in [0.003, 0.008, 0.02, 0.05, 0.1, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and rng.random() < 0.2:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 200 == 0:
                for _ in range(15):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.003, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def get_initial_sequence():
    best_seq = None
    best_score = float('inf')
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                best_seq = p_norm
                best_score = sc
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
    except Exception:
        pass

    if best_seq is None:
        n = 8000
        x = np.linspace(0, 1, n)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))
    return best_seq, best_score


def generate_diverse_inits(n, n_candidates=5):
    inits = []
    inits.append(norm_sum1(fft_magnitude_init(n, seed=42)))
    inits.append(norm_sum1(fft_magnitude_init(n, seed=123)))
    inits.append(norm_sum1(dithered_init(n, seed=77)))
    x = np.linspace(0, 1, n)
    inits.append(norm_sum1(np.sin(np.pi * x) ** 2))
    inits.append(norm_sum1(np.ones(n) * (1.0 + 0.3 * np.cos(2 * np.pi * 3 * x))))
    return inits


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = get_solver()
    best_seq, best_score = get_initial_sequence()
    n0 = len(best_seq)

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 0: Try diverse initializations at similar n ===
    print("\n=== Phase 0: Diverse inits (n=" + str(n0) + ") ===")
    inits = generate_diverse_inits(n0, 5)
    for i, init in enumerate(inits):
        sc = evaluate_safe(init)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = init.copy()
            print("  Better init #" + str(i) + ": score=" + repr(sc))

    # === Phase 1: LP-FW on best ===
    print("\n=== Phase 1: LP-FW (n=" + str(len(best_seq)) + ") ===")
    t1 = min(remaining_time() * 0.22, 320.0)
    if t1 > 40 and len(best_seq) <= 15000:
        max_K = 3000 if len(best_seq) <= 8000 else 2000
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t1, solver,
                                  initial_K=400, max_K=max_K, add_per_iter=50, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1", t_global)

    # === Phase 2: LP-FW on a different init ===
    print("\n=== Phase 2: LP-FW on alternate init (n=" + str(n0) + ") ===")
    t2 = min(remaining_time() * 0.12, 180.0)
    if t2 > 30 and n0 <= 15000:
        alt_init = inits[0].copy()
        if best_score < evaluate_safe(alt_init):
            alt_init = norm_sum1(0.5 * best_seq + 0.5 * alt_init)
        max_K = 2500 if n0 <= 8000 else 1500
        a_out, _ = lp_fw_optimize(alt_init, t2, solver,
                                  initial_K=300, max_K=max_K, add_per_iter=40, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_alt", t_global)

    # === Phase 3: LP-FW second pass on best ===
    print("\n=== Phase 3: LP-FW second (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining_time() * 0.20, 300.0)
    if t3 > 40 and len(best_seq) <= 15000:
        max_K = 3500 if len(best_seq) <= 8000 else 2500
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t3, solver,
                                  initial_K=500, max_K=max_K, add_per_iter=60, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2", t_global)

    # === Phase 4: Multi-n exploration ===
    print("\n=== Phase 4: Multi-n exploration ===")
    n_best = len(best_seq)
    try_ns = []
    for delta in [-2000, -1000, -500, 500, 1000, 2000, 3000, 5000, 7000]:
        nn = n_best + delta
        if 3000 <= nn <= 25000:
            try_ns.append(nn)
    try_ns = sorted(set(try_ns), key=lambda l: abs(l - n_best))

    for new_n in try_ns[:5]:
        rt = remaining_time()
        if rt < 100:
            break
        print("\n  Multi-n: n=" + str(new_n) + " (remaining=" + str(round(rt, 1)) + "s)")
        a_long = interpolate_seq(best_seq, new_n)
        t4 = min(rt * 0.12, 140.0)
        if t4 > 25 and new_n <= 20000:
            max_K = 2500 if new_n <= 8000 else 1500
            a_out, _ = lp_fw_optimize(a_long, t4, solver,
                                      initial_K=300, max_K=max_K, add_per_iter=40, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

    # === Phase 5: LP-FW third pass ===
    print("\n=== Phase 5: LP-FW third (n=" + str(len(best_seq)) + ") ===")
    t5 = min(remaining_time() * 0.18, 250.0)
    if t5 > 40 and len(best_seq) <= 20000:
        max_K = 4000 if len(best_seq) <= 8000 else 2500
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t5, solver,
                                  initial_K=600, max_K=max_K, add_per_iter=70, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw3", t_global)

    # === Phase 6: Coordinate polish ===
    print("\n=== Phase 6: Coordinate polish (n=" + str(len(best_seq)) + ") ===")
    t6 = min(remaining_time() * 0.10, 100.0)
    if t6 > 15:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t6, rng_seed=99, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord", t_global)

    # === Phase 7: Final LP-FW ===
    print("\n=== Phase 7: Final LP-FW (n=" + str(len(best_seq)) + ") ===")
    t7 = min(remaining_time() * 0.15, 180.0)
    if t7 > 30 and len(best_seq) <= 20000:
        max_K = 4000 if len(best_seq) <= 8000 else 2500
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t7, solver,
                                  initial_K=800, max_K=max_K, add_per_iter=80, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # Final coordinate polish
    rt = remaining_time()
    if rt > 15:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), min(rt * 0.6, 60.0), rng_seed=2024, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
