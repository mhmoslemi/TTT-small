def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / n
    return arr / s


def fft_conv(a):
    """Compute a * a (convolution) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def fft_cross(a, b):
    """Compute a * b (cross-convolution) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def hankel_mv(p, a):
    """Compute P@a where P[i,j] = p[i+j].
    p has length 2n-1, a has length n.
    Result has length n.
    Uses the identity: (Pa)[i] = (rev(a) * p)[i + n - 1]
    """
    n = a.size
    p = np.asarray(p, dtype=np.float64)
    a_rev = a[::-1].copy()
    total = (2*n - 1) + n - 1  # max length of conv of (2n-1) and n
    N = next_pow2(total)
    A = np.fft.rfft(a_rev, N)
    P = np.fft.rfft(p, N)
    c = np.fft.irfft(A * P, N)
    # (rev(a) * p)[k] for k = n-1, n, ..., 2n-2
    out = c[n-1:2*n-1]
    out = np.where(np.isfinite(out), out, 0.0)
    return out


def proj_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return out / s

def lse_objective_grad(a, beta, mask=None):
    """Compute log-sum-exp smoothed objective and gradient.
    f(a) = (1/beta) * log(sum_k exp(beta * c[k]))
    where c = a*a (convolution), restricted to mask if given.
    """
    n = a.size
    c = fft_conv(a)
    size = 2 * n - 1
    
    if mask is None:
        vals = c
    else:
        vals = c[mask]
    
    if vals.size == 0:
        return 0.0, np.zeros(n)
    
    m = float(np.max(vals))
    shifted = beta * (vals - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0 or not math.isfinite(se):
        e = np.zeros(vals.size)
        e[0] = 1.0
        se = 1.0
    p = e / se
    
    # Reconstruct full pressure vector
    pf = np.zeros(size, dtype=np.float64)
    if mask is None:
        pf[:] = p
    else:
        pf[mask] = p

    grad = np.where(np.isfinite(grad), grad, 0.0)
    
    f = m + math.log(se) / beta
    if not math.isfinite(f):
        f = 1e100
    return f, grad

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / n
    return arr / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def fft_cross(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if b.size != n:
        b = np.resize(b, n)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def hankel_mv(p, a):
    """Compute (P a)[i] = sum_j p[i+j] * a[j] using FFT."""
    n = a.size
    p = np.asarray(p, dtype=np.float64)
    a_rev = a[::-1].copy()
    total = (2 * n - 1) + n - 1
    N = next_pow2(total)
    A = np.fft.rfft(a_rev, N)
    P = np.fft.rfft(p, N)
    c = np.fft.irfft(A * P, N)
    out = c[n - 1:2 * n - 1]
    out = np.where(np.isfinite(out), out, 0.0)
    return out

def lse_obj_grad(a, beta, mask=None):
    n = a.size
    c = fft_conv(a)
    size = 2 * n - 1
    if mask is None:
        vals = c
    else:
        vals = c[mask]
    if vals.size == 0:
        return 0.0, np.zeros(n)
    m = float(np.max(vals))
    shifted = beta * (vals - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0 or not math.isfinite(se):
        e = np.zeros(vals.size)
        e[0] = 1.0
        se = 1.0
    p = e / se
    pf = np.zeros(size, dtype=np.float64)
    if mask is None:
        pf[:] = p
    else:
        pf[mask] = p
    grad = 2.0 * hankel_mv(pf, a)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    f = m + math.log(se) / beta
    if not math.isfinite(f):
        f = 1e100
    return f, grad


def softmax_pp(v):
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = 0
    if v.size > 0:
        u[0] = 0.0
        u[1:] = v
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0.0:
        return np.ones(n) / n
    return e / s


def make_log_param(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 1:
        return np.array([], dtype=np.float64)
    a_safe = np.maximum(a, 1e-300)
    loga = np.log(a_safe)
    loga -= loga[0]
    return loga[1:]


def from_log_param(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    if v.size > 0:
        u[1:] = v
    return softmax_pp(v)


def obj_prob(v, beta, mask=None):
    a = from_log_param(v)
    f, ga = lse_obj_grad(a, beta, mask)
    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)
    gv = np.zeros(v.size, dtype=np.float64)
    if v.size > 0:
        gv[:] = gu[1:]
    gv = np.where(np.isfinite(gv), gv, 0.0)
    return f, gv


def run_lbfegs(a0, beta, mask, maxfun=400):
    a = norm_sum1(a0)
    n = a.size
    if n <= 1 or not SCIPY_OK or maxfun <= 0:
        return a
    v0 = make_log_param(a)
    if v0.size == 0:
        return a
    options = {
        "maxfun": int(maxfun),
        "maxiter": int(maxfun + 30),
        "ftol": 1e-14,
        "gtol": 1e-8,
        "disp": False
    }
    try:
        res = scipy.optimize.minimize(
            obj_prob, v0, args=(beta, mask),
            method="L-BFGS-B", jac=True, options=options
        )
        if res is not None and hasattr(res, "x"):
            vx = np.asarray(res.x, dtype=np.float64)
            if vx.size == v0.size and np.all(np.isfinite(vx)):
                return norm_sum1(from_log_param(vx))
    except Exception:
        pass
    return a


def make_top_mask(a, top_count, base_mask=None):
    n = a.size
    size = 2 * n - 1
    c = fft_conv(a)
    if top_count <= 0:
        return base_mask
    if top_count >= size:
        return base_mask
    if base_mask is None:
        mask = np.ones(size, dtype=bool)
    else:
        mask = base_mask.copy()
    vals = c.copy()
    vals[~mask] = -np.inf
    top = np.argsort(vals)[::-1][:top_count]
    m = np.zeros(size, dtype=bool)
    m[top] = True
    m[~mask] = False
    if not np.any(m):
        return base_mask
    return m


def game_direction(f, active_idx, mw_iters=25, seed=0):
    f = norm_sum1(f)
    n = f.size
    size = 2 * n - 1
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K <= 0:
        return np.ones(n) / n

    rng = np.random.default_rng(seed)
    logits = rng.normal(0.0, 0.05, K)

    def norm_logits(loc):
        loc = np.where(np.isfinite(loc), loc, -1000.0)
        loc = loc - float(np.max(loc))
        w = np.exp(loc)
        s = float(w.sum())
        if s <= 0.0:
            return np.ones(K) / K
        return w / s

    lambda_w = norm_logits(logits)
    g = np.ones(n) / n
    for _ in range(max(1, mw_iters)):
        pf = np.zeros(size, dtype=np.float64)
        pf[active_idx] = lambda_w
        pressure = hankel_mv(pf, f)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)
        pmx = float(np.max(pressure))
        pmn = float(np.min(pressure))
        ptp = pmx - pmn
        if ptp <= 1e-30:
            g = np.ones(n) / n
        else:
            scaled = (pressure - pmn) / ptp
            w = np.exp(-8.0 * scaled)
            w = np.where(np.isfinite(w), w, 0.0)
            w += 1e-12
            g = w / float(w.sum())
            g = 0.97 * g + 0.03 * (np.ones(n) / n)
            g = g / float(g.sum())

        h = fft_cross(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmac_ = float(np.max(ha))
        hptp = hmac_ - float(np.min(ha))
        if hptp > 1e-30:
            eta = 12.0 / hptp
            logits += eta * (ha - float(np.mean(ha)))
        lambda_w = norm_logits(logits)
    return g


def line_search(a, g, cf, cg, fg, current_M, mask=None):
    base_alphas = [0.01, 0.02, 0.04, 0.08, 0.15, 0.25, 0.40, 0.60, 0.80, 1.0]
    best_alpha = 0.0
    best_m = current_M
    for alpha in base_alphas:
        one = 1.0 - alpha
        val = (one * one) * cf + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
        if mask is None:
            m = float(np.max(val))
        else:
            m = float(np.max(val[mask]))
        if math.isfinite(m) and m < best_m:
            best_m = m
            best_alpha = alpha
    if best_alpha <= 0.0:
        return a.copy(), best_m, 0.0
    new_a = (1.0 - best_alpha) * a + best_alpha * g
    np.maximum(new_a, 0.0, out=new_a)
    s = float(new_a.sum())
    if s < 1e-15:
        return a.copy(), best_m, 0.0
    return new_a / s, best_m, best_alpha


def frank_wolfe_search(a0, budget, active_max=2000, game_iters=20):
    t0 = time.monotonic()
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    active = np.argsort(c)[::-1][:min(active_max, 2 * n - 1)]
    no_improve = 0
    t = 0
    while time.monotonic() - t0 < budget - 5 and no_improve < 60:
        t += 1
        if time.monotonic() - t0 > budget - 10:
            break
        g = game_direction(a, active, mw_iters=game_iters, seed=5000 + t)
        if g is None:
            break
        cg = fft_conv(g)
        fg = fft_cross(a, g)
        new_a, new_M, alpha = line_search(a, g, c, cg, fg, M)
        if new_M < M - 1e-16 and alpha > 0:
            one = 1.0 - alpha
            c = (one * one) * c + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
            c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
            a = new_a
            M = new_M
            no_improve = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            active = np.argsort(c)[::-1][:min(active_max, 2 * n - 1)]
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                rng = np.random.default_rng(7777 + t)
                pert = rng.normal(0.0, 0.02, n) * best_a
                a = best_a + pert
                np.maximum(a, 1e-12, out=a)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = float(np.max(c))
                active = np.argsort(c)[::-1][:min(active_max, 2 * n - 1)]
                no_improve = 0
    return best_a


def apply_single_move(c, a, s, d, tmove):
    n = a.size
    c_new = c.copy()
    c_new[d:d + n] += 2.0 * tmove * a
    c_new[s:s + n] -= 2.0 * tmove * a
    t2 = tmove * tmove
    if 2 * d < c_new.size:
        c_new[2 * d] += t2
    if 2 * s < c_new.size:
        c_new[2 * s] += t2
    if s + d < c_new.size:
        c_new[s + d] -= 2.0 * t2
    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
    return c_new

        src_count = min(10, n)
        dst_count = min(15, n)
        src_score = pressure * (1.0 + float(n) * a)
        src_cands = np.argpartition(src_score, min(src_count - 1, n - 1))[:src_count]
        dst_score = pressure + 1e-6 * a
        dst_cands = np.argpartition(dst_score, min(dst_count - 1, n - 1))[:dst_count]

        improved = False

def interpolate_seq(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == 0:
        return np.ones(new_n) / new_n
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)

2*n*max(convolve(a,a)) / (sum(a))^2.

def next_pow2(x):
    x = int(x)

def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / n
    return arr / s

    return c
