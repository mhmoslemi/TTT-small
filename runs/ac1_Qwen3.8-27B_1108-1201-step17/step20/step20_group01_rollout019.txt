import numpy as np
import time


def fast_convolution(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def obj_value(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolution(a).max()


def eval_safe(a):

def compute_gradient_direct(a, weights):
    """
    Compute gradient of sum_k weights[k] * (a*a)[k] with respect to a.
    weights has length 2n-1. Returns vector of length n.
    grad[j] = 2 * sum_k weights[k] * a[k-j] (where a[m]=0 outside [0,n-1])
    """
    n = len(a)
    g = np.zeros(n, dtype=np.float64)
    for k in range(len(weights)):
        wk = weights[k]
        if abs(wk) < 1e-15:
            continue
        s = max(0, k - n + 1)
        e = min(n - 1, k)
        if s > e:
            continue
        # a[k-j] for j in [s, e] gives a[k-s], a[k-s-1], ..., a[k-e]
        # which is a[k-e : k-s+1] reversed
        g[s:e + 1] += wk * a[k - e:k - s + 1][::-1]
    return 2.0 * g


def compute_gradient_fft(a, weights):
    """
    Compute gradient using FFT for efficiency when weights have large support.
    grad[j] = 2 * sum_k weights[k] * a[k-j]
    """
    n = len(a)
    # a_ext has length 2n-1: a[0..n-1] followed by zeros
    a_ext = np.zeros(2 * n - 1, dtype=np.float64)
    a_ext[:n] = a
    # We want g[j] = sum_k weights[k] * a_ext[k-j] for j=0..n-1
    # This is cross-correlation of weights and a_ext
    # Use: g = ifft(fft(weights_flipped) * fft(a_ext)) appropriately indexed
    # Actually: define a_rev[m] = a_ext[2n-2-m]
    # g[j] = sum_k weights[k] * a_ext[k-j] = sum_k weights[k] * a_rev[2n-2-k+j]
    # Let l = 2n-2-k: g[j] = sum_l weights[2n-2-l] * a_rev[l+j]
    # = (weights_rev * a_rev)[j + 2n-2] where weights_rev = weights[::-1]
    # Convolution: (w * x)[s] = sum_l w[l] * x[s-l]
    # So (weights_rev * a_rev)[2n-2+j] = sum_l weights_rev[l] * a_rev[2n-2+j-l]
    # = sum_l weights[2n-2-l] * a_rev[2n-2-j-l+2n-2]... this is getting confusing.
    
    # Let me just use the direct approach - it's fast enough for our sizes.
    return compute_gradient_direct(a, weights)


def soft_grad_descent(a0, time_limit, T_start=30.0, T_end=300.0, lr0=0.08,
                      verbose=True, seed=42):
    """
    Soft-max parameterized gradient descent.
    Minimizes sum_k exp(T*b[k]) / sum exp(T*b[k]) * b[k] as T -> max b[k].
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_val = obj_value(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    polyak = a.copy()  # Polyak-Ruppert average
    total_t = int(1 + time_limit / 0.05)  # estimate total iterations
    
    for t in range(1, 2000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        # Anneal temperature
        T = T_start + (T_end - T_start) * min(1.0, elapsed / (time_limit * 0.7))
        
        # Compute convolution
        b = fast_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        
        # Track best
        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
        
        # Compute soft weights (Boltzmann distribution)
        shifted = T * (b - max_b)
        # Clip for numerical stability
        shifted = np.maximum(shifted, -50.0)
        weights_raw = np.exp(shifted)
        weight_sum = weights_raw.sum()
        if weight_sum < 1e-30:
            weights_raw = np.zeros_like(b)
            weights_raw[np.argmax(b)] = 1.0
            weight_sum = 1.0
        p = weights_raw / weight_sum
        
        # Compute gradient
        g = compute_gradient_direct(a, p)
        
        # Adaptive step size
        lr = lr0 / (1.0 + 0.002 * t)
        
        # Gradient step
        a_new = a - lr * g
        
        # Project onto simplex (sum=1, a>=0)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n, dtype=np.float64) / n
        else:
            a_new = a_new / s
        
        # Polyak averaging
        alpha = 1.0 / (t + 1.0)
        polyak = (1.0 - alpha) * polyak + alpha * a_new
        
        a = a_new
        
        # Periodically check Polyak average
        if t % 200 == 0:
            polyak_val = obj_value(polyak)
            if polyak_val < best_val - 1e-14:
                best_val = polyak_val
                best_a = polyak.copy()
            
            # Occasional random perturbation to escape local minima
            if t % 5000 == 0 and cur_val > best_val * 1.001:
                noise_scale = 0.01 * best_a
                a_pert = best_a + noise_scale * rng.standard_normal(n)
                a_pert = np.maximum(a_pert, 0.0)
                sp = a_pert.sum()
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    vp = obj_value(a_pert)
                    if vp < best_val:
                        best_val = vp
                        best_a = a_pert.copy()
                    a = a_pert
        
        if verbose and t % 2000 == 0:
            print("    SGD t={:6d} ({:.1f}s) T={:.1f} J={:.8f} best={:.8f}".format(
                t, elapsed, T, cur_val, best_val))
    
    elapsed = time.monotonic() - t0
    if verbose:
        print("    SGD done ({:.1f}s) best={:.8f}".format(elapsed, best_val))
    return best_a, best_val


def hard_grad_descent(a0, time_limit, K=50, lr0=0.02, verbose=True, seed=123):
    """
    Hard active-set subgradient descent with line search.
    Uses only top-K lags as active constraints.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_val = obj_value(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    for t in range(1, 1000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        if stall > stall_limit:
            break
        
        b = fast_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        
        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find active set: top-K lags and all lags within 1% of max
        threshold = max_b * 0.999
        active_mask = b >= threshold
        # Also include top K
        top_idx = np.argsort(b)[::-1][:K]
        active_mask[top_idx] = True
        
        active_idx = np.where(active_mask)[0]
        if len(active_idx) == 0:
            break
        
        # Compute subgradient (equal weight on active lags)
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[active_idx] = 1.0 / len(active_idx)
        g = compute_gradient_direct(a, w)
        
        # Backtracking line search
        lr = lr0 / (1.0 + 0.001 * t)
        best_step = lr
        best_trial_val = best_val
        
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 2.0, lr * 5.0]:
            if trial_lr > 1.0:
                continue
            a_trial = a - trial_lr * g
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            vt = 2.0 * n * fast_convolution(a_trial).max()
            if vt < best_trial_val - 1e-15:
                best_trial_val = vt
                best_step = trial_lr
        
        a = a - best_step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n, dtype=np.float64) / n
        else:
            a = a / s
        
        # Random restart if stuck
        if stall > 1500 and stall % 500 == 0:
            noise = 0.005 * best_a * rng.standard_normal(n)
            a_r = best_a + noise
            a_r = np.maximum(a_r, 0.0)
            sr = a_r.sum()
            if sr > 1e-10:
                a_r = a_r / sr
                vr = obj_value(a_r)
                if vr < best_val:
                    best_val = vr
                    best_a = a_r.copy()
                a = a_r
                stall = 0
        
        if verbose and t % 1000 == 0:
            print("    HGD t={:6d} ({:.1f}s) K={} J={:.8f} best={:.8f} stall={}".format(
                t, elapsed, len(active_idx), cur_val, best_val, stall))
    
    elapsed = time.monotonic() - t0
    if verbose:
        print("    HGD done ({:.1f}s) best={:.8f}".format(elapsed, best_val))
    return best_a, best_val

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss
