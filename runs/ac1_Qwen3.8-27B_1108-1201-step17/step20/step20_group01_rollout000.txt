import numpy as np
import time
import cvxpy as cp

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

def build_constraint_matrix(a, active_lags, n):
    """Build the matrix C where C[k,j] = a[|k-j|] for the given active lags."""
    K = len(active_lags)
    C = np.zeros((K, n), dtype=np.float64)
    for row, ki in enumerate(active_lags):
        ki = int(ki)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            C[row, idx] = a[ki - idx]
    return C

def lp_fw_optimize(a0, time_limit, solver, init_K=200, max_K=5000, verbose=True):

            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Refresh active set
        if t % 20 == 0:
            top_lags = np.argsort(ac)[::-1]
            target_K = min(max_K, init_K + t * 5)
            # Keep current active + add new top lags
            active_set = set(active_lags)
            for k in top_lags[:target_K]:
                active_set.add(int(k))
            if len(active_set) > max_K:
                active_list = sorted(active_set)
                margins = np.array([M_curr - ac[k] for k in active_list])
                keep = set(active_list[np.argsort(margins)[:max_K]])
                active_set = keep
            active_lags = list(active_set)
        
        active_lags.append(int(np.argmax(ac)))
        active_lags = list(set(active_lags))
        
        K = len(active_lags)

                best_alpha = alpha

def mirror_descent(a0, time_limit, n_top=150, lr0=0.05, verbose=True):
    """Mirror descent with adaptive temperature on top lags."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    stall = 0
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 10000:
            break
        
        ac = fast_autocorr(x)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr
        
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Select top-K lags
        k_sel = min(n_top, len(ac))
        idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = ac[idx]
        
        # Temperature weighting
        temp = 20.0 + 10.0 * (t / 500.0)
        w = np.exp(temp * (top_vals - top_vals[0]))
        w = w / w.sum()
        
        # Compute gradient via FFT
        # grad[j] = 2 * sum_k w_k * a[k-j] for valid k,j
        # This is 2 * (w * a)[j] (cross-correlation)
        n = len(x)
        w_full = np.zeros(2 * n - 1)
        w_full[idx] = w
        # Cross-correlation: (w_full * x)[j] = sum_k w_full[k] * x[k-j]
        # Use FFT: correlate(w_full, x)
        fft_size = 1 << ((2 * n - 1) + n - 1 - 1).bit_length()
        W = np.fft.rfft(w_full, fft_size)
        X = np.fft.rfft(x, fft_size)
        # Cross-correlation via FFT
        X_conj = np.conj(X)
        corr = np.fft.irfft(W * X_conj, fft_size)
        # corr[j] = sum_k w_full[k] * x[k-j] (circular)
        # We want linear: corr[j] for j in [0, n-1]
        grad = 2.0 * corr[:n]
        
        # Mirror descent update
        lr = lr0 / (1.0 + 0.01 * t)
        log_x = np.log(x + 1e-30) - lr * grad
        log_x -= log_x.max()
        x = np.exp(log_x)
        x = np.maximum(x, 1e-25)
        x = x / x.sum()
        
        # Periodic noise
        if stall > 5000 and stall % 1000 == 0:
            rng = np.random.default_rng(stall)
            noise = 0.01 * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x = x_r / x_r.sum()
            stall = 0
        
        if verbose and t % 500 == 0:
            print("    MirrorDescent t={} ({:.1f}s) J={:.7f} stall={}".format(
                t, time.monotonic() - t0, best_val, stall))
    
    return best_a, best_val

def coord_polish(a0, time_limit, n_targets=15, seed=42, verbose=True):
    """Coordinate descent: move mass from peak contributors to quiet positions."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 3000
    
    def compute_pressure(a_, top_lags_):
        pr = np.zeros(n, dtype=np.float64)
        for k in top_lags_:
            ki = int(k)
            ws = max(0, ki)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            pr[idx] += a_[ki - idx]
        return pr
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(n_targets, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False
        
        pr = compute_pressure(a, top_k[:10])
        quiet = np.argsort(pr)[:50]
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            ws = max(0, ki)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-25:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-15:
                continue
            
            for dst in quiet:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = src_val * frac
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t /= st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-16:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if not improved and stall > 500:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.005, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a /= st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
                            improved = True
                break
        
        if not improved:
            stall += 1
            if stall > 1000:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a /= st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
        
        if verbose and stall % 1000 == 0 and stall > 0:
            print("    CoordPolish ({:.1f}s) J={:.7f} stall={}".format(
                time.monotonic() - t0, 2 * n * best_m, stall))
    
    return best_a, 2.0 * n * best_m
