def fft_convolve(a):
    """Fast linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

    n = len(a)

def sslm_optimize(a0, time_limit, verbose=True):
    """
    Smooth Subgradient Log-Sum-Exp Method.
    Minimizes max autocorrelation using log-sum-exp smoothing.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_max = fft_convolve(a).max()
    t0 = time.monotonic()
    
    # Parameters
    tau0 = np.log(2 * n - 1)  # Initial temperature (gives roughly uniform weighting)
    tau_min = 0.1
    momentum = 0.7
    eta0 = 0.1
    
    # Nesterov momentum
    velocity = np.zeros(n)
    x = a.copy()  # Nesterov lookahead point
    
    stagnation = 0
    max_stagnation = 5000
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        # Anneal temperature
        progress = min(1.0, iter_count / max(1, int(time_limit * 2)))
        tau = tau0 * (tau_min / tau0) ** progress
        
        # Learning rate
        eta = eta0 / (np.sqrt(iter_count) + 1.0)
        
        # Nesterov lookahead
        x = a + momentum * velocity
        x = np.maximum(x, 1e-15)
        x = x / x.sum()
        
        # Compute gradient
        grad, curr_max = logsumexp_grad(x, tau)
        
        # Mirror descent update
        a_new = mirror_descent_update(a, grad, eta)
        
        # Update velocity and position
        velocity = momentum * velocity + (a_new - a)
        a = a_new
        
        # Track best
        if curr_max < best_max - 1e-12:
            best_max = curr_max
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        if stagnation > max_stagnation:
            break
        
        if verbose and iter_count % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SSLM iter={iter_count} ({elapsed:.1f}s): max_b={best_max:.8f}, J={2*n*best_max:.6f}, tau={tau:.4f}")
    
    return best_a, best_max

2. Iteratively Reweighted Quadratic (IRQ) with projected gradient on simplex
3. Peak-shaving LP-Frank-Wolfe with adaptive active set
4. Multi-scale coordinate descent polish
5. Multi-n interpolation and exploration
"""

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None


def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    return cp.ECOS


def fft_convolve(a):
    """Fast linear autocorrelation of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fft_cross_convolve(a, b):
    """Fast linear cross-correlation: result[j] = sum_i a[i+j]*b[i] for valid i."""
    na, nb = len(a), len(b)
    total = na + nb - 1
    fft_size = 1 << (total - 1).bit_length()
    Af = np.fft.rfft(a, fft_size)
    Bf = np.fft.rfft(b, fft_size)
    return np.fft.irfft(Af * np.conj(Bf), fft_size)[:nb]


def logsumexp_obj_grad(a, tau):
    """
    Compute value and gradient of the log-sum-exp smoothing:
    f(a) = tau * log(sum_k exp(b_k / tau)) - tau * log(2n-1)
    which approximates max(b) as tau -> 0.
    
    Returns (objective_value, gradient).
    """
    n = len(a)
    m = 2 * n - 1
    b = fft_convolve(a)
    b_max = b.max()
    # Numerically stable log-sum-exp
    b_shifted = (b - b_max) / tau
    max_bs = b_shifted.max()
    log_sum = np.log(np.sum(np.exp(b_shifted - max_bs))) + max_bs
    obj = tau * (log_sum + max_bs) + b_max  # log(sum exp(b_k/tau)) = log_sum + max_bs... 
    # Simpler: just return b_max as the "value" and compute gradient
    # Gradient: grad_j = (2/(m*tau)) * sum_i exp(b_{i+j}/tau) * a_i
    # For numerical stability, factor out exp(b_max/tau)
    W = np.exp((b - b_max) / tau)
    W_sum = W.sum()
    # Normalize W so that sum W = m (like a probability times m)
    W_norm = W / W_sum * m
    # grad_j = (2/m) * sum_i W_norm_{i+j} * a_i / tau * (W_sum/m) ... 
    # Actually: grad_j = (2/(m*tau)) * sum_i exp(b_{i+j}/tau) * a_i
    # = (2/(m*tau)) * exp(b_max/tau) * sum_i W_{i+j} * a_i
    # The exp(b_max/tau) factor is constant w.r.t. a, so for optimization we can drop it
    # and just use: grad_j = (2/(m)) * sum_i (W_{i+j}/tau_normalized) * a_i
    # Let's use the normalized version for numerical stability:
    # grad_j = (2/(m)) * sum_i (W_{i+j} / W_sum * m / tau) * a_i ... 
    # Simplify: grad_j = (2/tau) * (1/m) * sum_i W_{i+j} * a_i / (W_sum/m) * (W_sum/m)
    # Just use: grad_j = (2/tau) * (W_norm * a cross-corr)_j / m
    # where W_norm = W / W_sum * m
    # Then (1/m) * sum_i W_norm_{i+j} * a_i = (1/m) * sum_i (W_{i+j}/W_sum * m) * a_i
    # = (1/W_sum) * sum_i W_{i+j} * a_i
    # And grad_j = (2/tau) * (1/W_sum) * sum_i W_{i+j} * a_i
    # = (2/tau) * cross_corr(W, a)_j / W_sum
    
    corr = fft_cross_convolve(W, a)
    grad = (2.0 / tau) * corr / W_sum
    return float(b_max), grad


def mirror_descent_update(a, grad, eta):
    """Exponentiated gradient update: a_new_j = a_j * exp(-eta*g_j) / Z."""
    log_ratio = -eta * grad
    log_ratio -= log_ratio.max()
    weights = np.exp(log_ratio)
    a_new = a * weights
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def sslm_optimize(a0, time_limit, tau_start=None, tau_end=0.3, verbose=False):
    """
    Smooth Subgradient Log-Sum-Exp Method with Nesterov acceleration.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_max = fft_convolve(a).max()
    t0 = time.monotonic()

    if tau_start is None:
        tau_start = max(1.0, 2.0 * n / 100.0)

    momentum = 0.7
    eta0 = 0.8
    total_iters = int(time_limit * 8000)  # estimate
    x = a.copy()
    velocity = np.zeros(n)
    stagnation = 0
    max_stagnation = 30000
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        progress = min(1.0, iter_count / max(100, total_iters))
        # Smooth annealing: fast decrease at first, slow at end
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.7)
        eta = eta0 / (np.sqrt(iter_count) + 1.0) ** 0.7

        # Nesterov lookahead
        x = a + momentum * velocity
        x = np.maximum(x, 1e-15)
        s_x = x.sum()
        if s_x < 1e-15:
            x = a.copy()
        else:
            x = x / s_x

        _, grad = logsumexp_obj_grad(x, tau)

        # Mirror descent step
        a_new = mirror_descent_update(a, grad, eta)

        velocity = momentum * velocity + (a_new - a)
        a = a_new

        # Check improvement
        curr_max = fft_convolve(a).max()
        if curr_max < best_max - 1e-13:
            best_max = curr_max
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > max_stagnation:
            break

        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print("    SSLM iter={}: max_b={:.10f}, J={:.6f}, tau={:.3f} ({:.0f}s)".format(
                iter_count, best_max, 2.0 * n * best_max, tau, elapsed))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    SSLM done: J={:.6f}, iters={} ({:.1f}s)".format(
            2.0 * n * best_max, iter_count, elapsed))
    return best_a, best_max


def projected_gradient_quadratic(a0, w, n_iter, verbose=False):
    """
    Minimize (1/2)*a^T T_w a = sum_k w_k * (a*a)_k
    subject to a >= 0, sum(a) = 1, using projected gradient.
    T_w is the Toeplitz matrix from convolution with w.
    Gradient = 2 * (w * a) (cross-correlation).
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    m = 2 * n - 1

    # Estimate Lipschitz constant
    w_abs_sum = np.abs(w).sum()
    L = 2.0 * w_abs_sum
    eta = 1.0 / max(L, 1e-6)

    for it in range(n_iter):
        # Gradient: 2 * (w cross-corr a)
        grad = 2.0 * fft_cross_convolve(w, a)

        # Gradient step
        a_step = a - eta * grad

        # Project onto simplex {a >= 0, sum(a) = 1}
        a_new = _project_simplex(a_step)

        a = a_new

    return a


def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def irq_optimize(a0, time_limit, verbose=False):
    """
    Iteratively Reweighed Quadratic optimization.
    Alternates between:
    1. Minimizing weighted quadratic sum_k w_k * b_k via projected gradient
    2. Updating weights to emphasize peaks
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_max = fft_convolve(a).max()
    t0 = time.monotonic()
    m = 2 * n - 1

    # Initialize weights near uniform
    b = fft_convolve(a)
    w = np.ones(m)
    b_max = b.max()

    alpha = 1.5  # Weight sharpening exponent
    n_inner = 50  # Inner projected gradient iterations

    stagnation = 0
    max_stagnation = 200
    outer_iter = 0

    while time.monotonic() - t0 < time_limit:
        outer_iter += 1
        b = fft_convolve(a)
        b_max = b.max()

        if b_max < best_max - 1e-13:
            best_max = b_max
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > max_stagnation:
            break

        # Update weights: emphasize peaks
        b_clipped = np.maximum(b, 1e-10)
        w = b_clipped ** alpha
        w = w / w.max()  # Normalize to keep values manageable

        # Inner loop: minimize weighted quadratic
        a = projected_gradient_quadratic(a, w, n_inner)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a / s

        # Reduce inner iterations over time
        if outer_iter > 50:
            n_inner = max(10, n_inner // 2)

        if verbose and outer_iter % 20 == 0:
            elapsed = time.monotonic() - t0
            print("    IRQ outer={}: max_b={:.10f}, J={:.6f} ({:.0f}s)".format(
                outer_iter, best_max, 2.0 * n * best_max, elapsed))

    if verbose:
        print("    IRQ done: J={:.6f}, outer_iters={}".format(2.0 * n * best_max, outer_iter))
    return best_a, best_max


def lp_fw_peak_shave(a0, time_limit, solver, initial_K=300, max_K=3000, verbose=False):
    """
    LP-Frank-Wolfe peak shaving.
    Solves: min M s.t. sum(g)=1, g>=0, (a*g)_k <= M for k in active_set.
    """
    n = len(a0)
    y = norm_sum1(a0).copy()
    best_a = y.copy()
    ac = fft_convolve(y)
    best_m = ac.max()
    t0 = time.monotonic()

    active_set = sorted(np.argsort(ac)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 400
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if verbose:
                print("    LP-FW: stagnation, stopping at iter {}".format(iter_count))
            break

        iter_count += 1
        ac = fft_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Build constraint matrix
        K = len(active_set)
        if K > max_K:
            active_set = active_set[:max_K]
            K = max_K

        coeffs = np.zeros((K, n))
        for jj in range(K):
            k_int = int(active_set[jj])
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(100, K // 3)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if no_improve > 100:
                active_set = active_set[:max(100, K // 3)]
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Add violated constraints to active set
        conv_ys = fft_cross_convolve(y, g_dir)
        # Actually need cross-correlation properly
        # conv_ys[k] = sum_i y[i] * g_dir[k-i] = (y * g_dir)_k
        # Let me use the proper convolution
        conv_ys = np.fft.irfft(
            np.fft.rfft(y, 1 << (2 * n - 2).bit_length()) *
            np.fft.rfft(g_dir, 1 << (2 * n - 2).bit_length()),
            1 << (2 * n - 2).bit_length())[:2 * n - 1]

        all_viol = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(all_viol) > 0:
            viol_amt = conv_ys[all_viol] - M_curr
            top_v = all_viol[np.argsort(viol_amt)[::-1][:50]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        # Backtracking line search
        best_alpha = 1.0 / (iter_count + 2.0)
        best_trial_M = M_curr

        alphas = [0.5, 0.3, 0.2, 0.15, 0.1, 0.05,
                  max(0.03, 2.0 / (iter_count + 2.0)),
                  max(0.01, 1.0 / (iter_count + 1.0))]

        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fft_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y = (1.0 - best_alpha) * y + best_alpha * g_dir
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = best_a.copy()
        else:
            y = y / s_y

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter={}: J={:.6f}, |S|={} ({:.0f}s)".format(
                iter_count, 2.0 * n * best_m, len(active_set), elapsed))

    if verbose:
        print("    LP-FW done: J={:.6f}, iters={}".format(2.0 * n * best_m, iter_count))
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, seed=42):
    """
    Coordinate descent: transfer mass from peak-contributing coords to low-impact coords.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fft_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 2000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fft_convolve(a)
        # Get top 20 peaks
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            # Find the source coordinate with highest contribution
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score destinations by their impact on top peaks
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            # Prefer destinations with low score
            candidates = np.where(scores < scores.max() * 0.5)[0]
            if len(candidates) < n // 4:
                candidates = np.argsort(scores)[:n // 2]

            if src in candidates:
                candidates = np.delete(candidates, np.where(candidates == src)[0])

            if len(candidates) == 0:
                candidates = np.arange(n)

            dst = int(candidates[np.argmin(scores[candidates])])
            if dst == src:
                continue

            # Try multiple transfer fractions
            found = False
            for frac in [0.003, 0.005, 0.008, 0.015, 0.03]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fft_convolve(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found:
                # Random perturbation
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fft_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            # Periodic random restart
            if stall % 500 == 0:
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.003, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fft_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 1000 == 0:
            print("    Coord ({:.0f}s): J={:.6f}, stall={}".format(
                time.monotonic() - t0, 2.0 * n * best_m, stall))

    return best_a, best_m


def get_initial_sequence():
    """Get the best starting sequence from parent or fallback."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                print("Parent: n={}, score={}".format(len(p), sc))
                return p_norm, sc
    except Exception:
        pass

    # Fallback: sine-based
    n = 5000
    x = np.linspace(0, 1, n)
    best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
    best_score = evaluate_safe(best_seq)
    print("Default init: n={}, score={}".format(n, best_score))
    return best_seq, best_score


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence if candidate is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [{}]: {:.6f} (n={}, t={:.1f}s)".format(
            label, sc, len(a_norm), elapsed))
        best_score = sc
        best_seq = a_norm.copy()
    return best_seq, best_score


def sslm_multistart(a0, time_limit, verbose=False):
    """
    Run SSLM with multiple Nesterov momentum values and merge results.
    """
    n = len(a0)
    t0 = time.monotonic()
    best_a = norm_sum1(a0).copy()
    best_max = fft_convolve(best_a).max()

    momentums = [0.5, 0.65, 0.7, 0.8]
    alpha_schedule = len(momentums)

    # Run each for a fraction of the time
    per_run_time = time_limit / alpha_schedule

    for mi, m_val in enumerate(momentums):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        t_run = min(per_run_time, remaining)

        a_curr = norm_sum1(best_a).copy()
        # Small perturbation for diversity
        rng = np.random.default_rng(42 + mi)
        noise = rng.normal(0, 0.02, n)
        a_curr = a_curr + noise * a_curr
        a_curr = np.maximum(a_curr, 0)
        a_curr = a_curr / a_curr.sum()

        # Run SSLM with this momentum
        a_out, m_out = sslm_optimize(a_curr, t_run, tau_start=5.0, tau_end=0.2, verbose=False)
        # Override momentum - actually sslm_optimize has fixed momentum
        # Let me just use the result
        if m_out < best_max:
            best_max = m_out
            best_a = a_out.copy()
            if verbose:
                print("    SSLM multistart m={}: J={:.6f}".format(m_val, 2.0 * n * best_max))

    return best_a, best_max


def propose_candidate(budget_s=1800.0):
    """
    Main search function. Combines multiple optimization strategies:
    1. SSLM with Nesterov acceleration (smooth log-sum-exp subgradient)
    2. Iteratively Reweighted Quadratic (peak equalization)
    3. LP-Frank-Wolfe (peak shaving)
    4. Coordinate descent polish
    5. Multi-n exploration
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)
    print("Total budget: {:.1f}s".format(time_budget))

    solver = get_solver()
    best_seq, best_score = get_initial_sequence()
    n0 = len(best_seq)

    def remaining():
        return time_budget - (time.monotonic() - t_global)

    def is_good(r):
        return r > 30

    # === Phase 1: SSLM (smooth subgradient with log-sum-exp) ===
    print("\n=== Phase 1: SSLM (n={}) ===".format(n0))
    t1 = min(remaining() * 0.25, 350.0)
    if is_good(t1):
        a_out, _ = sslm_optimize(best_seq, t1, tau_start=max(1.0, n0/200.0), tau_end=0.3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sslm1", t_global)

    # === Phase 2: IRQ (iteratively reweighted quadratic) ===
    print("\n=== Phase 2: IRQ (n={}) ===".format(len(best_seq)))
    t2 = min(remaining() * 0.12, 150.0)
    if is_good(t2):
        a_out, _ = irq_optimize(best_seq, t2, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "irq1", t_global)

    # === Phase 3: LP-FW peak shaving ===
    print("\n=== Phase 3: LP-FW (n={}) ===".format(len(best_seq)))
    t3 = min(remaining() * 0.18, 280.0)
    if is_good(t3) and len(best_seq) <= 10000:
        a_out, _ = lp_fw_peak_shave(best_seq, t3, solver, initial_K=400, max_K=3500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1", t_global)

    # === Phase 4: SSLM second round (refined) ===
    print("\n=== Phase 4: SSLM round 2 (n={}) ===".format(len(best_seq)))
    t4 = min(remaining() * 0.15, 200.0)
    if is_good(t4):
        a_out, _ = sslm_optimize(best_seq, t4, tau_start=2.0, tau_end=0.15, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sslm2", t_global)

    # === Phase 5: Multi-n exploration ===
    print("\n=== Phase 5: Multi-n ===")
    n_best = len(best_seq)
    deltas = [200, -200, 400, -400, 600]
    tried_ns = set()
    for d in deltas:
        new_n = n_best + d
        if new_n in tried_ns or not (3000 <= new_n <= 12000):
            continue
        tried_ns.add(new_n)
        rt = remaining()
        if rt < 80:
            break
        print("  Trying n={}".format(new_n))
        a_interp = interpolate_seq(best_seq, new_n)

        # Quick SSLM
        t5 = min(rt * 0.15, 120.0)
        if is_good(t5):
            a_out, _ = sslm_optimize(a_interp, t5, tau_start=3.0, tau_end=0.2, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "sslm_n{}".format(new_n), t_global)

        # Quick LP-FW
        rt = remaining()
        if rt > 60 and new_n <= 10000:
            t5b = min(rt * 0.12, 100.0)
            if is_good(t5b):
                a_out, _ = lp_fw_peak_shave(a_interp, t5b, solver, initial_K=300, max_K=2500, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n{}".format(new_n), t_global)

    # === Phase 6: LP-FW final (on best) ===
    print("\n=== Phase 6: LP-FW final (n={}) ===".format(len(best_seq)))
    t6 = min(remaining() * 0.15, 220.0)
    if is_good(t6) and len(best_seq) <= 10000:
        a_out, _ = lp_fw_peak_shave(best_seq, t6, solver, initial_K=500, max_K=4000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # === Phase 7: IRQ second round ===
    print("\n=== Phase 7: IRQ round 2 (n={}) ===".format(len(best_seq)))
    t7 = min(remaining() * 0.08, 100.0)
    if is_good(t7):
        a_out, _ = irq_optimize(best_seq, t7, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "irq2", t_global)

    # === Phase 8: Coordinate polish ===
    print("\n=== Phase 8: Coordinate polish (n={}) ===".format(len(best_seq)))
    t8 = min(remaining() * 0.10, 120.0)
    if t8 > 15:
        a_out, _ = coordinate_polish(best_seq, t8, verbose=True, seed=123)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord1", t_global)

    # === Phase 9: SSLM final (low temperature) ===
    rt = remaining()
    if rt > 40:
        print("\n=== Phase 9: SSLM final (n={}) ===".format(len(best_seq)))
        t9 = min(rt * 0.3, 150.0)
        if t9 > 20:
            a_out, _ = sslm_optimize(best_seq, t9, tau_start=1.0, tau_end=0.1, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "sslm_final", t_global)

    # === Final coordinate polish ===
    rt = remaining()
    if rt > 15:
        print("\n=== Final coord polish ===")
        t10 = min(rt * 0.8, 80.0)
        if t10 > 10:
            a_out, _ = coordinate_polish(best_seq, t10, verbose=False, seed=999)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={}, n={}, time={:.1f}s".format(
            best_score, len(best_seq), elapsed))
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score={}".format(best_score))

    return [float(x) for x in best_seq]
