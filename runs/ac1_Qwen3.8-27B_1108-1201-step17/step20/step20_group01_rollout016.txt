    n = len(a)
    w_padded = np.zeros(2 * n - 1)
    for i in range(len(top_lags)):
        ki = int(top_lags[i])
        if 0 <= ki < 2 * n - 1:
            w_padded[ki] = top_weights[i]
    flipped_w = w_padded[::-1]

def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Project to simplex: non-negative, sum to 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def obj_val(a):
    """Compute 2*n*max(autocorr) for normalized a."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def eval_safe(a):

def fft_gradient_weighted(a, weights):
    """
    Compute sum_k weights[k] * d(b_k)/d(a) using FFT.
    b = conv(a,a), so d(sum_k w_k b_k)/d(a_j) can be computed via correlation.
    """
    n = len(a)
    size = 2 * n - 1
    # The gradient of sum_k w_k * b_k w.r.t. a is:
    # grad[j] = sum_k w_k * (a[k-j] + a[j+k - (2n-2)])  (with validity checks)
    # This equals: conv(w_flipped, a) appropriately sliced
    w_padded = np.zeros(size)
    w_padded[:] = weights
    flipped_w = w_padded[::-1]
    req_size = 3 * n - 2
    fft_size = 1 << (req_size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(flipped_w, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    # Slice to get the gradient of correct length n
    grad = conv_result[2 * n - 2 : n - 2 : -1]
    # Ensure correct length
    if len(grad) != n:
        if len(grad) < n:
            grad = np.concatenate([grad, np.zeros(n - len(grad))])
        else:
            grad = grad[:n]
    return grad


def softmax_weights(ac, T):
    """Compute softmax weights for the autocorrelation values."""
    ac_shifted = ac - ac.max()
    exponents = T * ac_shifted
    # Clip for numerical stability
    exponents = np.clip(exponents, -500, 500)
    w = np.exp(exponents)
    s = w.sum()
    if s < 1e-300:
        # Uniform fallback
        return np.ones_like(ac) / len(ac)
    return w / s


def mirror_descent_smooth(a0, time_limit, T_init=20.0, T_final=100.0,

    # Nesterov momentum in log-space
    v = np.zeros(n, dtype=np.float64)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)
    stall = 0
    total_iters = 0

    for t in range(1, 5000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        if stall > 5000:
            break

        # Adaptive temperature
        frac = min(1.0, t / 5000.0)
        T_cur = T_init + (T_final - T_init) * frac
        # Adaptive momentum (decay slowly)
        mu = momentum0 * (0.999 ** t)
        # Adaptive learning rate
        lr = lr0 / (1.0 + 0.002 * t)

        # Compute autocorrelation
        ac = fast_autocorr(x)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr

        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Compute softmax weights (top-K for efficiency)
        k_sel = min(n_top, len(ac))
        idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = ac[idx]
        # Softmax over top-K
        w = softmax_weights(top_vals, T_cur)
        # Full weight vector
        weights = np.zeros(len(ac))
        weights[idx] = w

        # Compute smooth gradient
        grad = fft_gradient_weighted(x, weights)

        # Nesterov update in log-space
        # log_x_new = log_x + mu * v - lr * grad
        v = mu * v - lr * grad
        log_x_new = log_x + v
        log_x_new -= log_x_new.max()
        x_new = np.exp(np.clip(log_x_new, -50, 50))
        x_new = np.maximum(x_new, 1e-30)

        # Line search: try a few step sizes
        best_alpha = 1.0
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2]:

            if Mt < M_curr:

                break

            x = (1 - best_alpha) * x + best_alpha * x_new
            x = np.maximum(x, 0)
            s = x.sum()
            if s > 1e-15:
                x = x / s
            # Update log-space
            log_x = np.log(x + 1e-30)
            log_x -= log_x.max()
            v = v * best_alpha
        else:
            x = x_new
            log_x = np.log(x + 1e-30)
            log_x -= log_x.max()

                    v = np.zeros(n, dtype=np.float64)

                else:

                    x = x_pert
                    v = np.zeros(n, dtype=np.float64)

def lp_fw_refined(a0, time_limit, solver, init_K=100, max_K=2000,
                  add_per=20, verbose=True):
    """
    Refined LP-based Frank-Wolfe with:
    - Smaller initial active set (faster LPs)
    - Gradual growth
    - Exact line search via ternary search
    - Check for violations more aggressively
    """
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:init_K])
    no_improve = 0
    stall_limit = 1500
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > stall_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update active set
        iter_count += 1
        if iter_count % 20 == 1 or len(active_set) < init_K:
            top_lags = np.argsort(ac)[::-1]
            target_size = min(max_K, init_K + iter_count * 3)
            for k in top_lags[:target_size]:
                active_set.add(int(k))
            if len(active_set) > max_K:
                clist = sorted(active_set)
                margins = np.array([M_curr - ac[k] for k in clist])
                keep = np.argsort(margins)[:max_K]
                active_set = set(int(clist[i]) for i in keep)
        active_set.add(int(np.argmax(ac)))

        clist = sorted(active_set)
        K = len(clist)
        if K == 0:
            break

        # Build LP
        try:
            coeffs = np.zeros((K, n), dtype=np.float64)
            for jj in range(K):
                ki = clist[jj]
                i_start = max(0, ki - n + 1)
                i_end = min(n, ki + 1)
                if i_start < i_end:
                    idx = np.arange(i_start, i_end)
                    coeffs[jj, idx] = y[ki - idx]

            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            prob.solve(solver=solver, verbose=False)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                # Shrink active set on failure
                if len(active_set) > init_K:
                    clist2 = sorted(active_set)
                    active_set = set(int(clist2[i]) for i in
                                     np.argsort([M_curr - ac[k] for k in clist2])[:init_K])
                continue
        except Exception:
            if len(active_set) > init_K:
                clist2 = sorted(active_set)
                active_set = set(int(clist2[i]) for i in
                                 np.argsort([M_curr - ac[k] for k in clist2])[:init_K])
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations to grow active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]
            for k in top_v:
                active_set.add(int(k))

        # Exact line search via ternary search (f is convex in alpha)
        lo, hi = 0.0, 1.0
        for _ in range(20):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            t1 = (1 - m1) * y + m1 * g_dir
            t2 = (1 - m2) * y + m2 * g_dir
            t1 = np.maximum(t1, 0)
            t2 = np.maximum(t2, 0)
            s1, s2 = t1.sum(), t2.sum()
            if s1 < 1e-15 or s2 < 1e-15:
                break
            m1_val = fast_autocorr(t1 / s1).max()
            m2_val = fast_autocorr(t2 / s2).max()
            if m1_val < m2_val:
                hi = m2
            else:
                lo = m1
        best_alpha = (lo + hi) / 2.0

        # Update
        if iter_count <= 2:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t={:5d} ({:.1f}s) J={:.7f} |S|={:5d}".format(

    elapsed = time.monotonic() - t0
    if verbose:
        print("    LP-FW done ({:.1f}s) J={:.7f} |S|={}".format(
            elapsed, 2 * n * best_m, len(active_set)))

def coord_polish(a0, time_limit, n_targets=20, seed=42, verbose=True):
    """
    Fast coordinate descent: for each active lag, find the best single-coordinate
    transfer that reduces the objective.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 3000

    # Precompute pressure map
    def compute_pressure(a_, ac_, lags_):
        pr = np.zeros(n, dtype=np.float64)
        for ki in lags_:
            ki = int(ki)
            ws = max(0, ki)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            pr[idx] += a_[ki - idx] * 2.0  # Two contributions
        return pr

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(n_targets, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        M_curr = ac.max()
        improved = False

        pr = compute_pressure(a, ac, top_k[:10])

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            ws = max(0, ki)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-25:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-15:
                continue

            # Find quiet destination
            pr_local = pr.copy()
            pr_local[src] = 1e10
            quiet = np.argsort(pr_local)[:50]

            for dst in quiet:
                dst = int(dst)
                if dst == src:
                    continue
                # Try several transfer fractions
                for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.2, 0.35]:
                    amt = src_val * frac
                    if amt < 1e-18:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-16:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        # Recompute pressure
                        pr = compute_pressure(a, fast_autocorr(a), top_k[:10])
                        break
                if improved:
                    break

            if not improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
                            improved = True
            if stall % 1000 == 0 and stall > 0 and verbose:
                print("    CoordPolish ({:.1f}s) J={:.7f} stall={}".format(
                    time.monotonic() - t0, 2 * n * best_m, stall))

    return best_a, 2.0 * n * best_m


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search function: combines multiple optimization strategies."""
    t_global = time.monotonic()
    # Leave 60s buffer for finalization
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent: n={} score={}".format(len(p), repr(best_score)))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_init = 12000
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(np.cos(np.pi * (x - 0.5)) ** 2)
        best_score = eval_safe(best_seq)
        print("Default init n={} score={}".format(n_init, repr(best_score)))

    n0 = len(best_seq)
    print("Starting: n={}, initial score={}".format(n0, repr(best_score)))

    # === Phase 1: Quick mirror descent (20% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Mirror Descent (n={}) ===".format(n0))
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        a_out, v_out = mirror_descent_smooth(
            normalize(best_seq), t1,
            T_init=15.0, T_final=60.0, n_top=150,
            lr0=0.08, momentum0=0.92, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P1 improved: {}".format(repr(best_score)))

    # === Phase 2: LP-FW (25% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P2: LP-FW (n={}) ===".format(len(best_seq)))
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 60:
        a_out, v_out = lp_fw_refined(
            normalize(best_seq), t2, solver,
            init_K=100, max_K=2500, add_per=30, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P2 improved: {}".format(repr(best_score)))

    # === Phase 3: Mirror descent again with higher T (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P3: MD refined (n={}) ===".format(len(best_seq)))
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_out, v_out = mirror_descent_smooth(
            normalize(best_seq), t3,
            T_init=50.0, T_final=200.0, n_top=100,
            lr0=0.04, momentum0=0.85, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P3 improved: {}".format(repr(best_score)))

    # === Phase 4: Coordinate polish (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P4: CoordPolish (n={}) ===".format(len(best_seq)))
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_out, v_out = coord_polish(
            normalize(best_seq), t4, n_targets=20, seed=42, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P4 improved: {}".format(repr(best_score)))

    # === Phase 5: Try longer length (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 2000
    if remaining > 100:
        print("\n=== P5: Long n={} ===".format(n_long))
        t5 = min(remaining * 0.15, 250.0)
        if t5 > 60:
            a_long = interpolate_seq(best_seq, n_long)
            # Quick MD on longer sequence
            a_out, v_out = mirror_descent_smooth(
                a_long, t5 * 0.5,
                T_init=20.0, T_final=80.0, n_top=200,
                lr0=0.05, momentum0=0.90, verbose=False)
            # Quick coord polish
            a_out2, v_out2 = coord_polish(
                a_out, t5 * 0.5, n_targets=15, seed=77, verbose=False)
            sc1 = eval_safe(a_out)
            sc2 = eval_safe(a_out2)
            best_sc_long = min(sc1, sc2)
            if np.isfinite(best_sc_long) and best_sc_long < best_score:
                best_score = best_sc_long
                best_seq = a_out if sc1 < sc2 else a_out2
                print("  P5 improved: {}".format(repr(best_score)))

    # === Phase 6: Final LP-FW polish (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P6: Final LP-FW (n={}) ===".format(n_cur))
    t6 = min(remaining * 0.15, 250.0)
    if t6 > 30:
        if n_cur <= 15000:
            a_out, v_out = lp_fw_refined(
                normalize(best_seq), t6, solver,
                init_K=200, max_K=3000, add_per=30, verbose=True)
        else:
            a_out, v_out = mirror_descent_smooth(
                normalize(best_seq), t6,
                T_init=80.0, T_final=300.0, n_top=100,
                lr0=0.02, momentum0=0.80, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P6 improved: {}".format(repr(best_score)))

    # === Phase 7: Final coordinate polish (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P7: Final Polish (n={}) ===".format(len(best_seq)))
    t7 = min(remaining * 0.95, 300.0)
    if t7 > 10:
        a_out, v_out = coord_polish(
            normalize(best_seq), t7, n_targets=25, seed=12345, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P7 improved: {}".format(repr(best_score)))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={} n={} t={:.1f}s".format(
            repr(best_score), len(best_seq), elapsed))
    else:
        print("\nERROR: falling back to default")
        n_fb = 12000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.cos(np.pi * (x - 0.5)) ** 2)
        best_score = eval_safe(best_seq)

    return [float(x) for x in best_seq]
