def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

    """Objective for polynomial parameterization with softmax."""
    b = basis @ c
    b = b - b.max()
    exp_b = np.exp(b)
    a = exp_b / exp_b.sum()
    return obj_val(a)

def poly_gradient(c, n, basis):
    """Gradient of poly_objective via finite differences (central)."""
    grad = np.zeros_like(c)
    eps = 1e-7
    f0 = poly_objective(c, n, basis)
    for i in range(len(c)):
        c_plus = c.copy()
        c_plus[i] += eps
        c_minus = c.copy()
        c_minus[i] -= eps
        grad[i] = (poly_objective(c_plus, n, basis) - poly_objective(c_minus, n, basis)) / (2 * eps)
    return grad

def poly_optimize(n, degree, time_limit, init_c=None, seed=42):
    """Optimize polynomial coefficients using L-BFGS-B."""
    basis = cosine_basis(n, degree)
    
    if init_c is None:
        rng = np.random.default_rng(seed)
        init_c = rng.standard_normal(degree + 1) * 0.1
        init_c[0] = 0.5  # bias towards uniform
    
    def f(c):
        return poly_objective(c, n, basis)
    
    def g(c):
        return poly_gradient(c, n, basis)
    
    t0 = time.monotonic()
    best_c = init_c.copy()
    best_f = f(init_c)
    
    # Try L-BFGS-B
    try:
        result = sp_min(
            fun=f,
            x0=init_c,
            jac=g,
            method='L-BFGS-B',
            options={'maxiter': 500, 'maxfun': 2000, 'gtol': 1e-8}
        )
        if result.fun < best_f:
            best_f = result.fun
            best_c = result.x.copy()
    except Exception:
        pass
    
    # Also try Nelder-Mead (doesn't need gradient)
    try:
        result2 = sp_min(
            fun=f,
            x0=init_c,
            method='Nelder-Mead',
            options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-10}
        )
        if result2.fun < best_f:
            best_f = result2.fun
            best_c = result2.x.copy()
    except Exception:
        pass
    
    elapsed = time.monotonic() - t0
    print("    Poly-opt degree=" + str(degree) + " (" + str(round(elapsed, 1)) + "s) obj=" + repr(best_f))
    
    # Convert to sequence
    b = basis @ best_c
    b = b - b.max()
    exp_b = np.exp(b)
    a = exp_b / exp_b.sum()
    
    return a, best_f

def build_lp_matrix(a, lags, n):
    """Build the constraint matrix for the LP."""
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs

def lp_fw_improved(a0, time_limit, solver, init_K=300, max_K=3000):
    """Improved LP-Frank-Wolfe with away step and adaptive active set."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    # Initial active set: top K lags
    ac = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac)[::-1][:init_K])
    
    no_improve = 0
    stall_limit = 2000
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    rng = np.random.default_rng(2024)
    
    # Store previous y for away step
    y_prev = y.copy()
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive active set management
        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)
        
        # Add newly violated constraints
        violations = np.where(ac > M_curr - 1e-10)[0]  # all peaks
        # Add top violators not already in set
        new_viol = np.setdiff1d(np.argsort(ac)[::-1][:init_K], np.array(list(active_set)))
        for v in new_viol[:10]:
            active_set.add(int(v))
        
        K = len(list(active_set))
        if K == 0:
            break
        
        clist = sorted(active_set)
        coeffs = build_lp_matrix(y, clist, n)
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        lp_ok = False
        g_val = None
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                lp_ok = True
                g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
                gs = g_val.sum()
                if gs < 1e-15:
                    g_val = None
                    lp_ok = False
                else:
                    g_val = g_val / gs
        except Exception:
            if len(active_set) > init_K:
                active_set = set(int(k) for k in clist[:init_K])
        
        if not lp_ok or g_val is None:
            if no_improve > stall_limit // 2:
                break
            if len(clist) > 200:
                active_set = set(int(k) for k in clist[:300])
                continue
            # Try a gradient-based step instead
            top_k = np.argsort(ac)[::-1][:5]
            eta = 0.1 / (1.0 + 0.01 * t)
            for k_idx in top_k:
                ki = int(k_idx)
                i_s = max(0, ki - n + 1)
                i_e = min(n, ki + 1)
                idx = np.arange(i_s, i_e)
                grad = 2.0 * y[ki - idx]
                y[idx] *= np.exp(-eta * grad)
            y = np.maximum(y, 0.0)
            ys = y.sum()
            if ys > 1e-15:
                y = y / ys
            continue
        
        # Check and add violations
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_val, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        viol = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(viol) > 0:
            viol_amt = conv_ys[viol] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:20]]
            for k in top_v:
                active_set.add(int(k))
        
        # Line search: try alpha = 1 first, then bisect
        best_alpha = 1.0
        trial = np.maximum((1.0 - 1.0) * y + 1.0 * g_val, 0.0)
        trial_sum = trial.sum()
        if trial_sum > 1e-15:
            trial = trial / trial_sum
            trial_m = fast_autocorr(trial).max()
            if trial_m >= M_curr:
                # Need to do line search
                lo, hi = 0.01, 1.0
                for _ in range(30):
                    mid = (lo + hi) / 2.0
                    trial = np.maximum((1.0 - mid) * y + mid * g_val, 0.0)
                    ts = trial.sum()
                    if ts < 1e-15:
                        hi = mid
                        continue
                    trial = trial / ts
                    if fast_autocorr(trial).max() < M_curr:
                        lo = mid
                    else:
                        hi = mid
                best_alpha = (lo + hi) / 2.0
        
        y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * g_val, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy
        else:
            y = g_val.copy()
        
        # Periodic restart with noise
        if no_improve > 500 and no_improve % 500 == 0:
            noise_scale = 0.005 + 0.01 * (no_improve / 1000.0)
            noise = noise_scale * rng.standard_normal(n) * best_a
            y_r = best_a + noise
            y_r = np.maximum(y_r, 1e-12)
            y_r = y_r / y_r.sum()
            y = y_r
            no_improve = 0
        
        if t % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW2 t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(2 * n * best_m) + " |S|=" + str(len(active_set)))
    
    print("    LP-FW2 done (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2 * n * best_m))
    return best_a, 2.0 * n * best_m

def grad_mult_v2(a0, time_limit, top_k=50, temp=30.0, c0=2.5, seed=42):
    """Multiplicative gradient descent v2 with better adaptive temperature."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    x_prev = x.copy()
    stall = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 15000:
            break
        
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]
        
        # Adaptive temperature: decrease over time
        cur_temp = temp * max(0.2, 1.0 - t / 30000.0)
        w = np.exp(cur_temp * (top_vals - top_vals[0]))
        w = w / w.sum()
        
        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            idx_arr = np.arange(i_s, i_e)
            grad[idx_arr] += 2.0 * wi * y[ki - idx_arr]
        
        eta_t = c0 / (np.sqrt(max(t, 1)) + 1.0)
        log_a = np.log(y) - eta_t * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        val = obj_val(x)
        if val < best_val - 1e-12:
            best_val = val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart with noise
        if stall > 5000 and stall % 2500 == 0:
            noise_scale = 0.01 + 0.02 * min(stall / 20000.0, 1.0)
            noise = noise_scale * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            stall = 0
        
        if t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print("    GradMult2 t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(best_val))
    
    return best_a, best_val

def freq_flatten(a0, time_limit, alpha_range=(1.05, 2.0), n_iter=200, seed=42):
    """Frequency-domain peak shaving via spectrum modification."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    for iteration in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute spectrum
        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
        power = np.abs(A) ** 2
        
        # Flatten: raise to 1/alpha power (reduces dynamic range)
        for alpha in alpha_range:
            alpha = float(alpha)
            power_flat = np.power(np.maximum(power, 1e-30), 1.0 / alpha)
            
            # Reconstruct: take sqrt of flattened power with original phase
            A_new = np.sqrt(power_flat) * (A / np.maximum(np.abs(A), 1e-30))
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            
            val = obj_val(a_new)
            if val < best_val - 1e-10:

                best_a = a_new.copy()

        A_pert = A * (1.0 + 0.01 * rng.standard_normal(len(A)))
        a_pert = np.fft.irfft(A_pert, fft_size)[:n]
        a_pert =
