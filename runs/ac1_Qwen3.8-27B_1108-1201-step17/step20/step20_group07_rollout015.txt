def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def peak_shaving(a0, time_limit, top_k=15, verbose=False):
    """Pressure-aware peak shaving: reduce mass at positions contributing to peaks,
    move to positions with lowest 'danger' score."""
    n = len(a0)

    best_m = fast_autocorr(a).max()

    while time.monotonic() - t0 < time_limit and stall < 300:
        b = fast_autocorr(a)
        n_peaks = min(top_k, len(b))
        peak_indices = np.argsort(b)[::-1][:n_peaks]

        # Compute danger score per position
        danger = np.zeros(n)
        for k_int in peak_indices[:8]:
            k_i = int(k_int)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                w = b[k_i] / max(b.max(), 1e-15)
                danger[idx] += a[k_i - idx] * w
                danger[k_i - idx] += a[idx] * w

        improved = False
        for k_int in peak_indices[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_i = int(k_int)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_i - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            partner = k_i - src
            if partner < 0 or partner >= n:
                continue

            # Choose which to reduce
            if a[src] >= a[partner]:
                reduce_idx = src
            else:
                reduce_idx = partner

            if a[reduce_idx] < 1e-14:
                continue

            # Find safest destination
            safe_score = danger.copy()
            safe_score[reduce_idx] = np.max(danger) * 2.0
            # Also avoid positions that are partners in other top peaks
            for k2_int in peak_indices[1:6]:
                k2_i = int(k2_int)
                p2 = k2_i - reduce_idx
                if 0 <= p2 < n:
                    safe_score[p2] += 0.5 * danger[p2]

            dst = int(np.argmin(safe_score))
            if dst == reduce_idx:
                continue

            found = False

                amt = a[reduce_idx] * frac

                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                b_trial = fast_autocorr(a_trial)
                m_trial = b_trial.max()
                if m_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

        if not improved:
            stall += 1
            if stall >= 30:
                break
            # Periodic random perturbation to escape
            if stall % 10 == 0:
                noise_scale = 0.01 + 0.005 * (stall // 10)
                pert = rng.normal(0, noise_scale, n) * best_a
                a_r = best_a + pert

def subgradient_smoothed(a0, time_limit, beta_start=30.0, beta_end=800.0,
                         momentum=0.9, verbose=False):

    x = norm_sum1(a0)
    best_a = x.copy()

    stall = 0

    while time.monotonic() - t0 < time_limit:

        b = fast_autocorr(x)

        b_max = b_top.max()

        shifted = np.clip(shifted, -200, 200)
        exp_vals = np.exp(shifted)
        p = exp_vals / exp_vals.sum()

        # Gradient computation via FFT
        # grad_i = (2/beta) * sum_k p_k * [a_{k-i} + a_i for valid k-i]
        # = (2/beta) * [conv_rev(p, a)_i + sum_k p_k * a_i]
        # where conv is the appropriate cross-correlation
        p_full = np.zeros(len(b))
        p_full[idx_top] = p
        # conv of rev(p_full) with x gives sum_k p_full[k] * x[k-i] for valid k
        fft_size = 1 << ((2 * n - 1) - 1).bit_length()
        p_rev = p_full[::-1]
        Pr = np.fft.rfft(p_rev, fft_size)
        xa = np.fft.rfft(x, fft_size)

        # conv1[j] = sum_k p_rev[2n-2-k] * x[j-k] ... let me just use direct correlation
        # Actually: sum_k p_k * x[k-i] for k where 0<=k-i<n and 0<=k<len(b)
        # This is the cross-correlation of p and x
        # Let's compute it as: for each i, sum_k p_k * x[k-i]
        # = flip correlation
        # Use FFT: if C = ifft(fft(p) * conj(fft(x))), then C[i] = sum_j p[j]*x[j+i]
        # We want sum_k p[k] * x[k-i] = sum_j p[j+i] * x[j] ... 
        # Let me just use the convolution approach
        # sum_k p[k] * x[k-i]: let j = k-i, so k = j+i
        # = sum_j p[j+i] * x[j] (for valid j)
        # This is (p reversed) convolved with x, evaluated at position ...
        # Actually the simplest: fft(p) * fft(flip(x)) gives cross-correlation
        p_f = np.fft.rfft(p_full, fft_size)
        x_f = np.fft.rfft(x, fft_size)
        # cross-corr: ifft(fft(p) * conj(fft(x)))
        cross = np.fft.irfft(p_f * np.conj(x_f), fft_size)
        # cross[i] = sum_j p[j] * conj(x[j-i]) = sum_j p[j] * x[j-i] (real case)
        # We want sum_k p[k] * x[k-i]. Let j=k, so sum_j p[j] * x[j-i] = cross[i]
        # But we need to handle the circular wrap-around. For our case with zero-padding,
        # cross[i] for i in [0, n-1] should be correct.
        term1 = cross[:n]
        term2 = p.sum() * x  # sum_k p_k * x_i (constant factor times x_i)
        grad = (2.0 / beta) * (term1 + term2)

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size with diminishing rate
        eta = 1.0 / (gnorm * (1.0 + 0.1 * beta / 100.0))
        eta = min(eta, 0.03)
        eta = max(eta, 1e-7)

        # Heavy-ball update
        v = momentum * v - eta * grad
        x_new = x + v
        x_new = simplex_project(x_new)

        # Evaluate
        b_new = fast_autocorr(x_new)
        m_new = b_new.max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0
            x = x_new
        else:
            stall += 1
            # Revert
            x = x
            if stall > 100:
                # Perturb and restart
                noise = 0.01 * best_a * rng.normal(0, 1, n)
                x_r = best_a + noise
                x_r = np.maximum(x_r, 1e-12)
                x_r = x_r / x_r.sum()
                x = x_r
                v = np.zeros(n)
                stall = 0
                beta = max(20.0, beta * 0.5)

        # Update beta
        if (it := it + 1) % 100 == 0:
            beta = min(beta * 1.008, beta_end)

        if verbose and it % 500 == 0:
            print(f"      SG it={it}: J={2.0*n*best_m:.6f}, beta={beta:.0f}")

    return best_a, best_m


def parametric_init(n, time_limit, best_seq, best_score):
    """Quick parametric sweep for better initialization."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    best_a = best_seq.copy()
    best_val = best_score
    rng = np.random.default_rng(42)

    # Beta-like distributions (flexible shape)
    for alpha in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0]:
        for beta_p in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0]:
            if time.monotonic() - t0 > time_limit:
                break
            a_raw = (x + 0.005) ** alpha * (1.0 - x + 0.005) ** beta_p
            a = norm_sum1(a_raw)
            m = fast_autocorr(a).max()
            if m < best_val:
                best_val = m
                best_a = a.copy()

    # U-shaped (edge-concentrated)
    for c in [2, 3, 5, 8, 12]:
        a_raw = 1.0 + c * (x ** 2 + (1 - x) ** 2)
        a = norm_sum1(a_raw)
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    # Gaussian mixtures
    for c1 in [0.15, 0.25, 0.35, 0.5]:
        for c2 in [0.5, 0.65, 0.75, 0.85]:
            for s1 in [0.05, 0.08, 0.12]:
                for s2 in [0.05, 0.08, 0.12]:
                    if time.monotonic() - t0 > time_limit * 0.6:
                        break
                    a_raw = np.exp(-((x - c1) ** 2) / (2 * s1 ** 2)) + np.exp(-((x - c2) ** 2) / (2 * s2 ** 2))
                    a = norm_sum1(a_raw)
                    m = fast_autocorr(a).max()
                    if m < best_val:
                        best_val = m
                        best_a = a.copy()

    # Random with structure
    for _ in range(20):
        if time.monotonic() - t0 > time_limit:
            break
        a = norm_sum1(rng.random(n) ** rng.uniform(0.5, 4.0))
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    return best_a, best_val


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score

    """Main search: LP-based optimization interleaved with peak shaving and subgradient."""
    t_global = time.monotonic()

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 7000
        x = np.linspace(0, 1, n_base)
        a_raw = (x + 0.01) ** 1.0 * (1.0 - x + 0.01) ** 1.0
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.02, 35.0)
