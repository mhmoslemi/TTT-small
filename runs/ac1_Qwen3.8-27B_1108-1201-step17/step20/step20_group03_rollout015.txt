def simplex_proj(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]

    return np.maximum(z - theta, 0.0)

            if pt < 1e-300:
                p = np.zeros_like(b)
                mask = b == b_max
                p[mask] = 1.0 / max(1, mask.sum())
            else:
                p = exp_v / pt
            a_rev = y[::-1]
            conv_len = n + len(p) - 1
            fft_s = 1 << (conv_len - 1).bit_length()
            Ar = np.fft.rfft(a_rev, fft_s)
            Pr = np.fft.rfft(p, fft_s)
            conv = np.fft.irfft(Ar * Pr, fft_s)
            g_piece = conv[n - 1:2 * n - 1]
            grad += wi * g_piece

        # BB step size
        if s_prev is not None and y_prev is not None:
            sy = y - y_prev
            gy = grad - s_prev
            sy_dot_gy = sy.dot(gy)
            gy_dot_gy = gy.dot(gy)
            if gy_dot_gy > 1e-20:
                bb_cand = sy_dot_gy / gy_dot_gy
                if bb_cand > 1e-8:
                    bb_step = max(1e-6, min(bb_cand, 3.0 / beta))
                else:
                    bb_step = max(bb_step * 0.9, 1e-6)
            s_prev = grad.copy()
            y_prev = y.copy()

        eta = min(bb_step, 0.15)
        z = y - eta * grad
        x_new = simplex_proj(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        avg_count += 1.0
        avg_x = avg_x + (x_new - avg_x) / avg_count

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 80 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.02 / (1.0 + 0.03 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 5.0)
            s_prev = None
            y_prev = None
            bb_step = 1.0 / (n ** 0.5)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    FISTA-MS restart t=%d (%.1fs): J=%.6f" % (t, elapsed, J))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    FISTA-MS t=%d (%.1fs): J=%.6f beta=%.1f" % (t, elapsed, J, beta))

    J_avg = 2.0 * n * fast_autocorr(avg_x).max()
    J_best = 2.0 * n * best_max_b
    if J_avg < J_best:
        return avg_x, fast_autocorr(avg_x).max()
    return best_a, best_max_b


def lp_fw_v2(a0, time_limit, solver, initial_K=500, max_K=4000,
             add_per_iter=30, verbose=True):
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    # Include lag 0 always, plus top peaks
    always_include = {0}
    top_idx = np.argsort(ac_y)[::-1][:initial_K]
    active_set = set(int(k) for k in top_idx)
    active_set |= always_include
    no_improve = 0
    max_no_improve = 600
    restart_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            # Try a restart
            restart_count += 1
            if restart_count > 10:
                break
            rng_r = np.random.default_rng(1000 + restart_count)
            noise = 0.01 * rng_r.normal(0, 1, n)
            y_r = best_a + noise
            y_r = np.maximum(y_r, 1e-10)
            y_r = y_r / y_r.sum()
            y = y_r
            ac_y = fast_autocorr(y)
            top_idx = np.argsort(ac_y)[::-1][:initial_K]
            active_set = set(int(k) for k in top_idx)
            active_set |= always_include
            no_improve = 0
            continue

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            active_set |= always_include
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                top_idx = np.argsort(fast_autocorr(y))[::-1][:initial_K // 2]
                active_set = set(int(k) for k in top_idx)
                active_set |= always_include
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations for new constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            # Prioritize adding violations with large amounts
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Adaptive alpha via best of several candidates
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.01, default_alpha * 0.5),
                  max(0.01, default_alpha * 2.0), 1.0 / (t + 1),
                  min(0.1, 0.2 / (1.0 + t * 0.001))]
        alphas = [max(0.005, min(1.0, a)) for a in alphas]
        alphas = list(set(alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW2 iter=%d (%.1fs): J=%.6f |S|=%d" % (t, elapsed, J, len(active_set)))

    return best_a, best_m


def freq_shape(a0, time_limit, n_iters=500, verbose=True):
    """Frequency-domain shaping: make |FFT(a)|^2 as flat as possible."""
    n = len(a0)
    a = norm1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        power_mean = power.mean()

        # Reduce peaks in power spectrum
        alpha_peak = 0.15
        power_new = power.copy()
        above = power > power_mean
        power_new[above] = power_mean + (power[above] - power_mean) * (1.0 - alpha_peak)
        # Slightly boost below-mean to maintain total power
        below = ~above
        if below.sum() > 0:
            boost = (power[above].sum() - power_new[above].sum()) / below.sum()
            power_new[below] += boost
        power_new = np.maximum(power_new, 0.0)

        # New magnitude (keep phase)
        mag_new = np.sqrt(power_new)
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        a = a_new

        if verbose and (it + 1) % 100 == 0:
            J = 2.0 * n * best_m
            print("    FreqShape it=%d: J=%.6f" % (it + 1, J))

    return best_a, best_m


def local_2ex(a0, time_limit, verbose=True):
    """2-exchange local search: move mass from peak contributors to low areas."""
    n = len(a0)
    a = norm1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:40]
        improved = False

        for _ in range(80):
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(top_peaks[rng.integers(len(top_peaks))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])

            # Find low-impact target
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:20]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src or a[src] < 1e-12:
                continue

            fracs = [0.003, 0.008, 0.015, 0.03, 0.05, 0.1]
            frac = fracs[int(rng.integers(len(fracs)))]
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-14:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                ac = fast_autocorr(a)
                top_peaks = np.argsort(ac)[::-1][:40]

        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * float(rng.uniform(0.005, 0.05))
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Local2ex (%.1fs): J=%.6f stall=%d" % (elapsed, J, stall))

    return best_a, best_m


def best_cyclic(a0):
    """Find best cyclic shift."""
    n = len(a0)
    a = norm1(a0)
    best_m = fast_autocorr(a).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    for s in range(1, n):
        a_shift = np.roll(a, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a, -best_shift), best_m
    return a.copy(), best_m


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = eval_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [%s]: %.6f (n=%d, t=%.1fs)" % (label, sc, len(a), elapsed))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Time budget: %.0fs, solver: %s" % (time_budget, solver))

    best_seq = None
    best_score = float('inf')

    # Load parent if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = eval_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm1(parent)
                best_score = p_score
                print("Parent: n=%d, score=%.6f" % (len(parent), p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 7000
        x = np.linspace(0, 1, n_base)
        best_seq = norm1(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)
        print("Default init: n=%d, score=%.6f" % (len(best_seq), best_score))

    n0 = len(best_seq)

    # Phase 1: Multi-scale FISTA on parent
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: FISTA-MultiScale (n=%d) ===" % n0)
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 60:
        a_out, _ = fista_multiscale(norm1(best_seq), t1,
                                    beta_start=10.0, beta_end=1000.0,
                                    beta_growth=1.004, restart_every=4000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista1", t_global)

    # Phase 2: Frequency shaping
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: Frequency Shaping (n=%d) ===" % len(best_seq))
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 30:
        a_out, _ = freq_shape(norm1(best_seq), t2, n_iters=300, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "freq1", t_global)

    # Phase 3: Cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: Cyclic Shift (n=%d) ===" % len(best_seq))
    if remaining > 60:
        a_shifted, m_shift = best_cyclic(norm1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyc1", t_global)

    # Phase 4: LP-Frank-Wolfe (main powerful step)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW v2 (n=%d) ===" % n_best)
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 60 and n_best <= 12000:
        a_out, _ = lp_fw_v2(norm1(best_seq), t4, solver,
                            initial_K=600, max_K=4000, add_per_iter=30, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 5: Local search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Local 2-exchange (n=%d) ===" % n_best)
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 30:
        a_out, _ = local_2ex(norm1(best_seq), t5, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "loc1", t_global)

    # Phase 6: Second FISTA pass (different params)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: FISTA pass 2 (n=%d) ===" % n_best)
    t6 = min(remaining * 0.20, 250.0)
    if t6 > 40:
        a_out, _ = fista_multiscale(norm1(best_seq), t6,
                                    beta_start=50.0, beta_end=2000.0,
                                    beta_growth=1.006, restart_every=3000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista2", t_global)

    # Phase 7: Try longer sequences
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Multi-length exploration ===")
    longer_lengths = [n_best + 500, n_best + 1000, n_best + 2000]
    longer_lengths = [l for l in longer_lengths if l <= 15000 and l > n_best]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("  Trying n=%d" % new_n)
        a_long = interpolate_seq(best_seq, new_n)

        t7a = min(remaining * 0.25, 200.0)
        if t7a > 40:
            a_out, _ = fista_multiscale(a_long, t7a,
                                        beta_start=15.0, beta_end=800.0,
                                        beta_growth=1.004, restart_every=3500, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_n%d" % new_n, t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80:
            a_shifted, m_shift = best_cyclic(norm1(best_seq))
            best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyc_n%d" % new_n, t_global)

    # Phase 8: Final LP burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final LP burst (n=%d) ===" % n_best)
    if remaining > 60 and n_best <= 12000:
        t8 = min(remaining * 0.25, 200.0)

            a_out, _ = lp_fw_v2(norm1(best_seq), t8, solver,
                                initial_K=800, max_K=5000, add_per_iter=40, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp2", t_global)

    # Phase 9: Final local search + shift
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 9: Final polish ===")
    if remaining > 40:
        t9 = min(remaining * 0.30, 100.0)
        if t9 > 20:
            a_out, _ = local_2ex(norm1(best_seq), t9, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "loc_final", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        a_shifted, m_shift = best_cyclic(norm1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyc_final", t_global)

    # Phase 10: Final FISTA micro-polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t10 = min(remaining * 0.50, 100.0)
        if t10 > 20:
            a_out, _ = fista_multiscale(norm1(best_seq), t10,
                                        beta_start=200.0, beta_end=3000.0,
                                        beta_growth=1.01, restart_every=2000, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=%.6f, n=%d, time=%.1fs" % (best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 7000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm1(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
