        if beta >= 200.0:
            top_count = max(20, size // 20)
        msk = _make_top_mask(a, top_count, None) if top_count else None
        a_new, _, _ = _run_lbfegs(a, beta, msk, maxfun=per_run)
        M_new = _proxy_max(a_new, None)
        if math.isfinite(M_new) and M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        if math.isfinite(M_new) and M_new <= best_M * 1.001 + 1e-15:
            a = a_new
        else:
            a = best_a
        if bi % 2 == 1:
            print("    L-BFGS beta=" + repr(beta) + " M=" + repr(best_M) +
                  " t=" + repr(time.monotonic() - t0))
    return best_a


def _spectral_equalize(a, n_iter, alpha_init=0.1):
    a = _norm1(a)
    n = a.size
    size = 2 * n - 1
    N = _next_pow2(size)
    best_a = a.copy()
    best_M = _proxy_max(a, None)
    c = _fft_conv(a)
    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        P = A.real ** 2 + A.imag ** 2
        P[P < 0] = 0
        pmax = float(np.max(P))
        if pmax <= 1e-30:
            break
        Pn = P / pmax
        target = np.ones_like(Pn) * 0.5
        Pn = Pn + 0.3 * (target - Pn)
        Pn = np.maximum(Pn, 0.01)
        phase = np.angle(A)
        new_P = np.maximum(Pn * pmax, 0.0)
        new_A = np.sqrt(new_P) * np.exp(1j * phase)
        new_a = np.fft.irfft(new_A, N)[:n]
        new_a = np.where(new_a > 0, new_a, 0.0)
        new_a = np.where(np.isfinite(new_a), new_a, 0.0)
        s = float(new_a.sum())
        if s < 1e-15:
            break
        new_a = new_a / s
        M_new = _proxy_max(new_a, None)
        if math.isfinite(M_new) and M_new < best_M:
            best_M = M_new
            best_a = new_a.copy()
            a = new_a
        else:
            a = best_a
            break
    return best_a


def _coord_polish(a0, budget, verbose=True):
    a = _norm1(a0)
    n = a.size
    c = _fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    rnd = 0
    while time.monotonic() - t0 < budget - 3 and stall < 200:
        rnd += 1
        if n <= 1:
            break
        top_k = min(30, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        p_top = np.zeros(2 * n - 1, dtype=np.float64)
        p_top[top_idx] = 1.0 / float(top_idx.size)
        N = _next_pow2(3 * n - 2)
        A = np.fft.rfft(a, N)
        Pr = np.fft.rfft(p_top[::-1], N)
        conv = np.fft.irfft(A * Pr, N)[:3 * n - 2]
        pressure = 2.0 * conv[n - 1:2 * n - 1][::-1]
        pressure = np.where(np.isfinite(pressure) & (pressure > 0), pressure, 0.0)
        src_count = min(10, n)
        dst_count = min(15, n)
        src_score = pressure
        src_cand = np.argpartition(src_score, min(src_count - 1, n - 1))[:src_count]
        src_cand = src_cand[np.argsort(src_score[src_cand])[::-1]]
        dst_score = -pressure
        dst_cand = np.argpartition(dst_score, min(dst_count - 1, n - 1))[:dst_count]
        dst_cand = dst_cand[np.argsort(dst_score[dst_cand])]
        best_cand = None
        for si in src_cand:
            s = int(si)
            if a[s] < 1e-12:
                continue
            for di in dst_cand:
                d = int(di)
                if d == s:
                    continue
                for frac in (0.003, 0.008, 0.015, 0.03, 0.06):
                    tm = a[s] * frac
                    tm = min(tm, a[s] * 0.99)
                    c_new = c.copy()
                    for off in range(n):
                        idx1 = s + off
                        idx2 = d + off
                        if idx1 < 2 * n - 1:
                            c_new[idx1] += 2.0 * tm * a[off]
                        if idx2 < 2 * n - 1:
                            c_new[idx2] -= 2.0 * tm * a[off]
                    if 2 * d < 2 * n - 1:
                        c_new[2 * d] += tm * tm
                    if 2 * s < 2 * n - 1:
                        c_new[2 * s] += tm * tm
                    if s + d < 2 * n - 1:
                        c_new[s + d] -= 2.0 * tm * tm
                    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
                    m_new = float(np.max(c_new))
                    if m_new < M - 1e-18:
                        if best_cand is None or m_new < best_cand[0]:
                            best_cand = (m_new, s, d, tm)
        if best_cand is not None:
            m_new, s, d, tm = best_cand
            a[s] -= tm
            a[d] += tm
            np.maximum(a, 0.0, out=a)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = _fft_conv(a)
            M = float(np.max(c))
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and (rnd <= 3 or rnd % 30 == 0):
                print("    Coord rnd=" + str(rnd) + " M=" + repr(best_M) +
                      " t=" + repr(time.monotonic() - t0))
        else:
            stall += 1
            if stall % 10 == 0:
                for _ in range(50):
                    s = int(rng.integers(n))
                    d = int(rng.integers(n))
                    if s == d or a[s] < 1e-12:
                        continue
                    tm = a[s] * float(rng.uniform(0.001, 0.02))
                    tm = min(tm, a[s] * 0.99)
                    if tm <= 1e-16:
                        continue
                    c_new = c.copy()
                    for off in range(n):
                        idx1 = s + off
                        idx2 = d + off
                        if idx1 < 2 * n - 1:
                            c_new[idx1] += 2.0 * tm * a[off]
                        if idx2 < 2 * n - 1:
                            c_new[idx2] -= 2.0 * tm * a[off]
                    if 2 * d < 2 * n - 1:
                        c_new[2 * d] += tm * tm
                    if 2 * s < 2 * n - 1:
                        c_new[2 * s] += tm * tm
                    if s + d < 2 * n - 1:
                        c_new[s + d] -= 2.0 * tm * tm
                    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
                    m_new = float(np.max(c_new))
                    if m_new < M - 1e-18:
                        a[s] -= tm
                        a[d] += tm
                        np.maximum(a, 0.0, out=a)
                        ss = float(a.sum())
                        if ss < 1e-15:
                            break
                        a = a / ss
                        c = _fft_conv(a)
                        M = float(np.max(c))
                        if M < best_M:
                            best_M = M
                            best_a = a.copy()
                        stall = 0
                        break
                else:
                    stall += 5
    return best_a


def _kkt_polish(a0, budget):
    a = _norm1(a0)
    n = a.size
    c = _fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()
    for it in range(200):
        dt = time.monotonic() - t0
        if dt > budget - 2:
            break
        top_k = min(15, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        c_top = c[top_idx]
        c_max = float(np.max(c_top))
        active = top_idx[c_top >= c_max * 0.999]
        K = active.size
        if K == 0:
            break
        S = np.where(a > 1e-10)[0]
        if S.size == 0:
            break
        JS = np.zeros((K, S.size), dtype=np.float64)
        for ki, k in enumerate(active):
            for si, j in enumerate(S):
                idx = int(k) - int(j)
                if 0 <= idx < n:
                    JS[ki, si] = 2.0 * a[int(idx)]
        rhs = np.zeros((K, 1), dtype=np.float64)
        for ki, k in enumerate(active):
            rhs[ki] = c_max - c[int(k)]
        ones_S = np.ones((S.size, 1), dtype=np.float64)
        A_mat = np.hstack([JS, ones_S])
        b_vec = rhs.flatten()
        try:
            sol, res_info = np.linalg.lstsq(A_mat, b_vec, rcond=None)
        except Exception:
            break
        delta_S = sol[:S.size]
        delta = np.zeros(n, dtype=np.float64)
        delta[S] = delta_S
        delta[S] -= float(np.mean(delta_S))
        step_fracs = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]
        improved = False
        for sf in step_fracs:
            new_a = a + sf * delta
            new_a = np.maximum(new_a, 0.0)
            ss = float(new_a.sum())
            if ss < 1e-15:
                continue
            new_a = new_a / ss
            new_c = _fft_conv(new_a)
            new_M = float(np.max(new_c))
            if math.isfinite(new_M) and new_M < M - 1e-15:
                a = new_a
                c = new_c
                M = new_M
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
                improved = True
                break
        if not improved:
            if it % 20 == 0 and it > 0:
                rng = np.random.default_rng(100 + it)
                pert = rng.normal(0, 0.005, n) * a
                a = a + pert
                np.maximum(a, 1e-12, out=a)
                a = a / float(a.sum())
                c = _fft_conv(a)
                M = float(np.max(c))
            else:
                break
    return best_a


def _interpolate_seq(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    new_n = int(max(2, new_n))
    if a.size == new_n:
        return _norm1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return _norm1(y)


def _comb_init(n, n_peaks=8):
    x = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for i in range(n_peaks):
        center = int((i + 0.5) * n / n_peaks)
        width = n / (n_peaks * 3)
        amp = 1.0 + 0.5 * math.sin(2 * math.pi * i / n_peaks)
        a += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _try_update(best_seq, best_score, a, label, t_start):
    if a is None:
        return best_seq, best_score
    try:
        arr = np.asarray(a, dtype=np.float64)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        np.maximum(arr, 0.0, out=arr)
        s = float(arr.sum())
        if arr.size == 0 or s < 0.01:
            return best_seq, best_score
        arr = arr / s
        sc = _safe_eval(arr)
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_seq = arr.copy()
            best_score = float(sc)
            print("  NEW BEST [" + label + "]: " + repr(sc) + " n=" + str(arr.size))
    except Exception:
        pass
    return best_seq, best_score
