def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

    s = a.sum()

        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z.copy()
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def smooth_max_grad(a, beta, top_k=None):
    """Compute gradient of smoothed max of autocorrelation."""
    n = len(a)
    b = fast_autocorr(a)
    size = 2 * n - 1
    
    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    
    if b_max <= 0:
        return np.zeros(n), b
    
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s
    
    # Gradient: g[j] = (1/beta) * sum_s p[j+s] * a[s] for j=0..n-1
    # = (1/beta) * correlate(p, a, 'valid')
    # Using FFT: (a_rev * p) where * is convolution, take [n-1 : 2n-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    a_rev = a[::-1]
    Ar = np.fft.rfft(a_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * P, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, b

        if it % 300 == 0:
            alpha = max(0.003, alpha * 0.93)
        
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        target = power[1:].mean() if len(power) > 1 else power[0]
        
        A_new = A.copy()
        # Vectorized peak suppression
        ratios = power[1:] / max(target, 1e-30)
        # Reduce peaks, slightly boost valleys
        suppress = np.where(ratios > 1.0, 1.0 - alpha * (ratios - 1.0), 1.0)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 100 == 0:
            # Periodic random phase perturbation
            A_cur = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A_cur)) * 0.015

            a_trial = np.fft.irfft(A_cur, fft_size)[:n]
