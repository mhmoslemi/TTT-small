"""
Search for non-negative coefficient sequences minimizing
    score = 2 * n * max(convolve(a, a)) / (sum(a))^2
with FFT-based local search, multiplicative mirror updates, Frank-Wolfe
directions, optional HiGHS LP directions, L-BFGS on a smoothed log-sum-exp
surrogate, exact single-transfer polishing, multi-resolution length search,
and final direct validation.
"""

def next_pow2(x):
    x = int(max(1, x))
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def clean_sequence(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        arr = np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-300:
        arr = np.ones(arr.size, dtype=np.float64) / float(arr.size)
    else:
        arr = arr / s
    return arr


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c), c, 0.0)
    np.maximum(c, 0.0, out=c)
    return c


def fft_conv2(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if b.size != n:
        if b.size == 0:
            b = np.zeros(n, dtype=np.float64)
        elif b.size == 1:
            b = np.full(n, float(b[0]), dtype=np.float64)
        else:
            old_x = np.linspace(0.0, 1.0, b.size)
            new_x = np.linspace(0.0, 1.0, n)
            b = np.interp(new_x, old_x, b).astype(np.float64)
    b = np.where(np.isfinite(b), b, 0.0)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c = np.where(np.isfinite(c), c, 0.0)
    np.maximum(c, 0.0, out=c)
    return c


def pressure_p_a(p, a, factor=1.0):
    """
    P_i = factor * sum_k p[k] a[k-i], for i=0..n-1.
    p has length 2n-1, a has length n.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    expected = 2 * n - 1
    if p.size != expected:
        if p.size == 0:
            p = np.zeros(expected, dtype=np.float64)
        else:
            p = np.interp(np.linspace(0.0, 1.0, expected),
                          np.linspace(0.0, 1.0, p.size), p).astype(np.float64)
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    r = np.fft.irfft(A * Pr, N)[:total]
    out = r[n - 1:2 * n - 1][::-1]
    out = np.where(np.isfinite(out), out, 0.0)
    return float(factor) * out


def smooth1d(x, w):
    x = np.asarray(x, dtype=np.float64)
    w = int(max(1, w))
    if w <= 1 or x.size <= 1:
        return x.copy()
    half = w // 2
    pad = np.pad(x, half, mode="wrap")
    kern = np.ones(w, dtype=np.float64) / float(w)
    y = np.convolve(pad, kern, mode="same")
    return y[half:half + x.size].copy()


def fast_score(a):
    arr = clean_sequence(a)
    n = arr.size
    if n == 0:
        return float("inf")
    c = fft_conv(arr)
    if c.size == 0:
        return float("inf")
    m = float(np.max(c))
    if not math.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def direct_actual_score(arr):
    arr = np.asarray(arr, dtype=np.float64).ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    np.maximum(arr, 0.0, out=arr)
    np.minimum(arr, 1000.0, out=arr)
    if arr.size == 0:
        return float("inf")
    s = float(arr.sum())
    if s < 0.01:
        return float("inf")
    c = np.convolve(arr, arr, mode="full")
    if c.size == 0:
        return float("inf")
    m = float(np.max(c))
    if not math.isfinite(m):
        return float("inf")
    return float(2.0 * arr.size * m / (s * s))

def make_profile(n, kind, rng=None):
    if rng is None:
        rng = np.random.default_rng(12345)
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    y = np.ones(n, dtype=np.float64)

    if kind == "uniform":
        y[:] = 1.0
    elif kind == "sin":
        y[:] = np.sin(np.pi * x) ** 2
    elif kind == "u":
        y[:] = np.cos(0.5 * np.pi * x) ** 2
    elif kind == "ripple2":
        y[:] = 1.0 + 0.70 * np.cos(2.0 * np.pi * x)
    elif kind == "ripple4":
        y[:] = 1.0 + 0.80 * np.cos(4.0 * np.pi * x)
    elif kind == "alt":
        b = np.where((np.arange(n) % 2) == 0, 1.0, -1.0)
        y[:] = 1.0 + 0.78 * b
    elif kind == "alt_sq":
        idx = np.arange(n)
        y[:] = (1.0 + 0.70 * np.cos(np.pi * idx + 0.25)) ** 2
    elif kind == "chirp":
        idx = np.arange(n, dtype=np.float64)
        phase = 2.0 * np.pi * 0.35 * idx * idx / float(n)
        y[:] = 2.0 + 2.0 * np.cos(phase)
    elif kind == "signed_anti":
        b = np.zeros(n, dtype=np.float64)
        half = n // 2
        for i in range(half):
            sgn = 1.0 if rng.random() < 0.5 else -1.0
            b[i] = sgn
            b[n - 1 - i] = -sgn
        y[:] = 1.0 + 0.86 * b
    elif kind == "random":
        y[:] = 1.0
        m = max(8, n // 24)
        coeffs = rng.normal(size=m)
        for k in range(1, m + 1):
            phase = rng.uniform(0.0, 2.0 * np.pi)
            amp = coeffs[k - 1] / math.sqrt(float(k))
            y += amp * np.cos(2.0 * np.pi * float(k) * x + phase)
    else:
        y[:] = 1.0

    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 1e-8, out=y)
    return clean_sequence(y)


def make_lambda(c, gamma=100.0):
    c = np.asarray(c, dtype=np.float64)
    size = c.size
    if size == 0:
        return np.array([], dtype=np.float64)
    m = float(np.max(c))
    if not math.isfinite(m) or m <= 0.0:
        return np.ones(size, dtype=np.float64) / float(size)
    x = np.minimum(c / m, 1.0)
    val = np.exp(np.clip(gamma * (x - 1.0), -700.0, 0.0))
    val = np.where(np.isfinite(val), val, 0.0)
    val += 1e-12
    s = float(val.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(size, dtype=np.float64) / float(size)
    return val / s


def top_indices(c, k):
    c = np.asarray(c, dtype=np.float64)
    size = c.size
    k = int(min(max(0, k), size))
    if k <= 0:
        return np.array([], dtype=int)
    if k >= size:
        return np.argsort(c)[::-1].astype(int)
    part = np.argpartition(c, size - k)[-k:]
    return part[np.argsort(c[part])[::-1]].astype(int)


def top_values(x, valid, k):
    x = np.asarray(x, dtype=np.float64)
    valid = np.asarray(valid, dtype=int)
    if valid.size == 0:
        return np.array([], dtype=int)
    k = int(min(max(0, k), valid.size))
    if k <= 0:
        return np.array([], dtype=int)
    xv = x[valid]
    if k >= xv.size:
        order = np.argsort(xv)[::-1]
        return valid[order].astype(int)
    part = np.argpartition(xv, -k)[-k:]
    part = part[np.argsort(xv[part])[::-1]]
    return valid[part].astype(int)


def bottom_indices(x, k):
    x = np.asarray(x, dtype=np.float64)
    size = x.size
    k = int(min(max(0, k), size))
    if k <= 0:
        return np.array([], dtype=int)
    if k >= size:
        return np.argsort(x).astype(int)
    part = np.argpartition(x, k - 1)[:k]
    return part[np.argsort(x[part])].astype(int)


def top_mask(c, k):
    c = np.asarray(c, dtype=np.float64)
    size = c.size
    k = int(min(max(0, k), size))
    if k <= 0 or k >= size:
        return None
    idx = top_indices(c, k)
    m = np.zeros(size, dtype=bool)
    m[idx] = True
    return m


def mix_max_value(alpha, c, cg, fg):
    one = 1.0 - alpha
    val = (one * one) * c + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
    val = np.where(np.isfinite(val) & (val > 0.0), val, 0.0)
    if val.size == 0:
        return 0.0
    return float(np.max(val))


def line_search_mix(a, g, c, cg, fg, current_M):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return a.copy(), current_M, 0.0
    if not math.isfinite(current_M):
        current_M = float(np.max(c)) if c.size else 0.0

    alphas = np.array([0.01, 0.02, 0.03, 0.05, 0.07, 0.10, 0.15,
                       0.20, 0.30, 0.45, 0.65, 0.85, 1.00], dtype=np.float64)
    best_alpha = 0.0
    best_M = current_M

    for al in alphas:
        al = float(al)
        m = mix_max_value(al, c, cg, fg)
        if math.isfinite(m) and m < best_M:
            best_M = m
            best_alpha = al

    if best_alpha > 0.0 and best_alpha < 1.0:
        refine = np.array([best_alpha * 0.5, best_alpha * 0.85,
                           best_alpha * 1.0, best_alpha * 1.15,
                           best_alpha * 1.35, best_alpha * 0.7,
                           best_alpha * 1.5], dtype=np.float64)
        for al in refine:
            al = float(al)
            if al <= 1e-6 or al >= 1.0 - 1e-9:
                continue
            al = min(1.0, max(1e-6, al))
            m = mix_max_value(al, c, cg, fg)
            if math.isfinite(m) and m < best_M:
                best_M = m
                best_alpha = al

    if best_alpha <= 0.0:
        return a.copy(), current_M, 0.0

    new_a = (1.0 - best_alpha) * a + best_alpha * g
    np.maximum(new_a, 0.0, out=new_a)
    s = float(new_a.sum())
    if s < 1e-15:
        return a.copy(), best_M, best_alpha
    new_a = new_a / s
    return new_a, best_M, best_alpha

        return float(np.max(c)) if size else 0.0
    cn = c.copy()
    cn[d:d + n] += 2.0 * tmove * a
    cn[s:s + n] -= 2.0 * tmove * a
    tt = tmove * tmove
    if 2 * d < size:
        cn[2 * d] += tt
    if 2 * s < size:
        cn[2 * s] += tt
    if s + d < size:
        cn[s + d] -= 2.0 * tt
    np.maximum(cn, 0.0, out=cn)
    return float(np.max(cn))


def multiplicative_descent(a0, overall_end, verbose=False):
    a = clean_sequence(a0)
    n = a.size
    if n <= 1:
        return a
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() < overall_end - 2.0:
        it += 1
        gamma = 40.0 + 25.0 * float(it)
        if gamma > 1800.0:
            gamma = 1800.0
        step = 0.38 + 0.02 * float(it)
        if step > 1.35:
            step = 1.35
        if 0 < stall < 12:
            step *= 0.65

        lam = make_lambda(c, gamma)
        P = pressure_p_a(lam, a)
        P = smooth1d(P, max(3, n // 2500))
        pmax = float(np.max(P))
        pmin = float(np.min(P))
        rngP = pmax - pmin
        if not math.isfinite(rngP) or rngP <= 1e-14 * (pmax + 1e-300):
            stall += 1
            if stall >= 30:
                r = np.random.default_rng(100001 + it)
                a = best_a + r.normal(0.0, 0.018, n) * best_a
                np.maximum(a, 1e-12, out=a)
                a = clean_sequence(a)
                c = fft_conv(a)
                M = float(np.max(c))
                stall = 0
            continue

        Pmean = float(np.mean(P))
        log_ratio = -step * (P - Pmean) / rngP
        log_ratio = np.clip(log_ratio, -2.6, 2.6)
        g = a * np.exp(log_ratio)
        gs = float(g.sum())
        if gs <= 0.0 or not math.isfinite(gs):
            g = np.ones(n, dtype=np.float64) / float(n)
        else:
            g = g / gs

        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        new_a, new_M, alpha = line_search_mix(a, g, c, cg, fg, M)

        if math.isfinite(new_M) and new_M < best_M - 1e-14:
            best_a = new_a.copy()
            best_M = new_M

        if alpha > 0.0 and math.isfinite(new_M) and new_M < M - 1e-14:
            a = new_a
            c = fft_conv(a)
            M = float(np.max(c))
            stall = 0
        else:
            stall += 1

        if stall >= 32:
            r = np.random.default_rng(100173 + it)
            a = best_a + r.normal(0.0, 0.02, n) * best_a
            np.maximum(a, 1e-12, out=a)
            a = clean_sequence(a)
            c = fft_conv(a)
            M = float(np.max(c))
            stall = 0

        if verbose and (it <= 2 or it % 25 == 0):
            print("  mult it=%d score=%.9f best=%.9f t=%.1f" %
                  (it, 2.0 * n * M, 2.0 * n * best_M, time.monotonic() - t0))

    return best_a


def game_direction(f, c, iters=25, seed=0, topK=150):
    f = clean_sequence(f)
    n = f.size
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    active = top_indices(c, topK)
    if active.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)

    val = c[active]
    mx = float(np.max(val))
    if mx <= 0.0 or not math.isfinite(mx):
        return np.ones(n, dtype=np.float64) / float(n)

    logits = np.log(val / mx + 1e-12)
    if seed is not None and seed != 0:
        r = np.random.default_rng(seed)
        logits += r.normal(0.0, 0.02, logits.size)
    logits = np.where(np.isfinite(logits), logits, -1000.0)
    logits -= float(np.max(logits))

    iters = int(max(3, iters))
    g = np.ones(n, dtype=np.float64) / float(n)

    for it in range(iters):
        tmp = np.where(np.isfinite(logits), logits, -1000.0)
        ln = float(np.max(tmp))
        w = np.exp(tmp - ln)
        ws = float(w.sum())
        if ws <= 0.0 or not math.isfinite(ws):
            w = np.ones(active.size, dtype=np.float64)
            ws = float(active.size)
        w = w / ws

        p = np.zeros(c.size, dtype=np.float64)
        p[active] = w

        P = pressure_p_a(p, f)
        P = smooth1d(P, 3)
        pmax = float(np.max(P))
        pmin = float(np.min(P))
        rngP = max(pmax - pmin, 1e-300)
        scaled = (P - pmin) / rngP

        gamma_g = 4.0 + 5.0 * float(it + 1) / float(iters)
        g = np.exp(-gamma_g * scaled)
        g += 0.015
        gs = float(g.sum())
        if gs <= 0.0 or not math.isfinite(gs):
            g = np.ones(n, dtype=np.float64) / float(n)
        else:
            g = g / gs

        h = fft_conv2(f, g)
        ha = h[active]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmax = float(np.max(ha))
        if hmax <= 0.0:
            hmax = 1.0
        ha_n = ha / hmax
        logits += 8.0 * (ha_n - float(np.mean(ha_n)))
        logits = np.where(np.isfinite(logits), logits, -1000.0)
        logits -= float(np.max(logits))

    return g


def lp_direction(f, active, max_k=120):
    if not SCIPY_OK:
        return None
    f = clean_sequence(f)
    n = f.size
    active = np.asarray(active, dtype=int)
    if active.size == 0:
        return None
    if active.size > max_k:
        order = np.argsort(active)
        active = active[order[:max_k]]
    K = active.size
    if K == 0 or n <= 1:
        return None

    row_parts = []
    col_parts = []
    val_parts = []
    for r, k in enumerate(active):
        k = int(k)
        s = max(0, k - n + 1)
        e = min(n, k + 1)
        if s >= e:
            continue
        cols = np.arange(s, e, dtype=np.int64)
        vals = f[k - cols].astype(np.float64).copy()
        if vals.size == 0:
            continue
        rows = np.full(cols.size, r, dtype=np.int64)
        row_parts.append(rows)
        col_parts.append(cols)
        val_parts.append(vals)

    if not row_parts:
        return None

    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[-1] = 1.0
    neg_one = scipy.sparse.csc_matrix(-np.ones((K, 1), dtype=np.float64))
    A_ub = scipy.sparse.hstack((A, neg_one), format="csr")
    A_eq = np.ones((1, n + 1), dtype=np.float64)
    A_eq[0, -1] = 0.0
    b_eq = np.ones(1, dtype=np.float64)
    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = scipy.optimize.linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=np.zeros(K, dtype=np.float64),
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="HiGHS",
            options={"disp": False}
        )
    except Exception:
        return None

    if res is None or not getattr(res, "success", False):
        return None

    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.where(np.isfinite(g) & (g > 0.0), g, 0.0)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs


def frank_wolfe_search(a0, overall_end, active_max=350, game_iters=24,
                       exact_every=8, verbose=False):
    a = clean_sequence(a0)
    n = a.size
    if n <= 1:
        return a
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() < overall_end - 3.0:
        it += 1
        topK = min(active_max, 60 + 12 * it)
        active = top_indices(c, topK)

        g = None
        if (exact_every > 0 and it % exact_every == 0 and
                overall_end - time.monotonic() > 45 and n <= 14000):
            g = lp_direction(a, active, max_k=min(130, topK))

        if g is None:
            g = game_direction(a, c, iters=game_iters, seed=910 + it,
                               topK=min(260, max(80, topK)))
        if g is None:
            break

        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        new_a, new_M, alpha = line_search_mix(a, g, c, cg, fg, M)

        if math.isfinite(new_M) and new_M < best_M - 1e-14:
            best_a = new_a.copy()
            best_M = new_M

        if math.isfinite(new_M) and alpha > 0.0 and new_M < M - 1e-14:
            a = new_a
            c = fft_conv(a)
            M = float(np.max(c))
            stall = 0
        else:
            stall += 1

        if stall >= 28:
            r = np.random.default_rng(200000 + it)
            a = best_a + r.normal(0.0, 0.018, n) * best_a
            np.maximum(a, 1e-12, out=a)
            a = clean_sequence(a)
            c = fft_conv(a)
            M = float(np.max(c))
            stall = 0

        if verbose and (it <= 2 or it % 10 == 0):
            print("  FW it=%d score=%.9f best=%.9f t=%.1f" %
                  (it, 2.0 * n * M, 2.0 * n * best_M, time.monotonic() - t0))

    return best_a


def make_v(a, ref=None):
    a = clean_sequence(a)
    n = a.size
    if n <= 1:
        return np.array([], dtype=np.float64), 0
    if ref is None or ref < 0 or ref >= n:
        ref = int(np.argmax(a))
    ref = int(ref)
    aval = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(aval)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.clip(v, -700.0, 700.0)
    v = np.where(np.isfinite(v), v, -700.0)
    return v, ref


def from_v(v, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if ref < 0:
        ref = n - 1
    if ref > n - 1:
        ref = 0
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    u = np.where(np.isfinite(u), u, 0.0)
    mx = float(np.max(u))
    e = np.exp(u - mx)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def lse_conv(a, beta, mask=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 0.0, np.array([], dtype=np.float64)
    c = fft_conv(a)
    size = 2 * n - 1

    if mask is None:
        cv = c
    else:
        cv = c[mask]

    if cv.size == 0:
        return 0.0, np.zeros(n, dtype=np.float64)

    m = float(np.max(cv))
    shifted = beta * (cv - m)
    shifted = np.clip(shifted, -700.0, 0.0)
    e = np.exp(shifted)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        f = m
        return f, np.zeros(n, dtype=np.float64)

    p_mask = e / s
    p = np.zeros(size, dtype=np.float64)
    if mask is None:
        p[:] = p_mask
    else:
        p[mask] = p_mask

    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    f = m + math.log(s) / float(beta)
    if not math.isfinite(f):
        f = 1e100
    return f, grad


def objgrad_v(v, beta, mask, reg_a, reg_u, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    if n <= 0:
        return 1e100, np.array([], dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return 1e100, np.zeros(n - 1, dtype=np.float64)

    a = from_v(v, ref)
    f, ga = lse_conv(a, beta, mask)
    if not math.isfinite(f):
        f = 1e100
        ga = np.zeros(n, dtype=np.float64)

    if reg_a > 0.0:
        f += 0.5 * reg_a * float(np.dot(a, a))
        ga = ga + reg_a * a

    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)
    gu = np.where(np.isfinite(gu), gu, 0.0)

    gv = np.zeros(n - 1, dtype=np.float64)
    ref = int(ref)
    if ref < 0:
        ref = n - 1
    if ref > n - 1:
        ref = 0
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]

    if reg_u > 0.0:
        f += 0.5 * reg_u * float(np.dot(v, v))
        gv = gv + reg_u * v

    gv = np.where(np.isfinite(gv), gv, 0.0)
    return f, gv


def run_lbfgs(a0, beta, mask, maxfun, reg_u, ref=None):
    a = clean_sequence(a0)
    v0, ref0 = make_v(a, ref)
    if not SCIPY_OK or maxfun <= 0 or v0.size == 0:
        return a
    options = {
        "maxfun": int(maxfun),
        "maxiter": int(maxfun) + 50,
        "ftol": 1e-12,
        "gtol": 1e-7,
        "disp": False
    }
    try:
        res = scipy.optimize.minimize(
            objgrad_v,
            v0,
            args=(beta, mask, 0.0, reg_u, ref0),
            method="L-BFGS-B",
            jac=True,
            options=options
        )
    except Exception:
        return a

    if res is not None and hasattr(res, "x"):
        vx = np.asarray(res.x, dtype=np.float64)
        if vx.size == v0.size and np.all(np.isfinite(vx)):
            out = from_v(vx, ref0)
            out = clean_sequence(out)
            return out
    return a


def lbfgs_phase(a0, overall_end, verbose=False):
    a = clean_sequence(a0)
    n = a.size
    best_a = a.copy()
    best_score = fast_score(a)
    if not SCIPY_OK or n > 14000 or overall_end - time.monotonic() < 45.0:
        return best_a

    v, ref = make_v(a)
    t0 = time.monotonic()
    try:
        objgrad_v(v, 30.0, None, 0.0, 0.0, ref)
        cost = time.monotonic() - t0
    except Exception:
        cost = 0.1
    if cost <= 0.0 or not math.isfinite(cost):
        cost = 0.1
    cost = max(0.03, cost)

    rem = overall_end - time.monotonic()
    if rem <= 0.0:
        return best_a
    total_evals = int(min(700, max(120, rem / (2.0 * cost))))
    schedules = [
        (30.0, 0),
        (80.0, 0),
        (220.0, 500),
        (600.0, 1000)
    ]
    per_base = max(25, total_evals // max(1, len(schedules)))

    for beta, topc in schedules:
        now = time.monotonic()
        if now > overall_end - 12.0:
            break
        per = int(min(180, max(25, (overall_end - now) / (2.5 * cost))))
        per = min(per, per_base)

        mask = None
        if topc > 0:
            ctmp = fft_conv(a)
            mask = top_mask(ctmp, min(topc, max(1, ctmp.size // 3)))

        out = run_lbfgs(a, float(beta), mask, per, 1e-12, ref)
        sc = fast_score(out)
        if math.isfinite(sc):
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = out.copy()
            if sc <= best_score * (1.0 + 1e-6):
                a = out
            else:
                a = best_a
        else:
            a = best_a

    return best_a


def coordinate_polish(a0, overall_end, source_count=0, dest_count=0,
                      verbose=False):
    a = clean_sequence(a0)
    n = a.size
    if n <= 1:
        return a

    if source_count <= 0:
        source_count = 4 if n <= 10000 else 3
    if dest_count <= 0:
        dest_count = 7 if n <= 10000 else 5

    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()
    stall = 0
    it = 0
    gamma = 35.0

    fracs_big = (0.005, 0.01, 0.02, 0.05, 0.10, 0.20)
    fracs_small = (0.01, 0.03, 0.08, 0.18)
    fracs = fracs_small if n > 10000 else fracs_big

    while time.monotonic() < overall_end - 2.0:
        it += 1
        gamma += 2.0
        if gamma > 450.0:
            gamma = 450.0

        lam = make_lambda(c, gamma)
        P = pressure_p_a(lam, a)
        P = smooth1d(P, 3)

        valid = np.nonzero(a > 1e-12)[0]
        if valid.size < 2:
            break

        src = top_values(P, valid, source_count)
        dst = bottom_indices(P, dest_count)

        best_move = None
        best_m = M
        for srr in src:
            s = int(srr)
            avail = float(a[s])
            if avail <= 1e-12:
                continue
            for drr in dst:
                d = int(drr)
                if d == s:
                    continue
                if P[s] <= P[d] + 1e-15:
                    continue
                for frac in fracs:
                    tmove = avail * float(frac)
                    if tmove <= 1e-16:
                        continue
                    m_new = max_after_move(c, a, s, d, tmove)
                    if math.isfinite(m_new) and m_new < best_m - 1e-15:
                        best_m = m_new
                        best_move = (s, d, tmove)

        if best_move is not None:
            s, d, tmove = best_move
            a[s] -= tmove
            a[d] += tmove
            np.maximum(a, 0.0, out=a)
            a = clean_sequence(a)
            c = fft_conv(a)
            M = float(np.max(c))
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 12 == 11:
                r = np.random.default_rng(300000 + it)
                a = best_a + r.normal(0.0, 0.012, n) * best_a
                np.maximum(a, 1e-12, out=a)
                a = clean_sequence(a)
                c = fft_conv(a)
                M = float(np.max(c))
                stall = 0
            if stall > 260:
                break

        if verbose and (it <= 3 or it % 40 == 0):
            print("  coord it=%d score=%.9f best=%.9f t=%.1f" %
                  (it, 2.0 * n * M, 2.0 * n * best_M, time.monotonic() - t0))

    return best_a


def refine_sequence(a0, overall_end, use_lbfgs=True, use_fw=True,
                    use_multiplicative=True, use_coord=True):
    a = clean_sequence(a0)
    best_a = a.copy()
    best_score = fast_score(a)

    t_ref = time.monotonic()
    rem0 = max(1.0, overall_end - t_ref)

    if use_lbfgs and rem0 > 120.0:
        dl = t_ref + 0.18 * rem0
        out = lbfgs_phase(a, dl)
        sc = fast_score(out)
        if math.isfinite(sc):
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = out.copy()
            if sc <= best_score * (1.0 + 1e-6):
                a = out

    if use_multiplicative and overall_end - time.monotonic() > 8.0:
        dl = t_ref + 0.55 * rem0
        out = multiplicative_descent(a, dl)
        sc = fast_score(out)
        if math.isfinite(sc):
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = out.copy()
            if sc <= best_score * (1.0 + 1e-6):
                a = out

    if use_fw and overall_end - time.monotonic() > 25.0:
        dl = t_ref + 0.82 * rem0
        out = frank_wolfe_search(a, dl,
                                 active_max=380 if a.size <= 10000 else 220,
                                 game_iters=22 if a.size <= 10000 else 16,
                                 exact_every=9 if a.size <= 9000 else 14)
        sc = fast_score(out)
        if math.isfinite(sc):
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = out.copy()
            if sc <= best_score * (1.0 + 1e-6):
                a = out

    if use_coord and overall_end - time.monotonic() > 5.0:
        out = coordinate_polish(a, overall_end)
        sc = fast_score(out)
        if math.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_a = out.copy()

    return best_a


def best_from_candidates(candidates):
    if not candidates:
        return float("inf"), None
    best_val = candidates[0][0]
    best_seq = candidates[0][1]
    for i in range(1, len(candidates)):
        v = candidates[i][0]
        if math.isfinite(v) and v < best_val:
            best_val = v
            best_seq = candidates[i][1]
    return best_val, best_seq


def remove_candidate(cand_list, seq):
    for i in range(len(cand_list)):
        if cand_list[i][1] is seq:
            del cand_list[i]
            return
    for i in range(len(cand_list)):
        if np.shares_memory(cand_list[i][1], seq):
            del cand_list[i]
            return


def add_candidate(candidates, seq):
    if seq is None:
        return
    sc = fast_score(seq)
    if math.isfinite(sc):
        candidates.append((sc, np.asarray(seq, dtype=np.float64).copy()))


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    budget = float(budget_s)
    reserve = 135.0 if budget > 500.0 else 75.0
    opt_end = t_start + max(60.0, budget - reserve)
    total = opt_end - t_start

    parent = get_parent_sequence()
    pn = parent.size if parent is not None else 0

    if total > 900.0:
        n_coarse = 2048
        if 4000 <= pn <= 12000:
            n_mid = int(round(float(pn) / 512.0) * 512)
            n_mid = int(max(5000, min(12000, n_mid)))
        else:
            n_mid = 8192
        n_fine = int(min(16384, max(n_mid + 4096, 2 * n_mid)))
    elif total > 450.0:
        n_coarse = 1024
        n_mid = 4096
        n_fine = 8192
    else:
        n_coarse = 1024
        n_mid = 2048
        n_fine = 4096

    print("propose_candidate budget_s=%r opt_time=%.1f n_coarse=%d n_mid=%d n_fine=%d parent=%s" %
          (budget_s, total, n_coarse, n_mid, n_fine, str(parent is not None)))

    seeds = []
    if parent is not None:
        seeds.append(interpolate_sequence(parent, n_coarse))
    else:
        seeds.append(make_profile(n_coarse, "sin", 11))
    seeds.append(make_profile(n_coarse, "signed_anti", 21))
    seeds.append(make_profile(n_coarse, "alt_sq", 22))
    seeds.append(make_profile(n_coarse, "ripple2", 23))
    seeds.append(make_profile(n_coarse, "chirp", 24))
    if total > 600.0:
        seeds.append(make_profile(n_coarse, "random", 25))

    candidates = []

    now = time.monotonic()
    if now < opt_end - 20.0:
        print("=== Coarse exploration ===")
        coarse_time = min(240.0, 0.13 * total)
        per = coarse_time / float(max(1, len(seeds)))
        for i, sd in enumerate(seeds):
            now = time.monotonic()
            if now > opt_end - 18.0:
                break
            end = min(opt_end, now + per)
            if end <= now + 5.0:
                break
            a = refine_sequence(sd, end,
                                use_lbfgs=False,
                                use_fw=False,
                                use_multiplicative=True,
                                use_coord=True)
            add_candidate(candidates, a)
    if not candidates and seeds:
        add_candidate(candidates, seeds[0])

    now = time.monotonic()
    if now < opt_end - 70.0 and candidates:
        print("=== Main mid-length refinement (n=%d) ===" % n_mid)
        _, coarse_best = best_from_candidates(candidates)
        a_mid = interpolate_sequence(coarse_best, n_mid)
        sc_mid = fast_score(a_mid)
        if parent is not None:
            a_parent = interpolate_sequence(parent, n_mid)
            sc_parent = fast_score(a_parent)
            if math.isfinite(sc_parent) and sc_parent < sc_mid - 1e-12:
                a_mid = a_parent
                sc_mid = sc_parent
        main_time = min((opt_end - now) * 0.60, 1050.0)
        mid_end = min(opt_end, now + main_time)
        a_mid = refine_sequence(a_mid, mid_end,
                                use_lbfgs=True,
                                use_fw=True,
                                use_multiplicative=True,
                                use_coord=True)
        add_candidate(candidates, a_mid)

    now = time.monotonic()
    if total > 900.0 and now < opt_end - 650.0 and candidates:
        print("=== Second mid-length basin probe ===")
        tas = list(candidates)
        score1, seq1 = best_from_candidates(tas)
        remove_candidate(tas, seq1)
        score2, seq2 = best_from_candidates(tas)
        if seq2 is not None and math.isfinite(score2) and score2 < score1 + 0.015:
            second_time = min(240.0, (opt_end - now) * 0.12)
            second_end = min(opt_end, now + second_time)
            a2 = interpolate_sequence(seq2, n_mid)
            a2 = refine_sequence(a2, second_end,
                                 use_lbfgs=False,
                                 use_fw=(n_mid <= 8000),
                                 use_multiplicative=True,
                                 use_coord=True)
            add_candidate(candidates, a2)

    now = time.monotonic()
    if now < opt_end - 70.0 and candidates:
        print("=== Fine-length upscale and refine (n=%d) ===" % n_fine)
        _, best_seq = best_from_candidates(candidates)
        fine_time = min((opt_end - now) * 0.50, 720.0)
        fine_end = min(opt_end, now + fine_time)
        a_fine = interpolate_sequence(best_seq, n_fine)
        a_fine = refine_sequence(a_fine, fine_end,
                                 use_lbfgs=(n_fine <= 9000 and SCIPY_OK),
                                 use_fw=(n_fine <= 10000),
                                 use_multiplicative=True,
                                 use_coord=True)
        add_candidate(candidates, a_fine)

    now = time.monotonic()
    if now < opt_end - 35.0 and candidates:
        print("=== Final quick polish ===")
        _, best_seq = best_from_candidates(candidates)
        rem = opt_end - now
        a_pol = best_seq.copy()
        mp_end = now + 0.35 * rem
        a_pol = multiplicative_descent(a_pol, mp_end)
        a_pol = coordinate_polish(a_pol, opt_end)
        add_candidate(candidates, a_pol)

    print("=== Validating top candidates ===")
    selected_seq = None
    selected_score = float("inf")
    tas = list(candidates)
    large_validated = 0
    small_validated = 0
    for _ in range(min(6, len(tas))):
        sc, seq = best_from_candidates(tas)
        remove_candidate(tas, seq)
        if seq is None or not math.isfinite(sc):
            continue
        now = time.monotonic()
        nseq = int(seq.size)
        if nseq > 12000:
            if (opt_end - now) < 85.0 or large_validated >= 2:
                continue
        else:
            if (opt_end - now) < 35.0 or small_validated >= 4:
                continue
        try:
            actual = direct_actual_score(seq)
        except Exception:
            actual = float("inf")
        if nseq > 12000:
            large_validated += 1
        else:
            small_validated += 1
        print("  candidate n=%d fast=%.10f direct=%.10f" %
              (nseq, sc, actual))
        if math.isfinite(actual) and actual < selected_score:
            selected_score = actual
            selected_seq = seq

    if selected_seq is None:
        _, selected_seq = best_from_candidates(candidates)
        if selected_seq is None:
            selected_seq = make_profile(8192, "sin", 998)
        selected_score = fast_score(selected_seq)
        print("  no direct validation, using best fast score=%.10f n=%d" %
              (selected_score, selected_seq.size))

    # If direct validation was skipped for the fastest small candidate, this
    # final consistency check is cheap enough.
    now = time.monotonic()
    if selected_seq is not None and selected_seq.size <= 9000 and (opt_end - now) > 30.0:
        try:
            actual = direct_actual_score(selected_seq)
            if math.isfinite(actual) and actual < selected_score:
                selected_score = actual
                print("  final direct recheck improved score=%.10f" % selected_score)
        except Exception:
            pass

    arr = np.asarray(selected_seq, dtype=np.float64).ravel()
    arr = clean_sequence(arr)
    print("FINAL internal score=%.12f n=%d elapsed=%.1f" %
          (selected_score, arr.size, time.monotonic() - t_start))
    return [float(x) for x in arr]
