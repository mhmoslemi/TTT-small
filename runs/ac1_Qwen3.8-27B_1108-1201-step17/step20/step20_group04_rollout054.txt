def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def init_dual_peak(n, ratio=0.5, seed=0):
    """Mass concentrated in two peaks."""

    c1 = int(n * ratio)
    c2 = int(n * (1 - ratio))

    s = a.sum()
    return a / s

        return best_a, best_cmax

        it += 1
        elapsed = time.monotonic() - t0

            a = np.maximum(a, 0)

            stall = 0

            prob = cp.Problem(cp.Minimize(t_var + 0.5 * reg * cp.sum_squares(delta)), constraints)

                        if c_try < best_c_try:
                            best_c_try = c_try
                            best_step = a_try

3. Nesterov-accelerated projected gradient with log-sum-exp smoothing.
4. Multi-length exploration via interpolation.
5. Multiple structured initializations for diversity.
Key: reformulate delta search as u=a+delta with u>=0, sum(u)=1, yielding a
clean convex QP with only variable bounds and peak constraints.
"""

import numpy as np
import time
import math

_module_solver = None


def get_solver():
    global _module_solver
    if _module_solver is not None:
        return _module_solver
    try:
        import cvxpy as cp
        for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                x = cp.Variable(10, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
                prob.solve(solver=s, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    _module_solver = s
                    print("  Solver: " + str(s))
                    return s
            except Exception:
                continue
        _module_solver = cp.ECOS
        return cp.ECOS
    except Exception:
        return None


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def init_uniform(n):
    return np.ones(n) / n


def init_antisymmetric(n, eps=0.3):
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    mod = 1.0 + eps * np.tanh(2.0 * (x - center) / n)
    return mod / mod.sum()


def init_cosine(n, gamma=1.0):
    x = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * x / (n + 1)) ** (2.0 * gamma)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_power(n, alpha=0.3):
    x = np.arange(n, dtype=np.float64) + 1
    a = x ** (-alpha)
    return a / a.sum()


def init_smoothed_random(n, seed=42):
    rng = np.random.default_rng(seed)
    raw = rng.random(n)
    k = max(3, n // 30)
    kernel = np.exp(-0.5 * (np.arange(-k, k + 1) / (k // 2 + 1)) ** 2)
    kernel /= kernel.sum()
    smoothed = np.convolve(raw, kernel, mode='same')
    smoothed = np.maximum(smoothed, 0)
    s = smoothed.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return smoothed / s


def qp_waterfilling(a0, time_limit, verbose=True, seed=42,
                    top_k_start=30, top_k_end=300,
                    reg_start=5.0, reg_end=5000.0,
                    max_iters=1000, stall_limit=100):
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_cmax = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    solver_name = get_solver()

    if solver_name is None:
        return best_a, best_cmax

    import cvxpy as cp

    it = 0
    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        elapsed = time.monotonic() - t0

        c = fast_autocorr(a)
        cmax = c.max()

        if cmax < best_cmax - 1e-16:
            best_cmax = cmax
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_limit:
            pert_scale = 0.01 + 0.02 * (stall // stall_limit)
            pert = rng.normal(0, pert_scale, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            stall = 0

        progress = min(1.0, it / max_iters)
        top_k = max(10, int(top_k_start + (top_k_end - top_k_start) * progress))
        reg = reg_start * (reg_end / reg_start) ** progress

        top_idx = np.argsort(c)[::-1][:top_k]

        try:
            u = cp.Variable(n, nonneg=True)
            t_var = cp.Variable()

            constraints = [cp.sum(u) == 1.0]

            for k in top_idx:
                k_int = int(k)
                i_start = max(0, k_int - (n - 1))
                i_end = min(n - 1, k_int)
                if i_start > i_end:
                    continue
                i_vals = np.arange(i_start, i_end + 1)
                d_idx = k_int - i_vals
                conv_term = a[i_start:i_end + 1] @ u[d_idx]
                constraints.append(2 * conv_term - t_var <= c[k_int])

            prob = cp.Problem(cp.Minimize(t_var + 0.5 * reg * cp.sum_squares(u - a)), constraints)
            prob.solve(solver=solver_name, verbose=False)

            if prob.status in ('optimal', 'optimal_inaccurate'):
                u_star = u.value
                if u_star is not None and np.all(np.isfinite(u_star)):
                    u_star = np.maximum(u_star, 0)
                    s_u = u_star.sum()
                    if s_u > 1e-15:
                        u_star = u_star / s_u

                    best_step = None
                    best_c_try = best_cmax
                    for ss in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
                        a_try = (1.0 - ss) * a + ss * u_star
                        a_try = np.maximum(a_try, 0)
                        s_try = a_try.sum()
                        if s_try < 1e-15:
                            continue
                        a_try = a_try / s_try
                        c_try = fast_autocorr(a_try).max()
                        if c_try < best_c_try:
                            best_c_try = c_try
                            best_step = a_try.copy()

                    if best_step is not None:
                        a = best_step
                        if best_c_try < best_cmax:
                            best_cmax = best_c_try
                            best_a = best_step.copy()
                    else:
                        a_try = a + rng.normal(0, 0.001, n) * a
                        a_try = np.maximum(a_try, 0)
                        s_try = a_try.sum()
                        if s_try > 1e-15:
                            a = a_try / s_try
        except Exception:
            if verbose and it % 100 == 0:
                print("    QP exception at iter " + str(it))
            break

        if verbose and it % 100 == 0:
            J = 2.0 * n * best_cmax
            print("    QPW it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " k=" + str(top_k) + " reg=" + str(round(reg, 1)))

    return best_a, best_cmax


def gp_frequency_filter(a0, time_limit, verbose=True, seed=13, n_peaks=5, alpha_start=0.3, alpha_end=0.05):
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_cmax = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(a)
        cmax = c.max()
        if cmax < best_cmax - 1e-16:
            best_cmax = cmax
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 100:
            pert = rng.normal(0, 0.03, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0

        A = np.fft.rfft(a, fft_size)
        progress = min(1.0, it / 2000)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        alpha = max(alpha, alpha_end)

        top_idx = np.argsort(c)[::-1][:n_peaks]
        modified = A.copy()
        for k in top_idx:
            k_int = int(k)
            freqs = np.arange(len(A))
            phase = np.cos(2.0 * np.pi * freqs * k_int / fft_size)
            reduction = 1.0 - alpha * np.maximum(phase, 0) ** 2
            modified = modified * reduction

        a_new = np.fft.irfft(modified, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        c_new = fast_autocorr(a_new).max()
        if c_new < cmax:
            a = a_new
        else:
            a = 0.5 * a + 0.5 * a_new
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and it % 200 == 0:
            J = 2.0 * n * best_cmax
            print("    GP it=" + str(it) + " (" + str(round(time.monotonic() - t0, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_cmax


def compute_smoothed_grad(y, beta, top_k=None):
    n = len(y)
    c = fast_autocorr(y)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total

    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(y, fft_size)
    P = np.fft.rfft(p, fft_size)
    a_rev = y[::-1]
    Conv = np.fft.irfft(P * np.fft.rfft(a_rev, fft_size), fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad


def nesterov_opt(a0, time_limit, beta_start=50.0, beta_end=3000.0,
                 beta_growth=1.006, eta_base=0.6, top_k_ratio=0.02,
                 restart_every=2000, verbose=True, seed=42):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad = compute_smoothed_grad(y, beta, top_k)
        eta = eta_base / max(1.0, beta * 0.5)
        eta = min(eta, 0.1)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.03 / (1.0 + 0.05 * (t // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_max_b


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

                                        verbose=False, seed=7)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

    # Phase 9: One last QP burst
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print("\n=== Phase 9: Last QP (n=" + str(n_best) + ") ===")
        t9 = min(remaining * 0.8, 100.0)
        if t9 > 15:
            a_out, _ = qp_waterfilling(norm_sum1(best_seq), t9, verbose=False,
                                       seed=600, top_k_start=100, top_k_end=400,
                                       reg_start=1.0, reg_end=5000.0,
                                       stall_limit=30)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "qp_last", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        best_seq = init_uniform(n_fb)
        best_score = evaluate_safe(best_seq)

    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)

    return [float(x) for x in best_seq]
