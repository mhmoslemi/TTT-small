                break
            
            a = v_to_a(v)
            f, grad_a = lse_objective(a, beta)
            grad_v = a_to_grad_v(a, grad_a)
            
            t += 1
            m = beta1 * m + (1 - beta1) * grad_v
            u = beta2 * u + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1 ** t)
            u_hat = u / (1 - beta2 ** t)
            
            v = v - lr * m_hat / (np.sqrt(u_hat) + eps)
            v = np.where(np.isfinite(v), v, -700.0)
            
            iters += 1
            # Adaptive learning rate: decrease if oscillating
            if iters % 50 == 0:
                lr = max(0.001, lr * 0.95)
    
    return v_to_a(v)

def game_direction(f, active_idx, mw_iters=50, seed=0):
    """Solve the zero-sum game for Frank-Wolfe direction."""
    f = norm_sum1(f)
    n = f.size
    size = 2 * n - 1
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K <= 0:
        return np.ones(n) / n
    
    rng = np.random.default_rng(seed)
    
    # Initialize lambda (row player strategy)
    logits = np.zeros(K)
    c = fft_conv(f)
    init_vals = c[active_idx]
    init_vals = np.where(init_vals > 1e-300, init_vals, 1e-300)
    logits = np.log(init_vals) - float(np.max(np.log(init_vals)))
    logits += rng.normal(0, 0.1, K)
    
    def normalize_logits(loc):
        loc = np.where(np.isfinite(loc), loc, -1000)
        loc = loc - float(np.max(loc))
        w = np.exp(loc)
        s = float(w.sum())
        if s <= 0:
            w = np.ones(K)
            s = K
        return w / s
    
    lambda_w = normalize_logits(logits)
    
    for it in range(mw_iters):
        # Update column player (g): pressure on each coordinate
        pf = np.zeros(size)
        pf[active_idx] = lambda_w
        
        # pressure[i] = sum_k pf[k] * f[(k-i) mod ...] 
        # Actually: (f*g)[k] for fixed k is sum_i f[k-i]*g[i]
        # pressure[i] = sum_k pf[k] * f[k-i] (where k-i is in [0, n-1])
        # This is cross-correlation
        total = 3 * n - 2
        N2 = next_pow2(total)
        A = np.fft.rfft(f, N2)
        Pr = np.fft.rfft(pf[::-1], N2)
        conv = np.fft.irfft(A * Pr, N2)[:total]
        pressure = conv[n-1:2*n-1][::-1]
        pressure = np.where(np.isfinite(pressure) & (pressure > 0), pressure, 0)
        
        # g is the column player's strategy: minimize pressure
        # In the game, the column player wants to minimize the payoff
        # So g should concentrate where pressure is LOW
        pmn = float(np.min(pressure)) if pressure.size > 0 else 0
        pmx = float(np.max(pressure)) if pressure.size > 0 else 0
        ptp = pmx - pmn
        if ptp < 1e-30:
            g = np.ones(n) / n
        else:
            scaled = (pressure - pmn) / ptp
            exp_val = np.exp(-10.0 * scaled)
            exp_val = np.where(np.isfinite(exp_val), exp_val, 0)
            exp_val += 1e-12
            g = exp_val / float(exp_val.sum())
            g = 0.95 * g + 0.05 * (np.ones(n) / n)
            g = g / float(g.sum())
        
        # Update row player (lambda): based on f*g at active indices
        h = fft_conv2(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmac_ = float(np.max(ha))
        hmin = float(np.min(ha))
        hptp = hmac_ - hmin
        if hptp > 1e-30:
            eta = 15.0 / hptp
            logits += eta * (ha - float(np.mean(ha)))
        lambda_w = normalize_logits(logits)
    
    return g

def fw_step(a, c, M, active_idx, seed):
    """One Frank-Wolfe step."""
    n = a.size
    g = game_direction(a, active_idx, mw_iters=40, seed=seed)
    if g is None:
        return a, M, 0.0
    
    cg = fft_conv(g)
    fg = fft_conv2(a, g)
    
    # Line search
    best_alpha = 0.0
    best_m = M
    for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
        one = 1.0 - alpha
        val = (one*one) * c + (2*alpha*one) * fg + (alpha*alpha) * cg
        m = float(np.max(val))
        if math.isfinite(m) and m < best_m:
            best_m = m
            best_alpha = alpha
    
    if best_alpha <= 0:
        return a, M, 0.0
    
    new_a = (1 - best_alpha) * a + best_alpha * g
    new_a = np.maximum(new_a, 0)
    s = float(new_a.sum())
    if s < 1e-15:
        return a, M, 0.0
    new_a = new_a / s
    return new_a, best_m, best_alpha

def coord_descent(a0, c, M, time_limit, verbose=False):
    """Pressure-guided coordinate descent."""
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()
    stall = 0
    rnd = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit - 2 and stall < 100:
        rnd += 1
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        # Find top convolution indices (active set)
        top_k = min(50, 2*n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        
        # Compute pressure
        p_top = np.zeros(2*n-1)
        p_top[top_idx] = 1.0 / top_k
        total = 3*n - 2
        N2 = next_pow2(total)
        A = np.fft.rfft(a, N2)
        Pr = np.fft.rfft(p_top[::-1], N2)
        conv = np.fft.irfft(A * Pr, N2)[:total]
        pressure = conv[n-1:2*n-1][::-1]
        pressure = np.where(np.isfinite(pressure) & (pressure > 0), pressure, 0)
        
        # Find source candidates (high a, high pressure = "expensive")
        # and destination candidates (low a, low pressure = "cheap")
        src_score = pressure * a  # move from here if high
        dst_score = pressure + 1e-5 * a  # move to here if low
        
        src_cands = np.argpartition(src_score, min(10, n-1))[:10]
        dst_cands = np.argpartition(dst_score, min(10, n-1))[:10]
        
        best_move = None
        best_new_M = M
        
        for s in src_cands:
            if a[s] < 1e-12: continue
            for d in dst_cands:
                if d == s: continue
                for frac in (0.01, 0.02, 0.05, 0.1):
                    tmove = a[s] * frac
                    tmove = min(tmove, a[s] * 0.99)
                    if tmove < 1e-16: continue
                    # Apply move and compute new max
                    c_new = c.copy()
                    c_new[d:d+n] += 2.0 * tmove * a
                    c_new[s:s+n] -= 2.0 * tmove * a
                    if 2*d < c_new.size: c_new[2*d] += tmove*tmove
                    if 2*s < c_new.size: c_new[2*s] += tmove*tmove
                    if s+d < c_new.size: c_new[s+d] -= 2.0*tmove*tmove
                    c_new = np.where(np.isfinite(c_new), c_new, 0)
                    m_new = float(np.max(c_new))
                    if math.isfinite(m_new) and m_new < best_new_M - 1e-18:
                        best_new_M = m_new
                        best_move = (s, d, tmove)
        
        if best_move is not None:
            s, d, tmove = best_move
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0)
            a = a / float(a.sum())
            c = fft_conv(a)
            M = float(np.max(c))
            stall = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
        else:
            stall += 1
            if stall % 10 == 0:
                # Random perturbation
                for _ in range(50):
                    s = int(rng.integers(n))
                    d = int(rng.integers(n))
                    if s == d or a[s] < 1e-12: continue
                    frac = float(rng.uniform(0.005, 0.04))
                    tmove = a[s] * frac
                    tmove = min(tmove, a[s] * 0.99)
                    c_new = c.copy()
                    c_new[d:d+n] += 2.0 * tmove * a
                    c_new[s:s+n] -= 2.0 * tmove * a
                    if 2*d < c_new.size: c_new[2*d] += tmove*tmove
                    if 2*s < c_new.size: c_new[2*s] += tmove*tmove
                    if s+d < c_new.size: c_new[s+d] -= 2.0*tmove*tmove
                    c_new = np.where(np.isfinite(c_new), c_new, 0)
                    m_new = float(np.max(c_new))
                    if math.isfinite(m_new) and m_new < M - 1e-17:
                        a[s] -= tmove
                        a[d] += tmove
                        a = np.maximum(a, 0)
                        a = a / float(a.sum())
                        c = fft_conv(a)
                        M = float(np.max(c))
                        stall = 0
                        if M < best_M:
                            best_M = M
                            best_a = a.copy()
                        break
    
    return best_a
