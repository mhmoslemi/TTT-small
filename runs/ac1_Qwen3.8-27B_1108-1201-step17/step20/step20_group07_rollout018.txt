def simplex_project(z):
    """Project z onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_subgrad_smoothed(a, b, top_k, beta):

    b_top = b[idx_top]
    
    # Softmax weights
    shifted = beta * (b_top - b_top.max())
    exp_vals = np.exp(np.clip(shifted, -500, 500))
    w = exp_vals / exp_vals.sum()
    
    # Build weight vector v (length size)
    v = np.zeros(size)
    for i, idx in enumerate(idx_top):
        v[int(idx)] = w[i]

    # = 2 * (v * a_rev)[i + n - 1]
    fft_size = 1 << (size - 1).bit_length()
    v_fft = np.fft.rfft(v, fft_size)
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(v_fft * a_rev_fft, fft_size)
    sg = 2.0 * conv[n-1:n-1+n]
    
    return sg

def fw_subproblem_mw(a, active_indices, n_mw_iter=150):
    n = len(a)
    k_list = np.array(active_indices, dtype=np.int64)
    K = len(k_list)
    if K == 0:
        return np.ones(n) / n, 0.0
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, fft_size)
    y = np.ones(K) / K
    eta = 1.0 / np.sqrt(np.log(max(K, 2)))
    Ct_y = None
    prev_min = None
    for t in range(n_mw_iter):
        v = np.zeros(size)
        for idx in range(K):
            v[int(k_list[idx])] = y[idx]
        v_fft = np.fft.rfft(v, fft_size)
        conv = np.fft.irfft(v_fft * a_rev_fft, fft_size)
        Ct_y = conv[n - 1:n - 1 + n]
        i_star = int(np.argmin(Ct_y))
        cur_min = Ct_y[i_star]
        if prev_min is not None and abs(cur_min - prev_min) < 1e-10:
            break
        prev_min = cur_min
        C_k_star = np.zeros(K)
        for idx in range(K):
            partner = int(k_list[idx]) - i_star
            if 0 <= partner < n:
                C_k_star[idx] = a[partner]
        y *= np.exp(-eta * C_k_star)
        ys = y.sum()
        if ys < 1e-300:
            y = np.ones(K) / K
        else:
            y /= ys
    max_val = Ct_y.max()
    temp = max(0.01, (max_val - Ct_y.min()) * 0.1)
    g = np.exp((Ct_y - max_val) / temp)
    gs = g.sum()
    if gs < 1e-300:
        g = np.ones(n) / n
    else:
        g /= gs
    return g, max_val


def fw_step(a, active_set, t_idx, n_mw_iter=100):
    n = len(a)
    b = fast_autocorr(a)
    M_curr = b.max()
    g, _ = fw_subproblem_mw(a, active_set, n_mw_iter)
    default_alpha = 2.0 / (t_idx + 2.0)
    best_alpha = default_alpha
    best_M = np.inf
    alphas = [default_alpha, max(0.01, default_alpha * 0.5), 1.0 / (t_idx + 1.0),
              max(0.01, default_alpha * 2.0)]
    for alpha in alphas:
        trial = (1.0 - alpha) * a + alpha * g
        trial = np.maximum(trial, 0.0)
        st = trial.sum()
        if st < 1e-15:
            continue
        trial = trial / st
        M_trial = fast_autocorr(trial).max()
        if M_trial < best_M:
            best_M = M_trial
            best_alpha = alpha
    a_new = (1.0 - best_alpha) * a + best_alpha * g
    a_new = np.maximum(a_new, 0.0)
    st = a_new.sum()
    if st > 1e-15:
        a_new = a_new / st
    # Update active set
    b_new = fast_autocorr(a_new)
    M_new = b_new.max()
    violations = np.where(b_new > M_curr + 1e-8)[0]
    if len(violations) > 0:
        viol_amt = b_new[violations] - M_curr
        top_v = violations[np.argsort(viol_amt)[::-1][:min(20, len(violations))]]
        for k in top_v:
            active_set.add(int(k))
    # Also keep top peaks in active set
    top_peaks = np.argsort(b_new)[::-1][:50]

        active_set.add(int(k))
    # Prune active set if too large

        keep = np.argsort(margins)[:600]
        active_list = sorted(active_set)
        active_set = set(int(active_list[i]) for i in keep)
    return a_new, active_set

            for k2 in peak_indices[:6]:
                k2i = int(k2)
                i2s = max(0, k2i - n + 1)
                i2e = min(n - 1, k2i)
                if i2s <= i2e:
                    idx2 = np.arange(i2s, i2e + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]
            pressure[src] = np.max(pressure) + 1.0
            dst = int(np.argmin(pressure))
            if dst == src:
                continue
            found = False
            for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            if not found and stall > 100:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-12:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 1e-14)
                    a = a / a.sum()
                    m_new = fast_autocorr(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
        if not improved:
            stall += 1
            if stall % 100 == 0:
                noise = 0.005 * min(stall / 100, 3.0)
                pert = rng.normal(0, noise, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-12)
                a_r = a_r / a_r.sum()
                m_r = fast_autocorr(a_r).max()
                if m_r < best_m:
                    best_m = m_r
                    best_a = a_r.copy()
                a = best_a.copy()
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakShave (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " stall=" + str(stall))
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n, rng_seed=42):
    x = np.linspace(0.0, 1.0, n)
    candidates = []
    for alpha in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]:
        for beta_p in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]:
            a_raw = (x + 0.01) ** alpha * (1.0 - x + 0.01) ** beta_p
            candidates.append(norm_sum1(a_raw))
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for s in [0.1, 0.15, 0.2, 0.3]:
            a_raw = np.exp(-((x - c) ** 2) / (2 * s ** 2))
            candidates.append(norm_sum1(a_raw))
    for c1 in [0.3, 0.4]:
        for c2 in [0.6, 0.7]:
            for s1 in [0.1, 0.15]:
                for s2 in [0.1, 0.15]:
                    for w in [0.5, 1.0, 2.0]:
                        a_raw = np.exp(-((x - c1) ** 2) / (2 * s1 ** 2)) + w * np.exp(-((x - c2) ** 2) / (2 * s2 ** 2))
                        candidates.append(norm_sum1(a_raw))
    rng = np.random.default_rng(rng_seed)
    for _ in range(20):
        a = norm_sum1(rng.random(n) ** rng.uniform(0.3, 4.0))
        candidates.append(a)
    best_a = None
    best_m = np.inf
    for a in candidates:
        m = fast_autocorr(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()
    return best_a, best_m


def lp_fw_step(a, active_set, solver, t_idx):
    n = len(a)
    b = fast_autocorr(a)
    M_curr = b.max()
    constraint_list = sorted(active_set)
    if len(constraint_list) > 400:
        margins = np.array([M_curr - b[k] for k in constraint_list])
        keep = np.argsort(margins)[:400]
        constraint_list = [constraint_list[i] for i in keep]
    K = len(constraint_list)
    if K == 0:
        return a, active_set
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            coeffs[jj, idx] = a[k_int - idx]
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        g_dir = np.ones(n) / n
    else:
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            g_dir = np.ones(n) / n
        else:
            g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
            gs = g_val.sum()
            if gs < 1e-15:
                g_dir = np.ones(n) / n
            else:
                g_dir = g_val / gs
    default_alpha = 2.0 / (t_idx + 2.0)
    best_alpha = default_alpha
    best_M = np.inf
    for alpha in [default_alpha, max(0.02, default_alpha * 0.5), 1.0 / (t_idx + 1.0)]:
        trial = (1.0 - alpha) * a + alpha * g_dir
        trial = np.maximum(trial, 0.0)
        st = trial.sum()
        if st < 1e-15:
            continue
        trial = trial / st
        M_trial = fast_autocorr(trial).max()
        if M_trial < best_M:
            best_M = M_trial
            best_alpha = alpha
    a_new = (1.0 - best_alpha) * a + best_alpha * g_dir
    a_new = np.maximum(a_new, 0.0)
    st = a_new.sum()
    if st > 1e-15:
        a_new = a_new / st
    b_new = fast_autocorr(a_new)
    M_new = b_new.max()
    violations = np.where(b_new > M_curr + 1e-7)[0]
    if len(violations) > 0:
        viol_amt = b_new[violations] - M_curr
        top_v = violations[np.argsort(viol_amt)[::-1][:30]]
        for k in top_v:
            active_set.add(int(k))
    top_peaks = np.argsort(b_new)[::-1][:60]
    for k in top_peaks:
        active_set.add(int(k))
    if len(active_set) > 500:
        margins = np.array([M_new - b_new[k] for k in active_set])
        keep = np.argsort(margins)[:500]
        active_list = sorted(active_set)
        active_set = set(int(active_list[i]) for i in keep)
    return a_new, active_set


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")
    print("Solver: " + str(solver))

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        a_init, _ = parametric_init(n_base)
        best_seq = a_init
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: Quick parametric sweep to see if better init exists
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.02, 30.0)
    if t1 > 10:
        print("\n=== Phase 1: Parametric init (" + str(n0) + ") ===")
        a_init, _ = parametric_init(n0, rng_seed=123)
        best_seq, best_score = try_update(best_seq, best_score, a_init, "parametric", t_global)

    # Phase 2: FW with MW subproblem (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: FW-MW (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.30, 400.0)
    if t2 > 60:
        a_out, _ = fw_main(best_seq, t2, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fwmw_n" + str(n_best), t_global)

    # Phase 3: FISTA fine-tuning
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: FISTA (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 40:
        a_out, _ = fista_optimize(best_seq, t3, beta_start=50.0, beta_growth=1.004,
                                  beta_max=800.0, top_k_start=50, top_k_max=300, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_n" + str(n_best), t_global)

    # Phase 4: FW with LP subproblem (more precise)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30 and n_best <= 7000:
        a = norm_sum1(best_seq)
        b = fast_autocorr(a)
        active_set = set(int(k) for k in np.argsort(b)[::-1][:60])
        best_a = a.copy()
        best_m = b.max()
        t0 = time.monotonic()
        t = 0
        stall = 0
        while time.monotonic() - t0 < t4:
            t += 1
            a_new, active_set = lp_fw_step(a, active_set, solver, t)
            b_new = fast_autocorr(a_new)
            m_new = b_new.max()
            if m_new < best_m - 1e-14:
                best_m = m_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
            a = a_new
            if stall > 40:
                stall = 0
                b_curr = fast_autocorr(a)
                active_set = set(int(k) for k in np.argsort(b_curr)[::-1][:80])
            if t % 10 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n_best * best_m
                print("    LP-FW t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))
        best_seq, best_score = try_update(best_seq, best_score, best_a, "lpfw_n" + str(n_best), t_global)

    # Phase 5: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Peak shaving (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20:
        a_out, _ = peak_shaving(best_seq, t5, top_k=12, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peaksav_n" + str(n_best), t_global)

    # Phase 6: FISTA with high beta (very fine)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: FISTA fine (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 20:
        a_out, _ = fista_optimize(best_seq, t6, beta_start=100.0, beta_growth=1.005,
                                  beta_max=2000.0, top_k_start=100, top_k_max=400, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_fine", t_global)

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Multi-length ===")
    nearby_lengths = []
    for dn in [-300, -100, 100, 300, 500, -500, 400]:
        nl = n_best + dn
        if 4000 < nl <= 9000 and nl not in nearby_lengths:
            nearby_lengths.append(nl)
    for new_n in nearby_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print("  Trying n=" + str(new_n) + " (remaining=" + str(round(remaining, 1)) + "s)")
        a_long = interpolate_sequence(best_seq, new_n)
        t7a = min(remaining * 0.3, 100.0)
        if t7a > 20:
            a_out, _ = fw_main(a_long, t7a, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fw_n" + str(new_n), t_global)
        remaining = time_budget - (time.monotonic() - t_global)
        t7b = min(remaining * 0.3, 80.0)
        if t7b > 15:
            a_out, _ = peak_shaving(best_seq, t7b, top_k=10, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ps_n" + str(new_n), t_global)

    # Phase 8: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final polish (n=" + str(n_best) + ") ===")
    if remaining > 40:
        t8a = min(remaining * 0.35, 100.0)
        if t8a > 15:
            a_out, _ = peak_shaving(best_seq, t8a, top_k=10, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_ps", t_global)
        remaining = time_budget - (time.monotonic() - t_global)
        t8b = min(remaining * 0.35, 80.0)
        if t8b > 10:
            a_out, _ = fista_optimize(best_seq, t8b, beta_start=200.0, beta_growth=1.01,
                                      beta_max=3000.0, top_k_start=150, top_k_max=400, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_fista", t_global)
        remaining = time_budget - (time.monotonic() - t_global)
        t8c = min(remaining * 0.3, 60.0)
        if t8c > 10 and n_best <= 7000:
            a_out, _ = fw_main(best_seq, t8c, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_fw", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        a_raw = (x + 0.01) ** 1.0 * (1.0 - x + 0.01) ** 1.0
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
