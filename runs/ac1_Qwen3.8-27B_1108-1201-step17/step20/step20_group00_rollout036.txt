4. Coordinate descent for final polish
5. Multi-length strategy with interpolation

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None

def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    return _MODULE_SOLVER

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def spectral_shape(a0, time_limit, alpha=0.3, n_iters=None, verbose=True):
    """Aktas-Kennedy style spectral shaping for flattening autocorrelation."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_obj = obj_val(a)
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    max_iters = n_iters if n_iters else 100000
    no_improve = 0
    cur_alpha = alpha
    
    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 5000:
            break
        
        # Compute FFT of zero-padded sequence
        A = np.fft.rfft(a, n=fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flatten magnitude
        target_mag = np.sqrt(np.mean(mag**2))

        A_new = new_mag * np.exp(1j * phase)
        a_candidate = np.fft.irfft(A_new, fft_size)[:n]

        a_candidate = np.maximum(a_candidate, 0.0)

            best_a = a_new.copy()

            cur_alpha = max(0.01, alpha * (1 - 0.99 * t / max_iters))
        else:

        a = a_new
        
        if verbose and t % 5000 == 0:
            print(f"    Spectral t={t} obj={obj_val(a):.6f} best={best_obj:.6f} alpha={cur_alpha:.4f}")

def spectral_shape_v2(a0, time_limit, verbose=True):
    """Improved spectral shaping with adaptive target and peak-aware flattening."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_obj = obj_val(a)
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    no_improve = 0
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 8000:
            break
        
        A = np.fft.rfft(a, n=fft_size)
        mag = np.abs(A)
        phase = np.exp(1j * np.angle(A))
        
        # Compute current autocorrelation to find peaks
        ac = fast_autocorr(a)
        max_ac = ac.max()
        
        # Strategy: reduce magnitude at frequencies that contribute to peaks
        # The contribution of frequency bin k to autocorrelation lag m is:
        # |A[k]|^2 * exp(-2πi*k*m/fft_size)
        # So to reduce the peak, we can reduce |A| at bins with high |A|^2
        
        # Simple approach: cap the magnitude
        cap = np.percentile(mag, 95) * 1.5
        new_mag = np.minimum(mag, cap)
        
        # Also apply partial flattening
        target = np.sqrt(np.mean(mag**2))
        blend = 0.1 + 0.2 * (1 - t / min(t, 50000))
        new_mag = new_mag + blend * (target - new_mag)
        new_mag = np.maximum(new_mag, 0)
        
        A_new = new_mag * phase
        a_candidate = np.fft.irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0.0)
        
        s = a_candidate.sum()
        if s < 1e-15:
            break
        a_new = a_candidate / s
        
        obj_new = obj_val(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if verbose and t % 5000 == 0:
            print(f"    Spectral2 t={t} obj={obj_val(a):.6f} best={best_obj:.6f}")
    
    return best_a, best_obj

def mult_grad(a0, time_limit, top_k=80, verbose=True):
    """Multiplicative gradient descent with multi-peak targeting and momentum."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    x_prev = x.copy()
    stall = 0
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Precompute helper arrays
    i_arr = np.arange(n, dtype=np.float64)
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 15000:
            break
        
        # Nesterov-like momentum
        beta = (t - 1.0) / (t + 3.0)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-18)
        y = y / y.sum()
        
        # Compute autocorrelation
        ac = fast_autocorr(y)
        
        # Weight the top peaks
        k_max = min(top_k, len(ac))
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]
        
        # Exponential weighting (temperature annealing)
        temp = 20.0 * max(0.1, 1.0 - t / 30000.0)
        w = np.exp(temp * (top_vals - top_vals[0]))
        w = w / w.sum()
        
        # Compute gradient
        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            idx_arr = np.arange(i_s, i_e)
            # d(conv[k])/da[i] = a[k-i] + a[i] (with appropriate bounds)
            mirrored = ki - idx_arr
            # Only valid where 0 <= mirrored < n
            valid = (mirrored >= 0) & (mirrored < n)
            grad[idx_arr[valid]] += wi * y[mirrored[valid]]
            # Self-contribution (when i = k-i, i.e., k = 2i)
            if ki % 2 == 0:
                mid = ki // 2
                if 0 <= mid < n:
                    grad[mid] += wi * y[mid]
        
        # Learning rate
        eta = 2.0 / (np.sqrt(max(t, 1)) + 1.0)
        
        # Multiplicative update
        log_ratio = -eta * grad
        # Clip for numerical stability
        log_ratio = np.clip(log_ratio, -5.0, 5.0)
        x_new = y * np.exp(log_ratio)
        x_new = np.maximum(x_new, 1e-18)
        x_new = x_new / x_new.sum()
        
        x_prev = x
        x = x_new
        
        val = obj_val(x)
        if val < best_val - 1e-12:
            best_val = val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 5000 and stall % 5000 == 0:
            # Restart with noise
            noise = 0.01 * rng.standard_normal(n)
            x_r = best_a * (1 + noise)
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            stall = 0
        
        if verbose and t % 1000 == 0:
            print(f"    MultGrad t={t} ({time.monotonic()-t0:.1f}s) J={best_val:.6f}")
    
    return best_a, best_val

def build_lp_matrix(a, lags, n):
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs

def lp_fw_best(a0, time_limit, solver, init_K=400, max_K=3000, add_per=40, verbose=True):
    """LP-based Frank-Wolfe with line search."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    ac_init = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_init)[-init_K:])
    no_improve = 0
    stall_limit = 2000
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)
        
        K = len(clist)
        if K == 0:
            break
        
        # Build LP
        coeffs = build_lp_matrix(y, clist, n)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        lp_ok = False
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                lp_ok = True
        except Exception:
            if len(active_set) > init_K:
                active_set = set(int(k) for k in clist[:init_K])
        
        if not lp_ok:
            if no_improve > stall_limit // 2:
                break
            if len(clist) > 100:
                active_set = set(int(k) for k in clist[:200])
                continue
            break
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Add violations to active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]
            for k in top_v:
                active_set.add(int(k))
        
        # Line search (golden section)
        lo = 0.01
        hi = 1.0
        
        def trial_obj(alpha):
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                return 1e10
            trial = trial / ts
            return fast_autocorr(trial).max()
        
        f1 = trial_obj(lo + inv_phi * (hi - lo))
        f2 = trial_obj(hi - inv_phi * (hi - lo))
        c1 = lo + inv_phi * (hi - lo)
        c2 = hi - inv_phi * (hi - lo)
        
        for _ in range(25):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = lo + inv_phi * (hi - lo)
                f1 = trial_obj(c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = hi - inv_phi * (hi - lo)
                f2 = trial_obj(c2)
        
        best_alpha = (lo + hi) / 2.0
        
        y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy
        else:
            y = g_dir.copy()
        
        if verbose and t % 200 == 0:
            print(f"    LP-FW t={t} ({time.monotonic()-t0:.1f}s) J={2*n*best_m:.6f} |S|={len(active_set)}")
    
    return best_a, 2.0 * n * best_m

def coord_polish(a0, time_limit, seed=42, verbose=True):
    """Coordinate descent with smart move selection."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 2000:
        ac = fast_autocorr(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False
        
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            top_srcs = idx[np.argsort(contrib)[::-1][:3]]
            for src in top_srcs:
                src = int(src)
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                
                scores = np.zeros(n)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                        scores[k2i - idx2] += a[idx2]
                scores[src] = 1e10
                
                candidates = np.argsort(scores)[:20]
                found = False
                for dst in candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.003, 0.007, 0.015, 0.03]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-15:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            found = True
                            break
                    if found:
                        break
        
        if not improved:
            stall += 1
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
        
        if verbose and stall % 500 == 0:
            print(f"    Coord ({time.monotonic()-t0:.1f}s) J={2*n*best_m:.6f} stall={stall}")
    
    return best_a, 2.0 * n * best_m

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

        return np.ones_like(a, dtype=np.float64) / len(a)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

    """Aktas-Kennedy style spectral shaping to flatten autocorrelation."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_obj = obj_val(a)
    t0 = time.monotonic()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    no_improve = 0
    cur_alpha = alpha

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 6000:
            break

        A = np.fft.rfft(a, n=fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        target_mag = np.sqrt(np.mean(mag**2))
        cur_alpha = alpha * max(0.05, 1.0 - 0.8 * t / 50000.0)
        new_mag = mag + cur_alpha * (target_mag - mag)
        new_mag = np.maximum(new_mag, 1e-10)

        A_new = new_mag * np.exp(1j * phase)
        a_candidate = np.fft.irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0.0)

        s = a_candidate.sum()
        if s < 1e-15:

        a_new = a_candidate / s

        if obj_new < best_obj - 1e-14:

            no_improve = 0

        k_max = min(top_k, len(ac))
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]

        cur_temp = 30.0 * max(0.15, 1.0 - t / 40000.0)
        w = np.exp(cur_temp * (top_vals - top_vals[0]))
        w = w / w.sum()

        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            idx_arr = np.arange(i_s, i_e)
            mirrored = ki - idx_arr
            valid = (mirrored >= 0) & (mirrored < n)
            grad[idx_arr[valid]] += wi * y[mirrored[valid]]
            if ki % 2 == 0:
                mid = ki // 2
                if 0 <= mid < n:
                    grad[mid] += wi * y[mid]

        eta_t = 2.5 / (np.sqrt(max(t, 1)) + 1.0)
        log_ratio = np.clip(-eta_t * grad, -4.0, 4.0)
        x_new = y * np.exp(log_ratio)
        x_new = np.maximum(x_new, 1e-18)
        x_new = x_new / x_new.sum()

        x_prev = x
        x = x_new

        val = obj_val(x)
        if val < best_val - 1e-12:
            best_val = val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 6000 and stall % 6000 == 0:
            noise = 0.015 * rng.standard_normal(n)
            x_r = best_a * (1.0 + noise)
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            stall = 0

        if verbose and t % 2000 == 0:
            print("    MultGrad t=" + str(t) + " (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(round(best_val, 6)))

    return best_a, best_val

def build_lp_matrix(a, lags, n):
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs

def lp_fw_best(a0, time_limit, solver, init_K=500, max_K=4500, add_per=50, verbose=True):
    """LP-based Frank-Wolfe with golden-section line search."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    ac_init = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_init)[-init_K:])
    no_improve = 0
    stall_limit = 3000

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)

        K = len(clist)
        if K == 0:
            break

        coeffs = build_lp_matrix(y, clist, n)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        lp_ok = False
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                lp_ok = True
        except Exception:
            if len(active_set) > init_K:
                active_set = set(int(k) for k in clist[:init_K])

        if not lp_ok:
            if no_improve > stall_limit // 2:
                break
            if len(clist) > 100:
                active_set = set(int(k) for k in clist[:300])
                continue
            break

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]
            for k in top_v:
                active_set.add(int(k))

        lo = 0.01
        hi = 1.0

        def trial_obj(alpha):
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                return 1e10
            trial = trial / ts
            return fast_autocorr(trial).max()

        c1 = lo + inv_phi * (hi - lo)
        c2 = hi - inv_phi * (hi - lo)
        f1 = trial_obj(c1)
        f2 = trial_obj(c2)

        for _ in range(25):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = lo + inv_phi * (hi - lo)
                f1 = trial_obj(c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = hi - inv_phi * (hi - lo)
                f2 = trial_obj(c2)

        best_alpha = (lo + hi) / 2.0

        y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy
        else:
            y = g_dir.copy()

        if verbose and t % 200 == 0:
            print("    LP-FW t=" + str(t) + " (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(round(2 * n * best_m, 6)) + " |S|=" + str(len(active_set)))

    return best_a, 2.0 * n * best_m
