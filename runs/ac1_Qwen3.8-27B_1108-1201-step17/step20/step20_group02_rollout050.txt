We want to minimize:
$$J = \frac{2n \cdot \max_b (a * a)}{(\sum a)^2}$$

where $a * a$ is the linear convolution of the sequence with itself.

This is related to the "flat-top" or "low autocorrelation" problem. The ideal case would be a sequence where the autocorrelation is as flat as possible (all peaks equal), which gives the minimal maximum value.

Key observations:
1. The objective is invariant to scaling of $a$ (since both numerator and denominator scale the same way). So we can normalize $\sum a = 1$ without loss of generality.
2. After normalization, $J = 2n \cdot \max(a*a)$.

        fft_size = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2*n-1]

        viol_mask = conv_ys > M_curr + 1e-10
        if np.any(viol_mask):
            viol_idx = np.where(viol_mask)[0]
            viol_amt = conv_ys[viol_idx] - M_curr
            top_v = viol_idx[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        ac_now = fast_convolve(y)
        current_M = ac_now.max()

        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_M = current_M
        alphas = [1.0, 0.8, 0.6, 0.4, 0.25,
                  max(0.1, 3.0/(iter_count+2)),
                  max(0.05, 2.0/(iter_count+2)),
                  max(0.02, 1.0/(iter_count+1))]

        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M - 1e-15:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 1:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        if iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed,1)) + "s): J=" +
                  repr(2.0*n*best_m) + " |S|=" + str(len(active_set)))

    return best_a, best_m

def smoothed_log_descent(a0, time_limit, T_init=0.08, T_final=0.002,
                         mom=0.8, verbose=True, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    a0 = norm_sum1(a0)
    a0 = np.maximum(a0, 1e-12)
    theta = np.log(a0 / a0.sum())
    vel = np.zeros(n)
    best_a = a0.copy()
    best_exact_M = fast_convolve(best_a).max()
    t0 = time.monotonic()
    t = 0
    eta = 0.15
    no_improve = 0
    max_no_improve = 3000

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        T = T_init * (T_final / T_init) ** elapsed_frac
        eta = max(0.01, 0.15 * (1.0 - 0.7 * elapsed_frac))

        t += 1
        m = min(mom, 0.5 + 0.5 * t / (t + 30))

        theta_look = theta + m * vel
        a_look = softmax_fn(theta_look)

        val, grad = smoothed_max_conv_val_grad(a_look, T)
        log_grad = a_look * (grad - a_look.dot(grad))

        best_trial_theta = None
        best_trial_val = val
        for eta_try in [eta, eta*0.5, eta*0.25]:
            theta_trial = theta_look - eta_try * log_grad
            a_trial = softmax_fn(theta_trial)
            trial_val, _ = smoothed_max_conv_val_grad(a_trial, T)
            if trial_val < best_trial_val - 1e-12:
                best_trial_val = trial_val
                best_trial_theta = theta_trial
                break

        if best_trial_theta is None:
            best_trial_theta = theta_look - eta * 0.1 * log_grad

        new_vel = m * vel + (best_trial_theta - theta_look)
        theta = best_trial_theta
        vel = new_vel
        a_curr = softmax_fn(theta)

        exact_M = fast_convolve(a_curr).max()
        if exact_M < best_exact_M - 1e-13:
            best_exact_M = exact_M
            best_a = a_curr.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if verbose and t % 30000 == 0:
            elapsed = time.monotonic() - t0
            print("    Smoothed t=" + str(t) + " (" + str(round(elapsed,1)) + "s): exact_J=" +
                  repr(2.0*n*best_exact_M) + " T=" + repr(round(T,5)))

    return best_a, best_exact_M

def coordinate_polish(a0, time_limit, rng_seed=77):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(20, 2*n-1)
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:10]:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n-1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je+1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n-1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2+1)
                    scores[idx2] += a[k2i - idx2]

            candidates = np.where(scores < scores.mean())[0]
            if len(candidates) < n//4:
                candidates = np.arange(n)
            dst = int(candidates[np.argmin(scores[candidates])])
            if dst == src:
                dst = int(rng.integers(n))

            found = False
            for frac in [0.01, 0.03, 0.05, 0.08, 0.12]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_M - 1e-15:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found:
                for _ in range(3):
                    j = int(rng.integers(n))
                    if j != src:
                        frac = rng.uniform(0.005, 0.02)
                        a_tmp = a.copy()
                        a_tmp[src] -= a_tmp[src] * frac
                        a_tmp[j] += a_tmp[src] * frac / (1 - frac) if frac < 1 else 0
                        a_tmp = np.maximum(a_tmp, 0.0)
                        s_t = a_tmp.sum()
                        if s_t > 1e-15:
                            a_tmp = a_tmp / s_t
                            M_new = fast_convolve(a_tmp).max()
                            if M_new < best_M:
                                a = a_tmp
                                best_M = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True

        if not improved:
            stall += 1
            if stall % 300 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        a_tmp = a.copy()
                        a_tmp[i] *= 0.9
                        a_tmp[j] += a[i] * 0.1
                        a_tmp = np.maximum(a_tmp, 0)
                        s_t = a_tmp.sum()
                        if s_t > 1e-15:
                            a_tmp = a_tmp / s_t
                            M_new = fast_convolve(a_tmp).max()
                            if M_new < best_M:
                                a = a_tmp
                                best_M = M_new
                                best_a = a.copy()
                                stall = 0

    return best_a, best_M

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def get_initial(t_global):
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
                return p_norm, sc
    except Exception:
        pass
    n = 5000
    x = np.linspace(0, 1, n)
    best_seq = norm_sum1(np.sin(np.pi * x)**2)
    best_score = evaluate_safe(best_seq)
    print("Default init: n=" + str(n) + ", score=" + repr(best_score))
    return best_seq, best_score

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed,1)) + "s)")
    return best_seq, best_score

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = get_solver()
    best_seq, best_score = get_initial(t_global)
    n0 = len(best_seq)

    def remaining():
        return time_budget - (time.monotonic() - t_global)

    # Phase 1: Smoothed gradient descent from multiple diverse starts
    print("\n=== Phase 1: Smoothed descent (multi-restart) ===")
    starts = [best_seq]

    # Add diverse starts
    x = np.linspace(0, 1, n0)
    starts.append(norm_sum1(np.sin(np.pi * x)**2))
    starts.append(norm_sum1(np.exp(-2.0 * (x - 0.5)**2 / (2*0.15**2)) + 0.3*np.ones(n0)))
    starts.append(norm_sum1(np.exp(-2.0*(x-0.3)**2/(2*0.1**2)) + np.exp(-2.0*(x-0.7)**2/(2*0.1**2)) + 0.1*np.ones(n0)))
    rng_init = np.random.default_rng(12345)
    starts.append(norm_sum1(rng_init.dirichlet(np.ones(n0) * 0.5)))

    n_starts = len(starts)
    t_per_start = min(80.0, time_budget * 0.22 / n_starts)

    for i, start in enumerate(starts):
        rt = remaining()
        if rt < 40:
            break
        t_alloc = min(t_per_start, rt * 0.9)
        print("  Start " + str(i+1) + "/" + str(n_starts) + " (t=" + str(round(t_alloc,1)) + "s)")
        a_out, _ = smoothed_log_descent(start, t_alloc, T_init=0.08, T_final=0.001,
                                         mom=0.8, verbose=False, seed=100+i)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "smooth_" + str(i), t_global)

    # Phase 2: LP-FW on best
    print("\n=== Phase 2: LP-FW main (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining() * 0.35, 600.0)
    if t2 > 60 and len(best_seq) <= 10000:
        a_out, _ = lp_fw_refine(norm_sum1(best_seq), t2, solver,
                                 initial_K=800, max_K=5000, add_per_iter=60)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_main", t_global)

    # Phase 3: Smoothed descent again (from LP-FW result, lower T)
    print("\n=== Phase 3: Smoothed polish (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining() * 0.15, 250.0)
    if t3 > 40:
        a_out, _ = smoothed_log_descent(norm_sum1(best_seq), t3, T_init=0.02, T_final=0.0005,
                                         mom=0.85, verbose=False, seed=777)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "smooth_polish", t_global)

    # Phase 4: Multi-n exploration
    print("\n=== Phase 4: Multi-n exploration ===")
    n_best = len(best_seq)
    try_ns = sorted(set([n_best-500, n_best+500, n_best+1000, n_best-1000]))
    try_ns = [l for l in try_ns if 3000 <= l <= 12000]
    try_ns.sort(key=lambda l: abs(l - n_best))

    for new_n in try_ns[:3]:
        rt = remaining()
        if rt < 100:
            break
        print("  Multi-n: n=" + str(new_n))
        a_long = interpolate_seq(best_seq, new_n)
        t5 = min(rt * 0.2, 150.0)
        if t5 > 40 and new_n <= 10000:
            a_out, _ = lp_fw_refine(a_long, t5, solver,
                                     initial_K=500, max_K=4000, add_per_iter=40)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

    # Phase 5: Final LP-FW (long)
    print("\n=== Phase 5: Final LP-FW (n=" + str(len(best_seq)) + ") ===")
    t5 = min(remaining() * 0.3, 400.0)
    if t5 > 60 and len(best_seq) <= 10000:
        a_out, _ = lp_fw_refine(norm_sum1(best_seq), t5, solver,
                                 initial_K=1000, max_K=6000, add_per_iter=80)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # Phase 6: Coordinate polish
    print("\n=== Phase 6: Coordinate polish ===")
    rt = remaining()
    if rt > 30:
        t6 = min(rt * 0.7, 120.0)
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t6, rng_seed=2024)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final: one more LP-FW if time
    rt = remaining()
    if rt > 50 and len(best_seq) <= 10000:
        print("\n=== Final LP-FW ===")
        a_out, _ = lp_fw_refine(norm_sum1(best_seq), min(rt*0.8, 100.0), solver,
                                 initial_K=1200, max_K=6000, add_per_iter=100)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_last", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed,1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x)**2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
