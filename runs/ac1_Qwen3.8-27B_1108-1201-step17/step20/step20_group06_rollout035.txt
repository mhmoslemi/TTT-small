def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        # Periodic randomization to escape local minima
        if it % 300 == 0:
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.05
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial
                    stall = 0
                elif m_t < m_new + 0.01 * best_m:
                    # Accept with some probability (simulated annealing-like)
                    if rng.random() < 0.1:
                        a = a_trial
                        m_new = m_t

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SpecWF it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)) + " alpha=" + str(round(alpha, 4)))

    return best_a, best_m

def lp_fw_improved(a0, time_limit, solver, initial_K=200, max_K=3000,
                   verbose=True, warm_start_set=None):
    """
    Improved LP-Frank-Wolfe with:
    - Dynamic active set management (add/remove constraints)
    - Better line search (Armijo-like)
    - Periodic active set refresh
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)

    if warm_start_set is not None:
        active_set = set(int(k) for k in warm_start_set)
    else:
        active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 800
    lp_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        size = 2 * n - 1

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Dynamic active set management
        constraint_list = sorted(active_set)
        
        # Remove very loose constraints (keep only those within 2*M_curr of max)
        if len(constraint_list) > initial_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            # Keep constraints that are within top 2x margin
            threshold = 0.5 * margins.max() if margins.max() > 0 else 0
            keep = np.where(margins <= max(threshold, 1e-10))[0]
            if len(keep) < max_K:
                active_set = set(int(constraint_list[i]) for i in keep)
                constraint_list = sorted(active_set)
        
        # Cap active set size
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Reset active set on failure
            active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 5
            continue

        lp_count += 1
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find new violations
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                active_set.add(int(k))

        # Periodically add top autocorrelation peaks to active set
        if t % 50 == 0:
            top_peaks = np.argsort(ac)[::-1][:20]
            for k in top_peaks:
                active_set.add(int(k))

        # Adaptive line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        alphas_to_try = [best_alpha, best_alpha * 0.5, best_alpha * 1.5,
                         1.0 / (t + 1), 0.05, 0.1, 0.15, 0.2, 0.3]
        for alpha_tr in alphas_to_try:
            alpha_tr = max(0.01, min(1.0, alpha_tr))
            trial = (1.0 - alpha_tr) * y + alpha_tr * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha_tr

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new
        ac_y = fast_autocorr(y)

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW it=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)) + " |S|=" + str(len(active_set)) + " LPs=" + str(lp_count))

    return best_a, best_m

def subgradient_momentum(a0, time_limit, verbose=True, seed=42):
    """
    Subgradient method with Nesterov acceleration for the non-smooth
    objective max_k b[k]. Uses exact subgradient at the maximizing lag.
    """
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    t_nest = 1.0
    eta = 0.3
    stall = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Compute autocorrelation and find max
        b = fast_autocorr(y)
        k_max = int(np.argmax(b))
        b_max = b[k_max]

        # Subgradient: gradient of b[k_max] w.r.t. y
        # b[k] = sum_i y[i]*y[k-i], so db[k]/dy[j] = y[k-j] + y[j-k] (with appropriate bounds)
        # But since we're on the simplex, we need the gradient of the objective 2n*b[k_max] / (sum(y))^2
        # Simplified: just use the subgradient of max(b) which is the gradient of b[k_max]
        grad = np.zeros(n)
        for j in range(n):
            i1 = k_max - j
            i2 = j - k_max
            if 0 <= i1 < n:
                grad[j] += y[i1]
            if 0 <= i2 < n:
                grad[j] += y[i2]
        
        # This is O(n^2) in the worst case, but k_max is typically O(n), so it's O(n)
        # Actually, let's use FFT for efficiency
        # grad[j] = y[k_max - j] + y[j - k_max] (with bounds checking)
        # This is y convolved with a delta at k_max, plus y convolved with a delta at -k_max
        
        # Better: use the fact that db[k]/dy[j] = y[k-j] (if k-j in [0,n-1]) + y[j-k] (if j-k in [0,n-1])
        # For j in [0, n-1]:
        #   k-j is in [k-(n-1), k] = [k-n+1, k]
        #   j-k is in [-k, n-1-k]
        
        # Actually let me just compute it directly for the range
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += y[k_max - idx]
            grad[k_max - idx] += y[idx]
        # Wait, this double-counts when idx == k_max - idx, i.e., 2*idx == k_max
        
        # Let me redo: grad[j] = sum of y[k_max - j] for valid (k_max - j) + y[j - k_max] for valid (j - k_max)
        # = y[k_max - j] (if 0 <= k_max - j < n) + y[j - k_max] (if 0 <= j - k_max < n)
        
        # For j in [max(0, k_max-n+1), min(n-1, k_max)]: k_max - j is in [0, n-1]
        # For j in [max(0, k_max), min(n-1, k_max+n-1)]: j - k_max is in [0, n-1]
        
        # Since 0 <= k_max <= 2n-2:
        # Range 1: j in [max(0, k_max-n+1), min(n-1, k_max)]
        # Range 2: j in [max(0, k_max), min(n-1, k_max+n-1)] = [max(0, k_max), n-1] (since k_max <= 2n-2, k_max+n-1 >= n-1)
        
        # So:
        # If k_max <= n-1:
        #   Range 1: j in [0, k_max], grad[j] += y[k_max - j]
        #   Range 2: j in [k_max, n-1], grad[j] += y[j - k_max]
        # If k_max >= n:
        #   Range 1: j in [k_max-n+1, n-1], grad[j] += y[k_max - j]
        #   Range 2: j in [k_max, n-1] (empty if k_max >= n), so empty
        #   Wait, if k_max >= n, then j - k_max >= 0 requires j >= k_max >= n, but j < n, so range 2 is empty.
        
        # So more carefully:
        grad = np.zeros(n)
        if k_max <= n - 1:
            # Range 1: j from 0 to k_max
            idx1 = np.arange(0, k_max + 1)
            grad[idx1] += y[k_max - idx1]
            # Range 2: j from k_max to n-1
            idx2 = np.arange(k_max, n)
            grad[idx2] += y[idx2 - k_max]
        else:
            # k_max >= n, only range 1: j from k_max-n+1 to n-1
            idx1 = np.arange(k_max - n + 1, n)
            grad[idx1] += y[k_max - idx1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            x_new = x.copy()
            m_new = best_m
        else:
            # Backtracking line search
            grad_normalized = grad / gnorm
            found = False
            eta_trial = eta
            current_m = fast_autocorr(y).max()
            for _ in range(8):
                z = y - eta_trial * grad_normalized
                x_new = simplex_project(z)
                m_new = fast_autocorr(x_new).max()
                if m_new <= current_m + 1e-10:
                    found = True
                    break
                eta_trial *= 0.5
            if not found:
                x_new = x.copy()
                m_new = best_m
                eta = eta_trial

        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.05, 1.0)
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if stall > 2000:
            stall = 0
            t_nest = 1.0
            # Random restart
            strategy = rng.integers(3)
            if strategy == 0:
                pert = rng.normal(0, 0.03, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                x_r = 0.7 * best_a + 0.3 * rng.dirichlet(np.ones(n) * 0.5)
            else:
                mask = rng.random(n) > 0.5
                x_r = best_a.copy()
                x_r[mask] *= 0.6
                x_r[~mask] *= 1.4
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            eta = 0.3

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SubGrad t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)))

    return best_a, best_m

def targeted_peak_reduction(a0, time_limit, verbose=True, seed=42):
    """
    Directly target the top peaks of the autocorrelation and construct
    moves that reduce them. For each peak at lag k, the gradient tells us
    which pairs (i, k-i) contribute most. We move mass away from those pairs.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 3000:
        b = fast_autocorr(a)
        size = 2 * n - 1
        # Get top peaks
        top_peaks = np.argsort(b)[::-1][:min(15, size)]
        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            lo = max(0, k_int - n + 1)
            hi = min(n - 1, k_int)
            if lo > hi:
                continue

            idx = np.arange(lo, hi + 1)
            # Contribution of each pair to b[k]
            contrib = a[idx] * a[k_int - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue
            
            # Find the pair with maximum contribution
            max_idx_local = np.argmax(contrib)
            src = int(idx[max_idx_local])
            partner = k_int - src
            if not (0 <= partner < n):
                continue

            # Try moving mass from src to reduce this peak
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                # Move to a position that doesn't increase other peaks
                # Try several positions
                best_dest = -1
                best_trial_m = a.copy()
                # Try random positions
                candidates = rng.integers(n, size=8)
                candidates = np.unique(candidates)
                for dst in candidates:
                    dst = int(dst)
                    if dst == src or dst == partner:
                        continue
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    s_t = a_t.sum()
                    if s_t < 1e-15:
                        continue
                    a_t = a_t / s_t
                    m_t = fast_autocorr(a_t).max()
                    if m_t < best_m - 1e-16:
                        a = a_t
                        best_m = m_t
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.005, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakRed (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m

def multi_scale_optimize(a0, time_limit, solver, verbose=True):
    """
    Multi-scale optimization: work at a coarser resolution first,
    then refine at full resolution.
    """
    n = len(a0)
    t0 = time.monotonic()
    
    # Downsample to 1/4 resolution
    coarse_n = max(100, n // 4)
    a_coarse = interpolate_sequence(a0, coarse_n)
    
    remaining = time_limit - (time.monotonic() - t0)
    
    # Optimize at coarse scale
    t_coarse = remaining * 0.15
    a_out, _ = spectral_waterfill(a_coarse, t_coarse, verbose=False, seed=42)
    a_coarse_best, _ = lp_fw_improved(a_out, t_coarse, solver, initial_K=100, max_K=500, verbose=False)
    
    # Upscale back
    a_up = interpolate_sequence(a_coarse_best, n)
    
    remaining = time_limit - (time.monotonic() - t0)
    
    # Refine at full scale
    a_full, _ = spectral_waterfill(a_up, remaining * 0.2, verbose=False, seed=88)
    a_full, _ = lp_fw_improved(a_full, remaining * 0.4, solver, initial_K=300, max_K=2000, verbose=True)
    
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 30:
        a_full, _ = targeted_peak_reduction(a_full, remaining * 0.8, verbose=True, seed=123)
    
    return a_full

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

def propose_candidate(budget_s=1800.0):
    """
    Multi-strategy optimization for minimizing 2*n*max(autocorr)/(sum)^2:
    1. Spectral water-filling with adaptive peak reduction
    2. LP-Frank-Wolfe with dynamic active set management
    3. Subgradient method with Nesterov acceleration
    4. Targeted peak reduction via pair swaps
    5. Multi-scale homotopy for escape from local minima
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Time budget: " + str(round(time_budget, 1)) + "s, solver: " + str(solver))

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + str(round(p_score, 6)))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6000
        x = np.linspace(0, 1, n_base)
        # Use a raised cosine window as initial guess (better than sin^2)
        best_seq = norm_sum1(0.5 * (1 + np.cos(2 * np.pi * x)))
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + str(round(best_score, 6)))

    n0 = len(best_seq)
    print("Start: n=" + str(n0) + ", J=" + str(round(best_score, 6)))

    # Phase 1: Spectral water-filling
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Spectral WF (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.08, 100.0)
    if t1 > 20:
        a_out, _ = spectral_waterfill(norm_sum1(best_seq), t1, verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec1", t_global)

    # Phase 2: LP-Frank-Wolfe (main optimizer)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P2: LP-FW (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.25, 380.0)
    if t2 > 30 and n_best <= 10000:
        a_out, _ = lp_fw_improved(norm_sum1(best_seq), t2, solver,
                                   initial_K=300, max_K=3000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 3: Subgradient with momentum
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P3: SubGrad (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20:
        a_out, _ = subgradient_momentum(norm_sum1(best_seq), t3, verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "subg1", t_global)

    # Phase 4: Targeted peak reduction
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P4: PeakRed (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        a_out, _ = targeted_peak_reduction(norm_sum1(best_seq), t4, verbose=True, seed=321)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak1", t_global)

    # Phase 5: Multi-scale exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P5: Multi-scale (base n=" + str(n_best) + ") ===")
