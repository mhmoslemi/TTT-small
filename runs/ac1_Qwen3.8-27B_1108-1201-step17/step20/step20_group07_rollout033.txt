Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (accelerated proximal gradient) on log-sum-exp smoothed objective
2. Multi-beta schedule for progressive refinement
3. Peak shaving for fine-grained local improvements
4. Multi-length exploration
5. Diverse restarts

import numpy as np
import time

def fast_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    """Project onto probability simplex."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.nonzero(u * np.arange(1, n + 1) > (cssv - 1)))
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(z - theta, 0.0)

def compute_gradient(a, beta, N2):
    """Compute gradient of smoothed objective."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = beta * (b - b_max)
    exp_vals = np.exp(np.clip(shifted, -500, 500))
    p = exp_vals / exp_vals.sum()

    A = np.fft.rfft(a, N2)
    cross = np.fft.irfft(P * np.conj(A), N2)[:n]
    grad = (2.0 / beta) * cross
    return grad, b, b_max

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (accelerated proximal gradient) on log-sum-exp smoothed objective with adaptive beta
2. Multi-scale peak shaving with pressure maps
3. Frank-Wolfe style updates with away-steps for final polish
4. Multi-length exploration with interpolation
5. Diverse multi-start with different initializations

import numpy as np
import time


def fast_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Euclidean projection onto probability simplex."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1)
    return np.maximum(z - theta, 0.0)


def compute_smoothed_grad_and_val(a, beta, N2=None):
    """Compute gradient and value of log-sum-exp smoothed max of autocorrelation."""
    n = len(a)
    if N2 is None:
        N2 = 1 << (3 * n - 4).bit_length()
    b = fast_conv(a)
    b_max = b.max()
    shifted = beta * (b - b_max)
    exp_vals = np.exp(np.clip(shifted, -700, 700))
    p = exp_vals / exp_vals.sum()
    P = np.fft.rfft(p, N2)
    A = np.fft.rfft(a, N2)
    cross = np.fft.irfft(P * np.conj(A), N2)[:n]
    grad = (2.0 / beta) * cross
    val = (1.0 / beta) * (np.log(exp_vals.sum()) + beta * b_max)
    return grad, val, b, b_max


def fista_optimize(a0, time_limit, beta_start=10.0, beta_end=400.0, verbose=True):
    """FISTA on the log-sum-exp smoothed objective with adaptive beta and backtracking."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    N2 = 1 << (3 * n - 4).bit_length()

    t0 = time.monotonic()
    y = a.copy()
    t_param = 1.0
    stall = 0
    step = 0
    L = 10.0
    rng = np.random.default_rng(7)

    while time.monotonic() - t0 < time_limit:
        step += 1
        frac = min(1.0, step / 3000.0)
        beta = beta_start * (beta_end / beta_start) ** frac

        gy, vy, _, _ = compute_smoothed_grad_and_val(y, beta, N2)

        # Backtracking line search
        for ls_iter in range(20):
            a_new = simplex_project(y - gy / L)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = np.ones(n) / n
                s_new = 1.0
            a_new = a_new / s_new

            ga_new, va_new, _, mn_new = compute_smoothed_grad_and_val(a_new, beta, N2)
            # Decrease condition
            pred = vy + gy.dot(a_new - y) + 0.5 * L * np.dot(a_new - y, a_new - y)
            if va_new <= pred + 1e-12:
                break
            L *= 2.0

        # Check improvement on true objective
        b_new = fast_conv(a_new)
        m_new = b_new.max()
        if m_new < best_m - 1e-16:
            best_m = m_new

        else:
            stall += 1
            L = min(L * 1.1, 10000.0)

        a = a_new

    return best_a, best_m


def peak_shaving(a0, time_limit, top_k=15, verbose=True):
    """Aggressive peak shaving: directly reduce top autocorrelation peaks."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 3000:
        b = fast_conv(a)
        peak_indices = np.argsort(b)[::-1][:top_k]
        peak_vals = b[peak_indices]
        b_max = peak_vals[0]

        # Build pressure map
        pressure = np.zeros(n)
        for k_i in peak_indices[:8]:
            ki = int(k_i)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                pressure[idx] += a[ki - idx]

        improved = False
        for k_i in peak_indices[:6]:
            ki = int(k_i)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            if a[src] < 1e-14:
                continue
            # Find safe destination
            safe = pressure < np.percentile(pressure, 60)
            safe[src] = False
            if safe.sum() < 10:
                safe = np.ones(n, dtype=bool)
                safe[src] = False
            dst_cand = np.where(safe)[0]
            if len(dst_cand) == 0:
                continue
            dst_pressure = pressure[dst_cand]
            dst = dst_cand[np.argmin(dst_pressure)]

            for frac in [0.003, 0.007, 0.015, 0.03, 0.06]:
                amt = a[src] * frac
                if amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 100 == 0:
                noise = rng.normal(0, 0.008, n) * best_a
                a_r = norm1(best_a + noise)
                m_r = fast_conv(a_r).max()
                if m_r < best_m:
                    best_m = m_r
                    best_a = a_r.copy()
                    stall = 0
                a = best_a.copy()

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave stale={stall} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def fw_away_steps(a0, time_limit, verbose=True):

    n = len(a0)

    best_m = fast_conv(a).max()

    step = 0

        step += 1
        b = fast_conv(a)
        b_max = b.max()

        if b_max < best_m - 1e-16:
            best_m = b_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Active set (top peaks)
        top_peaks = np.argsort(b)[::-1][:50]
        # Subgradient: average of gradients at active peaks
        g = np.zeros(n)
        for ki in top_peaks[:20]:
            ki = int(ki)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                g[idx] += a[ki - idx]
        g *= 2.0 / min(len(top_peaks), 20)

        # FW: towards min gradient
        vs = list(vertex_set) if len(vertex_set) > 0 else list(range(n))
        if len(vs) > 500:
            vs = [int(x) for x in np.random.default_rng(step).choice(n, 500, replace=False)]
        g_vs = np.array([g[i] for i in vs])
        i_plus = vs[int(np.argmin(g_vs))]
        i_minus = vs[int(np.argmax(g_vs))] if len(vs) > 1 else i_plus

        # Try multiple step sizes
        best_alpha = 0
        best_trial_m = best_m
        alphas = [2.0 / (step + 2.0), 1.0 / (step + 1), 0.1, 0.05, 0.02, 0.01]
        for alpha in alphas:

            trial = (1 - alpha) * a + alpha * (np.zeros(n) if True else None)

            trial[i_plus] += alpha

            a = best_trial
            if i_plus not in vertex_set and a[i_plus] > 1e-12:

            if i_minus in vertex_set and a[i_minus] < 1e-12:
                vertex_set.discard(i_minus)
            if len(vertex_set) > 200:
                # Prune

                keep = arr[np.argsort(a[arr])[::-1][:200]]
