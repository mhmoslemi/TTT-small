def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def fista_smoothed(a0, time_limit, beta_start=20.0, beta_end=1000.0,
                   beta_growth=1.004, top_k=15, restart_every=3000,
                   verbose=True):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(42)
    avg_x = x.copy()
    avg_count = 0.0
    bb_step = 1.0 / (n ** 0.5)

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        b = fast_autocorr(y)
        if top_k is not None and top_k < len(b) and top_k > 0:
            idx = np.argsort(b)[::-1][:top_k]
            b_use = np.zeros(len(b))
            b_use[idx] = b[idx]
            b_max = b[idx].max()
        else:
            b_use = b
            b_max = b.max()

        shifted = beta * (b_use - b_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        p_total = exp_vals.sum()
        if p_total < 1e-300:
            mask = (b_use == b_max)
            p = np.zeros_like(b_use)
            p[mask] = 1.0 / max(1, mask.sum())
        else:
            p = exp_vals / p_total

        a_rev = y[::-1]
        conv_len = n + len(p) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        Ar = np.fft.rfft(a_rev, fft_size)
        Pr = np.fft.rfft(p, fft_size)
        conv = np.fft.irfft(Ar * Pr, fft_size)
        grad = conv[n - 1:2 * n - 1] / beta

        eta = min(bb_step, 0.5)
        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        avg_count += 1.0
        avg_x = avg_x + (x_new - avg_x) / avg_count

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.02 / (1.0 + 0.03 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 5.0)
            bb_step = 1.0 / (n ** 0.5)
            avg_x = x_r.copy()
            avg_count = 0.0
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    FISTA restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    FISTA t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + repr(round(beta, 1)))

    J_best = 2.0 * n * best_max_b

            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        b = fast_autocorr(y)
        b_max = b.max()

        # Find all positions where b is within 1% of max (handle plateaus)
        threshold = b_max * 0.999
        active_idx = np.where(b >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(b)])

        # Compute subgradient: average of gradients at active positions
        g = np.zeros(n)
        for k_idx in active_idx:
            ki = int(k_idx)
            # dc[ki]/da[j] = y[ki-j] if 0<=ki-j<n, plus y[j-ki] if 0<=j-ki<n
            for j in range(n):
                val = 0.0
                if 0 <= ki - j < n:
                    val += y[ki - j]
                if 0 <= j - ki < n:
                    val += y[j - ki]
                g[j] += val
        g /= len(active_idx)

        # Step size: alpha = c / sqrt(t)
        alpha = 1.0 / np.sqrt(t + 1.0)

        z = y - alpha * g
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        avg_count += 1.0
        avg_x = avg_x + (x_new - avg_x) / avg_count

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.01 / (1.0 + 0.02 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            avg_x = x_r.copy()
            avg_count = 0.0

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    NSubG t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    J_avg = 2.0 * n * fast_autocorr(avg_x).max()
    J_best = 2.0 * n * best_max_b
    if J_avg < J_best:
        return avg_x, fast_autocorr(avg_x).max()
    return best_a, best_max_b

def peak_equalization_lp(a0, time_limit, solver, top_k=10, verbose=True):
    """Iterative peak equalization using small LPs on top-k peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    max_iters = 500

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()

        # Find top-k peaks
        top_idx = np.argsort(ac)[::-1][:top_k]

        # Build gradient matrix for top-k peaks
        # G[i, j] = dac[top_idx[i]]/da[j]
        G = np.zeros((len(top_idx), n))
        for ii, ki in enumerate(top_idx):
            ki = int(ki)
            for j in range(n):
                val = 0.0
                if 0 <= ki - j < n:
                    val += a[ki - j]
                if 0 <= j - ki < n:
                    val += a[j - ki]
                G[ii, j] = val

        # Solve LP: minimize t
        # s.t. G @ d <= t
        #      sum(d) = 0
        #      -1 <= d <= 1
        d_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            G @ d_var <= t_var,
            cp.sum(d_var) == 0,
            d_var >= -1,
            d_var <= 1
        ]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        d_val = np.asarray(d_var.value, dtype=np.float64)
        if d_val is None:
            continue

        # Line search
        best_alpha = 0.0
        best_trial_M = float('inf')
        alphas = [0.01, 0.005, 0.002, 0.02, 0.05]
        for alpha in alphas:
            trial = a + alpha * d_val
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if best_alpha == 0.0 or best_trial_M >= M_curr - 1e-14:
            # Try small random perturbations
            rng = np.random.default_rng(t)
            improved = False
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-12:
                    continue
                frac = float(rng.uniform(0.001, 0.01))
                amt = a[i] * frac
                trial = a.copy()
                trial[i] -= amt
                trial[j] += amt
                trial = np.maximum(trial, 0.0)
                s_t = trial.sum()
                if s_t < 1e-15:
                    continue
                trial = trial / s_t
                M_trial = fast_autocorr(trial).max()
                if M_trial < M_curr - 1e-14:
                    a = trial
                    improved = True
                    break
            if not improved:
                # Check if we should continue
                if t > 50 and t % 20 == 0:
                    # Check convergence
                    if t > 100:
                        break
            continue

        a = a + best_alpha * d_val
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakEQ LP t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m

def cyclic_shift_best(a0):
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a0, fft_size)
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m

def local_search(a0, time_limit, verbose=True):
    """2-exchange local search with adaptive strategy."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:30]
        improved = False

        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(top_peaks[int(rng.integers(len(top_peaks)))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])

            # Find destination: away from all top peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:15]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            dst = int(candidates[int(rng.integers(len(candidates)))])
            if dst == src or a[src] < 1e-12:
                continue

            frac = float(rng.choice([0.002, 0.005, 0.01, 0.02, 0.05]))
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-14:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                ac = fast_autocorr(a)
                top_peaks = np.argsort(ac)[::-1][:30]

        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.002, 0.02))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

    return best_a, best_m

def multiplicative_update(a0, time_limit, n_iters=500, verbose=True):
    """Multiplicative (sinkhorn-like) update to flatten convolution peaks."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        ac_max = ac.max()
        if ac_max < 1e-15:
            break

        # Compute "cost" for each element: how much it contributes to peaks
        cost = np.zeros(n)
        top_k = 20
        top_idx = np.argsort(ac)[::-1][:top_k]
        for ki in top_idx:
            ki = int(ki)
            weight = ac[ki] / ac_max
            for j in range(n):
                val = 0.0
                if 0 <= ki - j < n:
                    val += a[ki - j]
                if 0 <= j - ki < n:
                    val += a[j - ki]
                cost[j] += weight * val

        # Multiplicative update: reduce elements with high cost
        eps = 0.01 * (1.0 + 0.1 * (it // 100))
        a_new = a * (1.0 - eps * cost / (cost.max() + 1e-15))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = a_new  # Still update, but don't record as best

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    MultiUpd it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Try to use parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: FISTA with smoothed max (initial optimization)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: FISTA-smoothed (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 60:
        a_out, _ = fista_smoothed(norm_sum1(best_seq), t1,
                                  beta_start=15.0, beta_end=800.0,
                                  beta_growth=1.004, top_k=12,
                                  restart_every=3000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista1", t_global)

    # Phase 2: Cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: Cyclic shift ===")
    if remaining > 30:
        a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic1", t_global)

    # Phase 3: Nesterov subgradient on exact max
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Nesterov SubGrad (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 60:
        a_out, _ = nesterov_subgrad(norm_sum1(best_seq), t3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nsubg", t_global)

    # Phase 4: Peak equalization LP
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Peak Equalization LP (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30 and n_best <= 12000:
        a_out, _ = peak_equalization_lp(norm_sum1(best_seq), t4, solver,
                                        top_k=8, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peaklp", t_global)

    # Phase 5: Local search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Local search (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30:
        a_out, _ = local_search(norm_sum1(best_seq), t5, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "local1", t_global)

    # Phase 6: Cyclic shift again
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic2", t_global)

    # Phase 7: Multiplicative update
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Multiplicative update (n=" + str(n_best) + ") ===")
    t7 = min(remaining * 0.15, 200.0)
    if t7 > 30:
        a_out, _ = multiplicative_update(norm_sum1(best_seq), t7,
                                         n_iters=300, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "mult1", t_global)

    # Phase 8: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = [n_best + 500, n_best + 1500]
    longer_lengths = [l for l in longer_lengths if l <= 12000 and l > n_best]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("\n=== Phase 8: Scale n=" + str(new_n) + " ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t8a = min(remaining * 0.25, 200.0)
        if t8a > 40:
            a_out, _ = fista_smoothed(a_long, t8a,
                                      beta_start=20.0, beta_end=600.0,
                                      beta_growth=1.005, top_k=10,
                                      restart_every=2500, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
            best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_n" + str(new_n), t_global)

    # Phase 9: Second FISTA pass with higher beta
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 9: FISTA v2 (n=" + str(n_best) + ") ===")
    t9 = min(remaining * 0.20, 250.0)
    if t9 > 40:
        a_out, _ = fista_smoothed(norm_sum1(best_seq), t9,
                                  beta_start=50.0, beta_end=1500.0,
                                  beta_growth=1.006, top_k=8,
                                  restart_every=2000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista2", t_global)

    # Phase 10: Final local search polish
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 10: Final polish (n=" + str(len(best_seq)) + ") ===")
    if remaining > 30:
        t10 = min(remaining * 0.35, 180.0)
        if t10 > 20:
            a_out, _ = local_search(norm_sum1(best_seq), t10, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "local_final", t_global)

    # Final cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_final", t_global)

    # Final FISTA micro-polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t_f = min(remaining * 0.50, 120.0)
        if t_f > 20:
            a_out, _ = fista_smoothed(norm_sum1(best_seq), t_f,
                                      beta_start=100.0, beta_end=2500.0,
                                      beta_growth=1.01, top_k=6,
                                      restart_every=1500, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_final", t_global)

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
