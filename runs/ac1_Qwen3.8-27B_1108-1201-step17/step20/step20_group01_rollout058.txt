        # Nesterov momentum update
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y_new = x_new * (1.0 + beta_new) / beta - x * (beta_new - 1.0) / beta
        y_new = np.maximum(y_new, 1e-20)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = x_new.copy()
            sy = x_new.sum()
        y_new = y_new / sy
        x = x_new
        y = y_new
        beta = beta_new

        # Periodic restart if stuck
        if stall > 2000 and stall % 500 == 0:
            noise_scale = 0.05 * np.exp(-stall / 3000.0)
            noise = noise_scale * np.random.standard_normal(n)
            y_r = best_a + noise
            y_r = np.maximum(y_r, 1e-15)
            y = y_r / y_r.sum()
            x = y.copy()
            stall = 0

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    MD t={:6d} ({:.1f}s) obj={:.8f} stall={}".format(t, elapsed, best_val, stall))

    return best_a, best_val

def coord_descent(a0, time_limit, n_peaks=10, verbose=True):
    """Greedy coordinate descent with peak balancing."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_val = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    max_stall = 3000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_val = 2.0 * n * best_m
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Get top peak lags
        n_pk = min(n_peaks, len(ac) - 1)
        peak_lags = np.argpartition(ac, -n_pk)[-n_pk:]
        peak_lags = peak_lags[np.argsort(ac[peak_lags])[::-1]]

        improved = False

        # For each peak lag, find the coordinate contributing most and reduce it
        for ki in peak_lags[:min(5, n_pk)]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(ki)
            # Coordinates contributing to this lag: j where 0<=j<n and 0<=ki-j<n
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start >= j_end:
                continue
            js = np.arange(j_start, j_end)
            contrib = a[js] * a[ki - js]
            if contrib.max() < 1e-20:
                continue

            # Find the coordinate with max contribution
            src = int(js[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-15:
                continue

            # Compute "pressure" on each coordinate
            pressure = np.zeros(n)
            for pk in peak_lags[:n_pk]:
                pk = int(pk)
                ps = max(0, pk - n + 1)
                pe = min(n, pk + 1)
                if ps < pe:
                    pjs = np.arange(ps, pe)
                    pressure[pjs] += a[pk - pjs]

            # Find quiet destination
            pressure[src] = np.inf
            quiet = np.argsort(pressure)[:20]

            tried = False
            for dst in quiet:
                dst = int(dst)
                if dst == src:
                    continue
                # Try moving different fractions
                for frac in [0.05, 0.15, 0.3, 0.5, 0.7]:
                    amt = src_val * frac
                    if amt < 1e-18:
                        break
                    a_try = a.copy()
                    a_try[src] -= amt
                    a_try[dst] += amt
                    a_try = np.maximum(a_try, 0.0)
                    st = a_try.sum()
                    if st < 1e-15:
                        continue
                    a_try = a_try / st
                    Mt = fast_autocorr(a_try).max()
                    if Mt < M_curr - 1e-15:
                        a = a_try
                        improved = True
                        tried = True
                        break
                if tried:
                    break

        if not improved:
            # Random perturbation to escape
            if stall > 200:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_val = 2.0 * n * best_m
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 500 == 0 and stall > 0:
            print("    CD ({:.1f}s) obj={:.8f} stall={}".format(time.monotonic() - t0, best_val, stall))

    return best_a, best_val
