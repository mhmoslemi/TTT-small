def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c

def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)

    se = float(e.sum())
    if se <= 0.0 or not math.isfinite(se):
        e = np.zeros(cv.size, dtype=np.float64)
        e[0] = 1.0
        se = 1.0
    p = e / se

    pf = np.zeros(size, dtype=np.float64)
    if mask is None:
        pf[:] = p
    else:
        pf[mask] = p

    Pr = np.fft.rfft(pf[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    f = m + math.log(se) / beta

    if not math.isfinite(f):
        f = 1e100
    return f, grad


def softmax_from_u(u):
    u = np.asarray(u, dtype=np.float64)
    n = u.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(u)):
        return np.ones(n, dtype=np.float64) / float(n)
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def make_v(a, ref=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 1:
        return np.array([], dtype=np.float64), 0
    if ref is None:
        if np.any(a > 0.0):
            ref = int(np.argmax(a))
        else:
            ref = 0
    ref = int(ref)
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.where(np.isfinite(v), v, -700.0)
    return v, ref


def from_v(v, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    return softmax_from_u(u)


def objgrad_v(v, beta, mask=None, reg_u=0.0, ref=0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    if n <= 0:
        return 1e100, np.array([], dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return 1e100, np.zeros(v.size, dtype=np.float64)

    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1

    u = np.empty(n, dtype=np.float64)
    u[ref] = 0.0
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]

    a = softmax_from_u(u)
    f, ga = lse_a(a, beta, mask)
    if not math.isfinite(f):
        f = 1e100
        ga = np.zeros(n, dtype=np.float64)

    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)

    gv = np.empty(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]

    if reg_u > 0.0:
        f += 0.5 * reg_u * float(np.dot(v, v))
        gv = gv + reg_u * v

    if not np.all(np.isfinite(gv)):
        gv = np.zeros(v.size, dtype=np.float64)
    return f, gv


def run_lbfegs(a0, beta, mask, maxfun=500, reg_u=1e-14, verbose=False):
    a = norm_sum1(a0)
    n = a.size
    if n <= 1:
        return a

    v0, ref = make_v(a)
    if not SCIPY_OK or maxfun <= 0:
        return a

    options = {
        "maxfun": int(maxfun),
        "maxiter": int(maxfun + 50),
        "ftol": 1e-13,
        "gtol": 1e-8,
        "disp": False
    }
    try:
        res = scipy.optimize.minimize(
            objgrad_v,
            v0,
            args=(beta, mask, reg_u, ref),
            method="L-BFGS-B",
            jac=True,
            options=options
        )
        if res is not None and hasattr(res, "x"):
            vx = np.asarray(res.x, dtype=np.float64)
            if vx.size == v0.size and np.all(np.isfinite(vx)):
                a_out = from_v(vx, ref)
                return norm_sum1(a_out)
    except Exception:
        pass
    return a


def lbfegs_adaptive(a_init, mask, budget, verbose=True):
    """Adaptive L-BFGS: starts with low beta (flat landscape), increases to
    sharpen the max objective. Uses active-set refinement."""
    a = norm_sum1(a_init)
    n = a.size
    size = 2 * n - 1
    best_a = a.copy()
    c = fft_conv(a)
    best_M = proxy_max(c, mask)

    t0 = time.monotonic()

    # Schedule: (beta, top_fraction, maxfun_fraction)
    schedule = [
        (8.0, None, 0.12),
        (20.0, None, 0.12),
        (50.0, 0.25, 0.12),
        (100.0, 0.15, 0.10),
        (200.0, 0.08, 0.10),
        (400.0, 0.05, 0.10),
        (800.0, 0.03, 0.10),
        (1500.0, 0.02, 0.10),
        (3000.0, 0.015, 0.10),
    ]

    for i, (beta, top_frac, fun_frac) in enumerate(schedule):
        if time.monotonic() - t0 > budget - 5:
            break

        # Refine active set
        if top_frac is None:
            msk = mask
        else:
            top_count = max(20, int(size * top_frac))
            msk = make_top_mask(a, top_count, mask)

        # Estimate cost for this configuration
        maxfun = 400
        try:
            t1 = time.monotonic()
            objgrad_v(np.zeros(n - 1), beta, msk, 0.0, 0)
            dt = time.monotonic() - t1
            if dt > 0:
                remaining = budget - (time.monotonic() - t0)
                per_run = int(max(60, min(800, remaining / max(dt * 1.5, 0.05))))
                maxfun = per_run
        except Exception:
            maxfun = 300

        a_new = run_lbfegs(a, beta, msk, maxfun=maxfun, verbose=False)
        c_new = fft_conv(a_new)
        m_new = proxy_max(c_new, mask)

        if math.isfinite(m_new):
            if m_new <= best_M * (1.0 + 5e-5) + 1e-18:
                a = a_new
            if m_new < best_M - 1e-15:
                best_M = m_new
                best_a = a_new.copy()
                if verbose:
                    print("    LB adaptive beta=" + str(beta) +
                          " n=" + str(n) +
                          " M_proxy=" + repr(m_new) +
                          " score=" + repr(2.0 * n * best_M))

    return best_a


def game_direction(f, active_idx, mw_iters=30, seed=0):
    f = norm_sum1(f)
    n = f.size
    size = 2 * n - 1
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K <= 0:
        return np.ones(n, dtype=np.float64) / float(n)

    rng = np.random.default_rng(seed)
    logits = rng.normal(0.0, 0.1, K)

    for _ in range(max(1, int(mw_iters))):
        pf = np.zeros(size, dtype=np.float64)
        pf[active_idx] = np.exp(logits - logits.max())
        pf_sum = pf.sum()
        if pf_sum < 1e-30:
            pf[active_idx] = 1.0 / K
        else:
            pf = pf / pf_sum

        pressure = cross_convolve_p_a(pf, f)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)

        pmx = float(np.max(pressure))
        pmn = float(np.min(pressure))
        ptp = pmx - pmn
        if ptp <= 1e-30:
            g = np.ones(n, dtype=np.float64) / float(n)
        else:
            scaled = (pressure - pmn) / ptp
            w = np.exp(-10.0 * scaled)
            w = np.where(np.isfinite(w), w, 0.0)
            w += 1e-12
            g = w / float(w.sum())
            g = 0.95 * g + 0.05 * (np.ones(n, dtype=np.float64) / float(n))
            g = g / float(g.sum())

        h = fft_conv2(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmac_ = float(np.max(ha))
        hmin = float(np.min(ha))
        hptp = hmac_ - hmin
        if hptp > 1e-30:
            eta = 15.0 / hptp
            logits += eta * (ha - float(np.mean(ha)))

    return g


def line_search_mix(a, g, cf, cg, fg, mask, current_M):
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    n = a.size

    base_alphas = [0.02, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.50, 0.70, 0.90, 1.0]
    best_alpha = 0.0
    best_m = current_M
    for alpha in base_alphas:
        one = 1.0 - alpha
        val = (one * one) * cf + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
        if mask is None:
            m = float(np.max(val))
        else:
            m = float(np.max(val[mask]))
        if math.isfinite(m) and m < best_m:
            best_m = m
            best_alpha = alpha

    if best_alpha <= 0.0:
        return a.copy(), best_m, 0.0

    new_a = (1.0 - best_alpha) * a + best_alpha * g
    new_a = np.maximum(new_a, 0.0)
    s = float(new_a.sum())
    if s < 1e-15:
        return a.copy(), best_m, 0.0
    new_a = new_a / s
    return new_a, best_m, best_alpha


def frank_wolfe_search(a0, budget, mask, active_max=3000, game_iters=30,
                       verbose=True):
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = proxy_max(c, mask)
    best_a = a.copy()
    best_M = M
    active = active_top(c, mask, active_max)

    no_improve = 0
    t = 0
    while time.monotonic() - t0 < time_limit - 3 and no_improve < 100:
        t += 1
        time_left = time_limit - (time.monotonic() - t0)
        if time_left < 15:
            break

        g = game_direction(a, active, mw_iters=game_iters, seed=5000 + t)

        if g is None:
            break

        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        new_a, new_M, alpha = line_search_mix(a, g, c, cg, fg, mask, M)

        if new_M < M - 1e-16 and alpha > 0:
            one = 1.0 - alpha
            c = (one * one) * c + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
            c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
            a = new_a
            M = new_M
            no_improve = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            active = active_top(c, mask, active_max)
            if verbose and (t == 1 or t % 15 == 0):
                print("    FW t=" + str(t) +
                      " score_proxy=" + str(round(2.0 * n * best_M, 6)) +
                      " alpha=" + str(round(alpha, 4)))
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                rng = np.random.default_rng(7777 + t)
                pert = rng.normal(0.0, 0.02, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-12)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = proxy_max(c, mask)
                active = active_top(c, mask, active_max)
                no_improve = 0

    return best_a


def apply_single_move(c, a, s, d, tmove):
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = a.size
    c_new = c.copy()
    c_new[d:d + n] += 2.0 * tmove * a
    c_new[s:s + n] -= 2.0 * tmove * a
    t2 = tmove * tmove
    if 2 * d < c_new.size:
        c_new[2 * d] += t2
    if 2 * s < c_new.size:
        c_new[2 * s] += t2
    if s + d < c_new.size:
        c_new[s + d] -= 2.0 * t2
    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
    return c_new


def coordinate_polish(a0, budget, mask, verbose=True):
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    c = fft_conv(a)
    M = proxy_max(c, mask)
    best_a = a.copy()
    best_M = M
    n = a.size
    stall = 0
    rnd = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit - 2 and stall < 150:
        rnd += 1
        if n <= 1:
            break

        top_k = min(60, 2 * n - 1)
        top_idx = active_top(c, mask, top_k)
        if top_idx.size == 0:
            break

        p_top = np.zeros(2 * n - 1, dtype=np.float64)
        p_top[top_idx] = 1.0 / float(top_idx.size)
        pressure = cross_convolve_p_a(p_top, a)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)

        src_count = min(10, n)
        dst_count = min(15, n)

        src_score = pressure * (1.0 + float(n) * a)
        src_candidates = np.argpartition(src_score, min(src_count - 1, n - 1))[:src_count]
        dst_score = pressure + 1e-5 * a
        dst_candidates = np.argpartition(dst_score, min(dst_count - 1, n - 1))[:dst_count]

        best_cand = None

        for srr in src_candidates:
            s = int(srr)
            if a[s] < 1e-12:
                continue
            for drr in dst_candidates:
                d = int(drr)
                if d == s:
                    continue
                for frac in (0.003, 0.007, 0.015, 0.03, 0.06):
                    tmove = a[s] * frac
                    if tmove <= 1e-16:
                        continue
                    tmove = min(tmove, a[s] * 0.95)
                    c_new = apply_single_move(c, a, s, d, tmove)
                    m_new = proxy_max(c_new, mask)
                    if m_new < M - 1e-18:
                        if best_cand is None or m_new < best_cand[0]:
                            best_cand = (m_new, s, d, tmove)

        # Random exploratory moves
        for _ in range(30):
            s = int(rng.integers(n))
            d = int(rng.integers(n))
            if s == d or a[s] < 1e-12:
                continue
            frac = float(rng.uniform(0.002, 0.04))
            tmove = a[s] * frac
            tmove = min(tmove, a[s] * 0.95)
            if tmove <= 1e-16:
                continue
            c_new = apply_single_move(c, a, s, d, tmove)
            m_new = proxy_max(c_new, mask)
            if m_new < M - 1e-18:
                if best_cand is None or m_new < best_cand[0]:
                    best_cand = (m_new, s, d, tmove)

        if best_cand is not None:
            m_new, s, d, tmove = best_cand
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = fft_conv(a)
            M = proxy_max(c, mask)
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and (rnd <= 3 or rnd % 30 == 0):
                print("    Coord rnd=" + str(rnd) +
                      " score_proxy=" + str(round(2.0 * n * best_M, 6)))
        else:
            stall += 1
            if stall % 15 == 14 and stall < 150:
                for _ in range(200):
                    s = int(rng.integers(n))
                    d = int(rng.integers(n))
                    if s == d or a[s] < 1e-12:
                        continue
                    frac = float(rng.uniform(0.001, 0.08))
                    tmove = a[s] * frac
                    tmove = min(tmove, a[s] * 0.95)
                    c_new = apply_single_move(c, a, s, d, tmove)
                    m_new = proxy_max(c_new, mask)
                    if m_new < M - 1e-17:
                        a[s] -= tmove
                        a[d] += tmove
                        a = np.maximum(a, 0.0)
                        a = a / float(a.sum())
                        c = fft_conv(a)
                        M = proxy_max(c, mask)
                        stall = 0
                        break

    return best_a
