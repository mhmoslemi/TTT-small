def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_grad_smoothed(y, beta, top_k):

            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 80 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.01 / (1.0 + 0.03 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 8.0)
            s_prev = None
            y_prev = None
            bb_step = 1.0 / np.sqrt(n)

    J = 2.0 * n * best_max_b
    if verbose:
        elapsed = time.monotonic() - t0
        print("    FISTA done (n={}, {}s): J={}".format(n, round(elapsed, 1), round(J, 8)))
    return best_a, best_max_b


def lp_fw_optimize(a0, time_limit, solver, initial_K=300, max_K=2000,
                   add_per_iter=15, verbose=False):
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    ac = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 600

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        M_curr = best_m

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            ac_now = fast_autocorr(y)
            margins = np.array([M_curr - ac_now[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                ac_from = fast_autocorr(y)
                active_set = set(int(k) for k in np.argsort(ac_from)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5), 1.0]
        alphas = [max(0.01, min(1.0, a)) for a in alphas]

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        M_new = fast_autocorr(y_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = y_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter={} ({}s): J={} |S|={}".format(t, round(elapsed, 1), round(J, 8), len(active_set)))

    return best_a, best_m


def peak_targeted_search(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:40]
        peak_vals = ac[top_peaks]
        improved = False

        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(top_peaks[rng.integers(min(30, len(top_peaks)))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue
            src = int(idx[np.argmax(contrib)])

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:20]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src or a[src] < 1e-12:
                continue

            fracs = [0.002, 0.005, 0.01, 0.02, 0.05]
            frac = float(rng.choice(fracs))
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-14:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                ac = fast_autocorr(a)
                top_peaks = np.argsort(ac)[::-1][:40]

        if not improved:
            stall += 1
            noise_scale = min(0.05, 0.005 * (1 + stall // 100))
            for _ in range(5):
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                if i1 != i2 and a[i1] > 1e-10:
                    amt = a[i1] * noise_scale
                    a[i1] -= amt
                    a[i2] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print("    Peak search done (n={}, {}s): J={}".format(n, round(elapsed, 1), round(J, 8)))
    return best_a, best_m


def cyclic_shift_best(a0):
    n = len(a0)
    if n <= 1:
        return a0.copy(), fast_autocorr(a0).max()
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    A_base = np.fft.rfft(a0, fft_size)
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def coordinate_descent_polish(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)

    coord_order = rng.permutation(n)
    stall = 0
    max_stall = n * 2

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        stall = 0
        for ii in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(coord_order[ii])
            if a[i] < 1e-15:
                stall += max(1, n // 100)
                continue

            ac = fast_autocorr(a)
            M_old = ac.max()

            best_val = a[i]
            best_M = M_old

            vals_to_try = [a[i] * 0.5, a[i] * 0.7, a[i] * 1.3, a[i] * 1.5, 0.0]
            for v in vals_to_try:
                a_trial = a.copy()
                delta = v - a[i]
                a_trial[i] = v
                if delta != 0:
                    j = int(coord_order[(ii + n // 2) % n])
                    a_trial[j] = max(0.0, a_trial[j] - delta)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_M - 1e-14:
                    best_M = M_trial
                    best_val = v

            if best_M < M_old - 1e-14:
                a[i] = best_val
                delta = best_val - a[i] if False else 0
                a = a  # already set
                best_m = best_M
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        coord_order = rng.permutation(n)

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print("    Coord descent done (n={}, {}s): J={}".format(n, round(elapsed, 1), round(J, 8)))
    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {} (n={}, t={}s)".format(label, repr(round(sc, 8)), len(a), round(elapsed, 1)))
    return best_seq, best_score


def make_initializer(n, shape='sin2', seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if shape == 'sin2':
        f = np.sin(np.pi * x) ** 2
    elif shape == 'cos':
        f = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif shape == 'raised':
        f = 0.5 * (1 + np.cos(2 * np.pi * x)) ** 2
    elif shape == 'super':
        f = np.sin(np.pi * x) ** 2 + 0.1 * np.sin(2 * np.pi * x) ** 2
    elif shape == 'uniform':
        f = np.ones(n)
    elif shape == 'random_smooth':
        f = np.ones(n)
        for _ in range(3):
            freq = rng.uniform(1, 4)
            phase = rng.uniform(0, 2 * np.pi)
            f = f + 0.1 * np.sin(2 * np.pi * freq * x + phase)
        f = np.maximum(f, 0.01)
    else:
        f = np.sin(np.pi * x) ** 2
    return norm_sum1(f)
