    col_list = []
    val_list = []
    for jj in range(K):

        return sp.csc_matrix((K, n))

    cols = np.concatenate(col_list)

def spectral_polish(a0, time_limit, verbose=False):
    """Polish using alternating projections in frequency domain."""
    n = len(a0)
    a = normalize_seq(a0)

    nbins = fft_size // 2 + 1
    
    while time.monotonic() - t0 < time_limit:
        # Compute DFT
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # Project in frequency: reduce peaks in power spectrum
        # The autocorrelation is the inverse DFT of power
        # max autocorrelation ≈ max of inverse DFT of power
        # To reduce max autocorrelation, reduce peak power
        max_power = power.max()
        target_power = max_power * 0.95  # reduce peak by 5%
        
        # Find the frequency bins with highest power
        peak_bins = np.where(power > target_power)[0]
        
        if len(peak_bins) == 0:
            break
        
        # Reduce power at peak bins (proportionally)
        new_power = power.copy()
        for kb in peak_bins:
            reduction = (power[kb] - target_power) * 0.5
            new_power[kb] = target_power
        
        # Reconstruct amplitude with modified power
        # Keep phase, adjust magnitude
        new_mag = np.sqrt(new_power)
        phase = np.angle(A) if hasattr(np, 'angle') else np.arctan2(A.imag, A.real)
        # A is real FFT, so it's complex
        A_new = new_mag * np.exp(1j * np.angle(A))
        
        # Convert back to time domain
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        M_new = float(fast_convolve(a_new).max())
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Revert and try less aggressive reduction
            target_power = max_power * 0.99
            new_power = power.copy()
            for kb in peak_bins:
                new_power[kb] = target_power
            new_mag = np.sqrt(new_power)
            A_new = new_mag * np.exp(1j * np.angle(A))
            a_new = np.fft.irfft(A_new, fft_size)[:n]

            if s > 1e-15:

    return best_a, best_m

def get_initial(n, mode='sine'):
    """Generate an initial sequence of length n."""
    x = np.linspace(0, 1, n)
    if mode == 'sine':
        return normalize_seq(np.sin(np.pi * x))
    elif mode == 'gaussian':
        return normalize_seq(np.exp(-20 * (x - 0.5) ** 2))
    elif mode == 'cosine':
        return normalize_seq(np.maximum(np.cos(np.pi * (x - 0.5)), 0))
    elif mode == 'triangular':
        return normalize_seq(np.minimum(x, 1 - x) + 0.1)
    elif mode == 'uniform':
        return normalize_seq(np.ones(n))
    else:
        return normalize_seq(np.sin(np.pi * x))

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def evaluate_safe(a):
    try:
        a_list = [max(0.0, min(1000.0, float(x))) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
