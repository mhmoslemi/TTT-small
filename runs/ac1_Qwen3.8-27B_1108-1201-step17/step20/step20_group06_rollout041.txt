    s = a.sum()
    if s < 1e-15:

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

    n = len(f)
    size = 2 * n - 1

    G = np.fft.rfft(np.asarray(g, dtype=np.float64), fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]

We want to solve:
minimize M
subject to:

  g_i >= 0 for all i

where (A_k @ g) = sum_i f(k-i) * g_i

In standard LP form (minimize c^T x subject to constraints):
- Decision variables: x = [g_0, ..., g_{n-1}, M] (n+1 variables)
- Objective: minimize M → c = [0, 0, ..., 0, 1]
- Equality constraint: sum g_i = 1 → A_eq = [1, 1, ..., 1, 0], b_eq = [1]
- Inequality constraints: for each k in S: sum_i f(k-i)*g_i - M <= 0
  → A_ub[k, :] = [f(k-0), f(k-1), ..., f(k-(n-1)), -1] (with f(j)=0 if j not in [0,n-1])
  → b_ub[k] = 0
- Bounds: 0 <= g_i (free for M)

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using enhanced LP-Frank-Wolfe
with scipy HiGHS solver, adaptive active set management, combined with
Nesterov accelerated gradient on smoothed max, spectral refinement, and
targeted coordinate descent. Key improvements over previous: direct HiGHS LP
solves (avoiding cvxpy overhead), larger initial active sets, smarter
constraint management, and multi-phase optimization with sequence extension.

import numpy as np
import time
from scipy.optimize import linprog

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def conv_with(f, g):
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(np.asarray(f, dtype=np.float64), fft_size)
    G = np.fft.rfft(np.asarray(g, dtype=np.float64), fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def build_constraint_matrix(f, active_list, n):
    """Build the LP constraint matrix rows for given active lags."""
    K = len(active_list)
    A = np.zeros((K, n))
    for jj, k in enumerate(active_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            A[jj, idx] = f[k_int - idx]
    return A

def solve_lp_subproblem(f, active_set, n):
    """Solve LP: min M s.t. sum(g)=1, g>=0, (f*g)[k]<=M for k in active_set."""
    K = len(active_set)
    if K == 0:
        return np.ones(n) / n
    
    active_list = sorted(active_set)
    coeffs = build_constraint_matrix(f, active_list, n)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    A_ub = np.hstack([coeffs, -np.ones((K, 1))])
    b_ub = np.zeros(K)
    
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if not result.success:
        return None
    
    g_val = np.maximum(result.x[:n], 0.0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return None
    return g_val / g_sum

def lp_fw_enhanced(a0, time_limit, initial_K=1000, max_K=5000,
                   add_per_iter=80, verbose=True, seed=42):
    """Enhanced LP-Frank-Wolfe using HiGHS solver."""
    n = len(a0)
    f = norm_sum1(a0)
    best_f = f.copy()
    best_m = fast_autocorr(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    ac = fast_autocorr(f)
    active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])
    
    no_improve = 0
    max_no_improve = 1000
    it = 0
    alpha_base = 0.1
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if no_improve > max_no_improve:
            if verbose and it % 100 == 0:
                print("    LP-FW+ stuck at it=" + str(it))
            break
        
        f_curr = f
        M_curr = fast_autocorr(f_curr).max()
        
        g_dir = solve_lp_subproblem(f_curr, active_set, n)
        if g_dir is None:
            if len(active_set) > initial_K // 2:
                ac_curr = fast_autocorr(f_curr)
                active_set = set(int(k) for k in np.argsort(ac_curr)[::-1][:initial_K // 2])
            continue
        
        fg_conv = conv_with(f_curr, g_dir)
        size = 2 * n - 1
        violations = np.where(fg_conv > M_curr + 1e-12)[0]
        if len(violations) > 0:
            viol_amt = fg_conv[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))
        
        if len(active_set) > max_K:
            ac_curr = fast_autocorr(f_curr)
            margins = np.array([M_curr - ac_curr[k] for k in active_set])
            keep_idx = np.argsort(margins)[:max_K]
            active_list_sorted = sorted(active_set)
            active_set = set(int(active_list_sorted[i]) for i in keep_idx)
        
        best_alpha = 2.0 / (it + 2.0)
        best_trial_M = M_curr
        alphas = [best_alpha, best_alpha * 0.5, best_alpha * 2.0,
                  1.0 / (it + 1), 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
        for alpha in alphas:
            alpha = max(0.001, min(1.0, alpha))
            trial = (1.0 - alpha) * f_curr + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue

                best_alpha = alpha
        
        f = (1.0 - best_alpha) * f_curr + best_alpha * g_dir

            f = f_curr.copy()
        else:
            f = f / s_f
        
        M_new = fast_autocorr(f).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW+ it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)) + " alpha=" + repr(round(best_alpha, 4)))
    
    return best_f, best_m

def spectral_flatten(a0, time_limit, verbose=True, seed=777):
    """Spectral domain peak reduction."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    alpha = 0.03
    
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        target = power[1:].mean() if len(power) > 1 else power[0]
        
        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.0:
                red = 1.0 - alpha * min(ratio - 1.0, 3.0)
                A_new[k] *= max(0.2, np.sqrt(max(red, 0.0)))
            elif ratio < 0.3:
                inc = 1.0 + alpha * 0.5 * (0.3 - ratio)
                A_new[k] *= min(3.0, np.sqrt(inc))
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 100 == 0:
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.03
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial
        
        if it % 300 == 0:
            alpha = max(0.003, alpha * 0.93)
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
    
    return best_a, best_m

def targeted_local_search(a0, time_limit, verbose=True, seed=321):

            if js > je:
                continue
            
            idx = np.arange(js, je + 1)

rch (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score
