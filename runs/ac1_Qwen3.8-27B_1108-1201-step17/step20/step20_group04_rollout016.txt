def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

    if s < 1e-15:

def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def active_set_lp(a, n_constraints=200, blend=0.5):
    """Solve LP with top constraints, return blended solution."""
    n = len(a)
    c = fast_autocorr(a)
    top_idx = np.argsort(c)[::-1][:n_constraints]
    
    # Build constraint matrix
    m = len(top_idx)
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for ii, k in enumerate(top_idx):
        k = int(k)
        # For each j: A_ub[ii, j] = 2 * a[k-j] if 0 <= k-j < n
        # Vectorized:
        j_vals = np.arange(n)
        x_vals = k - j_vals
        mask = (x_vals >= 0) & (x_vals < n)
        A_ub[ii, mask] = 2 * a[x_vals[mask]]
        A_ub[ii, n] = -1
        b_ub[ii] = c[k]
    
    # Solve LP
    from scipy.optimize import linprog
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        if result.success:

    return None

def power_sum_subgrad(a, beta, top_k=None):
    """Compute subgradient of (1/beta)*log(sum top_k exp(beta*c_k)) w.r.t. a."""
    n = len(a)
    c = fast_autocorr(a)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    
    # Gradient: (2/beta) * conv(p, a_rev)[n-1 : 2n-1]
    fft_size = 1 << (2 * n).bit_length()
    P = np.fft.rfft(p, fft_size)
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, fft_size)
    Conv = np.fft.irfft(P * A_rev, fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad

        y = np.maximum(y, 1e-15)

        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        grad = power_sum_subgrad(y, beta, top_k)
        eta = eta_base / max(1.0, beta * 2.0)
        eta = min(eta, 0.5)
        
        z = y - eta * grad
        x_new = simplex_project(z)
        
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        if t % 20 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.02 / (1.0 + 0.02 * (t // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.3 * anti + 0.7 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("  NF restart t=" + str(t) + " J=" + repr(round(J, 6)))
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("  NF t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(round(J, 6)) + " beta=" + str(round(beta, 1)))
    
    return best_a, best_max_b

def lp_optimize(a0, time_limit, n_constraints=200, blend=0.4, verbose=True):
    """Iterative active-set LP optimization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 50
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        # Solve LP
        a_lp = active_set_lp(a, n_constraints=n_constraints, blend=blend)
        
        if a_lp is not None:
            M_lp = fast_autocorr(a_lp).max()
            if M_lp < best_M - 1e-15:
                best_M = M_lp
                best_a = a_lp.copy()
                a = a_lp.copy()
                stall = 0
            else:
                stall += 1
                # Try with more constraints or different blend
                if stall % 10 == 0:
                    a_lp2 = active_set_lp(a, n_constraints=min(n_constraints + 100, 500), blend=0.3)
                    if a_lp2 is not None:
                        M_lp2 = fast_autocorr(a_lp2).max()
                        if M_lp2 < best_M - 1e-15:
                            best_M = M_lp2
                            best_a = a_lp2.copy()
                            a = a_lp2.copy()
                            stall = 0
        else:
            stall += 1
        
        if verbose and (stall % 10 == 0 or stall == 0):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print("  LP (" + str(round(elapsed, 1)) + "s) J=" + repr(round(J, 6)) + " stall=" + str(stall))
    
    return best_a, best_M

def peak_surgery_v2(a0, time_limit, verbose=True, seed=77):
    """Improved peak surgery with better destination selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 2000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find top peaks
        n_top = min(50, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]
        
        # Compute global "pressure" map
        pressure = np.zeros(n)
        for k in top_idx[:20]:
            k_i = int(k)
            s = max(0, k_i - n + 1)
            e = min(n - 1, k_i)
            idx = np.arange(s, e + 1)
            pressure[idx] += a[k_i - idx]
            pressure[k_i - idx] += a[idx]
        
        improved = False
        for k in top_idx[:10]:
            if improved:
                break
            k_i = int(k)
            s = max(0, k_i - n + 1)
            e = min(n - 1, k_i)
            idx = np.arange(s, e + 1)
            products = a[idx] * a[k_i - idx]
            
            if products.max() < 1e-20:
                continue
            
            src_i = int(idx[np.argmax(products)])
            partner = k_i - src_i
            if partner < 0 or partner >= n:
                continue
            
            reduce_i = src_i if a[src_i] >= a[partner] else partner
            if a[reduce_i] < 1e-14:
                continue
            
            # Find safe destinations (low pressure, not in any peak region)
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_idx[:15]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 100:
                candidates = np.arange(n)
            mask = candidates != reduce_i
            candidates = candidates[mask]
            
            if len(candidates) == 0:
                continue
            
            cand_pressure = pressure[candidates]
            n_try = min(10, len(candidates))
            try_dsts = candidates[np.argsort(cand_pressure)[:n_try]]
            
            for dst in try_dsts:
                dst_i = int(dst)
                for frac in [0.003, 0.005, 0.01, 0.02, 0.03]:
                    amt = a[reduce_i] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved and no_improve > 200:
            # Random perturbation
            for _ in range(20):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
        
        if verbose and no_improve % 500 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  PS2 (" + str(round(elapsed, 1)) + "s) J=" + repr(round(J, 6)) + " stall=" + str(no_improve))
    
    return best_a, best_m
