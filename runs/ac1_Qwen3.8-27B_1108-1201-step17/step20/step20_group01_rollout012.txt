"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full softmax-smoothed gradient with Adam optimizer and simplex projection
2. Temperature annealing from smooth to sharp max approximation
3. LP-based Frank-Wolfe with warm start for local refinement
4. Fine-grained coordinate descent with adaptive pressure-based moves
5. Multi-start exploration and length-adaptive refinement
"""

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None

def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  Solver: ECOS (fallback)")
    return _MODULE_SOLVER

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def project_simplex(x):
    n = len(x)
    if x.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1.0) / (rho[-1] + 1.0)
    return np.maximum(x - theta, 0.0)

def full_softmax_gradient(a, tau, ac):
    n = len(a)
    size = 2 * n - 1
    shifted = (ac - ac.max()) / tau
    max_sh = shifted.max()
    exp_vals = np.exp(shifted - max_sh)
    w = exp_vals / exp_vals.sum()
    fft_size = 1 << (3 * n - 3 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w[::-1].copy(), fft_size)
    conv = np.fft.irfft(A * W, fft_size)[:3 * n - 2]
    grad = 2.0 * conv[n - 1:2 * n - 1:-1]
    return grad, w

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_matrix(a, active_list, n):
    K = len(active_list)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj in range(K):
        ki = int(active_list[jj])
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[ki - idx]
    return coeffs

def adam_smoothed(a0, time_limit, tau_init=None, tau_end=0.001, lr_init=0.02,
                   verbose=True, restart_every=3000, noise_scale=0.01):
    n = len(a0)
    x = normalize(a0).copy()
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    ac = fast_autocorr(x)
    if tau_init is None:
        tau_init = max(0.01, (ac.max() - ac.min()) * 0.5)
    
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    iter_count = 0
    stall = 0
    max_stall = restart_every
    lr = lr_init
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            x_r = best_a + noise_scale * rng.standard_normal(n) * best_a
            x = project_simplex(np.maximum(x_r, 1e-12))
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            iter_count = 0
            stall = 0
            tau = tau_init * 0.5
            if verbose:
                print("    Adam restart at t={} ({:.1f}s) J={:.7f}".format(
                    t, time.monotonic() - t0, best_val))
        
        progress = min(1.0, t / (time_limit * 2.0))
        tau = tau_init * (tau_end / tau_init) ** progress
        tau = max(tau, 1e-6)
        
        ac = fast_autocorr(x)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr
        
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        grad, w = full_softmax_gradient(x, tau, ac)
        iter_count += 1
        lr_t = lr_init / (1.0 + 0.001 * iter_count)
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** iter_count)
        v_hat = v / (1.0 - beta2 ** iter_count)
        x_new = project_simplex(x - lr_t * m_hat / (np.sqrt(v_hat) + eps))
        
        if not np.all(np.isfinite(x_new)) or x_new.sum() < 1e-15:
            x_new = x.copy()
        
        x = x_new
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    Adam t={} ({:.1f}s) J={:.7f} tau={:.4f} stall={}".format(
                t, elapsed, best_val, tau, stall))
    
    return best_a, best_val

def lp_fw(a0, time_limit, solver, init_K=200, max_K=4000, add_per=25, verbose=True):
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:init_K])
    no_improve = 0
    stall_limit = 2000
    prev_g = None

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t % 20 == 0 or t == 1:
            top_lags = np.argsort(ac)[::-1]
            target_size = min(max_K, init_K + t * 3)
            for k in top_lags[:target_size]:
                active_set.add(int(k))
            if len(active_set) > max_K:
                clist = sorted(active_set)
                margins = np.array([M_curr - ac[k] for k in clist])
                keep = np.argsort(margins)[:max_K]
                active_set = set(int(clist[i]) for i in keep)
        active_set.add(int(np.argmax(ac)))

        clist = sorted(active_set)
        K = len(clist)
        if K == 0:
            break

        try:
            coeffs = build_lp_matrix(y, clist, n)
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
            if prev_g is not None:
                constraints.append(cp.sum((g_var - prev_g) ** 2) <= 0.1 * n)
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            prob.solve(solver=solver, verbose=False)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                raise Exception("LP status: " + str(prob.status))
        except Exception:
            if prev_g is not None:
                constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
                prob = cp.Problem(cp.Minimize(M_var), constraints)
                prob.solve(solver=solver, verbose=False)
                if prob.status not in ('optimal', 'optimal_inaccurate'):
                    if len(active_set) > init_K // 2:
                        clist = sorted(active_set)
                        active_set = set(int(clist[i]) for i in np.argsort([M_curr - ac[k] for k in clist])[:init_K])
                    continue
                prev_g = None
            else:
                if len(active_set) > init_K // 2:
                    clist = sorted(active_set)
                    active_set = set(int(clist[i]) for i in np.argsort([M_curr - ac[k] for k in clist])[:init_K])
                continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        prev_g = g_dir.copy()

        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]
            for k in top_v:
                active_set.add(int(k))

        best_alpha = 2.0 / (t + 2.0)
        trial = np.maximum((1 - best_alpha) * y + best_alpha * g_dir, 0.0)
        ts = trial.sum()
        best_trial = fast_autocorr(trial / ts).max() if ts > 1e-15 else 1e10

        for alpha in [1.0/(t+1), 0.2, 0.4, 0.6, 0.8, 0.9, 1.0]:
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            Mt = fast_autocorr(trial / ts).max()
            if Mt < best_trial - 1e-15:
                best_trial = Mt
                best_alpha = alpha

        lo = max(0.0, best_alpha - 0.1)
        hi = min(1.0, best_alpha + 0.1)
        for _ in range(5):
            mid1 = lo + (hi - lo) / 3.0
            mid2 = hi - (hi - lo) / 3.0
            t1 = np.maximum((1 - mid1) * y + mid1 * g_dir, 0.0)
            t2 = np.maximum((1 - mid2) * y + mid2 * g_dir, 0.0)
            s1, s2 = t1.sum(), t2.sum()
            m1 = fast_autocorr(t1 / s1).max() if s1 > 1e-15 else 1e10
            m2 = fast_autocorr(t2 / s2).max() if s2 > 1e-15 else 1e10
            if m1 < m2:
                hi = mid2
            else:
                lo = mid1
        best_alpha = (lo + hi) / 2.0

        if t <= 2:
            y = g_dir.copy()
        else:
            y = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t={:5d} ({:.1f}s) J={:.7f} |S|={}".format(
                t, elapsed, 2 * n * best_m, len(active_set)))

    return best_a, 2.0 * n * best_m

def fine_coord_descent(a0, time_limit, n_targets=20, seed=42, verbose=True):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 5000
    fracs = [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(n_targets, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            ws = max(0, ki - (n - 1))
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-25:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-15:
                continue

            pr = np.zeros(n, dtype=np.float64)
            for kk in top_k[:10]:
                kkk = int(kk)
                wss = max(0, kkk - (n - 1))
                wee = min(n - 1, kkk)
                if wss <= wee:
                    iidx = np.arange(wss, wee + 1)
                    pr[iidx] += a[kkk - iidx]
            pr[src] = 1e10
            quiet_positions = np.argsort(pr)[:50]

            best_move = None
            best_move_val = best_m
            for dst in quiet_positions:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-20:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_move_val - 1e-16:
                        best_move_val = Mt
                        best_move = (src, dst, amt)
            
            if best_move is not None:
                src_i, dst_i, amt_i = best_move
                a[src_i] -= amt_i
                a[dst_i] += amt_i
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_move_val
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1
            if stall > 200:
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st < 1e-15:
                        continue
                    a = a / st
                    Mn = fast_autocorr(a).max()
                    if Mn < best_m:
                        best_m = Mn
                        best_a = a.copy()
                        stall = 0
                        break

        if verbose and t0 > 0 and (time.monotonic() - t0) % 5 < 0.01:
            pass

    if verbose:
        print("    CoordDesc ({:.1f}s) J={:.7f} stall={}".format(
            time.monotonic() - t0, 2 * n * best_m, stall))
    return best_a, 2.0 * n * best_m

            best_seq = a_out
            print("  P3 improved: {:.7f}".format(best_score))

    # Phase 4: Second Adam pass (finer temperature)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P4: Adam Fine (n={}) ===".format(len(best_seq)))
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        a_out, v_out = adam_smoothed(normalize(best_seq), t4,
                                     tau_init=0.01, tau_end=0.0001,
                                     lr_init=0.01, verbose=True,
                                     restart_every=2000, noise_scale=0.005)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P4 improved: {:.7f}".format(best_score))

    # Phase 5: Try slightly longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 1500
    if remaining > 120:
        print("\n=== P5: Long n={} ===".format(n_long))
        t5 = min(remaining * 0.20, 300.0)
        if t5 > 30:
            a_long = interpolate_seq(best_seq, n_long)
            a_out, v_out = adam_smoothed(a_long, t5 * 0.5,
                                         tau_init=0.02, tau_end=0.001,
                                         lr_init=0.02, verbose=False)
            a_out2, v_out2 = fine_coord_descent(a_out, t5 * 0.5,
                                                n_targets=15, seed=77, verbose=False)
            sc1 = eval_safe(a_out)
            sc2 = eval_safe(a_out2)
            sc_best = min(sc1, sc2)
            if np.isfinite(sc_best) and sc_best < best_score:
                best_a = a_out if sc1 < sc2 else a_out2
                best_score = sc_best
                best_seq = best_a
                print("  P5 improved: {:.7f}".format(best_score))

    # Phase 6: Final LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P6: Final LP-FW (n={}) ===".format(n_cur))
    t6 = min(remaining * 0.20, 300.0)
    if t6 > 30:
        if n_cur <= 13000:
            a_out, v_out = lp_fw(normalize(best_seq), t6, solver,
                                 init_K=400, max_K=6000, add_per=35, verbose=True)
        else:
            a_out, v_out = adam_smoothed(normalize(best_seq), t6,
                                         tau_init=0.005, tau_end=0.0001,
                                         lr_init=0.01, verbose=False)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P6 improved: {:.7f}".format(best_score))

    # Phase 7: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P7: Final Coord (n={}) ===".format(len(best_seq)))
    t7 = max(remaining * 0.30, 30.0)
    if t7 > 20:
        a_out, v_out = fine_coord_descent(normalize(best_seq), t7,
                                          n_targets=30, seed=999, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P7 improved: {:.7f}".format(best_score))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.7f} n={} t={:.1f}s".format(best_score, len(best_seq), elapsed))
    else:
        print("\nERROR fallback")
        n_fb = 10000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)

    return [float(x) for x in best_seq]
