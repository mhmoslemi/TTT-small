    # conv[m] = sum_k P_full[k] * a[m-k] for valid indices
    # We want for each j: sum_k P_full[k] * a[k-j]
    # Let m = k-j, then k = m+j: sum_m P_full[m+j] * a[m]
    # This is (a * P_full_shifted)[j] where P_full_shifted[m] = P_full[m+j]... 
    
    # Actually, let me use: sum_k P_full[k] * a[k-j] 
    # = sum_k P_full[k] * a[k-j]
    # Let's define R[j] = sum_k P_full[k] * a[k-j]
    # R[j] = (P_full_reversed * a)[j + n - 1] ... 
    
    # Simpler: R[j] = sum_k P_full[k] * a[k-j]
    # This is the cross-correlation of P_full and a:
    # R = np.correlate(P_full, a, mode='full')[n-1:]... not quite
    
    # Let me just do it with the FFT approach from the previous code:
    a_rev = a[::-1]
    fft_size2 = 1 << (n + size - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size2)
    Pr = np.fft.rfft(P_full, fft_size2)
    cov = np.fft.irfft(Ar * Pr, fft_size2)
    # cov[j] = sum_i a_rev[i] * P_full[j-i] = sum_i a[n-1-i] * P_full[j-i]
    # Let m = n-1-i: cov[j] = sum_m a[m] * P_full[j-(n-1-m)] = sum_m a[m] * P_full[j-n+1+m]
    # We want R[j] = sum_k P_full[k] * a[k-j]
    # Let m = k-j: R[j] = sum_m P_full[m+j] * a[m]
    # cov[n-1+j] = sum_m a[m] * P_full[n-1+j-n+1+m] = sum_m a[m] * P_full[j+m]
    # That's R[j]! (since R[j] = sum_m a[m] * P_full[m+j])
    
    R = cov[n-1:n-1+n]
    
    grad = (R + a) / beta
    return grad

def nesterov_subgradient(a0, time_limit, beta_start=20.0, beta_end=800.0,
                         beta_growth=1.004, verbose=True):
    """Nesterov-accelerated subgradient with heavy-ball momentum."""
    n = len(a0)
    y = norm_sum1(a0)  # Nesterov point
    x = y.copy()       # actual point
    best_a = y.copy()
    b = fast_autocorr(y)
    best_m = b.max()
    t0 = time.monotonic()
    beta = beta_start
    momentum = 0.85
    t = 0
    rng = np.random.default_rng(42)
    top_k = max(50, int(n * 0.05))
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(y)
        m_curr = b.max()
        
        if m_curr < best_m - 1e-15:
            best_m = m_curr
            best_a = x.copy()  # Note: x is the actual point, y is Nesterov
            stall = 0
        else:
            stall += 1
        
        # Compute gradient at y
        grad = compute_gradient_smoothed(y, b, beta, top_k)
        
        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        # Step size: eta = 1 / (grad_norm * sqrt(t+1)) approximately
        eta = 1.0 / (grad_norm + 1e-10)
        eta = min(eta * 0.3, 0.03)
        eta = max(eta, 1e-7)
        
        # Nesterov update
        mu = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (1.0 - 1.0/mu) / (1.0 + 1.0/(t+1))
        # theta ≈ 1 - 2/(t+3) for large t
        theta = min(0.95, max(0.5, 1.0 - 2.0/(t+3.0)))
        
        x_new = simplex_project(y - eta * grad)
        y = (1.0 - theta) * x_new + theta * y  # Nesterov extrapolation
        
        # Ensure y stays valid
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = best_a.copy()
        
        x = x_new
        
        # Adaptive beta (increase slowly for more accurate gradient)
        if t % 100 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        # Restart if stalled
        if stall > 800:
            stall = 0
            noise_scale = min(0.02, 0.003 * (stall + 1))
            # Structured perturbation: perturb more where the peaks are
            b_full = fast_autocorr(best_a)
            peak_idx = int(np.argmax(b_full))
            # Perturb coordinates involved in the top peak
            pressure = np.zeros(n)
            i_start = max(0, peak_idx - n + 1)
            i_end = min(n - 1, peak_idx)
            idx = np.arange(i_start, i_end + 1)
            pressure[idx] += best_a[peak_idx - idx]
            pressure[peak_idx - idx] += best_a[idx]
            
            noise = rng.normal(0, noise_scale, n) * (0.5 + pressure / max(1e-15, pressure.max()))
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            y = x.copy()
            beta = max(beta * 0.7, 10.0)
            
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    Nesterov restart t={t} ({elapsed:.1f}s): J={J:.6f}")
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Nesterov t={t} ({elapsed:.1f}s): J={J:.6f} beta={beta:.1f} stall={stall}")
    
    return best_a, best_m

def peak_shaving_direct(a0, time_limit, verbose=True):
    """Direct peak shaving: find top peak, move mass from contributing coords to safe coords."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 2000:
        b = fast_autocorr(a)
        max_idx = int(np.argmax(b))
        
        # Identify contributing coordinates for the max peak
        i_start = max(0, max_idx - n + 1)
        i_end = min(n - 1, max_idx)
        if i_start > i_end:
            break
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[max_idx - idx]
        
        # Also identify second and third peaks
        sorted_all = np.argsort(b)[::-1]
        peak2 = int(sorted_all[1]) if len(sorted_all) > 1 else -1
        peak3 = int(sorted_all[2]) if len(sorted_all) > 2 else -1
        
        improved = False
        
        # For the top few contributing positions, try to move mass
        top_contrib = np.argsort(contrib)[::-1][:10]
        
        for tc in top_contrib[:5]:
            src = int(idx[tc])
            partner = max_idx - src
            if partner < 0 or partner >= n:
                continue
            
            src_val = a[src]
            if src_val < 1e-14:
                continue
            
            # Find safe destination: low contribution to all top peaks
            danger = np.zeros(n)
            for pk in [max_idx, peak2, peak3]:
                if pk < 0:
                    continue
                ps_start = max(0, pk - n + 1)
                ps_end = min(n - 1, pk)
                if ps_start <= ps_end:
                    pidx = np.arange(ps_start, ps_end + 1)
                    danger[pidx] += a[pk - pidx] * a[pidx]
            
            # Exclude source and positions already high
            danger[src] = np.inf
            # Also exclude positions that would create new peaks
            if 0 <= max_idx - (max_idx - src) < n:  # partner
                danger[partner] = min(danger[partner], danger[src] - 1)  # allow but prefer others
            
            dst_candidates = np.where(danger < np.min(danger[np.isfinite(danger)]) * 2)[0]
            if len(dst_candidates) < 10:
                dst_candidates = np.argsort(danger)[:50]
            
            for dst in dst_candidates[:20]:
                if dst == src:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        a_trial[src] = 0
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m * (1 - 1e-10):
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if improved:
                stall = 0
                break
        
        if not improved:
            stall += 1
            # Random perturbation every 100 stalls
            if stall % 100 == 0:
                noise = rng.normal(0, 0.005, n) * best_a
                x_r = best_a + noise
                x_r = np.maximum(x_r, 1e-12)
                x_r = x_r / x_r.sum()
                m_r = fast_autocorr(x_r).max()
                if m_r < best_m:
                    best_m = m_r
                    best_a = x_r.copy()
                a = x_r.copy()
            
            if verbose and stall % 500 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    PeakShave ({elapsed:.1f}s): J={J:.6f} stall={stall}")
    
    return best_a, best_m

def ftrl_optimize(a0, time_limit, verbose=True):
    """FTRL (Follow-the-Regularized-Leader) with AdaGrad-style adaptive rates."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    # FTRL state
    z = np.zeros(n)       # cumulative gradient
    G = np.zeros(n)       # cumulative |grad|
    H = 1e-6              # regularization
    t = 0
    stall = 0
    beta = 30.0
    top_k = max(30, int(n * 0.03))
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(a)
        m_curr = b.max()
        
        if m_curr < best_m - 1e-15:
            best_m = m_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Smoothed gradient
        grad = compute_gradient_smoothed(a, b, beta, top_k)
        
        # FTRL update
        z += grad
        G += np.abs(grad)
        # Newton step: x_i = -(z_i - sqrt(z_i^2 + 2*H*G_i)) / (2*H) for z_i < 0
        # Simplified: x_i = max(0, (sqrt(z_i^2 + 2*H*G_i) - z_i)) / (2*H)
        xs = H * np.sqrt((z / H) ** 2 + 2.0 * G) - z
        xs = np.maximum(xs / (2.0 * H), 0.0)
        
        # Project to simplex
        a_new = simplex_project(xs)
        
        # Check
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            a = a_new
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            # Blend
            alpha = min(0.3, 2.0 / (t + 10.0))
            a_new2 = (1.0 - alpha) * a + alpha * a_new2 if False else (1.0 - alpha) * a + alpha * a_new
            a_new2 = np.maximum(a_new2, 0.0)
            s = a_new2.sum()
            if s > 1e-15:
                a_new2 = a_new2 / s
            a = a_new2
        
        # Adapt beta
        if t % 80 == 0:
            beta = min(beta * 1.01, 500.0)
        
        # Restart
        if stall > 500:
            stall = 0
            noise = rng.normal(0, 0.01, n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            a = x_r
            z = np.zeros(n)
            G = np.zeros(n)
            beta = max(beta * 0.5, 10.0)
            
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    FTRL restart t={t} ({elapsed:.1f}s): J={J:.6f}")
        
        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FTRL t={t} ({elapsed:.1f}s): J={J:.6f} stall={stall}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.1f}s)")
    return best_seq, best_score

def lp_refine(a0, time_limit, verbose=True):
    """LP-based Frank-Wolfe refinement focusing on the top peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Get top peaks
    b = fast_autocorr(a)
    
    # Try a few different LP formulations
    for n_active in [200, 500, 1000]:
        if time.monotonic() - t0 > time_limit * 0.7:
            break
        
        active = np.argsort(b)[::-1][:n_active]
        M = b.max()
        
        # Build constraint matrix: (a*g)[k] <= M for k in active
        # (a*g)[k] = sum_i a[i]*g[k-i] -> linear in g
        K = len(active)
        try:
            A = np.zeros((K, n))
            for jj, k in enumerate(active):
                k_int = int(k)
                i_start = max(0, k_int - n + 1)
                i_end = min(n - 1, k_int)
                if i_start <= i_end:
                    idx = np.arange(i_start, i_end + 1)
                    A[jj, idx] = a[k_int - idx]
            
            import cvxpy as cp
            g = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [cp.sum(g) == 1, A @ g <= M_var]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            
            try:
                prob.solve(solver=cp.CLARABEL, verbose=False)
            except:
                try:
                    prob.solve(solver=cp.SCS, verbose=False)
                except:
                    prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                continue
            
            g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
            g_sum = g_val.sum()
            if g_sum < 1e-15:
                continue
            g_dir = g_val / g_sum
            
            # Line search
            best_alpha = 0
            best_trial = best_m
            for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                trial = (1.0 - alpha) * a + alpha * g_dir
                trial = np.maximum(trial, 0.0)
                st = trial.sum()
                if st < 1e-15:
                    continue
                trial = trial / st
                M_trial = fast_autocorr(trial).max()
                if M_trial < best_trial:
                    best_trial = M_trial
                    best_alpha = alpha
            
            if best_alpha > 0:
                a_new = (1.0 - best_alpha) * a + best_alpha * g_dir
                a_new = np.maximum(a_new, 0.0)
                st = a_new.sum()
                if st > 1e-15:
                    a_new = a_new / st
                a = a_new
                if best_trial < best_m:
                    best_m = best_trial
                    best_a = a.copy()
        
        except Exception as e:
            print(f"    LP error: {e}")
            continue
    
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Load parent, try nearby lengths
    2. Direct peak shaving
    3. Nesterov subgradient with adaptive smoothing
    4. FTRL optimization
    5. More peak shaving
    6. LP refinement
    7. Multi-length exploration
    8. Final polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        parent = None
    
    if best_seq is None:
        n_base = 7000
        x = np.linspace(0, 1, n_base)
        a_raw = (x + 0.01) ** 1.0 * (1.0 - x + 0.01) ** 1.0
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # Phase 1: Quick peak shaving to get a quick improvement
    print(f"\n=== Phase 1: Peak shaving (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    if t1 > 20:
        a_out, _ = peak_shaving_direct(best_seq, t1, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps1", t_global)
    
    # Phase 2: Nesterov subgradient (main workhorse)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Nesterov subgrad (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 60:
        a_out, _ = nesterov_subgradient(best_seq, t2, beta_start=20.0, beta_end=600.0,
                                         beta_growth=1.004, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)
    
    # Phase 3: Direct peak shaving round 2
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Peak shaving 2 (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        a_out, _ = peak_shaving_direct(best_seq, t3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps3", t_global)
    
    # Phase 4: FTRL optimization
    n_best = len(best_seq)
    print(f"\n=== Phase 4: FTRL (n={n_best}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining *
