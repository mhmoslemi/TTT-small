                _MODULE_SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_cross_cov(a, w, fft_size=None):
    """Compute C_j = sum_{m=0}^{n-1} a[m] * w[j+m] for j=0..n-1 using FFT."""
    n = len(a)
    if fft_size is None:
        fft_size = 1 << (3 * n - 1 + 64).bit_length()
    A = np.fft.fft(a[::-1], fft_size)
    W = np.fft.fft(w, fft_size)
    C_full = np.fft.ifft(A * W).real[:3 * n - 1]
    return C_full[n - 1:n - 1 + n]


def smoothed_objective_and_grad(a, t):
    """Compute smoothed objective value and gradient. a should be normalized (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    conv_scaled = t * conv
    conv_shifted = conv_scaled - conv_scaled.max()
    exp_vals = np.exp(conv_shifted)
    log_sum_exp = conv_shifted.max() + np.log(exp_vals.sum())
    smooth_val = t * log_sum_exp / (t * t)  # = log_sum_exp / t
    w = exp_vals / exp_vals.sum()
    C = smooth_cross_cov(a, w)
    grad = 2.0 * C
    return smooth_val, grad, w, conv


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def smoothed_adam(a0, time_limit, t_start=20.0, t_end=200.0, momentum=0.9,
                  lr_start=0.02, lr_end=0.001, seed=42, verbose=True):
    """Adam with Nesterov momentum on smoothed log-sum-exp objective."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    conv_best = fast_convolve(best_a)
    best_m = conv_best.max()
    best_score = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Adam state
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    t_step = 0
    no_improve = 0
    max_no_improve = 2000
    stall_restarts = 0

    total_iters_expected = max(1, int(time_limit / 0.01))

    while time.monotonic() - t0 < time_limit:
        t_step += 1
        progress = min(1.0, t_step / max(1, total_iters_expected))
        
        # Anneal temperature
        t_temp = t_start + (t_end - t_start) * progress
        # Anneal learning rate
        lr = lr_start * (lr_end / lr_start) ** progress

        # Nesterov lookahead
        y_lookup = y + momentum * (y - (y - lr * (m_adam / (np.sqrt(v_adam) + eps))))
        y_lookup = np.maximum(y_lookup, 1e-15)
        s_lookup = y_lookup.sum()
        if s_lookup < 1e-15:
            y_lookup = y.copy()
        else:
            y_lookup = y_lookup / s_lookup

        # Compute smoothed gradient
        smooth_val, grad, w, conv = smoothed_objective_and_grad(y_lookup, t_temp)

        # Normalize gradient by sum to make it scale-invariant
        s_y = y.sum()
        if s_y > 1e-15:
            grad = grad / s_y

        # Adam update
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * grad * grad
        m_hat = m_adam / (1 - beta1 ** t_step)
        v_hat = v_adam / (1 - beta2 ** t_step)

        # Update
        y_new = y - lr * m_hat / (np.sqrt(v_hat) + eps)
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_new

        # Evaluate true objective
        conv_new = fast_convolve(y_new)
        M_new = conv_new.max()
        score_new = 2.0 * n * M_new

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_m = M_new
            best_a = y_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        y = y_new

        if no_improve > max_no_improve:
            stall_restarts += 1
            no_improve = 0
            # Restart with perturbation
            if stall_restarts <= 5:
                noise_scale = 0.05 * (stall_restarts ** 0.5)
                noise = rng.normal(0, noise_scale, n) * best_a
                y_p = best_a + noise
                y_p = np.maximum(y_p, 1e-10)
                sp = y_p.sum()
                if sp > 1e-12:
                    y = y_p / sp
                m_adam = np.zeros(n)
                v_adam = np.zeros(n)
                t_step = 0
                momentum = min(0.95, momentum * 1.02)
            else:
                if verbose:
                    print("    Smoothed Adam: giving up after " + str(stall_restarts) + " restarts")
                break

        if verbose and t_step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print("    Smoothed Adam iter=" + str(t_step) + " (" + str(round(elapsed, 1)) + "s): "
                  + "J=" + repr(best_score) + ", t_temp=" + repr(round(t_temp, 1)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    Smoothed Adam done (" + str(round(elapsed, 1)) + "s): J=" + repr(best_score))
    return best_a, best_score


def lp_fw_refined(a0, time_limit, solver, initial_K=300, max_K=5000,
                  add_per_iter=60, verbose=True, dual_prune=True):
    """Improved LP-FW with dual-based active set management and better line search."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac_y = fast_convolve(y)
    best_m = ac_y.max()
    t0 = time.monotonic()

    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 600
    iter_count = 0
    stagnation_count = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if stagnation_count > 4:
                if verbose:
                    print("    LP-FW: persistent stagnation, stopping")
                break
            # Perturb active set
            ac_curr = fast_convolve(y)
            n_pert = max(30, len(active_set) // 4)
            rng_local = np.random.default_rng(42 + stagnation_count)
            keep = list(rng_local.choice(len(active_set), size=len(active_set) - n_pert, replace=False))
            new_adds = np.argsort(ac_curr)[::-1][:n_pert * 2]
            new_set = set([active_set[i] for i in keep])
            for k in new_adds:
                new_set.add(int(k))
            active_set = sorted(new_set)
            no_improve = 0
            stagnation_count += 1
            continue

        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            stagnation_count = 0
        else:
            no_improve += 1

        # Trim active set using dual-informed criteria
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            # Keep constraints with smallest margins (most binding)
            keep_idx = np.argsort(margins)[:int(max_K * 0.75)]
            active_set_base = [active_set[i] for i in keep_idx]
            violations = np.where(ac > M_curr - 1e-8)[0]
            for v in violations[:100]:
                if int(v) not in active_set_base:
                    active_set_base.append(int(v))
            active_set = sorted(set(active_set_base))[:max_K]

        K = len(active_set)
        if K == 0:
            break

        coeffs = build_constraint_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(100, K // 3)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # Try with reduced active set
            active_set = active_set[:max(200, K // 2)]
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]

        all_viol = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(all_viol) > 0:
            viol_amt = conv_ys[all_viol] - M_curr
            top_v = all_viol[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        # Better line search: golden-section on the 1D problem
        best_alpha = 1.0 / (iter_count + 2.0)
        best_trial_M = fast_convolve(y).max()

        if iter_count > 5:
            # Try a range of alphas including aggressive ones
            alphas_to_try = [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1,
                            max(0.05, 2.0 / (iter_count + 2.0)),
                            max(0.02, 1.0 / (iter_count + 1.0))]
        else:
            alphas_to_try = [1.0, 0.5, 0.3, 0.1]

        for alpha in alphas_to_try:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 3:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): "
                  + "J=" + repr(2.0 * n * best_m) + ", |S|=" + str(len(active_set)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", iters=" + str(iter_count))
    return best_a, best_m


def coordinate_polish_v2(a0, time_limit, rng_seed=77, verbose=True):
    """Enhanced coordinate polish with smarter peak reduction."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    conv_a = fast_convolve(a)
    best_m = conv_a.max()
    best_score = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv_a = fast_convolve(a)
        # Focus on top peaks
        n_peaks = min(20, 2 * n - 1)
        top_peaks = np.argsort(conv_a)[::-1][:n_peaks]
        peak_vals = conv_a[top_peaks]
        M_curr = peak_vals[0]

        improved = False

        # Strategy 1: Reduce mass at positions contributing most to top peaks
        for peak_idx in range(min(8, n_peaks)):
            if time.monotonic() - t0 > time_limit:
                break
            k = int(top_peaks[peak_idx])
            if conv_a[k] < M_curr * 0.95:
                continue  # Only work on significant peaks

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # Find the position with highest contribution
            src_local = int(idx[np.argmax(contrib)])
            src_val = a[src_local]
            if src_val < 1e-10:
                continue

            # Find destination: position that minimizes impact on all top peaks
            # Score each position by how much it would affect the top peaks if we add mass there
            scores = np.zeros(n)
            for pk in top_peaks[:min(10, n_peaks)]:
                pk_i = int(pk)
                s2 = max(0, pk_i - n + 1)
                e2 = min(n - 1, pk_i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[pk_i - idx2]

            # Prefer destinations with low scores (less impact on peaks)
            # Also avoid the immediate neighborhood of the source
            exclude = set()
            for d in range(max(1, n // 50), max(2, n // 25)):
                if src_local - d >= 0:
                    exclude.add(src_local - d)
                if src_local + d < n:
                    exclude.add(src_local + d)
            exclude.add(src_local)

            candidate_mask = np.ones(n, dtype=bool)
            for e in exclude:
                candidate_mask[e] = False
            
            cand_scores = scores.copy()
            cand_scores[~candidate_mask] = np.inf

            dst = int(np.argmin(cand_scores))
            if dst == src_local or cand_scores[dst] == np.inf:
                continue

            # Try to transfer mass
            for frac in [0.005, 0.01, 0.02, 0.003, 0.015]:
                amt = src_val * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src_local] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_score = 2.0 * n * best_m
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        # Strategy 2: Random perturbations when stuck
        if not improved:
            stall += 1
            if stall % 100 == 0:
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.003, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_score = 2.0 * n * best_m
                                best_a = a.copy()
                                stall = 0
                                improved = True

        if verbose and stall > 0 and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    Coord v2 (" + str(round(elapsed, 1)) + "s): J=" + repr(best_score) + ", stall=" + str(stall))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    Coord v2 done (" + str(round(elapsed, 1)) + "s): J=" + repr(best_score))
    return best_a, best_score


def get_initial_sequence(t_global):
    """Get the best starting sequence."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
                return p_norm, sc
    except Exception:
        pass

    # Fallback: sinusoidal
    n = 5000
    x = np.linspace(0, 1, n)
    best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
    best_score = evaluate_safe(best_seq)
    print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))
    return best_seq, best_score


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

                a_out, _ = lp_fw_refined(norm_sum1(best_seq), t5b, solver,
                                        initial_K=300, max_K=3000, add_per_iter=50, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

    # === Phase 6: Final LP-FW on best ===
    print("\n=== Phase 6: Final LP-FW (n=" + str(len(best_seq)) + ") ===")
    t6 = min(remaining_time() * 0.20, 250.0)
    if t6 > 40 and len(best_seq) <= 12000:
        a_out, _ = lp_fw_refined(norm_sum1(best_seq), t6, solver,
                                initial_K=800, max_K=6000, add_per_iter=100, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw6_n" + str(len(best_seq)), t_global)

    # === Phase 7: Coordinate polish ===
    print("\n=== Phase 7: Coordinate Polish (n=" + str(len(best_seq)) + ") ===")
    t7 = min(remaining_time() * 0.10, 120.0)
    if t7 > 20:
        a_out, _ = coordinate_polish_v2(norm_sum1(best_seq), t7, rng_seed=2024, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord7_n" + str(len(best_seq)), t_global)

    # === Phase 8: Final smoothed Adam micro-optimization ===
    rt = remaining_time()
    if rt > 30:
        t8 = min(rt * 0.5, 100.0)
        a_out, _ = smoothed_adam(norm_sum1(best_seq), t8, t_start=200.0, t_end=600.0,
                                 lr_start=0.005, lr_end=0.0002, momentum=0.8, seed=999, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "smooth_final", t_global)

    # === Final coordinate polish ===
    rt = remaining_time()
    if rt > 20:
        a_out, _ = coordinate_polish_v2(norm_sum1(best_seq), min(rt * 0.6, 80.0), rng_seed=3141, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
