def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    n = arr.size
    if n == 0:
        return np.ones(1)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    np.maximum(arr, 0.0, out=arr)
    s = arr.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return arr / s


def fft_conv(a):
    n = a.size
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return np.where(np.isfinite(c) & (c > 0), c, 0.0)


def fft_conv2(a, b):
    n = a.size
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    return np.where(np.isfinite(c) & (c > 0), c, 0.0)


def grad_smooth_max(a, beta, top_k=None):
    """
    Compute the smoothed gradient of max_k conv(a,a)[k].
    Uses top-k lags if specified, otherwise all lags with softmax(beta*c).
    Returns gradient vector of length n.
    """
    n = a.size
    c = fft_conv(a)
    size = 2 * n - 1
    
    if top_k is not None and top_k < size:
        idx = np.argpartition(c, -top_k)[-top_k:]
        p = np.zeros(size)
        vals = c[idx]
        m = vals.max()
        e = np.exp(beta * (vals - m))
        p[idx] = e / e.sum()
    else:
        m = c.max()
        e = np.exp(beta * (c - m))
        p = e / e.sum()
    
    # Compute gradient: g_j = sum_k p_k * (a[j] + a[k-j]) * 1(0 <= k-j <= n-1)
    # Using FFT: g = 2 * cross_conv(p, a) approximately
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    out = conv[n-1:2*n-1][::-1]
    out = np.where(np.isfinite(out), out, 0.0)
    
    # Add the "a[j] * sum(valid p_k)" term

    # Approximate: g ≈ 2 * out (for interior)
    # More precisely: g_j = a[j] * S_j + out_j where S_j = sum of p_k for valid k
    # For now, use 2*out as approximation (good for interior coordinates)
    return 2.0 * out


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return np.array([])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

        se = e.sum()
        if se < 1e-30:
            p = np.zeros(size)
            p[np.argmax(c)] = 1.0
        else:
            p = e / se
        f = 2.0 * n * (m + math.log(se) / beta)
    
    # Gradient w.r.t. a
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    ga = 2.0 * conv[n-1:2*n-1][::-1]
    ga = np.where(np.isfinite(ga), ga, 0.0)
    
    # Chain rule: dv -> du -> da
    # a = softmax(u), so da/dv is the Jacobian of softmax
    mean_ga = np.dot(a, ga)
    gu = a * (ga - mean_ga)
    
    gv = np.empty(v.size)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref+1:]
    
    gv = np.where(np.isfinite(gv), gv, 0.0)
    return f, gv


def run_lbfegs(a0, beta, top_k, maxfun, ref):
    if not SCIPY_OK:
        return a0
    v0, ref = make_v(a0, ref)
    if v0.size == 0:
        return a0
    try:
        res = scipy.optimize.minimize(
            objgrad_v, v0,
            args=(beta, top_k, ref),
            method='L-BFGS-B',
            jac=True,
            options={'maxfun': maxfun, 'maxiter': maxfun + 50,
                     'ftol': 1e-14, 'gtol': 1e-8, 'disp': False}
        )
        if res is not None and hasattr(res, 'x'):
            vx = np.asarray(res.x, dtype=np.float64)
            if vx.size == v0.size and np.all(np.isfinite(vx)):
                return norm_sum1(from_v(vx, ref))
    except:
        pass
    return a0


def projected_gradient_search(a0, budget_s, max_k_init=500, max_k_min=30, verbose=True):
    """
    Projected subgradient method with adaptive active set.
    """
    t0 = time.monotonic()
    a = norm_sum1(a0)
    n = a.size
    score0 = proxy_score(a)
    best_a = a.copy()
    best_score = score0
    
    t = 0
    no_improve = 0
    alpha = 0.1  # initial step size
    
    while time.monotonic() - t0 < budget_s - 2:
        t += 1
        time_left = budget_s - (time.monotonic() - t0)
        if time_left < 5:
            break
        
        # Adaptive top-k
        progress = min(1.0, t / 500.0)
        top_k = int(max_k_init * (1 - progress) + max_k_min * progress)
        
        # Compute smoothed gradient
        beta = 50.0 + 200.0 * progress
        g = grad_smooth_max(a, beta, top_k)
        
        # Center for simplex
        g = g - g.mean()
        
        # Step size adaptation (BB-like)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Try a few step sizes
        improved = False
        for trial_alpha in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            new_a = norm_sum1(a - trial_alpha * g)
            new_score = proxy_score(new_a)
            if new_score < best_score - 1e-16:
                a = new_a
                best_a = a.copy()
                best_score = new_score
                alpha = trial_alpha  # keep this step size
                improved = True
                break
        
        if not improved:
            alpha *= 0.7
            no_improve += 1
            if no_improve > 100:
                # Reset with perturbation
                rng = np.random.default_rng(42 + t)
                a = norm_sum1(best_a + rng.normal(0, 0.01, n))
                alpha = 0.1
                no_improve = 0
            if no_improve > 300:
                break
        else:
            no_improve = 0
            alpha = min(alpha * 1.02, 0.5)
        
        if verbose and (t == 1 or t % 50 == 0):
            print(f"    PG t={t} score={best_score:.8f} alpha={alpha:.4f} top_k={top_k}")
    
    return best_a


def coordinate_descent(a0, budget_s, verbose=True):
    """
    Coordinate descent with rank-1 convolution updates.
    """
    t0 = time.monotonic()
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    score = 2.0 * n * c.max()
    best_a = a.copy()
    best_score = score
    
    rng = np.random.default_rng(12345)
    stall = 0
    
    while time.monotonic() - t0 < budget_s - 3:
        if stall > 200:
            break
        
        # Find top peaks
        top_count = min(20, 2*n-1)
        top_idx = np.argpartition(c, -top_count)[-top_count:]
        
        # Compute pressure (which coordinates affect the top peaks most)
        p_top = np.zeros(2*n-1)
        p_top[top_idx] = 1.0 / len(top_idx)
        
        # Pressure for each coordinate j: how much it contributes to top peaks
        total = 3*n - 2
        N = next_pow2(total)
        A = np.fft.rfft(a, N)
        Pr = np.fft.rfft(p_top[::-1], N)
        conv = np.fft.irfft(A * Pr, N)[:total]
        pressure = 2.0 * conv[n-1:2*n-1][::-1]
        pressure = np.where(np.isfinite(pressure), pressure, 0.0)
        
        # Try moves from high-pressure source to low-pressure destination
        src_candidates = np.argpartition(-pressure, min(10, n-1))[:10]
        dst_candidates = np.argpartition(pressure, min(15, n-1))[:15]
        
        best_move = None
        best_new_score = score
        
        for s_idx in src_candidates:
            s = int(s_idx)
            if a[s] < 1e-12:
                continue
            for d_idx in dst_candidates:
                d = int(d_idx)
                if d == s:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    tmove = a[s] * frac
                    if tmove < 1e-15:
                        continue
                    # Rank-1 update
                    c_new = c.copy()
                    c_new[d:d+n] += 2.0 * tmove * a
                    c_new[s:s+n] -= 2.0 * tmove * a
                    if 2*d < c_new.size:
                        c_new[2*d] += tmove*tmove
                    if 2*s < c_new.size:
                        c_new[2*s] += tmove*tmove
                    if s+d < c_new.size:
                        c_new[s+d] -= 2.0*tmove*tmove
                    c_new = np.where(np.isfinite(c_new) & (c_new > 0), c_new, 0.0)
                    new_score = 2.0 * n * c_new.max()
                    if new_score < best_new_score:
                        best_new_score = new_score
                        best_move = (s, d, tmove)
        
        # Also try some random moves
        for _ in range(20):
            s = int(rng.integers(n))
            d = int(rng.integers(n))
            if s == d or a[s] < 1e-12:
                continue
            frac = float(rng.uniform(0.005, 0.03))
            tmove = a[s] * frac
            c_new = c.copy()
            c_new[d:d+n] += 2.0 * tmove * a
            c_new[s:s+n] -= 2.0 * tmove * a
            if 2*d < c_new.size:
                c_new[2*d] += tmove*tmove
            if 2*s < c_new.size:
                c_new[2*s] += tmove*tmove
            if s+d < c_new.size:
                c_new[s+d] -= 2.0*tmove*tmove
            c_new = np.where(np.isfinite(c_new) & (c_new > 0), c_new, 0.0)
            new_score = 2.0 * n * c_new.max()
            if new_score < best_new_score:
                best_new_score = new_score
                best_move = (s, d, tmove)
        
        if best_move is not None:
            s, d, tmove = best_move
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            c = fft_conv(a)
            score = 2.0 * n * c.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            stall = 0
            if verbose and stall == 0 and (t0 and time.monotonic() - t0) % 50 < 1:
                pass  # skip verbose for simplicity
        else:
            stall += 1
            # Perturb to escape
            if stall % 30 == 0:
                a = norm_sum1(best_a + rng.normal(0, 0.005*best_score/100, n))
                c = fft_conv(a)
                score = 2.0 * n * c.max()
    
    return best_a


def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)))
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0, 1, a.size)
    new_x = np.linspace(0, 1, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines aggressive Frank-Wolfe (multiplicative weights), high-beta L-BFGS,
exhaustive coordinate descent with rank-1 updates, and length exploration.
Distinct from prior approach by using higher-beta polishing, more MW iterations,
and a more exhaustive descent phase.

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    n = arr.size
    if n == 0:
        return np.ones(1)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    np.maximum(arr, 0.0, out=arr)
    s = arr.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return arr / s


def fft_conv(a):
    n = a.size
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return np.where(np.isfinite(c) & (c > 0.0), c, 0.0)


def fft_conv2(a, b):
    n = a.size
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    return np.where(np.isfinite(c) & (c > 0.0), c, 0.0)


def gradient_at_lags(a, p, factor=2.0):
    """Compute factor * sum_k p[k] * d/d a_j conv(a,a)[k] via FFT."""
    n = a.size
    size = 2 * n - 1
    if p.size != size:
        p = np.resize(p, size)
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    out = conv[n - 1:2 * n - 1][::-1]
    out = np.where(np.isfinite(out), out, 0.0)
    return factor * out

def make_v(a, ref=None):
    n = a.size
    if n <= 1:
        return np.array([]), 0
    if ref is None:
        ref = int(np.argmax(a))
    ref = int(ref)
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.where(np.isfinite(v), v, -700.0)

def from_v(v, ref):
    n = v.size + 1
    u = np.zeros(n)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    um = float(u.max())
    e = np.exp(u - um)
    s = e.sum()
    if s < 1e-15 or not math.isfinite(s):
        return np.ones(n) / n
    return e / s


def objgrad_v(v, beta, top_k, ref):
    n = v.size + 1
    if n <= 0:
        return 1e100, np.array([])
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    u = np.empty(n)
    u[ref] = 0.0
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    a = from_v(v, ref)
    c = fft_conv(a)
    size = 2 * n - 1
    if top_k is not None and 0 < top_k < size:
        idx = np.argpartition(c, -top_k)[-top_k:]
        vals = c[idx]
        m = float(vals.max())
        shifted = np.clip(beta * (vals - m), -700.0, 700.0)
        e = np.exp(shifted)
        se = float(e.sum())
        if se < 1e-30:
            p = np.zeros(size)
            p[idx[0]] = 1.0
        else:
            p = np.zeros(size)
            p[idx] = e / se
        f = 2.0 * n * (m + math.log(se) / beta)
    else:
        m = float(c.max())
        shifted = np.clip(beta * (c - m), -700.0, 700.0)
        e = np.exp(shifted)
        se = float(e.sum())
        if se < 1e-30:
            p = np.zeros(size)
            p[int(np.argmax(c))] = 1.0
        else:
            p = e / se
        f = 2.0 * n * (m + math.log(se) / beta)
    ga = gradient_at_lags(a, p, 2.0)
    mean_ga = float(np.dot(a, ga))
    gu = a * (ga - mean_ga)
    gv = np.empty(v.size)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]
    gv = np.where(np.isfinite(gv), gv, 0.0)
    return f, gv


def run_lbfegs(a0, beta, top_k, maxfun, ref=None):
    if not SCIPY_OK:
        return a0
    v0, ref = make_v(a0, ref)
    if v0.size == 0:
        return a0
    try:
        res = scipy.optimize.minimize(
            objgrad_v, v0,
            args=(beta, top_k, ref),
            method='L-BFGS-B', jac=True,
            options={'maxfun': int(maxfun), 'maxiter': int(maxfun + 50),
                     'ftol': 1e-14, 'gtol': 1e-9, 'disp': False}
        )
        if res is not None and hasattr(res, 'x'):
            vx = np.asarray(res.x, dtype=np.float64)
            if vx.size == v0.size and np.all(np.isfinite(vx)):
                return norm_sum1(from_v(vx, ref))
    except Exception:
        pass
    return a0


def mw_direction(f, active_idx, mw_iters, seed, lr=12.0):
    """Solve FW subproblem via multiplicative weights."""
    f = norm_sum1(f)
    n = f.size
    size = 2 * n - 1
    active_idx = np.asarray(active_idx, dtype=np.int64)
    K = active_idx.size
    if K <= 0:
        return np.ones(n) / n
    rng = np.random.default_rng(seed)
    logits = np.zeros(K)
    logits += rng.normal(0, 0.1, K)
    for _ in range(max(1, int(mw_iters))):
        pf = np.zeros(size)
        lm = logits - logits.max()
        w = np.exp(np.clip(lm, -500, 500))
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(K)
            ws = K
        pf[active_idx] = w / ws
        pressure = gradient_at_lags(f, pf, 1.0)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0), pressure, 0.0)
        pmx = float(pressure.max())
        pmn = float(pressure.min())
        ptp = pmx - pmn
        if ptp < 1e-30:
            g = np.ones(n) / n
        else:
            scaled = (pressure - pmn) / ptp
            ew = np.exp(-lr * scaled)
            ew = np.where(np.isfinite(ew), ew, 0.0)
            ew += 1e-15
            g = ew / ew.sum()
            g = 0.98 * g + 0.02 * (np.ones(n) / n)
            g = g / g.sum()
        h = fft_conv2(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmac_ = float(ha.max())
        hmin = float(ha.min())
        hptp = hmac_ - hmin
        if hptp < 1e-30:
            break
        eta = lr / hptp
        logits += eta * (ha - ha.mean())
    return g
