def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_active_coeffs(a, active_list, n):
    K = len(active_list)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj in range(K):
        ki = int(active_list[jj])
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[ki - idx]
    return coeffs

        conv_ra = np.fft.irfft(Wr * Ar, fft_size2)
        
        # grad_j = conv_ra[len(w_rev)-1-j] for j=0..n-1
        offset = len(w_rev) - 1
        grad = conv_ra[offset-n+1:offset+1][::-1]
        # Wait, let me re-check the indexing.
        # conv_ra[m] for m = 0..conv_size-1
        # We want conv_ra[offset-j] for j = 0..n-1
        # So indices: offset, offset-1, ..., offset-n+1
        grad = conv_ra[offset-n+1:offset+1][::-1]
        grad = np.maximum(grad, 0)  # gradient of max should be non-negative
        
        # Actually wait - the gradient of max_k (a*a)[k] is not simply the weighted sum.
        # For the subgradient: at the maximum, the subgradient is a*a[k*] vector.
        # For a smoothed version, we use the weighted sum.
        # But we should NOT clip the gradient to non-negative. The gradient can be negative
        # for coordinates that don't contribute to the peak lags.
        
        # Actually, let me reconsider. The gradient of (a*a)[k] with respect to a[j] is:
        # d/d(a[j]) sum_i a[i]*a[k-i] = a[k-j] (if 0 <= k-j < n)
        # This is always non-negative! So the gradient IS non-negative.
        # But we're minimizing, so we want to decrease a where the gradient is large.
        # Mirror descent: log(a_j) -= lr * grad_j, which decreases a_j where grad_j is large.
        
        # Actually, I realize the gradient IS always non-negative because a >= 0.
        # So we don't need to clip.
        
        # Mirror descent with momentum
        lr = lr0 / (1.0 + 0.003 * t)
        v = beta * v + (1 - beta) * grad
        log_x = np.log(x + 1e-30) - lr * v
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        x_new = np.maximum(x_new, 1e-25)
        x = x_new / x_new.sum()
        
        # Periodic restart
        if stall > 4000 and stall % 1000 == 0:
            noise = 0.05 * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x = x_r / x_r.sum()
            v *= 0.1
            stall = 0
        
        if verbose and t % 500 == 0:
            print("    SubGrad t={} ({:.1f}s) J={:.7f} stall={}".format(t, time.monotonic() - t0, best_val, stall))
    
    return best_a, best_val
