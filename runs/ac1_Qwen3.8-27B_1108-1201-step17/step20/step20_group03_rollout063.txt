            b[ks] += 2.0 * delta * a[idx]
        if 0 <= 2 * i <= 2 * n - 2:
            b[2 * i] += delta * delta

        a[i] = x_new

        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 100 == 0:
            b = fast_autocorr(a)
            cur_m = b.max()
            if cur_m < best_m - 1e-15:
                best_m = cur_m
                best_a = a.copy()

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    TopK-CoordDesc it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    b_final = fast_autocorr(a)
    if b_final.max() < best_m:
        best_m = b_final.max()
        best_a = a.copy()

    return best_a, best_m


def mirror_descent(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = 1.0 / np.sqrt(n)

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        k_star = int(np.argmax(b))

        if k_star == 2 * np.argmax(a):
            pass

        grad = np.zeros(n)
        for i in range(n):
            if k_star == 2 * i:
                grad[i] = 2.0 * a[i]
            elif 0 <= k_star - i < n:
                grad[i] = 2.0 * a[k_star - i]

        log_a = np.log(np.maximum(a, 1e-30))
        log_new = log_a - eta * grad
        log_new -= np.max(log_new)
        a_new = np.exp(log_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        b_new = fast_autocorr(a_new)
        if b_new.max() < best_m - 1e-15:
            best_m = b_new.max()
            best_a = a_new.copy()

        a = a_new
        eta = 1.0 / np.sqrt(t + 1)

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    MirrorDesc t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2000, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 500

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[int(i)]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:20]]
            for k in top_v:
                active_set.add(int(k))

        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5),
                  max(0.02, default_alpha * 1.5), 1.0 / (t + 1)]
        alphas = [max(0.01, min(1.0, a)) for a in alphas]
        alphas = list(set(alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def cyclic_shift_best(a0):
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a0, fft_size)
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def peak_directed_search(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0).copy()
    b = fast_autocorr(a)
    best_a = a.copy()
    best_m = b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    stall = 0
    max_stall = 100

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:20]
        improved = False

        for _ in range(80):
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(top_peaks[int(rng.integers(len(top_peaks)))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:12]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src:
                continue
            if a[src] < 1e-12:
                continue

            frac = float(rng.choice([0.003, 0.008, 0.015, 0.03, 0.05, 0.08]))
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-14:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                ac = fast_autocorr(a)
                top_peaks = np.argsort(ac)[::-1][:20]

        if not improved:
            stall += 1
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.002, 0.04))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSearch (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s, solver: " + str(solver))

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: FISTA-BB (main optimizer)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: FISTA-BB (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 60:
        a_out, _ = fista_opt(norm_sum1(best_seq), t1,
                             beta_start=12.0, beta_end=700.0,
                             beta_growth=1.0035, top_k=12,
                             restart_every=2500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista1", t_global)

    # Phase 2: Coordinate descent (new)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: Coordinate Descent (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining * 0.12, 180.0)
    if t2 > 30:
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t2, n_sweeps=80, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord1", t_global)

    # Phase 3: Cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: Cyclic shift (n=" + str(len(best_seq)) + ") ===")
    if remaining > 30:
        a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic1", t_global)

    # Phase 4: Top-k peaks coordinate descent (new)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 4: TopK CoordDescent (n=" + str(len(best_seq)) + ") ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_out, _ = top_k_peaks_coord_descent(norm_sum1(best_seq), t4, top_k=8, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "topk1", t_global)

    # Phase 5: LP-Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: LP-FW (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 40 and n_best <= 8000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t5, solver,
                         initial_K=350, max_K=2000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 6: Peak-directed search
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 6: Peak-directed search (n=" + str(len(best_seq)) + ") ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 20:
        a_out, _ = peak_directed_search(norm_sum1(best_seq), t6, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak1", t_global)

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Multi-length exploration ===")
    longer_lengths = [n_best + 500, n_best + 1000, n_best + 1500]
    longer_lengths = [l for l in longer_lengths if l <= 10000 and l > n_best]
    shorter_lengths = [max(2000, n_best - 1000), max(3000, n_best - 500)]
    shorter_lengths = [l for l in shorter_lengths if l < n_best]

    for new_n in longer_lengths + shorter_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("  Trying n=" + str(new_n) + " (remaining=" + str(round(remaining, 1)) + "s)")
        a_long = interpolate_sequence(best_seq, new_n)

        t_a = min(remaining * 0.25, 180.0)
        if t_a > 40:
            a_out, _ = fista_opt(a_long, t_a,
                                 beta_start=20.0, beta_end=500.0,
                                 beta_growth=1.004, top_k=10,
                                 restart_every=2000, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t_b = min(remaining * 0.15, 100.0)
            if t_b > 20:
                a_out, _ = coordinate_descent(norm_sum1(best_seq), t_b, n_sweeps=40, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
            best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_n" + str(new_n), t_global)

    # Phase 8: Second FISTA pass with higher beta
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: FISTA v2 (n=" + str(n_best) + ") ===")
    t8 = min(remaining * 0.15, 200.0)
    if t8 > 40:
        a_out, _ = fista_opt(norm_sum1(best_seq), t8,
                             beta_start=50.0, beta_end=1500.0,
                             beta_growth=1.006, top_k=8,
                             restart_every=2000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista2", t_global)

    # Phase 9: Final coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 9: Final CoordDesc (n=" + str(len(best_seq)) + ") ===")
    t9 = min(remaining * 0.15, 150.0)
    if t9 > 20:
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t9, n_sweeps=100, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Phase 10: Final peak-directed polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t10 = min(remaining * 0.20, 100.0)
        if t10 > 15:
            a_out, _ = peak_directed_search(norm_sum1(best_seq), t10, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_final", t_global)

    # Final cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== Final: Cyclic shift ===")
        a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_final", t_global)

    # Final micro FISTA
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t_f = min(remaining * 0.50, 100.0)
        if t_f > 15:
            a_out, _ = fista_opt(norm_sum1(best_seq), t_f,
                                 beta_start=100.0, beta_end=2000.0,
                                 beta_growth=1.01, top_k=6,
                                 restart_every=1500, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_final", t_global)

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
