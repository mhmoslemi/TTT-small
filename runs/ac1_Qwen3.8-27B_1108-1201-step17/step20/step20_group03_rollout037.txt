"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with combined max-subgradient and variance (flatness) gradient
2. Multi-n exploration with asymmetric initializations
3. Adam-based local refinement for final polishing
4. Peak-shaving local search (mass redistribution from peak lags)
5. LP-Frank-Wolfe for structured improvement when n is moderate
"""

import numpy as np
import time

def fast_conv(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    """Project onto probability simplex."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_gradient(a, C, top_k, var_weight, beta):
    """
    Compute combined gradient: smoothed max-subgradient + variance gradient.
    Returns gradient vector of length n.
    """
    n = len(a)
    # Smoothed max subgradient via softmax over peaks
    C_max = C.max()
    if beta > 1e-10:
        shifted = (C - C_max) * beta
        shifted = np.clip(shifted, -500, 0)
        weights = np.exp(shifted)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.zeros_like(C)
            weights[np.argmax(C)] = 1.0
        else:
            weights /= w_sum
    else:
        weights = np.zeros_like(C)
        weights[np.argmax(C)] = 1.0

    # Restrict to top_k peaks if specified
    if top_k is not None and 0 < top_k < len(C):
        thresh = np.partition(C, -top_k)[-top_k]
        weights[C < thresh - 1e-15] = 0.0
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.zeros_like(C)
            weights[np.argmax(C)] = 1.0
        else:
            weights /= w_sum

    # Compute subgradient: grad[i] = sum_k weights[k] * a[k-i] for valid k-i
    # Using FFT: grad = ifft(weights_fft * conj(a_fft))[n-1:n-1+n] (with appropriate padding)
    fft_size = 1 << (2 * n - 1).bit_length()
    Wf = np.fft.rfft(weights, fft_size)
    Af = np.fft.rfft(a, fft_size)
    # We want: grad[i] = sum_k weights[k] * a[k-i] (for 0<=k-i<n)
    # This is cross-correlation: ifft(conj(Af) * Wf)
    # ifft(conj(Af)*Wf)[m] = sum_j a[j] * W[m+j] ... let me verify
    # Actually: conv(W, a_rev)[m] = sum_j W[j] * a_rev[m-j] = sum_j W[j] * a[n-1-m+j]
    # We want grad[i] = sum_k W[k] * a[k-i], let j=k-i: sum_j W[i+j] * a[j]
    # = conv(W, a)[i + offset]... 
    # Let me use: grad[i] = sum_k W[k]*a[k-i] = (a * W_rev)[n-1-i + (2n-2-n+1)]
    # Simpler: use the correlation theorem
    # ifft(conj(FFT(a)) * FFT(W))[m] = sum_j a[j] * W[j+m]
    # We want sum_k W[k] * a[k-i] = sum_j a[j] * W[j+i] (sub j=k-i)
    # = ifft(conj(FFT(a)) * FFT(W))[i] for i=0..n-1
    corr = np.fft.irfft(np.conj(Af) * Wf, fft_size)
    grad_max = corr[:n].copy()

    # Variance gradient (encourages flatness of C)
    if var_weight > 0:
        # grad_var[i] = 2 * sum_j a[j] * C[i+j]
        # = 2 * ifft(conj(FFT(a)) * FFT(C))[i]
        Cf = np.fft.rfft(C, fft_size)
        grad_var = np.fft.irfft(np.conj(Af) * Cf, fft_size)[:n]
        grad_max += var_weight * 2.0 * grad_var

    return grad_max

def fista_optimize(a0, time_limit, beta_start=10.0, beta_end=2000.0,
                   beta_growth=1.004, top_k=20, var_weight=0.05,
                   restart_every=4000, verbose=True, label=""):
    """
    FISTA with Nesterov acceleration, BB step size, and combined gradient.
    """
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_conv(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(42)
    s_prev = None
    y_prev = None
    bb_step = 1.0 / np.sqrt(n)
    avg_x = x.copy()
    avg_count = 0.0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        C = fast_conv(y)
        C_max = C.max()

        grad = compute_gradient(y, C, top_k, var_weight, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm  # normalize for stable BB

        # Barzilai-Borwein step size
        if s_prev is not None and y_prev is not None:
            sy = y - y_prev
            gy = grad - s_prev
            sy_gy = sy.dot(gy)
            gy_gy = gy.dot(gy)
            if gy_gy > 1e-20:
                bb_candidate = sy_gy / gy_gy
                if 1e-8 < bb_candidate < 10.0:
                    bb_step = 0.5 * bb_step + 0.5 * bb_candidate
                    bb_step = max(1e-7, min(bb_step, 3.0))
                else:
                    bb_step = max(bb_step * 0.85, 1e-7)
            s_prev = grad.copy()
            y_prev = y.copy()
        else:
            s_prev = grad.copy()
            y_prev = y.copy()

        eta = min(bb_step, 0.5)
        z = y - eta * gnorm * grad  # Use gnorm to scale
        x_new = simplex_project(z)

        C_new = fast_conv(x_new)
        max_new = C_new.max()

        avg_count += 1.0
        avg_x = avg_x + (x_new - avg_x) / avg_count

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 100 == 0:
            beta = min(beta * beta_growth, beta_end)
            var_weight = max(var_weight * 0.999, 0.0)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.01 / (1.0 + 0.03 * (t // restart_every))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 5.0)
            s_prev = None
            y_prev = None
            bb_step = 1.0 / np.sqrt(n)

            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print(f"    FISTA{label} restart t={t} ({elapsed:.1f}s): J={J:.6f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print(f"    FISTA{label} t={t} ({elapsed:.1f}s): J={J:.6f} beta={beta:.1f}")

    # Check averaged solution
    if avg_count > 100:
        J_avg = 2.0 * n * fast_conv(avg_x).max()
        if J_avg < 2.0 * n * best_max_b:
            return avg_x, fast_conv(avg_x).max()
    return best_a, best_max_b

def adam_optimize(a0, time_limit, lr=0.05, top_k=5, var_weight=0.1,
                  verbose=True, label=""):
    """Adam optimizer with simplex projection."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        t += 1
        C = fast_conv(a)
        C_max = C.max()
        cur_obj = 2.0 * n * C_max

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        grad = compute_gradient(a, C, top_k, var_weight, 50.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr / (np.sqrt(v_hat) + eps)
        a_new = simplex_project(a - step * m_hat)

        obj_new = 2.0 * n * fast_conv(a_new).max()
        if obj_new < cur_obj * 1.01:
            a = a_new

        if stall > 3000:
            noise = rng.normal(0, 0.005, n)
            a_pert = np.maximum(best_a + noise, 0)
            s = a_pert.sum()
            if s > 1e-10:
                a = a_pert / s
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            stall = 0

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam{label} t={t} ({elapsed:.1f}s): obj={best_obj:.6f}")

    return best_a, best_obj
