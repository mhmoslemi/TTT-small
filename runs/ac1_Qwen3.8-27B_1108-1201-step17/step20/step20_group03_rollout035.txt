def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
                elif M_trial < M_trial + 1:
                    # Try smaller fraction
                    pass

        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.005, 0.05))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            break

        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSearch (%.1fs): J=%.6f stall=%d" % (elapsed, J, stall))

    return best_a, best_m

def block_coordinate_refine(a0, time_limit, block_size=64, verbose=True):
    """Optimize blocks of coordinates while keeping others fixed."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    n_blocks = (n + block_size - 1) // block_size
    block_order = list(range(n_blocks))

    for outer in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        rng.shuffle(block_order)
        improved_outer = False

        for bi in block_order:
            if time.monotonic() - t0 > time_limit:
                break

            start = bi * block_size
            end = min(start + block_size, n)
            if start >= n:
                break

            block = a[start:end].copy()
            block_sum = block.sum()
            if block_sum < 1e-10:
                continue

            # Optimize this block using gradient descent (small problem)
            # The block values appear linearly in most autocorrelation terms
            # and quadratically in the diagonal terms
            for inner in range(30):
                a_full = a.copy()
                a_full[start:end] = block
                b = fast_autocorr(a_full)
                b_max = b.max()

                # Gradient wrt block variables
                grad_block = np.zeros(end - start)
                for offset, i in enumerate(range(start, end)):
                    # grad[i] = 2 * sum_k p[k] * a[k-i]
                    # For simplicity, use subgradient of max
                    # Find contributing k values
                    for k in range(max(0, i - n + 1), min(2 * n - 1, i + n)):
                        if 0 <= k - i < n:
                            if b[k] >= b_max * 0.99:
                                grad_block[offset] += 2.0 * a[k - i]
                        if 0 <= k - i < n:
                            if b[k] >= b_max * 0.99:
                                grad_block[offset] += 2.0 * a[k - i]

                # Actually this is O(block_size * n) which is fine for block_size=64
                # But let's just use a simpler approach: softmax gradient restricted to block

                # Project onto simplex (maintain block sum)
                step = 0.01
                block_new = block - step * grad_block / max(block.max(), 1e-10)
                block_new = np.maximum(block_new, 0)
                bs = block_new.sum()
                if bs < 1e-15:
                    break
                block_new = block_new * (block_sum / bs)

                a_full_new = a.copy()
                a_full_new[start:end] = block_new
                M_new = fast_autocorr(a_full_new).max()
                if M_new < b_max:
                    a = a_full_new
                    block = block_new
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        improved_outer = True
                else:
                    step *= 0.5

        if not improved_outer and outer > 5:
            break

    return best_a, best_m

def cyclic_shift_best(a0, max_shifts=None):
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    if max_shifts is None:
        max_shifts = n
    fft_size = 1 << (2 * n - 1).bit_length()
    for s in range(1, min(max_shifts, n)):
        a_shift = np.roll(a0, -s)
        b = fast_autocorr(a_shift)
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a)
    a_interp = np.maximum(a_interp, 0.0)
    s = a_interp.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / s

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [%s]: %.6f (n=%d, t=%.1fs)" % (label, sc, len(a), elapsed))
    return best_seq, best_score

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Time budget: %.1fs" % time_budget)

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            best_seq = parent.copy()
            best_score = p_score
            print("Parent: n=%d, score=%.6f" % (len(parent), p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        best_seq = make_initial(n_base, 'sinusoidal')
        best_score = evaluate_safe(best_seq)
        print("Default init: n=%d, score=%.6f" % (len(best_seq), best_score))

    n0 = len(best_seq)

    # Phase 1: Try multiple parametric initializations and pick best
    print("\n=== Phase 1: Parametric exploration ===")
    shapes = [('ramp', 1.0), ('ramp', 2.0), ('exp_ramp', 0),
              ('sinusoidal', 0), ('bell', 0), ('power_sym', 1.5),
              ('parabola', 0), ('gaussian', 0.25), ('gaussian', 0.4)]
    for shape, param in shapes:
        if time.monotonic() - t_global > time_budget - 600:
            break
        a_init = make_initial(n0, shape, power=param if shape in ('ramp', 'power_sym') else 1.0,
                              sigma=param if shape == 'gaussian' else 0.3)
        # Quick FISTA for 15 seconds
        remaining_here = min(15.0, time_budget - (time.monotonic() - t_global))
        if remaining_here < 5:
            continue
        a_out, _ = fista_optimize(a_init, remaining_here, tau_start=3.0, tau_end=1.0,
                                  tau_decay=0.998, restart_every=1000, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "init_%s" % shape, t_global)

    # Phase 2: Main FISTA optimization on best sequence
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: Main FISTA (n=%d) ===" % len(best_seq))
    t2 = min(remaining * 0.35, 450.0)
    if t2 > 30:
        a_out, _ = fista_optimize(norm_sum1(best_seq), t2,
                                  tau_start=5.0, tau_end=0.3,
                                  tau_decay=0.9993, restart_every=2500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_main", t_global)

    # Phase 3: Cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic", t_global)

    # Phase 4: Peak-directed local search
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 4: Peak-directed search (n=%d) ===" % len(best_seq))
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        a_out, _ = peak_directed_search(norm_sum1(best_seq), t4, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_search1", t_global)

    # Phase 5: FISTA with different parameters (higher initial tau for exploration)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 5: FISTA v2 (n=%d) ===" % len(best_seq))
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 30:
        # Start from a perturbed version to escape local minimum
        rng = np.random.default_rng(77)
        a_pert = norm_sum1(best_seq)
        noise = rng.normal(0, 0.01, len(a_pert)) * a_pert
        a_pert = norm_sum1(a_pert + noise)
        a_out, _ = fista_optimize(a_pert, t5,
                                  tau_start=10.0, tau_end=0.5,
                                  tau_decay=0.999, restart_every=2000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_v2", t_global)

    # Phase 6: Length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Length exploration (base n=%d) ===" % n_best)
    lengths_to_try = []
    for delta in [-500, -200, 200, 500, 1000, 1500]:
        new_n = n_best + delta
        if 3000 <= new_n <= 12000 and new_n != n_best:
            lengths_to_try.append(new_n)

    for new_n in sorted(lengths_to_try):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("  Trying n=%d..." % new_n)
        a_long = interpolate_seq(best_seq, new_n)

        t_sub = min(remaining * 0.25, 150.0)
        if t_sub < 20:
            continue
        a_out, _ = fista_optimize(a_long, t_sub,
                                  tau_start=5.0, tau_end=0.5,
                                  tau_decay=0.999, restart_every=2000, verbose=False)
        # Also try cyclic shift
        a_cs, _ = cyclic_shift_best(a_out)
        best_seq, best_score = try_update(best_seq, best_score, a_cs, "n=%d" % new_n, t_global)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "n=%d_pre" % new_n, t_global)

    # Phase 7: Final FISTA on best
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 7: Final FISTA (n=%d) ===" % len(best_seq))
    t7 = min(remaining * 0.25, 250.0)
    if t7 > 20:
        a_out, _ = fista_optimize(norm_sum1(best_seq), t7,
                                  tau_start=3.0, tau_end=0.2,
                                  tau_decay=0.999, restart_every=2000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_final", t_global)

    # Phase 8: Final peak-directed search
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 8: Final peak search (n=%d) ===" % len(best_seq))
    t8 = min(remaining * 0.30, 200.0)
    if t8 > 20:
        a_out, _ = peak_directed_search(norm_sum1(best_seq), t8, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_final", t_global)

    # Phase 9: Final cyclic shift + micro FISTA
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_final", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    t9 = min(remaining * 0.5, 100.0)
    if t9 > 15:
        a_out, _ = fista_optimize(norm_sum1(best_seq), t9,
                                  tau_start=1.0, tau_end=0.1,
                                  tau_decay=0.998, restart_every=1500, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_micro", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=%.6f, n=%d, time=%.1fs" % (best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        best_seq = make_initial(n_fb, 'sinusoidal')
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
