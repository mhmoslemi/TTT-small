**IPF details:**

If load[i] < 0, position i is "underloaded". We should increase a[i].

def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize a to sum 1, ensuring non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project z onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate sequence f to new length new_n, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_init(n, rng):
    """
    Initialize using spectral design: design A(omega) to have specific properties,
    then take inverse FFT and clip to non-negative.
    """
    # We want |A(omega)|^2 to be as "peaked" as possible in a way that
    # makes the time-domain autocorrelation flat.
    # Actually, for flat time-domain autocorrelation, we want |A(omega)| to be constant.
    # But that gives a delta in time domain (which has max = 1 at one point).
    # 
    # Instead, let's try: design A(omega) to have a specific shape, take real part
    # of inverse FFT, clip to non-negative.
    
    # Strategy: use a "raised cosine" or "Waterman" type spectrum
    fft_size = 1 << (2 * n - 2).bit_length()
    omega = np.fft.rfftfreq(fft_size)
    
    # Try several spectral shapes
    candidates = []
    
    # Power-law spectrum
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        A_spec = 1.0 / (1.0 + abs(omega) * n * 0.1) ** alpha
        a = np.fft.irfft(A_spec * fft_size / n, fft_size)[:n]
        a = np.maximum(a, 0.0)
        if a.sum() > 1e-10:
            candidates.append(a / a.sum())
    
    # Gaussian spectrum
    for sigma in [0.1, 0.2, 0.3, 0.5]:
        A_spec = np.exp(-2 * (np.pi * omega * n * sigma) ** 2)
        a = np.fft.irfft(A_spec * fft_size / n, fft_size)[:n]
        a = np.maximum(a, 0.0)
        if a.sum() > 1e-10:
            candidates.append(a / a.sum())
    
    # Random phase (gives flat-ish autocorrelation)
    for _ in range(10):
        mag = np.ones(len(omega))
        # Make it smooth
        for smooth in [0.5, 1.0, 2.0]:
            kernel = np.exp(-np.arange(len(mag)) ** 2 / (2 * smooth ** 2 * len(mag)))
            mag_s = np.convolve(mag, kernel, 'same')
            phase = rng.uniform(0, 2 * np.pi, len(omega))
            A_spec = mag_s * np.exp(1j * phase)
            A_spec[0] = np.abs(A_spec[0])  # Keep DC real and positive
            a = np.fft.irfft(A_spec, fft_size)[:n]
            a = np.maximum(a, 0.0)
            if a.sum() > 1e-10:
                candidates.append(a / a.sum())
    
    # Pick the best candidate
    best = None
    best_score = float('inf')
    for c in candidates:
        m = fast_autocorr(c).max()
        if m < best_score:
            best_score = m
            best = c
    return best, best_score

def nesterov_subgradient(a0, time_limit, beta_start=10.0, beta_end=500.0,
                         beta_growth=1.008, verbose=True):
    """
    Nesterov-accelerated subgradient method for the log-sum-exp smoothed
    max-autocorrelation objective, with adaptive step size and restarts.
    """
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()  # Nesterov's lookahead point
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    rng = np.random.default_rng(99)
    top_k = max(50, int(n * 0.05))
    t = 0
    alpha = 1.0  # Nesterov momentum
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute gradient at y
        b = fast_autocorr(y)
        idx_top = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros_like(b)
        b_use[idx_top] = b[idx_top]
        b_max = b_use.max()
        
        shifted = np.clip(beta * (b_use - b_max), -300, 300)
        exp_vals = np.exp(shifted)
        p = exp_vals / exp_vals.sum()
        
        # Gradient computation
        N = len(p)
        conv_len = n + N - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        a_rev = y[::-1]
        Ar = np.fft.rfft(a_rev, fft_size)
        Pr = np.fft.rfft(p, fft_size)
        conv = np.fft.irfft(Ar * Pr, fft_size)[:conv_len]
        grad = 2.0 * conv[n-1:2*n-1] / beta
        
        # Adaptive step size with backtracking
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        eta = 0.1 / (grad_norm + 1e-10)
        eta = min(eta, 0.03)
        eta = max(eta, 1e-7)
        
        # Nesterov update
        alpha_new = (1.0 + np.sqrt(1.0 + 4.0 * alpha ** 2)) / 2.0
        y_new = x - eta * grad + ((alpha - 1.0) / (alpha_new + 1.0)) * (y - x)
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y
        else:
            y_new = y_new / s_new
        
        # Check if y_new is better
        m_y = fast_autocorr(y_new).max()
        
        # Projected x update
        x_new = simplex_project(y_new)
        m_x = fast_autocorr(x_new).max()
        
        m_best_trial = min(m_y, m_x)
        a_trial = y_new if m_y <= m_x else x_new
        
        if m_best_trial < best_m - 1e-15:
            best_m = m_best_trial
            best_a = a_trial.copy()
            stall = 0
        else:
            stall += 1
        
        x = x_new
        y = y_new
        alpha = alpha_new
        
        # Adaptive beta
        if t % 30 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        # Restart
        if stall > 2000:
            stall = 0
            noise_scale = min(0.05, 0.01 * (1 + stall / 500))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            y = x_r.copy()
            alpha = 1.0
            beta = max(beta * 0.5, 5.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    Nesterov restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Nesterov t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))
    
    return best_a, best_m


def peak_shaving_v2(a0, time_limit, top_k=30, verbose=True):
    """
    Improved peak shaving: simultaneously consider all top peaks and find
    the global best (source, destination) pair to move mass.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        # Compute total pressure per position
        pressure = np.zeros(n)
        for k_i in peak_indices[:15]:
            ki = int(k_i)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                # contribution to this peak from position i
                contrib = a[idx] * a[ki - idx]
                weight = c[ki] / (c[peak_indices[0]] + 1e-30)
                pressure[idx] += weight * contrib
                pressure[ki - idx] += weight * contrib
        
        # Find the position with highest pressure that can be reduced
        # and the position with lowest pressure that can be increased
        
        # Mask out very small values
        active = a > 1e-10
        
        if not active.any():
            break
        
        # Source: highest pressure among active positions
        pressure_masked = np.where(active, pressure, -np.inf)
        src = int(np.argmax(pressure_masked))
        
        # Destination: lowest pressure, not the source
        pressure_dest = pressure.copy()
        pressure_dest[src] = np.inf
        dst = int(np.argmin(pressure_dest))
        
        if a[src] < 1e-12:
            stall += 1
            continue
        
        # Try different fractions
        improved = False
        for frac in [0.003, 0.008, 0.02, 0.05, 0.1, 0.2]:
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m - 1e-16:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                stall = 0
                improved = True
                break
        
        if not improved:
            stall += 1
            # Random perturbation when stuck
            if stall % 30 == 0:
                n_perturb = max(10, n // 100)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                noise = rng.normal(0, 0.005 * best_a[idx_perturb], n_perturb)
                a_perturb = a.copy()
                a_perturb[idx_perturb] += noise
                a_perturb = np.maximum(a_perturb, 1e-12)
                a_perturb = a_perturb / a_perturb.sum()
                m_perturb = fast_autocorr(a_perturb).max()
                if m_perturb < best_m:
                    best_m = m_perturb
                    best_a = a_perturb.copy()
                    a = a_perturb
                    stall = 0
                else:
                    a = best_a.copy()
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakShave2 (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " stall=" + str(stall))
    
    return best_a, best_m
