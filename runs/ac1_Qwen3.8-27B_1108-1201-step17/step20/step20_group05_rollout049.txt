def fft_conv(a):

    """
    Compute the gradient of (a*a)[peak_idx] with respect to a.
    (a*a)[k] = sum_i a[i] * a[i+k] (with boundary handling)
    d/dx_j (a*a)[k] = 2 * a[k-j] for valid j (when j != k-j)
    = a[j] + a[k-j] adjusted for self-term
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    k = int(peak_idx)
    grad = np.zeros(n, dtype=np.float64)

    # d/dx_j: a[j]*a[j+k] contributes if j+k < n: derivative w.r.t. a[j] is a[j+k], w.r.t. a[j+k] is a[j]

def coordinate_descent(a, c, peak_idx, n, top_k=20, iters_per_round=200, rng=None):
    """
    Coordinate descent: try to move small amounts of mass from high-gradient
    positions to low-gradient positions to reduce the peak at peak_idx.
    """
    if rng is None:
        rng = np.random.default_rng(42)
    
    a = a.copy()
    best_M = float(c[peak_idx])
    no_improve = 0
    
    for _ in range(iters_per_round):
        # Compute gradient at the peak
        g = gradient_at_peak(a, peak_idx)
        # We want to reduce a[j] where g[j] is high (it contributes more to the peak)
        # and increase a[j'] where g[j'] is low
        # But we need to be careful: the move also affects other convolution bins
        
        # Simple approach: pick the source as argmax of g*a (largest contribution)
        # and dest as argmin of g (least impact on this peak)
        contrib = g * a
        src = int(np.argmax(contrib))
        if a[src] < 1e-10:
            src = int(rng.integers(n))
            if a[src] < 1e-10:
                continue
        dst = int(np.argmin(g))
        if dst == src:
            continue
        
        # Try a small move
        frac = 0.01
        t = a[src] * frac
        if t < 1e-12:
            continue
        
        # Apply move
        a[src] -= t
        a[dst] += t
        
        # Check if the peak decreased
        new_peak_val = 0.0
        # Recompute just the peak value
        # (a*a)[peak_idx] = sum_i a[i]*a[i+peak_idx]
        k = int(peak_idx)
        if k <= n - 1:
            idx1 = slice(0, n - k)
            new_peak_val = float(np.dot(a[idx1], a[k:]))
        else:
            k2 = 2 * n - 1 - k
            idx1 = slice(0, n - k2)
            new_peak_val = float(np.dot(a[idx1], a[k2:]))
        
        if new_peak_val < best_M - 1e-15:
            best_M = new_peak_val
            no_improve = 0
        else:
            # Revert
            a[src] += t
            a[dst] -= t
            no_improve += 1
            if no_improve > 50:
                break
    
    return a, best_M

def fast_peak_update(a, c, src, dst, t):
    """
    Update the convolution c after moving t from index src to index dst.
    c_new = c - t*(a shifted by src, added twice) + t*(a shifted by dst, added twice) + t^2 corrections
    Specifically:
    a' = a - t*e_src + t*e_dst
    a'*a' = a*a - 2t*(e_src * a) + 2t*(e_dst * a) + t^2*(e_dst-e_src)*(e_dst-e_src)
    
    (e_j * a)[k] = a[k-j] for max(0, j) <= k < j+n (with a zero-padded)
    So the shift of a by j is: arr[j:j+n] = a, rest is 0.
    
    (e_dst-e_src)*(e_dst-e_src) in convolution = e_{2dst} - 2*e_{dst+src} + e_{2src}
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    c_new = c.copy()
    
    # Subtract 2t * (e_src * a): this is 2t*a placed starting at index src
    s1 = int(src)
    e1 = s1
    e2 = min(s1 + n, size)
    c_new[e1:e2] -= 2.0 * t * a[:e2-e1]
    
    # Add 2t * (e_dst * a): this is 2t*a placed starting at index dst
    s2 = int(dst)
    e1b = s2
    e2b = min(s2 + n, size)
    c_new[e1b:e2b] += 2.0 * t * a[:e2b-e1b]
    
    # Add t^2 * (e_{2dst} - 2*e_{dst+src} + e_{2src})
    t2 = t * t
    idx_2dst = 2 * int(dst)
    if idx_2dst < size:
        c_new[idx_2dst] += t2
    idx_dsrc = int(dst) + int(src)
    if idx_dsrc < size:
        c_new[idx_dsrc] -= 2.0 * t2
    idx_2src = 2 * int(src)
    if idx_2src < size:
        c_new[idx_2src] += t2
    
    c_new = np.where(np.isfinite(c_new) & (c_new > 0.0), c_new, 0.0)
    return c_new

            a[s] -= t
            a[d] += t

            # Revert and redo properly

def multi_start_initialization(n, n_starts=5, rng=None):
    """Generate multiple diverse starting points."""
    if rng is None:
        rng = np.random.default_rng(12345)
    
    candidates = []
    
    # Uniform
    candidates.append(np.ones(n, dtype=np.float64) / n)
    
    # Sine modulated
    for f in [1, 2, 3, 5, 8, 13]:
        x = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * x / n)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        candidates.append(a.copy())
    
    # Chirp
    for f in [2, 5]:
        x = np.arange(n, dtype=np.float64)
        phase = 2 * np.pi * f * (x / n) ** 2
        a = 1.0 + 0.4 * np.cos(phase)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        candidates.append(a.copy())
    
    # Gaussian bumps
    for center in [0.25, 0.5, 0.75]:
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-10 * (x / n - center) ** 2)
        a += 0.1  # floor
        a = a / a.sum()
        candidates.append(a.copy())
    
    # Random
    for _ in range(3):
        a = rng.dirichlet(np.ones(n))
        candidates.append(a)
    
    return candidates


def lse_objective(a, beta, mask=None):
    """Log-sum-exp smoothed objective for the max of the convolution."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    c = fft_conv(a)
    if mask is not None:
        c = c[mask]
    if c.size == 0:
        return 0.0
    m = float(np.max(c))
    shifted = beta * (c - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0 or not math.isfinite(se):
        return m
    return m + math.log(se) / beta


def lse_gradient(a, beta, mask=None):
    """Gradient of the LSE-smoothed max convolution."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    N = next_pow2(3 * n - 2)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    
    if mask is not None:
        cm = c[mask]
    else:
        cm = c
    
    if cm.size == 0:
        return np.zeros(n, dtype=np.float64)
    
    m = float(np.max(cm))
    shifted = beta * (cm - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0:
        return np.ones(n, dtype=np.float64) / n
    p = e / se
    
    # Scatter p back to full size
    pf = np.zeros(size, dtype=np.float64)
    if mask is not None:
        pf[mask] = p
    else:
        pf[:] = p
    
    # Gradient: 2 * (pf * a reversed, cropped)
    # (pf * a_rev)[k] for k = n-1, ..., 2n-2 gives the gradient
    Pr = np.fft.rfft(pf[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:3*n-2]
    grad = 2.0 * conv[n-1:2*n-1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad


def projected_gradient_step(a, grad, step_size):
    """Take a projected gradient step on the simplex."""
    n = a.size
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def run_gradient_descent(a0, beta, time_budget, mask=None, verbose=False):
    """Projected gradient descent with LSE smoothing."""
    a = norm_sum1(a0)
    n = a.size
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = lse_objective(a, beta, mask)
    
    # Adaptive step size
    step = 1.0 / (beta * float(n))
    
    while time.monotonic() - t0 < time_budget:
        grad = lse_gradient(a, beta, mask)
        a_new = projected_gradient_step(a, grad, step)
        obj_new = lse_objective(a_new, beta, mask)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            step = min(step * 1.05, 1.0 / (0.1 * float(n)))
        else:
            step *= 0.7
            if step < 1e-10 / float(n):
                break
        
        if verbose and (time.monotonic() - t0) % 10 < 0.1:
            print("    GD step=" + repr(step) + " obj=" + repr(obj_new))
    
    return best_a

Since `c = a*a` in time domain corresponds to `|A(ω)|²` in frequency domain (where A is the DFT), we have:
`c[k] = (1/N) Σ_ω |A(ω)|² e^{2πiωk/N}`

We want to minimize `max_k c[k]`. This is equivalent to finding a non-negative `a` such that `|A(ω)|²` is as "flat" as possible (in the sense that its inverse DFT has minimal peak).

But `a >= 0` constrains what `|A(ω)|²` can be (it must be the squared magnitude of the DFT of a non-negative sequence). This is a very complex constraint.

1. Start with some `a`.

3. Try to "flatten" `|A|²` by adjusting `a` (when `|A|²` has a peak at frequency ω, the corresponding spatial frequency in `a` should be adjusted).
4. This is essentially what gradient descent does, but we can make it more targeted.

Actually, I think the most practical approach is to improve the coordinate descent significantly. Let me focus on that.

**Key improvement: Batch coordinate descent with the rank-1 update**

The idea:
1. Compute the current convolution `c` and identify the top-K peaks.
2. Compute the "pressure" at each position: how much each position contributes to the active peaks.
3. For each position `j`, find the best "destination" for a small amount of mass from `j`.
4. Apply the best batch of non-conflicting moves simultaneously.

This is faster than one-at-a-time because we can evaluate many moves in parallel (vectorized) and apply a subset.

Let me also think about the **Greedy "peak shaving" algorithm**:
1. Identify the global maximum of the convolution, at position `k*`.
2. Compute the gradient of `c[k*]` with respect to `a`: `g[j] = 2*a[k*-j]` (appropriate shifts).
3. The component of `a` at position `j` that most contributes to `c[k*]` is proportional to `a[j]*g[j]`.
4. Move a small amount from the highest-contribution position to the lowest-contribution position.
5. Recompute the convolution update.
6. Repeat.
