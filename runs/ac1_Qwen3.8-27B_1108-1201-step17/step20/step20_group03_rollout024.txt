"""

2. Staircase beta (temperature) scheduling with exploration via beta decreases

def self_conv(a):
    """Compute linear self-convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_gradient(y, beta, top_k):
    """Compute gradient of smoothed max of self-convolution."""
    n = len(y)
    b = self_conv(y)
    if top_k is not None and top_k < len(b) and top_k > 0:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    if b_max < 1e-30:
        return np.zeros_like(y), b
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b_use)
        p[b_max == b_use] = 1.0 / max(1, (b_max == b_use).sum())
    else:
        p = exp_vals / p_total
    # grad[j] = sum_k p[k] * a[k-j] for valid k-j
    # This is a cross-correlation: grad = correlate(p, y)
    # Using FFT: correlate(p, y)[j] = sum_k p[k] * y[k-j]
    # = convolve(y_flipped, p)[j + n - 1]... let me use direct approach
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1]
    return grad, b


def adaptive_grad_descent(a0, time_limit, beta_start=20.0, beta_end=2000.0,
                          top_k=20, restart_every=3000, verbose=True):
    """
    Adaptive gradient descent with RMSProp and Nesterov momentum.
    """
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_b = self_conv(x)
    best_max_b = best_b.max()
    t0 = time.monotonic()

    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(42)

    # RMSProp state
    rms = np.ones(n) * 1e-8
    rms_decay = 0.9
    eps = 1e-8
    base_lr = 1.0 / np.sqrt(n)

    # Momentum
    momentum = np.zeros(n)
    mu = 0.9

    # Nesterov
    y = x.copy()

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, b = compute_gradient(y, beta, top_k)
        max_b_current = b.max()

        # RMSProp update
        rms = rms_decay * rms + (1 - rms_decay) * (grad ** 2)
        lr = base_lr / (np.sqrt(rms) + eps)

        # Nesterov momentum
        momentum = mu * momentum + (1 - mu) * (grad - mu * (grad - momentum))
        step = lr * momentum

        # Backtracking to ensure decrease
        z = y - step
        x_candidate = simplex_project(z)
        b_candidate = self_conv(x_candidate)
        max_candidate = b_candidate.max()

        if max_candidate > max_b_current * 1.01:
            # Step too large, shrink
            for attempt in range(5):
                step = step * 0.5
                z = y - step
                x_candidate = simplex_project(z)
                b_candidate = self_conv(x_candidate)
                max_candidate = b_candidate.max()
                if max_candidate <= max_b_current * 1.001:
                    break

        x_new = x_candidate

        if max_candidate < best_max_b - 1e-15:
            best_max_b = max_candidate
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        # Adaptive beta
        if t % 50 == 0:
            if stall < 100:
                beta = min(beta * 1.005, beta_end)
            elif stall > restart_every // 2:
                beta = max(beta * 0.85, 5.0)
                base_lr = base_lr * 1.1

        # Restart when stuck
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            momentum = np.zeros(n)
            noise_scale = 0.02 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise_scale, n) * best_x
            x_r = best_x + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 10.0)
            base_lr = 1.0 / np.sqrt(n)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    AGD restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + repr(round(beta, 1)))

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    AGD t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + repr(round(beta, 1)) + " lr=" + repr(round(base_lr, 6)))

    return best_x, best_max_b


def peak_shaving_local_search(a0, time_limit, verbose=True):
    """
    Local search that specifically targets reducing the top peaks of the self-convolution.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = self_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 150:
        b = self_conv(a)
        top_peaks = np.argsort(b)[::-1][:50]
        improved = False

        # For each top peak, try to reduce it
        for k_int in top_peaks[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            # Find the pair (i, k-i) that contributes most to this peak
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find a destination that doesn't contribute to any top peak
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:25]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            # Try multiple destinations
            best_dst = -1
            best_trial_m = best_m
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)

            # Sample a few candidates
            sample_size = min(20, len(candidates))
            sampled = rng.choice(candidates, size=sample_size, replace=False)

            for dst in sampled:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = self_conv(a_trial).max()
                    if M_trial < best_trial_m:
                        best_trial_m = M_trial
                        best_dst = dst
                        best_frac = frac

            if best_dst >= 0 and best_trial_m < best_m - 1e-14:
                amt = a[src] * best_frac
                a[src] -= amt
                a[best_dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_trial_m
                best_a = a.copy()
                improved = True
                break  # Restart peak loop with new b

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.005, 0.05))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = self_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break

        if verbose and stall % 50 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakShave (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=3000, verbose=True):
    """LP-Frank-Wolfe with improved active set management."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = self_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = self_conv(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 600

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = self_conv(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = set(int(k) for k in np.argsort(self_conv(y))[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check for violations and add to active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:20]]
            for k in top_v:
                active_set.add(int(k))

        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.01, default_alpha * 0.5),
                  max(0.01, default_alpha * 2.0), min(0.5, 1.0 / (t + 1))]
        alphas = list(set([max(0.005, min(1.0, a)) for a in alphas]))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = self_conv(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def cyclic_shift_best(a0):
    """Find the best cyclic shift of the sequence."""
    n = len(a0)
    best_m = self_conv(a0).max()
    best_shift = 0
    for s in range(1, min(n, 200)):  # Limit shifts for speed
        a_shift = np.roll(a0, -s)
        m = self_conv(a_shift).max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        # Generate a good initial sequence
        n_base = 6000
        x = np.linspace(0, 1, n_base)
        # Use a skewed distribution (helps reduce self-convolution peak)
        init = np.sin(np.pi * x ** 0.7) ** 2
        best_seq = norm_sum1(init)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # === Phase 1: Adaptive Gradient Descent (main optimizer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: AGD (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 80:
        a_out, _ = adaptive_grad_descent(norm_sum1(best_seq), t1,
                                          beta_start=15.0, beta_end=1500.0,
                                          top_k=20, restart_every=3000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "agd1", t_global)

    # === Phase 2: Cyclic shift ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic1", t_global)

    # === Phase 3: Peak Shaving Local Search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: Peak Shaving (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 50:
        a_out, _ = peak_shaving_local_search(norm_sum1(best_seq), t3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak1", t_global)

    # === Phase 4: LP-Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 40 and n_best <= 10000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t4, solver,
                         initial_K=400, max_K=3000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # === Phase 5: Second AGD pass with different parameters ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 5: AGD v2 (n=" + str(len(best_seq)) + ") ===")
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 50:
        a_out, _ = adaptive_grad_descent(norm_sum1(best_seq), t5,
                                          beta_start=50.0, beta_end=3000.0,
                                          top_k=12, restart_every=2500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "agd2", t_global)

    # === Phase 6: Try a larger n ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    new_n = min(n_best + 2000, 12000)
    if remaining > 200 and new_n > n_best:
        print("\n=== Phase 6: Scale n=" + str(new_n) + " ===")
        a_long = interpolate_sequence(best_seq, new_n)
        t6 = min(remaining * 0.30, 400.0)
        if t6 > 60:
            a_out, _ = adaptive_grad_descent(a_long, t6,
                                              beta_start=20.0, beta_end=1000.0,
                                              top_k=15, restart_every=2500, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "agd_n" + str(new_n), t_global)
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40:
                a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
                best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_n" + str(new_n), t_global)

    # === Phase 7: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 7: Final Peak Shaving (n=" + str(len(best_seq)) + ") ===")
    t7 = min(remaining * 0.25, 180.0)
    if t7 > 30:
        a_out, _ = peak_shaving_local_search(norm_sum1(best_seq), t7, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_final", t_global)

    # === Phase 8: AGD micro-polish with high beta ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 8: AGD polish (n=" + str(len(best_seq)) + ") ===")
    t8 = min(remaining * 0.30, 200.0)
    if t8 > 30:
        a_out, _ = adaptive_grad_descent(norm_sum1(best_seq), t8,
                                          beta_start=100.0, beta_end=5000.0,
                                          top_k=8, restart_every=1500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "agd_polish", t_global)

    # === Final: Cyclic shift ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== Final: Cyclic shift ===")
        a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
