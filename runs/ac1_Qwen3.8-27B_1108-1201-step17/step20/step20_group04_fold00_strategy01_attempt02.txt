import numpy as np

def norm_sum1(a):
    a_np = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a_np.sum()
    if s < 1e-15:
        return np.ones_like(a_np) / len(a_np)
    return a_np / s

def simplex_project(z, tol=1e-6):
    z = np.maximum(z, 0.0)
    s = z.sum()
    if s < 1e-15:
        return np.ones_like(z) / len(z)
    u = np.sort(z)[::-1]  # Decreasing order
    cssv = np.cumsum(u) - s / len(z)
    rho = np.nonzero(u * (np.arange(1, len(z)+1) > (cssv + u[0])/(s/len(z)+1)))[0]
    if len(rho) == 0:
        rho = len(z) - 1
    theta = (cssv[rho] + u[rho]*rho) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def convolve_safe(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project_optimized(z, tol=1e-6):
    n = len(z)
    u = np.sort(np.asarray(z, dtype=np.float64), axis=-1, kind='mergesort', order='F')[::-1]
    budget = np.arange(1, n+1)
    cond = np.cumsum(u) >= 1.0 + (budget * (budget+0.0) / (n*2) if n>1 else 1e-6)
    pivot = np.nonzero(cond)[0][-1] if cond.shape else 0
    theta = (np.cumsum(u)[pivot] - 1.0) / (pivot + 1.0) if pivot > 0 else (np.inf if any(u) else 0.0)
    return np.maximum(z - theta, 0.0)
