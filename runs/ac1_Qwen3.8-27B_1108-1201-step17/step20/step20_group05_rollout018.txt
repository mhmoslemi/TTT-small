def _next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = _next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return np.where(np.isfinite(c) & (c > 0.0), c, 0.0)


def _fft_conv2(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:

    size = 2 * n - 1
    N = _next_pow2(size)
    A = np.fft.rfft(a, N)

    c = np.fft.irfft(A * B, N)[:size]
    return np.where(np.isfinite(c) & (c > 0.0), c, 0.0)

def _interpolate_to_length(a, new_n):
    a = np.asarray(a, dtype=np.float64)

def _lse_obj_grad(a, beta, mask=None):
    """Log-sum-exp smoothed max of convolution and its gradient."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 0.0, np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = _next_pow2(2 * size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)

    if mask is not None and np.any(mask):
        cv = c[mask]
    else:
        cv = c

    if cv.size == 0:
        return 0.0, np.zeros(n, dtype=np.float64)

    m = float(np.max(cv))
    shifted = beta * (cv - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0 or not math.isfinite(se):
        e = np.zeros(cv.size, dtype=np.float64)
        e[0] = 1.0
        se = 1.0
    p = e / se

    # Build full pressure vector
    pf = np.zeros(size, dtype=np.float64)
    if mask is not None and np.any(mask):
        pf[mask] = p
    else:
        pf[:] = p

    # Gradient: grad_i = 2 * sum_k p_k * a[k-i] (valid range)
    N2 = _next_pow2(3 * n - 2)
    Pr = np.fft.rfft(pf[::-1], N2)
    A2 = np.fft.rfft(a, N2)
    conv = np.fft.irfft(A2 * Pr, N2)[:3 * n - 2]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)

    f = m + math.log(se) / beta
    if not math.isfinite(f):
        f = 1e100
    return f, grad


def _softmax_from_logu(u):
    u = np.asarray(u, dtype=np.float64)
    n = u.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(u)):
        return np.ones(n, dtype=np.float64) / float(n)
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def _make_log_coords(a):
    """Convert a to log coordinates with one reference fixed at 0."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 1:
        return np.array([], dtype=np.float64), 0
    ref = int(np.argmax(a))
    a_safe = np.maximum(a, 1e-300)
    loga = np.log(a_safe) - math.log(a_safe[ref])
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.where(np.isfinite(v), v, -700.0)
    return v, ref


def _from_log_coords(v, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    return _softmax_from_logu(u)


def _obj_in_log_coords(v, beta, mask=None, ref=0):
    n = v.size + 1
    if n <= 0:
        return 1e100, np.array([], dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return 1e100, np.zeros(v.size, dtype=np.float64)

    a = _from_log_coords(v, ref)
    f, ga = _lse_obj_grad(a, beta, mask)
    if not math.isfinite(f):
        f = 1e100
        ga = np.zeros(n, dtype=np.float64)

    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)

    gv = np.empty(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]

    gv = np.where(np.isfinite(gv), gv, 0.0)
    return f, gv


def _run_lbfegs(a0, beta, mask, maxfun=400, verbose=False):

    if n <= 1:

def _game_direction(f, active_idx, mw_iters=30, seed=0):
    """Game-theoretic Frank-Wolfe direction."""
    f = _norm_sum1(f)
    n = f.size
    size = 2 * n - 1
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K <= 0:
        return np.ones(n, dtype=np.float64) / float(n)

    rng = np.random.default_rng(seed)
    logits = np.zeros(K, dtype=np.float64) + rng.normal(0, 0.1, K)

    for _ in range(max(1, int(mw_iters))):
        # Player 1: choose distribution over active peaks
        loc = np.where(np.isfinite(logits), logits, -1000.0)
        loc = loc - float(np.max(loc))
        w = np.exp(loc)
        ws = float(w.sum())
        if ws <= 0:
            w = np.ones(K) / K
        else:
            w = w / ws

        pf = np.zeros(size, dtype=np.float64)
        pf[active_idx] = w

        # Player 2 best response: minimize pressure
        N = _next_pow2(3 * n - 2)
        A = np.fft.rfft(f, N)
        Pr = np.fft.rfft(pf[::-1], N)
        conv = np.fft.irfft(A * Pr, N)[:3 * n - 2]
        pressure = conv[n - 1:2 * n - 1][::-1]
        pressure = np.where(np.isfinite(pressure) & (pressure > 0), pressure, 0.0)

        pmx = float(np.max(pressure))
        pmn = float(np.min(pressure))
        ptp = pmx - pmn
        if ptp <= 1e-30:
            g = np.ones(n) / n
        else:
            scaled = (pressure - pmn) / ptp
            g = np.exp(-6.0 * scaled)
            g = np.where(np.isfinite(g), g, 0.0)
            g += 1e-12
            g = g / g.sum()
            g = 0.95 * g + 0.05 * (np.ones(n) / n)
            g = g / g.sum()

        # Update logits
        h = _fft_conv2(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmax = float(np.max(ha))
        hmin = float(np.min(ha))
        htp = hmax - hmin
        if htp > 1e-30:
            eta = 10.0 / htp
            logits += eta * (ha - float(np.mean(ha)))

    return g


def _fw_search(a0, budget, active_max=2000, verbose=True):
    """Frank-Wolfe search for peak minimization."""
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = _norm_sum1(a0)
    n = a.size
    c = _fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    active = np.argsort(c)[::-1][:min(active_max, c.size)]

    no_improve = 0
    t = 0
    while time.monotonic() - t0 < time_limit - 3 and no_improve < 60:
        t += 1
        time_left = time_limit - (time.monotonic() - t0)
        if time_left < 10:
            break

        g = _game_direction(a, active, mw_iters=25, seed=2000 + t)
        if g is None:
            break

        cg = _fft_conv(g)
        fg = _fft_conv2(a, g)

        # Line search
        best_alpha = 0.0
        best_m = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            one = 1.0 - alpha
            val = (one * one) * c + 2.0 * alpha * one * fg + (alpha * alpha) * cg
            m = float(np.max(val))
            if math.isfinite(m) and m < best_m:
                best_m = m
                best_alpha = alpha

        if best_alpha > 0 and best_m < M - 1e-16:
            new_a = (1.0 - best_alpha) * a + best_alpha * g
            new_a = np.maximum(new_a, 0.0)
            s = float(new_a.sum())
            if s > 1e-15:
                new_a /= s
                c = _fft_conv(new_a)
                M = float(np.max(c))
                a = new_a
                no_improve = 0
                if M < best_M - 1e-15:
                    best_M = M
                    best_a = a.copy()
                active = np.argsort(c)[::-1][:min(active_max, c.size)]
        else:
            no_improve += 1
            if no_improve % 10 == 0:

                M = float(np.max(c))
                active = np.argsort(c)[::-1][:min(active_max, c.size)]

def _coord_polish(a0, budget, verbose=False):
    """Coordinate descent with rank-1 convolution updates."""
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = _norm_sum1(a0)
    n = a.size
    c = _fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    stall = 0
    rnd = 0
    rng = np.random.default_rng(4242)

    while time.monotonic() - t0 < time_limit - 2 and stall < 80:
        rnd += 1
        if n <= 1:
            break

        # Focus on top peaks
        top_k = min(30, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]

        # Compute pressure from top peaks
        p_top = np.zeros(2 * n - 1, dtype=np.float64)
        p_top[top_idx] = 1.0 / top_k

        N = _next_pow2(3 * n - 2)
        A = np.fft.rfft(a, N)
        Pr = np.fft.rfft(p_top[::-1], N)
        conv = np.fft.irfft(A * Pr, N)[:3 * n - 2]
        pressure = 2.0 * conv[n - 1:2 * n - 1][::-1]
        pressure = np.where(np.isfinite(pressure) & (pressure > 0), pressure, 0.0)

        # Find best single move
        best_cand = None
        improved = False

        # Try moving from high-pressure source to low-pressure destination
        src_order = np.argsort(pressure)[::-1]
        dst_order = np.argsort(pressure)

        for _ in range(50):
            s = int(src_order[int(rng.integers(min(10, n)))])
            d = int(dst_order[int(rng.integers(min(10, n)))])
            if s == d or a[s] < 1e-14:
                continue
            for frac in (0.005, 0.01, 0.02, 0.05, 0.1):
                tm = a[s] * frac
                tm = min(tm, a[s] * 0.95)
                if tm <= 1e-16:
                    continue
                # Rank-1 update
                c_new = c.copy()
                c_new[d:d + n] += 2.0 * tm * a
                c_new[s:s + n] -= 2.0 * tm * a
                if 2 * d < c_new.size:
                    c_new[2 * d] += tm * tm
                if 2 * s < c_new.size:
                    c_new[2 * s] += tm * tm
                if s + d < c_new.size:
                    c_new[s + d] -= 2.0 * tm * tm

                m_new = float(np.max(c_new))
                if m_new < M - 1e-18:
                    if best_cand is None or m_new < best_cand[0]:
                        best_cand = (m_new, s, d, tm)
                        improved = True

        if best_cand is not None:
            m_new, s, d, tm = best_cand
            a[s] -= tm
            a[d] += tm
            a = np.maximum(a, 0.0)
            ssum = float(a.sum())
            if ssum < 1e-15:
                break
            a = a / ssum
            c = _fft_conv(a)
            M = float(np.max(c))
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 8 == 7:
                # Random exploration
                for _ in range(50):
                    s = int(rng.integers(n))
                    d = int(rng.integers(n))
                    if s == d or a[s] < 1e-14:
                        continue
                    tm = a[s] * float(rng.uniform(0.002, 0.04))
                    tm = min(tm, a[s] * 0.9)
                    if tm <= 1e-16:
                        continue
                    c_new = c.copy()
                    c_new[d:d + n] += 2.0 * tm * a
                    c_new[s:s + n] -= 2.0 * tm * a
                    if 2 * d < c_new.size:
                        c_new[2 * d] += tm * tm
                    if 2 * s < c_new.size:
                        c_new[2 * s] += tm * tm
                    if s + d < c_new.size:
                        c_new[s + d] -= 2.0 * tm * tm
                    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
                    m_new = float(np.max(c_new))
                    if m_new < M - 1e-18:
                        a[s] -= tm
                        a[d] += tm
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        c = _fft_conv(a)
                        M = float(np.max(c))
                        if M < best_M:
                            best_M = M
                            best_a = a.copy()
                        stall = 0
                        break

    return best_a


def _peak_equalization(a0, budget, top_k=20, verbose=False):
    """Try to equalize the top peaks using projected gradient."""
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = _norm_sum1(a0)
    n = a.size
    c = _fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M

    while time.monotonic() - t0 < time_limit - 2:
        active = np.argsort(c)[::-1][:top_k]
        c_active = c[active]
        c_max = float(np.max(c_active))
        c_mean = float(np.mean(c_active))
        spread = c_max - c_mean

        if spread < 1e-8 * c_max:
            break

        # Gradient direction: push down the highest, push up the rest
        # For each active peak k, grad_k = a reversed and shifted
        # We want to decrease max and increase min
        weights = c_active - c_mean  # positive for peaks above mean

        # Compute negative gradient of the max (push down)
        # and positive gradient of the min (push up)
        N = _next_pow2(3 * n - 2)
        A = np.fft.rfft(a, N)

        # Build weighted pressure
        pf = np.zeros(2 * n - 1, dtype=np.float64)
        pf[active] = weights
        Pr = np.fft.rfft(pf[::-1], N)
        conv = np.fft.irfft(A * Pr, N)[:3 * n - 2]
        grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
        grad = np.where(np.isfinite(grad), grad, 0.0)

        # Projected gradient step (stay on simplex)
        step_sizes = [0.001, 0.005, 0.01, 0.02, 0.05]
        improved = False
        for step in step_sizes:
            new_a = a - step * grad
            new_a = np.maximum(new_a, 0.0)
            s = float(new_a.sum())
            if s < 1e-15:
                continue
            new_a = new_a / s
            c_new = _fft_conv(new_a)
            M_new = float(np.max(c_new))
            if M_new < M - 1e-16:
                a = new_a
                c = c_new
                M = M_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
                break

        if not improved:
            # Try small perturbations
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            pert = rng.normal(0, 0.01, n) * a
            new_a = a + pert
            new_a = np.maximum(new_a, 0.0)
            s = float(new_a.sum())
            if s > 1e-15:
                new_a /= s
                c_new = _fft_conv(new_a)
                M_new = float(np.max(c_new))
                if M_new < M:
                    a = new_a
                    c = c_new
                    M = M_new
                    if M < best_M:
                        best_M = M
                        best_a = a.copy()

    return best_a


def _default_sequence(n=5000):

    # Smooth bump with some structure
    a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.1) * (1.0 + 0.3 * np.sin(4.0 * np.pi * x))
    a = np.maximum(a, 0.0)
    return _norm_sum1(a)


def _random_sequence(n, seed=0):
    rng = np.random.default_rng(seed)
    a = rng.lognormal(0, 0.5, size=n)
    return _norm_sum1(a)


def _optimize_at_length(parent, n, budget, verbose=False):
    """Try to optimize a sequence of length n starting from parent."""
    a = _interpolate_to_length(parent, n)
    score0 = _safe_eval(a)
    if not math.isfinite(score0):
        score0 = _proxy_score(a)

    best_a = a.copy()
    best_score = score0

    # Phase 1: L-BFGS with increasing beta
    if SCIPY_OK and budget > 30:
        t0 = time.monotonic()
        betas = [10.0, 40.0, 100.0, 300.0, 800.0]
        n_betas = len(betas)
        per_budget = budget * 0.35 / n_betas
        for beta in betas:
            if time.monotonic() - t0 > budget * 0.4:
                break
            a_new = _run_lbfegs(best_a, beta, None,
                                maxfun=int(max(50, per_budget / 0.05)),
                                verbose=False)
            sc = _proxy_score(a_new)
            if math.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new
            elif math.isfinite(sc) and sc < best_score + 0.01:
                best_a = a_new  # Accept small increases for exploration

        # Verify with actual evaluator
        actual = _safe_eval(best_a)
        if math.isfinite(actual) and actual < best_score:
            best_score = actual

    # Phase 2: Peak equalization
    if budget > 60:
        t0 = time.monotonic()
        eq_budget = min(budget * 0.25, 200.0)
        a_eq = _peak_equalization(best_a, eq_budget, top_k=20, verbose=False)
        sc = _safe_eval(a_eq)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_eq

    # Phase 3: Frank-Wolfe
    if budget > 80:
        t0 = time.monotonic()
        fw_budget = min(budget * 0.25, 250.0)
        a_fw = _fw_search(best_a, fw_budget, active_max=1500, verbose=False)
        sc = _safe_eval(a_fw)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_fw

    # Phase 4: Coordinate polish
    if budget > 40:
        t0 = time.monotonic()
        cp_budget = min(budget * 0.20, 150.0)
        a_cp = _coord_polish(best_a, cp_budget, verbose=False)
        sc = _safe_eval(a_cp)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_cp

    return best_a, best_score


def _multi_start_optimize(parent, n, budget, verbose=False):
    """Try multiple starting points at length n."""
    a_parent = _interpolate_to_length(parent, n)
    best_a = a_parent.copy()
    best_score = _safe_eval(a_parent)
    if not math.isfinite(best_score):
        best_score = _proxy_score(a_parent)

    # Starting points: parent interp, random, structured
    starts = [a_parent.copy()]

    # A few random starts
    for seed in [1, 7, 42]:
        starts.append(_random_sequence(n, seed=seed))

    # Structured: sinusoidal modulation of uniform
    x = np.linspace(0, 1, n)
    for freq in [2, 3, 5]:
        a_struct = np.ones(n) * (1.0 + 0.3 * np.cos(2 * np.pi * freq * x))
        starts.append(_norm_sum1(np.maximum(a_struct, 0.01)))

    per_budget = budget / len(starts)
    time_left = budget

    for i, start in enumerate(starts):
        if time.left() if False else (time_left < 10):
            break
        actual_budget = min(per_budget, time_left - 5)
        if actual_budget < 5:
            break
        a_opt, sc = _optimize_at_length(start, n, actual_budget, verbose=False)
        if sc < best_score:
            best_score = sc
            best_a = a_opt
        time_left = budget - (time.monotonic() - time.monotonic() + 0)  # just track

    return best_a, best_score


def propose_candidate():
    t_start = time.monotonic()
    TOTAL_BUDGET = 1720.0  # Leave margin for safety

    print("Starting optimization, budget=" + str(TOTAL_BUDGET))

    # Get parent sequence
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        parent_raw = np.where(np.isfinite(parent_raw) & (parent_raw >= 0), parent_raw, 0.0)
        if parent_raw.ndim != 1:
            parent_raw = parent_raw.ravel()
        if parent_raw.size == 0 or float(parent_raw.sum()) < 0.01:
            raise ValueError("Empty parent")
    except Exception:
        parent_raw = _default_sequence(5000)

    parent = _norm_sum1(parent_raw)
    parent_n = parent.size
    parent_score = _safe_eval(parent)
    print("Parent: n=" + str(parent_n) + " score=" + str(parent_score))

    best_seq = parent.copy()
    best_score = parent_score

    # Strategy: try a few lengths, with most time on the best one
    # First, quick probe of lengths
    probe_budget = min(60.0, TOTAL_BUDGET * 0.05)
    probe_lengths = sorted(set([
        parent_n,
        max(1000, parent_n - 500),
        parent_n + 500,
        max(1000, parent_n - 1000),
        parent_n + 1000,
    ]))
    probe_lengths = [l for l in probe_lengths if 500 <= l <= 15000]

    print("Probing lengths: " + str(probe_lengths))
    length_scores = {}
    t0 = time.monotonic()
    for n in probe_lengths:
        if time.monotonic() - t0 > probe_budget:
            break
        a = _interpolate_to_length(parent, n)
        sc = _safe_eval(a)
        length_scores[n] = sc
        print("  n=" + str(n) + " score=" + str(sc))

    # Sort lengths by probe score
    sorted_lengths = sorted(length_scores.keys(), key=lambda x: length_scores[x])

    # Main optimization: spend most time on best length, some on others
    elapsed = time.monotonic() - t_start
    remaining = TOTAL_BUDGET - elapsed
    print("Remaining after probe: " + str(remaining))

    # Spend 70% on best length, 15% on second, 10% on third, 5% on fourth
    allocations = [remaining * 0.60, remaining * 0.15, remaining * 0.10, remaining * 0.10, remaining * 0.05]

    for li, n in enumerate(sorted_lengths[:5]):
        if li >= len(allocations):
            break
        bud = allocations[li]
        elapsed = time.monotonic() - t_start
        bud = min(bud, TOTAL_BUDGET - elapsed - 10)
        if bud < 10:
            break

        print("\nOptimizing n=" + str(n) + " budget=" + str(bud))
        t0 = time.monotonic()

        # For the best length, do multi-start
        if li == 0:
            a_opt, sc = _multi_start_optimize(parent, n, bud, verbose=True)
        else:
            a_opt, sc = _optimize_at_length(parent, n, bud, verbose=False)

        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt
            print("  NEW BEST: score=" + str(sc) + " n=" + str(n))
        print("  Done n=" + str(n) + " best_score=" + str(sc) +
              " time=" + str(round(time.monotonic() - t0, 1)))

    # Final polish on the best sequence found
    elapsed = time.monotonic() - t_start
    remaining = TOTAL_BUDGET - elapsed - 5
    if remaining > 20:
        print("\nFinal polish: budget=" + str(remaining))
        n_best = best_seq.size
        a_polish = _coord_polish(best_seq, min(remaining * 0.5, 100.0))
        sc = _safe_eval(a_polish)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_polish
            print("  Polish improved: " + str(sc))

        elapsed = time.monotonic() - t_start
        remaining = TOTAL_BUDGET - elapsed - 5
        if remaining > 20 and SCIPY_OK:
            a_lb = _run_lbfegs(best_seq, 200.0, None, maxfun=200)
            sc = _safe_eval(a_lb)
            if math.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_lb
                print("  Final L-BFGS improved: " + str(sc))

    # Also try: if best is from a different length, try interpolating back
    if best_seq.size != parent_n and best_score < parent_score:
        # Try a length between
        mid_n = (best_seq.size + parent_n) // 2
        elapsed = time.monotonic() - t_start
        remaining = TOTAL_BUDGET - elapsed - 5
        if remaining > 30:
            a_mid = _interpolate_to_length(best_seq, mid_n)
            a_opt, sc = _optimize_at_length(a_mid, mid_n, min(remaining * 0.5, 80.0))
            if math.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt
                print("  Mid-length improved: " + str(sc))

    # Final validation
    final_val = _safe_eval(best_seq)
    if not math.isfinite(final_val):
        best_seq = parent
        final_val = parent_score

    arr = np.asarray(best_seq, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if arr.size == 0 or s < 0.01:
        arr = parent
    else:
        arr = arr / s

    elapsed = time.monotonic() - t_start
    print("\nFINAL: score=" + str(final_val) + " n=" + str(arr.size) +
          " elapsed=" + str(round(elapsed, 1)) + "s")

    return [float(x) for x in arr]
