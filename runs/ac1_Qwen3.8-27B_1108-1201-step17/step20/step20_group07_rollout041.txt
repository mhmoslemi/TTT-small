```python
    N = 1 << (2 * (2 * n - 1)).bit_length()  # power of 2 >= 4n-2
    P = np.zeros(N, dtype=complex)
    P[:2*n-1] = 1.0
    P_fft = np.fft.fft(P)
    # Take square root, choosing branch with positive value at DC
    P_sqrt = np.sqrt(P_fft)
    # Ensure DC is positive
    if P_sqrt[0].real < 0:
        P_sqrt = -P_sqrt
    # Inverse FFT
    Q = np.fft.ifft(P_sqrt)
    # Take real part, first n elements
    q = Q.real[:n]
    # Clean up small negative values
    q = np.maximum(q, 0.0)
    # Normalize
    s = q.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return q / s
```

Minimizes 2*n*max(autocorr(a))/(sum(a))^2 using:
1. FFT-based square root initialization (spectral flat convolution target)

5. Adaptive beta (smoothing parameter) schedule

import numpy as np
import time

def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fft_sqrt_init(n):
    """Initialize a sequence whose autocorrelation is as flat as possible.
    
    Computes the first n coefficients of sqrt(1 + x + x^2 + ... + x^{2n-2}).
    """
    size = 2 * n - 1
    # Need N >= 2*size - 1 for the square root computation to be correct
    # Actually, we need N >= size for the FFT of P, and the IFFT gives us Q
    # where Q*Q = P (circularly), so we need N >= 2*size - 1 to avoid aliasing
    N = 1 << (2 * size - 1).bit_length()
    P = np.zeros(N, dtype=np.float64)
    P[:size] = 1.0
    
    P_fft = np.fft.fft(P)
    # Complex square root
    P_sqrt = np.sqrt(P_fft)
    # Ensure DC component is positive real
    if P_sqrt[0].real < 0:
        P_sqrt = -P_sqrt
    Q = np.fft.ifft(P_sqrt).real
    a = Q[:size]  # Q should have length size (2n-1), take all
    # We want the first n coefficients
    a = a[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def simplex_project(z):
    """Project vector z onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1] + 1
    theta = (cssv[rho - 1] - 1) / rho
    return np.maximum(z - theta, 0.0)

def compute_gradient(a, p):
    """Compute gradient of smoothed max autocorrelation.
    
    grad[i] = 4*n * sum_j a[j] * p[j+i]
    """
    n = len(a)
    size = 2 * n - 1
    q = p[::-1]  # reverse p
    # conv = a * q (linear convolution, length n + size - 1 = 3n-2)
    conv_size = n + size - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    Q = np.fft.rfft(q, fft_size)
    conv = np.fft.irfft(A * Q, fft_size)[:conv_size]
    grad = 4.0 * n * conv[n-1:2*n-1][::-1]
    return grad

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
