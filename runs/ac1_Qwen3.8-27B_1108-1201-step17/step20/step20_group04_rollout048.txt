def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def topk_gradient(y, top_k):
    """Compute subgradient of sum of top-k autocorrelation values."""
    n = len(y)
    c = fast_autocorr(y)
    if top_k < len(c):
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        p = np.zeros(len(c))
        p[top_idx] = 1.0 / top_k
    else:
        p = np.ones(len(c)) / len(c)
    
    # Gradient: g[i] = 2 * sum_k p[k] * a[k-i]
    # = 2 * (cross-correlation of p with y)[i + n - 1]
    fft_size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    Y_rev = np.fft.rfft(y[::-1], fft_size)
    Conv = np.fft.irfft(P * Y_rev, fft_size)
    grad = 2.0 * Conv[n - 1:2 * n - 1]
    return grad, c, top_idx if top_k < len(c) else None

def nesterov_topk(a0, time_limit, k_start=100, k_end=5,

    t0 = time.monotonic()
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    k = k_start
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        # Anneal k
        if t % 50 == 0:
            k = max(k_end, int(k * k_decay))
        
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            # Spectral restart: perturb in Fourier domain
            noise_scale = 0.02 / (1.0 + 0.1 * (t // restart_every))
            # Random phase perturbation in frequency domain
            A = np.fft.rfft(best_a)
            # Perturb the magnitude slightly

            phase = np.angle(A)

            a_pert = a_pert / a_pert.sum()

def mirror_descent(a0, time_limit, top_k=50, eta_start=0.1, eta_end=0.001,
                   decay=0.999, verbose=True, seed=99):
    """Mirror descent (multiplicative updates) with top-k gradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_max_b = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = eta_start
    stall = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        grad, c, _ = topk_gradient(a, top_k)
        
        # Multiplicative update: a_new[i] = a[i] * exp(-eta * grad[i])
        # This automatically maintains positivity
        log_update = -eta * grad
        log_update = np.clip(log_update, -10, 10)  # numerical stability
        a_new = a * np.exp(log_update)
        
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Project to simplex (clip very small values)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        b_new = fast_autocorr(a_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        eta = max(eta * decay, eta_end)
        
        if stall > 2000:
            break
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Mirror t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " eta=" + str(round(eta, 6)))
    
    return best_a, best_max_b


def exact_1d_optimize(a, i, steps=20):
    """Find optimal value for coordinate i by 1D search, maintaining sum=1."""
    n = len(a)
    # We'll move mass between position i and a "pool" of other positions
    # Simple approach: try different values of a[i] and redistribute the rest proportionally
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    orig_val = a[i]
    # Try reducing a[i] by various fractions and redistributing
    for frac in np.linspace(0, 0.3, steps):
        new_val = orig_val * (1 - frac)
        if new_val < 0:
            continue
        a_trial = a.copy()
        a_trial[i] = new_val
        # Redistribute the removed mass to other positions proportionally
        removed = orig_val - new_val
        if removed < 1e-15:
            continue
        # Give to positions with low contribution to peaks
        others = np.ones(n)
        others[i] = 0
        others = np.maximum(others * a, 1e-15)  # proportional to current values
        others = others / others.sum()
        a_trial += removed * others
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s
        m = fast_autocorr(a_trial).max()
        if m < best_m:
            best_m = m
            best_a = a_trial.copy()
    
    return best_a, best_m


def cyclic_cd(a0, time_limit, top_peaks=10, n_trials=5, verbose=True, seed=77):
    """Cyclic coordinate descent: for each position, try to reduce the max peak."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    overall_stall = 0
    
    while time.monotonic() - t0 < time_limit and overall_stall < 200:
        c = fast_autocorr(a)
        # Find positions contributing most to the top peaks
        top_k = min(top_peaks, len(c))
        peak_idx = np.argsort(c)[::-1][:top_k]
        
        # For each peak, identify the top contributing pairs
        contrib = np.zeros(n)
        for k in peak_idx:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib[idx] += a[k_int - idx]
            contrib[k_int - idx] += a[idx]
        
        # Try to reduce the most contributing positions
        top_contrib = np.argsort(contrib)[::-1][:20]
        
        improved = False
        for i in top_contrib:
            i_int = int(i)
            if a[i_int] < 1e-12:
                continue
            
            # Try moving small amounts from i to various destinations
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02]:
                amt = a[i_int] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i_int] -= amt
                # Distribute to n_random positions
                for dst in rng.choice(n, size=n_trials, replace=False):
                    dst_int = int(dst)
                    if dst_int == i_int:
                        continue
                    a_try = a_trial.copy()
                    a_try[dst_int] += amt
                    a_try = np.maximum(a_try, 0.0)
                    s = a_try.sum()
                    if s < 1e-15:
                        continue
                    a_try = a_try / s
                    m = fast_autocorr(a_try).max()
                    if m < best_m - 1e-16:
                        a = a_try
                        best_m = m
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                overall_stall = 0
                break
        
        if not improved:
            overall_stall += 1
        
        if verbose and overall_stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    CCD (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(overall_stall))
    
    return best_a, best_m

def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def evaluate_safe(a):

    except Exception:

def norm_sum1(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_topk_grad(y, k):
    """Compute exact subgradient of sum of top-k autocorrelation values.
    Returns (gradient, c, max_c)."""
    n = len(y)
    c = fast_autocorr(y)
    n_c = len(c)
    
    if k < n_c:
        # Get top-k indices
        top_idx = np.argpartition(c, -k)[-k:]
        p = np.zeros(n_c)
        p[top_idx] = 1.0 / k

    Conv = np.fft.irfft(P * Y_rev, fft_size)
    grad = 2.0 * Conv[n - 1:2 * n - 1]

        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        grad, c, max_c = compute_topk_grad(y, k)
        
        # Adaptive step size
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-18:
            eta_t = eta_base
        else:
            eta_t = eta_base / (1.0 + 0.5 * g_norm)
        eta_t = min(eta_t, 0.05)
        
        z = y - eta_t * grad
        x_new = simplex_project(z)
        
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        # Anneal k (reduce to focus on top peaks)
        if t % 30 == 0:
            k = max(k_end, int(k * k_decay))
        
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            n_restarts += 1
            # Structured restart
            noise_scale = 0.015 / (1.0 + 0.08 * n_restarts)
            
            # Mix of anti-symmetric, spectral, and random perturbations
            anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
            anti = anti - anti.mean()
            
            A = np.fft.rfft(best_a)
            mag = np.abs(A)
            phase = np.angle(A)
            mag_p = mag * (1 + noise_scale * 0.5 * rng.normal(0, 1, len(mag)))
            mag_p = np.maximum(mag_p, 0)
            a_spec = np.fft.irfft(mag_p * np.exp(1j * phase), 2 * n)[:n]
            a_spec = np.maximum(a_spec, 1e-10)
            a_spec = a_spec / a_spec.sum()
            
            rand = rng.normal(0, 1, n)
            
            blend = rng.uniform(0, 1)
            if blend < 0.33:
                pert = noise_scale * anti * best_a
            elif blend < 0.66:
                pert = noise_scale * (a_spec - best_a)
            else:
                pert = noise_scale * rand * best_a
            
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            k = min(k_start, int(k * 2.0))
            
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b

    return best_a, best_max_b
