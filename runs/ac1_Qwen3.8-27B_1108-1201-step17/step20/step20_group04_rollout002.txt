def lp_perturbation(a, M, n_constraints=200):
    """
    Given a normalized sequence a and target M, solve:
    max 1^T g s.t. (conv(a,g))_k <= M for top n_constraints peaks, g >= 0
    Returns the optimal g (or None if infeasible).
    """
    n = len(a)
    c = fast_autocorr(a)
    # Get top constraint positions
    top_idx = np.argsort(c)[::-1][:n_constraints]
    
    # Build constraint matrix: for each peak position k, the constraint is
    # sum_i a[i] * g[k-i] <= M
    # where g[j] = 0 for j outside [0, n-1]
    # So the constraint coefficients for g[j] are: a[k-j] if 0<=k-j<n
    # i.e., coefficient of g[j] in constraint k is a[k-j] (if valid)
    
    # This is a matrix of size n_constraints x n
    A_mat = np.zeros((n_constraints, n))
    for row, k in enumerate(top_idx):
        k_int = int(k)
        # g[j] contributes a[k-j] to constraint k if 0 <= k-j < n
        # i.e., j ranges from max(0, k-n+1) to min(n-1, k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            A_mat[row, j_start:j_end+1] = a[k_int - (j_start:j_end+1)]
    
    # Solve LP: max 1^T g s.t. A_mat @ g <= M, g >= 0
    try:
        from scipy.optimize import linprog
        # Maximize 1^T g => minimize -1^T g
        c_obj = -np.ones(n)
        A_ub = A_mat
        b_ub = np.full(n_constraints, M)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            if g.sum() > 1e-10:
                return g / g.sum()
    except Exception as e:
        print("  LP error: " + str(e))
    return None

        if n_peaks >= len(c):
            peak_positions = np.array(range(len(c)))
        else:
            peak_positions = np.argsort(c)[::-1][:n_peaks]
        
        # Compute combined subgradient
        grad = np.zeros(n)
        for k in peak_positions:
            k_int = int(k)
            grad += subgradient_at_peak(y, k_int, n)
        grad /= len(peak_positions)
        
        # Adaptive step size
        eta = best_eta / t_new
        # Backtracking: try the step, if max increased, reduce eta
        z = y - eta * grad
        x_new = simplex_project(z)
        
        # Check improvement
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        
        if max_new > c_max * (1 + 1e-10):
            # Step too large, reduce
            for _ in range(5):
                eta *= 0.5
                z = y - eta * grad
                x_new = simplex_project(z)
                b_new = fast_autocorr(x_new)
                max_new = b_new.max()
                if max_new <= c_max * (1 + 1e-10):
                    break
            best_eta = eta
        else:
            best_eta = min(eta * 1.05, 0.1)
        
        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        # Periodic restart
        if stall > 3000:
            stall = 0
            t_nest = 1.0
            noise = 0.02 / (1.0 + 0.1 * (it // 3000))
            x_r = best_a * (1.0 + noise * rng.normal(0, 1, n))
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    NestSG restart (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    NestSG it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " eta=" + repr(round(best_eta, 6)))
    
    return best_a, best_max_b


def lp_perturb_step(a, M, n_constraints=100, verbose=False):
    """Solve LP to find direction that maintains conv constraints."""
    n = len(a)
    c = fast_autocorr(a)
    top_idx = np.argsort(c)[::-1][:n_constraints]
    
    # Build constraint matrix
    A_mat = np.zeros((n_constraints, n))
    for row, k in enumerate(top_idx):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            A_mat[row, j_start:j_end+1] = a[k_int - (j_start:j_end+1)]
    
    # LP: max 1^T g s.t. A_mat @ g <= M, g >= 0
    try:
        from scipy.optimize import linprog
        c_obj = -np.ones(n)
        A_ub = A_mat
        b_ub = np.full(n_constraints, M * 1.001)  # Tiny tolerance
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x.sum() > 1e-10:
            g = res.x / res.x.sum()
            if verbose:
                print("    LP perturb: sum(g)=" + repr(res.x.sum()) + ", obj=" + repr(res.fun))
            return g
    except Exception as e:
        if verbose:
            print("    LP error: " + str(e))
    return None


def spectral_randomize(a, strength=0.1, verbose=False):
    """Add random phase perturbation in frequency domain, project back to non-negative."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    # Randomize phase
    phase = np.exp(1j * strength * np.random.default_rng(0).uniform(0, 2*np.pi, len(A)))

    rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
    phase = np.exp(1j * strength * rng.uniform(0, 2*np.pi, len(A)))
    A_new = np.abs(A) * phase
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)

    return a_new / s
