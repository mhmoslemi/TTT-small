import numpy as np

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Project to probability simplex (non-negative, sums to 1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Algorithm from Wang & Carreira-Perpinan (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_max_objective(a, T):
    """Compute T * log(sum(exp(c/T))) where c = autocorr(a)."""
    c = fast_autocorr(a)
    c_max = c.max()
    return T * np.log(np.sum(np.exp((c - c_max) / T))) + c_max


def soft_max_gradient(a, T):
    """Compute gradient of T*log(sum(exp(c/T))) w.r.t. a."""
    n = len(a)
    c = fast_autocorr(a)
    c_max = c.max()
    p = np.exp((c - c_max) / T)
    p_sum = p.sum()
    p = p / p_sum
    # Grad[j] = sum_m p[m+j] * a[m] for j = 0..n-1
    # This is cross-correlation of p and a, first n elements
    N = 1 << (3 * n - 3).bit_length()
    P = np.fft.fft(p, N)
    A = np.fft.fft(a, N)
    grad = np.fft.ifft(P * np.conj(A)).real[:n]
    return grad


def exact_objective(a):
    """Compute 2*n*max(autocorr(a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_autocorr(a)
    return 2.0 * n * c.max() / (s * s)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (accelerated proximal gradient) with log-sum-exp smoothed objective
2. Geometric temperature annealing from smooth to sharp
3. Exact subgradient descent with active set tracking
4. Efficient mass-transfer coordinate descent
5. Multi-start with diverse initializations

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear convolution conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Project to probability simplex."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def project_simplex(v):
    """Project vector onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def exact_obj(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    c = fast_autocorr(a)
    return 2.0 * n * c.max()


def eval_safe(a):
    """Safely evaluate using the provided evaluate_sequence function."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_soft_gradient(a, c, T):
    """
    Compute gradient of T*log(sum(exp(c/T))) w.r.t. a.
    c = conv(a,a) already computed.
    Uses FFT for efficiency.
    """
    n = len(a)
    c_max = c.max()
    p = np.exp((c - c_max) / T)
    p_sum = p.sum()
    p = p / p_sum
    # grad[j] = sum_m p[m+j] * a[m] for j=0..n-1 (cross-correlation)
    N = 1 << (3 * n - 3).bit_length()
    P = np.fft.fft(p, N)
    A = np.fft.fft(a, N)
    grad = np.fft.ifft(P * np.conj(A)).real[:n]
    return grad, p


def fista_logsumexp(a0, time_limit, T_start=None, T_end=None, verbose=True):
    """
    FISTA with log-sum-exp smoothed objective and adaptive step size.
    Anneals temperature geometrically over the time budget.
    """
    n = len(a0)
    x = normalize(a0).copy()
    x_prev = x.copy()
    best_x = x.copy()
    c0 = fast_autocorr(x)
    best_M = c0.max()
    best_val = 2.0 * n * best_M
    
    if T_start is None:
        T_start = best_M * 0.1
    if T_end is None:
        T_end = best_M * 1e-5
    
    t0 = time.monotonic()
    t_mom = 1.0
    step = 0.5 / n  # conservative initial step
    step_min = 1e-8 / n
    no_improve = 0
    max_no_improve = 30000
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        # Temperature annealing (geometric)
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** elapsed_frac
        
        a = x
        c = fast_autocorr(a)
        M_curr = c.max()
        curr_val = 2.0 * n * M_curr
        
        if curr_val < best_val - 1e-15:
            best_val = curr_val
            best_M = M_curr
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute smoothed gradient
        grad, p = compute_soft_gradient(a, c, T)
        
        # FISTA momentum
        if iter_count > 0:
            z = x + ((t_mom - 1.0) / (t_mom + 1.0)) * (x - x_prev)
            z = project_simplex(z)
        else:
            z = x.copy()
        
        # Gradient step + projection
        x_new = project_simplex(z - step * grad)
        
        # Backtracking: check if objective decreased (every 50 iters)
        if iter_count % 50 == 0 and iter_count > 0:
            c_new = fast_autocorr(x_new)
            M_new = c_new.max()
            if M_new > M_curr * (1 + 1e-6) + 1e-15:
                step *= 0.5
                if step < step_min:
                    break
            elif M_new < M_curr * (1 - 1e-4):
                # Good progress, try increasing step
                step = min(step * 1.1, 2.0 / n)

                    if M_test < best_new_M - 1e-15:
                        best_new_M = M_test
                        best_frac = frac
                        best_dst = dst
                        best_new_a = a_test
                    break  # Only try first fraction per dst for speed
            
            if best_new_a is not None:
                a = best_new_a
                best_a = a.copy()
                best_M = best_new_M
                best_val = 2.0 * n * best_M
                no_improve = 0
                improved = True
                break  # Restart the loop with new a
        
        if not improved:
            no_improve += 1
            # Random perturbation to escape
            if no_improve % 200 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    delta = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= delta
                    a[j] += delta
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s > 1e-10:

                            best_M = c_new.max()
                            best_a = a.copy()
                            best_val = 2.0 * n * best_M
                            no_improve = 0
                            improved = True
        
        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            print("    CoordDT ({:.0f}s) J={:.7f} no_imp={}".format(elapsed, curr_val, no_improve))
    
    elapsed = time.monotonic() - t0
    if verbose:
        print("    CoordDT done ({:.1f}s) J={:.7f}".format(elapsed, best_val))
    return best_a, best_val
