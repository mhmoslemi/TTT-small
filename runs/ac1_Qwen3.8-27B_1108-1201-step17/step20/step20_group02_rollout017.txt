def fast_convolve(a):

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

def soft_mirror_descent(a0, time_limit, tau_start=0.01, tau_end=0.001, verbose=True):

    best_m = fast_convolve(y).max()

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def peak_reduction(a0, time_limit, verbose=True):
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_convolve(y).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(y)
        top_peaks = np.argsort(ac)[::-1][:12]

        improved = False
        for k in top_peaks[:6]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            contrib = y[idx] * y[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = y[src]
            if src_val < 1e-12:
                continue

            scores = np.zeros(n)
            for k2 in top_peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i)
                if s2 < e2:
                    idx2 = np.arange(s2, e2)
                    scores[idx2] += y[k2i - idx2]

            scores[src] = np.inf
            for dk in range(-5, 6):
                pos = src + dk
                if 0 <= pos < n:
                    scores[pos] = np.inf

            candidates = np.argsort(scores)[:min(30, n // 4)]

            found = False
            for frac in [0.012, 0.008, 0.018, 0.005, 0.025, 0.015]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                for dst in candidates[:10]:
                    dst = int(dst)
                    if dst == src:
                        continue
                    trial = y.copy()
                    trial[src] -= amt
                    trial[dst] += amt
                    trial = np.maximum(trial, 0.0)
                    s_t = trial.sum()
                    if s_t < 1e-15:
                        continue
                    trial = trial / s_t
                    M_trial = fast_convolve(trial).max()
                    if M_trial < best_m - 1e-15:
                        y = trial
                        best_m = M_trial
                        best_a = y.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break

        if not improved:
            stall += 1
            if stall % 50 == 0:
                for _ in range(rng.integers(5, 15)):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and y[i] > 1e-8:
                        amt = y[i] * rng.uniform(0.004, 0.02)
                        trial = y.copy()
                        trial[i] -= amt
                        trial[j] += amt
                        trial = np.maximum(trial, 0.0)
                        s_t = trial.sum()
                        if s_t > 1e-15:
                            trial = trial / s_t
                            M_trial = fast_convolve(trial).max()
                            if M_trial < best_m:
                                y = trial
                                best_m = M_trial
                                best_a = y.copy()
                                improved = True
                                stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    PeakRed: J=" + repr(2.0 * n * best_m) + " stall=" + str(stall) + " (" + str(round(elapsed, 1)) + "s)")

    return best_a, best_m

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

def get_initial_sequence(t_global):
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:

            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):

    except Exception:
        pass

    n = 6000

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = get_solver()
    best_seq, best_score = get_initial_sequence(t_global)
    n0 = len(best_seq)

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # Phase 1: Soft mirror descent warm-up
    print("\n=== Phase 1: Soft Mirror Descent ===")
    t1 = min(remaining_time() * 0.06, 80.0)
    if t1 > 15:
        a_out, _ = soft_mirror_descent(norm_sum1(best_seq), t1, tau_start=0.004, tau_end=0.0003, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "softmd", t_global)

    # Phase 2: Main LP-FW
    print("\n=== Phase 2: LP-FW (main) ===")
    t2 = min(remaining_time() * 0.28, 400.0)
    if t2 > 60 and len(best_seq) <= 12000:
        a_out, _ = lp_fw_improved(norm_sum1(best_seq), t2, solver, initial_K=500, max_K=4000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_main", t_global)

    # Phase 3: Peak reduction
    print("\n=== Phase 3: Peak Reduction ===")
    t3 = min(remaining_time() * 0.12, 180.0)
    if t3 > 25:
        a_out, _ = peak_reduction(norm_sum1(best_seq), t3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakred1", t_global)

    # Phase 4: LP-FW round 2
    print("\n=== Phase 4: LP-FW (round 2) ===")
    t4 = min(remaining_time() * 0.22, 300.0)
    if t4 > 50 and len(best_seq) <= 12000:
        a_out, _ = lp_fw_improved(norm_sum1(best_seq), t4, solver, initial_K=600, max_K=5000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2", t_global)

    # Phase 5: Multi-n exploration
    print("\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)
    try_ns = sorted(set([n_best - 500, n_best + 500, n_best - 1000, n_best + 1000, n_best + 1500, n_best - 1500]))
    try_ns = [l for l in try_ns if 2500 <= l <= 14000]
    try_ns.sort(key=lambda l: abs(l - n_best))

    for new_n in try_ns[:4]:
        rt = remaining_time()
        if rt < 130:
            break
        print("  Multi-n: n=" + str(new_n) + " (remaining=" + str(round(rt, 1)) + "s)")
        a_long = interpolate_seq(best_seq, new_n)

        t5 = min(rt * 0.20, 180.0)
        if t5 > 40 and new_n <= 12000:
            a_out, _ = lp_fw_improved(a_long, t5, solver, initial_K=300, max_K=2500, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

        rt = remaining_time()
        if rt > 50:
            t5b = min(rt * 0.15, 80.0)
            if t5b > 20:
                a_out, _ = peak_reduction(norm_sum1(best_seq), t5b, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "peakred_n" + str(new_n), t_global)

    # Phase 6: Final LP-FW on best
    print("\n=== Phase 6: Final LP-FW ===")
    t6 = min(remaining_time() * 0.18, 220.0)
    if t6 > 40 and len(best_seq) <= 12000:
        a_out, _ = lp_fw_improved(norm_sum1(best_seq), t6, solver, initial_K=700, max_K=5000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # Phase 7: Final peak reduction
    print("\n=== Phase 7: Final Peak Reduction ===")
    t7 = min(remaining_time() * 0.12, 120.0)
    if t7 > 20:
        a_out, _ = peak_reduction(norm_sum1(best_seq), t7, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakred_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + str(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + str(round(best_score, 6)))

    return [float(x) for x in best_seq]
