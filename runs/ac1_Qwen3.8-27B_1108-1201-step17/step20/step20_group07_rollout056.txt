    """Multiplicative weights update: a_new[j] = a[j] * exp(-eta * grad[j]) / Z"""

    log_a_new -= log_a_new.max()  # For numerical stability

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()

    return a_new / s

def simplex_project(z):
    """Project z onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho_val = rho_idx[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_smooth_gradient(a, beta, top_k=1000):
    n = len(a)
    conv = fast_conv(a)
    k = min(top_k, len(conv))
    idx = np.argpartition(conv, -k)[-k:]
    conv_vals = conv[idx]
    max_v = conv_vals.max()
    shifted = beta * (conv_vals - max_v)
    shifted = np.clip(shifted, -50, 50)
    exp_v = np.exp(shifted)
    p_top = exp_v / exp_v.sum()
    
    # p is sparse: only top_k entries are non-zero
    # Build full p vector for FFT
    size = 2 * n - 1
    p = np.zeros(size)
    p[idx] = p_top
    
    # Cross-correlation: h[j] = sum_k p[k] * a[k-j]
    # N >= len(p) + len(a) - 1 = (2n-1) + n - 1 = 3n - 2
    N = 1 << (3 * n - 2 - 1).bit_length()
    Pf = np.fft.rfft(p, N)
    Af = np.fft.rfft(a, N)
    h_full = np.fft.irfft(Pf * np.conj(Af), N)
    grad = 2.0 * h_full[:n] / beta
    return grad, conv.max()

def mirror_step(a, grad, eta):
    # Multiplicative update
    log_a = np.where(a > 1e-30, np.log(a), -30.0)
    log_a_new = log_a - eta * grad
    log_a_new -= log_a_new.max()
    a_new = np.exp(log_a_new)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho_val = rho_idx[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def obj_value(a):
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()

def phase_mirror_descent(a0, time_limit, beta_start=50.0, beta_end=500.0,
                         top_k=500, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = obj_value(a)
    t0 = time.monotonic()
    beta = beta_start
    rng = np.random.default_rng(999)
    stall = 0
    t = 0
    eta = 1.0 / (beta * best_val / (2.0 * n))  # Initial step size estimate
    eta = min(eta, 0.1)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, max_conv = compute_smooth_gradient(a, beta, top_k)
        
        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            eta = 1.0 / (grad_norm * 10.0)
            eta = min(eta, 0.05)
            eta = max(eta, 1e-6)
        
        a_new = mirror_step(a, grad, eta)
        val_new = obj_value(a_new)
        
        if val_new < best_val - 1e-15:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            # If stalling, try smaller eta
            if stall % 50 == 0:
                eta *= 0.5
            if stall > 500:
                # Restart with perturbation
                noise_scale = 0.01 + 0.01 * (stall // 500)
                pert = rng.normal(0, noise_scale, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-10)
                a_r = a_r / a_r.sum()
                val_r = obj_value(a_r)
                if val_r < best_val:
                    best_val = val_r
                    best_a = a_r.copy()
                a = a_r
                stall = 0
                eta = 0.02
                beta = max(beta * 0.7, beta_start)
        
        # Gradually increase beta
        if t % 100 == 0:
            beta = min(beta * 1.02, beta_end)
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc t={t} ({elapsed:.1f}s): val={best_val:.7f} beta={beta:.1f}")
    
    return best_a, best_val

def phase_peak_shaving(a0, time_limit, top_k=15, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = obj_value(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        conv = fast_conv(a)
        peak_indices = np.argsort(conv)[::-1][:top_k]
        
        # Compute pressure
        pressure = np.zeros(n)
        for k_i in peak_indices[:8]:
            k_int = int(k_i)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                pressure[idx] += a[k_int - idx]
                pressure[k_int - idx] += a[idx]
        
        improved = False
        for k_i in peak_indices[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k_i)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            if partner < 0 or partner >= n:
                continue
            
            # Choose which to reduce
            if a[src] >= a[partner]:
                reduce_idx = src
            else:
                reduce_idx = partner
            
            if a[reduce_idx] < 1e-14:
                continue
            
            # Find safe destination
            safe_mask = pressure < np.percentile(pressure, 40)
            if safe_mask.sum() < 10:
                safe_mask = np.ones(n, dtype=bool)
            safe_mask[reduce_idx] = False
            if safe_mask.sum() == 0:
                continue
            
            # Try to move mass to the safest position
            safe_pressure = pressure.copy()
            safe_pressure[~safe_mask] = np.max(pressure) + 1
            dst = int(np.argmin(safe_pressure))
            
            found = False
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[reduce_idx] * frac
                if amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                val_trial = obj_value(a_trial)
                if val_trial < best_val - 1e-16:
                    a = a_trial
                    best_val = val_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if found:
                break  # Found improvement, break inner loop
        
        if not improved:
            stall += 1
            if stall % 30 == 0:
                noise = min(0.03, 0.005 * stall / 30)
                pert = rng.normal(0, noise, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-10)
                a_r = a_r / a_r.sum()
                val_r = obj_value(a_r)
                if val_r < best_val:
                    best_val = val_r
                    best_a = a_r.copy()
                a = a_r
            if verbose and stall % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    PeakShave ({elapsed:.1f}s): val={best_val:.7f} stall={stall}")
    
    return best_a, best_val

def phase_simulated_annealing(a0, time_limit, T_start=None, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = obj_value(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    if T_start is None:
        T_start = best_val * 0.001
    
    T = T_start
    n_steps = 0
    n_accept = 0
    
    while time.monotonic() - t0 < time_limit:
        # Batch of steps to check time
        batch_end = time.monotonic() + 0.5  # 0.5 second batches
        current_val = obj_value(a)
        
        while time.monotonic() < batch_end:
            n_steps += 1
            # Propose a move
            move_type = rng.random()
            
            if move_type < 0.5:
                # Transfer from a peak-contributing position to a random safe position
                conv = fast_conv(a)
                top_peaks = np.argsort(conv)[::-1][:5]
                k_i = int(top_peaks[int(rng.integers(len(top_peaks)))])
                i_start = max(0, k_i - n + 1)
                i_end = min(n - 1, k_i)
                if i_start > i_end:
                    continue
                idx = np.arange(i_start, i_end + 1)
                contrib = a[idx] * a[k_i - idx]
                src = int(idx[rng.integers(len(idx))])
                if a[src] < 1e-14:
                    continue
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                frac = rng.uniform(0.01, 0.08)
                amt = a[src] * frac
            elif move_type < 0.8:
                # Random pairwise transfer
                src = int(rng.integers(n))
                if a[src] < 1e-14:
                    continue
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                frac = rng.uniform(0.01, 0.05)
                amt = a[src] * frac
            else:
                # Power-law perturbation
                src = int(np.argmax(a))
                if a[src] < 1e-14:
                    continue
                dst = int(np.argmin(a))
                frac = rng.uniform(0.005, 0.03)
                amt = a[src] * frac
            
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            val_trial = obj_value(a_trial)
            delta = val_trial - current_val
            
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                a = a_trial
                current_val = val_trial
                n_accept += 1
                if val_trial < best_val:
                    best_val = val_trial
                    best_a = a.copy()
        
        # Cool down
        T *= 0.995
        if T < best_val * 1e-5:
            T = best_val * 1e-5
        
        # Occasional reheat
        if n_steps % 5000 == 0:
            T = max(T * 2, best_val * 0.0001)
        
        if verbose and n_steps % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA steps={n_steps} ({elapsed:.1f}s): best={best_val:.7f} T={T:.6f} acc={n_accept/n_steps:.3f}")
    
    return best_a, best_val
