def solve_primal_lp(f, M, S):
    n = len(f)
    K = len(S)
    # c_objective = -1 (maximize sum(g))
    c_obj = -np.ones(n)
    # Inequality: A @ g <= b where A is K x n, b = M * ones(K)
    A = np.zeros((K, n))

        for j in range(n):
            kj = k_int - j
            if 0 <= kj < n:
                A[idx, j] = f[kj]
    b = M * np.ones(K)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return res.x if res.success else None

import numpy as np
import time
from scipy.optimize import linprog

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def lp_step(f, M, K=50):
    """
    Solve the LP: max sum(g) s.t. (f*g)[k] <= M for k in top K lags, g >= 0.
    Returns the solution g (sum-normalized) or None on failure.
    """
    n = len(f)
    c = fast_autocorr(f)
    S = np.argsort(c)[::-1][:K].astype(int)
    
    # Build constraint matrix: A[idx, j] = f[S[idx] - j] if 0 <= S[idx]-j < n
    A = np.zeros((K, n))
    for idx, k in enumerate(S):
        # (f*g)[k] = sum_j f[k-j] * g[j]
        # j ranges from max(0, k-n+1) to min(n-1, k)
        j_start = max(0, int(k) - n + 1)
        j_end = min(n - 1, int(k))
        if j_start <= j_end:
            A[idx, j_start:j_end+1] = f[int(k) - np.arange(j_start, j_end+1)][::-1]
            # Wait, let me be more careful
            # (f*g)[k] = sum_j f[j] * g[k-j]
            # For fixed k, j ranges where both j and k-j are in [0,n-1]
            # j in [max(0, k-n+1), min(n-1, k)]
            # (f*g)[k] = sum_j f[j] * g[k-j]
            # Let m = k-j, so j = k-m
            # = sum_m f[k-m] * g[m] where m = k-j
            # m in [k - min(n-1,k), k - max(0, k-n+1)] = [max(0, k-n+1), min(n-1, k)]
            # So (f*g)[k] = sum_{m=m_start}^{m_end} f[k-m] * g[m]
            # A[k_idx, m] = f[k-m] for m in [m_start, m_end]
            A[idx, j_start:j_end+1] = f[int(k) - np.arange(j_start, j_end+1)]
            # A[idx, m] = f[k-m] for m in [j_start, j_end]
            # f[int(k) - np.arange(j_start, j_end+1)] gives f[k-j_start], f[k-j_start-1], ..., f[k-j_end]
            # That's reversed! We want A[idx, m] = f[k-m], so:
            # A[idx, j_start] = f[k - j_start]
            # A[idx, j_start + 1] = f[k - j_start - 1]
            # ...
            # A[idx, m] = f[k - m]
            # f[k - np.arange(j_start, j_end+1)] gives exactly this!
            # f[k-j_start], f[k-j_start-1], ..., f[k-j_end]
            # That's A[idx, j_start] = f[k-j_start], A[idx, j_start+1] = f[k-j_start-1], etc.
            # Yes! This is correct.
    
    b = M * np.ones(K)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        s = g.sum()
        if s > 1e-10:
            return g / s
    return None

def lp_alternation(f_init, time_limit, K=50, t_param=0.3, verbose=True, seed=42):
    """LP-based alternating minimization."""
    n = len(f_init)
    f = norm_sum1(f_init)
    best_f = f.copy()
    best_M = fast_autocorr(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    K_current = K
    t_current = t_param
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 200:
            # Restart with perturbation
            noise = 0.05 * rng.normal(0, 1, n) * f
            f_r = f + noise
            f_r = np.maximum(f_r, 0)
            f_r = f_r / f_r.sum()
            f = f_r
            stall = 0
            K_current = int(rng.integers(20, 80))
            t_current = rng.uniform(0.1, 0.5)
            continue
        
        c = fast_autocorr(f)
        M = c.max()
        
        g = lp_step(f, M, K_current)
        if g is None:
            # Try smaller K
            g = lp_step(f, M, max(10, K_current // 2))
            if g is None:
                stall += 1
                continue
        
        f_new = (1 - t_current) * f + t_current * g
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            stall += 1
            continue
        f_new = f_new / s_new
        
        M_new = fast_autocorr(f_new).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
        
        f = f_new
        
        # Adaptive parameters
        if it % 50 == 0:
            K_current = int(min(100, K_current * 1.1))
            t_current = max(0.1, t_current * 0.98)
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    LP alt it={it} ({elapsed:.0f}s): J={J:.6f}, M={M_new:.8f}, stall={stall}, K={K_current}")
    
    return best_f, best_M

def nesterov_opt(a0, time_limit, beta_start=50.0, beta_end=3000.0,
                 beta_growth=1.008, eta_base=0.5, top_k_ratio=0.015,
                 restart_every=1500, verbose=True, seed=42):
    """Nesterov accelerated projected gradient."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_M = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        # Compute smoothed gradient
        c = fast_autocorr(y)
        if top_k < len(c):
            idx = np.argsort(c)[::-1][:top_k]
            c_use = np.zeros(len(c))
            c_use[idx] = c[idx]
            c_max = c[idx].max()
        else:
            c_use = c
            c_max = c.max()
        
        shifted = beta * (c_use - c_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        total = exp_vals.sum()
        if total < 1e-300:
            p = np.zeros_like(c_use)
            p[np.argmax(c)] = 1.0
        else:
            p = exp_vals / total
        
        # Gradient via FFT
        fft_size = 1 << (2 * n - 1).bit_length()
        P = np.fft.rfft(p, fft_size)
        a_rev = y[::-1]
        A_rev = np.fft.rfft(a_rev, fft_size)
        Conv = np.fft.irfft(P * A_rev, fft_size)
        grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
        
        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)
        
        z = y - eta * grad
        x_new = simplex_project(z)
        
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_M - 1e-15:
            best_M = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.03 / (1.0 + 0.05 * (t // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Nest t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}")
    
    return best_a, best_M

def peak_surgery(a0, time_limit, verbose=True, seed=77):
    """Targeted local search: reduce top peaks via mass redistribution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 2000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_M - 1e-15:
            best_M = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        n_top = min(20, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]
        
        improved = False
        found_move = False
        
        for k in top_idx[:10]:
            if found_move:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            if products.max() < 1e-20:
                continue
            
            src_i = int(idx[np.argmax(products)])
            partner = k_int - src_i
            if partner < 0 or partner >= n:
                continue
            
            if src_i == partner:
                # Diagonal element, can't split
                continue
            
            src_val = a[src_i]
            par_val = a[partner]
            
            if src_val >= par_val:
                reduce_i = src_i
            else:
                reduce_i = partner
            
            if a[reduce_i] < 1e-14:
                continue
            
            # Compute pressure
            pressure = np.zeros(n)
            for k2 in top_idx[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]
            
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_idx[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)
            
            mask = candidates != reduce_i
            candidates = candidates[mask]
            if len(candidates) == 0:
                continue
            
            cand_pressure = pressure[candidates]
            n_try = min(5, len(candidates))
            try_dsts = candidates[np.argsort(cand_pressure)[:n_try]]
            
            for dst in try_dsts:
                dst_int = int(dst)
                for frac in [0.003, 0.008, 0.015, 0.03, 0.05]:
                    amt = a[reduce_i] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_i] -= amt
                    a_trial[dst_int] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M - 1e-16:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        found_move = True
                        break
                if found_move:
                    break
        
        if not improved and no_improve > 300:
            # Random exploration
            for _ in range(20):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_M:
                            best_M = M_new
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
        
        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    PeakSurgery ({elapsed:.0f}s): J={J:.6f} stall={no_improve}")
    
    return best_a, best_M
