"""
Dual Descent + Spectral Refinement for minimizing 2*n*max(autocorr)/(sum)^2.

Key innovation: Exploits the KKT stationarity condition of the primal-dual problem.
The dual variables lambda[k] represent the 'importance' of each autocorrelation lag.
The primal is updated so that the 'dual convolution' c[i] = sum_j a[j]*lambda[i+j]
becomes constant on the support of a (the KKT condition). This naturally drives
the autocorrelation towards flatness.

Supporting methods: spectral domain flattening, peak-targeting moves,
multi-length homotopy, and adaptive restarts.
"""

import numpy as np
import time


def fast_autocorr(a):
    """FFT-based linear autocorrelation (full convolution a*a)."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a.astype(np.float64), size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def correlate_valid(lam, a):
    """Compute c[i] = sum_{j=0}^{n-1} a[j]*lam[i+j] for i=0..n-1."""
    return np.correlate(lam, a, mode='valid')


def normalize_sum1(a):
    """Project to non-negative simplex (sum=1)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def spectral_flatten(a, alpha=0.08, boost_low=0.3):
    """One step of spectral domain peak flattening."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, size)
    power = np.abs(A) ** 2

    if len(power) > 1:
        target = np.median(power[1:])
    else:
        target = power[0] if len(power) > 0 else 1.0

    A_new = A.copy()
    for k in range(1, len(A_new)):
        ratio = power[k] / max(target, 1e-30)
        if ratio > 1.0:
            A_new[k] *= np.sqrt(max(0.2, 1.0 - alpha * (ratio - 1.0)))
        elif ratio < 0.3:
            A_new[k] *= np.sqrt(min(2.0, 1.0 + alpha * boost_low * (0.3 - ratio)))

    a_new = np.fft.irfft(A_new, size)[:n]
    return normalize_sum1(a_new)


def dual_descent(a0, time_limit, verbose=True, seed=42,
                 eta_d_init=0.5, eta_p_init=0.08, lam_power=2.0):
    """
    Dual descent optimizer.
    
    Maintains primal a (non-negative, sum=1) and dual lambda (non-negative, sum=1,
    length 2n-1). Updates:
    - Dual: lambda[k] proportional to exp(eta_d * (b[k]/M - 1)), emphasizing peaks
    - Primal: a[i] updated using the KKT mismatch (nu - c[i]) where
      c[i] = sum_j a[j]*lambda[i+j] is the 'dual convolution'
    """
    n = len(a0)
    a = normalize_sum1(a0.copy())
    b = fast_autocorr(a)
    M = b.max()

    # Initialize dual from peaked autocorrelation
    lam = np.power(np.maximum(b, 1e-15), lam_power)
    lam_sum = lam.sum()
    if lam_sum < 1e-300:
        lam = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        lam = lam / lam_sum

    best_a = a.copy()
    best_M = M
    best_score = 2.0 * n * M

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    eta_d = eta_d_init
    eta_p = eta_p_init
    momentum = np.zeros(n)
    beta_mom = 0.7
    stall = 0
    stall_limit = 3000

    for it in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > stall_limit:
            break

        # Compute autocorrelation
        b = fast_autocorr(a)
        M = b.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Dual update: multiplicative weights emphasizing peaks
        b_scaled = b / max(M, 1e-15)
        lam = lam * np.exp(eta_d * (b_scaled - 1.0))
        lam_sum = lam.sum()
        if lam_sum < 1e-300:
            lam = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            lam = lam / lam_sum

        # Compute dual convolution: c[i] = sum_j a[j] * lam[i+j]
        c = correlate_valid(lam, a)

        # Target value nu (weighted average)
        nu = np.dot(a, c)

        # Primal update with momentum
        c_range = c.max() - c.min()
        c_scale = max(c_range, 0.05 * (np.abs(c).max() + 1e-15))
        grad = (nu - c) / c_scale

        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
        a_new = a + eta_p * momentum
        a = normalize_sum1(a_new)

        # Periodic spectral refinement (every 300 iterations)
        if it % 300 == 0:
            a_spec = spectral_flatten(a, alpha=0.06)
            b_spec = fast_autocorr(a_spec)
            M_spec = b_spec.max()
            if 2 * n * M_spec < score:
                a = a_spec
            # Occasionally add small noise to escape locals
            if stall > 500:
                noise = rng.normal(0, 0.005, n)
                a = normalize_sum1(a * (1.0 + noise))

        # Adaptive step sizes
        if it % 100 == 0:
            if stall < 50:
                eta_p = min(eta_p * 1.03, 0.3)
                eta_d = min(eta_d * 1.01, 2.0)
            elif stall < 500:
                eta_p = max(eta_p * 0.97, 0.02)
            else:
                eta_p = max(eta_p * 0.9, 0.01)
                eta_d = min(eta_d * 1.05, 4.0)

        # Restart strategy when stalled
        if stall > 800 and it % 200 == 0:
            strategy = rng.integers(4)
            if strategy == 0:
                # Gaussian noise
                a = normalize_sum1(best_a * (1.0 + 0.02 * rng.standard_normal(n)))
            elif strategy == 1:
                # Blend with spectral
                a = normalize_sum1(0.8 * best_a + 0.2 * spectral_flatten(best_a, 0.2))
            elif strategy == 2:
                # Local shuffle
                a = best_a.copy()
                for _ in range(10):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    amt = a[i] * 0.05
                    a[i] -= amt
                    a[j] += amt
                a = normalize_sum1(a)
            else:
                # Reinitialize dual with different power
                b_cur = fast_autocorr(a)
                new_power = lam_power * rng.uniform(0.5, 2.0)
                lam_new = np.power(np.maximum(b_cur, 1e-15), new_power)
                lam = lam_new / lam_new.sum()

            momentum = np.zeros(n)
            eta_p = eta_p_init
            eta_d = eta_d_init
            stall = 0

        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  DD it={it} ({elapsed:.1f}s): score={best_score:.6f} "
                  f"eta_p={eta_p:.4f} eta_d={eta_d:.3f} stall={stall}")

    return best_a, best_score


def peak_targeting(a0, time_limit, verbose=True, seed=99):
    """
    Directly target the top autocorrelation peaks by moving mass
    from contributing positions to non-contributing ones.
    """
    n = len(a0)
    a = normalize_sum1(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 2000:
        b = fast_autocorr(a)
        M = b.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Find top peaks
        top_lags = np.argsort(b)[::-1][:min(5, 2 * n - 1)]

        improved = False
        for k in top_lags:
            k = int(k)
            if b[k] <= M * 0.7:
                break

            # Positions contributing to lag k: a[i] * a[k-i]
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo > i_hi:
                continue

            idx = np.arange(i_lo, i_hi + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-15:
                continue

            src = int(idx[np.argmax(contrib)])
            partner = k - src

            # Find a "good" destination (low contribution to all top peaks)
            best_dst = -1
            best_improvement = 0

            for _ in range(20):
                dst = int(rng.integers(n))
                if dst == src or dst == partner:
                    continue

                # Try moving a small fraction
                for frac in [0.01, 0.03, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = normalize_sum1(a_trial)
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < M - 1e-10:
                        if M_trial < best_improvement or best_dst == -1:
                            best_improvement = M_trial
                            best_dst = dst
                            best_amt = amt

            if best_dst >= 0:
                a[src] -= best_amt
                a[best_dst] += best_amt
                a = normalize_sum1(a)
                improved = True

        if not improved:
            # Random exploration
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.005, 0.03)
                a[i] -= amt
                a[j] += amt
                a = normalize_sum1(a)
                M_new = fast_autocorr(a).max()
                if 2.0 * n * M_new < best_score:
                    stall = 0
                    best_a = a.copy()
                    best_score = 2.0 * n * M_new

        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"  PeakTarget ({elapsed:.1f}s): score={best_score:.6f} stall={stall}")

    return best_a, best_score


def interpolate_length(a, new_n):
    """Interpolate sequence to a new length, preserving sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a)
    return normalize_sum1(a_interp)
