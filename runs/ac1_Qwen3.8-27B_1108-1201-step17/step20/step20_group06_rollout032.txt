def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_grad_two_scale(y, beta_coarse, beta_fine, top_k_coarse, top_k_fine):
    """Compute gradient using two-scale softmax for better peak resolution."""
    n = len(y)
    b = fast_autocorr(y)
    size = 2 * n - 1

    # Coarse scale: broader view
    idx_c = np.argsort(b)[::-1][:top_k_coarse]
    b_use_c = np.zeros(size)
    b_use_c[idx_c] = b[idx_c]
    b_max_c = b[idx_c].max()

    # Fine scale: focused on top peaks
    idx_f = np.argsort(b)[::-1][:top_k_fine]
    b_use_f = np.zeros(size)
    b_use_f[idx_f] = b[idx_f]
    b_max_f = b[idx_f].max()

    def softmax_probs(b_use, b_max, beta):
        shifted = beta * (b_use - b_max)
        shifted = np.clip(shifted, -500, 0)
        exp_vals = np.exp(shifted)
        s = exp_vals.sum()
        if s < 1e-300:
            return np.ones(size) / size
        return exp_vals / s

    p_c = softmax_probs(b_use_c, b_max_c, beta_coarse)
    p_f = softmax_probs(b_use_f, b_max_f, beta_fine)
    p = 0.4 * p_c + 0.6 * p_f

    a_rev = y[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1: 2 * n - 1]
    return grad, b

def lp_fw_v3(a0, time_limit, solver, verbose=True):
    """LP-Frank-Wolfe with confidence-based active set and adaptive margin."""

    t0 = time.monotonic()

    min_K = 100
    max_K = 2500
    add_per_iter = 30
    y = a.copy()
    no_improve = 0
    max_no_improve = 800

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = float(ac.max())

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Confidence-based active set: include k where ac[k] > M_curr - margin
        threshold = M_curr - confidence_margin * M_curr
        active_mask = ac >= threshold
        active_indices = np.where(active_mask)[0]

        # Ensure minimum size
        if len(active_indices) < min_K:
            all_sorted = np.argsort(ac)[::-1]
            active_indices = all_sorted[:min_K]

        # Cap size
        if len(active_indices) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_indices])
            keep = np.argsort(margins)[:max_K]
            active_indices = active_indices[keep]

        K = len(active_indices)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(active_indices):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations and expand active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-6)[0]
        if len(violations) > 0:
            for k in violations[:add_per_iter]:
                if k not in active_indices:
                    active_indices = np.append(active_indices, k)

        # Adaptive line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        alphas = [best_alpha, best_alpha * 0.5, best_alpha * 1.5,
                  1.0 / (t + 1), 0.03, 0.07, 0.15, 0.25]
        for alpha in alphas:
            alpha = max(0.005, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = float(fast_autocorr(trial).max())
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        # Adapt confidence margin
        if no_improve > 200:
            confidence_margin = max(0.005, confidence_margin * 0.9)
        elif no_improve == 0:
            confidence_margin = min(0.05, confidence_margin * 1.05)

        if verbose and t % 80 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW3 it=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" +
                  repr(round(J, 6)) + " |S|=" + str(K) + " margin=" + str(round(confidence_margin, 4)))

    return best_a, best_m


def peak_shaving_cd(a0, time_limit, verbose=True, seed=789):
    """
    Novel peak-shaving coordinate descent.
    Directly identifies top contributors to max peaks and adjusts them.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 5000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        size = 2 * n - 1
        # Focus on top peaks
        top_peaks = np.argsort(ac)[::-1][:min(15, size)]
        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k_int - idx]

            # Find the top contributor
            top_contrib_idx = idx[np.argmax(contribs)]
            top_contrib_val = contribs[np.argmax(contribs)]

            if top_contrib_val < 1e-10:
                continue

            src = int(top_contrib_idx)
            partner = k_int - src
            if not (0 <= partner < n):
                continue

            # Try to move mass from src to a "safe" position
            # A safe position is one that doesn't contribute to any top peak
            src_ac_contrib = np.zeros(n)
            for kp in top_peaks[:8]:
                kp_int = int(kp)
                if 0 <= kp_int - src < n:
                    src_ac_contrib[kp_int - src] += a[kp_int - src]
                if 0 <= src - kp_int < n:
                    src_ac_contrib[src - kp_int] += a[src - kp_int]

            # Find positions with low contribution to peak k
            candidate_dsts = []
            for _ in range(30):
                dst = int(rng.integers(n))
                if dst == src or dst == partner:
                    continue
                # Check if dst would create a new peak
                new_contrib = a[src] * a[dst] if (src + dst) != k_int else 0
                candidate_dsts.append((dst, -src_ac_contrib[dst] + 0.1 * new_contrib))

            if candidate_dsts:
                candidate_dsts.sort(key=lambda x: x[1], reverse=True)
                best_dst = candidate_dsts[0][0]

                for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08]:
                    amt = a[src] * frac
                    if amt < 1e-15 or a[src] - amt < -1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[best_dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = float(fast_autocorr(a_trial).max())
                    if M_trial < best_m - 1e-17:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                    elif M_trial < best_m:
                        # Tiny improvement, accept with probability
                        if rng.random() < 0.3:
                            a = a_trial
                            stall = 0
                            improved = True
                            break

            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                # Larger random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = float(fast_autocorr(a).max())
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 1000 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakShave (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def spectral_refine_v3(a0, time_limit, verbose=True, seed=555):
    """Spectral refinement with adaptive peak/valley ratio and phase randomization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()

    alpha = 0.015
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Adaptive threshold
        p_sorted = np.sort(power[1:])[::-1]
        threshold_ratio = 1.3 if it < 500 else 1.1
        target = np.mean(power[1:])

        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > threshold_ratio:
                red = 1.0 - alpha * (ratio - 1.0)
                A_new[k] *= max(0.2, np.sqrt(red))
            elif ratio < 0.4:
                inc = 1.0 + alpha * 0.2 * (0.4 - ratio)
                A_new[k] *= min(3.0, np.sqrt(inc))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = float(fast_autocorr(a_new).max())
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 150 == 0:
            # Phase perturbation
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.01
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = float(fast_autocorr(a_trial).max())
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial

        if it % 400 == 0:
            alpha = max(0.003, alpha * 0.93)

        if verbose and it % 800 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spec3 it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def multi_scale_nesterov(a0, time_limit, verbose=True, seed=111):
    """
    Multi-scale approach: optimize at coarser grid, then interpolate and refine.
    """
    n = len(a0)
    best_a = norm_sum1(a0)
    best_score = evaluate_safe(best_a)
    t0 = time.monotonic()

    # Try several coarser resolutions
    for factor in [4, 3, 2]:
        if time.monotonic() - t0 > time_limit * 0.7:
            break
        n_coarse = max(100, n // factor)
        # Downsample
        a_coarse = interpolate_sequence(best_a, n_coarse)

        remaining = time_budget_remaining(t0, time_limit) * 0.3
        if remaining < 20:
            continue

        a_out, _ = nesterov_v4(a_coarse, remaining, verbose=False, seed=seed + factor)
        # Upsample back
        a_interp = interpolate_sequence(a_out, n)
        sc = evaluate_safe(a_interp)
        if sc < best_score:
            best_a = a_interp
            best_score = sc
            if verbose:
                print("    MultiScale factor=" + str(factor) + ": J=" + repr(round(sc, 6)))

    # Final refinement at full resolution
    remaining = time.monotonic() - t0 < time_limit and (time_limit - (time.monotonic() - t0))
    if remaining > 30:
        a_out, _ = nesterov_v4(best_a, min(remaining * 0.5, 100), verbose=False, seed=seed + 99)
        sc = evaluate_safe(a_out)
        if sc < best_score:
            best_a = a_out
            best_score = sc

    return best_a, best_score


def time_budget_remaining(t0, total_budget):
    return max(0, total_budget - (time.monotonic() - t0))


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Multi-strategy optimization pipeline:
    1. Two-scale Nesterov (primary)
    2. Confidence LP-Frank-Wolfe
    3. Peak-shaving coordinate descent (novel)
    4. Spectral refinement
    5. Multi-scale exploration
    6. Final polish with alternating methods
    Time-adaptive scheduling ensures we always have time for final validation.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(round(p_score, 6)))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(round(best_score, 6)))

    n0 = len(best_seq)
    print("Start: n=" + str(n0) + ", J=" + repr(round(best_score, 6)))

    # Phase 1: Two-scale Nesterov (primary workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Nesterov v4 (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 40:
        a_out, _ = nesterov_v4(norm_sum1(best_seq), t1, verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 2: LP-FW with confidence active set
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P2: LP-FW v3 (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.18, 250.0)
    if t2 > 30 and n_best <= 10000:
        a_out, _ = lp_fw_v3(norm_sum1(best_seq), t2, solver, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 3: Peak-shaving CD (novel)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P3: PeakShave CD (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20:
        a_out, _ = peak_shaving_cd(norm_sum1(best_seq), t3, verbose=True, seed=789)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak1", t_global)

    # Phase 4: Spectral refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P4: Spectral v3 (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 15:
        a_out, _ = spectral_refine_v3(norm_sum1(best_seq), t4, verbose=True, seed=555)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec1", t_global)

    # Phase 5: Second Nesterov round with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P5: Nesterov v4 r2 (n=" + str(n_best) + ") ===")
    if remaining > 50:
        t5 = min(remaining * 0.18, 250.0)
        if t5 > 30:
            a_out, _ = nesterov_v4(norm_sum1(best_seq), t5, verbose=True, seed=1234)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P6: Multi-length (base n=" + str(n_best) + ") ===")

    longer_lengths = [n_best + 400, n_best + 1000]
    longer_lengths = [l for l in longer_lengths if l <= 12000]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print("  -> n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.40, 150.0)
        if t6a > 25:
            a_out, _ = nesterov_v4(a_long, t6a, verbose=True, seed=new_n % 9973)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 10000:
            t6b = min(remaining * 0.35, 100.0)
            if t6b > 20:
                a_out, _ = peak_shaving_cd(norm_sum1(interpolate_sequence(best_seq, new_n)), t6b, verbose=False, seed=new_n % 4441)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_n" + str(new_n), t_global)

    # Phase 7: LP-FW second round
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P7: LP-FW v3 r2 (n=" + str(n_best) + ") ===")
    if remaining > 40 and n_best <= 10000:
        t7 = min(remaining * 0.15, 150.0)
        if t7 > 20:
            a_out, _ = lp_fw_v3(norm_sum1(best_seq), t7, solver, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp2", t_global)

    # Phase 8: Final polish - alternating peak-shaving and Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P8: Final polish (n=" + str(n_best) + ") ===")
    if remaining > 30:
        # Peak shaving
        t8a = min(remaining * 0.25, 80.0)
        if t8a > 10:
            a_out, _ = peak_shaving_cd(norm_sum1(best_seq), t8a, verbose=True, seed=2468)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_final", t_global)

        # Quick Nesterov
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8b = min(remaining * 0.30, 80.0)
            if t8b > 10:
                a_out, _ = nesterov_v4(norm_sum1(best_seq), t8b, verbose=True, seed=8642)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

        # Final spectral touch
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t8c = min(remaining * 0.30, 60.0)
            if t8c > 8:
                a_out, _ = spectral_refine_v3(norm_sum1(best_seq), t8c, verbose=False, seed=3141)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "spec_final", t_global)

        # Ultra-final peak shaving
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t8d = min(remaining * 0.50, 60.0)
            if t8d > 5:
                a_out, _ = peak_shaving_cd(norm_sum1(best_seq), t8d, verbose=False, seed=9999)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_ultra", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: J=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
