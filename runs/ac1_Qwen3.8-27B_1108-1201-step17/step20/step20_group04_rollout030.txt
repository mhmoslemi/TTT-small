import numpy as np

def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_active_subgradient(a, c, active_frac=0.97, beta_ws=30.0):
    """Compute subgradient of max(conv) using active-set with softmax weights."""
    n = len(a)
    c_max = c.max()
    threshold = c_max * active_frac
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    
    active_vals = c[active]
    shifted = beta_ws * (active_vals - active_vals.max())
    shifted = np.clip(shifted, -300, 300)
    weights = np.exp(shifted)
    total = weights.sum()
    if total < 1e-300:
        weights = np.ones_like(active_vals) / len(active_vals)
    else:
        weights = weights / total
    
    # grad_j = 2 * sum_i w_i * a[k_i - j] (with a[x]=0 outside [0,n-1])
    # = 2 * cross-correlation of (w at active positions) with a
    w_ext = np.zeros(2 * n - 1)
    w_ext[active] = weights
    fft_size = 1 << (2 * n - 2).bit_length()
    W = np.fft.rfft(w_ext, fft_size)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, fft_size)
    corr = np.fft.irfft(W * AR, fft_size)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, active, c_max


def compute_smoothed_gradient(a, c, beta, top_k=None):
    """Compute gradient of smoothed max via log-sum-exp."""
    n = len(a)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    
    fft_size = 1 << (2 * n - 2).bit_length()
    P = np.fft.rfft(p, fft_size)
    a_rev = a[::-1]
    AR = np.fft.rfft(a_rev, fft_size)
    corr = np.fft.irfft(P * AR, fft_size)
    grad = (2.0 / beta) * corr[n - 1:2 * n - 1]
    return grad

def peak_equalization(a0, time_limit, verbose=True, seed=99):
    """Targeted peak reduction: identify top peaks and try to reduce each via mass redistribution."""
    n = len(a0)
    a = norm_sum1(a0)

    no_improve = 0

        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-16:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find top peaks
        top_n = min(20, len(c))
        top_idx = np.argsort(c)[::-1][:top_n]
        top_vals = c[top_idx]
        
        improved = False
        
        for k_i in range(min(12, len(top_idx))):
            if improved:
                break
            k = int(top_idx[k_i])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k - idx]
            if products.max() < 1e-20:
                continue
            
            # Find the dominant product pair
            src_i = int(idx[np.argmax(products)])
            partner = k - src_i
            if partner < 0 or partner >= n or partner == src_i:
                continue
            
            # Determine which to reduce
            if a[src_i] >= a[partner]:
                reduce_i = src_i
            else:
                reduce_i = partner
            
            if a[reduce_i] < 1e-14:
                continue
            
            # Compute pressure map: how much each position contributes to top peaks
            pressure = np.zeros(n)
            for k2 in top_idx[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]
            
            # Find low-pressure destinations
            exclude = np.zeros(n, dtype=bool)
            exclude[reduce_i] = True
            # Also exclude positions that are in the "partner" set of top peaks
            for k2 in top_idx[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)

                candidates = np.where(~exclude[reduce_i])[0]

                candidates = np.arange(n)
                candidates = candidates[candidates != reduce_i]
            
            # Sort by pressure (lowest first)
            cand_pressure = pressure[candidates]
            n_try = min(8, len(candidates))
            try_dsts = candidates[np.argsort(cand_pressure)[:n_try]]
            
            for dst in try_dsts:
                if improved:
                    break
                dst_int = int(dst)
                for frac in [0.002, 0.005, 0.01, 0.02, 0.03, 0.05]:
                    amt = a[reduce_i] * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_i] -= amt
                    a_trial[dst_int] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-17:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
        
        if not improved:
            # Try multi-move: swap mass between two positions
            if no_improve > 500 and no_improve % 10 == 0:

                    i = int(rng.integers(n))
                    j = int(rng.integers(n))

                        a[i] -= amt

                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()

                            improved = True

                print("    PeakEq (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 8)) + " stall=" + str(no_improve))

    return best_a, best_m

        # Focus on blocks near peak-contributing positions
        top_peaks = np.argsort(c)[::-1][:15]
        
        # Generate candidate block positions weighted by peak relevance
        block_positions = set()
        for k in top_peaks:
            k_i = int(k)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            for pos in range(i_start, min(i_start + 40, i_end + 1)):
                if pos <= n - block_size:
                    block_positions.add(pos)
        
        # Add some random blocks
        for _ in range(20):
            block_positions.add(int(rng.integers(0, max(1, n - block_size))))
        
        block_list = list(block_positions)
        rng.shuffle(block_list)
        
        for bp in block_list[:30]:
            if time.monotonic() - t0 > time_limit:
                break
            if improved and stall > 100:
                break
                
            # Block is a[bp:bp+block_size]
            block = a[bp:bp + block_size].copy()
            if block.sum() < 1e-15:
                continue
            
            # Try to optimize this block via inner iterations
            # The contribution to autocorrelation from this block:
            # For each lag k, the block contributes terms where at least one index is in [bp, bp+block_size-1]
            # We'll use a simple gradient step within the block
            
            for inner_iter in range(5):
                c_current = fast_autocorr(a)
                # Compute gradient only for block positions
                c_max_cur = c_current.max()
                # Use top-k smoothing for gradient
                top_k_inner = max(10, int(len(c_current) * 0.01))
                idx_top = np.argsort(c_current)[::-1][:top_k_inner]
                c_smooth = np.zeros(len(c_current))
                c_smooth[idx_top] = c_current[idx_top]
                beta_inner = 100.0
                c_max_use = c_smooth.max()
                shifted = beta_inner * (c_smooth - c_max_use)
                shifted = np.clip(shifted, -300, 300)
                exp_vals = np.exp(shifted)
                total = exp_vals.sum()
                if total < 1e-300:
                    p = np.zeros_like(c_smooth)
                    p[np.argmax(c_current)] = 1.0
                else:
                    p = exp_vals / total
                
                # Gradient at block positions
                fft_size = 1 << (2 * n - 2).bit_length()
                P = np.fft.rfft(p, fft_size)
                a_rev = a[::-1]
                AR = np.fft.rfft(a_rev, fft_size)
                corr = np.fft.irfft(P * AR, fft_size)
                grad_full = (2.0 / beta_inner) * corr[n - 1:2 * n - 1]
                grad_block = grad_full[bp:bp + block_size]
                
                # Update block with projection onto non-negative and fixed sum
                block_sum = block.sum()
                # Simple: block - eta * grad_block, then project to non-negative with same sum
                eta_block = 0.1 / (np.abs(grad_block).max() + 1e-10)
                block_trial = block - eta_block * grad_block
                block_trial = np.maximum(block_trial, 0.0)
                bt_sum = block_trial.sum()
                if bt_sum > 1e-15:
                    block_trial = block_trial * (block_sum / bt_sum)
                else:
                    block_trial = block.copy()
                
                if np.allclose(block_trial, block, atol=1e-12):
                    break
                
                a[bp:bp + block_size] = block_trial
                block = block_trial
                improved = True
                stall = 0
        
        if not improved:
            stall += 1
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    BlockCD (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 8)) + " stall=" + str(stall))

    return best_a, best_m


def fourier_refine(a0, time_limit, verbose=True, seed=5):
    """Refine by adding small Fourier components that reduce peak autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Try adding Fourier modes with frequencies near the peak locations
    c = fast_autocorr(a)
    peak_lags = np.argsort(c)[::-1][:12]
    
    for _ in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-16:
            best_m = c_max
            best_a = a.copy()
        
        # Choose a frequency near a peak
        k_peak = int(peak_lags[int(rng.integers(min(8, len(peak_lags))))])
        # Try frequencies near k_peak/n
        base_freq = k_peak / n
        for _ in range(20):
            freq = base_freq + rng.uniform(-0.01, 0.01)
            freq = max(0.001, min(0.999, freq))
            phase = rng.uniform(0, 2 * np.pi)
            sign = 1.0 if rng.random() > 0.5 else -1.0
            
            for amp in [0.005, 0.01, 0.02, 0.03, 0.05]:
                pert = sign * amp * np.cos(2.0 * np.pi * freq * np.arange(n) + phase)
                a_try = a + pert * a
                a_try = np.maximum(a_try, 1e-15)
                a_try = a_try / a_try.sum()
                m_try = fast_autocorr(a_try).max()
                if m_try < best_m - 1e-16:
                    a = a_try
                    best_m = m_try
                    best_a = a.copy()
                    break
        
        # Also try sine modes
        for _ in range(5):
            freq = base_freq + rng.uniform(-0.02, 0.02)
            freq = max(0.001, min(0.999, freq))
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.01, 0.04)
            pert = amp * np.sin(2.0 * np.pi * freq * np.arange(n) + phase)
            a_try = a + pert * a
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum()
            m_try = fast_autocorr(a_try).max()
            if m_try < best_m - 1e-16:
                a = a_try
                best_m = m_try
                best_a = a.copy()
        
        if verbose and best_m < fast_autocorr(a0).max():
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    FourRefine (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 8)))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search combining active-set, Nesterov annealing, peak equalization, block CD, and Fourier refinement."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None
    
    if best_seq is None:
        n_base = 6700
        x = np.linspace(0, 1, n_base)
        a_init = (1.0 - x) ** 1.5 + 0.1
        best_seq = norm_sum1(a_init)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(n_base) + ", score=" + repr(best_score))
    
    n0 = len(best_seq)
    print("Starting: n=" + str(n0) + ", score=" + repr(best_score))
    
    # Phase 1: Active-set subgradient (fast initial improvement)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Active-Set Subgradient (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.15, 300.0)
    if t1 > 60:
        a_out, _ = active_set_optimize(norm_sum1(best_seq), t1, verbose=True, seed=10)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "act1", t_global)
    
    # Phase 2: Nesterov with beta annealing (global refinement)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov Annealing (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.20, 400.0)
    if t2 > 90:
        a_out, _ = nesterov_annealing(norm_sum1(best_seq), t2, verbose=True, seed=2)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)
    
    # Phase 3: Peak equalization (targeted moves)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Peak Equalization (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 40:
        a_out, _ = peak_equalization(norm_sum1(best_seq), t3, verbose=True, seed=33)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peq3", t_global)
    
    # Phase 4: Fourier refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Fourier Refinement (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.08, 120.0)
    if t4 > 30:
        a_out, _ = fourier_refine(norm_sum1(best_seq), t4, verbose=True, seed=44)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "four4", t_global)
    
    # Phase 5: Block CD (polynomial convergence polish)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Block CD (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 40:
        a_out, _ = block_cd(norm_sum1(best_seq), t5, block_size=6, verbose=True, seed=55)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "bcd5", t_global)
    
    # Phase 6: Active-set again (exploit phase with tighter active set)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Active-Set Tight (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.12, 200.0)
    if t6 > 40:
        a_out, _ = active_set_optimize(norm_sum1(best_seq), t6, verbose=True, seed=66)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "act6", t_global)
    
    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Multi-Length (base n=" + str(n_best) + ") ===")
    lengths_to_try = [n_best + 300, n_best + 800, n_best - 500, n_best + 1500]
    lengths_to_try = [l for l in lengths_to_try if 3000 <= l <= 12000 and l != n_best]
    
    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("  Trying n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)
        
        t7a = min(remaining * 0.30, 150.0)
        if t7a > 30:
            a_out, _ = nesterov_annealing(a_long, t7a, verbose=False, seed=77)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ml_n" + str(new_n), t_global)
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t7b = min(remaining * 0.25, 100.0)
            if t7b > 30:
                a_try = interpolate_sequence(best_seq, new_n)
                a_out, _ = peak_equalization(a_try, t7b, verbose=False, seed=88)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "mlp_n" + str(new_n), t_global)
    
    # Phase 8: Final Nesterov burst with high precision
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final Nesterov (n=" + str(n_best) + ") ===")
    if remaining > 60:

        if t8 > 30:
            a_out, _ = nesterov_annealing(norm_sum1(best_seq), t8, verbose=True, seed=99)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest8", t_global)
    
    # Phase 9: Final block CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 9: Final Block CD (n=" + str(n_best) + ") ===")
    if remaining > 40:
        t9 = min(remaining * 0.30, 150.0)
        if t9 > 20:
            a_out, _ = block_cd(norm_sum1(best_seq), t9, block_size=4, verbose=True, seed=111)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "bcd9", t_global)
    
    # Phase 10: One more peak equalization + active set
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_best = len(best_seq)
        print("\n=== Phase 10: Final Polish (n=" + str(n_best) + ") ===")
        t10a = min(remaining * 0.4, 100.0)
        if t10a > 30:
            a_out, _ = peak_equalization(norm_sum1(best_seq), t10a, verbose=False, seed=222)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peq10", t_global)
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t10b = min(remaining * 0.5, 80.0)
            if t10b > 20:
                a_out, _ = active_set_optimize(norm_sum1(best_seq), t10b, verbose=False, seed=333)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "act10", t_global)
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")

        n_fb = 6700
        x = np.linspace(0, 1, n_fb)

    # Ensure all values are valid
    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)
    
    return [float(x) for x in best_seq]
