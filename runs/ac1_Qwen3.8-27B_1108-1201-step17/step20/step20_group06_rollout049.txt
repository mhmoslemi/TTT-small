def simplex_project(z):
    """Project onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def smoothed_grad(a, beta, top_k=None):
    """Gradient of the smoothed log-sum-exp of autocorrelation."""
    n = len(a)
    b = fast_autocorr(a)
    size = 2 * n - 1
    
    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
    else:
        b_use = b
    
    b_max = b_use.max()
    if b_max <= 0:
        return np.zeros(n), b
    
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s
    
    # Gradient: d/db_k (sum_p(p_k * b_k)) = p_k * (d b_k / d a)
    # b_k = sum_j a_j * a_{k-j}
    # db_k / da_i = a_{k-i} + a_{(k-i)} where second term exists only if k-i != i
    # More precisely: db_k/da_i = a_{k-i} (if 0 <= k-i < n) + a_{k-i} (if 0 <= k-i < n and k-i != i)
    # Actually for the autocorrelation b_k = sum_j a_j * a_{k-j}:
    # db_k/da_i = a_{k-i} (if valid) + a_{k-i} (if valid and k-i != i)
    # Wait no: b_k = sum_j a_j * a_{k-j}. 
    # db_k/da_i = a_{k-i} (j=i term) + sum over j where k-j = i, i.e., j = k-i: a_j = a_{k-i}
    # So db_k/da_i = a_{k-i} + a_{k-i} = 2*a_{k-i} (if 0 <= k-i < n and i != k-i)
    # Or db_k/da_i = a_i (if i = k-i, i.e., k = 2i)
    
    # Use FFT for efficiency:
    # grad_i = sum_k p_k * db_k/da_i = sum_k p_k * (a_{k-i} [1_{0<=k-i<n}] + a_{k-i} [1_{0<=k-i<n} and k-i != i])
    # = 2 * sum_k p_k * a_{k-i} [1_{0<=k-i<n}] - sum_k p_k * a_i [1_{k=2i}]
    # = 2 * (p * a_rev)_i - p_{2i} * a_i (if 2i < size)
    # where (p * a_rev)_i = sum_k p_k * a_{k-i} = (convolve(p, a_rev))[n-1+i]... 
    
    # Simpler: use the fact that grad = 2 * ifft(fft(p) * fft(a)) restricted to valid range
    # Actually, let's just use the direct formula:
    # grad_i = sum_k p_k * (a_{k-i} + a_{k-i}) for valid k-i, minus the diagonal correction
    
    # Most efficient: convolve p with reversed a
    a_rev = a[::-1]
    conv_len = n + size - 1
    fft_sz = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_sz)
    Pr = np.fft.rfft(p, fft_sz)
    conv = np.fft.irfft(Ar * Pr, fft_sz)
    grad = 2.0 * conv[n-1:2*n-1]
    
    # Subtract the diagonal correction: for positions where k=2i
    # Actually, the factor of 2 is correct for non-diagonal terms.
    # For diagonal terms (k=2i), db_k/da_i = a_i (not 2*a_i).
    # The convolve approach gives 2*a_i for these, so we need to subtract p_{2i}*a_i.
    for i in range(n):
        k = 2 * i
        if k < size:
            grad[i] -= p[k] * a[i]
    
    return grad / beta, b  # Actually we want the gradient of the smoothed objective
