def compute_objectives_and_grad(a, beta):
    """
    Compute C = conv(a,a), the smoothed max value, and the gradient.
    
    Parameters:
        a: array of length n, non-negative, sum = 1
        beta: smoothing parameter (large beta → sharp max)
    
    Returns:
        C: full convolution (length 2n-1)
        obj: smoothed max value (≈ max(C) for large beta)
        grad: gradient of obj w.r.t. a (length n)
    """
    n = len(a)
    # FFT size for gradient computation (need 3n-2)
    M = 1 << (3*n - 3).bit_length()  # next power of 2 >= 3n-2
    
    # Step 1: Compute C = conv(a, a)
    A = np.fft.rfft(a, M)
    C = np.fft.irfft(A * A, M)[:2*n-1]
    
    # Step 2: Compute smoothed max and weights
    C_max = C.max()
    shifted = beta * (C - C_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    # Smoothed obj = C_max + (1/beta) * log(sum(exp_vals))
    obj = C_max + np.log(exp_vals.sum()) / beta
    
    # Step 3: Compute gradient
    # grad[i] = 2 * sum_m a[m] * p[i+m] (p out of range = 0)
    # = 2 * (a_rev cross-correlated with p)[i + n - 1]
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, M)
    p_padded = np.zeros(M)
    p_padded[:2*n-1] = p
    p_fft = np.fft.rfft(p_padded, M)
    cross = np.fft.irfft(np.conj(a_rev_fft) * p_fft, M)
    grad = 2.0 * cross[n-1:2*n-1]
    
    return C, obj, grad
