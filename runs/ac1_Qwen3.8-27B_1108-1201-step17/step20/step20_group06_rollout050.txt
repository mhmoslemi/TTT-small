def compute_conv_grad(a, p):
    """
    Compute gradient of sum_k p[k]*conv(a,a)[k] w.r.t. a.
    grad[i] = 2 * sum_j p[i+j]*a[j] using FFT cross-correlation.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size + n - 2).bit_length()
    Pf = np.fft.rfft(p, fft_size)
    Af = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(Pf * np.conj(Af), fft_size)
    return 2.0 * corr[:n]


def softmax_weights(conv_vals, beta, top_k=None):
    """Compute softmax weights over convolution values for smoothed max."""
    size = len(conv_vals)
    if top_k is not None and top_k < size:
        idx = np.argsort(conv_vals)[::-1][:top_k]
        x_use = conv_vals[idx]
    else:
        idx = np.arange(size)
        x_use = conv_vals
    max_x = x_use.max()
    shifted = beta * (x_use - max_x)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        w = np.zeros(size)
        w[idx[0]] = 1.0
    else:
        w = np.zeros(size)
        w[idx] = exp_vals / s
    return w


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s
