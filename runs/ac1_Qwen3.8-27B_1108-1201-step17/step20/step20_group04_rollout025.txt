**Key insight for local search:**

At a local optimum of max_k c_k, the "active" peaks (those at the maximum) form a set S. For any feasible direction d (i.e., sum d = 0, and a + εd >= 0 for small ε), we need:
sum_{k in S} w_k * d(c_k)/d(ε) <= 0

where w_k are the weights (probabilities) on the active peaks.

A direction d that reduces all active peaks simultaneously would be an improving direction. If no such direction exists, we're at a local optimum.

To find such a direction, we can solve:
minimize sum_{k in S} w_k * d(c_k)
subject to: sum d = 0, a + d >= 0 (for small enough step)

This is a linear program in d! (Since d(c_k)/d(ε) = sum_j (∂c_k/∂a_j) * d_j is linear in d.)

The LP would be:
minimize c_S^T d (where c_S is the combined gradient)
subject to: 1^T d = 0
             d_j >= -a_j (for feasibility, to first order)

If the optimal value is < 0, we have an improving direction. If = 0, we're at a local optimum.

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project vector z onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_smoothed_grad(a, beta, top_k=None):
    """
    Compute gradient of (1/beta)*log(sum of top_k exp(beta*c_k)) w.r.t. a.
    Returns (gradient, probability weights, convolution).
    """
    n = len(a)
    c = fast_autocorr(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    
    # grad_j = (2/beta) * sum_k p_k * a[k-j]
    # = (2/beta) * conv(p, a_flipped)[n-1+j]
    a_flipped = a[::-1]
    P = np.fft.rfft(p, fft_size)
    Af = np.fft.rfft(a_flipped, fft_size)
    Conv = np.fft.irfft(P * Af, fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    
    return grad, p, c


def exact_subgradient(a, n_top=5):
    """
    Compute a subgradient of max_k c_k, focusing on the top n_top peaks.
    Returns (subgradient, max_value, convolution).
    """
    n = len(a)
    c = fast_autocorr(a)
    c_max = c.max()
    
    # Find top peaks
    top_idx = np.argsort(c)[::-1][:n_top]
    
    # Weight peaks by their values (peaks closer to max get more weight)
    weights = np.zeros(len(c))
    for idx in top_idx:
        idx = int(idx)
        # Weight: how close to the max?
        w = c[idx] / c_max  # between 0 and 1
        weights[idx] = w
    weights /= weights.sum()
    
    # Compute gradient as weighted sum of individual peak gradients
    fft_size = 1 << (2 * n - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    Af = np.fft.rfft(a[::-1], fft_size)
    Conv = np.fft.irfft(W * Af, fft_size)
    grad = 2.0 * Conv[n - 1:2 * n - 1]
    
    return grad, c_max, c


def nesterov_smoothed(a0, time_limit, beta_start=30.0, beta_end=3000.0,
                       beta_growth=1.006, eta_base=0.6, top_k_ratio=0.02,
                       restart_every=2000, verbose=True, seed=42):
    """Nesterov accelerated projected gradient with log-sum-exp smoothing."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        grad, p, c = compute_smoothed_grad(y, beta, top_k)
        
        # Adaptive step size
        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)
        
        z = y - eta * grad
        x_new = simplex_project(z)
        
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            # Deflation-based restart
            noise_scale = 0.05 / (1.0 + 0.1 * (t // restart_every))
            # Get subgradient of current max
            c_cur = fast_autocorr(best_a)
            max_idx = int(np.argmax(c_cur))
            # Deflation vector: gradient of the max peak
            j_start = max(0, max_idx - n + 1)
            j_end = min(n - 1, max_idx)
            deflation = np.zeros(n)
            if j_start <= j_end:
                idx = max_idx - np.arange(j_start, j_end + 1)
                deflation[j_start:j_end+1] = 2.0 * best_a[idx]
            # Random component
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.3 * deflation / (np.linalg.norm(deflation) + 1e-10) * best_a + 0.7 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nesterov restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
        
        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nesterov t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + ", beta=" + str(round(beta, 1)))
    
    return best_a, best_max_b


def nesterov_subgradient(a0, time_limit, n_top=5, verbose=True, seed=99):
    """Nesterov with exact subgradient (no smoothing)."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        grad, c_max, c = exact_subgradient(y, n_top=n_top)
        
        # Step size: proportional to 1/sqrt(t)
        eta = 2.0 / (np.sqrt(t_nest) * (np.linalg.norm(grad) + 1e-10))
        eta = min(eta, 0.05)
        
        z = y - eta * grad
        x_new = simplex_project(z)
        
        max_new = fast_autocorr(x_new).max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        if stall > 500:
            # Small random perturbation
            x_pert = best_a + 0.01 * rng.normal(0, 1, n) * best_a
            x_pert = np.maximum(x_pert, 1e-10)
            x_pert = x_pert / x_pert.sum()
            x = x_pert
            x_prev = x_pert.copy()
            t_nest = 1.0
            stall = 0
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    SubGrad t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
    
    return best_a, best_max_b


def systematic_local_search(a0, time_limit, verbose=True, seed=123):
    """
    Systematic local search: for each top peak, try all plausible single-pair moves.
    Much more thorough than random local search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 50
    
    iteration = 0
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iteration += 1
        c = fast_autocorr(a)
        c_max = c.max()
        
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find top peaks (up to 10)
        n_peaks = min(10, len(c))
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        improved = False
        
        for k in top_idx[:8]:
            if improved:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            
            if products.max() < 1e-20:
                continue
            
            # Get top 5 contributing pairs
            top_pairs = np.argsort(products)[::-1][:5]
            
            for pi in top_pairs:
                if improved:
                    break
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n:
                    continue
                
                # Determine which to reduce
                src_val = a[src_i]
                par_val = a[partner]
                
                if src_val >= par_val:
                    reduce_i = src_i
                else:
                    reduce_i = partner
                
                if a[reduce_i] < 1e-14:
                    continue
                
                # Compute "pressure" at each position from all top peaks
                pressure = np.zeros(n)
                for k2 in top_idx[:10]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        pressure[idx2] += a[k2i - idx2]
                        pressure[k2i - idx2] += a[idx2]
                
                # Find safe destinations (low pressure)
                # Exclude positions that are in any peak's contributing range
                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_idx[:10]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2 + 1] = True
                
                candidates = np.where(~in_peak)[0]
                if len(candidates) < 100:
                    candidates = np.arange(n)
                
                mask = candidates != reduce_i
                if not mask.any():
                    continue
                candidates = candidates[mask]
                
                # Try the 10 lowest pressure destinations
                cand_pressure = pressure[candidates]
                n_try = min(10, len(candidates))
                try_dsts = candidates[np.argsort(cand_pressure)[:n_try]]
                
                for dst in try_dsts:
                    if improved:
                        break
                    dst_int = int(dst)
                    
                    # Try different fractions with binary search
                    for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                        amt = a[reduce_i] * frac
                        if amt < 1e-18:
                            continue
                        a_trial = a.copy()
                        a_trial[reduce_i] -= amt
                        a_trial[dst_int] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
        
        if not improved:
            # Try random exploration
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
