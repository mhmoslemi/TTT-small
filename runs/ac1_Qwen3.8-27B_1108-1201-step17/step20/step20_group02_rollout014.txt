def fft_interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    fft_size = 1 << (max(old_n, new_n) * 2 - 1).bit_length()
    F = np.fft.rfft(f_norm, fft_size)
    freqs = np.fft.rfftfreq(fft_size)
    mask = np.abs(freqs) <= 0.48
    F *= mask
    f_long = np.fft.irfft(F, fft_size)[:new_n]
    f_long = np.maximum(f_long, 0.0)
    ss = f_long.sum()
    if ss < 1e-15:

    return f_long / ss

            keep_idx = rng_pert.choice(len(active_set), size=len(active_set) - n_pert, replace=False)
            new_adds = np.argsort(ac_curr)[::-1][:n_pert * 2]
            new_set = set([active_set[i] for i in keep_idx])
            for k in new_adds:
                new_set.add(int(k))
            active_set = sorted(new_set)
            no_improve = 0
            stagnation_count += 1
            continue

        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            stagnation_count = 0
        else:
            no_improve += 1

        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep_count = int(max_K * 0.75)
            keep_idx = np.argsort(margins)[:keep_count]
            active_set_base = [active_set[i] for i in keep_idx]
            violations = np.where(ac > M_curr - 1e-10)[0]
            for v in violations[:150]:
                v_int = int(v)
                if v_int not in active_set_base:
                    active_set_base.append(v_int)
            active_set = sorted(set(active_set_base))[:max_K]

        K = len(active_set)
        if K == 0:
            break

        coeffs = build_active_constraints(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(100, K // 3)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        cross_conv = fast_cross_convolve(y, g_dir)
        all_viol = np.where(cross_conv > M_curr + 1e-10)[0]
        if len(all_viol) > 0:
            viol_amt = cross_conv[all_viol] - M_curr
            top_v = all_viol[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        best_alpha = 1.0 / (iter_count + 2.0)
        best_trial_M = M_curr

        lo = 0.0
        hi = 1.0
        for _ in range(8):
            mid = (lo + hi) / 2.0
            trial = (1.0 - mid) * y + mid * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                hi = mid
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = mid
            if M_trial < M_curr:
                lo = mid
            else:
                hi = mid

        if iter_count <= 2:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", |S|=" + str(len(active_set)) + ", alpha=" + repr(best_alpha))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", iters=" + str(iter_count))
    return best_a, best_m


def spectral_peak_shave(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_convolve(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)

    fft_size = 1 << (2 * n - 1).bit_length()
    half = fft_size // 2 + 1
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        ac = fast_convolve(a)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()

        top_k = np.argsort(ac)[::-1][:min(20, len(ac))]

        A = np.fft.rfft(a, fft_size)
        A2 = A * A

        peak_contrib = np.zeros(half, dtype=np.float64)
        for k in top_k[:10]:
            k_int = int(k)
            phase_target = np.exp(-2j * np.pi * k_int / fft_size)
            contrib = np.abs(A2 * phase_target)
            peak_contrib += contrib

        mean_contrib = peak_contrib.mean()
        excess = peak_contrib - mean_contrib
        excess = np.maximum(excess, 0)

        A2_mod = A2 * (1.0 - 0.1 * excess / (mean_contrib + 1e-10))
        A2_mod[0] = A2[0]

        A_new = np.sqrt(np.abs(A2_mod)) * np.exp(1j * np.angle(A2_mod))
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        M_new = fast_convolve(a_new).max()
        if M_new < M_curr - 1e-14:
            a = a_new
        else:
            alpha = 0.1 / (1.0 + iter_count / 200.0)
            a_blend = (1.0 - alpha) * a + alpha * a_new
            a_blend = np.maximum(a_blend, 0.0)
            s_b = a_blend.sum()
            if s_b > 1e-15:
                a_blend = a_blend / s_b
                M_blend = fast_convolve(a_blend).max()
                if M_blend < M_curr:
                    a = a_blend

        if iter_count % 50 == 0 and time.monotonic() - t0 > 5.0:
            noise = rng.normal(0, 0.005, n) * a
            a_p = a + noise
            a_p = np.maximum(a_p, 0.0)
            sp = a_p.sum()
            if sp > 1e-15:
                a_p = a_p / sp
                M_p = fast_convolve(a_p).max()
                if M_p < best_m:
                    a = a_p
                    best_m = M_p
                    best_a = a.copy()

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    SpecShave iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))

        if iter_count > 5000 and time.monotonic() - t0 > time_limit * 0.8:
            break

    if verbose:
        elapsed = time.monotonic() - t0
        print("    SpecShave done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))
    return best_a, best_m


def peak_targeted_cd(a0, time_limit, rng_seed=77, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 4:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])

            if dst == src:
                continue

            found = False
            for frac in [0.003, 0.007, 0.015, 0.03, 0.06]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found:
                for _ in range(3):
                    j = int(rng.integers(n))
                    if j != src and a[src] > 1e-10:
                        frac = rng.uniform(0.002, 0.01)
                        amt = a[src] * frac
                        a[src] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0:
                for _ in range(15):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.01)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 300 == 0:
            elapsed = time.monotonic() - t0
            print("    CD (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", stall=" + str(stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def get_initial_sequence(t_global):
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
                return p_norm, sc
    except Exception:
        pass

    n = 5000
    x = np.linspace(0, 1, n)
    best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
    best_score = evaluate_safe(best_seq)
    print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))
    return best_seq, best_score


def diverse_initialization(base_seq, n, t_limit):
    t0 = time.monotonic()
    x = np.linspace(0, 1, n)
    candidates = []

    c1 = norm_sum1(np.sin(np.pi * x) ** 2)
    candidates.append(c1)

    c2 = norm_sum1(np.sin(np.pi * x) ** 4)
    candidates.append(c2)

    c3 = norm_sum1(np.cos(np.pi * x) ** 2 + 0.1)
    candidates.append(c3)

    c4 = norm_sum1(x ** 2 * (1 - x) ** 2)
    candidates.append(c4)

    c5 = norm_sum1(np.exp(-5 * (x - 0.5) ** 2))
    candidates.append(c5)

    best_local = None
    best_local_score = np.inf
    for i, c in enumerate(candidates):
        if time.monotonic() - t0 > t_limit:
            break
        c_interp = fft_interpolate(base_seq, len(c)) if len(base_seq) != len(c) else base_seq
        sc = evaluate_safe(c)
        if np.isfinite(sc) and sc < best_local_score:
            best_local_score = sc
            best_local = c.copy()

    return best_local


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = get_solver()
    best_seq, best_score = get_initial_sequence(t_global)
    n0 = len(best_seq)

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: LP-FW initial ===
    print("\n=== Phase 1: LP-FW initial (n=" + str(n0) + ") ===")
    t1 = min(remaining_time() * 0.20, 300.0)
    if t1 > 40 and n0 <= 10000:
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t1, solver,
                                  initial_K=600, max_K=4000, add_per_iter=60, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1_n" + str(n0), t_global)

    # === Phase 2: Spectral peak shaving ===
    print("\n=== Phase 2: Spectral peak shaving (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining_time() * 0.10, 150.0)
    if t2 > 30:
        a_out, _ = spectral_peak_shave(norm_sum1(best_seq), t2, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec_n" + str(len(best_seq)), t_global)

    # === Phase 3: LP-FW second pass ===
    print("\n=== Phase 3: LP-FW second (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining_time() * 0.18, 280.0)
    if t3 > 40 and len(best_seq) <= 10000:
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t3, solver,
                                  initial_K=700, max_K=5000, add_per_iter=80, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2_n" + str(len(best_seq)), t_global)

    # === Phase 4: Multi-n exploration ===
    print("\n=== Phase 4: Multi-n exploration ===")
    n_best = len(best_seq)
    offsets = [400, -400, 800, -800, 1200, -1200]
    try_ns = []
    for off in offsets:
        new_n = n_best + off
        if 3000 <= new_n <= 12000:
            try_ns.append(new_n)
    try_ns = list(dict.fromkeys(try_ns))
    try_ns.sort(key=lambda l: abs(l - n_best))

    for new_n in try_ns[:4]:
        rt = remaining_time()
        if rt < 100:
            break
        print("  Multi-n: n=" + str(new_n) + " (remaining=" + str(round(rt, 1)) + "s)")
        a_interp = fft_interpolate(best_seq, new_n)

        t5 = min(rt * 0.18, 180.0)
        if t5 > 30 and new_n <= 10000:
            a_out, _ = lp_fw_optimize(a_interp, t5, solver,
                                      initial_K=500, max_K=3500, add_per_iter=50, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

        rt = remaining_time()
        if rt > 60 and len(best_seq) <= 10000:
            t5b = min(rt * 0.08, 70.0)
            if t5b > 15:
                a_out, _ = spectral_peak_shave(norm_sum1(best_seq), t5b, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "spec_n" + str(len(best_seq)), t_global)

    # === Phase 5: LP-FW third pass on best ===
    print("\n=== Phase 5: LP-FW third (n=" + str(len(best_seq)) + ") ===")
    t6 = min(remaining_time() * 0.18, 260.0)
    if t6 > 40 and len(best_seq) <= 10000:
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t6, solver,
                                  initial_K=800, max_K=5500, add_per_iter=100, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw3_n" + str(len(best_seq)), t_global)

    # === Phase 6: Spectral + Coordinate ===
    print("\n=== Phase 6: Spectral + Coordinate (n=" + str(len(best_seq)) + ") ===")
    t7 = min(remaining_time() * 0.08, 100.0)
    if t7 > 20:
        a_out, _ = spectral_peak_shave(norm_sum1(best_seq), t7 * 0.5, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec2_n" + str(len(best_seq)), t_global)

    t8 = min(remaining_time() * 0.10, 120.0)
    if t8 > 20:
        a_out, _ = peak_targeted_cd(norm_sum1(best_seq), t8, rng_seed=99, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "cd_n" + str(len(best_seq)), t_global)

    # === Phase 7: Final LP-FW ===
    print("\n=== Phase 7: Final LP-FW (n=" + str(len(best_seq)) + ") ===")
    t9 = min(remaining_time() * 0.22, 220.0)
    if t9 > 30 and len(best_seq) <= 10000:
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t9, solver,
                                  initial_K=900, max_K=6000, add_per_iter=120, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # === Final coordinate polish ===
    rt = remaining_time()
    if rt > 15:
        a_out, _ = peak_targeted_cd(norm_sum1(best_seq), min(rt * 0.6, 60.0), rng_seed=2024, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
