```python
def multiplicative_weights(a, beta, top_k_frac=0.05, eta=0.1):
    """One step of multiplicative weights on the smoothed max autocorrelation."""
    n = len(a)
    b = fast_autocorr(a)
    size = 2*n - 1
    
    # Get top k peaks
    k = max(5, int(size * top_k_frac))
    top_idx = np.argsort(b)[::-1][:k]
    b_top = b[top_idx]
    
    # Soft max weights
    b_shift = b_top - b_top.max()
    weights = np.exp(b_shift * beta)
    weights /= weights.sum()
    
    # Gradient: for each peak k*, d(b[k*])/da[i] = a[k*-i] (if valid)
    grad = np.zeros(n)
    for w, k_star in zip(weights, top_idx):
        k_int = int(k_star)
        # d(a*a)[k_int]/da[i] = a[k_int - i] for 0 <= k_int-i < n
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += w * a[k_int - idx]
    
    # Multiplicative update
    a_new = a * np.exp(-eta * grad / max(eta, 1e-10))
    # Actually, let's be more careful
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    return a_new
```

```python
def compute_smoothed_grad_fft(a, beta, top_k=None):
    n = len(a)
    size = 2*n - 1
    b = fast_autocorr(a)
    
    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    
    if b_max <= 0:
        return np.zeros(n)
    
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s
```

```python
def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def compute_smoothed_grad(a, beta, top_k=None):
    """Compute gradient of the beta-smoothed max of autocorrelation."""
    n = len(a)
    b = fast_autocorr(a)
    size = 2 * n - 1
    
    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    
    if b_max <= 0:
        return np.zeros(n), b, 0.0
    
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s
    
    # grad[i] = sum_k p[k] * a[k-i] for valid indices
    # Use FFT: grad = (p * a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_len = n + size - 1
    fft_size_g = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size_g)
    Pr = np.fft.rfft(p, fft_size_g)
    conv = np.fft.irfft(Ar * Pr, fft_size_g)
    grad = conv[n - 1:2 * n - 1]
    
    return grad, b, b_max
```

    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def make_palindromic(a):
    """Make sequence palindromic by averaging with its reverse."""
    n = len(a)
    a_rev = a[::-1]
    return (a + a_rev) / 2.0

def compute_smoothed_grad(a, beta, top_k=None):
    n = len(a)
    b = fast_autocorr(a)
    size = 2 * n - 1

    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()

    if b_max <= 0:
        return np.zeros(n), b, 0.0

    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s

    a_rev = a[::-1]
    conv_len = n + size - 1
    fft_size_g = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size_g)
    Pr = np.fft.rfft(p, fft_size_g)
    conv = np.fft.irfft(Ar * Pr, fft_size_g)
    grad = conv[n - 1:2 * n - 1]

    return grad, b, b_max

def spectral_flatten(a, n_iters=200, alpha_init=0.05, verbose=False):
    """Iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = norm_sum1(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    fft_size = 1 << (2 * n - 2).bit_length()
    alpha = alpha_init
    rng = np.random.default_rng(12345)

    for it in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Target: geometric mean of non-DC power
        non_dc = power[1:]
        if len(non_dc) > 0 and non_dc.min() > 0:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        elif len(non_dc) > 0:
            target = non_dc.mean()
        else:
            target = power[0] * 0.1

        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.05:
                red = 1.0 - alpha * min(ratio - 1.0, 5.0)
                A[k] *= np.sqrt(max(0.2, red))
            elif ratio < 0.7:
                inc = 1.0 + alpha * 0.5 * (0.7 - ratio)
                A[k] *= np.sqrt(min(3.0, inc))

        a_new = np.fft.irfft(A, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 50 == 0 and it > 0:
            # Phase perturbation to escape local minima
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.01
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial

        alpha = max(0.005, alpha * 0.98)

        if verbose and it % 50 == 0:
            print(f"    Spectral it={it}: J={2*n*best_m:.6f}")

    return best_a, best_m

def smooth_gradient_optimize(a0, time_limit, beta_start=50.0, beta_end=3000.0,
                             eta_init=0.4, top_k_frac=0.02, seed=42,
                             palindromic=True, verbose=True):
    """Nesterov-accelerated smoothed gradient optimization."""
    n = len(a0)
    x = norm_sum1(a0)
    if palindromic:
        x = make_palindromic(x)
        x = norm_sum1(x)

    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    beta = beta_start
    eta = eta_init
    t_nest = 1.0
    x_prev = x.copy()
    stall = 0
    restart_count = 0
    use_mult = True  # Alternate between multiplicative and additive

    while time.monotonic() - t0 < time_limit:
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest ** 2)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        top_k = max(10, int((2 * n - 1) * top_k_frac))
        grad, b, b_max = compute_smoothed_grad(y, beta, top_k)

        if b_max < best_m - 1e-16:
            best_m = b_max
            best_a = y.copy()
            stall = 0

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            x_new = y
        else:
            grad_normed = grad / gnorm
            founded = False
            eta_trial = eta
            for _ in range(8):
                if use_mult:
                    log_update = -eta_trial * grad_normed
                    log_update = np.clip(log_update, -5, 5)
                    x_new = y * np.exp(log_update)
                    x_new = np.maximum(x_new, 1e-15)
                    x_new = x_new / x_new.sum()
                else:
                    z = y - eta_trial * grad_normed
                    x_new = simplex_project(z)

                m_new = fast_autocorr(x_new).max()
                if m_new <= b_max + 1e-14:
                    founded = True
                    break
                eta_trial *= 0.5

            if not founded:
                x_new = y
                eta = max(eta_trial, 0.01)

        m_final = fast_autocorr(x_new).max()
        if m_final < best_m - 1e-16:
            best_m = m_final
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.03, 1.5)
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new
        use_mult = not use_mult

        if stall % 30 == 0:
            beta = min(beta * (1.01 if stall < 100 else 1.05), beta_end)

        if stall > 400:
            stall = 0
            restart_count += 1
            t_nest = 1.0
            eta = eta_init
            beta = max(beta * 0.6, beta_start)

            strategy = rng.integers(6)
            if strategy == 0:
                x_r = best_a * np.exp(0.015 * rng.standard_normal(n))
            elif strategy == 1:
                shift = int(rng.integers(1, max(2, n // 30)))
                x_r = np.roll(best_a, shift)
            elif strategy == 2:
                # Scale subsets
                mask = rng.random(n) > 0.6
                x_r = best_a.copy()
                x_r[mask] *= rng.uniform(0.7, 0.9)
                x_r[~mask] *= rng.uniform(1.1, 1.4)
            elif strategy == 3:
                # Blend with uniform
                x_r = 0.85 * best_a + 0.15 * np.ones(n) / n
            elif strategy == 4:
                # Local smoothing
                kernel_size = min(21, n // 50)
                if kernel_size % 2 == 0:
                    kernel_size += 1
                kernel = np.ones(kernel_size) / kernel_size
                x_smooth = np.convolve(best_a, kernel, mode='same')
                x_r = 0.7 * best_a + 0.3 * x_smooth
            else:
                # Mirror perturbation
                x_r = best_a.copy()
                mid = n // 2
                pert = 0.02 * rng.standard_normal(mid)
                x_r[:mid] *= np.exp(pert)
                x_r[mid:] *= np.exp(pert[::-1])

            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            if palindromic:
                x_r = make_palindromic(x_r)
                x_r = norm_sum1(x_r)
            x = x_r
            x_prev = x_r.copy()

            elapsed = time.monotonic() - t0
            if verbose and elapsed < time_limit:
                print(f"    SmoothOpt restart {restart_count} ({elapsed:.0f}s): J={2*n*best_m:.6f}, beta={beta:.0f}")

        if verbose and stall % 2000 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"    SmoothOpt ({elapsed:.0f}s): J={2*n*best_m:.6f}, beta={beta:.0f}, eta={eta:.3f}, stall={stall}")

    return best_a, best_m

def lp_fw_optimize(a0, time_limit, solver, initial_K=400, max_K=3000,
                   verbose=True, palindromic=True):
    """LP-Frank-Wolfe with dynamic active set."""
    n = len(a0)
    a = norm_sum1(a0)
    if palindromic:
        a = make_palindromic(a)
        a = norm_sum1(a)

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    ac = fast_autocorr(a)
    active = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 800

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set size
        active_sorted = sorted(active)
        if len(active_sorted) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_sorted])
            keep_idx = np.argsort(margins)[:max_K]
            active = set(int(active_sorted[i]) for i in keep_idx)
            active_sorted = sorted(active)

        K_cur = len(active_sorted)
        if K_cur == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K_cur, n), dtype=np.float64)
        for jj, k in enumerate(active_sorted):
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = a[k - idx]

        # If palindromic, add constraint g[i] = g[n-1-i]
        if palindromic and n > 1:
            palin_constraints = [g_var[:n//2] == g_var[n-1-n//2:-1][::-1]] if n % 2 == 0 else [g_var[:n//2] == g_var[n-n//2-1:-1][::-1]]
        else:
            palin_constraints = []

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        constraints.extend(palin_constraints)

        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active) > initial_K // 2:
                ac_tmp = fast_autocorr(a)
                active = set(int(k) for k in np.argsort(ac_tmp)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Detect violations and add to active set
        fft_size = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(a, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ay = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]

        violations = np.where(conv_ay > M_curr + 1e-7)[0]
        if len(violations) > 0:
            add_k = min(30, len(violations))
            top_v = violations[np.argsort(conv_ay[violations])[::-1][:add_k]]
            for k in top_v:
                active.add(int(k))

        # Periodically refresh with current top peaks
        if t % 300 == 0:
            new_top = np.argsort(ac)[::-1][:initial_K // 2]
            for k in new_top:
                active.add(int(k))

        # Adaptive line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        candidates = [best_alpha, best_alpha * 0.5, best_alpha * 1.5,
                      1.0 / (t + 1), 0.03, 0.05, 0.08, 0.1, 0.15]
        for alpha in candidates:
            alpha = max(0.005, min(0.9, alpha))
            trial = (1.0 - alpha) * a + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        if palindromic:
            a = make_palindromic(a)
            a = norm_sum1(a)

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW it={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, |S|={len(active)}")

    return best_a, best_m

def peak_coord_descent(a0, time_limit, verbose=True, seed=777):
    """Coordinate descent targeting top autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 3000:
        ac = fast_autocorr(a)
        size = 2 * n - 1
        top_k = np.argsort(ac)[::-1][:min(30, size)]
        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            if not (0 <= partner < n):
                continue

            # Try moving mass from src to reduce this peak
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue

                # Try multiple destinations
                destinations = []
                # Position that minimizes contribution to this peak
                best_dst = None
                best_val = float('inf')
                for _ in range(20):
                    dst = int(rng.integers(n))
                    if dst == src or dst == partner:
                        continue
                    # Check if dst reduces the peak
                    if 0 <= k_int - dst < n and a[k_int - dst] > 0:
                        score = a[k_int - dst]
                    else:
                        score = -1  # Good: doesn't contribute
                    if score < best_val:
                        best_val = score
                        best_dst = dst
                    destinations.append(dst)

                if best_dst is not None:
                    destinations = [best_dst] + destinations[:5]
                else:
                    destinations = destinations[:10]

                found_better = False
                for dst in destinations:
                    if dst == src or dst == partner:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-17:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        found_better = True
                        break

                if found_better:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 50:
                # Random exploration
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_new = fast_autocorr(a_trial).max()
                    if M_new < best_m:
                        a = a_trial
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break

        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc ({elapsed:.0f}s): J={2*n*best_m:.6
