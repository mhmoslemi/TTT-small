def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def generate_initializations(n, parent_seq=None):
    """Generate diverse initial sequences for n."""
    inits = []
    x = np.linspace(0, 1, n)
    j = np.arange(n)

    # Parent (interpolated)
    if parent_seq is not None and len(parent_seq) > 0:
        inits.append(interpolate_seq(parent_seq, n))

    # Sin^2 taper
    inits.append(norm_sum1(np.sin(np.pi * x) ** 2))

    # Sin^4 taper (sharper)
    inits.append(norm_sum1(np.sin(np.pi * x) ** 4))

    # Chirp-like: (1+cos(2*pi*x))^0.5
    inits.append(norm_sum1((1.0 + np.cos(2.0 * np.pi * x)) ** 0.5))

    # Gaussian
    inits.append(norm_sum1(np.exp(-2.0 * (x - 0.5) ** 2 / 0.08)))

    # Asymmetric: more mass on left
    inits.append(norm_sum1(np.exp(-4.0 * (x - 0.35) ** 2 / 0.06)))

    # Triangle
    inits.append(norm_sum1(1.0 - np.abs(2.0 * x - 1.0)))

    # Flat with edges tapered
    taper = 0.1
    mask = np.ones(n)
    edge = int(taper * n)
    ramp = np.linspace(0, 1, edge)
    mask[:edge] = ramp
    mask[-edge:] = ramp[::-1]
    inits.append(norm_sum1(mask))

    # Structured random (low-frequency noise)
    rng = np.random.default_rng(42)
    freqs = rng.uniform(0.5, 3.0, 5)
    phases = rng.uniform(0, 2 * np.pi, 5)
    sig = np.ones(n)
    for f, p in zip(freqs, phases):
        sig += 0.3 * np.cos(2 * np.pi * f * x + p)
    inits.append(norm_sum1(np.abs(sig)))

    # Pure chirp real part
    inits.append(norm_sum1(np.abs(np.cos(np.pi * j * j / n) * 0.5 + 0.5)))

    # Two-lobe
    inits.append(norm_sum1(np.exp(-30 * (x - 0.3) ** 2) + np.exp(-30 * (x - 0.7) ** 2)))

    return inits


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw(a0, time_limit, solver, tol_frac=0.005, max_K=8000, verbose=True):
    """LP-Frank-Wolfe with tolerance-based active set and parabolic line search."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac_y = fast_convolve(y)
    best_m = ac_y.max()
    t0 = time.monotonic()

    # Tolerance-based active set: all lags within tol_frac of max
    threshold = best_m * (1.0 - tol_frac)
    active_set = set(np.where(ac_y >= threshold)[0].tolist())
    # Also include top 200 by value
    top200 = np.argsort(ac_y)[::-1][:min(200, 2 * n - 1)]
    for k in top200:
        active_set.add(int(k))
    active_set = sorted(active_set)

    no_improve = 0
    max_no_improve = 500
    iter_count = 0
    stagnation_count = 0
    best_alpha_history = []

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if stagnation_count > 5:
                if verbose:
                    print("    LP-FW: persistent stagnation, stopping")
                break
            # Restart: expand active set
            ac_curr = fast_convolve(y)
            M_c = ac_curr.max()
            new_threshold = M_c * (1.0 - tol_frac * 0.5)
            active_set = set(np.where(ac_curr >= new_threshold)[0].tolist())
            top300 = np.argsort(ac_curr)[::-1][:min(300, 2 * n - 1)]
            for k in top300:
                active_set.add(int(k))
            active_set = sorted(active_set)[:max_K]
            no_improve = 0
            stagnation_count += 1
            continue

        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            stagnation_count = 0
        else:
            no_improve += 1

        # Update active set: tolerance-based
        if iter_count % 20 == 1:
            threshold = M_curr * (1.0 - tol_frac)
            new_set = set(np.where(ac >= threshold)[0].tolist())
            # Always keep current active set members that are still relevant
            for k in active_set:
                if ac[k] >= M_curr * (1.0 - tol_frac * 2):
                    new_set.add(k)
            # Add top contributors
            topK = np.argsort(ac)[::-1][:min(100, 2 * n - 1)]
            for k in topK:
                new_set.add(int(k))
            active_set = sorted(new_set)
            if len(active_set) > max_K:
                # Prune: keep tightest constraints
                margins = np.array([M_curr - ac[k] for k in active_set])
                order = np.argsort(margins)
                active_set = [active_set[i] for i in order[:max_K]]

        K = len(active_set)
        if K == 0:
            break

        coeffs = build_constraint_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(100, K // 3)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations: add significantly violated constraints
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        # Parabolic line search
        # Evaluate at alpha = 1.0, 0.5, and adaptive point
        def eval_alpha(alpha):
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                return float('inf')
            trial = trial / s_t
            return fast_convolve(trial).max()

        a1, a2, a3 = 1.0, 0.5, 0.25
        m1, m2, m3 = eval_alpha(a1), eval_alpha(a2), eval_alpha(a3)

        # Fit parabola through 3 points to find minimum
        if m1 < m2 and m1 < m3:
            best_alpha = a1
        elif m3 < m1 and m3 < m2:
            best_alpha = a3
        else:
            # Standard parabolic interpolation
            denom = (m1 - 2.0 * m2 + m3)
            if abs(denom) > 1e-20:
                delta = (a1 - a2) * (a1 - m2 + (a2 - a3) * (m1 - m2) / (m1 - 2.0 * m2 + m3)) / (2.0 * (a1 - a2) * (m1 - m2 - (m1 - m3) * (a2 - a3) / (a1 - a3)))
                # Simpler: use the symmetric formula
                alpha_min = a2 + 0.5 * (a1 - a3) * (m1 - m3) / (m1 - 2.0 * m2 + m3)
                alpha_min = np.clip(alpha_min, 0.05, 1.0)
            else:
                alpha_min = 0.5
            best_alpha = alpha_min

        # Also try a few more points around the best
        candidates = [best_alpha, best_alpha * 0.8, best_alpha * 1.2,
                      1.0 / (iter_count + 2.0), 0.1, 0.05]
        candidates = [c for c in candidates if 0.01 <= c <= 1.0]
        best_trial_M = min(m1, m2, m3)
        best_trial_alpha = best_alpha
        for ca in candidates:
            if ca == best_alpha or ca == a1 or ca == a2 or ca == a3:
                continue
            mc = eval_alpha(ca)
            if mc < best_trial_M:
                best_trial_M = mc
                best_trial_alpha = ca

        if iter_count <= 2:
            y = g_dir.copy()
        else:
            y = (1.0 - best_trial_alpha) * y + best_trial_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        best_alpha_history.append(best_trial_alpha)

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", |S|=" + str(len(active_set)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", iters=" + str(iter_count))
    return best_a, best_m


def pairwise_coordinate_descent(a0, time_limit, n_top_lags=30, n_dest=200, step_fracs=None, rng_seed=123, verbose=False):
    """Fine-grained pairwise mass transfer coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    if step_fracs is None:
        step_fracs = [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]

    stall = 0
    max_stall = 800
    total_evals = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:n_top_lags]

        # Find top contributing elements for the peak lag
        k_peak = int(top_lags[0])
        i_start = max(0, k_peak - n + 1)
        i_end = min(n, k_peak + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[k_peak - idx]

        # Get top contributing source indices
        top_sources = idx[np.argsort(contrib)[::-1][:20]]
        top_sources = top_sources[contrib[np.argsort(contrib)[::-1][:20]] > 1e-12]

        # Precompute impact vectors for top sources (only check top_lags)
        # For source i moving to dest j: delta at lag k = a[k-j] - a[k-i]
        improved = False

        for src in top_sources[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(src)
            if a[src] < 1e-12:
                continue

            # Score destinations by their effect on top lags (want to decrease max)
            dest_scores = np.zeros(n)
            for k2 in top_lags[:10]:
                k2i = int(k2)
                src_at_k2 = 0.0
                si = k2i - src
                if 0 <= si < n:
                    src_at_k2 = a[si]
                # For each dest j, the effect at lag k2 is a[k2-j] - a[k2-src]
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    a_vals = a[k2i - idx2]
                    dest_scores[idx2] += a_vals
                    # We want dest where a[k2-dest] is small (reduces this lag)
                    # But also need to not increase other lags
            # We want to move to where the sum a[k2-j] is small for top lags
            # i.e., dest_scores[j] = sum over top lags of a[k2-j], minimize this
            # But exclude the source itself
            dest_scores[src] = np.inf

            valid_dests = np.where(a > 1e-15)[0]
            if len(valid_dests) > n_dest:
                # Sample destinations: mix of low-score and random
                valid_scores = dest_scores[valid_dests]
                n_low = n_dest // 2
                low_idx = valid_dests[np.argsort(valid_scores)[:n_low]]
                rand_idx = valid_dests[rng.integers(0, len(valid_dests), size=n_dest - n_low)]
                dest_candidates = np.concatenate([low_idx, rand_idx])
            else:
                dest_candidates = valid_dests

            found_improve = False
            for dst in dest_candidates[:n_dest]:
                if time.monotonic() - t0 > time_limit:
                    break
                dst = int(dst)
                if dst == src:
                    continue

                for frac in step_fracs:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_convolve(a_trial).max()
                    total_evals += 1
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found_improve = True
                        break
                if found_improve:
                    break

        if not improved:
            stall += 1
            # Random perturbation every 200 stalls
            if stall % 200 == 0:
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.01)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            total_evals += 1
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    CoordDescent (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", stall=" + str(stall))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    CoordDescent done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", evals=" + str(total_evals))
    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
                return p_norm, sc
    except Exception:
        pass

    n = 5000
    x = np.linspace(0, 1, n)
    best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
    best_score = evaluate_safe(best_seq)
    print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = get_solver()
    best_seq, best_score = get_initial_sequence()
    n0 = len(best_seq)

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: Diverse initialization search ===
    print("\n=== Phase 1: Initialization search ===")
    try_ns_init = [n0, max(3000, n0 - 1000), min(20000, n0 + 1000), max(3000, n0 - 2000), min(20000, n0 + 2000)]
    try_ns_init = sorted(set([l for l in try_ns_init if 2000 <= l <= 25000]))

    best_init_seq = best_seq.copy()
    best_init_score = best_score

    for trial_n in try_ns_init[:4]:
        rt = remaining_time()
        if rt < 60:
            break
        inits = generate_initializations(trial_n, best_seq)
        for ii, ini in enumerate(inits):
            if rt < 30:
                break
            sc = evaluate_safe(ini)
            if np.isfinite(sc) and sc < best_init_score:
                best_init_score = sc
                best_init_seq = ini.copy()
                print("  Better init: n=" + str(trial_n) + ", type=" + str(ii) + ", score=" + repr(sc))
        best_seq, best_score = try_update(best_seq, best_score, best_init_seq, "init_n" + str(trial_n), t_global)

    print("  Best init after phase 1: score=" + repr(best_score) + ", n=" + str(len(best_seq)))

    # === Phase 2: LP-FW main optimization ===
    print("\n=== Phase 2: LP-FW main (n=" + str(len(best_seq)) + ") ===")
    n_main = len(best_seq)
    t2 = min(remaining_time() * 0.30, 480.0)
    if t2 > 60 and n_main <= 20000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t2, solver,
                         tol_frac=0.005, max_K=min(10000, 3 * n_main), verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_main_n" + str(n_main), t_global)

    # === Phase 3: Coordinate descent ===
    print("\n=== Phase 3: Coordinate descent (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining_time() * 0.15, 250.0)
    if t3 > 40:
        a_out, _ = pairwise_coordinate_descent(norm_sum1(best_seq), t3,
                                                n_top_lags=30, n_dest=300,
                                                step_fracs=[0.001, 0.002, 0.005, 0.01, 0.02],
                                                rng_seed=77, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord1_n" + str(len(best_seq)), t_global)

    # === Phase 4: LP-FW second pass (refined active set) ===
    print("\n=== Phase 4: LP-FW refined (n=" + str(len(best_seq)) + ") ===")
    t4 = min(remaining_time() * 0.20, 350.0)
    if t4 > 60 and len(best_seq) <= 20000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t4, solver,
                         tol_frac=0.003, max_K=min(12000, 4 * len(best_seq)), verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2_n" + str(len(best_seq)), t_global)

    # === Phase 5: Multi-n exploration ===
    print("\n=== Phase 5: Multi-n exploration ===")
    n_best = len(best_seq)
    try_ns = sorted(set([
        n_best - 500, n_best - 1000, n_best + 500, n_best + 1000,
        n_best - 2000, n_best + 2000, n_best + 3000, n_best - 3000
    ]))
    try_ns = [l for l in try_ns if 2000 <= l <= 25000]
    # Sort by proximity to best
    try_ns.sort(key=lambda l: abs(l - n_best))

    for new_n in try_ns[:5]:
        rt = remaining_time()
        if rt < 100:
            break
        print("  Multi-n: n=" + str(new_n) + " (remaining=" + str(round(rt, 1)) + "s)")
        a_long = interpolate_seq(best_seq, new_n)

        # Quick LP-FW
        t5 = min(rt * 0.18, 150.0)
        if t5 > 40 and new_n <= 20000:
            a_out, _ = lp_fw(a_long, t5, solver,
                             tol_frac=0.005, max_K=min(8000, 3 * new_n), verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

        # Quick coordinate descent
        rt = remaining_time()
        if rt > 40:
            t5b = min(rt * 0.10, 80.0)
            if t5b > 15:
                a_out, _ = pairwise_coordinate_descent(norm_sum1(best_seq), t5b,
                                                       n_top_lags=20, n_dest=200,
                                                       step_fracs=[0.002, 0.005, 0.01],
                                                       rng_seed=200 + new_n, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n" + str(new_n), t_global)

    # === Phase 6: LP-FW third pass (tighter tolerance) ===
    print("\n=== Phase 6: LP-FW tight (n=" + str(len(best_seq)) + ") ===")
    t6 = min(remaining_time() * 0.20, 300.0)
    if t6 > 50 and len(best_seq) <= 20000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t6, solver,
                         tol_frac=0.002, max_K=min(15000, 5 * len(best_seq)), verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw3_n" + str(len(best_seq)), t_global)

    # === Phase 7: Final coordinate descent (very fine) ===
    print("\n=== Phase 7: Fine coordinate descent (n=" + str(len(best_seq)) + ") ===")
    t7 = min(remaining_time() * 0.15, 200.0)
    if t7 > 30:
        a_out, _ = pairwise_coordinate_descent(norm_sum1(best_seq), t7,
                                                n_top_lags=40, n_dest=400,
                                                step_fracs=[0.0005, 0.001, 0.002, 0.003, 0.005, 0.008],
                                                rng_seed=555, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final_n" + str(len(best_seq)), t_global)

    # === Phase 8: Final LP-FW + coord ===
    rt = remaining_time()
    if rt > 30 and len(best_seq) <= 20000:
        print("\n=== Phase 8: Final LP-FW (n=" + str(len(best_seq)) + ") ===")
        t8 = min(rt * 0.5, 120.0)
        if t8 > 20:
            a_out, _ = lp_fw(norm_sum1(best_seq), t8, solver,
                             tol_frac=0.002, max_K=min(12000, 4 * len(best_seq)), verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

        rt = remaining_time()
        if rt > 20:
            a_out, _ = pairwise_coordinate_descent(norm_sum1(best_seq), min(rt * 0.7, 60.0),
                                                    n_top_lags=30, n_dest=300,
                                                    step_fracs=[0.001, 0.002, 0.005],
                                                    rng_seed=999, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_very_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
