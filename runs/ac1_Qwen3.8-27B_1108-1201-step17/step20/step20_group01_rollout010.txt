    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))

def spectral_flatten(a0, time_limit, alpha=0.3, verbose=True):
    """
    Iterative spectral flattening: work in frequency domain to reduce
    the dynamic range of |FFT(a)|^2, which corresponds to flattening
    the linear convolution.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
            
        # Compute FFT
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        mag[mag < 1e-30] = 1e-30
        
        # Flatten the spectrum: reduce peaks
        # A_new = A / |A|^(2*alpha) => |A_new| = |A|^(1-2*alpha)
        A_mod = A / (mag ** (2 * alpha))
        
        # Inverse FFT
        a_new = np.fft.irfft(A_mod, fft_size)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check improvement
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and t % 100 == 0:
            print("  SpecFlat t={} ({:.1f}s) max_acf={:.7f} obj={:.7f}".format(
                t, time.monotonic() - t0, best_m, 2*n*best_m))
    
    return best_a, 2.0 * n * best_m

def grad_descent_smooth(a0, time_limit, n_top=50, lr0=0.1, verbose=True):
    """
    Gradient descent on smoothed objective: sum of top-K convolution values.
    Uses FFT to compute the gradient efficiently.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution
        c = fast_autocorr(a)
        M_curr = c.max()
        cur_val = 2.0 * n * M_curr
        
        if cur_val < best_val - 1e-13:
            best_val = cur_val
            best_a = a.copy()
        
        # Top-K smoothing
        k_sel = min(n_top, len(c))
        idx = np.argsort(c)[::-1][:k_sel]
        top_vals = c[idx]
        
        # Soft weighting (exponential)
        temp = 50.0
        w = np.exp(temp * (top_vals - top_vals[0]))
        w = w / w.sum()
        
        # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation)
        # This is 2 * corr(w, a)[j] = 2 * conv(a_reversed, w)[j]... 
        # Actually: grad[j] = 2 * sum_m a[m] * w[j+m]
        # = 2 * (a * w_flipped)[j] where w_flipped[i] = w[n-1-i]... 
        # Let me think more carefully.
        # grad[j] = 2 * sum_{k} w_k * a[k-j] for valid k-j
        # Let m = k-j, so k = m+j:
        # grad[j] = 2 * sum_m a[m] * w[m+j]
        # This is the cross-correlation of w and a, evaluated at lag j.
        # corr(w,a)[j] = sum_m w[m+j] * a[m]
        # In FFT: ifft(FFT(w, L) * conj(FFT(a, L))) gives corr for circular
        # For linear: pad to L >= n + k_sel - 1
        
        # Simpler: just use the convolution formula
        # grad[j] = 2 * sum_{k: k-j in [0,n-1]} w[k] * a[k-j]
        # = 2 * sum_{m=0}^{n-1} w[m+j] * a[m]
        # This is cross-correlation. Compute via FFT.
        
        L = 2 * fft_size  # enough for linear cross-correlation
        W = np.fft.rfft(w, L)
        A_fft = np.fft.rfft(a, L)
        # Cross-correlation: ifft(W * conj(A))
        corr = np.fft.irfft(W * np.conj(A_fft), L)
        grad = 2.0 * corr[:n]
        
        # Or equivalently: conv(a, w_reversed)
        # w_rev = w[::-1], then conv(a, w_rev)[j] = sum_m a[m] * w_rev[j-m] = sum_m a[m] * w[n-1-j+m]...
        # That's not quite right. Let me just use the FFT cross-correlation.
        
        # Mirror descent update
        lr = lr0 / (1.0 + 0.01 * t)
        logits = np.log(a + 1e-30) - lr * grad
        logits -= logits.max()
        a_new = np.exp(logits)
        a_new = a_new / a_new.sum()
        a = a_new
        
        if verbose and t % 200 == 0:
            print("  GradDesc t={} ({:.1f}s) obj={:.7f}".format(t, time.monotonic()-t0, best_val))
    
    return best_a, best_val

def coord_peel(a0, time_limit, n_targets=10, seed=42, verbose=True):
    """Coordinate peeling: move mass from peak contributors to quiet positions."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 5000
    fracs = [0.001, 0.002, 0.005, 0.01, 0.02]
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        top_k = np.argsort(c)[::-1][:min(n_targets, len(c))]
        improved = False
        
        for ki in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(ki)
            ws = max(0, ki - n + 1)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            contrib = a[idx] * a[ki - idx]
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-15:
                continue
            
            # Find quiet positions (low pressure)
            pressure = np.zeros(n)
            for k2 in top_k[:6]:
                k2 = int(k2)
                ws2 = max(0, k2 - n + 1)
                we2 = min(n - 1, k2)
                if ws2 > we2:
                    continue
                idx2 = np.arange(ws2, we2 + 1)
                pressure[idx2] += a[idx2] * a[k2 - idx2]
            pressure[src] = 1e10
            quiet = np.argsort(pressure)[:20]
            
            for dst in quiet:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-18:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-15:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            if stall > 200:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
                stall = stall  # keep counting
                if stall % 1000 == 0:
                    print("  Peel ({:.1f}s) obj={:.7f} stall={}".format(time.monotonic()-t0, 2*n*best_m, stall))
    
    return best_a, 2.0 * n * best_m

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

Hmm wait, I need to be careful with the FFT convention. Let me think again.

The linear cross-correlation: (w ⋆ a)[m] = sum_n w[n+m] * a[n]

In the frequency domain: FFT(w ⋆ a) = conj(FFT(a)) * FFT(w)

So: w ⋆ a = ifft(conj(FFT(a)) * FFT(w))

Or equivalently: w ⋆ a = ifft(FFT(w) * conj(FFT(a)))

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based gradient descent with smooth-max (softmax) approximation
2. Spectral shaping (iterative magnitude compression in frequency domain)
3. Coordinate peeling for local refinement
4. Multi-length exploration and multi-start

import numpy as np
import time


def fast_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_conv(a).max()


def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_gradient(a, w_full):
    """
    Compute gradient of sum_k w_full[k] * c[k] where c = conv(a,a).
    grad[j] = 2 * sum_k w_full[k] * a[k-j] (for valid k-j in [0,n-1])
    = 2 * cross_correlation(w_full, a)[j]
    Computed via FFT.
    """
    n = len(a)
    len_w = len(w_full)
    L = 1 << (len_w + n - 1).bit_length()
    W = np.fft.rfft(w_full, L)
    A = np.fft.rfft(a, L)
    # Cross-correlation: ifft(conj(A) * W)
    corr = np.fft.irfft(np.conj(A) * W, L)
    return 2.0 * corr[:n]


def smooth_max_gradient(a, beta, n_top=None):
    """
    Compute gradient of smooth approximation to max(conv(a,a)).
    Uses softmax weighting over top conv values.
    """
    n = len(a)
    c = fast_conv(a)
    if n_top is not None and n_top < len(c):
        # Focus on top values
        threshold = np.sort(c)[::-1][min(n_top, len(c)) - 1]
        w_full = np.exp(beta * (c - c.max()))
        w_full[c < threshold - 1e-10] *= 0.1
    else:
        w_full = np.exp(beta * (c - c.max()))
    w_sum = w_full.sum()
    if w_sum < 1e-30:
        w_full = np.zeros(len(c))
        w_full[np.argmax(c)] = 1.0
        return w_full, c.max()
    w_full = w_full / w_sum
    grad = compute_gradient(a, w_full)
    return grad, c.max()


def grad_descent(a0, time_limit, beta0=20.0, beta_end=100.0, n_top=50,
                 lr0=0.06, verbose=True):
    """
    Mirror-descent gradient minimization with smooth-max objective.
    Adaptive beta (annealing) and learning rate.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    n_iter = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(elapsed / max(time_limit, 1.0), 1.0)

        # Anneal beta
        beta = beta0 + (beta_end - beta0) * (progress ** 0.5)
        # Anneal learning rate
        lr = lr0 / (1.0 + 0.02 * n_iter)

        grad, M_curr = smooth_max_gradient(a, beta, n_top)
        cur_val = 2.0 * n * M_curr

        if cur_val < best_val - 1e-13:
            best_val = cur_val
            best_a = a.copy()

        # Mirror descent (multiplicative update)
        logits = np.log(a + 1e-30) - lr * grad
        logits -= logits.max()
        a_new = np.exp(logits)
        a_new = a_new / a_new.sum()
        a = a_new
        n_iter += 1

        if verbose and n_iter % 300 == 0:
            print("  GD iter={} ({:.1f}s) beta={:.0f} lr={:.4f} obj={:.7f}".format(
                n_iter, elapsed, beta, lr, best_val))

    return best_a, best_val


def spectral_shape(a0, time_limit, alpha=0.3, verbose=True):
    """
    Iterative spectral shaping: compress the dynamic range of |FFT(a)|^2
    to flatten the convolution in time domain.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    n_iter = 0

    while time.monotonic() - t0 < time_limit:
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        mag[mag < 1e-30] = 1e-30

        # Compress magnitude: |A_new| = |A|^(1-2*alpha)
        A_mod = A / (mag ** (2.0 * alpha))

        a_new = np.fft.irfft(A_mod, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        m_new = fast_conv(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()

        a = a_new
        n_iter += 1

        if verbose and n_iter % 200 == 0:
            print("  SpecShape iter={} ({:.1f}s) obj={:.7f}".format(
                n_iter, time.monotonic() - t0, 2 * n * best_m))

    return best_a, 2.0 * n * best_m

        improved = False

        for ki in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(ki)
            ws = max(0, ki - n + 1)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            contrib = a[idx] * a[ki - idx]
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # Compute pressure
            pressure = np.zeros(n)
            for k2 in top_k[:8]:
                k2 = int(k2)
                ws2 = max(0, k2 - n + 1)
                we2 = min(n - 1, k2)
                if ws2 > we2:
                    continue
                idx2 = np.arange(ws2, we2 + 1)
                pressure[idx2] += a[idx2] * a[k2 - idx2]
            pressure[src] = 1e10
            quiet = np.argsort(pressure)[:25]

            for dst in quiet:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-18:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_conv(a_t).max()
                    if Mt < best_m - 1e-15:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 300 and stall % 100 == 0:
                # Random restart perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_conv(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
            if verbose and stall % 2000 == 0:
                print("  Peel ({:.1f}s) obj={:.7f} stall={}".format(
                    time.monotonic() - t0, 2 * n * best_m, stall))

    return best_a, 2.0 * n * best_m


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_alternating(a0, time_limit, verbose=False):
    """Alternate between spectral shaping and gradient descent."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    n_cycles = 0

    while time.monotonic() - t0 < time_limit:
        # Spectral shaping pass (short)
        pass_time = min(20.0, (time_limit - (time.monotonic() - t0)) * 0.3)
        if pass_time > 2:
            a, _ = spectral_shape(a, pass_time, alpha=0.25, verbose=False)

        # Gradient descent pass
        pass_time = min(30.0, (time_limit - (time.monotonic() - t0)) * 0.5)
        if pass_time > 3:
            a, _ = grad_descent(a, pass_time, beta0=30, beta_end=80,
                                n_top=60, lr0=0.06, verbose=False)

        sc = obj_val(a)
        if sc < best_val - 1e-13:
            best_val = sc
            best_a = a.copy()
        a = a
        n_cycles += 1

        if verbose and n_cycles % 3 == 0:
            print("  Alt cycle={} ({:.1f}s) obj={:.7f}".format(
                n_cycles, time.monotonic() - t0, best_val))

    return best_a, best_val


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent: n={} score={:.7f}".format(len(p), best_score))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_init = 10000
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = obj_val(best_seq)
        print("Default init n={} score={:.7f}".format(n_init, best_score))

    n0 = len(best_seq)

    # Phase 1: Quick spectral shaping to improve initial point
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: SpecShape (n={}) ===".format(n0))
    t1 = min(remaining * 0.08, 120.0)
    if t1 > 15:
        a_out, _ = spectral_shape(best_seq, t1, alpha=0.35, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P1: {:.7f}".format(best_score))

    # Phase 2: Main gradient descent (long)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P2: GradDescent main (n={}) ===".format(len(best_seq)))
    t2 = min(remaining * 0.22, 380.0)
    if t2 > 30:
        a_out, _ = grad_descent(best_seq, t2, beta0=15, beta_end=80,
                                n_top=60, lr0=0.07, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P2: {:.7f}".format(best_score))

    # Phase 3: Coordinate peeling
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P3: CoordPeel (n={}) ===".format(len(best_seq)))
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_out, _ = coord_peel(best_seq, t3, n_targets=12, seed=42, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P3: {:.7f}".format(best_score))

    # Phase 4: Gradient descent with different params
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P4: GradDescent v2 (n={}) ===".format(len(best_seq)))
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_out, _ = grad_descent(best_seq, t4, beta0=40, beta_end=150,
                                n_top=30, lr0=0.04, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P4: {:.7f}".format(best_score))

    # Phase 5: Spectral shaping with different alpha
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P5: SpecShape alpha=0.45 (n={}) ===".format(len(best_seq)))
    t5 = min(remaining * 0.08, 130.0)
    if t5 > 15:
        a_out, _ = spectral_shape(best_seq, t5, alpha=0.45, verbose=False)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P5: {:.7f}".format(best_score))

    # Phase 6: Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = min(len(best_seq) + 3000, 15000)
    if n_long != len(best_seq) and remaining > 150:
        print("\n=== P6: Long n={} ===".format(n_long))
        t6 = min(remaining * 0.18, 280.0)
        if t6 > 30:
            a_long = interpolate_seq(best_seq, n_long)
            a_out1, _ = spectral_shape(a_long, t6 * 0.35, alpha=0.3, verbose=False)
            a_out2, _ = grad_descent(a_out1, t6 * 0.65, beta0=20, beta_end=70,
                                     n_top=80, lr0=0.05, verbose=False)
            sc1 = eval_safe(a_out1)
            sc2 = eval_safe(a_out2)
            if np.isfinite(min(sc1, sc2)) and min(sc1, sc2) < best_score:
                best_score = min(sc1, sc2)
                best_seq = a_out1 if sc1 < sc2 else a_out2
                print("  P6: {:.7f}".format(best_score))

    # Phase 7: Alternating spectral + gradient
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P7: Alternating (n={}) ===".format(len(best_seq)))
    t7 = min(remaining * 0.15, 250.0)
    if t7 > 30:
        a_out, _ = try_alternating(best_seq, t7, verbose=False)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P7: {:.7f}".format(best_score))

    # Phase 8: Final coordinate peeling (long)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P8: Final Peel (n={}) ===".format(len(best_seq)))
    t8 = min(remaining * 0.25, 400.0)
    if t8 > 20:
        a_out, _ = coord_peel(best_seq, t8, n_targets=15, seed=123, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P8: {:.7f}".format(best_score))

    # Phase 9: Final gradient polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print("\n=== P9: Final GD (n={}) ===".format(len(best_seq)))
        a_out, _ = grad_descent(best_seq, remaining * 0.7, beta0=50,
                                beta_end=120, n_top=25, lr0=0.03, verbose=False)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P9: {:.7f}".format(best_score))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.7f} n={} t={:.1f}s".format(best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: using fallback")
        n_fb = 10000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)

    return [float(x) for x in best_seq]
