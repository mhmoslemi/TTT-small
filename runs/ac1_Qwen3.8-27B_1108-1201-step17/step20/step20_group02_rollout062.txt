def get_solver():
    global _SOLVER
    if _SOLVER is not None:
        return _SOLVER
    if cp is None:
        _SOLVER = None
        return None

    candidates = []
    for name in ("CLARABEL", "MOSEK", "PDLP", "ECOS", "SCS", "CVXOPT", "ECOS_PP"):
        if hasattr(cp, name):
            candidates.append(getattr(cp, name))

    for s in candidates:
        try:
            v = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(v)), [cp.sum(v) == 1])

def normalize_sum1(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)[:size]


def fast_cross(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = b.size
    size = n + m - 1
    L = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(A * B, L)[:size]


def score_after_sum_one(a):

def top_indices(c, k):
    c = np.asarray(c, dtype=np.float64)
    m = c.size
    if k <= 0:
        return np.empty(0, dtype=int)
    if k >= m:
        return np.arange(m, dtype=int)
    idx = np.argpartition(c, -k)[-k:]
    return idx


def interpolate_seq(f, new_n):
    f = np.asarray(f, dtype=np.float64).ravel()
    old_n = f.size
    if old_n == 0:
        return np.ones(max(1, new_n), dtype=np.float64)
    if new_n <= 0:
        return np.ones(1, dtype=np.float64)
    s = float(f.sum())
    if s <= 1e-15:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    f = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    g = np.interp(new_x, old_x, f)
    g = np.maximum(g, 0.0)
    gs = float(g.sum())
    if gs <= 1e-15:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    return g / gs

def line_search(a, g, current_m, alphas=None):
    if alphas is None:
        alphas = (0.05, 0.10, 0.20, 0.35, 0.55, 0.80)
    best_a = a
    best_m = current_m
    for alpha in alphas:
        trial = (1.0 - float(alpha)) * a + float(alpha) * g
        trial = normalize_sum1(trial)
        m = fast_conv(trial).max()
        if m < best_m:
            best_m = m
            best_a = trial
    return best_a, best_m


def weighted_sensitivity(a, active, mu):
    n = a.size
    s = np.zeros(n, dtype=np.float64)
    active = list(active)
    mu = np.asarray(mu, dtype=np.float64)
    for idx in range(len(active)):
        w = float(mu[idx])
        if w <= 0.0:
            continue
        k = int(active[idx])
        lo = k - n + 1
        if lo < 0:
            lo = 0
        hi = k
        if hi >= n:
            hi = n - 1
        if lo > hi:
            continue
        s[lo:hi + 1] += w * a[k - hi:k - lo + 1][::-1]
    return s


def fast_fw(a, time_limit, k_active=90, beta=12.0, verbose=False):
    a = normalize_sum1(a)
    n = a.size
    t0 = time.monotonic()
    if n == 0:
        return a

    best_a = a.copy()
    best_c = fast_conv(best_a)
    best_m = float(best_c.max())

    no_improve = 0
    iter_ct = 0
    beta_cur = float(beta)
    limit = max(0.0, time_limit)

    while time.monotonic() - t0 < limit:
        iter_ct += 1
        c = fast_conv(a)
        m = float(c.max())

        if m < best_m - 1e-15:
            best_m = m
            best_a = a.copy()
            no_improve = 0

        K = min(n, max(30, int(k_active)))
        active = top_indices(c, K)
        peaks = c[active]
        mu = peaks + 1e-12
        mu /= float(mu.sum())

        s = weighted_sensitivity(a, active, mu)
        if s.max() <= s.min() + 1e-30:
            s = np.ones(n, dtype=np.float64)

        p05 = float(np.percentile(s, 5))
        p95 = float(np.percentile(s, 95))
        rng_s = max(p95 - p05, 1e-15)
        u = (s - p05) / rng_s
        u = np.clip(u, 0.0, 1.6)

        g = np.exp(-beta_cur * u)
        low_mask = u <= 0.25
        g = g + np.where(low_mask, 1.0, 0.0)
        g = np.maximum(g, 0.0)
        gs = float(g.sum())
        if gs <= 1e-15:
            g = np.ones(n, dtype=np.float64) / float(n)
        else:
            g /= gs

        trial, tm = line_search(a, g, best_m)
        if tm < best_m - 1e-15:
            a = trial
            best_a = trial.copy()
            best_m = tm
            no_improve = 0
        else:
            no_improve += 1

        if no_improve == 60 or no_improve % 120 == 0:
            scale = 0.015 + 0.01 * min(8.0, float(no_improve) / 120.0)
            rng = np.random.default_rng(1234 + iter_ct)
            pert = best_a + scale * best_a * rng.standard_normal(n)
            pert = normalize_sum1(pert)
            pm = float(fast_conv(pert).max())
            if pm < best_m - 1e-12:
                a = pert
                best_a = pert.copy()
                best_m = pm
                no_improve = 0
            else:
                beta_cur *= 1.28
                if beta_cur > 100.0:
                    beta_cur = 5.0
                if beta_cur < 3.0:
                    beta_cur = 3.0

        if no_improve > 1200:
            break

        if verbose and iter_ct % 80 == 0:
            el = time.monotonic() - t0
            print("    fast_fw it=%d t=%.1fs M2n=%.10f noimp=%d" % (iter_ct, el, 2.0 * n * best_m, no_improve))

    if verbose:
        el = time.monotonic() - t0
        print("    fast_fw done t=%.1fs best=%.10f it=%d" % (el, 2.0 * n * best_m, iter_ct))
    return best_a


def build_coefficients(a, active, n):
    K = len(active)
    C = np.zeros((K, n), dtype=np.float64)
    for r, kk in enumerate(active):
        k = int(kk)
        lo = k - n + 1
        if lo < 0:
            lo = 0
        hi = k
        if hi >= n:
            hi = n - 1
        if lo > hi:
            continue
        C[r, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return C


def lp_fw(a, time_limit, solver, k_init=110, k_max=300, verbose=False):

    a = normalize_sum1(a)

def coordinate_polish(a, time_limit, top_p=18, max_pairs=35, verbose=False):
    a = normalize_sum1(a)
    n = a.size
    t0 = time.monotonic()
    if n == 0:
        return a

    best_a = a.copy()
    best_c = fast_conv(best_a)
    best_m = float(best_c.max())
    rng = np.random.default_rng(2024)
    no_improve = 0
    limit = max(0.0, time_limit)
    frac_list = (0.08, 0.18, 0.35, 0.60)

    while time.monotonic() - t0 < limit:
        c = fast_conv(a)
        m = float(c.max())
        if m < best_m - 1e-15:
            best_m = m
            best_a = a.copy()

        active = top_indices(c, min(n, max(8, int(top_p))))
        peaks = c[active]
        mu = peaks + 1e-12
        mu /= float(mu.sum())
        s = weighted_sensitivity(a, active, mu)

        threshold = max(1e-12, float(np.median(a)) * 0.05)
        src_mask = a > threshold
        if int(src_mask.sum()) < 40:
            src_mask = a > 1e-12
        src_candidates = np.where(src_mask)[0]
        if src_candidates.size == 0:
            break
        s_src = s[src_candidates]
        order = np.argsort(s_src)[::-1][:45]
        src_idx = src_candidates[order]
        src_bool = np.zeros(n, dtype=bool)
        src_bool[src_idx] = True

        dest_candidates = np.where(~src_bool)[0]
        if dest_candidates.size == 0:
            dest_candidates = np.arange(n)
        s_dest = s[dest_candidates]
        d_order = np.argsort(s_dest)[:45]
        dest_idx = dest_candidates[d_order]

        pairs = []
        for si in src_idx:
            for di in dest_idx:
                if si != di:
                    pred = float(s[si] - s[di])
                    if pred > 0.0:
                        pairs.append((pred, int(si), int(di)))
        pairs.sort(reverse=True)

        best_trial = None
        best_trial_m = best_m

        for pred, si, di in pairs[:max_pairs]:
            base_amt = float(a[si])
            if base_amt <= 1e-14:
                continue
            for fr in frac_list:
                amt = base_amt * float(fr)
                trial = a.copy()
                trial[si] -= amt
                trial[di] += amt
                trial = np.maximum(trial, 0.0)
                ss = float(trial.sum())
                if ss <= 1e-15:
                    continue
                trial /= ss
                tm = float(fast_conv(trial).max())
                if tm < best_trial_m - 1e-15:
                    best_trial_m = tm
                    best_trial = trial

        if best_trial is not None:
            a = best_trial
            best_a = a.copy()
            best_m = best_trial_m
            no_improve = 0
        else:
            improved = False
            for _ in range(12):
                si = int(rng.integers(0, n))
                di = int(rng.integers(0, n))
                if si == di or a[si] <= 1e-12:
                    continue
                amt = a[si] * float(rng.uniform(0.05, 0.25))
                trial = a.copy()
                trial[si] -= amt
                trial[di] += amt
                trial = np.maximum(trial, 0.0)
                ss = float(trial.sum())
                if ss <= 1e-15:
                    continue
                trial /= ss
                tm = float(fast_conv(trial).max())
                if tm < best_m - 1e-15:
                    a = trial
                    best_a = trial.copy()
                    best_m = tm
                    no_improve = 0
                    improved = True
                    break

            if not improved:
                no_improve += 1
                if no_improve > 30:
                    nit = max(1, n // 8)
                    idx = rng.choice(n, size=nit, replace=False)
                    trial = best_a.copy()
                    trial[idx] += 0.02 * trial[idx] * rng.standard_normal(nit)
                    trial = normalize_sum1(trial)
                    tm = float(fast_conv(trial).max())
                    if tm < best_m - 1e-15:
                        a = trial
                        best_a = trial.copy()
                        best_m = tm
                        no_improve = 0
                if no_improve > 400:
                    break

        if verbose and (iter_ct if 'iter_ct' in locals() else 0) % 40 == 0:
            el = time.monotonic() - t0
            print("    coord t=%.1fs 2nM=%.10f noimp=%d" % (el, 2.0 * n * best_m, no_improve))

    if verbose:
        el = time.monotonic() - t0
        print("    coord done t=%.1fs best=%.10f" % (el, 2.0 * n * best_m))
    return best_a

        th = 2.0 * np.pi * m * x
        cols.append(np.sin(th)[:, None])
        cols.append(np.cos(th)[:, None])
    return np.hstack(cols)

                best_c = cand.copy()
                best_a = log_shape_from_global(ccur)
                step = max(0.01, step * 0.88)
                improved = True
                break
        if not improved:
            step = min(0.8, step * 1.12)

    if verbose:
        el = time.monotonic() - t0
        print("    lowdim_log t=%.1fs n=%d 2nM=%.10f" % (el, n, scorer_best))
    return best_a


def small_n_search(n, time_limit, initial=None, verbose=False):
    t0 = time.monotonic()
    if n <= 0:
        return np.ones(1)
    if initial is not None:
        a = normalize_sum1(interpolate_seq(initial, n))
    else:
        a = np.ones(n, dtype=np.float64) / float(n)

    half = max(5.0, 0.55 * time_limit)
    a = fast_fw(a, half, k_active=min(100, n), beta=18.0, verbose=False)
    a = coordinate_polish(a, max(5.0, time_limit - half), top_p=12, max_pairs=30, verbose=False)
    if verbose:
        print("    small_n t=%.1fs n=%d 2nM=%.10f" % (time.monotonic() - t0, n, 2.0 * n * fast_conv(a).max()))
    return a


def get_initial_sequence(t0):
    # Parent, if present.
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        arr = arr[np.isfinite(arr)]
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size > 0 and float(arr.sum()) > 1e-12:
            a = normalize_sum1(arr)
            sc = evaluate_safe(a)
            if np.isfinite(sc):
                print("Parent: n=%d score=%.10f" % (a.size, sc))
                return a, sc
    except Exception:
        pass

    # Fallback templates.
    candidates = []
    for n in (1024, 2048, 4096):
        candidates.append(np.ones(n, dtype=np.float64) / float(n))

    # Two-end block templates, experimentally can be useful as seeds.
    for n in (2048, 4096):
        for frac in (0.30, 0.35, 0.42):
            m = max(2, int(n * frac))
            if 2 * m >= n:
                m = n // 3
            for ratio in (1.0, 1.35, 1.6):
                W = ratio / (1.0 + ratio)
                R = 1.0 / (1.0 + ratio)
                a = np.zeros(n, dtype=np.float64)
                a[:m] = W / float(m)
                a[-m:] = R / float(m)
                a = normalize_sum1(a)
                candidates.append(a)

        # Three endpoint-ish blocks
        m = max(2, n // 6)
        for wmid in (0.08, 0.15, 0.25):
            a = np.zeros(n, dtype=np.float64)
            wl = (1.0 - wmid) * 0.62
            wr = (1.0 - wmid) * 0.38
            a[:m] = wl / float(m)
            a[m:n - m] = wmid / float(max(1, n - 2 * m))
            a[-m:] = wr / float(m)
            a = normalize_sum1(a)
            candidates.append(a)

    best = None
    best_sc = float("inf")
    for a in candidates:
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a

    if best is None:
        best = np.ones(2048, dtype=np.float64) / 2048.0
        best_sc = evaluate_safe(best)

    print("Initial fallback: n=%d score=%.10f" % (best.size, best_sc))
    return best, best_sc


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    total_budget = min(float(budget_s) - 45.0, 1745.0)
    if total_budget <= 60.0:
        a, _ = get_initial_sequence(t_global)
        return [float(x) for x in a]

    print("Total search budget: %.1fs" % total_budget)
    solver = get_solver()

    best_seq, best_score = get_initial_sequence(t_global)

    def remaining():
        return max(0.0, total_budget - (time.monotonic() - t_global))

    # 1) Fast active-set dual heuristic on parent length.
    t1 = min(remaining() * 0.18, 280.0)
    if t1 > 60:
        print("\nPhase 1: fast_fw initial")
        a2 = fast_fw(best_seq, t1, k_active=95, beta=13.0, verbose=True)
        best_seq, best_score = update_best(best_seq, best_score, a2, "fast1", t_global)

    # 2) Exact LP Frank-Wolfe burst.
    t2 = min(remaining() * 0.15, 240.0)
    if t2 > 60 and len(best_seq) <= 10000 and solver is not None:
        print("\nPhase 2: LP-FW initial")
        a2 = lp_fw(best_seq, t2, solver, k_init=110, k_max=320, verbose=True)
        best_seq, best_score = update_best(best_seq, best_score, a2, "lp1", t_global)

    # 3) Coordinate polish.
    t3 = min(remaining() * 0.12, 180.0)
    if t3 > 30:
        print("\nPhase 3: coordinate polish")
        a2 = coordinate_polish(best_seq, t3, top_p=20, max_pairs=40, verbose=False)
        best_seq, best_score = update_best(best_seq, best_score, a2, "coord1", t_global)

    # 4) Low-dimensional log-polynomial refinement.
    t4 = min(remaining() * 0.10, 130.0)
    if t4 > 40:
        print("\nPhase 4: lowdim log search")
        n_search = min(len(best_seq), 6144)
        a2 = lowdim_log_search(best_seq, t4, n_search, verbose=True)
        if a2 is not None:
            a2 = interpolate_seq(a2, len(best_seq))
            a2 = coordinate_polish(a2, min(remaining() * 0.05, 70.0), top_p=18, max_pairs=35, verbose=False)
            best_seq, best_score = update_best(best_seq, best_score, a2, "lowdim1", t_global)

    # 5) Small-n basin exploration, then interpolate back.
    t5_total = min(remaining() * 0.12, 180.0)
    if t5_total > 90:
        print("\nPhase 5: small-n basin")
        for small_n in (512, 1024):
            if remaining() < 60:
                break
            t_for_n = min(remaining() * 0.35, 140.0)
            a2 = small_n_search(small_n, t_for_n, initial=best_seq, verbose=True)
            a2 = interpolate_seq(a2, len(best_seq))
            a2 = fast_fw(a2, min(remaining() * 0.07, 50.0), k_active=90, beta=15.0, verbose=False)
            a2 = coordinate_polish(a2, min(remaining() * 0.04, 35.0), top_p=16, max_pairs=30, verbose=False)
            best_seq, best_score = update_best(best_seq, best_score, a2, "smalln%d" % small_n, t_global)

    # 6) Multi-length exploration.
    current_n = len(best_seq)
    candidate_ns = set()
    for f in (0.7, 0.85, 1.15, 1.35):
        nn = int(round(current_n * f))
        if 900 <= nn <= 14000:
            candidate_ns.add(nn)
    candidate_ns.add(current_n)
    candidate_ns = sorted(candidate_ns)
    # Prefer near current and powers of 2-ish.
    candidate_ns.sort(key=lambda nn: abs(nn - current_n))

    t6_total = min(remaining() * 0.18, 260.0)
    if t6_total > 90:
        print("\nPhase 6: multi-n")
        for nn in candidate_ns[:4]:
            if remaining() < 75:
                break
            a_interp = interpolate_seq(best_seq, nn)
            a2 = fast_fw(a_interp, min(remaining() * 0.18, 110.0), k_active=95, beta=12.0, verbose=False)
            a2 = coordinate_polish(a2, min(remaining() * 0.09, 55.0), top_p=20, max_pairs=35, verbose=False)
            best_seq, best_score = update_best(best_seq, best_score, a2, "multi_n%d" % nn, t_global)

    # 7) Second exact LP burst, if worthwhile.
    t7 = min(remaining() * 0.22, 300.0)
    if t7 > 70 and len(best_seq) <= 10000 and solver is not None:
        print("\nPhase 7: LP-FW second")
        a2 = lp_fw(best_seq, t7, solver, k_init=130, k_max=380, verbose=True)
        best_seq, best_score = update_best(best_seq, best_score, a2, "lp2", t_global)

    # 8) Final fast + coordinate polishing.
    t8 = min(remaining() * 0.22, 260.0)
    if t8 > 40:
        print("\nPhase 8: final fast_fw")
        a2 = fast_fw(best_seq, min(t8 * 0.6, 150.0), k_active=85, beta=14.0, verbose=True)
        best_seq, best_score = update_best(best_seq, best_score, a2, "fast_final", t_global)

    t9 = min(remaining() * 0.30, 180.0)
    if t9 > 20:
        print("\nPhase 9: final coordinate polish")
        a2 = coordinate_polish(best_seq, t9, top_p=24, max_pairs=45, verbose=True)
        best_seq, best_score = update_best(best_seq, best_score, a2, "coord_final", t_global)

    # Final validation and cleanup.
    best_seq = normalize_sum1(best_seq)
    final_score = evaluate_safe(best_seq)
    if not np.isfinite(final_score):
        fallback = np.ones(4096, dtype=np.float64) / 4096.0
        final_score = evaluate_safe(fallback)
        best_seq = fallback

    final_score2 = score_after_sum_one(best_seq)
    if np.isfinite(final_score2) and final_score2 < final_score:
        best_seq = normalize_sum1(best_seq)

    elapsed = time.monotonic() - t_global
    print("\nFINAL: score=%.10f n=%d elapsed=%.1fs" % (final_score, len(best_seq), elapsed))
    return [float(x) for x in best_seq]
