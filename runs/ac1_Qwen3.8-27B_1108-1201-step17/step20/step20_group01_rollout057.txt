```python
def simplex_project(y):
    """Project y onto the probability simplex {a: a >= 0, sum(a) = 1}."""
    n = len(y)
    if n == 0:
        return y
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(y - theta, 0)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (Nesterov-accelerated proximal gradient) with log-sum-exp smoothed objective
2. Adaptive temperature schedule for the smoothing
3. Simplex projection for non-negativity and normalization constraints
4. Coordinate descent polish for final refinement
5. Multi-restart with length variation
"""

import numpy as np
import time
import cvxpy as cp


def fast_autocorr(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def simplex_project(y):
    """Project y onto the probability simplex {a: a >= 0, sum(a) = 1}."""
    n = len(y)
    if n <= 0:
        return y
    u = np.sort(y, kind='mergesort')[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1) / rho_val
    return np.maximum(y - theta, 0)


def normalize(a):
    """Normalize a to sum to 1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_val(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def eval_safe(a):
    """Safely evaluate the sequence using the global evaluate_sequence function."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_smoothed_grad(a, tau, fft_size_cross):
    """Compute gradient of (1/tau)*log(sum_k exp(tau*b[k])) where b=conv(a,a)."""
    n = len(a)
    # Compute autocorrelation
    fft_size_auto = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size_auto)
    b = np.fft.irfft(A * A, fft_size_auto)[:2 * n - 1]
    
    # Softmax weights (numerically stable)
    b_max = b.max()
    exp_vals = np.exp(tau * (b - b_max))
    p = exp_vals / exp_vals.sum()
    
    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j]
    P = np.fft.rfft(p, fft_size_cross)
    A_cross = np.fft.rfft(a, fft_size_cross)
    cross = np.fft.irfft(P * np.conj(A_cross), fft_size_cross)
    grad = 2.0 * cross[:n]
    
    smoothed_obj = b_max + np.log(exp_vals.sum()) / tau
    return grad, p, smoothed_obj


def fista_optimize(a_init, time_limit, tau_start=50.0, tau_end=5000.0, 
                   max_iter=500000, verbose=True):
    """FISTA with log-sum-exp smoothing and adaptive temperature."""
    n = len(a_init)
    x = normalize(a_init)
    y = x.copy()
    t_mom = 1.0
    
    fft_size_cross = 1 << (3 * n - 2).bit_length()
    
    best_x = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    no_improve = 0
    stall_limit = 5000
    
    for k in range(1, max_iter + 1):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        # Adaptive temperature: increase over time
        frac = min(1.0, k / (max_iter * 0.5))
        tau = tau_start + (tau_end - tau_start) * frac
        
        # Compute gradient at y
        grad, p, f_y = compute_smoothed_grad(y, tau, fft_size_cross)
        
        # Step size: use 1/L where L is estimated from the Hessian bound
        # L <= 2/tau + correction; use backtracking
        alpha = tau / 2.0  # initial step size estimate
        
        # Try the step with Armijo backtracking (limited)
        best_step = alpha
        best_f = float('inf')
        for _ in range(5):
            z = y - best_step * grad
            x_new = simplex_project(z)
            # Evaluate smoothed objective at x_new
            fft_size_auto = 1 << (2 * n - 1).bit_length()
            A_new = np.fft.rfft(x_new, fft_size_auto)
            b_new = np.fft.irfft(A_new * A_new, fft_size_auto)[:2 * n - 1]
            b_new_max = b_new.max()
            f_new = b_new_max + np.log(np.sum(np.exp(tau * (b_new - b_new_max)))) / tau
            if f_new <= f_y + 0.5 * (1.0 / best_step) * np.sum((x_new - y) ** 2) + 1e-12:
                break
            best_step *= 0.5
        
        x_new = simplex_project(y - best_step * grad)
        
        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y_new = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
        y_new = simplex_project(y_new)
        
        x = x_new
        y = y_new
        t_mom
