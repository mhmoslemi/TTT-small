Minimizes 2*n*max(autocorr(a))/(sum(a))^2 using:
1. Nesterov-accelerated subgradient with top-K gradient approximation
2. LP-based Frank-Wolfe with golden section line search
3. Second-order peak shaving
4. Multi-length exploration
5. Final fine-grained polish
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0])
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(z - theta, 0.0)

def compute_gradient_topk(a, p, K):
    """Compute gradient using top-K approximation.
    grad[i] = sum_{k in topK} p[k] * (a[k-i] + a[i-k])
    """
    n = len(a)
    grad = np.zeros(n)
    # Only iterate over top K indices of p
    top_indices = np.argsort(p)[::-1][:K]
    for k in top_indices:
        k = int(k)
        pk = p[k]
        if pk < 1e-15:
            continue
        # a[k-i] is non-zero for i in [k-n+1, k] ∩ [0, n-1]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            grad[idx] += pk * a[k - idx]  # a[k-i] for i in idx
            # a[i-k] is non-zero for i in [k, k+n-1] ∩ [0, n-1]
            # i-k in [0, n-1] means i in [k, k+n-1]
        i_start2 = max(0, k)
        i_end2 = min(n - 1, k + n - 1)
        if i_start2 <= i_end2:
            idx2 = np.arange(i_start2, i_end2 + 1)
            grad[idx2] += pk * a[idx2 - k]  # a[i-k] for i in idx2
    
    return grad

                return 1e10
            trial = trial / st
            return fast_autocorr(trial).max()
        
        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f_c1 = line_obj(c1)
        f_c2 = line_obj(c2)
        for _ in range(10):
            if f_c1 < f_c2:
                hi = c2
                c2 = c1
                f_c2 = f_c1
                c1 = hi - gr * (hi - lo)
                f_c1 = line_obj(c1)
            else:
                lo = c1
                c1 = c2
                f_c1 = f_c2
                c2 = lo + gr * (hi - lo)
                f_c2 = line_obj(c2)
        
        alpha_best = (lo + hi) / 2.0
        a_new = (1.0 - alpha_best) * a + alpha_best * g_dir
        a_new = np.maximum(a_new, 0.0)
        st = a_new.sum()
        if st > 1e-15:
            a_new = a_new / st
        a = a_new
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    FW-LP it=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " alpha=" + repr(round(alpha_best, 4)))
    
    return best_a, best_m

def peak_shave_2nd(a0, time_limit, top_k=15, verbose=True):
    """Second-order peak shaving: for each peak, find optimal (src, dst, amount)."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        b = fast_autocorr(a)
        n_peaks = min(top_k, len(b))
        peak_indices = np.argsort(b)[::-1][:n_peaks]
        
        # Compute pressure map
        pressure = np.zeros(n)
        for k_int in peak_indices[:8]:
            ki = int(k_int)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                pressure[idx] += a[ki - idx]
        
        improved = False
        
        for k_int in peak_indices[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k_int)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[ki - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = int(idx[np.argmax(contrib)])
            partner = ki - src
            if partner < 0 or partner >= n:
                partner = src
            
            reduce_idx = src if a[src] >= a[partner] else partner
            
            if a[reduce_idx] < 1e-14:
                continue
            
            # Find best destination using second-order analysis
            # Moving amount 'amt' from reduce_idx to dst changes b[k] by:
            # delta_b[k] = -amt*(a[k-ri] + a[ri-k]) + amt*(a[k-dst] + a[dst-k])
            # We want to minimize the max over k after the move.
            
            safe_mask = pressure < np.percentile(pressure, 60)
            safe_mask[reduce_idx] = False
            if safe_mask.sum() < 50:
                safe_mask = np.ones(n, dtype=bool)
                safe_mask[reduce_idx] = False
            
            candidate_dsts = np.where(safe_mask)[0]
            # Sample a subset for speed
            if len(candidate_dsts) > 500:
                candidate_dsts = rng.choice(candidate_dsts, 500, replace=False)
            
            best_dst = -1
            best_frac = 0
            best_trial_m = best_m
            
            for dst in candidate_dsts[:200]:
                dst = int(dst)
                if dst == reduce_idx:
                    continue
                # Try several fractions
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amt = a[reduce_idx] * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_idx] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_trial_m:
                        best_trial_m = m_trial
                        best_dst = dst
                        best_frac = frac
            
            if best_dst >= 0 and best_trial_m < best_m - 1e-16:
                amt = a[reduce_idx] * best_frac
                a[reduce_idx] -= amt
                a[best_dst] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                best_m = best_trial_m
                best_a = a.copy()
                improved = True
                stall = 0
        
        if not improved:
            stall += 1
            if stall % 30 == 0:
                noise_scale = min(0.02, 0.003 * stall / 30)
                pert = rng.normal(0, noise_scale, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-12)
                a_r = a_r / a_r.sum()
                m_r = fast_autocorr(a_r).max()
                if m_r < best_m:
                    best_m = m_r
                    best_a = a_r.copy()
                a = a_r if m_r < best_m + 0.005 else best_a.copy()
        
        if verbose and stall > 0 and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakShave2 stall=" + str(stall) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def parametric_init(n, seed=42):
    """Try several parametric families and return the best."""
    x = np.linspace(0.0, 1.0, n)
    best_a = None
    best_m = float('inf')
    rng = np.random.default_rng(seed)
    
    # Power laws
    for alpha in [0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
        for beta_p in [0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
            a_raw = (x + 0.005) ** alpha * (1.0 - x + 0.005) ** beta_p
            a = norm_sum1(a_raw)
            m = fast_autocorr(a).max()
            if m < best_m:
                best_m = m
                best_a = a.copy()
    
    # Gaussian mixtures
    for c1 in [0.3, 0.4, 0.5]:
        for s1 in [0.1, 0.15, 0.2]:
            a_raw = np.exp(-((x - c1) ** 2) / (2 * s1 ** 2))
            a = norm_sum1(a_raw)
            m = fast_autocorr(a).max()
            if m < best_m:
                best_m = m
                best_a = a.copy()
    
    # Random
    for _ in range(20):
        a = norm_sum1(rng.random(n) ** rng.uniform(0.5, 3.0))
        m = fast_autocorr(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()
    
    return best_a, best_m

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

def propose_candidate(budget_s=1800.0):
    """Main search orchestration."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Time budget: " + repr(time_budget) + "s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None
    
    # If no parent, use parametric init
    if best_seq is None:
        n_base = 6700
        best_seq, _ = parametric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))
    
    n0 = len(best_seq)
    
    # Phase 1: Quick L-BFGS (fast convergence)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: L-BFGS (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        a_out, _ = lbfgs_smooth(best_seq, t1, beta_start=30.0, beta_end=300.0, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs", t_global)
    
    # Phase 2: Nesterov subgradient (fine-tuning)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov subgradient (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 60:
        a_out, _ = nesterov_subgrad(best_seq, t2, beta_start=50.0, beta_end=1500.0, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nesterov", t_global)
    
    # Phase 3: LP Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Frank-Wolfe LP (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30 and n_best <= 8000:
        a_out, _ = frank_wolfe_lp(best_seq, t3, active_size=300, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fwlp", t_global)
    
    # Phase 4: Peak shaving 2nd order
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Peak shaving 2nd (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, _ = peak_shave_2nd(best_seq, t4, top_k=15, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakshave2", t_global)
    
    # Phase 5: Nesterov round 2 (higher beta)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Nesterov 2 (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        a_out, _ = nesterov_subgrad(best_seq, t5, beta_start=200.0, beta_end=3000.0, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nesterov2", t_global)
    
    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-length ===")
    nearby_lengths = []
    for dn in [-300, -100, 100, 300, 500]:
        nl = n_best + dn
        if 5000 < nl <= 9000 and nl != n_best:
            nearby_lengths.append(nl)
    
    for new_n in nearby_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 90:
            break
        print("  Trying n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)
        
        # Quick Nesterov
        t6 = min(remaining * 0.3, 120.0)
        if t6 > 20:
            a_out, _ = nesterov_subgrad(a_long, t6, beta_start=100.0, beta_end=1000.0, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "n" + str(new_n), t_global)
        
        # Quick peak shave
        remaining = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining * 0.3, 80.0)
        if t6b > 15:
            a_out, _ = peak_shave_2nd(best_seq, t6b, top_k=10, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ps_n" + str(new_n), t_global)
    
    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final polish (n=" + str(n_best) + ") ===")
    if remaining > 40:
        t7a = min(remaining * 0.4, 120.0)
        if t7a > 20:
            a_out, _ = peak_shave_2nd(best_seq, t7a, top_k=10, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_ps", t_global)
        
        remaining = time_budget - (time.monotonic() - t_global)
        t7b = min(remaining * 0.4, 100.0)
        if t7b > 15:
            a_out, _ = nesterov_subgrad(best_seq, t7b, beta_start=500.0, beta_end=5000.0, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_ng", t_global)
    
    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        a_raw = (x + 0.01) ** 1.0 * (1.0 - x + 0.01) ** 1.0
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)
    
    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
