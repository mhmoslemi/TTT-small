def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def build_lp_matrix(a, active_list, n):
    K = len(active_list)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj in range(K):
        ki = int(active_list[jj])
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[ki - idx]
    return coeffs

def lp_fw(a0, time_limit, solver, init_K=300, max_K=8000, add_per=40, verbose=True, random_frac=0.1):
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:init_K])
    no_improve = 0
    stall_limit = 3000
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Refresh active set periodically
        if t % 15 == 0 or t == 1:
            top_lags = np.argsort(ac)[::-1]
            target_size = min(max_K, init_K + t * 5)
            for k in top_lags[:target_size]:
                active_set.add(int(k))
            # Add some random lags
            n_rand = int(random_frac * target_size)
            rand_lags = rng.integers(0, 2*n-1, n_rand)
            for k in rand_lags:
                active_set.add(int(k))
            if len(active_set) > max_K:
                clist = sorted(active_set)
                margins = np.array([M_curr - ac[k] for k in clist])
                keep = np.argsort(margins)[:max_K]
                active_set = set(int(clist[i]) for i in keep)
        
        # Always include the argmax
        active_set.add(int(np.argmax(ac)))
        
        clist = sorted(active_set)
        K = len(clist)
        if K == 0:
            break
        
        try:
            coeffs = build_lp_matrix(y, clist, n)
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            prob.solve(solver=solver, verbose=False)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                raise Exception("LP status: " + str(prob.status))
        except Exception:
            # Reduce active set on failure
            if len(active_set) > init_K // 2:
                clist2 = sorted(active_set)
                margins = np.array([M_curr - ac[k] for k in clist2])
                keep = np.argsort(margins)[:init_K]
                active_set = set(int(clist2[i]) for i in keep)
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check violations
        size_fft = 2 * n - 1
        fft_size = 1 << (size_fft - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size_fft]
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]
            for k in top_v:
                active_set.add(int(k))
        
        # Line search: try multiple alphas
        best_alpha = 2.0 / (t + 2.0)
        best_trial = fast_autocorr(normalize(np.maximum((1 - best_alpha) * y + best_alpha * g_dir, 0.0))).max()
        
        alpha_candidates = [1.0/(t+1), 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for alpha in alpha_candidates:
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            Mt = fast_autocorr(trial / ts).max()
            if Mt < best_trial - 1e-15:
                best_trial = Mt
                best_alpha = alpha
        
        # Bisection refinement
        lo = max(0.0, best_alpha - 0.1)
        hi = min(1.0, best_alpha + 0.1)
        for _ in range(8):
            mid1 = lo + (hi - lo) / 3.0
            mid2 = hi - (hi - lo) / 3.0
            t1 = np.maximum((1 - mid1) * y + mid1 * g_dir, 0.0)
            t2 = np.maximum((1 - mid2) * y + mid2 * g_dir, 0.0)
            s1, s2 = t1.sum(), t2.sum()
            m1 = fast_autocorr(t1 / s1).max() if s1 > 1e-15 else 1e10
            m2 = fast_autocorr(t2 / s2).max() if s2 > 1e-15 else 1e10
            if m1 < m2:
                hi = mid2
            else:
                lo = mid1
        best_alpha = (lo + hi) / 2.0
        
        if t <= 2:
            y = g_dir.copy()
        else:
            y = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t={:5d} ({:.1f}s) J={:.7f} |S|={}".format(t, elapsed, 2 * n * best_m, len(active_set)))
    
    elapsed = time.monotonic() - t0
    if verbose:
        print("    LP-FW done ({:.1f}s) J={:.7f} |S|={}".format(elapsed, 2 * n * best_m, len(active_set)))
    return best_a, 2.0 * n * best_m

def coord_refine(a0, time_limit, n_targets=20, seed=42, verbose=True):
    """Coordinate refinement: move mass from peak contributors to quiet positions."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 5000
    fracs = [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(n_targets, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            ws = max(0, ki)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-25:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-15:
                continue
            
            # Find quiet positions (low pressure from all top lags)
            pr = np.zeros(n, dtype=np.float64)
            for kk in top_k[:10]:
                kkk = int(kk)
                ws2 = max(0, kkk)
                we2 = min(n - 1, kkk)
                if ws2 > we2:
                    continue
                idx2 = np.arange(ws2, we2 + 1)
                pr[idx2] += a[idx2] * a[kkk - idx2]
            pr[src] = 1e10
            quiet_positions = np.argsort(pr)[:50]
            
            for dst in quiet_positions:
                dst = int(dst)
                if dst == src:
                    continue
                found = False
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-18:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-16:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break
            
            if not improved and stall > 200:
                # Random move to escape
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
                            improved = True
                break
        
        if not improved:
            stall += 1
            if stall > 300:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
        
        if verbose and stall % 500 == 0 and stall > 0:
            print("    CoordRef ({:.1f}s) J={:.7f} stall={}".format(time.monotonic() - t0, 2 * n * best_m, stall))
    
    return best_a, 2.0 * n * best_m

def iterated_search(a0, time_limit, solver, n_iter_max=20, verbose=True):
    """Iterated local search: perturb + short LP-FW + coordinate refine."""
    n = len(a0)
    best_a = normalize(a0)
    best_m = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break
        
        # Perturb the best solution
        sigma = max(0.005, 0.05 * (1 - it / n_iter_max))
        noise = sigma * rng.standard_normal(n) * best_a
        perturbed = best_a + noise
        perturbed = np.maximum(perturbed, 0.0)
        ps = perturbed.sum()
        if ps < 1e-15:
            continue
        perturbed = perturbed / ps
        
        # Short LP-FW
        t_lp = min(remaining * 0.4, 60.0)
        a_out, v_out = lp_fw(perturbed, t_lp, solver, init_K=200, max_K=4000, add_per=20, verbose=False)
        sc = eval_safe(a_out)
        
        # Coordinate refine
        t_cr = min(remaining * 0.3, 30.0)
        a_out2, v_out2 = coord_refine(a_out, t_cr, n_targets=15, seed=42+it, verbose=False)
        sc2 = eval_safe(a_out2)
        
        if np.isfinite(sc) and sc < 2*n*best_m:
            best_m = sc / (2*n)
            best_a = a_out
            print("    Iter{} LP improved: J={:.7f}".format(it, sc))
        if np.isfinite(sc2) and sc2 < 2*n*best_m:
            best_m = sc2 / (2*n)
            best_a = a_out2
            print("    Iter{} CR improved: J={:.7f}".format(it, sc2))
    
    return best_a, 2.0 * n * best_m
