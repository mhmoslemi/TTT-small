            sine = np.sin(2.0 * np.pi * 3.0 * pos / n) - np.sin(2.0 * np.pi * 3.0 * pos / n).mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.3 * anti + 0.2 * sine + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            p = max(p * 0.6, float(p_start))

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    NP t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " p=" + str(round(p, 0)))

    return best_a, best_max_b


def pressure_leveling(a0, time_limit, verbose=True, seed=99):
    """
    Global pressure-field local search.
    Computes a 'pressure' at each position from all significant peaks simultaneously,
    then moves mass from high-pressure to low-pressure positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 5000

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-16:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Identify significant peaks (above 90th percentile)
        thresh = np.percentile(c, 90)
        peak_idx = np.where(c > thresh)[0]
        if len(peak_idx) == 0:
            peak_idx = np.argsort(c)[::-1][:30]
        peak_idx = peak_idx[:50]  # Cap at 50 peaks

        # Compute pressure field: pressure[j] = sum over peaks k of w_k * (a[k-j] + valid partner)
        w = c[peak_idx] - thresh  # Weight by excess over threshold
        w = np.maximum(w, 0)
        if w.sum() < 1e-30:
            no_improve += 10
            continue
        w = w / w.sum()

        pressure = np.zeros(n)
        for ii, k in enumerate(peak_idx):
            k_i = int(k)
            s = max(0, k_i - n + 1)
            e = min(n - 1, k_i)
            if s > e:
                continue
            idx = np.arange(s, e + 1)
            partners = k_i - idx
            pressure[idx] += w[ii] * a[partners]
            pressure[partners] += w[ii] * a[idx]

        # Also add self-pressure (positions that appear in many peak convolutions)
        appear_count = np.zeros(n)
        for k in peak_idx:
            k_i = int(k)
            s = max(0, k_i - n + 1)
            e = min(n - 1, k_i)
            if s <= e:
                appear_count[s:e + 1] += 1
        pressure += 0.1 * appear_count * a

        # Find high-pressure source and low-pressure destination
        # Source: position with high pressure AND nonzero value
        src_candidates = np.where(a > 1e-10)[0]
        if len(src_candidates) == 0:
            break
        src_pressure = pressure[src_candidates]
        top_srcs = src_candidates[np.argsort(src_pressure)[::-1][:10]]

        # Destination: low pressure, not in peak regions
        in_peak = np.zeros(n, dtype=bool)
        for k in peak_idx[:20]:
            k_i = int(k)
            s = max(0, k_i - n + 1)
            e = min(n - 1, k_i)
            in_peak[s:e + 1] = True
        dst_candidates = np.where(~in_peak)[0]
        if len(dst_candidates) < 50:
            dst_candidates = np.arange(n)
        dst_pressure = pressure[dst_candidates]
        bottom_dsts = dst_candidates[np.argsort(dst_pressure)[:10]]

        improved = False
        for src in top_srcs[:5]:
            if improved:
                break
            src_i = int(src)
            for dst in bottom_dsts[:5]:
                dst_i = int(dst)
                if dst_i == src_i:
                    continue
                # Try several fractions
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src_i] * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-17:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Random exploration
            n_random = 5
            for _ in range(n_random):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
            if not improved:
                no_improve += 1

        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PressureLevel (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve))

    return best_a, best_m


def systematic_cd(a0, time_limit, verbose=True, seed=200):
    """
    Systematic coordinate descent with line search.
    For each peak, try reducing the top contributing pairs via tiny moves
    to many candidate destinations (not just random).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        # Focus on top peaks
        n_peaks = min(25, len(c))
        peak_idx = np.argsort(c)[::-1][:n_peaks]

        improved = False
        for k in peak_idx:
            if (time.monotonic() - t0) > time_limit:
                break
            if improved:
                continue
            k_i = int(k)
            s = max(0, k_i - n + 1)
            e = min(n - 1, k_i)
            if s > e:
                continue
            idx = np.arange(s, e + 1)
            products = a[idx] * a[k_i - idx]
            if products.max() < 1e-20:
                continue

            # Top 5 contributing pairs
            top_pairs = np.argsort(products)[::-1][:5]
            for pi in top_pairs:
                if improved:
                    break
                src_i = int(idx[pi])
                partner = k_i - src_i
                if partner < 0 or partner >= n:
                    continue
                # Reduce the larger of the pair
                if a[src_i] >= a[partner]:
                    reduce_pos = src_i
                else:
                    reduce_pos = partner
                if a[reduce_pos] < 1e-14:
                    continue

                # Candidate destinations: positions NOT in any top peak's support
                in_any = np.zeros(n, dtype=bool)
                for k2 in peak_idx[:15]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_any[s2:e2 + 1] = True
                safe_positions = np.where(~in_any)[0]
                if len(safe_positions) < 20:
                    safe_positions = np.arange(n)
                # Pick 8 diverse safe positions
                if len(safe_positions) > 8:
                    safe_positions = safe_positions[rng.choice(len(safe_positions), 8, replace=False)]

                for dst in safe_positions:
                    if improved:
                        break
                    dst_i = int(dst)
                    if dst_i == reduce_pos:
                        continue
                    for frac in [0.001, 0.003, 0.005, 0.01]:
                        amt = a[reduce_pos] * frac
                        if amt < 1e-17:
                            continue
                        a_trial = a.copy()
                        a_trial[reduce_pos] -= amt
                        a_trial[dst_i] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break

        if not improved:
            stall += 1
            # Structured nudge every 200 stalls
            if stall % 200 == 0:
                eps = 0.005 / (1.0 + stall / 300.0)
                pos = np.arange(n, dtype=np.float64)
                # Try different structured perturbations
                patterns = [
                    np.sin(2 * np.pi * pos / n),
                    np.cos(2 * np.pi * 2 * pos / n),
                    (pos / n - 0.5),
                    rng.normal(0, 1, n)
                ]
                for pat in patterns:
                    a_try = a + eps * pat * a
                    a_try = np.maximum(a_try, 0.0)
                    s_t = a_try.sum()
                    if s_t > 1e-15:
                        a_try = a_try / s_t
                        M_try = fast_autocorr(a_try).max()
                        if M_try < best_m:
                            a = a_try
                            best_m = M_try
                            best_a = a.copy()
                            stall = 0
                            break

        if verbose and stall % 400 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SysCD (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search: p-norm Nesterov + pressure leveling + systematic CD + multi-length."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent loaded: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        x = np.linspace(0, 1, n_base)
        a_init = (1.0 - x) ** 1.2 + 0.05
        best_seq = norm_sum1(a_init)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(n_base) + ", score=" + repr(best_score))

    n0 = len(best_seq)
    print("Starting: n=" + str(n0) + ", score=" + repr(best_score))

    # Phase 1: Nesterov p-norm (initial refinement)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Nesterov p-norm (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.20, 330.0)
    if t1 > 60:
        a_out, _ = nesterov_pnorm(norm_sum1(best_seq), t1,
                                  p_start=16, p_end=256, p_growth=1.005,
                                  eta_base=0.55, top_k_ratio=0.02,
                                  restart_every=1500, verbose=True, seed=1)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "np1", t_global)

    # Phase 2: Pressure leveling
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Pressure Leveling (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 40:
        a_out, _ = pressure_leveling(norm_sum1(best_seq), t2, verbose=True, seed=11)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pl1", t_global)

    # Phase 3: Nesterov with higher p (sharp focus on max)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Nesterov high-p (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.18, 300.0)
    if t3 > 60:
        a_out, _ = nesterov_pnorm(norm_sum1(best_seq), t3,
                                  p_start=64, p_end=1024, p_growth=1.006,
                                  eta_base=0.4, top_k_ratio=0.01,
                                  restart_every=1200, verbose=True, seed=2)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "np2", t_global)

    # Phase 4: Systematic CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Systematic CD (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 40:
        a_out, _ = systematic_cd(norm_sum1(best_seq), t4, verbose=True, seed=33)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "scd1", t_global)

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-length (base n=" + str(n_best) + ") ===")
    lengths = [n_best + 800, n_best + 2000, max(4000, n_best - 1000)]
    lengths = [l for l in lengths if 2000 <= l <= 12000]
    seen_lengths = set()
    for new_n in lengths:
        if new_n in seen_lengths or abs(new_n - n_best) < 200:
            continue
        seen_lengths.add(new_n)
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("  Trying n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.30, 180.0)
        if t5a > 40:
            a_out, _ = nesterov_pnorm(a_long, t5a,
                                      p_start=16, p_end=256, p_growth=1.004,
                                      eta_base=0.5, top_k_ratio=0.02,
                                      restart_every=1500, verbose=True, seed=3)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ml_np_" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80:
            t5b = min(remaining * 0.35, 120.0)
            if t5b > 30:
                a_try = interpolate_sequence(best_seq, new_n)
                a_out, _ = pressure_leveling(a_try, t5b, verbose=False, seed=44)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "ml_pl_" + str(new_n), t_global)

    # Phase 6: Final Nesterov (very high p, fine tuning)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Final Nesterov (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 40:
        a_out, _ = nesterov_pnorm(norm_sum1(best_seq), t6,
                                  p_start=128, p_end=2048, p_growth=1.008,
                                  eta_base=0.3, top_k_ratio=0.005,
                                  restart_every=800, verbose=True, seed=5)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "np_final", t_global)

    # Phase 7: Systematic CD (final polish)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final SysCD (n=" + str(n_best) + ") ===")
    t7 = min(remaining * 0.20, 250.0)
    if t7 > 30:
        a_out, _ = systematic_cd(norm_sum1(best_seq), t7, verbose=True, seed=55)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "scd_final", t_global)

    # Phase 8: Pressure leveling (final)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final Pressure Leveling (n=" + str(n_best) + ") ===")
    t8 = min(remaining * 0.15, 150.0)
    if t8 > 30:
        a_out, _ = pressure_leveling(norm_sum1(best_seq), t8, verbose=False, seed=66)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pl_final", t_global)

    # Phase 9: Ultra-fine Nesterov burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print("\n=== Phase 9: Ultra-fine Nesterov (n=" + str(n_best) + ") ===")
        t9 = min(remaining * 0.40, 120.0)
        if t9 > 20:
            a_out, _ = nesterov_pnorm(norm_sum1(best_seq), t9,
                                      p_start=256, p_end=4096, p_growth=1.01,
                                      eta_base=0.2, top_k_ratio=0.003,
                                      restart_every=600, verbose=True, seed=7)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "np_ultra", t_global)

    # Phase 10: Final systematic CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print("\n=== Phase 10: Final SysCD (n=" + str(n_best) + ") ===")
        t10 = min(remaining * 0.60, 100.0)
        if t10 > 20:
            a_out, _ = systematic_cd(norm_sum1(best_seq), t10, verbose=True, seed=88)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "scd_ulte", t_global)

    # One final pressure leveling if time allows
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== Phase 11: Last Pressure Leveling ===")
        t11 = min(remaining * 0.70, 80.0)
        if t11 > 15:
            a_out, _ = pressure_leveling(norm_sum1(best_seq), t11, verbose=False, seed=99)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "pl_ulte", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence! Using fallback.")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1((1.0 - x) ** 1.5 + 0.1)
        best_score = evaluate_safe(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)

    return [float(x) for x in best_seq]
