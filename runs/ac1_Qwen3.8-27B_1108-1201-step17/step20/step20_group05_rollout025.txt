    arr = np.asarray(a, dtype=np.float64).copy()

def solve_lp_direction(y, active_idx=None):
    """Solve the LP subproblem to find the optimal direction z.
    
    minimize  t
    subject to (y*z)[k] <= t for all k (or just active k)
               z >= 0
               sum(z) = 1
    """
    n = y.size
    size = 2 * n - 1
    
    if active_idx is None:
        active_idx = np.arange(size)
    
    K = active_idx.size
    
    # Build constraint matrix for active constraints only
    # For constraint k: sum_j y[k-j] * z[j] - t <= 0
    rows = []
    cols = []
    vals = []
    
    for r, k in enumerate(active_idx):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start < j_end:
            js = np.arange(j_start, j_end)
            rows.append(np.full(js.size, r, dtype=np.int32))
            cols.append(js.astype(np.int32))
            vals.append(y[k - js])
        # Add -1 in column n (for t)
        rows.append(np.array([r], dtype=np.int32))
        cols.append(np.array([n], dtype=np.int32))
        vals.append(np.array([-1.0]))
    
    if not rows:
        return None
    
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    vals = np.concatenate(vals)
    
    A_ub = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n + 1))
    b_ub = np.zeros(K)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)

        z = np.maximum(z, 0)
        s = z.sum()
        if s > 0:
            return z / s

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def fft_conv(a, b=None):
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if b.size != n:
        b = np.resize(b, n)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c = np.where(np.isfinite(c), c, 0.0)
    return c

def solve_lp_direction_fast(y, max_k=None, time_limit=20.0):
    if not SCIPY_OK:
        return None
    
    n = y.size
    size = 2 * n - 1
    
    if max_k is not None and max_k < size:
        c = fft_conv(y)
        active = np.argsort(c)[::-1][:max_k]
    else:
        active = np.arange(size)
    
    K = active.size
    if K == 0:
        return None
    
    k_arr = active.astype(np.int64)
    j_starts = np.maximum(0, k_arr - n + 1).astype(np.int64)
    j_ends = np.minimum(n, k_arr + 1).astype(np.int64)
    lengths = j_ends - j_starts  # number of non-zeros in each row
    
    total_nnz_z = int(lengths.sum())
    total_nnz = total_nnz_z + K
    
    # Compute column indices vectorized
    offsets = np.zeros(K + 1, dtype=np.int64)
    np.cumsum(lengths, out=offsets[1:])
    
    pos = np.arange(total_nnz_z, dtype=np.int64)
    row_of_pos = np.searchsorted(offsets, pos, side='right') - 1
    offsets_within = pos - offsets[row_of_pos]
    all_j = (j_starts[row_of_pos] + offsets_within).astype(np.int32)
    all_k = k_arr[row_of_pos]
    all_vals = y[all_k - all_j].astype(np.float64)
    r_indices = row_of_pos.astype(np.int32)
    
    # Build full arrays including t column
    rows = np.empty(total_nnz, dtype=np.int32)
    cols = np.empty(total_nnz, dtype=np.int32)
    vals = np.empty(total_nnz, dtype=np.float64)
    
    rows[:total_nnz_z] = r_indices
    cols[:total_nnz_z] = all_j
    vals[:total_nnz_z] = all_vals
    
    rows[total_nnz_z:] = np.arange(K, dtype=np.int32)
    cols[total_nnz_z:] = n

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1, dtype=np.float64)
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0
    
    try:
        res = scipy.optimize.linprog(
