def simplex_project(z):
    """Project a vector onto the probability simplex (non-negative, sum to 1).
    Uses the method by Dunn and Kass (1996) and is the standard method."""
    n = len(z)
    if n == 0:
        return z
    # Remove negative values
    z = np.maximum(z, 1e-10)
    # Calculate cumulative sums
    s = np.cumsum(z)
    # Find the active subsystem
    t = np.arange(1, n+1)
    # Construct the dual variable
    # This is the core projection operation
    lambda_val = np.true_divide(np.minimum(s / t - 1, 0), 1, where=True, 
        otherwise= np.divide(np.minimum(s, 1) / t, s[-1] / n, 
        out=np.full_like(s/ t, fill_value=0))) 
    # The optimization step is the typical projected gradient method
    return np.maximum(z - lambda_val, 0)
