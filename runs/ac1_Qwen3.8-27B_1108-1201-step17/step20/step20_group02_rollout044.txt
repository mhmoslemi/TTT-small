def _convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

def _interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _smoothed_grad_descent(a0, time_limit, t_initial=50.0, t_final=500.0,
                           eta_scale=0.5, verbose=True):
    """Smoothed gradient descent with log-sum-exp objective and multiplicative updates."""
    n = len(a0)
    y = _normalize(a0)
    best_a = y.copy()
    best_val = _eval(y)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    temperature = t_initial
    total_iters = 0
    no_improve = 0
    max_no_improve = 200
    stall_restarts = 0

    while time.monotonic() - t0 < time_limit:
        # Anneal temperature
        elapsed_frac = min((time.monotonic() - t0) / time_limit, 1.0)
        temperature = t_initial * (t_final / t_initial) ** elapsed_frac

        # Compute gradient
        conv = _convolve(y)
        max_c = conv.max()
        exp_c = np.exp(temperature * (conv - max_c))
        Z = exp_c.sum()

        if Z < 1e-300:
            temperature *= 0.5
            continue

        cc = _cross_correlate(exp_c, y)
        grad = (2.0 / (temperature * Z)) * cc[:n]

        # Add small regularization to avoid zeros
        grad = grad + 1e-10 * y

        # Adaptive step size based on gradient magnitude
        g_norm = np.sqrt(np.sum(grad * y))
        eta = eta_scale / (temperature * g_norm + 1e-10)
        eta = min(eta, 1.0 / temperature)

        # Multiplicative (mirror descent) update
        log_update = -eta * grad
        # Clip to avoid overflow
        log_update = np.clip(log_update, -5.0, 5.0)
        y_new = y * np.exp(log_update)
        y_new = np.maximum(y_new, 1e-15)
        s_new = y_new.sum()
        y_new = y_new / s_new

        # Evaluate
        val_new = _eval(y_new)

        if val_new < best_val - 1e-12:
            best_val = val_new
            best_a = y_new.copy()
            no_improve = 0
            y = y_new
        else:
            no_improve += 1
            # Try smaller step
            for shrink in [0.5, 0.25, 0.1]:
                eta_s = eta * shrink
                log_update_s = np.clip(-eta_s * grad, -5.0, 5.0)
                y_s = y * np.exp(log_update_s)
                y_s = np.maximum(y_s, 1e-15)
                y_s = y_s / y_s.sum()
                val_s = _eval(y_s)
                if val_s < best_val - 1e-12:
                    best_val = val_s
                    best_a = y_s.copy()
                    y = y_s
                    no_improve = 0
                    break

        total_iters += 1

        if no_improve > max_no_improve:
            stall_restarts += 1
            if stall_restarts > 15:
                if verbose:
                    print("    Smoothed GD: too many stalls, stopping")
                break
            # Restart with perturbation
            noise = rng.normal(0, 0.02, n) * best_a
            y_p = best_a + noise
            y = _normalize(y_p)
            no_improve = 0
            # Also try reducing temperature
            temperature = max(t_initial, temperature * 0.7)

        if verbose and total_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    Smoothed GD iter=" + str(total_iters) +
                  " (t=" + str(round(elapsed, 1)) + "s, temp=" + str(round(temperature, 1)) +
                  "): J=" + repr(best_val))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    Smoothed GD done (" + str(round(elapsed, 1)) + "s): J=" + repr(best_val))
    return best_a, best_val


def _lp_fw(a0, time_limit, solver, initial_K=400, max_K=3000, verbose=True):
    """LP-Frank-Wolfe with adaptive active set."""
    n = len(a0)
    y = _normalize(a0)
    best_a = y.copy()
    best_m = _convolve(y).max()
    t0 = time.monotonic()
    nfft = 1 << (2 * n - 2).bit_length()

    conv_y = _convolve(y)
    active_set = sorted(np.argsort(conv_y)[::-1][:initial_K].tolist())
    no_improve = 0
    iter_count = 0
    restarts = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 600:
            restarts += 1
            if restarts > 5:
                break
            # Perturb active set
            conv_curr = _convolve(y)
            keep_frac = 0.6
            keep = active_set[:int(len(active_set) * keep_frac)]
            new_k = np.argsort(conv_curr)[::-1][:initial_K]
            active_set = sorted(set(keep) | set(new_k.tolist()))[:max_K]
            no_improve = 0
            continue

        iter_count += 1
        conv = _convolve(y)
        M = conv.max()

        if M < best_m - 1e-14:
            best_m = M
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Trim active set
        if len(active_set) > max_K:
            margins = np.array([M - conv[k] for k in active_set])
            keep_idx = np.argsort(margins)[:int(max_K * 0.7)]
            active_set = sorted([active_set[i] for i in keep_idx])
            # Keep current violations
            viol = np.where(conv > M - 1e-8)[0]
            for v in viol[:50]:
                iv = int(v)
                if iv not in active_set:
                    active_set.append(iv)
            active_set = sorted(active_set)[:max_K]

        K = len(active_set)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj in range(K):
            k_int = int(active_set[jj])
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(100, K // 2)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations, add to active set
        Yf = np.fft.rfft(y, nfft)
        Gf = np.fft.rfft(g_dir, nfft)
        conv_ys = np.fft.irfft(Yf * Gf, nfft)[:2 * n - 1]
        viol_idx = np.where(conv_ys > M + 1e-10)[0]
        if len(viol_idx) > 0:
            viol_amt = conv_ys[viol_idx] - M
            top_v = viol_idx[np.argsort(viol_amt)[::-1][:50]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        # Line search
        best_alpha = 1.0 / (iter_count + 2.0)
        best_trial = M
        alphas = [1.0, 0.7, 0.5, 0.3, 0.15, 0.05,
                  max(0.02, 2.0 / (iter_count + 2.0))]
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            Mt = _convolve(trial).max()
            if Mt < best_trial:
                best_trial = Mt
                best_alpha = alpha

        if iter_count <= 2:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = best_a.copy()
            else:
                y = y / sy

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) +
                  " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) +
                  ", |S|=" + str(len(active_set)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))
    return best_a, best_m


def _peak_shave(a0, time_limit, verbose=False):
    """Direct peak-shaving: iteratively move mass away from peak-contributing elements."""
    n = len(a0)
    y = _normalize(a0)
    best_a = y.copy()
    best_m = _convolve(y).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv = _convolve(y)
        # Find top peaks
        n_peaks = min(20, len(conv))
        top_k = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for k in top_k[:10]:
            ki = int(k)
            # Contributions to this peak
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = y[idx] * y[ki - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-20:
                continue

            # Find the largest contributor
            src_i = int(idx[np.argmax(contrib)])
            partner_i = ki - src_i

            if y[src_i] < 1e-12:
                continue

            # Score each position by how much it would affect the peaks
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += y[k2i - idx2]

            # Move to position with lowest score
            scores[src_i] = np.inf
            dst = int(np.argmin(scores))

            if dst == src_i:
                continue

            # Try different amounts
            for frac in [0.05, 0.03, 0.01, 0.005]:
                amt = y[src_i] * frac
                if amt < 1e-15:
                    continue
                y_trial = y.copy()
                y_trial[src_i] -= amt
                y_trial[dst] += amt
                y_trial = np.maximum(y_trial, 0.0)
                st = y_trial.sum()
                if st < 1e-15:
                    continue
                y_trial = y_trial / st
                M_trial = _convolve(y_trial).max()
                if M_trial < best_m - 1e-14:
                    y = y_trial
                    best_m = M_trial
                    best_a = y.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 400:
                # Random perturbation
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and y[i] > 1e-10:
                        frac = rng.uniform(0.01, 0.05)
                        amt = y[i] * frac
                        y[i] -= amt
                        y[j] += amt
                        y = _normalize(y)
                        M_new = _convolve(y).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = y.copy()
                            stall = 0
                if stall > 600:
                    break
            if stall % 100 == 0:
                # More aggressive random moves
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and y[i] > 1e-10:
                        frac = rng.uniform(0.05, 0.15)
                        amt = y[i] * frac
                        y_trial = y.copy()
                        y_trial[i] -= amt
                        y_trial[j] += amt
                        y_trial = _normalize(y_trial)
                        M_new = _convolve(y_trial).max()
                        if M_new < best_m:
                            best_m = M_new
                            y = y_trial
                            best_a = y.copy()
                            stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    PeakShave (" + str(round(elapsed, 1)) + "s): J=" +
                  repr(2.0 * n * best_m) + ", stall=" + str(stall))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    PeakShave done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))
    return best_a, best_m


def _gradient_flow(a0, time_limit, verbose=True):
    """Multi-peak gradient flow with adaptive step size."""
    n = len(a0)
    y = _normalize(a0)
    best_a = y.copy()
    best_val = _eval(y)
    t0 = time.monotonic()
    rng = np.random.default_rng(7)

    # Track multiple peaks
    n_track = 30
    stall = 0
    max_stall = 500
    eta = 0.3
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        if stall > max_stall:
            # Restart
            noise = rng.normal(0, 0.03, n) * best_a
            y_p = _normalize(best_a + noise)
            y = y_p
            eta *= 1.3
            stall = 0
            if eta > 2.0:
                break

        conv = _convolve(y)
        M = conv.max()

        # Find active peaks (within 5% of max)
        threshold = M * 0.95
        active_peaks = np.where(conv >= threshold)[0]
        if len(active_peaks) == 0:
            active_peaks = np.array([int(np.argmax(conv))])

        # Compute combined subgradient
        grad = np.zeros(n)
        for k in active_peaks:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += y[ki - idx]

        # Normalize gradient
        g_norm = np.sqrt(np.sum(grad * y) + 1e-30)
        if g_norm < 1e-15:
            break

        # Use smoothed max weights
        scaled = (conv - M) / max(1e-10, M * 0.01)
        weights = np.exp(scaled * 10)
        weights /= weights.sum()

        weighted_grad = np.zeros(n)
        for k in range(len(conv)):
            ki = int(k)
            if weights[k] < 1e-10:
                continue
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                weighted_grad[idx] += weights[k] * y[ki - idx]

        # Mirror descent step
        eff_eta = eta / (1.0 + iter_count * 0.001)
        log_update = -eff_eta * weighted_grad
        log_update = np.clip(log_update, -3.0, 3.0)
        y_new = y * np.exp(log_update)
        y_new = np.maximum(y_new, 1e-15)
        y_new = y_new / y_new.sum()

        val_new = _eval(y_new)
        iter_count += 1

        if val_new < best_val - 1e-12:
            best_val = val_new
            best_a = y_new.copy()
            y = y_new
            stall = 0
            eta = min(eta * 1.02, 2.0)
        else:
            stall += 1
            # Try smaller step
            for shrink in [0.5, 0.2]:
                lu_s = np.clip(-eff_eta * shrink * weighted_grad, -3.0, 3.0)
                y_s = y * np.exp(lu_s)
                y_s = np.maximum(y_s, 1e-15)
                y_s = y_s / y_s.sum()
                val_s = _eval(y_s)
                if val_s < best_val - 1e-12:
                    best_val = val_s
                    best_a = y_s.copy()
                    y = y_s
                    stall = 0
                    break
            eta *= 0.95

        if verbose and iter_count % 300 == 0:
            elapsed = time.monotonic() - t0
            print("    GradFlow iter=" + str(iter_count) +
                  " (" + str(round(elapsed, 1)) + "s): J=" + repr(best_val) +
                  ", n_peaks=" + str(len(active_peaks)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    GradFlow done (" + str(round(elapsed, 1)) + "s): J=" + repr(best_val))
    return best_a, best_val


def _random_walk(a0, time_limit, n_steps=5000, verbose=False):
    """Random walk with acceptance criterion for exploring the landscape."""
    n = len(a0)
    y = _normalize(a0)
    best_a = y.copy()
    best_val = _eval(y)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Random move: transfer mass between two positions
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        frac = rng.uniform(0.001, 0.05)
        amt = y[i] * frac
        if amt < 1e-15:
            continue

        y_trial = y.copy()
        y_trial[i] -= amt
        y_trial[j] += amt
        y_trial = np.maximum(y_trial, 0.0)
        st = y_trial.sum()
        if st < 1e-15:
            continue
        y_trial = y_trial / st

        val_trial = _eval(y_trial)

        # Accept if better or with small probability (simulated annealing)
        if val_trial < best_val:
            best_val = val_trial
            best_a = y_trial.copy()
            y = y_trial
        elif rng.random() < 0.01:
            # Occasional exploration step
            y = y_trial

    if verbose:
        elapsed = time.monotonic() - t0
        print("    RandWalk done (" + str(round(elapsed, 1)) + "s): J=" + repr(best_val))
    return best_a, best_val


def _fourier_init(n, mode=0):
    """Generate initial sequences using different patterns."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        return _normalize(np.sin(np.pi * x) ** 2)
    elif mode == 1:
        return _normalize(np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2)))
    elif mode == 2:
        return _normalize(np.sin(np.pi * x) * (1 + 0.3 * np.cos(2 * np.pi * x)))
    elif mode == 3:
        return _normalize(np.ones(n) * 0.5 + 0.5 * np.sin(4 * np.pi * x))
    else:
        return _normalize(np.ones(n))


def _try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = _eval_safe_list(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) +
              " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main search function. Uses a combination of:
    1. Smoothed gradient descent (log-sum-exp)
    2. LP-Frank-Wolfe
    3. Multi-peak gradient flow
    4. Peak shaving
    5. Multi-n exploration
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = _get_solver()

    # Get initial sequence
    best_seq = None
    best_score = float('inf')
    n0 = 5000

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = _eval_safe_list(p_norm)
            if np.isfinite(sc):
                best_seq = p_norm
                best_score = sc
                n0 = len(p_norm)
                print("Parent: n=" + str(n0) + ", score=" + repr(sc))
    except Exception:
        pass

    if best_seq is None:
        best_seq = _fourier_init(n0, mode=0)
        best_score = _eval_safe_list(best_seq)
        print("Default init: n=" + str(n0) + ", score=" + repr(best_score))

    def remaining():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: Smoothed gradient descent ===
    print("\n=== Phase 1: Smoothed GD (n=" + str(len(best_seq)) + ") ===")
    t1 = min(remaining() * 0.15, 240.0)
    if t1 > 30:
        a_out, _ = _smoothed_grad_descent(_normalize(best_seq), t1,
                                          t_initial=30.0, t_final=300.0,
                                          eta_scale=0.4, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "smoothgd1", t_global)

    # === Phase 2: LP-FW ===
    print("\n=== Phase 2: LP-FW (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining() * 0.20, 330.0)
    if t2 > 40 and len(best_seq) <= 9000:
        a_out, _ = _lp_fw(_normalize(best_seq), t2, solver,
                          initial_K=500, max_K=3500, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "lpfw1", t_global)

    # === Phase 3: Gradient flow ===
    print("\n=== Phase 3: Gradient Flow (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining() * 0.12, 200.0)
    if t3 > 30:
        a_out, _ = _gradient_flow(_normalize(best_seq), t3, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "gradflow1", t_global)

    # === Phase 4: Peak shaving ===
    print("\n=== Phase 4: Peak Shave (n=" + str(len(best_seq)) + ") ===")
    t4 = min(remaining() * 0.10, 160.0)
    if t4 > 20:
        a_out, _ = _peak_shave(_normalize(best_seq), t4, verbose=False)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "peakshave1", t_global)

    # === Phase 5: Multi-n exploration ===
    print("\n=== Phase 5: Multi-n exploration ===")
    n_best = len(best_seq)
    candidate_ns = set()
    for delta in [-1000, -500, -200, 200, 500, 1000, 1500, 2000]:
        nn = n_best + delta
        if 2000 <= nn <= 12000:
            candidate_ns.add(nn)
    candidate_ns = sorted(candidate_ns, key=lambda x: abs(x - n_best))

    for new_n in candidate_ns[:4]:
        rt = remaining()
        if rt < 100:
            break
        print("\n  Multi-n: n=" + str(new_n) + " (remaining=" + str(round(rt, 1)) + "s)")
        a_interp = _interpolate(best_seq, new_n)

        # Smoothed GD
        t5 = min(rt * 0.20, 150.0)
        if t5 > 25:
            a_out, _ = _smoothed_grad_descent(a_interp, t5,
                                              t_initial=50.0, t_final=400.0,
                                              eta_scale=0.3, verbose=False)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "smoothgd_n" + str(new_n), t_global)

        # LP-FW
        rt = remaining()
        if rt > 80 and new_n <= 9000:
            t5b = min(rt * 0.25, 180.0)
            if t5b > 30:
                a_out, _ = _lp_fw(_normalize(best_seq), t5b, solver,
                                  initial_K=400, max_K=3000, verbose=False)
                best_seq, best_score = _try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

    # === Phase 6: LP-FW second pass on best ===
    print("\n=== Phase 6: LP-FW pass 2 (n=" + str(len(best_seq)) + ") ===")
    t6 = min(remaining() * 0.18, 280.0)
    if t6 > 40 and len(best_seq) <= 9000:
        a_out, _ = _lp_fw(_normalize(best_seq), t6, solver,
                          initial_K=600, max_K=4000, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "lpfw2", t_global)

    # === Phase 7: Smoothed GD with different schedule ===
    print("\n=== Phase 7: Smoothed GD v2 (n=" + str(len(best_seq)) + ") ===")
    t7 = min(remaining() * 0.10, 150.0)
    if t7 > 25:
        a_out, _ = _smoothed_grad_descent(_normalize(best_seq), t7,
                                          t_initial=100.0, t_final=800.0,
                                          eta_scale=0.2, verbose=False)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "smoothgd2", t_global)

    # === Phase 8: Peak shaving final ===
    print("\n=== Phase 8: Peak Shave final (n=" + str(len(best_seq)) + ") ===")
    t8 = min(remaining() * 0.08, 100.0)
    if t8 > 20:
        a_out, _ = _peak_shave(_normalize(best_seq), t8, verbose=False)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "peakshave2", t_global)

    # === Phase 9: Final LP-FW ===
    print("\n=== Phase 9: Final LP-FW (n=" + str(len(best_seq)) + ") ===")
    t9 = min(remaining() * 0.15, 200.0)
    if t9 > 30 and len(best_seq) <= 9000:
        a_out, _ = _lp_fw(_normalize(best_seq), t9, solver,
                          initial_K=800, max_K=5000, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # === Final: Short peak shave ===
    rt = remaining()
    if rt > 15:
        a_out, _ = _peak_shave(_normalize(best_seq), min(rt * 0.5, 60.0), verbose=False)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "peakshave_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = _eval_safe_list(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) +
              ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        best_seq = _fourier_init(5000, mode=0)
        best_score = _eval_safe_list(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
