import numpy as np
import time

def fast_conv(a):
    """Compute linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_gradient(a, beta):
    """Compute gradient of log-sum-exp(beta * conv(a,a)) w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    # Softmax
    shifted = beta * (b - b.max())
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(b)
        p[np.argmax(b)] = 1.0
    else:
        p /= p_sum
    # Cross-correlation: g[j] = sum_k p[k] * a[k-j]
    # Use FFT: g = ifft(FFT(p, N) * conj(FFT(a, N)))[:n]
    N = 1 << (3 * n - 2 - 1).bit_length()
    p_pad = np.zeros(N)
    p_pad[:len(p)] = p
    a_pad = np.zeros(N)
    a_pad[:n] = a
    P = np.fft.rfft(p_pad)
    A = np.fft.rfft(a_pad)
    g = np.fft.irfft(P * np.conj(A), N)[:n]
    grad = 2.0 * g
    max_b = b.max()
    return grad, max_b, p

def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def objective(a):
    """Compute J = 2*n*max(b)/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return 2.0 * n * b.max() / (s * s)

def norm_sum1(a):

def optimize_gradient(a0, time_limit, beta_start=10.0, beta_end=500.0,
                      patience=500, verbose=True):
    """Main optimization: projected gradient with adaptive beta and BB step size."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_J = objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    # BB step size variables
    a_prev = a.copy()
    grad_prev = None
    eta = 0.01  # initial step size
    
    stall = 0
    beta = beta_start
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient
        grad, max_b, p = compute_gradient(a, beta)
        
        # Check improvement
        J_curr = 2.0 * n * max_b  # since sum(a) = 1
        if J_curr < best_J - 1e-12:
            best_J = J_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Barzilai-Borwein step size
        if grad_prev is not None:
            s_bb = a - a_prev
            y_bb = grad - grad_prev
            ss = np.dot(s_bb, s_bb)
            ys = np.dot(s_bb, y_bb)
            yy = np.dot(y_bb, y_bb)
            if yy > 1e-20 and ss > 1e-20:
                if np.random.random() < 0.5:
                    eta_new = ss / ys if abs(ys) > 1e-15 else eta
                else:
                    eta_new = ys / yy if yy > 1e-15 else eta
                eta = np.clip(eta_new, 1e-6, 0.5)
        
        a_prev = a.copy()
        grad_prev = grad.copy()
        
        # Projected gradient step
        a_new = simplex_project(a - eta * grad)
        
        # Check if the step was good
        J_new = 2.0 * n * fast_conv(a_new).max()
        if J_new > J_curr * 1.01:
            # Step was bad, reduce step size
            eta *= 0.5
            a_new = simplex_project(a - eta * grad)
        
        a = a_new
        
        # Adaptive beta
        if it % 100 == 0:
            progress = min(1.0, it / 5000.0)
            beta = beta_start * (beta_end / beta_start) ** progress
        
        # Restart
        if stall > patience:
            stall = 0
            noise_scale = 0.01 + 0.02 * (stall / patience)
            # Actually use a temperature that decreases

            pert = rng.normal(0, temp, n)
            a_r = best_a + pert
            a_r = np.maximum(a_r, 0.0)
            s_r = a_r.sum()
            if s_r > 1e-15:
                a_r /= s_r
            J_r = 2.0 * n * fast_conv(a_r).max()
            if J_r < best_J:
                best_J = J_r
                best_a = a_r.copy()
                a = a_r
            else:
                # Simulated annealing: accept with probability exp(-delta/temp)
                delta = J_r - best_J
                if delta < 0 or np.random.random() < np.exp(-delta / max(temp * 0.1, 1e-6)):
                    a = a_r
                else:
                    a = best_a.copy()
            eta = 0.01
            grad_prev = None
            beta = max(beta * 0.5, beta_start)
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradOpt it={it} ({elapsed:.1f}s): J={best_J:.6f} beta={beta:.1f} eta={eta:.6f} stall={stall}")
    
    return best_a, best_J
