"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FISTA with adaptive log-sum-exp smoothing and Nesterov momentum
2. Peak-directed local search (targeted mass reallocation)
3. Multi-scale interpolation (try larger n for better lower bounds)
4. Cyclic shift enumeration
5. Multiple restart strategies with diverse initializations
"""

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, ensure non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def smoothed_gradient(y, beta, top_k=None):
    """
    Compute smoothed gradient of max(b) where b = convolve(y,y).
    Uses log-sum-exp smoothing over top-k autocorrelation values.
    Returns: (gradient, b, max_b)
    """
    n = len(y)
    b = fast_autocorr(y)
    
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    
    # Log-sum-exp smoothing
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b_use)
        mask = (b_use == b_max)
        p[mask] = 1.0 / max(1, mask.sum())
    else:
        p = exp_vals / p_total
    
    # Gradient: grad[i] = sum_j a[j] * p[i+j] (computed via FFT)
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    
    return grad, p, b


def fista_optimize(a0, time_limit, beta_start=10.0, beta_end=800.0,
                   beta_growth=1.004, top_k=25, restart_every=3000,
                   step_decay=0.9999, verbose=True):
    """
    FISTA (accelerated projected gradient) with adaptive smoothing.
    Minimizes max(convolve(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(42)
    step = 2.0 / (beta + 1.0)
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov acceleration
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute smoothed gradient
        grad, p, b = smoothed_gradient(y, beta, top_k)
        
        # Gradient step + projection
        step = min(step, 0.5 / (beta + 0.1))
        z = y - step * grad
        x_new = simplex_project(z)
        
        # Evaluate
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Decay step size slowly
        step *= step_decay
        step = max(step, 1e-8)
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        # Increase beta (decrease smoothing) periodically
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)
            step = min(step * beta_growth, 2.0 / (beta + 1.0))
        
        # Restart if stalled
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            step = 2.0 / (beta * 0.5 + 1.0)
            beta = max(beta * 0.7, 5.0)
            # Perturb from best
            noise_scale = 0.01 / (1.0 + 0.02 * (t // restart_every))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    FISTA restart t={:.0f} ({:.1f}s): J={:.6f}".format(t, elapsed, J))
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    FISTA t={:.0f} ({:.1f}s): J={:.6f} beta={:.1f}".format(t, elapsed, J, beta))
    
    return best_a, best_max_b


def peak_directed_search(a0, time_limit, verbose=True):
    """
    Local search that directly targets the highest autocorrelation peaks.
    Moves mass from positions contributing to peaks to cold positions.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        
        # Find top peaks
        num_peaks = min(20, len(ac))
        peak_pos = np.argsort(ac)[::-1][:num_peaks]
        
        # Identify "hot" positions (contributing to peaks)
        hot = np.zeros(n, dtype=bool)
        for k in peak_pos[:10]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            if s <= e:
                # Find which positions contribute most to this peak
                idx = np.arange(s, e + 1)
                contrib = a[idx] * a[ki - idx]
                top_contrib = idx[np.argsort(contrib)[::-1][:max(1, n // 20)]]
                hot[top_contrib] = True
        
        # Find "cold" positions (not contributing to any peak)
        cold = np.where(~hot)[0]
        if len(cold) < 10:
            cold = np.arange(n)
        
        improved = False
        
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick a random top peak
            k_int = int(peak_pos[rng.integers(min(5, len(peak_pos)))])
            s = max(0, k_int - n + 1)
            e = min(n - 1, k_int)
            if s > e:
                continue
            
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            # Source: position with highest contribution to this peak
            src = int(idx[np.argmax(contrib)])
            if a[src] < 1e-12:
                continue
            
            # Destination: random cold position
            dst = int(cold[rng.integers(len(cold))])
            if dst == src:
                continue
            
            # Try different transfer fractions
            best_frac = None
            best_m_trial = ac.max()
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m_trial:
                    best_m_trial = m_trial
                    best_frac = frac
            
            if best_frac is not None and best_m_trial < best_m - 1e-14:
                amt = a[src] * best_frac
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = fast_autocorr(a).max()
                best_a = a.copy()
                improved = True
                stall = 0
                # Recompute peaks
                ac = fast_autocorr(a)
                peak_pos = np.argsort(ac)[::-1][:num_peaks]
                hot = np.zeros(n, dtype=bool)
                for k in peak_pos[:10]:
                    ki = int(k)
                    s2 = max(0, ki - n + 1)
                    e2 = min(n - 1, ki)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        contrib2 = a[idx2] * a[ki - idx2]
                        top_c2 = idx2[np.argsort(contrib2)[::-1][:max(1, n // 20)]]
                        hot[top_c2] = True
                cold = np.where(~hot)[0]
                if len(cold) < 10:
                    cold = np.arange(n)
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.001, 0.02))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        m_new = fast_autocorr(a).max()
                        if m_new < best_m:
                            best_m = m_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSearch ({:.1f}s): J={:.6f} stall={}".format(elapsed, J, stall))
    
    return best_a, best_m


def cyclic_shift_best(a0):
    """Try all cyclic shifts and return the best one."""
    n = len(a0)
    b = fast_autocorr(a0)
    best_m = b.max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a0, fft_size)
    
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b_s = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b_s.max()
        if m < best_m:
            best_m = m
            best_shift = s
    
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_to(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = norm_sum1(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_flatten(a, n_iters=50, verbose=False):
    """
    Iteratively flatten the power spectrum to reduce autocorrelation peaks.
    Works by reducing spectral components that contribute to high autocorrelation.
    """
    n = len(a)
    a_work = norm_sum1(a).copy()
    best_a = a_work.copy()
    best_m = fast_autocorr(a_work).max()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        A = np.fft.rfft(a_work, fft_size)
        power = np.abs(A) ** 2
        b = np.fft.irfft(power, fft_size)[:2 * n - 1]
        b_max = b.max()
        
        if b_max < 1e-15:
            break
        
        # Compute "excess" at each autocorrelation bin
        b_avg = b.mean()
        excess = np.maximum(b - b_avg, 0)
        
        # Transform excess to frequency domain to modulate spectrum
        E = np.fft.rfft(excess, fft_size)
        
        # Dampen frequencies that contribute to excess
        phase_E = np.angle(E)
        mag_E = np.abs(E)
        mag_E_norm = mag_E / (mag_E.max() + 1e-30)
        
        # Reduce components proportionally to their contribution to excess
        A_new = A * (1.0 - 0.1 * mag_E_norm)
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        
        a_work = a_new
    
    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update the best sequence if 'a' is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {:.6f} (n={}, t={:.1f}s)".format(label, sc, len(a), elapsed))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main optimization routine. Searches for the best non-negative sequence
    minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print("Time budget: {:.1f}s".format(time_budget))
    
    best_seq = None
    best_score = float('inf')
    
    # --- Get starting point ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n={}, score={:.6f}".format(len(parent), p_score))
    except Exception:
        parent = None
    
    if best_seq is None:
        # Generate initial sinusoidal sequence
        n_base = 8000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n={}, score={:.6f}".format(len(best_seq), best_score))
    
    n0 = len(best_seq)
    
    # --- Phase 1: FISTA at original n ---
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: FISTA (n={}) ===".format(n0))
    t1 = min(remaining * 0.30, 400.0)
    if t1 > 60:
        a_out, _ = fista_optimize(best_seq, t1,
                                   beta_start=10.0, beta_end=500.0,
                                   beta_growth=1.004, top_k=20,
                                   restart_every=2500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista1", t_global)
    
    # --- Phase 2: Cyclic shift ---
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: Cyclic shift ===")
    if remaining > 30:
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic1", t_global)
    
    # --- Phase 3: Peak-directed local search ---
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: Peak-directed search (n={}) ===".format(len(best_seq)))
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_out, _ = peak_directed_search(best_seq, t3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak1", t_global)
    
    # --- Phase 4: Interpolate to larger n and optimize ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    
    # Try 1.5x and 2x the current length
    target_lengths = []
    for mult in [1.5, 2.0]:
        new_n = int(n_best * mult)
        if new_n > n_best and new_n <= 20000:
            target_lengths.append(new_n)
    
    # Also try fixed large sizes
    for nn in [12000, 15000]:
        if nn > n_best and nn not in target_lengths and nn <= 20000:
            target_lengths.append(nn)
    
    target_lengths = sorted(set(target_lengths))
    
    for new_n in target_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        
        print("\n=== Phase 4: Interpolate to n={} ===".format(new_n))
        a_long = interpolate_to(best_seq, new_n)
        
        # FISTA on the longer sequence
        t4a = min(remaining * 0.35, 300.0)
        if t4a > 50:
            a_out, _ = fista_optimize(a_long, t4a,
                                       beta_start=10.0, beta_end=400.0,
                                       beta_growth=1.005, top_k=15,
                                       restart_every=2000, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_n{}".format(new_n), t_global)
        
        # Cyclic shift
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
            best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_n{}".format(new_n), t_global)
        
        # Peak search on longer sequence
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80:
            t4b = min(remaining * 0.2, 100.0)
            if t4b > 30:
                a_out, _ = peak_directed_search(best_seq, t4b, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_n{}".format(new_n), t_global)
    
    # --- Phase 5: Second FISTA pass with different parameters ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: FISTA v2 (n={}, different params) ===".format(n_best))
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 40:
        a_out, _ = fista_optimize(best_seq, t5,
                                   beta_start=50.0, beta_end=1500.0,
                                   beta_growth=1.008, top_k=10,
                                   restart_every=2000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista2", t_global)
    
    # --- Phase 6: Spectral flattening ---
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 6: Spectral flattening ===")
    if remaining > 30:
        a_out, _ = spectral_flatten(best_seq, n_iters=80, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spectral", t_global)
    
    # --- Phase 7: Final peak search ---
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 7: Final peak search ===")
    if remaining > 30:
        t7 = min(remaining * 0.35, 150.0)
        if t7 > 20:
            a_out, _ = peak_directed_search(best_seq, t7, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_final", t_global)
    
    # --- Phase 8: Final FISTA micro-polish ---
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 8: FISTA micro-polish ===")
    if remaining > 20:
        t8 = min(remaining * 0.50, 120.0)
        if t8 > 15:
            a_out, _ = fista_optimize(best_seq, t8,
                                       beta_start=100.0, beta_end=2000.0,
                                       beta_growth=1.01, top_k=8,
                                       restart_every=1500, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_final", t_global)
    
    # Final cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print("\n=== Final: Cyclic shift ===")
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_final", t_global)
    
    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.6f}, n={}, time={:.1f}s".format(best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence! Using fallback.")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
    
    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
