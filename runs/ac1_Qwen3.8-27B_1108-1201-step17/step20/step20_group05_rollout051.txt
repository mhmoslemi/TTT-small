def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def fft_conv2(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if b.size != n:
        b = np.resize(b, n)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def cross_convolve_p_a(p, a, factor=1.0):
    p = np.asarray(p, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    total = 3 * n - 2
    if p.size != 2 * n - 1:
        p = np.resize(p, 2 * n - 1)
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    out = conv[n - 1:2 * n - 1][::-1]
    out = np.where(np.isfinite(out), out, 0.0)
    return factor * out

def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)


def lse_objective(a, beta):
    """Compute LSE-smoothed max of self-convolution and its gradient."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 0.0, np.array([], dtype=np.float64)
    size = 2 * n - 1
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)

    m = float(np.max(c))
    shifted = beta * (c - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0 or not math.isfinite(se):
        return 1e100, np.zeros(n, dtype=np.float64)
    p = e / se

    pf = np.zeros(size, dtype=np.float64)
    pf[:] = p

    Pr = np.fft.rfft(pf[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    f = m + math.log(se) / beta
    if not math.isfinite(f):
        f = 1e100
    return f, grad


def softmax_from_u(u):
    u = np.asarray(u, dtype=np.float64)
    n = u.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(u)):
        return np.ones(n, dtype=np.float64) / float(n)
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def make_v(a, ref=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 1:
        return np.array([], dtype=np.float64), 0
    if ref is None:
        if np.any(a > 0.0):
            ref = int(np.argmax(a))
        else:
            ref = 0
    ref = int(ref)
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.where(np.isfinite(v), v, -700.0)
    return v, ref


def from_v(v, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    return softmax_from_u(u)


def objgrad_v(v, beta, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    if n <= 0:
        return 1e100, np.array([], dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return 1e100, np.zeros(v.size, dtype=np.float64)

    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1

    u = np.empty(n, dtype=np.float64)
    u[ref] = 0.0
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]

    a = softmax_from_u(u)
    f, ga = lse_objective(a, beta)
    if not math.isfinite(f):
        f = 1e100
        ga = np.zeros(n, dtype=np.float64)

    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)

    gv = np.empty(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]

    if not np.all(np.isfinite(gv)):
        gv = np.zeros(v.size, dtype=np.float64)
    # Add small regularization for numerical stability
    gv = gv + 1e-12 * v
    return f, gv

def game_direction(f, active_idx, mw_iters=30, seed=0):
    f = norm_sum1(f)
    n = f.size
    size = 2 * n - 1
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K <= 0:
        return np.ones(n, dtype=np.float64) / float(n)

    rng = np.random.default_rng(seed)
    logits = rng.normal(0.0, 0.1, K)

    def normalize_logits(loc):
        loc = np.where(np.isfinite(loc), loc, -1000.0)
        loc = loc - float(np.max(loc))
        w = np.exp(loc)
        s = float(w.sum())
        if s <= 0.0:
            w = np.ones(K, dtype=np.float64)
            s = float(K)
        return w / s

    lambda_w = normalize_logits(logits)
    g = None
    for it in range(max(1, int(mw_iters))):
        pf = np.zeros(size, dtype=np.float64)
        pf[active_idx] = lambda_w

        pressure = cross_convolve_p_a(pf, f, 1.0)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)

        pmx = float(np.max(pressure))
        pmn = float(np.min(pressure))
        ptp = pmx - pmn
        if ptp <= 1e-30:
            g = np.ones(n, dtype=np.float64) / float(n)
        else:
            scaled = (pressure - pmn) / ptp
            # Use adaptive temperature
            temp = 6.0 + 2.0 * (it / max(1, mw_iters))
            w = np.exp(-temp * scaled)
            w = np.where(np.isfinite(w), w, 0.0)
            w += 1e-12
            g = w / float(w.sum())
            g = 0.95 * g + 0.05 * (np.ones(n, dtype=np.float64) / float(n))
            g = g / float(g.sum())

        h = fft_conv2(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmac_ = float(np.max(ha))
        hmin = float(np.min(ha))
        hptp = hmac_ - hmin
        if hptp > 1e-30:
            eta = 10.0 / hptp
            logits += eta * (ha - float(np.mean(ha)))
        lambda_w = normalize_logits(logits)

    return g


def line_search_fw(a, g, cf, cg, fg, current_M):
    """Find best mixing alpha in [0,1] for a + alpha*(g-a)."""
    n = a.size
    base_alphas = [
        float(2.0 / (30.0)),  # theoretical FW rate
        0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0
    ]
    best_alpha = 0.0
    best_m = current_M
    for alpha in base_alphas:
        one = 1.0 - alpha
        val = (one * one) * cf + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
        m = float(np.max(val))
        if math.isfinite(m) and m < best_m:
            best_m = m
            best_alpha = alpha

    if best_alpha <= 0.0:
        return a.copy(), best_m, 0.0

    new_a = (1.0 - best_alpha) * a + best_alpha * g
    new_a = np.maximum(new_a, 0.0)
    s = float(new_a.sum())
    if s < 1e-15:
        return a.copy(), best_m, 0.0
    new_a = new_a / s
    return new_a, best_m, best_alpha

def apply_single_move(c, a, s, d, tmove):
    """Apply rank-1 update for moving mass from position s to d."""
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = a.size
    c_new = c.copy()
    c_new[d:d + n] += 2.0 * tmove * a
    c_new[s:s + n] -= 2.0 * tmove * a

    t2 = tmove * tmove
    if 2 * d < c_new.size:
        c_new[2 * d] += t2
    if 2 * s < c_new.size:
        c_new[2 * s] += t2
    if s + d < c_new.size:
        c_new[s + d] -= 2.0 * t2
    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
    return c_new

                    continue
                for frac in (0.003, 0.008, 0.015, 0.03, 0.06):
                    tmove = a[s] * frac
                    if tmove <= 1e-16:
                        continue
                    tmove = min(tmove, a[s] * 0.99)
                    c_new = apply_single_move(c, a, s, d, tmove)
                    m_new = proxy_max(c_new)
                    if m_new < best_m_new - 1e-18:
                        best_m_new = m_new
                        best_cand = (m_new, s, d, tmove)

        # If no improvement from structured search, try random
        if best_cand is None:
            for _ in range(40):
                s = int(rng.integers(n))
                d = int(rng.integers(n))
                if s == d or a[s] < 1e-12:
                    continue
                frac = float(rng.uniform(0.002, 0.04))
                tmove = a[s] * frac
                tmove = min(tmove, a[s] * 0.99)
                if tmove <= 1e-16:
                    continue
                c_new = apply_single_move(c, a, s, d, tmove)
                m_new = proxy_max(c_new)
                if m_new < best_m_new - 1e-18:
                    best_m_new = m_new
                    best_cand = (m_new, s, d, tmove)

        if best_cand is not None:
            m_new, s, d, tmove = best_cand
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = fft_conv(a)
            M = proxy_max(c)
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and (rnd <= 5 or rnd % 30 == 0):
                print("    Coord rnd=" + str(rnd) +
                      " proxy=" + repr(2.0 * n * best_M) +
                      " s=" + str(s) + " d=" + str(d))
        else:
            stall += 1
            if stall % 15 == 14:
                # Perturbation to escape local minimum
                scale = min(0.03, 0.08 / (1.0 + stall * 0.05))
                pert = rng.normal(0.0, scale, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-12)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = proxy_max(c)
                stall = max(0, stall - 5)

    return best_a


def spectral_init(n, seed=0):
    """Generate a good initial sequence using spectral methods."""
    rng = np.random.default_rng(seed)
    # Random phases
    phases = rng.uniform(0, 2 * np.pi, n)
    # Inverse FFT gives approximately flat autocorrelation
    a = np.real(np.fft.ifft(np.exp(1j * phases)))
    a = np.abs(a)  # Take absolute value for non-negativity
    # Mix with a smooth component
    x = np.linspace(0, 1, n)
    smooth = np.sin(np.pi * x) ** 2
    a = 0.5 * a + 0.5 * smooth
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def spectral_init_v2(n, seed=0):
    """Better spectral init: use overlapping chirp structure."""
    rng = np.random.default_rng(seed)
    # Create a chirp-like signal
    x = np.linspace(0, 1, n)
    freq = rng.uniform(0.5, 2.0)
    a = np.sin(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    a = np.abs(a)
    # Add random modulation
    mod = 0.5 + 0.5 * np.sin(2 * np.pi * rng.uniform(1, 5) * x + rng.uniform(0, 2 * np.pi))
    a = a * mod
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def default_sequence(n=5000):
    x = np.linspace(0.0, 1.0, int(n))
    return norm_sum1(np.sin(np.pi * x) ** 2)


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    total_budget = max(60.0, min(float(budget_s) - 75.0, 1725.0))
    print("Start: total_budget=" + repr(total_budget) + "s")

    # Get parent sequence
    seq_best = None
    try:
        parent_list = height_sequence_1
        p = np.asarray(parent_list, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        np.clip(p, 0.0, 1000.0, out=p)
        if p.ndim != 1:
            p = p.ravel()
        if p.size > 0 and float(p.sum()) > 1e-10:
            seq_best = norm_sum1(p)
    except Exception:
        pass

    if seq_best is None:
        seq_best = default_sequence(5000)

    best_score = safe_eval(seq_best)
    if not math.isfinite(best_score):
        best_score = 2.0 * seq_best.size * proxy_max(fft_conv(seq_best))

    print("Parent: n=" + str(seq_best.size) + " score=" + repr(best_score))

    # Work at a manageable length
    work_n = min(seq_best.size, 8000)
    if seq_best.size > 8000:
        work_seq = interpolate_sequence(seq_best, 8000)
    else:
        work_seq = seq_best.copy()

    seq_best, best_score = try_update(seq_best, best_score, work_seq, "initial", t_start)

    # Generate alternative initializations
    inits = [work_seq.copy()]
    inits.append(spectral_init(work_n, seed=7))
    inits.append(spectral_init(work_n, seed=13))
    inits.append(spectral_init_v2(work_n, seed=23))

    # Evaluate all initializations
    for i, init in enumerate(inits):
        sc = safe_eval(init)
        print("  Init " + str(i) + ": score=" + repr(sc) if math.isfinite(sc) else "  Init " + str(i) + ": invalid")

    # Phase 1: Multi-scale L-BFGS on each initialization (time-limited)
    print("\n=== Phase 1: Multi-scale L-BFGS ===")
    t0 = time.monotonic()
    phase1_budget = min(total_budget * 0.35, 600.0)
    per_init_budget = phase1_budget / len(inits)

    best_after_p1 = work_seq.copy()
    best_p1_score = safe_eval(work_seq)
    if not math.isfinite(best_p1_score):
        best_p1_score = 1e6

    for i, init in enumerate(inits):
        time_left = phase1_budget - (time.monotonic() - t0)
        if time_left < 30:
            break
        given_budget = min(per_init_budget * 1.3, time_left - 10)
        if given_budget < 20:
            break
        print("  L-BFGS on init " + str(i) + " with budget=" + repr(given_budget))
        a_new = multiscale_lbfegs(init, given_budget, verbose=(i == 0))
        sc = safe_eval(a_new)
        if math.isfinite(sc) and sc < best_p1_score:
            best_p1_score = sc
            best_after_p1 = a_new.copy()
            print("    Improved to " + repr(sc))

    seq_best, best_score = try_update(seq_best, best_score, best_after_p1, "phase1", t_start)
    work_seq = best_after_p1.copy()

    # Phase 2: Frank-Wolfe search
    print("\n=== Phase 2: Frank-Wolfe ===")
    t0 = time.monotonic()
    phase2_budget = min(total_budget * 0.25, 420.0)
    time_left_total = total_budget - (time.monotonic() - t_start)
    phase2_budget = min(phase2_budget, time_left_total * 0.35)

    if phase2_budget > 50:
        a_fw = frank_wolfe_search(work_seq, phase2_budget,
                                  active_max=2500, game_iters=28, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_fw, "fw", t_start)
        work_seq = a_fw.copy()

    # Phase 3: Coordinate polish
    print("\n=== Phase 3: Coordinate Polish ===")
    t0 = time.monotonic()
    phase3_budget = min(total_budget * 0.15, 250.0)
    time_left_total = total_budget - (time.monotonic() - t_start)
    phase3_budget = min(phase3_budget, time_left_total * 0.30)

    if phase3_budget > 30:
        a_cp = coordinate_polish(work_seq, phase3_budget, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_cp, "coord1", t_start)
        work_seq = a_cp.copy()

    # Phase 4: Length exploration
    print("\n=== Phase 4: Length Exploration ===")
    t0 = time.monotonic()
    time_left_total = total_budget - (time.monotonic() - t_start)
    phase4_budget = min(time_left_total * 0.25, 300.0)

    if phase4_budget > 60:
        base_len = work_seq.size
        # Try a few nearby lengths
        length_candidates = []
        for delta in [-200, -100, 100, 200, 300, -300]:
            target = base_len + delta
            if target >= 1000 and target <= 12000:
                length_candidates.append(target)
        # Also try the original parent length if different
        if seq_best.size != base_len and seq_best.size >= 1000:
            length_candidates.append(seq_best.size)

        # Deduplicate and sort by distance from base
        seen = set()
        unique_lens = []
        length_candidates.sort(key=lambda x: abs(x - base_len))
        for ln in length_candidates:
            if ln not in seen:
                seen.add(ln)
                unique_lens.append(ln)

        for target_n in unique_lens[:4]:
            time_left_total = total_budget - (time.monotonic() - t_start)
            if time_left_total < 80:
                break
            per_len_budget = min(phase4_budget / max(1, len(unique_lens)), time_left_total - 20)
            if per_len_budget < 30:
                break

            print("  Trying n=" + str(target_n) + " budget=" + repr(per_len_budget))
            a_interp = interpolate_sequence(work_seq, target_n)

            # Quick L-BFGS
            if SCIPY_OK and per_len_budget > 40:
                a_lb = run_lbfegs(a_interp, 100.0, maxfun=150, verbose=False)
                sc = safe_eval(a_lb)
                if math.isfinite(sc):
                    print("    L-BFGS score=" + repr(sc))
                else:
                    a_lb = a_interp
                a_interp = a_lb

            # Coordinate polish
            if per_len_budget > 25:
                a_cp = coordinate_polish(a_interp, min(per_len_budget * 0.4, 60.0), verbose=False)
                seq_best, best_score = try_update(seq_best, best_score, a_cp, "len_" + str(target_n), t_start)

    # Phase 5: Final intensive polish on the best
    print("\n=== Phase 5: Final Polish ===")
    t0 = time.monotonic()
    time_left_total = total_budget - (time.monotonic() - t_start)
    phase5_budget = min(time_left_total * 0.40, 200.0)

    if phase5_budget > 20:
        # Use the best sequence found
        final_work = seq_best.copy()
        if final_work.size > 9000:
            final_work = interpolate_sequence(final_work, 8000)

        # Coordinate polish
        a_final = coordinate_polish(final_work, phase5_budget * 0.6, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_final, "final_coord", t_start)

        # One more L-BFGS pass
        time_left_total = total_budget - (time.monotonic() - t_start)
        if SCIPY_OK and time_left_total > 30:
            a_final2 = run_lbfegs(a_final, 500.0, maxfun=100, verbose=False)
            seq_best, best_score = try_update(seq_best, best_score, a_final2, "final_lb", t_start)

            # Tiny coordinate polish
            if time_left_total > 20:
                a_final3 = coordinate_polish(a_final2, min(time_left_total - 10, 40.0), verbose=False)
                seq_best, best_score = try_update(seq_best, best_score, a_final3, "final_coord2", t_start)

    # Final validation
    final_val = safe_eval(seq_best)
    if math.isfinite(final_val):
        if final_val < best_score:
            best_score = final_val
    else:
        seq_best = default_sequence(5000)
        best_score = safe_eval(seq_best)

    arr = np.asarray(seq_best, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if arr.size == 0 or s < 0.01:
        arr = default_sequence(5000)
        s = float(arr.sum())
    else:
        arr = arr / s

    elapsed = time.monotonic() - t_start
    print("\nFINAL: score=" + repr(best_score) +
          " n=" + str(arr.size) +
          " elapsed=" + repr(elapsed))

    return [float(x) for x in arr]
