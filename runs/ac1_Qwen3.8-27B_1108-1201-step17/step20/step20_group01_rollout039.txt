def random_spec_seq(n, seed, harmonics=6):
    rng=np.random.default_rng(seed)
    x=np.arange(n)/float(n)

    f=np.maximum(f, 1e-4)

def random_spec_seq(n, seed):
    rng=np.random.default_rng(seed)
    x=np.arange(n,dtype=np.float64)/float(n)
    f=np.ones(n)

"""
Minimizer for 2*n*max(conv(a,a))/(sum(a))^2 with a_i >= 0.

Strategy:
- Work with a fast FFT-based self-convolution proxy that matches np.convolve.
- Multi-start from the parent construction, Beta-family shapes, and random
  positive spectral shapes.
- Main driver: Adam-like exponentiated-gradient descent in log-space on the
  simplex, using softmax weights over currently large convolution lags.
- Polish with a batch pressure-based mass mover, active-set LP/Frank-Wolfe
  direction, and exact pairwise coordinate moves.
- Explore a longer length by interpolation and continued optimization.
"""

import numpy as np
import time
import math

try:
    import cvxpy as cp
except Exception:
    cp = None


_SOLVER = None


def get_solver():
    global _SOLVER
    if _SOLVER is not None:
        return _SOLVER
    if cp is None:
        return None

    names = ("CLARABEL", "PDLP", "SCS", "ECOS")
    candidates = []
    for name in names:
        try:
            candidates.append(getattr(cp, name))
        except Exception:
            pass

    for s in candidates:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                _SOLVER = s
                print("  LP solver: " + str(s))
                return _SOLVER
        except Exception:
            pass

    try:
        _SOLVER = cp.ECOS
        print("  LP solver: ECOS fallback")
    except Exception:
        _SOLVER = None
    return _SOLVER


def normalize(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return a
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    c = np.maximum(c, 0.0)
    return c


def quick_score_norm(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    n = int(a.size)
    c = fast_conv(a)
    return float(2.0 * n * float(c.max()))


def score_and_lse(a, temp):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return float("inf"), float("inf")
    c = fast_conv(a)
    score = float(2.0 * n * float(c.max()))
    d = c * n
    if d.size == 0:
        return score, float("inf")
    maxd = float(d.max())
    if temp <= 1e-15:
        lse = maxd
    else:
        e = np.exp((d - maxd) / temp)
        s = float(e.sum())
        if not np.isfinite(s) or s <= 0.0:
            lse = maxd
        else:
            lse = maxd + temp * math.log(s)
    return score, lse


def eval_safe(a):
    try:
        arr = np.asarray(a, dtype=np.float64).reshape(-1)
        lst = [float(x) for x in arr]
        val = evaluate_sequence(lst)
        return float(val)
    except Exception:
        return float("inf")


def interpolate_seq(f, new_n):
    f = np.asarray(f, dtype=np.float64).reshape(-1)
    old_n = int(f.size)
    new_n = int(new_n)
    if new_n <= 0:
        return f
    if old_n == new_n:
        return f.copy()
    s = float(f.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    out = np.interp(new_x, old_x, fn)
    return normalize(out)


def beta_seq(n, p, q):
    n = int(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    if abs(p) < 1e-15:
        pi = np.ones(n, dtype=np.float64)
    else:
        pi = np.power(i, p)
    if abs(q) < 1e-15:
        pj = np.ones(n, dtype=np.float64)
    else:
        pj = np.power(j, q)
    return normalize(pi * pj)


def random_spec_seq(n, seed):
    n = int(n)
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64) / float(n)
    f = np.ones(n, dtype=np.float64)
    harmonics = 8
    two_pi = 2.0 * np.pi
    for k in range(1, harmonics + 1):
        amp = rng.uniform(0.0, 1.8 / float(k))
        phase = rng.uniform(0.0, two_pi)
        f = f + amp * np.cos(two_pi * k * x + phase)
    f = np.maximum(f, 1e-4)
    power = rng.uniform(0.5, 3.0)
    f = np.power(f, power)
    f = np.where(np.isfinite(f), f, 1.0)
    return normalize(f)


def generate_alternatives(n, parent):
    n = int(n)
    cands = []
    if parent is not None:
        cands.append(parent.copy())

    # Small scan over Beta-family shapes.
    n_scan = 512
    p_vals = (-0.25, 0.0, 0.25, 0.5, 1.0, 2.0)
    q_vals = (-0.25, 0.0, 0.25, 0.5, 1.0, 2.0)
    best_beta = []
    for p in p_vals:
        for q in q_vals:
            seq = beta_seq(n_scan, p, q)
            sc = quick_score_norm(seq)
            best_beta.append((sc, p, q))
    best_beta.sort()
    for _sc, p, q in best_beta[:2]:
        cands.append(beta_seq(n, p, q))

    # A few safe fixed sinusoidal shapes.
    x = np.arange(n, dtype=np.float64) / float(n)
    two_pi = 2.0 * np.pi

    f1 = 1.0 + 0.35 * np.cos(two_pi * x) + 0.15 * np.cos(4.0 * np.pi * x + 0.7)
    if float(f1.min()) > 1e-3:
        cands.append(normalize(f1))

    f2 = 1.0 + 0.50 * np.cos(two_pi * (x + 0.25))
    if float(f2.min()) > 1e-3:
        cands.append(normalize(f2))

    f3 = 1.0 + 0.25 * np.cos(2.0 * two_pi * x) - 0.20 * np.cos(3.0 * two_pi * x)
    if float(f3.min()) > 1e-3:
        cands.append(normalize(f3))

    for seed in (0, 1, 2):
        cands.append(random_spec_seq(n, 100 + seed))

    return cands


def top_weights(c, K, temp):
    c = np.asarray(c, dtype=np.float64)
    size = int(c.size)
    K = int(K)
    if size <= 0 or K <= 0:
        return np.zeros(0, dtype=np.int64), np.zeros(0, dtype=np.float64)
    if K >= size:
        idx = np.arange(size, dtype=np.int64)
        vals = c
    else:
        part = np.argpartition(c, size - K)
        idx = part[size - K:].astype(np.int64, copy=False)
        vals = c[idx]
        order = np.argsort(vals)[::-1]
        idx = idx[order]
        vals = vals[order]

    k = int(idx.size)
    if k == 0:
        return idx, np.zeros(0, dtype=np.float64)

    maxv = float(vals[0])
    if temp <= 1e-15:
        w = np.ones(k, dtype=np.float64) / float(k)
    else:
        e = np.exp((vals - maxv) / temp)
        s = float(e.sum())
        if not np.isfinite(s) or s <= 0.0:
            w = np.ones(k, dtype=np.float64) / float(k)
        else:
            w = e / s
    return idx, w


def sparse_grad(a, lags, weights):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    g = np.zeros(n, dtype=np.float64)
    lags = np.asarray(lags, dtype=np.int64)
    weights = np.asarray(weights, dtype=np.float64)
    for ii in range(int(lags.size)):
        m = int(lags[ii])
        w = float(weights[ii])
        if w <= 0.0:
            continue
        lo = m - (n - 1)
        if lo < 0:
            lo = 0
        hi = m
        if hi > n - 1:
            hi = n - 1
        if lo > hi:
            continue
        seg = a[m - hi: m - lo + 1]
        if seg.size != hi - lo + 1:
            continue
        g[lo:hi + 1] += w * seg[::-1]
    return g


def apply_log_update(log_a, step, eta):
    log_new = log_a - eta * step
    maxv = float(np.max(log_new))
    if not np.isfinite(maxv):
        return None
    log_new = log_new - maxv
    trial = np.exp(log_new)
    s = float(trial.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return trial / s


def entropy_optimize(a0, time_limit, verbose=False, seed=1,
                     K0=80, Kmax=500, temp0=0.30, temp1=0.04,
                     lr_init=0.35, max_stall=450):
    t0 = time.monotonic()
    if time_limit <= 0.0:
        a = normalize(a0)
        return a, quick_score_norm(a)

    a = normalize(a0)
    n = int(a.size)
    log_a = np.log(np.maximum(a, 1e-300))
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    best_a = a.copy()
    best_score = quick_score_norm(a)
    eta = float(lr_init)
    stall = 0
    it = 0
    rng = np.random.default_rng(seed)

    if verbose:
        print("  entropy n={} start={:.8f}".format(n, best_score))

    while True:
        elapsed = time.monotonic() - t0
        if elapsed >= time_limit:
            break
        it += 1
        progress = min(1.0, elapsed / max(1e-9, time_limit))

        c = fast_conv(a)
        score = float(2.0 * n * float(c.max()))
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
            if verbose and it % 100 == 0:
                print("    entropy it={} t={:.1f}s best={:.8f}".format(it, elapsed, best_score))
        else:
            stall += 1
            if verbose and it % 250 == 0:
                print("    entropy it={} t={:.1f}s best={:.8f} stall={}".format(
                    it, elapsed, best_score, stall))

        if stall > int(max_stall):
            sigma = max(0.03, 0.18 * (1.0 - progress))
            ar = best_a * (1.0 + sigma * rng.standard_normal(n))
            ar = np.maximum(ar, 1e-12)
            ar = normalize(ar)
            a = ar
            log_a = np.log(np.maximum(a, 1e-300))
            m.fill(0.0)
            v.fill(0.0)
            stall = 0
            continue

        d = c * float(n)
        K = int(K0 + (Kmax - K0) * min(1.0, progress * 1.1))
        K = max(1, min(K, 2 * n - 1))
        temp = temp1 + (temp0 - temp1) * ((1.0 - progress) ** 1.5)

        idx, w = top_weights(d, K, temp)
        if idx.size == 0:
            break

        g = sparse_grad(a, idx, w)
        gm = float(np.mean(g))
        gs = float(np.std(g))
        if not np.isfinite(gs) or gs <= 1e-14:
            ar = best_a * (1.0 + 0.02 * rng.standard_normal(n))
            ar = normalize(np.maximum(ar, 1e-12))
            a = ar
            log_a = np.log(np.maximum(a, 1e-300))
            m.fill(0.0)
            v.fill(0.0)
            stall = 0
            continue

        u = (g - gm) / gs
        beta1 = 0.9
        beta2 = 0.99
        m = beta1 * m + (1.0 - beta1) * u
        v = beta2 * v + (1.0 - beta2) * (u * u)

        bc1 = 1.0 - beta1 ** it
        bc2 = 1.0 - beta2 ** it
        step = (m / bc1) / (np.sqrt(v / bc2) + 1e-6)
        step = step - float(np.mean(step))

        maxd = float(d.max())
        if temp <= 1e-15:
            current_lse = maxd
        else:
            ec = np.exp((d - maxd) / temp)
            sc = float(ec.sum())
            current_lse = maxd + temp * math.log(sc) if sc > 0.0 else maxd

        cetas = [0.5 * eta, eta, 2.0 * eta]
        best_trial = None
        best_trial_score = score
        best_trial_lse = current_lse
        best_eta = eta

        for e in cetas:
            if not np.isfinite(e) or e <= 0.0:
                continue
            trial = apply_log_update(log_a, step, e)
            if trial is None:
                continue
            sc_t, lse_t = score_and_lse(trial, temp)
            exact_improve = (sc_t < score - 1e-13)
            lse_improve = (lse_t < current_lse - 1e-10) and (sc_t <= score + 1e-8)
            if exact_improve or lse_improve:
                if (best_trial is None or
                    sc_t < best_trial_score - 1e-14 or
                    (abs(sc_t - best_trial_score) < 1e-9 and lse_t < best_trial_lse - 1e-12)):
                    best_trial = trial
                    best_trial_score = sc_t
                    best_trial_lse = lse_t
                    best_eta = float(e)

        if best_trial is not None:
            a = best_trial
            log_a = np.log(np.maximum(a, 1e-300))
            eta = max(0.02, min(5.0, best_eta))
        else:
            eta = max(0.02, eta * 0.6)

    return best_a, best_score


def build_lp_coeffs(a, lags):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    lags = np.asarray(lags, dtype=np.int64)
    K = int(lags.size)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for ii in range(K):
        m = int(lags[ii])
        lo = m - (n - 1)
        if lo < 0:
            lo = 0
        hi = m
        if hi > n - 1:
            hi = n - 1
        if lo <= hi:
            coeffs[ii, lo:hi + 1] = a[m - hi: m - lo + 1][::-1]
    return coeffs


def lp_fw_optimize(a0, time_limit, verbose=False):
    if cp is None:
        a = normalize(a0)
        return a, quick_score_norm(a)
    solver = get_solver()
    if solver is None:
        a = normalize(a0)
        return a, quick_score_norm(a)

    t0 = time.monotonic()
    y = normalize(a0)
    n = int(y.size)
    best_a = y.copy()
    best_score = quick_score_norm(y)

    max_K = min(300, max(30, int(2500000 // max(1, n))))
    init_K = min(80, max_K)
    stall = 0
    it = 0

    if verbose:
        print("  LP-FW n={} start={:.8f}".format(n, best_score))

    while True:
        elapsed = time.monotonic() - t0
        if elapsed >= time_limit or it > 500:
            break
        it += 1

        c = fast_conv(y)
        score = float(2.0 * n * float(c.max()))
        if score < best_score - 1e-12:
            best_score = score
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        if stall > 40:
            break

        K = min(max_K, init_K + 2 * it)
        d = c * float(n)
        idx, _ = top_weights(d, K, 0.1)
        if idx.size == 0:
            break

        coeffs = build_lp_coeffs(y, idx)
        solved = False
        for attempt in range(2):
            try:
                g_var = cp.Variable(n, nonneg=True)
                m_var = cp.Variable()
                prob = cp.Problem(
                    cp.Minimize(m_var),
                    [cp.sum(g_var) == 1, m_var >= 0, coeffs @ g_var <= m_var]
                )
                prob.solve(solver=solver, verbose=False)
                if prob.status in ("optimal", "optimal_inaccurate"):
                    solved = True
                    break
            except Exception:
                pass
            K = max(30, K // 2)
            idx, _ = top_weights(d, K, 0.1)
            if idx.size == 0:
                break
            coeffs = build_lp_coeffs(y, idx)

        if not solved:
            break

        try:
            g = np.asarray(g_var.value, dtype=np.float64)
        except Exception:
            break
        g = np.maximum(g, 0.0)
        gs = float(g.sum())
        if not np.isfinite(gs) or gs <= 1e-15:
            continue
        g = g / gs

        alphas = (1.0 / (it + 1), 0.05, 0.15, 0.30, 0.50, 0.80, 1.00)
        best_trial = None
        best_trial_score = score
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g
            trial = np.maximum(trial, 0.0)
            ts = float(trial.sum())
            if ts <= 1e-15:
                continue
            trial = trial / ts
            sc = quick_score_norm(trial)
            if sc < best_trial_score - 1e-14:
                best_trial_score = sc
                best_trial = trial

        if best_trial is not None:
            y = best_trial
            if verbose and it % 5 == 0:
                print("    LP-FW it={} t={:.1f}s best={:.8f}".format(it, time.monotonic() - t0, best_score))
        else:
            stall += 1

    return best_a, best_score


def batch_mover(a0, time_limit, verbose=False, seed=21):
    t0 = time.monotonic()
    a = normalize(a0)
    n = int(a.size)
    best_a = a.copy()
    best_score = quick_score_norm(a)
    rng = np.random.default_rng(seed)

    if verbose:
        print("  batch n={} start={:.8f}".format(n, best_score))

    scales = (0.05, 0.025, 0.012, 0.006, 0.003)
    for scale in scales:
        stall = 0
        while True:
            elapsed = time.monotonic() - t0
            if elapsed >= time_limit:
                break
            if stall >= 40:
                break

            c = fast_conv(a)
            score = float(2.0 * n * float(c.max()))
            if score < best_score - 1e-13:
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

            d = c * float(n)
            idx, w = top_weights(d, 80, 0.05)
            if idx.size == 0:
                break
            g = sparse_grad(a, idx, w)

            src_order = np.argsort(g)[::-1]
            dst_order = np.argsort(g)

            sources = []
            for val in src_order:
                iv = int(val)
                if a[iv] > 1e-9 and len(sources) < 10:
                    sources.append(iv)
            dests = []
            for val in dst_order:
                iv = int(val)
                if len(dests) < 20:
                    dests.append(iv)

            if not sources or not dests:
                break

            a_new = a.copy()
            for s in sources:
                amt = a_new[s] * float(scale)
                if amt <= 0.0:
                    continue
                cand_list = []
                for d_idx in dests:
                    if d_idx != s:
                        cand_list.append(d_idx)
                if not cand_list:
                    continue
                cand_arr = np.asarray(cand_list, dtype=np.int64)
                pref = np.maximum(-float(g[cand_arr].max()) - g[cand_arr], 0.0) + 1e-5
                ps = float(pref.sum())
                if ps <= 0.0:
                    continue
                frac_arr = pref / ps
                for ii in range(cand_arr.size):
                    mv = amt * float(frac_arr[ii])
                    a_new[s] -= mv
                    a_new[cand_arr[ii]] += mv

            a_new = np.maximum(a_new, 0.0)
            st = float(a_new.sum())
            if st <= 1e-300:
                continue
            a_new = a_new / st
            sc = quick_score_norm(a_new)
            if sc < best_score - 1e-13:
                a = a_new
                best_a = a_new
                best_score = sc
                stall = 0
            else:
                a = best_a.copy()

            if verbose and stall % 20 == 0:
                print("    batch t={:.1f}s best={:.8f} stall={}".format(
                    time.monotonic() - t0, best_score, stall))

    return best_a, best_score


def pair_polish(a0, time_limit, verbose=False):
    t0 = time.monotonic()
    a = normalize(a0)
    n = int(a.size)
    best_a = a.copy()
    best_score = quick_score_norm(a)
    rng = np.random.default_rng(13)
    fracs = (0.002, 0.005, 0.01, 0.02, 0.05, 0.10)

    if verbose:
        print("  pair n={} start={:.8f}".format(n, best_score))

    sweep = 0
    stall = 0
    while True:
        elapsed = time.monotonic() - t0
        if elapsed >= time_limit or stall >= 30:
            break
        sweep += 1

        c = fast_conv(a)
        score = float(2.0 * n * float(c.max()))
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if verbose and sweep % 25 == 0:
            print("    pair sweep={} t={:.1f}s best={:.8f}".format(sweep, elapsed, best_score))

        d = c * float(n)
        idx, w = top_weights(d, 60, 0.05)
        if idx.size == 0:
            break
        g = sparse_grad(a, idx, w)

        src_order = np.argsort(g)[::-1][:8]
        dst_order = np.argsort(g)[:8]

        found = False
        for sii in range(src_order.size):
            s = int(src_order[sii])
            if found or a[s] <= 1e-300:
                continue
            for djj in range(dst_order.size):
                didx = int(dst_order[djj])
                if found or didx == s:
                    continue
                for frac in fracs:
                    amt = a[s] * float(frac)
                    if amt <= 1e-300:
                        continue
                    a_t = a.copy()
                    a_t[s] -= amt
                    a_t[didx] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = float(a_t.sum())
                    if st <= 1e-300:
                        continue
                    a_t = a_t / st
                    sc = quick_score_norm(a_t)
                    if sc < best_score - 1e-13:
                        a = a_t
                        best_a = a_t
                        best_score = sc
                        found = True
                        break

        if not found:
            if sweep % 7 == 6 and n >= 2:
                s = int(rng.integers(n))
                didx = int(rng.integers(n))
                if s != didx and a[s] > 1e-300:
                    for frac in (0.01, 0.03, 0.08):
                        amt = a[s] * float(frac)
                        a_t = a.copy()
                        a_t[s] -= amt
                        a_t[didx] += amt
                        a_t = normalize(np.maximum(a_t, 0.0))
                        sc = quick_score_norm(a_t)
                        if sc < best_score - 1e-13:
                            a = a_t
                            best_a = a_t
                            best_score = sc
                            found = True
                            break
            if not found:
                a = best_a.copy()

    return best_a, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    try:
        total_budget = max(60.0, float(budget_s) - 30.0)
    except Exception:
        total_budget = 1750.0
    time_budget = min(total_budget, 1750.0)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if p.size >= 100 and np.all(np.isfinite(p)):
            p = np.clip(p, 0.0, 1000.0)
            if float(p.sum()) > 1e-10:
                parent = normalize(p)
    except Exception:
        parent = None

    if parent is not None:
        parent_eval = eval_safe(parent)
        print("Parent: n={} eval={}".format(parent.size, repr(parent_eval)))
    else:
        parent_eval = float("inf")
        print("No usable parent.")

    n_main = int(parent.size) if parent is not None else 12000
    cands = generate_alternatives(n_main, parent)

    multi = []
    if len(cands) > 0:
        s0 = quick_score_norm(cands[0])
        multi.append((s0, 0, cands[0]))
    alts = []
    for ii in range(1, len(cands)):
        sc = quick_score_norm(cands[ii])
        alts.append((sc, ii, cands[ii]))
    alts.sort()
    for item in alts[:3]:
        multi.append(item)
    multi.sort()

    if not multi:
        fb = beta_seq(10000, 0.0, 0.0)
        multi = [(quick_score_norm(fb), 0, fb)]

    best_seq = None
    best_score = float("inf")

    # Quick multi-start.
    n_multi = min(4, len(multi))
    quick_each = min(30.0, max(10.0, (time_budget * 0.07) / max(1, n_multi)))
    for ci in range(n_multi):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < quick_each + 60.0:
            quick_each = max(5.0, (remaining - 30.0) / 2.0)
        if quick_each <= 0.0:
            break
        _sc, _tag, seq = multi[ci]
        a_out, s_out = entropy_optimize(
            seq, quick_each, verbose=False, seed=30 + 2 * ci,
            K0=60, Kmax=300, temp0=0.25, temp1=0.05,
            lr_init=0.30, max_stall=150
        )
        if np.isfinite(s_out) and s_out < best_score:
            best_score = s_out
            best_seq = a_out
        print("Quick cand {} proxy={:.8f} t={:.1f}s".format(ci, best_score, time.monotonic() - t_global))

    if best_seq is None:
        best_seq = multi[0][2].copy()
        best_score = quick_score_norm(best_seq)

    # Main entropy optimization.
    remaining = time_budget - (time.monotonic() - t_global)
    t_main = min(remaining * 0.35, 550.0)
    if t_main > 30.0:
        a_out, s_out = entropy_optimize(
            best_seq, t_main, verbose=True, seed=7,
            K0=100, Kmax=550, temp0=0.35, temp1=0.03,
            lr_init=0.40, max_stall=450
        )
        if np.isfinite(s_out) and s_out < best_score:
            best_score = s_out
            best_seq = a_out
        print("After main entropy proxy={:.8f} n={}".format(best_score, best_seq.size))

    # Batch pressure mover.
    remaining = time_budget - (time.monotonic() - t_global)
    t_batch = min(remaining * 0.15, 180.0)
    if t_batch > 30.0:
        a_out, s_out = batch_mover(best_seq, t_batch, verbose=True, seed=21)
        if np.isfinite(s_out) and s_out < best_score:
            best_score = s_out
            best_seq = a_out
        print("After batch proxy={:.8f} n={}".format(best_score, best_seq.size))

    # LP/Frank-Wolfe polish.
    remaining = time_budget - (time.monotonic() - t_global)
    t_lp = min(remaining * 0.20, 300.0)
    if t_lp > 60.0:
        a_out, s_out = lp_fw_optimize(best_seq, t_lp, verbose=True)
        if np.isfinite(s_out) and s_out < best_score:
            best_score = s_out
            best_seq = a_out
        print("After LP proxy={:.8f} n={}".format(best_score, best_seq.size))

    # Pairwise exact polish.
    remaining = time_budget - (time.monotonic() - t_global)
    t_pair = min(remaining * 0.15, 200.0)
    if t_pair > 30.0:
        a_out, s_out = pair_polish(best_seq, t_pair, verbose=True)
        if np.isfinite(s_out) and s_out < best_score:
            best_score = s_out
            best_seq = a_out
        print("After pairs proxy={:.8f} n={}".format(best_score, best_seq.size))

    # Explore a longer length.
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 180.0 and int(best_seq.size) < 16000:
        n_new = min(int(best_seq.size) + 2500, 16000)
        if n_new != int(best_seq.size):
            a_new = interpolate_seq(best_seq, n_new)
            t_exp1 = min(remaining * 0.50, 300.0)
            if t_exp1 > 60.0:
                a_out, s_out = entropy_optimize(
                    a_new, t_exp1, verbose=True, seed=99,
                    K0=120, Kmax=520, temp0=0.35, temp1=0.03,
                    lr_init=0.40, max_stall=350
                )
                if np.isfinite(s_out) and s_out < best_score:
                    best_score = s_out
                    best_seq = a_out
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 60.0 and int(best_seq.size) == n_new:
                t_exp2 = min(remaining * 0.50, 120.0)
                if t_exp2 > 30.0:
                    a_out, s_out = pair_polish(best_seq, t_exp2, verbose=False)
                    if np.isfinite(s_out) and s_out < best_score:
                        best_score = s_out
                        best_seq = a_out
            print("After length explore proxy={:.8f} n={}".format(best_score, best_seq.size))

    # Final validation and safe fallback.
    final_score = eval_safe(best_seq)
    if not np.isfinite(final_score):
        if parent is not None:
            ps = eval_safe(parent)
            if np.isfinite(ps):
                best_seq = parent
                final_score = ps
        if not np.isfinite(final_score):
            best_seq = beta_seq(10000, 0.0, 0.0)
            final_score = eval_safe(best_seq)

    print("FINAL score={} n={} t={:.1f}s".format(repr(final_score), int(best_seq.size), time.monotonic() - t_global))

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64).reshape(-1)]
