def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A @ g)[k] = conv(f, g)[k]."""
    n = len(f)
    # For each diagonal offset i (0 to n-1), the contribution is f[i] on that diagonal
    # Row k, col j: A[k,j] = f[k-j] if 0 <= k-j < n
    # Diagonal i = k-j, so k = j+i
    # For diagonal i: rows j+i, cols j, for j in range(n), i in range(n)
    #   valid when j+i < 2n-1, i.e., always for j<n, i<n (max j+i = 2n-2)
    data = []
    rows = []
    cols = []
    for i in range(n):
        # Diagonal i: A[j+i, j] = f[i] for j=0..n-1
        js = np.arange(n)
        ks = js + i
        # All valid since 0 <= j < n and i <= n-1, so k <= 2n-2
        data.extend([f[i]] * n)
        rows.extend(ks.tolist())
        cols.extend(js.tolist())
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A


def lp_iteration(f, M, solver=None):
    """Solve LP: max sum(g) s.t. g >= 0, conv(f,g) <= M."""
    import cvxpy as cp
    n = len(f)
    A = build_conv_matrix_sparse(f)
    g = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g))
    constraints = [A @ g <= M]
    prob = cp.Problem(objective, constraints)
    if solver is None:
        solver = get_solver()
    prob.solve(solver=solver, verbose=False)
    if g.value is None:
        return None
    return np.asarray(g.value, dtype=np.float64)


def lp_iteration_reduce(f, M, solver=None, top_constraints=2000):
    """Solve LP with only top-k most constraining rows for speed."""
    import cvxpy as cp
    n = len(f)
    A = build_conv_matrix_sparse(f)
    g = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g))
    
    # Use all constraints if n is small, else top-k
    if 2*n - 1 <= top_constraints:
        constraints = [A @ g <= M]
    else:
        # Find which rows are most constraining (have largest sum of f[i]*f[j] for valid pairs)
        # The "tightness" of row k is conv(f,f)[k] / M (higher = tighter)
        conv_ff = fast_autocorr(f)
        # Select top rows by conv_ff value
        order = np.argsort(conv_ff)[::-1][:top_constraints]
        A_sub = A[order, :]
        constraints = [A_sub @ g <= np.sort(conv_ff[order])[::-1] / M * M]
        # Actually, constraint is conv(f,g)[k] <= M for the selected k's
        constraints = [A_sub @ g <= M]
    
    prob = cp.Problem(objective, constraints)
    if solver is None:
        solver = get_solver()
    prob.solve(solver=solver, verbose=False)
    if g.value is None:
        return None
    return np.asarray(g.value, dtype=np.float64)


def lp_based_optimize(a0, time_limit, solver=None, verbose=True, t_start=0.15, t_decay=0.98):
    """Iterative LP-based optimization to flatten autocorrelation."""
    a = norm_sum1(a0)
    n = len(a)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    t_step = t_start
    M = best_M
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP
        g = lp_iteration(a, M, solver=solver)
        if g is None:
            # Try with slightly larger M
            g = lp_iteration(a, M * 1.01, solver=solver)
            if g is None:
                break
        
        g = np.maximum(g, 0.0)
        gsum = g.sum()
        if gsum < 1e-15:
            break
        
        # Rescale g to match sum(a)
        g = g * (a.sum() / gsum)
        
        # Convex combination
        t = min(t_step, 0.3)
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        M_new = fast_autocorr(a_new).max()
        
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
        M = M_new
        t_step *= t_decay
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0

            print("    LP it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + ", t=" + repr(round(t, 4)))

def nesterov_opt(a0, time_limit, beta_start=50.0, beta_end=3000.0,
                 beta_growth=1.006, eta_base=0.5, top_k=None,
                 restart_every=1500, verbose=True, seed=42):
    """Nesterov accelerated projected gradient on log-sum-exp smoothed objective."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    if top_k is None:
        top_k = max(10, int(n * 0.02))

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Compute gradient
        c = fast_autocorr(y)
        if top_k < len(c):
            idx = np.argsort(c)[::-1][:top_k]
            c_use = np.zeros(len(c))
            c_use[idx] = c[idx]
        else:
            c_use = c
        c_max = c_use.max()
        shifted = beta * (c_use - c_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        total = exp_vals.sum()
        if total < 1e-300:
            p = np.zeros_like(c_use)
            p[np.argmax(c)] = 1.0
        else:
            p = exp_vals / total
        
        # Gradient computation via FFT
        fft_size = 1 << (2 * n - 1).bit_length()
        P = np.fft.rfft(p, fft_size)
        A_rev = np.fft.rfft(y[::-1], fft_size)
        Conv = np.fft.irfft(P * A_rev, fft_size)
        grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
        
        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.03 / (1.0 + 0.05 * (t // restart_every))
            anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + ", beta=" + str(round(beta, 1)))

    return best_a, best_max_b


def peak_surgery(a0, time_limit, verbose=True, seed=77):
    """Targeted local search to reduce convolution peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a =
