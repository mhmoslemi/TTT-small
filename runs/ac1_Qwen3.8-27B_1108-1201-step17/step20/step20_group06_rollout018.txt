def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_waterfill(a, time_limit, verbose=True, seed=42):
    """
    Iterative spectral water-filling: reduce peaks in |A(f)|^2 by redistributing
    energy to lower-power bins. This is different from multiplicative shrinking -
    we explicitly conserve total spectral energy while flattening.
    """
    n = len(a)
    a = norm_sum1(a).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    nf = fft_size // 2 + 1

    it = 0
    for it in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Target: mean power of non-DC bins
        mean_power = power[1:].mean() if nf > 1 else power[0]

        # Water-filling: reduce bins above target, boost bins below
        above = power[1:] > mean_power * 1.1
        below = power[1:] < mean_power * 0.9

        n_above = above.sum()
        n_below = below.sum()

        if n_above == 0:
            # Nearly flat spectrum - do a small perturbation
            phase_pert = rng.standard_normal(nf) * 0.01
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial
            continue

        # Shrink above, boost below (conserve energy approximately)
        excess = (power[1:][above] - mean_power).sum()
        deficit = (mean_power - power[1:][below]).sum() if n_below > 0 else excess

        A_new = A.copy()
        for k_global in range(1, nf):
            k_local = k_global - 1
            if above[k_local]:
                # Reduce amplitude
                target_ratio = mean_power / power[k_global]
                frac = 1.0 - 0.15 * (1.0 - target_ratio)
                A_new[k_global] *= max(0.5, np.sqrt(frac))
            elif below[k_local]:
                # Boost amplitude
                target_ratio = mean_power / max(power[k_global], 1e-30)
                frac = 1.0 + 0.05 * min(target_ratio - 1.0, 2.0)
                A_new[k_global] *= min(1.5, np.sqrt(frac))

        a_trial = np.fft.irfft(A_new, fft_size)[:n]
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t

        m_t = fast_autocorr(a_trial).max()
        if m_t < best_m:
            best_m = m_t
            best_a = a_trial.copy()
            a = a_trial
        elif it % 500 == 0:
            # Perturb to escape local minimum
            A = np.fft.rfft(a, fft_size)
            magnitudes = np.abs(A)
            phase = np.angle(A)
            mag_pert = 1.0 + rng.standard_normal(nf) * 0.02
            magnitudes = np.maximum(magnitudes * mag_pert, 0.0)
            A_new = magnitudes * np.exp(1j * phase)
            a_trial = np.fft.irfft(A_new, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Waterfill it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def smooth_max_grad_direct(a, beta, top_k):
    """
    Compute gradient of smoothed max of top-k autocorrelation values.
    Uses analytical formula: grad = sum_j w_j * shift(a, k_j)
    where w_j are softmax weights.
    """
    n = len(a)
    b = fast_autocorr(a)
    size = 2 * n - 1

    idx = np.argsort(b)[::-1][:top_k]
    b_top = b[idx]
    b_max = b_top.max()

    if b_max <= 0:
        return np.zeros(n), b, b_max

    shifted = beta * (b_top - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        w = np.ones(len(idx)) / len(idx)
    else:
        w = exp_vals / s

    # Gradient: sum_j w[j] * shift(a, k_j)
    # shift(a, k) has (shift(a,k))[i] = a[k-i] for valid i
    grad = np.zeros(n)
    for jj in range(len(idx)):
        k = int(idx[jj])
        if w[jj] < 1e-10:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            # grad[i] += w[jj] * a[k - i] for i in [i_lo, i_hi]
            grad[i_lo:i_hi + 1] += w[jj] * a[k - i_lo:k - i_hi + 1][::-1]

    return grad / beta, b, b_max


def gradient_descent_direct(a0, time_limit, beta_start=30.0, beta_end=2000.0,
                            beta_growth=1.004, top_k=50, verbose=True, seed=123):
    """
    Direct gradient descent on smoothed max with aggressive backtracking.
    Different from Nesterov - uses simple step with careful line search.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    beta = beta_start
    step = 0.1
    stall = 0
    it = 0

    for it in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        grad, b, b_max = smooth_max_grad_direct(a, beta, top_k)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-12:
            break

        # Backtracking line search
        found = False
        step_trial = step
        for _ in range(15):
            a_trial = a - step_trial * grad
            a_trial = simplex_project(a_trial)
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < b_max + 1e-12:
                a = a_trial
                a = norm_sum1(a)
                found = True
                break
            step_trial *= 0.5

        if not found:
            # Accept smallest improvement or stay
            a_trial = a - step * 1e-3 * grad
            a_trial = simplex_project(a_trial)
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m:
                a = norm_sum1(a_trial)
            step = max(step * 0.3, 1e-6)
        else:
            m_new = fast_autocorr(a).max()
            if m_new < best_m - 1e-14:
                best_m = m_new
                best_a = a.copy()
                stall = 0
                step = min(step * 1.1, 0.5)
            else:
                stall += 1
                step = max(step * 0.7, 1e-6)

        # Grow beta
        if it % 40 == 0:
            beta = min(beta * beta_growth, beta_end)

        # Restart if stalled
        if stall > 500:
            stall = 0
            strategy = rng.integers(4)
            if strategy == 0:
                pert = rng.normal(0, 0.01, n) * best_a
                a = norm_sum1(best_a + pert)
            elif strategy == 1:
                a = 0.85 * best_a + 0.15 * rng.dirichlet(np.ones(n) * 0.5)
                a = norm_sum1(a)
            elif strategy == 2:
                # Focus on peak regions
                b = fast_autocorr(a)
                peak_lag = int(np.argmax(b))
                i_lo = max(0, peak_lag - n // 3)
                i_hi = min(n - 1, peak_lag + n // 3)
                local_pert = np.zeros(n)
                local_pert[i_lo:i_hi] = rng.normal(0, 0.005, i_hi - i_lo) * a[i_lo:i_hi]
                a = norm_sum1(a + local_pert)
            else:
                # Random shift
                shift = int(rng.integers(1, max(2, n // 20)))
                a = norm_sum1(np.roll(best_a, shift))
            step = 0.1
            beta = max(beta * 0.5, 10.0)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    GradDesc it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + str(int(beta)))

    return best_a, best_m


def analytical_peak_target(a0, time_limit, verbose=True, seed=77):
    """
    For the current peak in autocorrelation, find the optimal 2-coordinate
    move (move mass from position i to position j) that maximally reduces the peak.
    Uses analytical computation of the effect on the peak.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 3000:
        b = fast_autocorr(a)
        size = 2 * n - 1
        # Get top peaks
        top_peaks = np.argsort(b)[::-1][:min(20, size)]
        improved = False

        for pk in top_peaks[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(pk)
            # Peak at lag k: b[k] = sum_i a[i]*a[k-i]
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo > i_hi:
                continue

            # Find the pair (i, k-i) that contributes most
            pairs = np.arange(i_lo, i_hi + 1)
            contribs = a[pairs] * a[k - pairs]
            src = int(pairs[np.argmax(contribs)])
            partner = k - src

            if not (0 <= partner < n):
                continue

            # Try moving small mass from src (or partner) to a random destination
            best_move_improvement = 1e-15
            best_move = None

            for _ in range(15):
                dst = int(rng.integers(n))
                if dst == src or dst == partner:
                    continue

                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    # Estimate effect on b[k]:
                    # Old contribution from src: a[src]*a[k-src] = a[src]*a[partner]
                    # After move: (a[src]-amt)*a[partner] + a[dst]*a[k-dst] (new term if k-dst valid)
                    # Change in b[k] = -amt * a[partner] + (possibly new term)
                    delta = -amt * a[partner]
                    # New term: a[dst]*a[k-dst] if 0 <= k-dst < n
                    kd = k - dst
                    if 0 <= kd < n and kd != dst:
                        delta += a[dst] * a[kd] * 0  # was 0 before, now a[dst]+amt * a[kd]
                        # Actually: new a[dst] = a[dst]+amt, so new term = (a[dst]+amt)*a[kd]
                        # Old term at this position was a[dst]*a[kd] (if dst was in range)
                        # This is getting complex - just compute directly
                        pass

                    # Just do the actual computation
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-16:
                        best_m = m_trial
                        best_a = a_trial.copy()
                        a = a_trial
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 50:
                # Random exploration
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        m_new = fast_autocorr(a).max()
                        if m_new < best_m:
                            best_m = m_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakTarget (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def lp_fw_improved(a0, time_limit, solver, initial_K=400, max_K=3000,
                   add_per_iter=30, verbose=True):
    """LP-Frank-Wolfe with better active set management and warm starts."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 800
    g_prev = None

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Value()

        if g_prev is not None:
            g_var = cp.Variable(n, nonneg=True, value=g_prev)

        constraints = [cp.sum(g_var) == 1]
        # Use only a subset of constraints for speed if K is large
        if K > 1500:
            # Use the most binding constraints
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            top_idx = np.argsort(margins)[:1500]
            sub_coeffs = coeffs[top_idx]
            constraints.append(sub_coeffs @ g_var <= M_var)
        else:
            constraints.append(coeffs @ g_var <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        g_prev = g_val.copy()

        # Find violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Backtracking line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        for alpha in [best_alpha, best_alpha * 0.5, best_alpha * 2.0,
                      0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
            alpha = max(0.005, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and t % 150 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW it=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def diverse_initializations(n, seed=0):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    inits = []

    # Raised cosine
    inits.append(np.sin(np.pi * x) ** 2)

    # Gaussian (various widths)
    for sigma in [0.1, 0.2, 0.3, 0.4]:
        center = 0.5
        inits.append(np.exp(-0.5 * ((x - center) / sigma) ** 2))

    # Sinc-squared
    W = n * 0.3
    t = (x - 0.5) * W
    inits.append(np.sinc(t / W) ** 2)

    # Triangular
    inits.append(np.maximum(0, 1 - 2 * np.abs(x - 0.5)))

    # Power law
    inits.append((x * (1 - x)) ** 0.5)

    # Random smoothed
    r = rng.random(n)
    kernel = np.exp(-0.5 * (np.arange(-20, 21) / 5.0) ** 2)
    kernel /= kernel.sum()
    inits.append(np.convolve(r, kernel, mode='same'))

    return [norm_sum1(ii) for ii in inits]

    parent = None

        p = np.clip(p, 0.0, 1000.0)

            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(round(p_score, 6)))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6000
        x = np.linspace(0, 1, n_base)

        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(round(best_score, 6)))

    n0 = len(best_seq)
    print("Start: n=" + str(n0) + ", J=" + repr(round(best_score, 6)))

    # === Phase 1: Quick diverse initialization search ===
    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 4: LP-Frank-Wolfe ===

            best_seq, best_score = try_update(best_seq, best_score, a_out, "grad_n" + str(new_n), t_global)
