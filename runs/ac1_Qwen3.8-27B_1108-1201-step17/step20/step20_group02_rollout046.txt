def fast_convolve(a):
    """Compute a * a (self-convolution) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def l2_gradient(a):
    """
    Compute gradient of sum_k (h[k] - mean(h))^2 where h = conv(a,a).
    Assumes sum(a) = 1.
    Returns gradient vector of length n.
    """
    n = len(a)
    h = fast_convolve(a)
    c = 1.0 / (2 * n - 1)  # mean of h when sum(a)=1
    h_c = h - c
    # grad[i] = 4 * sum_k h_c[k] * a[k-i] for valid k-i
    # This is 4 * cross-correlation of h_c and a
    full_corr = np.correlate(h_c, a, mode='full')
    # full_corr[k] = sum_n h_c[n+k] * a[n]
    # We want w[i] = sum_k h_c[k] * a[k-i] = full_corr[i] (by the formula above)
    return 4.0 * full_corr[:n]


def l2_mirror_descent(a0, time_limit, verbose=True):
    """
    Minimize variance of autocorrelation using multiplicative weights.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = 2.0 * n * fast_convolve(best_a).max()
    # Also track L2 objective
    h = fast_convolve(a)
    l2_obj = np.sum((h - 1.0/(2*n-1))**2)
    best_l2 = l2_obj
    t0 = time.monotonic()
    
    eta0 = 0.1
    patience = 0
    max_patience = 500
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        if patience > max_patience:
            break
        
        iter_count += 1
        eta = eta0 / (np.sqrt(iter_count) + 1.0)
        
        grad = l2_gradient(a)
        
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
        log_a -= log_a.max()  # for numerical stability
        a_new = np.exp(log_a)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Check improvement
        h_new = fast_convolve(a_new)
        l2_new = np.sum((h_new - 1.0/(2*n-1))**2)
        max_new = h_new.max()
        obj_new = 2.0 * n * max_new
        
        if obj_new < best_val - 1e-12:
            best_val = obj_new
            best_a = a_new.copy()
            patience = 0
        elif l2_new < best_l2 - 1e-12:
            best_l2 = l2_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    L2-MD iter=%d (%.1fs): obj=%.6f, l2=%.8f" % (iter_count, elapsed, best_val, best_l2))
    
    if verbose:
        elapsed = time.monotonic() - t0
        print("    L2-MD done (%.1fs): obj=%.6f, iters=%d" % (elapsed, best_val, iter_count))
    return best_a, best_val


def smoothed_max_anneal(a0, time_limit, verbose=True):
    """
    Use log-sum-exp with annealing temperature as a bridge between L2 and L-inf.
    Minimize T * log(sum(exp(h_k / T))) which approximates max(h) as T->0.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()
    
    # Temperature schedule: start high (similar to L2), end low (similar to L-inf)
    n_stages = 8
    temps = np.exp(np.linspace(np.log(5.0), np.log(0.05), n_stages))
    
    for stage, T in enumerate(temps):
        stage_time = time_limit / n_stages
        stage_t0 = time.monotonic()
        eta = 0.05 / (T + 0.1)
        iter_count = 0
        
        while time.monotonic() - stage_t0 < stage_time:
            iter_count += 1
            
            h = fast_convolve(a)
            h_shifted = h - h.max()  # for numerical stability
            log_sum_exp = h.max() + T * np.log(np.exp(h_shifted / T).sum())
            
            # Gradient of T * log(sum(exp(h/T))) = sum(h_k / sum(exp(h/T))) * grad(h_k)
            # = sum_k softmax(h/T)_k * grad(h_k)
            weights = np.exp(h_shifted / T)
            weights /= weights.sum()
            
            # Gradient: sum_k weights_k * 2 * (correlation with a shifted)
            # grad[i] = 2 * sum_k weights_k * 2 * a[k-i] = 4 * sum_k weights_k * a[k-i]
            grad = 4.0 * np.correlate(weights, a, mode='full')[:n]
            
            # Multiplicative update
            eta_t = eta / (np.sqrt(iter_count) + 1.0)
            log_a = np.log(np.maximum(a, 1e-30)) - eta_t * grad
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            
            h_new = fast_convolve(a_new)
            obj_new = 2.0 * n * h_new.max()
            if obj_new < best_val - 1e-12:
                best_val = obj_new
                best_a = a_new.copy()
            
            a = a_new
        
        if verbose:
            elapsed = time.monotonic() - t0
            print("    SmoothMax stage %d/%d (T=%.4f, %.1fs): obj=%.6f" % (stage+1, n_stages, T, elapsed, best_val))
    
    if verbose:
        elapsed = time.monotonic() - t0
        print("    SmoothMax done (%.1fs): obj=%.6f" % (elapsed, best_val))
    return best_a, best_val


def build_constraint_matrix(y, constraint_list, n):
    """Build the constraint matrix for the LP."""
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2500, add_per_iter=30, verbose=True):
    """
    LP-based Frank-Wolfe with dual-informed active set management.
    """
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac_y = fast_convolve(y)
    best_m = ac_y.max()
    t0 = time.monotonic()
    
    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 600
    iter_count = 0
    stagnation_count = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if stagnation_count > 5:
                if verbose:
                    print("    LP-FW: persistent stagnation, stopping")
                break
            # Perturb and restart active set
            ac_curr = fast_convolve(y)
            n_pert = max(30, len(active_set) // 4)
            rng = np.random.default_rng(hash(str(stagnation_count)))
            keep = list(rng.choice(len(active_set), size=len(active_set) - n_pert, replace=False))
            new_adds = np.argsort(ac_curr)[::-1][:n_pert * 3]
            new_set = set([active_set[i] for i in keep])
            for k in new_adds:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]
            no_improve = 0
            stagnation_count += 1
            continue
        
        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            stagnation_count = 0
        else:
            no_improve += 1
        
        # Manage active set size
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            # Keep constraints closest to being violated
            keep_idx = np.argsort(margins)[:int(max_K * 0.8)]
            active_set_base = [active_set[i] for i in keep_idx]
            # Always keep top violations
            violations = np.where(ac > M_curr - 1e-6)[0]
            for v in violations[:50]:
                vi = int(v)
                if vi not in active_set_base:
                    active_set_base.append(vi)
            active_set = sorted(set(active_set_base))[:max_K]
        
        K = len(active_set)
        if K == 0:
            break
        
        coeffs = build_constraint_matrix(y, active_set, n)
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(100, K // 3)]
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Dual-informed active set update
        # Get dual values for the inequality constraints
        try:
            # In cvxpy, dual values for constraints are in constraint.dual_value
            dual_vals = np.zeros(K)
            for ii in range(min(K, len(constraints) - 1)):
                dv = constraints[ii + 1].dual_value
                if dv is not None and np.isfinite(dv):
                    dual_vals[ii] = max(0.0, float(dv))
        except Exception:
            dual_vals = np.abs(dual_vals)  # keep as zeros
        
        # Prune active set: remove constraints with very small dual values
        if K > initial_K * 2:
            dual_sorted = np.sort(dual_vals)[::-1]
            threshold = dual_sorted[min(K-1, int(K*0.3))] * 0.01  # 1% of the 30th percentile
            keep_mask = dual_vals > max(threshold, 1e-8)
            # Always keep top M_curr constraints
            top_mask = np.array([ac[k] > M_curr - 1e-6 for k in active_set])
            keep_mask = keep_mask | top_mask
            if keep_mask.sum() < K:
                active_set = [active_set[i] for i in range(K) if keep_mask[i]]
        
        # Add new violations
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]
        
        all_viol = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(all_viol) > 0:
            viol_amt = conv_ys[all_viol] - M_curr
            top_v = all_viol[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]
        
        # Line search
        best_alpha = 1.0 / (iter_count + 2.0)
        best_trial_M = fast_convolve(y).max()
        
        # Geometric line search
        alphas = []
        base_alpha = min(1.0, 2.0 / (np.sqrt(iter_count) + 1.0))
        for i in range(8):
            alphas.append(base_alpha * (0.7 ** i))
        alphas = sorted(set([max(0.01, a) for a in alphas]), reverse=True)
        
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        y = (1.0 - best_alpha) * y + best_alpha * g_dir
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = best_a.copy()
        else:
            y = y / s_y
        
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=%d (%.1fs): J=%.6f, |S|=%d" % (iter_count, elapsed, 2.0*n*best_m, len(active_set)))
    
    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done (%.1fs): J=%.6f, iters=%d" % (elapsed, 2.0*n*best_m, iter_count))
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=77, verbose=True):
    """
    Gradient-guided coordinate descent.
    Move mass from high-peak-contributing positions to low-impact positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 400
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        total_iters += 1
        ac = fast_convolve(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        # For each top peak, try to reduce it by moving mass
        for ki in range(min(8, len(top_k))):
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(top_k[ki])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            # Find the source that most contributes to this peak
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find destination: minimize impact on all top peaks
            # Score each position by how much it would affect the peaks
            scores = np.zeros(n)
            for k2 in top_k[:min(10, len(top_k))]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
            
            # Prefer positions with low score (low impact on peaks)
            # and that are not the source
            scores[src] = np.inf
            dst = int(np.argmin(scores))
            
            # Try different fractions
            for frac in [0.01, 0.03, 0.05, 0.02, 0.005, 0.08]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            
            if improved:
                break  # Restart the top_k loop with new a
        
        if not improved:
            stall += 1
            # Random perturbations to escape
            if stall % 50 == 0:
                n_pert = min(n // 10, 50)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.003, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                    M_new = fast_convolve(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and total_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    Coord iter=%d (%.1fs): J=%.6f, stall=%d" % (total_iters, elapsed, 2.0*n*best_m, stall))
    
    if verbose:
        elapsed = time.monotonic() - t0
        print("    Coord done (%.1fs): J=%.6f, iters=%d" % (elapsed, 2.0*n*best_m, total_iters))
    return best_a, best_m


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update the best sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [%s]: %.6f (n=%d, t=%.1fs)" % (label, sc, len(a), elapsed))
    return best_seq, best_score
