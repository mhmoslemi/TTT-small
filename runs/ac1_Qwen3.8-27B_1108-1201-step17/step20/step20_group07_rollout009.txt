def fast_conv(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)


def exact_subgradient(a, weights):

    Compute exact subgradient of sum_k weights[k] * (a*a)[k] w.r.t. a.
    Uses: grad[j] = a[j]*S[j] + (W_cross_A)[j]

    W_pad[:size] = W
    cumW = np.cumsum(W_pad)
    S = np.zeros(n)
    for j in range(n):
        lo = j
        hi = min(j + n, size)
        if hi > lo:
            S[j] = cumW[hi - 1] - (cumW[lo - 1] if lo > 0 else 0)

    grad1 = a * S

    # (W_cross_A)[j] = sum_{m=0}^{n-1} W[j+m] * a[m]
    fft_size = 1 << (size + n - 1 + 1).bit_length()
    W_p = np.zeros(fft_size)
    W_p[:size] = W
    A_p = np.zeros(fft_size)
    A_p[:n] = a
    Wf = np.fft.rfft(W_p)
    Af = np.fft.rfft(A_p)
    cross = np.fft.irfft(Wf * np.conj(Af), fft_size)[:n]

    return grad1 + cross


def subgradient_opt(a0, time_limit, verbose=True):
    """Subgradient with exact gradients, adaptive beta, momentum."""
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_m = fast_conv(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)
    stall = 0
    t = 0
    beta = 20.0
    momentum = 0.85
    v = np.zeros(n)
    top_k = max(50, int(n * 0.03))

    while time.monotonic() - t0 < time_limit:
        t += 1
        c = fast_conv(x)
        c_max = c.max()

        if c_max < best_m - 1e-14:
            best_m = c_max
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Compute weights for log-sum-exp smoothing
        idx_top = np.argsort(c)[::-1][:top_k]
        c_top = c[idx_top]
        beta_eff = min(beta, 50.0 / (c_max + 1e-10))
        shifted = beta_eff * (c_top - c_max)
        shifted = np.clip(shifted, -50, 50)
        exp_w = np.exp(shifted)
        p = exp_w / exp_w.sum()

        # Build weight vector
        weights = np.zeros(2 * n - 1)
        weights[idx_top] = p

        # Exact subgradient
        grad = exact_subgradient(x, weights)
        grad /= beta_eff

        # Adaptive step size
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        eta = min(0.3 / gnorm, 0.05)
        eta = max(eta, 1e-7)

        # Heavy-ball momentum
        v = momentum * v - eta * grad
        x_new = simplex_project(x + v)

        # Accept if improvement
        c_new = fast_conv(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-15:
            x = x_new
        elif m_new < c_max + 0.02 * best_m:
            # Allow small increase (annealing)
            if rng.random() < max(0.05, 0.5 * (1 - stall / 2000)):
                x = x_new
        else:
            x = x  # Revert
            v *= 0.5

        # Adaptive parameters
        if t % 30 == 0:
            beta = min(beta * 1.01, 300.0)
            momentum = 0.85 + 0.08 * min(1.0, t / 5000)

        # Restart if stuck
        if stall > 800:
            stall = 0
            noise = 0.015 / (1.0 + 0.1 * (t // 800))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            m_r = fast_conv(x_r).max()
            x = x_r
            v = np.zeros(n)
            beta = max(10.0, beta * 0.7)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    SubGrad restart t={} ({:.0f}s): J={:.6f}".format(t, elapsed, J))

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SubGrad t={} ({:.0f}s): J={:.6f} stall={}".format(t, elapsed, J, stall))

    return best_a, best_m


def frank_wolfe_lp(a0, time_limit, solver, verbose=True):
    """Frank-Wolfe with LP direction finding."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    t = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 600:
        t += 1
        c = fast_conv(a)
        M = c.max()

        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Active set: top peaks
        K = min(800, 2 * n - 1)
        if t < 20:
            K = min(200, 2 * n - 1)
        idx_active = np.argsort(c)[::-1][:K]

        # Build constraint matrix: (a*g)[k] = sum_i a[i]*g[k-i] <= M
        A_mat = np.zeros((K, n))
        for ii, k in enumerate(idx_active):
            k_i = int(k)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                A_mat[ii, idx] = a[k_i - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A_mat @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-10:
            continue
        g_dir = g_val / g_sum

        # Check for violations (peaks not in active set that would be violated)
        conv_ag = fast_conv_combined(a, g_dir)
        violations = conv_ag > M + 1e-5
        if violations.sum() > K // 4:
            # Too many violations, skip this step
            no_improve += 1
            continue

        # Line search
        best_alpha = 1.0 / (t + 2.0)
        best_trial_M = float('inf')
        for alpha in [best_alpha, min(1.0, best_alpha * 2), 0.5 / (t + 1)]:
            alpha = min(1.0, max(0.01, alpha))
            trial = (1.0 - alpha) * a + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-10:
                continue
            trial = trial / st
            M_trial = fast_conv(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        a_new = (1.0 - best_alpha) * a + best_alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        st = a_new.sum()
        if st > 1e-10:
            a_new = a_new / st
        a = a_new

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    FW iter={} ({:.0f}s): J={:.6f} K={}".format(t, elapsed, J, K))

    return best_a, best_m


def fast_conv_combined(a, b):
    """Compute convolution of a and b."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def peak_shaving_v2(a0, time_limit, verbose=True):
    """Improved peak shaving with pressure-aware moves and local search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    top_k = 30

    while time.monotonic() - t0 < time_limit and stall < 800:
        c = fast_conv(a)
        c_max = c.max()
        if c_max < best_m - 1e-14:
            best_m = c_max
            best_a = a.copy()
            stall = 0

        # Identify top peaks
        peak_idx = np.argsort(c)[::-1][:top_k]

        # Compute pressure map
        pressure = np.zeros(n)
        for k_i in peak_idx[:15]:
            k_int = int(k_i)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                pressure[idx] += a[k_int - idx] * a[idx]

        improved = False

        # For each top peak, find worst contributor pair and try to fix
        for k_i in peak_idx[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k_i)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]

            # Try top-3 contributor pairs
            top_contrib = np.argsort(contrib)[::-1][:3]
            for tc in top_contrib:
                src = int(idx[tc])
                partner = k_int - src
                if partner < 0 or partner >= n:
                    continue

                reduce_at = src if a[src] >= a[partner] else partner
                if a[reduce_at] < 1e-12:
                    continue

                # Find destination: low pressure, far from peaks
                safe = pressure < np.percentile(pressure, 40)
                if safe.sum() < 100:
                    safe = pressure < np.percentile(pressure, 70)
                if safe.sum() < 50:
                    safe = np.ones(n, dtype=bool)
                safe[reduce_at] = False

                if not safe.any():
                    continue

                # Pick best destination (lowest pressure)
                dst_candidates = np.where(safe)[0]
                dst_scores = pressure[dst_candidates]
                best_dst = dst_candidates[np.argmin(dst_scores)]

                # Try moving mass
                for frac in [0.01, 0.03, 0.06, 0.1, 0.15]:
                    amt = a[reduce_at] * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_at] -= amt
                    a_trial[best_dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-10:
                        continue
                    a_trial = a_trial / st
                    m_trial = fast_conv(a_trial).max()
                    if m_trial < best_m - 1e-15:
                        a = a_trial
                        improved = True
                        break

                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation / annealing
            if stall % 30 == 0:
                scale = min(0.05, 0.005 * (stall / 30))
                n_pert = max(10, n // 50)
                positions = rng.choice(n, size=n_pert, replace=False)
                a_pert = a.copy()
                for pos in positions:
                    if a_pert[pos] > 1e-10:
                        a_pert[pos] *= (1 + rng.normal(0, scale))
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                m_pert = fast_conv(a_pert).max()
                if m_pert < best_m:
                    best_m = m_pert
                    best_a = a_pert.copy()
                    a = a_pert
                    stall = 0
                elif rng.random() < 0.1:
                    a = a_pert
            if stall > 400:
                # Bigger restart
                a = best_a + rng.normal(0, 0.02, n) * best_a
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                v = np.zeros(n)
                stall = 200

            if verbose and stall % 200 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    PSv2 ({:.0f}s): J={:.6f} stall={}".format(elapsed, J, stall))

    return best_a, best_m


def local_search(a0, time_limit, window=50, verbose=True):
    """Try all pairwise swaps within a window around peak contributors."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        c = fast_conv(a)
        c_max = c.max()
        if c_max < best_m - 1e-14:
            best_m = c_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        peak_idx = np.argsort(c)[::-1][:15]
        improved = False

        for k_i in peak_idx[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k_i)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            src = int(idx[np.argmax(contrib)])

            # Try moving a small amount from src to nearby positions
            lo = max(0, src - window)
            hi = min(n - 1, src + window)
            candidates = np.arange(lo, hi + 1)
            candidates = candidates[candidates != src]

            # Sort by current pressure (low = good destination)
            dst_pressure = np.array([
                a[d] * a.get(k_int - d, 0) if 0 <= k_int - d < n else 0
                for d in candidates
            ])
            # Simpler: just try a few
            for frac in [0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-13:
                    break
                best_dst = -1
                best_m_trial = best_m
                for d in candidates[np.argsort(a[candidates])[:20]]:
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[int(d)] += amt
                    a_t = np.maximum(a_t, 0)
                    st = a_t.sum()
                    if st < 1e-10:
                        continue
                    a_t = a_t / st
                    m_t = fast_conv(a_t).max()
                    if m_t < best_m_trial:
                        best_m_trial = m_t
                        best_dst = int(d)
                if best_dst >= 0:
                    a[src] -= amt
                    a[best_dst] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 100 == 0:
                a = best_a + rng.normal(0, 0.01, n) * best_a
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                if stall > 300:
                    stall = 100  # Reset partially

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def diversity_init(n, rng):
    """Generate diverse asymmetric initializations."""
    x = np.linspace(0, 1, n)
    candidates = []

    # Power law from left
    for alpha in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]:
        for beta in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]:
            a = (x + 0.005) ** alpha * (1.0 - x + 0.005) ** beta
            candidates.append(a)

    # Sine modulation
    for freq in [1, 2, 3, 5]:
        for phase in [0, np.pi / 4, np.pi / 2]:
            a = 1.0 + 0.5 * np.sin(freq * np.pi * x + phase)
            a = np.maximum(a, 0.1)
            candidates.append(a)

    # Gaussian mixtures (asymmetric)
    for c1 in [0.15, 0.25, 0.35, 0.5]:
        for c2 in [0.5, 0.65, 0.75, 0.85]:
            for w in [0.5, 1.0, 2.0]:
                a = np.exp(-((x - c1) ** 2) / 0.02) + w * np.exp(-((x - c2) ** 2) / 0.01)
                candidates.append(a)

    # Random smooth
    for _ in range(20):
        a = np.cumsum(rng.random(n) ** rng.uniform(0.5, 3.0))
        a = a / a[-1]
        candidates.append(a)

    # Edge-concentrated (key for breaking symmetry)
    for eps in [0.01, 0.05, 0.1]:
        a = np.ones(n) * eps + (1 - eps) * np.exp(-x * 3)
        candidates.append(a)
        a2 = np.ones(n) * eps + (1 - eps) * np.exp(-(1 - x) * 3)
        candidates.append(a2)

    best_a = None
    best_m = float('inf')
    for a_raw in candidates:
        a = norm_sum1(a_raw)
        m = fast_conv(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {:.6f} (n={}, t={:.0f}s)".format(
            label, sc, len(a), elapsed))
    return best_seq, best_score

        t7a = min(remaining * 0.3, 120.0)
        if t7a > 20:
            a_out, _ = peak_shaving_v2(best_seq, t7a, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fps", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        t7b = min(remaining * 0.3, 120.0)
        if t7b > 20:
            a_out, _ = subgradient_opt(best_seq, t7b, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fsg", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        t7c = min(remaining * 0.3, 100.0)
        if t7c > 20 and n_best <= 8000:
            a_out, _ = frank_wolfe_lp(best_seq, t7c, solver, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fwf", t_global)

    # Phase 8: Final micro-peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t8 = min(remaining * 0.6, 80.0)
        a_out, _ = peak_shaving_v2(best_seq, t8, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "final_ps", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.6f}, n={}, time={:.0f}s".format(best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 7000
        x = np.linspace(0, 1, n_fb)
        a_raw = (x + 0.01) ** 1.0 * (1.0 - x + 0.01) ** 0.5
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
