def npow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1)
    arr[~np.isfinite(arr)] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n) / float(n)
    return arr / s


def fconv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    N = npow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c[np.isnan(c) | (c <= 0.0)] = 0.0
    return c


def fconv2(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([])
    if b.size != n:
        b = np.resize(b, n)
    size = 2 * n - 1
    N = npow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c[np.isnan(c) | (c <= 0.0)] = 0.0
    return c


def cross_p_a(p, a):
    """Gradient-like: g_i = sum_k p_k * a_{k-i} for valid i."""
    p = np.asarray(p, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([])
    total = 3 * n - 2
    N = npow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    out = conv[n - 1:2 * n - 1][::-1]
    out[~np.isfinite(out)] = 0.0
    return out

def interp_seq(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    new_n = int(max(2, new_n))

    y = np.interp(new_x, old_x, a)
    y[~np.isfinite(y)] = 0.0
    np.maximum(y, 0.0, out=y)
    return norm1(y)


def lse_obj(a, beta, active_idx):
    """Log-sum-exp smoothed max over active set, with gradient."""
    c = fconv(a)
    size = c.size
    if active_idx is None or active_idx.size == 0:
        active_idx = np.arange(size)
    cv = c[active_idx]
    if cv.size == 0:
        return 0.0, np.zeros(a.size)
    m = float(np.max(cv))
    shifted = beta * (cv - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0 or not math.isfinite(se):
        return float(np.max(c) * 2 * a.size), np.zeros(a.size)
    p = e / se
    pf = np.zeros(size, dtype=np.float64)
    pf[active_idx] = p
    grad = 2.0 * cross_p_a(pf, a)
    grad[~np.isfinite(grad)] = 0.0
    f = m + math.log(se) / beta
    return f, grad

        # grad_v = a_loc * (grad_a - dot(a_loc, grad_a))
        mean_g = float(np.dot(a_loc, ga))
        gv = a_loc * (ga - mean_g)
        gv[0] = 0.0
        if not np.all(np.isfinite(gv)):
            gv = np.zeros(n)
        if not math.isfinite(f):
            f = 1e100
        return f, gv

    options = {"maxfun": int(maxfun), "maxiter": int(maxfun + 20),
               "ftol": 1e-14, "gtol": 1e-8, "disp": False}
    try:
        res = scipy.optimize.minimize(obj_v, v0, jac=True,
                                       method="L-BFGS-B", options=options)
        vx = np.asarray(res.x, dtype=np.float64)
        vx[~np.isfinite(vx)] = 0.0
        vx[0] = 0.0
        e = np.exp(vx)
        e = np.where(np.isfinite(e), e, 1e-300)
        a_out = e / e.sum()
        return norm1(a_out)
    except Exception:
        return a


def lp_fw_direction(f, active_idx, time_limit=5.0):
    """Solve LP: min t s.t. (f*g)[k] <= t, g>=0, sum(g)=1. Return g."""
    if not SCIPY_OK or not hgshs_ok():
        return None
    f = norm1(f)
    n = f.size
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K == 0 or K > 2000:
        return None

    row_parts = []
    col_parts = []
    val_parts = []
    for r, k in enumerate(active_idx):
        k = int(k)
        s = max(0, k - n + 1)
        e = min(n, k + 1)
        if s >= e:
            continue
        cols = np.arange(s, e, dtype=np.int32)
        row_parts.append(np.full(cols.size, r, dtype=np.int32))
        col_parts.append(cols)
        # (f*g)[k] = sum_{i} f[k-i] * g[i] for valid i
        # f[k-i] where i ranges from s to e-1
        # k-i ranges from k-(e-1) to k-s
        vals = f[k - cols]
        val_parts.append(vals)

    if not row_parts:
        return None

    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))

    # Variables: g_0,...,g_{n-1}, t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t

    # Constraints: A@g - t <= 0
    neg_t_col = scipy.sparse.csc_matrix(-np.ones((K, 1)))
    A_ub = scipy.sparse.hstack((A, neg_t_col), format="csr")
    b_ub = np.zeros(K)

    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.ones(1)

    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                                      A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                                      method="HiGHS",
                                      options={"disp": False, "time_limit": float(time_limit)})
        if res is None or not getattr(res, "success", False):
            return None
        g = np.asarray(res.x[:n], dtype=np.float64)
        g[~np.isfinite(g) | (g < 0.0)] = 0.0
        gs = float(g.sum())
        if gs < 1e-15:
            return None
        return g / gs
    except Exception:
        return None


def game_fw_direction(f, active_idx, iters=30, seed=42):
    """Approximate FW direction via multiplicative weights (zero-sum game)."""
    f = norm1(f)
    n = f.size
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K == 0:
        return np.ones(n) / float(n)

    rng = np.random.default_rng(seed)
    logits = np.zeros(K)
    logits += rng.normal(0, 0.1, K)

    g = np.ones(n) / float(n)
    for it in range(iters):
        # Update g: minimize max pressure
        lam = np.exp(logits - np.max(logits))
        lam = lam / lam.sum()
        pf = np.zeros(2 * f.size - 1)
        pf[active_idx] = lam
        pressure = cross_p_a(pf, f)
        pressure = np.maximum(pressure, 1e-30)
        pmin = float(np.min(pressure))
        pmax = float(np.max(pressure))
        ptp = pmax - pmin
        if ptp < 1e-30:
            g = np.ones(n) / float(n)
        else:
            scaled = (pressure - pmin) / ptp
            w = np.exp(-6.0 * scaled)
            w = np.where(np.isfinite(w), w, 0.0)
            w += 1e-10
            g = w / w.sum()
            g = 0.98 * g + 0.02 * (np.ones(n) / float(n))
            g /= g.sum()

        # Update lambdas: concentrate on high (f*g) values
        h = fconv2(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmax = float(np.max(ha))
        hmin = float(np.min(ha))
        htp = hmax - hmin
        if htp > 1e-30:
            eta = 10.0 / htp
            logits += eta * (ha - np.mean(ha))

    return g


def fw_search(a0, budget, active_max=2000):
    """Main Frank-Wolfe loop with LP/game directions."""
    t0 = time.monotonic()
    a = norm1(a0)
    n = a.size
    c = fconv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    active = topk_idx(c, active_max)
    no_improve = 0
    t = 0
    seed = 12345

    while time.monotonic() - t0 < budget - 5 and no_improve < 60:
        t += 1
        time_left = budget - (time.monotonic() - t0)
        if time_left < 20:
            break

        # Get direction
        g = None
        use_lp = (hgshs_ok() and time_left > 80 and active.size <= 1500)
        if use_lp and (t % 3 == 1):
            g = lp_fw_direction(a, active, time_limit=min(5.0, time_left * 0.1))
        if g is None:
            g = game_fw_direction(a, active, iters=25, seed=seed + t)

        if g is None:
            break

        # Line search
        cg = fconv(g)
        fg = fconv2(a, g)
        best_alpha = 0.0
        best_m = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            one = 1.0 - alpha
            val = (one * one) * c + 2.0 * alpha * one * fg + (alpha * alpha) * cg
            m = float(np.max(val))
            if m < best_m:
                best_m = m
                best_alpha = alpha

        if best_alpha <= 0.0:
            no_improve += 1
            if no_improve % 10 == 0:
                rng = np.random.default_rng(999 + t)
                pert = rng.normal(0, 0.02, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                c = fconv(a)
                M = float(np.max(c))
                active = topk_idx(c, active_max)
                no_improve = 0
            continue

        one = 1.0 - best_alpha
        c = (one * one) * c + 2.0 * best_alpha * one * fg + (best_alpha * best_alpha) * cg
        c = np.where(np.isfinite(c) & (c > 0), c, 0.0)
        a = one * a + best_alpha * g
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        M = float(np.max(c))
        no_improve = 0
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
        active = topk_idx(c, active_max)
        if t % 10 == 0:
            print("    FW t=%d M=%.8f score=%.6f alpha=%.2f" % (t, M, 2.0 * n * M, best_alpha))

    return best_a


def apply_move(c, a, s, d, tmove):
    """Rank-1 update of autocorrelation when moving tmove from s to d."""
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64).copy()
    n = a.size
    c[d:d + n] += 2.0 * tmove * a
    c[s:s + n] -= 2.0 * tmove * a
    t2 = tmove * tmove
    if 2 * d < c.size:
        c[2 * d] += t2
    if 2 * s < c.size:
        c[2 * s] += t2
    if s + d < c.size:
        c[s + d] -= 2.0 * t2
    c[~np.isfinite(c)] = 0.0
    return c


def coord_polish(a0, budget, verbose=True):
    """Intensive coordinate descent with broad move exploration."""
    t0 = time.monotonic()
    a = norm1(a0)
    n = a.size
    c = fconv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    stall = 0
    rnd = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < budget - 3 and stall < 200:
        rnd += 1
        if n <= 1:
            break

        # Compute pressure from top peaks
        top_k = min(30, 2 * n - 1)
        top = topk_idx(c, top_k)
        p_top = np.zeros(2 * n - 1)
        p_top[top] = np.ones(top_k) / float(top_k)
        pressure = cross_p_a(p_top, a)
        pressure = np.maximum(pressure, 0.0)
        pressure = np.where(np.isfinite(pressure), pressure, 0.0)

        # Find candidate source and destination positions
        # Sources: high pressure + high value (moving from here reduces peak)
        # Destinations: low pressure (moving here doesn't increase peak much)
        src_score = pressure * (1.0 + n * a)
        dst_score = pressure + 1e-4 * a

        n_src = min(12, n)
        n_dst = min(16, n)
        src_cands = np.argpartition(-src_score, n_src - 1)[:n_src]
        dst_cands = np.argpartition(dst_score, n_dst - 1)[:n_dst]

        best_move = None
        best_m = M

        # Try structured moves
        for s in src_cands:
            s = int(s)
            if a[s] < 1e-12:
                continue
            for d in dst_cands:
                d = int(d)
                if d == s:
                    continue
                for frac in (0.005, 0.01, 0.02, 0.03, 0.05, 0.08):
                    tmove = a[s] * frac
                    tmove = min(tmove, a[s] * 0.95)
                    if tmove <= 1e-16:
                        continue
                    c_new = apply_move(c, a, s, d, tmove)
                    m_new = float(np.max(c_new))
                    if m_new < best_m:
                        best_m = m_new
                        best_move = (s, d, tmove, m_new)

        # Try random moves
        n_rand = 50
        for _ in range(n_rand):
            s = int(rng.integers(n))
            d = int(rng.integers(n))
            if s == d or a[s] < 1e-12:
                continue
            frac = float(rng.uniform(0.003, 0.06))
            tmove = a[s] * frac
            tmove = min(tmove, a[s] * 0.9)
            if tmove <= 1e-16:
                continue
            c_new = apply_move(c, a, s, d, tmove)
            m_new = float(np.max(c_new))
            if m_new < best_m:
                best_m = m_new
                best_move = (s, d, tmove, m_new)

        if best_move is not None:
            s, d, tmove, m_new = best_move
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = fconv(a)
            M = float(np.max(c))
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and (rnd <= 5 or rnd % 50 == 0):
                print("    Coord rnd=%d M=%.8f score=%.6f" % (rnd, M, 2.0 * n * M))
        else:
            stall += 1
            if stall % 15 == 0:
                # Perturbation escape
                rng2 = np.random.default_rng(777 + rnd)
                scale = min(0.03, 0.01 * math.sqrt(stall / 15.0))
                pert = rng2.normal(0, scale, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                c = fconv(a)
                M = float(np.max(c))
                stall = 0

    return best_a


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    total_budget = max(60.0, min(float(budget_s) - 80.0, 1720.0))
    print("Start: budget=%.0fs" % total_budget)

    # Initialize from parent
    try:
        parent = height_sequence_1
    except Exception:
        parent = None

    seq_best = None
    if parent is not None:
        try:
            p = np.asarray(parent, dtype=np.float64)
            p[~np.isfinite(p)] = 0.0
            np.clip(p, 0.0, 1000.0, out=p)
            if p.ndim != 1:
                p = p.ravel()
            if p.size > 0 and float(p.sum()) > 1e-10:
                seq_best = norm1(p)
        except Exception:
            seq_best = None

    if seq_best is None:
        n_init = 5000
        x = np.linspace(0.0, 1.0, n_init)
        seq_best = norm1(np.sin(np.pi * x) ** 2)

    best_score = safe_eval(seq_best)
    if not math.isfinite(best_score):
        best_score = 2.0 * seq_best.size * float(np.max(fconv(seq_best)))
    print("Init: n=%d score=%.8f" % (seq_best.size, best_score))

    # Work sequence (compressed if too long)
    if seq_best.size > 9000:
        work = interp_seq(seq_best, 8000)
    else:
        work = seq_best.copy()

    seq_best, best_score = try_update(seq_best, best_score, work, "init_work", t_start)

    n_work = work.size
    print("Work: n=%d" % n_work)

    # === Phase 1: L-BFGS smoothed optimization ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if SCIPY_OK and time_left > 400:
        phase_budget = min(time_left * 0.35, 600.0)
        print("\n=== Phase 1: L-BFGS (%.0fs) ===" % phase_budget)
        active = topk_idx(fconv(work), 300)
        a_lb = run_lbfegs(work, 50.0, active, maxfun=400)
        seq_best, best_score = try_update(seq_best, best_score, a_lb, "lbfegs", t_start)
        if seq_best.size == n_work:
            work = seq_best.copy()
        else:
            work = interp_seq(seq_best, n_work)

    # === Phase 2: Frank-Wolfe ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 300:
        phase_budget = min(time_left * 0.35, 600.0)
        print("\n=== Phase 2: Frank-Wolfe (%.0fs) ===" % phase_budget)
        a_fw = fw_search(work, phase_budget, active_max=1500)
        seq_best, best_score = try_update(seq_best, best_score, a_fw, "fw", t_start)
        if seq_best.size == n_work:
            work = seq_best.copy()
        else:
            work = interp_seq(seq_best, n_work)

    # === Phase 3: Intensive coordinate polish ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 200:
        phase_budget = min(time_left * 0.30, 400.0)
        print("\n=== Phase 3: Coordinate polish (%.0fs) ===" % phase_budget)
        a_cp = coord_polish(work, phase_budget, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_cp, "coord1", t_start)
        if seq_best.size == n_work:
            work = seq_best.copy()
        else:
            work = interp_seq(seq_best, n_work)

    # === Phase 4: Length probes ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 300:
        print("\n=== Phase 4: Length probes (%.0fs) ===" % time_left)
        base_n = seq_best.size if seq_best.size <= 9000 else n_work
        probe_lens = []
        for delta in [300, -300, 600, -600, 1000, -1000]:
            ln = base_n + delta
            if 1000 <= ln <= 10000 and ln not in [p for p in probe_lens]:
                probe_lens.append(ln)

        for target_n in probe_lens[:4]:
            time_left = total_budget - (time.monotonic() - t_start)
            if time_left < 150:
                break
            print("  Probing n=%d (remaining %.0fs)" % (target_n, time_left))
            a_init = interp_seq(seq_best, target_n)
            # Quick FW
            b_fw = min(60.0, time_left * 0.4)
            a_try = fw_search(a_init, b_fw, active_max=1000)
            # Quick coord
            b_cp = min(40.0, time_left * 0.25)
            a_try = coord_polish(a_try, b_cp, verbose=False)
            seq_best, best_score = try_update(seq_best, best_score, a_try, "probe_%d" % target_n, t_start)

    # === Phase 5: Final polish on best ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 80:
        work_n = seq_best.size if seq_best.size <= 9000 else n_work
        work = seq_best.copy() if seq_best.size == work_n else interp_seq(seq_best, work_n)
        phase_budget = min(time_left * 0.7, 120.0)
        print("\n=== Phase 5: Final coord polish (%.0fs) ===" % phase_budget)
        a_final = coord_polish(work, phase_budget, verbose=False)
        seq_best, best_score = try_update(seq_best, best_score, a_final, "final_coord", t_start)

    # === Phase 6: Multi-restart perturbation ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 60 and seq_best.size <= 9000:
        print("\n=== Phase 6: Multi-restart (%.0fs) ===" % time_left)
        per_restart = min(30.0, (time_left - 10) / 4.0)
        n_base = seq_best.size
        for ri in range(4):
            time_left = total_budget - (time.monotonic() - t_start)
            if time_left < 40:
                break
            rng = np.random.default_rng(500 + ri)
            scale = 0.01 * (ri + 1)
            pert = rng.normal(0, scale, n_base) * seq_best
            a_pert = seq_best + pert
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / a_pert.sum()
            a_p = coord_polish(a_pert, min(per_restart, time_left * 0.6), verbose=False)
            seq_best, best_score = try_update(seq_best, best_score, a_p, "restart_%d" % ri, t_start)

    # Final validation
    final_val = safe_eval(seq_best)
    if math.isfinite(final_val):
        best_score = min(best_score, final_val)
    else:
        seq_best = norm1(np.sin(np.pi * np.linspace(0, 1, 5000)) ** 2)
        best_score = safe_eval(seq_best)

    arr = np.asarray(seq_best, dtype=np.float64)
    arr[~np.isfinite(arr)] = 0.0
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if arr.size == 0 or s < 0.01:
        arr = norm1(np.sin(np.pi * np.linspace(0, 1, 5000)) ** 2)
    else:
        arr = arr / s

    elapsed = time.monotonic() - t_start
    print("\nFINAL: score=%.8f n=%d elapsed=%.1fs" % (best_score, arr.size, elapsed))
    return [float(x) for x in arr]
