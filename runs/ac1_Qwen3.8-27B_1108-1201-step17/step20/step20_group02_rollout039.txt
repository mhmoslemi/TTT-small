    n = len(a)

    # Gradient is weighted sum of individual gradients

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1) / rho > 0
    rho_hat = np.max(rho[cond])
    theta = (cssv[rho_hat - 1] - 1) / rho_hat
    return np.maximum(v - theta, 0)


def smooth_max_gradient(a, T):
    """
    Compute the value and gradient of the smooth max of conv(a,a).
    Uses the softmax approximation: (1/T) log(sum exp(T * f_k))
    where f_k = (conv(a,a))_k.
    
    Returns (smooth_value, gradient_vector).
    """
    n = len(a)
    conv = fast_convolve(a)
    # Softmax weights (numerically stable)
    shifted = conv - conv.max()
    weights = np.exp(T * shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
        w_sum = 1.0
    else:
        weights /= w_sum
    
    smooth_val = conv.max() + np.log(w_sum) / T
    
    # Gradient: grad[j] = 2 * sum_k weights[k] * a[k-j] (for valid k-j)
    # = 2 * cross_correlation(a, weights)[j]
    # Computed as: 2 * (a_full * flip(weights))[n-1+j] (in the appropriate range)
    
    # FFT-based cross-correlation
    fft_size = 1 << (4 * n - 2).bit_length()  # enough for full correlation
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(weights, fft_size)
    # cross_corr[j] = sum_i a[i] * weights[i+j]
    # = (a * flip(weights))[j + (2n-2)]
    # flip(weights) has length 2n-1
    W_flip = np.fft.rfft(np.flip(weights), fft_size)
    corr = np.fft.irfft(A * W_flip, fft_size)
    # We want corr at indices (2n-2) + j for j=0..n-1
    grad = 2.0 * corr[(2 * n - 2):(3 * n - 2)]
    
    return smooth_val, grad


def smooth_adam_optimize(a0, time_limit, T_start=5.0, T_end=80.0, 
                         lr_start=0.1, lr_end=0.01, beta1=0.9, beta2=0.999,
                         verbose=True):
    """Optimize using smooth max + Adam with simplex projection."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_exact_val = fast_convolve(best_a).max()
    t0 = time.monotonic()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Interpolate T
        progress = min(1.0, t / max(1, int(time_limit // 0.1)))  # ramp over first portion
        # Actually, let's ramp over the whole time
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        T_current = T_start + (T_end - T_start) * frac
        lr_current = lr_start + (lr_end - lr_start) * frac
        
        # Compute gradient
        _, grad = smooth_max_gradient(a, T_current)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr_current * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_simplex(a_new)
        
        # Check improvement
        exact_val = fast_convolve(a_new).max()
        if exact_val < best_exact_val - 1e-14:
            best_exact_val = exact_val
            best_a = a_new.copy()
        a = a_new
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print("  SmoothAdam t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): "
                  + "T=" + str(round(T_current, 1)) + " J=" + str(2 * n * best_exact_val))
    
    if verbose:
        elapsed = time.monotonic() - t0
        print("  SmoothAdam done (" + str(round(elapsed, 1)) + "s): J=" + str(2 * n * best_exact_val))
    return best_a, best_exact_val

def fast_convolve(a):

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_hat = idx[-1] + 1
    theta = (cssv[rho_hat - 1] - 1.0) / rho_hat
    return np.maximum(v - theta, 0.0)


def smooth_max_gradient(a, T):
    n = len(a)
    conv = fast_convolve(a)
    cmax = conv.max()
    shifted = conv - cmax
    exp_part = np.exp(T * shifted)
    w_sum = exp_part.sum()
    if w_sum < 1e-300:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    else:
        weights = exp_part / w_sum

    smooth_val = cmax + np.log(max(w_sum, 1e-300)) / T

    ar = a[::-1]

    conv_ar_w = np.fft.irfft(AR * W, fft_size)
    grad = 2.0 * conv_ar_w[n - 1:2 * n - 1]
    return smooth_val, grad

        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1.0))
        T_current = T_start + (T_end - T_start) * (frac ** 0.7)
        lr_current = lr_start + (lr_end - lr_start) * frac

        _, grad = smooth_max_gradient(a, T_current)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr_current * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_simplex(a_new)

        exact_val = fast_convolve(a_new).max()
        if exact_val < best_val - 1e-14:
            best_val = exact_val
            best_a = a_new.copy()
            best_t = t
            stagnant = 0
        else:
            stagnant += 1

        a = a_new

        if stagnant > 5000:
            break

        if verbose and t % 1000 == 0:
            print("  SmoothAdam t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): "
                  + "T=" + str(round(T_current, 1)) + " J=" + str(2 * n * best_val))

    if verbose:
        elapsed = time.monotonic() - t0
        print("  SmoothAdam done (" + str(round(elapsed, 1)) + "s): J=" + str(2 * n * best_val)
              + ", iters=" + str(t))
    return best_a, best_val


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2500, add_per_iter=50, verbose=True):
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac_y = fast_convolve(y)
    best_m = ac_y.max()
    t0 = time.monotonic()

    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 500
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            break

        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep_idx = np.argsort(margins)[:int(max_K * 0.7)]
            active_set_base = [active_set[i] for i in keep_idx]
            violations = np.where(ac > M_curr - 1e-8)[0]
            for vv in violations[:100]:
                if int(vv) not in active_set_base:
                    active_set_base.append(int(vv))
            active_set = sorted(set(active_set_base))

        K = len(active_set)
        if K == 0:
            break
        if K > max_K:
            active_set = sorted(active_set[:max_K])
            K = max_K

        coeffs = build_constraint_matrix(y, active_set, n)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > 100:
                active_set = active_set[:max(50, len(active_set) // 3)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        fft_size = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]

        all_viol = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(all_viol) > 0:
            viol_amt = conv_ys[all_viol] - M_curr
            top_v = all_viol[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        best_alpha = 1.0 / (iter_count + 2.0)
        best_trial_M = fast_convolve(y).max()

        alphas = [1.0, 0.8, 0.6, 0.4, 0.25, 0.12, max(0.05, 1.5 / (iter_count + 1.0))]
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 1:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): "
                  + "J=" + str(2.0 * n * best_m) + ", |S|=" + str(len(active_set)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + str(2.0 * n * best_m)
              + ", iters=" + str(iter_count))
    return best_a, best_m


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def coordinate_polish(a0, time_limit, rng_seed=77, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        top_k = np.argsort(ac)[::-1][:20]
        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            candidates = np.where(scores < scores.max() * 0.3)[0]
            if len(candidates) < n // 5:
                candidates = np.arange(n)
            dst = int(candidates[np.argmin(scores[candidates])])

            if dst == src:
                continue

            for frac in [0.01, 0.005, 0.002, 0.02]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if not improved:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1

    if verbose:
        elapsed = time.monotonic() - t0
        print("    Coord polish done (" + str(round(elapsed, 1)) + "s): J=" + str(2 * n * best_m))
    return best_a, best_m

                return [p_norm], sc

        pass

        sc = evaluate_safe(s)

            best_sc = sc

        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
        seqs.append(s)

    print("Default init: best score=" + repr(best_sc))
    return seqs, best_sc
