"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Parametric warm-start with multiple shape families
2. Nesterov accelerated gradient with adaptive top-k smoothed max
3. Backtracking line search for step size control
4. Iterative peak cancellation for targeted reduction
5. Multi-length exploration with structured interpolation
6. Final pairwise coordinate descent polish
"""

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    if n <= 64:
        return np.convolve(a, a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    u = np.sort(z)[::-1]
    css = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (css - 1)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (css[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_smoothed_max_and_grad(y, beta, top_k=None):
    """Compute smoothed max of autocorrelation and its gradient."""
    n = len(y)
    c = fast_autocorr(y)
    m = len(c)
    if top_k is not None and top_k < m:
        # Get top-k indices
        # Use partition for speed
        if top_k < m // 4:
            top_idx = np.argpartition(c, -top_k)[-top_k:]
        else:
            top_idx = np.argsort(c)[::-1][:top_k]
        c_masked = np.full(m, -np.inf)
        c_masked[top_idx] = c[top_idx]
    else:
        c_masked = c.copy()
    
    c_max = c_masked.max()
    shifted = beta * (c_masked - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros(m)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    
    smoothed_val = c_max + np.log(total) / beta
    
    # Gradient: d/da[j] of smoothed_val = (1/beta) * sum_k p_k * d(c_k)/da[j]
    # d(c_k)/da[j] = 2*a[k-j] (with a[x]=0 outside [0,n-1])
    # So grad[j] = (2/beta) * sum_k p_k * a[k-j]
    # = (2/beta) * (cross_correlation of p and a at lag j)
    # Using FFT:
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(y, fft_size)
    P = np.fft.rfft(p, fft_size)
    # We need sum_k p[k] * a[k-j] for each j
    # This is the cross-correlation: (p * a_rev)[j + n - 1] where a_rev = a[::-1]
    # conv(p, a_rev)[m] = sum_k p[k] * a_rev[m-k] = sum_k p[k] * a[n-1-m+k]
    # We want: sum_k p[k] * a[k-j]. Set n-1-m+k = k-j => m = n-1+j
    a_rev = y[::-1]
    Conv = np.fft.irfft(P * np.fft.rfft(a_rev, fft_size), fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    
    return smoothed_val, grad, p, c

def parametric_init(n, family, params, rng):
    """Generate a parametric initial sequence."""
    x = np.linspace(0, 1, n)
    if family == 'power':
        alpha, beta = params
        a = (1 + alpha * x) ** beta
    elif family == 'exp':
        gamma = params[0]
        a = np.exp(gamma * (x - 1))  # decaying exponential
    elif family == 'bell':
        mu, sigma = params
        a = np.exp(-((x - mu) ** 2) / (2 * sigma ** 2)) + 0.01
    elif family == 'u_shape':
        alpha = params[0]
        a = 1 + alpha * (2 * x - 1) ** 2
    elif family == 'gauss_decay':
        mu, sigma, decay = params
        a = np.exp(-((x - mu) ** 2) / (2 * sigma ** 2)) * np.exp(decay * x) + 0.01
    elif family == 'pow_decay':
        p, q = params
        a = (1 - x) ** p * (0.1 + 0.9 * x ** q) + 0.01
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def nesterov_opt(a0, time_limit, beta_start=50.0, beta_end=3000.0,
                  beta_growth=1.006, top_k_frac=0.02, restart_every=1500,
                  verbose=True, seed=42, use_backtrack=True):
    """Nesterov accelerated projected gradient with adaptive top-k and optional backtracking."""
    n = len(a0)
    x = normalize(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(5, int(n * top_k_frac))
    eta = 0.5
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        val, grad, p, c = compute_smoothed_max_and_grad(y, beta, top_k)
        
        # Adaptive top-k: if top peak dominates, reduce k
        c_sorted = np.sort(c)[::-1]
        if len(c_sorted) > 10:
            ratio = c_sorted[0] / max(c_sorted[min(top_k, len(c_sorted)-1)], 1e-30)
            if ratio > 10:
                top_k = max(5, top_k // 2)
            elif ratio < 2:
                top_k = min(int(n * 0.05), top_k * 2)
        
        # Backtracking line search
        if use_backtrack:
            f_curr = val
            eta_try = eta
            for bt in range(10):
                z = y - eta_try * grad
                x_trial = simplex_project(z)
                val_trial, _, _, _ = compute_smoothed_max_and_grad(x_trial, beta, top_k)
                if val_trial < f_curr - 1e-10 * eta_try * np.sum(grad * grad):
                    break
                eta_try *= 0.5
            eta = eta_try
            x_new = x_trial
        else:
            # Fixed step with damping
            eta_fixed = min(0.5, 1.0 / max(1.0, beta * 2.0 * p.max()))
            z = y - eta_fixed * grad
            x_new = simplex_project(z)
        
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        if t % 30 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            eta = min(eta * 0.5, 0.5)
            # Structured restart
            noise_scale = 0.02 / (1.0 + 0.1 * (t // restart_every))
            # Sinusoidal modulation
            freq = rng.uniform(3, 12)
            phase = rng.uniform(0, 2 * np.pi)
            pert = noise_scale * np.sin(freq * np.arange(n) / n * 2 * np.pi + phase) * best_a
            # Anti-symmetric component
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            pert += noise_scale * 0.5 * anti * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + ", beta=" + str(round(beta, 1)) + ", top_k=" + str(top_k))
    
    return best_a, best_max_b

def peak_cancellation(a0, time_limit, verbose=True, seed=99):
    """Iteratively cancel the top peaks using gradient information."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 5000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Get top peaks
        n_top = min(10, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]
        
        # Compute combined gradient pressure
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            # d(c[k])/d(a[j]) = 2*a[k-j] (with bounds)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                grad[j] += 2.0 * a[k - j]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-30:
            break
        grad = grad / grad_norm
        
        # Try several step sizes
        improved = False
        for eta in [0.1, 0.05, 0.02, 0.01, 0.005]:
            a_trial = a - eta * np.dot(a, grad_norm) * grad  # scale by a to keep magnitude reasonable
            # Actually, let's use a simpler step
            a_trial = a - eta * a * grad / max(a.max(), 1e-15)
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-16:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try random pairwise moves
            for _ in range(20):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.001, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
        
        if verbose and no_improve % 1000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakCancel (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve))
    
    return best_a, best_m

def pairwise_cd(a0, time_limit, verbose=True, seed=123):
    """Fine-grained pairwise coordinate descent."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        c = fast_autocorr(a)
        # Focus on peaks above 90th percentile
        threshold = np.percentile(c, 90)
        peak_positions = np.where(c > threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.argsort(c)[::-1][:20]
        
        improved = False
        
        for k in peak_positions[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            
            top_pairs = np.argsort(products)[::-1][:3]
            for pi in top_pairs:
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n or partner == src_i:
                    continue
                
                larger_i = src_i if a[src_i] >= a[partner] else partner
                smaller_i = partner if a[src_i] >= a[partner] else src_i
                if a[larger_i] < 1e-14:
                    continue
                
                for frac in [0.001, 0.002, 0.005, 0.01]:
                    amt = a[larger_i] * frac
                    if amt < 1e-18:
                        continue
                    # Try several destinations: prefer low "pressure" positions
                    # Compute pressure at each position from this peak
                    pressure = np.zeros(n)
                    s2 = max(0, k_int - n + 1)
                    e2 = min(n - 1, k_int)
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k_int - idx2]
                    pressure[k_int - idx2] += a[idx2]
                    
                    # Best destinations: low pressure, not the source
                    candidates = np.argsort(pressure)
                    for dst in candidates[:5]:
                        dst = int(dst)
                        if dst == larger_i:
                            continue
                        a_trial = a.copy()
                        a_trial[larger_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall % 100 == 0:
                # Structured nudge
                eps = 0.005 / (1.0 + stall / 50.0)
                anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
                a_try = a + eps * anti * a
                a_try = np.maximum(a_try, 0.0)
                s_t = a_try.sum()
                if s_t > 1e-15:
                    a_try = a_try / s_t
                    M_try = fast_autocorr(a_try).max()
                    if M_try < best_m:
                        a = a_try
                        best_m = M_try
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PairCD (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent_score = evaluate_safe(p)
            if np.isfinite(parent_score):
                best_seq = normalize(p)
                best_score = parent_score
                print("Parent: n=" + str(len(p)) + ", score=" + str(round(parent_score, 6)))
    except Exception:
        pass
