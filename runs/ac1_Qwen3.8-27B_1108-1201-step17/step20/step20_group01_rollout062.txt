def _fast_conv(a):

    fft_n = 1 << (size - 1).bit_length()

def _normalize(a):

def _correct_fft_grad(a, wp):
    """Compute gradient of sum_k wp[k]*c[k] w.r.t. a, where c=conv(a,a).
    wp has length 2n-1. Returns gradient of length n."""
    n = len(a)
    c = 2 * n - 1
    w_rev = wp[::-1]
    req = 3 * n - 2
    fft_n = 1 << (req - 1).bit_length()
    A = np.fft.rfft(a, fft_n)

    T2 = conv_aw[:n]

def _gradient_top_k(a, K, soft_beta=0.0):
    """Compute gradient using top-K lags with optional soft weighting."""
    n = len(a)
    c = len(_fast_conv(a))
    cv = _fast_conv(a)
    K_use = min(K, c)
    idx = np.argsort(cv)[::-1][:K_use]
    top_vals = cv[idx]
    if soft_beta > 0:
        w = np.exp(soft_beta * (top_vals - top_vals[0]))
        w = w / w.sum()
    else:
        w = np.ones(K_use) / K_use
    wp = np.zeros(2 * n - 1)
    for i in range(K_use):
        wp[int(idx[i])] = w[i]
    return _correct_fft_grad(a, wp)


def _mirror_descent(a0, time_limit, K=100, lr0=0.02, soft_beta=30.0,
                    n_restarts=3, verbose=True):
    """Adaptive mirror descent with backtracking and restarts."""
    n = len(a0)
    best_a = a0.copy()
    best_val = _obj_val(a0)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        x = best_a.copy()
        if restart > 0:
            noise_level = 0.03 * restart
            noise = rng.standard_normal(n) * noise_level
            x = x + noise * best_a
            x = np.maximum(x, 1e-15)
            x = x / x.sum()

        lr = lr0
        prev_val = _obj_val(x)
        stall = 0

        for t in range(1, 500000):
            if time.monotonic() - t0 > time_limit:
                break
            if stall > 5000:
                break

            grad = _gradient_top_k(x, K, soft_beta)
            log_x = np.log(x + 1e-30) - lr * grad
            log_x -= log_x.max()
            x_new = np.exp(log_x)
            x_new = np.maximum(x_new, 1e-25)
            x_new = x_new / x_new.sum()

            val_new = _obj_val(x_new)
            if val_new < prev_val - 1e-14:
                x = x_new
                prev_val = val_new
                lr = min(lr * 1.05, 0.5)
                if val_new < best_val - 1e-12:
                    best_val = val_new
                    best_a = x.copy()
                stall = 0
            else:
                lr *= 0.7
                if lr < 1e-8:
                    lr = lr0 * 0.5
                    x = x + 0.01 * rng.standard_normal(n) * best_a
                    x = np.maximum(x, 1e-15)
                    x = x / x.sum()
                    prev_val = _obj_val(x)
                    lr = lr0
                stall += 1

            if verbose and t % 2000 == 0:
                print("    MD r={} t={} ({:.1f}s) J={:.7f} lr={:.5f} stall={}".format(
                    restart, t, time.monotonic() - t0, best_val, lr, stall))

    return best_a, best_val

    n = len(a0)
    if max_K is None:
        max_K = min(3000, 2 * n)
    y = _normalize(a0)
    best_a = y.copy()
    best_m = _fast_conv(y).max()
    t0 = time.monotonic()
    stall = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 2000:
            break

        ac = _fast_conv(y)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr

        if cur_val < 2.0 * n * best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        K_target = min(max_K, init_K + t * 3)
        top_lags = np.argsort(ac)[::-1][:max(K_target, 50)]
        active_set = set(int(k) for k in top_lags[:min(K_target, 500)])
        active_set.add(int(np.argmax(ac)))

        clist = sorted(active_set)
        K = len(clist)

        try:
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [cp.sum(g_var) == 1]
            for ki in clist:
                i_start = max(0, ki - n + 1)
                i_end = min(n, ki + 1)
                if i_start < i_end:
                    idx = np.arange(i_start, i_end)
                    row = y[ki - idx]
                    constraints.append(cp.sum(g_var[idx] * row) <= M_var)
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            prob.solve(solver=solver, verbose=False)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                raise RuntimeError("LP: " + str(prob.status))
        except Exception:
            if K > init_K:
                clist_short = [int(k) for k in top_lags[:init_K]]
                active_set = set(clist_short)
                if t > 10:
                    continue
            else:
                continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = g_val.sum()
        if gs < 1e-15:
            continue
        g_dir = g_val / gs

        # Check violations to potentially grow set (implicit in next iteration)
        # Line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial = 1e10
        for alpha in [best_alpha, 1.0, 0.5, 0.7, 0.3, 0.1]:
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            Mt = _fast_conv(trial / ts).max()
            if Mt < best_trial:
                best_trial = Mt
                best_alpha = alpha

        # Bisection refinement
        lo = max(0.0, best_alpha - 0.1)
        hi = min(1.0, best_alpha + 0.1)
        for _ in range(5):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            t1 = np.maximum((1 - m1) * y + m1 * g_dir, 0.0)
            t2 = np.maximum((1 - m2) * y + m2 * g_dir, 0.0)
            s1, s2 = t1.sum(), t2.sum()
            v1 = _fast_conv(t1 / s1).max() if s1 > 1e-15 else 1e10
            v2 = _fast_conv(t2 / s2).max() if s2 > 1e-15 else 1e10
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        best_alpha = (lo + hi) / 2.0

        if t <= 2:
            y = g_dir.copy()
        else:
            y = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy

        if verbose and t % 50 == 0:
            print("    LP-FW t={:5d} ({:.1f}s) J={:.7f} K={}".format(
                t, time.monotonic() - t0, 2 * n * best_m, K))

    return best_a, 2.0 * n * best_m


def _coord_refine(a0, time_limit, n_targets=10, seed=77, verbose=True):
    """Coordinate refinement: move mass from peak contributors to quiet positions."""
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_m = _fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 3000
    fracs = [0.002, 0.005, 0.01, 0.02, 0.04]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = _fast_conv(a)
        K = min(n_targets, len(ac))
        top_k = np.argsort(ac)[::-1][:K]
        improved = False

        for ki in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(ki)
            ws = max(0, ki)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-25:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-15:
                continue

            # Find quiet positions (low pressure)
            pressure = np.zeros(n)
            for kk in top_k[:6]:
                kk = int(kk)
                w2s = max(0, kk)
                w2e = min(n - 1, kk)
                if w2s > w2e:
                    continue
                id2 = np.arange(w2s, w2e + 1)
                pressure[id2] += a[kk - id2]
            pressure[src] = 1e10
            quiet = np.argsort(pressure)[:20]

            loop_improved = False
            for dst_idx in quiet:
                if time.monotonic() - t0 > time_limit:
                    break
                dst = int(dst_idx)
                if dst == src:
                    continue
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-18:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = _fast_conv(a_t).max()
                    if Mt < best_m - 1e-15:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        loop_improved = True
                        break
                if loop_improved:
                    break

            if not improved and stall > 200:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    amt = a[src] * rng.uniform(0.005, 0.02)
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = _fast_conv(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
                            improved = True
                break

        if not improved:
            stall += 1
            if stall > 500:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.005, 0.03)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = _fast_conv(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

        if verbose and stall > 0 and stall % 1000 == 0:
            print("    Coord ({:.1f}s) J={:.7f} stall={}".format(
                time.monotonic() - t0, 2 * n * best_m, stall))

    return best_a, 2.0 * n * best_m


def _interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return fi / ss


def propose_candidate(budget_s=1800.0):
    """Main search function. Returns best sequence found within time budget."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1740.0)
    solver = _get_solver()

    best_seq = None
    best_score = float('inf')

    # Initialize from parent if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = _normalize(p)
            best_score = _eval_safe(best_seq)
            print("Parent: n={} score={}".format(len(p), repr(best_score)))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_init = 12000
        x = np.linspace(0, 1, n_init)
        # Use a slightly skewed initialization (geometric-ish)
        base = np.exp(-0.3 * x)
        base = base / base.sum()
        best_seq = base
        best_score = _eval_safe(best_seq)
        print("Default init n={} score={}".format(n_init, repr(best_score)))

    n0 = len(best_seq)
    print("Starting: n={}, score={}".format(n0, repr(best_score)))

    # Phase 1: Quick mirror descent (get to good local min fast)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 250.0)
    if t1 > 30:
        print("\n=== P1: Mirror Descent (n={}) t={}s ===".format(n0, int(t1)))
        a_out, v_out = _mirror_descent(_normalize(best_seq), t1,
                                        K=80, lr0=0.03, soft_beta=20.0,
                                        n_restarts=1, verbose=True)
        sc = _eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P1 improved: {}".format(repr(best_score)))

    # Phase 2: LP-FW main (cutting plane)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 60:
        n_cur = len(best_seq)
        print("\n=== P2: LP-FW (n={}) t={}s ===".format(n_cur, int(t2)))
        a_out, v_out = _lp_fw(_normalize(best_seq), t2, solver,
                               init_K=80, max_K=min(2000, n_cur), verbose=True)
        sc = _eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P2 improved: {}".format(repr(best_score)))

    # Phase 3: Mirror descent with correct gradient (finer)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 30:
        n_cur = len(best_seq)
        print("\n=== P3: MD fine (n={}) t={}s ===".format(n_cur, int(t3)))
        a_out, v_out = _mirror_descent(_normalize(best_seq), t3,
                                        K=150, lr0=0.015, soft_beta=50.0,
                                        n_restarts=2, verbose=True)
        sc = _eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P3 improved: {}".format(repr(best_score)))

    # Phase 4: LP-FW round 2 (with potentially larger active set)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 300.0)
    if t4 > 60:
        n_cur = len(best_seq)
        print("\n=== P4: LP-FW round2 (n={}) t={}s ===".format(n_cur, int(t4)))
        a_out, v_out = _lp_fw(_normalize(best_seq), t4, solver,
                               init_K=150, max_K=min(3000, n_cur), verbose=True)
        sc = _eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P4 improved: {}".format(repr(best_score)))

    # Phase 5: Coordinate refinement
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 200.0)
    if t5 > 20:
        n_cur = len(best_seq)
        print("\n=== P5: Coord refine (n={}) t={}s ===".format(n_cur, int(t5)))
        a_out, v_out = _coord_refine(_normalize(best_seq), t5,
                                      n_targets=12, seed=55, verbose=True)
        sc = _eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P5 improved: {}".format(repr(best_score)))

    # Phase 6: Try slightly longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_long = min(len(best_seq) + 3000, 20000)
        if n_long != len(best_seq):
            print("\n=== P6: Long n={} ===".format(n_long))
            t6 = min(remaining * 0.25, 300.0)
            a_long = _interpolate(best_seq, n_long)
            a_out1, _ = _mirror_descent(a_long, t6 * 0.5, K=100, lr0=0.02,
                                         soft_beta=30.0, n_restarts=1, verbose=False)
            a_out2, _ = _coord_refine(a_out1, t6 * 0.3, n_targets=10, seed=88,
                                       verbose=False)
            a_out3, _ = _lp_fw(a_out2, t6 * 0.2, solver, init_K=100,
                                max_K=1000, verbose=False)
            sc1 = _eval_safe(a_out1)
            sc2 = _eval_safe(a_out2)
            sc3 = _eval_safe(a_out3)
            best_sc = min(sc1, sc2, sc3)
            if np.isfinite(best_sc) and best_sc < best_score:
                if best_sc == sc1:
                    best_seq = a_out1
                elif best_sc == sc2:
                    best_seq = a_out2
                else:
                    best_seq = a_out3
                best_score = best_sc
                print("  P6 improved: {}".format(repr(best_score)))

    # Phase 7: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        print("\n=== P7: Final polish (n={}) ===".format(n_cur))
        t7 = remaining * 0.6
        a_out, v_out = _coord_refine(_normalize(best_seq), t7,
                                      n_targets=8, seed=200, verbose=False)
        sc = _eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P7 improved: {}".format(repr(best_score)))

    # Final validation
    if best_seq is not None:
        fs = _eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={} n={} t={:.1f}s".format(
            repr(best_score), len(best_seq), elapsed))
    else:
        print("\nERROR: using fallback")
        n_fb = 10000
        x = np.linspace(0, 1, n_fb)
        best_seq = _normalize(np.exp(-0.3 * x))
        best_score = _eval_safe(best_seq)

    return [float(x) for x in best_seq]
