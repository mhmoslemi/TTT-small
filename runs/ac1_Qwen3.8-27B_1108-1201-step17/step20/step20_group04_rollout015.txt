def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def spectral_flattening(a0, time_limit, verbose=True, seed=42, alpha=0.5, n_iter_max=10000):
    """
    Spectral flattening: iteratively modify the power spectrum to flatten
    the autocorrelation, while projecting back to non-negative sequences.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    stall = 0
    max_stall = 200
    
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break
        
        # Compute current autocorrelation
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        c = np.fft.irfft(power, fft_size)[:2*n-1]
        c_max = c.max()
        
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive alpha: increase when stalling
        alpha_eff = alpha * (1.0 + stall / 100.0)
        alpha_eff = min(alpha_eff, 2.0)
        
        # Modify power spectrum to flatten autocorrelation
        # Where c is large, reduce power; where c is small, increase power
        # This is like a "water-filling" in frequency domain
        c_shifted = c - c.min() + 1e-10
        # We want to reduce the peaks, so we want the modification to be
        # inversely proportional to c
        target = 1.0 / (c_shifted ** alpha_eff)
        target = target / target.max()  # normalize
        
        # Blend with current power spectrum
        # Use geometric mean: new_power = |A|^2 * (target / (c^alpha))^beta
        # Actually, let's do: new_power = |A|^2 * (1/c^alpha)^beta
        # where beta controls the strength of the update
        
        # Simpler approach: new_power = |A|^2 * (mean_c / c)^alpha
        # This reduces power at frequencies that contribute to high c values
        
        # Actually, the correct approach for spectral flattening:
        # c = IFFT(|A|^2)
        # We want c to be flat. The gradient of max(c) w.r.t. |A|^2 is
        # the DFT of the indicator of where c is maximum.
        # For a smooth version: weight = 1/c (or 1/c^alpha)
        
        # Let's use: new_power = power * (mean(c) / c)^order
        # where order controls the strength
        order = 1.0  # fixed order for now
        weight = (c.mean() + 1e-10) / (c + 1e-15)
        weight = weight ** order
        # Limit the weight to avoid extreme changes
        weight = np.clip(weight, 0.1, 10.0)
        
        new_power = power * weight
        
        # Convert back to sequence
        # Keep the phase of A, modify the amplitude
        new_A = A * np.sqrt(weight)  # A * sqrt(weight) has |A|^2 * weight = new_power
        a_new = np.fft.irfft(new_A, fft_size)[:n].real
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Check if this improved things
        c_new = fast_autocorr(a_new)
        m_new = c_new.max()
        
        if m_new < c_max - 1e-15:
            a = a_new
        else:
            # Try a smaller step
            for step in [0.5, 0.25, 0.1]:
                a_trial = a + step * (a_new - a)
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < c_max - 1e-15:
                    a = a_trial
                    break
            else:
                # No improvement, try random perturbation
                if stall > 50:
                    pert = rng.normal(0, 0.02, n) * a
                    a_try = a + pert
                    a_try = np.maximum(a_try, 0.0)
                    s_t = a_try.sum()
                    if s_t > 1e-15:
                        a_try = a_try / s_t
                        m_try = fast_autocorr(a_try).max()
                        if m_try < c_max:
                            a = a_try
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("   SpecFlat it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))
    
    return best_a, best_m


def lp_direction_finding(f, M, time_per_lp=2.0, verbose=False):
    """
    Find direction g using LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns g normalized to sum 1, or None if LP fails.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Constraint: sum(g) = 1 (to make it bounded)
    # Actually, let's not constrain sum, just maximize it
    # But then the LP might be unbounded... 
    # Let's add sum(g) <= some upper bound
    # Actually, if f has all positive entries, then (f*g)[0] = f[0]*g[0] <= M
    # so g[0] <= M/f[0], which bounds each g[i] individually
    # So the LP is bounded.
    
    objective = cp.Maximize(cp.sum(g))
    constraints = [cp.sum(g) == 1]  # Normalize to make it a proper LP
    
    # Add convolution constraints: (f*g)[k] <= M for all k
    # (f*g)[k] = sum_i f[i] * g[k-i]
    # This is a linear constraint in g
    # For each k, the constraint is: sum_{i: 0<=i<n, 0<=k-i<n} f[i] * g[k-i] <= M
    
    # Build the constraint matrix
    # For constraint k: coefficient of g[j] is f[k-j] if 0<=k-j<n, else 0
    # This is a Toeplitz-like matrix
    
    # For efficiency, let's only add the most binding constraints
    c = fast_autocorr(f)  # This is f*f, not f*g, but gives us an idea of where peaks are
    # Actually, we need ALL constraints for correctness. But we can start with a subset.
    
    # Let's add constraints for the top 50% of lags (by f*f value) plus all lags near the peaks
    # Actually, for correctness, we need all 2n-1 constraints.
    # Let's build them efficiently.
    
    # For k = 0, ..., 2n-2:
    #   sum_{i=max(0,k-n+1)}^{min(k,n-1)} f[i] * g[k-i] <= M
    # Let j = k-i, so i = k-j
    #   sum_{j=max(0,k-n+1)}^{min(k,n-1)} f[k-j] * g[j] <= M
    
    # The coefficient of g[j] in constraint k is f[k-j] (if 0<=k-j<n)
    # This forms a matrix where entry (k,j) = f[k-j] if valid
    
    # For n=6700, this is a 13399 x 6700 matrix. Too large to build explicitly.
    # Let's use a sparse representation or just add constraints one by one.
    
    # Alternative: Use the FFT structure to check constraints after solving
    # But cvxpy needs explicit constraints...
    
    # Let's try a different approach: use only a subset of constraints
    # Start with the most critical ones (where f*f is largest)
    
    # Actually, for the LP to be useful, we need the right constraints.
    # Let's use all constraints but build them efficiently.
    
    # For cvxpy, we can write:
    # cp.sum(cp.multiply(f_pad[k:n-1+k if ...], g[...])) <= M
    # This is cumbersome. Let me try a different formulation.
    
    # Alternative formulation using convolution:
    # f * g = IFFT(FFT(f) * FFT(g))
    # But this is not linear in the cvxpy sense for the IFFT...
    # Actually, FFT is linear, so FFT(g) is linear in g.
    # But the product FFT(f)*FFT(g) is not linear (it's bilinear).
    # However, since FFT(f) is fixed, FFT(f)*FFT(g) IS linear in g!
    # So: (f*g)[k] = IFFT(FFT(f) * FFT(g))[k] = sum_m exp(2πi(k-m)) * FFT(f)[m] * FFT(g)[m] / N
    # This is linear in FFT(g), which is linear in g.
    # But expressing this in cvxpy is complex.
    
    # Let me just build the constraints explicitly for a subset of critical lags.
    
    # Find the lags where f*f is largest (these are likely to be binding)
    c_ff = fast_autocorr(f)
    n_constraints = min(2*n-1, 5000)  # Limit number of constraints
    # Choose the lags with highest f*f values
    top_lags = np.argsort(c_ff)[::-1][:n_constraints]
    
    for k in top_lags:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        # (f*g)[k] = sum_{i=i_start}^{i_end} f[i] * g[k-i]
        # = sum_{j=k-i_end}^{k-i_start} f[k-j] * g[j]
        # where j ranges from k-i_end to k-i_start
        j_start = k_int - i_end
        j_end = k_int - i_start
        # Constraint: sum_{j=j_start}^{j_end} f[k-j] * g[j] <= M
        coeffs = f[k_int - np.arange(j_start, j_end + 1)]
        g_slice = g[j_start:j_end + 1]
        constraints.append(cp.sum(cp.multiply(coeffs, g_slice)) <= M)
    
    prob = cp.Problem(objective, constraints)
    try:
        # Try different solvers
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.CBC]:
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = g.value
                    if g_val is not None and np.all(g_val >= -1e-6):
                        g_val = np.maximum(g_val, 0.0)
                        s = g_val.sum()
                        if s > 1e-15:
                            return g_val / s
                    break
            except Exception:
                continue
    except Exception:
        pass
    
    return None


def lp_iteration(a0, time_limit, verbose=True, seed=42, n_lp_iters=50, t_blend=0.1):
    """
    LP-based iteration: repeatedly find LP direction and blend.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    for it in range(n_lp_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_autocorr(a).max()
        
        # Find LP direction
        g = lp_direction_finding(a, M, verbose=verbose)
        if g is None:
            if verbose:
                print("   LP failed at iter " + str(it))
            break
        
        # Blend
        a_new = (1 - t_blend) * a + t_blend * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        
        M_new = fast_autocorr(a_new).max()
        
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        elif M_new < M:
            # Improved over current but not over best
            a = a_new
        else:
            # Try smaller blend
            for t in [t_blend * 0.5, t_blend * 0.25, t_blend * 0.1]:
                a_try = (1 - t) * a + t * g
                a_try = np.maximum(a_try, 0.0)
                s_t = a_try.sum()
                if s_t < 1e-15:
                    continue
                a_try = a_try / s_t
                M_try = fast_autocorr(a_try).max()
                if M_try < M - 1e-15:
                    a = a_try
                    if M_try < best_m:
                        best_m = M_try
                        best_a = a.copy()
                    break
        
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("   LP it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
    
    return best_a, best_m


def subgradient_opt(a0, time_limit, verbose=True, seed=42):
    """
    Subgradient method for the (non-convex) problem of minimizing max(conv(a,a)).
    Uses a smoothed approximation with adaptive parameters.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    # Nesterov momentum
    x = a.copy()
    x_prev = a.copy()
    t_nest = 1.0
    beta = 50.0  # smoothing parameter
    stall = 0
    
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        # Compute smoothed gradient
        c = fast_autocorr(y)
        c_max = c.max()
        shifted = beta * (c - c_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        total = exp_vals.sum()
        if total < 1e-300:
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p = exp_vals / total
        
        # Gradient: grad_j = (2/beta) * sum_k p[k] * a[k-j]
        # = (2/beta) * conv(p, y_rev)[n-1+j]
        P = np.fft.rfft(p, fft_size)
        Y_rev = np.fft.rfft(y[::-1], fft_size)
        Conv = np.fft.irfft(P * Y_rev, fft_size)
        grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
        
        # Step size
        eta = 0.5 / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)
        
        # Gradient step + projection
        z = y - eta * grad
        x_new = simplex_project(z)
        
        # Check improvement
        m_new = fast_autocorr(x_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        # Adapt beta
        if it % 30 == 0:
            beta = min(beta * 1.008, 5000.0)
        
        # Restart if stalled
        if stall > 1000:
            stall = 0
            t_nest = 1.0
            noise = rng.normal(0, 0.03, n) * best_a
            x = best_a + noise
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            beta = max(beta * 0.5, 20.0)
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("   SubGrad it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + str(round(beta, 1)))
    
    return best_a, best_m


def peak_surgery_v2(a0, time_limit, verbose=True, seed=77):
    """Improved peak surgery with better destination selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit and no_improve < 5000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find top peaks
        n_top = min(20, 2*n-1)
        top_idx = np.argsort(c)[::-1][:n_top]
        
        improved = False
        found_move = False
        
        for k in top_idx[:10]:
            if found_move:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            if products.max() < 1e-20:
                continue
            
            # Top contributing pairs
            top_pairs = np.argsort(products)[::-1][:5]
            
            for pi in top_pairs:
                if found_move:
                    break
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n:
                    continue
                
                # Try reducing the larger of the pair
                if a[src_i] >= a[partner]:
                    reduce_i = src_i
                else:
                    reduce_i = partner
                
                if a[reduce_i] < 1e-14:
                    continue
                
                # Compute pressure: how much each position contributes to top peaks
                pressure = np.zeros(n)
                for k2 in top_idx[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        pressure[idx2] += a[k2i - idx2]
                        pressure[k2i - idx2] += a[idx2]
                
                # Find low-pressure destinations
                candidates = np.where(pressure < pressure[reduce_i] * 0.5)[0]
                if len(candidates) < 20:
                    candidates = np.where(pressure < np.percentile(pressure, 30))[0]
                if len(candidates) < 10:
                    candidates = np.arange(n)
                
                candidates = candidates[candidates != reduce_i]
                if len(candidates) == 0:
                    continue
                
                # Try the best destinations
                cand_pressure = pressure[candidates]
                n_try = min(8, len(candidates))
                try_dsts = candidates[np.argsort(cand_pressure)[:n_try]]
                
                for dst in try_dsts:
                    dst_int = int(dst)
                    for frac in [0.01, 0.02, 0.05, 0.1]:
                        amt = a[reduce_i] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[reduce_i] -= amt
                        a_trial[dst_int] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            found_move = True
                            break
                    if found_move:
                        break
                
        if not improved:
            # Random exploration
            if no_improve > 200:
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.1)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
