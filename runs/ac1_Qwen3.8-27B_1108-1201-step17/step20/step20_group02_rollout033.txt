One more thought: the "away-step" Frank-Wolfe variant can significantly speed up convergence. The idea is:
- Instead of y_new = (1-t)*y + t*g, use:
  - If there's an "away" vertex v (a vertex of the simplex that we want to move away from):
    y_new = (1-t)*y + t*(g - v)/(1 - <v, y>)
  - Otherwise: y_new = (1-t)*y + t*g

In our case, the "simplex" is the probability simplex {a >= 0, sum(a) = 1}, and the vertices are the standard basis vectors e_0, ..., e_{n-1}. The LP solution g is always in the interior (or on the boundary) of the simplex, so the "away" vertex would be a coordinate where y_i is large but g_i is small.

Actually, in the standard FW on the probability simplex, the "vertex" chosen by the LP is g (a point in the simplex, not necessarily a vertex). The away-step variant applies when the LP is:
min <c, v> subject to v in simplex
The optimum is v = e_j where j = argmin c_j.

But in our case, the LP is:
min M subject to sum(g) = 1, g >= 0, Cg <= M
The optimum g is generally NOT a vertex of the simplex. So the standard away-step FW doesn't directly apply.

However, we can still use a modified away-step: if a coordinate y_i is very large (say, in the top-1% of coordinates), we can try to reduce it by adding a constraint (a*g)_k <= M - epsilon for the k where a_{k-i} is large.

This is essentially what the active set management already does. So I'll skip the away-step and focus on other improvements.

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_of(a):
    a = norm_sum1(a)
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_lp_matrix(y, active_set, n, use_bilinear=True):
    K = len(active_set)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(active_set[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            if use_bilinear:
                coeffs[jj, idx] = y[k_int - idx]
                mid = k_int // 2
                if k_int % 2 == 0 and i_start <= mid < i_end:
                    coeffs[jj, mid] += y[mid]
            else:
                coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw_optimize(a0, time_limit, solver, initial_K=300, max_K=3000,
                   add_per_iter=50, verbose=False):
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac_y = fast_convolve(y)
    best_m = ac_y.max()
    t0 = time.monotonic()

    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 600
    iter_count = 0
    stagnation_count = 0
    use_bilinear = True

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if stagnation_count > 5:
                if verbose:
                    print("    LP-FW: persistent stagnation, stopping")
                break
            ac_curr = fast_convolve(y)
            n_pert = max(30, len(active_set) // 4)
            rng_pert = np.random.default_rng(stagnation_count * 1000 + 7)
            keep = list(rng_pert.choice(len(active_set), size=len(active_set) - n_pert, replace=False))
            new_adds = np.argsort(ac_curr)[::-1][:n_pert * 2]
            new_set = set(int(active_set[i]) for i in keep)
            for k in new_adds:
                new_set.add(int(k))
            active_set = sorted(new_set)
            no_improve = 0
            stagnation_count += 1
            continue

        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-13:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            stagnation_count = 0
        else:
            no_improve += 1

        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            n_keep = int(max_K * 0.6)
            keep_idx = np.argsort(margins)[:n_keep]
            active_set_base = [active_set[i] for i in keep_idx]
            near_max = np.where(ac > M_curr - 1e-6)[0]
            for v in near_max[:50]:
                if int(v) not in active_set_base:
                    active_set_base.append(int(v))
            active_set = sorted(set(active_set_base))

        K = min(len(active_set), max_K)
        if K == 0:
            break

        active_set_use = active_set[:K]
        coeffs = build_lp_matrix(y, active_set_use, n, use_bilinear)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(100, K // 3)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.asarray(g_var.value, dtype=np.float64)
        g_val = np.maximum(g_val, 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Add new violations
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]
        all_viol = np.where(conv_ys > M_curr + 1e-9)[0]
        if len(all_viol) > 0:
            viol_amt = conv_ys[all_viol] - M_curr
            top_v = all_viol[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(int(k) for k in active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        # Step size: try multiple alphas
        best_alpha = 1.0 / max(iter_count, 2)
        best_trial_M = M_curr
        alphas = [1.0, 0.8, 0.6, 0.4, 0.25, 0.15,
                  max(0.1, 3.0 / (iter_count + 1)),
                  max(0.05, 2.0 / (iter_count + 1))]
        for alpha in alphas:
            if alpha >= 1.0:
                trial = g_dir.copy()
            else:
                trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 3:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) +
                  "s): score=" + repr(2.0 * n * best_m) + " |S|=" + str(len(active_set)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): score=" +
              repr(2.0 * n * best_m) + ", iters=" + str(iter_count))
    return best_a, best_m


def reweighting_descent(a0, time_limit, verbose=False):
    """Iterative reweighting: flatten the convolution by adjusting a."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_convolve(y).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    beta = 0.5
    iter_count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 2000:
        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-13:
            best_m = M_curr
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Find the argmax and use it to guide updates
        k_star = int(np.argmax(ac))

        # Compute subgradient direction
        grad = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] = y[k_star - idx]

        # Mirror descent update with learning rate decay
        eta = 0.02 / (1.0 + iter_count / 500.0)
        log_a = np.log(np.maximum(y, 1e-30)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        # Blending with current
        blend = 0.3
        y = (1.0 - blend) * y + blend * a_new
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = best_a.copy()
        else:
            y = y / s_y

        # Periodic random perturbation to escape local minima
        if stall > 500 and stall % 500 == 0:
            noise = rng.normal(0, 0.02, n)
            y_p = y + noise
            y_p = np.maximum(y_p, 0.0)
            sp = y_p.sum()
            if sp > 1e-12:
                y = y_p / sp
            eta = eta * 1.5
            stall = 0

        if verbose and iter_count % 1000 == 0:
            elapsed = time.monotonic() - t0
            print("    Reweight t=" + str(iter_count) + " (" + str(round(elapsed, 1)) +
                  "s): score=" + repr(2.0 * n * best_m))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    Reweight done (" + str(round(elapsed, 1)) + "s): score=" +
              repr(2.0 * n * best_m) + ", iters=" + str(iter_count))
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=77, verbose=False):
    """Peak-directed coordinate transfer polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # Find least-impact destination
            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            # Exclude peak contributors
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            dst = int(candidates[np.argmin(scores[candidates])])
            if dst == src:
                continue

            found = False
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and rng.random() < 0.3:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    a[src] -= a[src] * frac
                    a[j] += a[src] * frac / (1.0 - frac)
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0:

                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:

                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True
                                break

        if verbose and stall % 500 == 0:
