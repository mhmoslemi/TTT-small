```python
def lp_direction(a, active_indices, solver):
    """Solve min_{s>=0, sum(s)=1} max_k (a*s)[k] using only active constraints."""
    n = len(a)
    K = len(active_indices)
    
    # Build constraint matrix
    # A[k, j] = a[k-j] if 0 <= k-j < n else 0
    # For efficiency, only build rows for active indices
    A = np.zeros((K, n))
    for idx, k in enumerate(active_indices):
        k_int = int(k)
        # (a*s)[k_int] = sum_j a[k_int - j] * s[j] for valid j
        # Valid j: 0 <= k_int - j < n => k_int - n + 1 <= j <= k_int
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            A[idx, j_start:j_end+1] = a[k_int - np.arange(j_start, j_end+1)]
    
    # Solve LP: min M s.t. A @ s <= M*1, sum(s) = 1, s >= 0
    s_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [
        cp.sum(s_var) == 1,
        A @ s_var <= M_var
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
```

```python
def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lp_direction(a, active_indices, solver):
    """Solve min_M s.t. A@S <= M, sum(S)=1, S>=0 where A[k,j]=a[k-j] for valid k-j."""
    n = len(a)
    K = len(active_indices)
    if K == 0:
        return None

    # Build constraint matrix: row k has A[k,j] = a[k-j] for valid j
    A = np.zeros((K, n))
    for idx, k_val in enumerate(active_indices):
        k_int = int(k_val)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            A[idx, j_start:j_end + 1] = a[k_int - js]

    s_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [
        cp.sum(s_var) == 1,
        A @ s_var <= M_var
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    s_val = np.maximum(np.asarray(s_var.value, dtype=np.float64), 0.0)
    ss = s_val.sum()
    if ss < 1e-15:
        return None
    return s_val / ss


def frank_wolfe_lp(a0, time_limit, solver, max_active=400, verbose=True):
    """Frank-Wolfe with active-set LP subproblem."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Initial active set: top peaks
    b = fast_autocorr(a)
    active_set = set(int(k) for k in np.argsort(b)[::-1][:200])
    it = 0
    no_improve = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        if elapsed > time_limit - 2:
            break

        b = fast_autocorr(a)
        M_curr = b.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 2000:
            break

        # Update active set: top peaks + most violated
        top_peaks = np.argsort(b)[::-1][:min(150, len(b))]
        for k in top_peaks:
            active_set.add(int(k))

        # Remove constraints that are far from active
        if len(active_set) > max_active:
            margins = np.array([M_curr - b[k] for k in active_set])
            keep = np.argsort(margins)[:max_active]
            active_set = set(int(list(active_set)[i]) for i in keep)

        active_list = sorted(active_set)

        # Solve LP for direction
        s_dir = lp_direction(a, active_list, solver)
        if s_dir is None:
            no_improve += 100
            continue

        # Check for violations and add to active set
        cross = fast_crossconv(a, s_dir)
        violations = np.where(cross > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = cross[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                active_set.add(int(k))

        # Line search: try multiple step sizes
        best_alpha = 2.0 / (it + 2.0)
        best_trial_M = fast_autocorr(a).max()
        alphas = [best_alpha, best_alpha * 0.5, best_alpha * 1.5, 1.0 / (it + 1), 0.01, 0.05]
        for alpha in alphas:
            if alpha <= 0 or alpha > 1:
                continue
            trial = (1.0 - alpha) * a + alpha * s_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        a_new = (1.0 - best_alpha) * a + best_alpha * s_dir
        a_new = np.maximum(a_new, 0.0)
        st = a_new.sum()
        if st > 1e-15:
            a_new = a_new / st
        a = a_new

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    FW-LP iter=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " |S|=" + str(len(active_set)))

    return best_a, best_m

            m_x = b_x.max()
            if m_x < best_m - 1e-14:
                best_m = m_x
                best_a = x.copy()
                stall = 0
            else:
                stall += 1
                # Revert Nesterov if stalling too much
                if stall > 500:
                    x = best_a.copy()
                    y = best_a.copy()

            # Update beta
            if t % 20 == 0:
                beta = min(beta * beta_growth, beta_end)

            # Random restart
            if stall > 1000:
                stall = 0
                noise = 0.03 / (1.0 + 0.1 * (stall // 1000))
                pert = rng.normal(0, noise, n) * best_a
                x_r = best_a + pert
                x_r = np.maximum(x_r, 1e-12)
                x_r = x_r / x_r.sum()
                x = x_r
                y = x_r.copy()
                beta = max(beta * 0.5, 10.0)
                if verbose:
                    elapsed = time.monotonic() - t0
                    J = 2.0 * n * best_m
                    print("    Nesterov restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

            if verbose and t % 500 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    Nesterov t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " beta=" + str(round(beta, 1)))

    return best_a, best_m


def peak_shaving_fast(a0, time_limit, top_k=30, verbose=True):
    """Fast greedy peak shaving: directly reduce the top autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if stall > 800:
            break

        b = fast_autocorr(a)
        peak_indices = np.argsort(b)[::-1][:top_k]
        M_curr = b.max()

        # Compute pressure map
        pressure = np.zeros(n)
        for k_int in peak_indices[:15]:
            k_i = int(k_int)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                pressure[idx] += a[k_i - idx]

        improved = False

        # Try to reduce top peaks
        for k_int in peak_indices[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_i = int(k_int)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_i - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            partner = k_i - src
            if partner < 0 or partner >= n:
                continue

            # Choose which to reduce (the larger one)
            if a[src] >= a[partner]:
                reduce_idx = src
            else:
                reduce_idx = partner

            if a[reduce_idx] < 1e-14:
                continue

            # Find safe destination
            safe_pressure = pressure.copy()
            safe_pressure[reduce_idx] = np.max(pressure) + 1.0
            # Also avoid positions in top peaks
            for k2 in peak_indices[:8]:
                k2i = int(k2)
                i2s = max(0, k2i - n + 1)
                i2e = min(n - 1, k2i)
                # Don't increase positions that contribute to other peaks
                # (approximate: just avoid high pressure)
            safe_pressure[reduce_idx] = np.max(pressure) + 2.0

            # Pick lowest pressure position
            dst = int(np.argmin(safe_pressure))
            if dst == reduce_idx:
                continue

            # Try to move mass
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.4]:
                amt = a[reduce_idx] * frac
                if amt < 1e-16:
                    break
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Periodic perturbation
            if stall % 100 == 0:
                noise_scale = min(0.05, 0.01 * stall / 100)
                pert = rng.normal(0, noise_scale, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-12)
                a_r = a_r / a_r.sum()
                m_r = fast_autocorr(a_r).max()
                if m_r < best_m:
                    best_m = m_r
                    best_a = a_r.copy()
                a = a_r if m_r < best_m + 0.02 else best_a.copy()

        if verbose and stall % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakShave (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " stall=" + str(stall))

    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate-wise optimization targeting top peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break

        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:20]
        M_curr = b.max()

        # For each top peak, try to reduce it by adjusting coordinates
        for k_int in top_peaks[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_i = int(k_int)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)

            # Find the most contributing pair
            contrib = a[idx] * a[k_i - idx]
            top_pairs = np.argsort(contrib)[::-1][:5]

            for pi in top_pairs:
                src = int(idx[pi])
                partner = k_i - src
                if partner < 0 or partner >= n:
                    continue
                if a[src] < 1e-14:
                    continue

                # Try moving some mass from src to a random low-pressure position
                n_dest = 5
                dests = rng.choice(n, size=n_dest, replace=False)
                for dst in dests:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        m_trial = fast_autocorr(a_trial).max()
                        if m_trial < best_m - 1e-14:
                            a = a_trial
                            best_m = m_trial
                            best_a = a.copy()
                            break

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    CoordDesc it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search: Frank-Wolfe LP + Nesterov subgradient + peak shaving + multi-length."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print("Solver: " + str(solver))
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 7000
        x = np.linspace(0, 1, n_base)
        a_raw = (x + 0.01) ** 1.5 * (1.0 - x + 0.01) ** 0.5
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: Quick peak shaving to get initial improvement
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Initial peak shaving (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.08, 120.0)
    if t1 > 15:
        a_out, _ = peak_shaving_fast(best_seq, t1, top_k=25, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps1_n" + str(n0), t_global)

    # Phase 2: Frank-Wolfe with LP (main optimization)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Frank-Wolfe LP (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.35, 600.0)
    if t2 > 60:
        a_out, _ = frank_wolfe_lp(best_seq, t2, solver, max_active=500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fwlp_n" + str(n_best), t_global)

    # Phase 3: Nesterov subgradient
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Nesterov subgradient (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 60:
        a_out, _ = nesterov_subgradient(best_seq, t3, beta_start=30.0, beta_end=3000.0,
                                        beta_growth=1.012, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(n_best), t_global)

    # Phase 4: Second Frank-Wolfe round
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: FW-LP round 2 (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.20, 350.0)
    if t4 > 60:
        a_out, _ = frank_wolfe_lp(best_seq, t4, solver, max_active=500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fwlp2_n" + str(n_best), t_global)

    # Phase 5: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Peak shaving (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_out, _ = peak_shaving_fast(best_seq, t5, top_k=20, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps2_n" + str(n_best), t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-length exploration ===")
    nearby_lengths = []
    for dn in [-300, -100, 100, 300, 500, 700]:
        nl = n_best + dn
        if 4000 < nl <= 9000:
            nearby_lengths.append(nl)

    for new_n in nearby_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("  Exploring n=" + str(new_n) + " (remaining=" + str(round(remaining, 1)) + "s)")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick optimization at new length
        t6a = min(remaining * 0.25, 120.0)
        if t6a > 20:
            a_out, _ = peak_shaving_fast(a_long, min(t6a * 0.5, 60.0), top_k=20, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ps_ml_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining * 0.35, 180.0)
        if t6b > 40:
            a_out, _ = frank_wolfe_lp(best_seq, min(t6b, 150.0), solver, max_active=400, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fwlp_ml_n" + str(new_n), t_global)

    # Phase 7: Final Nesterov polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final polish (n=" + str(n_best) + ") ===")
    if remaining > 40:
        t7a = min(remaining * 0.4, 120.0)
        if t7a > 20:
            a_out, _ = nesterov_subgradient(best_seq, t7a, beta_start=100.0, beta_end=5000.0,
                                            beta_growth=1.02, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_nest", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        t7b = min(remaining * 0.4, 100.0)
        if t7b > 20:
            a_out, _ = peak_shaving_fast(best_seq, t7b, top_k=15, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_ps", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        t7c = min(remaining * 0.4, 80.0)
        if t7c > 20 and n_best <= 8000:
            a_out, _ = frank_wolfe_lp(best_seq, t7c, solver, max_active=400, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_fwlp", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 7000
        x = np.linspace(0, 1, n_fb)
        a_raw = (x + 0.01) ** 1.0 * (1.0 - x + 0.01) ** 1.0
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
```
