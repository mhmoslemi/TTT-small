"""

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_subgradient(a, b, top_k=10):
    """Compute subgradient of max(b) w.r.t. a, targeting top-k peaks."""
    n = len(a)
    size = 2 * n - 1
    # Find the top-k lag positions
    top_idx = np.argsort(b)[::-1][:top_k]
    # For each top lag k, the gradient of b[k] w.r.t. a[i] is a[k-i] (with bounds)
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += a[j]
    return grad / len(top_idx)

def fista_optimize(a0, time_limit, beta_start=20.0, beta_end=1000.0,
                   beta_growth=1.003, top_k=15, restart_every=3000,
                   verbose=True):
    """FISTA with BB step size and Nesterov momentum."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(42)
    avg_x = x.copy()
    avg_count = 0.0
    s_prev = None
    y_prev = None
    bb_step = 1.0 / (n ** 0.5)

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        b = fast_autocorr(y)
        b_max = b.max()
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_use_max = b_use[idx].max()
        
        shifted = beta * (b_use - b_use_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        p_total = exp_vals.sum()
        if p_total < 1e-300:
            p = np.zeros_like(b_use)
            p[idx] = 1.0 / top_k
        else:
            p = exp_vals / p_total

        # Gradient: convolve reversed a with p
        a_rev = y[::-1]
        conv_len = n + len(p) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        Ar = np.fft.rfft(a_rev, fft_size)
        Pr = np.fft.rfft(p, fft_size)
        conv = np.fft.irfft(Ar * Pr, fft_size)
        grad = conv[n - 1:2 * n - 1] / beta

            y_prev = y.copy()

        x_new = simplex_project(z)

        max_new = b_new.max()

        avg_count += 1.0
        avg_x = avg_x + (x_new - avg_x) / avg_count

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x

def subgradient_descent(a0, time_limit, top_k=5, lr_base=0.01, verbose=True):
    """Exact subgradient descent targeting the max autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    lr = lr_base
    stall = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 2000:
            break

        b = fast_autocorr(a)
        M = b.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            lr = max(lr * 0.999, 1e-8)

        # Compute subgradient
        top_idx = np.argsort(b)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            grad[idx] += a[k - idx]
        grad /= len(top_idx)

        # Move
        a_new = a - lr * grad
        a_new = simplex_project(a_new)

        # Check improvement
        M_new = fast_autocorr(a_new).max()
        if M_new < M:
            a = a_new
        else:
            # Try smaller step
            a_new = a - lr * 0.1 * grad
            a_new = simplex_project(a_new)
            M_new2 = fast_autocorr(a_new).max()
            if M_new2 < M:
                a = a_new
                lr *= 0.5
            else:
                lr *= 0.8

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  Subgrad t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m

        total_contrib = contrib.sum()

        if total_contrib < 1e-20:
            break

        i_hot = int(idx[np.argmax(contrib)])
        j_hot = k_max - i_hot

        # Positions that don't contribute to the top peaks
        top_peaks = np.argsort(b)[::-1][:20]

            s2 = max(0, k2 - n + 1)

        if len(cold_positions) < 5:
            cold_positions = np.arange(n)

            i_cold = int(cold_positions[rng.integers(len(cold_positions))])

                a_trial[i_hot] -= amt

                if s_t < 1e-15:
                    continue

def cyclic_shift_best(a0):
    """Find the best cyclic shift."""
    n = len(a0)
    best_m = fast_autocorr(a0).max()

def interp_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def coordinate_descent(a0, time_limit, n_coords=None, verbose=True):

    stall = 0

        if time.monotonic() - t0 > time_limit:
            break
        if stall > 50:
            break

        for ci in coord_idx:
            ci = int(ci)

                continue

            lo = 0.0

Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:

        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_smoothed_grad(y, beta, top_k):
    n = len(y)
    b = fast_autocorr(y)
    idx = np.argsort(b)[::-1][:top_k]
    b_use = np.zeros(len(b))
    b_use[idx] = b[idx]
    b_use_max = b[idx].max()
    shifted = beta * (b_use - b_use_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b_use)
        p[idx] = 1.0 / top_k
    else:
        p = exp_vals / p_total
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad

            y = y / s_y

        grad = compute_smoothed_grad(y, beta, top_k)

        # Fixed BB step size update
        if s_prev is not None and y_prev is not None:
            sy = y - y_prev
            gy = grad - s_prev
            sy_gy = sy.dot(gy)
            gy_gy = gy.dot(gy)
            if gy_gy > 1e-20:
                bb_candidate = sy_gy / gy_gy
                bb_max = 5.0 / beta
                if bb_candidate > 1e-8 and bb_candidate < bb_max:
                    bb_step = 0.6 * bb_step + 0.4 * bb_candidate
                    bb_step = max(1e-6, min(bb_step, bb_max))
                else:
                    bb_step = max(bb_step * 0.85, 1e-6)
            s_prev = grad.copy()
            y_prev = y.copy()

        eta = min(bb_step, 0.3)
        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        avg_count += 1.0
        avg_x = avg_x + (x_new - avg_x) / avg_count

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 100 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.02 / (1.0 + 0.03 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 10.0)
            s_prev = None
            y_prev = None
            bb_step = 1.0 / (n ** 0.5)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("  FISTA restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("  FISTA t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + repr(round(beta, 1)))

    J_avg = 2.0 * n * fast_autocorr(avg_x).max()
    J_best = 2.0 * n * best_max_b
    if J_avg < J_best:
        return avg_x, fast_autocorr(avg_x).max()
    return best_a, best_max_b


def subgradient_descent(a0, time_limit, top_k=5, lr_base=0.005, verbose=True):
    """Exact subgradient descent on the max autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    lr = lr_base
    stall = 0

    for t in range(1, 3000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 3000:
            break

        b = fast_autocorr(a)
        M = b.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 100 == 0:
                lr = max(lr * 0.95, 1e-9)

        # Compute subgradient from top-k peaks
        top_idx = np.argsort(b)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            grad[idx] += a[k - idx]
        grad /= len(top_idx)

        # Try multiple step sizes
        tried = False
        for lr_try in [lr, lr * 0.3, lr * 0.1, lr * 0.5]:
            a_new = a - lr_try * grad
            a_new = simplex_project(a_new)
            M_new = fast_autocorr(a_new).max()
            if M_new < M - 1e-16:
                a = a_new
                lr = lr_try
                tried = True
                break
        if not tried:
            lr = max(lr * 0.7, 1e-10)

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  Subgrad t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def deflation_search(a0, time_limit, n_trials=200, verbose=True):
    """Targeted deflation: move mass away from peak-contributing elements."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for t in range(n_trials * 5 + 100):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        M = b.max()
        k_max = int(np.argmax(b))

        i_start = max(0, k_max - n + 1)
        i_end = min(n, k_max + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[k_max - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-20:
            break

        i_hot = int(idx[np.argmax(contrib)])

        # Find cold positions
        top_peaks = np.argsort(b)[::-1][:20]
        in_peak = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            k2 = int(k2)
            s2 = max(0, k2 - n + 1)
            e2 = min(n, k2 + 1)
            in_peak[s2:e2] = True

        cold_positions = np.where(~in_peak)[0]
        if len(cold_positions) < 10:
            cold_positions = np.arange(n)

        found_improve = False
        for attempt in range(8):
            i_cold = int(cold_positions[rng.integers(len(cold_positions))])
            if i_cold == i_hot:
                continue
            for frac in [0.003, 0.008, 0.02, 0.05]:
                amt = a[i_hot] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i_hot] -= amt
                a_trial[i_cold] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    found_improve = True
                    break
            if found_improve:
                break

        if not found_improve and t % 20 == 0:
            # Random perturbation to escape local minimum
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.002, 0.02))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            found_improve = True

        if verbose and t % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  Deflate t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def cyclic_shift_best(a0):
    """Find the best cyclic shift among all possible shifts."""
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        b = fast_autocorr(a_shift)
        m = b.max()
        if m < best_m - 1e-15:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def local_cyclic_shift(a0, half_window=100):
    """Try cyclic shifts in a local window for faster search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_m = fast_autocorr(a).max()
    best_shift = 0
    for s in list(range(1, half_window + 1)) + list(range(-half_window, 0)):
        s_mod = s % n
        if s_mod == 0:
            continue
        a_shift = np.roll(a, -s_mod)
        b = fast_autocorr(a_shift)
        m = b.max()
        if m < best_m - 1e-15:
            best_m = m
            best_shift = s_mod
    if best_shift != 0:
        return np.roll(a, -best_shift), best_m
    return a, best_m


def interp_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def systematic_pair_exchange(a0, time_limit, verbose=True):
    """Systematic pair exchange: target the peak, move contributing elements."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_autocorr(a)
        M = b.max()
        top_peaks = np.argsort(b)[::-1][:15]
        improved = False

        for k_peak in top_peaks[:8]:
            k_peak = int(k_peak)
            i_start = max(0, k_peak - n + 1)
            i_end = min(n, k_peak + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_peak - idx]

            if contrib.max() < 1e-20:
                continue

            # Sort by contribution
            order = np.argsort(contrib)[::-1]
            for oi in order[:5]:
                src = int(idx[oi])
                dst_match = k_peak - src

                # Build "in-peak" mask
                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_peaks[:10]:
                    k2 = int(k2)
                    s2 = max(0, k2 - n + 1)
                    e2 = min(n, k2 + 1)
                    in_peak[s2:e2] = True

                candidates = np.where(~in_peak)[0]
                if len(candidates) < 5:
                    candidates = np.arange(n)

                # Try a few destinations
                for _ in range(4):
                    dst = int(candidates[rng.integers(len(candidates))])
                    if dst == src:
                        continue
                    if a[src] < 1e-12:
                        break

                    for frac in [0.005, 0.015, 0.04]:
                        amt = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-14:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Escaping perturbation
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if
