def fast_autocorr(a):

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_matrix(a, lags, n):
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs

def subgradient_fw(a0, time_limit, n_steps=500, verbose=True):
    """Subgradient Frank-Wolfe: use the peak lag to define a linearized direction."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    decay = 2.0 / (n_steps + 2.0)

    for t in range(1, n_steps + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()

        # Find top few lags
        top_lags = np.argsort(ac)[::-1][:5]
        # Subgradient: average of the "active" gradient directions
        grad = np.zeros(n)
        for k in top_lags:
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            idx = np.arange(i_s, i_e)
            grad[idx] += y[ki - idx]
        grad /= len(top_lags)

        # FW step: minimize grad · g over simplex => g = delta at argmin grad
        g_dir = np.zeros(n)
        g_dir[int(np.argmin(grad))] = 1.0

        alpha = 2.0 / (t + 2.0)
        y_new = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy

    return best_a, 2.0 * n * best_m

                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-14:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            found = True
                            break
                    if found:
                        break
                if found:
                    break

        if not improved:
            stall += 1
            if stall > 100:
                # Random perturbation
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                if i1 != i2 and a[i1] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i1] * frac
                    a[i1] -= amt
                    a[i2] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 500 == 0:
            print("    Equalize (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2 * n * best_m) + " stall=" + str(stall))

    return best_a, 2.0 * n * best_m

def diverse_init(n, parent_seq):
    """Generate diverse initial candidates."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)

    candidates = []

    # Linear ramps (exploit asymmetry)
    for p in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]:
        candidates.append(normalize((i_arr + 1.0) ** p))
        candidates.append(normalize((n - i_arr) ** p))

    # Asymmetric Gaussians
    for c in [0.3, 0.4, 0.5, 0.6]:
        for sg in [0.1, 0.15, 0.2, 0.3]:
            candidates.append(normalize(np.exp(-((x - c) ** 2) / (2 * sg * sg))))

    # Skewed shapes
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        shape = x ** alpha * (1 - x)
        shape = np.maximum(shape, 1e-10)
        candidates.append(normalize(shape))
        shape2 = x * (1 - x) ** alpha
        shape2 = np.maximum(shape2, 1e-10)
        candidates.append(normalize(shape2))

    # Sine-based
    for f in [1, 2, 3]:
        candidates.append(normalize(np.sin(np.pi * x * f) ** 2))

    # Blend with parent
    if parent_seq is not None:
        bi = normalize(parent_seq)
        for w in [0.3, 0.5, 0.7, 0.9]:
            candidates.append(normalize(w * bi + (1 - w) * candidates[0]))
            candidates.append(normalize(w * bi + (1 - w) * candidates[len(candidates) // 2]))

    best_a = normalize(parent_seq) if parent_seq is not None else normalize(np.sin(np.pi * x) ** 2)
    best_val = obj_val(best_a)

    for c in candidates:
        v = obj_val(c)
        if v < best_val:
            best_val = v
            best_a = c.copy()

    return best_a, best_val

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1755.0)
    solver = get_solver()
    print("Total budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent loaded: n=" + str(len(p)) + " score=" + repr(best_score))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_init = 10000
        x = np.linspace(0, 1, n_init)
        # Start with asymmetric shape
        best_seq = normalize((np.arange(n_init, dtype=float) + 1.0) ** 1.2)
        best_score = eval_safe(best_seq)
        print("Default init n=" + str(n_init) + " score=" + repr(best_score))

    n_original = len(best_seq)

    # === Phase 1: Quick diverse init check ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.03, 40.0)
    print("\n=== P1: Diverse Init ===")
    if t1 > 5:
        a_p, v_p = diverse_init(len(best_seq), best_seq)
        sc = eval_safe(a_p)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_p
            print("  Init improved: " + repr(best_score))

    # === Phase 2: LP-FW at moderate length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_lp = min(len(best_seq), 8000)
    a_lp = interpolate_seq(best_seq, n_lp) if n_lp != len(best_seq) else best_seq.copy()
    print("\n=== P2: LP-FW (n=" + str(n_lp) + ") ===")
    t2 = min(remaining * 0.30, 480.0)
    if t2 > 60:
        a_out, v_out = lp_fw_fast(normalize(a_lp), t2, solver, init_K=400, max_K=3500, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = interpolate_seq(a_out, n_original) if n_original > n_lp else a_out
            print("  LP-FW improved: " + repr(best_score))

    # === Phase 3: Peak equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_eq = min(len(best_seq), 10000)
    a_eq = interpolate_seq(best_seq, n_eq) if n_eq != len(best_seq) else best_seq.copy()
    print("\n=== P3: Peak Equalize (n=" + str(n_eq) + ") ===")
    t3 = min(remaining * 0.15, 230.0)
    if t3 > 30:
        a_out, v_out = peak_equalize(normalize(a_eq), t3, n_targets=12, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = interpolate_seq(a_out, n_original) if n_original > n_eq else a_out
            print("  Equalize improved: " + repr(best_score))

    # === Phase 4: Subgradient FW kick + LP-FW again ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_sg = min(len(best_seq), 8000)
    a_sg = interpolate_seq(best_seq, n_sg) if n_sg != len(best_seq) else best_seq.copy()
    print("\n=== P4: Subgrad + LP-FW (n=" + str(n_sg) + ") ===")
    t4 = min(remaining * 0.20, 320.0)
    if t4 > 40:
        # Brief subgradient to perturb
        t_sg = min(t4 * 0.15, 30.0)
        a_sg_out, _ = subgradient_fw(normalize(a_sg), t_sg, n_steps=200, verbose=False)
        # Then LP-FW from the perturbed point
        t_lp = t4 - t_sg - 5
        if t_lp > 30:
            a_out, v_out = lp_fw_fast(normalize(a_sg_out), t_lp, solver, init_K=400, max_K=3000, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = interpolate_seq(a_out, n_original) if n_original > n_sg else a_out
                print("  Subgrad+LP improved: " + repr(best_score))

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P5: Multi-length ===")
    if remaining > 100:
        for n_try in [len(best_seq) + 1000, len(best_seq) - 1000, len(best_seq) + 3000]:
            if n_try < 2000 or n_try > 20000:
                continue
            if time.monotonic() - t_global > time_budget - 100:
                break
            a_try = interpolate_seq(best_seq, n_try)
            n_lp_try = min(n_try, 7000)
            a_try_lp = interpolate_seq(a_try, n_lp_try) if n_lp_try != n_try else a_try
            t_try = min(remaining * 0.25, 200.0)
            if t_try > 30:
                a_out, v_out = lp_fw_fast(normalize(a_try_lp), t_try, solver, init_K=300, max_K=2500, verbose=False)
                a_full = interpolate_seq(a_out, n_try) if n_try > n_lp_try else a_out
                sc = eval_safe(a_full)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_full
                    print("  n=" + str(n_try) + " improved: " + repr(best_score))

    # === Phase 6: Final peak equalization at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_fin = min(len(best_seq), 12000)
    a_fin = interpolate_seq(best_seq, n_fin) if n_fin != len(best_seq) else best_seq.copy()
    print("\n=== P6: Final Equalize (n=" + str(n_fin) + ") ===")
    t6 = min(remaining * 0.20, 250.0)
    if t6 > 20:
        a_out, v_out = peak_equalize(normalize(a_fin), t6, n_targets=15, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = interpolate_seq(a_out, len(best_seq)) if len(best_seq) > n_fin else a_out
            print("  Final Equalize improved: " + repr(best_score))

    # === Phase 7: Final LP-FW polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_pol = min(len(best_seq), 8000)
    a_pol = interpolate_seq(best_seq, n_pol) if n_pol != len(best_seq) else best_seq.copy()
    print("\n=== P7: Final LP (n=" + str(n_pol) + ") ===")
    t7 = min(remaining * 0.35, 400.0)
    if t7 > 30:
        a_out, v_out = lp_fw_fast(normalize(a_pol), t7, solver, init_K=500, max_K=4000, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = interpolate_seq(a_out, len(best_seq)) if len(best_seq) > n_pol else a_out
            print("  Final LP improved: " + repr(best_score))

    # === Phase 8: Last coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20 and best_seq is not None:
        n_cp = min(len(best_seq), 10000)
        a_cp = interpolate_seq(best_seq, n_cp) if n_cp != len(best_seq) else best_seq.copy()
        t8 = remaining - 10
        a_out, v_out = peak_equalize(normalize(a_cp), t8, n_targets=10, verbose=False)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = interpolate_seq(a_out, len(best_seq)) if len(best_seq) > n_cp else a_out
            print("  Final polish: " + repr(best_score))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + " n=" + str(len(best_seq)) + " t=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: fallback")
        n_fb = 10000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in np.asarray(best_seq, dtype=np.float64)]
    return result
