    candidates.append(np.cos(np.pi * (x - 0.5)) ** 2)
    candidates.append(1.0 / (0.3 + 0.7 * np.sin(np.pi * x) ** 2))
    candidates.append(np.exp(-2 * (x - 0.5) ** 2 / 0.05))
    candidates.append(0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x))
    f1 = np.exp(-8 * (x - 0.3) ** 2 / 0.04)
    f2 = np.exp(-8 * (x - 0.7) ** 2 / 0.04)
    candidates.append(0.5 * f1 + 0.5 * f2)
    mi = 5
    rr = rng.uniform(0.5, 1.5, mi)
    cs = np.linspace(0.15, 0.85, mi)
    ss = rng.uniform(0.04, 0.12, mi)
    mb = np.zeros(n)
    for i in range(mi):
        mb += rr[i] * np.exp(-((x - cs[i]) ** 2) / (2.0 * ss[i] ** 2))
    candidates.append(mb)
    best_a = None
    best_sc = float('inf')
    for c in candidates:
        cn = _norm1(c)
        sc = _eval_safe(cn)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = cn.copy()
    if best_a is None:
        best_a = _norm1(candidates[0])
    return best_a, best_sc


def _gs_iter(a_in, time_limit, alphabetas, verbose=False):
    t0 = time.monotonic()
    a = _norm1(a_in)
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    bc = np.ones(fs // 2 + 1) / size
    best_a = a.copy()
    bcg = _fast_conv(a).max()
    a_b = [1.0] + list(alphabetas)
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        if (it + 1) % len(a_b) == 0:
            a_b.append(a_b[-1] * 0.9)
        beta = a_b[it % len(a_b)]
        A = np.fft.rfft(a, fs)
        B = A.real ** 2 + A.imag ** 2
        B = B / B.sum()
        Bn = (1.0 - beta) * B + beta * bcg
        A2 = np.sqrt(np.maximum(Bn, 0)) * np.exp(1j * np.angle(A))
        an = np.fft.irfft(A2, fs)[:n]
        an = np.maximum(an, 0.0)
        s = an.sum()
        if s < 1e-12:
            break
        an = an / s
        m = _fast_conv(an).max()
        if m < bcg:
            bcg = m
            best_a = an.copy()
        a = an
    if verbose:
        print("    GS: J=" + repr(2.0 * n * bcg) + " it=" + str(it))
    return best_a, bcg


def _build_lp_mat(y, aklass, n):
    K = len(aklass)
    if K == 0:
        return np.zeros((0, n))
    C = np.zeros((K, n))
    for j in range(K):
        k = int(aklass[j])
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        if i0 < i1:
            idx = np.arange(i0, i1)
            C[j, idx] = y[k - idx]
    return C


def _lp_fw(a0, time_limit, solver, iK, xK, vrb=False):
    t0 = time.monotonic()
    n = len(a0)
    y = _norm1(a0)
    bay = y.copy()
    yc = _fast_conv(y)
    bm = yc.max()
    aset = sorted(np.argsort(yc)[::-1][:iK].tolist())
    ni = 0
    mxni = 400
    ic = 0
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 3.0:
            break
        if ni > mxni:
            break
        ic += 1
        yc = _fast_conv(y)
        Mc = yc.max()
        if Mc < bm - 1e-14:
            bm = Mc
            bay = y.copy()
            ni = 0
        else:
            ni += 1
        if len(aset) > xK:
            mg = np.array([Mc - yc[k] for k in aset])
            ki = np.argsort(mg)[:int(xK * 0.6)]
            ab = list(aset[ki])
            vor = np.where(yc > Mc - 1e-7)[0]
            for vv in vor[:50]:
                vvi = int(vv)
                if vvi not in ab:
                    ab.append(vvi)
            aset = sorted(set(ab))[:xK]
        K = len(aset)
        if K < 5:
            aset = sorted(np.argsort(yc)[::-1][:iK].tolist())
            K = iK
        Cm = _build_lp_mat(y, aset, n)
        gv = cp.Variable(n, nonneg=True)
        Mv = cp.Variable()
        cons = [cp.sum(gv) == 1, -Mv + Cm @ gv <= 0]
        prb = cp.Problem,cp.Minimize(Mv), cons)
        try:
            prb.solve(solver=solver, verbose=False)
        except Exception:
            aset = aset[:max(50, K // 3)]
            continue
        if prb.status not in ('optimal', 'optimal_inaccurate'):
            continue
        gval = np.maximum(np.asarray(gv.value, dtype=np.float64), 0.0)
        gs = gval.sum()
        if gs < 1e-14:
            continue
        gd = gval / gs
        fs2 = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(y, fs2)
        Gf = np.fft.rfft(gd, fs2)
        cconv = np.fft.irfft(Yf * Gf, fs2)[:2 * n - 1]
        vor = np.where(cconv > Mc + 1e-9)[0]
        if len(vor) > 0:
            va = cconv[vor] - Mc
            tv = vor[np.argsort(va)[::-1][:80]]
            ns = set(aset)
            for kk in tv:
                ns.add(int(kk))
            aset = sorted(ns)[:xK]
        ba = 1.0 / (ic + 2.0)
        bM = _fast_conv(y).max()
        alphs = [1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.1, 0.05,
                 max(0.03, 2.0 / (ic + 2)), max(0.01, 1.0 / (ic + 1))]
        for al in alphs:
            tr = (1.0 - al) * y + al * gd
            tr = np.maximum(tr, 0.0)
            st = tr.sum()
            if st < 1e-14:
                continue
            tr = tr / st
            Mt = _fast_conv(tr).max()
            if Mt < bM:
                bM = Mt
                ba = al
        if ic <= 2:
            y = gd.copy()
        else:
            y = (1.0 - ba) * y + ba * gd
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-14:
                y = bay.copy()
            else:
                y = y / sy
        if vrb and ic % 100 == 0:
            el = time.monotonic() - t0
            print("    LP-FW it=" + str(ic) + " (" + str(round(el, 1)) + "s) J=" + repr(2.0 * n * bm) + " |S|=" + str(len(aset)))
    if vrb:
        el = time.monotonic() - t0
        print("    LP-FW done (" + str(round(el, 1)) + "s) J=" + repr(2.0 * n * bm) + " it=" + str(ic))
    return bay, bm


def _nesterov(a0, time_limit, vrb=False):
    t0 = time.monotonic()
    n = len(a0)
    y = _norm1(a0)
    ba = y.copy()
    bm = _fast_conv(y).max()
    rng = np.random.default_rng(123)
    x = y.copy()
    vel = np.zeros(n)
    mom = 0.65
    eta0 = 0.06
    stall = 0
    mxs = 3000
    for t in range(1, 3000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > mxs:
            break
        eta = eta0 / (np.sqrt(t) + 1.0)
        yl = x + mom * vel
        yl = np.maximum(yl, 1e-12)
        sl = yl.sum()
        if sl < 1e-12:
            yl = x.copy()
        else:
            yl = yl / sl
        ac = _fast_conv(yl)
        kma = int(np.argmax(ac))
        grad = np.zeros(n)
        i0 = max(0, kma - n + 1)
        i1 = min(n, kma + 1)
        if i0 < i1:
            idx = np.arange(i0, i1)
            grad[idx] = yl[kma - idx]
        ps = 0.01 / (1.0 + t / 500.0)
        grad += ps * rng.standard_normal(n) * yl
        la = np.log(np.maximum(y, 1e-30)) - eta * grad
        ml = la.max()
        an = np.exp(la - ml)
        an = np.maximum(an, 1e-30)
        sn = an.sum()
        if sn < 1e-15:
            an = y.copy()
        else:
            an = an / sn
        vel = mom * vel + (an - x)
        x = an
        y = x
        Mn = _fast_conv(x).max()
        if Mn < bm - 1e-14:
            bm = Mn
            ba = x.copy()
            stall = 0
        else:
            stall += 1
        if stall > 800 and stall % 800 == 0:
            noise = rng.normal(0, 0.03, n) * ba
            xp = ba + noise
            xp = np.maximum(xp, 1e-10)
            sp = xp.sum()
            if sp > 1e-12:
                x = xp / sp
                vel = np.zeros(n)
            eta0 = eta0 * 1.2
        if vrb and t % 5000 == 0:
            el = time.monotonic() - t0
            print("    Nest t=" + str(t) + " (" + str(round(el, 1)) + "s) J=" + repr(2.0 * n * bm))
    if vrb:
        el = time.monotonic() - t0
        print("    Nest done (" + str(round(el, 1)) + "s) J=" + repr(2.0 * n * bm))
    return ba, bm


def _coord_polish(a0, time_limit, rseed=77, vrb=False):
    t0 = time.monotonic()
    a = _norm1(a0)
    ba = a.copy()
    bm = _fast_conv(a).max()
    n = len(a)
    rng = np.random.default_rng(rseed)
    stall = 0
    mxstall = max(300, int(time_limit * 2))
    ssc = [0.003, 0.006, 0.01, 0.015, 0.025, 0.04]
    while time.monotonic() - t0 < time_limit and stall < mxstall:
        ac = _fast_conv(a)
        nt = min(25, len(ac))
        tk = np.argsort(ac)[::-1][:nt]
        imp = False
        for kk in tk[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(kk)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            srcv = a[src]
            if srcv < 1e-12:
                continue
            scores = np.zeros(n)
            for k2 in tk[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
            inpk = np.zeros(n, dtype=bool)
            for k2 in tk[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    inpk[s2:e2 + 1] = True
            cand = np.where(~inpk)[0]
            if len(cand) < n // 5:
                cand = np.arange(n)
            cs = scores[cand]
            dst = int(cand[np.argmin(cs)])
            if dst == src:
                continue
            found = False
            for fr in ssc:
                amt = srcv * fr
                if amt < 1e-15:
                    continue
                at = a.copy()
                at[src] -= amt
                at[dst] += amt
                at = np.maximum(at, 0.0)
                st = at.sum()
                if st < 1e-15:
                    continue
                at = at / st
                Mt = _fast_conv(at).max()
                if Mt < bm - 1e-15:
                    a = at
                    bm = Mt
                    ba = a.copy()
                    imp = True
                    found = True
                    stall = 0
                    break
            if not found:
                jj = int(rng.integers(n))
                if jj != src and a[src] > 1e-10:
                    fr = rng.uniform(0.002, 0.01)
                    a[src] -= a[src] * fr
                    a[jj] += a[jj] + a[src] * fr
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = _fast_conv(a).max()
                        if Mn < bm:
                            bm = Mn
                            ba = a.copy()
                            stall = 0
                            imp = True
        if not imp:
            stall += 1
            if stall % 200 == 0:
                for _r in range(8):
                    i1 = int(rng.integers(n))
                    i2 = int(rng.integers(n))
                    if i1 != i2 and a[i1] > 1e-10:
                        fr = rng.uniform(0.002, 0.015)
                        amt = a[i1] * fr
                        a[i1] -= amt
                        a[i2] += amt
                        a = np.maximum(a, 0.0)
                        st = a.sum()
                        if st > 1e-15:
                            a = a / st
                            Mn = _fast_conv(a).max()
                            if Mn < bm:
                                bm = Mn
                                ba = a.copy()
                                stall = 0
        if vrb and stall % 400 == 0:
            el = time.monotonic() - t0
            print("    Coord (" + str(round(el, 1)) + "s) J=" + repr(2.0 * n * bm) + " st=" + str(stall))
    return ba, bm


def _try_upd(bs, bsc, aa, lbl, tg):
    if aa is None:
        return bs, bsc
    aa = np.maximum(np.asarray(aa, dtype=np.float64), 0.0)
    s = aa.sum()
    if s < 0.01:
        return bs, bsc
    an = aa / s
    sc = _eval_safe(an)
    if np.isfinite(sc) and sc < bsc:
        bsc = sc
        bs = an.copy()
        el = time.monotonic() - tg
        print("*** NEW BEST [" + lbl + "]: " + repr(sc) + " n=" + str(len(aa)) + " t=" + str(round(el, 1)) + "s")
    return bs, bsc


def propose_candidate(budget_s=1800.0):
    t0 = time.monotonic()
    T = min(float(budget_s) - 60.0, 1740.0)
    print("Budget: " + repr(T) + "s")

    def _rem():
        return T - (time.monotonic() - t0)

    solver = _get_solver()

    # Phase 0: diverse init
    print("\n=== P0: Diverse init ===")
    n0 = 5000
    a1, s1 = _diverse_init(n0, 0)
    best_seq, best_score = a1, s1
    a2, s2 = _diverse_init(n0 + 1000, 7)
    if s2 < best_score:
        best_score = s2
        best_seq = a2.copy()
    n0 = len(best_seq)
    print("  Best init: n=" + str(n0) + " s=" + repr(best_score))

    # Phase 1: GS on best
    print("\n=== P1: GS (n=" + str(n0) + ") ===")
    t1 = min(_rem() * 0.05, 70.0)
    if t1 > 15:
        ao, _ = _gs_iter(best_seq, t1, [0.03, 0.1, 0.3], True)
        best_seq, best_score = _try_upd(best_seq, best_score, ao, "gs1", t0)

    # Phase 2: LP-FW round 1
    print("\n=== P2: LP-FW 1 (n=" + str(len(best_seq)) + ") ===")
    t2 = min(_rem() * 0.16, 250.0)
    if t2 > 40:
        ao, _ = _lp_fw(_norm1(best_seq), t2, solver, 500, 3500, True)
        best_seq, best_score = _try_upd(best_seq, best_score, ao, "lpfw1", t0)

    # Phase 3: Nesterov
    print("\n=== P3: Nesterov ===")
    t3 = min(_rem() * 0.09, 130.0)
    if t3 > 25:
        ao, _ = _nesterov(_norm1(best_seq), t3, False)
        best_seq, best_score = _try_upd(best_seq, best_score, ao, "nest1", t0)

    # Phase 4: LP-FW round 2
    print("\n=== P4: LP-FW 2 ===")
    t4 = min(_rem() * 0.16, 250.0)
    if t4 > 40:
        ao, _ = _lp_fw(_norm1(best_seq), t4, solver, 700, 4500, True)
        best_seq, best_score = _try_upd(best_seq, best_score, ao, "lpfw2", t0)

    # Phase 5: GS refinement
    print("\n=== P5: GS ref ===")
    t5 = min(_rem() * 0.04, 50.0)
    if t5 > 10:
        ao, _ = _gs_iter(best_seq, t5, [0.02, 0.06, 0.15], False)
        best_seq, best_score = _try_upd(best_seq, best_score, ao, "gs2", t0)

    # Phase 6: Multi-n
    print("\n=== P6: Multi-n ===")
    nb = len(best_seq)
    try_ns = sorted(set([nb - 500, nb - 1000, nb + 500, nb + 1000, nb + 2000]))
    try_ns = [l for l in try_ns if 2000 <= l <= 12000]
    try_ns.sort(key=lambda l: abs(l - nb))
    count = 0
    for nn in try_ns[:3]:
        rt = _rem()
        if rt < 100:
            break
        count += 1
        print("  Multi-n: n=" + str(nn) + " rem=" + str(round(rt, 1)) + "s")
        al = _interp_seq(best_seq, nn)
        t6 = min(rt * 0.22, 160.0)
        if t6 > 30 and nn <= 10000:
            ao, _ = _lp_fw(al, t6, solver, 400, 3000, False)
            best_seq, best_score = _try_upd(best_seq, best_score, ao, "lpfw_mn" + str(nn), t0)
        rt = _rem()
        if rt > 40 and len(best_seq) <= 9000:
            t7 = min(rt * 0.10, 80.0)
            if t7 > 20:
                ao, _ = _nesterov(_norm1(best_seq), t7, False)
                best_seq, best_score = _try_upd(best_seq, best_score, ao, "nest_mn" + str(nn), t0)

    # Phase 7: LP-FW round 3
    print("\n=== P7: LP-FW 3 ===")
    t8 = min(_rem() * 0.16, 230.0)
    if t8 > 35 and len(best_seq) <= 10000:
        ao, _ = _lp_fw(_norm1(best_seq), t8, solver, 800, 5000, True)
        best_seq, best_score = _try_upd(best_seq, best_score, ao, "lpfw3", t0)

    # Phase 8: Coordinate polish
    print("\n=== P8: Coord ===")
    t9 = min(_rem() * 0.12, 150.0)
    if t9 > 20:
        ao, _ = _coord_polish(_norm1(best_seq), t9, 99, True)
        best_seq, best_score = _try_upd(best_seq, best_score, ao, "coord1", t0)

    # Phase 9: Final LP-FW
    print("\n=== P9: LP-FW final ===")
    t10 = min(_rem() * 0.22, 180.0)
    if t10 > 30 and len(best_seq) <= 10000:
        ao, _ = _lp_fw(_norm1(best_seq), t10, solver, 900, 5500, True)
        best_seq, best_score = _try_upd(best_seq, best_score, ao, "lpfwF", t0)

    # Phase 10: Final polish
    rt = _rem()
    if rt > 20 and len(best_seq) > 0:
        ao, _ = _coord_polish(_norm1(best_seq), min(rt * 0.7, 60.0), 2024, False)
        best_seq, best_score = _try_upd(best_seq, best_score, ao, "coordF", t0)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        bs = _eval_safe(best_seq)
        if np.isfinite(bs):
            best_score = bs
        el = time.monotonic() - t0
        print("\nFINAL: score=" + repr(best_score) + " n=" + str(len(best_seq)) + " t=" + str(round(el, 1)) + "s")
    else:
        nfb = 5000
        xx = np.linspace(0, 1, nfb)
        best_seq = _norm1(np.sin(np.pi * xx) ** 2)
        best_score = _eval_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    out = [float(x) for x in best_seq]
    for i in range(len(out)):
        if out[i] < 0:
            out[i] = 0.0
        if out[i] > 1000.0:
            out[i] = 1000.0
    return out
