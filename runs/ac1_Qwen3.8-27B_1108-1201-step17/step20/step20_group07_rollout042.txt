def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_gradient(p, a, fft_size):
    """
    Compute gradient of smoothed max objective.
    grad[i] = (2/beta) * sum_j p[i+j] * a[j]  (cross-correlation)
    Using FFT: grad = ifft(FFT(p) * conj(FFT(a)))
    """
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    # Cross-correlation: ifft(FFT(p) * conj(FFT(a)))
    # For rfft, we need to handle the conjugate properly
    # Full FFT approach for correctness:
    P_full = np.fft.fft(p, fft_size)
    A_full = np.fft.fft(a, fft_size)
    corr = np.fft.ifft(P_full * np.conj(A_full)).real
    return corr[:len(a)]


def fista_optimize(a0, time_limit, beta_start=10.0, beta_end=500.0,

    stall = 0

        t += 1
        b = fast_conv(y)
        b_max = b.max()

            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Compute peak weights (top-k softmax)
        if top_k >= len(b):

            p = p / p.sum()
        else:
            idx_top = np.argsort(b)[::-1][:top_k]
            p = np.zeros_like(b)
            p[idx_top] = np.exp(beta * (b[idx_top] - b[idx_top][0]))
            p = p / p.sum()

        # Compute gradient via FFT cross-correlation
        grad = compute_gradient(p, y, fft_size)
        # Center gradient (for sum constraint)

        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:

        eta = 0.5 / (grad_norm * grad_norm)

        # Nesterov update
        alpha_new = (1.0 + np.sqrt(1.0 + 4.0 * prev_t * prev_t)) / 2.0

            x = y_new
        else:

            y_new = simplex_project(y_new)

        y = y_new
        prev_t = alpha_new

        # Adaptive beta
        if t % 30 == 0:
            beta = min(beta * beta_growth, beta_end)

            x = x_r
            y = x_r
            prev_t = 1.0

def fp_refine(a0, time_limit, solver, verbose=True):
    """Frank-Wolfe with LP oracle for direction finding."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    t = 0
    no_improve = 0
    max_active = 2000

    while time.monotonic() - t0 < time_limit and no_improve < 500:
        t += 1
        b = fast_conv(a)
        M_curr = b.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Active set: top constraints
        active = np.argsort(b)[::-1][:min(max_active, len(b))]
        K = len(active)
        if K == 0:
            break

        # Build LP constraints: (a * g)[k] <= M for k in active
        # (a*g)[k] = sum_i a[i]*g[k-i], linear in g
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(active):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = a[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Line search
        alpha = 2.0 / (t + 2.0)
        best_alpha = alpha
        best_trial_M = float('inf')
        for test_alpha in [alpha, alpha * 0.5, 1.0 / (t + 1)]:
            trial = (1.0 - test_alpha) * a + test_alpha * g_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            M_trial = fast_conv(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = test_alpha

        a_new = (1.0 - best_alpha) * a + best_alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        st = a_new.sum()
        if st > 1e-15:
            a_new = a_new / st
        a = a_new

        # Add violations to active set (implicit via recomputation)
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    FP t={} ({}s): J={}".format(t, round(elapsed, 1), repr(J)))

    return best_a, best_m


def parametric_init(n, rng):
    """Generate diverse parametric initializations."""
    x = np.linspace(0.0, 1.0, n)
    candidates = []

    # Power laws
    for alpha in [0.5, 0.8, 1.2, 1.5, 2.0]:
        for beta_p in [0.5, 0.8, 1.2, 1.5, 2.0]:
            a = (x + 0.005) ** alpha * (1.0 - x + 0.005) ** beta_p
            candidates.append(norm_sum1(a))

    # Two-bump with gap
    for gap_frac in [0.15, 0.25, 0.35]:
        gap_start = int(n * (0.5 - gap_frac / 2))
        gap_end = int(n * (0.5 + gap_frac / 2))
        a = np.ones(n)
        a[gap_start:gap_end] = 0.0
        candidates.append(norm_sum1(a))

    # Gaussian mixtures
    for c1 in [0.3, 0.5, 0.7]:
        for s1 in [0.08, 0.12, 0.18]:
            a = np.exp(-((x - c1) ** 2) / (2 * s1 ** 2))
            candidates.append(norm_sum1(a))

    # Random
    for _ in range(10):
        a = rng.random(n) ** rng.uniform(0.3, 4.0)
        candidates.append(norm_sum1(a))

    # Smoothed random (low-pass filtered)
    for _ in range(5):
        a_raw = rng.random(n)
        # Simple moving average for smoothing
        kernel_size = max(3, n // 100)
        kernel = np.ones(kernel_size) / kernel_size
        a_smooth = np.convolve(a_raw, kernel, mode='same')
        candidates.append(norm_sum1(a_smooth))

    return candidates


def try_update(best_seq, best_score, a, label, t_global):

    return best_seq, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def multi_start(num_starts, n, rng):

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                n_base = len(parent)
                print("Parent: n={}, score={}".format(n_base, repr(p_score)))
    except Exception:
        pass

    if best_seq is None:
        x = np.linspace(0, 1, n_base)
        a_raw = (x + 0.01) ** 1.0
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)
        print("Default init: n={}, score={}".format(n_base, repr(best_score)))

    n0 = len(best_seq)

    # === Phase 1: Multi-start FISTA ===
    print("\n=== Phase 1: Multi-start FISTA (n={}) ===".format(n0))
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 500.0)

    if t1 > 60:
        # Sub-allocate time to multiple starts
        n_starts = 3
        time_per_start = t1 / n_starts

        # Start 1: from parent/best
        t1a = min(time_per_start, remaining - 120)
        if t1a > 30:
            a_out, _ = fista_optimize(best_seq, t1a, beta_start=15.0, beta_end=400.0,
                                      beta_growth=1.007, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista1_n{}".format(n0), t_global)

        # Start 2: from best parametric
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 100:
            t1b = min(remaining * 0.3, 200.0)
            if t1b > 30:
                a_init, m_init, _ = multi_start(2, n0, rng)
                if m_init < fast_conv(best_seq).max():
                    best_seq, best_score = try_update(best_seq, best_score, a_init, "param_init", t_global)
                a_out, _ = fista_optimize(a_init, t1b, beta_start=10.0, beta_end=350.0,
                                          beta_growth=1.006, verbose=True)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "fista2_n{}".format(n0), t_global)

        # Start 3: from random
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 100:
            t1c = min(remaining * 0.25, 150.0)
            if t1c > 30:
                a_rand = norm_sum1(rng.random(n0) ** rng.uniform(0.5, 3.0))
                a_out, _ = fista_optimize(a_rand, t1c, beta_start=8.0, beta_end=300.0,
                                          beta_growth=1.006, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "fista3_n{}".format(n0), t_global)

    # === Phase 2: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Peak shaving (n={}) ===".format(n_best))
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_out, _ = peak_shaving_global(best_seq, t2, top_k=25, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps1_n{}".format(n_best), t_global)

    # === Phase 3: FISTA round 2 (higher beta) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: FISTA round 2 (n={}) ===".format(n_best))
    t3 = min(remaining * 0.2, 300.0)
    if t3 > 30:
        a_out, _ = fista_optimize(best_seq, t3, beta_start=30.0, beta_end=800.0,
                                  beta_growth=1.005, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_r2_n{}".format(n_best), t_global)

    # === Phase 4: Peak shaving round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Peak shaving 2 (n={}) ===".format(n_best))
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 20:
        a_out, _ = peak_shaving_global(best_seq, t4, top_k=20, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps2_n{}".format(n_best), t_global)

    # === Phase 5: FP refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: FP/LP (n={}) ===".format(n_best))
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 40 and n_best <= 8000:
        a_out, _ = fp_refine(best_seq, t5, solver, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fp_n{}".format(n_best), t_global)

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-length (base n={}) ===".format(n_best))
    nearby = []
    for dn in [-300, 200, 500, -500, 800]:
        nl = n_best + dn
        if 5000 < nl <= 9000 and nl != n_best:
            nearby.append(nl)

    for new_n in nearby[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("  Trying n={}".format(new_n))
        a_interp = interpolate_sequence(best_seq, new_n)

        # Quick FISTA
        t6 = min(remaining * 0.3, 180.0)
        if t6 > 30:
            a_out, _ = fista_optimize(a_interp, t6, beta_start=20.0, beta_end=500.0,
                                      beta_growth=1.006, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "multilens_n{}".format(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and len(best_seq) == new_n:
            t6b = min(remaining * 0.2, 100.0)
            if t6b > 20:
                a_out, _ = peak_shaving_global(best_seq, t6b, top_k=15, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "mnl_ps_n{}".format(new_n), t_global)

    # === Phase 7: Final intensive optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final optimization (n={}) ===".format(n_best))

    # 7a: FISTA with high beta
    if remaining > 60:
        t7a = min(remaining * 0.35, 200.0)
        if t7a > 30:
            a_out, _ = fista_optimize(best_seq, t7a, beta_start=50.0, beta_end=1500.0,
                                      beta_growth=1.004, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_fista", t_global)

    # 7b: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t7b = min(remaining * 0.3, 150.0)
        if t7b > 20:
            a_out, _ = peak_shaving_global(best_seq, t7b, top_k=15, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_ps", t_global)

    # 7c: FP
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50 and len(best_seq) <= 8000:
        t7c = min(remaining * 0.4, 200.0)
        if t7c > 30:
            a_out, _ = fp_refine(best_seq, t7c, solver, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_fp", t_global)

    # 7d: One more peak shave
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t7d = min(remaining * 0.5, 120.0)
        if t7d > 15:
            a_out, _ = peak_shaving_global(best_seq, t7d, top_k=10, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_ps2", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={}, n={}, time={:.1f}s".format(repr(best_score), len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence!")
        x = np.linspace(0, 1, 7000)
        best_seq = norm_sum1(x ** 1.0)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
