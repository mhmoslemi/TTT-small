def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

def simplex_project(z):
    """Project onto probability simplex."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(a, beta, top_k=None):
    """Compute smoothed gradient of max autocorrelation using log-sum-exp."""
    n = len(a)
    b = fast_autocorr(a)
    size = 2 * n - 1

    if top_k is not None and top_k < size and top_k > 0:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
    else:
        b_use = b

    b_max = b_use.max()
    if b_max <= 0:
        return np.zeros(n), b

    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s

    # Gradient: d(max_b)/d(a[i]) approximated via soft max
    # b[k] = sum_j a[j]*a[k-j], so db[k]/da[i] = a[k-i] (for valid indices)
    # grad[i] = sum_k p[k] * db[k]/da[i]
    # db[k]/da[i] = a[k-i] if 0<=k-i<n, else 0
    # This is a cross-correlation of p with a
    a_rev = a[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1]
    return grad, b


def nesterov_optimize(a0, time_limit, beta_start=30.0, beta_end=1000.0,
                      eta_init=0.5, top_k_ratio=0.02,
                      restart_every=1000, seed=42):
    """Nesterov accelerated gradient descent with log-sum-exp smoothing."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    eta = eta_init

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, b = compute_smoothed_grad(y, beta, top_k)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            x_new = y
            max_new = fast_autocorr(x_new).max()
        else:
            grad_normed = grad / gnorm
            eta_trial = eta
            found = False
            y_max = fast_autocorr(y).max()
            for _ in range(8):
                z = y - eta_trial * grad_normed
                x_new = simplex_project(z)
                max_new = fast_autocorr(x_new).max()
                if max_new <= y_max + 1e-14:
                    found = True
                    break
                eta_trial *= 0.5
            if not found:
                x_new = x
                max_new = fast_autocorr(x).max()
                eta = max(eta_trial, 0.01)

        if max_new < best_m - 1e-16:
            best_m = max_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.05, 3.0)
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 40 == 0:
            beta = min(beta * 1.01, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = rng.integers(6)
            if strategy == 0:
                noise = 0.02 * (1.0 + 0.5 * rng.random())
                pert = rng.normal(0, noise, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                alpha_r = rng.uniform(0.1, 0.4)
                x_r = (1 - alpha_r) * best_a + alpha_r * rng.dirichlet(np.ones(n) * 0.5)
            elif strategy == 2:
                x_r = best_a * np.exp(0.02 * rng.standard_normal(n))
            elif strategy == 3:
                mask = rng.random(n) > 0.5
                x_r = best_a.copy()
                x_r[mask] *= rng.uniform(0.5, 0.8)
                x_r[~mask] *= rng.uniform(1.2, 1.5)
            elif strategy == 4:
                shift = int(rng.integers(1, max(2, n // 20)))
                x_r = np.roll(best_a, shift)
            else:
                # Spectral perturbation
                fft_size = 1 << (2 * n - 2).bit_length()
                A = np.fft.rfft(best_a, fft_size)
                A[1:] *= np.exp(1j * 0.03 * rng.standard_normal(len(A) - 1))
                x_r = np.fft.irfft(A, fft_size)[:n]
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 10.0)
            eta = eta_init

    return best_a, best_m


def spectral_flatten(a0, time_limit, seed=777, alpha_init=0.05):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    alpha = alpha_init

    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        target = power[1:].mean() if len(power) > 1 else power[0]

        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.0:
                red = 1.0 - alpha * min(ratio - 1.0, 2.0)
                A_new[k] *= max(0.2, math.sqrt(red))
            elif ratio < 0.3:
                inc = 1.0 + alpha * 0.5 * (0.3 - ratio)
                A_new[k] *= min(3.0, math.sqrt(inc))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-17:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 100 == 0:
            # Random phase perturbation
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.05
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial

        if it % 300 == 0:
            alpha = max(0.01, alpha * 0.97)

    return best_a, best_m


def coordinate_descent(a0, time_limit, seed=321):
    """Targeted coordinate descent: adjust elements that contribute most to peak."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 3000:
        b = fast_autocorr(a)
        size = 2 * n - 1
        max_k = int(np.argmax(b))
        M = b[max_k]

        # Find elements contributing most to the peak
        k = max_k
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            # Peak at lag 0
            top_idx = np.argsort(a)[::-1][:20]
        else:
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            top_local = np.argsort(contrib)[::-1][:10]
            top_idx = idx[top_local]

        for i in top_idx[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            i = int(i)
            if a[i] < 1e-15:
                continue

            # Try moving some mass from i to a random location
            for frac in [0.01, 0.02, 0.05, 0.1]:
                amt = a[i] * frac
                if amt < 1e-15:
                    continue
                best_trial = None
                best_trial_m = best_m
                for _ in range(10):
                    j = int(rng.integers(n))
                    if j == i:
                        continue
                    a_t = a.copy()
                    a_t[i] -= amt
                    a_t[j] += amt
                    a_t = np.maximum(a_t, 0.0)
                    s_t = a_t.sum()
                    if s_t < 1e-15:
                        continue
                    a_t = a_t / s_t
                    m_t = fast_autocorr(a_t).max()
                    if m_t < best_trial_m:
                        best_trial_m = m_t
                        best_trial = a_t.copy()

                if best_trial is not None and best_trial_m < best_m - 1e-17:
                    a = best_trial
                    best_m = best_trial_m
                    best_a = a.copy()
                    stall = 0
                    break
            else:
                continue
            break
        else:
            stall += 1
            if stall > 200:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    amt = a[i] * rng.uniform(0.005, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                        m = fast_autocorr(a).max()
                        if m < best_m:
                            best_m = m
                            best_a = a.copy()
                            stall = 0

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, initial_K=300, max_K=1500):
    """LP-Frank-Wolfe method for minimizing max autocorrelation."""
    if not _HAS_CVXPY or solver is None:
        return a0, fast_autocorr(a0).max()

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 400:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:15]]
            for k in top_v:
                active_set.add(int(k))

        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        for alpha_cand in [best_alpha, 0.5 / (t + 1), 0.1, 0.2, 0.05]:
            alpha_cand = max(0.01, min(1.0, alpha_cand))
            trial = (1.0 - alpha_cand) * y + alpha_cand * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha_cand

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

    return best_a, best_m


def multiple_mass_transfer(a0, time_limit, seed=456):
    """Move mass from high-contribution positions to low-contribution ones."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 2000:
        b = fast_autocorr(a)
        M = b.max()
        max_k = int(np.argmax(b))

        # Identify "hot" positions (contributing to peak)
        if max_k < n:
            js = max(0, max_k - n + 1)
            je = min(n - 1, max_k)
            if js <= je:
                idx = np.arange(js, je + 1)
                contrib = a[idx] * a[max_k - idx]
            else:
                contrib = a ** 2
        else:
            offset = max_k - (n - 1)
            idx = np.arange(offset, n)
            partner = np.arange(max_k - idx)
            contrib = a[idx] * a[partner]

        hot_positions = np.argsort(contrib)[::-1][:min(20, len(contrib))]
        cold_positions = np.argsort(contrib)[:min(20, len(contrib))]

        improved = False
        for i in hot_positions[:6]:
            if time.monotonic() - t0 > time_limit:
                break
            i = int(i)
            if a[i] < 1e-14:
                continue
            for j in cold_positions[:6]:
                j = int(j)
                if j == i:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    amt = a[i] * frac
                    a_t = a.copy()
                    a_t[i] -= amt
                    a_t[j] += amt
                    a_t = np.maximum(a_t, 0.0)
                    s_t = a_t.sum()
                    if s_t < 1e-15:
                        continue
                    a_t = a_t / s_t
                    m_t = fast_autocorr(a_t).max()
                    if m_t < best_m - 1e-17:
                        a = a_t
                        best_m = m_t
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 100:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    amt = a[i] * rng.uniform(0.01, 0.08)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                        m = fast_autocorr(a).max()
                        if m < best_m:
                            best_m = m
                            best_a = a.copy()
                            stall = 0

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best if candidate is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) +
              " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Multi-strategy optimization combining spectral methods, Nesterov,
    coordinate descent, LP-FW, and multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Time budget: " + str(round(time_budget, 1)) + "s")

    best_seq = None
    best_score = float('inf')

    # Try parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + str(round(p_score, 6)))
    except Exception:
        parent = None

    # Try uniform as baseline
    for n_try in [len(best_seq) if best_seq is not None else 5000]:
        unif = np.ones(n_try) / n_try
        u_score = evaluate_safe(unif)
        if np.isfinite(u_score) and u_score < best_score:
            best_score = u_score
            best_seq = unif.copy()
            print("Uniform n=" + str(n_try) + " score=" + str(round(u_score, 6)))

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + str(round(best_score, 6)))

    n0 = len(best_seq)
    print("Start: n=" + str(n0) + ", score=" + str(round(best_score, 6)))

    # Phase 1: Spectral flattening
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Spectral (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 10:
        a_out, _ = spectral_flatten(best_seq, t1, seed=777)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec1", t_global)

    # Phase 2: Nesterov (primary)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P2: Nesterov (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_out, _ = nesterov_optimize(best_seq, t2,
                                     beta_start=30.0, beta_end=800.0,
                                     eta_init=0.5, top_k_ratio=0.02,
                                     restart_every=800, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 3: LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P3: LP-FW (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20 and n_best <= 8000 and solver is not None:
        a_out, _ = lp_frank_wolfe(best_seq, t3, solver,
                                   initial_K=300, max_K=1500)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P4: CoordDescent (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 10:
        a_out, _ = coordinate_descent(best_seq, t4, seed=321)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "cd1", t_global)

    # Phase 5: Multiple mass transfer
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P5: MassTransfer (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 10:
        a_out, _ = multiple_mass_transfer(best_seq, t5, seed=456)
        best_seq, best_score = try_update(best_seq, best_score, "mt1", t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P6: Multi-length (base n=" + str(n_best) + ") ===")

    longer_lengths = [n_best + 500, n_best + 1000, n_best + 2000]
    longer_lengths = [l for l in longer_lengths if l <= 10000 and l > n_best]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print("  -> n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.40, 200.0)
        if t6a > 20:
            a_out2, _ = nesterov_optimize(a_long, t6a,
                                          beta_start=25.0, beta_end=600.0,
                                          eta_init=0.4, top_k_ratio=0.015,
                                          restart_every=1000, seed=new_n % 7919)
            best_seq, best_score = try_update(best_seq, best_score, a_out2,
                                              "nest_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 8000 and solver is not None:
            t6b = min(remaining * 0.40, 120.0)
            if t6b > 15:
                a_long2 = interpolate_sequence(best_seq, new_n)
                a_out3, _ = lp_frank_wolfe(a_long2, t6b, solver,
                                            initial_K=250, max_K=1200)
                best_seq, best_score = try_update(best_seq, best_score, a_out3,
                                                  "lp_n" + str(new_n), t_global)

    # Phase 7: Second Nesterov with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P7: Nesterov r2 (n=" + str(n_best) + ") ===")
    if remaining > 40:
        t7 = min(remaining * 0.25, 250.0)
        if t7 > 20:
            a_out, _ = nesterov_optimize(best_seq, t7,
                                         beta_start=50.0, beta_end=1200.0,
                                         eta_init=0.3, top_k_ratio=0.01,
                                         restart_every=600, seed=999)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)

    # Phase 8: Spectral + coordinate combo
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P8: Spectral+CD (n=" + str(n_best) + ") ===")
    if remaining > 20:
        t8a = min(remaining * 0.20, 60.0)
        if t8a > 8:
            a_out, _ = spectral_flatten(best_seq, t8a, seed=888, alpha_init=0.03)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "spec2", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t8b = min(remaining * 0.25, 60.0)
            if t8b > 8:
                a_out, _ = multiple_mass_transfer(best_seq, t8b, seed=789)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "mt2", t_global)

    # Phase 9: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P9: Final polish (n=" + str(n_best) + ") ===")
    if remaining > 20:
        t9a = min(remaining * 0.30, 80.0)
        if t9a > 10:
            a_out, _ = nesterov_optimize(best_seq, t9a,
                                         beta_start=100.0, beta_end=2000.0,
                                         eta_init=0.2, top_k_ratio=0.005,
                                         restart_every=500, seed=555)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest3", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15 and n_best <= 8000 and solver is not None:
            t9b = min(remaining * 0.50, 80.0)
            if t9b > 8:
                a_out, _ = lp_frank_wolfe(best_seq, t9b, solver,
                                           initial_K=400, max_K=2000)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + str(round(best_score, 6)) +
              ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
