def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(arr.size, dtype=np.float64) / max(arr.size, 1)
    return arr / s

def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c

def compute_pressure(c, a, top_k):
    """Compute the pressure: how much does each coordinate contribute to the top-k convolution peaks."""
    n = a.size
    size = 2 * n - 1
    top_idx = np.argsort(c)[::-1][:min(top_k, size)]
    p = np.zeros(size, dtype=np.float64)
    p[top_idx] = 1.0 / len(top_idx)

    # Using FFT: pressure = cross-correlation of p and a
    # c[k] = sum_i a[i]*a[k-i]
    # dc[k]/da[j] = 2*a[k-j] (for the symmetric case)

def game_direction(a, c, active_idx, iterations=20, seed=0):
    """Zero-sum game to find the Frank-Wolfe direction."""
    n = a.size
    size = 2 * n - 1
    rng = np.random.default_rng(seed)
    K = len(active_idx)
    if K <= 0:
        return np.ones(n) / n
    
    # Initialize lambda (distribution over active constraints)
    logits = np.zeros(K, dtype=np.float64) + rng.normal(0, 0.1, K)
    
    for it in range(iterations):
        # Compute pressure from current lambda
        p = np.zeros(size, dtype=np.float64)
        exp_l = np.exp(logits - logits.max())
        lam = exp_l / exp_l.sum()
        p[active_idx] = lam
        
        N = next_pow2(3 * n - 2)
        A = np.fft.rfft(a, N)
        Pr = np.fft.rfft(p[::-1], N)
        conv = np.fft.irfft(A * Pr, N)
        pressure = 2.0 * conv[n-1:2*n-1]
        pressure = np.where(np.isfinite(pressure), pressure, 0.0)
        
        # g minimizes pressure (the "best response" of the column player)
        pmin = pressure.min()
        pmax = pressure.max()
        if pmax - pmin < 1e-30:
            g = np.ones(n) / n
        else:
            scaled = (pressure - pmin) / (pmax - pmin)

            w = np.where(np.isfinite(w), w, 0.0)
            w += 1e-15
            g = w / w.sum()
            g = 0.95 * g + 0.05 * (np.ones(n) / n)

        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmax = ha.max()
        hmin = ha.min()

            eta = 10.0 / (hmax - hmin)
            logits += eta * (ha - ha.mean())
    
    # Final lambda

    lam = exp_l / exp_l.sum()
    p = np.zeros(size, dtype=np.float64)
    p[active_idx] = lam
    N = next_pow2(3 * n - 2)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)

def frank_wolfe(a0, time_budget, active_max=2000, game_iters=20, verbose=True):

        t += 1
        time_left = time_budget - (time.monotonic() - t0)
        if time_left < 10:

        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        
        # Line search
        best_alpha = 0.0

            c = fft_conv(a)

def coordinate_polish(a0, time_budget, verbose=True):

        top_idx = np.argsort(c)[::-1][:top_k]
        p = np.zeros(2*n-1, dtype=np.float64)
        p[top_idx] = 1.0 / top_k

        A = np.fft.rfft(a, N)
        Pr = np.fft.rfft(p[::-1], N)

        pressure = 2.0 * conv[n-1:2*n-1]

        # Score for source: high pressure, high value
        # Score for destination: low pressure
        src_score = pressure + n * a  # want to move FROM these
        dst_score = -pressure  # want to move TO these
        
        n_src = min(5, n)
        n_dst = min(8, n)
        src_cands = np.argpartition(src_score, n_src-1)[:n_src] if n_src < n else np.arange(n)
        dst_cands = np.argpartition(dst_score, n_dst-1)[:n_dst] if n_dst < n else np.arange(n)

        improved = False

            s = int(s_idx)
            if a[s] < 1e-12:
                continue

                d = int(d_idx)
                if d == s:
                    continue

                    if 2*s < c_new.size:
                        c_new[2*s] -= tmove*tmove
                    if s+d < c_new.size:
                        c_new[s+d] -= 2*tmove*tmove
                    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
                    m_new = float(np.max(c_new))
                    if m_new < M - 1e-18:
                        if best_cand is None or m_new < best_cand[0]:
                            best_cand = (m_new, s, d, tmove)
                            improved = True

                        best_cand = (m_new, s, d, tmove)
                        improved = True
        
        if best_cand is not None:
            m_new, s, d, tmove = best_cand
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            s_sum = a.sum()
            a = a / s_sum
            c = fft_conv(a)
            M = float(np.max(c))
            if M < best_M - 1e-16:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and (stall == 0 and t <= 5):
                print(f"  Coord: M={best_M:.8f}")
        else:
            stall += 1
            if stall % 15 == 14:
                # Larger perturbation
                rng2 = np.random.default_rng(999 + stall)
                a = best_a * (1 + rng2.normal(0, 0.02, n))
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                c = fft_conv(a)
                M = float(np.max(c))
                stall = 0
    
    return best_a

def lse_objective(a, beta, mask=None):
    """Log-sum-exp smoothed objective."""
    n = a.size
    c = fft_conv(a)
    if mask is not None:
        cv = c[mask]
    else:
        cv = c
    if cv.size == 0:
        return 0.0, np.zeros(n)
    m = float(np.max(cv))
    shifted = beta * (cv - m)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0 or not math.isfinite(se):
        return m, np.zeros(n)
    p = e / se
    if mask is not None:
        pf = np.zeros(len(c), dtype=np.float64)
        pf[mask] = p
    else:
        pf = p.copy()
    # Gradient
    N = next_pow2(3*n-2)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(pf[::-1], N)
    conv = np.fft.irfft(A * Pr, N)
    grad = 2.0 * conv[n-1:2*n-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    f = m + math.log(se) / beta
    return f, grad

def lbfegs_optimize(a0, beta, maxfun=400, mask=None):
    """L-BFGS optimization on the log-sum-exp smoothed objective."""
    if not SCIPY_OK:
        return a0
    a = norm_sum1(a0)
    n = a.size
    if n <= 1:
        return a
    
    # Use log-coordinate parameterization
    # Actually, let's just do projected gradient / L-BFGS on the simplex
    # using the trick: parametrize a = softmax(u)
    u = np.full(n, math.log(n)/n)  # uniform start in log-space
    # Actually, better to start from current a
    loga = np.log(np.maximum(a, 1e-30))
    u = loga - loga.mean()  # center it
    
    def obj_grad(u_vec):
        u_vec = np.where(np.isfinite(u_vec), u_vec, 0.0)
        u_centered = u_vec - u_vec.mean()
        e = np.exp(u_centered)
        a_curr = e / e.sum()
        f, _ = lse_objective(a_curr, beta, mask)
        # Chain rule: da/du = a*(1 - a) for softmax
        _, ga = lse_objective(a_curr, beta, mask)
        mean_ga = float(np.dot(a_curr, ga))
        gu = a_curr * (ga - mean_ga)
        gu = np.where(np.isfinite(gu), gu, 0.0)
        return f, gu
    
    try:
        res = scipy.optimize.minimize(
            obj_grad, u, method='L-BFGS-B', jac=True,
            options={'maxiter': maxfun, 'ftol': 1e-12, 'gtol': 1e-8, 'disp': False}
        )
        if res.x is not None and res.x.size == n:
            u_opt = np.where(np.isfinite(res.x), res.x, 0.0)
            u_centered = u_opt - u_opt.mean()
            e = np.exp(u_centered)
            a_out = e / e.sum()
            a_out = norm_sum1(a_out)
            return a_out
    except Exception as e:
        print(f"    L-BFGS error: {e}")
    
    return a

def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0, 1, a.size)
    new_x = np.linspace(0, 1, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)
