def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    if K == 0:
        return np.zeros((0, n))
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2500, add_per_iter=40, verbose=True, seed=0):
    """LP-Frank-Wolfe method for minimizing max autocorrelation."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac_y = fast_convolve(y)
    best_m = ac_y.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 500
    iter_count = 0
    stagnation_count = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if stagnation_count > 5:
                if verbose:
                    print("    LP-FW: persistent stagnation, stopping at iter " + str(iter_count))
                break

            n_pert = max(30, len(active_set) // 4)
            keep = list(rng.choice(len(active_set), size=len(active_set) - n_pert, replace=False))
            new_adds = np.argsort(ac_curr)[::-1][:n_pert * 3]
            new_set = set([active_set[i] for i in keep])
            for k in new_adds:
                new_set.add(int(k))
            active_set = sorted(new_set)
            no_improve = 0
            stagnation_count += 1
            # Also try a random perturbation of y
            noise = rng.normal(0, 0.01, n) * y
            y_p = y + noise
            y_p = np.maximum(y_p, 0.0)
            sp = y_p.sum()
            if sp > 1e-12:
                y_p = y_p / sp
                M_p = fast_convolve(y_p).max()
                if M_p < best_m:
                    best_m = M_p
                    best_a = y_p.copy()
                    y = y_p
            continue

        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            stagnation_count = 0
        else:
            no_improve += 1

        # Adaptive active set management
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep_idx = np.argsort(margins)[:int(max_K * 0.6)]
            active_set_base = [active_set[i] for i in keep_idx]
            # Keep top violations
            top_viols = np.argsort(ac)[::-1][:100]
            for v in top_viols:
                if int(v) not in active_set_base:
                    active_set_base.append(int(v))
            active_set = sorted(set(active_set_base))

        if len(active_set) > max_K:
            active_set = sorted(active_set)[:max_K]

        K = len(active_set)
        if K == 0:
            break

        # Don't solve LP if active set is too small (degenerate)
        if K < 10:
            active_set = sorted(list(set(active_set) | set(np.argsort(ac)[::-1][:initial_K].tolist())))
            K = len(active_set)
            if K < 10:
                active_set = sorted(np.argsort(ac)[::-1][:initial_K].tolist())
                continue

        coeffs = build_constraint_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(50, K // 3)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if prob.status == 'infeasible':
                # Shrink active set
                active_set = active_set[:max(50, K // 2)]
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check for new violations and add to active set
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]

        all_viol = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(all_viol) > 0:
            viol_amt = conv_ys[all_viol] - M_curr
            top_v = all_viol[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)
            if len(active_set) > max_K:
                # Prune: keep most violated
                ac_new = fast_convolve(g_dir)
                margins_new = ac_new[active_set]
                active_set = [active_set[i] for i in np.argsort(margins_new)[::-1][:max_K]]
                active_set = sorted(active_set)

        best_alpha = 1.0 / (iter_count + 2.0)
        best_trial_M = M_curr

        if iter_count <= 3:
            alphas_to_try = [1.0, 0.8, 0.5]
        else:

            base_step = 2.0 / (iter_count + 2.0)
            alphas_to_try = [
                min(1.0, base_step * 3.0),
                base_step * 2.0,
                base_step,
                base_step * 0.5,
                base_step * 0.2,
                0.01,

            alphas_to_try = [a for a in alphas_to_try if 0 < a <= 1.0]

        for alpha in alphas_to_try:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y = (1.0 - best_alpha) * y + best_alpha * g_dir
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = best_a.copy()
        else:
            y = y / s_y

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): " +
                  "J=" + repr(2.0 * n * best_m) + ", |S|=" + str(len(active_set)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): " +
              "J=" + repr(2.0 * n * best_m) + ", iters=" + str(iter_count))
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=77, verbose=False):
    """Coordinate descent polish targeting peak reduction."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()

    t0 = time.monotonic()

    stall = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination to move mass to
            # Target: positions that don't contribute to top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    peak_mask[s2:e2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < max(10, n // 10):
                candidates = np.where(a < a.mean())[0]
            if len(candidates) < 5:
                candidates = np.arange(n)

            # Prefer positions with low current value
            cand_vals = a[candidates]
            dst = int(candidates[np.argmin(cand_vals)])

            if dst == src:
                continue

            found = False
            for frac in [0.01, 0.005, 0.002, 0.02]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found:
                # Random small perturbation
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.008)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1

    if verbose:
        elapsed = time.monotonic() - t0
        print("    Coord polish (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))
    return best_a, best_m


def diverse_initialization(n, seed=0):
    """Generate diverse initial sequences for exploration."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)

    candidates = []

    # Uniform (theoretical optimum for this objective)
    candidates.append(np.ones(n) / n)

    # Sine shapes
    candidates.append(norm_sum1(np.sin(np.pi * x) ** 2))
    candidates.append(norm_sum1(np.sin(np.pi * x * 2) ** 2))
    candidates.append(norm_sum1(0.5 + 0.5 * np.cos(2 * np.pi * x)))

    # Gaussian mixture
    for trial in range(3):
        a = np.zeros(n)
        n_bumps = rng.integers(2, 6)
        for _ in range(n_bumps):
            c = rng.uniform(0.1, 0.9)
            s = rng.uniform(0.03, 0.15)
            w = rng.uniform(0.5, 1.5)
            a += w * np.exp(-((x - c) ** 2) / (2 * s * s))
        candidates.append(norm_sum1(a))

    # Random smooth
    for trial in range(2):
        a = np.abs(rng.standard_normal(n))
        # Smooth with moving average
        k = max(3, n // 100)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        candidates.append(norm_sum1(a))

    return candidates

                best_seq, best_score = try_update(best_seq, best_score, a_out, "restart" + str(r), t_global)

    # === Phase 6: Final LP-FW ===
    print("\n=== Phase 6: Final LP-FW (n=" + str(len(best_seq)) + ") ===")
    t6 = min(remaining_time() * 0.25, 250.0)
    if t6 > 40 and len(best_seq) <= 9000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t6, solver,
                         initial_K=800, max_K=5000, add_per_iter=100, verbose=True, seed=2024)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # === Phase 7: Final coordinate polish ===
    rt = remaining_time()
    if rt > 20:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), min(rt * 0.7, 80.0), rng_seed=2024, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        # Absolute fallback
        n_fb = 5000
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print("Fallback uniform: score=" + repr(best_score))

    return [float(x) for x in best_seq]
