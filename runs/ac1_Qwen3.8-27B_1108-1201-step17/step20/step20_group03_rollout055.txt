def simplex_project(z):
    """Project z onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def simplex_project(z):
    n = len(z)
    if n == 0:

    if len(rho) == 0:
        return np.ones(n) / n

    return np.maximum(z - theta, 0.0)

Actually, I want to add one more clever technique: **frequency-domain shaping**.

In the frequency domain, the autocorrelation's DFT is `|A(f)|^2`. The peak of the autocorrelation is related to the "concentration" of `|A(f)|^2`. If `|A(f)|^2` is very peaked (concentrated at one frequency), the autocorrelation will be oscillatory with high peaks. If `|A(f)|^2` is flat, the autocorrelation will be concentrated at lag 0.

So we want `|A(f)|^2` to be as "uniform" as possible (in a certain sense). This is related to the "minimum entropy" or "maximum flatness" of the power spectrum.

A simple heuristic: iteratively modify the spectrum to flatten `|A(f)|^2` while maintaining non-negativity in the time domain.

Algorithm:
1. Compute `A(f) = FFT(a)`
2. Compute `P(f) = |A(f)|^2`
3. Compute `b = IFFT(P)` (this is the autocorrelation)
4. Compute a "flattening" factor: `F(f) = 1 / (1 + λ * P(f) / max(P))`
5. Update: `A_new(f) = A(f) * F(f)`
6. `a_new = IFFT(A_new)`
7. Clip to non-negative: `a_new = max(a_new, 0)`
8. Renormalize
9. Repeat

This is similar to the "spectral_reshape" in the previous solution but with a different flattening strategy.

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_topk_gradient(a, b, K):
    n = len(a)
    K = min(K, len(b))
    top_idx = np.argsort(b)[::-1][:K]
    g = np.zeros(n)
    for k in top_idx:
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            g[j_lo:j_hi + 1] += a[k - j_hi:k - j_lo + 1][::-1]
    return g / K


def fista_topk(a0, time_limit, K=10, restart_every=2500, verbose=True):
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    x_prev = x.copy()
    t_val = 1.0
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    step = 5.0 / n
    min_step = 1e-8
    max_step = 10.0

    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        alpha = (t_val - 1.0) / t_new
        y = x + alpha * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        b_y = fast_autocorr(y)
        g = compute_topk_gradient(y, b_y, K)
        g_mean = g.mean()
        g = g - g_mean

        g_norm = np.sqrt(np.dot(g, g))
        if g_norm < 1e-15:
            break

        step_trial = step
        b_max_current = b_y.max()
        accepted = False
        for _ls in range(8):
            z = y - step_trial * g
            x_cand = simplex_project(z)
            m_cand = fast_autocorr(x_cand).max()
            if m_cand < b_max_current - 1e-12:
                accepted = True
                break
            step_trial *= 0.5

        if accepted:
            step = min(step * 1.05, max_step)
        else:
            step = max(step * 0.6, min_step)

        x_prev = x
        x = x_cand
        t_val = t_new

        m_new = fast_autocorr(x).max()
        if m_new < best_m:
            best_m = m_new
            best_a = x.copy()

        if it % restart_every == 0 and m_new > best_m * (1.0 - 1e-6):
            noise_scale = 0.003 + 0.002 * (it // restart_every)
            noise = noise_scale * rng.standard_normal(n) * best_a
            x = best_a + noise
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            y = x.copy()
            x_prev = x.copy()
            t_val = 1.0
            step = 5.0 / n

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print("    FISTA-TopK it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(2.0 * n * best_m, 6)) + " step=" + repr(round(step, 4)))

    return best_a, best_m


def fista_multik(a0, time_limit, K_list, verbose=True):
    n = len(a0)
    best_a = norm_sum1(a0)
    best_m = fast_autocorr(best_a).max()
    t0 = time.monotonic()

    for K in K_list:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break
        t_alloc = min(remaining * 0.25, 300.0)
        print("    FISTA-TopK K=" + str(K) + " alloc=" + str(round(t_alloc, 1)) + "s")
        a_out, m_out = fista_topk(best_a, t_alloc, K=K, verbose=False)
        m_check = fast_autocorr(a_out).max()
        if m_check < best_m:
            best_m = m_check
            best_a = a_out.copy()
            elapsed = time.monotonic() - t0
            print("      Improved: J=" + repr(round(2.0 * n * best_m, 6)) + " (" + str(round(elapsed, 1)) + "s)")

    return best_a, best_m


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=3000, verbose=True):
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    ac = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 500
    g_prev = None

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                active_set.add(int(k))

        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5),
                  max(0.02, default_alpha * 1.5), 1.0 / (t + 1),
                  max(0.01, default_alpha * 2.0)]
        alphas = [max(0.005, min(1.0, a)) for a in alphas]
        alphas = list(set(round(a, 6) for a in alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t <= 2:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(2.0 * n * best_m, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def local_search(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:40]
        improved = False

        for _round in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(top_peaks[rng.integers(len(top_peaks))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:20]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src:
                continue
            if a[src] < 1e-12:
                continue

            frac = float(rng.choice([0.003, 0.005, 0.01, 0.02, 0.05, 0.1]))
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-13:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                ac = fast_autocorr(a)
                top_peaks = np.argsort(ac)[::-1][:40]

        if not improved:
            stall += 1
            n_pert = min(10, n)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.002, 0.05))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print("    LocalSearch (" + str(round(elapsed, 1)) + "s): J=" + repr(round(2.0 * n * best_m, 6)) + " stall=" + str(stall))

    return best_a, best_m


def cyclic_shift_best(a0):
    n = len(a0)
    if n > 20000:
        return a0.copy(), fast_autocorr(a0).max()
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a0, fft_size)
    for s in range(1, n):
        phase = np.exp(-2j * np.pi * np.arange(len(A)) * s / fft_size)
        AS = A * phase
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

    # Phase 8: Final local search
