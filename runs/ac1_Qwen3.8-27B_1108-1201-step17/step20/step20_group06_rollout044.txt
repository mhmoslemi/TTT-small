def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_smoothed_grad(y, beta, top_k=None):
    n = len(y)
    b = fast_autocorr(y)
    size = 2 * n - 1
    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    if b_max <= 0:
        return np.zeros(n), b
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s
    a_rev = y[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, b

def fw_with_active_set(y, active_set, solver, max_K=3000):
    """Solve the FW subproblem: minimize max_k (y*g)[k] s.t. g>=0, sum(g)=1, (y*g)[k]<=M for active k"""
    n = len(y)
    K = len(active_set)
    if K == 0:
        return None
    constraint_list = list(active_set)[:max_K]
    K = len(constraint_list)
    if K == 0:
        return None
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return None
    return g_val / g_sum

            target = power[0] / n
        
        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.5:
                A_new[k] *= np.sqrt(max(0.2, 1.0 - alpha * (ratio - 1.0)))
            elif ratio < 0.3:
                A_new[k] *= np.sqrt(min(3.0, 1.0 + alpha * 0.5))
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 500 == 0:
            # Random phase perturbation
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.01
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial
            # Also try random amplitude perturbation
            A = np.fft.rfft(a, fft_size)
            amp_pert = 1.0 + rng.standard_normal(len(A)) * 0.01
            A[1:] *= amp_pert[1:]
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial
        
        if it % 1000 == 0:
            alpha = max(0.003, alpha * 0.97)
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
    
    return best_a, best_m

def nesterov_refined(a0, time_limit, beta_start=50.0, beta_end=2000.0,
                      beta_growth=1.004, eta_init=0.4, top_k_ratio=0.02,
                      restart_every=1000, verbose=True, seed=42):
    """Refined Nesterov with aggressive beta schedule."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    eta = eta_init
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        grad, b = compute_smoothed_grad(y, beta, top_k)
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-15:
            x_new = x
            max_new = fast_autocorr(x_new).max()
        else:
            eta_trial = eta
            found = False
            y_max = fast_autocorr(y).max()
            for _ in range(12):
                z = y - eta_trial * grad / gnorm * np.sqrt(gnorm)
                x_new = simplex_project(z)
                max_new = fast_autocorr(x_new).max()
                if max_new <= y_max + 1e-15:
                    found = True
                    break
                eta_trial *= 0.5
            if not found:
                x_new = x
                max_new = fast_autocorr(x).max()
                eta = eta_trial
        
        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.05, 1.5)
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        if t % 30 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = rng.integers(6)
            if strategy == 0:
                noise_level = 0.01 + 0.02 * rng.random()
                pert = rng.normal(0, noise_level, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                # Blend with sparse random
                sp = np.zeros(n)
                sp[rng.choice(n, size=n//10, replace=False)] = 1.0
                sp = sp / sp.sum()
                x_r = 0.7 * best_a + 0.3 * sp
            elif strategy == 2:
                x_r = best_a * np.exp(0.05 * rng.standard_normal(n))
            elif strategy == 3:
                # Block perturbation
                block = n // 5
                bidx = rng.integers(0, 5) * block
                x_r = best_a.copy()
                x_r[bidx:bidx+block] *= rng.uniform(0.5, 1.5)
            elif strategy == 4:
                shift = int(rng.integers(1, max(2, n // 20)))
                x_r = np.roll(best_a, shift)
            else:
                # Mirror reflection
                x_r = best_a[::-1].copy()
            
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 20.0)
            eta = eta_init
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nesterov+ t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " b=" + str(int(beta)))
    
    return best_a, best_max_b

def multi_direction_fw(a0, time_limit, solver, initial_K=200, max_K=2000, verbose=True):
    """Frank-Wolfe with multiple directions per iteration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 500
    t_step = 1.0
    directions_cache = []
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        size = 2 * n - 1
        
        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Generate 2-3 different active sets
        n_dirs = 2
        best_dir_M = M_curr
        best_dir = None
        
        for d in range(n_dirs):
            if d == 0:
                # Standard: top peaks
                top_k_for_set = initial_K + d * 100
                top_idx = np.argsort(ac)[::-1][:min(top_k_for_set, size)]
                active_set = set(int(k) for k in top_idx)
            else:
                # Offset: top peaks shifted
                offset = d * 7
                top_idx = np.argsort(np.roll(ac, offset))[::-1][:initial_K]
                active_set = set(int(k) for k in top_idx)
            
            g_dir = fw_with_active_set(a, active_set, solver, max_K)
            if g_dir is None:
                continue
            
            # Evaluate this direction
            for alpha in [2.0/(t+2), 0.1, 0.2]:
                alpha = max(0.01, min(0.99, alpha))
                trial = (1.0 - alpha) * a + alpha * g_dir
                trial = np.maximum(trial, 0.0)
                s_t = trial.sum()
                if s_t < 1e-15:
                    continue
                trial = trial / s_t
                M_trial = fast_autocorr(trial).max()
                if M_trial < best_dir_M:
                    best_dir_M = M_trial
                    best_dir = (g_dir, alpha)
        
        if best_dir is None:
            no_improve += 50
            if no_improve > max_no_improve:
                break
            continue
        
        g_dir, alpha = best_dir
        a_new = (1.0 - alpha) * a + alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        s_y = a_new.sum()
        if s_y < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_y
        a = a_new
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    MultiFW it=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " no_imp=" + str(no_improve))
    
    return best_a, best_m

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score
