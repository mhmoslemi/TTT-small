def lp_fw(a0, time_limit, solver, initial_K=600, max_K=6000, verbose=True):
    """
    Enhanced LP-Frank-Wolfe method.
    At each iteration:
    1. Solve LP: min M s.t. sum(g)=1, g>=0, (A_k @ g) <= M for k in active set
    2. Add violated constraints to active set
    3. Prune inactive constraints
    4. Line search along direction g_dir
    """
    n = len(a0)
    y = norm1(a0)
    best_a = y.copy()
    ac = fast_convolve(y)
    best_m = ac.max()
    t0 = time.monotonic()
    
    # Initialize active set with top peaks
    active_set = list(np.argsort(ac)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 500
    iter_count = 0
    restarts = 0
    max_restarts = 5
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if restarts >= max_restarts:
                break
            # Restart with perturbed active set
            restarts += 1
            no_improve = 0
            ac_curr = fast_convolve(y)
            rng = np.random.default_rng(restarts * 12345)
            n_keep = max(200, len(active_set) // 2)
            keep = list(rng.choice(len(active_set), size=n_keep, replace=False))
            # Add new top peaks
            new_peaks = np.argsort(ac_curr)[::-1][:n_keep]
            active_set = sorted(set([active_set[i] for i in keep]) | set(int(k) for k in new_peaks))
            active_set = active_set[:max_K]
            if verbose:
                print("    LP-FW restart " + str(restarts) + ", |S|=" + str(len(active_set)))
            continue
        
        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-12:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Prune active set
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            # Keep constraints with smallest margins (most active)
            keep_idx = np.argsort(margins)[:max_K]
            active_set = [active_set[i] for i in keep_idx]
        
        K = len(active_set)
        if K < 10:
            active_set = list(np.argsort(ac)[::-1][:min(100, 2*n-1)])
            K = len(active_set)
        
        # Build constraint matrix
        # For each k in active_set, the constraint is: sum_i g_i * y_{k-i} <= M
        # where y_{k-i} = 0 if k-i out of range
        coeffs = np.zeros((K, n))
        for jj in range(K):
            k_int = int(active_set[jj])
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Reduce active set and retry
            active_set = active_set[:max(100, K // 4)]
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violations
        conv_ys = np.zeros(2 * n - 1)
        fft_size = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]
        
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            # Add top violations
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            active_set_set = set(active_set)
            for k in top_v:
                active_set_set.add(int(k))
            active_set = sorted(active_set_set)[:max_K]
        
        # Line search: find best alpha in [0, 1]
        # Use backtracking
        best_alpha = 0.5
        best_trial_M = M_curr
        
        # Try a range of alphas
        alphas = [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05,
                  1.0 / (iter_count + 2), max(0.01, 2.0 / (iter_count + 5))]
        seen = set()
        for alpha in alphas:
            alpha_r = round(alpha, 4)
            if alpha_r in seen or alpha_r <= 0:
                continue
            seen.add(alpha_r)
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        # Update
        if iter_count <= 1:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y
        
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", |S|=" + str(len(active_set)))
    
    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", iters=" + str(iter_count))
    return best_a, best_m


def smooth_mirror_descent(a0, time_limit, T_start=0.01, T_end=0.0001, verbose=True):
    """
    Smooth optimization using log-sum-exp approximation and mirror descent.
    """
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    # Anneal T from T_start to T_end
    total_iters = 5000
    iters_done = 0
    eta0 = 0.02
    
    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal T
        frac = it / total_iters
        T = T_start * (T_end / T_start) ** frac
        
        # Compute smooth objective and gradient
        ac = fast_convolve(a)
        
        # Softmax weights
        ac_shifted = ac - ac.max()
        p = np.exp(ac_shifted / T)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(2 * n - 1)
            p[np.argmax(ac)] = 1.0
        else:
            p = p / p_sum
        
        # Gradient of smooth objective:
        # grad_i = 2 * sum_k p_k * a_{k-i} (with bounds)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-15:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * p[k] * a[k - idx]
        
        # Mirror descent step (multiplicative)
        eta = eta0 / (np.sqrt(it + 1) + 1)
        log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        M_new = fast_convolve(a).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a.copy()
        
        iters_done += 1
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    Smooth iter=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", T=" + repr(T))
    
    if verbose:
        print("    Smooth done: J=" + repr(2.0 * n * best_m))
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True):
    """Coordinate descent polish targeting peak reduction."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    stall = 0
    max_stall = 2000
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        # Get top peaks
        n_top = min(20, 2 * n - 1)
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_k[:10]:
            k_int = int(k)
            # Find the pair (i, j) with i+j=k that contributes most to this peak
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-25:
                continue
            
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination (low impact on peaks)
            # Score each position by how much it contributes to top peaks
            peak_impact = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                idx2 = np.arange(s2, e2)
                peak_impact[idx2] += a[k2i - idx2]
            
            # Choose destination with lowest peak impact
            candidates = np.argsort(peak_impact)[:n // 4]
            dst = int(candidates[0])
            if dst == src:
                dst = int(candidates[1]) if len(candidates) > 1 else int(rng.integers(n))
            
            # Try transfer
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            
            if not improved:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if not improved:
            stall += 1
        
        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", stall=" + str(stall))
    
    return best_a, best_m


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    if hasattr(cp, 'SCIP'):
        candidates.append(cp.SCIP)
    for s in candidates:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _SOLVER = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _SOLVER


def _fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        sc = evaluate_sequence([float(x) for x in a_norm])
    except Exception:
        return best_seq, best_score
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: score=" + repr(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def _interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _lp_fw(a0, time_limit, solver, initial_K, max_K, verbose):
    n = len(a0)
    y = _norm1(a0)
    best_a = y.copy()
    ac = _fast_convolve(y)
    best_m = ac.max()
    t0 = time.monotonic()

    active_set = sorted(int(k) for k in np.argsort(ac)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 600
    iter_count = 0
    restarts = 0
    max_restarts = 8
    last_coeffs_hash = None

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if restarts >= max_restarts:
                break
            restarts += 1
            no_improve = 0
            ac_curr = _fast_convolve(y)
            rng = np.random.default_rng(restarts * 99991 + 7)
            n_half = max(200, len(active_set) // 2)
            keep_idx = rng.choice(len(active_set), size=n_half, replace=False)
            keep = set(int(active_set[i]) for i in keep_idx)
            new_peaks = set(int(k) for k in np.argsort(ac_curr)[::-1][:n_half])
            active_set = sorted(keep | new_peaks)[:max_K]
            if verbose:
                print("    LP-FW restart " + str(restarts) + ", |S|=" + str(len(active_set)))
            continue

        iter_count += 1
        ac = _fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-13:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Prune active set
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep_idx = np.argsort(margins)[:int(max_K * 0.8)]
            active_set_kept = [active_set[i] for i in keep_idx]
            # Always keep the most violated
            viol = np.where(ac > M_curr - 1e-9)[0]
            for v in viol[:200]:
                iv = int(v)
                if iv not in active_set_kept:
                    active_set_kept.append(iv)
            active_set = sorted(set(active_set_kept))[:max_K]

        K = len(active_set)
        if K < 20:
            active_set = sorted(int(k) for k in np.argsort(ac)[::-1][:min(200, 2 * n - 1)])
            K = len(active_set)

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj in range(K):
            k_int = int(active_set[jj])
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(100, K // 4)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations and update active set
        fft_size = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:80]]
            active_set_set = set(active_set)
            for k in top_v:
                active_set_set.add(int(k))
            active_set = sorted(active_set_set)[:max_K]

        # Line search
        best_alpha = 1.0 / (iter_count + 2.0)
        best_trial_M = M_curr

        alphas = [1.0, 0.9, 0.7, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05,
                  1.0 / (iter_count + 2), max(0.01, 2.0 / (iter_count + 5))]
        seen_alphas = set()
        for alpha in alphas:
            ar = round(alpha, 5)
            if ar in seen_alphas or ar <= 0 or ar > 1:
                continue
            seen_alphas.add(ar)
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = _fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 2:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", |S|=" + str(len(active_set)) + ", M=" + repr(M_curr))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", iters=" + str(iter_count))
    return best_a, best_m


def _smooth_mirror_descent(a0, time_limit, T_start, T_end, verbose):
    n = len(a0)
    a = _norm1(a0)
    best_a = a.copy()
    best_m = _fast_convolve(a).max()
    t0 = time.monotonic()

    total_iters = 8000
    eta_base = 0.03

    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, it / float(total_iters))
        T = T_start * (T_end / T_start) ** frac

        ac = _fast_convolve(a)

        # Softmax weights (numerically stable)
        ac_shifted = ac - ac.max()
        p = np.exp(ac_shifted / T)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(2 * n - 1)
            p[int(np.argmax(ac))] = 1.0
        else:
            p = p / p_sum

        # Gradient: grad_i = 2 * sum_k p_k * a_{k-i}
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-14:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * pk * a[k - idx]

        # Mirror descent (multiplicative update)
        eta = eta_base / (np.sqrt(it + 1) + 1.0)
        log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        M_new = _fast_convolve(a).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a.copy()

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print("    Smooth iter=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", T=" + repr(T))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    Smooth done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))
    return best_a, best_m


def _subgradient_polish(a0, time_limit, verbose):
    """Plain subgradient with Polyak averaging for final polish."""
    n = len(a0)
    a = _norm1(a0)
    best_a = a.copy()
    best_m = _fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(31415)

    # Polyak averaging
    avg_a = a.copy()
    eta0 = 0.05
    stall = 0
    max_stall = 5000
    iter_count = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break
        iter_count += 1
        eta = eta0 / (np.sqrt(t) + 1.0)

        ac = _fast_convolve(a)
        k_star = int(np.argmax(ac))

        # Subgradient
        grad = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        idx = np.arange(i_start, i_end)
        grad[idx] = 2.0 * a[k_star - idx]

        # Mirror descent step
        log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Polyak averaging
        alpha_avg = 1.0 / (1.0 + t / 100.0)
        avg_a = (1.0 - alpha_avg) * avg_a + alpha_avg * a_new

        a = a_new
        M_new = _fast_convolve(a).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Also check the averaged point
        M_avg = _fast_convolve(avg_a).max()
        if M_avg < best_m - 1e-14:
            best_m = M_avg
            best_a = avg_a.copy()
            stall = 0

        # Perturb if stuck
        if stall > 1000 and stall % 1000 == 0:
            noise = rng.normal(0, 0.02, n) * best_a
            a_p = best_a + noise
            a_p = np.maximum(a_p, 1e-10)
            sp = a_p.sum()
            if sp > 1e-12:
                a = a_p / sp
                avg_a = a.copy()
            eta0 *= 1.3

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print("    SubGrad t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    SubGrad done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))
    return best_a, best_m


def _coordinate_polish(a0, time_limit, verbose):
    n = len(a0)
    a = _norm1(a0)
    best_a = a.copy()
    best_m = _fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    stall = 0
    max_stall = 3000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = _fast_convolve(a)
        n_top = min(15, 2 * n - 1)
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:8]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-25:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score positions by peak impact
            peak_impact = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                idx2 = np.arange(s2, e2)
                peak_impact[idx2] += a[k2i - idx2]

            candidates = np.argsort(peak_impact)[:max(10, n // 10)]
            dst = int(candidates[0])
            if dst == src:
                dst = int(candidates[1]) if len(candidates) > 1 else int(rng.integers(n))
            if dst == src:
                continue

            found = False
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = _fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = _fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 500 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.003, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = _fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 1000 == 0:
            elapsed = time.monotonic() - t0
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", stall=" + str(stall))

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = _get_solver()

    # Get initial sequence from parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_sequence([float(x) for x in p_norm])
            if np.isfinite(sc):
                best_seq = p_norm
                best_score = sc
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
            else:
                raise Exception("invalid score")
        else:
            raise Exception("invalid shape")
    except Exception:
        n = 5000
        x = np.linspace(0, 1, n)
        best_seq = _norm1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_sequence([float(x) for x in best_seq])
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # === Phase 1: LP-FW primary ===
    print("\n=== Phase 1: LP-FW primary (n=" + str(n0) + ") ===")
    t1 = min((time_budget - (time.monotonic() - t_global)) * 0.22, 350.0)
    if t1 > 60 and n0 <= 12000:
        a_out, _ = _lp_fw(_norm1(best_seq), t1, solver, initial_K=800, max_K=8000, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "lpfw1", t_global)

    # === Phase 2: Smooth mirror descent ===
    print("\n=== Phase 2: Smooth MD (n=" + str(len(best_seq)) + ") ===")
    t2 = min((time_budget - (time.monotonic() - t_global)) * 0.10, 150.0)
    if t2 > 30:
        a_out, _ = _smooth_mirror_descent(_norm1(best_seq), t2, T_start=0.003, T_end=0.00005, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "smooth", t_global)

    # === Phase 3: LP-FW second pass with larger active set ===
    print("\n=== Phase 3: LP-FW pass 2 (n=" + str(len(best_seq)) + ") ===")
    t3 = min((time_budget - (time.monotonic() - t_global)) * 0.22, 350.0)
    if t3 > 60 and len(best_seq) <= 12000:
        a_out, _ = _lp_fw(_norm1(best_seq), t3, solver, initial_K=1200, max_K=10000, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "lpfw2", t_global)

    # === Phase 4: Multi-n exploration ===
    print("\n=== Phase 4: Multi-n ===")
    n_best = len(best_seq)
    candidates_n = [n_best - 400, n_best + 400, n_best + 800, n_best - 800, n_best + 1500, n_best - 1500]
    candidates_n = sorted(set(l for l in candidates_n if 2000 <= l <= 14000), key=lambda x: abs(x - n_best))

    for new_n in candidates_n[:4]:
        rt = time_budget - (time.monotonic() - t_global)
        if rt < 120:
            break
        print("\n  Multi-n: n=" + str(new_n) + " (remaining=" + str(round(rt, 1)) + "s)")
        a_interp = _interpolate_seq(best_seq, new_n)
        t = min(rt * 0.12, 180.0)
        if t > 40 and new_n <= 12000:
            a_out, _ = _lp_fw(a_interp, t, solver, initial_K=600, max_K=6000, verbose=False)
            best_seq, best_score = _try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

    # === Phase 5: Subgradient with Polyak averaging ===
    print("\n=== Phase 5: SubGrad+Polyak (n=" + str(len(best_seq)) + ") ===")
    t5 = min((time_budget - (time.monotonic() - t_global)) * 0.10, 150.0)
    if t5 > 30:
        a_out, _ = _subgradient_polish(_norm1(best_seq), t5, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "subgrad", t_global)

    # === Phase 6: LP-FW third pass ===
    print("\n=== Phase 6: LP-FW pass 3 (n=" + str(len(best_seq)) + ") ===")
    t6 = min((time_budget - (time.monotonic() - t_global)) * 0.18, 280.0)
    if t6 > 60 and len(best_seq) <= 12000:
        a_out, _ = _lp_fw(_norm1(best_seq), t6, solver, initial_K=1500, max_K=12000, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "lpfw3", t_global)

    # === Phase 7: Coordinate polish ===
    print("\n=== Phase 7: Coord polish (n=" + str(len(best_seq)) + ") ===")
    t7 = min((time_budget - (time.monotonic() - t_global)) * 0.08, 120.0)
    if t7 > 20:
        a_out, _ = _coordinate_polish(_norm1(best_seq), t7, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "polish", t_global)

    # === Phase 8: Final LP-FW ===
    print("\n=== Phase 8: Final LP-FW (n=" + str(len(best_seq)) + ") ===")
    t8 = min((time_budget - (time.monotonic() - t_global)) * 0.20, 250.0)
    if t8 > 40 and len(best_seq) <= 12000:
        a_out, _ = _lp_fw(_norm1(best_seq), t8, solver, initial_K=2000, max_K=12000, verbose=True)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # === Final coordinate polish ===
    rt = time_budget - (time.monotonic() - t_global)
    if rt > 30:
        a_out, _ = _coordinate_polish(_norm1(best_seq), min(rt * 0.5, 100.0), verbose=False)
        best_seq, best_score = _try_update(best_seq, best_score, a_out, "polish_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        final_score = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(final_score):
            best_score = final_score
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = _norm1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_sequence([float(x) for x in best_seq])
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
