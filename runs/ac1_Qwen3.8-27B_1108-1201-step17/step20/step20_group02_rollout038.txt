    n = len(v)
    if n == 0:

    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

    z is the log-parameterization, a = softmax(z)."""

    max_c = c.max()
    val = (1.0 / beta) * np.log(np.sum(np.exp(beta * (c - max_c)))) + max_c
    return val, a, c

Let me also consider: should I use cvxpy's quadratic solver? The problem is:
- Minimize $\max_k c_k$ subject to $\sum a_i = 1, a_i \geq 0$

This can be written as:
- Minimize $M$
- Subject to $c_k \leq M$ for all $k$
- $\sum a_i = 1$
- $a_i \geq 0$

The constraint $c_k \leq M$ is $a^T C_k a \leq M$, which is a quadratic constraint. This is a QCQP (quadratically constrained quadratic program). It's non-convex because $C_k$ for $k \neq 0$ is indefinite.

However, for $k = 0$: $c_0 = \sum a_i^2 \leq M$, which IS convex.
For $k \neq 0$: $c_k = \sum_i a_i a_{i+k} \leq M$, which is NOT convex.

So the full problem is non-convex and can't be solved directly by standard QP solvers.

But we can use an **SDP relaxation**:
- Lift to $X = aa^T + \text{diag}(a) \geq 0$ (or just $X = aa^T \succeq 0$ with $X_{ii} = a_i$)
- The constraint $a^T C_k a \leq M$ becomes $\text{tr}(C_k X) \leq M$ (linear in $X$!)
- $\sum a_i = 1$ becomes $\sum X_{ii} = 1$
- $a_i \geq 0$ becomes $X_{ii} \geq 0$

So the SDP relaxation is:
- Minimize $M$
- Subject to $\text{tr}(C_k X) ≤ M$ for all $k$
- $\sum X_{ii} = 1$
- $X_{ii} \geq 0$
- $X \succeq 0$

This is a linear program in $X$ (all constraints are linear, objective is linear)! The only issue is the matrix $X$ is $n \times n$, so for $n = 5000$, it's a 25M variable SDP, which is way too large.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp (smoothed max) optimization with L-BFGS on log-parameterization
2. Adaptive temperature annealing
3. Peak-aware mass-transfer coordinate descent
4. Multi-n exploration with interpolation
5. Small-n SDP relaxation for guidance
6. Final coordinate polish

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def log_to_seq(z):
    """Convert log-parameterization to sequence (softmax)."""
    z = z - z.max()
    a = np.exp(z)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(z)) / len(z)
    return a / s

def seq_to_log(a):
    """Convert sequence to log-parameterization."""
    a = np.maximum(a, 1e-30)
    return np.log(a)

def smoothed_max_obj_and_grad(z, beta):
    """Objective and gradient of smoothed max autocorrelation.
    
    Log-sum-exp of the autocorrelation, parameterized via z (a = softmax(z)).
    Returns (value, gradient w.r.t. z).
    """
    a = log_to_seq(z)
    n = len(a)
    c = fast_convolve(a)
    
    # Compute log-sum-exp: (1/beta) * log(sum(exp(beta * c))) + max(c)
    max_c = c.max()
    shifted = beta * (c - max_c)
    # Handle numerical issues
    shifted = np.clip(shifted, -700, 700)
    log_sum = np.log(np.sum(np.exp(shifted)))
    value = max_c + log_sum / beta
    
    # Softmax weights
    p = np.exp(shifted) / np.exp(log_sum)
    
    # Gradient w.r.t. a: g_a[j] = 2 * sum_m a[m] * p[m+j]
    p_rev = p[::-1]
    fft_size = 1 << (2*n - 1 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p_rev, fft_size)
    conv_ap = np.fft.irfft(A * P, fft_size)
    g_a = conv_ap[2*n-1-n : 2*n-1][::-1] * 2.0
    
    # Transform to z-gradient: g_z = a * (g_a - dot(g_a, a))
    offset = np.dot(g_a, a)
    g_z = a * (g_a - offset)
    
    return value, g_z

def smoothed_max_optimize(a0, time_limit, verbose=True):
    """Optimize the smoothed max autocorrelation using L-BFGS with adaptive temperature."""
    n = len(a0)
    best_a = norm_sum1(a0)
    best_score = evaluate_safe(best_a)
    best_M = fast_convolve(best_a).max()
    t0 = time.monotonic()
    
    z0 = seq_to_log(best_a)
    
    # Temperature schedule: start low, increase
    betas = [20, 50, 100, 200, 500, 1000, 2000, 5000]
    
    current_z = z0.copy()
    
    for beta in betas:
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_end = time.monotonic() - t0 > time_limit * 0.85
        max_iter = 200 if not beta_end else 1000
        
        if verbose:
            print(f"    SMX: beta={beta}, n={n}, starting...")
        
        result = minimize(
            fun=lambda z: smoothed_max_obj_and_grad(z, beta)[0],
            x0=current_z,
            jac=lambda z: smoothed_max_obj_and_grad(z, beta)[1],
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'gtol': 1e-8, 'ftol': 1e-10,
                     'maxfun': max_iter * 20}
        )
        
        if result.x is not None:
            current_z = result.x
            a_new = log_to_seq(current_z)
            M_new = fast_convolve(a_new).max()
            score_new = evaluate_safe(a_new)
            
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
                best_M = M_new
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"    SMX: NEW BEST beta={beta}: score={best_score:.6f} (t={elapsed:.1f}s)")
        
        if beta_end:
            break
    
    # Try increasing beta further if time remains
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 30:
        for beta in [10000, 50000]:
            if time.monotonic() - t0 > time_limit:
                break
            result = minimize(
                fun=lambda z: smoothed_max_obj_and_grad(z, beta)[0],
                x0=current_z,
                jac=lambda z: smoothed_max_obj_and_grad(z, beta)[1],
                method='L-BFGS-B',
                options={'maxiter': 500, 'gtol': 1e-10, 'ftol': 1e-12}
            )
            if result.x is not None:
                current_z = result.x
                a_new = log_to_seq(current_z)
                score_new = evaluate_safe(a_new)
                if score_new < best_score:
                    best_score = score_new
                    best_a = a_new.copy()
                    if verbose:
                        elapsed = time.monotonic() - t0
                        print(f"    SMX-high: NEW BEST beta={beta}: score={best_score:.6f} (t={elapsed:.1f}s)")
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    SMX done (t={elapsed:.1f}s): best score={best_score:.6f}")
    
    return best_a, best_score

    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def log_to_seq(z):
    z = np.clip(z, -30, 30)
    z = z - z.max()
    a = np.exp(z)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(z)) / len(z)
    return a / s

def seq_to_log(a):
    a = np.maximum(a, 1e-25)
    return np.log(a)

def smoothed_max_value_grad(z, beta):
    a = log_to_seq(z)
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    shifted = np.clip(beta * (c - max_c), -700, 700)
    log_sum = np.log(np.sum(np.exp(shifted)))
    value = 2.0 * n * (max_c + log_sum / beta)
    p = np.exp(shifted) / np.exp(log_sum)
    p_rev = p[::-1]
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p_rev, fft_size)
    conv_ap = np.fft.irfft(A * P, fft_size)

    pc = np.dot(p, c)
    g_a = 4.0 * n * a * (g_a0 - pc)
    return value, g_a
