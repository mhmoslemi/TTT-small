    objective = cp.Minimize(cp.sum_squares(x - x_cur))
    
    constraints = [cp.sum(x) == 1]
    
    # Linearized constraints
    for k in active_peaks:
        k_int = int(k)
        # Constraint: 2 * (x_cur[k-j] for valid j) @ x <= M + c_cur[k]
        #系数 of x[j] is 2*x_cur[k-j] if 0 <= k-j <= n-1, else 0

        if j_min > j_max:
            continue
        js = np.arange(j_min, j_max + 1)
        coeffs = 2.0 * x_cur[k_int - js]
        rhs = M + c_cur[k_int]
        constraints.append(coeffs @ x <= rhs)
    
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver, verbose=False)
    
    if x.value is None:
        return None
    x_val = np.maximum(x.value, 0)
    s = x_val.sum()
    if s < 1e-15:
        return None
    return x_val / s

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    """Compute gradient of smoothed max objective."""
    n = len(y)
    c = fast_autocorr(y)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total

    fft_size = 1 << (3 * n).bit_length()
    P = np.fft.rfft(p, fft_size)
    X = np.fft.rfft(y[::-1], fft_size)
    Conv = np.fft.irfft(P * X, fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad, p, c


def nesterov_opt(a0, time_limit, beta_start=50.0, beta_end=2000.0,
                 beta_growth=1.006, eta_base=0.5, top_k_ratio=0.02,
                 restart_every=1500, verbose=False, seed=42):
    """Nesterov accelerated projected gradient with adaptive beta."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(5, int(n * top_k_ratio))

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, c = compute_smoothed_grad(y, beta, top_k)

        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.05)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 30 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.02 / (1.0 + 0.1 * (t // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 15.0)

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" +
                  repr(round(J, 8)) + " beta=" + str(round(beta, 1)))

    return best_a, best_max_b


def sqp_improve(a0, time_limit, reduction_factor=0.002, n_peaks=30,
                max_iters=5000, verbose=False, seed=1):
    """SQP-based improvement: solve QP with linearized peak constraints."""
    import cvxpy as cp
    solver = get_qp_solver()
    if solver is None:
        return a0, fast_autocorr(a0).max()

    n = len(a0)
    x = norm_sum1(a0).copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if it % 50 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SQP it=" + str(it) + " (" + str(round(elapsed, 1)) +
                  "s): J=" + repr(round(J, 8)) + " stall=" + str(stall))

        c = fast_autocorr(x)
        M = c.max()
        if M < best_m - 1e-15:
            best_m = M
            best_x = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 500:
            # Restart with perturbation
            noise = rng.normal(0, 0.005, n) * x
            x_r = best_x + noise
            x_r = np.maximum(x_r, 1e-12)
            x = x_r / x_r.sum()
            stall = 0
            continue

        # Find active peaks (top n_peaks)
        top_idx = np.argsort(c)[::-1][:n_peaks]

        # Build QP
        x_var = cp.Variable(n, nonneg=True)
        target_M = M * (1 - reduction_factor)

        objective = cp.Minimize(cp.sum_squares(x_var - x))
        constraints = [cp.sum(x_var) == 1]

        # Linearized constraints for each active peak
        for k in top_idx:
            k_int = int(k)
            j_min = max(0, k_int - n + 1)
            j_max = min(n - 1, k_int)
            if j_min > j_max:
                continue
            js = np.arange(j_min, j_max + 1)
            coeffs = 2.0 * x[k_int - js]
            rhs = target_M + c[k_int]
            constraints.append(coeffs @ x_var <= rhs)

        # Trust region: limit the step size
        constraints.append(cp.sum_squares(x_var - x) <= 0.001)

        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=solver, verbose=False)

            if prob.status not in ('optimal', 'optimal_inaccurate'):
                stall += 1
                if stall > 200:
                    # Try smaller problem with fewer peaks
                    n_peaks = max(10, n_peaks // 2)
                    stall = 0
                continue

            x_new = np.maximum(x_var.value, 0.0)
            s_new = x_new.sum()
            if s_new < 1e-15:
                stall += 1
                continue
            x_new = x_new / s_new

            # Verify improvement
            c_new = fast_autocorr(x_new)
            M_new = c_new.max()

            if M_new < M - 1e-15:
                x = x_new
                reduction_factor = min(0.01, reduction_factor * 1.01)
            else:
                # Try smaller reduction
                reduction_factor = max(0.0005, reduction_factor * 0.8)
                stall += 5

        except Exception as e:
            stall += 50
            if stall > 300:
                reduction_factor = 0.001
                n_peaks = max(10, n_peaks - 5)
                stall = 100

    return best_x, best_m


def lp_improve(a0, time_limit, n_peaks=30, max_iters=100, verbose=False, seed=2):
    """LP-guided improvement: find direction g that maximizes sum while controlling cross-correlation."""
    import cvxpy as cp
    solver = get_lp_solver()
    if solver is None:
        return a0, fast_autocorr(a0).max()

    n = len(a0)
    x = norm_sum1(a0).copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if it % 5 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP it=" + str(it) + " (" + str(round(elapsed, 1)) +
                  "s): J=" + repr(round(J, 8)))

        c = fast_autocorr(x)
        M = c.max()
        top_idx = np.argsort(c)[::-1][:n_peaks]

        # Solve LP: max sum(g) s.t. g >= 0, (x*g)[k] <= M for top peaks
        g_var = cp.Variable(n, nonneg=True)
        objective = cp.Maximize(cp.sum(g_var))
        constraints = [g_var <= 1.0]

        for k in top_idx:
            k_int = int(k)
            j_min = max(0, k_int - n + 1)
            j_max = min(n - 1, k_int)
            if j_min > j_max:
                continue
            js = np.arange(j_min, j_max + 1)
            coeffs = x[k_int - js]
            constraints.append(coeffs @ g_var <= M)

        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=solver, verbose=False)

            if prob.status not in ('optimal', 'optimal_inaccurate'):
                continue

            g_val = np.maximum(g_var.value, 0.0)
            s_g = g_val.sum()
            if s_g < 1e-15:
                continue
            g_norm = g_val / s_g

            # Check g's own autocorrelation
            c_g = fast_autocorr(g_norm)
            M_g = c_g.max()

            # Try different blend factors
            found = False
            for t_blend in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
                x_try = (1 - t_blend) * x + t_blend * g_norm
                x_try = np.maximum(x_try, 0.0)
                s_try = x_try.sum()
                if s_try < 1e-15:
                    continue
                x_try = x_try / s_try
                M_try = fast_autocorr(x_try).max()
                if M_try < best_m - 1e-15:
                    x = x_try
                    best_m = M_try
                    best_x = x_try.copy()
                    found = True
                    break

            if not found:
                # Try smaller blends
                for t_blend in [0.01, 0.02, 0.03]:
                    x_try = (1 - t_blend) * x + t_blend * g_norm
                    x_try = np.maximum(x_try, 0.0)
                    s_try = x_try.sum()
                    if s_try < 1e-15:
                        continue
                    x_try = x_try / s_try
                    M_try = fast_autocorr(x_try).max()
                    if M_try < best_m - 1e-15:
                        x = x_try
                        best_m = M_try
                        best_x = x_try.copy()
                        break
                else:
                    # Stuck, add perturbation
                    noise = rng.normal(0, 0.01, n) * x
                    x_r = x + noise
                    x_r = np.maximum(x_r, 1e-10)
                    x = x_r / x_r.sum()

        except Exception:
            continue

    return best_x, best_m


def subgradient_polish(a0, time_limit, top_k=5, verbose=False, seed=3):
    """Subgradient method with averaging for final polish."""
    n = len(a0)
    x = norm_sum1(a0).copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x_avg = x.copy() * x
    count = 0
    step_scale = 1.0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(x)
        M = c.max()

        if M < best_m - 1e-15:
            best_m = M
            best_x = x.copy()

        # Find top peaks
        top_idx = np.argsort(c)[::-1][:top_k]

        # Compute subgradient
        p = np.zeros(2 * n - 1)
        for k in top_idx:
            p[int(k)] = 1.0 / top_k

        fft_size = 1 << (3 * n).bit_length()
        P = np.fft.rfft(p, fft_size)
        X = np.fft.rfft(x[::-1], fft_size)
        Conv = np.fft.irfft(P * X, fft_size)
        subgrad = 2.0 * Conv[n - 1:2 * n - 1]

        # Adaptive step size
        step_size = step_scale / (np.sqrt(t + 1) * np.sqrt(n))
        step_size = min(step_size, 0.01)

        z = x - step_size * subgrad
        x_new = simplex_project(z)

        # Update running average
        count += 1
        x_avg = x_avg * (count - 1) / count + x_new / count

        # Check average
        c_avg = fast_autocorr(x_avg)
        if c_avg.max() < best_m - 1e-15:
            best_m = c_avg.max()
            best_x = x_avg.copy()

        x = x_new

        # Occasionally check if we should restart
        if t % 500 == 0:
            # Decay step scale
            step_scale = max(0.1, step_scale * 0.95)

            if verbose and t % 2000 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    SG t=" + str(t) + " (" + str(round(elapsed, 1)) +
                      "s): J=" + repr(round(J, 8)))

    return best_x, best_m


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence if new one is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 8)) +
              " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

    if remaining > 30 and qp_solver is not None:
        t8 = min(remaining * 0.30, 150.0)
        if t8 > 20:
            a_out, _ = sqp_improve(norm_sum1(best_seq), t8,
                                   reduction_factor=0.0005, n_peaks=15,
                                   max_iters=5000, verbose=False, seed=8)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "sqp_final", t_global)

    # Phase 9: Final Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 9: Final Nesterov (n=" + str(n_best) + ") ===")
    if remaining > 30:
        t9 = min(remaining * 0.40, 120.0)
        if t9 > 20:
            a_out, _ = nesterov_opt(norm_sum1(best_seq), t9,
                                    beta_start=300.0, beta_end=8000.0,
                                    beta_growth=1.012, eta_base=0.2,
                                    top_k_ratio=0.005, restart_every=600,
                                    verbose=False, seed=9)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

    # Phase 10: Final subgradient
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 10: Final subgradient (n=" + str(n_best) + ") ===")
    if remaining > 20:
        t10 = min(remaining * 0.50, 80.0)
        if t10 > 10:
            a_out, _ = subgradient_polish(norm_sum1(best_seq), t10,
                                          top_k=3, verbose=False, seed=10)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "sg_final", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(round(best_score, 8)) +
              ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1((1.0 - x) ** 1.5 + 0.05)
        best_score = evaluate_safe(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)

    return [float(x) for x in best_seq]
