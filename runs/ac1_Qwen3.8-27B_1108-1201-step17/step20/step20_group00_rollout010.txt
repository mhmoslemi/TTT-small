"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Multi-length diverse initialization
2. Smoothed-max (log-sum-exp) gradient descent with momentum
3. Peak-targeted deflation
4. LP-free Frank-Wolfe with FFT-based subgradient
5. Coordinate descent polish
"""

import numpy as np
import time

def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def diverse_init(n, parent_seq=None):
    """Generate diverse initial candidates."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    
    best_a = normalize(np.sin(np.pi * x) ** 2)
    best_val = obj_val(best_a)
    
    candidates = []
    
    # Sine window
    for f in [1, 2, 3]:
        candidates.append(normalize(np.sin(np.pi * x) ** (2 * f)))
    
    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for sg in [0.1, 0.2, 0.3]:
            candidates.append(normalize(np.exp(-((x - c) ** 2) / (2 * sg * sg))))
    
    # Cosine taper
    for alpha in [0.1, 0.2, 0.3, 0.5]:
        cands = np.ones(n)
        m = max(1, int(alpha * n))
        taper = 0.5 * (1 + np.cos(np.pi * np.arange(m) / m))
        cands[:m] = taper
        cands[-m:] = taper[::-1]
        candidates.append(normalize(cands))
    
    # Triangular
    candidates.append(normalize(np.minimum(x, 1 - x)))
    
    # Plateau
    for w in [0.3, 0.5, 0.7]:
        cands = np.ones(n)
        cands[:int(w*n/4)] = 0
        cands[-int(w*n/4):] = 0
        candidates.append(normalize(cands))
    
    if parent_seq is not None:
        bi = normalize(parent_seq)
        for w in [0.5, 0.7, 0.9]:
            for c in candidates[:5]:
                candidates.append(normalize(w * bi + (1 - w) * c))
    
    for c in candidates:
        v = obj_val(c)
        if v < best_val:
            best_val = v
            best_a = c.copy()
    
    return best_a, best_val

3. Repeat until c is flat

def interpolate_seq(f, new_n):

        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)

        K = len(clist)
        if K == 0:
            break

        g_dir = lp_direction_cvxpy(y, active_set, solver)
        if g_dir is None:
            # Fallback: use the uniform direction on active set complement
            if no_improve > stall_limit // 3:
                break
            # Try with smaller set
            if len(clist) > 100:
                active_set = set(int(clist[i]) for i in np.argsort(ac)[-200:])
                clist = sorted(active_set)
                continue
            break

        # Check violations and add to active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]
            for k in top_v:
                active_set.add(int(k))

        # Golden section line search
        lo = 0.0
        hi = 1.0
        c1 = lo + inv_phi * (hi - lo)
        c2 = hi - inv_phi * (hi - lo)

        def trial_obj(alpha):
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                return 1e10
            trial = trial / ts
            return fast_autocorr(trial).max()

        f1 = trial_obj(c1)
        f2 = trial_obj(c2)
        for _ in range(18):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = lo + inv_phi * (hi - lo)
                f1 = trial_obj(c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = hi - inv_phi * (hi - lo)
                f2 = trial_obj(c2)

        best_alpha = (lo + hi) / 2.0
        y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy
        else:
            y = g_dir.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(2 * n * best_m) + " |S|=" + str(len(active_set)))

    if verbose:
        print("    LP-FW done (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2 * n * best_m))
    return best_a, 2.0 * n * best_m

def smoothed_max_grad(a0, time_limit, beta_start=5.0, beta_end=100.0, lr_base=0.5, momentum=0.85, verbose=True):
    """Gradient descent on log-sum-exp smoothed max of autocorrelation."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    m = np.zeros(n, dtype=np.float64)
    stall = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 3000:
            # Random restart
            rng = np.random.default_rng(t)
            noise = 0.02 * rng.standard_normal(n) * best_a
            a = best_a + noise
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            m = np.zeros(n)
            stall = 0

        b = fast_autocorr(a)
        b_max = b.max()
        b_shift = b - b_max
        beta = beta_start + (beta_end - beta_start) * min(1.0, t / 3000.0)
        weights = np.exp(beta * b_shift)
        weights /= weights.sum()

        # Gradient: grad[j] = 2 * sum_k weights[k] * a[k-j] (where 0<=k-j<n)
        # Use FFT for the convolution-like operation
        # c[j] = sum_k w[k] * a[k-j] = (w_flip * a)[2n-2-j] in full conv
        # Equivalently: cross-correlation
        W_f = np.fft.rfft(weights, n=fft_size)
        A_f = np.fft.rfft(a, n=fft_size)
        # First moment: sum_k w[k] (this is 1) -> contributes a[j] * 1 to grad
        # Second: sum_k w[k] * a[k-j] for valid k-j
        # cross_corr = ifft(fft(w) * conj(fft(a))) gives (w * a_rev)[shift]
        # Let me compute: c[j] = sum_k w[k] * a[k-j] = (w ⊛ a)[j]
        # Using: ifft(fft(w) * conj(fft(a))) = sum_k w[k+l] * a[k] ... hmm
        # Let me just do it with a direct FFT convolution:
        # c[j] = sum_{i=0}^{n-1} a[i] * w[j+i] (where j+i < 2n-1)
        # Define w_ext of length 2n-1 (the full autocorr has this length)
        # c[j] = sum_i a[i] * w[j+i] = (a * w_flip)[j + 2n - 2] where w_flip[i] = w[2n-2-i]
        w_flip = weights[::-1]
        ac_f = np.fft.rfft(a, n=fft_size)
        wf_f = np.fft.rfft(w_flip, n=fft_size)
        c_full = np.fft.irfft(ac_f * wf_f, fft_size)
        # c[j] = c_full[j + n - 1] (shifted)
        # Actually: (a * w_flip)[m] = sum_i a[i] * w_flip[m-i] = sum_i a[i] * w[2n-2-(m-i)]
        # We want sum_i a[i] * w[j+i], so 2n-2-(m-i) = j+i -> m = 2n-2-j
        # So c[j] = c_full[2n-2-j]
        c = c_full[2*n-2 - np.arange(n)]
        c = np.maximum(c, 0)
        
        grad = 2.0 * (c + a)  # a[j] from the "identity" part + c[j] from cross term
        
        # Project gradient onto sum=0 subspace (normalization constraint)
        grad = grad - grad.mean()
        
        # Momentum update
        lr = lr_base / (1.0 + 0.05 * np.sqrt(t))
        m = momentum * m + lr * grad
        a_new = a - m
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        val = obj_val(a_new)
        if val < best_val - 1e-14:
            best_val = val
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Half step
            a_half = a - 0.3 * m
            a_half = np.maximum(a_half, 1e-15)
            a_half = a_half / a_half.sum()
            val_half = obj_val(a_half)
            if val_half < val:
                a = a_half
            stall += 1

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print("  SmMax t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(best_val) + " beta=" + repr(round(beta, 1)))

    return best_a, best_val

def peak_deflation(a0, time_limit, n_targets=8, verbose=True):
    """Target specific peak lags and reduce them."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for rnd in range(300):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        targets = np.argsort(ac)[::-1][:n_targets]
        improved = False

        for k in targets:
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-activity destinations
            scores = np.zeros(n)
            for k2 in targets:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
            scores[src] = 1e10
            scores += 0.01 * a  # prefer low-mass destinations

            candidates = np.argsort(scores)[:40]
            best_m_trial = best_m
            best_frac = None
            best_dst = None

            for dst in candidates:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in [0.003, 0.008, 0.015, 0.03]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m_trial - 1e-15:
                        best_m_trial = Mt
                        best_frac = frac
                        best_dst = dst

            if best_frac is not None:
                amt = a[src] * best_frac
                a[src] -= amt
                a[best_dst] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    best_m = fast_autocorr(a).max()
                    best_a = a.copy()
                    improved = True

        if not improved:
            for _ in range(8):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            improved = True
                            break

        if verbose and rnd % 100 == 0:
            print("    Deflate rnd=" + str(rnd) + " J=" + repr(2 * n * best_m))

    return best_a, 2.0 * n * best_m

def coord_polish(a0, time_limit, seed=42, verbose=False):
    """Coordinate descent with smart move selection."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            top_srcs = idx[np.argsort(contrib)[::-1][:2]]
            for src in top_srcs:
                src = int(src)
                if a[src] < 1e-12:
                    continue
                scores = np.zeros(n)
                for k2 in top_k[:6]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                scores[src] = 1e10
                candidates = np.argsort(scores)[:25]
                found = False
                for dst in candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.003, 0.007, 0.015, 0.03]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-15:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            found = True
                            break
                    if found:
                        break

        if not improved:
            stall += 1
            if stall > 100:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.012)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

    return best_a, 2.0 * n * best_m

def diverse_init(n, parent_seq=None):
    """Generate diverse initial candidates and return the best."""
    x = np.linspace(0.0, 1.0, n)
    best_a = normalize(np.sin(np.pi * x) ** 2)
    best_val = obj_val(best_a)

    candidates = [best_a]
    for f in [1, 2, 3]:
        candidates.append(normalize(np.sin(np.pi * x) ** (2 * f)))
    for c in [0.3, 0.5, 0.7]:
        for sg in [0.15, 0.25, 0.35]:
            candidates.append(normalize(np.exp(-((x - c) ** 2) / (2 * sg * sg))))
    for alpha in [0.1, 0.2, 0.4]:
        cands = np.ones(n)
        m = max(1, int(alpha * n))
        taper = 0.5 * (1 + np.cos(np.pi * np.arange(m) / m))
        cands[:m] = taper
        cands[-m:] = taper[::-1]
        candidates.append(normalize(cands))

    if parent_seq is not None:
        bi = normalize(parent_seq)
        for w in [0.5, 0.75, 0.9]:
            for c in candidates[:6]:
                candidates.append(normalize(w * bi + (1 - w) * c))

    for c in candidates:
        v = obj_val(c)
        if v < best_val:
            best_val = v
            best_a = c.copy()

    return best_a, best_val

def propose_candidate(budget_s=1800.0):
    """Main search function: multi-phase optimization with length exploration."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print("Total budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            n_parent = len(best_seq)
            print("Parent loaded: n=" + str(n_parent) + " score=" + repr(best_score))
    except Exception:
        n_parent = 13700
        x = np.linspace(0, 1, n_parent)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)
        print("Default init n=" + str(n_parent) + " score=" + repr(best_score))

    # === Phase 1: Multi-length LP-FW (main optimization) ===
    # Try the parent's length and a couple of neighbors
    lengths_to_try = []
    if len(best_seq) > 1000:
        n_base = len(best_seq)
        lengths_to_try = [n_base]
        if n_base > 2000:
            lengths_to_try.append(int(n_base * 1.15))
        if n_base < 20000:
            lengths_to_try.append(int(n_base * 0.85))
    else:
        lengths_to_try = [n_parent]

    # Remove duplicates and cap
    lengths_to_try = sorted(set(lengths_to_try))
    lengths_to_try = [l for l in lengths_to_try if 3000 <= l <= 25000]
    if not lengths_to_try:
        lengths_to_try = [n_parent]

    print("\n=== P1: Multi-length LP-FW ===")
    for n_try in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break

        a_init = interpolate_seq(best_seq, n_try)
        t_phase = min(remaining * 0.35, 500.0)
        if t_phase < 60:
            t_phase = remaining * 0.4

        print("  LP-FW at n=" + str(n_try) + " budget=" + repr(round(t_phase, 0)) + "s")
        a_out, v_out = lp_fw_optimize(a_init, t_phase, solver,
                                       init_K=500, max_K=1000, add_per=40, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  LP-FW improved: " + repr(best_score))
        else:
            # Still try the output at its own length
            if np.isfinite(v_out) and v_out < best_score * 1.005:
                # Close enough to keep as candidate for further optimization
                candidate_at_len = a_out
                sc2 = eval_safe(candidate_at_len)
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = candidate_at_len

    # === Phase 2: Smoothed-max gradient at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P2: Smoothed-Max Grad (n=" + str(n_cur) + ") ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 40:
        a_out, v_out = smoothed_max_grad(best_seq, t2,
                                         beta_start=8.0, beta_end=120.0,
                                         lr_base=0.4, momentum=0.85, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  SmMax improved: " + repr(best_score))

    # === Phase 3: Peak deflation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_defl = min(n_cur, 15000)
    a_defl = interpolate_seq(best_seq, n_defl) if n_defl != n_cur else best_seq.copy()
    print("\n=== P3: Peak Deflation (n=" + str(n_defl) + ") ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_out, v_out = peak_deflation(a_defl, t3, n_targets=10, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = interpolate_seq(a_out, n_cur) if n_cur > n_defl else a_out
            print("  Deflation improved: " + repr(best_score))

    # === Phase 4: LP-FW again (second round, refined active set) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_lp2 = min(n_cur, 15000)
    a_lp2 = interpolate_seq(best_seq, n_lp2) if n_lp2 != n_cur else best_seq.copy()
    print("\n=== P4: LP-FW Round 2 (n=" + str(n_lp2) + ") ===")
    t4 = min(remaining * 0.25, 350.0)
    if t4 > 50:
        a_out, v_out = lp_fw_optimize(a_lp2, t4, solver,
                                       init_K=700, max_K=1200, add_per=50, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = interpolate_seq(a_out, n_cur) if n_cur > n_lp2 else a_out
            print("  LP-FW2 improved: " + repr(best_score))

    # === Phase 5: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_pol = min(n_cur, 15000)
    a_pol = interpolate_seq(best_seq, n_pol) if n_pol != n_cur else best_seq.copy()
    print("\n=== P5: Coord Polish (n=" + str(n_pol) + ") ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 20:
        a_out, v_out = coord_polish(a_pol, t5, seed=123, verbose=False)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = interpolate_seq(a_out, n_cur) if n_cur > n_pol else a_out
            print("  Polish improved: " + repr(best_score))

    # === Phase 6: If time remains, try a random restart + quick optimize ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        rng = np.random.default_rng(2024)
        print("\n=== P6: Random Restart ===")
        for attempt in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            noise_scale = 0.01 + 0.005 * attempt
            noise = noise_scale * rng.standard_normal(n_cur) * best_seq
            a_restart = best_seq + noise
            a_restart = np.maximum(a_restart, 1e-10)
            a_restart = a_restart / a_restart.sum()
            
            # Quick LP-FW
            t_fast = min(remaining * 0.3, 120.0)
            if t_fast < 15:
                break
            a_out, v_out = lp_fw_optimize(a_restart, t_fast, solver,
                                           init_K=400, max_K=600, add_per=20, verbose=False)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  Restart" + str(attempt+1) + " improved: " + repr(best_score))
            else:
                # Try the smoothed gradient
                t_fast2 = min(remaining * 0.2, 80.0)
                if t_fast2 > 15:
                    a_out2, v_out2 = smoothed_max_grad(a_restart, t_fast2,
                                                        beta_start=10, beta_end=80,
                                                        lr_base=0.5, momentum=0.8, verbose=False)
                    sc2 = eval_safe(a_out2)
                    if np.isfinite(sc2) and sc2 < best_score:
                        best_score = sc2
                        best_seq = a_out2
                        print("  Restart" + str(attempt+1) + "-SmMax improved: " + repr(best_score))

    # === Final validation and return ===
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        else:
            # Fallback
            n_fb = n_parent if 'n_parent' in dir() else 13700
            x = np.linspace(0, 1, n_fb)
            best_seq = normalize(np.sin(np.pi * x) ** 2)
            best_score = eval_safe(best_seq)
        
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + " n=" + str(len(best_seq)) + " t=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: fallback")
        n_fb = 13700
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in np.asarray(best_seq, dtype=np.float64)]
    return result
