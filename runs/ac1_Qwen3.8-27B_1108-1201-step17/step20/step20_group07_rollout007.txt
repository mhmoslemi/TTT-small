"""
Minimizes 2*n*max(autocorr(a))/(sum(a))^2 using:
1. FFT-based smooth gradient descent (O(n log n) per iteration)
2. Adaptive beta (logsumexp smoothing parameter)
3. Peak-shaving heuristics for direct peak reduction
4. Multi-length exploration with interpolated transfers
5. Simultaneous equality enforcement (all top peaks at same level)

import numpy as np
import time

def next_power2(x):
    return 1 << (x - 1).bit_length()

def fast_autocorr(a):
    """Compute autocorrelation (linear convolution of a with itself)."""
    n = len(a)
    size = 2 * n - 1
    N = next_power2(size)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]

def simplex_project(z):
    """Project z onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))

def smooth_gradient_descent(a0, time_limit, beta_start=10.0, beta_end=2000.0,
                            beta_rate=1.003, step_scale=0.5, verbose=True):
    """
    Minimize max(autocorr(p)) using logsumexp-smoothed gradient descent.
    p is constrained to the probability simplex.
    """
    n = len(a0)
    p = norm_sum1(a0)
    best_p = p.copy()
    best_m = fast_autocorr(p).max()
    t0 = time.monotonic()
    beta = beta_start
    t = 0
    stall = 0
    stall_threshold = 300
    rng = np.random.default_rng(999)
    
    # Precompute FFT size
    N_fft = next_power2(3 * n - 2)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute autocorrelation
        b = fast_autocorr(p)
        
        # Check for improvement
        m_curr = b.max()
        if m_curr < best_m - 1e-15:
            best_m = m_curr
            best_p = p.copy()
            stall = 0
        
        # Compute softmax weights (numerically stable)
        b_shifted = beta * (b - b.max())
        b_shifted = np.clip(b_shifted, -500, 500)
        exp_b = np.exp(b_shifted)
        w = exp_b / exp_b.sum()
        
        # Compute gradient via FFT
        # g[j] = (1/beta) * (h[j] + p[j])
        # h[j] = sum_k w[k]*p[k-j] (zero-padded)
        W = np.fft.rfft(w, N_fft)
        P = np.fft.rfft(p, N_fft)
        h = np.fft.irfft(np.conj(W) * P, N_fft)[:n]
        g = (h + p) / beta
        
        # Adaptive step size
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            eta = 0.01
        else:
            eta = step_scale / g_norm
        
        # Limit step size to prevent oscillation
        eta = min(eta, 0.1)
        
        # Update
        p_new = p - eta * g
        p_new = simplex_project(p_new)
        
        # Verify improvement
        b_new = fast_autocorr(p_new)
        m_new = b_new.max()
        
        if m_new > m_curr * (1 + 0.01) + 1e-12:
            # Step was too aggressive, halve and retry
            eta /= 3
            p_new = p - eta * g
            p_new = simplex_project(p_new)
            b_new = fast_autocorr(p_new)
            m_new = b_new.max()
        
        p = p_new
        
        # Adaptive beta
        if t % 20 == 0:
            beta = min(beta * beta_rate, beta_end)
        
        # Stall detection and restart
        if t > stall_threshold and m_curr >= best_m:
            stall += 1
        else:
            stall = 0
        
        if stall > 500:
            # Perturb and restart
            noise_scale = 0.01 + 0.003 * (stall - 500) / 100
            pert = rng.normal(0, noise_scale, n) * best_p
            p_r = best_p + pert
            p_r = np.maximum(p_r, 1e-15)
            p_r = p_r / p_r.sum()
            m_r = fast_autocorr(p_r).max()
            if m_r < best_m:
                best_m = m_r
                best_p = p_r.copy()
            p = p_r
            stall = 0
            beta = max(beta * 0.5, 5.0)
            
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    SGD restart t=" + str(t) + " (" + str(round(elapsed,1)) + "s): J=" + repr(J))
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SGD t=" + str(t) + " (" + str(round(elapsed,1)) + "s): J=" + repr(J) + " beta=" + repr(round(beta,1)))
    
    return best_p, best_m


def peak_equalization(a0, time_limit, verbose=True):
    """
    Directly equalize the top peaks by moving mass from peak-contributing
    positions to low-pressure positions.
    """
    n = len(a0)
    p = norm_sum1(a0)
    best_p = p.copy()
    best_m = fast_autocorr(p).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        b = fast_autocorr(p)
        m_curr = b.max()
        
        if m_curr < best_m - 1e-15:
            best_m = m_curr
            best_p = p.copy()
            stall = 0
        else:
            stall += 1
        
        # Find top peaks
        top_k = min(20, len(b))
        peak_idx = np.argsort(b)[::-1][:top_k]
        peak_val = b[peak_idx[0]]
        
        # For each top peak, try to reduce it
        improved = False
        for k_i in peak_idx[:10]:
            k_int = int(k_i)
            # Find the pair (i, k-i) with maximum contribution to this peak
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            contrib = p[idx] * p[k_int - idx]
            max_contrib_idx = idx[np.argmax(contrib)]
            src = int(max_contrib_idx)
            partner = k_int - src
            
            if partner < 0 or partner >= n:
                continue
            
            # Find a safe destination (low autocorrelation contribution)
            # Compute "pressure" at each position
            pressure = np.zeros(n)
            for k2 in peak_idx[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += p[idx2] * p[k2i - idx2]
                    pressure[k2i - idx2] += p[idx2] * p[k2i - idx2]
            
            pressure[src] = np.inf  # Don't move to self
            pressure[partner] = np.inf  # Don't move to partner
            
            dst = int(np.argmin(pressure))
            if pressure[dst] >= np.inf:
                continue
            
            # Try to move mass
            for frac in [0.01, 0.02, 0.05, 0.1]:
                amt = p[src] * frac
                if amt < 1e-15:
                    continue
                p_trial = p.copy()
                p_trial[src] -= amt
                p_trial[dst] += amt
                p_trial = np.maximum(p_trial, 0.0)
                s_t = p_trial.sum()
                if s_t < 1e-15:
                    continue
                p_trial = p_trial / s_t
                m_trial = fast_autocorr(p_trial).max()
                if m_trial < m_curr - 1e-16:
                    p = p_trial
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation when stuck
            if stall % 30 == 0:
                noise = 0.005 * min(stall / 30, 5.0)
                pert = rng.normal(0, noise, n) * best_p
                p_r = best_p + pert
                p_r = np.maximum(p_r, 1e-15)
                p_r = p_r / p_r.sum()
                m_r = fast_autocorr(p_r).max()
                if m_r < best_m:
                    best_m = m_r
                    best_p = p_r.copy()
                p = p_r if m_r < m_curr + 1e-6 else best_p.copy()
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakEq stall=" + str(stall) + " (" + str(round(elapsed,1)) + "s): J=" + repr(J))
    
    return best_p, best_m


def local_search(a0, time_limit, verbose=True):
    """
    Local search: try small perturbations (move mass between pairs) and accept

    """
    n = len(a0)
    p = norm_sum1(a0)
    best_p = p.copy()
    best_m = fast_autocorr(p).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    # Precompute FFT size
    N_fft = next_power2(2 * n - 1)
    
    def fast_eval(p_in):
        b = fast_autocorr(p_in)
        return b.max()
    
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        b = fast_autocorr(p)
        m_curr = b.max()
        
        if m_curr < best_m - 1e-15:
            best_m = m_curr
            best_p = p.copy()
        
        # Identify top peaks
        top_peaks = np.argsort(b)[::-1][:15]
        
        # For the top peak, find the worst contributing pair
        k_top = int(top_peaks[0])
        i_start = max(0, k_top - n + 1)
        i_end = min(n - 1, k_top)
        idx = np.arange(i_start, i_end + 1)
        contrib = p[idx] * p[k_top - idx]
        
        # Try several random moves
        for trial in range(5):
            # Source: one of the top contributing positions
            src_candidates = idx[np.argsort(contrib)[::-1][:max(3, len(idx)//10)]]
            src = int(rng.choice(src_candidates))
            partner = k_top - src
            if partner < 0 or partner >= n:
                continue
            
            # Destination: random position with low value
            weights = 1.0 + p  # prefer positions with low p
            dst = int(rng.choice(n, p=weights/weights.sum()))
            
            if dst == src or dst == (partner if 0<=partner<n else -1):
                continue
            
            # Move a small fraction
            frac = rng.uniform(0.01, 0.1)
            amt = p[src] * frac
            if amt < 1e-16:
                continue
            
            p_trial = p.copy()
            p_trial[src] = max(0, p[src] - amt)
            p_trial[dst] += amt
            p_trial = np.maximum(p_trial, 0.0)
            s_t = p_trial.sum()
            if s_t < 1e-15:
                continue
            p_trial = p_trial / s_t
            m_trial = fast_eval(p_trial)
            if m_trial < m_curr - 1e-16:
                p = p_trial
                break  # Accept and move to next outer iteration
        
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LocalSearch it=" + str(iter_count) + " (" + str(round(elapsed,1)) + "s): J=" + repr(J))
    
    return best_p, best_m


def interpolate_sequence(f, new_n):
    """Interpolate sequence f to new length new_n, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed,1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main search function. Uses FFT-based smooth gradient descent as the core
    optimizer, with peak equalization and local search for refinement.
    Explores multiple sequence lengths.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print("Parent loaded: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        pass
    
    if best_seq is None:
        # Default initialization
        n_base = 7000
        x = np.linspace(0, 1, n_base)
        # Use a raised-cosine-like shape
        a_raw = 0.5 + 0.5 * np.cos(np.pi * (x - 0.5))
        a_raw = np.maximum(a_raw, 0.01)
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))
    
    n0 = len(best_seq)
    print("Starting: n=" + str(n0) + ", J=" + repr(best_score))
    
    # === Phase 1: Main gradient descent on parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Smooth GD (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 30:
        a_out, _ = smooth_gradient_descent(best_seq, t1, beta_start=8.0, beta_end=3000.0,
                                           beta_rate=1.004, step_scale=0.4, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sgd_n" + str(n0), t_global)

    # === Phase 3: Second gradient descent (higher beta) ===
