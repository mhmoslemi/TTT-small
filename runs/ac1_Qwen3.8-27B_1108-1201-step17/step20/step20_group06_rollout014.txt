def simplex_project(z):
    z=np.asarray(z,dtype=np.float64)
    n=z.size
    u=np.sort(z)[::-1]
    cssv=np.cumsum(u)
    rho=np.nonzero(u*np.arange(1,n+1) > (cssv-1.0))[0]
    if len(rho)==0:
       return np.ones(n)/n
    r=rho[-1]
    theta=(cssv[r]-1.0)/(r+1.0)
    return np.maximum(z-theta,0.0)
```

"""
Nonnegative linear-convolution peak minimizer.

The scored quantity is 2*n*max(np.convolve(a,a))/(sum(a))^2.  After
normalizing sum(a)=1, this is 2*n*max_conv(a).  Optimizations below use
exact FFT-based linear convolutions, not circular autocorrelation.

Main components:
- Smoothed peak-gradient steps on the simplex.
- Coordinate descent on a weighted-quadratic surrogate sum(p_k b_k).
- Spectrum-shaping projection toward a flat target convolution spectrum.
- Multi-start / multi-length / mixture search around the supplied parent.
"""

import numpy as np
import time

MAX_N = 20000

P_SPECS = [
    {"mode": "soft", "rel_tau": 0.005, "topK": 25},
    {"mode": "violation", "target_frac": 0.005, "q": 1.0},

def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def to_return_list(a):
    arr = np.asarray(a, dtype=np.float64).reshape(-1)
    out = []
    for x in arr:
        xx = float(x)
        if not np.isfinite(xx):
            xx = 0.0
        if xx < 0.0:
            xx = 0.0
        out.append(float(xx))
    return out


def score_key(item):
    return item[0]


def conv_same_full(a, F=None):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    if F is None:
        F = _next_pow2(L)
    A = np.fft.rfft(a, F)
    b = np.fft.irfft(A * A, F)[:L]
    b = np.maximum(b, 0.0)
    return b


def conv_same_max(a, F=None):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return float("inf")
    L = 2 * n - 1
    if F is None:
        F = _next_pow2(L)
    A = np.fft.rfft(a, F)
    b = np.fft.irfft(A * A, F)[:L]
    b = np.maximum(b, 0.0)
    return float(np.max(b))


def simplex_project(z):
    z = np.asarray(z, dtype=np.float64).reshape(-1)
    n = z.size
    if n == 0:
        return np.array([], dtype=np.float64)
    z = np.nan_to_num(z, nan=0.0, posinf=1e6, neginf=-1e6)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if rho.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    r = int(rho[-1])
    theta = (cssv[r] - 1.0) / float(r + 1)
    return np.maximum(z - theta, 0.0)


def hankel_product(p, a, F):
    """
    h_i = sum_j p_{i+j} a_j, with p length 2n-1 and a length n.
    This is (1/2) of the gradient of sum_k p_k b_k.
    """
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    pr = p[::-1]
    A = np.fft.rfft(a, F)
    P = np.fft.rfft(pr, F)
    c = np.fft.irfft(A * P, F)
    h = c[(n - 1):(2 * n - 1)][::-1]
    h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)
    return np.maximum(h, 0.0)


def make_weights(b, M, mode="soft", rel_tau=0.02, topK=0, target_frac=0.02, q=1.0):
    b = np.asarray(b, dtype=np.float64).reshape(-1)
    L = b.size
    p = np.zeros(L, dtype=np.float64)
    if L == 0:
        return p
    if M <= 0.0 or not np.isfinite(M):
        p[0] = 1.0
        return p

    if mode == "soft":
        k = int(max(1, min(int(topK), L)))
        if k < L:
            idx = np.argpartition(b, -k)[-k:]
        else:
            idx = np.arange(L)
        tau = max(float(M) * max(float(rel_tau), 1e-8), 1e-300)
        vals = b[idx]
        w = np.exp(np.clip((vals - float(M)) / tau, -700.0, 0.0))
        s = float(w.sum())
        if s <= 0.0:
            pm = idx[int(np.argmax(vals))]
            p[pm] = 1.0
            return p
        p[idx] = w / s
    elif mode == "violation":
        target = float(M) * (1.0 - float(target_frac))
        if target < 0.0:
            target = 0.0
        v = np.maximum(b - target, 0.0)
        if q != 1.0:
            v = np.power(v, float(q))
        s = float(v.sum())
        if s <= 0.0:
            k = int(max(1, min(20, L)))
            if k < L:
                idx = np.argpartition(b, -k)[-k:]
            else:
                idx = np.arange(L)
            p[idx] = 1.0 / float(idx.size)
        else:
            p = v / s
    elif mode == "uniform_top":
        k = int(max(1, min(int(topK), L)))
        if k < L:
            idx = np.argpartition(b, -k)[-k:]
        else:
            idx = np.arange(L)
        p[idx] = 1.0 / float(k)
    else:
        p = np.ones(L, dtype=np.float64) / float(L)

    s = float(p.sum())
    if s <= 0.0:
        p[0] = 1.0
    else:
        p = p / s
    return p


def smooth_fft(a, width):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return a
    F = _next_pow2(2 * n)
    A = np.fft.rfft(a, F)
    f = np.fft.rfftfreq(F)
    sigma = max(float(width) * float(n), 1e-6)
    A = A * np.exp(-2.0 * np.pi * np.pi * (f * sigma) ** 2)
    out = np.fft.irfft(A, F)[:n]
    return np.maximum(out, 0.0)


def modulate_sequence(base, freq, eps, phase):
    base = np.asarray(base, dtype=np.float64).reshape(-1)
    n = base.size
    if n == 0:
        return base
    x = np.arange(n, dtype=np.float64)
    m = 1.0 + float(eps) * np.sin(2.0 * np.pi * float(freq) * x / float(n) + float(phase))
    m = np.maximum(m, 0.0)
    out = base * m
    s = float(out.sum())
    if s < 1e-15:
        return base.copy()
    return out / s


def interpolate_sequence(f, new_n):
    f = np.asarray(f, dtype=np.float64).reshape(-1)
    new_n = int(max(1, new_n))
    old_n = f.size
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    if old_n == new_n:
        return f.copy()
    s = float(f.sum())
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    out = np.interp(new_x, old_x, fn)
    out = np.maximum(out, 0.0)
    return normalize(out)


def spectral_flat_step(a, alpha):
    """
    Move the squared DFT toward the DFT of a flat nonnegative convolution,
    take a principal square root, invert, and project to nonnegative simplex.
    """
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return a
    L = 2 * n - 1
    F = _next_pow2(L)
    A = np.fft.rfft(a, F)
    B = A * A

    w = 2.0 * np.pi * np.arange(B.size, dtype=np.float64) / float(F)
    denom = np.sin(0.5 * w)
    ratio = np.empty_like(w)
    mask = np.abs(denom) > 1e-14
    ratio[~mask] = float(L)
    ratio[mask] = np.sin(0.5 * w[mask] * float(L)) / denom[mask]
    target = np.exp(-0.5j * w * (L - 1)) * ratio / float(L)

    Bmix = (1.0 - float(alpha)) * B + float(alpha) * target
    Anew = np.sqrt(Bmix)
    aout = np.fft.irfft(Anew, F)[:n]
    return normalize(np.maximum(aout, 0.0))


def make_random_init(n, seed=0):
    n = int(max(2, min(int(n), MAX_N)))
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    width = 0.004 + 0.08 * float(rng.random())
    a = smooth_fft(a, width)
    a = 0.75 * a + 0.25 * (1.0 / float(n))
    return normalize(a)


def default_init(n, seed=0):
    n = int(max(100, min(int(n), MAX_N)))
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 + 0.5 * np.sin(np.pi * x) ** 2

def make_perturbed(a, seed, strength=1.0):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return a
    rng = np.random.default_rng(seed)
    strength = max(0.05, min(float(strength), 2.0))
    method = int(rng.integers(6))

    if method == 0:
        noise = rng.normal(0.0, 0.05 * strength / float(n), size=n)
        out = a + noise
    elif method == 1:
        r = make_random_init(n, seed + 17)
        out = (1.0 - 0.4 * strength) * a + 0.4 * strength * r
    elif method == 2:
        freq = int(rng.integers(1, 8)) + 1
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        eps = (0.08 + 0.35 * float(rng.random())) * strength
        out = modulate_sequence(a, freq, eps, phase)
    elif method == 3:
        max_shift = max(2, n // 80)
        shift = int(rng.integers(1, max_shift + 1))
        rolled = np.roll(a, shift)
        out = 0.75 * a + 0.25 * rolled
    elif method == 4:
        alpha = 0.25 + 0.75 * strength
        out = spectral_flat_step(a, alpha)
    else:
        new_n = int(max(2, min(round(float(n) * (0.75 + 0.55 * float(rng.random()))), float(MAX_N))))
        out = interpolate_sequence(interpolate_sequence(a, new_n), n)

    return normalize(np.maximum(out, 0.0))


def gradient_trial(a, b, M, p, Fc, Fh):
    pa = hankel_product(p, a, Fh)
    grad = 2.0 * pa
    d = grad - float(np.mean(grad))
    scale = float(np.max(np.abs(d)))
    typical = 1.0 / float(a.size)
    if scale < 1e-12 * typical:
        return None

    rels = (1e-5, 3e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 1e-1)
    best_x = None
    best_m = float("inf")

    for rel in rels:
        alpha = float(rel) * typical / scale
        if alpha <= 0.0:
            continue
        z = a - alpha * d
        xn = simplex_project(z)
        mx = conv_same_max(xn, Fc)
        if mx < best_m:
            best_m = mx
            best_x = xn

    if best_x is not None and best_m < float(M) - 1e-15:
        b_new = conv_same_full(best_x, Fc)
        return best_x, b_new, best_m
    return None


def coordinate_cycle(a, b, M, p, Fc, Fh, time_sub):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return None
    if p.size != 2 * n - 1:
        return None

    t0 = time.monotonic()
    a_cur = a.copy()
    Pa = hankel_product(p, a_cur, Fh)
    G = float(np.dot(a_cur, Pa))
    G_start = G
    best_tol = 1e-15 * max(1.0, float(M))

    max_moves = 38
    sort_every = 7
    list_size = 45

    order = np.argsort(Pa)
    high = order[::-1][:list_size]
    low = order[:list_size]

    for mv in range(max_moves):
        if time.monotonic() - t0 > time_sub:
            break

        best = None
        best_dg = 0.0

        for ii in high:
            iint = int(ii)
            ai = float(a_cur[iint])
            if ai <= 1e-15:
                continue
            pai = float(Pa[iint])
            p2i = float(p[2 * iint])
            for jj in low:
                jint = int(jj)
                if iint == jint:
                    continue
                diff = pai - float(Pa[jint])
                if diff <= 1e-15:
                    continue
                denom = p2i + float(p[2 * jint]) - 2.0 * float(p[iint + jint])
                maxd = 0.5 * ai
                if denom > 1e-18:
                    d = diff / denom
                    if d > maxd:
                        d = maxd
                    if d <= 1e-15:
                        d = 0.0
                else:
                    d = maxd
                if d <= 1e-15:
                    continue
                for fac in (1.0, 0.5, 0.1):
                    dd = d * float(fac)
                    if dd <= 1e-15:
                        continue
                    dg = 2.0 * dd * (float(Pa[jint]) - pai) + dd * dd * denom
                    if dg < best_dg - 1e-22:
                        best_dg = float(dg)
                        best = (iint, jint, dd)

        if best is None:
            break

        iint, jint, dd = best
        old_pai = float(Pa[iint])
        old_paj = float(Pa[jint])
        old_denom = float(p[2 * iint]) + float(p[2 * jint]) - 2.0 * float(p[iint + jint])

        a_cur[iint] -= dd
        a_cur[jint] += dd
        if a_cur[iint] < 0.0:
            a_cur[iint] = 0.0

        G += 2.0 * dd * (old_paj - old_pai) + dd * dd * old_denom
        Pa += dd * (p[jint:jint + n] - p[iint:iint + n])

        if (mv + 1) % sort_every == 0:
            order = np.argsort(Pa)
            high = order[::-1][:list_size]
            low = order[:list_size]

    b_cur = conv_same_full(a_cur, Fc)
    M_cur = float(np.max(b_cur))

    s = float(a_cur.sum())
    if s < 1e-12 or abs(s - 1.0) > 1e-10:
        a_cur = normalize(a_cur)
        b_cur = conv_same_full(a_cur, Fc)
        M_cur = float(np.max(b_cur))

    if M_cur < float(M) - best_tol:
        return a_cur, b_cur, M_cur

    G_tol = max(1e-15, 1e-8 * abs(G_start))
    if M_cur <= float(M) * (1.0 + 1e-4) + best_tol and G < G_start - G_tol:
        return a_cur, b_cur, M_cur

    return None


def spectral_trial(a, b, M, Fc, alphas=(0.2, 0.5, 0.8, 1.0)):
    best = None
    for al in alphas:
        if al <= 0.0:
            continue
        an = spectral_flat_step(a, al)
        mx = conv_same_max(an, Fc)
        if best is None or mx < best[1]:
            best = (an, mx)

    if best is not None and best[1] < float(M) - 1e-15:
        bnew = conv_same_full(best[0], Fc)
        return best[0], bnew, best[1]
    return None


def local_optimizer(a0, time_limit, seed=0, tag=""):
    a = normalize(a0)
    n = a.size
    if n == 0:
        return a, float("inf")
    if n > MAX_N:
        a = interpolate_sequence(a, MAX_N)
        n = a.size

    t0 = time.monotonic()
    Fc = _next_pow2(2 * n - 1)
    Fh = _next_pow2(3 * n - 3)

    b = conv_same_full(a, Fc)
    M = float(np.max(b))
    best_a = a.copy()
    best_M = M
    last_print = 0.0
    fail = 0

    if time_limit > 5.0:
        print("  local " + str(tag) + " n=" + str(n) + " start J=" +
              repr(round(2.0 * n * best_M, 6)))

    while True:
        elapsed = time.monotonic() - t0
        remaining = float(time_limit) - elapsed
        if remaining <= 0.5:
            break

        if M < best_M - 1e-16:
            best_M = M
            best_a = a.copy()

        spec = P_SPECS[int(fail) % len(P_SPECS)]
        p = make_weights(b, M, spec["mode"], spec["rel_tau"], spec["topK"],
                         spec.get("target_frac", 0.02), spec.get("q", 1.0))

        improved = False

        sub = min(remaining, max(0.4, min(6.0, 0.8 + 0.3 * float(n) / 1000.0)))
        out = coordinate_cycle(a, b, M, p, Fc, Fh, sub)
        if out is not None:
            a, b, M = out
            fail = 0
            improved = True

        if not improved:
            out = gradient_trial(a, b, M, p, Fc, Fh)
            if out is not None:
                a, b, M = out
                fail = 0
                improved = True

        if not improved:
            fail += 1
            if int(fail) % 25 == 0:
                out = spectral_trial(a, b, M, Fc, (0.2, 0.5, 0.8, 1.0))
                if out is not None:
                    a, b, M = out
                    fail = 0
            if int(fail) % 60 == 0:
                strength = 0.5 + 0.1 * float(min(int(fail), 20))
                a = make_perturbed(best_a, int(seed) * 100000 + int(fail), strength)
                b = conv_same_full(a, Fc)
                M = float(np.max(b))
                fail = 0

        elapsed = time.monotonic() - t0
        if elapsed - last_print > 60.0:
            last_print = elapsed
            print("  local " + str(tag) + " elapsed=" + repr(round(elapsed, 1)) +
                  " curJ=" + repr(round(2.0 * n * M, 6)) +
                  " bestJ=" + repr(round(2.0 * n * best_M, 6)))

    if M < best_M:
        best_M = M
        best_a = a.copy()

    if time.monotonic() - t0 > 0.5:
        print("  local " + str(tag) + " end J=" + repr(round(2.0 * n * best_M, 6)))

    return best_a, best_M


def sequences_close(a, b):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    b = np.asarray(b, dtype=np.float64).reshape(-1)
    if a.size != b.size:
        return False
    return float(np.linalg.norm(a - b)) < 1e-7


def select_top_candidates(scored, max_count):
    out = []
    for item in scored:
        if len(out) >= int(max_count):
            break
        dup = False
        for existing in out:
            if sequences_close(item[1], existing[1]):
                dup = True
                break
        if not dup:
            out.append(item)
    return out


def record_candidate(score, a, archive, best_score, best_a):
    s = float(score)
    if not np.isfinite(s):
        return best_score, best_a
    arr = normalize(a)
    if arr.size == 0:
        return best_score, best_a
    if s < float(best_score):
        best_score = s
        best_a = arr.copy()
    archive.append((s, arr.copy()))
    return best_score, best_a


def evaluate_safe(a):
    try:
        lst = to_return_list(a)
        val = evaluate_sequence(lst)
        val = float(val)
        if np.isfinite(val):
            return val
        return float("inf")
    except Exception:
        return float("inf")


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    try:
        total_time = min(float(budget_s) - 45.0, 1740.0)
    except Exception:
        total_time = 1740.0
    if not np.isfinite(total_time) or total_time < 5.0:
        total_time = 5.0

    base = None
    parent_score = float("inf")
    try:
        p = height_sequence_1
        if p is not None:
            arr = np.asarray(p, dtype=np.float64).reshape(-1)
            arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
            arr = np.maximum(arr, 0.0)
            if arr.size > 0 and float(arr.sum()) > 1e-10:
                arr = normalize(arr)
                if arr.size > MAX_N:
                    arr = interpolate_sequence(arr, MAX_N)
                base = arr
    except Exception:
        base = None

    if base is None:
        base = default_init(8000, seed=0)

    best_a = base.copy()
    best_M = conv_same_max(best_a)
    best_score = 2.0 * float(len(best_a)) * best_M
    print("Initial/base: n=" + str(len(best_a)) + " J=" + repr(round(best_score, 9)))

    archive = [(best_score, best_a.copy())]

    candidates = []
    candidates.append(best_a)

    for w in (0.003, 0.008, 0.015, 0.03, 0.06):
        candidates.append(smooth_fft(best_a, w))

    mod_params = (
        (1, 0.0, 0.20),
        (1, np.pi, 0.35),
        (2, 0.0, 0.28),
        (3, 0.5 * np.pi, 0.32),
        (4, 0.0, 0.25),
        (5, np.pi, 0.20),
    )
    for f, ph, eps in mod_params:
        candidates.append(modulate_sequence(best_a, f, eps, ph))

    candidates.append(spectral_flat_step(best_a, 0.5))
    candidates.append(spectral_flat_step(best_a, 1.0))

    sg = best_a.copy()
    for alpha in (1.0, 0.85, 0.75, 0.65, 0.55):
        sg = spectral_flat_step(sg, alpha)
        sg = normalize(sg)
    candidates.append(sg)

    cand_n = len(best_a)
    candidates.append(make_random_init(cand_n, seed=12345))
    candidates.append(make_random_init(cand_n, seed=67890))

    comp = np.maximum(1.0 / float(cand_n) - best_a, 0.0)
    if float(comp.sum()) > 1e-10:
        candidates.append(normalize(comp))

    scored = []
    for c in candidates:
        cc = normalize(c)
        if cc.size == 0:
            continue
        M = conv_same_max(cc)
        sc = 2.0 * float(len(cc)) * M
        scored.append((sc, cc, M))

    scored.sort(key=score_key)
    for item in scored:
        best_score, best_a = record_candidate(item[0], item[1], archive, best_score, best_a)

    top_cands = select_top_candidates(scored, 6)
    local_fracs = (0.20, 0.14, 0.11, 0.09, 0.07, 0.06)

    for idx, item in enumerate(top_cands):
        rem = total_time - (time.monotonic() - t_global)
        if rem < 60.0:
            break
        frac = local_fracs[idx % len(local_fracs)]
        alloc = min(rem * frac, 320.0 if idx < 2 else 180.0)
        if alloc < 20.0:
            continue
        a_out, M_out = local_optimizer(item[1], alloc, seed=1000 + 17 * (idx + 1),
                                       tag="init" + str(idx + 1))
        sc_out = 2.0 * float(len(a_out)) * M_out
        best_score, best_a = record_candidate(sc_out, a_out, archive, best_score, best_a)

    multi_lengths = []
    best_n = len(best_a)
    for factor in (1.04, 1.12, 1.25, 1.50):
        ln = int(round(float(best_n) * factor))
        if ln > best_n + 20 and ln <= MAX_N:
            multi_lengths.append(ln)
    for ln in (10000, 13000, 16000, 20000):
        if ln > best_n + 20 and ln <= MAX_N:
            multi_lengths.append(ln)

    uniq = []
    for ln in multi_lengths:
        if ln not in uniq:
            uniq.append(ln)
    uniq.sort()
    multi_lengths = uniq[:4]

    rem = total_time - (time.monotonic() - t_global)
    if rem > 120.0 and len(multi_lengths) > 0:
        multi_budget = min(rem * 0.45, 650.0)
        per_alloc = multi_budget / float(len(multi_lengths))
        for idx, ln in enumerate(multi_lengths):
            rem = total_time - (time.monotonic() - t_global)
            if rem < 70.0:
                break
            alloc = min(per_alloc, rem)
            if alloc < 50.0:
                continue
            a_new = interpolate_sequence(best_a, ln)
            a_out, M_out = local_optimizer(a_new, alloc, seed=5000 + 31 * ln,
                                           tag="n" + str(ln))
            sc_out = 2.0 * float(len(a_out)) * M_out
            best_score, best_a = record_candidate(sc_out, a_out, archive, best_score, best_a)

    rem = total_time - (time.monotonic() - t_global)
    if rem > 90.0:
        archive.sort(key=score_key)
        other = None
        for sc, aa in archive:
            if aa.size != best_a.size or not sequences_close(aa, best_a):
                other = aa
                break
        if other is not None:
            other_m = interpolate_sequence(other, len(best_a))
            mix_alloc = min(70.0, rem * 0.25)
            for alpha in (0.35, 0.65):
                rem = total_time - (time.monotonic() - t_global)
                if rem < 35.0:
                    break
                mix = normalize((1.0 - alpha) * best_a + alpha * other_m)
                a_out, M_out = local_optimizer(mix, mix_alloc,
                                               seed=9000 + int(alpha * 100),
                                               tag="mix" + str(int(alpha * 100)))
                sc_out = 2.0 * float(len(a_out)) * M_out
                best_score, best_a = record_candidate(sc_out, a_out, archive, best_score, best_a)

    rem = total_time - (time.monotonic() - t_global)
    if rem > 80.0:
        alloc = min(rem * 0.55, 400.0)
        a_out, M_out = local_optimizer(best_a, alloc, seed=12345, tag="final1")
        sc_out = 2.0 * float(len(a_out)) * M_out
        best_score, best_a = record_candidate(sc_out, a_out, archive, best_score, best_a)

    rem = total_time - (time.monotonic() - t_global)
    if rem > 40.0:
        alloc = min(rem * 0.70, 250.0)
        a_out, M_out = local_optimizer(best_a, alloc, seed=54321, tag="final2")
        sc_out = 2.0 * float(len(a_out)) * M_out
        best_score, best_a = record_candidate(sc_out, a_out, archive, best_score, best_a)

    final_check = evaluate_safe(best_a)
    if np.isfinite(final_check) and final_check < best_score:
        best_score = final_check

    out = to_return_list(best_a)
    print("FINAL: J=" + repr(round(best_score, 9)) +
          ", n=" + str(len(out)) +
          ", elapsed=" + repr(round(time.monotonic() - t_global, 1)) + "s")
    return out
