"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FISTA with corrected Barzilai-Borwein step, adaptive top-k smoothing, Nesterov averaging
2. LP-Frank-Wolfe with warm-start, adaptive active set, and aggressive constraint management
3. Targeted peak-shaving coordinate descent on autocorrelation peaks
4. Cyclic shift enumeration
5. Multi-restart with diverse frequency-domain initializations
6. Adaptive n exploration
"""

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None


def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    return _MODULE_SOLVER


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_sm_grad(y, beta, top_k):
    n = len(y)
    b = fast_autocorr(y)
    size = len(b)
    if top_k is not None and top_k < size and top_k > 0:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = (b_use - b_max) * beta
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros(size)
        ties = (b_use == b_max)
        p[ties] = 1.0 / max(1, ties.sum())
    else:
        p = exp_vals / p_total
    # grad[j] = 2 * sum_{i=0}^{n-1} p[i+j] * y[i]
    L = 1 << (max(3 * n - 2, 1) - 1).bit_length()
    Pf = np.fft.rfft(p.astype(np.float64), L)
    Yrev = y[::-1].copy()
    Yrf = np.fft.rfft(Yrev, L)
    conv = np.fft.irfft(Pf * Yrf, L)[:3 * n - 2]
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, p, b


def fista_opt(a0, time_limit, beta_start=20.0, beta_end=1000.0,
              beta_growth=1.003, top_k_start=40, top_k_end=5,
              restart_every=3000, seed=42, verbose=True):
    n = len(a0)
    x = norm_sum1(a0).copy()
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    avg_x = x.copy()
    avg_count = 0.0
    s_prev = None
    y_prev = None
    bb_step = 1.0 / np.sqrt(n)

    for t in range(1, 80000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta_mom = (t_nest - 1.0) / t_new
        y = x + theta_mom * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Adaptive top-k: start high, decrease over iterations
        frac = min(1.0, t / (restart_every * 2.0))
        top_k = max(top_k_end, int(top_k_start * (1.0 - frac) + top_k_end * frac))

        grad, p, b = compute_sm_grad(y, beta, top_k)

        # Barzilai-Borwein step size
        if s_prev is not None and y_prev is not None:
            sy = y - y_prev
            gy = grad - s_prev
            sy_gy = sy.dot(gy)
            gy_gy = gy.dot(gy)
            if gy_gy > 1e-20:
                bb_candidate = sy_gy / gy_gy
                if bb_candidate > 1e-8:
                    bb_step = 0.5 * bb_step + 0.5 * bb_candidate
                    bb_step = max(1e-6, min(bb_step, 5.0 / beta))
                else:
                    bb_step = max(bb_step * 0.95, 1e-6)
            s_prev = grad.copy()
            y_prev = y.copy()

        eta = min(bb_step, 0.3)
        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        avg_count += 1.0
        avg_x = avg_x + (x_new - avg_x) / avg_count

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.02 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 8.0)
            s_prev = None
            y_prev = None
            bb_step = 1.0 / np.sqrt(n)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("  FISTA restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("  FISTA t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + repr(round(beta, 1)) + " tk=" + str(top_k))

    if verbose:
        J_avg = 2.0 * n * fast_autocorr(avg_x).max()
        J_best = 2.0 * n * best_max_b
        if J_avg < J_best:
            print("  FISTA: Nesterov avg better, using it. J=" + repr(round(J_avg, 6)))
            return avg_x, fast_autocorr(avg_x).max()
    return best_a, best_max_b


def lp_fw_opt(a0, time_limit, solver, initial_K=400, max_K=3000, add_per_iter=30, verbose=True):
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 600

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5),
                  max(0.02, default_alpha * 1.5), 1.0 / (t + 1), max(0.01, default_alpha * 3.0)]
        alphas = [max(0.01, min(1.0, a)) for a in alphas]
        alphas = list(set(alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  LP-FW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        # Focus on top 20 peaks
        top_peaks = np.argsort(ac)[::-1][:20]
        improved = False

        for _ in range(80):
            if time.monotonic() - t0 > time_limit:
                break

            idx_choice = rng.integers(len(top_peaks))
            k_int = int(top_peaks[idx_choice])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-20:
                continue

            # Find the index with max contribution in this peak
            src = int(idx[np.argmax(contrib)])

            # Find a good destination: low contribution to all top peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:12]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)

            # Score candidates by how much they contribute to peaks
            dst_scores = np.zeros(len(candidates))
            for ci, c in enumerate(candidates):
                for k2 in top_peaks[:8]:
                    k2i = int(k2)
                    if 0 <= k2i - c < n or 0 <= c - k2i < n:
                        dst_scores[ci] += a[c] * a[k2i - c] if 0 <= k2i - c < n else a[c] * a[c - k2i]
            dst = int(candidates[np.argmin(dst_scores)])

            if dst == src or a[src] < 1e-12:
                continue

            # Try different fractions
            best_frac = 0.01
            best_m_trial = best_m
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m_trial:
                    best_m_trial = M_trial
                    best_frac = frac
                    a_best_trial = a_trial.copy()

            if best_m_trial < best_m - 1e-14:
                amt = a[src] * best_frac
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                best_m = best_m_trial
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.002, 0.02))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  PeakShave (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def cyclic_shift_best(a0):
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_init(n, seed):
    """Create diverse initialization using frequency domain shaping."""
    rng = np.random.default_rng(seed)
    # Start with sinusoidal base
    x = np.linspace(0, 1, n)
    base = np.sin(np.pi * x) ** 2

    # Add random spectral components
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(base, fft_size)
    # Add random phase and amplitude perturbation
    n_freq = len(A)
    phase = rng.uniform(-0.3, 0.3, n_freq)
    amp = 1.0 + rng.uniform(-0.1, 0.1, n_freq)
    A_new = A * amp
    A_new[1:] *= np.exp(1j * phase[1:])
    A_new[0] *= amp[0]
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return base / base.sum()
    return a_new / s


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):

    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()

            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

        n_base = 6000
        x = np.linspace(0, 1, n_base)

        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    remaining = time_budget - (time.monotonic() - t_global)

    t1 = min(remaining * 0.32, 500.0)

    remaining = time_budget - (time.monotonic() - t_global)

    t4 = min(remaining * 0.12, 200.0)
    if t4 > 30:
        a_out, _ = peak_shaving(norm_sum1(best_seq), t4, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak1", t_global)

    # === Phase 5: FISTA v2 with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    t5 = min(remaining * 0.18, 300.0)
    if t5 > 50:

                             beta_growth=1.005, top_k_start=20, top_k_end=3,
                             restart_every=2000, seed=12345, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista2", t_global)

    # === Phase 6: Multi-n exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    candidate_ns = []
    for delta in [500, 1000, 1500, 2000]:
        nn = n_best + delta
        if nn <= 12000 and nn > n_best:
            candidate_ns.append(nn)
        nn2 = n_best - delta
        if nn2 >= 3000 and nn2 < n_best:
            candidate_ns.append(nn2)

    for new_n in candidate_ns[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("\n=== Phase 6: n=" + str(new_n) + " ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.25, 180.0)
        if t6a > 40:
            a_out, _ = fista_opt(a_long, t6a,
                                 beta_start=25.0, beta_end=600.0,
                                 beta_growth=1.004, top_k_start=35, top_k_end=5,
                                 restart_every=3000, seed=777, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
            best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_n" + str(new_n), t_global)

    # === Phase 7: LP burst ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: LP burst (n=" + str(n_best) + ") ===")
    t7 = min(remaining * 0.18, 250.0)
    if t7 > 30 and n_best <= 10000:
        a_out, _ = lp_fw_opt(norm_sum1(best_seq), t7, solver,
                             initial_K=600, max_K=3500, add_per_iter=40, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp2", t_global)

    # === Phase 8: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== Phase 8: Final peak shaving ===")
        t8 = min(remaining * 0.25, 150.0)
        if t8 > 20:
            a_out, _ = peak_shaving(norm_sum1(best_seq), t8, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_final", t_global)

    # === Final: Cyclic shift ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== Final: Cyclic shift ===")
        a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_final", t_global)

    # === Final: FISTA micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print("\n=== Final: FISTA micro-polish ===")
        t_f = min(remaining * 0.50, 100.0)
        if t_f > 20:
            a_out, _ = fista_opt(norm_sum1(best_seq), t_f,
                                 beta_start=150.0, beta_end=2500.0,
                                 beta_growth=1.01, top_k_start=10, top_k_end=2,
                                 restart_every=1500, seed=2024, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence! Fallback.")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
