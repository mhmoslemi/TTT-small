def simplex_project(z):
    """Project z onto {x >= 0, sum(x) = 1}"""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_grad_multi(a, beta, top_k=None):
    """Compute gradient using multi-scale smoothed max."""
    n = len(a)
    b = fast_autocorr(a)
    size = 2 * n - 1

    b_max = b.max()
    if b_max <= 0:
        return np.zeros(n), b, b_max

    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1]
        top_idx = idx[:top_k]
    else:
        top_idx = np.arange(size)

    # Primary gradient at given beta
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)

    # Suppress non-top-k to focus
    if top_k is not None and top_k < size:
        exp_vals[np.setdiff1d(np.arange(size), top_idx, assume_unique=False)] *= 0.3

    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s

    # Gradient via FFT: grad[i] = sum_k p[k] * a[k-i] (with bounds)
    a_rev = a[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1]

    return grad, b, b_max

            found = False

                x_new = simplex_project(z)

                eta = max(eta_trial * 0.8, min_eta)

        if max_new < best_max_b - 1e-16:
            best_max_b = max_new

            stall = 0

        x = x_new
        t_nest = t_new

            elif strategy == 2:

def spectral_refine(a0, time_limit, verbose=True, seed=777):
    """Spectral domain: iteratively flatten power spectrum above DC."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    fft_size = 1 << (2 * n - 2).bit_length()

    alpha = 0.03
    for it in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Target: match mean of non-DC bins
        target = power[1:].mean() if len(power) > 1 else power[0]
        target = max(target, 1e-30)

        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / target
            if ratio > 1.0:
                reduction = 1.0 - alpha * min(2.0, ratio - 1.0)
                A_new[k] *= max(0.2, np.sqrt(reduction))
            elif ratio < 0.3:
                boost = 1.0 + alpha * 0.2 * (0.3 - ratio)
                A_new[k] *= min(1.5, np.sqrt(boost))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 500 == 0:
            # Random phase reheat
            A = np.fft.rfft(a, fft_size)
            n_freq = len(A)
            phase_pert = rng.standard_normal(n_freq) * 0.03
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial

        if it % 300 == 0:
            alpha = max(0.005, alpha * 0.97)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def peak_coordinate_descent(a0, time_limit, verbose=True, seed=42):
    """
    Novel: Peak-targeted coordinate descent.
    For the top peaks in autocorrelation, identify contributing pairs (i, k-i)
    and redistribute mass to reduce the peak while monitoring other peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 3000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        size = 2 * n - 1

        # Focus on top 15 peaks
        top_peaks = np.argsort(ac)[::-1][:15]
        improved = False

        for k in top_peaks[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            if k_int >= size:
                continue

            # Find contributing pairs for this lag
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            partners = k_int - idx
            contrib = a[idx] * a[partners]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # Find the pair with maximum contribution
            local_max_idx = int(np.argmax(contrib))
            src_i = int(idx[local_max_idx])
            src_p = int(partners[local_max_idx])

            # Try moving mass from the larger of the pair to other locations
            if a[src_i] >= a[src_p]:
                donor = src_i
            else:
                donor = src_p

            # Try several destinations
            best_trial = None
            best_trial_m = best_m

            # Strategic destinations: positions that don't align with current peaks
            # and positions that can "split" the peak
            candidates = []
            # Near the donor (local moves)
            for d in [-50, -20, -10, -5, 5, 10, 20, 50]:
                dc = donor + d
                if 0 <= dc < n and dc != donor:
                    candidates.append(dc)
            # Far from donor (global moves)
            for _ in range(20):
                candidates.append(int(rng.integers(n)))

            for dst in candidates[:30]:
                if dst == donor:
                    continue
                frac = 0.03
                amt = a[donor] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[donor] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_trial_m - 1e-16:
                    best_trial_m = m_trial
                    best_trial = a_trial

            if best_trial is not None:
                a = best_trial
                best_m = best_trial_m
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            # When stalled, try a more aggressive random move
            if stall > 50 and stall % 10 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.01, 0.04)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 500 == 0 and stall > 50:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    CoordDesc (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def lp_fw_optimize(a0, time_limit, solver, initial_K=250, max_K=1500,
                   add_per_iter=20, verbose=True):
    """LP-Frank-Wolfe with constraint generation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 500

    for t in range(1, 300000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Adaptive line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        alphas = [best_alpha, best_alpha * 0.5, best_alpha * 2.0,
                  0.05, 0.1, 0.2, 0.3, 0.5]
        for alpha in alphas:
            alpha = max(0.01, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW it=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(round(p_score, 6)))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(round(best_score, 6)))

    n0 = len(best_seq)
    print("Start: n=" + str(n0) + ", J=" + repr(round(best_score, 6)))

    # Phase 1: Spectral refinement (fast, sets good initial conditions)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Spectral (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 10:
        a_out, _ = spectral_refine(norm_sum1(best_seq), t1, verbose=True, seed=777)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec1", t_global)

    # Phase 2: Nesterov with aggressive parameters (primary optimizer)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P2: Nesterov (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.30, 400.0)
    if t2 > 40:
        a_out, _ = nesterov_optimize(norm_sum1(best_seq), t2,
                                      beta_start=12.0, beta_end=1000.0,
                                      beta_growth=1.004, eta_init=0.6,
                                      top_k_ratio=0.02, restart_every=1000,
                                      verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 3: Peak coordinate descent (novel method)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P3: PeakCoord (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        a_out, _ = peak_coordinate_descent(norm_sum1(best_seq), t3, verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord1", t_global)

    # Phase 4: Nesterov round 2 with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P4: Nesterov r2 (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 30:
        a_out, _ = nesterov_optimize(norm_sum1(best_seq), t4,
                                      beta_start=30.0, beta_end=2000.0,
                                      beta_growth=1.006, eta_init=0.4,
                                      top_k_ratio=0.015, restart_every=800,
                                      verbose=True, seed=999)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)

    # Phase 5: LP-Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P5: LP-FW (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20 and n_best <= 10000:
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t5, solver,
                                   initial_K=250, max_K=1500,
                                   add_per_iter=25, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P6: Multi-length (base n=" + str(n_best) + ") ===")

    # Try a few nearby lengths
    candidate_lengths = []
    for delta in [-500, -200, 200, 500, 1000]:
        cl = n_best + delta
        if 3000 <= cl <= 12000:
            candidate_lengths.append(cl)
    candidate_lengths = sorted(set(candidate_lengths))

    for new_n in candidate_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        print("  -> n=" + str(new_n) + " (remaining=" + str(round(remaining, 1)) + "s)")

        a_long = interpolate_sequence(best_seq, new_n)

        # Quick Nesterov on new length
        t6a = min(remaining * 0.35, 120.0)
        if t6a > 15:
            a_out, _ = nesterov_optimize(a_long, t6a,
                                          beta_start=20.0, beta_end=800.0,
                                          beta_growth=1.005, eta_init=0.5,
                                          top_k_ratio=0.02, restart_every=800,
                                          verbose=False, seed=new_n % 7919)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(new_n), t_global)

        # Quick coordinate descent
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.25, 60.0)
            if t6b > 10:
                a_out, _ = peak_coordinate_descent(norm_sum1(best_seq), t6b,
                                                   verbose=False, seed=new_n % 3163)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n" + str(new_n), t_global)

        # Quick spectral
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6c = min(remaining * 0.20, 40.0)
            if t6c > 8:
                a_out, _ = spectral_refine(norm_sum1(best_seq), t6c,
                                           verbose=False, seed=new_n % 5243)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "spec_n" + str(new_n), t_global)

    # Phase 7: Final polish - Nesterov with very high beta
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P7: Final polish (n=" + str(n_best) + ") ===")
    if remaining > 30:
        t7a = min(remaining * 0.40, 150.0)
        if t7a > 15:
            a_out, _ = nesterov_optimize(norm_sum1(best_seq), t7a,
                                          beta_start=100.0, beta_end=3000.0,
                                          beta_growth=1.01, eta_init=0.3,
                                          top_k_ratio=0.008, restart_every=600,
                                          verbose=True, seed=555)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

        # Final coordinate descent
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.35, 80.0)
            if t7b > 10:
                a_out, _ = peak_coordinate_descent(norm_sum1(best_seq), t7b,
                                                   verbose=True, seed=654)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

        # Final LP if time and size allow
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20 and n_best <= 10000:
            t7c = min(remaining * 0.30, 80.0)
            if t7c > 10:
                a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t7c, solver,
                                           initial_K=300, max_K=1200,
                                           add_per_iter=20, verbose=True)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: J=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
