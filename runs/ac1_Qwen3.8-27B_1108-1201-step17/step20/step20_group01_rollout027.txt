def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def obj_fast(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:

        except Exception:

# --- LP Frank-Wolfe ---

def build_lp_matrix(y, clist, n):
    K = len(clist)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj in range(K):
        ki = int(clist[jj])
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[ki - idx]
    return coeffs

def lp_fw(a0, time_limit, solver, init_K=200, max_K=3000, verbose=True):
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    ac = fast_autocorr(y)
    M_curr = ac.max()
    active_set = set(int(k) for k in np.argsort(ac)[::-1][:init_K])
    no_improve = 0
    stall_limit = 3000
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-13:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Refresh active set
        if t % 20 == 0 or t == 1:

                active_set.add(int(k))

                margins = np.array([M_curr - ac[k] for k in clist_sorted])
                keep_idx = np.argsort(margins)[:max_K]

        active_set.add(int(np.argmax(ac)))
        clist = sorted(active_set)
        K = len(clist)
        if K == 0:
            break

            coeffs = build_lp_matrix(y, clist, n)
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()

            prob = cp.Problem(cp.Minimize(M_var), constraints)
            prob.solve(solver=solver, verbose=False)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                raise Exception("LP status: " + str(prob.status))
        except Exception:
            if len(active_set) > init_K:
                clist_sorted = sorted(active_set)
                margins = np.array([M_curr - ac[k] for k in clist_sorted])

        g_sum = g_val.sum()

            continue

        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:20]]
            for k in top_v:
                if len(active_set) < max_K:
                    active_set.add(int(k))
        
        # Line search
        best_alpha = 2.0 / (t + 2.0)
        trial = np.maximum((1 - best_alpha) * y + best_alpha * g_dir, 0.0)
        ts = trial.sum()
        best_trial_M = fast_autocorr(trial / ts).max() if ts > 1e-15 else 1e10
        
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            Mt = fast_autocorr(trial / ts).max()
            if Mt < best_trial_M - 1e-14:
                best_trial_M = Mt
                best_alpha = alpha
        
        # Bisection
        lo = max(0.0, best_alpha - 0.1)
        hi = min(1.0, best_alpha + 0.1)
        for _ in range(5):

            mid2 = hi - (hi - lo) / 3.0
            t1 = np.maximum((1 - mid1) * y + mid1 * g_dir, 0.0)

            else:
                lo = mid1
        best_alpha = (lo + hi) / 2.0
        
        if t <= 1:
            y = g_dir.copy()
        else:
            y = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)

# --- Smooth max gradient ---

def smooth_max_grad(a, beta):
    """Gradient of (1/beta)*log(sum_k exp(beta*(a*a)_k)) w.r.t. a."""
    n = len(a)
    ac = fast_autocorr(a)
    size = 2 * n - 1
    
    beta_c = beta * (ac - ac.max())
    p = np.exp(beta_c)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros(size)
        p[int(np.argmax(ac))] = 1.0
    else:
        p /= ps
    
    # grad[j] = 2 * sum_{k=j}^{min(j+n-1, 2n-2)} p[k] * a[k-j]
    # = 2 * (a * p_rev)[2n-2-j] where p_rev[k] = p[2n-2-k]
    p_rev = p[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    A = np.fft.fft(a, fft_size)
    PR = np.fft.fft(p_rev, fft_size)
    conv = np.fft.ifft(A * PR).real
    
    # cc[j] = conv[2n-2-j]
    grad = 2.0 * conv[2*n-2 : n-2 : -1]
    # Wait, let me index this correctly.
    # We need conv[2n-2-0], conv[2n-2-1], ..., conv[2n-2-(n-1)]
    # = conv[2n-2], conv[2n-3], ..., conv[n-1]
    # That's conv[n-1 : 2n-1] reversed
    grad = 2.0 * conv[n-1 : 2*n-1][::-1]
    return grad

def smooth_max_val(a, beta):
    ac = fast_autocorr(a)
    shifted = beta * (ac - ac.max())
    log_sum = ac.max() + np.log(np.exp(shifted).sum()) / beta
    return log_sum  # This is the smooth max value

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_grad_opt(a0, time_limit, beta_start=100, beta_end=2000, verbose=True):
    """Optimize using smooth max approximation with projected gradient + momentum."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_fast(x)
    t0 = time.monotonic()
    
    # FISTA parameters
    t_k = 1.0
    x_prev = x.copy()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta
        progress = min(1.0, t / 5000.0)
        beta = beta_start + (beta_end - beta_start) * progress
        
        y = x + ((t_k - 1) / (t_k + 2)) * (x - x_prev)
        y = np.maximum(y, 0)
        s = y.sum()
        if s < 1e-15:
            y = x
        else:
            y = y / s
        
        # Gradient
        g = smooth_max_grad(y, beta)
        
        # Backtracking line search for step size
        lr = 1.0 / (2.0 * n * beta)  # Rough initial guess
        best_step = lr
        best_x_new = None
        best_f = obj_fast(x) + 1e10
        
        for attempt in range(10):
            x_new = proj_simplex(y - lr * g)
            f_new = obj_fast(x_new)
            if f_new < best_f:
                best_f = f_new
                best_x_new = x_new
                best_step = lr
            lr *= 0.5

            x_prev = x
            x = best_x_new
        else:
            x_prev = x
            x = proj_simplex(y - best_step * g)

def coord_peel(a0, time_limit, n_targets=15, seed=42, verbose=True):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 3000
    fracs = [0.001, 0.002, 0.005, 0.01, 0.02, 0.03, 0.05]
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(n_targets, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            ws = max(0, ki - n + 1)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-25:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-15:
                continue
            
            # Find quiet positions
            pr = np.zeros(n)
            for kk in top_k[:6]:
                iki = int(kk)
                iws = max(0, iki - n + 1)
                iwe = min(n - 1, iki)
                if iws > iwe:
                    continue
                ii = np.arange(iws, iwe + 1)
                pr[ii] += a[ii] * a[iki - ii]
            pr[src] = 1e10
            quiet = np.argsort(pr)[:40]
            
            for dst in quiet:
                dst = int(dst)
                if dst == src:
                    continue
                found = False
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-18:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-15:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True

                    break

                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()

                            best_a = a.copy()
                            stall = 0
        
        if verbose and stall % 500 == 0 and stall > 0:
