def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]

    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_max_grad(c, beta):
    """Compute softmax weights for smoothed max and the gradient w.r.t. a."""
    weights = np.exp(beta * (c - c.max()))
    weights /= weights.sum()
    return weights

def compute_grad_a(a, weights):
    """
    Compute gradient of sum_k w_k * c[k] w.r.t. a.
    c[k] = sum_i a[i]*a[k-i]
    d(c[k])/d(a[j]) = 2*a[k-j] if 0 <= k-j < n
    grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    """
    n = len(a)
    # grad[j] = 2 * sum_{k: 0 <= k-j < n} w[k] * a[k-j]
    # = 2 * sum_{m: 0 <= m < n} w[m+j] * a[m]
    # This is the cross-correlation of w and a
    w_full = np.zeros(2 * n - 1)
    w_full[:len(weights)] = weights
    # Using FFT for efficiency
    fft_size = 1 << (3 * n - 3).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_full, fft_size)
    # cross-correlation: r[j] = sum_m a[m] * w[m+j]
    # = sum_m a[m] * conj(w[m+j]) for real = same
    # Using: ifft(fft(a) * conj(fft(w))) gives circular cross-correlation
    # For linear cross-correlation:
    # r[j] = sum_{m=0}^{n-1} a[m] * w_full[m+j]
    # This is the 'valid' part of the cross-correlation
    # 
    # Alternative: r = ifft(conj(fft(a)) * fft(w)) shifted
    # Actually let's just use the correlation directly
    # r[j] = sum_m a[m] * w[m+j] for m=0..n-1
    # This is np.correlate(w_full, a, 'valid') if len(w_full) >= len(a)
    # np.correlate(a1, a2, 'valid')[k] = sum_n a1[n+k]*a2[n] for n=0..len(a2)-1
    # So np.correlate(w_full, a, 'valid')[j] = sum_n w_full[n+j]*a[n] = r[j] ✓
    r = np.correlate(w_full, a, 'valid')[:n]
    return 2.0 * r

def grad_descent_smoothed(a0, time_limit, beta_init=50, beta_max=2000, lr_init=0.02, verbose=True):
    """
    Smoothed gradient descent on the simplex using log-sum-exp approximation of max.
    """
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    max_stall = 5000
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break
        
        c = fast_conv(x)
        cur_val = 2.0 * n * c.max()
        
        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Increase beta over time (sharpening the max approximation)
        beta = min(beta_max, beta_init * (1 + t / 500.0))
        
        # Compute smoothed gradient
        weights = smooth_max_grad(c, beta)
        grad = compute_grad_a(x, weights)
        
        # Gradient descent step with projection onto simplex
        lr = lr_init / (1.0 + 0.001 * t)
        x_new = project_simplex(x - lr * grad)
        x = x_new
        
        # Occasionally add noise to escape local minima
        if stall > 2000 and stall % 500 == 0:
            noise_scale = 0.01 * (1.0 + 0.1 * (stall // 500))
            noise = rng.standard_normal(n) * noise_scale
            x_pert = x + noise
            x = project_simplex(x_pert)
            stall = 0
        
        if verbose and t % 200 == 0:
            print("    GradDescent t={} ({:.1f}s) J={:.7f} beta={:.0f} stall={}".format(
                t, time.monotonic() - t0, best_val, beta, stall))
    
    return best_a, best_val

def lp_fw_improved(a0, time_limit, solver, init_K=200, max_K=5000, verbose=True):
    """LP-based Frank-Wolfe with margin-based active set."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_conv(y).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 3000
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break
        
        c = fast_conv(y)
        M_curr = c.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Active set: lags where c[k] is close to M_curr
        margin = 0.05 * M_curr + 1e-10
        active_lags = np.where(c >= M_curr - margin)[0]
        
        # Also include top K lags
        K = min(max_K, max(init_K, len(active_lags)))
        top_lags = np.argsort(c)[::-1][:K]
        active_lags = np.unique(np.concatenate([active_lags, top_lags]))
        active_lags = active_lags[:max_K]
        
        # Solve LP: min M s.t. a >= 0, sum(a) = 1, (y*a)[k] <= M for k in active
        clist = sorted(int(k) for k in active_lags)
        K_act = len(clist)
        
        # Build constraint matrix
        cols = []
        rows = []
        for jj, ki in enumerate(clist):
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            for i in range(i_start, i_end):
                rows.append(jj)
                cols.append(i)
        
        if len(rows) == 0:
            break
        
        A_mat = np.zeros((K_act, n), dtype=np.float64)
        for jj, ki in enumerate(clist):
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            A_mat[jj, idx] = y[ki - idx]
        
        try:
            a_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [cp.sum(a_var) == 1, A_mat @ a_var <= M_var]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            prob.solve(solver=solver, verbose=False)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                raise Exception("LP failed")
        except Exception:
            # Reduce active set and retry
            clist = sorted(int(k) for k in top_lags[:init_K])
            K_act = len(clist)
            A_mat = np.zeros((K_act, n), dtype=np.float64)
            for jj, ki in enumerate(clist):
                i_start = max(0, ki - n + 1)
                i_end = min(n, ki + 1)
                idx = np.arange(i_start, i_end)
                A_mat[jj, idx] = y[ki - idx]
            try:
                a_var = cp.Variable(n, nonneg=True)
                M_var = cp.Variable()
                constraints = [cp.sum(a_var) == 1, A_mat @ a_var <= M_var]
                prob = cp.Problem(cp.Minimize(M_var), constraints)
                prob.solve(solver=solver, verbose=False)
                if prob.status not in ('optimal', 'optimal_inaccurate'):
                    continue
            except Exception:
                continue
        
        g_val = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Line search: try several step sizes
        best_alpha = 2.0 / (t + 2.0)
        best_trial_m = 1e10
        for alpha in [best_alpha, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            trial = np.maximum((1 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            Mt = fast_conv(trial / ts).max()
            if Mt < best_trial_m:
                best_trial_m = Mt
                best_alpha = alpha
        
        # Bisection refinement
        lo = max(0.0, best_alpha - 0.2)
        hi = min(1.0, best_alpha + 0.2)
        for _ in range(8):
            mid = (lo + hi) / 2.0
            trial = np.maximum((1 - mid) * y + mid * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                hi = mid
                continue
            Mt = fast_conv(trial / ts).max()
            if Mt < best_trial_m:
                best_trial_m = Mt
                best_alpha = mid
                lo = mid
            else:
                hi = mid
        
        y = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        if verbose and t % 100 == 0:
            print("    LP-FW t={} ({:.1f}s) J={:.7f} |S|={}".format(
                t, time.monotonic() - t0, 2 * n * best_m, len(clist)))
    
    return best_a, 2.0 * n * best_m
