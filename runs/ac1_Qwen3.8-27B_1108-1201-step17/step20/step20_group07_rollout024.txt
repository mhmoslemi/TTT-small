    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def evaluate_safe(a):
    try:
        return float(evaluate_sequence([float(x) for x in a]))
    except Exception:
        return float('inf')

def compute_grad(a, beta, top_k, fft_3n):
    """Compute gradient of log-sum-exp smoothed max of autocorrelation."""
    n = len(a)
    size = 2 * n - 1
    b = fast_autocorr(a)
    if top_k < size:
        idx = np.argpartition(b, -top_k)[-top_k:]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
    else:
        b_use = b
    b_max = b_use.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_v = np.exp(shifted)
    p = exp_v / exp_v.sum()
    # grad[j] = (2/beta) * sum_k p[k] * a[k-j] (when valid)
    # = (2/beta) * (p * a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_p = np.fft.irfft(np.fft.rfft(p, fft_3n) * np.fft.rfft(a_rev, fft_3n), fft_3n)
    grad = 2.0 * conv_p[n-1:2*n-1] / beta
    return grad, b, b_max

def nesterov_prox_grad(a0, time_limit, beta_start=20.0, beta_end=5000.0,
                        beta_growth=1.003, top_k_ratio=0.04, verbose=True):
    """Nesterov-accelerated proximal gradient method."""
    n = len(a0)
    size = 2 * n - 1
    fft_2n = 1 << (size - 1).bit_length()
    fft_3n = 1 << (3 * n - 2 - 1).bit_length()
    top_k = max(30, int(n * top_k_ratio))
    
    x = norm_sum1(a0)
    y = x.copy()
    best_a = x.copy()
    b = fast_autocorr(x)
    best_m = b.max()
    t0 = time.monotonic()
    beta = beta_start
    t = 0
    rng = np.random.default_rng(7)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        
        # Nesterov momentum: y_t = x_t + (t-1)/(t+2) * (x_t - x_{t-1})
        # For t=1: y = x
        # For t>1: y = x + momentum * (x - x_old)
        # We track x_old implicitly
        
        # Compute gradient at y
        grad, b_y, m_y = compute_grad(y, beta, top_k, fft_3n)
        
        # Update best
        m_y_actual = b_y.max()
        if m_y_actual < best_m - 1e-15:
            best_m = m_y_actual
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Step size: use 1/L where L is Lipschitz constant estimate
        # L ≈ (2/beta) * max(p) * 2 * max(a) * something
        # Heuristic: eta = c / (beta * max(a))
        max_a = y.max()
        eta = 1.0 / (beta * max_a + 1e-10)
        eta = min(eta, 2.0 / (t + 2))  # Nesterov step size
        
        # Backtracking line search
        f_y = m_y + np.log(np.exp(np.clip(beta * (b_y - m_y), -300, 300)).sum()) / beta
        
        accepted = False
        for c in [1.0, 0.5, 0.25, 0.1, 0.05]:
            trial_eta = eta * c
            x_new = simplex_project(y - trial_eta * grad)
            b_new = fast_autocorr(x_new)
            f_new = b_new.max()
            if f_new < f_y + 1e-10:  # Allow tiny increase
                accepted = True
                break
        
        if not accepted:
            x_new = simplex_project(y - eta * 0.01 * grad)
        
        # Nesterov update
        if t == 1:
            y = x_new
            x = x_new
        else:
            # x_{t+1} = x_new (the prox step result)
            # y_{t+1} = x_{t+1} + (t)/(t+2) * (x_{t+1} - x_t)
            x_old = x
            x = x_new
            momentum = t / (t + 2.0)
            y = x + momentum * (x - x_old)
        
        # Adaptive beta
        if t % 200 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        # Restart
        if stall > 500:
            stall = 0
            scale = 0.02 / (1.0 + 0.1 * (t // 500))
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-15)
            x_r = x_r / x_r.sum()
            y = x_r.copy()
            x = x_r.copy()
            beta = max(beta * 0.7, beta_start)
        
        if verbose and t % 2000 == 0:
            J = 2.0 * n * best_m
            print(f"    NG t={t} ({elapsed:.1f}s): J={J:.6f} m={best_m:.7e} beta={beta:.0f} stall={stall}")
    
    return best_a, best_m

def peak_deflation(a0, time_limit, verbose=True):
    """Directly reduce the top autocorrelation peaks by moving mass."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        m = b.max()
        if m < best_m - 1e-15:
            best_m = m
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find top peaks
        K = min(15, 2 * n - 1)
        top_idx = np.argsort(b)[::-1][:K]
        
        # Compute contribution matrix for top peaks
        # For each peak k, find the dominant contributing pair
        improved = False
        
        for k_int in top_idx[:8]:
            k = int(k_int)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k - idx]
            max_c = contrib.max()
            if max_c < 1e-20:
                continue
            
            src = int(idx[np.argmax(contrib)])
            partner = k - src
            if partner < 0 or partner >= n:
                continue
            
            # Determine which to reduce
            if a[src] >= a[partner]:
                reduce_at = src
            else:
                reduce_at = partner
            
            if a[reduce_at] < 1e-15:
                continue
            
            # Find safe destination: minimize increase in top peaks
            # For candidate j, the increase in peak k is approximately a[reduce_at Fraction] * a[k-j] * 2
            # We want to minimize sum over top peaks of a[k_j - j] (where valid)
            
            # Sample a subset of candidates for efficiency
            n_samples = min(200, n // 10)
            candidates = rng.choice(n, size=n_samples, replace=False)
            
            # Exclude positions near the top peaks' contributing ranges
            scores = np.zeros(n_samples)
            for c_idx, j in enumerate(candidates):
                j = int(j)
                if j == reduce_at:
                    scores[c_idx] = np.inf
                    continue
                score = 0.0
                for k2 in top_idx[:5]:
                    k2i = int(k2)
                    pj = k2i - j
                    if 0 <= pj < n:
                        score += a[pj]
                scores[c_idx] = score
            
            best_cand = int(candidates[np.argmin(scores)])
            
            # Try to move mass
            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[reduce_at] * frac
                if amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[reduce_at] -= amt
                a_trial[best_cand] += amt
                a
