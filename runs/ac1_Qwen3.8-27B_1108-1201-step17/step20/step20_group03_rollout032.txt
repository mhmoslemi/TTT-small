def fast_autocorr(a):
    """Compute linear autocorrelation (a*a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project z onto the unit simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_lse_gradient(a, beta):
    """Compute gradient of (2n/beta)*log(sum(exp(beta*(a*a))) ) via FFT."""
    n = len(a)
    b = fast_autocorr(a)
    # Compute probabilities p_k = exp(beta*b_k) / sum(exp(beta*b_k))
    b_shifted = beta * (b - b.max())
    exp_vals = np.exp(b_shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b)
        idx_max = np.argwhere(b == b.max()).flatten()
        p[idx_max] = 1.0 / len(idx_max)
    else:
        p = exp_vals / p_total
    # Gradient: g_i = (4n/beta) * sum_k p_k * a_{k-i}
    # = (4n/beta) * (p * a_rev)_{n-1+i}
    a_rev = a[::-1]
    conv_len = len(p) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Pf = np.fft.rfft(p, fft_size)
    Arf = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(Pf * Arf, fft_size)[:len(p) + n - 1]
    grad = (4.0 * n / beta) * conv[n - 1:2 * n - 1]
    return grad, b, p

    lp_fail_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break
        if lp_fail_count > 20:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Active set management
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        # Also add some "spread" constraints (not just peaks)
        if len(constraint_list) < initial_K:
            all_idx = np.arange(2 * n - 1)
            not_in_active = np.array([k for k in all_idx if k not in active_set])
            if len(not_in_active) > 0:
                # Add some from non-peak region
                n_add = min(50, len(not_in_active), initial_K - len(constraint_list))
                add_idx = rng_choice(not_in_active, n_add)
                for k in add_idx:
                    active_set.add(int(k))
                constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build LP
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail_count += 1
            if len(active_set) > initial_K // 2:
                ac_now = fast_autocorr(y)
                active_set = set(int(k) for k in np.argsort(ac_now)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail_count += 1
            continue
        lp_fail_count = 0

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations and add to active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-6)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                active_set.add(int(k))

        # Backtracking line search
        default_alpha = 2.0 / (t + 2.0)
        best_alpha = default_alpha
        best_trial_M = M_curr
        for alpha in [default_alpha, max(0.02, default_alpha * 0.3),
                      max(0.1, default_alpha * 2.0), 1.0 / (t + 1), 0.05, 0.1]:
            alpha = max(0.01, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def rng_choice(arr, k):
    """Randomly choose k elements from arr."""
    rng = np.random.default_rng(12345)
    if k >= len(arr):
        return arr
    idx = rng.choice(len(arr), size=k, replace=False)
    return arr[idx]


def peak_deflation(a0, time_limit, n_peaks=5, verbose=True):
    """Targeted deflation of top peaks in the convolution."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:n_peaks]
        improved = False

        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(top_peaks[rng.integers(len(top_peaks))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])

            # Find quiet positions
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src:
                continue
            if a[src] < 1e-12:
                continue

            frac = float(rng.choice([0.002, 0.005, 0.01, 0.02, 0.05]))
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-14:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                ac = fast_autocorr(a)
                top_peaks = np.argsort(ac)[::-1][:n_peaks]

        if not improved:
            stall += 1
            # Random kicks to escape local minima
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.001, 0.02))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Deflation (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def cyclic_shift_best(a0):
    """Find the best cyclic shift."""
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Try to use parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: FISTA (main optimizer, 300s)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: FISTA (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.20, 350.0)
    if t1 > 60:
        a_out, _ = fista_optimize(norm_sum1(best_seq), t1,
                                  beta_start=15.0, beta_end=600.0,
                                  beta_growth=1.004, verbose=True, top_k=20)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista1", t_global)

    # Phase 2: Cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: Cyclic shift ===")
    if remaining > 60:
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic", t_global)

    # Phase 3: LP-FW (main LP optimizer, 400s)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP-FW (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.25, 450.0)
    if t3 > 60 and n_best <= 10000:
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t3, solver,
                                  initial_K=500, max_K=3000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, "lp1", t_global) if False else try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 4: Peak deflation
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Peak deflation (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 30:
        a_out, _ = peak_deflation(norm_sum1(best_seq), t4, n_peaks=8, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "deflate1", t_global)

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-length ===")
    longer_lengths = [n_best + 1000, n_best + 2000]
    longer_lengths = [l for l in longer_lengths if l <= 12000]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("  Trying n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.25, 180.0)
        if t5a > 40:
            a_out, _ = fista_optimize(a_long, t5a,
                                      beta_start=20.0, beta_end=500.0,
                                      beta_growth=1.004, top_k=15, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and len(best_seq) <= 10000:
            t5b = min(remaining * 0.20, 120.0)
            if t5b > 30:
                a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t5b, solver,
                                          initial_K=400, max_K=2000, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(new_n), t_global)

    # Phase 6: Second FISTA pass with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: FISTA v2 (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 40:
        a_out, _ = fista_optimize(norm_sum1(best_seq), t6,
                                  beta_start=80.0, beta_end=1500.0,
                                  beta_growth=1.006, top_k=10, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista2", t_global)

    # Phase 7: Second LP burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: LP v2 (n=" + str(n_best) + ") ===")
    if remaining > 60 and n_best <= 10000:
        t7 = min(remaining * 0.25, 250.0)
        if t7 > 40:
            a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t7, solver,
                                      initial_K=600, max_K=3500, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp2", t_global)

    # Phase 8: Final deflation
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 8: Final deflation ===")
    if remaining > 40:
        t8 = min(remaining * 0.20, 120.0)
        if t8 > 20:
            a_out, _ = peak_deflation(norm_sum1(best_seq), t8, n_peaks=5, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "deflate_final", t_global)

    # Final cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print("\n=== Final: Cyclic shift ===")
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_final", t_global)

    # Final FISTA micro-polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== Final: FISTA polish ===")
        t_f = min(remaining * 0.40, 100.0)
        if t_f > 20:
            a_out, _ = fista_optimize(norm_sum1(best_seq), t_f,
                                      beta_start=150.0, beta_end=2000.0,
                                      beta_growth=1.008, top_k=8, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_final", t_global)

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
