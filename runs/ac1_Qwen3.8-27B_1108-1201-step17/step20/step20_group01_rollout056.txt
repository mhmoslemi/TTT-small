def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def subgradient_at_lag(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a"""
    n = len(a)
    g = np.zeros(n, dtype=np.float64)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    for i in range(i_start, i_end):
        j = k - i
        g[i] += a[j]
    return g

def subgradient_fft(a, k):
    """FFT-based computation of subgradient at lag k."""
    n = len(a)
    # g[i] = a[k-i] for valid i
    # This is a "reversed and shifted" version of a
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    g = np.zeros(n, dtype=np.float64)
    if i_start < i_end:
        idx = np.arange(i_start, i_end)
        g[idx] = a[k - idx]
    return g

def multi_peak_gradient(a, top_lags, weights):
    """Compute weighted sum of subgradients at multiple lags."""
    n = len(a)
    g = np.zeros(n, dtype=np.float64)
    for w, k in zip(weights, top_lags):
        g += w * subgradient_fft(a, k)
    return g

def spectral_smooth(a, sigma):
    """Apply Gaussian low-pass filter in frequency domain."""
    n = len(a)
    f = np.fft.rfft(a)
    freqs = np.fft.rfftfreq(n)
    # Gaussian window in frequency domain
    # sigma in cycles: larger sigma = more smoothing
    mask = np.exp(-2.0 * np.pi**2 * sigma**2 * freqs**2)
    a_smooth = np.fft.irfft(f * mask, n)
    return np.maximum(a_smooth, 0.0)

def mirror_step(a, grad, lr):
    """One mirror descent step in log-space."""
    log_a = np.log(a + 1e-30)
    log_a -= lr * grad
    log_a -= log_a.max()  # numerical stability
    a_new = np.exp(log_a)
    a_new = np.maximum(a_new, 1e-25)
    return a_new / a_new.sum()

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        # Build the linear coefficient vector
        coeff = np.zeros(m)

            if 0 <= j2 < n:
                coeff[j] = a[j2]
        constraints.append(coeff @ g_var <= M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver='ECOS', verbose=False)

        return None

    g_vals = np.maximum(np.asarray(g_var.value), 0)

        g_full[ci] = g_vals[j]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Hybrid Small-LP + Gradient Frank-Wolfe with adaptive line search
2. Multi-peak subgradient with temperature annealing
3. Spectral domain smoothing for peak control
4. Coordinate-level final polish
5. Multi-start with diverse asymmetric initializations

import numpy as np
import time
import math

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def subgradient_at_lag(a, k):
    n = len(a)
    g = np.zeros(n, dtype=np.float64)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    if i_start < i_end:
        idx = np.arange(i_start, i_end)
        g[idx] = a[k - idx]
    return g

def multi_peak_gradient(a, top_lags, weights):
    n = len(a)
    g = np.zeros(n, dtype=np.float64)
    for i in range(len(top_lags)):
        g += weights[i] * subgradient_at_lag(a, int(top_lags[i]))
    return g

def spectral_smooth(a, sigma_frac):
    n = len(a)
    if n < 4:
        return a.copy()
    f = np.fft.rfft(a)
    freqs = np.fft.rfftfreq(n)
    sigma = sigma_frac * n
    mask = np.exp(-0.5 * (freqs / sigma) ** 2)
    a_smooth = np.fft.irfft(f * mask, n)
    return np.maximum(a_smooth, 0.0)

def small_lp_direction(a, active_lags, n_candidates=80):
    """Solve a small LP to find a good direction supported on few coordinates."""
    try:
        import cvxpy as cp
    except ImportError:
        return None

    n = len(a)
    ac = fast_autocorr(a)
    M_curr = ac.max()

    if M_curr < 1e-15:
        return None

    candidate_set = set()
    for k in active_lags:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        top_in_range = idx[np.argsort(a[idx])[::-1][:8]]
        for i in top_in_range:
            candidate_set.add(int(i))
        bottom_in_range = idx[np.argsort(a[idx])[:5]]
        for i in bottom_in_range:
            candidate_set.add(int(i))

    candidates = list(candidate_set)[:n_candidates]
    if len(candidates) < 5:
        return None

    m = len(candidates)
    K = len(active_lags)

    g_var = cp.Variable(m, nonneg=True)
    M_var = cp.Variable()

    constraints = [cp.sum(g_var) == 1]

    for k in active_lags:
        k = int(k)
        coeff = np.zeros(m, dtype=np.float64)
        for j, ci in enumerate(candidates):
            j2 = k - ci
            if 0 <= j2 < n:
                coeff[j] = a[j2]
        constraints.append(coeff @ g_var <= M_var)

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    g_vals = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
    g_sum = g_vals.sum()
    if g_sum < 1e-15:
        return None
    g_full = np.zeros(n, dtype=np.float64)
    for j, ci in enumerate(candidates):
        g_full[ci] = g_vals[j] / g_sum
    return g_full

def line_search_alpha(a, direction, base_alpha):
    """Find best step size alpha by trying several values."""
    best_alpha = base_alpha
    best_val = float('inf')
    alphas = [base_alpha * 0.25, base_alpha * 0.5, base_alpha, base_alpha * 1.5, base_alpha * 2.0]
    for alpha in alphas:
        alpha = min(max(alpha, 0.01), 0.95)
        trial = (1.0 - alpha) * a + alpha * direction
        trial = np.maximum(trial, 0.0)
        ts = trial.sum()
        if ts < 1e-15:
            continue
        trial = trial / ts
        val = fast_autocorr(trial).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    return best_alpha

def main_optimize(a0, time_limit, n_top=10, lp_every=8, lr0=0.05,
                  momentum=0.8, smooth_every=300, smooth_sigma=0.01,
                  perturb_every=600, verbose=True):
    """Main optimization: hybrid LP + gradient with momentum and line search."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    v = np.zeros(n, dtype=np.float64)
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if stall > 20000:
            pert = 0.15 * rng.standard_normal(n)
            a = normalize(best_a * (1.0 + pert))
            v = np.zeros(n, dtype=np.float64)
            stall = 0
            it = 0
            continue

        ac = fast_autocorr(a)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr

        if cur_val < best_val - 1e-13:
            best_val = cur_val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        n_top_eff = min(n_top, len(ac))
        top_idx = np.argsort(ac)[::-1][:n_top_eff]
        top_vals = ac[top_idx]

        temp = 5.0 / (1.0 + 0.0005 * it)
        w = np.exp((top_vals - top_vals[0]) / temp)
        w = w / w.sum()

        # Determine direction
        use_lp = (it % lp_every == 0) and (n <= 15000) and (time.monotonic() - t0 < time_limit * 0.8)
        if use_lp:
            direction = small_lp_direction(a, top_idx.tolist(), n_candidates=60)
            if direction is None:
                use_lp = False

        if not use_lp:
            grad = multi_peak_gradient(a, top_idx, w)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                grad = np.ones(n) / n
            else:
                grad = grad / grad_norm
            v = momentum * v + grad
            v_norm = np.linalg.norm(v)
            if v_norm > 1e-10:
                direction = v / v_norm
                direction = np.maximum(direction, 0.0)
                ds = direction.sum()
                if ds > 1e-15:
                    direction = direction / ds
                else:
                    direction = np.ones(n) / n

        # Adaptive step size
        base_alpha = lr0 / (1.0 + 0.001 * it)
        if stall > 500:
            base_alpha *= 2.0
        if stall > 2000:
            base_alpha *= 4.0
        base_alpha = min(base_alpha, 0.5)

        # Line search
        best_alpha = line_search_alpha(a, direction, base_alpha)

        a = (1.0 - best_alpha) * a + best_alpha * direction
        a = np.maximum(a, 0.0)
        sa = a.sum()
        if sa < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / sa

        # Periodic spectral smoothing
        if it % smooth_every == 0:
            a_smooth = spectral_smooth(a, smooth_sigma)
            ss = a_smooth.sum()
            if ss > 1e-15:
                a = 0.4 * a + 0.6 * (a_smooth / ss)
                a = normalize(a)

        # Periodic perturbation
        if it % perturb_every == 0 and it > 0:
            freq = rng.uniform(1.0, 6.0)
            phase = rng.uniform(0.0, 2.0 * np.pi)
            pert = 0.03 * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
            a_pert = a * (1.0 + pert)
            a = normalize(a_pert)
            v *= 0.5

        if verbose and it % 500 == 0:
            print("  MainOpt it={:5d} ({:.1f}s) val={:.7f} stall={:d}".format(
                it, time.monotonic() - t0, best_val, stall))

    return best_a, best_val

def coord_polish(a0, time_limit, n_targets=15, verbose=True):
    """Coordinate-level polish: targeted mass transfer from peak contributors."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stagnation = 0

    while time.monotonic() - t0 < time_limit and stagnation < 300:
        ac = fast_autocorr(a)
        M = ac.max()
        n_top = min(n_targets, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[ki - idx]
            top_src = idx[np.argsort(contrib)[::-1][:15]]

            for src in top_src:
                src = int(src)
                if a[src] < 1e-14:
                    continue
                for dst in np.random.default_rng(src * 7 + 3).integers(0, n, size=8):
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                        amt = a[src] * frac
                        if amt < 1e-18:
                            break
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < M - 1e-15:
                            val_t = 2.0 * n * Mt
                            if val_t < best_val:
                                best_val = val_t
                                best_a = a_t.copy()
                            a = a_t
                            M = Mt
                            improved = True
                            break

        if not improved:
            stagnation += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    Mt = fast_autocorr(a).max()
                    if 2.0 * n * Mt < best_val:
                        best_val = 2.0 * n * Mt
                        best_a = a.copy()
                        stagnation = 0
        else:
            stagnation = 0

        if verbose and stagnation % 300 == 0 and stagnation > 0:
            print("  CoordPolish ({:.1f}s) val={:.7f} stagn={}".format(
                time.monotonic() - t0, best_val, stagnation))

    return best_a, best_val

def parametric_init(n, seed=0):
    """Diverse parametric initialization with asymmetric shape."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64) / n
    shift = rng.uniform(-0.15, 0.15)
    power = rng.uniform(1.0, 2.5)
    a = np.maximum(1.0 + rng.uniform(0.3, 0.8) * np.cos(2.0 * np.pi * (x + shift)), 0.0) ** power
    tilt = rng.uniform(-0.4, 0.4)
    a *= np.exp(tilt * (x - 0.5))
    a = np.maximum(a, 1e-6)
    return normalize(a)
