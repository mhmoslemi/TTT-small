            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  Solver fallback: ECOS")
    return _MODULE_SOLVER

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def soft_grad_ac(a, beta):
    """Compute gradient of soft-max objective w.r.t. a.
    Returns (soft_obj, grad) where soft_obj = (1/beta)*log(sum(exp(beta*ac)))
    and grad = d(soft_obj)/d(a_i).
    """
    n = len(a)
    ac = fast_autocorr(a)
    # Numerically stable log-sum-exp
    ac_max = ac.max()
    shifted = beta * (ac - ac_max)
    exp_shifted = np.exp(shifted)
    log_sum = beta * ac_max + np.log(exp_shifted.sum())
    soft_obj = log_sum / beta
    w = exp_shifted / exp_shifted.sum()
    # Gradient: (2/beta) * (w cross-correlate a)
    # (a cross-correlate w)[k] = sum_j a[j] * w[j+k]
    # We need c[i] = sum_j a[j] * w[j+i] for i=0..n-1
    # via FFT: c = irfft(conj(rfft(a,N)) * rfft(w,N))
    N = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    c = np.fft.irfft(np.conj(A) * W, N)
    grad = (2.0 / beta) * c[:n]
    return soft_obj, grad

def smooth_grad_multi(a0, budget_s, beta_start=5.0, beta_end=500.0,
                      base_lr=1.0, momentum=0.9, verbose=True):
    """Multiplicative weight update with log-sum-exp smoothed gradient and Nesterov momentum."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    v = np.zeros(n, dtype=np.float64)  # velocity for Nesterov
    iter_count = 0
    stall = 0
    max_stall = 3000

    total_iters_est = 50000
    for t in range(1, 10000000):
        if time.monotonic() - t0 > budget_s:
            break
        if stall > max_stall:
            break

        # Adaptive beta: ramp up over the course of optimization
        progress = min(t / float(total_iters_est), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        # Nesterov lookahead
        x_look = x + momentum * v
        x_look = np.maximum(x_look, 1e-15)
        x_look = x_look / x_look.sum()

        # Compute gradient at lookahead point
        _, grad = soft_grad_ac(x_look, beta)

        # Scale gradient to be comparable across betas
        # Use a normalized step: the gradient of the actual objective is 2n * grad
        # We want the step to be reasonable relative to the current values
        g_norm = np.sqrt(np.sum(grad ** 2))
        if g_norm < 1e-15:
            break

        # Adaptive learning rate
        x_mean = x.mean()
        lr = base_lr * x_mean / (g_norm + 1e-10)
        lr = min(lr, 0.5)  # cap step size

        # Multiplicative update
        v = momentum * v + (1.0 - momentum) * (-grad / g_norm)
        log_update = np.exp(-lr * v * g_norm)
        x_new = x * log_update
        x_new = np.maximum(x_new, 1e-30)
        x_new = x_new / x_new.sum()

        x = x_new
        iter_count += 1

        # Evaluate actual objective periodically
        if iter_count % 50 == 0:
            val = obj_val(x)
            if val < best_val - 1e-10:
                best_val = val
                best_a = x.copy()
                stall = 0
            else:
                stall += 1

        if verbose and iter_count % 2000 == 0:
            elapsed = time.monotonic() - t0
            val_now = obj_val(x)
            print("    SmoothGrad t=" + str(iter_count) + " (" + str(round(elapsed, 1)) +
                  "s) best=" + repr(best_val) + " cur=" + repr(val_now) +
                  " beta=" + str(round(beta, 1)))

    return best_a, best_val

def smooth_grad_multi_v2(a0, budget_s, beta_schedule=None, verbose=True):
    """Improved multiplicative gradient with two-phase beta schedule and backtracking."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    stall = 0
    max_stall = 2000
    iter_count = 0

    # Two-phase: phase 1 uses small beta (smooth), phase 2 ramps up
    phase1_frac = 0.4
    beta_p1 = 3.0
    beta_p2_start = 20.0
    beta_p2_end = 800.0
    est_iters = 40000

    for t in range(1, 10000000):
        if time.monotonic() - t0 > budget_s:
            break
        if stall > max_stall:
            # Reset with slight perturbation
            rng = np.random.default_rng(t)
            x = best_a * (1.0 + 0.01 * rng.standard_normal(n))
            x = np.maximum(x, 1e-12)
            x = x / x.sum()
            stall = 0
            continue

        progress = min(t / float(est_iters), 1.0)
        if progress < phase1_frac:
            beta = beta_p1
        else:
            p2_prog = (progress - phase1_frac) / (1.0 - phase1_frac)
            beta = beta_p2_start * (beta_p2_end / beta_p2_start) ** p2_prog

        # Compute gradient
        _, grad = soft_grad_ac(x, beta)
        g_norm = np.sqrt(np.sum(grad ** 2))
        if g_norm < 1e-15:
            break

        # Backtracking line search on multiplicative update
        # Try: x_trial = x * exp(-eta * grad / g_norm), then normalize
        # Objective: actual hard-max objective (not smooth)
        eta = 0.5
        best_alpha = 0.0
        best_trial_val = obj_val(x)
        for _ in range(8):
            log_step = -eta * grad / g_norm
            # Clip log_step to avoid overflow
            log_step = np.clip(log_step, -5.0, 5.0)
            x_trial = x * np.exp(log_step)
            x_trial = np.maximum(x_trial, 1e-30)
            x_trial = x_trial / x_trial.sum()
            trial_val = obj_val(x_trial)
            if trial_val < best_trial_val - 1e-12:
                best_trial_val = trial_val
                best_alpha = eta
                x = x_trial
                break
            eta *= 0.5

        iter_count += 1

        if iter_count % 20 == 0:
            val = obj_val(x)
            if val < best_val - 1e-10:
                best_val = val
                best_a = x.copy()
                stall = 0
            else:
                stall += 1

        if verbose and iter_count % 3000 == 0:
            elapsed = time.monotonic() - t0
            print("    SmoothGrad2 t=" + str(iter_count) + " (" + str(round(elapsed, 1)) +
                  "s) best=" + repr(best_val) + " beta=" + str(round(beta, 1)) +
                  " stall=" + str(stall))

    return best_a, best_val

def lp_fw_best(a0, budget_s, solver, init_K=400, max_K=4000, add_per=50, verbose=True):
    """LP-based Frank-Wolfe with periodic full re-scan and backtracking line search."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Initialize active set with top lags
    ac_init = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_init)[-init_K:])
    no_improve = 0
    stall_limit = 800
    full_rescan_interval = 50

    for t in range(1, 5000000):
        if time.monotonic() - t0 > budget_s:
            break
        if no_improve > stall_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic full re-scan: refresh active set with top lags
        if t % full_rescan_interval == 0:
            top_lags = np.argsort(ac)[-init_K:]
            active_set = set(int(k) for k in top_lags)

        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)

        K = len(clist)
        if K == 0:
            break

        # Build LP constraint matrix
        coeffs = np.zeros((K, n), dtype=np.float64)
        for jj, k in enumerate(clist):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP: min M' s.t. sum(g)=1, g>=0, coeffs@g <= M'
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        lp_ok = False
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                lp_ok = True
        except Exception:
            pass

        if not lp_ok:
            if no_improve > stall_limit // 2:
                break
            if len(clist) > 100:
                active_set = set(int(k) for k in clist[:200])
                continue
            break

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations and add to active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]
            for k in top_v:
                active_set.add(int(k))

        # Backtracking line search
        best_alpha = 0.0
        cur_alpha = 1.0
        obj_start = fast_autocorr(y).max()
        for _ in range(15):
            x_trial = (1.0 - cur_alpha) * y + cur_alpha * g_dir
            x_trial = np.maximum(x_trial, 0.0)
            s_trial = x_trial.sum()
            if s_trial < 1e-15:
                cur_alpha *= 0.5
                continue
            x_trial = x_trial / s_trial
            m_trial = fast_autocorr(x_trial).max()
            # Armijo condition: m_trial <= obj_start - c * alpha * (obj_start - M_lp)
            # Simplified: just require improvement
            if m_trial < obj_start - 1e-14:
                best_alpha = cur_alpha
                break
            cur_alpha *= 0.5

        if best_alpha < 1e-6:
            best_alpha = 0.01  # minimum step to ensure progress

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy
        else:
            y = g_dir.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) " +
                  "J=" + repr(2 * n * best_m) + " |S|=" + str(len(active_set)))

    if verbose:
        print("    LP-FW done (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2 * n * best_m))
    return best_a, 2.0 * n * best_m

def peak_shave(a0, budget_s, n_targets=10, verbose=True):
    """Target specific peak lags and reduce them via smart mass redistribution."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0

    while time.monotonic() - t0 < budget_s and stall < 500:
        ac = fast_autocorr(a)
        targets = np.argsort(ac)[::-1][:n_targets]
        improved = False

        for k in targets:
            if time.monotonic() - t0 > budget_s:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            # Find top contributing sources
            top_srcs = idx[np.argsort(contrib)[::-1][:5]]
            for src in top_srcs:
                src = int(src)
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Score destinations: minimize increase in all target peaks
                scores = np.zeros(n)
                for k2 in targets[:6]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                        scores[k2i - idx2] += a[idx2]
                scores[src] = np.inf

                # Try best destinations with various fractions
                candidates = np.argsort(scores)[:40]
                found = False
                for dst in candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.001, 0.003, 0.007, 0.015, 0.03]:
                        amt = src_val * frac
                        if amt < 1e-15:
                            continue
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-15:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            found = True
                            break
                    if found:
                        break

        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            if stall % 20 == 0:
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        st = a.sum()
                        if st > 1e-15:
                            a = a / st
                            Mn = fast_autocorr(a).max()
                            if Mn < best_m:
                                best_m = Mn
                                best_a = a.copy()
                                stall = 0
                                break

        if verbose and stall % 100 == 0:
            print("    PeakShave (" + str(round(time.monotonic() - t0, 1)) +
                  "s) J=" + repr(2 * n * best_m) + " stall=" + str(stall))

    return best_a, 2.0 * n * best_m

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def spectral_init(n, seed=0):
    """Generate a sequence with approximately flat spectrum (made non-negative)."""
    rng = np.random.default_rng(seed)
    # Random phases
    theta = rng.uniform(0, 2 * np.pi, n)
    # Complex sequence with unit magnitude DFT
    Z = np.exp(1j * theta)
    # Inverse DFT to get time-domain sequence
    z = np.fft.irfft(Z * np.sqrt(n), n)  # real part
    # Take absolute value and normalize
    a = np.abs(z)
    a = normalize(a)
    return a

def diverse_init(n, parent_seq=None):
    """Generate diverse initial candidates and return the best."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)

    candidates = []

    # Sine-squared (classic)
    candidates.append(normalize(np.sin(np.pi * x) ** 2))
    # Sine
    candidates.append(normalize(np.sin(np.pi * x)))
    # Triangle
    candidates.append(normalize(1.0 - np.abs(2.0 * x - 1.0)))
    # Gaussian centered
    for sg in [0.15, 0.2, 0.25, 0.3, 0.4]:
        candidates.append(normalize(np.exp(-((x - 0.5) ** 2) / (2 * sg * sg))))
    # Double bump
    for sg in [0.08, 0.12, 0.15]:
        cands = np.exp(-((x - 0.3) ** 2) / (2 * sg * sg)) + np.exp(-((x - 0.7) ** 2) / (2 * sg * sg))
        candidates.append(normalize(cands))
    # Power taper
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(normalize(np.maximum(x * (1 - x), 1e-10) ** p))
    # Cosine windows
    for alpha in [0.1, 0.2, 0.3, 0.5]:
        cands = np.ones(n)
        m = int(alpha * n)
        if m > 0:
            taper = 0.5 * (1 + np.cos(np.pi * np.arange(m) / m))
            cands[:m] = taper
            cands[-m:] = taper[::-1]
        candidates.append(normalize(cands))
    # Spectral inits
    for seed in [0, 1, 2, 7, 42]:
        candidates.append(spectral_init(n, seed))
    # Modulated sine (non-negative envelope)
    for f in [1, 2, 3, 5]:
        candidates.append(normalize(np.sin(np.pi * x) * np.maximum(1 + 0.5 * np.cos(2 * np.pi * f * x), 0.1)))

    # Blend with parent
    if parent_seq is not None:
        bi = normalize(parent_seq)
        for w in [0.3, 0.5, 0.7]:
            for c_idx in [0, 3, 6, 9]:
                if c_idx < len(candidates):
                    candidates.append(normalize(w * bi + (1 - w) * candidates[c_idx]))

    best_a = candidates[0]
    best_val = obj_val(candidates[0])
    for c in candidates:
        v = obj_val(c)
        if v < best_val:
            best_val = v
            best_a = c.copy()

    return best_a, best_val
