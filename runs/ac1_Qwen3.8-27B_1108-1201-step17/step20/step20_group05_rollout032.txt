    # Then press[k] = sum_j A_ext[j] * W_ext[j+k] for appropriate ranges
    # = cross-correlation at lag k
    # = (A_ext * W_ext_reversed)[k + 2n - 2]
    
    # Let me just use the direct convolution approach:
    # I'll reverse w, convolve with a, and extract the right part
    wr = weights[::-1]  # length 2n-1
    # conv of a (length n) and wr (length 2n-1): length 3n-1... that's too long
    # Actually: np.convolve(a, wr) has length n + (2n-1) - 1 = 3n-1
    # (a * wr)[m] = sum_j a[j] * wr[m-j] = sum_j a[j] * w[2n-2-m+j]
    # We want sum_j a[j] * w[j+k]
    # So: 2n-2-m+j = j+k => m = 2n-2-k
    # press[k] = (a * wr)[2n-2-k]
    # Let me verify: (a*wr)[2n-2-k] = sum_j a[j] * wr[2n-2-k-j] = sum_j a[j] * w[2n-2-(2n-2-k-j)] = sum_j a[j] * w[k+j] ✓
    
    # So I need indices 2n-2-k for k=0,...,n-1, i.e., indices from 2n-2 down to n-1
    # conv = np.convolve(a, wr) would work but is O(n^2)
    # Using FFT:
    total = n + (2 * n - 1) - 1  # 3n - 2
    N2 = next_pow2(total + 1)
    Af = np.fft.rfft(a, N2)
    Wr = np.fft.rfft(wr, N2)
    conv_full = np.fft.irfft(Af * Wr, N2)[:total + 1]
    
    # press[k] = conv_full[2*n-2-k] for k = 0, ..., n-1
    idx = 2 * n - 2 - np.arange(n)
    out = conv_full[idx]
    out = np.where(np.isfinite(out), out, 0.0)
    return out


def softmax_weights(conv, beta):
    """Compute softmax weights over convolution values."""
    c = np.asarray(conv, dtype=np.float64)
    c_max = float(np.max(c))
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        w = np.zeros(c.size, dtype=np.float64)
        w[np.argmax(c)] = 1.0
        return w
    return e / s


def safe_eval(a):
    try:
        arr = np.asarray(a, dtype=np.float64)
        if arr.size == 0:
            return float("inf")
        arr = np.clip(arr, 0.0, 1000.0)
        if not np.all(np.isfinite(arr)):
            return float("inf")
        s = float(arr.sum())
        if s < 0.01:
            return float("inf")
        lst = [float(x) for x in arr]
        val = evaluate_sequence(lst)
        if val is None:
            return float("inf")
        val = float(val)
        if not math.isfinite(val):
            return float("inf")
        return val
    except Exception:
        return float("inf")


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    try:
        arr = np.asarray(a, dtype=np.float64)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        np.maximum(arr, 0.0, out=arr)
        s = float(arr.sum())
        if arr.size == 0 or s < 0.01:
            return best_seq, best_score
        arr = arr / s
        sc = safe_eval(arr)
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_seq = arr.copy()
            best_score = float(sc)
            print("  NEW BEST [" + label + "]: " + repr(sc) +
                  " n=" + str(arr.size) +
                  " t=" + repr(time.monotonic() - t_global))
    except Exception:
        pass
    return best_seq, best_score


def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    new_n = int(max(2, int(new_n)))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)


def default_sequence(n=5000):
    x = np.linspace(0.0, 1.0, int(n))
    return norm_sum1(np.sin(np.pi * x) ** 2)


def asymmetric_sequence(n):
    """Create an asymmetric initial sequence."""
    x = np.linspace(0.0, 1.0, int(n))
    # Weight towards one side
    y = np.exp(-3.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-3.0 * (x - 0.7) ** 2)
    return norm_sum1(y)


def comb_sequence(n, teeth=5):
    """Create a comb-like pattern."""
    x = np.linspace(0.0, 1.0, int(n))
    y = 1.0 + 0.5 * np.sin(2.0 * np.pi * teeth * x)
    return norm_sum1(np.maximum(y, 0.0))


def lse_objective_and_grad(a, beta):
    """
    Compute log-sum-exp smoothed objective and gradient.
    Returns (f, grad) where f = beta * log(sum(exp(beta*conv))) and
    grad = d(f * 2n) / da (the actual gradient of 2n * smoothed_max).
    """
    n = a.size
    conv = fft_conv(a)
    c_max = float(np.max(conv))
    shifted = beta * (conv - c_max)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return c_max * 2 * n, np.zeros(n, dtype=np.float64)
    
    p = e / s
    f_smooth = c_max + math.log(s) / beta
    f = 2.0 * n * f_smooth
    
    # Gradient: 2n * sum_i p_i * 2 * a[i-k] = 4n * pressure
    press = compute_pressure(p, a)
    grad = 4.0 * n * press
    
    return f, grad


def lse_objective_v(v, beta, ref):
    """
    Objective in softmax parameterization.
    v has size n-1, ref is the reference index.
    a_k = exp(u_k) / sum exp(u_j) where u[ref]=0, u[other]=v[...].
    """
    n = v.size + 1
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    
    u = np.zeros(n, dtype=np.float64)
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    
    u_max = float(np.max(u))
    e = np.exp(u - u_max)
    a = e / float(e.sum())
    
    f, ga = lse_objective_and_grad(a, beta)
    
    # Chain rule: dL/dv_j = a_j * (ga_j - sum_k a_k * ga_k)
    mean_grad = float(np.dot(a, ga))
    gu = a * (ga - mean_grad)
    
    gv = np.zeros(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]
    
    return f, gv


def run_lbfegs(a0, beta, maxfun=500, verbose=False):
    """Run L-BFGS-B on the smoothed objective in softmax coordinates."""
    a = norm_sum1(a0)
    n = a.size
    if n <= 2 or not SCIPY_OK or maxfun <= 0:
        return a
    
    # Choose ref as argmax for numerical stability
    ref = int(np.argmax(a))
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    
    v0 = np.zeros(n - 1, dtype=np.float64)
    if ref > 0:
        v0[:ref] = loga[:ref]
    if ref < n - 1:
        v0[ref:] = loga[ref + 1:]
    v0 = np.where(np.isfinite(v0), v0, -50.0)
    
    t0 = time.monotonic()
    try:
        res = scipy.optimize.minimize(
            lse_objective_v,
            v0,
            args=(beta, ref),
            method="L-BFGS-B",
            jac=True,
            options={"maxfun": int(maxfun), "maxiter": int(maxfun + 50),
                     "ftol": 1e-14, "gtol": 1e-8, "disp": False}
        )
    except Exception as e:
        if verbose:
            print("    L-BFGS error: " + repr(e))
        return a
    
    if res is not None and hasattr(res, "x"):
        vx = np.asarray(res.x, dtype=np.float64)
        if vx.size == v0.size and np.all(np.isfinite(vx)):
            n = vx.size + 1
            u = np.zeros(n, dtype=np.float64)
            u[ref] = 0.0
            if ref > 0:
                u[:ref] = vx[:ref]
            if ref < n - 1:
                u[ref + 1:] = vx[ref:]
            um = float(np.max(u))
            e = np.exp(u - um)
            a_out = e / float(e.sum())
            a_out = norm_sum1(a_out)
            if verbose:
                print("    L-BFGS beta=" + repr(beta) + " n=" + str(n) +
                      " time=" + repr(time.monotonic() - t0))
            return a_out
    
    return a


def mirror_descent_step(a, beta, eta):
    """
    One mirror descent step using the probability simplex with negative entropy.
    a_k <- a_k * exp(-eta * grad_k) / sum
    where grad is the gradient of the smoothed objective.
    """
    n = a.size
    _, grad = lse_objective_and_grad(a, beta)
    # Normalize gradient to have zero mean (tangent to simplex)
    grad_tangent = grad - float(np.mean(grad))
    # Multiply
    log_mult = -eta * grad_tangent
    log_mult = np.clip(log_mult, -10.0, 10.0)
    mult = np.exp(log_mult)
    a_new = a * mult
    a_new = np.maximum(a_new, 0.0)
    s = float(a_new.sum())
    if s < 1e-30:
        return a.copy()
    return a_new / s


def pressure_coordinate_descent(a, budget, beta_initial=50.0, beta_final=2000.0, verbose=True):
    """
    Coordinate descent guided by the pressure (gradient of smoothed objective).
    Moves mass from high-pressure to low-pressure positions.
    """
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a)
    n = a.size
    conv = fft_conv(a)
    M = float(np.max(conv))
    best_a = a.copy()
    best_M = M
    stall = 0
    total_iters = 0
    
    # Adaptive beta: start moderate, increase over time
    progress_basis = 0.5
    
    while time.monotonic() - t0 < time_limit - 5 and stall < 200:
        total_iters += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = beta_initial * (beta_final / beta_initial) ** (0.3 * progress)
        
        # Compute softmax weights and pressure
        w = softmax_weights(conv, beta)
        press = compute_pressure(w, a)
        
        # Find candidate source and destination
        # Source: positions with high pressure AND non-trivial mass
        # Destination: positions with low pressure
        mask_pos = a > 1e-10
        if not np.any(mask_pos):
            break
        
        # Score for source: pressure * log(a + eps) — want high pressure, enough mass
        src_score = np.where(mask_pos, press * np.log(a + 1e-300), -np.inf)
        # Score for dest: negative pressure + small regularization
        dst_score = -press + 1e-6 * a
        
        # Get top candidates
        n_src = min(20, n)
        n_dst = min(30, n)
        src_candidates = np.argpartition(src_score, n_src - 1)[:n_src]
        src_candidates = src_candidates[np.argsort(src_score[src_candidates])[::-1]]
        dst_candidates = np.argpartition(dst_score, n_dst - 1)[:n_dst]
        dst_candidates = dst_candidates[np.argsort(dst_score[dst_candidates])]
        
        best_move = None
        best_gain = 0
        
        for s in src_candidates:
            s = int(s)
            if a[s] < 1e-12:
                continue
            for d in dst_candidates:
                d = int(d)
                if d == s:
                    continue
                # Try several fractions
                for frac in (0.01, 0.03, 0.08, 0.15, 0.3, 0.5):
                    t = a[s] * frac
                    if t <= 0:
                        continue
                    # Quick estimate of new max using rank-1 update
                    # conv_new[i] = conv[i] + 2*t*(a[i-d] - a[i-s]) + t^2*(delta terms)
                    # For speed, just check the change at the current peak and nearby
                    # Use full update for accuracy
                    a_new = a.copy()
                    a_new[s] -= t
                    a_new[d] += t
                    a_new = np.maximum(a_new, 0.0)
                    sn = float(a_new.sum())
                    if sn < 1e-15:
                        continue
                    a_new = a_new / sn
                    conv_new = fft_conv(a_new)
                    M_new = float(np.max(conv_new))
                    gain = M - M_new
                    if gain > best_gain:
                        best_gain = gain
                        best_move = (M_new, s, d, t, frac)
        
        if best_move is not None:
            M_new, s, d, t, frac = best_move
            a[s] -= t
            a[d] += t
            a = np.maximum(a, 0.0)
            sn = float(a.sum())
            if sn < 1e-15:
                break
            a = a / sn
            conv = fft_conv(a)
            M = float(np.max(conv))
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and total_iters % 50 == 1:
                print("    PCO it=" + str(total_iters) + " M=" + repr(M) +
                      " beta=" + repr(beta) + " move=(" + str(s) + "," + str(d) + ")")
        else:
            stall += 1
            # Periodic perturbation to escape local minima
            if stall % 30 == 29:
                rng = np.random.default_rng(42 + stall)
                # Perturb a random subset
                k = max(3, n // 50)
                idx = rng.choice(n, size=k, replace=False)
                a[idx] *= np.exp(rng.normal(0, 0.1, k))
                a = np.maximum(a, 0.0)
                s = float(a.sum())
                if s > 1e-15:
                    a = a / s
                    conv = fft_conv(a)
                    M = float(np.max(conv))
                stall = 0
    
    return best_a


def game_direction(f, active_idx, mw_iters=30, seed=0, init_logits=None):
    """
    Game-theoretic Frank-Wolfe direction.
    Solves min_g max_i (f*g)[i] approximately via multiplicative weights.
    """
    f = norm_sum1(f)
    n = f.size
    size = 2 * n - 1
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K <= 0:
        return np.ones(n, dtype=np.float64) / float(n)
    
    rng = np.random.default_rng(seed)
    if init_logits is None:
        logits = np.zeros(K, dtype=np.float64)
    else:
        logits = np.asarray(init_logits, dtype=np.float64).copy()
        if logits.size != K:
            logits = np.zeros(K, dtype=np.float64)
        logits = np.where(np.isfinite(logits), logits, 0.0)
        lm = float(np.max(logits))
        if math.isfinite(lm):
            logits = logits - lm
    logits += rng.normal(0, 0.03, K)
    
    def norm_logits(loc):
        loc = np.where(np.isfinite(loc), loc, -1000.0)
        loc = loc - float(np.max(loc))
        wv = np.exp(loc)
        s = float(wv.sum())
        if s <= 0:
            wv = np.ones(K, dtype=np.float64)
            s = float(K)
        return wv / s
    
    lambda_w = norm_logits(logits)
    
    for _ in range(max(1, int(mw_iters))):
        # Compute g from current lambda
        pf = np.zeros(size, dtype=np.float64)
        pf[active_idx] = lambda_w
        pressure = compute_pressure(pf, f)
        pressure = np.maximum(pressure, 0.0)
        
        pmx = float(np.max(pressure))
        pmn = float(np.min(pressure))
        ptp = pmx - pmn
        if ptp <= 1e-30:
            g = np.ones(n, dtype=np.float64) / float(n)
        else:
            scaled = (pressure - pmn) / ptp
            wv = np.exp(-10.0 * scaled)
            wv = np.where(np.isfinite(wv), wv, 0.0)
            wv += 1e-12
            g = wv / float(wv.sum())
            g = 0.95 * g + 0.05 * (np.ones(n, dtype=np.float64) / float(n))
            g = g / float(g.sum())
        
        # Update lambda
        h = fft_conv2(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmac_ = float(np.max(ha))
        hmin = float(np.min(ha))
        hptp = hmac_ - hmin
        if hptp > 1e-30:
            eta = 15.0 / hptp
            logits += eta * (ha - float(np.mean(ha)))
        lambda_w = norm_logits(logits)
    
    return g


def frank_wolfe_search(a0, budget, active_max=2000, game_iters=30, verbose=True):
    """Frank-Wolfe search with game-theoretic directions."""
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    n = a.size
    conv = fft_conv(a)
    M = float(np.max(conv))
    best_a = a.copy()
    best_M = M
    
    # Active set: top convolution indices
    active = np.argsort(conv)[::-1][:active_max]
    
    no_improve = 0
    t = 0
    while time.monotonic() - t0 < time_limit - 10 and no_improve < 60:
        t += 1
        time_left = time_limit - (time.monotonic() - t0)
        if time_left < 20:
            break
        
        # Get game direction
        init_log = None
        if active.size > 0:
            av = conv[active]
            av = np.maximum(av, 1e-300)
            init_log = np.log(av)
            init_log -= float(np.max(init_log))
        g = game_direction(a, active, mw_iters=game_iters, seed=5000 + t, init_logits=init_log)
        
        if g is None:
            break
        
        # Line search
        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        
        best_alpha = 0.0
        best_m = M
        for alpha in (0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0):
            one = 1.0 - alpha
            val = (one * one) * conv + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
            m = float(np.max(val))
            if m < best_m:
                best_m = m
                best_alpha = alpha
        
        if best_alpha <= 0:
            no_improve += 1
            if no_improve % 20 == 19:
                rng = np.random.default_rng(9999 + t)
                pert = rng.normal(0, 0.02, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-12)
                a = a / float(a.sum())
                conv = fft_conv(a)
                M = float(np.max(conv))
                active = np.argsort(conv)[::-1][:active_max]
            continue
        
        new_a = (1.0 - best_alpha) * a + best_alpha * g
        new_a = np.maximum(new_a, 0.0)
        sn = float(new_a.sum())
        if sn < 1e-15:
            break
        new_a = new_a / sn
        new_conv = fft_conv(new_a)
        new_M = float(np.max(new_conv))
        
        if new_M < M - 1e-16:
            a = new_a
            conv = new_conv
            M = new_M
            no_improve = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            active = np.argsort(conv)[::-1][:active_max]
            if verbose and (t <= 3 or t % 15 == 0):
                print("    FW t=" + str(t) + " proxy_score=" + repr(2.0 * n * best_M) +
                      " alpha=" + repr(best_alpha))
        else:
            no_improve += 1
            if verbose and (t <= 3 or t % 15 == 0):
                print("    FW stall t=" + str(t) + " best_proxy=" + repr(2.0 * n * best_M))
    
    return best_a


def block_coordinate_move(a, budget, verbose=False):
    """
    Block coordinate descent: move mass between contiguous blocks.
    """
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a)
    n = a.size
    conv = fft_conv(a)
    M = float(np.max(conv))
    best_a = a.copy()
    best_M = M
    stall = 0
    
    block_sizes = [max(2, n // 100), max(5, n // 40), max(10, n // 20)]
    
    while time.monotonic() - t0 < time_limit - 5 and stall < 100:
        rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
        bsize = block_sizes[int(rng.integers(len(block_sizes)))]
        
        s = int(rng.integers(max(1, n - bsize)))
        d = int(rng.integers(max(1, n - bsize)))
        if abs(s - d) < bsize:
            continue
        
        # Get mass in source block
        src_block = a[s:s + bsize]
        src_sum = float(src_block.sum())
        if src_sum < 1e-12:
            continue
        
        # Try several fractions
        best_gain = 0
        best_frac = 0
        for frac in (0.02, 0.05, 0.1, 0.2, 0.4):
            t_total = src_sum * frac
            if t_total <= 0:
                continue
            a_new = a.copy()
            # Distribute proportionally from source
            a_new[s:s + bsize] -= src_block * frac
            # Add to destination block
            a_new[d:d + bsize] += t_total / bsize
            a_new = np.maximum(a_new, 0.0)
            sn = float(a_new.sum())
            if sn < 1e-15:
                continue
            a_new = a_new / sn
            conv_new = fft_conv(a_new)
            M_new = float(np.max(conv_new))
            gain = M - M_new
            if gain > best_gain:
                best_gain = gain
                best_frac = frac
        
        if best_gain > 0:
            frac = best_frac
            t_total = src_sum * frac
            a[s:s + bsize] -= src_block * frac
            a[d:d + bsize] += t_total / bsize
            a
