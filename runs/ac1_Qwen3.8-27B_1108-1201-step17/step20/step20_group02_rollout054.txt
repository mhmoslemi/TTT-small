    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def _compute_ideal(n):
    """Compute first n coefficients of sqrt(1+x+...+x^(2n-2))."""
    b = np.zeros(n)
    b[0] = 1.0
    for k in range(1, n):

        b[k] = max(0.0, val)

def _lp_fw(a_init, time_limit, solver, initial_K=500, max_K=3000, verbose=False):
    """LP-Frank-Wolfe with adaptive active set."""
    n = len(a_init)
    y = _norm1(a_init)
    best_a = y.copy()
    best_m = _fast_conv(y).max()
    t0 = time.monotonic()
    
    # Initialize active set with top violations
    ac = _fast_conv(y)
    M = ac.max()
    threshold = M * 0.95
    active_set = set(np.where(ac >= threshold)[0].tolist())
    if len(active_set) > initial_K:
        top = np.argsort(ac)[::-1][:initial_K]
        active_set = set(top.tolist())
    active_set = sorted(active_set)
    
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 500:
            break
        
        iter_count += 1
        ac = _fast_conv(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-13:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update active set
        if iter_count % 10 == 1 or no_improve > 100:
            new_threshold = M_curr * (0.90 if no_improve > 200 else 0.95)
            candidates = np.where(ac >= new_threshold)[0]
            if len(candidates) > max_K:
                # Keep top max_K
                order = np.argsort(ac[candidates])[::-1][:max_K]
                active_set = sorted(candidates[order].tolist())
            else:
                active_set = sorted(set(active_set).union(candidates.tolist()))
                if len(active_set) > max_K:
                    # Trim: keep those closest to M
                    margins = np.array([M_curr - ac[k] for k in active_set])
                    keep = np.argsort(margins)[:max_K]
                    active_set = sorted([active_set[i] for i in keep])
        
        K = len(active_set)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k_int in enumerate(active_set):
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(100, K // 2)]
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()

        g_dir = g_val / g_sum
        
        # Check new violations

        Yf = np.fft.rfft(y, fft_size)

        violations = np.where(conv_ys > M_curr + 1e-10)[0]

            active_set = sorted(new_set)[:max_K]

def _coord_polish(a_init, time_limit, seed=42, verbose=False):
    """Targeted coordinate descent: move mass from peak contributors to low-impact positions."""
    n = len(a_init)
    a = _norm1(a_init)
    best_a = a.copy()
    best_m = _fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 1000:
        ac = _fast_conv(a)
        top_k = np.argsort(ac)[::-1][:20]
        improved = False
        
        for k_int in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find low-impact destination
            impact = np.zeros(n)
            for k2 in top_k[:10]:
                s2 = max(0, int(k2) - n + 1)
                e2 = min(n - 1, int(k2))
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    impact[idx2] += a[int(k2) - idx2]
            
            candidates = np.where(impact < np.percentile(impact, 25))[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src:
                continue
            
            for frac in [0.05, 0.03, 0.01, 0.005, 0.001]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_t = _fast_conv(a_trial).max()
                if M_t < best_m - 1e-14:
                    a = a_trial
                    best_m = M_t
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            if stall % 100 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = _fast_conv(a).max()
                        if M_new < best
