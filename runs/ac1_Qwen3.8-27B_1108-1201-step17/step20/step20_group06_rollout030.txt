def project_simplex(v):
    n=len(v)
    if n==0: return v
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    rho=np.arange(1,n+1)
    cond = u * rho > (cssv - 1)
    I = np.nonzero(cond)[0]
    if len(I)==0: return np.ones(n)/n
    rho = I[-1]
    theta = (cssv[rho]-1)/(rho+1.0)
    return np.maximum(v-theta,0.0)

```python
"""
Search for non-negative step-function heights minimizing

    J(a) = 2 * n * max_k conv(a,a)_k / (sum a)^2.

The code normalizes to sum(a)=1 and minimizes 2*n*max conv(a,a).  It uses:

- FFT-based exact convolution and gradients for polynomial-square coefficients.
- A smooth log-sum-exp objective over the convolution coefficients, optimized
  implicitly on the simplex via a softmax parameterization and L-BFGS-B.
- A custom Nesterov-style projected subgradient method with active-lag weighting
  for exact final polishing.
- Diverse analytic starts and short homotopy runs to nearby lengths.

The final returned sequence is chosen among several candidates by the visible
evaluate_sequence() when time allows, otherwise by a local FFT proxy that is
equivalent up to floating-point roundoff.
"""

import numpy as np
import time
import math

try:
    from scipy import optimize as scipy_opt
    HAVE_SCIPY_OPT = True
except Exception:
    HAVE_SCIPY_OPT = False


class SearchTimeLimit(Exception):
    pass


# -----------------------------------------------------------------------------
# Global state used by top-level Scipy callbacks (no closures/lambdas allowed).
# -----------------------------------------------------------------------------
OPT_N = 0
OPT_FFTc = 0
OPT_FFTg = 0

LBFGS_BETA = 10.0
LBFGS_DEADLINE = 0.0
LBFGS_STOP = False

CACHE_Z = None
CACHE_GRAD = None
CACHE_BETA = -1.0


# -----------------------------------------------------------------------------
# Basic utilities
# -----------------------------------------------------------------------------
def next_pow2(x):
    x = int(max(1, x))
    return 1 << (x - 1).bit_length()


def set_opt_sizes(n):
    global OPT_N, OPT_FFTc, OPT_FFTg
    OPT_N = int(n)
    OPT_FFTc = next_pow2(2 * OPT_N - 1)
    OPT_FFTg = next_pow2(3 * OPT_N - 2)


def norm_sum1(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = np.where(np.isfinite(a), a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    a = a / s
    # Clean possible tiny negative zeros.
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def interpolate_sequence(f, new_n):
    f = norm_sum1(f)
    old_n = int(f.size)
    new_n = int(max(1, new_n))
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    g = np.interp(new_x, old_x, f)
    return norm_sum1(g)


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = int(v.size)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    idx = np.nonzero(cond)[0]
    if idx.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(idx[-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    out = v - theta
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return out / s


# -----------------------------------------------------------------------------
# Convolution and objective/gradient helpers
# -----------------------------------------------------------------------------
def conv_square(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if n == 1:
        return np.array([float(a[0] * a[0])], dtype=np.float64)
    if OPT_N != n:
        set_opt_sizes(n)
    A = np.fft.rfft(a, OPT_FFTc)
    c = np.fft.irfft(A * A, OPT_FFTc)[:2 * n - 1]
    c = np.abs(c)
    # True convolution of a nonnegative sequence is nonnegative; clip roundoff.
    c[c < 1e-15] = 0.0
    return c


def fast_score(a):
    a = norm_sum1(a)
    n = int(a.size)
    if n == 0:
        return float("inf")
    c = conv_square(a)
    if c.size == 0:
        return float("inf")
    return float(2.0 * n * max(0.0, float(np.max(c))))


def gradient_weights(a, w):
    """
    For a and weights w over convolution lags k=0..2n-2, return

        g_j = sum_i a_i * w_{i+j},  j=0..n-1.

    The gradient of the polynomial-square objective uses 4*n*g for
    2*n*max(c_k).
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if n == 1:
        if w.size == 1:
            return np.array([float(a[0] * w[0])], dtype=np.float64)
        return np.array([float(a[0])] * max(1, min(n, w.size)), dtype=np.float64)
    if OPT_N != n:
        set_opt_sizes(n)

    m = 2 * n - 1
    if w.size != m:
        # Defensive fallback: truncate/pad weights.
        tmp = np.zeros(m, dtype=np.float64)
        r = min(m, w.size)
        tmp[:r] = w[:r]
        w = tmp

    # FFT correlation / convolution trick:
    # q = reversed(w).  conv(a,q)[m-1-j] = sum_i a_i w_{i+j}.
    q = w[::-1]
    Af = np.fft.rfft(a, OPT_FFTg)
    Qf = np.fft.rfft(q, OPT_FFTg)
    corr = np.fft.irfft(Af * Qf, OPT_FFTg)

    # Need indices m-1, m-2, ..., m-n = n-1.
    # Python slice: start=m-1, stop=n-2, step=-1.
    g = corr[m - 1 : n - 2 : -1]
    if g.size != n:
        # Extremely defensive direct fallback.
        g = np.zeros(n, dtype=np.float64)
        for k in range(m):
            wk = float(w[k])
            if wk == 0.0:
                continue
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if hi < lo:
                continue
            i = np.arange(lo, hi + 1, dtype=np.int64)
            j = k - i
            g[j] += a[i] * wk
    return np.asarray(g, dtype=np.float64)


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z)) if z.size else z
    z = np.clip(z, -700.0, 700.0)
    a = np.exp(z)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(max(1, a.size))
    a = a / s
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(max(1, a.size))
    return a / s


def compute_z_data(z, beta):
    """
    Return value, gradient w.r.t. z, a, c, weights for the smoothed objective

        2 * smooth_max(n * conv(a,a)).

    Gradient is for the softmax parameterization a = softmax(z).
    """
    beta = float(beta)
    a = softmax_from_z(z)
    n = int(a.size)
    if OPT_N != n:
        set_opt_sizes(n)

    c = conv_square(a)
    d = n * c
    if d.size == 0:
        val = 1e100
        gz = np.zeros_like(z, dtype=np.float64)
        return val, gz, a, c, np.zeros(0, dtype=np.float64)

    md = float(np.max(d))
    if beta <= 1e-12:
        w = np.ones(d.size, dtype=np.float64) / float(d.size)
        val = 2.0 * md
    else:
        shift = beta * (d - md)
        shift = np.clip(shift, -50.0, 0.0)
        e = np.exp(shift)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 1e-300:
            w = np.ones(d.size, dtype=np.float64) / float(d.size)
            val = 2.0 * md
        else:
            w = e / se
            val = 2.0 * (md + float(np.log(se)) / beta)

    gw = gradient_weights(a, w)
    # Objective = 2 * smooth_max(d), d = n*c.
    # d/d a [smooth_max(d)] = 2*n*g, times leading 2 => 4*n*g.
    ga = 4.0 * n * gw
    ga = np.where(np.isfinite(ga), ga, 0.0)

    sdot = float(np.dot(ga, a))
    gz = a * (ga - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    return float(val), gz, a, c, w

# Scipy L-BFGS-B wrappers (top-level, using globals)

def _lbfgs_fun(z):
    global LBFGS_STOP, CACHE_Z, CACHE_GRAD, CACHE_BETA
    if LBFGS_STOP or time.monotonic() > LBFGS_DEADLINE:
        raise SearchTimeLimit
    val, gz, a, c, w = compute_z_data(z, LBFGS_BETA)
    CACHE_Z = np.asarray(z, dtype=np.float64).copy()
    CACHE_GRAD = gz
    CACHE_BETA = float(LBFGS_BETA)
    if not np.isfinite(val):
        return 1e100
    return float(val)


def _lbfgs_jac(z):
    global LBFGS_STOP, CACHE_Z, CACHE_GRAD, CACHE_BETA
    if LBFGS_STOP or time.monotonic() > LBFGS_DEADLINE:
        raise SearchTimeLimit

    # Reuse the cached gradient if scipy asks for it immediately after fun(z).
    if CACHE_Z is not None and abs(float(CACHE_BETA) - float(LBFGS_BETA)) < 1e-12:
        try:
            if np.array_equal(np.asarray(z, dtype=np.float64), CACHE_Z):
                return np.asarray(CACHE_GRAD, dtype=np.float64).copy()
        except Exception:
            pass

    val, gz, a, c, w = compute_z_data(z, LBFGS_BETA)
    CACHE_Z = np.asarray(z, dtype=np.float64).copy()
    CACHE_GRAD = gz
    CACHE_BETA = float(LBFGS_BETA)
    return np.asarray(gz, dtype=np.float64)


def _lbfgs_callback(xk):
    global LBFGS_STOP
    if LBFGS_STOP or time.monotonic() > LBFGS_DEADLINE:
        LBFGS_STOP = True
        return False
    return None


def run_lbfgs(a0, time_limit, betas=None, maxiter_per_beta=500, verbose=True):
    if not HAVE_SCIPY_OPT:
        a = norm_sum1(a0)
        return a.copy(), fast_score(a)

    t_start = time.monotonic()
    deadline_total = t_start + max(1.0, float(time_limit))
    a = norm_sum1(a0)
    n = int(a.size)
    if OPT_N != n:
        set_opt_sizes(n)

    z0 = np.log(np.maximum(a, 1e-300))
    if z0.size:
        z0 = z0 - float(np.mean(z0))
    z0 = np.clip(z0, -20.0, 20.0)

    best_a = a.copy()
    best_score = fast_score(best_a)
    if verbose:
        print(f"    L-BFGS start n={n} score={best_score:.9f}")

    if betas is None:
        if time_limit > 900.0:
            betas = [10.0, 30.0, 100.0, 300.0, 1000.0]
        elif time_limit > 400.0:
            betas = [20.0, 80.0, 300.0, 1000.0]
        elif time_limit > 150.0:
            betas = [40.0, 200.0, 800.0]
        else:
            betas = [100.0, 500.0]

    bounds = [(-20.0, 20.0)] * n
    options = {"maxiter": int(maxiter_per_beta), "gtol": 1e-9, "ftol": 1e-13, "maxls": 20}

    for bi, beta in enumerate(betas):
        if time.monotonic() - t_start >= time_limit:
            break
        remaining = time_limit - (time.monotonic() - t_start)
        per_budget = remaining / float(len(betas) - bi)
        per_budget = min(per_budget, 300.0)
        if per_budget < 8.0:
            break

        global LBFGS_BETA, LBFGS_DEADLINE, LBFGS_STOP
        global CACHE_Z, CACHE_GRAD, CACHE_BETA

        LBFGS_BETA = float(beta)
        LBFGS_DEADLINE = min(deadline_total, time.monotonic() + per_budget)
        LBFGS_STOP = False
        CACHE_Z = None
        CACHE_GRAD = None
        CACHE_BETA = -1.0

        if verbose:
            print(f"    L-BFGS beta={LBFGS_BETA:.1f} budget={per_budget:.1f}s")

        try:
            res = scipy_opt.minimize(
                _lbfgs_fun,
                z0,
                jac=_lbfgs_jac,
                method="L-BFGS-B",
                bounds=bounds,
                options=options,
                callback=_lbfgs_callback,
            )
            if res.x is not None:
                a_res = softmax_from_z(np.clip(np.asarray(res.x, dtype=np.float64), -20.0, 20.0))
                sc = fast_score(a_res)
                if np.isfinite(sc) and sc < best_score - 1e-15:
                    best_score = sc
                    best_a = a_res.copy()
                    z0 = np.asarray(res.x, dtype=np.float64).copy()
                    if verbose:
                        print(f"    L-BFGS beta={LBFGS_BETA:.1f} new score={best_score:.9f}")
        except SearchTimeLimit:
            pass
        except Exception as e:
            if verbose:
                print(f"    L-BFGS beta={LBFGS_BETA:.1f} stopped: {e}")

    return norm_sum1(best_a), best_score


# -----------------------------------------------------------------------------
# Active-lag projected Nesterov-style exact polishing
# -----------------------------------------------------------------------------
def active_search(a0, time_limit, K=None, beta=100.0, eta_scale=5.0, verbose=True, seed=98765):
    t_start = time.monotonic()
    a = norm_sum1(a0)
    n = int(a.size)
    if OPT_N != n:
        set_opt_sizes(n)

    x = a.copy()
    y = a.copy()
    x_prev = a.copy()
    best_a = x.copy()
    best_score = fast_score(x)

    t_mom = 1.0
    eta = max(1e-12 / max(1, n), float(eta_scale) / max(1, n))
    eta_min = 1e-10 / max(1, n)
    eta_max = 300.0 / max(1, n)
    stall = 0
    it = 0

    size = 2 * n - 1
    if K is None:
        K = int(max(96, min(512, 0.02 * n)))
    K = int(max(1, min(size, K)))

    rng = np.random.default_rng(int(seed))

    if verbose:
        print(f"    Active search start n={n} score={best_score:.9f} K={K} beta={beta:.1f}")

    while time.monotonic() - t_start < float(time_limit):
        c_y = conv_square(y)
        d_y = n * c_y
        cur = 2.0 * n * float(np.max(d_y)) if d_y.size else float("inf")

        if K < d_y.size:
            idx = np.argpartition(d_y, -K)[-K:]
        else:
            idx = np.arange(d_y.size)

        top = d_y[idx]
        mtop = float(np.max(top)) if top.size else 0.0
        shift = np.clip(beta * (top - mtop), -50.0, 0.0)
        e = np.exp(shift)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 1e-300:
            w_top = np.ones(K, dtype=np.float64) / float(K)
        else:
            w_top = e / se

        w = np.zeros(d_y.size, dtype=np.float64)
        w[idx] = w_top
        ga = 4.0 * n * gradient_weights(y, w)

        found = False
        best_trial = None
        best_trial_score = cur
        best_eta = eta
        e_trial = eta

        for _ in range(8):
            zproj = y - e_trial * ga
            trial = project_simplex(zproj)
            ss = float(trial.sum())
            if ss <= 1e-300:
                trial = y.copy()
            else:
                trial = trial / ss
            c_t = conv_square(trial)
            vt = 2.0 * n * float(np.max(c_t)) if c_t.size else float("inf")
            if vt <= cur + 1e-14:
                found = True
                best_trial = trial
                best_trial_score = vt
                best_eta = e_trial
                break
            e_trial *= 0.5

        if found and best_trial is not None:
            # Accept if it decreases y slightly or if line search found no worse.
            if best_trial_score <= cur + 1e-14:
                x_prev = x
                x = np.asarray(best_trial, dtype=np.float64).copy()
                t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_mom * t_mom))
                theta = (t_mom - 1.0) / t_new
                t_mom = t_new
                y = x + theta * (x - x_prev)
                y = np.maximum(y, 0.0)
                sy = float(y.sum())
                if sy <= 1e-300:
                    y = x.copy()
                else:
                    y = y / sy

                if best_trial_score < best_score - 1e-15:
                    best_score = float(best_trial_score)
                    best_a = x.copy()
                    stall = 0
                    if verbose and (it % 500 == 0):
                        el = time.monotonic() - t_start
                        print(f"    Active t={it} ({el:.1f}s) score={best_score:.9f}")
                else:
                    if best_trial_score < cur - 1e-14:
                        stall = 0
                    else:
                        stall += 1
                eta = min(eta * 1.08, eta_max)
            else:
                eta = max(eta * 0.5, eta_min)
                stall += 1
        else:
            eta = max(eta * 0.5, eta_min)
            stall += 1

        if stall > 350:
            strat = int(rng.integers(4))
            if strat == 0:
                x = best_a * (1.0 + 0.05 * rng.standard_normal(n))
            elif strat == 1:
                x = 0.85 * best_a + 0.15 * rng.dirichlet(np.ones(n) * 0.2)
            elif strat == 2:
                x = best_a * np.exp(0.07 * rng.standard_normal(n))
            else:
                x = best_a.copy()
                mask = rng.random(n) > 0.5
                x[mask] *= 0.70
                x[~mask] *= 1.45
            x = np.maximum(x, 1e-20)
            sx = float(x.sum())
            if sx <= 1e-300:
                x = best_a.copy()
            else:
                x = x / sx
            y = x.copy()
            x_prev = x.copy()
            t_mom = 1.0
            eta = max(eta_min, float(eta_scale) / max(1, n))
            stall = 0
            beta = max(20.0, beta * 0.6)

        if (it % 1000 == 999) and stall > 120:
            if beta < 600.0:
                beta *= 1.25
            else:
                beta *= 0.6

        it += 1

    return norm_sum1(best_a), best_score


# -----------------------------------------------------------------------------
# Candidate generation
# -----------------------------------------------------------------------------
def add_candidate_to_list(cands, a):
    try:
        an = norm_sum1(a)
        if an.size == 0:
            return
        if not np.all(np.isfinite(an)):
            return
        if np.max(an) < 0.0:
            return
        sc = fast_score(an)
        if np.isfinite(sc):
            cands.append((float(sc), np.asarray(an, dtype=np.float64).copy()))
    except Exception:
        pass


def make_candidates(n, seed_a):
    n = int(max(1, n))
    cands = []

    # Seed and uniform baseline.
    add_candidate_to_list(cands, seed_a)
    add_candidate_to_list(cands, np.ones(n, dtype=np.float64) / float(n))

    t = np.linspace(0.0, 1.0, n)
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx) / max(1.0, n / 50.0)

    # Two-sided endpoint-heavy exponential layers.
    for lam in [0.0, 0.1, 0.2, 0.35, 0.6, 1.0, 1.8, 3.0, 5.0, 9.0]:
        add_candidate_to_list(cands, np.exp(-lam * dist))

    # Sums of endpoint powers.
    for p in [0.0, 0.25, 0.5, 1.0, 2.0, 4.0, 8.0]:
        a = np.power(t + 1e-6, p) + np.power(1.0 - t + 1e-6, p)
        add_candidate_to_list(cands, a)

    # Cosine endpoint peaks.
    for delta in [0.0, 0.2, 0.45, 0.75, 1.0, 1.35]:
        a = 1.0 + delta * np.cos(2.0 * np.pi * t)
        if float(np.min(a)) <= 0.0:
            a = np.maximum(a, 0.05)
        add_candidate_to_list(cands, a)

    # Quadratic endpoint peaks.
    for delta in [0.0, 0.25, 0.6, 1.2, 2.5, 5.0]:
        a = 1.0 + delta * ((2.0 * t - 1.0) ** 2)
        add_candidate_to_list(cands, a)

    # Smooth random low-pass bumps.
    for sd in [11, 23, 37, 53, 71]:
        rng = np.random.default_rng(sd)
        z = rng.normal(0.0, 0.35, n)
        fs = next_pow2(2 * n)
        A = np.fft.rfft(z, fs)
        k = np.arange(A.shape[0])
        cutoff = max(1, n // 4)
        A[k > cutoff] = 0.0
        z = np.fft.irfft(A, fs)[:n]
        a = 1.0 + 0.8 * z
        a = np.maximum(a, 1e-3)
        add_candidate_to_list(cands, a)

    cands = sort_candidates(cands, max_keep=16)
    return cands


def sort_candidates(cands, max_keep=12):
    if not cands:
        return []
    try:
        scores = np.array([float(x[0]) for x in cands], dtype=np.float64)
        order = np.argsort(scores)
    except Exception:
        return []
    out = []
    for i in order[:max_keep]:
        out.append(cands[int(i)])
    return out

def exact_score(a):
    try:
        an = norm_sum1(a)
        if an.size == 0:
            return float("inf")
        lst = [float(x) for x in an]
        return float(evaluate_sequence(lst))
    except Exception:
        return float("inf")

# Main search routine

def propose_candidate(budget_s=1800.0):
    t0 = time.monotonic()
    budget = min(float(budget_s) - 45.0, 1740.0)
    budget = max(60.0, budget)

    def elapsed():
        return time.monotonic() - t0

    def remaining():
        return max(0.0, budget - elapsed())

    print(f"Total internal search budget: {budget:.1f}s")

    # Load parent sequence if available.
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.ndim == 1 and p.size > 0:
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.clip(p, 0.0, 1000.0)
            if float(p.sum()) > 1e-300:
                parent = norm_sum1(p)
    except Exception:
        parent = None

    if parent is None:
        n_init = 8192
        tt = np.linspace(0.0, 1.0, n_init)
        parent = norm_sum1(np.sin(np.pi * tt) ** 2)
        print("No parent found; using default sine-square init.")

    base_n = int(parent.size)
    if base_n < 4096:
        base_n = 8192
    elif base_n > 10000:
        base_n = 10000

    # Prefer a modest, FFT-friendly length.  8192 is safe and often better than 6700.
    if base_n <= 8192:
        main_n = 8192
    else:
        main_n = base_n

    a_main = interpolate_sequence(parent, main_n)
    sc_main = fast_score(a_main)
    print(f"Parent original n={int(parent.size)}; working n={main_n}; fast score={sc_main:.9f}")

    # Candidate pool.
    cands = []
    add_candidate_to_list(cands, parent)
    add_candidate_to_list(cands, a_main)
    cands = make_candidates(main_n, a_main)
    add_candidate_to_list(cands, parent)
    add_candidate_to_list(cands, a_main)
    cands = sort_candidates(cands, max_keep=16)

    if not cands:
        best_seq = norm_sum1(np.ones(main_n))
    else:
        best_seq = cands[0][1].copy()
        best_score = cands[0][0]

    print(f"Initial best fast score={best_score:.9f}, n={int(best_seq.size)}")

    # ---------------- Phase 1: L-BFGS on best main-length candidate ----------------
    rem = remaining()
    if rem > 260.0 and HAVE_SCIPY_OPT:
        lb_budget = min(rem * 0.55, 820.0)
        print(f"\n=== Phase 1: L-BFGS (budget {lb_budget:.1f}s) ===")
        out, sc = run_lbfgs(a_main, lb_budget, betas=[10.0, 40.0, 150.0, 500.0, 1500.0],
                            maxiter_per_beta=700, verbose=True)
        if np.isfinite(sc) and sc < best_score - 1e-15:
            best_score = sc
            best_seq = out.copy()
            add_candidate_to_list(cands, out)

        # Run a second L-BFGS from a different strong candidate if available.
        rem = remaining()
        second_done = 0
        for cand_sc, cand_a in cands:
            if int(cand_a.size) != main_n:
                continue
            if np.array_equal(np.asarray(cand_a, dtype=np.float64),
                              np.asarray(a_main, dtype=np.float64)):
                continue
            if rem < 220.0:
                break
            b_budget = min(rem * 0.30, 320.0)
            print(f"\n=== Phase 1b: L-BFGS alternative (budget {b_budget:.1f}s) ===")
            out2, sc2 = run_lbfgs(cand_a, b_budget, betas=[30.0, 150.0, 600.0, 2000.0],
                                  maxiter_per_beta=500, verbose=True)
            if np.isfinite(sc2) and sc2 < best_score - 1e-15:
                best_score = sc2
                best_seq = out2.copy()
            add_candidate_to_list(cands, out2)
            second_done += 1
            if second_done >= 1:
                break

    # ---------------- Phase 2: exact active-lag polishing ----------------
    rem = remaining()
    if rem > 140.0:
        act_budget = min(rem * 0.35, 430.0)
        n_cur = int(best_seq.size)
        K_cur = int(max(128, min(512, 0.018 * n_cur)))
        print(f"\n=== Phase 2: Active polish (budget {act_budget:.1f}s, n={n_cur}) ===")
        out, sc = active_search(best_seq, act_budget, K=K_cur, beta=120.0, eta_scale=5.0,
                                verbose=True, seed=424242)
        if np.isfinite(sc) and sc < best_score - 1e-15:
            best_score = sc
            best_seq = out.copy()
        add_candidate_to_list(cands, out)

    # ---------------- Phase 3: nearby homotopy lengths ----------------
    rem = remaining()
    last_n = int(best_seq.size)
    proposed_lengths = []
    if last_n < 8192:
        proposed_lengths = [8192, 9500, 10000]
    elif last_n < 10000:
        proposed_lengths = [last_n + 1000, 10000]
    else:
        proposed_lengths = [10000]

    proposed_lengths = [x for x in proposed_lengths if last_n < x <= 10000]

    for new_n in proposed_lengths:
        rem = remaining()
        if rem < 200.0:
            break
        print(f"\n=== Phase 3: Length {new_n} (remaining {rem:.1f}s) ===")
        a_long = interpolate_sequence(best_seq, new_n)
        sc_long = fast_score(a_long)
        if np.isfinite(sc_long) and sc_long < best_score - 1e-15:
            best_score = sc_long
            best_seq = a_long.copy()
        add_candidate_to_list(cands, a_long)

        b_budget = min(rem * 0.30, 320.0)
        if HAVE_SCIPY_OPT and b_budget > 120.0:
            out, sc = run_lbfgs(a_long, b_budget, betas=[30.0, 150.0, 600.0, 1800.0],
                                maxiter_per_beta=400, verbose=False)
        else:
            out, sc = active_search(a_long, b_budget, K=int(max(96, 0.015 * new_n)),
                                    beta=150.0, eta_scale=5.0, verbose=False,
                                    seed=707 + int(new_n % 9973))
        if np.isfinite(sc) and sc < best_score - 1e-15:
            best_score = sc
            best_seq = out.copy()
        add_candidate_to_list(cands, out)

    # ---------------- Phase 4: final local polish ----------------
    rem = remaining()
    if rem > 90.0:
        act_budget2 = min(rem * 0.45, 260.0)
        n_cur = int(best_seq.size)
        print(f"\n=== Phase 4: Final active polish (budget {act_budget2:.1f}s, n={n_cur}) ===")
        out, sc = active_search(best_seq, act_budget2, K=int(max(64, 0.012 * n_cur)),
                                beta=250.0, eta_scale=3.0, verbose=True, seed=31337)
        if np.isfinite(sc) and sc < best_score - 1e-15:
            best_score = sc
            best_seq = out.copy()
        add_candidate_to_list(cands, out)

    # Sort final candidate pool by FAST score and exact-evaluate a few bests.
    final_pool = []
    add_candidate_to_list(final_pool, best_seq)
    for sc, a in cands:
        add_candidate_to_list(final_pool, a)
    final_pool = sort_candidates(final_pool, max_keep=8)

    chosen_seq = np.asarray(best_seq, dtype=np.float64).copy()
    chosen_score = best_score

    if remaining() > 60.0 and final_pool:
        # Evaluate at most 3 candidates; direct convolution is O(n^2).
        exact_best_score = float("inf")
        exact_best_seq = None
        for sc, a in final_pool[:3]:
            if remaining() < 35.0:
                break
            esc = exact_score(a)
            print(f"  Exact eval candidate: fast={sc:.9f}, exact={esc:.9f}, n={int(a.size)}")
            if np.isfinite(esc) and esc < exact_best_score:
                exact_best_score = esc
                exact_best_seq = np.asarray(a, dtype=np.float64).copy()
        if exact_best_seq is not None and np.isfinite(exact_best_score):
            chosen_seq = exact_best_seq
            chosen_score = exact_best_score

    # Safety final exact check on the chosen one if time is plentiful.
    if remaining() > 40.0:
        esc_final = exact_score(chosen_seq)
        print(f"  Final exact check: {esc_final:.9f}")
        if np.isfinite(esc_final):
            chosen_score = esc_final

    chosen = np.asarray(chosen_seq, dtype=np.float64)
    chosen = np.maximum(chosen, 0.0)
    chosen = np.where(np.isfinite(chosen), chosen, 0.0)
    ss = float(chosen.sum())
    if ss <= 0.01:
        # Extremely defensive: if sum is too small, scale up.
        if chosen.size == 0:
            chosen = np.ones(8192, dtype=np.float64) / 8192.0
        else:
            chosen = chosen * (0.01 / max(1e-300, ss))

    out_list = []
    for x in chosen.ravel():
        xf = float(x)
        if not np.isfinite(xf):
            xf = 0.0
        if xf < 0.0:
            xf = 0.0
        if xf > 1000.0:
            xf = 1000.0
        out_list.append(xf)

    print(f"FINAL returned n={len(out_list)} score={chosen_score:.9f} time={elapsed():.1f}s")
    return out_list
```
