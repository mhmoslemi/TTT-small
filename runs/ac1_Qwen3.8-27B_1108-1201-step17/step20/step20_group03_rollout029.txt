        grad, p, b = compute_smoothed_grad(y, beta, top_k)

        # Barzilai-Borwein step size
        if s_prev is not None and y_prev is not None:
            sy = y - y_prev
            gy = grad - s_prev
            sy_gy = sy.dot(gy)
            gy_gy = gy.dot(gy)
            if gy_gy > 1e-20:
                bb_candidate = sy_gy / gy_gy
                if 1e-8 < bb_candidate < 10.0:
                    bb_step = max(1e-6, min(bb_candidate, 5.0 / beta))
                else:
                    bb_step = max(bb_step * 0.9, 1e-6)
            s_prev = grad.copy()
            y_prev = y.copy()

        eta = min(bb_step, 0.15)
        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.01 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 10.0)
            s_prev = None
            y_prev = None
            bb_step = 1.0 / np.sqrt(n)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            if t % 25000 == 0:
                print("  FISTA t={} ({}s): J={} beta={}".format(t, round(elapsed,1), round(J,6), round(beta,1)))

    return best_a, best_max_b

def targeted_local_search(a0, time_limit, verbose=False):
    """Local search that specifically targets the top autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    max_stall = 100
    
    ftp_size = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_autocorr(a)
        # Find top 5 peaks
        top_idx = np.argsort(b)[::-1][:5]
        top_val = b[top_idx[0]]
        
        # Compute sensitivity for each position w.r.t. top peaks
        # Sensitivity of b[k] w.r.t. a[j] is approximately a[j-k] + a[j+k] (for the peak at k)
        # But for the full convolution, it's more complex.
        # Use the gradient approach: grad_j = sum_k p[k] * (a[j-k] + a[j+k])
        # For simplicity, focus on the single top peak
        
        k_star = int(top_idx[0])
        
        # Positions that contribute most to peak at k_star
        # b[k_star] = sum_i a[i] * a[k_star - i] (with bounds)
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_star - idx]
            total_contrib = contrib.sum()
            if total_contrib > 1e-20:
                # Top contributing positions
                local_top = np.argsort(contrib)[::-1][:20]
                src_candidates = idx[local_top]
            
            # Find positions that DON'T contribute to this peak (safe to add mass)
            in_peak = np.zeros(n, dtype=bool)
            in_peak[i_start:i_end + 1] = True
            # Also mark positions that affect nearby peaks
            for k2 in top_idx[1:]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True
            
            dst_candidates = np.where(~in_peak)[0]
            if len(dst_candidates) < 20:
                dst_candidates = np.arange(n)
        else:
            src_candidates = np.argsort(a)[::-1][:20]
            dst_candidates = np.argsort(a)[:20]
        
        improved = False
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            if len(src_candidates) == 0 or len(dst_candidates) == 0:
                break
            
            src = int(src_candidates[rng.integers(len(src_candidates))])
            dst = int(dst_candidates[rng.integers(len(dst_candidates))])
            if src == dst or a[src] < 1e-12:
                continue
            
            frac = float(rng.choice([0.002, 0.005, 0.01, 0.02, 0.05]))
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-15:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
        
        if not improved:
            stall += 1
            # Try random perturbations
            for _ in range(10):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.001, 0.02))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
            if not improved and stall == max_stall:
                break

    return best_a, best_m

def cyclic_shift_best(a0):
    n = len(a0)
    b0 = fast_autocorr(a0)
    best_m = b0.max()
    best_shift = 0
    # Sample shifts for efficiency
    if n > 2000:
        shifts = np.random.RandomState(0).choice(n, size=min(n, 500), replace=False)
    else:
        shifts = range(n)
    for s in shifts:
        if s == 0:
            continue
        a_s = np.roll(a0, -s)
        b_s = fast_autocorr(a_s)
        m = b_s.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

Wait, I think there's an issue with my local search. The `fast_autocorr` call is O(n log n), and I'm calling it many times in the inner loop. For n=6700, each call takes maybe 1ms, and I'm doing 200 per outer iteration. So each outer iteration takes ~200ms, and with 100 outer iterations (max_stall), that's 20 seconds. That's too slow for a 300s budget.

Let me optimize: instead of computing the full autocorrelation for each trial, I can compute just the value at the critical lags.

For a trial `a_trial = a - amt*e_src + amt*e_dst`:
The change in b[k] = convolve(a,a)[k] is:
delta_b[k] = amt * (a_trial_related terms)

b_trial[k] = b[k] + amt*(a[k-src] + a[k+src] - a[k-dst] - a[
