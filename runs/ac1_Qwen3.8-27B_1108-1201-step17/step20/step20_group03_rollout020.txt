def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_gradient(a, beta, n_peaks=None):
    """
    Compute smoothed gradient of max(autocorr) w.r.t. a.
    Uses softmax over the autocorrelation values.
    """
    n = len(a)
    b = fast_autocorr(a)
    b_max = b.max()
    
    if n_peaks is not None and 0 < n_peaks < len(b):
        # Use top-k peaks
        idx = np.argsort(b)[::-1][:n_peaks]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
    else:
        b_use = b
    
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -50, 0)  # prevent overflow
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b)
        p[b == b_max] = 1.0 / max(1, (b == b_max).sum())
    else:
        p = exp_vals / p_total
    
    # Gradient: convolve reversed(a) with p, take the middle part
    a_rev = a[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1]
    return grad / beta, b, b_max

def fista_optimize(a0, time_limit, beta_start=10.0, beta_end=1000.0,
                   beta_growth=1.005, n_peaks=20, restart_every=3000,
                   step_decay=0.9999, verbose=False):
    """FISTA with adaptive step and Nesterov momentum."""
    global _best_seq, _best_score
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(42)
    step = 1.0 / (n ** 0.5)
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        grad, b, b_max = compute_gradient(y, beta, n_peaks)
        
        # Adaptive step size
        step = max(step * step_decay, 1e-8 / beta)
        
        z = y - step * grad
        x_new = simplex_project(z)
        
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x.copy()
        x = x_new
        t_nest = t_new
        
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        # Check global best
        J = 2.0 * n * best_max_b
        if J < _best_score:
            _best_score = J
            _best_seq = best_a.copy()
        
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.02 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 5.0)
            step = 1.0 / (n ** 0.5)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print(f"    FISTA t={t} ({elapsed:.1f}s): J={J:.6f} beta={beta:.1f} step={step:.2e}")
    
    if verbose:
        J = 2.0 * n * best_max_b
        print(f"    FISTA done ({time.monotonic()-t0:.1f}s): J={J:.6f}")
    
    return best_a, best_max_b

        ac = fast_autocorr(a)
        # Find top peaks
        peak_threshold = ac.max() * 0.95
        peaks = np.where(ac >= peak_threshold)[0]
        
        improved = False
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick a random peak
            k_int = int(peaks[rng.integers(len(peaks))])

            je = min(n - 1, k_int)
            if js > je:
                continue

            contrib = a[idx] * a[k_int - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-20:
                continue
            
            # Find the position with highest contribution
            src = int(idx[np.argmax(contrib)])
            
            # Find a good destination (low autocorrelation contribution)
            # Try positions that don't contribute to any peak
            in_any_peak = np.zeros(n, dtype=bool)
            for k2 in peaks[:20]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_any_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_any_peak)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            
            # Try multiple destinations and keep the best
            best_improvement = best_m
            best_a_trial = None
            best_frac = None
            best_dst = None
            
            for _ in range(20):
                dst = int(candidates[rng.integers(len(candidates))])
                if dst == src or a[src] < 1e-12:
                    continue

                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_improvement:
                        best_improvement = M_trial
                        best_a_trial = a_trial.copy()
                        best_frac = frac
                        best_dst = dst
            
            if best_a_trial is not None and best_improvement < best_m - 1e-14:
                a = best_a_trial
                best_m = best_improvement
                best_a = a.copy()
                improved = True
                stall = 0
                ac = fast_autocorr(a)
                peak_threshold = ac.max() * 0.95
                peaks = np.where(ac >= peak_threshold)[0]
                J = 2.0 * n * best_m
                if J < _best_score:
                    _best_score = J
                    _best_seq = best_a.copy()
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.005, 0.05))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:

                            best_a = a.copy()
                            stall = 0
                            improved = True
                            J = 2.0 * n * best_m

                                _best_score = J
                                _best_seq = best_a.copy()
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LocalSearch ({elapsed:.1f}s): J={J:.6f} stall={stall}")
    
    return best_a, best_m

1. FISTA with adaptive beta and Nesterov momentum
2. Structured peak-suppression local search with lookahead

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_grad(a, beta, n_peaks=None):
    n = len(a)
    b = fast_autocorr(a)
    b_max = b.max()

    if n_peaks is not None and 0 < n_peaks < len(b):
        idx = np.argsort(b)[::-1][:n_peaks]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
    else:
        b_use = b

    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -50.0, 0.0)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b)
        p[b == b_max] = 1.0 / max(1, (b == b_max).sum())
    else:
        p = exp_vals / p_total

    a_rev = a[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1]
    return grad / beta, b, b_max

    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(12345)
    step = 0.8 / (n ** 0.4)
    restart_count = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, b, b_max = compute_grad(y, beta, n_peaks)

        # Adaptive step: decrease slowly, reset on restart
        step *= 0.99995
        step = max(step, 1e-9 / beta)

        z = y - step * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x.copy()
        x = x_new
        t_nest = t_new

        if t % 100 == 0:
            beta = min(beta * beta_growth, beta_end)

        J = 2.0 * n * best_max_b
        if J < _best_score:
            _best_score = J
            _best_seq = best_a.copy()

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            restart_count += 1
            noise = 0.025 / (1.0 + 0.08 * restart_count)
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 5.0)
            step = 0.8 / (n ** 0.4)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print(f"    FISTA t={t} ({elapsed:.1f}s): J={J:.6f} b={beta:.0f} st={step:.1e}")

    return best_a, best_max_b


def local_search(a0, time_limit, verbose=False):
    global _best_seq, _best_score
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 400:
        ac = fast_autocorr(a)
        ac_max = ac.max()
        peak_thresh = ac_max * 0.98
        peaks = np.where(ac >= peak_thresh)[0]
        if len(peaks) == 0:
            peaks = np.argsort(ac)[::-1][:10]

        improved = False
        n_trials = 100 if len(peaks) < 30 else 50

        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break

            k_int = int(peaks[rng.integers(len(peaks))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            if a[src] < 1e-12:
                continue

            # Find positions not contributing to any peak
            in_peak = np.zeros(n, dtype=bool)
            for k2 in peaks[:25]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)

            # Try several destination/fraction combos
            best_trial_m = best_m
            best_trial_a = None
            for trial in range(15):
                dst = int(candidates[rng.integers(len(candidates))])
                if dst == src:
                    continue
                frac = float(rng.choice([0.003, 0.008, 0.02, 0.05, 0.1, 0.2]))
                amt = a[src] * frac
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                s_t = a_t.sum()
                if s_t < 1e-15:
                    continue
                a_t = a_t / s_t
                M_t = fast_autocorr(a_t).max()
                if M_t < best_trial_m:
                    best_trial_m = M_t
                    best_trial_a = a_t

            if best_trial_a is not None and best_trial_m < best_m - 1e-15:
                a = best_trial_a
                best_m = best_trial_m
                best_a = a.copy()
                improved = True
                stall = 0
                J = 2.0 * n * best_m
                if J < _best_score:
                    _best_score = J
                    _best_seq = best_a.copy()

        if not improved:
            stall += 1
            # Escape: random perturbation
            for _ in range(8):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.005, 0.08))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            J = 2.0 * n * best_m
                            if J < _best_score:
                                _best_score = J
                                _best_seq = best_a.copy()

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LS ({elapsed:.1f}s): J={J:.6f} stall={stall}")

    return best_a, best_m


def cyclic_shift_best(a0):
    n = len(a0)
    a = norm_sum1(a0)
    best_m = fast_autocorr(a).max()
    best_shift = 0

    # Coarse search
    step_c = max(1, n // 300)
    for s in range(step_c, n, step_c):
        a_s = np.roll(a, -s)
        m = fast_autocorr(a_s).max()
        if m < best_m:
            best_m = m
            best_shift = s

    # Refine
    if best_shift > 0:
        step_f = max(1, n // 5000)
        lo = max(1, best_shift - step_c * 2)
        hi = min(n - 1, best_shift + step_c * 2)
        for s in range(lo, hi + 1, step_f):
            a_s = np.roll(a, -s)
            m = fast_autocorr(a_s).max()
            if m < best_m:
                best_m = m
                best_shift = s

    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def coordinate_descent_polish(a0, time_limit, verbose=False):
    """Polish by trying to reduce individual peak contributions."""
    global _best_seq, _best_score
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(31337)

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        # Find the top peak
        k_peak = int(np.argmax(ac))
        peak_val = ac[k_peak]

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]

        if contrib.max() < 1e-20:
            break

        # Try reducing the highest-contributing position
        src_rank = np.argsort(contrib)[::-1]
        improved = False

        for rank in src_rank[:5]:
            src = int(idx[rank])
            if a[src] < 1e-12:
                continue

            # Try moving small amounts to random positions
            for _ in range(30):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.1]:
                    amt = a[src] * frac
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    s_t = a_t.sum()
                    if s_t < 1e-15:
                        continue
                    a_t = a_t / s_t
                    M_t = fast_autocorr(a_t).max()
                    if M_t < best_m - 1e-15:
                        a = a_t
                        best_m = M_t
                        best_a = a.copy()
                        improved = True
                        J = 2.0 * n * best_m
                        if J < _best_score:
                            _best_score = J
                            _best_seq = best_a.copy()
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random pair moves
            n_rand = 50
            any_improve = False
            for _ in range(n_rand):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = float(rng.uniform(0.005, 0.15))
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        any_improve = True
                        J = 2.0 * n * best_m
                        if J < _best_score:
                            _best_score = J
                            _best_seq = best_a.copy()
            if not any_improve:
                # Extra smooth perturbation
                pert = rng.normal(0, 0.002, n) * a
                a_p = a + pert
                a_p = np.maximum(a_p, 0.0)
                s_p = a_p.sum()
                if s_p > 1e-15:
                    a_p = a_p / s_p
                    M_p = fast_autocorr(a_p).max()
                    if M_p < best_m:
                        a = a_p
                        best_m = M_p
                        best_a = a.copy()

    if verbose:
        print(f"    CD polish done ({time.monotonic()-t0:.1f}s): J={2.0*n*best_m:.6f}")
    return best_a, best_m


def propose_candidate(budget_s=1800.0):

            parent = p.copy()
            p_score = evaluate_safe(parent)

                             beta_growth=1.006, n_peaks=12,

        if np.isfinite(sc) and sc < _best_score:
            _best_score = sc
            _best_seq = a_shifted.copy()

    # Final FISTA micro-polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:

        if np.isfinite(fs):

    return [float(x) for x in np.asarray(_best_seq, dtype=np.float64)]
