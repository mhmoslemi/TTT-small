Now let me think about the full algorithm more carefully and write clean code.

One concern: the mirror descent update `a_new[i] = a[i] * exp(-eta * grad[i])` can underflow if eta * grad[i] is very negative (making the exponential very large for small grad, or very small for large grad). To handle this, I'll use log-space:

log_a_new[i] = log(a[i]) - eta * grad[i]
Then normalize: a_new = exp(log_a_new - max(log_a_new)) / sum(exp(log_a_new - max(log_a_new)))

But a[i] could be 0, making log(a[i]) = -inf. I'll add a small floor: a[i] = max(a[i], 1e-30).

Actually, for the mirror descent to work well, we need all a[i] > 0 initially. If some a[i] = 0, the multiplicative update will keep them at 0 (since 0 * anything = 0). This is actually fine - it means we're restricting to the face of the simplex where those coordinates are zero.

But it might also limit our ability to find the global optimum. To handle this, I can:
1. Start with all a[i] > 0 (add a small epsilon)
2. Periodically "reactivate" zero coordinates by adding a small amount

Let me also think about the step size. For the mirror descent:
- The objective f_beta is L-Lipschitz on the simplex (with respect to L∞ norm)
- The optimal step size is eta ≈ 1/L
- L can be bounded by: L <= (2 * max_k p[k] * ||a||_∞ * n) / beta ≈ (2n * ||a||∞) / beta (rough bound)

In practice, I'll use adaptive step size: start with eta = 1.0 and decrease if the objective doesn't decrease.

Let me also consider using **accelerated mirror descent** (Nesterov's method) for faster convergence. This uses a "lookahead" step:
- y_t = (t/(t+3)) * a_t + (3/(t+3)) * a_{t-1} (for strong convexity, different formula)
- a_{t+1} = argmin_{a in simplex} <grad(y_t), a> + (1/(2*eta_t)) * D(a, a_t)

For negative entropy, D(a, b) = sum a[i] * log(a[i]/b[i]) + sum b[i] - sum a[i] = sum a[i] * log(a[i]/b[i]) (on the simplex).

The minimizer is a_{t+1}[i] = b[i] * exp(-grad(y_t)[i] / eta_t) / Z.

This is similar to the regular mirror descent but with the "blended" gradient evaluation point.

Actually, for our problem, the standard (non-accelerated) mirror descent should be sufficient given the number of iterations we can afford. Let me keep it simple.

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_soft_gradient(a, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*c[k])) where c = autocorr(a)."""
    n = len(a)
    size = 2 * n - 1
    
    # Compute autocorrelation
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    
    # Softmax weights (numerically stable)
    c_shift = c - c.max()
    exp_c = np.exp(beta * c_shift)
    p = exp_c / exp_c.sum()
    
    # Compute S1[i] = sum_j a[j] * p[i+j] via FFT convolution
    conv_len = n + size - 1  # = 3n-2
    conv_fft_size = 1 << (conv_len - 1).bit_length()
    
    a_pad = np.zeros(conv_fft_size)
    a_pad[:n] = a
    p_rev = np.zeros(conv_fft_size)
    p_rev[:size] = p[::-1]
    
    conv = np.fft.irfft(np.fft.rfft(a_pad) * np.fft.rfft(p_rev), conv_fft_size)[:conv_len]
    S1 = conv[n-1:2*n-1][::-1]
    
    # Compute S2[i] = a[i] * sum_{k=i}^{i+n-1} p[k]
    cumsum_p = np.concatenate([[0.0], np.cumsum(p)])
    window_sums = cumsum_p[n:] - cumsum_p[:n]  # cumsum_p[i+n] - cumsum_p[i] for i=0..n-1
    S2 = a * window_sums
    
    grad = (S1 + S2) / beta
    return grad, c.max()

def mirror_descent_step(a, grad, eta):
    """Multiplicative weights update: a_new ∝ a * exp(-eta * grad)."""
    # Log-space for numerical stability
    log_a = np.log(np.maximum(a, 1e-300))
    log_new = log_a - eta * grad
    log_new -= log_new.max()
    a_new = np.exp(log_new)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s

def simplex_project(z):
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

"""
Minimizes 2*n*max(autocorr(a))/(sum(a))^2 using:
1. Smoothed max objective (softmax) with FFT-based O(n log n) gradient computation
2. Mirror descent (multiplicative weights) for natural simplex handling
3. Adaptive beta annealing (smooth-to-sharp transition)
4. Exact subgradient method at high beta (projected gradient on simplex)
5. Peak shaving for final polish
6. Multi-length exploration

import numpy as np
import time


def fast_autocorr(a):
    """Compute linear autocorrelation (self-convolution) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_soft_gradient(a, beta):
    """
    Compute gradient of f_beta(a) = (1/beta)*log(sum_k exp(beta * c(k)))
    where c = autocorrelation(a).
    Returns (grad, max_c).
    """
    n = len(a)
    size = 2 * n - 1

    # Step 1: Compute autocorrelation
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]

    # Step 2: Softmax weights (numerically stable)
    c_shift = c - c.max()
    exp_c = np.exp(beta * c_shift)
    p = exp_c / exp_c.sum()

    # Step 3: S1[i] = sum_j a[j] * p[i+j] for i=0..n-1
    # Via FFT: S1[i] = (a * p_rev)[2n-2-i]
    conv_len = n + size - 1
    conv_fft_size = 1 << (conv_len - 1).bit_length()

    a_pad = np.zeros(conv_fft_size)
    a_pad[:n] = a
    p_rev_pad = np.zeros(conv_fft_size)
    p_rev_pad[:size] = p[::-1]

    conv = np.fft.irfft(np.fft.rfft(a_pad) * np.fft.rfft(p_rev_pad), conv_fft_size)[:conv_len]
    S1 = conv[n - 1:2 * n - 1][::-1]

    # Step 4: S2[i] = a[i] * sum_{k=i}^{i+n-1} p[k]
    cumsum_p = np.concatenate([[0.0], np.cumsum(p)])
    window_sums = cumsum_p[n:] - cumsum_p[:n]
    S2 = a * window_sums

    grad = (S1 + S2) / beta
    return grad, c.max()


def mirror_descent_step(a, grad, eta):
    """Multiplicative weights: a_new ∝ a * exp(-eta * grad), normalized."""
    log_a = np.log(np.maximum(a, 1e-300))
    log_new = log_a - eta * grad
    log_new -= log_new.max()
    a_new = np.exp(log_new)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s


def simplex_project(z):
    """Project onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length (preserving shape)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={best_score:.6f}")
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        best_seq = parametric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Mirror descent (main optimization) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Mirror descent (n={n0}, t_remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.45, 750.0)
    if t1 > 60:
        try:
            a_out, _ = mirror_descent_optimize_v2(best_seq, t1,
                                                  beta_start=3.0, beta_end=200.0,
                                                  n_beta_steps=6, verbose=True)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print(f"  MD improved: {sc:.6f}")
        except Exception as e:
            print(f"  MD error: {e}")

    # === Phase 2: Exact subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Exact subgradient (n={n_best}, t_remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.2, 250.0)
    if t2 > 30:
        try:
            a_out, _ = subgradient_exact(best_seq, t2, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print(f"  SubGrad improved: {sc:.6f}")
        except Exception as e:
            print(f"  SubGrad error: {e}")

    # === Phase 3: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Peak shaving (n={n_best}, t_remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 180.0)
    if t3 > 20:
        try:
            a_out = peak_shaving(best_seq, t3, top_k=20, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print(f"  PeakShave improved: {sc:.6f}")
        except Exception as e:
            print(f"  PeakShave error: {e}")

    # === Phase 4: Mirror descent round 2 (higher beta) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: MD round 2 (n={n_best}, t_remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.2, 250.0)
    if t4 > 30:
        try:
            a_out, _ = mirror_descent_optimize_v2(best_seq, t4,
                                                  beta_start=50.0, beta_end=2000.0,
                                                  n_beta_steps=4, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print(f"  MD2 improved: {sc:.6f}")
        except Exception as e:
            print(f"  MD2 error: {e}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    for dn in [-400, 400, -800, 800, -200, 200]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        new_n = n_best + dn
        if new_n < 3000 or new_n > 10000:
            continue

        print(f"\n=== Phase 5: n={new_n} (t_remaining={remaining:.0f}s) ===")
        a_interp = interpolate_sequence(best_seq, new_n)

        try:
            t5 = min(remaining * 0.35, 200.0)
            if t5 > 20:
                a_out, _ = mirror_descent_optimize_v2(a_interp, t5,
                                                      beta_start=10.0, beta_end=300.0,
                                                      n_beta_steps=4, verbose=False)
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out
                    n_best = new_n
                    print(f"  n={new_n} improved: {sc:.6f}")
        except Exception as e:
            print(f"  n={new_n} error: {e}")

    # === Phase 6: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}, t_remaining={remaining:.0f}s) ===")
    if remaining > 40:
        t6 = min(remaining * 0.5, 120.0)
        if t6 > 15:
            try:
                a_out = peak_shaving(best_seq, t6, top_k=15, verbose=False)
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out
                    print(f"  Final PS: {sc:.6f}")
            except Exception as e:
                print(f"  Final PS error: {e}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.6, 80.0)
            if t6b > 15:
                try:
                    a_out, _ = subgradient_exact(best_seq, t6b, verbose=False)
                    sc = evaluate_safe(a_out)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = a_out
                        print(f"  Final SG: {sc:.6f}")
                except Exception as e:
                    print(f"  Final SG error: {e}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        a_fb = (x + 0.01) ** 1.0 * (1 - x + 0.01) ** 1.0
        best_seq = norm_sum1(a_fb)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
