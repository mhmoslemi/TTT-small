            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def compute_subgradient(y, k_argmax, n):
    grad = np.zeros(n)
    i_start = max(0, k_argmax - n + 1)
    i_end = min(n, k_argmax + 1)
    if i_start < i_end:
        idx = np.arange(i_start, i_end)
        grad[idx] = y[k_argmax - idx]
    return grad


def compute_multi_subgradient(y, active_indices, weights, n):
    grad = np.zeros(n)
    for jj in range(len(active_indices)):
        k_int = int(active_indices[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += weights[jj] * y[k_int - idx]
    return grad


def lp_fw_2wf(a0, time_limit, solver, initial_K=500, max_K=4000, add_per_iter=50, verbose=True):
    """2-Step Frank-Wolfe with adaptive active set."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac_y = fast_convolve(y)
    best_m = ac_y.max()
    t0 = time.monotonic()

    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 600
    iter_count = 0
    stagnation_count = 0
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if stagnation_count > 5:
                if verbose:
                    print("    2-WF: persistent stagnation, stopping")
                break
            ac_curr = fast_convolve(y)
            n_pert = max(50, len(active_set) // 4)
            rng_restart = np.random.default_rng(42 + stagnation_count)
            keep = list(rng_restart.choice(len(active_set), size=len(active_set) - n_pert, replace=False))
            new_adds = np.argsort(ac_curr)[::-1][:n_pert * 3]
            new_set = set([active_set[i] for i in keep])
            for k in new_adds:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]
            no_improve = 0
            stagnation_count += 1
            continue

        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            stagnation_count = 0
        else:
            no_improve += 1

        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep_idx = np.argsort(margins)[:int(max_K * 0.6)]
            active_set_base = [active_set[i] for i in keep_idx]
            violations = np.where(ac > M_curr - 1e-8)[0]
            for v in violations[:100]:
                if int(v) not in active_set_base:
                    active_set_base.append(int(v))
            active_set = sorted(set(active_set_base))

        K = len(active_set)
        if K == 0:
            break
        if K > max_K:
            active_set = active_set[:max_K]
            K = max_K

        coeffs = build_constraint_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(100, K // 3)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]

        all_viol = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(all_viol) > 0:
            viol_amt = conv_ys[all_viol] - M_curr
            top_v = all_viol[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        # FW step with t = 2/(iter+2)
        t_step = 2.0 / (iter_count + 2.0)
        y_fw = (1.0 - t_step) * y + t_step * g_dir
        y_fw = np.maximum(y_fw, 0.0)
        s_fw = y_fw.sum()
        if s_fw < 1e-15:
            continue
        y_fw = y_fw / s_fw

        # 2-WF: pull-forward step along direction (g_dir - y)
        direction = g_dir - y
        dir_norm_sq = np.dot(direction, direction)

        if dir_norm_sq > 1e-20:
            ac_fw = fast_convolve(y_fw)
            k_max_fw = int(np.argmax(ac_fw))
            grad_fw = compute_subgradient(y_fw, k_max_fw, n)
            proj_dir = np.dot(grad_fw, direction)
            s_pull = max(0.0, -proj_dir / dir_norm_sq)
            s_pull = min(s_pull, 1.0 / max(t_step, 1e-6))

            # Try several pull amounts
            best_trial_M = M_curr
            best_s = 0.0
            candidates_s = [0.0, 0.5, 1.0, 1.5, s_pull, min(s_pull * 1.5, 1.0 / max(t_step, 1e-6)), 1.0 / max(t_step, 1e-6)]
            candidates_s = [max(0.0, min(cs, 1.0 / max(t_step, 1e-6))) for cs in candidates_s]
            seen = set()
            for cs in candidates_s:
                cs_round = round(cs, 4)
                if cs_round in seen:
                    continue
                seen.add(cs_round)
                trial = y_fw + cs * direction
                trial = np.maximum(trial, 0.0)
                s_t = trial.sum()
                if s_t < 1e-15:
                    continue
                trial = trial / s_t
                M_trial = fast_convolve(trial).max()
                if M_trial < best_trial_M:
                    best_trial_M = M_trial
                    best_s = cs

            y = y_fw + best_s * direction
        else:
            y = y_fw

        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = best_a.copy()
        else:
            y = y / s_y

        if iter_count <= 2:
            y = g_dir.copy()

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            M_now = fast_convolve(y).max()
            print("    2-WF iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + " cur=" + repr(2.0 * n * M_now) + " |S|=" + str(len(active_set)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    2-WF done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + " iters=" + str(iter_count))
    return best_a, best_m


def mirror_descent_multi(a0, time_limit, top_k=30, verbose=False):
    """Mirror descent with multi-activeset subgradient in log-space."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_convolve(y).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)

    f_log = np.log(np.maximum(y, 1e-30))
    best_f_log = f_log.copy()
    momentum = 0.7
    vel = np.zeros(n)
    eta0 = 0.08
    stall = 0
    max_stall = 5000
    iter_count = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break

        iter_count += 1
        eta = eta0 / (np.sqrt(t) + 1.0)

        # Nesterov lookahead
        f_look = f_log + momentum * vel
        a_look = np.exp(f_look - f_look.max())
        a_look = np.maximum(a_look, 1e-30)
        s_look = a_look.sum()
        a_look = a_look / s_look

        # Multi-activeset subgradient
        ac = fast_convolve(a_look)
        thresh = ac.max() - 0.1 * (ac.max() - np.median(ac))
        active_idx = np.where(ac > thresh)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(ac)])
        if len(active_idx) > top_k:
            active_idx = active_idx[np.argsort(ac[active_idx])[::-1][:top_k]]

        # Inverse-gaussian weights on active constraints
        weights = np.exp(-10.0 * (ac.max() - ac[active_idx]) / (ac.max() + 1e-10))
        weights = weights / weights.sum()

        grad = compute_multi_subgradient(a_look, active_idx, weights, n)

        # Small random perturbation
        perturb = 0.005 / (1.0 + t / 1000.0)
        grad += perturb * rng.standard_normal(n) * a_look

        # Mirror descent update in log-space
        f_new = f_look - eta * grad
        a_new = np.exp(f_new - f_new.max())
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new
        f_log = np.log(a_new)

        vel = momentum * vel + (f_log - f_look)
        f_log = f_look - momentum * vel  # Actually just use f_new
        # Simpler: just update directly
        f_log = f_new
        vel = momentum * vel + (f_log - (f_log + eta * grad))  # simplified
        # Reset velocity update
        vel = momentum * vel  # decay only

        M_new = fast_convolve(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
            best_f_log = f_log.copy()
            stall = 0
        else:
            stall += 1

        if stall > 500 and stall % 500 == 0:
            f_log = best_f_log + rng.normal(0, 0.05, n)
            a_test = np.exp(f_log - f_log.max())
            a_test = np.maximum(a_test, 1e-30)
            f_log = np.log(a_test / a_test.sum())
            vel = np.zeros(n)
            eta0 = eta0 * 1.15

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print("    Mirror t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    Mirror done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=77, verbose=False):
    """Coordinate descent with local optimization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(40, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find target with minimal impact
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < max(10, n // 10):
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])

            if dst == src:
                continue

            found = False
            for frac in [0.003, 0.008, 0.015, 0.03, 0.06]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and a[src] > 1e-10:
                j = int(rng.integers(n))
                if j != src:
                    frac = rng.uniform(0.003, 0.012)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0:
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + " stall=" + str(stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) + ")")
    return best_seq, best_score


def get_initial_sequence(t_global):
    """Get the best starting sequence from parent or fallback."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
                return p_norm, sc
    except Exception:
        pass

    n = 6000
    x = np.linspace(0, 1, n)
    best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
    best_score = evaluate_safe(best_seq)
    print("Default init: n=" + str(n) + ", score=" + repr(best_score))
    return best_seq, best_score


def diverse_init(n, seed=0):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    inits = []

    # Sine squared
    inits.append(norm_sum1(np.sin(np.pi * x) ** 2))
    # Cosine bell
    inits.append(norm_sum1(np.cos(np.pi * (x - 0.5)) ** 4))
    # Two bumps
    inits.append(norm_sum1(np.exp(-20*(x-0.3)**2) + np.exp(-20*(x-0.7)**2)))
    # Gaussian
    inits.append(norm_sum1(np.exp(-8*(x-0.5)**2)))
    # Triangular
    inits.append(norm_sum1(1.0 - np.abs(x - 0.5) * 2))
    # Cubic bump
    inits.append(norm_sum1(np.maximum(0, 1.0 - (x - 0.5)**6)))
    # Random smooth
    f = rng.normal(0, 1, n)
    from scipy.ndimage import gaussian_filter1d
    f_smooth = np.exp(gaussian_filter1d(f, sigma=10))
    inits.append(norm_sum1(f_smooth))
    # Power law shape
    inits.append(norm_sum1((1.0 - np.abs(x - 0.5) * 2) ** 1.5))

    return inits


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = get_solver()
    best_seq, best_score = get_initial_sequence(t_global)
    n0 = len(best_seq)

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: Multi-start initialization ===
    print("\n=== Phase 1: Multi-start (n=" + str(n0) + ") ===")
    t1 = min(remaining_time() * 0.08, 100.0)
    if t1 > 20:
        inits = diverse_init(n0, seed=42)
        # Evaluate all inits, pick best
        for ii, init in enumerate(inits):
            sc = evaluate_safe(init)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = init.copy()
                print("  Init " + str(ii) + ": " + repr(sc))

    # === Phase 2: 2-WF on best ===
    print("\n=== Phase 2: 2-WF initial (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining_time() * 0.20, 300.0)
    if t2 > 50 and len(best_seq) <= 10000:
        a_out, _ = lp_fw_2wf(norm_sum1(best_seq), t2, solver,
                             initial_K=500, max_K=4000, add_per_iter=60, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "2wf1_n" + str(len(best_seq)), t_global)

    # === Phase 3: Mirror descent ===
    print("\n=== Phase 3: Mirror descent (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining_time() * 0.10, 150.0)
    if t3 > 30:
        a_out, _ = mirror_descent_multi(norm_sum1(best_seq), t3, top_k=40, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "mirror_n" + str(len(best_seq)), t_global)

    # === Phase 4: 2-WF second pass ===
    print("\n=== Phase 4: 2-WF second (n=" + str(len(best_seq)) + ") ===")
    t4 = min(remaining_time() * 0.20, 300.0)
    if t4 > 50 and len(best_seq) <= 10000:
        a_out, _ = lp_fw_2wf(norm_sum1(best_seq), t4, solver,
                             initial_K=600, max_K=5000, add_per_iter=80, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "2wf2_n" + str(len(best_seq)), t_global)

    # === Phase 5: Multi-n exploration ===
    print("\n=== Phase 5: Multi-n ===")
    n_best = len(best_seq)
    try_ns = sorted(set([n_best - 500, n_best + 500, n_best + 1000, n_best - 1000,
                         max(3000, n_best - 2000), min(12000, n_best + 2000)]))
    try_ns = [l for l in try_ns if 3000 <= l <= 10000]
    try_ns.sort(key=lambda l: abs(l - n_best))

    for new_n in try_ns[:4]:
        rt = remaining_time()
        if rt < 100:
            break
        print("\n  Multi-n: n=" + str(new_n) + " (rt=" + str(round(rt, 1)) + "s)")
        a_long = interpolate_seq(best_seq, new_n)

        rt = remaining_time()
        if rt > 60 and new_n <= 10000:
            t5 = min(rt * 0.15, 150.0)
            a_out, _ = lp_fw_2wf(a_long, t5, solver,
                                 initial_K=400, max_K=3000, add_per_iter=40, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "2wf_n" + str(new_n), t_global)

        rt = remaining_time()
        if rt > 40:
            t5b = min(rt * 0.08, 60.0)
            if t5b > 15:
                a_out, _ = mirror_descent_multi(norm_sum1(best_seq), t5b, top_k=30, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "mirror_n" + str(new_n), t_global)

    # === Phase 6: 2-WF third pass (long) ===
    print("\n=== Phase 6: 2-WF third (n=" + str(len(best_seq)) + ") ===")
    t6 = min(remaining_time() * 0.22, 350.0)
    if t6 > 50 and len(best_seq) <= 10000:
        a_out, _ = lp_fw_2wf(norm_sum1(best_seq), t6, solver,
                             initial_K=800, max_K=5000, add_per_iter=100, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "2wf3_n" + str(len(best_seq)), t_global)

    # === Phase 7: Mirror descent round 2 ===
    print("\n=== Phase 7: Mirror round 2 (n=" + str(len(best_seq)) + ") ===")
    t7 = min(remaining_time() * 0.08, 100.0)
    if t7 > 25:
        a_out, _ = mirror_descent_multi(norm_sum1(best_seq), t7, top_k=50, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "mirror2_n" + str(len(best_seq)), t_global)

    # === Phase 8: Coordinate polish ===
    print("\n=== Phase 8: Coordinate polish (n=" + str(len(best_seq)) + ") ===")
    t8 = min(remaining_time() * 0.10, 120.0)
    if t8 > 20:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t8, rng_seed=99, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_n" + str(len(best_seq)), t_global)

    # === Phase 9: Final 2-WF ===
    print("\n=== Phase 9: Final 2-WF (n=" + str(len(best_seq)) + ") ===")
    t9 = min(remaining_time() * 0.20, 200.0)
    if t9 > 30 and len(best_seq) <= 10000:
        a_out, _ = lp_fw_2wf(norm_sum1(best_seq), t9, solver,
                             initial_K=1000, max_K=6000, add_per_iter=120, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "2wf_final", t_global)

    # === Final coordinate polish ===
    rt = remaining_time()
    if rt > 25:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), min(rt * 0.7, 80.0), rng_seed=2024, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
