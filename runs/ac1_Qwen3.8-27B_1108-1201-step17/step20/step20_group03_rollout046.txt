def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def top_k_gradient(a, k):
    """Compute gradient of sum of top-k autocorrelation values."""
    n = len(a)
    b = fast_autocorr(a)
    top_idx = np.argsort(b)[::-1][:k]
    grad = np.zeros(n)
    for kk in top_idx:
        kk = int(kk)
        idx = kk - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        grad[valid] += 2.0 * a[idx[valid]]
    max_val = b[n - 1]  # lag 0 is always the max for non-negative
    return grad, b


def fista_optimize(a0, time_limit, top_k=8, beta=0.3, restart_every=3000,
                   verbose=True):
    """FISTA with top-k gradient and backtracking line search."""
    n = len(a0)
    x = norm_sum1(a0).copy()
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(42)
    step = 1.0 / (2.0 * np.sqrt(n))
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        grad, b = top_k_gradient(y, top_k)
        
        # Backtracking line search
        eta = step
        for _ls in range(30):
            z = y - eta * grad
            x_new = simplex_project(z)
            b_new = fast_autocorr(x_new)
            m_new = b_new.max()
            # Check descent
            if m_new < best_m + 1e-14 or eta < 1e-10:
                break
            eta *= 0.7
        
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x.copy()
        x = x_new
        t_nest = t_new
        
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_level = 0.02 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise_level, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            step = 1.0 / (2.0 * np.sqrt(n))
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    FISTA t={:.0f} ({:.1f}s): J={:.6f}".format(t, elapsed, J))
    
    return best_a, best_m


def lp_fw_optimize(a0, time_limit, solver, initial_K=300, max_K=3000,
                   add_per_iter=20, verbose=True):
    """LP-Frank-Wolfe with adaptive active set."""
    n = len(a0)
    if n > 12000:
        return norm_sum1(a0), fast_autocorr(norm_sum1(a0)).max()
    
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 1500
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:max(50, initial_K // 2)])
            no_improve += 50
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 10
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Add violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))
        
        # Line search for step size
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5),
                  max(0.02, default_alpha * 1.5), 1.0 / (t + 1), 0.1, 0.05]
        alphas = sorted(set([max(0.01, min(1.0, a)) for a in alphas]))
        
        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y
        
        y = y_new
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter={:.0f} ({:.1f}s): J={:.6f} |S|={}".format(
                t, elapsed, J, len(active_set)))
    
    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=True):
    """Coordinate descent that directly reduces peak autocorrelation contributions."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:50]
        improved = False
        
        for _iter in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            
            k_int = int(top_peaks[rng.integers(len(top_peaks))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            
            # Find a good destination: low autocorrelation contribution
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:20]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src or a[src] < 1e-12:
                continue
            
            frac = float(rng.choice([0.002, 0.005, 0.01, 0.02, 0.05]))
            amt = a[src] * frac * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-14:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                ac = fast_autocorr(a)
                top_peaks = np.argsort(ac)[::-1][:50]
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.001, 0.01))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakShave ({:.1f}s): J={:.6f} stall={}".format(elapsed, J, stall))
    
    return best_a, best_m


def cyclic_shift_best(a0):
    """Find the best cyclic shift."""
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for s in range(0, n, max(1, n // 200)):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {:.6f} (n={}, t={:.1f}s)".format(
            label, sc, len(a), elapsed))
    return best_seq, best_score


def smoothed_fista(a0, time_limit, top_k=5, restart_every=2000, verbose=True):
    """FISTA using softmax-smoothed max for better gradient quality."""
    n = len(a0)
    x = norm_sum1(a0).copy()
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(7)
    
    beta = 50.0
    step = 1.0 / (4.0 * np.sqrt(n))
    beta_growth = 1.005
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        # Compute smoothed gradient
        b = fast_autocorr(y)
        b_shifted = b - b.max()
        b_shifted = np.clip(b_shifted, -500, 0)
        weights = np.exp(b_shifted * beta)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.zeros_like(b)
            weights[b == b.max()] = 1.0 / max(1, (b == b.max()).sum())
        else:
            weights = weights / w_sum
        
        # Gradient = sum_k weights[k] * grad(b[k])
        grad = np.zeros(n)
        top_w = np.argsort(weights)[::-1][:top_k + 5]
        for kk in top_w:
            w_k = weights[kk]
            if w_k < 1e-10:
                break
            kk_int = int(kk)
            idx = kk_int - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            grad[valid] += 2.0 * w_k * y[idx[valid]]
        
        # Gradient step with projection
        z = y - step * grad
        x_new = simplex_project(z)
        
        m_new = fast_autocorr(x_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x.copy()
        x = x_new
        t_nest = t_new
        
        if t % 200 == 0:
            beta = min(beta * beta_growth, 500.0)
        
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_level = 0.015 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise_level, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = 30.0
            step = 1.0 / (4.0 * np.sqrt(n))
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SmoothFISTA t={:.0f} ({:.1f}s): J={:.6f} beta={:.0f}".format(
                t, elapsed, J, beta))
    
    return best_a, best_m

            a_out, _ = smoothed_fista(a_long, t6a, top_k=5,
                                       restart_every=2000, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "long1", t_global)
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
            best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_long", t_global)
    
    # === Phase 7: LP-FW burst on best sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: LP-FW burst (n={}) ===".format(n_best))
    if remaining > 60 and n_best <= 10000:
        t7 = min(remaining * 0.20, 180.0)
        if t7 > 30:
            a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t7, solver,
                                       initial_K=500, max_K=3500,
                                       add_per_iter=30, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp2", t_global)
    
    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final Polish (n={}) ===".format(n_best))
    
    # Peak shaving
    t8a = min(remaining * 0.25, 100.0)
    if t8a > 20:
        a_out, _ = peak_shaving(norm_sum1(best_seq), t8a, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_final", t_global)
    
    # Cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_final", t_global)
    
    # Final micro FISTA
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t9 = min(remaining * 0.50, 100.0)
        if t9 > 20:
            a_out, _ = smoothed_fista(norm_sum1(best_seq), t9, top_k=3,
                                       restart_every=1000, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_final", t_global)
    
    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.6f}, n={}, time={:.1f}s".format(
            best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 8000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
    
    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
