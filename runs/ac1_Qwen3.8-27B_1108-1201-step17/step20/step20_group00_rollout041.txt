"""
Search for non-negative step-function heights minimizing
    2*n*max(convolve(a,a))/(sum(a))**2.

The program starts from the known parent construction if available and then
uses a combination of:
  1. multiresolution/shape exploration,
  2. an approximate primal-dual Frank-Wolfe/LP method for peak autocorrelation,
  3. a self-adaptive mirror-descent/softmax peak-deflation method,
  4. exact coordinate polish moves,
  5. low-resolution global reheating and interpolation back to the working length.

All internal optimization uses FFT autocorrelation.  The returned sequence is
kept non-negative and normalized to sum 1, which is safe for the evaluator.
"""

import time
import numpy as np


def _pow2_next(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        if a.size == 0:
            return np.ones(1, dtype=np.float64)
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def _autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    need = 2 * n - 1
    size = _pow2_next(need)
    P = np.fft.rfft(a, size)
    c = np.fft.irfft(P * np.conj(P), size)[:need]
    np.maximum(c, 0.0, out=c)
    return c


def _obj(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    return float(2.0 * a.size * _autocorr(a).max())


def _interp(a, m):
    a = np.asarray(a, dtype=np.float64)
    old_n = int(a.size)
    m = int(m)
    if old_n <= 0:
        return np.ones(m, dtype=np.float64) / float(m)
    if old_n == m:
        return _normalize(a).copy()
    f = _normalize(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, m)
    y = np.interp(new_x, old_x, f)
    y = np.maximum(y, 0.0)
    s = float(y.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(m, dtype=np.float64) / float(m)
    return y / s


def _param_seq(n, kind=0, seed=0):
    n = int(n)
    x = np.linspace(0.0, 1.0, n)
    if kind == 0:
        y = np.sin(np.pi * x) ** 2
    elif kind == 1:
        y = 1.0 + 0.6 * np.cos(2.0 * np.pi * x)
    elif kind == 2:
        y = 1.0 + 0.8 * np.cos(4.0 * np.pi * x)
    elif kind == 3:
        y = (
            np.exp(-((x - 0.5) ** 2) / (2 * 0.15 * 0.15))
            + 0.6 * np.exp(-((x - 0.2) ** 2) / (2 * 0.08 * 0.08))
            + 0.6 * np.exp(-((x - 0.8) ** 2) / (2 * 0.08 * 0.08))
        )
    elif kind == 4:
        y = 1.0 + 0.35 * np.sin(10.0 * np.pi * x)
    else:
        rng = np.random.default_rng(int(seed))
        y = rng.uniform(0.0, 1.0, n)
    y = np.maximum(y, 1e-6)
    return _normalize(y)

def _anti_peak(a0, time_limit, seed=0, beta0=10.0, beta1=300.0, mom=0.8,
               verbose=False):
    p = _normalize(a0)
    n = int(p.size)
    if n < 2:
        return p.copy(), _obj(p)
    t0 = time.monotonic()
    c = _autocorr(p)
    best = p.copy()
    best_obj = float(2.0 * n * c.max())
    rng = np.random.default_rng(int(seed))
    m = np.zeros(n, dtype=np.float64)
    stall = 0
    it = 0
    L = _pow2_next(2 * n - 1)
    factors = (0.05, 0.15, 0.4, 1.0, 2.5)

    while time.monotonic() - t0 < time_limit:
        c = _autocorr(p)
        cmax = float(c.max())
        cur_obj = 2.0 * n * cmax
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best = p.copy()
            stall = 0
        else:
            pass

        it += 1
        beta = beta0 + (beta1 - beta0) * min(1.0, it / 8000.0)
        if stall > 100:
            beta = min(1500.0, beta * 1.25)

        qn = n * c
        qmax = float(qn.max())
        if qmax <= 1e-15:
            break
        ex = beta * (qn - qmax)
        w = np.exp(ex)
        ws = float(w.sum())
        if not np.isfinite(ws) or ws <= 1e-300:
            K = min(1000, len(qn))
            w = np.zeros(len(qn), dtype=np.float64)
            if K > 0:
                idx = np.argpartition(qn, K - 1)[-K:]
                w[idx] = 1.0 / float(K)
        else:
            w = w / ws

        P = np.fft.fft(p, L)
        W = np.fft.fft(w, L)
        grad = 2.0 * np.real(np.fft.ifft(W * np.conj(P)))[:n]

        if not np.all(np.isfinite(grad)):
            break
        g = grad - grad.min()
        gmax = float(g.max())
        if gmax <= 1e-15:
            break
        g = g / gmax

        m = mom * m + (1.0 - mom) * g
        m = m - m.min()
        mmax = float(m.max())
        if mmax <= 1e-15:
            break
        m = m / mmax

        best_trial = None
        best_trial_c = cmax
        for f in factors:
            logp = np.log(p + 1e-30)
            logp = logp - f * m
            logp = logp - logp.max()
            tr = np.exp(logp)
            s = float(tr.sum())
            if not np.isfinite(s) or s <= 1e-300:
                continue
            tr = tr / s
            cm = float(_autocorr(tr).max())
            if cm < best_trial_c - 1e-15:
                best_trial_c = cm
                best_trial = tr

        if best_trial is not None:
            p = best_trial
            obj = float(2.0 * n * best_trial_c)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best = p.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                scale = 0.04 * (1.0 + min(1.0, stall / 2000.0))
                p = _normalize(best * np.exp(rng.normal(scale=scale, size=n)))
                m = 0.5 * m
                stall = 100

        if verbose and it % 200 == 0:
            print("    anti_peak it=" + str(it) +
                  " t=" + str(round(time.monotonic() - t0, 1)) +
                  " cur=" + repr(cur_obj) +
                  " best=" + repr(best_obj))

    best_obj = float(2.0 * n * _autocorr(best).max())
    return best, best_obj


def _lp_mw(a0, time_limit, seed=0, active_frac=0.35, k_cap=6000,
           dual_iters=14, bottom_count=100, gamma=30.0, mix=0.08,
           verbose=False):
    p = _normalize(a0)
    n = int(p.size)
    if n < 2:
        return p.copy(), _obj(p)

    t0 = time.monotonic()
    c = _autocorr(p)
    best = p.copy()
    best_obj = float(2.0 * n * c.max())
    rng = np.random.default_rng(int(seed))
    stall = 0
    it = 0

    Lact = 2 * n - 1
    L = _pow2_next(Lact)
    S = _pow2_next(3 * n - 2)
    n_idx = np.arange(n)
    active_size = int(min(k_cap, max(500, int(active_frac * Lact))))

    while time.monotonic() - t0 < time_limit:
        P_L = np.fft.rfft(p, L)
        c = np.fft.irfft(P_L * np.conj(P_L), L)[:Lact]
        np.maximum(c, 0.0, out=c)
        cmax = float(c.max())
        cur_obj = 2.0 * n * cmax
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best = p.copy()
            stall = 0

        it += 1
        qn = n * c

        if active_size >= Lact:
            active = np.arange(Lact, dtype=np.int64)
        else:
            active = np.argpartition(qn, active_size - 1)[-active_size:]
            active = active.astype(np.int64, copy=False)
        K = int(active.size)

        P_S = np.fft.rfft(p, S)
        loglam = np.zeros(K, dtype=np.float64)
        rev_pos = Lact - 1 - active
        wR_s = np.zeros(S, dtype=np.float64)
        b = np.zeros(n, dtype=np.float64)
        prev_bottom = None
        q = None

        for du in range(int(max(1, dual_iters))):
            lm = loglam - loglam.max()
            wa = np.exp(lm)
            sw = float(wa.sum())
            if not np.isfinite(sw) or sw <= 1e-300:
                wa = np.ones(K, dtype=np.float64) / float(K)
            else:
                wa = wa / sw

            if du > 0:
                wR_s[rev_pos] = 0.0
            wR_s[rev_pos] = wa

            Wr = np.fft.rfft(wR_s, S)
            conv = np.fft.irfft(P_S * Wr, S)
            q = conv[Lact - 1 - n_idx]
            np.maximum(q, 0.0, out=q)

            B = int(min(bottom_count, n))
            if B >= n:
                bottom = n_idx
            else:
                part = np.argpartition(q, B)[:B]
                bottom = part[np.argsort(q[part])][:B]
            bcnt = max(1, int(bottom.size))

            if du > 0 and prev_bottom is not None:
                b[prev_bottom] = 0.0
            b[bottom] = 1.0 / float(bcnt)
            prev_bottom = bottom.copy()

            Bh = np.fft.rfft(b, L)
            h = np.fft.irfft(P_L * Bh, L)[:Lact]
            np.maximum(h, 0.0, out=h)
            h_act = h[active]
            hmax = float(h_act.max())
            if hmax > 1e-15:
                step = 0.5 * h_act / hmax
                loglam = loglam + step

        if q is None or len(q) != n:
            q = np.ones(n, dtype=np.float64)

        qmin = float(q.min())
        qrx = float(q.max())
        den = qrx - qmin
        if den > 1e-15:
            gx = np.exp(-gamma * (q - qmin) / den)
        else:
            gx = np.ones(n, dtype=np.float64)
        gx = gx + 1e-12
        gx = (1.0 - mix) * gx + mix * (1.0 / float(n))
        gs = float(gx.sum())
        if not np.isfinite(gs) or gs <= 1e-300:
            g_dir = np.ones(n, dtype=np.float64) / float(n)
        else:
            g_dir = gx / gs

        best_trial = None
        best_trial_c = cmax
        for al in (0.02, 0.05, 0.1, 0.2, 0.4, 0.7):
            tr = (1.0 - al) * p + al * g_dir
            tr = np.maximum(tr, 0.0)
            s = float(tr.sum())
            if s <= 1e-300:
                continue
            tr = tr / s
            cm = float(_autocorr(tr).max())
            if cm < best_trial_c - 1e-15:
                best_trial_c = cm
                best_trial = tr

        if best_trial is not None:
            p = best_trial
            obj = float(2.0 * n * best_trial_c)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best = p.copy()
            stall = 0
        else:
            stall += 1
            if stall > 120:
                active_size = min(active_size + 500, int(k_cap))
                gamma = min(gamma * 1.2, 90.0)
                dual_iters = min(int(dual_iters) + 2, 30)
                mix = max(0.03, mix * 0.9)
            if stall > 350:
                p = _normalize(best * np.exp(rng.normal(scale=0.035, size=n)))
                stall = 120

        if verbose and it % 40 == 0:
            print("    lp_mw it=" + str(it) +
                  " t=" + str(round(time.monotonic() - t0, 1)) +
                  " cur=" + repr(cur_obj) +
                  " best=" + repr(best_obj) +
                  " active=" + str(K))

    best_obj = float(2.0 * n * _autocorr(best).max())
    return best, best_obj


def _coord_polish(a0, time_limit, seed=0, active_frac=0.35, beta=100.0,
                  top_c=2, fracs=(0.004, 0.01, 0.02), verbose=False):
    p = _normalize(a0)
    n = int(p.size)
    if n < 2:
        return p.copy(), _obj(p)

    t0 = time.monotonic()
    c = _autocorr(p)
    best = p.copy()
    best_obj = float(2.0 * n * c.max())
    rng = np.random.default_rng(int(seed))
    Lact = 2 * n - 1
    L = _pow2_next(Lact)
    top_c = int(max(1, top_c))
    fracs = tuple(float(x) for x in fracs)
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        c = _autocorr(p)
        cmax = float(c.max())
        cur_obj = 2.0 * n * cmax
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best = p.copy()
            stall = 0

        it += 1
        qn = n * c
        K = int(min(Lact, max(500, int(active_frac * Lact))))
        if K >= Lact:
            active = np.arange(Lact, dtype=np.int64)
        else:
            active = np.argpartition(qn, K - 1)[-K:]

        w = np.zeros(Lact, dtype=np.float64)
        vals = qn[active]
        vmax = float(vals.max())
        wa = np.exp(beta * (vals - vmax))
        ws = float(wa.sum())
        if ws <= 1e-300 or not np.isfinite(ws):
            wa = np.ones(K, dtype=np.float64) / float(K)
        else:
            wa = wa / ws
        w[active] = wa

        P = np.fft.fft(p, L)
        W = np.fft.fft(w, L)
        grad = 2.0 * np.real(np.fft.ifft(W * np.conj(P)))[:n]
        if not np.all(np.isfinite(grad)):
            break

        if n > top_c:
            src_part = np.argpartition(grad, -top_c)[-top_c:]
            dst_part = np.argpartition(grad, top_c)[:top_c]
        else:
            src_part = np.arange(n)
            dst_part = np.arange(n)

        srcs = src_part[np.argsort(grad[src_part])[::-1]][:top_c]
        dsts = dst_part[np.argsort(grad[dst_part])][:top_c]

        improved = False
        for src in srcs:
            src = int(src)
            sv = float(p[src])
            if sv <= 1e-18:
                continue
            for dst in dsts:
                dst = int(dst)
                if dst == src:
                    continue
                if not np.isfinite(grad[src]) or not np.isfinite(grad[dst]):
                    continue
                if grad[src] <= grad[dst]:
                    continue
                for frac in fracs:
                    d = sv * frac
                    if d <= 1e-18 or d >= sv:
                        continue
                    tr = p.copy()
                    tr[src] -= d
                    tr[dst] += d
                    s = float(tr.sum())
                    if s <= 1e-300:
                        continue
                    tr = tr / s
                    cm = float(_autocorr(tr).max())
                    if cm < cmax - 1e-15:
                        p = tr
                        cmax = cm
                        obj = 2.0 * n * cm
                        if obj < best_obj - 1e-12:
                            best_obj = obj
                            best = p.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and len(srcs) > 0 and len(dsts) > 0:
            for _ in range(8):
                src = int(srcs[int(rng.integers(len(srcs)))])
                dst = int(dsts[int(rng.integers(len(dsts)))])
                if src == dst or p[src] <= 1e-18:
                    continue
                frac = float(rng.uniform(0.002, 0.02))
                d = float(p[src]) * frac
                tr = p.copy()
                tr[src] -= d
                tr[dst] += d
                s = float(tr.sum())
                if s <= 1e-300:
                    continue
                tr = tr / s
                cm = float(_autocorr(tr).max())
                if cm < cmax - 1e-15:
                    p = tr
                    cmax = cm
                    obj = 2.0 * n * cm
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best = p.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 250:
                break

        if verbose and it % 200 == 0:
            print("    coord it=" + str(it) +
                  " t=" + str(round(time.monotonic() - t0, 1)) +
                  " cur=" + repr(cur_obj) +
                  " best=" + repr(best_obj) +
                  " stall=" + str(stall))

    best_obj = float(2.0 * n * _autocorr(best).max())
    return best, best_obj


def _lowres_search(parent, n_lr, time_total, target_n, seed=2024):

        return None, float("inf")
    t0 = time.monotonic()

    for kind in (0, 1, 3, 4):
        starts.append(_param_seq(n_lr, kind, seed=seed + kind * 31))
    starts.append(_param_seq(n_lr, 5, seed=seed + 999))

    best = None
    best_obj = float("inf")
    alloc = min(30.0, time_total / max(1, len(starts)))

    for i, s in enumerate(starts):
        rem = time_total - (time.monotonic() - t0)
        if rem < 12:
            break
        out, _ = _anti_peak(s, max(5.0, min(alloc, rem - 8)),
                            seed=seed + 17 * (i + 1),

        lifted = _interp(out, target_n)
        obj = float(_obj(lifted))

            best_obj = obj
            best = lifted.copy()
            print("    lowres cand=" + str(i) +
                  " obj=" + repr(best_obj))

    if best is not None:
        best = _normalize(best)
        best_obj = float(_obj(best))
    return best, best_obj


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    try:
        bud = float(budget_s)
    except Exception:
        bud = 1800.0
    if not np.isfinite(bud) or bud <= 0.0:
        bud = 1800.0
    TIME_LIMIT = min(bud - 25.0, 1765.0)
    if TIME_LIMIT < 30.0:
        TIME_LIMIT = max(10.0, bud - 2.0)

    parent = _load_parent()
    if parent is None:
        parent = _param_seq(13700, 0)
        parent_norm = _normalize(parent)
        parent_official = float("inf")
    else:
        parent_norm = _normalize(parent)
        parent_official = float("inf")
        if time.monotonic() - t_start < 120.0 and parent_norm.size <= 20000:
            parent_official = _eval_safe(parent_norm)
            print("Parent official=" + repr(parent_official))

    n_parent = int(parent_norm.size)
    max_n = int(min(max(16384, n_parent), 18000))

    vals = (8000, int(0.75 * n_parent), n_parent, int(1.2 * n_parent), max_n)
    n_list = []
    for vv in vals:
        vv = int(vv)
        if 2000 <= vv <= max_n:
            n_list.append(vv)
    n_list = sorted(set(n_list))

    best_start = None
    best_obj = float("inf")
    for n in n_list:
        a = _interp(parent_norm, n)
        obj = float(_obj(a))
        if np.isfinite(obj) and obj < best_obj:
            best_obj = obj
            best_start = a
            print("Quick length n=" + str(n) + " obj=" + repr(obj))

    if best_start is None:
        best_start = _parent if False else parent_norm.copy()
        best_obj = float(_obj(best_start))

    a_cur = best_start.copy()
    obj_cur = best_obj

    print("Initial best: obj=" + repr(obj_cur) + " n=" + str(a_cur.size))

    if time.monotonic() - t_start < 180.0:
        lr_time = min(90.0, TIME_LIMIT - (time.monotonic() - t_start) - 120.0)
        if lr_time > 30.0:
            lr_best, lr_obj = _lowres_search(parent_norm, 2048, lr_time,
                                             int(a_cur.size), seed=7114)
            if lr_best is not None and np.isfinite(lr_obj) and lr_obj < best_obj - 1e-8:
                best_obj = lr_obj
                best_start = lr_best.copy()
                a_cur = best_start.copy()
                obj_cur = best_obj
                print("Lowres lift improved: obj=" + repr(best_obj) +
                      " n=" + str(int(a_cur.size)))

    def remaining():
        return TIME_LIMIT - (time.monotonic() - t_start)

    # Pass 1: primal-dual LP/FW-like optimizer on the current best.
    rem = remaining()
    if rem > 140:
        alloc = min(520.0, rem * 0.38, rem - 180.0)
        if alloc > 70.0:
            print("Phase LP-MW start")
            out, ob = _lp_mw(a_cur, alloc, seed=101, active_frac=0.35,
                             k_cap=6000, dual_iters=16, bottom_count=120,
                             gamma=30.0, mix=0.08, verbose=True)
            if np.isfinite(ob) and ob < obj_cur - 1e-9:
                obj_cur = ob
                a_cur = out.copy()
                best_start = a_cur.copy()
                best_obj = obj_cur
                print("LP-MW improved: obj=" + repr(obj_cur) +
                      " n=" + str(a_cur.size))

    # Pass 2: mirror-descent peak deflation.
    rem = remaining()
    if rem > 120:
        alloc = min(360.0, rem * 0.30, rem - 160.0)
        if alloc > 50.0:
            print("Phase Anti-peak start")
            out, ob = _anti_peak(a_cur, alloc, seed=202,
                                 beta0=10.0, beta1=300.0, mom=0.85,
                                 verbose=True)
            if np.isfinite(ob) and ob < obj_cur - 1e-9:
                obj_cur = ob
                a_cur = out.copy()
                best_start = a_cur.copy()
                best_obj = obj_cur
                print("Anti-peak improved: obj=" + repr(obj_cur) +
                      " n=" + str(a_cur.size))

    # Pass 3: coordinate polish.
    rem = remaining()
    if rem > 100:
        alloc = min(220.0, rem * 0.22, rem - 120.0)
        if alloc > 40.0:
            print("Phase Coord start")
            out, ob = _coord_polish(a_cur, alloc, seed=303,
                                    active_frac=0.35, beta=100.0,
                                    top_c=2,
                                    fracs=(0.004, 0.01, 0.02),
                                    verbose=True)
            if np.isfinite(ob) and ob < obj_cur - 1e-9:
                obj_cur = ob
                a_cur = out.copy()
                best_start = a_cur.copy()
                best_obj = obj_cur
                print("Coord improved: obj=" + repr(obj_cur) +
                      " n=" + str(a_cur.size))

    # Pass 4: if current length is below the allowed long length, try a long
    # fresh lift from both parent and current best.
    rem = remaining()
    if rem > 180 and int(a_cur.size) < max_n:
        alt_n = int(max_n)
        a_alt1 = _interp(parent_norm, alt_n)
        a_alt2 = _interp(a_cur, alt_n)
        o1 = float(_obj(a_alt1))
        o2 = float(_obj(a_alt2))
        a_alt = a_alt1 if o1 <= o2 else a_alt2
        ob_alt_base = min(o1, o2)
        print("Alt length base obj=" + repr(ob_alt_base) + " n=" + str(alt_n))
        alloc = min(520.0, rem * 0.38, rem - 200.0)
        if alloc > 70.0:
            print("Phase long LP-MW start")
            out, ob = _lp_mw(a_alt, alloc, seed=404, active_frac=0.40,
                             k_cap=6000, dual_iters=18, bottom_count=150,
                             gamma=40.0, mix=0.07, verbose=True)
            if np.isfinite(ob) and ob < obj_cur - 1e-9:
                obj_cur = ob
                a_cur = out.copy()
                best_start = a_cur.copy()
                best_obj = obj_cur
                print("Long LP-MW improved: obj=" + repr(obj_cur) +
                      " n=" + str(a_cur.size))
            elif np.isfinite(ob_alt_base) and ob_alt_base < obj_cur - 1e-9:
                obj_cur = ob_alt_base
                a_cur = a_alt.copy()
                best_start = a_cur.copy()
                best_obj = obj_cur
                print("Long raw improved: obj=" + repr(obj_cur) +
                      " n=" + str(a_cur.size))

    # Pass 5: sharp anti-peak on the current best.
    rem = remaining()
    if rem > 120:
        alloc = min(280.0, rem * 0.28, rem - 120.0)
        if alloc > 50.0:
            print("Phase sharp Anti-peak start")
            out, ob = _anti_peak(a_cur, alloc, seed=505,
                                 beta0=40.0, beta1=700.0, mom=0.88,
                                 verbose=True)
            if np.isfinite(ob) and ob < obj_cur - 1e-9:
                obj_cur = ob
                a_cur = out.copy()
                best_start = a_cur.copy()
                best_obj = obj_cur
                print("Sharp anti-peak improved: obj=" + repr(obj_cur) +
                      " n=" + str(a_cur.size))

    # Pass 6: blended global重新开始 and anti-peak at current length.
    rem = remaining()
    if rem > 160:
        base_n = int(a_cur.size)
        rng = np.random.default_rng(917)
        rand = _param_seq(base_n, 5, seed=int(rng.integers(10**9)))
        blend = _normalize(0.72 * a_cur + 0.28 * rand)
        alloc = min(300.0, rem * 0.30, rem - 140.0)
        if alloc > 60.0:
            print("Phase blend anti-peak start")
            out, ob = _anti_peak(blend, alloc, seed=606,
                                 beta0=8.0, beta1=350.0, mom=0.84,
                                 verbose=True)
            if np.isfinite(ob) and ob < obj_cur - 1e-9:
                obj_cur = ob
                a_cur = out.copy()
                best_start = a_cur.copy()
                best_obj = obj_cur
                print("Blend anti-peak improved: obj=" + repr(obj_cur) +
                      " n=" + str(a_cur.size))

    # Final short coordinate polish.
    rem = remaining()
    if rem > 80:
        alloc = min(180.0, rem - 40.0)
        if alloc > 30.0:
            print("Phase final coord start")
            out, ob = _coord_polish(a_cur, alloc, seed=707,
                                    active_frac=0.45, beta=160.0,
                                    top_c=2,
                                    fracs=(0.003, 0.008, 0.015),
                                    verbose=True)
            if np.isfinite(ob) and ob < obj_cur - 1e-9:
                obj_cur = ob
                a_cur = out.copy()
                best_start = a_cur.copy()
                best_obj = obj_cur
                print("Final coord improved: obj=" + repr(obj_cur) +
                      " n=" + str(a_cur.size))

    final = _normalize(a_cur if a_cur is not None else parent_norm)
    final_obj_internal = float(_obj(final))
    print("Final internal obj=" + repr(final_obj_internal) +
          " n=" + str(final.size))

    # Conservative official check only if enough time and length is moderate.
    rem = remaining()
    use_official_check = (
        parent_norm is not None and
        np.isfinite(parent_official) and
        rem > 80 and
        final.size <= 20000
    )
    if use_official_check:
        final_official = _eval_safe(final)
        print("Final official obj=" + repr(final_official))
        if np.isfinite(final_official) and final_official > parent_official + 1e-8:
            final = parent_norm
            final_obj_internal = parent_official
            print("Official fallback to parent.")
    elif np.isfinite(parent_official):
        print("Skipping official fallback check due to time/length.")

    result = np.asarray(final, dtype=np.float64).ravel()
    result = np.nan_to_num(result, nan=0.0, posinf=1000.0, neginf=0.0)
    result = np.clip(result, 0.0, 1000.0)
    if result.size == 0:
        result = np.ones(8000, dtype=np.float64) / 8000.0

    return [float(x) for x in result.tolist()]
