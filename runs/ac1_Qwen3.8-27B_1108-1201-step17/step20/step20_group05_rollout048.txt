"""
Multi-start, multi-length optimization for minimizing the self-convolution
concentration ratio. Uses L-BFGS with log-sum-exp smoothing, coordinate
descent, and random restarts across multiple sequence lengths.
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import minimize
    SCIPY_OK = True
except:
    SCIPY_OK = False

def next_pow2(x):
    x = int(x)
    if x <= 1: return 1
    return 1 << (x - 1).bit_length()

def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    np.maximum(arr, 0.0, out=arr)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    s = arr.sum()
    if s < 1e-15:
        return np.ones(len(arr), dtype=np.float64) / len(arr)
    return arr / s

def fft_conv(a):
    n = len(a)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return np.maximum(c, 0.0)

def objective_and_grad(a, beta, top_k=None):
    """Compute smoothed objective and gradient."""
    n = len(a)
    c = fft_conv(a)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_work = c[idx]
        # Build full gradient
        p = np.zeros(len(c))
        m = c_work.max()
        if m < 1e-30: m = 1e-30
        exp_vals = np.exp(beta * (c_work - m))
        p[idx] = exp_vals / exp_vals.sum()
    else:
        p = c.copy()
        m = p.max()
        if m < 1e-30: m = 1e-30
        exp_vals = np.exp(beta * (p - m))
        p = exp_vals / exp_vals.sum()
    
    # Gradient: 2 * (p * a) cross-correlated properly
    # grad_i = 2 * sum_k p_k * a_{k-i}
    # This is 2 * cross_correlate(p_reversed, a) at the right offset
    total = len(p) + n - 1  # = 3n - 2
    N2 = next_pow2(total)
    A = np.fft.rfft(a, N2)
    Pr = np.fft.rfft(p[::-1], N2)
    conv = np.fft.irfft(A * Pr, N2)[:total]
    grad = 2.0 * conv[n-1:2*n-1][::-1]
    
    f = m + math.log(exp_vals.sum()) / beta
    return f, grad

"""
Multi-start optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS with log-sum-exp smoothing, adaptive active sets, coordinate
descent with rank-1 convolution updates, and random restarts across
multiple sequence lengths to escape local minima.

import numpy as np
import time
import math

try:
    from scipy.optimize import minimize
    SCIPY_OK = True
except Exception:
    SCIPY_OK = False


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    np.maximum(arr, 0.0, out=arr)
    s = arr.sum()
    if s < 1e-15:
        return np.ones(len(arr), dtype=np.float64) / max(1, len(arr))
    return arr / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return np.where(np.isfinite(c) & (c > 0.0), c, 0.0)


def safe_eval(a):
    try:
        arr = np.asarray(a, dtype=np.float64)
        if arr.size == 0:
            return float('inf')
        arr = np.clip(arr, 0.0, 1000.0)
        if not np.all(np.isfinite(arr)):
            return float('inf')
        s = arr.sum()
        if s < 0.01:
            return float('inf')
        lst = [float(x) for x in arr]
        val = evaluate_sequence(lst)
        if val is None:
            return float('inf')
        val = float(val)
        if not math.isfinite(val):
            return float('inf')
        return val
    except Exception:
        return float('inf')


def lse_obj_grad(a, beta, top_k=None):
    """Log-sum-exp smoothed objective and gradient."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 1e100, np.array([], dtype=np.float64)
    c = fft_conv(a)
    size = 2 * n - 1

    if top_k is not None and top_k < size and top_k > 0:
        idx = np.argpartition(c, size - top_k)[size - top_k:]
        c_work = c[idx]
        m = float(c_work.max())
        if m < 1e-300:
            m = 1e-300
        shifted = beta * (c_work - m)
        shifted = np.clip(shifted, -500.0, 500.0)
        e = np.exp(shifted)
        se = float(e.sum())
        if se <= 0 or not math.isfinite(se):
            e = np.ones(top_k) / top_k
            se = 1.0
        p_work = e / se
        p = np.zeros(size, dtype=np.float64)
        p[idx] = p_work
        f = m + math.log(se) / beta
    else:
        m = float(c.max())
        if m < 1e-300:
            m = 1e-300
        shifted = beta * (c - m)
        shifted = np.clip(shifted, -500.0, 500.0)
        e = np.exp(shifted)
        se = float(e.sum())
        if se <= 0 or not math.isfinite(se):
            e = np.ones(size) / size
            se = 1.0
        p = e / se
        f = m + math.log(se) / beta

    # Gradient: grad_i = 2 * sum_k p_k * a_{k-i} (cross-correlation)
    total = size + n - 1
    N2 = next_pow2(total)
    A2 = np.fft.rfft(a, N2)
    Pr = np.fft.rfft(p[::-1], N2)
    conv = np.fft.irfft(A2 * Pr, N2)[:total]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)

    if not math.isfinite(f):
        f = 1e100
        grad = np.zeros(n, dtype=np.float64)
    return f, grad


def run_lbfegs(a0, beta, top_k, time_budget=100.0, verbose=True):
    """Run L-BFGS-B on the log-sum-exp smoothed objective."""
    a = norm_sum1(a0)
    n = a.size
    if n <= 1 or not SCIPY_OK:
        return a

    t0 = time.monotonic()
    maxfun = max(50, int(time_budget / 0.05))
    maxfun = min(maxfun, 2000)

    def obj_and_grad(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-15:
            return 1e100, np.zeros(n, dtype=np.float64)
        ax = ax / s
        f, g = lse_obj_grad(ax, beta, top_k)
        # Adjust gradient for the normalization constraint
        mean_g = float(np.dot(ax, g))
        g = g - mean_g  # project onto tangent space of simplex
        return f, g

    options = {
        'maxfun': maxfun,
        'maxiter': maxfun + 50,
        'ftol': 1e-14,
        'gtol': 1e-6,
        'disp': False
    }

    try:
        res = minimize(obj_and_grad, a, method='L-BFGS-B', jac=True, options=options)
        if res is not None and hasattr(res, 'x'):
            a_out = norm_sum1(res.x)
        else:
            a_out = a
    except Exception:
        a_out = a

    if verbose:
        elapsed = time.monotonic() - t0
        print("    L-BFGS beta={} top_k={} time={:.1f}s".format(beta, top_k, elapsed))

    return a_out


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, new_n), dtype=np.float64)
    new_n = int(max(2, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)


def apply_move(c, a, s, d, t):
    """Apply rank-1 convolution update for moving mass t from s to d."""
    n = a.size
    c_new = c.copy()
    # c_new[k] changes by: 2t*a[k-d] - 2t*a[k-s] + t^2*(delta(2d) - 2*delta(s+d) + delta(2s))
    lo_d = max(0, d)
    hi_d = min(2 * n - 1, d + n - 1)
    lo_s = max(0, s)
    hi_s = min(2 * n - 1, s + n - 1)

    # Add 2t * a shifted by d
    for i in range(n):
        k = d + i
        if 0 <= k < 2 * n - 1:
            c_new[k] += 2.0 * t * a[i]
        k = s + i
        if 0 <= k < 2 * n - 1:
            c_new[k] -= 2.0 * t * a[i]

    # Quadratic correction
    t2 = t * t
    if 2 * d < 2 * n - 1:
        c_new[2 * d] += t2
    if 2 * s < 2 * n - 1:
        c_new[2 * s] += t2
    if s + d < 2 * n - 1:
        c_new[s + d] -= 2.0 * t2

    return np.where(np.isfinite(c_new), c_new, 0.0)


def coord_descent(a0, time_budget=100.0, verbose=True):
    """Coordinate descent with rank-1 convolution updates."""
    t0 = time.monotonic()
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = float(c.max())
    best_a = a.copy()
    best_M = M
    stall = 0
    rnd = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_budget - 2 and stall < 80:
        rnd += 1
        if n <= 1:
            break

        # Identify top pressure points
        top_count = min(30, 2 * n - 1)
        top_idx = np.argpartition(c, 2 * n - 1 - top_count)[2 * n - 1 - top_count:]
        p_top = np.zeros(2 * n - 1, dtype=np.float64)
        p_top[top_idx] = 1.0 / top_count

        # Compute "pressure" at each position
        # pressure_i = sum_k p_k * a_{k-i}
        total = (2 * n - 1) + n - 1
        N2 = next_pow2(total)
        A2 = np.fft.rfft(a, N2)
        Pr = np.fft.rfft(p_top[::-1], N2)
        conv_p = np.fft.irfft(A2 * Pr, N2)[:total]
        pressure = conv_p[n - 1:2 * n - 1][::-1]
        pressure = np.where(np.isfinite(pressure), pressure, 0.0)

        # Find good source-destination pairs
        # Source: high pressure and high value (moving mass away reduces peak)
        # Dest: low pressure (adding mass there doesn't increase peak much)
        src_score = pressure * a
        dst_score = -pressure

        src_count = min(10, n)
        dst_count = min(15, n)
        src_cands = np.argpartition(src_score, n - src_count)[n - src_count:]
        dst_cands = np.argpartition(dst_score, n - dst_count)[n - dst_count:]

        best_improve = None
        best_dt = 0.0

        for s in src_cands:
            s = int(s)
            if a[s] < 1e-10:
                continue
            for d in dst_cands:
                d = int(d)
                if d == s:
                    continue
                # Try several fractions
                for frac in (0.01, 0.03, 0.08, 0.15, 0.3):
                    t = a[s] * frac
                    if t < 1e-14:
                        continue
                    t = min(t, a[s] * 0.95)
                    c_new = apply_move(c, a, s, d, t)
                    m_new = float(c_new.max())
                    if m_new < M - 1e-16:
                        improve = M - m_new
                        if best_improve is None or improve > best_improve:
                            best_improve = improve
                            best_dt = t
                            best_s, best_d = s, d

        if best_dt > 0:
            a[best_s] -= best_dt
            a[best_d] += best_dt
            np.maximum(a, 0.0, out=a)
            s_total = a.sum()
            if s_total < 1e-15:
                break
            a = a / s_total
            c = fft_conv(a)
            M = float(c.max())
            stall = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
        else:
            stall += 1
            # Random exploration
            if stall % 5 == 0:
                for _ in range(50):
                    s = int(rng.integers(n))
                    d = int(rng.integers(n))
                    if s == d or a[s] < 1e-10:
                        continue
                    frac = float(rng.uniform(0.005, 0.1))
                    t = a[s] * frac
                    t = min(t, a[s] * 0.9)
                    c_new = apply_move(c, a, s, d, t)
                    m_new = float(c_new.max())
                    if m_new < M - 1e-16:
                        a[s] -= t
                        a[d] += t
                        np.maximum(a, 0.0, out=a)
                        a = a / a.sum()
                        c = fft_conv(a)
                        M = float(c.max())
                        stall = 0
                        if M < best_M:
                            best_M = M
                            best_a = a.copy()
                        break

        if verbose and (rnd <= 3 or rnd % 30 == 0):
            print("    Coord rnd={} M={:.10f} best_M={:.10f}".format(rnd, M, best_M))

    if verbose:
        print("    Coord done: {} rounds, best_M={:.10f}".format(rnd, best_M))
    return best_a


def lbfegs_chain(a0, time_budget, verbose=True):
    """Run L-BFGS with increasing beta values."""
    a = norm_sum1(a0)
    n = a.size
    size = 2 * n - 1

    # Adaptive top-k schedule
    betas = [10, 30, 80, 200, 500, 1500, 5000]
    top_ks = [size, size, max(100, size // 5), max(50, size // 10),
              max(30, size // 20), max(20, size // 30), max(10, size // 50)]

    t0 = time.monotonic()
    per_beta = time_budget / len(betas)

    for i, (beta, tk) in enumerate(zip(betas, top_ks)):
        elapsed = time.monotonic() - t0
        remaining = time_budget - elapsed
        if remaining < 10:
            break
        bt = min(per_beta * (1 + i * 0.2), remaining - 5)
        if bt < 10:
            break
        a = run_lbfegs(a, beta, tk, time_budget=bt, verbose=verbose)

    return a


def perturb_seq(a, sigma, rng):
    """Add Gaussian perturbation and renormalize."""
    a = norm_sum1(a)
    n = a.size
    noise = rng.normal(0.0, sigma, n)
    a_new = a * (1.0 + noise)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

        if remaining < 20:
            break
        bt = min(per_len * 1.5, remaining - 10)
        if bt < 15:
            continue

        print("  Trying n={} (budget={:.0f}s)".format(target_n, bt))
        a_interp = interpolate_seq(best_seq, target_n)
        n = target_n
        size = 2 * n - 1
        # Quick L-BFGS
        a_tmp = run_lbfegs(a_interp, 100, max(30, size // 10), time_budget=bt * 0.6, verbose=False)
        a_tmp = coord_descent(a_tmp, bt * 0.4, verbose=False)
        try_update(a_tmp, "len_{}".format(target_n))

    # Phase 5: Final intensive polish on best
    t0 = time.monotonic()
    budget_p5 = max(30.0, total_budget - (time.monotonic() - t_start) - 20.0)
    budget_p5 = min(budget_p5, 200.0)
    print("\n=== Phase 5: Final polish (budget={:.0f}s) ===".format(budget_p5))

    if budget_p5 > 20:
        work = best_seq.copy()
        # L-BFGS with high beta
        n = work.size
        size = 2 * n - 1
        a_f = run_lbfegs(work, 3000, max(10, size // 50), time_budget=budget_p5 * 0.5, verbose=True)
        try_update(a_f, "final_lb")
        work = best_seq.copy()
        # Coordinate descent
        a_f = coord_descent(work, budget_p5 * 0.5, verbose=True)
        try_update(a_f, "final_coord")

    # Final validation
    final_val = safe_eval(best_seq)
    if math.isfinite(final_val):
        if final_val < best_score:
            best_score = final_val
    else:
        best_seq = get_initial_sequence()
        best_score = safe_eval(best_seq)

    arr = np.asarray(best_seq, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if arr.size == 0 or s < 0.01:
        arr = get_initial_sequence()
    else:
        arr = arr / s

    elapsed = time.monotonic() - t_start
    print("\nFINAL: score={:.10f} n={} elapsed={:.1f}s".format(safe_eval(arr), arr.size, elapsed))
    return [float(x) for x in arr]
