"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiple structured initializations (tilted Gaussians, power laws, parent)
2. Fast Frank-Wolfe for rapid initial convergence
3. Nesterov accelerated subgradient with adaptive top-k peak weighting
4. Targeted peak surgery via mass redistribution
5. Fine-grained pairwise coordinate descent
6. Multi-length exploration
Key insight: the optimal shape breaks left-right symmetry to reduce c[n-1].
"""

import numpy as np
import time

def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_subgrad(a, c, top_k=10, sharpness=50.0):
    """Compute subgradient of max(c) w.r.t. a using top_k peaks with softmax weighting."""
    n = len(a)
    size = 2 * n - 1
    
    # Find top_k peaks
    idx = np.argsort(c)[::-1][:top_k]
    c_vals = c[idx]
    c_max = c_vals[0]
    
    # Softmax weights (sharper = more focus on top peak)
    shifted = -sharpness * (c_max - c_vals)
    shifted = np.clip(shifted, -300, 300)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Build weight vector
    w_vec = np.zeros(size)
    w_vec[idx] = weights
    
    # Compute grad[j] = 2 * sum_k w_vec[k] * a[k-j] (with a[x]=0 outside [0,n-1])
    # = 2 * conv(a_rev, w_vec)[j + n - 1]
    fft_size = 1 << (size - 1).bit_length()
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w_vec, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    grad = 2.0 * conv_result[n-1:2*n-1]
    
    return grad, idx

def nesterov_subgrad(a0, time_limit, top_k=10, sharpness=50.0,
                      eta_base=0.5, restart_every=3000, verbose=True, seed=1):
    """Nesterov accelerated subgradient method on the simplex."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        # Compute subgradient
        c = fast_autocorr(y)
        c_max = c.max()
        
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        grad, active_idx = compute_subgrad(y, c, top_k=top_k, sharpness=sharpness)
        
        # Step size: adapt to gradient magnitude
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            eta = eta_base
        else:
            eta = eta_base / (g_norm * np.sqrt(t))
            eta = min(eta, 0.1)
        
        z = y - eta * grad
        x_new = simplex_project(z)
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        # Gradually increase sharpness (focus more on top peak)
        if t % 500 == 0:
            sharpness = min(sharpness * 1.05, 500.0)
            top_k = max(3, top_k - 1) if top_k > 3 else top_k
        
        # Restart if stalled
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.02 / (1.0 + 0.1 * (t // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            sharpness = max(sharpness * 0.5, 20.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    NS restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    NS t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " sharp=" + str(round(sharpness, 1)))
    
    return best_a, best_m

def frank_wolfe(a0, time_limit, top_k=5, verbose=True, seed=1):
    """Frank-Wolfe (conditional gradient) on the simplex."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Track support set for accelerated FW
    support = [np.argmax(a)]
    
    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
        
        # Subgradient
        grad, _ = compute_subgrad(a, c, top_k=top_k, sharpness=100.0)
        
        # FW step
        s_idx = np.argmin(grad)
        t_step = 2.0 / (it + 2)
        
        a_new = (1 - t_step) * a
        a_new[s_idx] += t_step
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        
        a = a_new
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    FW it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
    
    return best_a, best_m

def peak_surgery(a0, time_limit, verbose=True, seed=77):
    """Targeted local search: reduce top peaks via mass redistribution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 5000
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find top peaks
        n_top = min(20, 2*n-1)
        top_idx = np.argsort(c)[::-1][:n_top]
        
        improved = False
        found_move = False
        
        for k in top_idx[:10]:
            if found_move:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            if products.max() < 1e-20:

            top_pairs = np.argsort(products)[::-1][:5]

                    break
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n:
                    continue
                
                # Determine which to reduce
                if a[src_i] >= a[partner]:
                    reduce_i = src_i
                else:
                    reduce_i = partner
                
                if a[reduce_i] < 1e-14:
                    continue
                
                # Compute pressure
                pressure = np.zeros(n)
                for k2 in top_idx[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        pressure[idx2] += a[k2i - idx2]
                        pressure[k2i - idx2] += a[idx2]
                
                # Find low-pressure destinations
                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_idx[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2+1] = True
                
                candidates = np.where(~in_peak)[0]
                if len(candidates) < 50:
                    candidates = np.arange(n)
                
                mask = candidates != reduce_i
                candidates = candidates[mask]
                if len(candidates) == 0:
                    continue
                
                cand_pressure = pressure[candidates]
                n_try = min(8, len(candidates))
                try_dsts = candidates[np.argsort(cand_pressure)[:n_try]]
                
                for dst in try_dsts:
                    if found_move:
                        break
                    dst_int = int(dst)
                    for frac in [0.003, 0.005, 0.01, 0.02, 0.03]:
                        amt = a[reduce_i] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[reduce_i] -= amt
                        a_trial[dst_int] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            found_move = True
                            break
            
        if not improved:
            # Random exploration
            if no_improve > 500:
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                break
        
        if verbose and no_improve > 0 and no_improve % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSurgery (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve))
    
    return best_a, best_m
