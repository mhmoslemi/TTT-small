def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:

def norm_sum1(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """_project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_grad_max(a, beta, top_k):
    """
    Compute gradient of smoothed max via log-sum-exp of top-k peaks.
    Returns (gradient, peak_values, peak_probs).
    """
    n = len(a)
    c = fast_autocorr(a)
    fft_size = 1 << (2 * n - 1).bit_length()

    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
    else:
        c_use = c.copy()

    c_max = c_use.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total

    # Gradient: (2/beta) * Conv[p, a_rev][n-1:2n-1]
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    a_rev = a[::-1]
    Conv = np.fft.irfft(P * np.fft.rfft(a_rev, fft_size), fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad, c, p


def nesterov_opt(a0, time_limit, beta_start=50.0, beta_end=3000.0,
                 beta_growth=1.006, eta_base=0.5, top_k_ratio=0.02,
                 restart_every=1500, verbose=True, seed=42):
    """Nesterov accelerated projected gradient with adaptive beta and top-k."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, c, p = compute_grad_max(y, beta, top_k)

        # Adaptive step size based on gradient magnitude
        g_norm = np.sqrt((grad * y).sum())
        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.08)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 30 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            # Structured restart: mix of anti-symmetric, zigzag, and random
            noise_scale = 0.04 / (1.0 + 0.03 * (t // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            zigzag = np.sin(2 * np.pi * np.arange(n) / max(1, n // 7))
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.3 * anti + 0.3 * zigzag + 0.4 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + ", beta=" + str(round(beta, 1)))

    return best_a, best_max_b


def analytical_peak_surgery(a0, time_limit, verbose=True, seed=77):
    """
    Targeted local search using analytical computation of optimal moves.
    For each top peak, finds the best (source, dest) pair and optimal step size.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 5000

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-16:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Find top peaks
        n_top = min(20, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:n_top]

        improved = False
        found_move = False

        for k in top_idx[:12]:
            if found_move:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]

            if products.max() < 1e-20:
                continue

            # Find the dominant source position
            src_i = int(idx[np.argmax(products)])
            partner = k_int - src_i
            if partner < 0 or partner >= n:
                continue

            # Determine which element to reduce
            src_val = a[src_i]
            par_val = a[partner]
            if src_val >= par_val:
                reduce_i = src_i
            else:
                reduce_i = partner

            if a[reduce_i] < 1e-14:
                continue

            # Compute pressure at each position from all top peaks
            pressure = np.zeros(n)
            in_critical = np.zeros(n, dtype=bool)
            for k2 in top_idx[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]
            for k2 in top_idx[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_critical[s2:e2 + 1] = True

            # Find best destination: low pressure, not in critical region
            candidates = np.where(~in_critical)[0]
            if len(candidates) < 30:
                candidates = np.arange(n)
            mask = candidates != reduce_i
            candidates = candidates[mask]

            if len(candidates) == 0:
                continue

            cand_pressure = pressure[candidates]
            n_try = min(8, len(candidates))
            try_dsts = candidates[np.argsort(cand_pressure)[:n_try]]

            for dst in try_dsts:
                dst_int = int(dst)
                # Try multiple step sizes
                for frac in [0.003, 0.007, 0.015, 0.03, 0.06]:
                    amt = a[reduce_i] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_i] -= amt
                    a_trial[dst_int] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-17:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found_move = True
                        break
                if found_move:
                    break

        if not improved:
            # Random exploration when stuck
            if no_improve > 300:
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.04)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                break

        if verbose and no_improve > 0 and no_improve % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSurgery (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve))

    return best_a, best_m


def multi_peak_equalization(a0, time_limit, verbose=True, seed=200):
    """
    Try to equalize the top peaks by iterative mass redistribution.
    When multiple peaks are near the max, reduce the largest by moving mass
    to positions that don't increase other peaks.
    """

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 3000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-16:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Identify top peaks and check if they're close together
        top_idx = np.argsort(c)[::-1][:10]
        top_vals = c[top_idx]
        max_val = top_vals[0]
        # Peak spread: how much the 2nd, 3rd peaks are below the max
        spread = max_val - top_vals.min()

        # For the single largest peak, try to reduce it
        k_main = int(top_idx[0])
        i_start = max(0, k_main - n + 1)
        i_end = min(n - 1, k_main)
        if i_start > i_end:
            no_improve += 1
            continue

        idx = np.arange(i_start, i_end + 1)
        products = a[idx] * a[k_main - idx]
        max_prod_idx = np.argmax(products)
        src_i = int(idx[max_prod_idx])
        partner = k_main - src_i
        if partner < 0 or partner >= n:
            no_improve += 1
            continue

        # Reduce the dominant term
        reduce_i = src_i if a[src_i] >= a[partner] else partner
        if a[reduce_i] < 1e-13:
            no_improve += 1
            continue

        # Compute sensitivity: for each destination q, how much does it affect peaks?
        # delta_c[k2] ≈ 2*amt*(a[k2-q] - a[k2-reduce_i])
        # We want delta_c[k2] < 0 for all top peaks k2
        # i.e., a[k2-q] < a[k2-reduce_i] for all top peaks

        # Sample a set of candidate destinations
        n_cand = min(100, n // 10)
        cand_positions = rng.choice(n, n_cand, replace=False)

        best_dst = -1
        best_score = 0.0

        for q in cand_positions:
            q_int = int(q)
            if q_int == reduce_i:
                continue
            # Check if this destination is good for all top peaks
            good = True
            total_reduction = 0.0
            for k2 in top_idx[:6]:
                k2i = int(k2)
                a_k2_q = a[k2i - q_int] if 0 <= k2i - q_int < n else 0.0
                a_k2_r = a[k2i - reduce_i] if 0 <= k2i - reduce_i < n else 0.0
                if a_k2_q >= a_k2_r:
                    good = False
                    break
                total_reduction += (a_k2_r - a_k2_q)
            if good and total_reduction > best_score:
                best_score = total_reduction
                best_dst = q_int

        if best_dst >= 0:
            # Try moving mass to best_dst
            for frac in [0.005, 0.01, 0.02, 0.04]:
                amt = a[reduce_i] * frac
                a_trial = a.copy()
                a_trial[reduce_i] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-17:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    no_improve = 0
                    break
        else:
            no_improve += 1

        if verbose and no_improve > 0 and no_improve % 2000 == 0:

            J = 2.0 * n * best_m
            print("    MultiPeakEq (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve))

        k_max = int(c.argmax())
        c_max = c[k_max]
        if c_max < best_m - 1e-17:
            best_m = c_max
            best_a = a.copy()
            stall = 0
            continue

        i_start = max(0, k_max - n + 1)
        i_end = min(n - 1, k_max)
        if i_start > i_end:
            break

        idx = np.arange(i_start, i_end + 1)
        products = a[idx] * a[k_max - idx]

        # Try reducing top 5 product terms
        top_pairs = np.argsort(products)[::-1][:5]
        improved = False

        for pi in top_pairs:
            if improved:
                break
            src_i = int(idx[pi])
            partner = k_max - src_i
            if partner < 0 or partner >= n:
                continue

            reduce_i = src_i if a[src_i] >= a[partner] else partner
            if a[reduce_i] < 1e-13:
                continue

            # Try very small moves to several random destinations
            for _ in range(8):
                dst = int(rng.integers(n))
                if dst == reduce_i:
                    continue
                for frac in [0.001, 0.003, 0.005]:
                    amt = a[reduce_i] * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_i] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-18:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

        if not improved:
            stall += 1

        if verbose and stall > 0 and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    FinePolish (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update the best sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score
