                y_mid = y_mid / sm
                f_mid = trial_max(y_mid)

                y_eval = np.maximum((1.0 - (lo + hi) / 4) * y + ((lo + hi) / 4) * g_dir, 0.0)
                se = y_eval.sum()
                if se > 1e-15:
                    y_eval = y_eval / se
                    f_q1 = trial_max(y_eval)
                else:
                    f_q1 = f_mid

                if f_mid <= f_q1:
                    hi = (lo + hi) / 2.0
                else:
                    lo = (lo + hi) / 2.0

            best_alpha = (lo + hi) / 2.0

        y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy
        else:
            y = g_dir.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t=%d (%.1fs) obj=%.7f |S|=%d" % (t, elapsed, best_obj, len(active_set)))

    if verbose:
        print("    LP-FW done (%.1fs) obj=%.7f" % (time.monotonic() - t0, best_obj))
    return best_a, best_obj


def gradient_refine(a0, time_limit, verbose=True):
    """
    Gradient-based refinement using subgradient of L-infinity autocorrelation
    with Barzilai-Borwein step size.
    """
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_obj = objective_val(y)
    t0 = time.monotonic()

    y_prev = y.copy()
    grad_prev = None
    step_prev = None
    bb_count = 0
    stall = 0
    rng = np.random.default_rng(2024)

    top_k_count = 20

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 8000:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        obj_now = 2.0 * n * M_curr

        if obj_now < best_obj - 1e-13:
            best_obj = obj_now
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Compute subgradient: sum of gradients at top peak lags
        k_max = min(top_k_count, len(ac))
        peak_idx = np.argsort(ac)[::-1][:k_max]
        peak_vals = ac[peak_idx]
        # Weight by proximity to max
        weights = np.exp(5.0 * (peak_vals - peak_vals[0]))
        weights /= weights.sum()

        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(weights, peak_idx):
            k_i = int(k)
            j_s = max(0, k_i - n + 1)
            j_e = min(n, k_i + 1)
            if j_s >= j_e:
                continue
            idx = np.arange(j_s, j_e)
            # d/dy_j (y*y)_k = y[k-j] + y[j-k] (symmetric contribution)
            # Actually (y*y)_k = sum_j y_j * y_{k-j}
            # d/dy_j = y_{k-j} (for the first occurrence) + y_{k-j} (for the second, when j' = k-j)
            # But since we sum over all j, each pair is counted once
            # So d/dy_j (y*y)_k = y_{k-j} if k-j is in [0,n-1]
            gj = y[k_i - idx]
            grad[idx] += 2.0 * wi * gj

        # Barzilai-Borwein step size
        if grad_prev is not None and step_prev > 1e-15:
            dy = y - y_prev
            dg = grad - grad_prev
            dy_dg = np.dot(dy, dg)
            dg_dg = np.dot(dg, dg)
            dy_dy = np.dot(dy, dy)
            dg_dy = np.dot(dg, dy)
            if dg_dg > 1e-15:
                bb_up = dy_dg / dg_dg
                bb_dn = dy_dy / dg_dy if dg_dy > 1e-15 else 1e-4
                if bb_count % 2 == 0:
                    step = bb_up
                else:
                    step = bb_dn
                step = np.clip(step, 1e-8, 0.5)
            else:
                step = 1.0 / (np.sqrt(t) + 1.0)
            bb_count += 1
        else:
            step = 3.0 / (np.sqrt(t) + 5.0)

        # Gradient step with projection
        y_new = y - step * grad
        y_new = np.maximum(y_new, 0.0)
        # Project onto sum=1
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = np.ones(n) / n
        else:
            y_new = y_new / sy

        # Ensure we're not increasing the objective too much (backtracking)
        obj_new = 2.0 * n * fast_autocorr(y_new).max()
        if obj_new > M_curr + 1e-10:
            # Reject and try smaller step
            for shrink in [0.5, 0.25, 0.1]:
                y_trial = y - step * shrink * grad
                y_trial = np.maximum(y_trial, 0.0)
                st = y_trial.sum()
                if st > 1e-15:
                    y_trial = y_trial / st
                    obj_trial = 2.0 * n * fast_autocorr(y_trial).max()
                    if obj_trial <= M_curr + 1e-10:
                        y_new = y_trial
                        step *= shrink
                        break
            else:
                y_new = y  # No move

        y_prev = y
        grad_prev = grad
        y = y_new

        # Periodic restart with noise if stalled
        if stall > 3000 and stall % 3000 == 0:
            noise = 0.005 * rng.standard_normal(n) * best_a
            y_r = best_a + noise
            y_r = np.maximum(y_r, 1e-12)
            y_r = y_r / y_r.sum()
            y = y_r
            y_prev = y_r.copy()
            grad_prev = None
            stall = 0

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    GradRef t=%d (%.1fs) obj=%.7f stall=%d" % (t, elapsed, best_obj, stall))

    return best_a, best_obj


def local_search(a0, time_limit, verbose=True):
    """
    Pairwise mass-relocation local search.
    For each pair (i, j), try moving a small amount of mass from i to j.
    Use FFT-based scoring to avoid full convolution at each step.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_obj = objective_val(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    Af = np.fft.rfft(a, fft_size)

    stall = 0
    max_stall = 500

    for rnd in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        obj_now = 2.0 * n * M_curr
        if obj_now < best_obj - 1e-13:
            best_obj = obj_now
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Focus on top peak lags
        n_peaks = min(10, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_peaks]

        # For each peak lag, find the most contributing source and best destination
        improved = False
        for k in top_lags[:5]:
            k_i = int(k)
            j_s = max(0, k_i - n + 1)
            j_e = min(n, k_i + 1)
            if j_s >= j_e:
                continue
            idx = np.arange(j_s, j_e)
            contrib = a[idx] * a[k_i - idx]
            if contrib.max() < 1e-18:
                continue

            # Top 2 sources
            top_srcs = idx[np.argsort(contrib)[::-1][:2]]
            for src in top_srcs:
                src_i = int(src)
                src_val = a[src_i]
                if src_val < 1e-10:
                    continue

                # Score destinations: estimate impact on all top peaks
                scores = np.zeros(n)
                for k2 in top_lags:
                    k2_i = int(k2)
                    s2 = max(0, k2_i - n + 1)
                    e2 = min(n, k2_i + 1)
                    if s2 < e2:
                        idx2 = np.arange(s2, e2)
                        # Adding mass at dst affects peak k2 by: a[k2-dst] (if valid)
                        # Subtracting from src affects by: -a[k2-src]
                        dst_valid = (k2_i - idx2)
                        # Contribution to peak k2 from adding at position p:
                        # a[p] * a[k2-p], so marginal = a[k2-p]
                        # For src removal: marginal = -a[k2-src_i]
                        scores[idx2] += a[k2_i - idx2]
                scores[src_i] = 1e20

                # Best candidates
                candidates = np.argsort(scores)[:40]
                found = False
                for dst in candidates:
                    dst_i = int(dst)
                    if dst_i == src_i:
                        continue
                    for frac in [0.001, 0.003, 0.008, 0.02]:
                        amt = src_val * frac
                        if amt < 1e-15:
                            continue
                        a_t = a.copy()
                        a_t[src_i] -= amt
                        a_t[dst_i] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < M_curr - 1e-14:
                            a = a_t
                            M_curr = Mt
                            obj_now = 2.0 * n * Mt
                            if obj_now < best_obj - 1e-14:
                                best_obj = obj_now
                                best_a = a.copy()
                            stall = 0
                            improved = True
                            found = True
                            break
                    if found:
                        break

        if not improved:
            # Random perturbation
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.001, 0.005)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < M_curr:
                            improved = True
                            stall = 0
                            if 2.0 * n * Mn < best_obj:
                                best_obj = 2.0 * n * Mn
                                best_a = a.copy()
                            break

        if verbose and rnd % 200 == 0:
            print("    LocSearch rnd=%d (%.1fs) obj=%.7f" % (rnd, time.monotonic() - t0, best_obj))

    return best_a, best_obj


def diverse_init(n, parent=None):
    """Generate diverse initial candidates."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)

    candidates = []

    # Truncated cosine (often near-optimal shape)
    for width in [0.8, 0.9, 0.95, 1.0]:
        centered = (x - 0.5) / (width / 2.0)
        mask = np.abs(centered) <= 1.0
        c = np.zeros(n)
        c[mask] = 0.5 * (1.0 + np.cos(np.pi * centered[mask]))
        if c.sum() > 0:
            candidates.append(normalize(c))

    # Sine power
    for p in [1, 1.5, 2, 2.5, 3]:
        candidates.append(normalize(np.sin(np.pi * x) ** p))

    # Gaussian
    for sigma in [0.15, 0.2, 0.25, 0.3, 0.35, 0.4]:
        candidates.append(normalize(np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)))

    # Double bump
    for c1, c2 in [(0.3, 0.7), (0.25, 0.75), (0.35, 0.65)]:
        for sg in [0.08, 0.12, 0.18]:
            c = np.exp(-((x - c1) ** 2) / (2 * sg ** 2)) + np.exp(-((x - c2) ** 2) / (2 * sg ** 2))
            candidates.append(normalize(c))

    # Tukey window
    for alpha in [0.1, 0.2, 0.3, 0.5]:
        c = np.ones(n)
        m = max(1, int(alpha * n))
        taper = 0.5 * (1 + np.cos(np.pi * np.arange(m) / m))
        c[:m] = taper
        c[-m:] = taper[::-1]
        candidates.append(normalize(c))

    # If parent available, blend
    if parent is not None:
        p_norm = normalize(parent)
        for w in [0.7, 0.85]:
            candidates.append(normalize(w * p_norm + (1 - w) * candidates[0]))
            candidates.append(normalize(w * p_norm + (1 - w) * candidates[len(candidates) // 2]))

    best_c = candidates[0]
    best_v = objective_val(best_c)
    for c in candidates:
        v = objective_val(c)
        if v < best_v:
            best_v = v
            best_c = c.copy()

    return best_c, best_v


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1750.0)
    solver = get_solver()
    print("Budget: %.1fs" % time_budget)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 0.01:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            parent_loaded = True
            print("Parent: n=%d score=%.7f" % (len(p), best_score))
    except Exception:
        pass

    if best_seq is None:
        n_init = 5000
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(0.5 * (1 + np.cos(np.pi * (2 * x - 1))))
        best_score = eval_safe(best_seq)
        print("Default init: n=%d score=%.7f" % (n_init, best_score))

    el = lambda: time.monotonic() - t_global

    # ========== Phase 1: Multi-length LP-FW ==========
    # Try optimization at several lengths and keep the best
    lengths_to_try = [3000, 5000, 7000]
    if parent_loaded and len(best_seq) > 7000:
        lengths_to_try.append(10000)

    for target_n in lengths_to_try:
        remaining = time_budget - el()
        if remaining < 60:
            break

        # Determine time allocation for this length
        frac = 0.25 if target_n <= 5000 else 0.15
        t_alloc = min(remaining * frac, 300.0)
        if t_alloc < 30:
            t_alloc = min(remaining * 0.3, 200.0)

        print("\n=== LP-FW at n=%d (t=%.0fs) ===" % (target_n, t_alloc))
        a_start = interp_seq(best_seq, target_n)
        a_out, v_out = lp_fw_optimize(a_start, t_alloc, solver,
                                       init_K=200, max_K=2000 if target_n <= 5000 else 1500,
                                       add_batch=25, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("  Improved to %.7f at n=%d" % (best_score, target_n))

    # ========== Phase 2: Gradient refinement ==========
    remaining = time_budget - el()
    if remaining > 60:
        n_grad = min(len(best_seq), 8000)
        a_grad = interp_seq(best_seq, n_grad)
        t_grad = min(remaining * 0.2, 200.0)
        print("\n=== GradRefine n=%d (t=%.0fs) ===" % (n_grad, t_grad))
        a_out, v_out = gradient_refine(a_grad, t_grad, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("  Improved to %.7f" % best_score)

    # ========== Phase 3: Local search ==========
    remaining = time_budget - el()
    if remaining > 40:
        n_ls = min(len(best_seq), 6000)
        a_ls = interp_seq(best_seq, n_ls)
        t_ls = min(remaining * 0.25, 250.0)
        print("\n=== LocalSearch n=%d (t=%.0fs) ===" % (n_ls, t_ls))
        a_out, v_out = local_search(a_ls, t_ls, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("  Improved to %.7f" % best_score)

    # ========== Phase 4: Second round LP-FW at best length ==========
    remaining = time_budget - el()
    if remaining > 60:
        n_cur = len(best_seq)
        n_lp2 = min(n_cur, 7000)
        a_lp2 = interp_seq(best_seq, n_lp2)
        t_lp2 = min(remaining * 0.3, 300.0)
        print("\n=== LP-FW2 n=%d (t=%.0fs) ===" % (n_lp2, t_lp2))
        a_out, v_out = lp_fw_optimize(a_lp2, t_lp2, solver,
                                       init_K=300, max_K=2500,
                                       add_batch=30, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("  Improved to %.7f" % best_score)

    # ========== Phase 5: Final local search ==========
    remaining = time_budget - el()
    if remaining > 30:
        n_fin = min(len(best_seq), 5000)
        a_fin = interp_seq(best_seq, n_fin)
        t_fin = remaining - 15
        print("\n=== Final LocalSearch n=%d (t=%.0fs) ===" % (n_fin, t_fin))
        a_out, v_out = local_search(a_fin, t_fin, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("  Improved to %.7f" % best_score)

    # ========== Phase 6: Try longer sequence with gradient ==========
    remaining = time_budget - el()
    if remaining > 40:
        n_long = min(len(best_seq) + 3000, 12000)
        a_long = interp_seq(best_seq, n_long)
        t_long = remaining - 10
        print("\n=== Long-grad n=%d (t=%.0fs) ===" % (n_long, t_long))
        a_out, v_out = gradient_refine(a_long, t_long, verbose=False)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("  Improved to %.7f at n=%d" % (best_score, n_long))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print("\nFINAL: score=%.7f n=%d t=%.1fs" % (best_score, len(best_seq), el()))
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(0.5 * (1 + np.cos(np.pi * (2 * x - 1))))
        best_score = eval_safe(best_seq)
        print("FALLBACK: score=%.7f" % best_score)

    result = [float(max(0.0, min(1000.0, x))) for x in np.asarray(best_seq, dtype=np.float64)]
    return result
