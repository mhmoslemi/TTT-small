Let me also think about the spectral approach more carefully.

**Spectral Peak Reduction (SPR):**

The idea: in the frequency domain, identify which frequency components contribute most to the autocorrelation peaks, and reduce them.

Given `A(f) = FFT(a)`, the autocorrelation is `b = IFFT(|A(f)|^2)`.

The contribution of frequency bin p to `b[k]` is `|A(p)|^2 * exp(2πi*p*k/N)` (for appropriate indexing).

The real part (which is what matters for the autocorrelation of a real signal) is `|A(p)|^2 * cos(2π*p*k/N)`.

So to reduce `b[k*]` (the peak), we should reduce `|A(p)|^2` for frequencies p where `cos(2π*p*k*/N)` is large and positive.

Algorithm:
1. Compute `A = FFT(a, N)` where N >= 2n-1
2. Compute `b = IFFT(|A|^2, N)[:2n-1]`
3. Find peak at k*
4. Compute `contrib(p) = |A(p)|^2 * cos(2π*p*k*/N)` for each p
5. Reduce |A(p)| proportionally to max(0, contrib(p)):
   `|A(p)|_new = |A(p)| * (1 - alpha * contrib(p) / max_contrib)`
6. `a_new = IFFT(A_new)[:n]`
7. Project: `a_new = max(a_new, 0)`, normalize
8. Repeat

def compute_smoothed_grad(y, beta, top_k):
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and top_k < len(b) and top_k > 0:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b_use)
        p[b_max == b_use] = 1.0 / max(1, (b_max == b_use).sum())
    else:
        p = exp_vals / p_total
    # Compute gradient via correlation
    fft_size = 1 << (2 * n + 1).bit_length()
    Y_f = np.fft.rfft(y, fft_size)
    P_f = np.fft.rfft(p, fft_size)
    # grad[i] = 2 * sum_k p[k] * a[k-i] (valid indices)
    # = 2 * cross_correlation(y, p) at lag i
    corr = np.fft.irfft(Y_f * P_f, fft_size)
    # corr[i] = sum_j y[j] * p[j+i] (approximately, need to check indexing)
    # Actually: ifft(FFT(y) * FFT(p))[i] = sum_j y[j] * p[(i-j) mod N]
    # We want: sum_k p[k] * y[k-i] = sum_k p[k] * y[k-i]
    # Let m = k-i: sum_m p[m+i] * y[m] = cross_correlation at lag i
    # ifft(FFT(y) * conj(FFT(p)))[i] = sum_j y[j] * p[j+i]... 
    # Let me just use the direct approach for correctness
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_s2 = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_s2)
    Pr = np.fft.rfft(p, fft_s2)
    conv = np.fft.irfft(Ar * Pr, fft_s2)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, p, b

def fista_optimize(a0, time_limit, beta_start=20.0, beta_end=1000.0,
                   beta_growth=1.004, top_k=12, restart_every=3000,
                   verbose=True):

    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()

    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(42)
    avg_x = x.copy()
    avg_count = 0.0
    s_prev = None
    y_prev = None
    bb_step = 1.0 / (n ** 0.5)
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        grad, p, b = compute_smoothed_grad(y, beta, top_k)
        
        if s_prev is not None and y_prev is not None:
            sy = y - y_prev
            gy = grad - s_prev
            sy_gy = sy.dot(gy)
            gy_gy = gy.dot(gy)
            if gy_gy > 1e-20:
                bb_candidate = sy_gy / gy_gy
                if bb_candidate > 1e-8 and bb_candidate < 10.0:
                    bb_step = 0.5 * bb_step + 0.5 * bb_candidate
                    bb_step = max(1e-6, min(bb_step, 5.0 / beta))
                else:
                    bb_step = max(bb_step * 0.9, 1e-6)
            s_prev = grad.copy()
            y_prev = y.copy()
        
        eta = min(bb_step, 0.15)
        z = y - eta * grad
        x_new = simplex_project(z)
        
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        
        avg_count += 1.0
        avg_x = avg_x + (x_new - avg_x) / avg_count

            best_a = x_new.copy()
            stall = 0

            stall += 1

        x = x_new
        t_nest = t_new

def coord_descent(a0, time_limit, verbose=True):
    """Incremental coordinate descent with exact local optimization."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    b = fast_autocorr(a)
    best_a = a.copy()
    best_m = b.max()
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(777)
    n_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < 1e-14:
                continue
            
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            # Compute unchanged max
            m_unchanged = 0.0
            if k_lo > 0:
                m_unchanged = max(m_unchanged, b[:k_lo].max())
            if k_hi < 2 * n - 2:
                m_unchanged = max(m_unchanged, b[k_hi + 1:].max())
            
            # Gather affected data
            ks = np.arange(k_lo, k_hi + 1)
            other_idx = ks - i
            is_double = (ks == 2 * i)
            
            # For k != 2i: b[k] = base + 2*ai*a[k-i] => new: base + 2*x*a[k-i]
            # For k == 2i: b[k] = base2 + ai^2 => new: base2 + x^2
            linear_mask = ~is_double
            if linear_mask.any():
                c_vals = a[other_idx[linear_mask]]
                d_vals = b[ks[linear_mask]] - 2.0 * ai * c_vals
            else:
                c_vals = np.array([])
                d_vals = np.array([])
            
            if is_double.any():
                d_quad = b[2 * i] - ai * ai
            
            # Try candidate x values
            candidates = np.array([0.0, ai * 0.3, ai * 0.6, ai, ai * 1.4, ai * 2.0])
            
            if len(c_vals) > 0 and c_vals.max() > 1e-20:
                x_cross = (m_unchanged - d_vals) / (2.0 * np.maximum(c_vals, 1e-30))
                x_cross = x_cross[(x_cross > 0) & (x_cross < ai * 3)]
                if len(x_cross) > 0:
                    candidates = np.concatenate([candidates, [x_cross.min(), x_cross.max()]])
            
            if is_double.any():
                quad_cross = m_unchanged - d_quad
                if quad_cross > 0:
                    candidates = np.concatenate([candidates, [np.sqrt(quad_cross) * 0.7, np.sqrt(quad_cross) * 1.3]])
            
            best_x = ai
            best_obj = best_m
            
            for x in candidates:
                x = max(0.0, x)
                new_max_aff = m_unchanged
                if len(c_vals) > 0:
                    new_lin = 2.0 * x * c_vals + d_vals
                    new_max_aff = max(new_max_aff, new_lin.max())
                if is_double.any():
                    new_max_aff = max(new_max_aff, x * x + d_quad)
                if new_max_aff < best_obj - 1e-14:
                    best_obj = new_max_aff
                    best_x = x
            
            if best_x != ai:
                delta = best_x - ai
                a[i] = best_x
                if linear_mask.any():
                    b[ks[linear_mask]] += 2.0 * delta * c_vals
                if is_double.any():
                    b[2 * i] += 2.0 * delta * ai + delta * delta
                if best_obj < best_m - 1e-13:
                    best_m = best_obj
                    best_a = a.copy()
                    improved = True
        
        n_passes += 1
        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0
        
        if verbose and n_passes % 2 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    CD pass=%d (%.1fs): J=%.6f" % (n_passes, elapsed, J))
    
    return best_a, best_m

def spectral_peak_reduction(a0, time_limit, n_iters=500, alpha=0.1, verbose=True):
    """Reduce autocorrelation peaks via frequency-domain spectral shaping."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    rng = np.random.default_rng(555)
    no_improve = 0
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        b = np.fft.irfft(power, fft_size)[:2 * n - 1]
        b_max = b.max()
        
        if b_max < 1e-15:
            break
        
        if b_max < best_m - 1e-12:
            best_m = b_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break
        
        # Find top peaks
        top_k = np.argsort(b)[::-1][:3]
        
        # For each peak, find frequencies that contribute positively
        freqs = np.arange(len(A))
        for k_peak in top_k:
            k_int = int(k_peak)
            # Contribution of freq p to b[k] is |A(p)|^2 * cos(2*pi*p*k/N)
            phase = 2.0 * np.pi * freqs * k_int / fft_size
            contrib = power * np.cos(phase)
            # Reduce frequencies with high positive contribution
            max_contrib = contrib.max()
            if max_contrib > 0:
                reduction = np.maximum(0, contrib / max_contrib)
                A = A * (1.0 - alpha * reduction) ** 0.5
        
        a_new = np.fft.irfft(A, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print("    SPR done (%.1fs): J=%.6f" % (elapsed, J))
    
    return best_a, best_m

def cyclic_shift_best(a0):
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m

def get_solver():

        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [%s]: %.6f (n=%d, t=%.1fs)" % (label, sc, len(a), elapsed))
    return best_seq, best_score

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score

    except Exception:
        parent = None

        best_seq = norm_sum1(a_init)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=%d, score=%.6f" % (len(best_seq), best_score))

    # Phase 1: FISTA (main optimizer) - 35% of time
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: FISTA (n=%d) ===" % n0)
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 60:

                                  restart_every=3000, verbose=True)

        best_seq, best_score = try_update(best_seq, best_score, a_out, "spr1", t_global)
    
    # Phase 5: LP-Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:

    # Final polish: CD
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Final: CD polish ===")
    if remaining > 20:
        t_f = min(remaining * 0.4, 100.0)
        if t_f > 15:
            a_out, _ = coord_descent(norm_sum1(best_seq), t_f, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "cd_final", t_global)
    
    # Final cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_final", t_global)
    
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=%.6f, n=%d, time=%.1fs" % (best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 7000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
    
    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
