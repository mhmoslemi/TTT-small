    # The max of (c[j] + t*h[j]) over j is a piecewise-linear convex function of t.
    # We want to find t that minimizes it.
    # For t=0, the max is at j* = argmax(c).
    # As t increases, if h[j*] < 0, the max decreases.
    # The function stops decreasing when another j catches up.

    # c[j] + t*h[j] = c[j_star] + t*h[j_star]
    # t = (c[j] - c[j_star]) / (h[j_star] - h[j])

    valid = denom > 0
    if not np.any(valid):
        return t_max
    
    t_candidates = np.full_like(c, np.inf)

    t_catch = np.min(t_candidates[t_candidates > 0]) if np.any(t_candidates > 0) else np.inf

    t_opt = min(t_max, t_catch)

"""
Optimized sequence search using analytical coordinate descent with water-filling,
subgradient-based Frank-Wolfe directions, and adaptive multi-length search.
Key improvement: computes exact optimal transfer amounts via 1D minimization
of the max-convolution objective, rather than trying fixed fractions.

import numpy as np
import time
import math

try:
    import scipy.optimize
    SCIPY_OK = True
except Exception:
    SCIPY_OK = False


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def conv_full(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return np.where(np.isfinite(c) & (c > 0), c, 0.0)


def cross_conv_f(a, b):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    if len(b) != n:
        b = np.resize(b, n)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    return np.where(np.isfinite(c) & (c > 0), c, 0.0)


def compute_pressure(a, active_idx, weights):
    """
    Compute pressure[i] = sum_j w_j * (a[j-i] + a[i-j]) for valid indices.
    This is the gradient of sum_j w_j * c[j] with respect to a[i].
    """
    n = len(a)
    pressure = np.zeros(n, dtype=np.float64)
    for jj, wj in zip(active_idx, weights):
        jj = int(jj)
        wj = float(wj)
        if wj <= 0:
            continue
        # a[jj - i] valid when 0 <= jj-i < n => jj-n+1 <= i <= jj
        lo1 = max(0, jj - n + 1)
        hi1 = min(n - 1, jj)
        if lo1 <= hi1:
            pressure[lo1:hi1 + 1] += wj * a[jj - np.arange(lo1, hi1 + 1)]
        # a[i - jj] valid when 0 <= i-jj < n => jj <= i <= jj+n-1
        lo2 = max(0, jj)
        hi2 = min(n - 1, jj + n - 1)
        if lo2 <= hi2:
            pressure[lo2:hi2 + 1] += wj * a[np.arange(lo2, hi2 + 1) - jj]
    return pressure


def find_optimal_t(c, h, t_max, top_k=60):
    """
    Find t in [0, t_max] minimizing max(c + t*h).
    Uses piecewise-linear analysis on top-k coefficients.
    Returns (optimal_t, resulting_max).
    """
    n = len(c)
    if n == 0 or t_max <= 1e-15:
        return 0.0, float(np.max(c)) if n > 0 else 0.0

    # Get top-k by c value
    k = min(top_k, n)
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx].copy()
    h_top = h[idx].copy()

    # Also include positions with negative h that are close to max (they could become max)
    max_c = c_top[0]
    spread = max_c - c_top[-1]
    threshold = max_c - spread * 0.3
    candidates_mask = (c >= threshold) & (h < -1e-15) & (~np.isin(np.arange(n), idx))
    extra_idx = np.where(candidates_mask)[0][:30]
    if len(extra_idx) > 0:
        all_idx = np.concatenate([idx, extra_idx])
        c_all = c[all_idx]
        h_all = h[all_idx]
    else:
        c_all = c_top
        h_all = h_top

    # Build candidate t values
    candidates = [0.0, t_max]
    m = len(c_all)
    for i in range(m):
        for j in range(i + 1, m):
            denom = h_all[i] - h_all[j]
            if abs(denom) < 1e-18:
                continue
            t_cross = (c_all[j] - c_all[i]) / denom
            if 1e-12 < t_cross < t_max * 1.001:
                candidates.append(min(t_cross, t_max))

    # Evaluate
    best_t = 0.0
    best_val = np.inf
    for t in candidates:
        t = max(0.0, min(t, t_max))
        val = float(np.max(c_all + t * h_all))
        if val < best_val - 1e-16:
            best_val = val
            best_t = t

    return best_t, best_val


def apply_move(c, a, s, d, t):
    """Apply transfer of t from s to d, return new c."""
    n = len(a)
    c_new = c.copy()
    c_new[d:d + n] += 2.0 * t * a
    c_new[s:s + n] -= 2.0 * t * a
    t2 = t * t
    if 2 * d < c_new.size:
        c_new[2 * d] += t2
    if 2 * s < c_new.size:
        c_new[2 * s] -= t2
    if s + d < c_new.size:
        c_new[s + d] -= 2.0 * t2
    return np.where(np.isfinite(c_new), c_new, 0.0)

        arr = np.clip(arr, 0.0, 1000.0)

        if s < 0.01:

        val = evaluate_sequence(lst)

    except Exception:

def normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    np.maximum(a, 0, out=a)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def interpolate_seq(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    new_n = int(max(2, new_n))
    if len(a) == new_n:
        return normalize(a)
    old_x = np.linspace(0, 1, len(a))
    new_x = np.linspace(0, 1, new_n)
    y = np.interp(new_x, old_x, a)
    return normalize(np.maximum(y, 0))


def coordinate_descent(a, c, budget, verbose=True):
    """
    Main coordinate descent with analytical optimal transfer.
    Returns improved sequence.
    """
    t0 = time.monotonic()
    n = len(a)
    a = a.copy()
    c = c.copy()
    best_a = a.copy()
    best_c = c.copy()
    best_M = float(np.max(c))
    stall = 0
    rng = np.random.default_rng(12345)
    iters = 0

    while time.monotonic() - t0 < budget - 2 and stall < 200:
        iters += 1
        M = float(np.max(c))

        # Active set: top K coefficients
        K = min(40, 2 * n - 1)
        active_idx = np.argsort(c)[::-1][:K]
        active_c = c[active_idx]
        # Weights: inverse gap from max (more weight on coefficients closer to max)
        gaps = active_c[0] - active_c + 1e-15
        weights = 1.0 / gaps
        weights /= weights.sum()

        # Compute pressure
        pressure = compute_pressure(a, active_idx, weights)
        pressure = np.where(np.isfinite(pressure), pressure, 0.0)

        # Source candidates: high a[i] * pressure[i]
        src_score = a * pressure
        n_src = min(8, n)
        src_candidates = np.argpartition(-src_score, n_src - 1)[:n_src]

        # Destination candidates: low pressure (prefer positions with some mass or zero)
        dst_score = pressure.copy()
        # Slightly prefer positions that are currently small (easier to add to)
        dst_score += 1e-6 * a
        n_dst = min(15, n)
        dst_candidates = np.argpartition(dst_score, n_dst - 1)[:n_dst]

        # Evaluate all (s, d) pairs
        best_move = None
        best_new_M = M
        for s in src_candidates:
            s = int(s)
            if a[s] < 1e-10:
                continue
            for d in dst_candidates:
                d = int(d)
                if d == s:
                    continue
                # Compute gradient h[j] for this transfer
                # h[j] = a[j-d] + a[d-j] - a[j-s] - a[s-j] (with bounds)
                h = np.zeros(2 * n - 1, dtype=np.float64)
                # Contribution from +t at d: a[j-d] for j in [d, d+n-1], a[d-j] for j in [max(0,d-n+1), d]
                lo1 = max(0, d)
                hi1 = min(2 * n - 2, d + n - 1)
                h[lo1:hi1 + 1] += a[lo1 - d:hi1 - d + 1]
                lo2 = max(0, d - n + 1)
                hi2 = min(2 * n - 2, d)
                if lo2 <= hi2:
                    h[lo2:hi2 + 1] += a[d - lo2:d - hi2 - 1:-1]
                # Contribution from -t at s: subtract a[j-s] for j in [s, s+n-1], a[s-j] for j in [max(0,s-n+1), s]
                lo3 = max(0, s)
                hi3 = min(2 * n - 2, s + n - 1)
                h[lo3:hi3 + 1] -= a[lo3 - s:hi3 - s + 1]
                lo4 = max(0, s - n + 1)
                hi4 = min(2 * n - 2, s)
                if lo4 <= hi4:
                    h[lo4:hi4 + 1] -= a[s - lo4:s - hi4 - 1:-1]

                t_max = a[s] * 0.5
                t_opt, new_M_est = find_optimal_t(c, h, t_max, top_k=50)

                if t_opt > 1e-10 and new_M_est < best_new_M - 1e-14:
                    # Verify with actual convolution update
                    c_new = apply_move(c, a, s, d, t_opt)
                    actual_M = float(np.max(c_new))
                    if actual_M < best_new_M - 1e-14:
                        best_new_M = actual_M
                        best_move = (s, d, t_opt, c_new)

        if best_move is not None:
            s, d, t_opt, c_new = best_move
            a[s] -= t_opt
            a[d] += t_opt
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = c_new / (ss * ss) * (float(best_a.sum()) ** 2) if False else apply_move(c, a, 0, 0, 0)
            # Recompute c properly after normalization
            c = conv_full(a)
            M = float(np.max(c))
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
                best_c = c.copy()
            stall = 0
            if verbose and iters % 100 == 0:
                score = 2.0 * n * best_M
                print(f"    CD iter={iters} score_proxy={score:.6f} t={time.monotonic()-t0:.1f}s")
        else:
            stall += 1
            if stall % 20 == 19:
                # Perturb to escape local minimum
                strength = 0.005 * min(1.0, stall / 50.0)
                pert = rng.normal(0, strength, n) * a
                a = best_a + pert
                a = np.maximum(a, 0)
                a = a / a.sum()
                c = conv_full(a)
                M = float(np.max(c))
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
                stall = 0

        if time.monotonic() - t0 > budget - 5:

    # Final recompute
    c_final = conv_full(best_a)
    return best_a, c_final


def frank_wolfe_step(a, c, active_idx, weights):
    """
    Frank-Wolfe: find direction g on simplex minimizing max of f*g over active set.
    Uses iterative approximation.
    """
    n = len(a)
    size = 2 * n - 1
    # Pressure = gradient of weighted sum of active c's
    pressure = compute_pressure(a, active_idx, weights)
    pressure = np.where(np.isfinite(pressure) & (pressure > 0), pressure, 0.0)

    # The FW direction minimizes <pressure, g> over simplex g
    # The minimum is at g = e_{argmin(pressure)}
    # But we smooth it a bit for better mixing
    pmn = float(np.min(pressure))
    pmx = float(np.max(pressure))
    if pmx - pmn < 1e-15:
        return np.ones(n) / n

    scaled = (pressure - pmn) / (pmx - pmn)
    temperature = 5.0
    w = np.exp(-temperature * scaled)
    w = np.where(np.isfinite(w), w, 0.0)
    w += 1e-15
    g = w / w.sum()
    # Blend with uniform for stability
    g = 0.9 * g + 0.1 * (np.ones(n) / n)
    g = g / g.sum()
    return g


def fw_search(a0, c0, budget, verbose=True):
    """Frank-Wolfe with line search."""
    t0 = time.monotonic()
    a = a0.copy()
    c = c0.copy()
    n = len(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    stall = 0
    iters = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < budget - 3 and stall < 60:
        iters += 1
        K = min(30, 2 * n - 1)
        active_idx = np.argsort(c)[::-1][:K]
        active_c = c[active_idx]
        gaps = active_c[0] - active_c + 1e-15
        weights = 1.0 / gaps
        weights /= weights.sum()

        # Get FW direction
        g = frank_wolfe_step(a, c, active_idx, weights)

        # Compute convolution of g and cross-convolution
        cg = conv_full(g)
        fg = cross_conv_f(a, g)

        # Line search: minimize max((1-alpha)^2 * c + 2*alpha*(1-alpha)*fg + alpha^2*cg)
        M_old = M
        best_alpha = 0.0
        best_new_M = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            val = (1 - alpha) ** 2 * c + 2 * alpha * (1 - alpha) * fg + alpha ** 2 * cg
            new_M = float(np.max(val))
            if new_M < best_new_M - 1e-14:
                best_new_M = new_M
                best_alpha = alpha

        if best_alpha > 0 and best_new_M < M - 1e-14:
            a = (1 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s
            c = conv_full(a)
            M = float(np.max(c))
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 15 == 14:
                pert = rng.normal(0, 0.01, n) * best_a
                a = np.maximum(best_a + pert, 0)
                a = a / a.sum()
                c = conv_full(a)
                M = float(np.max(c))
                stall = 0

        if verbose and iters % 50 == 0:
            print(f"    FW iter={iters} score_proxy={2*n*best_M:.6f} t={time.monotonic()-t0:.1f}s")

    return best_a, conv_full(best_a)


def lse_objective(params, a_fixed, beta, active_mask=None):
    """Log-sum-exp smoothed max of convolution, for use with scipy."""
    # params are the free variables (all n elements, but we'll handle constraints)
    a = np.exp(params - np.max(params))
    a = a / a.sum()
    c = conv_full(a)
    if active_mask is not None:
        c_use = c[active_mask]
    else:
        c_use = c
    m = np.max(c_use)
    shifted = beta * (c_use - m)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    se = e.sum()
    if se < 1e-300:
        return 1e100, np.zeros(len(params))
    val = m + np.log(se) / beta
    # Gradient (approximate, using softmax parameterization)
    p = e / se
    pf = np.zeros(2 * len(a) - 1)
    if active_mask is not None:
        pf[active_mask] = p
    else:
        pf[:] = p
    # grad of sum w_j c[j] w.r.t. a[i]
    n = len(a)
    N = next_pow2(3 * n - 2)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(pf[::-1], N)
    grad_conv = np.fft.irfft(A * Pr, N)[:3 * n - 2]
    grad = 2.0 * grad_conv[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    # Chain rule through softmax
    mean_grad = float(np.dot(a, grad))
    grad_params = a * (grad - mean_grad)
    grad_params = np.where(np.isfinite(grad_params), grad_params, 0.0)
    return val, grad_params


def lbfegs_optimize(a0, beta, budget, verbose=True):
    """L-BFGS optimization in log-parameterization."""
    if not SCIPY_OK:
        return a0
    t0 = time.monotonic()
    n = len(a0)
    a0 = normalize(a0)

    def obj(params):
        val, grad = lse_objective(params, None, beta)
        return val, grad

    # Initialize with log of a0 (shifted so max is 0)
    params0 = np.log(np.maximum(a0, 1e-300))
    params0 -= params0.max()

    maxfun = int(min(500, max(50, budget / 2.0)))
    try:
        res = scipy.optimize.minimize(
            obj, params0, method='L-BFGS-B', jac=True,
            options={'maxfun': maxfun, 'ftol': 1e-14, 'gtol': 1e-8, 'disp': False}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            a_new = np.exp(res.x - res.x.max())
            a_new = a_new / a_new.sum()
            if verbose:
                print(f"    L-BFGS: success={res.success} fun={res.fun:.6f} t={time.monotonic()-t0:.1f}s")
            return a_new
    except Exception as e:
        if verbose:
            print(f"    L-BFGS failed: {e}")
    return a0
