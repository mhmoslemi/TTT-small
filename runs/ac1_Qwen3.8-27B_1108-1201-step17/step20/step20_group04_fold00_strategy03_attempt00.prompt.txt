import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_autocorr(a):
    """Compute full linear convolution of a with itself."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    """Compute gradient of (1/beta)*log(sum top_k exp(beta*c)) w.r.t. a."""
    n = len(y)
    c = fast_autocorr(y)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    # Gradient: (1/beta) * sum_k p_k * d(c_k)/d(a_j)
    # d(c_k)/d(a_j) = sum_i p_k * (a[k-j] + a[k-j]) ... 
    # Actually c_k = sum_i a[i]*a[k-i], so d(c_k)/d(a_j) = sum_i [delta(i,j)*a[k-i] + delta(k-i,j)*a[i]]
    # = a[k-j] + a[j] (if both in range) ... no wait
    # d(c_k)/d(a_j) = a[k-j] + a[j] only if we're being careful
    # Actually: c_k = sum_{i=0}^{n-1} a[i]*a[k-i] (with a[x]=0 for x outside [0,n-1])
    # d(c_k)/d(a_j) = a[k-j] + a[k-j] ... no
    # d(c_k)/d(a_j) = sum_i [delta(i,j)*a[k-i] + a[i]*delta(k-i,j)]
    # = a[k-j] + a[k-j] = 2*a[k-j] (if j != k-j)
    # = 2*a[j] if j = k-j (i.e., k = 2j)
    # Wait no: the first term is a[k-j] (when i=j), the second term is a[i] when k-i=j, i.e., i=k-j, so a[k-j].
    # So d(c_k)/d(a_j) = a[k-j] + a[k-j] = 2*a[k-j] (when j != k-j)
    # And when j = k-j (k=2j): d(c_k)/d(a_j) = a[j] + a[j] = 2*a[j] = 2*a[k-j]
    # So in all cases: d(c_k)/d(a_j) = 2*a[k-j] (with a[x]=0 if x outside [0,n-1])
    # 
    # But wait, this is only correct if we consider c_k = sum a[i]*a[k-i] as a function of a.
    # Let's verify: c_0 = a[0]^2. d(c_0)/d(a_0) = 2*a[0]. Formula: 2*a[0-0] = 2*a[0]. Correct.
    # c_1 = a[0]*a[1] + a[1]*a[0] = 2*a[0]*a[1]. d(c_1)/d(a_0) = 2*a[1]. Formula: 2*a[1-0] = 2*a[1]. Correct.
    # c_1: d(c_1)/d(a_1) = 2*a[0]. Formula: 2*a[1-1] = 2*a[0]. Correct.
    # Great, so grad_j = (1/beta) * sum_k p_k * 2*a[k-j]
    # = (2/beta) * sum_k p_k * a[k-j]
    # This is (2/beta) * conv(p, a)[j] where conv is the full convolution
    # But we need to be careful with indexing.
    # sum_k p_k * a[k-j] for k from 0 to 2n-2, a[x] = 0 outside [0,n-1]
    # = sum_{k=j}^{j+2n-2} p_k * a[k-j] where a[k-j] is valid for k-j in [0,n-1]
    # = sum_{m=0}^{n-1} p_{j+m} * a[m] (where p is extended with 0s outside [0,2n-2])
    # This is (conv(p_ext, a))[j] where p_ext has length 2n-1
    # Actually it's a cross-correlation: sum_m a[m] * p[j+m]
    # In numpy: this is np.convolve(a, p_flipped)[j + len(p) - 1] ... let me just use FFT
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(y, fft_size)
    P = np.fft.rfft(p, fft_size)
    # We need sum_k p[k] * a[k-j] which is (p * a_flipped)[j + n - 1] ... 
    # Actually let's think: conv(p, a_rev)[j + n - 1] where a_rev = a[::-1]
    # conv(p, a_rev)[m] = sum_k p[k] * a_rev[m-k] = sum_k p[k] * a[n-1-m+k]
    # We want sum_k p[k] * a[k-j]. Set n-1-m+k = k-j => m = n-1+j
    # So it's conv(p, a_rev)[n-1+j]
    a_rev = y[::-1]
    Conv = np.fft.irfft(P * np.fft.rfft(a_rev, fft_size), fft_size)
    # grad_j = (2/beta) * Conv[n-1+j]
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad, p, c


def nesterov_opt(a0, time_limit, beta_start=30.0, beta_end=2000.0,
                 beta_growth=1.006, eta_base=0.6, top_k_ratio=0.02,
                 restart_every=2000, verbose=True, seed=42):
    """Nesterov accelerated projected gradient with adaptive beta."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, c = compute_smoothed_grad(y, beta, top_k)

        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            # Structured restart: mix anti-symmetric and random
            noise_scale = 0.03 / (1.0 + 0.05 * (t // restart_every))
            # Anti-symmetric component
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            # Random component
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + ", beta=" + str(round(beta, 1)))

    return best_a, best_max_b


def peak_surgery(a0, time_limit, verbose=True, seed=77):
    """Targeted local search: identify top peaks and reduce them via mass redistribution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 3000

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Find top peaks
        n_top = min(30, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]

        improved = False
        found_move = False

        for k in top_idx[:15]:
            if found_move:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            if products.max() < 1e-20:
                continue

            # For the top product pair, try to reduce it
            src_i = int(idx[np.argmax(products)])
            partner = k_int - src_i
            if partner < 0 or partner >= n:
                continue

            src_val = a[src_i]
            par_val = a[partner]

            # Determine which to reduce (the larger one)
            if src_val >= par_val:
                reduce_i = src_i
                keep_i = partner
            else:
                reduce_i = partner
                keep_i = src_i

            if a[reduce_i] < 1e-14:
                continue

            # Compute "pressure" at each position from top peaks
            pressure = np.zeros(n)
            for k2 in top_idx[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]

            # Find best destination: low pressure, not in peak region
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_idx[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)

            # Exclude the reduce position
            mask = candidates != reduce_i
            if not mask.any():
                continue
            candidates = candidates[mask]

            cand_pressure = pressure[candidates]
            # Try the 5 lowest pressure destinations
            n_try = min(5, len(candidates))
            try_dsts = candidates[np.argsort(cand_pressure)[:n_try]]

            for dst in try_dsts:
                dst_int = int(dst)
                if dst_int == reduce_i:
                    continue
                # Try different fractions
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                    amt = a[reduce_i] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_i] -= amt
                    a_trial[dst_int] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found_move = True
                        break
                if found_move:
                    break

        if not improved:
            # Random exploration when stuck
            if no_improve > 500:
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                break

        if verbose and no_improve % 1000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSurgery (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve))

    return best_a, best_m


def pairwise_cd(a0, time_limit, verbose=True, seed=123):
    """Fine-grained pairwise coordinate descent for final polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        c = fast_autocorr(a)
        # Focus on peaks above 95th percentile
        threshold = np.percentile(c, 95)
        peak_positions = np.where(c > threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.argsort(c)[::-1][:20]

        improved = False

        for k in peak_positions[:20]:
            if improved and (time.monotonic() - t0 > time_limit):
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]

            # Try reducing the top 3 product pairs
            top_pairs = np.argsort(products)[::-1][:3]
            for pi in top_pairs:
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n or partner == src_i:
                    continue
                if a[src_i] < 1e-14 and a[partner] < 1e-14:
                    continue

                # Try moving a tiny amount from the larger to a random safe position
                larger_i = src_i if a[src_i] >= a[partner] else partner
                smaller_i = partner if a[src_i] >= a[partner] else src_i
                if a[larger_i] < 1e-14:
                    continue

                for frac in [0.001, 0.002, 0.003, 0.005]:
                    amt = a[larger_i] * frac
                    if amt < 1e-18:
                        continue
                    # Try several random destinations
                    for _ in range(3):
                        dst = int(rng.integers(n))
                        if dst == larger_i:
                            continue
                        a_trial = a.copy()
                        a_trial[larger_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Occasionally try a structured move
            if stall % 100 == 0:
                # Anti-symmetric nudge
                anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
                eps = 0.01 / (1.0 + stall / 100.0)
                a_try = a + eps * anti * a
                a_try = np.maximum(a_try, 0.0)
                s_t = a_try.sum()
                if s_t > 1e-15:
                    a_try = a_try / s_t
                    M_try = fast_autocorr(a_try).max()
                    if M_try < best_m:
                        a = a_try
                        best_m = M_try
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PairCD (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
