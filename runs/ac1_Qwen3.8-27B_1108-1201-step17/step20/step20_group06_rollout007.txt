    # Compute heat at each coordinate
    h = np.zeros(n)
    for k in range(size):
        if ba[k] <= 0:
            continue
        for i in range(max(0, k-n+1), min(n, k+1)):
            j = k - i
            if 0 <= j < n:

    a_new = np.maximum(a_new, 0)

    if s > 0:

    return a_new

Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FFT-based softmax gradient descent with momentum (O(n log n) per step)
2. Multi-scale homotopy (solve at increasing n)
3. Spectral domain shaping
4. Coordinate descent polish on peak lags
5. Adaptive beta annealing with restarts

import numpy as np
import time

def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def cross_corr_trunc(p, a):
    """
    Compute h[i] = sum_j p[i+j] * a[j] for valid indices.
    This is the cross-correlation of p and a, truncated.
    p has length 2n-1, a has length n.
    h has length n.
    """
    n = len(a)
    size_p = len(p)  # 2n-1
    # h[i] = sum_j p[i+j] * a[j] where 0<=j<n and 0<=i+j<size_p
    # This is: h = IFFT(conj(FFT(a_padded)) * FFT(p_padded)) ... 
    # Actually, let's think of it as:
    # h[i] = sum_j a[j] * p[i+j]
    # If we define a_rev[j] = a[n-1-j], then:
    # h[i] = sum_j a[n-1-j] * p[i+n-1-j] ... that's not quite right either.
    # 
    # Let's use direct FFT approach:
    # h[i] = sum_j a[j] * p[i+j]
    # = (a * p_rev)[i + ?] ... 
    # 
    # Actually, h[i] = sum_j a[j] * p[i+j] is the cross-correlation of a and p at lag i.
    # cross_corr(a, p)[i] = sum_j a[j] * conj(p[j+i]) (for real: sum_j a[j]*p[j+i])
    # which is exactly h[i].
    # 
    # We can compute this via:
    # h = IFFT(conj(FFT(a, N)) * FFT(p, N)) for N >= len(a) + len(p) - 1
    # But we need the "full" cross-correlation, then take the first n elements.
    
    N = 1 << (n + size_p - 1 - 1).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    # cross-correlation: IFFT(conj(A) * P)
    result = np.fft.irfft(np.conj(A) * P, N)
    return result[:n]

def softmax_grad_step(a, beta, top_k=None):
    """
    Compute the smoothed gradient of max(conv(a,a)).
    Returns gradient (length n) and the autocorrelation.
    """
    n = len(a)
    size = 2*n - 1
    ba = fast_conv(a)
    
    if top_k is not None and top_k < size:
        idx = np.argsort(ba)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = ba[idx]
        b_max = ba[idx].max()
    else:
        b_use = ba
        b_max = ba.max()
    
    # Softmax weights
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -100, 0)
    exp_v = np.exp(shifted)
    s = exp_v.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_v / s
    
    # Gradient: g[i] = 2 * sum_k p[k] * a[k-i] (valid range)
    g = 2.0 * cross_corr_trunc(p, a)
    
    return g, ba

def nesterov_softmax(a0, time_limit, beta_start=50.0, beta_end=2000.0,
                      beta_growth=1.005, lr_init=0.1, top_k_ratio=0.05,
                      momentum=0.9, verbose=True, seed=42):
    """Nesterov accelerated gradient descent with softmax smoothing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    beta = beta_start
    top_k = max(10, int(n * top_k_ratio))
    lr = lr_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute gradient
        g, ba = softmax_grad_step(a, beta, top_k)
        
        # Nesterov step
        lr_adj = lr / (1.0 + it * 0.001)  # decay
        v = momentum * v - lr_adj * g
        a_new = a + v
        
        # Project to simplex
        a_new = simplex_project(a_new)
        
        # Check improvement
        ba_new = fast_conv(a_new)
        m_new = ba_new.max()
        
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            # Slightly increase lr
            lr = min(lr * 1.05, lr_init * 2)
        else:
            lr = max(lr * 0.9, lr_init * 0.1)
        
        a = a_new
        
        # Anneal beta
        if it % 100 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  NS it=%d (%.1fs): J=%.6f beta=%d lr=%.4f" % (it, elapsed, J, int(beta), lr))

            rng = np.random.default_rng(it)
            noise = rng.normal(0, 0.01, n) * best_a
            a = best_a + noise
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            v = np.zeros(n)
            beta = max(beta * 0.5, beta_start)
            lr = lr_init
    
    return best_a, best_m

    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        theta = (it - 1) / t_new
        y = a + theta * (a - a_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        
        # Compute gradient
        size = 2 * n - 1
        ba_y = fast_conv(y)
        if top_k < size:
            idx = np.argsort(ba_y)[::-1][:top_k]
            b_use = np.zeros(size)
            b_use[idx] = ba_y[idx]
            b_max = ba_y[idx].max()
        else:
            b_use = ba_y
            b_max = ba_y.max()
        
        shifted = beta * (b_use - b_max)
        shifted = np.clip(shifted, -200.0, 0.0)
        exp_v = np.exp(shifted)
        s = exp_v.sum()
        if s < 1e-300:
            p = np.ones(size) / size
        else:
            p = exp_v / s
        
        g = 2.0 * cross_corr_full(y, p)
        
        # Gradient step with momentum
        v = momentum * v - lr * g
        a_new = y + v
        a_new = simplex_project(a_new)
        
        # Evaluate
        ba_new = fast_conv(a_new)
        m_new = ba_new.max()
        obj_new = 2.0 * n * m_new
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            lr = min(lr * 1.03, lr_init * 3.0)
        else:
            stall += 1
            if stall % 50 == 0:
                lr = max(lr * 0.9, lr_init * 0.1)
        
        a_prev = a
        a = a_new
        
        if it % 80 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        if stall > 3000:
            # Restart
            a = norm_sum1(best_a + rng.normal(0, 0.015, n) * best_a)
            v = np.zeros(n)
            beta = max(beta * 0.4, beta_start)
            lr = lr_init
            stall = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print("    NSWM restart it=%d (%.1fs): J=%.6f" % (it, elapsed, best_obj))
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print("    NSWM it=%d (%.1fs): J=%.6f beta=%d lr=%.5f" % (it, elapsed, best_obj, int(beta), lr))
    
    return best_a, best_obj

def spectral_shape(a0, time_limit, steps=50, alpha=0.05, verbose=False):
    """Shape the sequence in spectral domain to reduce peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    rng = np.random.default_rng(123)
    
    for step in range(steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # Identify frequency bins contributing most to time-domain peaks
        # Reduce high-power bins
        target = np.median(power[1:]) if len(power) > 1 else power[0]
        
        A_new = A.copy()
        for k in range(1, len(A_new)):
            ratio = power[k] / max(target, 1e-20)
            if ratio > 1.5:
                A_new[k] *= np.sqrt(max(0.5, 1.0 - alpha * (ratio - 1.0)))
            elif ratio < 0.3:
                A_new[k] *= np.sqrt(min(2.0, 1.0 + alpha * 0.5 * (0.3 - ratio)))
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = norm_sum1(a_new)
        
        obj = 2.0 * n * fast_conv(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        elif step % 10 == 0:
            # Add random phase perturbation
            A_pert = A * np.exp(1j * rng.standard_normal(len(A)) * 0.03)
            a_pert = np.fft.irfft(A_pert, fft_size)[:n]
            a_pert = norm_sum1(a_pert)
            obj_pert = 2.0 * n * fast_conv(a_pert).max()
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert.copy()
                a = a_pert
        
        alpha *= 0.98
    
    return best_a, best_obj

def coord_descent_polish(a0, time_limit, verbose=True, seed=99):
    """Coordinate descent focusing on peak-reducing moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        ba = fast_conv(a)
        size = 2 * n - 1
        
        # Focus on top peak lags
        top_lags = np.argsort(ba)[::-1][:5]
        improved = False
        
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_lo = max(0, k_int - n + 1)
            i_hi = min(n - 1, k_int)
            
            # Find the pair (i, k-i) with maximum product
            idx = np.arange(i_lo, i_hi + 1)
            products = a[idx] * a[k_int - idx]
            if len(products) == 0:
                continue
            i_max = int(idx[np.argmax(products)])
            j_max = k_int - i_max
            
            # Try reducing mass at i_max or j_max
            for src in [i_max, j_max]:
                if a[src] < 1e-10:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    dest = int(rng.integers(n))
                    if dest == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dest] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial / s_t
                    obj_t = 2.0 * n * fast_conv(a_trial).max()
                    if obj_t < best_obj - 1e-14:
                        a = a_trial
                        best_obj = obj_t
                        best_a = a_trial.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Exploratory move
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = norm_sum1(a)
                    obj_new = 2.0 * n * fast_conv(a).max()
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    CD (%.1fs): J=%.6f stall=%d" % (elapsed, best_obj, stall))
    
    return best_a, best_obj
