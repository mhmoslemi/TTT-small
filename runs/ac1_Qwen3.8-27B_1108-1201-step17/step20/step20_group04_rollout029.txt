def adam_subgrad(a0, time_limit, beta_start=200.0, beta_end=8000.0,

    """Adam-optimized subgradient with adaptive smoothing."""
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Adam state
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t_adam = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    beta = beta_start
    top_k = max(10, int(n * top_k_ratio))

            break
        
        # Adaptive lr (decay over time)
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        lr = lr_start * (lr_end / lr_start) ** progress
        
        # Compute gradient
        grad, p, c = compute_grad(x, beta, top_k)
        
        # Adam update
        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)
        
        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x_new = simplex_project(x_new)
        
        # Evaluate
        c_new = fast_autocorr(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-18:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x = x_new
        
        # Grow beta periodically
        if it % 100 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        # Restart if stalled
        if stall > restart_every:
            stall = 0
            t_adam = 0
            m[:] = 0
            v[:] = 0
            beta = max(beta * 0.5, 50.0)
            # Structured restart

            x_r = np.maximum(x_r, 1e-10)
            x = x_r / x_r.sum()
            if verbose:

                J = 2.0 * n * best_m

def spectral_repair(a0, time_limit, verbose=True, seed=55):
    """Spectral peak reduction: modify frequency domain to reduce autocorrelation peaks."""
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    # FFT size
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current autocorrelation
        c = fast_autocorr(x)
        c_max = c.max()
        
        if c_max < best_m - 1e-18:
            best_m = c_max
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 2000:
            break
        
        # Find worst peak
        k_peak = int(np.argmax(c))
        
        # Compute FFT
        A = np.fft.rfft(x, fft_size)
        P = np.abs(A) ** 2
        
        # The peak at k_peak gets contribution from frequency m proportional to:
        # P[m] * cos(2*pi*m*k_peak/fft_size)
        # To reduce c[k_peak], we want to reduce P[m] where cos(...) > 0
        freqs = np.arange(len(P))
        phase = 2 * np.pi * freqs * k_peak / fft_size
        cos_term = np.cos(phase)
        
        # Attenuate frequencies that contribute positively to the peak
        attenuation = 1.0 - 0.01 * np.maximum(cos_term, 0)
        P_new = P * (attenuation ** 2)  # modify amplitude squared
        
        # Convert back: A_new has same phase as A, modified magnitude
        A_new = A * (attenuation)  # this modifies the amplitude
        # Actually, we want |A_new|^2 = P_new, so |A_new| = sqrt(P_new)
        # Keep the phase of A
        phase_A = np.angle(A)
        mag_new = np.sqrt(P_new)
        A_new = mag_new * np.exp(1j * phase_A)
        
        # Transform back
        x_new = np.fft.irfft(A_new, fft_size)[:n]
        x_new = np.maximum(x_new, 0.0)
        s = x_new.sum()
        if s < 1e-15:
            continue
        x_new = x_new / s
        
        c_new = fast_autocorr(x_new)
        if c_new.max() < best_m:
            x = x_new
        elif rng.random() < 0.1:
            x = x_new  # occasional acceptance of slightly worse for exploration
    
    return best_a, best_m

def adam_subgrad(a0, time_limit, beta_start=300.0, beta_end=10000.0,
                 beta_growth=1.003, lr_start=0.03, lr_end=0.001,
                 top_k_ratio=0.008, restart_every=1200, verbose=True, seed=42):
    """Adam-adaptive subgradient descent with log-sum-exp smoothing."""
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    b1 = 0.85
    b2 = 0.999
    eps = 1e-10

    beta = beta_start
    top_k = max(20, int(n * top_k_ratio))
    stall = 0

    for it in range(50000000):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        lr = lr_start * (lr_end / lr_start) ** progress

        grad, p, c = compute_grad(x, beta, top_k)

        t_adam += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1 - b1 ** t_adam)
        v_hat = v / (1 - b2 ** t_adam)

        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x_new = simplex_project(x_new)

        c_new = fast_autocorr(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-18:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x = x_new

        if it % 200 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_adam = 0
            m[:] = 0
            v[:] = 0
            beta = max(100.0, beta * 0.4)
            # Structured restart: combine anti-symmetric and random
            anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            noise_scale = 0.015 / (1.0 + 0.1 * (it // restart_every))
            pert = noise_scale * (0.6 * anti + 0.4 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x = x_r / x_r.sum()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    Adam restart ({}s): J={}".format(round(elapsed, 1), round(J, 7)))

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Adam t={} ({}s): J={}, beta={}".format(it, round(elapsed, 1), round(J, 7), round(beta, 0)))

    return best_a, best_m


def mirror_descent_opt(a0, time_limit, beta=1000.0, lr_start=0.005, lr_end=0.0001,
                       top_k=100, restart_every=400, verbose=True, seed=77):
    """Mirror descent with entropy regularization (multiplicative weights update)."""
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    for it in range(50000000):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        lr = lr_start * (lr_end / lr_start) ** progress

        grad, p, c = compute_grad(x, beta, top_k)

        # Multiplicative update
        log_update = -lr * grad
        log_update = np.clip(log_update, -30, 30)
        x_new = x * np.exp(log_update)
        s = x_new.sum()
        if s < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / s

        c_new = fast_autocorr(x_new)
        m_new = c_new.max()
        if m_new < best_m - 1e-18:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x = x_new

        if stall > restart_every:
            stall = 0
            beta = max(200.0, beta * 0.6)
            # Multiplicative noise restart
            noise = rng.normal(0, 0.02, n)
            x_r = best_a * np.exp(noise)
            x = x_r / x_r.sum()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    Mirror restart ({}s): J={}".format(round(elapsed, 1), round(J, 7)))

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Mirror t={} ({}s): J={}".format(it, round(elapsed, 1), round(J, 7)))

    return best_a, best_m


def nesterov_opt(a0, time_limit, beta_start=500.0, beta_end=15000.0,
                 beta_growth=1.005, eta_base=0.4, top_k_ratio=0.005,
                 restart_every=1000, verbose=True, seed=42):
    """Nesterov accelerated projected gradient with log-sum-exp smoothing."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(20, int(n * top_k_ratio))

    for t in range(50000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, c = compute_grad(y, beta, top_k)

        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.05)
        # Adaptive: reduce eta if we're making progress slowly
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        eta *= (1.0 - 0.5 * progress)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_m - 1e-18:
            best_m = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 100 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            # Diverse restart strategies
            strategy = rng.integers(3)
            if strategy == 0:
                # Anti-symmetric perturbation
                anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
                anti -= anti.mean()
                noise_scale = 0.01
                x_r = best_a + noise_scale * anti * best_a
            elif strategy == 1:
                # Random multiplicative
                noise = rng.normal(0, 0.01, n)
                x_r = best_a * np.exp(noise)
            else:
                # Block perturbation
                block_size = n // 10
                block = rng.integers(0, n - block_size)
                x_r = best_a.copy()
                x_r[block:block + block_size] *= rng.uniform(0.7, 1.3)
            x_r = np.maximum(x_r, 1e-12)
            x = x_r / x_r.sum()
            x_prev = x.copy()
            beta = max(beta * 0.5, 50.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    Nest restart t={} ({}s): J={}".format(t, round(elapsed, 1), round(J, 7)))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Nest t={} ({}s): J={}, beta={}".format(t, round(elapsed, 1), round(J, 7), round(beta, 0)))

    return best_a, best_m


def targeted_local_search(a0, time_limit, verbose=True, seed=33):
    """Targeted local search: for each top peak, try optimal small perturbations."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 5000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-18:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Focus on top peaks
        n_top = min(20, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]

        improved = False

        for k in top_idx[:10]:
            if improved:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]

            # Try reducing top contributing pairs
            top_pairs = np.argsort(products)[::-1][:5]
            for pi in top_pairs:
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n or partner == src_i:
                    continue

                # Determine larger
                if a[src_i] >= a[partner]:
                    reduce_i = src_i
                else:
                    reduce_i = partner
                if a[reduce_i] < 1e-14:
                    continue

                # Compute pressure landscape
                pressure = np.zeros(n)
                for k2 in top_idx[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        pressure[idx2] += a[k2i - idx2]
                        pressure[k2i - idx2] += a[idx2]

                # Find low-pressure destinations
                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_idx[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2 + 1] = True

                candidates = np.where(~in_peak)[0]
                if len(candidates) < 100:
                    candidates = np.arange(n)
                mask = candidates != reduce_i
                candidates = candidates[mask]

                if len(candidates) == 0:
                    continue

                cand_pressure = pressure[candidates]
                n_try = min(8, len(candidates))
                try_dsts = candidates[np.argsort(cand_pressure)[:n_try]]

                for dst in try_dsts:
                    dst_int = int(dst)
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.03, 0.05]:
                        amt = a[reduce_i] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[reduce_i] -= amt
                        a_trial[dst_int] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-18:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            # Random exploration when stuck
            if no_improve > 500:
                found = False
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.03)
                        a[i] -= a[i] * frac
                        a[j] += a[i] * frac / (1 - frac)  # approximate
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                found = True
                                break
                if not found:
                    no_improve += 100

        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LocalSearch ({}s): J={} stall={}".format(round(elapsed, 1), round(J, 7), no_improve))

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def generate_diverse_init(n, seed, variant=0):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if variant == 0:
        # Power law decay
        a = (1.0 - x) ** 1.2 + 0.05
    elif variant == 1:
        # Gaussian-like bump
        center = 0.4
        a = np.exp(-((x - center) ** 2) / 0.05) + 0.05
    elif variant == 2:
        # Anti-symmetric modulated
        base = 0.5 + 0.5 * np.sin(2 * np.pi * 3 * x)
        a = base * (1 - x) + 0.1
    elif variant == 3:
        # Chirp magnitude
        a = np.abs(np.sin(np.pi * x * x * 10)) + 0.1
    elif variant == 4:
        # Two-bump
        a = np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02) + 0.05
    else:
        # Random smooth
        a = np.ones(n)
        n_modes = 5
        for _ in range(n_modes):
            freq = rng.uniform(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.3, 1.0)
            a += amp * np.sin(2 * np.pi * freq * x + phase)
        a = np.abs(a) + 0.1

    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s


def propose_candidate(budget_s=1800.0):
    """Main search: combines Adam, mirror descent, Nesterov, and local search."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent loaded: n={}, score={}".format(len(parent), repr(p_score)))
    except Exception:
        pass

    if best_seq is None:
        n_base = 6700
        a_init = generate_diverse_init(n_base, 42, variant=0)
        best_seq = a_init
        best_score = evaluate_safe(a_init)
        print("Default init: n={}, score={}".format(n_base, repr(best_score)))

    n0 = len(best_seq)
    print("Starting: n={}, score={}".format(n0, repr(best_score)))

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print("  *** NEW BEST [{}]: {:.7f} (n={}, t={:.1f}s)".format(
                label, sc, len(a_norm), elapsed))

    # === Phase 1: Adam subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Adam subgradient (n={}) ===".format(n0))
    t1 = min(remaining * 0.22, 350.0)
    if t1 > 60:
        a_out, _ = adam_subgrad(norm_sum1(best_seq), t1,
                                beta_start=300.0, beta_end=10000.0,
                                beta_growth=1.003, lr_start=0.03, lr_end=0.001,
                                top_k_ratio=0.008, restart_every=1200,
                                verbose=True, seed=1)
        try_update(a_out, "adam1")

    # === Phase 2: Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov (n={}) ===".format(n_best))
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 60:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t2,
                                beta_start=500.0, beta_end=15000.0,
                                beta_growth=1.005, eta_base=0.4,
                                top_k_ratio=0.005, restart_every=1000,
                                verbose=True, seed=2)
        try_update(a_out, "nest1")

    # === Phase 3: Mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Mirror descent (n={}) ===".format(n_best))
    t3 = min(remaining * 0.18, 250.0)
    if t3 > 40:
        a_out, _ = mirror_descent_opt(norm_sum1(best_seq), t3,
                                      beta=1000.0, lr_start=0.005, lr_end=0.0001,
                                      top_k=100, restart_every=400,
                                      verbose=True, seed=3)
        try_update(a_out, "mirror1")

    # === Phase 4: Targeted local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Targeted local search (n={}) ===".format(n_best))
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        a_out, _ = targeted_local_search(norm_sum1(best_seq), t4, verbose=True, seed=4)
        try_update(a_out, "local1")

    # === Phase 5: Adam again with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Adam refined (n={}) ===".format(n_best))
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 40:
        a_out, _ = adam_subgrad(norm_sum1(best_seq), t5,
                                beta_start=2000.0, beta_end=20000.0,
                                beta_growth=1.002, lr_start=0.01, lr_end=0.0005,
                                top_k_ratio=0.003, restart_every=800,
                                verbose=True, seed=5)
        try_update(a_out, "adam2")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-length (n={}) ===".format(n_best))

    lengths_to_try = []
    for delta in [400, 800, 1200, -500]:
        nl = n_best + delta
        if 3000 <= nl <= 12000:
            lengths_to_try.append(nl)
    lengths_to_try = list(set(lengths_to_try))[:4]

    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print("  Trying n={}".format(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.30, 180.0)
        if t6a > 40:
            a_out, _ = adam_subgrad(a_long, t6a,
                                    beta_start=300.0, beta_end=8000.0,
                                    beta_growth=1.004, lr_start=0.02, lr_end=0.001,
                                    top_k_ratio=0.008, restart_every=1000,
                                    verbose=True, seed=10 + new_n % 100)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t6b = min(remaining * 0.25, 100.0)
            if t6b > 20:
                a_try = interpolate_sequence(best_seq, new_n)
                a_out, _ = targeted_local_search(a_try, t6b, verbose=False, seed=20 + new_n % 100)
                try_update(a_out, "ml_local_n{}".format(new_n))

    # === Phase 7: Final Nesterov burst ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final Nesterov (n={}) ===".format(n_best))
    t7 = min(remaining * 0.20, 150.0)
    if t7 > 30:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t7,
                                beta_start=3000.0, beta_end=30000.0,
                                beta_growth=1.003, eta_base=0.2,
                                top_k_ratio=0.003, restart_every=600,
                                verbose=True, seed=7)
        try_update(a_out, "nest_final")

    # === Phase 8: Final local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final local search (n={}) ===".format(n_best))
    t8 = min(remaining * 0.35, 120.0)
    if t8 > 20:
        a_out, _ = targeted_local_search(norm_sum1(best_seq), t8, verbose=True, seed=8)
        try_update(a_out, "local_final")

    # === Phase 9: Mirror descent final ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        t9 = min(remaining * 0.6, 80.0)
        if t9 > 20:
            print("\n=== Phase 9: Final mirror (n={}) ===".format(n_best))
            a_out, _ = mirror_descent_opt(norm_sum1(best_seq), t9,
                                          beta=5000.0, lr_start=0.002, lr_end=0.0001,
                                          top_k=50, restart_every=300,
                                          verbose=False, seed=9)
            try_update(a_out, "mirror_final")

    # === Phase 10: One more Adam if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        t10 = min(remaining * 0.7, 100.0)
        if t10 > 20:
            print("\n=== Phase 10: Final Adam (n={}) ===".format(len(best_seq)))
            a_out, _ = adam_subgrad(norm_sum1(best_seq), t10,
                                    beta_start=5000.0, beta_end=30000.0,
                                    beta_growth=1.002, lr_start=0.005, lr_end=0.0003,
                                    top_k_ratio=0.003, restart_every=500,
                                    verbose=False, seed=11)
            try_update(a_out, "adam_ultimate")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={}, n={}, time={:.1f}s".format(
            repr(best_score), len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1((1.0 - x) ** 1.5 + 0.1)
        best_score = evaluate_safe(best_seq)

    # Ensure validity
    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)

    return [float(x) for x in best_seq]
