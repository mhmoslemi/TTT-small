    SCIPY_OK = False

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s

def fft_conv(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c

def fft_conv2(a, b):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c

def solve_fw_lp(a, active_idx, time_limit=5.0):
    """Solve the Frank-Wolfe LP subproblem.
    Returns the optimal g (step direction on the simplex).
    """
    if not SCIPY_OK:
        return None
    n = a.size
    K = active_idx.size
    if K == 0:
        return np.ones(n) / n
    
    # Build constraint matrix (sparse)
    row_parts = []
    col_parts = []
    val_parts = []
    for r, k in enumerate(active_idx):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start > j_end:
            continue
        cols = np.arange(j_start, j_end + 1, dtype=np.int32)
        vals = a[k - cols]  # a[k - j] for j in cols
        if np.any(vals > 0):
            row_parts.append(np.full(cols.size, r, dtype=np.int32))
            col_parts.append(cols)
            val_parts.append(vals.astype(np.float64))
    
    if not row_parts:
        return np.ones(n) / n
    
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    
    # A_ub: [constraint_matrix | -1] (the -1 for c)
    A_con = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))
    neg_c = scipy.sparse.csc_matrix(-np.ones((K, 1)))
    A_ub = scipy.sparse.hstack([A_con, neg_c]).tocsr()
    b_ub = np.zeros(K)
    
    # Equality: sum(g) = 1, c is free
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize c
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0.0, None)] * n + [(0.0, None)]  # c >= 0 since all a_i >= 0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='HiGHS',
                      options={'disp': False, 'time_limit': time_limit})
    except Exception:
        return None
    
    if res is None or not res.success:
        return None
    
    g = res.x[:n]
    g = np.where(np.isfinite(g) & (g > 0), g, 0.0)
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs

One more thought: the "away step" in Frank-Wolfe. The idea is:
- Instead of just moving toward `v` (the LP minimizer), also consider moving away from the vertex `w` that has the largest "pressure."
- The update is: `a <- (1-t/a)_a * a + (t/a) * v - (t/w) * w` where `t/a + t/w <= 1`.
- This is equivalent to: `a <- a + t/a * (v - a) - t/w * (a - w)` = `a + t/a*(v-a) - t/w*(a-w)`.

In practice, the away step helps when the current point is "stuck" near a vertex that's not in the optimal solution.

For simplicity, I'll implement the basic FW first and add away steps if there's time.

Actually, let me think about this more carefully. The standard FW update is:
`a_{t+1} = (1 - gamma_t) * a_t + gamma_t * v_t`

where `v_t` is the LP solution and `gamma_t = 2/(t+2)` (or from line search).

The away step FW (ASF) is:
1. Find `v_t` (toward) as before
2. Find `w_t` = vertex index where `s_j` is maximum (away direction)

4. This is a 2D line search

for minimizing 2*n*max(convolve(a,a))/(sum(a))^2 over non-negative sequences.

    from scipy.optimize import linprog, minimize

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def fft_conv(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def fft_conv2(a, b):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if b.size != n:
        b = np.resize(b, n)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c

def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    new_n = int(max(1, new_n))

        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)


def lse_obj_grad(a, beta):
    """Log-sum-exp smoothed objective and gradient for max_k (a*a)_k."""
    n = a.size
    if n == 0:
        return 0.0, np.array([], dtype=np.float64)
    c = fft_conv(a)
    size = c.size
    m = float(np.max(c))
    shifted = beta * (c - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0 or not math.isfinite(se):
        f = 1e100
        grad = np.zeros(n)
        return f, grad
    p = e / se
    # grad = 2 * (p * a) cross-correlation
    # (p convolve reversed a) gives the gradient
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    f = m + math.log(se) / beta
    if not math.isfinite(f):
        f = 1e100
        grad = np.zeros(n)
    return f, grad


def lse_optimize(a0, beta, maxiter=200, time_limit=300.0):
    """Optimize using L-BFGS on the log-sum-exp smoothed objective."""
    if not SCIPY_OK:
        return norm_sum1(a0)
    a = norm_sum1(a0)
    n = a.size
    if n <= 1:
        return a
    t0 = time.monotonic()

    def obj_wrapper(v):
        if time.monotonic() - t0 > time_limit:
            return 1e100, np.zeros(v.size)
        a_tmp = np.exp(v - np.max(v))
        a_tmp = a_tmp / a_tmp.sum()
        a_tmp = np.maximum(a_tmp, 0.0)
        s = a_tmp.sum()
        a_tmp = a_tmp / s
        f, ga = lse_obj_grad(a_tmp, beta)
        # Convert to v-space gradient via softmax Jacobian
        mean = float(np.dot(a_tmp, ga))
        gv = a_tmp * (ga - mean)
        gv = np.where(np.isfinite(gv), gv, 0.0)
        return f, gv

    v0 = np.log(np.maximum(a, 1e-30))
    v0 = v0 - np.max(v0)

    try:
        res = minimize(obj_wrapper, v0, method='L-BFGS-B', jac=True,
                       options={'maxiter': int(maxiter), 'ftol': 1e-12,
                                'gtol': 1e-8, 'disp': False})
        if res.success and res.x is not None:
            a_out = np.exp(res.x - np.max(res.x))
            a_out = a_out / a_out.sum()
            a_out = np.maximum(a_out, 0.0)
            a_out = a_out / a_out.sum()
            return a_out
    except Exception:
        pass
    return a


def solve_fw_lp(a, active_idx, time_limit=5.0):
    """Solve the Frank-Wolfe LP: min c s.t. (a*g)[k] <= c for active k, g on simplex."""
    if not SCIPY_OK:
        return None
    n = a.size
    K = active_idx.size
    if K == 0:
        return np.ones(n) / n
    if K > 5000:
        active_idx = active_idx[:5000]
        K = 5000

    row_parts = []
    col_parts =
    val_parts = []
    for r_idx, k in enumerate(active_idx):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start > j_end:
            continue
        cols = np.arange(j_start, j_end + 1, dtype=np.int32)
        vals = a[k - cols]
        mask = vals > 1e-15
        if np.any(mask):
            r = np.full(int(np.sum(mask)), r_idx, dtype=np.int32)
            row_parts.append(r)
            col_parts.append(cols[mask])
            val_parts.append(vals[mask].astype(np.float64))

    if not row_parts:
        return np.ones(n) / n

    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)

    A_con = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))
    neg_c = scipy.sparse.csc_matrix(-np.ones((K, 1)))
    A_ub = scipy.sparse.hstack([A_con, neg_c]).tocsr()
    b_ub = np.zeros(K)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='HiGHS',
                      options={'disp': False, 'time_limit': float(time_limit)})
    except Exception:
        return None

    if res is None or not getattr(res, 'success', False):
        return None

    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.where(np.isfinite(g) & (g > 0), g, 0.0)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs


def subgradient_direction(a, c, active_idx, top_count=50):
    """Compute subgradient-based FW direction (fallback when LP fails)."""
    n = a.size
    top = active_idx[:min(top_count, active_idx.size)]
    s = np.zeros(n)
    for k in top:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            # s[j] += 2*a[k-j] for j in [j_start, j_end]
            # a[k-j] for j=j_start is a[k-j_start], for j=j_end is a[k-j_end]
            # We need the slice a[k-j_start], a[k-j_start-1], ..., a[k-j_end]
            # = a[k-j_end : k-j_start+1 : -1] reversed
            # = a[k-j_start:-k+j_end] ... let me just do it directly
            idx = np.arange(j_start, j_end + 1)
            s[idx] += 2.0 * a[k - idx]
    # FW direction is e_{argmin s}
    g = np.zeros(n)
    g[int(np.argmin(s))] = 1.0
    return g


def line_search_fw(a, g, c, fg, cg, mask=None):
    """Find best step size for FW update."""
    n = a.size
    best_alpha = 0.0
    best_m = float(np.max(c)) if mask is None else float(np.max(c[mask]))

    # Build candidate alphas
    candidates = [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]

    # Add optimal t for top active constraints
    if mask is None:
        top_idx = np.argsort(c)[::-1][:10]
    else:
        masked_c = np.where(mask, c, -np.inf)
        top_idx = np.argsort(masked_c)[::-1][:10]

    for k in top_idx:
        k = int(k)
        ck = c[k]
        fk = fg[k]
        gk = cg[k]
        denom = ck - 2.0 * fk + gk
        if denom > 1e-15:
            t_opt = (ck - fk) / denom
            if 0.001 < t_opt < 1.0:
                candidates.append(float(t_opt))

    for alpha in candidates:
        alpha = max(0.0, min(1.0, float(alpha)))
        one = 1.0 - alpha
        val = one * one * c + 2.0 * alpha * one * fg + alpha * alpha * cg
        val = np.maximum(val, 0.0)
        if mask is None:
            m = float(np.max(val))
        else:
            m = float(np.max(val[mask]))
        if math.isfinite(m) and m < best_m:
            best_m = m
            best_alpha = alpha

    return best_alpha, best_m


def frank_wolfe_main(a0, budget, max_active=3000, lp_time=4.0,
                     lp_every=2, verbose=True):
    """Main Frank-Wolfe loop with LP directions and line search."""
    t0 = time.monotonic()
    n = a0.size
    a = norm_sum1(a0)
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    no_improve = 0
    t = 0

    while time.monotonic() - t0 < budget - 10 and no_improve < 120:
        t += 1
        time_left = budget - (time.monotonic() - t0)
        if time_left < 40:
            break

        # Get active set
        K = min(max_active, 2 * n - 1)
        active_idx = np.argsort(c)[::-1][:K]

        # Get direction
        g = None
        use_lp = (t % lp_every == 0) and time_left > 60
        if use_lp:
            g = solve_fw_lp(a, active_idx, time_limit=lp_time)
        if g is None:
            g = subgradient_direction(a, c, active_idx, top_count=80)

        if g is None:
            break

        # Compute convolutions for line search
        cg = fft_conv(g)
        fg = fft_conv2(a, g)

        # Line search
        alpha, new_M = line_search_fw(a, g, c, fg, cg)

        if alpha <= 0:
            no_improve += 1
            if no_improve > 30 and t > 50:
                # Perturb to escape
                rng = np.random.default_rng(t * 7919 + 42)
                pert = rng.normal(0.0, 0.02, n) * a
                a = a + pert
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                c = fft_conv(a)
                M = float(np.max(c))
                no_improve = 0
            continue

        # Update
        a = (1.0 - alpha) * a + alpha * g
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if s < 1e-15:
            break
        a = a / s
        c = fft_conv(a)
        M = float(np.max(c))

        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            no_improve = 0
            if verbose and (t <= 3 or t % 25 == 0):
                print("  FW t=%d score=%.6f M=%.8f alpha=%.4f K=%d time=%.0fs" % (
                    t, 2.0 * n * best_M, best_M, alpha, K,
                    time.monotonic() - t0))
        else:
            no_improve += 1
            if verbose and (t <= 3 or t % 25 == 0):
                print("  FW total-improve t=%d best_score=%.6f alpha=%.4f time=%.0fs" % (
                    t, 2.0 * n * best_M, alpha, time.monotonic() - t0))

    return best_a


def coordinate_polish(a0, budget, verbose=True):
    """Exact single-coordinate transfer polish."""
    t0 = time.monotonic()
    n = a0.size
    a = norm_sum1(a0)
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    stall = 0
    rnd = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < budget - 3 and stall < 100:
        rnd += 1
        if n <= 1:
            break

        # Find top active positions
        top_k = min(30, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]

        # Compute pressure
        pressure = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                pressure[idx] += c[k] * a[k - idx]

        # Try transfers from high-pressure to low-pressure
        src_candidates = np.argpartition(pressure, -min(10, n))[-min(10, n):]
        dst_candidates = np.argpartition(pressure, min(10, n))[:min(10, n)]

        best_cand = None
        for srr in src_candidates:
            s = int(srr)
            if a[s] < 1e-14:
                continue
            for drr in dst_candidates:
                d = int(drr)
                if d == s:
                    continue
                for frac in (0.005, 0.01, 0.02, 0.05, 0.1):
                    tmove = a[s] * frac
                    if tmove <= 1e-16:
                        continue
                    tmove = min(tmove, a[s] * 0.99)
                    # Apply move: a[s]-=tmove, a[d]+=tmove
                    # Update convolution using rank-1 formula
                    c_new = c.copy()
                    # (a*a)[k] changes by:
                    # -2*tmove*(a[k-s] + a[k-d]) + tmove^2*(delta_{k,2s} - 2*delta_{k,s+d} + delta_{k,2d})
                    # More precisely:
                    # New a = a - t*e_s + t*e_d
                    # New (a*a)[k] = (a*a)[k] - 2t*a[k-s] + 2t*a[k-d] - 2t^2*delta... 
                    # Let me use the direct formula:
                    # (a-t*es+te_d) * (a-t*es+te_d)
                    # = a*a - t*a*e_s - t*e_s*a + t^2*e_s*e_s
                    #   + t*a*e_d + t*e_d*a - 2t^2*e_s*e_d + t^2*e_d*e_d
                    # (a*e_s)[k] = a[k] (delta at k=0 shifted by s) = a_s if k=s
                    # Actually (a * e_s)[k] = a[k-s] (if in bounds)
                    # So (a*e_s + e_s*a)[k] = a[k-s] + a[s-k] = a[k-s] + a[k-s] = 2*a[k-s]
                    # (e_s * e_s)[k] = 1 if k=2s
                    # (e_d * e_d)[k] = 1 if k=2d
                    # (e_s * e_d)[k] = 1 if k=s+d
                    for k in range(2 * n - 1):
                        val = c_new[k]
                        # subtract 2t * (a*e_s)[k] and 2t * (e_s*a)[k]... 
                        # (a*e_s)[k] = a[k-s] if 0<=k-s<n
                        if 0 <= k - s < n:
                            val -= 2.0 * tmove * a[k - s]
                        if 0 <= k - d < n:
                            val += 2.0 * tmove * a[k - d]
                        if k == 2 * s:
                            val += tmove * tmove
                        if k == 2 * d:
                            val += tmove * tmove
                        if k == s + d:
                            val -= 2.0 * tmove * tmove
                        # Cross terms: (-t es)(t ed) + (t ed)(-t es) = -2t^2 es*ed
                        # Already handled above
                        c_new[k] = max(0.0, val)

                    m_new = float(np.max(c_new))
                    if m_new < M - 1e-15:
                        if best_cand is None or m_new < best_cand[0]:
                            best_cand = (m_new, s, d, tmove)

        # Also try random moves
        for _ in range(30):
            s = int(rng.integers(n))
            d = int(rng.integers(n))
            if s == d or a[s] < 1e-14:
                continue
            frac = float(rng.uniform(0.003, 0.05))
            tmove = a[s] * frac
            tmove = min(tmove, a[s] * 0.99)
            if tmove <= 1e-16:
                continue
            c_new = c.copy()
            for k in range(2 * n - 1):
                val = c_new[k]
                if 0 <= k - s < n:
                    val -= 2.0 * tmove * a[k - s]
                if 0 <= k - d < n:
                    val += 2.0 * tmove * a[k - d]
                if k == 2 * s:
                    val += tmove * tmove
                if k == 2 * d:
                    val += tmove * tmove
                if k == s + d:
                    val -= 2.0 * tmove * tmove
                c_new[k] = max(0.0, val)
            m_new = float(np.max(c_new))
            if m_new < M - 1e-15:
                if best_cand is None or m_new < best_cand[0]:
                    best_cand = (m_new, s, d, tmove)

        if best_cand is not None:
            m_new, s, d, tmove = best_cand
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            c = fft_conv(a)
            M = float(np.max(c))
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and (rnd <= 3 or rnd % 30 == 0):
                print("  Coord rnd=%d score=%.6f time=%.0fs" % (
                    rnd, 2.0 * n * best_M, time.monotonic() - t0))
        else:
            stall += 1

    return best_a


def try_update(best_seq, best_score, candidate, label, t_start):
    if candidate is None:
        return best_seq, best_score
    try:
        arr = np.asarray(candidate, dtype=np.float64)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        s = float(arr.sum())
        if arr.size == 0 or s < 0.01:
            return best_seq, best_score
        arr = arr / s
        sc = safe_eval(arr)
        if math.isfinite(sc) and sc < best_score - 1e-15:
            print("  NEW BEST [%s]: %.6f n=%d t=%.0fs" % (
                label, sc, arr.size, time.monotonic() - t_start))
            return arr.copy(), float(sc)
    except Exception:
        pass
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search function."""
    t_start = time.monotonic()
    total_budget = max(60.0, min(float(budget_s) - 70.0, 1730.0))

    # Get initial sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.where(np.isfinite(parent), parent, 0.0)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and float(parent.sum()) > 0.01:
            a_work = parent / parent.sum()
        else:
            a_work = default_sequence(5000)
    except Exception:
        a_work = default_sequence(5000)

    # Cap working length
    if a_work.size > 9000:
        a_work = interpolate_sequence(a_work, 7000)

    seq_best = a_work.copy()
    best_score = safe_eval(seq_best)
    if not math.isfinite(best_score):
        best_score = 2.0 * a_work.size * float(np.max(fft_conv(a_work)))

    print("Start: n=%d score=%.6f" % (a_work.size, best_score))

    n = a_work.size

    # === Phase 1: L-BFGS on log-sum-exp ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if SCIPY_OK and time_left > 300:
        phase_budget = min(time_left * 0.25, 400.0)
        print("\n=== Phase 1: L-BFGS log-sum-exp (budget=%.0fs) ===" % phase_budget)
        for beta in [20.0, 50.0, 100.0]:
            time_left = total_budget - (time.monotonic() - t_start)
            if time_left < 100:
                break
            a_new = lse_optimize(a_work, beta, maxiter=150,
                                 time_limit=min(phase_budget / 3.0, time_left - 50))
            seq_best, best_score = try_update(seq_best, best_score, a_new,
                                              "lse_b%d" % int(beta), t_start)
            a_work = seq_best.copy() if seq_best.size == n else a_work
            print("  After LSE beta=%d: score=%.6f" % (beta, best_score))

    # === Phase 2: Frank-Wolfe with LP ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 200:
        phase_budget = min(time_left * 0.50, 800.0)
        print("\n=== Phase 2: Frank-Wolfe LP (budget=%.0fs) ===" % phase_budget)
        a_fw = frank_wolfe_main(a_work, phase_budget, max_active=3000,
                                lp_time=4.0, lp_every=2, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_fw, "fw", t_start)

        # If n changed, interpolate
        if seq_best.size != a_work.size:
            a_work = interpolate_sequence(seq_best, 5000)
        else:
            a_work = seq_best.copy()

    # === Phase 3: Coordinate polish ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 100:
        phase_budget = min(time_left * 0.15, 250.0)
        print("\n=== Phase 3: Coordinate polish (budget=%.0fs) ===" % phase_budget)
        a_cp = coordinate_polish(a_work, phase_budget, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_cp, "coord", t_start)
        a_work = seq_best.copy() if seq_best.size == a_work.size else interpolate_sequence(seq_best, a_work.size)

    # === Phase 4: Another FW round on polished result ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 150:
        phase_budget = min(time_left * 0.25, 350.0)
        print("\n=== Phase 4: FW polish (budget=%.0fs) ===" % phase_budget)
        a_fw2 = frank_wolfe_main(a_work, phase_budget, max_active=2500,
                                 lp_time=3.0, lp_every=2, verbose=False)
        seq_best, best_score = try_update(seq_best, best_score, a_fw2, "fw2", t_start)

    # === Phase 5: Coordinate polish again ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 80:
        phase_budget = min(time_left * 0.15, 150.0)
        print("\n=== Phase 5: Final coord polish (budget=%.0fs) ===" % phase_budget)
        a_cp2 = coordinate_polish(seq_best, phase_budget, verbose=False)
        seq_best, best_score = try_update(seq_best, best_score, a_cp2, "coord2", t_start)

    # === Phase 6: Try a couple different lengths ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 200:
        base_n = seq_best.size
        for target_n in [base_n - 500, base_n + 500]:
            time_left = total_budget - (time.monotonic() - t_start)
            if time_left < 120:
                break
            target_n = int(max(2000, min(target_n, 10000)))
            print("\n  Trying n=%d..." % target_n)
            a_interp = interpolate_sequence(seq_best, target_n)
            if SCIPY_OK and time_left > 180:
                a_new = lse_optimize(a_interp, 50.0, maxiter=100,
                                     time_limit=min(60.0, time_left * 0.3))
                seq_best, best_score = try_update(seq_best, best_score, a_new,
                                                  "len%d_lse" % target_n, t_start)
            time_left = total_budget - (time.monotonic() - t_start)
            if time_left > 80:
                a_fw3 = frank_wolfe_main(a_interp if seq_best.size != target_n
                                         else seq_best,
                                         min(40.0, time_left * 0.4),
                                         max_active=1500, lp_time=3.0,
                                         lp_every=3, verbose=False)
                seq_best, best_score = try_update(seq_best, best_score, a_fw3,
                                                  "len%d_fw" % target_n, t_start)

    # Final validation
    arr = np.asarray(seq_best, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if arr.size == 0 or s < 0.01:
        arr = default_sequence(5000)
    else:
        arr = arr / s
        final_val = safe_eval(arr)
        if math.isfinite(final_val):
            best_score = min(best_score, final_val)

    elapsed = time.monotonic() - t_start
    print("\nFINAL: score=%.6f n=%d elapsed=%.0fs" % (best_score, arr.size, elapsed))
    return [float(x) for x in arr]
