def simplex_project(z):
    """Project onto probability simplex."""

    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_grad_smoothed(a, p):
    """Compute gradient of smoothed max given probability weights p on peaks."""
    n = len(a)
    # grad[j] = 2 * sum_k p[k] * a[k-j]
    # = 2 * conv(p, a_rev)[n-1+j]
    N = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, N)
    A_rev = np.fft.rfft(a[::-1], N)
    grad_conv = np.fft.irfft(P * A_rev, N)
    return 2.0 * grad_conv[n - 1:2 * n - 1]

def nesterov_opt(a0, time_limit, beta_start=30.0, beta_end=5000.0,
                 beta_growth=1.008, eta_base=0.5, top_k_start=None,
                 top_k_end=5, restart_every=1500, seed=42, verbose=False):
    """Nesterov accelerated projected gradient on top-k log-sum-exp."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    v_best = fast_autocorr(x).max()
    best_a = x.copy()
    t0 = time.monotonic()
    beta = float(beta_start)
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    total_iters = 10000000
    
    for t in range(1, total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute smoothed gradient
        c = fast_autocorr(y)
        if top_k_start is not None:
            frac = min(1.0, t / (restart_every * 3))
            top_k = max(top_k_end, int(top_k_start - frac * (top_k_start - top_k_end)))
        else:
            top_k = top_k_end
        
        top_k = min(top_k, len(c))
        idx = np.argsort(c)[::-1][:top_k]
        c_max = c[idx].max()
        shifted = beta * (c[idx] - c_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        total = exp_vals.sum()
        p = np.zeros(len(c))
        if total > 1e-300:
            p_vals = exp_vals / total
            p[idx] = p_vals
        else:
            p[np.argmax(c)] = 1.0
        
        grad = compute_grad_smoothed(y, p)
        
        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            eta = eta_base
        else:
            eta = eta_base / (beta * p.max() * 2.0 + grad_norm)
        eta = min(eta, 0.5)
        
        z = y - eta * grad
        x_new = simplex_project(z)
        
        # Check improvement
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < v_best - 1e-15:
            v_best = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        # Adaptive beta
        if t % 30 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        # Restart if stalled
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.05 / (1.0 + 0.1 * (t // restart_every))
            anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * v_best
                print(f"    Restart t={t} ({elapsed:.1f}s): J={J:.6f}")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * v_best
            print(f"    Nest t={t} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}, stall={stall}")
    
    return best_a, v_best


def simulated_anneal(a0, time_limit, seed=99, verbose=False):
    """Simulated annealing with structured perturbations."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Temperature schedule
    T_start = 0.01
    T_end = 0.0001
    total_steps = 50000  # estimate
    best_no_improve = 0
    max_no_improve = 5000
    
    for step in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if best_no_improve > max_no_improve:
            break
        
        # Temperature (geometric cooling)
        frac = min(1.0, step / total_steps)
        T = T_start * (T_end / T_start) ** frac
        
        # Compute current autocorrelation
        c = fast_autocorr(a)
        c_max = c.max()
        
        # Choose perturbation type
        r = rng.random()
        if r < 0.3:
            # Gaussian perturbation
            scale = T * (1.0 + rng.random())
            pert = rng.normal(0, scale, n)
            a_new = a + pert * a  # multiplicative noise
        elif r < 0.5:
            # Anti-symmetric perturbation
            anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
            scale = T * (0.5 + rng.random())
            a_new = a + scale * anti * a
        elif r < 0.7:
            # Peak-aware move: reduce mass from peak-contributing positions
            # Find the dominant peak
            k_peak = int(np.argmax(c))
            i_start = max(0, k_peak - n + 1)
            i_end = min(n - 1, k_peak)
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_peak - idx]
            # Move mass from a high-product position to a low-contribution position
            src = int(idx[np.argmax(products)])
            dst = int(rng.integers(n))
            if dst != src and a[src] > 1e-10:
                frac_move = rng.uniform(0.01, 0.1)
                a_new = a.copy()
                amt = a[src] * frac_move
                a_new[src] -= amt
                a_new[dst] += amt
                a_new = np.maximum(a_new, 0.0)
                a_new = a_new / a_new.sum()
            else:
                a_new = a + rng.normal(0, T, n) * a
        elif r < 0.85:
            # Block swap: swap mass between two blocks
            block_size = max(10, n // 100)
            b1 = int(rng.integers(0, n - 2 * block_size))
            b2 = int(rng.integers(b1 + block_size, n - block_size))
            a_new = a.copy()
            swap_amt = min(a[b1:b1+block_size].sum(), a[b2:b2+block_size].sum()) * 0.3
            if swap_amt > 1e-10:
                # Move some mass from block 1 to block 2
                for i in range(block_size):
                    move = min(a_new[b1+i], swap_amt / block_size)
                    a_new[b1+i] -= move
                    a_new[b2+i] += move
                a_new = np.maximum(a_new, 0.0)
                a_new = a_new / a_new.sum()
        else:
            # Large jump: interpolate with a random structured sequence
            gamma = rng.uniform(0.5, 3.0)
            x = np.linspace(0, 1, n)
            structured = (1.0 - x) ** gamma + 0.01
            structured = structured / structured.sum()
            alpha = rng.uniform(0.1, 0.4)
            a_new = (1 - alpha) * a + alpha * structured
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
        
        # Ensure validity
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        
        # Evaluate
        c_new = fast_autocorr(a_new)
        max_new = c_new.max()
        delta = max_new - c_max
        
        # Metropolis acceptance
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_new
            c = c_new
            c_max = max_new
            if c_max < best_m - 1e-15:
                best_m = c_max
                best_a = a.copy()
                best_no_improve = 0
            else:
                best_no_improve += 1
        else:
            best_no_improve += 1
        
        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SA step={step} ({elapsed:.1f}s): J={J:.6f}, T={T:.6f}, stall={best_no_improve}")
    
    return best_a, best_m


def peak_reduction(a0, time_limit, seed=55, verbose=False):
    """Aggressive peak reduction via targeted mass moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 3000
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        if c.max() < best_m - 1e-15:
            best_m = c.max()
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find top peaks
        top_k = min(20, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        
        improved = False
        
        for k in top_idx[:10]:
            if improved:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            
            # Try to reduce the top contributing products
            top_prods = np.argsort(products)[::-1][:5]
            for pi in top_prods:
                if improved:
                    break
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n:
                    continue
                
                # Compute "danger" map: how much each position contributes to top peaks
                danger = np.zeros(n)
                for k2 in top_idx[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        danger[idx2] += a[k2i - idx2]
                        danger[k2i - idx2] += a[idx2]
                
                # Find safe destinations (low danger)
                danger[src_i] = np.inf  # can't move to self
                n_safe = min(5, n)
                safe_dsts = np.argsort(danger)[:n_safe]
                
                for dst in safe_dsts:
                    dst_int = int(dst)
                    for frac in [0.01, 0.02, 0.03, 0.05]:
                        amt = a[src_i] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src_i] -= amt
                        a_trial[dst_int] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                if improved:
                    break
        
        if not improved and stall > 500:
            # Random restart perturbation
            noise = 0.03 * rng.normal(0, 1, n) * a
            a_try = a + noise
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()
            M_try = fast_autocorr(a_try).max()
            if M_try < best_m:
                a = a_try
                best_m = M_try
                best_a = a.copy()
                stall = 0
            else:
                # Accept anyway (annealing-style) to explore
                if rng.random() < 0.3:
                    a = a_try
    
    return best_a, best_m


def block_cd(a0, time_limit, seed=77, verbose=False):
    """Block coordinate descent: optimize blocks of coordinates together."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    block_size = max(5, n // 500)
    
    while time.monotonic() - t0 < time_limit and stall < 2000:
        c = fast_autocorr(a)
        if c.max() < best_m - 1e-15:
            best_m = c.max()
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Pick a random block to optimize
        # Focus on blocks that contribute to peaks
        top_idx = np.argsort(c)[::-1][:10]
        
        # Choose blocks near peak-contributing positions
        candidate_blocks = []
        for k in top_idx[:5]:
            k_int = int(k)
            i_start = max(0, k_int - n // 4)
            i_end = min(n, k_int + n // 4)
            # Add some blocks in this range
            for _ in range(3):
                b = int(rng.integers(max(0, i_start), max(0, i_end - block_size)))
                candidate_blocks.append(b)
        
        # Also add some random blocks
        for _ in range(5):
            candidate_blocks.append(int(rng.integers(0, n - block_size)))
        
        improved = False
        for b in candidate_blocks:
            if improved:
                break
            if b + block_size > n:
                continue
            
            # Try different perturbations of this block
            for scale in [0.01, 0.02, 0.05]:
                # Gaussian perturbation of the block
                pert = rng.normal(0, scale, block_size) * a[b:b+block_size]
                a_trial = a.copy()
                a_trial[b:b+block_size] = np.maximum(a_trial[b:b+block_size] + pert, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_initialization(n, seed=0):
    """Create a diverse initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    gamma = rng.uniform(0.5, 3.0)
    a = (1.0 - x) ** gamma + 0.05
    # Add some structure
    freq = rng.uniform(2, 20)
    a *= (1.0 + 0.2 * np.sin(2 * np.pi * freq * x))
    a = np.maximum(a, 0.0)
    return a / a.sum()

        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = norm_sum1(p)
            best_score = evaluate_safe(best_seq)

    except Exception:
        pass
    
    if best_seq is None:
        n_init = 7000

    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)

def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto probability simplex {x>=0, sum(x)=1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_grad_smoothed(a, p):
    """Compute gradient of smoothed objective given probability weights p on peaks.
    grad[j] = 2 * sum_k p[k] * a[k-j]
    Computed via FFT as: 2 * conv(p, a_rev)[n-1 : 2n-1]
    """
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, N)
    A_rev = np.fft.rfft(a[::-1], N)
    grad_conv = np.fft.irfft(P * A_rev, N)
    return 2.0 * grad_conv[n - 1:2 * n - 1]


def nesterov_opt(a0, time_limit, beta_start=30.0, beta_end=5000.0,
                 beta_growth=1.008, eta_base=0.5, top_k_start=50,
                 top_k_end=5, restart_every=1200, seed=42, verbose=False):
    """Nesterov accelerated projected gradient on top-k log-sum-exp smoothed objective."""

    best_a = x.copy()

    beta = float(beta_start)
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)

        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0

        y = np.maximum(y, 1e-15)
        s_y = y.sum()

            y = y / s_y
        else:
            y = x.copy()

        top_k = min(top_k, len(c))

def simulated_anneal(a0, time_limit, seed=99, verbose=False):
    """Simulated annealing with diverse structured perturbations."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    T_start = 0.008
    T_end = 0.00005
    best_no_improve = 0
    max_no_improve = 8000

    for step in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if best_no_improve > max_no_improve:
            break

        frac = min(1.0, step / 30000.0)
        T = T_start * (T_end / T_start) ** frac

        c = fast_autocorr(a)
        c_max = c.max()

        r = rng.random()
        if r < 0.25:
            # Multiplicative Gaussian
            scale = T * (1.0 + rng.random() * 2.0)
            a_new = a * (1.0 + rng.normal(0, scale, n))
        elif r < 0.40:
            # Anti-symmetric
            anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
            scale = T * (0.5 + rng.random())
            a_new = a + scale * anti * a
        elif r < 0.60:
            # Peak-aware: move mass from peak contributor to safe location
            k_peak = int(np.argmax(c))
            i_start = max(0, k_peak - n + 1)
            i_end = min(n - 1, k_peak)
            idx = np.arange(i_start, i_end + 1)
            if len(idx) > 0:
                products = a[idx] * a[k_peak - idx]
                src = int(idx[np.argmax(products)])
                dst = int(rng.integers(n))
                if dst != src and a[src] > 1e-12:
                    frac_move = rng.uniform(0.02, 0.15)
                    a_new = a.copy()
                    amt = a[src] * frac_move
                    a_new[src] -= amt
                    a_new[dst] += amt
                else:
                    a_new = a * (1.0 + rng.normal(0, T, n))
            else:
                a_new = a * (1.0 + rng.normal(0, T, n))
        elif r < 0.75:
            # Block mass transfer
            block_size = max(10, n // 80)
            b1 = int(rng.integers(0, n - 2 * block_size))
            b2 = int(rng.integers(b1 + block_size, n - block_size))
            a_new = a.copy()
            amt = min(a[b1:b1+block_size].mean(), a[b2:b2+block_size].mean()) * T * 10
            if amt > 1e-12:
                a_new[b1:b1+block_size] -= amt / block_size
                a_new[b2:b2+block_size] += amt / block_size
        else:
            # Large structured jump
            gamma = rng.uniform(0.5, 3.0)
            x = np.linspace(0, 1, n)
            structured = (1.0 - x) ** gamma + 0.01
            structured = structured / structured.sum()
            alpha = rng.uniform(0.1, 0.5)
            a_new = (1.0 - alpha) * a + alpha * structured

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            best_no_improve += 1
            continue
        a_new = a_new / s

        max_new = fast_autocorr(a_new).max()
        delta = max_new - c_max

        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(min(0, -delta / T))):
            a = a_new
            c = fast_autocorr(a)
            c_max = c.max()
            if c_max < best_m - 1e-15:
                best_m = c_max
                best_a = a.copy()
                best_no_improve = 0
            else:
                best_no_improve += 1
        else:
            best_no_improve += 1

        if verbose and step % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SA step={step} ({elapsed:.1f}s): J={J:.6f}, T={T:.6f}, stall={best_no_improve}")

    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_initialization(n, seed=0):

    amp = rng.uniform(0.1, 0.3)
    a *= (1.0 + amp * np.sin(2 * np.pi * freq * x))

        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t0
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.1f}s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search function combining multiple optimization strategies."""
    t0 = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = norm_sum1(p)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 7000
        best_seq = make_initialization(n_init, seed=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Nesterov (primary length) ===
    remaining = time_budget - (time.monotonic() - t0)
    print(f"\n=== Phase 1: Nesterov (n={n0}) ===")
    t1 = min(remaining * 0.22, 350.0)
    if t1 > 60:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t1,
                                beta_start=40.0, beta_end=4000.0,
                                beta_growth=1.009, eta_base=0.5,
                                top_k_start=60, top_k_end=3,
                                restart_every=1000, verbose=True, seed=1)
        best_seq, best_score = check_update(best_seq, best_score, a_out, "nest1", t0)

    # === Phase 2: Peak Reduction ===
    remaining = time_budget - (time.monotonic() - t0)
    print(f"\n=== Phase 2: Peak Reduction (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.15, 220.0)
    if t2 > 40:
        a_out, _ = peak_reduction(norm_sum1(best_seq), t2,
                                  seed=11, verbose=True)
        best_seq, best_score = check_update(best_seq, best_score, a_out, "peak1", t0)

    # === Phase 3: Simulated Annealing ===
    remaining = time_budget - (time.monotonic() - t0)
    print(f"\n=== Phase 3: Simulated Annealing (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 50:
        a_out, _ = simulated_anneal(norm_sum1(best_seq), t3,
                                    seed=99, verbose=True)
        best_seq, best_score = check_update(best_seq, best_score, a_out, "sa1", t0)

    # === Phase 4: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t0)
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    n_best = len(best_seq)
    lengths_to_try = sorted(set([
        max(4000, n_best - 1500),
        n_best + 500,
        n_best + 1200,
        n_best + 2000,
        min(10000, n_best + 3000)
    ]))

    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 70:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)

        # Nesterov on new length
        t4a = min(remaining * 0.22, 130.0)
        if t4a > 30:
            a_out, _ = nesterov_opt(a_long, t4a,
                                    beta_start=30.0, beta_end=2500.0,
                                    beta_growth=1.008, eta_base=0.5,
                                    top_k_start=40, top_k_end=5,
                                    restart_every=800, verbose=False, seed=len(new_n) % 100)
            best_seq, best_score = check_update(best_seq, best_score, a_out, f"ml_{new_n}", t0)

        # Peak reduction on new length
        remaining = time_budget - (time.monotonic() - t0)
        if remaining > 50:
            t4b = min(remaining * 0.18, 90.0)
            if t4b > 25:
                a_try = interpolate_sequence(best_seq, new_n)
                a_out, _ = peak_reduction(a_try, t4b, seed=33 + new_n % 10, verbose=False)
                best_seq, best_score = check_update(best_seq, best_score, a_out, f"mlpk_{new_n}", t0)

    # === Phase 5: Block CD ===
    remaining = time_budget - (time.monotonic() - t0)
    print(f"\n=== Phase 5: Block CD (n={len(best_seq)}) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 30:
        a_out, _ = block_cd(norm_sum1(best_seq), t5, verbose=True, seed=77)
        best_seq, best_score = check_update(best_seq, best_score, a_out, "blockcd", t0)

    # === Phase 6: Final Nesterov (high beta, focus on max) ===
    remaining = time_budget - (time.monotonic() - t0)
    print(f"\n=== Phase 6: Final Nesterov (n={len(best_seq)}) ===")
    if remaining > 60:
        t6 = min(remaining * 0.25, 200.0)
        if t6 > 30:
            a_out, _ = nesterov_opt(norm_sum1(best_seq), t6,
                                    beta_start=300.0, beta_end=15000.0,
                                    beta_growth=1.012, eta_base=0.3,
                                    top_k_start=10, top_k_end=2,
                                    restart_every=600, verbose=True, seed=4)
            best_seq, best_score = check_update(best_seq, best_score, a_out, "nest_final", t0)

    # === Phase 7: Final Peak Reduction + Annealing ===
    remaining = time_budget - (time.monotonic() - t0)
    if remaining > 50:
        print(f"\n=== Phase 7: Final Polish ===")
        t7a = min(remaining * 0.35, 120.0)
        if t7a > 20:
            a_out, _ = peak_reduction(norm_sum1(best_seq), t7a, seed=44, verbose=False)
            best_seq, best_score = check_update(best_seq, best_score, a_out, "peak_final", t0)

        remaining = time_budget - (time.monotonic() - t0)
        if remaining > 40:
            t7b = min(remaining * 0.40, 100.0)
            if t7b > 20:
                a_out, _ = simulated_anneal(norm_sum1(best_seq), t7b, seed=88, verbose=False)
                best_seq, best_score = check_update(best_seq, best_score, a_out, "sa_final", t0)

        remaining = time_budget - (time.monotonic() - t0)
        if remaining > 30:
            t7c = min(remaining * 0.40, 80.0)
            if t7c > 15:
                a_out, _ = block_cd(norm_sum1(best_seq), t7c, seed=99, verbose=False)
                best_seq, best_score = check_update(best_seq, best_score, a_out, "block_final", t0)

    # === Phase 8: One more Nesterov burst if time allows ===
    remaining = time_budget - (time.monotonic() - t0)
    if remaining > 50:
        print(f"\n=== Phase 8: Ultimate Nesterov ===")
        t8 = min(remaining * 0.60, 150.0)
        if t8 > 25:
            a_out, _ = nesterov_opt(norm_sum1(best_seq), t8,
                                    beta_start=500.0, beta_end=20000.0,
                                    beta_growth=1.015, eta_base=0.2,
                                    top_k_start=5, top_k_end=1,
                                    restart_every=500, verbose=True, seed=7)
            best_seq, best_score = check_update(best_seq, best_score, a_out, "ultimate", t0)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t0
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = make_initialization(7000, seed=0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    # Ensure validity
    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)

    return [float(x) for x in best_seq]
