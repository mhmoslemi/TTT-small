def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, n=fft_size)[:size]

        return np.ones_like(a) / len(a)

"""

2. Adaptive exponential tilting on the probability simplex
3. FFT-accelerated autocorrelation

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, n=fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def obj_val_fast(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def compute_gradient_lag(a, k):
    """Compute gradient of c[k] = [conv(a,a)][k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    # S_k = [max(0, k-n+1), min(n-1, k)]
    L1 = max(0, k - n + 1)
    R1 = min(n - 1, k)
    L2 = max(0, n - 1 - k)
    R2 = min(n - 1, k)
    # For m in [L1, R1]: first indicator active -> grad[m] += a[k-m]
    for m in range(L1, R1 + 1):
        km = k - m
        if 0 <= km < n:
            grad[m] += a[km]
    # For m in [L2, R2]: second indicator active -> grad[m] += a[k-m]
    for m in range(L2, R2 + 1):
        km = k - m
        if 0 <= km < n:
            grad[m] += a[km]
    return grad

def compute_gradient_topk(a, top_lags, weights):
    """Compute weighted sum of gradients for top-K lags."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    for pos in range(len(top_lags)):
        k = int(top_lags[pos])
        w = weights[pos]
        L1 = max(0, k - n + 1)
        R1 = min(n - 1, k)
        L2 = max(0, n - 1 - k)
        R2 = min(n - 1, k)
        if L1 <= R1:
            ms = np.arange(L1, R1 + 1)
            kms = k - ms
            valid = (kms >= 0) & (kms < n)
            grad[ms[valid]] += w * a[kms[valid]]
        if L2 <= R2:
            ms = np.arange(L2, R2 + 1)
            kms = k - ms
            valid = (kms >= 0) & (kms < n)
            grad[ms[valid]] += w * a[kms[valid]]
    return grad

def subgrad_mirror_descent(a0, time_limit, K=30, eta0=0.05, verbose=True):
    """Subgradient mirror descent with exponential tilting on simplex."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_val = obj_val_fast(a)
    t0 = time.monotonic()
    eta = eta0
    stall = 0
    max_stall = 3000
    rng = np.random.default_rng(12345)
    iter_count = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_autocorr(a)
        M = b.max()
        cur_val = 2.0 * n * M
        iter_count += 1

        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            stall = 0
            eta = min(eta * 1.05, 0.5)
        else:
            stall += 1
            if stall % 200 == 0:
                eta = max(eta * 0.75, 1e-5)

        # Top-K lags with soft weighting
        n_lags = len(b)
        K_use = min(K, n_lags)
        top_idx = np.argsort(b)[::-1][:K_use]
        top_vals = b[top_idx]

        # Weight: emphasize peaks above average
        avg_b = b.mean()
        excess = np.maximum(top_vals - avg_b, 0)
        if excess.max() < 1e-15:
            excess = np.ones(K_use)
        # Boost the very top
        excess[0] += top_vals[0] * 0.5
        w = excess
        w = w / w.sum()

        # Compute gradient
        grad = compute_gradient_topk(a, top_idx, w)

        # Exponential tilting update (mirror descent on simplex)
        # Normalize gradient to have appropriate scale
        grad_scale = grad.max()
        if grad_scale < 1e-15:
            break
        grad_norm = grad / grad_scale

        # Update in log-space for numerical stability
        log_a = np.log(a + 1e-30)
        log_a -= eta * grad_norm
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        # Check improvement
        b_new = fast_autocorr(a_new)
        val_new = 2.0 * n * b_new.max()
        if val_new < best_val:
            a = a_new
        else:
            # Try smaller step
            if eta > 1e-6:
                eta_trial = eta * 0.5
                log_a2 = np.log(a + 1e-30)
                log_a2 -= eta_trial * grad_norm
                log_a2 -= log_a2.max()
                a_trial = np.exp(log_a2)
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial / st
                    val_trial = 2.0 * n * fast_autocorr(a_trial).max()
                    if val_trial < best_val:
                        a = a_trial
                        eta = eta_trial
                        continue
            a = a_new  # accept anyway to explore

        # Periodic random perturbation
        if stall > 1500 and stall % 500 == 0:
            noise = rng.standard_normal(n) * 0.01
            a_pert = a + noise * a
            a_pert = np.maximum(a_pert, 0.0)
            sp = a_pert.sum()
            if sp > 1e-15:
                a_pert = a_pert / sp
                vp = 2.0 * n * fast_autocorr(a_pert).max()
                if vp < best_val:
                    a = a_pert
                else:
                    # Blend
                    blend = 0.5
                    a_blend = (1 - blend) * a + blend * a_pert
                    a_blend = np.maximum(a_blend, 0.0)
                    sb = a_blend.sum()
                    if sb > 1e-15:
                        a = a_blend / sb

        if verbose and iter_count % 500 == 1:
            elapsed = time.monotonic() - t0
            print("    SMD t={:6d} ({:.1f}s) J={:.7f} eta={:.5f} stall={}".format(
                iter_count, elapsed, best_val, eta, stall))

    elapsed = time.monotonic() - t0
    if verbose:
        print("    SMD done ({:.1f}s) J={:.7f} iters={}".format(elapsed, best_val, iter_count))
    return best_a, best_val

def coord_descent_polish(a0, time_limit, n_top=20, verbose=True):
    """Aggressive coordinate descent: move mass from peak contributors to quiet positions."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_val = obj_val_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    stall = 0
    max_stall = 10000
    fracs = [0.0005, 0.001, 0.002, 0.004, 0.008, 0.015, 0.025, 0.04]

    iter_count = 0
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_autocorr(a)
        M = b.max()
        cur_val = 2.0 * n * M
        iter_count += 1

        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Find top peaks and their contributing pairs
        K_use = min(n_top, len(b))
        top_k = np.argsort(b)[::-1][:K_use]
        improved = False

        for k_pos in range(len(top_k)):
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(top_k[k_pos])
            L1 = max(0, ki - n + 1)
            R1 = min(n - 1, ki)
            if L1 > R1:
                continue

            # Find the pair (i, ki-i) with maximum product contribution
            idx = np.arange(L1, R1 + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-15:
                continue

            # Find quiet positions (low pressure from all top peaks)
            pressure = np.zeros(n)
            for kk in top_k[:8]:
                kk_i = int(kk)
                ps = max(0, kk_i - n + 1)
                pe = min(n - 1, kk_i)
                pr = a[ps:pe+1] * a[kk_i - pe:kk_i - ps+1][::-1]
                pressure[ps:pe+1] += pr

            pressure[src] = 1e10
            # Get quietest positions
            quiet = np.argsort(pressure)[:40]

            for dst in quiet:
                dst = int(dst)
                if dst == src:
                    continue
                found = False
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-18:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt * 2.0 * n < best_val - 1e-15:
                        a = a_t
                        best_val = Mt * 2.0 * n
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break

            if improved:
                break

        if not improved:
            # Random pair swap to escape
            if a.any():
                nonzero = np.where(a > 1e-10)[0]
                if len(nonzero) >= 2:
                    i = int(rng.integers(len(nonzero)))
                    j = int(rng.integers(n))
                    src_i = int(nonzero[i])
                    if src_i != j:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a[src_i] * frac
                        a[src_i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        st = a.sum()
                        if st > 1e-15:
                            a = a / st
                            Mn = fast_autocorr(a).max() * 2.0 * n
                            if Mn < best_val:
                                best_val = Mn
                                best_a = a.copy()
                                stall = 0
                                improved = True

    elapsed = time.monotonic() - t0
    if verbose:
        print("    CoordDescent done ({:.1f}s) J={:.7f} iters={}".format(elapsed, best_val, iter_count))
    return best_a, best_val

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def gen_initializations(n, n_seeds=15):

    # Random smooth shapes
    rng = np.random.default_rng(999)
    for s in range(n_seeds - len(inits)):

            inits.append(normalize(f))

        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / a_pert.sum()
        a_out, v_out = subgrad_mirror_descent(a_pert, t5 * 0.6, K=50, eta0=0.02, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P5a improved: {}".format(repr(best_score)))

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_out2, v_out2 = subgrad_mirror_descent(normalize(best_seq), remaining * 0.4, K=10, eta0=0.01, verbose=True)
            sc2 = eval_safe(a_out2)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_out2
                print("  P5b improved: {}".format(repr(best_score)))

    # Phase 6: Length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and len(best_seq) <= 12000:
        n_long = min(len(best_seq) + 3000, 16000)
        print("\n=== P6: Long n={} ===".format(n_long))
        t6 = min(remaining * 0.20, 200.0)
        if t6 > 30:
            a_long = interpolate_seq(best_seq, n_long)
            a_out, v_out = subgrad_mirror_descent(a_long, t6 * 0.6, K=40, eta0=0.03, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  P6 improved: {}".format(repr(best_score)))

    # Phase 7: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.25, 300.0)
    print("\n=== P7: Final CoordPolish (n={}) ===".format(len(best_seq)))
    if t7 > 15:
        a_out, v_out = coord_descent_polish(normalize(best_seq), t7 * 0.5, n_top=20, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out2, v_out2 = pair_swap_refine(normalize(best_seq), min(remaining * 0.5, 120), verbose=False)
            sc2 = eval_safe(a_out2)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_out2
            print("  P7 final: {}".format(repr(best_score)))

    # Phase 8: One more SMD with very small eta for fine tuning
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== P8: Fine SMD (n={}) ===".format(len(best_seq)))
        a_out, v_out = subgrad_mirror_descent(normalize(best_seq), remaining * 0.8, K=5, eta0=0.005, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P8 improved: {}".format(repr(best_score)))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={} n={} t={:.1f}s".format(repr(best_score), len(best_seq), elapsed))
    else:
        print("\nERROR: fallback")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)

    return [float(x) for x in best_seq]
