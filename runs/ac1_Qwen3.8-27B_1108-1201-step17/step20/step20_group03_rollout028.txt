def softmax_gradient(a, beta):

    n = len(a)
    b = fast_autocorr(a)
    b_shifted = b - b.max()

    p /= p.sum()

    # This is 2 * (p_reversed * a) centered appropriately
    # np.convolve(p, a) gives sum_i p[i]*a[k-i] for the (p*a)[k]

    Pr = np.fft.rfft(p[::-1], fft_size)  # reversed p for correlation
    Ar = np.fft.rfft(a, fft_size)

    # conv[j] = sum_i p_rev[i] * a[j-i] = sum_i p[n_b-1-i] * a[j-i]
    # where n_b = len(p) = 2n-1
    # = sum_m p[m] * a[j-(n_b-1-m)] = sum_m p[m] * a[m - (n_b-1) + j]
    # Hmm, let me just verify with a small example or use a different approach
    # 
    # Actually the simplest: g[j] = 2 * sum_k p[k] * a[k-j] for valid indices
    # This is the same as 2 * (p * a_flip)[j + n - 1] where a_flip = a[::-1]

    # So g[j] = 2 * (a ⊛ p)[j] where (a ⊛ p)[j] = sum_i a[i]*p[i+j]
    # In numpy: this is np.correlate(p, a, mode='valid') if len(p) >= len(a)
    # Or we can compute via FFT:
    # correlate(p, a) = ifft(fft(p) * conj(fft(a)))
    # But we need the full correlation, not just the valid part.

    # g[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]

    # g[j] = 2 * sum_m a[m] * p[j+m]

    # But m+k >= 2n-1 means m >= 2n-1-k, and since m < n, we need k >= n-1

    # So for k in [0, n-1], m+k is always < 2n-1 = len(p). 
    # Great, so corr[j] for j=0,...,n-1 gives exactly sum_m a[m]*p[m+j].
    
    grad = 2.0 * corr[:n] / beta  # normalize by beta for step size purposes
    return grad, p, b

Wait, I need to double-check my gradient formula. Let me re-derive:

f(a) = max_k b_k where b_k = sum_i a[i]*a[k-i]

The softmax approximation: f_smooth(a) = (1/beta) * log(sum_k exp(beta * b_k))

The gradient: df_smooth/da[j] = sum_k p_k * db_k/da[j] where p_k = exp(beta*b_k) / sum_l exp(beta*b_l)

And db_k/da[j] = 2*a[k-j] (if 0 <= k-j < n)

So df_smooth/da[j] = sum_k p_k * 2*a[k-j] = 2 * sum_k p_k * a[k-j]

Let me reindex: let m = k-j, so k = m+j:
= 2 * sum_m p[m+j] * a[m] (where m ranges over valid values, i.e., 0 <= m < n and 0 <= m+j < 2n-1)

So g[j] = 2 * sum_m a[m] * p[m+j]

This confirms my formula above. And the FFT computation gives:
corr[j] = sum_m a[m] * p[m+j] for j = 0, ..., n-1

So grad[j] = 2 * corr[j].

L ≈ 2 * max_j sum_k p_k * |db_k/da[j]| ... this is hard to bound tightly.

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FISTA with full-softmax smoothed gradient and adaptive step size
2. Spectral power compression (flattening the power spectrum)
3. Targeted peak-pushing local search
4. Multi-scale exploration with interpolation
5. Cyclic shift optimization

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize_simplex(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_gradient_full(a, beta):
    """Compute smoothed gradient using full softmax over all lags."""
    n = len(a)
    b = fast_autocorr(a)
    b_shift = b - b.max()
    b_shift = np.clip(b_shift, -400.0, 0.0)
    p = np.exp(beta * b_shift)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(b)
        p[b == b.max()] = 1.0 / (b == b.max()).sum()
    else:
        p = p / p_sum
    # grad[j] = 2 * sum_m a[m] * p[m+j]
    fft_size = 1 << (n + len(p) - 1 - 1).bit_length()
    Af = np.fft.rfft(a, fft_size)
    Pf = np.fft.rfft(p, fft_size)
    corr = np.fft.irfft(np.conj(Af) * Pf, fft_size)
    grad = 2.0 * corr[:n]
    return grad, p, b

def simplex_project(z):
    """Project z onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho_cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.nonzero(rho_cond)[0]) if rho_cond.any() else 0
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def fista_optimize(a0, time_limit, beta_start=20.0, beta_end=1000.0,
                   beta_growth=1.004, verbose=True):
    """FISTA with full-softmax smoothing and adaptive step size."""
    n = len(a0)
    x = normalize_simplex(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(42)
    
    # Adaptive step size via Backtracking
    L = 1.0 / n  # initial Lipschitz estimate
    step = 1.0 / L
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = x.copy()
        
        # Compute gradient at y
        grad, p, b = compute_gradient_full(y, beta)
        
        # Backtracking line search
        # f(y - step*grad) <= f(y) - grad.dot(y - y_new)*... 
        # For simplicity, use a heuristic step
        grad_norm = np.sqrt(grad.dot(grad))
        if grad_norm < 1e-15:
            step = 1.0 / L
        else:
            # Scale step by gradient norm
            f_y = (1.0 / beta) * np.log(p.sum() * np.exp(beta * b.max()))  # not quite right
            # Just use: step = c / (beta * max(|grad|))
            step = 0.5 / (beta * np.max(np.abs(grad)) + 1e-10)
            step = min(step, 5.0)
        
        z = y - step * grad
        x_new = simplex_project(z)
        
        # Check improvement
        b_new = fast_autocorr(x_new)
        m_new = b_new.max()
        
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        # Update beta
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        # Periodic restart
        if stall > 3000:
            stall = 0
            t_nest = 1.0
            noise_level = 0.02 / (1.0 + 0.1 * (t // 3000))
            pert = rng.normal(0, noise_level, n) * best_x
            x_r = best_x + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 5.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    FISTA restart t={t} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.1f}")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FISTA t={t} ({elapsed:.1f}s): J={J:.6f}, beta={beta:.0f}")
    
    return best_x, best_m

def spectral_reshape(a, n_iters=50, alpha=0.5, verbose=True):
    """Compress the power spectrum to flatten the autocorrelation."""
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    for it in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compress magnitude: new_mag = mag^alpha (alpha < 1 compresses)
        mag_clipped = np.maximum(mag, 1e-10)
        new_mag = mag_clipped ** alpha
        new_mag /= new_mag.mean()
        
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        
        a = a_new
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Spectral reshape: J={J:.6f}")
    
    return best_a, best_m

def peak_push_search(a0, time_limit, verbose=True):
    """Targeted local search that pushes mass away from peak-contributing pairs."""
    n = len(a0)
    a = normalize_simplex(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        b = fast_autocorr(a)
        # Find top contributing lags
        top_lags = np.argsort(b)[::-1][:20]
        improved = False
        
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            
            k = int(top_lags[rng.integers(len(top_lags))])
            # Find the pair (i, k-i) with maximum contribution
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_min > i_max:
                continue
            idx = np.arange(i_min, i_max + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = int(idx[np.argmax(contrib)])
            
            # Find a destination that doesn't contribute to top peaks
            # Consider positions where the local "autocorrelation pressure" is low
            pressure = np.zeros(n)
            for k2 in top_lags[:10]:
                k2 = int(k2)
                s2 = max(0, k2 - n + 1)
                e2 = min(n - 1, k2)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2 - idx2]
            
            # Sample destination from low-pressure region
            candidates = np.where(pressure < np.median(pressure))[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src:
                continue
            if a[src] < 1e-15:
                continue
            
            # Try different transfer fractions
            for frac in [0.01, 0.03, 0.05, 0.1]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    b = fast_autocorr(a)
                    top_lags = np.argsort(b)[::-1][:20]
                    break
        
        if not improved:
            stall += 1
            # Add random perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.005, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        m_new = fast_autocorr(a).max()
                        if m_new < best_m:
                            best_m = m_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Peak-push ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def cyclic_shift_optimize(a0):
    """Try all cyclic shifts and keep the best."""
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        m = fast_autocorr(a_shift).max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m

def interpolate_to_length(a, new_n):
    """Interpolate sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = normalize_simplex(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a)
    return normalize_simplex(a_interp)

def toepitz_matvec_fft(a, n):
