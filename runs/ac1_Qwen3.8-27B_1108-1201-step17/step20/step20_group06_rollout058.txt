I think the optimal shape is a smooth, slowly-varying function that's slightly asymmetric. Something like a*exp(b*x) * sin(pi*x) or similar.

Let me code this up:

```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Improved LP-Frank-Wolfe with adaptive active set
2. Nesterov smoothed-max with tilt-aware restarts
3. Spectral domain refinement with amplitude shaping
4. Coordinate descent targeting specific peaks
5. Multi-length and multi-shape exploration
"""

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None

def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    return _MODULE_SOLVER

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_smoothed_grad(y, beta, top_k=None):
    n = len(y)
    b = fast_autocorr(y)
    size = 2 * n - 1
    
    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    
    if b_max <= 0:
        return np.zeros(n), b
    
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s
    
    # Gradient: sum_k p[k] * grad_k
    # grad_k[i] = 2*a[k-i] if valid and i != k/2, a[k-i] if i == k/2
    # Actually, let's use the FFT-based approach:
    # sum_k p[k] * (a * a_shifted_k) [linearized]
    # = sum_k p[k] * sum_i a[i] * a[k-i] (the bilinear form)
    # Gradient w.r.t. a: sum_k p[k] * (2 * p-convolved a)
    # More precisely, if b = a*a (full conv), then db/da[i] = 2*sum_k p[k]*a[k-i] (adjusting for boundary)
    # This is 2 * (p_rev * a) where * is full convolution, evaluated at the right indices
    
    # Using FFT:
    # grad = 2 * irfft(rfft(p_rev) * rfft(a)) [cropped to length n]
    # where p_rev is p reversed
    
    # Actually, (a*a)[k] = sum_j a[j]*a[k-j]
    # d/da[i] sum_k p[k]*(a*a)[k] = sum_k p[k] * (a[k-i] + a[k-i]) [if valid]
    # = 2 * sum_k p[k]*a[k-i] [for i where k-i is valid]
    
    # Let's define: for each i, grad[i] = 2 * sum_{k: 0<=k-i<n} p[k]*a[k-i]
    # = 2 * sum_{j: 0<=j<n} p[i+j]*a[j]  (substituting k = i+j)
    # = 2 * (cross-correlation of p and a, evaluated at shift i)
    
    # This is: 2 * sum_j p[i+j] * a[j]
    # Which is the cross-correlation (p ⊛ a)[i] = sum_j p[i+j]*a[j]
    
    # Using FFT: cross-correlation can be computed as:
    # IFFT(FFT(a) * conj(FFT(p))) [with appropriate indexing]
    
    # Actually, let me use the direct approach from the previous code:
    a_rev = y[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    # This computes conv(a_rev, p)
    # conv(a_rev, p)[m] = sum_j a_rev[j]*p[m-j] = sum_j a[n-1-j]*p[m-j]
    # Let k = n-1-j, so j = n-1-k:
    # = sum_k a[k]*p[m-(n-1-k)] = sum_k a[k]*p[m-n+1+k]
    # We want: sum_k p[i+k]*a[k] for each i
    # = sum_k a[k]*p[k+i]
    # From above: conv(a_rev, p)[m] = sum_k a[k]*p[m-n+1+k]
    # So m-n+1 = i → m = i+n-1
    # grad[i] = 2 * conv(a_rev, p)[i+n-1]
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    # But we need to handle the factor of 1 for the "diagonal" (i = k/2) case
    # This is a second-order correction that's usually negligible
    # The /beta from the softmax is already in p
    return grad, b

def nesterov_improved(a0, time_limit, beta_start=20.0, beta_end=800.0,
                       beta_growth=1.004, eta_init=0.5, top_k_ratio=0.03,
                       restart_every=800, verbose=True, seed=42):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    eta = eta_init

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, b = compute_smoothed_grad(y, beta, top_k)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            x_new = x
            max_new = b.max()
        else:
            # Backtracking line search with Armijo
            y_max = b.max()
            eta_trial = eta
            found = False
            for _ in range(12):
                z = y - eta_trial * grad
                x_new = simplex_project(z)
                b_new = fast_autocorr(x_new)
                max_new = b_new.max()
                if max_new <= y_max - 1e-10 or max_new <= y_max + 1e-15:
                    found = True
                    break
                eta_trial *= 0.4
            if not found:
                x_new = x
                max_new = y_max
                eta = eta_trial
            else:
                eta = min(eta * 1.05, 3.0)

        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 40 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = rng.integers(7)
            if strategy == 0:
                noise = 0.02 + 0.02 * rng.random()
                x_r = best_a * np.exp(noise * rng.standard_normal(n))
            elif strategy == 1:
                # Tilt: multiply by linear ramp
                ramp = np.linspace(0.7, 1.3, n)
                x_r = best_a * ramp
            elif strategy == 2:
                # Blend with random
                x_r = 0.7 * best_a + 0.3 * rng.dirichlet(np.ones(n) * 0.2)
            elif strategy == 3:
                # Local flip: reverse a random segment
                i = rng.integers(0, n - 1)
                j = rng.integers(i + 1, n)
                x_r = best_a.copy()
                x_r[i:j] = best_a[i:j][::-1]
            elif strategy == 4:
                # Gaussian bump perturbation
                center = rng.integers(n // 4, 3 * n // 4)
                width = int(rng.integers(n // 20, n // 5))
                x_r = best_a.copy()
                x_r[center] *= 1 + 0.1 * rng.random()
                # Smooth decay
                for d in range(1, width):
                    decay = np.exp(-d / (width / 3))
                    if center - d >= 0:
                        x_r[center - d] *= (1 + 0.05 * decay * rng.random())
                    if center + d < n:
                        x_r[center + d] *= (1 + 0.05 * decay * rng.random())
            elif strategy == 5:
                # Frequency domain perturbation
                A = np.fft.rfft(best_a)
                A[1:] *= np.exp(1j * 0.03 * rng.standard_normal(len(A) - 1))
                x_r = np.fft.irfft(A, len(best_a))
            else:
                # Random subset scaling
                mask = rng.random(n) > 0.5
                x_r = best_a.copy()
                x_r[mask] *= rng.uniform(0.7, 0.9)
                x_r[~mask] *= rng.uniform(1.1, 1.3)
            
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 10.0)
            eta = eta_init

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("  Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + str(int(beta)))

    return best_a, best_max_b


def lp_fw_improved(a0, time_limit, solver, initial_K=500, max_K=3000,
                   add_per_iter=30, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    # Initialize active set with top peaks
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:min(initial_K, 2*n-1)])

    no_improve = 0
    max_no_improve = 800
    round_t = 0

    round_t = 1
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set size
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K // 2])
            no_improve += 50
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 10
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations and add to active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Also periodically add new top peaks not in active set
        round_t += 1
        if round_t % 50 == 0:
            top_peaks = set(int(k) for k in np.argsort(ac)[::-1][:int(max_K * 0.5)])
            active_set.update(top_peaks)

        # Adaptive line search
        best_alpha = 2.0 / (round_t + 1.0)
        best_trial_M = M_curr
        for alpha in [best_alpha, best_alpha * 0.5, best_alpha * 1.5,
                      1.0 / (round_t + 1), 0.02, 0.05, 0.1, 0.2, 0.3]:
            alpha = max(0.005, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and round_t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  LP-FW t=" + str(round_t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def spectral_refine(a0, time_limit, verbose=True, seed=777):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()

    alpha = 0.03
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        target = power[1:].mean() if len(power) > 1 else power[0]

        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.0:
                red = 1.0 - alpha * (ratio - 1.0)
                A_new[k] *= max(0.2, np.sqrt(red))
            elif ratio < 0.4:
                inc = 1.0 + alpha * 0.5 * (0.4 - ratio)
                A_new[k] *= min(2.5, np.sqrt(inc))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 300 == 0:
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.03
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial

        if it % 400 == 0:
            alpha = max(0.005, alpha * 0.93)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  Spec it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def targeted_peak_reduction(a0, time_limit, verbose=True, seed=42):
    """Directly target the worst peaks with local modifications."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 3000:
        ac = fast_autocorr(a)
        size = 2 * n - 1
        # Get top 5 peaks
        top_k = np.argsort(ac)[::-1][:5]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            # Partner indices
            partners = k_int - idx
            contrib = a[idx] * a[partners]
            
            # The peak is dominated by the largest contribution
            top_contrib = np.argsort(contrib)[::-1][:3]
            
            for ci in top_contrib:
                src_i = int(idx[ci])
                src_p = int(partners[ci])
                if not (0 <= src_i < n and 0 <= src_p < n):
                    continue
                if src_i == src_p:
                    continue
                
                # Try reducing the product by moving mass
                # Move some mass from the larger of a[src_i], a[src_p]
                # to a random location
                if a[src_i] > a[src_p]:
                    src = src_i
                else:
                    src = src_p
                
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    # Try several destinations
                    for _ in range(8):
                        dst = int(rng.integers(n))
                        if dst == src:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                        elif M_trial < ac[k_int] * 0.999:
                            # Accept if it reduces this specific peak significantly
                            # even if global max doesn't improve (to help with local structure)
                            pass
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                # Larger random perturbation
                i1, i2 = int(rng.integers(n)), int(rng.integers(n))
                if i1 != i2 and a[i1] > 1e-12:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i1] * frac
                    a[i1] -= amt
                    a[i2] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  PeakRed (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def multi_length_search(base_seq, time_limit, verbose=True):
    """Try different lengths around the base length."""
    best_a = base_seq.copy()
    best_score = 2.0 * len(base_seq) * fast_autocorr(base_seq).max()
    t0 = time.monotonic()
    n_base = len(base_seq)

    # Try a few different lengths
    candidate_lengths = [
        n_base - 500, n_base - 200, n_base,
        n_base + 200, n_base + 500, n_base + 1000, n_base + 2000
    ]
    candidate_lengths = [l for l in candidate_lengths if 1000 <= l <= 12000]
    # Remove duplicates
    seen = set()
    unique_lengths = []
    for l in candidate_lengths:
        if l not in seen:
            seen.add(l)
            unique_lengths.append(l)
```

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Improved LP-Frank-Wolfe with adaptive active set management
2. Nesterov smoothed-max with diverse tilt-aware restarts
3. Targeted peak reduction via local mass redistribution
4. Spectral domain amplitude shaping
5. Multi-length and multi-shape exploration
Key improvements: LP-FW prioritized, better active set growth,
segment-reversal restarts, tilt perturbations, and focused time allocation.
"""

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None


def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    n = len(y)
    b = fast_autocorr(y)
    size = 2 * n - 1

    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()

    if b_max <= 0:
        return np.zeros(n), b

    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s

    a_rev = y[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, b


def nesterov_v4(a0, time_limit, beta_start=20.0, beta_end=1000.0,
                beta_growth=1.004, eta_init=0.5, top_k_ratio=0.03,
                restart_every=700, verbose=True, seed=42):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    eta = eta_init

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, b = compute_smoothed_grad(y, beta, top_k)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            x_new = x
            max_new = b.max()
        else:
            y_max = b.max()
            eta_trial = eta
            found = False
            for _ in range(12):
                z = y - eta_trial * grad
                x_new = simplex_project(z)
                b_new = fast_autocorr(x_new)
                max_new = b_new.max()
                if max_new <= y_max - 1e-12:
                    found = True
                    break
                eta_trial *= 0.4
            if not found:
                x_new = x
                max_new = y_max
                eta = max(eta_trial, 0.01)
            else:
                eta = min(eta * 1.05, 3.0)

        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 40 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = rng.integers(8)
            if strategy == 0:
                noise = 0.02 + 0.02 * rng.random()
                x_r = best_a * np.exp(noise * rng.standard_normal(n))
            elif strategy == 1:
                tilt = rng.uniform(0.5, 1.5)
                ramp = np.linspace(1.0 - (tilt - 1.0), 1.0 + (tilt - 1.0), n)
                x_r = best_a * ramp
            elif strategy == 2:
                x_r = 0.7 * best_a + 0.3 * rng.dirichlet(np.ones(n) * 0.2)
            elif strategy == 3:
                i = rng.integers(0, max(1, n - 100))
                seg_len = int(rng.integers(100, max(200, n // 5)))
                j = min(i + seg_len, n)
                x_r = best_a.copy()
                x_r[i:j] = best_a[i:j][::-1]
            elif strategy == 4:
                center = rng.integers(n // 4, 3 * n // 4)
                width = int(rng.integers(n // 20, n // 5))
                x_r = best_a.copy()
                for d in range(width):
                    decay = np.exp(-d / max(1, width // 3))
                    scale = 1.0 + 0.1 * decay * rng.uniform(-1, 1)
                    if center - d >= 0:
                        x_r[center - d] *= scale
                    if center + d < n:
                        x_r[center + d] *= scale
            elif strategy == 5:
                A = np.fft.rfft(best_a)
                A[1:] *= np.exp(1j * 0.04 * rng.standard_normal(len(A) - 1))
                x_r = np.fft.irfft(A, len(best_a))
            elif strategy == 6:
                x_r = best_a.copy()
                mask = rng.random(n) > 0.5
                x_r[mask] *= rng.uniform(0.7, 0.95)
                x_r[~mask] *= rng.uniform(1.05, 1.3)
            else:
                # Quadratic tilt
                x_vals = np.linspace(0, 1, n)
                c = rng.uniform(-0.3, 0.3)
                x_r = best_a * (1.0 + c * (2 * x_vals - 1))

            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 10.0)
            eta = eta_init

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("  Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + str(int(beta)))

def lp_fw_v3(a0, time_limit, solver, initial_K=400, max_K=3000,
             add_per_iter=30, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:min(initial_K, 2 * n - 1)])

    no_improve = 0
    max_no_improve = 1000
    round_t = 0
    prev_lp_M = None

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K // 2])
            no_improve += 50
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 10
            continue

        lp_obj_val = float(M_var.value) if M_var.value is not None else None
        if prev_lp_M is not None and lp_obj_val is not None and abs(lp_obj_val - prev_lp_M) < 1e-10:
            no_improve += 20
        prev_lp_M = lp_obj_val

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        round_t += 1
        if round_t % 30 == 0:
            top_peaks = set(int(k) for k in np.argsort(ac)[::-1][:min(int(max_K * 0.4), 2 * n - 1)])
            active_set.update(top_peaks)

        best_alpha = 2.0 / (round_t + 1.0)
        best_trial_M = M_curr
        alphas_to_try = [best_alpha, best_alpha * 0.5, best_alpha * 1.5,
                         1.0 / (round_t + 1), 0.02, 0.05, 0.1, 0.15, 0.2, 0.3]
        for alpha in alphas_to_try:
            alpha = max(0.005, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and round_t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  LP-FW t=" + str(round_t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m
