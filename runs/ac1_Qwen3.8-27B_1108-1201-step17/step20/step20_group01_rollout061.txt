import numpy as np
import time
import scipy.optimize as so

def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_obj(a):
    """Compute 2*n*max_conv/(sum)^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    c = fast_autocorr(a)
    return 2.0 * n * c.max() / (s * s)

def normalize_simplex(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def softmax_gradient(a, C, lam):
    """Compute gradient of (1/lam)*log(sum(exp(lam*C))) w.r.t. a.
    Returns gradient of sum_k w_k * C[k] where w_k = softmax(lam*C)."""
    n = len(a)
    # Compute softmax weights
    C_shifted = C - C.max()
    w = np.exp(lam * C_shifted)
    w /= w.sum()
    
    # Compute correl[j] = sum_k w[k] * a[k-j] (approximately)
    # Using FFT: correl = (w * a_flipped)[n-1 : 2n-1]
    a_flip = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, n=fft_size)
    AF = np.fft.rfft(a_flip, n=fft_size)
    conv = np.fft.irfft(W * AF, n=fft_size)
    correl = conv[n-1 : 2*n-1]
    
    # Gradient is approximately 2 * correl (for interior peaks)
    return 2.0 * correl

    K = len(S)

        return None

        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[idx, j] = a[k - j]
    
    # Variables: [g_0, ..., g_{n-1}, M]

    #   A @ g <= M  =>  A @ g - M <= 0  =>  [A, -1] @ x <= 0
    #   g >= 0, M >= 0 (handled by bounds)
    
    c = np.zeros(n + 1)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent with FFT-accelerated softmax/p-power gradients
2. LP-based Frank-Wolfe with adaptive active set and warm-starting
3. Targeted coordinate descent on peak-contributing pairs
4. Multi-length exploration with interpolation

import numpy as np
import time
import scipy.optimize as so

# Module-level state to avoid recomputation
_mod_cache = {}

def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, n=fft_size)[:size]

def normalize_simplex(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def eval_obj_normalized(a):
    """Compute 2*n*max_conv (assuming sum(a)=1)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    c = fast_autocorr(a)
    return float(2.0 * n * c.max() / (s * s))

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    mask = u * rho_candidates > (cssv - 1)
    if not mask.any():
        return np.ones(n, dtype=np.float64) / n
    rho = rho_candidates[mask][-1]
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

def compute_smoothed_gradient(a, C, lam):
    """
    Compute gradient of (1/lam)*log(sum(exp(lam*C[k]))) w.r.t. a.
    Uses FFT for efficiency.
    """
    n = len(a)
    # Softmax weights (shifted for numerical stability)
    C_shifted = C - C.max()
    w = np.exp(lam * C_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(C))
        w[np.argmax(C)] = 1.0
    else:
        w /= w_sum

    # Compute grad[j] = 2 * sum_k w[k] * a[k-j]
    # = 2 * (w * a_flipped)[n-1+j]
    a_flip = a[::-1].copy()
    req = len(w) + n - 1
    fft_size = 1 << (req - 1).bit_length()
    W = np.fft.rfft(w, n=fft_size)
    AF = np.fft.rfft(a_flip, n=fft_size)
    conv = np.fft.irfft(W * AF, n=fft_size)
    grad = 2.0 * conv[n - 1: 2 * n - 1]
    return grad

def lp_frank_wolfe(a, active_set, max_time=3.0):
    """
    Solve LP: min M s.t. sum(g)=1, g>=0, (a*g)[k] <= M for k in active_set.
    Returns g (normalized direction) or None on failure.
    Uses scipy linprog with HiGHS.
    """
    n = len(a)
    S = sorted([int(k) for k in active_set if 0 <= k < 2 * n - 1])
    K = len(S)
    if K == 0:
        return None

    # Build constraint matrix
    A = np.zeros((K, n), dtype=np.float64)
    for idx, k in enumerate(S):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start < j_end:
            A[idx, j_start:j_end] = a[k - np.arange(j_start, j_end)]

    # LP: min [0,...,0,1] @ [g; M]
    # s.t. sum(g) = 1 (equality)
    #      A@g - M <= 0 (inequality)
    #      g >= 0, M >= 0 (bounds)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_ub = np.hstack([A, -np.ones((K, 1))])
    b_ub = np.zeros(K)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        result = so.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                            bounds=bounds, method='highs',
                            options={'time_limit': max_time, 'presolve': True})
        if result.success and result.x is not None:
            g = np.maximum(result.x[:n], 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

    """
    Main optimization: smoothed gradient descent with momentum,
    adaptive lambda, and adaptive learning rate.
    """
    a = normalize_simplex(a0.copy())
    n = len(a)
    best_a = a.copy()
    best_val = eval_obj_normalized(a)
    t0 = time.monotonic()
    velocity = np.zeros(n)
    no_improve = 0
    rng = np.random.default_rng(12345)

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 15000:
            break

        C = fast_autocorr(a)
        M_curr = C.max()
        cur_val = 2.0 * n * M_curr

        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive lambda: start with many peaks considered, narrow to top peaks
        progress = min(1.0, t / 4000.0)
        lam = 10.0 + 300.0 * progress * progress

        # Compute smoothed gradient
        grad = compute_smoothed_gradient(a, C, lam)

        # Center the gradient (to preserve sum approximately)
        grad -= grad.mean()

        # Adaptive learning rate with cosine schedule
        base_lr = 0.08
        lr = base_lr * (0.5 + 0.5 * np.cos(np.pi * progress)) / (1.0 + 0.002 * t)

        # Nesterov-style momentum
        momentum = 0.85 + 0.1 * progress
        velocity = momentum * velocity + (1.0 - momentum) * lr * grad
        a_new = a - velocity

        # Project to simplex
        a_new = project_simplex(a_new)

        # Evaluate
        new_C = fast_autocorr(a_new)
        new_val = 2.0 * n * new_C.max()

        # Accept/reject with simulated annealing
        temperature = max(0.001, 0.05 * np.exp(-t / 2000.0))
        if new_val < cur_val or (rng.random() < np.exp(-(new_val - cur_val) / temperature)):
            a = a_new
        else:
            # Try a smaller step
            half_step = project_simplex(a - 0.1 * velocity)
            half_val = 2.0 * n * fast_autocorr(half_step).max()
            if half_val < cur_val:
                a = half_step
            velocity *= 0.3

        if no_improve > 3000 and no_improve % 500 == 0:
            perturbation = rng.standard_normal(n) * 0.02
            a_pert = a + perturbation
            a_pert = project_simplex(a_pert)
            pert_val = 2.0 * n * fast_autocorr(a_pert).max()
            if pert_val < best_val:
                a = a_pert
                best_val = pert_val
                best_a = a.copy()
                no_improve = 0
            velocity = np.zeros(n)

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print("    SGD t={:6d} ({:.1f}s) obj={:.7f} lam={:.0f} no_imp={}".format(
                t, elapsed, best_val, lam, no_improve))

    return best_a, best_val

def lp_fw_iteration(a0, time_limit, solver_time=2.0, init_K=100, max_K=2000,
                    verbose=True):
    """
    LP-based Frank-Wolfe with adaptive active set.
    """
    a = normalize_simplex(a0.copy())
    n = len(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_val = 2.0 * n * best_m
    t0 = time.monotonic()
    active_set = set()
    no_improve = 0

    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 3000:
            break

        C = fast_autocorr(a)
        M_curr = C.max()
        cur_val = 2.0 * n * M_curr

        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_m = M_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update active set
        if t == 1 or t % 50 == 0:
            top_idx = np.argsort(C)[::-1]
            target_K = min(max_K, init_K + t * 3)
            for k in top_idx[:target_K]:
                active_set.add(int(k))
            # Prune: keep only those close to max
            if len(active_set) > max_K:
                clist = sorted(active_set)
                margins = np.array([M_curr - C[k] for k in clist])
                keep = np.argsort(margins)[:max_K]
                active_set = set(int(clist[i]) for i in keep)
        active_set.add(int(np.argmax(C)))

        # Solve LP
        g = lp_frank_wolfe(a, active_set, max_time=solver_time)
        if g is None:
            if len(active_set) > init_K // 2:
                clist = sorted(active_set)
                C_arr = fast_autocorr(a)
                active_set = set(int(clist[i]) for i in np.argsort(
                    [C_arr.max() - C_arr[k] for k in clist])[:init_K])
            continue

        # Check violations and grow active set
        from scipy.signal import convolve as sp_conv
        conv_ag = sp_conv(a, g, mode='full')
        violations = np.where(conv_ag > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ag[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:20]]
            for k in top_v:
                active_set.add(int(k))

        # Line search: try multiple step sizes
        best_alpha = 0.0
        best_trial_val = cur_val
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            trial = np.maximum(a * (1 - alpha) + alpha * g, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            Mt = fast_autocorr(trial).max()
            if Mt < best_trial_val:
                best_trial_val = Mt
                best_alpha = alpha

        # Bisection refinement around best_alpha
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(5):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                t1 = np.maximum(a * (1 - m1) + m1 * g, 0.0)
                t2 = np.maximum(a * (1 - m2) + m2 * g, 0.0)
                s1, s2 = t1.sum(), t2.sum()
                if s1 < 1e-15 or s2 < 1e-15:
                    break
                v1 = fast_autocorr(t1 / s1).max()
                v2 = fast_autocorr(t2 / s2).max()
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            best_alpha = (lo + hi) / 2.0

        if best_alpha > 0:
            a = np.maximum(a * (1 - best_alpha) + best_alpha * g, 0.0)
            a = a / a.sum()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t={:5d} ({:.1f}s) obj={:.7f} |S|={}".format(
                t, elapsed, 2.0 * n * best_m, len(active_set)))

    return best_a, 2.0 * n * best_m

def coord_descent_polish(a0, time_limit, n_peaks=8, verbose=True):
    """
    Targeted coordinate descent: move mass from peak contributors to quiet positions.
    """
    a = normalize_simplex(a0.copy())
    n = len(a)
    best_a = a.copy()
    best_val = eval_obj_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0

    fracs = [0.0005, 0.001, 0.002, 0.004, 0.008, 0.015, 0.03]

    while time.monotonic() - t0 < time_limit and stall < 3000:
        C = fast_autocorr(a)
        M_curr = C.max()
        cur_val = 2.0 * n * M_curr

        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Get top peaks
        top_k = np.argsort(C)[::-1][:n_peaks]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start >= j_end:
                continue

            idx = np.arange(j_start, j_end)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            # Find source (highest contributor)
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find quiet positions (low pressure from all top peaks)
            pressure = np.zeros(n)
            for kk in top_k[:6]:
                kk = int(kk)
                ps = max(0, kk)
                pe = min(n, kk + 1)
                if ps < pe:
                    ii = np.arange(ps, pe)
                    pressure[ii] += a[ii] * a[kk - ii]

            pressure[src] = 1e10
            quiet = np.argsort(pressure)[:50]

            # Try moving mass from src to quiet positions
            for dst in quiet:
                dst = int(dst)
                if dst == src:
                    continue
                found = False
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-16:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_a_is_best:  # placeholder, will fix
                        pass
                    if Mt * 2.0 * n < best_val * (1 - 1e-10):
                        a = a_t
                        best_val = 2.0 * n * Mt
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break

            if not improved and stall > 200:
                # Random perturbation
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if 2.0 * n * Mn < best_val:
                            best_val = 2.0 * n * Mn
                            best_a = a.copy()
                            stall = 0
                            improved = True
                break

        if not improved:
            stall += 1

        if verbose and stall % 500 == 0 and stall > 0:
            print("    CoordDescent ({:.1f}s) obj={:.7f} stall={}".format(
                time.monotonic() - t0, best_val, stall))

    return best_a, best_val

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape and normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss
