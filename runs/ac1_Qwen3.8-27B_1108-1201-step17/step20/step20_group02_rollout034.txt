def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def subgradient_at_peak(p, k_star):
    """Subgradient of (p*p)[k_star] w.r.t. p."""
    n = len(p)
    g = np.zeros(n)
    i_start = max(0, k_star - n + 1)
    i_end = min(n, k_star + 1)

        idx = np.arange(i_start, i_end)
        g[idx] = 2.0 * p[k_star - idx]
        if k_star % 2 == 0:

            if 0 <= c < n:
                g[c] = p[c]
    return g

def mirror_descent_fast(p0, time_limit, eta0=0.2, verbose=True):
    """Multiplicative weights with adaptive step and occasional large restarts."""
    n = len(p0)
    p = norm_sum1(p0)
    best_p = p.copy()
    best_m = fast_convolve(p).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    eta = eta0
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        if stall > 2000:
            break
        it += 1
        ac = fast_convolve(p)
        k_star = int(np.argmax(ac))
        m_curr = ac[k_star]

        if m_curr < best_m - 1e-14:
            best_m = m_curr
            best_p = p.copy()
            stall = 0
        else:
            stall += 1

        g = subgradient_at_peak(p, k_star)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-20:
            break

        # Normalized step for better conditioning
        eta_eff = eta / max(g_norm, 1e-10)

        # Multiplicative update
        log_p = np.log(np.maximum(p, 1e-30)) - eta_eff * g
        log_p -= log_p.max()
        p_new = np.exp(log_p)
        p_new = np.maximum(p_new, 0.0)
        s = p_new.sum()
        if s < 1e-15:
            eta *= 0.5
            continue
        p_new = p_new / s

        m_new = fast_convolve(p_new).max()
        if m_new < m_curr * (1.0 - 1e-8):
            p = p_new
            eta = min(eta * 1.005, 2.0)
        else:
            eta *= 0.85
            if eta < 1e-7:
                # Try a random restart

                    p = best_p + noise
                    p = np.maximum(p, 0)
                    s = p.sum()
                    if s > 1e-10:
                        p = p / s
                        eta = eta0
                        stall = 0
                else:
                    break

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print("    MD it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", eta=" + repr(eta))

    return best_p, best_m

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
