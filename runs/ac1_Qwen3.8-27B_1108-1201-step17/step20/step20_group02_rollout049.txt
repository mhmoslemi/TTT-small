def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def proj_simplex(v, s=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_constraint_matrix_sparse(y, constraint_list, n):
    """Build sparse constraint matrix for LP."""
    K = len(constraint_list)
    rows = []
    cols = []
    vals = []
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            cv = y[k_int - idx]
            mask = cv > 1e-15
            if mask.any():
                rows.extend([jj] * mask.sum())
                cols.extend((idx[mask]).tolist())
                vals.extend(cv[mask].tolist())
    if len(rows) == 0:
        return sp.csr_matrix((K, n))
    mat = sp.coo_matrix((vals, (rows, cols)), shape=(K, n))
    return mat.tocsr()

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def proj_simplex(v, s=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_constraint_matrix_sparse(y, constraint_list, n):
    K = len(constraint_list)
    rows = []
    cols = []
    vals = []
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            cv = y[k_int - idx]
            mask = cv > 1e-15
            if mask.any():
                rows.extend([jj] * int(mask.sum()))
                cols.extend(idx[mask].tolist())
                vals.extend(cv[mask].tolist())
    if len(rows) == 0:
        return sp.csr_matrix((K, n))
    mat = sp.coo_matrix((vals, (rows, cols)), shape=(K, n))
    return mat.tocsr()


def lp_fw(a0, time_limit, solver, initial_K=600, max_K=4500, add_per_iter=80, verbose=True):
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac_y = fast_convolve(y)
    best_m = ac_y.max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 500
    iter_count = 0
    stagnation_count = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if stagnation_count > 6:
                break
            ac_curr = fast_convolve(y)
            n_pert = max(100, len(active_set) // 4)
            rng_s = np.random.default_rng(stagnation_count * 7919)
            keep = rng_s.choice(len(active_set), size=len(active_set) - n_pert, replace=False)
            new_adds = np.argsort(ac_curr)[::-1][:n_pert * 3]
            new_set = set([active_set[i] for i in keep])
            for k in new_adds:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]
            no_improve = 0
            stagnation_count += 1
            continue

        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            stagnation_count = 0
        else:
            no_improve += 1

        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep_count = int(max_K * 0.55)
            keep_idx = np.argsort(margins)[:keep_count]
            active_set_base = [active_set[i] for i in keep_idx]
            top50 = np.argsort(ac)[::-1][:50]
            for v in top50:
                v_int = int(v)
                if v_int not in active_set_base:
                    active_set_base.append(v_int)
            active_set = sorted(set(active_set_base))[:max_K]

        K = len(active_set)
        if K < 10:
            break

        coeffs = build_constraint_matrix_sparse(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(200, K // 3)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]

        max_viol = conv_ys.max() - M_curr
        if max_viol > 1e-9:
            viol = np.where(conv_ys > M_curr + 1e-10)[0]
            viol_amt = conv_ys[viol] - M_curr
            top_v = viol[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        best_alpha = 1.0 / (iter_count + 2.0)
        best_trial_M = fast_convolve(y).max()

        alphas_to_try = [1.0, 0.85, 0.7, 0.55, 0.4, 0.3, 0.2, 0.12,
                         max(0.04, 1.0 / (iter_count + 1.0))]

        for alpha in alphas_to_try:
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 2:
            y = g_dir.copy()
        else:
            y = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", |S|=" + str(len(active_set)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", iters=" + str(iter_count))
    return best_a, best_m


def fista_smoothed(a0, time_limit, tau_start=0.1, tau_max=150.0, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    y = a.copy()
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    fft_s = 1 << (2 * n + 4).bit_length()
    t_mom = 1.0
    l0 = max(1.0, tau_max * best_m * n)
    lr0 = 1.0 / max(l0, 1e6)
    rng = np.random.default_rng(314)
    k = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        k += 1
        frac = min(1.0, k / (time_limit * 60))
        tau = tau_start * (tau_max / tau_start) ** frac
        ac = fast_convolve(y)
        ac_shift = ac - ac.max()
        tau_ac = np.clip(tau * ac_shift, -400, 400)
        weights = np.exp(tau_ac)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            p = np.ones(len(ac)) / len(ac)
        else:
            p = weights / w_sum
        P = np.fft.rfft(p, fft_s)
        Yc = np.conj(np.fft.rfft(y, fft_s))
        z = np.fft.irfft(P * Yc, fft_s)[:n]
        grad = 2.0 * z
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        lr = lr0 / t_new
        a_new = proj_simplex(y - lr * grad)
        y = a_new + (t_mom - 1.0) / t_new * (a_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a_new.copy()
        else:
            y = y / s_y
        t_mom = t_new
        M_new = fast_convolve(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        if stall > 3000:
            noise = rng.normal(0, 0.01, n) * best_a
            y = np.maximum(best_a + noise, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y / s_y
            a_new = y.copy()
            t_mom = 1.0
            stall = 0
        if verbose and k % 5000 == 0:
            elapsed = time.monotonic() - t0
            print("    FISTA k=" + str(k) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", tau=" + repr(round(tau, 2)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    FISTA done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", k=" + str(k))
    return best_a, best_m


def subgrad_nesterov(a0, time_limit, verbose=True):
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    best_a = x.copy()
    best_m = fast_convolve(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    t_mom = 1.0
    eta0 = 0.08
    stall = 0
    max_stall = 2500
    k = 0

    while time.monotonic() - t0 < time_limit:
        if stall > max_stall:
            break
        k += 1
        eta = eta0 / (np.sqrt(k) + 1.0)
        ac = fast_convolve(y)
        k_argmax = int(np.argmax(ac))
        grad = np.zeros(n)
        i_start = max(0, k_argmax - n + 1)
        i_end = min(n, k_argmax + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] = 2.0 * y[k_argmax - idx]
        perturb = 0.005 / (1.0 + k / 300.0)
        grad += perturb * rng.standard_normal(n) * y
        x_new = proj_simplex(y - eta * grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x_new.copy()
        else:
            y = y / s_y
        t_mom = t_new
        x = x_new
        M_new = fast_convolve(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        if stall > 800 and stall % 800 == 0:
            noise = rng.normal(0, 0.02, n) * best_a
            x_p = np.maximum(best_a + noise, 0.0)
            sp_ = x_p.sum()
            if sp_ > 1e-12:
                x = x_p / sp_
                y = x.copy()
            t_mom = 1.0
            eta0 = eta0 * 1.15
        if verbose and k % 5000 == 0:
            elapsed = time.monotonic() - t0
            print("    SubGrad k=" + str(k) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    SubGrad done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", k=" + str(k))
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=77, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 500
    step_scales = [0.001, 0.003, 0.008, 0.015, 0.03, 0.06]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue
            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    in_peak[s2:e2 + 1] = True
            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 5:
                candidates = np.arange(n)
            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])
            if dst == src:
                continue
            found = False
            for frac in step_scales:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a
