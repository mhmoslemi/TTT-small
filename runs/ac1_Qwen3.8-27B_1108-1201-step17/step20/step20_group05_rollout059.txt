def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def fft_conv2(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if b.size != n:
        b = np.resize(b, n)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def cross_convolve_p_a(p, a):
    p = np.asarray(p, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    total = 3 * n - 2
    if p.size != 2 * n - 1:
        p = np.resize(p, 2 * n - 1)
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    out = conv[n - 1:2 * n - 1][::-1]
    out = np.where(np.isfinite(out), out, 0.0)
    return out

def make_v(a, ref=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 1:
        return np.array([], dtype=np.float64), 0
    if ref is None:
        if np.any(a > 0.0):
            ref = int(np.argmax(a))
        else:
            ref = 0
    ref = int(ref)
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.where(np.isfinite(v), v, -700.0)
    return v, ref


def from_v(v, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def lse_objective(v, beta, top_idx, ref):
    n = v.size + 1
    if n <= 1:
        return 1e100, np.array([], dtype=np.float64)

    u = np.empty(n, dtype=np.float64)
    if ref > 0:
        u[:ref] = v[:ref]
    u[ref] = 0.0
    if ref < n - 1:
        u[ref + 1:] = v[ref:]

    um = float(np.max(u))
    a = np.exp(u - um)
    s = float(a.sum())
    if s <= 0.0:
        a = np.ones(n) / n
    else:
        a = a / s

    size = 2 * n - 1
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)

    K = len(top_idx)
    if K <= 0:
        K = 10
        top_idx = np.argsort(c)[::-1][:K]

    cv = c[top_idx]
    m = float(np.max(cv))
    shifted = beta * (cv - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0 or not math.isfinite(se):
        f = 1e100
        return f, np.zeros(v.size, dtype=np.float64)
    p = e / se

    pf = np.zeros(size, dtype=np.float64)
    pf[top_idx] = p

    Pr = np.fft.rfft(pf[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    ga = 2.0 * conv[n - 1:2 * n - 1][::-1]
    ga = np.where(np.isfinite(ga), ga, 0.0)

    f = m + math.log(se) / beta

    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)

    gv = np.empty(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]

    gv = np.where(np.isfinite(gv), gv, 0.0)
    if not math.isfinite(f):
        f = 1e100
    return f, gv


def run_lbfegs(a0, beta, top_count, maxfun=300, verbose=True):

def game_direction(f, active_idx, mw_iters=30, seed=0):
    f = norm_sum1(f)
    n = f.size
    size = 2 * n - 1
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K <= 0:
        return np.ones(n, dtype=np.float64) / float(n)

    rng = np.random.default_rng(seed)
    logits = np.zeros(K, dtype=np.float64)
    logits += rng.normal(0.0, 0.03, K)

    def normalize_logits(loc):
        loc = np.where(np.isfinite(loc), loc, -1000.0)
        logmx = float(np.max(loc))
        w = np.exp(loc - logmx)
        s = float(w.sum())
        if s <= 0.0:
            w = np.ones(K) / K
        else:
            w = w / s
        return w

    lambda_w = normalize_logits(logits)
    g = None

    for _ in range(max(1, int(mw_iters))):
        pf = np.zeros(size, dtype=np.float64)
        pf[active_idx] = lambda_w

        pressure = cross_convolve_p_a(pf, f)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)

        pmx = float(np.max(pressure))
        pmn = float(np.min(pressure))
        ptp = pmx - pmn
        if ptp <= 1e-30:
            g = np.ones(n, dtype=np.float64) / float(n)
        else:
            scaled = (pressure - pmn) / ptp
            w = np.exp(-10.0 * scaled)
            w = np.where(np.isfinite(w), w, 0.0)
            w += 1e-14
            g = w / float(w.sum())
            # Add small diffuse component
            g = 0.95 * g + 0.05 * (np.ones(n, dtype=np.float64) / float(n))
            g = g / float(g.sum())

        h = fft_conv2(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmx = float(np.max(ha))
        hmn = float(np.min(ha))
        htp = hmx - hmn
        if htp > 1e-30:
            eta = 15.0 / htp
            logits += eta * (ha - float(np.mean(ha)))
        lambda_w = normalize_logits(logits)

    return g

        cols = np.arange(s, e, dtype=np.int32)
        row_parts.append(np.full(cols.size, r, dtype=np.int32))
        col_parts.append(cols)
        vals = f[int(k) - cols].astype(np.float64)
        val_parts.append(vals)

    if not row_parts:
        return None

    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))

    c_obj = np.zeros(n, dtype=np.float64)
    c_obj[:] = 1.0
    c_obj[:] = -1.0  # Minimize -sum(g) = maximize sum(g)

    neg_one = scipy.sparse.csc_matrix(-np.ones((K, 1)), dtype=np.float64)
    A_ub = scipy.sparse.hstack((A, neg_one), format="csr")
    A_eq = np.ones((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 0.0
    A_eq[0, n] = 1.0
    b_eq = np.ones(1, dtype=np.float64)
    b_ub = np.zeros(K, dtype=np.float64)

    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = scipy.optimize.linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="HiGHS",
            options={"disp": False, "time_limit": float(time_limit)}
        )
    except Exception:
        return None

    if res is None or not getattr(res, "success", False):
        return None

    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.where(np.isfinite(g) & (g > 0.0), g, 0.0)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs


def line_search_mix(a, g, cf, cg, fg, current_M, t):
    n = a.size
    base_alphas = [
        max(0.01, 1.0 / (t + 1.0)),
        max(0.02, 2.0 / (t + 2.0)),
        0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.50, 0.70, 1.0
    ]
    alphas = []
    seen = set()
    for al in base_alphas:
        al = float(al)
        if not math.isfinite(al) or al <= 0.0 or al > 1.0:
            continue
        key = int(round(al * 1000.0))
        if key in seen:
            continue
        seen.add(key)
        alphas.append(al)
    if not alphas:
        alphas = [0.1]

    best_alpha = 0.0
    best_m = current_M
    for alpha in alphas:
        one = 1.0 - alpha
        val = (one * one) * cf + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
        m = float(np.max(val))
        if math.isfinite(m) and m < best_m - 1e-18:
            best_m = m
            best_alpha = alpha

    if best_alpha <= 0.0:
        return a.copy(), best_m, 0.0

    new_a = (1.0 - best_alpha) * a + best_alpha * g
    new_a = np.maximum(new_a, 0.0)
    s = float(new_a.sum())
    if s < 1e-15:
        return a.copy(), best_m, 0.0
    new_a = new_a / s
    return new_a, best_m, best_alpha


def frank_wolfe_search(a0, budget, active_max=2000, game_iters=28,
                       exact_every=5, exact_time=8.0, verbose=True):
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = proxy_max(c, None)
    best_a = a.copy()
    best_M = M
    active = active_top(c, None, active_max)

    no_improve = 0
    t = 0
    while time.monotonic() - t0 < time_limit - 5 and no_improve < 120:
        t += 1
        time_left = time_limit - (time.monotonic() - t0)
        if time_left < 20:
            break

        g = None
        # Try exact LP periodically
        if exact_every > 0 and (t % exact_every == 0) and time_left > 80 and active.size <= 800:
            g = lp_direction(a, active, exact_time, max_k=800)

        if g is None:
            g = game_direction(a, active, mw_iters=game_iters, seed=5000 + 37 * t)

        if g is None:
            break

        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        new_a, new_M, alpha = line_search_mix(a, g, c, cg, fg, M, t)

        if new_M < M - 1e-17:
            one = 1.0 - alpha
            c = (one * one) * c + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
            c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
            a = new_a
            M = new_M
            no_improve = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            active = active_top(c, None, active_max)
            if verbose and (t <= 3 or t % 15 == 0):
                print("    FW t=" + str(t) +
                      " score_proxy=" + repr(2.0 * n * best_M) +
                      " alpha=" + repr(alpha))
        else:
            no_improve += 1
            if no_improve % 10 == 9:
                # Perturb to escape
                rng = np.random.default_rng(12345 + t)
                scale = 0.01 + 0.01 * (no_improve // 10)
                perturb = rng.normal(0.0, scale, n) * best_a
                a = best_a + perturb
                a = np.maximum(a, 1e-14)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = proxy_max(c, None)
                active = active_top(c, None, active_max)
                no_improve = 0

    return best_a


def apply_single_move(c, a, s, d, tmove):
    c_new = c.copy()
    n = a.size
    # Transfer tmove from position s to position d
    # c_new[k] = c[k] + 2*tmove*a[k-d] (for valid range) - 2*tmove*a[k-s] (for valid range)
    # + tmove^2 at 2d, 2s; - 2*tmove^2 at s+d
    lo_d = max(0, d - n + 1)
    hi_d = min(2 * n - 1, d + n - 1)
    if hi_d >= lo_d:
        c_new[lo_d:hi_d + 1] += 2.0 * tmove * a[d - lo_d:d - lo_d + (hi_d - lo_d + 1)]

    lo_s = max(0, s - n + 1)
    hi_s = min(2 * n - 1, s + n - 1)
    if hi_s >= lo_s:
        c_new[lo_s:hi_s + 1] -= 2.0 * tmove * a[s - lo_s:s - lo_s + (hi_s - lo_s + 1)]

    t2 = tmove * tmove
    if 0 <= 2 * d < 2 * n - 1:
        c_new[2 * d] += t2
    if 0 <= 2 * s < 2 * n - 1:
        c_new[2 * s] += t2
    if 0 <= s + d < 2 * n - 1:
        c_new[s + d] -= 2.0 * t2

    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
    return c_new


def coordinate_polish_aggressive(a0, budget, verbose=True):
    """More aggressive coordinate polish with larger moves and multi-coordinate updates."""
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = proxy_max(c, None)
    best_a = a.copy()
    best_M = M
    rng = np.random.default_rng(42)
    stall = 0
    rnd = 0

    while time.monotonic() - t0 < time_limit - 2 and stall < 200:
        rnd += 1
        if n <= 2:
            break

        # Get top peaks and compute pressure
        top_k = min(30, 2 * n - 1)
        top_idx = active_top(c, None, top_k)
        if top_idx.size == 0:
            break

        p_top = np.zeros(2 * n - 1, dtype=np.float64)
        p_top[top_idx] = 1.0 / float(top_idx.size)
        pressure = cross_convolve_p_a(p_top, a, 1)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)

        # Source candidates: high pressure AND has mass
        src_score = pressure * a
        n_src = min(12, n)
        src_candidates = np.argpartition(src_score, n_src - 1)[:n_src]

        # Destination candidates: low pressure relative
        dst_score = -pressure
        n_dst = min(16, n)
        dst_candidates = np.argpartition(dst_score, n_dst - 1)[:n_dst]

        best_cand = None

        # Try transfers with various fractions
        for srr in src_candidates:
            s = int(srr)
            if a[s] < 1e-12:
                continue
            for drr in dst_candidates:
                d = int(drr)
                if d == s:
                    continue
                for frac in (0.01, 0.02, 0.05, 0.10, 0.15, 0.25):
                    tmove = a[s] * frac
                    tmove = min(tmove, a[s] * 0.95)
                    if tmove <= 1e-16:
                        continue
                    c_new = apply_single_move(c, a, s, d, tmove)
                    m_new = proxy_max(c_new, None)
                    if m_new < (best_cand[0] if best_cand else M) - 1e-18:
                        best_cand = (m_new, s, d, tmove)

        # Also try random exploratory moves
        for _ in range(40):
            s = int(rng.integers(n))
            d = int(rng.integers(n))
            if s == d or a[s] < 1e-12:
                continue
            frac = float(rng.uniform(0.005, 0.15))
            tmove = a[s] * frac
            tmove = min(tmove, a[s] * 0.95)
            if tmove <= 1e-16:
                continue
            c_new = apply_single_move(c, a, s, d, tmove)
            m_new = proxy_max(c_new, None)
            if m_new < (best_cand[0] if best_cand else M) - 1e-18:
                best_cand = (m_new, s, d, tmove)

        if best_cand is not None:
            m_new, s, d, tmove = best_cand
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = fft_conv(a)
            M = proxy_max(c, None)
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Periodic perturbation when stalled
            if stall % 15 == 14:
                scale = min(0.05, 0.01 * (stall // 15))
                perturb = rng.normal(0.0, scale, n) * best_a
                a = best_a + perturb
                a = np.maximum(a, 1e-14)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = proxy_max(c, None)
                stall = 0

    return best_a


def peaking_repair(a0, budget, verbose=True):
    """
    Targeted repair: iteratively reduce the top peaks by finding coordinates
    whose removal reduces the peak the most.
    """
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = proxy_max(c, None)
    best_a = a.copy()
    best_M = M
    rng = np.random.default_rng(99)

    for _ in range(500):
        if time.monotonic() - t0 > time_limit - 2:
            break
        if n <= 2:
            break

        # Find the worst peak
        worst_k = int(np.argmax(c))
        worst_val = c[worst_k]

        # Find which coordinates contribute most to this peak
        # c[worst_k] = sum_i a[i] * a[worst_k - i] where 0 <= i < n, 0 <= worst_k - i < n
        lo = max(0, worst_k - n + 1)
        hi = min(n - 1, worst_k)
        if lo > hi:
            break

        # Contribution of coordinate i to c[worst_k] is a[i] * a[worst_k - i]
        contrib = a[lo:hi + 1] * a[worst_k - hi:worst_k - lo + 1]

        # Try removing a small amount from the coordinate with highest contribution
        # and redistributing to a low-pressure coordinate
        if contrib.max() <= 1e-15:
            break

        # Find the best coordinate to reduce
        idx_in_range = int(np.argmax(contrib))
        s = lo + idx_in_range

        # Find a good destination: low overall pressure
        other_peaks = c.copy()
        other_peaks[worst_k] = 0
        p_other = np.zeros(2 * n - 1, dtype=np.float64)
        top_other = active_top(other_peaks, None, min(20, 2 * n - 1))
        if top_other.size > 0:
            p_other[top_other] = 1.0 / top_other.size
        else:
            p_other[:] = 1.0 / (2 * n - 1)

        pressure = cross_convolve_p_a(p_other, a)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)

        # Try several destinations
        n_dst = min(10, n)
        dst_candidates = np.argpartition(pressure, n_dst - 1)[:n_dst]

        best_cand = None
        for drr in dst_candidates:
            d = int(drr)
            if d == s:
                continue
            for frac in (0.01, 0.03, 0.05, 0.10):
                tmove = a[s] * frac
                tmove = min(tmove, a[s] * 0.9)
                if tmove <= 1e-16:
                    continue
                c_new = apply_single_move(c, a, s, d, tmove)
                m_new = proxy_max(c_new, None)
                if m_new < (best_cand[0] if best_cand else M) - 1e-18:
                    best_cand = (m_new, s, d, tmove)

        if best_cand is not None:
            m_new, s, d, tmove = best_cand
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = fft_conv(a)
            M = proxy_max(c, None)
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
        else:
            # No improvement found in several rounds; add small noise
            if rng.random() < 0.3:
                s = int(rng.integers(n))
                if a[s] < 1e-12:
                    continue
                d = int(rng.integers(n))
                if d == s:
                    continue
                tmove = a[s] * 0.01
                a[s] -= tmove
                a[d] += tmove
                a = np.maximum(a, 0.0)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = proxy_max(c, None)

    return best_a


def multi_start_explore(best_a, budget, verbose=True):
    """Try several perturbations of the best and re-optimize briefly."""
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a_best = norm_sum1(best_a)
    n = a_best.size
    score_best = 2.0 * n * proxy_max(fft_conv(a_best), None)

    rng = np.random.default_rng(777)
    n_starts = min(5, max(2, int(time_limit / 120)))

    for i in range(n_starts):
        time_left = time_limit - (time.monotonic() - t0)
        if time_left < 60:
            break

        # Generate perturbation
        scale = 0.02 + 0.01 * i
        perturb = rng.normal(0.0, scale, n) * a_best
        a_try = a_best + perturb
        a_try = np.maximum(a_try, 1e-14)
        a_try = a_try / float(a_try.sum())

        # Brief L-BFGS polish
        c_try = fft_conv(a_try)
        top_count = max(50, (2 * n - 1) // 15)
        a_polished = run_lbfegs(a_try, 80.0, top_count, maxfun=80, verbose=False)

        # Brief coordinate polish
        time_left = time_limit - (time.monotonic() - t0)
        if time_left > 30:
            a_polished = coordinate_polish_aggressive(a_polished, min(20.0, time_left / 2), verbose=False)

        score_try = 2.0 * a_polished.size * proxy_max(fft_conv(a_polished), None)
        if verbose:
            print("  Multi-start " + str(i) + ": score=" + repr(score_try) +
                  " (best so far: " + repr(score_best) + ")")

        if score_try < score_best:
            score_best = score_try
            a_best = a_polished

    return a_best


def propose_candidate():
    t_start = time.monotonic()
    total_budget = 1700.0  # Leave buffer

    print("=== Starting optimization, budget=" + repr(total_budget) + "s ===")

    # Initialize
    try:
        parent_list = height_sequence_1
    except Exception:
        parent_list = None

    seq_best = None
    if parent_list is not None:
        try:
            p = np.asarray(parent_list, dtype=np.float64)
            p = np.where(np.isfinite(p), p, 0.0)
            np.clip(p, 0.0, 1000.0, out=p)
            if p.ndim != 1:
                p = p.ravel()
            if p.size > 0 and float(p.sum()) > 1e-10:
                seq_best = norm_sum1(p)
        except Exception:
            seq_best = None

    if seq_best is None:
        # Default: sine-squared
        n = 5000
        x = np.linspace(0.0, 1.0, n)
        seq_best = norm_sum1(np.sin(np.pi * x) ** 2)

    best_score = safe_eval(seq_best)
    if not math.isfinite(best_score):
        best_score = 2.0 * seq_best.size * proxy_max(fft_conv(seq_best), None)

    print("Initial: n=" + str(seq_best.size) + " score=" + repr(best_score))

    # Work with the best sequence directly
    work = seq_best.copy()
    n = work.size

    # Phase 1: Adaptive L-BFGS (45% of budget)
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 300:
        phase_budget = min(time_left * 0.42, 700.0)
        print("\n=== Phase 1: Adaptive L-BFGS (budget=" + repr(phase_budget) + "s) ===")
        a_new = lbfegs_adaptive(work, phase_budget, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_new, "lbfegs_adapt", t_start)
        work = seq_best.copy() if seq_best.size <= 9000 else work
        print("  After Phase 1: score=" + repr(best_score))

    # Phase 2: Frank-Wolfe (30% of budget)
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 200:
        phase_budget = min(time_left * 0.32, 500.0)
        print("\n=== Phase 2: Frank-Wolfe (budget=" + repr(phase_budget) + "s) ===")
        a_fw = frank_wolfe_search(work, phase_budget,
                                  active_max=2000,
                                  game_iters=30,
                                  exact_every=5,
                                  exact_time=10.0,
                                  verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_fw, "fw", t_start)
        work = seq_best.copy() if seq_best.size <= 9000 else work
        print("  After Phase 2: score=" + repr(best_score))

    # Phase 3: Aggressive coordinate polish (15% of budget)
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 120:
        phase_budget = min(time_left * 0.18, 300.0)
        print("\n=== Phase 3: Aggressive coordinate polish (budget=" + repr(phase_budget) + "s) ===")
        a_cp = coordinate_polish_aggressive(work, phase_budget, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_cp, "coord_agg", t_start)
        work = seq_best.copy() if seq_best.size <= 9000 else work
        print("  After Phase 3: score=" + repr(best_score))

    # Phase 4: Peaking repair (8% of budget)
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 80:
        phase_budget = min(time_left * 0.10, 150.0)
        print("\n=== Phase 4: Peaking repair (budget=" + repr(phase_budget) + "s) ===")
        a_pk = peaking_repair(work, phase_budget, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_pk, "peaking", t_start)
        work = seq_best.copy() if seq_best.size <= 9000 else work
        print("  After Phase 4: score=" + repr(best_score))

    # Phase 5: Multi-start exploration (5% of budget)
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 150:
        phase_budget = min(time_left * 0.10, 200.0)
        print("\n=== Phase 5: Multi-start exploration (budget=" + repr(phase_budget) + "s) ===")
        a_ms = multi_start_explore(work, phase_budget, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_ms, "multi_start", t_start)
        print("  After Phase 5: score=" + repr(best_score))

    # Phase 6: Final polish with remaining time
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 60:
        phase_budget = min(time_left * 0.6, 120.0)
        print("\n=== Phase 6: Final polish (budget=" + repr(phase_budget) + "s) ===")
        # Run coordinate polish again on the best
        a_final = coordinate_polish_aggressive(work, phase_budget * 0.6, verbose=False)
        a_final = peaking_repair(a_final, phase_budget * 0.4, verbose=False)
        seq_best, best_score = try_update(seq_best, best_score, a_final, "final", t_start)
        print("  After Phase 6: score=" + repr(best_score))

    # Final validation
    final_val = safe_eval(seq_best)
    if math.isfinite(final_val):
        if final_val < best_score:
            best_score = final_val
    else:
        seq_best = norm_sum1(np.sin(np.pi * np.linspace(0, 1, 5000)) ** 2)
        best_score = safe_eval(seq_best)

    arr = np.asarray(seq_best, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if arr.size == 0 or s < 0.01:
        arr = norm_sum1(np.sin(np.pi * np.linspace(0, 1, 5000)) ** 2)
    else:
        arr = arr / s

    elapsed = time.monotonic() - t_start
    print("\nFINAL: score=" + repr(best_score) +
          " n=" + str(arr.size) +
          " elapsed=" + repr(elapsed) + "s")

    return [float(x) for x in arr]
