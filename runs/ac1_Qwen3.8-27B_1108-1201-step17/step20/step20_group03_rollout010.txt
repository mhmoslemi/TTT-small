        AS = np.fft.rfft(a_shift, fft_size)

            best_M = m
            best_shift = s

        return np.roll(a, -best_shift), best_M
    return a.copy(), best_M

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64).flatten(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z.copy()
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def fista_topk(a0, time_limit, top_k=3, beta=100.0, eta0=0.5, restart_every=4000, verbose=True, seed=42):

def cyclic_shift_best(a0):
    n = len(a0)
    a = norm_sum1(a0)
    best_M = fast_autocorr(a).max()
    best_shift = 0
    sz = 2 * n - 1
    fft_size = 1 << (sz - 1).bit_length()
    
    for s in range(n):
        a_s = np.roll(a, -s)
        AF = np.fft.rfft(a_s, fft_size)
        b = np.fft.irfft(AF * AF, fft_size)[:sz]
        m = b.max()
        if m < best_M - 1e-16:
            best_M = m
            best_shift = s
    
    if best_shift > 0:
        return np.roll(a, -best_shift), best_M
    return a.copy(), best_M

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_i = np.interp(new_x, old_x, f_n)
    f_i = np.maximum(f_i, 0.0)
    ss = f_i.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_i / ss
