1. Spectral-shaping and parametric initialization
2. Nesterov-accelerated smoothed gradient descent with top-k peaks
3. Adaptive temperature (beta) scheduling

import numpy as np
import time
import math

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum 1, ensuring non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def simplex_project(z):
    """Project onto the probability simplex."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def interp_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_gradient(a, b, peak_weights):
    """
    Compute gradient of weighted sum of autocorrelation peaks.
    grad[i] = 2 * sum_k w_k * a[k-i] for valid k-i.
    """
    n = len(a)
    size = 2 * n - 1
    grad = np.zeros(n, dtype=np.float64)
    for k, w in zip(peak_weights[0], peak_weights[1]):
        ki = int(k)
        wv = float(w)
        if wv < 1e-15:
            continue
        i_start = max(0, ki - n + 1)
        i_end = min(n - 1, ki)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        grad[idx] += 2.0 * wv * a[ki - idx]
    return grad

def nesterov_smoothed_grad(a0, time_limit, beta_start=5.0, beta_end=800.0,
                            growth=1.008, verbose=True):
    """
    Nesterov-accelerated smoothed gradient descent on the simplex.
    Minimize (1/beta)*log(sum exp(beta*b_k)) subject to sum(a)=1, a>=0.
    """
    n = len(a0)
    x = norm_sum1(a0).copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    y = x.copy()  # Nesterov lookahead
    t = 0
    stall = 0
    max_stall = 2000
    rng = np.random.default_rng(999)
    top_k_ratio = 0.02
    top_k = max(20, int(n * top_k_ratio))

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        t += 1
        b = fast_autocorr(y)
        n_b = len(b)
        idx_all = np.argsort(b)[::-1][:top_k]
        b_top = b[idx_all]
        b_max_val = b_top[0]

        # Softmax weights
        shifted = beta * (b_top - b_max_val)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        exp_sum = exp_vals.sum()
        if exp_sum < 1e-300:
            p = np.ones(len(idx_all)) / len(idx_all)
        else:
            p = exp_vals / exp_sum

        # Gradient
        grad = compute_gradient(y, b, (idx_all, p))
        grad_norm = np.linalg.norm(grad)

        # Step size: eta = 2 / (L * (t+2)) approximately, but use adaptive
        eta = 2.0 / (grad_norm * (t + 2) + 1e-10)
        eta = min(eta, 0.3)
        eta = max(eta, 1e-7)

        # Nesterov update
        momentum = t / (t + 3.0)
        x_old = x.copy()
        x_new = simplex_project(y - eta * grad)
        y = x_new + momentum * (x_new - x_old)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = x_new.copy()
        else:
            y = y / ys
        x = x_new

        # Check actual max
        b_x = fast_autocorr(x)
        m_x = b_x.max()
        if m_x < best_m - 1e-16:
            best_m = m_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive beta
        if t % 40 == 0:
            beta = min(beta * growth, beta_end)
            top_k = max(20, int(n * top_k_ratio * (1 + 0.5 * t // 1000)))

        # Periodic restart if stalled
        if stall > 600 and stall % 300 == 0:
            noise_scale = min(0.05, 0.01 * (stall // 600))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            m_r = fast_autocorr(x_r).max()
            if m_r < best_m * 1.01:
                x = x_r.copy()
                y = x_r.copy()
                stall = max(0, stall - 300)

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    NeSG t={:<6d} beta={:<8.1f} J={:.6f} stall={}".format(t, beta, J, stall))

    return best_a, best_m

def peak_shaving_multi(a0, time_limit, verbose=True):
    """
    Aggressive multi-peak shaving: simultaneously work on top peaks,
    using pressure maps to find safe destinations.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    max_stall = 800
    top_k = max(10, n // 300)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_autocorr(a)
        peak_indices = np.argsort(b)[::-1][:top_k]

        # Build pressure map
        pressure = np.zeros(n)
        for k_pick in peak_indices[:8]:
            ki = int(k_pick)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[ki - idx]
            pressure[idx] += contrib
            pressure[ki - idx] += contrib

        # Find safe positions (low pressure)
        p_thresh = np.percentile(pressure, 30)
        safe_mask = pressure < p_thresh
        if safe_mask.sum() < n // 4:
            safe_mask = pressure < np.percentile(pressure, 60)
        if safe_mask.sum() < n // 8:
            safe_mask = np.ones(n, dtype=bool)

        improved = False

        # Work on top peaks
        for k_pick in peak_indices[:6]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k_pick)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[ki - idx]
            max_c = contrib.max()
            if max_c < 1e-25:
                continue

            # Find top 3 contributors
            top_contrib = np.argsort(contrib)[::-1][:3]
            for tc in top_contrib:
                src = int(idx[tc])
                partner = ki - src
                if partner < 0 or partner >= partner:
                    continue

                # Determine which to reduce (the larger one)
                if a[src] >= a[partner]:
                    red = src
                else:
                    red = partner

                if a[red] < 1e-15:
                    continue

                # Find best safe destination
                safe_pressure = pressure.copy()
                safe_pressure[~safe_mask] = np.max(pressure) + 1.0
                safe_pressure[red] = np.max(pressure) + 2.0
                # Also avoid positions that contribute to other top peaks
                for k_other in peak_indices[:4]:
                    ko = int(k_other)
                    if ko == ki:
                        continue
                    i_s2 = max(0, ko - n + 1)
                    i_e2 = min(n - 1, ko)
                    if i_s2 <= i_e2:
                        safe_pressure[i_s2:i_e2 + 1] += 0.5

                dst = int(np.argmin(safe_pressure))
                if dst == red:
                    continue

                # Try moving mass
                found = False
                for frac in [0.01, 0.03, 0.06, 0.1, 0.15, 0.25]:
                    amt = a[red] * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[red] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-18:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break

                if found:
                    break

        if not improved:
            stall += 1
            # Random exploration when stalled
            if stall % 40 == 0:
                n_moves = rng.integers(3, 15)
                for _ in range(n_moves):
                    src = int(rng.integers(n))
                    dst = int(rng.integers(n))
                    if src == dst or a[src] < 1e-15:
                        continue
                    frac = rng.uniform(0.01, 0.08)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[dst] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    m_new = fast_autocorr(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        stall = 0
                    elif m_new > best_m * 1.05:
                        a = best_a.copy()

            # Larger perturbation
            if stall % 200 == 0:
                noise_scale = min(0.08, 0.01 * (stall // 200 + 1))
                pert = rng.normal(0, noise_scale, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-12)
                a_r = a_r / a_r.sum()
                m_r = fast_autocorr(a_r).max()
                if m_r < best_m:
                    best_m = m_r
                    best_a = a_r.copy()
                    a = best_a.copy()
                    stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakShave ({:.1f}s): J={:.6f} stall={}".format(elapsed, J, stall))

    return best_a, best_m

def fw_step(a, active_lags, n_constraints=None):
    """
    Frank-Wolfe step: find direction g that minimizes max active constraint
    via LP (linearized at current a). Returns the FW direction.
    """
    n = len(a)
    b = fast_autocorr(a)
    M = b.max()

    if n_constraints is None:
        n_constraints = 300
    active_lags = np.argsort(b)[::-1][:n_constraints]

    # Build constraint matrix: A[g_v] = sum_i a[k-i] * g[i] for each active k
    K = len(active_lags)
    A_mat = np.zeros((K, n))
    for jj, k_val in enumerate(active_lags):
        ki = int(k_val)
        i_start = max(0, ki - n + 1)
        i_end = min(n - 1, ki)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            A_mat[jj, idx] = a[ki - idx]

    # Solve: min M_s.t. A_mat @ g <= M_s, sum(g)=1, g>=0
    # Since n can be large, use a simple approach: coordinate descent on the dual
    # Or use scipy linprog if available
    try:
        from scipy.optimize import linprog
        c = np.zeros(n)
        # Minimize M -> add M as variable (index n)
        c_full = np.zeros(n + 1)
        c_full[n] = 1.0

        A_ub = np.zeros((K, n + 1))
        A_ub[:, :n] = A_mat
        A_ub[:, n] = -1.0
        b_ub = np.zeros(K)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(None, None)]

        res = linprog(c_full, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 5000})

        if res.success and res.x is not None:
            g = np.maximum(res.x[:n], 0.0)
            g_sum = g.sum()
            if g_sum > 1e-15:
                return g / g_sum
    except Exception:
        pass

    # Fallback: move toward uniform on least-constrained positions
    row_sums = A_mat.sum(axis=1)
    if row_sums.max() > 0:
        # Weight inversely to constraint tightness
        w = 1.0 / (row_sums + 1e-10)
        g = np.zeros(n)
        for jj in range(K):
            ki = int(active_lags[jj])
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                g[idx] += w[jj] / len(idx)
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs > 1e-15:
            return g / gs
    return None

def fw_refine(a0, time_limit, verbose=True):
    """Frank-Wolfe style refinement using active set LP."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    t = 0
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        t += 1
        b = fast_autocorr(a)
        M_curr = b.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Get FW direction
        g = fw_step(a, None, n_constraints=min(200, n // 20))
        if g is None:
            break

        # Line search
        default_alpha = 2.0 / (t + 2.0)
        best_alpha = default_alpha
        best_trial = float('inf')
        for alpha in [default_alpha, 0.5 * default_alpha, 0.25 * default_alpha,
                      max(0.02, 0.1 * default_alpha)]:
            trial = (1.0 - alpha) * a + alpha * g
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial:
                best_trial = M_trial
                best_alpha = alpha

        a_new = (1.0 - best_alpha) * a + best_alpha * g
        a_new = np.maximum(a_new, 0.0)
        st = a_new.sum()
        if st > 1e-15:
            a = a_new / st

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    FW iter={:<5d} ({:.1f}s): J={:.6f}".format(t, elapsed, J))

    return best_a, best_m

def parametric_init(n, rng):
    """Generate diverse parametric initializations."""
    candidates = []
    x = np.linspace(0.0, 1.0, n)

    # Power laws
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        for beta_p in [0.5, 1.0, 1.5, 2.0, 3.0]:
            a_raw = (x + 0.01) ** alpha * (1.0 - x + 0.01) ** beta_p
            candidates.append(norm_sum1(a_raw))

    # Gaussian
    for center in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3, 0.4]:
            a_raw = np.exp(-((x - center) ** 2) / (2 * sigma ** 2))
            candidates.append(norm_sum1(a_raw))

    # Dual Gaussian (dip in middle)
    for c in [0.3, 0.7]:
        for s in [0.15, 0.25]:
            a_raw = np.exp(-((x - 0.5) ** 2) / (2 * s ** 2)) + np.exp(-((x - c) ** 2) / (2 * (s * 0.7) ** 2))
            candidates.append(norm_sum1(a_raw))

    # W-shaped
    a_raw = np.exp(-((x - 0.2) ** 2) / 0.01) + np.exp(-((x - 0.5) ** 2) / 0.02) + np.exp(-((x - 0.8) ** 2) / 0.01)
    candidates.append(norm_sum1(a_raw))

    # Random smooth (band-limited noise)
    for _ in range(10):
        freqs = rng.uniform(0, 5, 5)
        phases = rng.uniform(0, 2 * np.pi, 5)
        a_raw = np.ones(n)
        for f, p in zip(freqs, phases):
            a_raw += 0.5 * np.cos(2 * np.pi * f * x + p)
        a_raw = np.maximum(a_raw, 0.1)
        candidates.append(norm_sum1(a_raw))

    return candidates

def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best if candidate is good."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {:.6f} (n={}, t={:.1f}s)".format(label, sc, len(a_norm), elapsed))
    return best_seq, best_score

                best_score = sc
                best_seq = cand.copy()
                print("  Parametric init {} beats: J={:.6f}".format(i, sc))

    # === Phase 2: Nesterov smoothed gradient (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: Nesterov SGD (n={}) ===".format(n0))
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 60:
        a_out, _ = nesterov_smoothed_grad(best_seq, t2, beta_start=3.0, beta_end=500.0,
                                          growth=1.01, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nesterov", t_global)

    # === Phase 3: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: Peak shaving (n={}) ===".format(len(best_seq)))
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_out, _ = peak_shaving_multi(best_seq, t3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakshave", t_global)

    # === Phase 4: Nesterov round 2 (higher beta) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 4: Nesterov r2 (n={}) ===".format(len(best_seq)))
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 40:
        a_out, _ = nesterov_smoothed_grad(best_seq, t4, beta_start=30.0, beta_end=1500.0,
                                          growth=1.012, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nesterov2", t_global)

    # === Phase 5: FW refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: FW (n={}) ===".format(n_best))
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 40 and n_best <= 9000:
        try:
            a_out, _ = fw_refine(best_seq, t5, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fw", t_global)
        except Exception as e:
            print("    FW failed: {}".format(str(e)))

    # === Phase 6: Peak shaving round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 6: Peak shaving r2 (n={}) ===".format(len(best_seq)))
    t6 = min(remaining * 0.12, 160.0)
    if t6 > 20:
        a_out, _ = peak_shaving_multi(best_seq, t6, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakshave2", t_global)

    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    nearby_lengths = []
    for dn in [-300, -100, 100, 300, 500, -500, 600]:
        nl = n_best + dn
        if 5000 < nl <= 9000 and nl not in nearby_lengths:
            nearby_lengths.append(nl)

    for new_n in nearby_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("\n=== Phase 7: n={} ===".format(new_n))
        a_long = interp_sequence(best_seq, new_n)

        # Quick Nesterov
        t7a = min(remaining * 0.4, 200.0)
        if t7a > 40:
            a_out, _ = nesterov_smoothed_grad(a_long, t7a, beta_start=10.0, beta_end=400.0,
                                              growth=1.008, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nesterov_n{}".format(new_n), t_global)

        # Peak shaving
        remaining = time_budget - (time.monotonic() - t_global)
        t7b = min(remaining * 0.4, 120.0)
        if t7b > 20:
            a_out, _ = peak_shaving_multi(best_seq, t7b, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ps_n{}".format(new_n), t_global)

    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print("\n=== Phase 8: Final polish (n={}) ===".format(n_final))
    if remaining > 80:
        t8a = min(remaining * 0.35, 200.0)
        if t8a > 30:
            a_out, _ = nesterov_smoothed_grad(best_seq, t8a, beta_start=100.0, beta_end=3000.0,
                                              growth=1.015, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_ne", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        t8b = min(remaining * 0.35, 150.0)
        if t8b > 20:
            a_out, _ = peak_shaving_multi(best_seq, t8b, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_ps", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        t8c = min(remaining * 0.3, 100.0)
        if t8c > 30 and n_final <= 8000:
            try:
                a_out, _ = fw_refine(best_seq, t8c, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "final_fw", t_global)
            except Exception:
                pass

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.6f}, n={}, time={:.1f}s".format(best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence! Using fallback.")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        a_raw = (x + 0.01) ** 1.0 * (1.0 - x + 0.01) ** 1.0
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)

    # Ensure we return valid floats
    result = [float(max(0.0, min(1000.0, x))) for x in np.asarray(best_seq, dtype=np.float64)]
    return result
