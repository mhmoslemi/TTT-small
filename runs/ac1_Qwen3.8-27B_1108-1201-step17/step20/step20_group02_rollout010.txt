"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed-max gradient descent with FFT-based gradient computation
2. Beta annealing (temperature schedule for softmax over autocorrelation peaks)
3. Multiplicative weights update (mirror descent) for positivity constraint
4. Multi-start with diverse initializations (sine, cosine, B-spline, random)
5. Coordinate descent polish at the end
6. Multi-n interpolation and re-optimization
"""

import numpy as np
import time

def fast_convolve(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_xcorr(w, a):
    """Compute cross-correlation of w (length 2n-1) with a (length n) at lags 0..n-1.
    Returns g of length n where g[j] = sum_k w[k] * a[k-j] for valid indices."""
    n = len(a)
    lw = len(w)  # 2n-1
    size = lw + n - 1  # 3n-2
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    # Cross-correlation: ifft(conj(A) * W) gives sum_k a[k-j]*w[k]
    # Actually: ifft(fft(w) * conj(fft(a))) gives cross-correlation
    # Let me verify: 
    # DFT: W[k] = sum_j w[j] e^{-2πijk/K}
    # conj(A[k]) = sum_j a[j] e^{+2πijk/K}
    # Product: sum_j sum_l w[j] a[l] e^{-2πik(j-l)/K}
    # IFFT: (1/K) sum_k [sum_j sum_l w[j] a[l] e^{-2πik(j-l)/K}] e^{+2πim k/K}
    # = sum_j sum_l w[j] a[l] delta_{m, j-l} (circular)
    # = sum_j w[j] a[j-m] (for circular conv)
    # = (w cross-corr a)[m] = sum_j w[j] * a[j-m]
    # 
    # We want g[m] = sum_k w[k] * a[k-m] for m=0..n-1
    # This matches (w cross-corr a)[m] with the above definition
    # So g = ifft(fft(w) * conj(fft(a)))[0:n]
    xcorr = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    return xcorr

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax_step(a, beta, eta):
    """One step of smoothed-max gradient descent with multiplicative update."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    # Softmax weights (shifted for stability)
    w = np.exp(beta * (b - max_b))
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Fallback: uniform weights
        w = np.ones_like(b) / len(b)
    else:
        w /= w_sum
    # Gradient via cross-correlation
    g = 2.0 * fast_xcorr(w, a)
    # Multiplicative update
    # a_new[i] proportional to a[i] * exp(-eta * g[i])
    # For numerical stability: subtract max
    log_update = -eta * g
    log_update -= log_update.max()
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a, max_b, g
    a_new /= s
    return a_new, max_b, g

def gradient_descent_optimize(a0, time_limit, beta_start=5.0, beta_end=200.0, 
                                n_anneal_steps=200, min_iters_per_beta=50,
                                verbose=True, seed=42):
    """Main gradient optimization with beta annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    best_score = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    total_iters = 0
    
    # Beta annealing schedule
    betas = np.linspace(beta_start, beta_end, n_anneal_steps)
    eta0 = 1.0
    
    for beta_idx, beta in enumerate(betas):
        # Adaptive eta: start larger, decrease
        eta = eta0 / (1.0 + beta_idx * 0.5)
        
        iters_this_beta = 0
        while time.monotonic() - t0 < time_limit:
            iters_this_beta += 1
            total_iters += 1
            
            b = fast_convolve(a)
            M = b.max()
            
            if M < best_m - 1e-14:
                best_m = M
                best_a = a.copy()
                best_score = 2.0 * n * M
            
            # Compute gradient
            max_b = M
            w = np.exp(beta * (b - max_b))
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.ones_like(b) / len(b)
            else:
                w /= w_sum
            
            g = 2.0 * fast_xcorr(w, a)
            
            # Multiplicative update with backtracking
            improved = False
            for trial_eta in [eta, eta*0.5, eta*0.25, eta*0.1]:
                log_update = -trial_eta * g
                log_update -= log_update.max()
                a_new = a * np.exp(log_update)
                a_new = np.maximum(a_new, 1e-30)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    continue
                a_new /= s_new
                M_new = fast_convolve(a_new).max()
                if M_new < M - 1e-16:
                    a = a_new
                    improved = True
                    break
            
            if not improved:
                # Take a small step anyway to keep moving
                log_update = -trial_eta * 0.5 * g
                log_update -= log_update.max()
                a_new = a * np.exp(log_update)
                a_new = np.maximum(a_new, 1e-30)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a = a_new / s_new
            
            # Periodic random perturbation
            if iters_this_beta % 500 == 0:
                noise_scale = 0.01 * np.exp(-beta_idx / 50.0)
                if noise_scale > 1e-6:
                    noise = rng.standard_normal(n) * noise_scale
                    a_p = a + noise * a
                    a_p = np.maximum(a_p, 1e-10)
                    s_p = a_p.sum()
                    if s_p > 1e-10:
                        a_p /= s_p
                        M_p = fast_convolve(a_p).max()
                        if M_p > M * 1.5:  # Only accept if not too bad
                            # Don't update, just explore
                            pass
            
            if verbose and iters_this_beta % 2000 == 0:
                elapsed = time.monotonic() - t0
                print("    GD bs=" + str(round(beta,1)) + " iter=" + str(iters_this_beta) + 
                      " (" + str(round(elapsed,1)) + "s): score=" + repr(2.0*n*best_m))
        
        # End of this beta level
        if verbose and beta_idx % 20 == 0:
            elapsed = time.monotonic() - t0
            print("    GD beta=" + str(round(beta,1)) + " done (" + str(round(elapsed,1)) + 
                  "s): score=" + repr(2.0*n*best_m))
    
    if verbose:
        elapsed = time.monotonic() - t0
        print("    GD done (" + str(round(elapsed,1)) + "s): score=" + repr(best_score) + 
              ", iters=" + str(total_iters))
    
    return best_a, best_score

def coordinate_polish(a0, time_limit, verbose=False, seed=77):
    """Coordinate descent polish: move mass from high-peak-contributing positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 200
    steps = [0.001, 0.003, 0.01, 0.03, 0.1]
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_convolve(a)
        M = b.max()
        
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
        
        # Find top peak positions
        n_peaks = min(20, 2*n-1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]
        
        improved = False
        
        for pk in top_peaks[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            pk_int = int(pk)
            # Positions that contribute to this peak
            i_start = max(0, pk_int - n + 1)
            i_end = min(n, pk_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[pk_int - idx]
            
            if contrib.max() < 1e-15:
                continue
            
            # Find the most contributing position
            src_j = idx[np.argmax(contrib)]
            src_val = a[src_j]
            if src_val < 1e-12:
                continue
            
            # Try moving a small amount to a "quiet" position
            # Quiet positions: those that don't contribute to top peaks
            in_peak = np.zeros(n, dtype=bool)
            for pk2 in top_peaks[:5]:
                pk2i = int(pk2)
                s2 = max(0, pk2i - n + 1)
                e2 = min(n, pk2i + 1)
                in_peak[s2:e2] = True
            
            quiet = np.where(~in_peak)[0]
            if len(quiet) < n // 10:
                quiet = np.where(a < a.max() * 0.3)[0]
            if len(quiet) == 0:
                quiet = np.arange(n)
            
            # Pick a quiet destination
            dst_j = quiet[rng.integers(len(quiet))]
            
            if dst_j == src_j:
                continue
            
            # Try several fractions
            for frac in steps:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src_j] -= amt
                a_trial[dst_j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial /= s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < M - 1e-15:
                    a = a_trial
                    best_m = M_trial if M_trial < best_m else best_m
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation when stuck
            if stall % 50 == 0:
                for _ in range(5):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a /= s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    Coord (" + str(round(elapsed,1)) + "s): score=" + repr(2.0*n*best_m) + 
                  ", stall=" + str(stall))
    
    return best_a, best_m

def make_sine_init(n):
    """Sine squared initialization."""
    i = np.arange(n)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    return norm_sum1(a)

def make_cosine_init(n):
    """Raised cosine initialization."""
    i = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * i / n))
    a = np.maximum(a, 0)
    return norm_sum1(a)

def make_gaussian_init(n, sigma_frac=0.3):
    """Gaussian bump initialization."""
    i = np.arange(n)
    center = n / 2
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    return norm_sum1(a)

def make_bispline_init(n, degree=2):
    """B-spline-like smooth initialization."""
    # Use a few control points and interpret as B-spline
    n_cp = max(3, n // 50)
    cp = np.ones(n_cp) * 0.5 + 0.5 * np.sin(np.linspace(0, np.pi, n_cp))
    # Simple linear interpolation and smoothing
    x_cp = np.linspace(0, 1, n_cp)
    x_n = np.linspace(0, 1, n)
    a = np.interp(x_n, x_cp, cp)
    # Smooth with a small kernel
    k = max(3, n // 200)
    kernel = np.ones(k) / k
    a = np.convolve(a, kernel, mode='same')
    return norm_sum1(a)

Wait, I need to double-check the cross-correlation computation. Let me be very careful.

We have:
- a of length n (indices 0 to n-1)
- b = conv(a, a) of length 2n-1 (indices 0 to 2n-2)
  where b[k] = Σ_{i=0}^{n-1} a[i] * a[k-i] (with a[j]=0 for j outside [0,n-1])

We want the gradient of soft_max(b) w.r.t. a:
g[j] = Σ_k (∂soft_max/∂b[k]) * (∂b[k]/∂a[j])

Where ∂soft_max/∂b[k] = w[k] = exp(β*b[k]) / Σ exp(β*b)

And ∂b[k]/∂a[j] = 2 * a[k-j] (if 0 ≤ k-j ≤ n-1, else 0)

So: g[j] = 2 * Σ_{k: 0≤k-j≤n-1} w[k] * a[k-j]
         = 2 * Σ_{l=0}^{n-1} w[j+l] * a[l]  (where w[k]=0 for k outside [0,2n-2])

This is: g[j] = 2 * (w ⊛ a)[j] where ⊛ denotes cross-correlation:
(w ⊛ a)[j] = Σ_l w[j+l] * a[l]

In FFT terms, the cross-correlation (w ⊛ a)[j] for j=0,...,n-1 can be computed as:
- Pad w to length L (L ≥ 2n-1+n-1 = 3n-2)

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed-max (softmax) gradient descent with FFT-based gradient computation
2. Geometric beta annealing for progressive peak focusing
3. Multiplicative weights (mirror descent) update for positivity
4. Backtracking line search with adaptive step size
5. Multi-start with diverse initializations (sine, cosine, Gaussian, B-spline)
6. Multi-n interpolation and re-optimization
7. Coordinate descent polish and random perturbation restarts
"""

import numpy as np
import time


def fast_convolve(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_xcorr(w, a):
    """Compute cross-correlation (w xcorr a)[j] = sum_l w[j+l]*a[l] for j=0..n-1 via FFT."""
    n = len(a)
    lw = len(w)
    size = lw + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(W * np.conj(A), fft_size)
    return result[:n]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_sine_init(n):
    """Sine squared initialization."""
    i = np.arange(n)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    return norm_sum1(a)


def make_cosine_init(n):
    """Raised cosine initialization."""
    i = np.arange(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * i / n))
    a = np.maximum(a, 1e-10)
    return norm_sum1(a)


def make_gaussian_init(n, sigma_frac):
    """Gaussian bump initialization."""
    i = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    return norm_sum1(a)


def make_bispline_init(n):
    """B-spline-like smooth initialization via interpolation of control points."""
    n_cp = max(4, n // 80)
    cp = 0.5 + 0.4 * np.sin(np.pi * np.linspace(0, 1, n_cp))
    x_cp = np.linspace(0, 1, n_cp)
    x_n = np.linspace(0, 1, n)
    a = np.interp(x_n, x_cp, cp)
    k = max(3, n // 300)
    kernel = np.ones(k) / k
    a = np.convolve(a, kernel, mode='same')
    return norm_sum1(a)


def make_weighted_sine_init(n, alpha):
    """Weighted combination of sine frequencies."""
    i = np.arange(n)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    if alpha > 0.
        a += alpha * np.sin(3.0 * np.pi * (i + 0.5) / n) ** 2
    a = np.maximum(a, 1e-10)
    return norm_sum1(a)


def gradient_descent_optimize(a0, time_limit, beta_start, beta_end,
                              n_anneal_steps, eta0, verbose, seed):
    """Smoothed-max gradient descent with geometric beta annealing and multiplicative updates."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    best_score = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    total_iters = 0
    no_improve_count = 0
    max_no_improve = 50000

    # Geometric beta schedule
    if n_anneal_steps <= 1:
        betas = [beta_end]
    else:
        ratio = (beta_end / beta_start) ** (1.0 / (n_anneal_steps - 1))
        betas = [beta_start * (ratio ** i) for i in range(n_anneal_steps)]

    for beta_idx, beta in enumerate(betas):
        eta = eta0 / (1.0 + 0.3 * beta_idx)
        iters_this_beta = 0

        while time.monotonic() - t0 < time_limit and no_improve_count < max_no_improve:
            iters_this_beta += 1
            total_iters += 1

            b = fast_convolve(a)
            M = b.max()

            if M < best_m - 1e-14:
                best_m = M
                best_a = a.copy()
                best_score = 2.0 * n * M
                no_improve_count = 0
            else:
                no_improve_count += 1

            # Softmax weights
            w = np.exp(beta * (b - M))
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.ones_like(b) / len(b)
            else:
                w /= w_sum

            # Gradient via cross-correlation
            g = 2.0 * fast_xcorr(w, a)

            # Backtracking multiplicative update
            improved = False
            for trial_scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
                trial_eta = eta * trial_scale
                log_update = -trial_eta * g
                log_max = log_update.max()
                if log_max > 700:
                    log_update -= log_max
                a_new = a * np.exp(log_update)
                a_new = np.maximum(a_new, 1e-30)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    continue
                a_new /= s_new
                M_new = fast_convolve(a_new).max()
                if M_new < M - 1e-16:
                    a = a_new
                    improved = True
                    break

            if not improved:
                # Take a tiny step to keep moving
                tiny_eta = eta * 0.03
                log_update = -tiny_eta * g
                log_max = log_update.max()
                if log_max > 700:
                    log_update -= log_max
                a_new = a * np.exp(log_update)
                a_new = np.maximum(a_new, 1e-30)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a = a_new / s_new

            # Periodic random perturbation (less aggressive at high beta)
            if iters_this_beta % 300 == 0:
                noise_scale = 0.02 * np.exp(-beta / 100.0)
                if noise_scale > 1e-8:
                    noise = rng.standard_normal(n) * noise_scale
                    a_p = a + noise * a
                    a_p = np.maximum(a_p, 1e-12)
                    s_p = a_p.sum()
                    if s_p > 1e-10:
                        a_p /= s_p
                        M_p = fast_convolve(a_p).max()
                        # Accept exploration if not too bad
                        if M_p < M * 1.2:
                            a = a_p

            # Safety: reset if stuck too long
            if no_improve_count == max_no_improve:
                # Restart from best with perturbation
                a = best_a + 0.01 * rng.standard_normal(n) * best_a
                a = norm_sum1(a)
                no_improve_count = 0

            if verbose and iters_this_beta % 5000 == 0:
                elapsed = time.monotonic() - t0
                print("    GD b=%.1f it=%d (%.1fs): score=%.6f" % (beta, iters_this_beta, elapsed, best_score))

        if verbose and (beta_idx % max(1, n_anneal_steps // 10) == 0):
            elapsed = time.monotonic() - t0
            print("    GD beta=%.2f done (%.1fs): score=%.6f" % (beta, elapsed, best_score))

        # Early exit if we've been stuck
        if no_improve_count > max_no_improve // 2:
            break

    elapsed = time.monotonic() - t0
    if verbose:
        print("    GD complete (%.1fs): score=%.6f, iters=%d" % (elapsed, best_score, total_iters))

    return best_a, best_score


def coordinate_polish(a0, time_limit, verbose, seed):
    """Coordinate descent: redistribute mass away from autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 500
    steps = [0.001, 0.003, 0.01, 0.03, 0.08]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_convolve(a)
        M = b.max()

        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()

        n_peaks = min(20, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]

        improved = False

        # Build "in_peak" mask
        in_peak = np.zeros(n, dtype=bool)
        for pk in top_peaks[:8]:
            pk_i = int(pk)
            s = max(0, pk_i - n + 1)
            e = min(n, pk_i + 1)
            in_peak[s:e] = True

        for pk in top_peaks[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            pk_i = int(pk)
            i_start = max(0, pk_i - n + 1)
            i_end = min(n, pk_i + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[pk_i - idx]

            if contrib.max() < 1e-15:
                continue

            src_j = idx[np.argmax(contrib)]
            src_val = a[src_j]
            if src_val < 1e-12:
                continue

            # Find quiet destination
            quiet = np.where(~in_peak)[0]
            if len(quiet) < n // 10:
                quiet = np.where(a < a.max() * 0.4)[0]
            if len(quiet) == 0:
                quiet = np.arange(n)

            for trial_dst in [quiet[rng.integers(len(quiet))]]:
                dst_j = int(trial_dst)
                if dst_j == src_j:
                    continue

                for frac in steps:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_t = a.copy()
                    a_t[src_j] -= amt
                    a_t[dst_j] += amt
                    a_t = np.maximum(a_t, 0.0)
                    s_t = a_t.sum()
                    if s_t < 1e-15:
                        continue
                    a_t /= s_t
                    M_t = fast_convolve(a_t).max()
                    if M_t < M - 1e-15:
                        a = a_t
                        if M_t < best_m:
                            best_m = M_t
                            best_a = a.copy()
                        improved = True
                        stall = 0
                        break

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation when stuck
            if stall % 50 == 0:
                for _ in range(8):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a /= s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                break

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    Coord (%.1fs): score=%.6f, stall=%d" % (elapsed, 2.0 * n * best_m, stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update the best sequence if the candidate is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [%s]: %.6f (n=%d, t=%.1fs)" % (label, sc, len(a_norm), elapsed))
    return best_seq, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 1e-15)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                best_seq = p_norm
                best_score = sc
                print("Parent: n=%d, score=%.6f" % (len(p_norm), sc))
    except Exception:
        pass

        best_seq = make_sine_init(5000)
        best_score = evaluate_safe(best_seq)
        print("Default sine init: n=5000, score=%.6f" % best_score)

    def remaining():
        return time_budget - (time.monotonic() - t_global)

    n0 = len(best_seq)

    # === Phase 1: Quick GD on parent ===
    print("\n=== Phase 1: GD on parent (n=%d) ===" % n0)
    t1 = min(remaining() * 0.15, 240.0)
    if t1 > 60:
