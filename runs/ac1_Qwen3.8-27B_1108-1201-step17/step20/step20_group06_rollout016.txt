def get_fft_size(n):

def normalize_pos(a):

```
def mult_step(x,h,eta,clip=5.0):
```

```
def smooth_stats(a,b,beta,N):
    n=int(a.size)
    M=float(np.max(b)) if b.size else 0.0
    if not np.isfinite(M): return np.zeros(n), M, inf
    if beta <= 1e-8:
       p=np.full(2*n-1,1.0/(2*n-1))
       F=M
    else:
       shifted=beta*(b-M)
       np.clip(shifted,-70.0,0.0,out=shifted)
       e=np.exp(shifted)
       S=float(e.sum())
       if np.isfinite(S) and S > 0:
          p=e/S
          F=M + float(np.log(S))/beta
       else:
          p=np.zeros_like(b); idx=int(np.argmax(b)); p[idx]=1.0; F=M
    Ar=np.fft.rfft(a[::-1],N)
    Pr=np.fft.rfft(p,N)
    c=np.fft.irfft(Ar*Pr,N)
    c=c[:2*n-1]
    c=c[n-1:2*n-1]
    h=2.0*c
    return h, M, F
```

```python
"""
Search for a nonnegative sequence minimizing

    2 * n * max(convolve(a, a)) / (sum(a))**2

using a combination of:
  * FFT convolution and log-sum-exp smoothing of the peak,
  * multiplicative gradient (Bregman) steps,
  * Frank-Wolfe-like low-gradient directions,
  * vectorized spectral-domain peak flattening for restarts,
  * diverse randomized restarts and length resampling.

The final sequence is normalized to sum 1, strictly nonnegative, and capped to
valid finite floats.
"""

import numpy as np
import time

_FFT_SIZE_CACHE = {}


def fft_size_for(n):
    """Power-of-two FFT length sufficient for linear convolution of length n with itself."""
    n = int(n)
    if n in _FFT_SIZE_CACHE:
        return _FFT_SIZE_CACHE[n]
    size = 2 * n - 1
    if size <= 1:
        N = 1
    else:
        N = 1 << (size - 1).bit_length()
    _FFT_SIZE_CACHE[n] = N
    return N


def normalize_pos(a):
    """Clip nonfinite/negative values, cap at 1000, and normalize sum to 1."""
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def conv_with_N(a, N=None):
    """Linear convolution a*a via zero-padded FFT, clipped to nonnegative."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if N is None:
        N = fft_size_for(n)
    A = np.fft.rfft(a, N)
    C = np.fft.irfft(A * A, N)
    C = C[: 2 * n - 1]
    return np.maximum(C, 0.0)


def internal_score(a):
    """Fast internal objective assuming sum(a)=1 after normalization."""
    a = normalize_pos(a)
    if a.size == 0 or float(a.sum()) < 0.01:
        return float("inf")
    b = conv_with_N(a)
    M = float(np.max(b)) if b.size else 0.0
    return 2.0 * float(a.size) * M


def smooth_F_from_b(b, beta):
    """Log-sum-exp smoothed maximum of b."""
    if b.size == 0:
        return 0.0
    if not np.all(np.isfinite(b)):
        return float("inf")
    M = float(np.max(b))
    if beta <= 1e-8:
        return M
    shifted = beta * (b - M)
    np.clip(shifted, -70.0, 0.0, out=shifted)
    e = np.exp(shifted)
    S = float(e.sum())
    if not np.isfinite(S) or S <= 0.0:
        return M
    return M + float(np.log(S)) / beta


def smooth_stats(a, b, beta, N):
    """
    Compute normalized softmax weights over b and the gradient of the
    log-sum-exp objective with respect to a.

    Returns:
        h: gradient direction (2 * sum_k p_k a_{k-i})
        M: max(b)
        F: smoothed objective value
    """
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64), 0.0, 0.0
    if b.size == 0:
        return np.zeros(n, dtype=np.float64), 0.0, 0.0

    M = float(np.max(b))
    if not np.isfinite(M):
        return np.zeros(n, dtype=np.float64), M, float("inf")

    if beta <= 1e-8:
        p = np.full(2 * n - 1, 1.0 / float(2 * n - 1))
        F = M
    else:
        shifted = beta * (b - M)
        np.clip(shifted, -70.0, 0.0, out=shifted)
        e = np.exp(shifted)
        S = float(e.sum())
        if np.isfinite(S) and S > 0.0:
            p = e / S
            F = M + float(np.log(S)) / beta
        else:
            p = np.zeros_like(b, dtype=np.float64)
            idx = int(np.argmax(b))
            p[idx] = 1.0
            F = M

    ar = a[::-1]
    Ar = np.fft.rfft(ar, N)
    Pr = np.fft.rfft(p, N)
    c = np.fft.irfft(Ar * Pr, N)
    c = c[: 2 * n - 1]
    c = c[n - 1: 2 * n - 1]
    h = 2.0 * c
    return h, M, F


def mult_step(x, h, eta, clip=5.0):
    """Multiplicative Bregman step: x_i <- x_i * exp(-eta * centered_h_i / scale)."""
    x = np.asarray(x, dtype=np.float64)
    h = np.asarray(h, dtype=np.float64)
    if not np.all(np.isfinite(h)):
        return x.copy()
    hc = h - float(np.mean(h))
    scale = float(np.max(np.abs(hc)))
    if scale <= 1e-14:
        return x.copy()
    z = -float(eta) * hc / scale
    np.clip(z, -clip, clip, out=z)
    y = x * np.exp(z)
    y = np.maximum(y, 0.0)
    s = float(y.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return x.copy()
    return y / s


def direction_weights_from_h(h):
    """Probability vector over low gradient coordinates."""
    h = np.asarray(h, dtype=np.float64)
    n = int(h.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if not np.all(np.isfinite(h)):
        return np.ones(n, dtype=np.float64) / float(n)

    med = float(np.median(h))
    hc = h - med
    scale = float(np.max(np.abs(hc)))
    if not np.isfinite(scale) or scale <= 1e-14:
        return np.ones(n, dtype=np.float64) / float(n)

    z = -hc / scale
    np.clip(z, -6.0, 6.0, out=z)
    w = np.exp(z)
    w = 0.6 * w + 0.4 / float(n)
    w = np.maximum(w, 1e-300)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return w / s


def spectral_refine(a0, time_limit, alpha=0.06, N=None, verbose=False):
    """
    Fast vectorized spectral-domain peak flattening.

    It repeatedly damps high power-spectrum bins and slightly boosts low bins,
    then projects back to the nonnegative simplex.
    """
    x = normalize_pos(a0)
    n = int(x.size)
    if n == 0:
        return x
    if N is None:
        N = fft_size_for(n)

    t0 = time.monotonic()
    b = conv_with_N(x, N)
    M = float(np.max(b)) if b.size else 0.0
    best_x = x.copy()
    best_M = M
    cur_M = M
    a_cur = float(alpha)
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        A = np.fft.rfft(x, N)
        P = np.abs(A) ** 2

        if P.size > 1:
            target = float(np.mean(P[1:]))
        else:
            target = float(P[0])
        if not np.isfinite(target) or target <= 1e-300:
            target = float(np.max(P)) * 1e-6
            if not np.isfinite(target) or target <= 0.0:
                target = 1.0

        ratio = P / target
        amp = np.ones_like(P)

        high = ratio > 1.0
        if np.any(high):
            r = ratio[high] - 1.0
            amp[high] = 1.0 / np.sqrt(1.0 + 2.0 * a_cur * np.maximum(r, 0.0))

        low = ratio < 0.5
        if np.any(low):
            low_margin = 0.5 - ratio[low]
            amp[low] = np.sqrt(1.0 + 0.5 * a_cur * np.maximum(low_margin, 0.0))

        if amp.size > 0:
            amp[0] = 1.0
        amp = np.clip(amp, 0.75, 1.35)

        A2 = A * amp
        y = np.fft.irfft(A2, N)[:n]
        y = np.maximum(y, 0.0)
        s = float(y.sum())
        if not np.isfinite(s) or s <= 1e-15:
            break
        y = y / s

        b2 = conv_with_N(y, N)
        M2 = float(np.max(b2)) if b2.size else 0.0

        if M2 < best_M - 1e-14:
            best_M = M2
            best_x = y.copy()

        if M2 <= cur_M + 1e-12:
            x = y
            cur_M = M2
            no_improve = 0
        else:
            no_improve += 1
            a_cur *= 0.7
            if no_improve > 4000:
                break

        if it % 1000 == 0:
            a_cur = max(0.01, a_cur * 0.95)

        if verbose and it % 2000 == 0:
            print(
                "    spectral it=%d t=%.1fs J=%.6f"
                % (it, time.monotonic() - t0, 2.0 * n * best_M)
            )

    return best_x


def make_default_sequence(n=6700, seed=1):
    """Fallback initialization if no parent construction is available."""
    n = max(2, int(n))
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-2.0 * x) + 0.35 * np.exp(-2.0 * (1.0 - x)) + 0.5
    a = a + 0.02 * rng.standard_normal(a.size)
    return normalize_pos(a)


def resample_sequence(a, new_n):
    """Linear resampling to another length, then normalize."""
    a = normalize_pos(a)
    new_n = int(new_n)
    old_n = int(a.size)
    if new_n <= 0:
        return a.copy()
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    return normalize_pos(y)


def perturb_sequence(x, best_x, strategy, rng):
    """Create a perturbed sequence for restarts."""
    x = np.asarray(x, dtype=np.float64)
    best_x = np.asarray(best_x, dtype=np.float64)
    n = int(x.size)
    if n == 0:
        return x.copy()
    best_x = best_x.reshape(-1)
    if best_x.size != n:
        best_x = np.ones(n, dtype=np.float64) / float(n)

    if strategy == 0:
        y = x * (1.0 + 0.04 * rng.standard_normal(n))
        y = np.maximum(y, 0.0)
    elif strategy == 1:
        alpha = 0.05 + 0.25 * float(rng.random())
        y = (1.0 - alpha) * best_x + alpha / float(n)
        y = np.maximum(y, 0.0)
    elif strategy == 2:
        y = 0.5 * x + 0.5 * best_x[::-1]
        y = y * (1.0 + 0.02 * rng.standard_normal(n))
        y = np.maximum(y, 0.0)
    elif strategy == 3:
        y = best_x.copy()
        k = int(max(5, n // 25))
        k = min(k, n)
        idx = rng.choice(n, size=k, replace=False)
        y[idx] = y[idx] * (0.25 + 0.75 * float(rng.random()))
        idx2 = rng.choice(n, size=k, replace=False)
        y[idx2] = y[idx2] * (1.0 + 0.5 * float(rng.random()))
        y = np.maximum(y, 0.0)
    else:
        y = x.copy()

    return normalize_pos(y)


def mirror_optimize(
    a0,
    time_limit,
    beta_start=40.0,
    beta_rate=12.0,
    eta_init=0.15,
    alpha_init=0.06,
    max_eta=0.70,
    max_alpha=0.25,
    restart_every=4000,
    verbose=True,
    seed=1,
):
    """
    Main local optimizer.

    Uses smooth peak gradient, multiplicative projected steps, low-gradient
    direction moves, beta annealing, and restarts.
    """
    x = normalize_pos(a0)
    n = int(x.size)
    if n == 0:
        return x, float("inf")

    N = fft_size_for(n)
    beta_end = min(500000.0, max(100000.0, 1000000.0 / float(n)))
    beta_start = min(float(beta_start), beta_end * 0.99)
    beta = max(10.0, beta_start)

    rng = np.random.default_rng(seed)
    b = conv_with_N(x, N)
    M = float(np.max(b)) if b.size else 0.0
    best_x = x.copy()
    best_M = M
    eta = max(0.01, float(eta_init))
    alpha = max(0.005, float(alpha_init))

    t0 = time.monotonic()
    tol_M = max(1e-13, 1e-15 * float(n))
    no_accept = 0
    no_best = 0
    it = 0
    last_print = t0

    while time.monotonic() - t0 < time_limit:
        it += 1
        h, M, F = smooth_stats(x, b, beta, N)
        if not np.isfinite(M) or not np.isfinite(F) or F > 1e100:
            break
        if not np.all(np.isfinite(h)):
            break

        best_y = None
        best_b = None
        best_M2 = M
        best_F2 = F
        best_type = -1

        # Multiplicative Bregman steps.
        for s in (eta, 0.5 * eta):
            if s <= 1e-6:
                continue
            y = mult_step(x, h, s, 5.0)
            b2 = conv_with_N(y, N)
            if b2.size == 0:
                continue
            M2 = float(np.max(b2))
            if not np.isfinite(M2):
                continue
            F2 = smooth_F_from_b(b2, beta)
            if not np.isfinite(F2):
                continue

            if M2 < M - tol_M or (M2 <= M + tol_M and F2 < F - 1e-12):
                if (
                    best_y is None
                    or M2 < best_M2
                    or (abs(M2 - best_M2) <= tol_M and F2 < best_F2)
                ):
                    best_y = y
                    best_b = b2
                    best_M2 = M2
                    best_F2 = F2
                    best_type = 0

        # Low-gradient Frank-Wolfe-like direction, used periodically or on stall.
        if (it % 3 == 0) or no_accept > 100 or no_best > 500:
            d = direction_weights_from_h(h)
            for aa in (alpha, 0.5 * alpha):
                if aa <= 1e-6:
                    continue
                yy = (1.0 - aa) * x + aa * d
                yy = np.maximum(yy, 0.0)
                ss = float(yy.sum())
                if not np.isfinite(ss) or ss <= 1e-300:
                    continue
                yy = yy / ss
                b2 = conv_with_N(yy, N)
                if b2.size == 0:
                    continue
                M2 = float(np.max(b2))
                if not np.isfinite(M2):
                    continue
                F2 = smooth_F_from_b(b2, beta)
                if not np.isfinite(F2):
                    continue

                if M2 < M - tol_M or (M2 <= M + tol_M and F2 < F - 1e-12):
                    if (
                        best_y is None
                        or M2 < best_M2
                        or (abs(M2 - best_M2) <= tol_M and F2 < best_F2)
                    ):
                        best_y = yy
                        best_b = b2
                        best_M2 = M2
                        best_F2 = F2
                        best_type = 1

        if best_y is not None:
            x = best_y
            b = best_b
            M = best_M2
            F = best_F2
            no_accept = 0
            if best_type == 0:
                eta = min(max_eta, max(eta * 1.05, 0.02))
            else:
                alpha = min(max_alpha, max(alpha * 1.05, 0.01))
        else:
            eta = max(0.01, eta * 0.6)
            alpha = max(0.008, alpha * 0.6)
            no_accept += 1

        if M < best_M - tol_M:
            best_M = M
            best_x = x.copy()
            no_best = 0
        else:
            no_best += 1

        beta = min(beta_end, beta + float(beta_rate))

        if no_best > restart_every or no_accept > max(500, restart_every // 4):
            strat = int((no_best // max(1, restart_every)) % 4)
            x = perturb_sequence(x, best_x, strat, rng)
            b = conv_with_N(x, N)
            M = float(np.max(b)) if b.size else 0.0
            beta = max(beta_start * 0.25, beta * 0.5)
            eta = max(0.05, eta * 0.5)
            alpha = max(0.02, alpha * 0.5)
            no_best = 0
            no_accept = 0

        if verbose and time.monotonic() - last_print > 25.0:
            elapsed = time.monotonic() - t0
            cur_max = float(np.max(b)) if b.size else 0.0
            print(
                "    mirror n=%d t=%.1fs it=%d curJ=%.8f bestJ=%.8f beta=%.0f eta=%.3f alpha=%.3f no_best=%d"
                % (
                    n,
                    elapsed,
                    it,
                    2.0 * n * cur_max,
                    2.0 * n * best_M,
                    beta,
                    eta,
                    alpha,
                    no_best,
                )
            )
            last_print = time.monotonic()

    best_x = normalize_pos(best_x)
    bbest = conv_with_N(best_x)
    final_M = float(np.max(bbest)) if bbest.size else 0.0
    return best_x, 2.0 * n * final_M


def consider_candidate(best_seq, best_score, a, label, t_global):
    """Validate a candidate with the real evaluator when worthwhile."""
    if a is None:
        return best_seq, best_score

    try:
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1:
            a = a.ravel()
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if not np.isfinite(s) or s < 0.01:
            return best_seq, best_score
        a = a / s
    except Exception:
        return best_seq, best_score

    # Cheap FFT check: skip full cvx/numpy convolve if clearly worse.
    try:
        int_sc = internal_score(a)
    except Exception:
        int_sc = float("inf")

    if best_seq is not None and np.isfinite(int_sc) and int_sc > best_score + 1e-7:
        return best_seq, best_score

    try:
        sc = evaluate_sequence([float(v) for v in a])
    except Exception:
        sc = int_sc

    if np.isfinite(sc) and (best_seq is None or sc < best_score - 1e-12):
        elapsed = time.monotonic() - t_global
        print(
            "  *** NEW BEST [%s]: score=%.8f n=%d t=%.1fs"
            % (label, float(sc), int(a.size), elapsed)
        )
        return a.copy(), float(sc)

    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Top-level search. Returns the best low-peak nonnegative sequence found
    within the time budget.
    """
    t_global = time.monotonic()
    total_budget = min(float(budget_s) - 120.0, 1680.0)
    if total_budget < 90.0:
        total_budget = 90.0

    best_seq = None
    best_score = float("inf")

    # Load parent construction if available.
    arr = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim != 1:
            p = p.ravel()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if p.size > 0 and np.isfinite(s) and s > 1e-8:
            arr = normalize_pos(p)
    except Exception:
        arr = None

    if arr is None:
        arr = make_default_sequence(6700, seed=1)

    # Keep initial length in a practical range.
    if arr.size > 20000:
        arr = resample_sequence(arr, 15000)
    elif arr.size < 1000:
        arr = resample_sequence(arr, 6700)

    best_seq, best_score = consider_candidate(best_seq, best_score, arr, "init", t_global)
    if best_seq is None:
        best_seq = arr.copy()
        best_score = internal_score(arr)
        print("Initial score (internal): %.8f n=%d" % (best_score, len(best_seq)))
    else:
        print("Initial score: %.8f n=%d" % (best_score, len(best_seq)))

    # Warm spectral flattening.
    rem = total_budget - (time.monotonic() - t_global)
    tspec = min(rem * 0.04, 60.0)
    if tspec > 20.0:
        try:
            warm = spectral_refine(best_seq, tspec, alpha=0.06, verbose=False)
            best_seq, best_score = consider_candidate(best_seq, best_score, warm, "spec_warm", t_global)
        except Exception as e:
            print("spectral warm failed: %s" % e)

    # Primary mirror-optimize on the current best length.
    rem = total_budget - (time.monotonic() - t_global)
    t1 = min(rem * 0.55, 800.0)
    if t1 > 40.0:
        print("Phase 1: mirror main n=%d budget=%.1fs" % (len(best_seq), t1))
        try:
            out, _ = mirror_optimize(
                best_seq,
                t1,
                beta_start=40.0,
                beta_rate=12.0,
                eta_init=0.15,
                alpha_init=0.06,
                max_eta=0.70,
                max_alpha=0.25,
                restart_every=4000,
                verbose=True,
                seed=101,
            )
            best_seq, best_score = consider_candidate(best_seq, best_score, out, "mirror_main", t_global)
        except Exception as e:
            print("mirror_main failed: %s" % e)

    # A second mirror pass with different parameters / perturb.
    rem = total_budget - (time.monotonic() - t_global)
    t2 = min(rem * 0.18, 220.0)
    if t2 > 40.0 and rem > 220.0:
        print("Phase 2: mirror perturb n=%d budget=%.1fs" % (len(best_seq), t2))
        try:
            rng = np.random.default_rng(7)
            start = perturb_sequence(best_seq, best_seq, 0, rng)
            out, _ = mirror_optimize(
                start,
                t2,
                beta_start=90.0,
                beta_rate=22.0,
                eta_init=0.10,
                alpha_init=0.05,
                max_eta=0.55,
                max_alpha=0.18,
                restart_every=3000,
                verbose=True,
                seed=715,
            )
            best_seq, best_score = consider_candidate(best_seq, best_score, out, "mirror_perturb", t_global)
        except Exception as e:
            print("mirror_perturb failed: %s" % e)

    # Another spectral polish.
    rem = total_budget - (time.monotonic() - t_global)
    tsp = min(rem * 0.05, 50.0)
    if tsp > 15.0:
        try:
            out = spectral_refine(best_seq, tsp, alpha=0.03, verbose=False)
            best_seq, best_score = consider_candidate(best_seq, best_score, out, "spec_polish", t_global)
        except Exception as e:
            print("spectral polish failed: %s" % e)

    # Length exploration.
    base_len = int(len(best_seq))
    cap_len = 15000
    targets = []
    for mult in (1.25, 1.55, 1.90, 2.25):
        L = int(base_len * mult)
        if L > cap_len:
            L = cap_len
        if L > int(len(best_seq)) + 300 and L <= cap_len:
            if not targets or abs(L - targets[-1]) > 250:
                targets.append(L)
    for L in (9000, 12000, 15000):
        if L > int(len(best_seq)) + 300 and L <= cap_len:
            if not targets or abs(L - targets[-1]) > 250:
                targets.append(L)
    targets = sorted(set(targets))[:3]

    phase_budgets = (300.0, 250.0, 200.0)
    for idx, L in enumerate(targets):
        rem = total_budget - (time.monotonic() - t_global)
        if rem < 120.0:
            break
        tL = min(rem * 0.40, phase_budgets[min(idx, len(phase_budgets) - 1)])
        if tL < 70.0:
            tL = min(rem, tL)
        if tL < 60.0:
            break

        print("Length phase: target n=%d budget=%.1fs" % (L, tL))
        try:
            aL = resample_sequence(best_seq, L)

            tsL = min(25.0, tL * 0.15)
            if tsL > 10.0:
                aL = spectral_refine(aL, tsL, alpha=0.05, verbose=False)
                best_seq, best_score = consider_candidate(best_seq, best_score, aL, "spec_n%d" % L, t_global)

            used = (time.monotonic() - t_global) - (total_budget - tL)
            tRM = tL - used - 20.0
            if tRM > 40.0:
                out, _ = mirror_optimize(
                    aL,
                    tRM,
                    beta_start=80.0,
                    beta_rate=12.0,
                    eta_init=0.12,
                    alpha_init=0.05,
                    max_eta=0.55,
                    max_alpha=0.18,
                    restart_every=3500,
                    verbose=True,
                    seed=(1000 + L) % 1000000,
                )
                best_seq, best_score = consider_candidate(best_seq, best_score, out, "mirror_n%d" % L, t_global)
        except Exception as e:
            print("length phase failed for n=%d: %s" % (L, e))

    # Final high-beta polish.
    rem = total_budget - (time.monotonic() - t_global)
    tfin = min(rem * 0.60, 260.0)
    if tfin > 35.0:
        print("Final polish: n=%d budget=%.1fs" % (len(best_seq), tfin))
        try:
            out, _ = mirror_optimize(
                best_seq,
                tfin,
                beta_start=20000.0,
                beta_rate=40.0,
                eta_init=0.035,
                alpha_init=0.02,
                max_eta=0.12,
                max_alpha=0.08,
                restart_every=3000,
                verbose=True,
                seed=999,
            )
            best_seq, best_score = consider_candidate(best_seq, best_score, out, "final_polish", t_global)
        except Exception as e:
            print("final polish failed: %s" % e)

    # Final validation and safety cleanup.
    if best_seq is None:
        best_seq = make_default_sequence(6700, seed=3)

    final = np.asarray(best_seq, dtype=np.float64)
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.clip(final, 0.0, 1000.0)

    # Tiny floor for strict positivity safety (negligibly affects objective).
    final = final + 1e-12
    s = float(final.sum())
    if not np.isfinite(s) or s <= 0.0:
        final = np.ones_like(final, dtype=np.float64) / max(1, final.size)
        s = 1.0
    else:
        final = final / s

    try:
        final_sc = evaluate_sequence([float(v) for v in final])
    except Exception:
        final_sc = internal_score(final)

    if not np.isfinite(final_sc):
        # Absolute fallback that is always valid.
        final = make_default_sequence(6700, seed=42)
        final = np.clip(final, 0.0, 1000.0) + 1e-12
        final = final / float(final.sum())
        final_sc = internal_score(final)

    elapsed = time.monotonic() - t_global
    print(
        "FINAL: score=%.8f n=%d elapsed=%.1fs"
        % (float(final_sc), int(final.size), elapsed)
    )

    return [float(v) for v in np.asarray(final, dtype=np.float64)]
```
