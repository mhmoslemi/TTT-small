   Let me think about the LP-FW in more detail. The key step is:
   Given current y (normalized, sum=1), solve:
   ```
   minimize M
   subject to: sum(g) = 1
                g >= 0
                (y*g)[k] <= M for k in active_set
   ```
   
   This LP has n+1 variables (g_0,...,g_{n-1}, M) and |active_set|+2 constraints. For n=10000 and |active_set|=5000, this is a 10001-variable LP with ~5002 constraints. This should be solvable in a few seconds with a good solver.

   The Frank-Wolfe step is then: `y_new = (1-t)*y + t*g_opt`.

   The line search: minimize `max_k ((1-t)*y + t*g)^T C_k ((1-t)*y + t*g)` over t in [0,1].

   For the line search, note that `((1-t)*y + t*g)^T C_k ((1-t)*y + t*g)` is a quadratic in t:
   `= (1-t)^2 * y^T C_k y + 2t(1-t) * y^T C_k g + t^2 * g^T C_k g`
   `= y^T C_k y + t(2*y^T C_k g - 2*y^T C_k y) + t^2(g^T C_k g - 2*y^T C_k g + y^T C_k y)`

   - `B_k = 2*((y*g)[k] - (y*y)[k])`
   - `C_k = ((g*g)[k] - 2*(y*g)[k] + (y*y)[k])`

   The max of these quadratics over t in [0,1] can be found by:
   1. Evaluating at t=0, t=1, and at all critical points of the individual quadratics.
   2. Or by bisection (simpler and robust).

        new_ac = fast_autocorr(x_new)
        new_val = 2.0 * n * new_ac.max()
        if new_val < cur_val + 0.5 * lr * np.max(np.abs(grad)) * x.max():
            x = x_new
        else:
            lr *= 0.5
            continue

        lr = min(lr * 1.02, 0.5)
        total_iter += 1

        if total_iter % iters_per_tau == 0:
            tau *= tau_mult
            lr *= 0.7
            if verbose:
                print("    SGD tau={:.1f} lr={:.4f} t={} ({:.1f}s) J={:.7f}".format(
                    tau, lr, total_iter, time.monotonic() - t0, best_val))

        if stall > 2000 and stall % 500 == 0:
            rng = np.random.default_rng(stall)
            x = best_a + 0.01 * rng.standard_normal(n) * best_a
            x = np.maximum(x, 1e-12)
            x = x / x.sum()
            stall = 0

    return best_a, best_val

def lp_fw(a0, time_limit, solver, init_K=300, max_K=6000, verbose=True):
    """LP-based Frank-Wolfe with adaptive active set and bisection line search."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:init_K])
    no_improve = 0
    stall_limit = 1500

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Refresh active set periodically
        if t % 20 == 0 or t == 1:
            top_lags = np.argsort(ac)[::-1]
            target_size = min(max_K, init_K + t * 15)
            current_size = len(active_set)
            if current_size < target_size:
                for k in top_lags[:target_size]:
                    active_set.add(int(k))
            # Remove stale constraints
            if len(active_set) > max_K // 2:
                clist = sorted(active_set)
                margins = np.array([M_curr - ac[k] for k in clist])
                # Keep the most binding ones
                keep_count = min(max_K, len(clist))
                keep = np.argsort(margins)[:keep_count]
                active_set = set(int(clist[i]) for i in keep)

        # Always include the current argmax
        active_set.add(int(np.argmax(ac)))

        clist = np.array(sorted(active_set), dtype=np.int64)
        K = len(clist)
        if K == 0:
            break

        # Build and solve LP
        try:
            coeffs = build_constraint_matrix(y, clist, n)
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            prob.solve(solver=solver, verbose=False)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                raise Exception("LP status: " + str(prob.status))
        except Exception:
            if len(active_set) > init_K:
                clist = sorted(active_set)
                margins = [M_curr - ac[k] for k in clist]
                active_set = set(int(clist[i]) for i in np.argsort(margins)[:init_K])
            else:
                # Fallback: use gradient direction
                ac_peak = int(np.argmax(ac))
                g_dir = np.zeros(n)
                ws = max(0, ac_peak)
                we = min(n - 1, ac_peak)
                contrib = y[ws:we+1] * y[ac_peak-ws:ac_peak-we+1]
                g_dir[1 - (ac_peak - np.arange(ws, we+1))[np.argmin(contrib)]] = 1.0
                if g_dir.sum() < 1e-10:
                    g_dir = y.copy()
                y_new = np.maximum(0.7 * y + 0.3 * g_dir, 0.0)
                y = y_new / y_new.sum()
                continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            g_dir = y.copy()
        else:
            g_dir = g_val / g_sum

        # Check violations and grow active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                active_set.add(int(k))

        # Line search: minimize max_k ((1-t)y + t*g)^T C_k ((1-t)y + t*g)
        # Use bisection
        def trial_max(tval):
            trial = (1.0 - tval) * y + tval * g_dir
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                return 1e10
            return fast_autocorr(trial / ts).max()

        # First try some candidate alphas
        best_alpha = 1.0 / (t + 1.0)
        best_trial = trial_max(best_alpha)
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            Mt = trial_max(alpha)
            if Mt < best_trial:
                best_trial = Mt
                best_alpha = alpha

        # Bisection refinement
        lo = max(0.0, best_alpha - 0.2)
        hi = min(1.0, best_alpha + 0.2)
        for _ in range(15):
            mid1 = lo + (hi - lo) / 3.0
            mid2 = hi - (hi - lo) / 3.0
            m1 = trial_max(mid1)
            m2 = trial_max(mid2)
            if m1 < m2:
                hi = mid2
            else:
                lo = mid1
        best_alpha = (lo + hi) / 2.0

        # Update
        if t <= 1:
            y = g_dir.copy()
        else:
            y = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t={:4d} ({:.1f}s) J={:.7f} |S|={:5d} alpha={:.3f}".format(
                t, elapsed, 2 * n * best_m, len(active_set), best_alpha))

    elapsed = time.monotonic() - t0
    if verbose:
        print("    LP-FW done ({:.1f}s) J={:.7f} |S|={}".format(elapsed, 2 * n * best_m, len(active_set)))
    return best_a, 2.0 * n * best_m

def coord_peel(a0, time_limit, n_targets=10, seed=42, verbose=True):
    """Coordinate peeling: move mass from peak contributors to quiet positions."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 3000
    fracs = [0.002, 0.005, 0.01, 0.02, 0.035, 0.05]

    def pressure_calc(a_, top_lags_):
        pr = np.zeros(n, dtype=np.float64)
        for k in top_lags_:
            ki = int(k)
            ws = max(0, ki)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            pr[idx] += a_[ki - idx]
        return pr

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(n_targets, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            ws = max(0, ki)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-15:
                continue

            pr = pressure_calc(a, top_k[:6])
            pr[src] = 1e10
            quiet_positions = np.argsort(pr)[:40]

            for dst in quiet_positions:
                dst = int(dst)
                if dst == src:
                    continue
                found = False
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-18:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-16:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break

            if not found and stall > 100:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
                            improved = True
                break

        if not improved:
            stall += 1
            if stall > 200:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 1000 == 0 and stall > 0:
            print("    Peel ({:.1f}s) J={:.7f} stall={}".format(
                time.monotonic() - t0, 2 * n * best_m, stall))

    return best_a, 2.0 * n * best_m

def make_initialization(n, kind='sine'):
    x = np.linspace(0, 1, n)
    if kind == 'sine':
        f = np.sin(np.pi * x) ** 2
    elif kind == 'cosine':
        f = np.cos(np.pi * x / 2) ** 2
    elif kind == 'asym_sine':
        f = np.sin(np.pi * x ** 0.7) ** 2
    elif kind == 'asym_exp':
        f = np.exp(-2.0 * (x - 0.4) ** 2)
    elif kind == 'ramp':
        f = x ** 0.5 * (1 - x) ** 0.3
    elif kind == 'bimodal':
        f = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    else:
        f = np.sin(np.pi * x) ** 2
    f = np.maximum(f, 1e-10)
    return f / f.sum()

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1750.0)
    solver = get_solver()

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent_norm = normalize(p)
            best_seq = parent_norm
            best_score = eval_safe(best_seq)
            print("Parent loaded: n={} score={:.7f}".format(len(p), best_score))
    except Exception:
        parent_norm = None
        print("No parent available, using fallback initialization")

    # Phase 1: Multi-initialization exploration (quick)
    print("\n=== Phase 1: Multi-init exploration ===")
    t1_start = time.monotonic()
    t1_budget = min(150.0, time_budget * 0.1)

    n_base = len(best_seq) if best_seq is not None else 10000
    init_kinds = ['sine', 'asym_sine', 'asym_exp', 'ramp', 'bimodal', 'cosine']
    candidates = []

    if best_seq is not None:
        candidates.append(('parent', best_seq.copy()))

    for kind in init_kinds:
        elapsed = time.monotonic() - t1_start
        if elapsed > t1_budget * 0.7:
            break
        f = make_initialization(n_base, kind)
        candidates.append((kind, f))
        sc = eval_safe(f)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = f.copy()
            print("  Init '{}' n={} score={:.7f}".format(kind, n_base, sc))

    # Quick LP-FW on each candidate
    per_cand_time = t1_budget / max(len(candidates), 1) * 0.5
    for name, cand in candidates:
        remaining_cand = t1_budget - (time.monotonic() - t1_start)
        if remaining_cand < 10:
            break
        t_cand = min(per_cand_time, remaining_cand)
        if t_cand > 5:
            a_out, v_out = lp_fw(cand, t_cand, solver, init_K=150, max_K=1500, verbose=False)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print("  Phase1 '{}' improved to {:.7f}".format(name, best_score))

    print("Phase 1 done ({:.1f}s): best={:.7f}".format(time.monotonic() - t1_start, best_score))

    # Phase 2: Main LP-FW optimization
    print("\n=== Phase 2: Main LP-FW (n={}) ===".format(len(best_seq)))
    t2_start = time.monotonic()
    t2_budget = min(900.0, time_budget * 0.5)
    a_out, v_out = lp_fw(best_seq, t2_budget, solver, init_K=300, max_K=5000, verbose=True)
    sc = eval_safe(a_out)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_out.copy()
        print("  Phase 2 improved: {:.7f}".format(best_score))
    print("Phase 2 done ({:.1f}s): best={:.7f}".format(time.monotonic() - t2_start, best_score))

    # Phase 3: Smooth gradient refinement
    print("\n=== Phase 3: Smooth Grad (n={}) ===".format(len(best_seq)))
    t3_start = time.monotonic()
    t3_budget = min(300.0, time_budget * 0.15)
    a_out, v_out = smooth_grad_descent(best_seq, t3_budget, verbose=True)
    sc = eval_safe(a_out)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_out.copy()
        print("  Phase 3 improved: {:.7f}".format(best_score))
    print("Phase 3 done ({:.1f}s): best={:.7f}".format(time.monotonic() - t3_start, best_score))

    # Phase 4: Coordinate peeling
    print("\n=== Phase 4: Coord Peel (n={}) ===".format(len(best_seq)))
    t4_start = time.monotonic()
    t4_budget = min(250.0, time_budget * 0.12)
    a_out, v_out = coord_peel(best_seq, t4_budget, n_targets=12, seed=42, verbose=True)
    sc = eval_safe(a_out)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_out.copy()
        print("  Phase 4 improved: {:.7f}".format(best_score))
    print("Phase 4 done ({:.1f}s): best={:.7f}".format(time.monotonic() - t4_start, best_score))

    # Phase 5: Try alternative length (interpolate and refine)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 200:
        n_long = min(len(best_seq) + 3000, 15000)
        if n_long != len(best_seq):
            print("\n=== Phase 5: Long n={} ===".format(n_long))
            t5_budget = min(200.0, remaining * 0.4)
            a_long = interpolate_seq(best_seq, n_long)
            a_out1, _ = lp_fw(a_long, t5_budget * 0.5, solver, init_K=200, max_K=3000, verbose=False)
            a_out2, _ = smooth_grad_descent(a_out1, t5_budget * 0.3, verbose=False)
            a_out3, _ = coord_peel(a_out2, t5_budget * 0.2, n_targets=10, seed=99, verbose=False)
            sc1 = eval_safe(a_out1)
            sc2 = eval_safe(a_out2)
            sc3 = eval_safe(a_out3)
            sc_best = min(sc1, sc2, sc3)
            if np.isfinite(sc_best) and sc_best < best_score:
                best_score = sc_best
                best_seq = a_out1 if sc_best == sc1 else (a_out2 if sc_best == sc2 else a_out3)
                print("  Phase 5 improved: {:.7f}".format(best_score))

    # Phase 6: Final LP-FW push
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        print("\n=== Phase 6: Final LP-FW (n={}) ===".format(len(best_seq)))
        t6_budget = min(remaining * 0.5, 400.0)
        a_out, v_out = lp_fw(best_seq, t6_budget, solver, init_K=400, max_K=6000, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("  Phase 6 improved: {:.7f}".format(best_score))

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== Phase 7: Final Peel (n={}) ===".format(len(best_seq)))
        t7_budget = min(remaining * 0.6, 200.0)
        a_out, v_out = coord_peel(best_seq, t7_budget, n_targets=15, seed=123, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("  Phase 7 improved: {:.7f}".format(best_score))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.7f} n={} t={:.1f}s".format(best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: using fallback")
        n_fb = 10000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    return [float(x) for x in best_seq]
