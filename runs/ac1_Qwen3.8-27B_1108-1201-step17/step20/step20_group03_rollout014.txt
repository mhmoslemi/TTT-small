def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_grad_smoothed(y, beta, top_k=None):
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and top_k < len(b) and top_k > 0:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b_use)
        mask = (b_use == b_max)
        p[mask] = 1.0 / max(1, mask.sum())
    else:
        p = exp_vals / p_total
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1]
    return grad, p


def fft_flat_top_iteration(a, n_iters=500, alpha=0.5, verbose=False):
    """POCS-like iteration: alternate between flattening spectrum and projecting to non-negativity."""
    n = len(a)
    a = norm_sum1(a.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-300)

        # Flatten: raise low-magnitude bins, lower high-magnitude bins
        mag_target = mag ** (-alpha)
        # Normalize to preserve total energy
        mag_target = mag_target / mag_target.mean() * mag.mean()
        A_new = phase * mag_target

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
        a = a_new

        if verbose and it % 100 == 0:
            J = 2.0 * n * best_m
            print("      FlatTop it=" + str(it) + ": J=" + repr(round(J, 6)))

    return best_a, best_m


def fista_optimize(a0, time_limit, beta_start=20.0, beta_end=1500.0,
                   beta_growth=1.004, top_k=None, restart_every=3000,
                   verbose=True):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(42)
    avg_x = x.copy()
    avg_count = 0.0
    amp = 0.5 / beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p = compute_grad_smoothed(y, beta, top_k)
        amp = 0.5 / beta
        z = y - amp * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        avg_count += 1.0
        avg_x = avg_x + (x_new - avg_x) / avg_count

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.02 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 5.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    FISTA restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 5000 == 0:

            print("    FISTA t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + repr(round(beta, 1)))

    J_avg = 2.0 * n * fast_autocorr(avg_x).max()
    J_best = 2.0 * n * best_max_b
    if J_avg < J_best:
        return avg_x, fast_autocorr(avg_x).max()
    return best_a, best_max_b


def coordinate_descent(a0, time_limit, n_sweeps=100, verbose=True):
    """Coordinate descent: for each position, try adjusting its value."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        indices = np.arange(n)
        rng.shuffle(indices)
        improved = False

        for i in indices:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue

            current_val = a[i]
            # Try several multipliers
            fracs = [0.5, 0.75, 1.25, 1.5, 0.25, 1.75]
            best_trial = current_val
            best_trial_m = best_m

            for f in fracs:
                trial_val = current_val * f
                a[i] = trial_val
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-15:
                    a[i] = current_val
                    continue
                a = a / s
                m = fast_autocorr(a).max()
                if m < best_trial_m - 1e-15:
                    best_trial_m = m
                    best_trial = a.copy()

            if best_trial_m < best_m - 1e-15:
                a = best_trial
                best_m = best_trial_m
                best_a = a.copy()
                improved = True
            else:
                a[i] = current_val
                s = a.sum()
                if s > 1e-15:
                    a = a / s

        if not improved and sweep > 5:
            break

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    CoordDesc sweep=" + str(sweep) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def simulated_annealing(a0, time_limit, verbose=True):
    """SA with structured moves to escape local minima."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    temp = 0.01
    cooling = 0.995

    while time.monotonic() - t0 < time_limit:
        # Generate a move
        move_type = rng.integers(3)
        a_trial = a.copy()

        if move_type == 0:
            # Move mass from one position to another
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = float(rng.uniform(0.01, 0.15))
                amt = a[i] * frac
                a_trial[i] -= amt
                a_trial[j] += amt
        elif move_type == 1:
            # Perturb a random position
            i = int(rng.integers(n))
            mult = float(rng.uniform(0.7, 1.3))
            a_trial[i] *= mult
        else:
            # Block perturbation
            i = int(rng.integers(n - 5))
            block_len = int(rng.integers(3, 10))
            block = a_trial[i:i + block_len]
            if block.sum() > 1e-10:
                block = block * rng.uniform(0.8, 1.2, block_len)
                a_trial[i:i + block_len] = np.maximum(block, 0)

        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s

        m_trial = fast_autocorr(a_trial).max()
        delta = m_trial - a_m if (a_m := fast_autocorr(a).max()) < float('inf') else 0

        if m_trial < best_m - 1e-15:
            best_m = m_trial
            best_a = a_trial.copy()

        # Accept if better or with probability exp(-delta/temp)
        if m_trial < best_m or rng.random() < np.exp(-min(m_trial - best_m, 10) / max(temp, 1e-10)):
            a = a_trial

        temp *= cooling

        if verbose and (int((time.monotonic() - t0) * 10) % 50 == 0):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SA (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " T=" + repr(round(temp, 5)))

    return best_a, best_m


def cyclic_shift_best(a0):
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== Final: FISTA micro ===")
        a_out, _ = fista_optimize(norm_sum1(best_seq), min(remaining * 0.6, 80.0),
                                   beta_start=500.0, beta_end=8000.0,
                                   beta_growth=1.015, top_k=5,
                                   restart_every=1000, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_micro", t_global)
        # Cyclic shift again
        a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_very_final", t_global)

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
