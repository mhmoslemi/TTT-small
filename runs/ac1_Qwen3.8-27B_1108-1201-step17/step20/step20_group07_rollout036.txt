But this lower bound is not achievable because of the non-negativity constraint and the specific structure of autocorrelations.

For non-negative sequences, b[0] = a[0]^2 and b[2n-2] = a[n-1]^2. These must be non-negative and are "pinned" to the edges. The middle values b[n-1] = sum a[i]*a[n-1-i] is the "reversed inner product."

For a well-spread sequence, b[n-1] ≈ (sum a)^2 / n = 1/n (by some averaging argument), giving J ≈ 2.

To get below 2, we need the maximum to be at a position other than the center, or the center value needs to be reduced below 1/n.

Actually wait - for the uniform sequence, b[n-1] = n*(1/n)^2 = 1/n, and max = 1/n, so J = 2n*(1/n) = 2.

To get J < 2, we need max_b < 1/n. This means ALL autocorrelation values must be < 1/n.

        x_prev = x
        x = x_new

        M_x = fast_autocorr(x).max()
        if M_x < best_m - 1e-14:
            best_m = M_x
            best_a = x.copy()

        # Adapt p
        if t % 200 == 0:
            p = min(p * 1.08, p_end)

        # Periodic restart
        if t % 500 == 0:
            if fast_autocorr(best_a).max() < best_m + 1e-10:
                pass
            else:
                pass
            # Small perturbation
            pert = rng.normal(0, 0.005, n) * best_a
            x_p = best_a + pert
            x_p = np.maximum(x_p, 1e-12)
            x_p = x_p / x_p.sum()
            if fast_autocorr(x_p).max() < best_m:
                best_m = fast_autocorr(x_p).max()
                best_a = x_p.copy()
            x = x_p.copy()
            y = x.copy()
            x_prev = x.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  PNorm t=" + str(t) + " p=" + str(int(p)) + " (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)))

    return best_a, best_m


def hybrid_polish(a0, time_limit, solver, verbose=True):
    """Alternate between LP-Frank-Wolfe and p-norm gradient for polishing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    p = 20.0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Alternate: odd steps = LP, even steps = gradient
        if t % 3 == 0:
            # LP step
            b = fast_autocorr(a)
            active = set(int(k) for k in np.argsort(b)[::-1][:min(1500, 2 * n - 1)])
            g_dir, M_lp, violations = solve_lp_subproblem(a, active, solver, max_k=1500)
            if g_dir is not None:
                for alpha in [0.3, 0.2, 0.1, 0.05]:
                    trial = (1.0 - alpha) * a + alpha * g_dir
                    trial = np.maximum(trial, 0.0)
                    st = trial.sum()
                    if st < 1e-15:
                        continue
                    trial = trial / st
                    M_trial = fast_autocorr(trial).max()
                    if M_trial < best_m:
                        a = trial
                        best_m = M_trial
                        best_a = trial.copy()
                        break
        else:
            # Gradient step
            grad_a = pnorm_gradient(a, p)
            step = min(0.03, 1.0 / (t + 10))
            a_new = simplex_project(a - step * grad_a)
            M_new = fast_autocorr(a_new).max()
            if M_new < best_m:
                a = a_new
                best_m = M_new
                best_a = a_new.copy()
            else:
                a = simplex_project(a - step * 0.5 * grad_a)

        if t % 100 == 0:
            p = min(p * 1.03, 150.0)

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  Hybrid t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)))

    return best_a, best_m


def parametric_init(n, seed=42):
    """Generate diverse parametric initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    candidates = []

    # Power laws
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        for beta_p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            a_raw = (x + 0.005) ** alpha * (1.0 - x + 0.005) ** beta_p
            candidates.append(norm_sum1(a_raw))

    # Gaussian mixtures
    for c1 in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for c2 in [0.3, 0.5, 0.7, 0.8]:
            for s in [0.08, 0.12, 0.18]:
                a_raw = np.exp(-((x - c1) ** 2) / (2 * s ** 2)) + 0.7 * np.exp(-((x - c2) ** 2) / (2 * s ** 2))
                candidates.append(norm_sum1(a_raw))

    # Smoothed uniform with edge tapering
    for taper in [0.1, 0.2, 0.3, 0.5]:
        a_raw = np.ones(n)
        edge = int(n * taper)
        if edge > 0:
            a_raw[:edge] = np.linspace(0.3, 1.0, edge)
            a_raw[-edge:] = np.linspace(1.0, 0.3, edge)
        candidates.append(norm_sum1(a_raw))

    # Cosine window
    a_raw = 0.5 * (1 + np.cos(2 * np.pi * x))
    candidates.append(norm_sum1(a_raw))

    # Sine-shaped
    a_raw = 1.0 + 0.5 * np.sin(np.pi * x)
    candidates.append(norm_sum1(a_raw))

    # Random smooth
    for _ in range(5):
        a_raw = np.abs(rng.normal(1.0, 0.5, n))
        # Smooth it
        kernel_size = max(3, n // 50)
        kernel = np.ones(kernel_size) / kernel_size
        a_smooth = np.convolve(a_raw, kernel, mode='same')
        candidates.append(norm_sum1(a_smooth))

    return candidates


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print("Time budget: " + str(round(time_budget, 1)) + "s, solver: " + str(getattr(solver, '__name__', solver)))

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + str(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        x = np.linspace(0, 1, n_base)
        a_raw = (x + 0.005) ** 1.0 * (1.0 - x + 0.005) ** 1.0
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + str(best_score))

    n0 = len(best_seq)

    # === Phase 1: Quick parametric sweep to find good init ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.02, 30.0)
    if t1 > 10:
        print("\n=== Phase 1: Parametric sweep ===")
        cands = parametric_init(n0, seed=123)
        best_cand_m = fast_autocorr(best_seq).max()
        for i, c in enumerate(cands):
            if time.monotonic() - t_global > time_budget - 100:
                break
            m_c = fast_autocorr(c).max()
            if m_c < best_cand_m:
                best_cand_m = m_c
        # Try top candidates
        for i, c in enumerate(cands[:10]):
            best_seq, best_score = try_update(best_seq, best_score, c, "param_"+str(i), t_global)

    # === Phase 2: FW with LP at parent n ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: FW-LP (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 60:
        a_out, _ = fw_with_momentum(best_seq, t2, solver, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fwlp_n" + str(n_best), t_global)

    # === Phase 3: Nesterov p-norm gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Nesterov p-norm (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 50:
        a_out, _ = nesterov_pnorm(best_seq, t3, p_start=8, p_end=80, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pnorm_n" + str(n_best), t_global)

    # === Phase 4: Hybrid polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Hybrid polish (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 40:
        a_out, _ = hybrid_polish(best_seq, t4, solver, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "hybrid_n" + str(n_best), t_global)

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-length exploration ===")
    target_ns = []
    for dn in [-300, -100, 100, 300, 500, 700, 1000]:
        nl = n_best + dn
        if 5000 <= nl <= 9000:
            target_ns.append(nl)

    for new_n in target_ns[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("\n  --- n=" + str(new_n) + " ---")
        a_interp = interpolate_to_len(best_seq, new_n)

        # Quick FW at new length
        t5a = min(remaining * 0.3, 150.0)
        if t5a > 40:
            a_out, _ = fw_with_momentum(a_interp, t5a, solver, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fw_n" + str(new_n), t_global)

        # Quick gradient at new length
        remaining = time_budget - (time.monotonic() - t_global)
        t5b = min(remaining * 0.3, 100.0)
        if t5b > 30:
            a_out, _ = nesterov_pnorm(best_seq, t5b, p_start=10, p_end=100, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "pnorm_n" + str(new_n), t_global)

    # === Phase 6: Second round of FW with larger active set ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: FW round 2 (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.20, 250.0)
    if t6 > 50:
        a_out, _ = fw_with_momentum(best_seq, t6, solver, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fw2_n" + str(n_best), t_global)

    # === Phase 7: Final gradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final gradient (n=" + str(n_best) + ") ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 30:
        a_out, _ = nesterov_pnorm(best_seq, t7, p_start=20, p_end=200, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "final_pnorm", t_global)

    # === Phase 8: Final hybrid ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final hybrid (n=" + str(n_best) + ") ===")
    t8 = min(remaining * 0.25, 200.0)
    if t8 > 40 and n_best <= 8500:
        a_out, _ = hybrid_polish(best_seq, t8, solver, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "final_hybrid", t_global)

    # === Validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + str(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        a_raw = (x + 0.005) ** 1.0 * (1.0 - x + 0.005) ** 1.0
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
