def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def fft_conv2(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if b.size != n:
        b = np.resize(b, n)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def cross_convolve_p_a(p, a):
    p = np.asarray(p, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    total = 3 * n - 2
    if p.size != 2 * n - 1:
        p = np.resize(p, 2 * n - 1)
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    out = conv[n - 1:2 * n - 1][::-1]
    out = np.where(np.isfinite(out), out, 0.0)
    return out

def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    new_n = int(max(1, int(new_n)))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)

def mirror_descent_step(a, c, top_k, eta, beta_soft=0.0):
    """
    One step of mirror descent with negative entropy on the simplex.
    c = conv(a, a), top_k = indices of top k peaks
    If beta_soft > 0, use log-sum-exp weighting instead of hard top-k.
    """
    n = a.size
    size = 2 * n - 1
    if c.size != size:
        c = fft_conv(a)
    
    if beta_soft > 0:
        # Soft weighting: p[k] ∝ exp(beta * c[k])
        m = float(np.max(c))
        shifted = np.clip(beta_soft * (c - m), -500.0, 500.0)
        w = np.exp(shifted)
        sw = float(w.sum())
        if sw <= 0 or not math.isfinite(sw):
            w = np.zeros(size)
            w[top_k] = 1.0 / len(top_k) if len(top_k) > 0 else 0.0
            sw = 1.0
        p = w / sw
    else:
        # Hard top-k
        p = np.zeros(size)
        if len(top_k) > 0:
            p[top_k] = 1.0 / float(len(top_k))
    
    # Gradient-like: g[i] = sum_k p[k] * a[k-i] (cross correlation)
    g = cross_convolve_p_a(p, a)
    g = np.where(np.isfinite(g) & (g > 0.0), g, 0.0)
    
    # Multiplicative update: a[i] *= exp(-eta * g[i])
    loga = np.log(np.maximum(a, 1e-300))
    loga_new = loga - eta * g
    # Normalize
    loga_new -= float(np.max(loga_new))
    a_new = np.exp(loga_new)
    s = float(a_new.sum())
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    return a_new

                    print("  MD iter=" + str(iters) + " M=" + repr(M) +
                          " score=" + repr(2 * n * M))
        else:
            stall += 1
            if stall > 200:
                # Perturb and restart
                rng = np.random.default_rng(42 + iters)
                pert = 1.0 + rng.normal(0.0, 0.01, n)
                a = best_a * pert
                a = np.maximum(a, 1e-12)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = float(np.max(c))
                stall = 0
    
    return best_a


# ==================== L-BFGS on log-sum-exp ====================

def lse_objective_and_grad(v, beta, ref, n):
    """
    Objective in log-coordinates: a = softmax(v padded with 0 at ref).
    Returns (f, grad_v) where f = log-sum-exp(beta * c) of convolution.
    """
    if not np.all(np.isfinite(v)):
        return 1e100, np.zeros(v.size, dtype=np.float64)
    
    # Reconstruct a from v
    u = np.zeros(n, dtype=np.float64)
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return 1e100, np.zeros(v.size, dtype=np.float64)
    a = e / s
    
    # Convolution
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    
    # Log-sum-exp
    m = float(np.max(c))
    shifted = np.clip(beta * (c - m), -500.0, 500.0)
    ex = np.exp(shifted)
    se = float(ex.sum())
    if se <= 0 or not math.isfinite(se):
        return 1e100, np.zeros(v.size, dtype=np.float64)
    
    p = ex / se  # Soft max weights
    
    # Gradient w.r.t. a: grad_a[i] = 2 * sum_k p[k] * a[k-i]
    total = 3 * n - 2
    N2 = next_pow2(total)
    A2 = np.fft.rfft(a, N2)
    Pr = np.fft.rfft(p[::-1], N2)
    conv_grad = np.fft.irfft(A2 * Pr, N2)[:total]
    grad_a = 2.0 * conv_grad[n - 1:2 * n - 1][::-1]
    grad_a = np.where(np.isfinite(grad_a), grad_a, 0.0)
    
    # Chain rule: grad_v = softmax_jacobian * grad_a
    # a = softmax(u), u[ref] = 0, u[other] = v
    # d a_i / d v_j = a_i * (delta_ij - a_j) for i, j != ref
    # d a_i / d u_j = a_i * (delta_ij - a_j)
    mean_g = float(np.dot(a, grad_a))
    grad_u = a * (grad_a - mean_g)
    
    grad_v = np.zeros(v.size, dtype=np.float64)
    if ref > 0:
        grad_v[:ref] = grad_u[:ref]
    if ref < n - 1:
        grad_v[ref:] = grad_u[ref + 1:]
    
    f = m + math.log(se) / beta
    if not math.isfinite(f):
        f = 1e100
        grad_v = np.zeros(v.size, dtype=np.float64)
    
    return f, grad_v


def run_lbfegs_custom(a0, beta, maxfun, ref):
    """Run L-BFGS-B on log-sum-exp objective in log-coordinates."""
    a = norm_sum1(a0)
    n = a.size
    if n <= 1:
        return a
    
    # Extract v (log-coordinates excluding ref)
    if ref > 0:
        v_part1 = np.log(np.maximum(a[:ref], 1e-300))
    else:
        v_part1 = np.array([], dtype=np.float64)
    if ref < n - 1:
        v_part2 = np.log(np.maximum(a[ref + 1:], 1e-300))
    else:
        v_part2 = np.array([], dtype=np.float64)
    # Subtract log(a[ref]) for gauge fixing
    log_ref = math.log(max(float(a[ref]), 1e-300))
    v = np.concatenate((v_part1, v_part2)) - log_ref
    
    if not SCIPY_OK or maxfun <= 0:
        return a
    
    def objfun(v_in):
        f, g = lse_objective_and_grad(v_in, beta, ref, n)
        return f, g
    
    try:
        res = scipy.optimize.minimize(
            objfun, v, method="L-BFGS-B", jac=True,
            options={"maxfun": int(maxfun), "maxiter": int(maxfun + 50),
                     "ftol": 1e-13, "gtol": 1e-8, "disp": False}
        )
        if res is not None and hasattr(res, "x"):
            vx = np.asarray(res.x, dtype=np.float64)
            if vx.size == v.size and np.all(np.isfinite(vx)):
                u = np.zeros(n, dtype=np.float64)
                if ref > 0:
                    u[:ref] = vx[:ref]
                if ref < n - 1:
                    u[ref + 1:] = vx[ref:]
                um = float(np.max(u))
                e = np.exp(u - um)
                a_out = e / float(e.sum())
                return norm_sum1(a_out)
    except Exception as e:
        print("  L-BFGS error: " + repr(e))
    
    return a


# ==================== Frank-Wolfe with game direction ====================

def game_direction(f, active_idx, mw_iters=20, seed=0):
    f = norm_sum1(f)
    n = f.size
    size = 2 * n - 1
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K <= 0:
        return np.ones(n, dtype=np.float64) / float(n)
    
    rng = np.random.default_rng(seed)
    logits = rng.normal(0.0, 0.1, K)
    
    for _ in range(mw_iters):
        pf = np.zeros(size, dtype=np.float64)
        logits_n = logits - float(np.max(logits))
        w = np.exp(logits_n)
        w = w / float(w.sum())
        pf[active_idx] = w
        
        pressure = cross_convolve_p_a(pf, f)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)
        
        pmx = float(np.max(pressure))
        pmn = float(np.min(pressure))
        ptp = pmx - pmn
        if ptp <= 1e-30:
            g = np.ones(n, dtype=np.float64) / float(n)
        else:
            scaled = (pressure - pmn) / ptp
            w_g = np.exp(-10.0 * scaled)
            w_g = np.where(np.isfinite(w_g), w_g, 0.0)
            w_g += 1e-12
            g = w_g / float(w_g.sum())
            g = 0.97 * g + 0.03 * (np.ones(n, dtype=np.float64) / float(n))
            g = g / float(g.sum())
        
        h = fft_conv2(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmac_ = float(np.max(ha))
        hmin = float(np.min(ha))
        hptp = hmac_ - hmin
        if hptp > 1e-30:
            eta = 15.0 / hptp
            logits += eta * (ha - float(np.mean(ha)))
            logits = np.clip(logits, -50.0, 50.0)
    
    return g


def frank_wolfe_phase(a0, budget, active_max=2000, verbose=True):
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    active = np.argsort(c)[::-1][:active_max]
    
    no_improve = 0
    t = 0
    while time.monotonic() - t0 < time_limit - 3 and no_improve < 60:
        t += 1
        time_left = time_limit - (time.monotonic() - t0)
        if time_left < 15:
            break
        
        g = game_direction(a, active, mw_iters=18, seed=1000 + t)
        
        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        
        # Line search
        best_alpha = 0.0
        best_m = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            one = 1.0 - alpha
            val = (one * one) * c + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
            m = float(np.max(val))
            if math.isfinite(m) and m < best_m:
                best_m = m
                best_alpha = alpha
        
        if best_alpha > 0 and best_m < M - 1e-16:
            one = 1.0 - best_alpha
            a = one * a + best_alpha * g
            a = np.maximum(a, 0.0)
            s = float(a.sum())
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
                c = fft_conv(a)
                M = float(np.max(c))
            no_improve = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            active = np.argsort(c)[::-1][:active_max]
            if verbose and (t == 1 or t % 10 == 0):
                print("  FW t=" + str(t) + " score=" + repr(2 * n * best_M))
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                rng = np.random.default_rng(9999 + t)
                pert = rng.normal(0.0, 0.02, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-12)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = float(np.max(c))
                active = np.argsort(c)[::-1][:active_max]
                no_improve = 0
    
    return best_a


# ==================== Coordinate descent with rank-1 updates ====================

def apply_single_move(c, a, s, d, tmove):
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = a.size
    c_new = c.copy()
    c_new[d:d + n] += 2.0 * tmove * a
    c_new[s:s + n] -= 2.0 * tmove * a
    t2 = tmove * tmove
    if 2 * d < c_new.size:
        c_new[2 * d] += t2
    if 2 * s < c_new.size:
        c_new[2 * s] += t2
    if s + d < c_new.size:
        c_new[s + d] -= 2.0 * t2
    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
    return c_new


def coordinate_polish(a0, budget, verbose=True):
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    n = a.size
    stall = 0
    rnd = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit - 2 and stall < 80:
        rnd += 1
        
        top_k = min(30, 2 * n - 1)
        if top_k <= 0:
            break
        top_idx = np.argsort(c)[::-1][:top_k]
        p_top = np.zeros(c.size)
        p_top[top_idx] = 1.0 / float(top_k)
        pressure = cross_convolve_p_a(p_top, a)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)
        
        # Find best (s, d, frac) transfer
        src_score = pressure * (1.0 + float(n) * a)
        dst_score = pressure + 1e-5 * a
        src_cands = np.argpartition(src_score, min(8, n - 1))[:min(8, n)]
        dst_cands = np.argpartition(dst_score, min(12, n - 1))[:min(12, n)]
        
        best_cand = None
        for srr in src_cands:
            s = int(srr)
            if a[s] < 1e-12:
                continue
            for drr in dst_cands:
                d = int(drr)
                if d == s:
                    continue
                for frac in (0.003, 0.008, 0.02, 0.05):
                    tmove = a[s] * frac
                    if tmove <= 1e-16:
                        continue
                    tmove = min(tmove, a[s] * 0.99)
                    c_new = apply_single_move(c, a, s, d, tmove)
                    m_new = float(np.max(c_new))
                    if m_new < M - 1e-18:
                        if best_cand is None or m_new < best_cand[0]:
                            best_cand = (m_new, s, d, tmove)
        
        # Random exploratory moves
        if best_cand is None:
            for _ in range(50):
                s = int(rng.integers(n))
                d = int(rng.integers(n))
                if s == d or a[s] < 1e-12:
                    continue
                frac = float(rng.uniform(0.002, 0.03))
                tmove = a[s] * frac
                tmove = min(tmove, a[s] * 0.99)
                if tmove <= 1e-16:
                    continue
                c_new = apply_single_move(c, a, s, d, tmove)
                m_new = float(np.max(c_new))
                if m_new < M - 1e-18:
                    if best_cand is None or m_new < best_cand[0]:
                        best_cand = (m_new, s, d, tmove)
        
        if best_cand is not None:
            m_new, s, d, tmove = best_cand
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = fft_conv(a)
            M = float(np.max(c))
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and (rnd <= 3 or rnd % 50 == 0):
                print("  Coord rnd=" + str(rnd) + " score=" + repr(2 * n * best_M))
        else:
            stall += 1
            if stall % 8 == 7:
                # Larger random perturbation
                a = best_a + rng.normal(0.0, 0.01, n) * best_a
                a = np.maximum(a, 1e-12)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = float(np.max(c))
                stall = 0
    
    return best_a
