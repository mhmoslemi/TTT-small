def hankel_matvec(a, q):
    """Compute Q @ a where Q_{ij} = q_{i+j} (Hankel matrix)."""
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (2*n - 2).bit_length()
    A = np.fft.rfft(a_rev, fft_size)
    Qf = np.fft.rfft(q, fft_size)  # q has length 2n-1, padded to fft_size
    conv = np.fft.irfft(A * Qf, fft_size)
    return conv[n-1:2*n-1]  # indices n-1 to 2n-2, which is 2n-2 - (n-1) + 1 = n values
```

def fista_qp(a_init, q, max_iter=50):
    n = len(a_init)
    x = a_init.copy()
    x_prev = a_init.copy()
    t = 1.0
    step = 0.5  # 1/L, conservative
    
    for i in range(max_iter):
        t_new = (1 + sqrt(1 + 4*t*t)) / 2
        theta = (t - 1) / t_new
        y = x + theta*(x - x_prev)
        y = np.maximum(y, 0)
        y /= y.sum()
        
        grad = hankel_matvec(y, q)
        z = y - step * grad
        x_new = simplex_project(z)
        
        x_prev = x
        x = x_new
        t = t_new
    
    return x
```

"""

import numpy as np
import time
import math


def fast_autocorr(a):
    """Compute full linear convolution of a with itself."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def hankel_matvec(a, q):
    """Compute Q @ a where Q_{ij} = q_{i+j} (Hankel matrix, q has length 2n-1)."""
    n = len(a)
    a_rev = a[::-1]
    total_len = n + len(q) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a_rev, fft_size)
    Qf = np.fft.rfft(q, fft_size)
    conv = np.fft.irfft(A * Qf, fft_size)
    return conv[n - 1:2 * n - 1]


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def fista_qp(a_init, q, max_iter=30, step=0.8):
    """Solve min a^T Q a s.t. a>=0, sum a=1 using FISTA, Q_{ij}=q_{i+j}."""
    n = len(a_init)
    x = a_init.copy()
    x_prev = a_init.copy()
    t_val = 1.0
    current_step = step

    for _ in range(max_iter):
        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        theta = (t_val - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        grad = hankel_matvec(y, q)
        z = y - current_step * grad
        x_new = simplex_project(z)

        # Simple backtracking: if progress is poor, reduce step
        x_prev = x
        x = x_new
        t_val = t_new

    return x


def alternating_qp(a0, time_limit, beta_start=20.0, beta_growth=1.05,
                   inner_iters=25, verbose=True, seed=42):
    """Alternating: softmax peak weights + FISTA on Hankel QP."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    c = fast_autocorr(a)
    best_max_b = c.max()
    t0 = time.monotonic()
    beta = beta_start
    stall = 0
    rng = np.random.default_rng(seed)
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        c_max = c.max()

        if c_max < best_max_b - 1e-12:
            best_max_b = c_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Compute softmax weights focused on peaks
        shifted = beta * (c - c_max)
        shifted = np.clip(shifted, -400, 0)
        q = np.exp(shifted)
        q_sum = q.sum()
        if q_sum < 1e-300:
            q = np.zeros(len(c))
            q[np.argmax(c)] = 1.0
        else:
            q = q / q_sum

        # Solve QP subproblem with FISTA
        a_new = fista_qp(a, q, max_iter=inner_iters, step=0.7)

        # Blend for stability
        blend = 0.6
        a = (1 - blend) * a + blend * a_new
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        iteration += 1

        # Adapt beta
        if iteration % 10 == 0:
            beta *= beta_growth
            if beta > 5000:
                beta = 5000

        # Restart if stalled
        if stall > 200:
            stall = 0
            beta = max(beta * 0.5, 10.0)
            # Structured perturbation
            noise_scale = 0.02 / (1.0 + iteration / 500.0)
            anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.4 * anti + 0.6 * rand) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()

        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    AltQP iter=" + str(iteration) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + ", beta=" + str(round(beta, 1)))

    return best_a, best_max_b


def multi_source_peak_surgery(a0, time_limit, verbose=True, seed=99):
    """Multi-source mass redistribution to reduce top peaks simultaneously."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 5000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-14:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Find top peaks
        n_top = min(20, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]

        improved = False

        # For each top peak, find contributing positions and try to reduce
        for k in top_idx[:10]:
            if improved:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]

            if products.max() < 1e-20:
                continue

            # Find top 3 contributing pairs
            top_pairs = np.argsort(products)[::-1][:3]

            for pi in top_pairs:
                if improved:
                    break
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n:
                    continue

                # Determine which to reduce
                if a[src_i] >= a[partner]:
                    reduce_i = src_i
                else:
                    reduce_i = partner

                if a[reduce_i] < 1e-14:
                    continue

                # Compute pressure map
                pressure = np.zeros(n)
                for k2 in top_idx[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        pressure[idx2] += a[k2i - idx2]
                        pressure[k2i - idx2] += a[idx2]

                # Find safe destinations (low pressure, not in peak regions)
                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_idx[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2 + 1] = True

                candidates = np.where(~in_peak)[0]
                if len(candidates) < 20:
                    candidates = np.arange(n)
                candidates = candidates[candidates != reduce_i]

                if len(candidates) == 0:
                    continue

                # Try best low-pressure destinations
                cand_press = pressure[candidates]
                n_try = min(4, len(candidates))
                try_dsts = candidates[np.argsort(cand_press)[:n_try]]

                for dst in try_dsts:
                    if improved:
                        break
                    for frac in [0.003, 0.008, 0.015, 0.025, 0.04]:
                        amt = a[reduce_i] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[reduce_i] -= amt
                        a_trial[int(dst)] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-15:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            break

        if not improved:
            # Random exploration
            if no_improve > 200:
                for _ in range(15):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                break

        if verbose and no_improve % 1000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSurgery (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve))

    return best_a, best_m


def pairwise_cd(a0, time_limit, verbose=True, seed=123):

    n = len(a0)

    t0 = time.monotonic()

        c = fast_autocorr(a)
        threshold = np.percentile(c, 93)
        peak_positions = np.where(c > threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.argsort(c)[::-1][:15]

        improved = False

        for k in peak_positions[:15]:
            if improved and (time.monotonic() - t0 > time_limit):
                break
            k_int = int(k)

            i_end = min(n - 1, k_int)
            if i_start > i_end:

            idx = np.arange(i_start, i_end + 1)

                    break
                src_i = int(idx[pi])

                if a[src_i] < 1e-14 and a[partner] < 1e-14:

                        a_trial = a.copy()

                        a_trial[dst] += amt

                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t

                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break

        if not improved:
            stall += 1

                eps = 0.005 / (1.0 + stall / 100.0)

                if s_t > 1e-15:
                    a_try = a_try / s_t
                    M_try = fast_autocorr(a_try).max()
                    if M_try < best_m:
                        a = a_try
                        best_m = M_try
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 300 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PairCD (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def simulated_annealing(a0, time_limit, T_start=0.05, T_end=0.001,

    """Simulated annealing with targeted moves on peak positions."""

    T = T_start
    iteration = 0

    while time.monotonic() - t0 < time_limit and T > T_end:
        # Compute current peaks
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-14:

            best_a = a.copy()

        # Choose a move type

            k_int = int(top_idx[int(rng.integers(len(top_idx)))])
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                products = a[idx] * a[k_int - idx]
                # Pick proportional to product
                total_p = products.sum()
                if total_p > 0:
                    probs = products / total_p
                    src_pos = int(idx[rng.choice(len(idx), p=probs)])
                    dst = int(rng.integers(n))
                    if dst != src_pos and a[src_pos] > 1e-12:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a[src_pos] * frac
                        a_trial = a.copy()
                        a_trial[src_pos] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        a_trial = a_trial / a_trial.sum()
                        M_trial = fast_autocorr(a_trial).max()
                        delta = M_trial - c_max
                        if delta < 0 or rng.random() < math.exp(-delta / T):
                            a = a_trial
        elif move_type == 1:
            # Swap: swap values between two positions
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                a_trial = a.copy()
                a_trial[i], a_trial[j] = a_trial[j], a_trial[i]
                M_trial = fast_autocorr(a_trial).max()
                delta = M_trial - c_max
                if delta < 0 or rng.random() < math.exp(-delta / T):
                    a = a_trial
        else:
            # Small Gaussian perturbation on a random subset
            subset_size = max(5, n // 50)
            positions = rng.choice(n, size=subset_size, replace=False)
            a_trial = a.copy()
            a_trial[positions] += rng.normal(0, 0.001, subset_size) * a
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                delta = M_trial - c_max
                if delta < 0 or rng.random() < math.exp(-delta / T):
                    a = a_trial

        T *= cooling
        iteration += 1

        if verbose and iteration % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SA iter=" + str(iteration) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + ", T=" + str(round(T, 6)))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score
