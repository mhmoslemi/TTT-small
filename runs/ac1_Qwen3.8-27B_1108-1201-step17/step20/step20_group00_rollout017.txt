    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_matrix(y, lags, n):
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs

def lp_fw_optimize(a0, time_limit, solver, init_K=800, max_K=6000, verbose=True):
    """LP-FW with dual-guided active set management."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    ac = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac)[::-1][:init_K])
    no_improve = 0
    stall_limit = 2000
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    Yf = np.fft.rfft(y, fft_size)

    # Track LP solver iterations for stall detection
    lp_fail_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        if lp_fail_count > 20:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            lp_fail_count = 0
        else:
            no_improve += 1

        clist = sorted(active_set)
        # Prune active set: keep lags with highest (a*a) values
        if len(clist) > max_K:
            margins = np.array([ac[k] for k in clist])
            keep = np.argsort(margins)[::-1][:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)

        K = len(clist)
        if K == 0:
            break

        # Build LP
        coeffs = build_lp_matrix(y, clist, n)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        lp_ok = False
        lp_M = None
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                lp_ok = True
                lp_M = float(M_var.value) if M_var.value is not None else None
        except Exception:
            lp_ok = False

        if not lp_ok:
            lp_fail_count += 1
            if no_improve > stall_limit // 3:
                break
            # Reduce active set and retry
            if len(clist) > 200:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:500])
            else:
                break
            continue

        lp_fail_count = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Update active set using dual info and violations
        Yf_new = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf_new * Gf, fft_size)[:size]

        # Add lags where (y*g) exceeds LP bound significantly
        threshold = (lp_M + M_curr) / 2.0 if lp_M is not None else M_curr * 0.95
        viol = np.where(conv_yg > threshold)[0]
        if len(viol) > 0:
            top_v = viol[np.argsort(conv_yg[viol])[::-1][:200]]
            for k in top_v:
                active_set.add(int(k))

        # Also ensure top peaks of (y*y) are in active set
        top_peaks = np.argsort(ac)[::-1][:min(500, len(ac))]
        for k in top_peaks[:200]:
            active_set.add(int(k))

        # Periodic refresh: every 200 iterations, reset to top max_K
        if t % 200 == 0 and len(active_set) > max_K // 2:
            top_refresh = np.argsort(ac)[::-1][:max_K]
            active_set = set(int(k) for k in top_refresh)

        # Golden section line search with more iterations
        lo = 0.005
        hi = 1.0
        if t <= 3:
            # For first few iterations, try aggressive steps
            candidates = [0.5, 0.8, 1.0, 0.3, 0.15]
            best_alpha = candidates[0]
            best_f = None
            for alpha in candidates:
                trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts
                f_trial = fast_autocorr(trial).max()
                if best_f is None or f_trial < best_f:
                    best_f = f_trial
                    best_alpha = alpha
        else:
            c1 = lo + inv_phi * (hi - lo)
            c2 = hi - inv_phi * (hi - lo)

            def trial_obj(alpha):
                trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
                ts = trial.sum()
                if ts < 1e-15:
                    return 1e10
                trial = trial / ts
                return fast_autocorr(trial).max()

            f1 = trial_obj(c1)
            f2 = trial_obj(c2)
            for _ in range(28):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_phi * (hi - lo)
                    f1 = trial_obj(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = hi - inv_phi * (hi - lo)
                    f2 = trial_obj(c2)
            best_alpha = (lo + hi) / 2.0

        y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy
        else:
            y = g_dir.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(2 * n * best_m) + " |S|=" + str(len(active_set)) + " alpha=" + repr(round(best_alpha, 4)))

    if verbose:
        print("    LP-FW done (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2 * n * best_m))
    return best_a, 2.0 * n * best_m

def grad_peak_deflation(a0, time_limit, top_k=80, verbose=True):
    """Gradient-based peak deflation with adaptive temperature."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    x_prev = x.copy()
    stall = 0
    temp = 60.0

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 15000:
            break

        beta_t = (t - 1.0) / (t + 3.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]

        # Adaptive temperature
        cur_temp = temp * max(0.2, 1.0 - t / 30000.0)
        w = np.exp(cur_temp * (top_vals - top_vals[0]))
        w = w / w.sum()

        # Compute gradient
        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            idx_arr = np.arange(i_s, i_e)
            grad[idx_arr] += 2.0 * wi * y[ki - idx_arr]

        eta_t = 2.5 / (np.sqrt(max(t, 1)) + 1.0)
        log_a = np.log(y) - eta_t * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        val = obj_val(x)
        if val < best_val - 1e-13:
            best_val = val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 5000 and stall % 5000 == 0:
            noise_scale = 0.005 + 0.01 * (stall / 20000.0)
            noise = noise_scale * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            stall = 0

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print("    GradPeak t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(best_val))

    return best_a, best_val

def coord_polish(a0, time_limit, seed=42, verbose=True):
    """Coordinate descent with contribution-based destination selection."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 2000

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            top_srcs = idx[np.argsort(contrib)[::-1][:3]]
            for src in top_srcs:
                src = int(src)
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Score destinations based impact on top peaks
                scores = np.zeros(n)
                for k2 in top_k[:6]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                        scores[k2i - idx2] += a[idx2]
                scores[src] = 1e10

                candidates = np.argsort(scores)[:40]
                found = False
                for dst in candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.001, 0.003, 0.007, 0.015, 0.03]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-15:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            found = True
                            break
                    if found:
                        break

        if not improved:
            stall += 1
            if stall > 50 and stall % 10 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.008)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 500 == 0:
            print("    Coord (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2 * n * best_m) + " stall=" + str(stall))

    return best_a, 2.0 * n * best_m

def diverse_init(n, parent_seq):
    """Generate diverse initial candidates."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)

    candidates = []

    # Sine family
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(normalize(np.sin(np.pi * x) ** p))

    # Asymmetric: concentrated at one end
    for c in [0.1, 0.2, 0.3, 0.7, 0.8, 0.9]:
        for sg in [0.05, 0.1, 0.15, 0.2]:
            candidates.append(normalize(np.exp(-((x - c) ** 2) / (2 * sg * sg))))

    # Double bumps (asymmetric)
    for c1 in [0.1, 0.2, 0.3]:
        for c2 in [0.6, 0.7, 0.8]:
            for sg in [0.05, 0.1, 0.15]:
                cands = np.exp(-((x - c1) ** 2) / (2 * sg * sg)) + np.exp(-((x - c2) ** 2) / (2 * sg * sg))
                candidates.append(normalize(cands))

    # Power-law edges
    for p in [0.3, 0.5, 1.0, 2.0, 3.0]:
        candidates.append(normalize((i_arr + 1.0) ** p * (n - i_arr + 1.0) ** p))
        candidates.append(normalize((i_arr + 1.0) ** p))
        candidates.append(normalize((n - i_arr + 1.0) ** p))

    # Cosine windows
    for f in [1, 2, 3, 4]:
        cands = np.maximum(1.0 + np.cos(2 * np.pi * f * x), 0.01)
        candidates.append(normalize(cands))

    # Tukey windows
    for alpha in [0.1, 0.2, 0.4, 0.6]:
        cands = np.ones(n)
        m = int(alpha * n)
        if m > 0:
            taper = 0.5 * (1 + np.cos(np.pi * np.arange(m) / m))
            cands[:m] = taper
            cands[-m:] = taper[::-1]
        candidates.append(normalize(cands))

    # Blend with parent
    if parent_seq is not None:
        bi = normalize(parent_seq)
        for w in [0.5, 0.7, 0.9]:
            for c in candidates[:5]:
                candidates.append(normalize(w * bi + (1 - w) * c))

    best_a = None
    best_val = float('inf')
    for c in candidates:
        v = obj_val(c)
        if v < best_val:
            best_val = v
            best_a = c.copy()

    if parent_seq is not None:
        pv = obj_val(normalize(parent_seq))
        if pv < best_val:
            best_val = pv
            best_a = normalize(parent_seq)

    return best_a, best_val

            best_seq = normalize(p)

    result = [float(max(0.0, min(1000.0, x))) for x in np.asarray(best_seq, dtype=np.float64)]
    return result
