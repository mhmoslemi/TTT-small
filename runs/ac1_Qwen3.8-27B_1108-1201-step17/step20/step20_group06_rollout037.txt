"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. L-BFGS-B with softmax parameterization and adaptive smoothed objective (continuation method)
2. Spectral domain equalization
3. Exact top-K peak gradient descent with momentum
4. Multi-length homotopy exploration
5. Final polish with high-precision gradient

Key innovations over previous approach:
- L-BFGS-B leverages curvature information for faster convergence
- Softmax parameterization ensures non-negativity and sum=1 automatically
- Continuation method (beta annealing) for global-to-local optimization
- Spectral equalization for global structure improvement
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_autocorr(a):
    """Compute linear autocorrelation (convolve(a,a)) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative simplex (sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def _softmax(x):
    """Stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def _obj_grad_lbfgs(x, n, beta, top_k):
    """Objective and gradient for L-BFGS-B with softmax parameterization."""
    a = _softmax(x)
    c = fast_autocorr(a)
    size = 2 * n - 1
    
    if top_k < size:
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_use = np.zeros(size)
        c_use[idx] = c[idx]
    else:
        c_use = c.copy()
    
    c_max = c_use.max()
    if c_max < 1e-30:
        return 2.0 * n * c_max, np.zeros(n)
    
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s
    
    smooth_max = c_max + np.log(s) / beta
    obj = 2.0 * n * smooth_max
    
    # Gradient w.r.t. a: G[i] = (2/beta) * sum_j p[j+i] * a[j]
    a_rev = a[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    G = (2.0 / beta) * conv[n-1:2*n-1]
    
    # Chain rule through softmax: dx_grad[i] = a[i] * (G[i] - G·a)
    Ga = np.dot(G, a)
    grad_x = a * (G - Ga)
    
    return obj, grad_x


def lbfgs_continuation(a0, time_limit, beta_schedule=None, verbose=True, seed=42):
    """L-BFGS-B with continuation method (annealing beta)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_max = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    
    if beta_schedule is None:
        beta_schedule = [20, 50, 100, 200, 400, 800, 1600, 3200]
    
    # Initialize x from a
    x = np.log(a + 1e-15)
    x -= x.mean()
    
    for beta_idx, beta in enumerate(beta_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        
        top_k = max(20, int((2*n-1) * 0.05))
        remaining = time_limit - (time.monotonic() - t0)
        max_iter = max(30, min(200, int(remaining * 50)))
        
        try:
            result = scipy_minimize(
                _obj_grad_lbfgs, x, method='L-BFGS-B', jac=True,
                args=(n, beta, top_k),
                options={'maxiter': max_iter, 'maxfun': max_iter * 2, 'gtol': 1e-8}
            )
        except Exception:
            continue
        
        x = np.asarray(result.x, dtype=np.float64)
        x -= x.mean()  # Re-center for numerical stability
        a_new = _softmax(x)
        
        c_new = fast_autocorr(a_new)
        m_new = c_new.max()
        if m_new < best_max - 1e-15:
            best_max = m_new
            best_a = a_new.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max
            print(f"    L-BFGS beta={beta} ({elapsed:.1f}s): J={J:.6f} max_b={best_max:.2e}")
    
    return best_a, best_max

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. L-BFGS-B with softmax parameterization and continuation method (beta annealing)
2. Spectral domain equalization for global structure
3. Exact top-K peak gradient descent with momentum and SA acceptance
4. Multi-length homotopy exploration with structured interpolation
5. Peak-directed coordinate descent for final polish

Key innovations: L-BFGS leverages curvature for faster convergence;
softmax ensures feasibility; continuation provides global-to-local path.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_autocorr(a):
    """Compute linear autocorrelation convolve(a,a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative simplex (sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def _softmax(x):
    """Stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def _obj_grad_lbfgs(x, n, beta, top_k):
    """Objective and gradient for L-BFGS-B with softmax parameterization."""
    a = _softmax(x)
    c = fast_autocorr(a)
    size = 2 * n - 1

    if top_k < size:
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_use = np.zeros(size)
        c_use[idx] = c[idx]
    else:
        c_use = c.copy()

    c_max = c_use.max()
    if c_max < 1e-30:
        return 2.0 * n * c_max, np.zeros(n)

    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s

    smooth_max = c_max + np.log(s) / beta
    obj = 2.0 * n * smooth_max

    a_rev = a[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    G = (2.0 / beta) * conv[n - 1:2 * n - 1]

    Ga = np.dot(G, a)
    grad_x = a * (G - Ga)

    return obj, grad_x


def lbfgs_continuation(a0, time_limit, beta_schedule=None, verbose=True, seed=42):
    """L-BFGS-B with continuation method (annealing beta from low to high)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_max = fast_autocorr(best_a).max()
    t0 = time.monotonic()

    if beta_schedule is None:
        beta_schedule = [15, 30, 60, 120, 250, 500, 1000, 2000, 4000]

    x = np.log(a + 1e-15)
    x -= x.mean()

    for beta_idx, beta in enumerate(beta_schedule):
        if time.monotonic() - t0 > time_limit:
            break

        top_k = max(20, int((2 * n - 1) * 0.04))
        remaining = time_limit - (time.monotonic() - t0)
        max_iter = max(20, min(150, int(remaining * 30)))

        try:
            result = scipy_minimize(
                _obj_grad_lbfgs, x, method='L-BFGS-B', jac=True,
                args=(n, beta, top_k),
                options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                         'gtol': 1e-10, 'ftol': 1e-12}
            )
        except Exception:
            continue

        x = np.asarray(result.x, dtype=np.float64)
        x -= x.mean()
        a_new = _softmax(x)
        c_new = fast_autocorr(a_new)
        m_new = c_new.max()
        if m_new < best_max - 1e-15:
            best_max = m_new
            best_a = a_new.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max
            print("    L-BFGS beta=" + str(beta) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " max_b=" + str(best_max) + " status=" + str(result.status))

    return best_a, best_max

def peak_directed_cd(a0, time_limit, verbose=True, seed=456):

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def try_update(best_seq, best_score, a, label, t_global):

        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

    t2 = min(remaining * 0.08, 90.0)
    if t2 > 15:
        a_out, _ = spectral_equalize(best_seq.copy(), t2, verbose=True, seed=777)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec1", t_global)

    # Phase 3: Top-K momentum descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P3: TopK Momentum (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_out, _ = topk_momentum_descent(best_seq.copy(), t3,
                                          top_k=50, momentum=0.85,
                                          step_init=0.3, verbose=True, seed=123)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "topk1", t_global)

    # Second L-BFGS round with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P3b: L-BFGS Round 2 (n=" + str(n_best) + ") ===")
    t3b = min(remaining * 0.15, 200.0)
    if t3b > 20:
        a_out, _ = lbfgs_continuation(best_seq.copy(), t3b,
                                       beta_schedule=[50, 100, 200, 500, 1000, 2000, 4000],
                                       verbose=True, seed=999)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs2", t_global)

    # Phase 4: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P4: Multi-length (base n=" + str(n_best) + ") ===")

    longer_lengths = [n_best + 400, n_best + 800, n_best + 1500]
    # Also try slightly shorter
    shorter_lengths = [n_best - 500]
    all_lengths = [l for l in shorter_lengths + longer_lengths if 3000 <= l <= 12000]

    for new_n in all_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print("  -> n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        # L-BFGS on new length
        t4a = min(remaining * 0.4, 200.0)
        if t4a > 20:
            a_out, _ = lbfgs_continuation(a_long, t4a,
                                           beta_schedule=[20, 50, 100, 200, 500, 1000],
                                           verbose=True, seed=new_n % 7919)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs_n" + str(new_n), t_global)

        # Top-K on new length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t4b = min(remaining * 0.3, 120.0)
            if t4b > 15:
                a_out, _ = topk_momentum_descent(best_seq.copy(), t4b,
                                                  top_k=40, momentum=0.85,
                                                  step_init=0.25, verbose=False, seed=new_n % 3163)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "topk_n" + str(new_n), t_global)

    # Phase 5: Peak-directed coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P5: Peak-Directed CD (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 15:
        a_out, _ = peak_directed_cd(best_seq.copy(), t5, verbose=True, seed=456)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakcd1", t_global)

    # Phase 6: Final polish - short L-BFGS + spectral
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P6: Final Polish (n=" + str(n_best) + ") ===")

    if remaining > 30:
        t6a = min(remaining * 0.35, 150.0)
        if t6a > 15:
            a_out, _ = lbfgs_continuation(best_seq.copy(), t6a,
                                           beta_schedule=[100, 300, 1000, 3000],
                                           verbose=True, seed=555)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs_final", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6b = min(remaining * 0.25, 80.0)
            if t6b > 10:
                a_out, _ = spectral_equalize(best_seq.copy(), t6b, verbose=False, seed=888)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "spec_final", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6c = min(remaining * 0.3, 80.0)
            if t6c > 10:
                a_out, _ = topk_momentum_descent(best_seq.copy(), t6c,
                                                  top_k=30, momentum=0.9,
                                                  step_init=0.15, verbose=False, seed=321)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "topk_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: J=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
