def next_pow2(x):

def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size

        return np.ones(1, dtype=np.float64)

    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s

def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size

    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c

def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)

    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)

def lse_a(a, beta, mask=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 0.0, np.array([], dtype=np.float64)
    size = 2 * n - 1
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    if mask is None:
        cv = c
    else:
        cv = c[mask]
    if cv.size == 0:
        return 0.0, np.zeros(n, dtype=np.float64)
    m = float(np.max(cv))
    shifted = beta * (cv - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0 or not math.isfinite(se):
        e = np.zeros(cv.size, dtype=np.float64)
        e[0] = 1.0
        se = 1.0
    p = e / se
    pf = np.zeros(size, dtype=np.float64)
    if mask is None:
        pf[:] = p
    else:
        pf[mask] = p
    Pr = np.fft.rfft(pf[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    f = m + math.log(se) / beta
    if not math.isfinite(f):
        f = 1e100
    return f, grad

def softmax_from_u(u):
    u = np.asarray(u, dtype=np.float64)

    if n == 0:

    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s

def cross_convolve_p_a(p, a, factor=1.0):
    p = np.asarray(p, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    total = 3 * n - 2
    if p.size != 2 * n - 1:
        p = np.resize(p, 2 * n - 1)
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    out = conv[n - 1:2 * n - 1][::-1]
    out = np.where(np.isfinite(out), out, 0.0)
    return factor * out

def game_direction(f, active_idx, mw_iters=30, seed=0, init_logits=None):

            logits += eta * (ha - float(np.mean(ha)))

def lp_direction_hihts(f, active_idx, time_limit=10.0, max_k=1500):

        return None
    f = norm_sum1(f)

    if active_idx.size == 0:
        return None
    if active_idx.size > max_k:
        active_idx = active_idx[:max_k]
    K = active_idx.size
    row_parts = []
    col_parts =
    val_parts = []
    for r, k in enumerate(active_idx):
        s = max(0, int(k) - n + 1)
        e = min(n, int(k) + 1)
        if s >= e:
            continue
        cols = np.arange(s, e, dtype=np.int32)
        row_parts.append(np.full(cols.size, r, dtype=np.int32))
        col_parts.append(cols)
        vals = f[int(k) - cols].astype(np.float64)
        val_parts.append(vals)
    if not row_parts:
        return None
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0
    neg_one = scipy.sparse.csc_matrix(-np.ones((K, 1)), dtype=np.float64)
    A_ub = scipy.sparse.hstack((A, neg_one), format="csr")
    A_eq = np.ones((1, n + 1), dtype=np.float64)
    A_eq[0, n] = 0.0
    b_eq = np.ones(1, dtype=np.float64)
    b_ub = np.zeros(K, dtype=np.float64)
    bounds = [(0.0, None)] * n + [(None, None)]
    try:
        res = scipy.optimize.linprog(
            c_obj, A_ub=A_ub, b_ub=b_ub,
            A_eq=A_eq, b_eq=b_eq, bounds=bounds,
            method="HiGHS",
            options={"disp": False, "time_limit": float(time_limit)}
        )
    except Exception:
        return None
    if res is None or not getattr(res, "success", False):
        return None
    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.where(np.isfinite(g) & (g > 0.0), g, 0.0)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs

def bisection_line_search(a, g, cf, cg, fg, current_M, tol=1e-12, max_iter=60):
    """Convex along line: find best alpha in [0,1] minimizing max of (1-a)^2 cf + 2a(1-a)fg + a^2 cg."""
    best_alpha = 0.0
    best_m = current_M
    lo = 0.0
    hi = 1.0
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        mid = (lo + hi) / 2.0
        a1 = 1.0 - mid
        val = (a1 * a1) * cf + (2.0 * mid * a1) * fg + (mid * mid) * cg
        m = float(np.max(val))
        if not math.isfinite(m):
            lo = mid
            continue
        # Check derivative sign at mid to determine direction
        # d/da of ((1-a)^2 cf + 2a(1-a)fg + a^2 cg) at coefficient level
        # = -2(1-a)cf + 2(1-2a)fg + 2a*cg elementwise
        # We use a finite difference on the max
        da = 1e-8
        mid2 = min(mid + da, 1.0)
        a1b = 1.0 - mid2
        val2 = (a1b * a1b) * cf + (2.0 * mid2 * a1b) * fg + (mid2 * mid2) * cg
        m2 = float(np.max(val2))
        if not math.isfinite(m2):
            m2 = m + 1
        if m2 > m:
            # Function increasing at mid, so optimum is to the left
            hi = mid
        else:
            lo = mid
        mid_final = (lo + hi) / 2.0
        a1f = 1.0 - mid_final
        valf = (a1f * a1f) * cf + (2.0 * mid_final * a1f) * fg + (mid_final * mid_final) * cg
        mf = float(np.max(valf))
        if math.isfinite(mf) and mf < best_m:
            best_m = mf
            best_alpha = mid_final
    # Also check alpha=1
    m1 = float(np.max(cg))
    if math.isfinite(m1) and m1 < best_m:
        best_m = m1
        best_alpha = 1.0
    if best_alpha <= 0.0:
        return a.copy(), current_M, 0.0
    new_a = (1.0 - best_alpha) * a + best_alpha * g
    new_a = np.maximum(new_a, 0.0)
    s = float(new_a.sum())
    if s < 1e-15:
        return a.copy(), current_M, 0.0
    new_a = new_a / s
    return new_a, best_m, best_alpha

def frank_wolfe_search(a0, budget, mask, active_max=3000, game_iters=25,
                       exact_every=3, exact_time=10.0, verbose=True):
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = proxy_max(c, mask)
    best_a = a.copy()
    best_M = M
    active = active_top(c, mask, active_max)
    no_improve = 0
    t = 0
    while time.monotonic() - t0 < time_limit - 5 and no_improve < 120:
        t += 1
        time_left = time_limit - (time.monotonic() - t0)
        if time_left < 30:
            break
        g = None
        if hihts_available() and (t % exact_every == 0) and time_left > 120:
            max_k_use = 1200 if time_left > 300 else 600
            g = lp_direction_hihts(a, active, exact_time, max_k=max_k_use)
        if g is None:
            init_log = None
            if active.size > 0:
                av = c[active]
                av = np.where(np.isfinite(av) & (av > 0.0), av, 1e-300)
                init_log = np.log(av)
                init_log -= float(np.max(init_log))
            g = game_direction(a, active, mw_iters=game_iters, seed=2000 + t, init_logits=init_log)
        if g is None:
            break
        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        new_a, new_M, alpha = bisection_line_search(a, g, c, cg, fg, M)
        if new_M < M - 1e-16 and alpha > 0:
            a = new_a
            c = fft_conv(a)
            M = proxy_max(c, mask)
            no_improve = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            active = active_top(c, mask, active_max)
            if verbose and (t <= 3 or t % 20 == 0):
                print("    FW t=" + str(t) + " M=" + repr(M) +
                      " best_M=" + repr(best_M) + " alpha=" + repr(alpha))
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                rng = np.random.default_rng(5555 + t)
                pert = rng.normal(0.0, 0.02, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-12)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = proxy_max(c, mask)
                active = active_top(c, mask, active_max)
                no_improve = 0
            if verbose and (t <= 3 or t % 20 == 0):
                print("    FW stall t=" + str(t) + " best_M=" + repr(best_M))
    return best_a

def apply_single_move(c, a, s, d, tmove):
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = a.size
    c_new = c.copy()
    c_new[d:d + n] += 2.0 * tmove * a
    c_new[s:s + n] -= 2.0 * tmove * a
    t2 = tmove * tmove
    if 2 * d >= 0 and 2 * d < c_new.size:
        c_new[2 * d] += t2
    if 2 * s >= 0 and 2 * s < c_new.size:
        c_new[2 * s] += t2
    if s + d >= 0 and s + d < c_new.size:
        c_new[s + d] -= 2.0 * t2
    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
    return c_new

def coordinate_polish(a0, budget, mask, verbose=True):
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    c = fft_conv(a)
    M = proxy_max(c, mask)
    best_a = a.copy()
    best_M = M
    n = a.size
    stall = 0
    rnd = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit - 3 and stall < 200:
        rnd += 1
        if n <= 1:
            break
        top_k = min(60, 2 * n - 1)
        top_idx = active_top(c, mask, top_k)
        if top_idx.size == 0:
            break
        p_top = np.zeros(2 * n - 1, dtype=np.float64)
        p_top[top_idx] = 1.0 / float(top_idx.size)
        pressure = cross_convolve_p_a(p_top, a, 2.0)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)

        src_count = min(12, n)
        dst_count = min(16, n)
        src_score = pressure * (1.0 + float(n) * a)
        src_candidates = np.argpartition(src_score, min(src_count - 1, n - 1))[:src_count]
        dst_score = pressure + 1e-5 * a
        dst_candidates = np.argpartition(dst_score, min(dst_count - 1, n - 1))[:dst_count]

        improved_any = False
        best_cand_val = 1e300
        best_cand_s = -1
        best_cand_d = -1
        best_cand_t = 0.0

        for srr in src_candidates:
            s = int(srr)
            if a[s] < 1e-14:
                continue
            for drr in dst_candidates:
                d = int(drr)
                if d == s:
                    continue
                for frac in (0.003, 0.007, 0.015, 0.04, 0.08):
                    tmove = a[s] * frac
                    if tmove <= 1e-18:
                        continue
                    tmove = min(tmove, a[s] * 0.99)
                    c_new = apply_single_move(c, a, s, d, tmove)
                    m_new = proxy_max(c_new, mask)
                    if m_new < best_cand_val:
                        best_cand_val = m_new
                        best_cand_s = s
                        best_cand_d = d
                        best_cand_t = tmove
                        improved_any = True

        if not improved_any or best_cand_val >= M:
            for _ in range(40):
                s = int(rng.integers(n))
                d = int(rng.integers(n))
                if s == d or a[s] < 1e-14:
                    continue
                frac = float(rng.uniform(0.002, 0.06))
                tmove = a[s] * frac
                tmove = min(tmove, a[s] * 0.99)
                if tmove <= 1e-18:
                    continue
                c_new = apply_single_move(c, a, s, d, tmove)
                m_new = proxy_max(c_new, mask)
                if m_new < best_cand_val:
                    best_cand_val = m_new
                    best_cand_s = s
                    best_cand_d = d
                    best_cand_t = tmove
                    improved_any = True

        if improved_any and best_cand_val < M - 1e-18:
            a[best_cand_s] -= best_cand_t
            a[best_cand_d] += best_cand_t
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = fft_conv(a)
            M = proxy_max(c, mask)
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and (rnd <= 3 or rnd % 50 == 0):
                print("    Coord rnd=" + str(rnd) + " M=" + repr(M) + " best_M=" + repr(best_M))
        else:
            stall += 1
            if stall % 15 == 14:
                # Random restart within neighborhood
                for _ in range(200):
                    s = int(rng.integers(n))
                    d = int(rng.integers(n))
                    if s == d or a[s] < 1e-14:
                        continue
                    frac = float(rng.uniform(0.001, 0.04))
                    tmove = a[s] * frac
                    tmove = min(tmove, a[s] * 0.99)
                    c_new = apply_single_move(c, a, s, d, tmove)
                    m_new = proxy_max(c_new, mask)
                    if m_new < M - 1e-17:
                        a[s] -= tmove
                        a[d] += tmove
                        a = np.maximum(a, 0.0)
                        ss = float(a.sum())
                        if ss < 1e-15:
                            break
                        a = a / ss
                        c = fft_conv(a)
                        M = proxy_max(c, mask)
                        if M < best_M:
                            best_M = M
                            best_a = a.copy()
                        stall = 0
                        break
    return best_a

def lbfegs_phase(a_init, mask, budget, lb_cost):
    a = norm_sum1(a_init)
    n = a.size
    size = 2 * n - 1
    best_a = a.copy()
    best_M = proxy_max(fft_conv(a), mask)
    current_a = a.copy()
    t1 = int(min(1200, max(60, size // 8)))
    t2 = int(min(800, max(50, size // 15)))
    t3 = int(min(500, max(40, size // 25)))
    t4 = int(max(20, size // 60))
    t5 = int(max(10, size // 120))
    schedules = [
        (5.0, None),
        (15.0, None),
        (40.0, None),
        (80.0, t1),
        (150.0, t2),
        (300.0, t3),
        (600.0, t4),
        (1200.0, t5),
        (2500.0, None)
    ]
    total_eval_budget = int(budget / max(3.0 * lb_cost, 0.05))
    per_run = int(max(100, min(800, max(1, total_eval_budget // max(1, len(schedules))))))
    t0 = time.monotonic()
    for beta, top_count in schedules:
        if time.monotonic() - t0 > budget - 5:
            break
        if top_count is None:
            msk = None
        else:
            msk = make_top_mask(current_a, top_count, mask)
        a_new = run_lbfegs(current_a, beta, msk, maxfun=per_run, verbose=True)
        c_new = fft_conv(a_new)
        m_new = proxy_max(c_new, mask)
        if math.isfinite(m_new):
            if m_new <= best_M * (1.0 + 1e-5) + 1e-18:
                current_a = a_new
        else:
            current_a = best_a
        if math.isfinite(m_new) and m_new < best_M - 1e-15:
            best_M = m_new
            best_a = a_new.copy()
    return best_a

def default_sequence(n=5000):
    x = np.linspace(0.0, 1.0, int(n))
    return norm_sum1(np.sin(np.pi * x) ** 2)

def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    try:
        total_budget = max(60.0, min(float(budget_s) - 60.0, 1740.0))
    except Exception:
        total_budget = 1740.0

    print("Start: budget=" + repr(total_budget))

    seq_best = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        np.clip(p, 0.0, 1000.0, out=p)
        if p.ndim != 1:
            p = p.ravel()
        if p.size > 0 and float(p.sum()) > 1e-10:
            seq_best = norm_sum1(p)
    except Exception:
        seq_best = None

    if seq_best is None:
        seq_best = default_sequence(5000)

    best_score = safe_eval(seq_best)
    if not math.isfinite(best_score):
        best_score = 2.0 * seq_best.size * proxy_max(fft_conv(seq_best), None)
    print("Initial: n=" + str(seq_best.size) + " score=" + repr(best_score))

    if seq_best.size > 10000:
        work_seq = interpolate_sequence(seq_best, 8000)
    else:
        work_seq = seq_best.copy()

    seq_best, best_score = try_update(seq_best, best_score, work_seq, "init", t_start)
    if seq_best.size > 10000:
        work_seq = interpolate_sequence(seq_best, min(seq_best.size, 9000))
    else:
        work_seq = seq_best.copy()

    work_score = safe_eval(work_seq)
    if not math.isfinite(work_score):
        work_score = 2.0 * work_seq.size * proxy_max(fft_conv(work_seq), None)

    # Estimate L-BFGS cost
    lb_cost = 0.15
    v0, ref0 = make_v(work_seq)
    try:
        t_e = time.monotonic()
        objgrad_v(v0, 30.0, None, ref0)
        dt = time.monotonic() - t_e
        if dt > 0.01 and math.isfinite(dt):
            lb_cost = dt
    except Exception:
        pass
    print("L-BFGS cost estimate: " + repr(lb_cost))

    def refresh_work():
        nonlocal work_seq
        if seq_best.size > 10000:
            work_seq = interpolate_sequence(seq_best, 8000)
        elif seq_best.size != work_seq.size:
            work_seq = interpolate_sequence(seq_best, min(seq_best.size, 9000))
        else:
            work_seq = seq_best.copy()

    refresh_work()

    # Phase 1: L-BFGS smoothed optimization
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if SCIPY_OK and time_left > 400:
        phase_budget = min(time_left * 0.42, 700.0)
        print("\n=== Phase 1: L-BFGS (budget=" + repr(phase_budget) + "s) ===")
        a_lb = lbfegs_phase(work_seq, None, phase_budget, lb_cost)
        seq_best, best_score = try_update(seq_best, best_score, a_lb, "lbfegs", t_start)
        refresh_work()

    # Phase 2: Frank-Wolfe with exact LP
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 300:
        phase_budget = min(time_left * 0.30, 480.0)
        print("\n=== Phase 2: Frank-Wolfe (budget=" + repr(phase_budget) + "s) ===")
        a_fw = frank_wolfe_search(work_seq, phase_budget, None,
                                  active_max=3000, game_iters=28,
                                  exact_every=3, exact_time=10.0, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_fw, "fw", t_start)
        refresh_work()

    # Phase 3: Coordinate polish
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 200:
        phase_budget = min(time_left * 0.22, 350.0)
        print("\n=== Phase 3: Coordinate polish (budget=" + repr(phase_budget) + "s) ===")
        a_cp = coordinate_polish(work_seq, phase_budget, None, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_cp, "coord", t_start)
        refresh_work()

    # Phase 4: Second round FW + coord
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 200:
        print("\n=== Phase 4: Refinement (budget) ===")
        fw_b = min(time_left * 0.35, 200.0)
        a_fw2 = frank_wolfe_search(work_seq, fw_b, None,
                                   active_max=2500, game_iters=22,
                                   exact_every=4, exact_time=8.0, verbose=False)
        seq_best, best_score = try_update(seq_best, best_score, a_fw2, "fw2", t_start)
        refresh_work()
        t0 = time.monotonic()
        time_left = total_budget - (t0 - t_start)
        if time_left > 100:
            cp_b = min(time_left * 0.5, 150.0)
            a_cp2 = coordinate_polish(work_seq, cp_b, None, verbose=False)
            seq_best, best_score = try_update(seq_best, best_score, a_cp2, "coord2", t_start)
            refresh_work()

    # Phase 5: Length probes
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 250:
        print("\n=== Phase 5: Length probes ===")
        base_len = seq_best.size
        if base_len > 10000:
            base_len = 8000
        candidates = []
        for delta in [200, 400, -200, -400, 600, -600]:
            tl = base_len + delta
            if 2000 <= tl <= 10000 and tl != base_len:
                candidates.append(tl)
        tried = set()
        for target in candidates:
            time_left = total_budget - (time.monotonic() - t_start)
            if time_left < 120:
                break
            target = int(target)
            if target in tried:
                continue
            tried.add(target)
            print("  Trying n=" + str(target))
            a0 = interpolate_sequence(work_seq, target)
            if SCIPY_OK and time_left > 160:
                per = int(max(80, min(300, int(time_left / max(4.0 * lb_cost, 0.1)))))
                a_new = run_lbfegs(a0, 100.0, None, maxfun=per, verbose=False)
                seq_best, best_score = try_update(seq_best, best_score, a_new, "len_lb_" + str(target), t_start)
                if time_left > 80:
                    a_fw3 = frank_wolfe_search(seq_best, min(time_left * 0.3, 60.0), None,
                                               active_max=1500, game_iters=15,
                                               exact_every=4, exact_time=5.0, verbose=False)
                    seq_best, best_score = try_update(seq_best, best_score, a_fw3, "len_fw_" + str(target), t_start)
                    refresh_work()

    # Phase 6: Final polish
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 60:
        phase_budget = min(time_left * 0.7, 120.0)
        print("\n=== Phase 6: Final polish (budget=" + repr(phase_budget) + "s) ===")
        a_final = coordinate_polish(work_seq, phase_budget, None, verbose=False)
        seq_best, best_score = try_update(seq_best, best_score, a_final, "final", t_start)
        refresh_work()

    # Final validation
    final_val = safe_eval(seq_best)
    if math.isfinite(final_val):
        if final_val < best_score:
            best_score = final_val
    else:
        seq_best = default_sequence(5000)
        best_score = safe_eval(seq_best)

    arr = np.asarray(seq_best, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if arr.size == 0 or s < 0.01:
        arr = default_sequence(5000)
    else:
        arr = arr / s

    elapsed = time.monotonic() - t_start
    print("\nFINAL: score=" + repr(best_score) + " n=" + str(arr.size) + " elapsed=" + repr(elapsed))
    return [float(x) for x in arr]
