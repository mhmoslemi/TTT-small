    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def flat_conv_seq(n):
    i = np.arange(n, dtype=np.float64)
    log_a = np.zeros(n, dtype=np.float64)
    for k in range(1, n):
        log_a[k] = log_a[k-1] + np.log(2*k-1) - np.log(2.0*k)
    a = np.exp(log_a)
    return a / a.sum()

def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fft_gradient_smooth(a, conv, n_top, temperature):
    n = len(a)
    size = 2 * n - 1
    top_idx = np.argsort(conv)[::-1][:n_top]
    top_vals = conv[top_idx]
    weights = np.exp(temperature * (top_vals - top_vals[0]))
    weights = weights / (weights.sum() + 1e-30)
    w_padded = np.zeros(size, dtype=np.float64)
    for i in range(len(top_idx)):
        w_padded[int(top_idx[i])] = weights[i]
    w_rev = w_padded[::-1]
    req_size = 3 * n - 2
    fft_size = 1 << (req_size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    grad = conv_result[2*n-2 : n-2 : -1]
    return grad * 2.0

def subgradient_descent(a0, time_limit, n_top=200, temp0=50.0, lr0=0.02, verbose=True):
    n = len(a0)
    x = normalize(a0).copy()
    best_a = x.copy()
    best_val = 2.0 * n * fast_autocorr(x).max()
    t0 = time.monotonic()
    y = x.copy()
    t_cnt = 0
    momentum = 0.9

    for t in range(1, 10000000):
        t_cnt = t
        if time.monotonic() - t0 > time_limit:
            break
        sr = t / (t + 1.0)
        y_new = (1.0 - sr) * x + sr * y
        ac = fast_autocorr(y_new)
        M = ac.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val - 1e-13:
            best_val = cur_val
            best_a = y_new.copy()
        temp = temp0 * (1.0 + 0.1 * (t / 500.0))
        grad = fft_gradient_smooth(y_new, ac, n_top, temp)
        lr = lr0 / (1.0 + 0.01 * t)
        x_new = project_simplex(y_new - lr * grad)
        y = x_new
        if verbose and t % 2000 == 0:
            print("    SubGrad t={} ({:.1f}s) J={:.7f}".format(t, time.monotonic() - t0, best_val))

    elapsed = time.monotonic() - t0
    if verbose:
        print("    SubGrad done t={} ({:.1f}s) J={:.7f}".format(t_cnt, elapsed, best_val))
    return best_a, best_val
