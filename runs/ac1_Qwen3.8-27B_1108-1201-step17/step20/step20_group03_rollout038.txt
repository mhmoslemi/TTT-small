Let me also think about the **spectral peak shaving** approach more carefully:

Given a (self-convolution), we want to reduce its peak. In the frequency domain:
- A = DFT(a)
- |A|^2 = DFT(a*a) (with appropriate scaling for the full convolution)

To reduce the peak of a*a at position k*, we want to reduce the contribution of the frequency components that construct that peak. The peak at k* is:
(a*a)[k*] = (1/N) * sum_m |A[m]|^2 * exp(2πi*m*k*/N)

where N is the FFT size. To reduce this, we can reduce |A[m]| for the frequencies m where exp(2πi*m*k*/N) has positive real part (i.e., where the contribution is constructive at position k*).

But we also need to maintain a >= 0, which constrains the possible A values.

A simpler approach: iteratively do:
1. Compute c = a*a
2. Find the peak position k*
3. Compute the "gradient" of c[k*] with respect to a: g = 2 * (reverse of a, shifted by k*)
4. Update: a_new = a - eta * g (projected onto simplex)
5. Repeat

This is essentially a gradient descent on the single constraint c[k*] <= max. It's similar to what FISTA does but focused on the single peak.

2. Compute p_k = softmax(beta * c) for all k

4. Update: a_new = a - eta * g (projected onto simplex)

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_grad_smoothed(y, beta, top_k=None):
    """Compute smoothed gradient of max(a*a) using softmax weighting."""
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and 0 < top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max_local = b[idx].max()
    else:
        b_use = b
        b_max_local = b.max()
    shifted = beta * (b_use - b_max_local)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b_use)
        mask = (b_use == b_max_local)
        p[mask] = 1.0 / max(1, mask.sum())
    else:
        p = exp_vals / p_total
    # Gradient: sum_k p[k] * 2 * circshift(reverse(y), k)
    # For full convolution, grad_a[j] = 2 * sum_k p[k] * a[k-j] (with valid bounds)
    # Use FFT: grad = 2 * ifft(fft(p_rev) * fft(y)) restricted to [0, n)
    p_full = np.zeros(2 * n - 1)
    p_full[:len(p)] = p
    p_rev = p_full[::-1]
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    Yf = np.fft.rfft(y, fft_size)
    Prf = np.fft.rfft(p_full, fft_size)
    # grad[j] = 2 * sum_k p[k] * y[k-j] for valid k-j
    # = 2 * (p * y_flipped) restricted
    # Actually: d(a*a)[k]/da[j] = 2*a[k-j] (if valid)
    # sum_k p[k] * 2*a[k-j] = 2 * (p * a)[j] where (p*a) is cross-correlation
    # cross-correlation: (p ⋆ a)[j] = sum_k p[k] * a[k-j]
    # = ifft(fft(p) * conj(fft(a))) ... let me just use direct computation via FFT
    # (p ⋆ a)[j] = ifft(fft(p, N) * conj(fft(a, N)))[j] for circular cross-corr
    # But we need linear, so zero-pad
    Pf = np.fft.rfft(p_full, fft_size)
    Af = np.fft.rfft(y, fft_size)
    # cross_corr[j] = sum_k p[k] * a[k-j] = ifft(fft(p) * conj(fft(a)))[j + offset]
    # Actually for linear cross-correlation with zero padding:
    xcorr = np.fft.irfft(Pf * np.conj(Af), fft_size)
    # xcorr[k] = sum_j p[j] * a[k-j] (circular, but linear due to padding)
    # We want sum_k p[k] * a[k-j] for j = 0,...,n-1
    # = xcorr shifted: the cross-correlation (p ⋆ a)[j] = sum_k p[k] a[k-j]
    # In the circular case with zero-padding, xcorr[m] = sum_k p[k] a[(m-k) mod N]
    # For linear: we need (p ⋆ a)[j] = sum_{k} p[k] a[k-j] where 0<=k-j<n
    # This equals xcorr[j + (len(p_full) - 1)] ... let me just verify with small example
    # Actually, let me use a simpler approach:
    # grad[j] = 2 * sum_k p[k] * a[k-j] (where 0 <= k-j < n)
    # = 2 * sum_i a[i] * p[i+j] (where i = k-j, so k = i+j, 0<=i<n, 0<=i+j<len(p))
    # This is a cross-correlation of a and p, evaluated at position j
    # cross_corr_ap[j] = sum_i a[i] * p[i+j]
    # Using FFT: cross_corr = ifft(conj(fft(a)) * fft(p)) ... 
    # Let me just compute it directly using the convolution approach
    # (a * p_rev)[j + len(p)-1 - n + 1] ... this is getting confusing
    # Simplest: use the fact that grad = 2 * (a convolved with reversed p) at specific positions
    p_rev = p_full[::-1]
    conv = np.fft.irfft(np.fft.rfft(y, fft_size) * np.fft.rfft(p_rev, fft_size), fft_size)
    # conv[m] = sum_i y[i] * p_rev[m-i]
    # p_rev[k] = p_full[len(p_full)-1-k]
    # conv[m] = sum_i y[i] * p_full[len(p_full)-1-m+i]
    # Let j = m - n + 1 (so m = j + n - 1):
    # conv[j+n-1] = sum_i y[i] * p_full[len(p_full)-1-j-n+1+i] = sum_i y[i] * p_full[2n-2-1-j-n+1+i]
    # = sum_i y[i] * p_full[n-2-j+i] ... hmm this is getting messy
    # Let me just use a simple approach: compute the gradient as 2 * cross_correlation
    # The gradient of (a*a)[k] w.r.t. a is: g_k where g_k[j] = 2*a[k-j] if 0<=k-j<n else 0
    # Total gradient = sum_k p[k] * g_k
    # grad[j] = 2 * sum_k p[k] * a[k-j] (for valid k-j)
    # = 2 * sum_i a[i] * p[i+j] (substituting i = k-j, k = i+j)
    # This is the cross-correlation of a and p at lag j

def fista_optimize(a0, time_limit, beta_start=20.0, beta_end=1000.0,
                   beta_growth=1.004, top_k=20, restart_every=3000,
                   verbose=True, seed=42):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    avg_x = x.copy()
    avg_count = 0.0
    s_prev = None
    y_prev = None
    bb_step = 1.0 / np.sqrt(n)

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, b = compute_grad_smoothed(y, beta, top_k)

        # Barzilai-Borwein step size
        if s_prev is not None and y_prev is not None:
            sy = y - y_prev
            gy = grad - s_prev
            sy_gy = sy.dot(gy)
            gy_gy = gy.dot(gy)
            if gy_gy > 1e-20:
                bb_candidate = abs(sy_gy) / gy_gy
                if bb_candidate > 1e-8 and bb_candidate < 20.0:
                    bb_step = 0.7 * bb_step + 0.3 * bb_candidate
                else:
                    bb_step = max(bb_step * 0.95, 1e-6)
            s_prev = grad.copy()
            y_prev = y.copy()

        eta = min(bb_step, 1.0 / beta * 2.0)
        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        avg_count += 1.0
        avg_x += (x_new - avg_x) / avg_count

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.02 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            s_r = x_r.sum()
            if s_r > 1e-15:
                x_r = x_r / s_r
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 10.0)
            s_prev = None
            y_prev = None
            bb_step = 1.0 / np.sqrt(n)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    FISTA restart t={} ({}s): J={:.6f}".format(t, round(elapsed, 1), J))

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    FISTA t={} ({}s): J={:.6f} beta={:.1f}".format(t, round(elapsed, 1), J, beta))

    # Check Nesterov average
    if avg_count > 100:
        J_avg = 2.0 * n * fast_autocorr(avg_x).max()
        if J_avg < 2.0 * n * best_max_b:
            return avg_x, fast_autocorr(avg_x).max()
    return best_a, best_max_b


def spectral_peak_shave(a0, time_limit, n_iters=None, verbose=True):
    """Iteratively shave peaks in the autocorrelation domain."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    if n_iters is None:
        n_iters = 50000

    fft_size = 1 << (2 * n - 1).bit_length()
    step_size = 0.01
    decay = 0.9995

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        b_max = b.max()
        peak_idx = np.where(b >= b_max * 0.999)[0]

        if len(peak_idx) == 0:
            break

        # Compute gradient for the top peaks
        n_peaks = min(len(peak_idx), 10)
        top_peaks = peak_idx[np.argsort(b[peak_idx])[::-1]][:n_peaks]

        grad = np.zeros(n)
        for k in top_peaks:
            k_int = int(k)
            # d(a*a)[k]/da[j] = 2*a[k-j] for valid j
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js <= je:
                idx = np.arange(js, je + 1)
                grad[idx] += 2.0 * a[k_int - idx]

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Projected gradient step
        a_new = a - step_size * grad
        a_new = simplex_project(a_new)

        b_new = fast_autocorr(a_new)
        m_new = b_new.max()

        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_try = a - step_size * 0.3 * grad
            a_try = simplex_project(a_try)
            b_try = fast_autocorr(a_try)
            if b_try.max() < best_m:
                a = a_try
                best_m = b_try.max()
                best_a = a.try if hasattr(a, 'try') else a
                best_a = a.copy()
            else:
                # Random perturbation
                if it % 100 == 0:
                    pert_pos = rng.integers(n, size=min(5, n))
                    for pp in pert_pos:
                        amt = a[pp] * rng.uniform(0.001, 0.01)
                        if amt < 1e-15:
                            continue
                        dst = int(rng.integers(n))
                        a[pp] -= amt
                        a[dst] += amt
                    a = np.maximum(a, 0.0)
                    s_a = a.sum()
                    if s_a > 1e-15:
                        a = a / s_a
                        m_check = fast_autocorr(a).max()
                        if m_check < best_m:
                            best_m = m_check
                            best_a = a.copy()

        step_size *= decay

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral it={} ({}s): J={:.6f} step={:.5f}".format(it, round(elapsed, 1), J, step_size))

    return best_a, best_m

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations and add to active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-6)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Adaptive step size with backtracking
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.01, default_alpha * 0.5),
                  min(1.0, default_alpha * 2.0), 1.0 / (t + 1.0), 0.5]
        alphas = list(set([max(0.01, min(1.0, a)) for a in alphas]))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter={} ({}s): J={:.6f} |S|={}".format(t, round(elapsed, 1), J, len(active_set)))

    return best_a, best_m


def local_search_refined(a0, time_limit, verbose=True):
    """Systematic local search: 1D line search along gradient + 2-exchange."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if stall > 300:
            break

        ac = fast_autocorr(a)
        b_max = ac.max()
        peak_idx = np.where(ac >= b_max * 0.995)[0]
        if len(peak_idx) == 0:
            peak_idx = np.argsort(ac)[::-1][:20]
        top_peaks = peak_idx[np.argsort(ac[peak_idx])[::-1]][:min(20, len(peak_idx))]

        improved = False

        # Strategy 1: Move mass away from peak-contributing pairs
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(top_peaks[int(rng.integers(len(top_peaks)))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            total = contrib.sum()
            if total < 1e-15:
                continue
            # Find the pair contributing most
            src = int(idx[np.argmax(contrib)])
            partner = k_int - src

            # Find destination: a position that doesn't contribute to top peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            dst = int(candidates[int(rng.integers(len(candidates)))])
            if dst == src:
                continue
            if a[src] < 1e-12:
                continue

            # Try multiple fractions
            best_trial_m = best_m
            best_frac = 0
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_trial_m:
                    best_trial_m = M_trial
                    best_frac = frac

            if best_frac > 0 and best_trial_m < best_m - 1e-15:
                amt = a[src] * best_frac
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                best_m = best_trial_m
                best_a = a.copy()
                improved = True
                ac = fast_autocorr(a)
                b_max = ac.max()
                peak_idx = np.where(ac >= b_max * 0.995)[0]
                top_peaks = peak_idx[np.argsort(ac[peak_idx])[::-1]][:min(20, len(peak_idx))]

        # Strategy 2: Smooth 1D moves (redistribute along a segment)
        if not improved:
            for _ in range(10):
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(rng.integers(n - 2))
                j = i + 2 + int(rng.integers(max(1, n - i - 3)))
                if j >= n:
                    j = n - 1
                # Try to smooth a[i:j+1]
                segment = a[i:j + 1].copy()
                seg_sum = segment.sum()
                if seg_sum < 1e-15:
                    continue
                # Move towards uniform on segment
                alpha_smooth = rng.uniform(0.05, 0.3)
                uniform_val = seg_sum / (j - i + 1)
                segment_new = (1 - alpha_smooth) * segment + alpha_smooth * uniform_val
                a_trial = a.copy()
                a_trial[i:j + 1] = segment_new
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True

        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = min(10, n)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.001, 0.02)
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Local ({}s): J={:.6f} stall={}".format(round(elapsed, 1), J, stall))

    return best_a, best_m


def cyclic_shift_best(a0):
    """Try all cyclic shifts and return the best one."""
    n = len(a0)
    if n <= 1:
        return a0.copy(), fast_autocorr(a0).max()
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m - 1e-15:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def gradient_line_search(a0, time_limit, verbose=True):
    """Do 1D line searches along the negative gradient direction."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(7)

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()
        peak_idx = np.argwhere(b >= b_max * 0.998)[:, 0]
        if len(peak_idx) == 0:
            peak_idx = np.argsort(b)[::-1][:5]

        # Compute gradient at the top peaks
        top = peak_idx[np.argsort(b[peak_idx])[::-1]][:min(5, len(peak_idx))]
        grad = np.zeros(n)
        for k in top:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js <= je:
                idx = np.arange(js, je + 1)
                grad[idx] += a[k_int - idx]

        if np.abs(grad).max() < 1e-15:
            break

        # Line search: a(t) = project(a - t * grad / |grad|^2, simplex)
        g_norm_sq = grad.dot(grad)
        if g_norm_sq < 1e-20:
            break
        direction = grad / g_norm_sq

        # Try several step sizes
        best_step = 0
        best_trial_m = best_m
        for step in [1e-5, 5e-5, 1e-4, 5e-4, 1e-3, 5e-3, 0.01]:
            a_trial = a - step * direction
            a_trial = simplex_project(a_trial)
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_trial_m:
                best_trial_m = M_trial
                best_step = step

        if best_step > 0:
            a = a - best_step * direction
            a = simplex_project(a)
            M_new = fast_autocorr(a).max()
            if M_new < best_m - 1e-15:
                best_m = M_new
                best_a = a.copy()
            else:
                a = best_a.copy()
                break
        else:
            # Stuck, try random perturbation
            if rng.random() < 0.5:
                break
            perm = rng.permutation(n)
            for _ in range(min(5, n)):
                src = int(perm[_])
                dst = int(perm[_ + 1] % n)
                if src != dst and a[src] > 1e-10:
                    amt = a[src] * 0.01
                    a[src] -= amt
                    a[dst] += amt
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a
    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {:.6f} (n={}, t={:.1f}s)".format(label, sc, len(a), elapsed))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Time budget: {:.1f}s, solver: {}".format(time_budget, solver))

    best_seq = None
    best_score = float('inf')

    # Try to use parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n={}, score={:.6f}".format(len(parent), p_score))
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n={}, score={:.6f}".format(len(best_seq), best_score))

    n0 = len(best_seq)

    # === Phase 1: FISTA (main optimizer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: FISTA (n={}) ===".format(n0))
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 60:
        a_out, _ = fista_optimize(norm_sum1(best_seq), t1,
                                  beta_start=15.0, beta_end=800.0,
                                  beta_growth=1.003, top_k=15,
                                  restart_every=3000, verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista1", t_global)

    # === Phase 2: Cyclic shift ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: Cyclic shift ===")
    if remaining > 30:
        a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "shift1", t_global)

    # === Phase 3: Spectral peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: Spectral peak shaving (n={}) ===".format(len(best_seq)))
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_out, _ = spectral_peak_shave(norm_sum1(best_seq), t3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spectral1", t_global)

    # === Phase 4: LP-Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW (n={}) ===".format(n_best))
    t4 = min(remaining * 0.18, 280.0)
    if t4 > 40 and n_best <= 10000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t4, solver,
                         initial_K=400, max_K=3000, add_per_iter=25, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # === Phase 5: Gradient line search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 5: Gradient line search (n={}) ===".format(len(best_seq)))
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_out, _ = gradient_line_search(norm_sum1(best_seq), t5, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "gls1", t_global)

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-length exploration ===")
    longer_lengths = [n_best + 500, n_best + 1000, n_best + 2000]
    longer_lengths = [l for l in longer_lengths if l <= 12000 and l > n_best]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("  Trying n={}...".format(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.25, 180.0)
        if t6a > 40:
            a_out, _ = fista_optimize(a_long, t6a,
                                      beta_start=20.0, beta_end=600.0,
                                      beta_growth=1.004, top_k=12,
                                      restart_every=2500, verbose=True, seed=123)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_n{}".format(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
            best_seq, best_score = try_update(best_seq, best_score, a_shifted, "shift_n{}".format(new_n), t_global)

    # === Phase 7: Second FISTA with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: FISTA v2 (n={}) ===".format(n_best))
    t7 = min(remaining * 0.15, 200.0)
    if t7 > 40:
        a_out, _ = fista_optimize(norm_sum1(best_seq), t7,
                                  beta_start=50.0, beta_end=1500.0,
                                  beta_growth=1.006, top_k=10,
                                  restart_every=2000, verbose=True, seed=777)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista2", t_global)

    # === Phase 8: Local search polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 8: Local search (n={}) ===".format(len(best_seq)))
    t8 = min(remaining * 0.15, 180.0)
    if t8 > 20:
        a_out, _ = local_search_refined(norm_sum1(best_seq), t8, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "local1", t_global)

    # === Phase 9: Final LP burst ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 9: Final LP (n={}) ===".format(n_best))
    if remaining > 50 and n_best <= 10000:
        t9 = min(remaining * 0.20, 150.0)
        if t9 > 30:
            a_out, _ = lp_fw(norm_sum1(best_seq), t9, solver,
                             initial_K=500, max_K=3500, add_per_iter=30, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # === Phase 10: Final spectral + shift + local ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 10: Final polish ===")
    if remaining > 30:
        a_out, _ = spectral_peak_shave(norm_sum1(best_seq), min(remaining * 0.3, 60.0), verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec_final", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "shift_final", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        a_out, _ = local_search_refined(norm_sum1(best_seq), min(remaining * 0.5, 80.0), verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "local_final", t_global)

    # === Final: One more FISTA micro-polish with high beta ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== Final: FISTA micro-polish ===")
        t_f = min(remaining * 0.50, 100.0)
        if t_f > 20:
            a_out, _ = fista_optimize(norm_sum1(best_seq), t_f,
                                      beta_start=200.0, beta_end=3000.0,
                                      beta_growth=1.01, top_k=8,
                                      restart_every=1500, verbose=True, seed=2024)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.6f}, n={}, time={:.1f}s".format(best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
