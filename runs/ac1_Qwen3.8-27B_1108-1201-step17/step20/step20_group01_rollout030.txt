4. Coordinate peeling for fine-tuning

import numpy as np
import time
import cvxpy as cp

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

    ac = fast_autocorr(a)

    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def mirror_descent(a0, time_limit, n_peaks=50, lr0=1.0, verbose=True):

    a = normalize(a0.copy())
    best_a = a.copy()
    best_obj = obj_val(a)
    t0 = time.monotonic()
    stall = 0

        if time.monotonic() - t0 > time_limit:
            break
        if stall > 80000:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        cur_obj = 2.0 * n * M_curr

            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        top_vals = ac[top_idx]

        w = np.exp(-temp * (top_vals - top_vals[0]) / (top_vals[0] + 1e-15))
        w = w / w.sum()

a[j] *= max(exp(-lr * SG[j] / sg_max), floor)

Alternatively, use a "soft" update:
a[j] *= exp(-lr * (SG[j] - SG.mean()) / sg_max)

This centers the gradient so that the update doesn't change the "overall scale" (before normalization). This is similar to using the natural gradient on the simplex.

a[j] *= exp(-lr * (SG[j] - SG.mean()) / sg_max)

This is a better update because it maintains the "shape" of a more naturally.

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

def eval_safe(a):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def get_solver():
    for s in [cp.ECOS, cp.SCS, cp.PDLP]:

            prob.solve(solver=s, verbose=False)

    t0 = time.monotonic()
    stall = 0
    iters = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

            break

        ac = fast_autocorr(a)
