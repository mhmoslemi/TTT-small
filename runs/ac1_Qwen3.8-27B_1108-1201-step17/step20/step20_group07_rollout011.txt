def fast_autocorr(a):
    """Full convolution autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project z onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def irls_exponentiated(a0, time_limit, beta_start=5.0, beta_end=500.0,
                       beta_growth=1.02, inner_steps=15, eta_base=0.3, verbose=True):
    """IRLS with exponentiated gradient for spectral equalization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    beta = beta_start
    rng = np.random.default_rng(99)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        b = fast_autocorr(a)
        m_curr = b.max()
        
        if m_curr < best_m - 1e-15:
            best_m = m_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Softmax weights on autocorrelation
        shifted = np.clip(beta * (b - m_curr), -400, 400)
        exp_vals = np.exp(shifted)
        z = exp_vals.sum()
        if z < 1e-30:
            w = np.zeros_like(b)
            w[np.argmin(b)] = 1.0
        else:
            w = exp_vals / z
        
        # Exponentiated gradient inner loop
        for inner in range(inner_steps):
            if time.monotonic() - t0 > time_limit:
                break
            # Compute pressure p[i] = sum_k w[k] * (a[k-i] + a[i])
            # = cross_corr(w, a)[i] + a[i] * sum(w[i:i+n])
            fft_size = 1 << (len(w) + n - 1).bit_length()
            Wf = np.fft.rfft(w, fft_size)
            Af = np.fft.rfft(a, fft_size)
            # Cross-correlation: sum_k w[k] * a[k-i]
            corr = np.fft.irfft(Wf * np.conj(Af), fft_size)[:len(w) + n - 1]
            # Shift to get d[i] = sum_k w[k] * a[k-i] for i=0..n-1
            # corr[m] = sum_k w[k] * a[k-m], we want d[i] = corr[i] for i in 0..n-1
            # But corr is offset: corr[i] = sum_k w[k]*a[k-i]
            # We need indices 0..n-1 of this
            d = corr[n - 1:2 * n - 1]  # shift by n-1 to center
            
            # Windowed sum of w for the second term
            w_cumsum = np.concatenate([[0], np.cumsum(w)])
            w_window = w_cumsum[np.minimum(n, np.arange(n) + n)] - w_cumsum[np.arange(n)]
            # Actually: sum_{m=0}^{n-1} w[i+m] for i=0..n-1
            w_window = np.array([w[i:i+n].sum() for i in range(n)]) if n < 2000 else \
                       (w_cumsum[min(2*n-2, np.arange(n) + n - 1)] - w_cumsum[np.arange(n)])
            
            p = d + a * w_window
            p = np.maximum(p, 0)
            
            # Adaptive step size
            eta = eta_base / (np.max(p) + 1e-10)
            eta = min(eta, 0.5)
            
            # Exponentiated gradient update
            log_a = np.log(np.maximum(a, 1e-300))
            log_a -= eta * p
            log_a -= log_a.max()  # numerical stability
            a_new = np.exp(log_a)
            a_new = a_new / a_new.sum()
            a = a_new
        
        # Update beta
        beta = min(beta * beta_growth, beta_end)
        
        if stall > 50 and stall % 30 == 0:
            # Perturbation to escape
            noise = 0.01
            pert = rng.normal(0, noise, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-12)
            a_r = a_r / a_r.sum()
            m_r = fast_autocorr(a_r).max()
            if m_r < best_m:
                best_m = m_r
                best_a = a_r.copy()
                stall = 0
            a = a_r
            beta = max(beta * 0.5, beta_start)
        
        if verbose and (stall == 0 or stall % 50 == 0):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    IRLS stall={}: J={} (t={:.1f}s, beta={:.1f})".format(stall, repr(J), elapsed, beta))
    
    return best_a, best_m


def peak_shaving(a0, time_limit, top_k=20, verbose=True):
    """Multi-scale peak shaving with random restarts."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit and stall < 600:
        b = fast_autocorr(a)
        n_peaks = min(top_k, len(b))
        peak_indices = np.argsort(b)[::-1][:n_peaks]
        
        # Compute pressure
        pressure = np.zeros(n)
        for k_int in peak_indices[:10]:
            k_i = int(k_int)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                pressure[idx] += a[k_i - idx]
                if k_i - idx != idx:
                    pressure[k_i - idx] += a[idx]
        
        dangerous = np.zeros(n, dtype=bool)
        for k_int in peak_indices[:8]:
            k_i = int(k_int)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            if i_start <= i_end:
                dangerous[i_start:i_end + 1] = True

            if time.monotonic() - t0 > time_limit:
                break
            k_i = int(k_int)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_i - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = int(idx[np.argmax(contrib)])
            partner = k_i - src
            if partner < 0 or partner >= n:
                continue
            reduce_idx = src if a[src] >= a[partner] else partner
            if a[reduce_idx] < 1e-14:
                continue
            
            safe_mask = ~dangerous
            if safe_mask.sum() < n // 4:
                safe_mask = pressure < np.percentile(pressure, 60)
            if safe_mask.sum() < 20:
                safe_mask = np.ones(n, dtype=bool)
            safe_mask[reduce_idx] = False
            
            safe_pressure = pressure.copy()
            safe_pressure[~safe_mask] = np.max(pressure) + 1.0
            
            dst = int(np.argmin(safe_pressure))
            if not safe_mask[dst]:
                continue
            
            found = False
            for frac in [0.003, 0.008, 0.02, 0.04, 0.08, 0.15, 0.3]:
                amt = a[reduce_idx] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if not found and stall > 100:
                j = int(rng.integers(n))
                if j != reduce_idx and a[reduce_idx] > 1e-12:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[reduce_idx] * frac
                    a[reduce_idx] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        m_new = fast_autocorr(a).max()
                        if m_new < best_m:
                            best_m = m_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if not improved:
            stall += 1
            if stall % 60 == 0:
                noise_scale = min(0.05, 0.003 * stall / 60)

                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-12)
                a_r = a_r / a_r.sum()
                m_r = fast_autocorr(a_r).max()
                if m_r < best_m:
                    best_m = m_r
                    best_a = a_r.copy()
                a = a_r if m_r < best_m + 0.005 else best_a.copy()
            
            if verbose and stall % 150 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    PeakShave stall={}: J={} (t={:.1f}s)".format(stall, repr(J), elapsed))
    
    return best_a, best_m


def frank_wolfe_lp(a0, time_limit, solver, verbose=True):
    """Frank-Wolfe with LP inner solver and adaptive active set."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    b = fast_autocorr(a)
    active_set = list(np.argsort(b)[::-1][:800])
    no_improve = 0
    t = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 600:
        t += 1
        b = fast_autocorr(a)
        M_curr = b.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update active set
        top_k_active = min(2000, max(300, int(300 + t * 50)))
        active_set = list(np.argsort(b)[::-1][:top_k_active])
        
        K = len(active_set)
        if K == 0:
            break
        
        # Build constraint matrix C where C[jj, idx] = a[k_int - idx]
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(active_set):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = a[k_int - idx]
        
        # LP: minimize M subject to sum(g)=1, g>=0, C@g <= M
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            no_improve += 10
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 5
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check violations and grow active set
        fft_size = 1 << (2 * n - 1).bit_length()
        Af = np.fft.rfft(a, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ag = np.fft.irfft(Af * Gf, fft_size)[:2 * n - 1]
        
        violations = np.where(conv_ag > M_curr + 1e-5)[0]
        if len(violations) > 0:
            viol_amt = conv_ag[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                if int(k) not in active_set:
                    active_set.append(int(k))
        
        # Line search with backtracking
        best_alpha = 1.0 / (t + 1.0)
        best_trial_M = fast_autocorr(a).max()
        alphas = [min(1.0, 2.0 / (t + 2.0)), 1.0 / (t + 1.0), 0.5 / (t + 1.0), 0.1 / (t + 1.0)]
        for alpha in alphas:
            trial = (1.0 - alpha) * a + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        a_new = (1.0 - best_alpha) * a + best_alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        st = a_new.sum()
        if st > 1e-15:
            a_new = a_new / st
        a = a_new
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    FW iter={}: J={} |S|={} (t={:.1f}s)".format(t, repr(J), len(active_set), elapsed))
    
    return best_a, best_m


def spectral_init(n, n_gaussians=3):
    """Generate initial sequence using spectral considerations."""
    x = np.linspace(0.0, 1.0, n)
    rng = np.random.default_rng(42)
    
    # Try several parametric families
    candidates = []
    
    # Power law asymmetric
    for alpha in [0.5, 0.8, 1.0, 1.5, 2.0]:
        for beta_p in [0.5, 0.8, 1.0, 1.5, 2.0]:
            a_raw = (x + 0.01) ** alpha * (1.0 - x + 0.01) ** beta_p
            candidates.append(a_raw)
    
    # Skewed gaussian
    for c in [0.2, 0.3, 0.4, 0.5]:
        for s in [0.1, 0.15, 0.2]:
            candidates.append(np.exp(-((x - c) ** 2) / (2 * s ** 2)))
    
    # Two-component
    for c1, c2 in [(0.2, 0.7), (0.3, 0.6), (0.25, 0.65)]:
        for s in [0.08, 0.12, 0.15]:
            candidates.append(np.exp(-((x - c1) ** 2) / (2 * s ** 2)) + 0.5 * np.exp(-((x - c2) ** 2) / (2 * s ** 2)))
    
    best_a = None
    best_m = float('inf')
    for a_raw in candidates:
        a = norm_sum1(a_raw)
        m = fast_autocorr(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()
    
    return best_a, best_m

        remaining = time_budget - (time.monotonic() - t_global)
        t5b = min(remaining * 0.3, 150.0)
        if t5b > 40 and new_n <= 8000:
            a_out, _ = frank_wolfe_lp(best_seq, t5b, solver, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fw_n{}".format(new_n), t_global)
    
    # Phase 6: Final IRLS polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Final IRLS polish (n={}) ===".format(n_best))
    if remaining > 80:
        t6 = min(remaining * 0.3, 200.0)
        a_out, _ = irls_exponentiated(best_seq, t6, beta_start=20.0, beta_end=1000.0,
                                       beta_growth=1.05, inner_steps=20, eta_base=0.2, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "irls2_n{}".format(n_best), t_global)
    
    # Phase 7: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final peak shaving (n={}) ===".format(n_best))
    if remaining > 40:
        t7 = min(remaining * 0.4, 150.0)
        a_out, _ = peak_shaving(best_seq, t7, top_k=10, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps_final", t_global)
    
    # Phase 8: Final LP burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60 and n_best <= 8000:
        print("\n=== Phase 8: Final LP burst (n={}) ===".format(n_best))
        t8 = min(remaining * 0.8, 200.0)
        a_out, _ = frank_wolfe_lp(best_seq, t8, solver, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.6f}, n={}, time={:.1f}s".format(best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence found!")
        x = np.linspace(0, 1, n_base)
        a_raw = (x + 0.01) ** 1.0 * (1.0 - x + 0.01) ** 1.0
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)
    
    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
