import time
import math

    SCIPY_OK = True

    SCIPY_OK = False

_hi_available = None

def check_hi():
    global _hi_available
    if _hi_available is None:
        _hi_available = False
        if SCIPY_OK:
            try:
                r = scipy.optimize.linprog([-1.0], bounds=[(0.0, None)], method="HiGHS", options={"disp": False})
                _hi_available = bool(r.success)
            except Exception:
                _hi_available = False
    return _hi_available


def npow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm1(a):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.ndim != 1:
        a = a.ravel()
    n = a.size
    if n == 0:
        return np.ones(1)
    a[~np.isfinite(a)] = 0.0
    np.maximum(a, 0.0, out=a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def fconv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([])
    sz = 2 * n - 1
    N = npow2(sz)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:sz]
    return np.where(np.isfinite(c) & (c > 0), c, 0.0)


def fconv2(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([])
    sz = 2 * n - 1
    N = npow2(sz)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:sz]
    return np.where(np.isfinite(c) & (c > 0), c, 0.0)


def px_max(c, mask=None):
    if c.size == 0:
        return 0.0
    v = c if mask is None else c[mask]
    v = v[np.isfinite(v)]
    if v.size == 0:
        return 0.0
    return float(np.max(v))


def top_k(c, mask, k):
    sz = c.size
    k = min(int(k), sz)
    if k <= 0:
        return np.array([], dtype=int)
    v = c.copy()
    if mask is not None:
        v[~mask] = -1e18
    return np.argpartition(v, -k)[-k:] if k < sz else np.argsort(v)[::-1][:k]


def top_mask(c, base_mask, cnt):
    if cnt <= 0:
        return base_mask
    m = np.zeros(c.size, dtype=bool)
    idx = top_k(c, base_mask, cnt)
    if idx.size > 0:
        m[idx] = True
    if base_mask is not None:
        m[~base_mask] = False
    return m


def safe_eval(a):
    try:
        arr = np.asarray(a, dtype=np.float64)
        if arr.size == 0:
            return float('inf')
        arr = np.clip(arr, 0.0, 1000.0)
        if not np.all(np.isfinite(arr)):
            return float('inf')
        lst = [float(x) for x in arr]
        v = evaluate_sequence(lst)
        v = float(v)
        return v if math.isfinite(v) else float('inf')
    except Exception:
        return float('inf')


def interpl(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return norm1(a)
    ox = np.linspace(0, 1, a.size)
    nx = np.linspace(0, 1, new_n)
    y = np.interp(nx, ox, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0, out=y)
    return norm1(y)


def lse_obj_grad(a, beta, mask=None):
    n = a.size
    c = fconv(a)
    cv = c if mask is None else c[mask]
    if cv.size == 0:
        return 0.0, np.zeros(n)
    m = float(np.max(cv))
    sh = np.clip(beta * (cv - m), -700, 700)
    e = np.exp(sh)
    se = float(e.sum())
    if se <= 0 or not math.isfinite(se):
        return 1e100, np.zeros(n)
    p = e / se
    pf = np.zeros(c.size)
    if mask is None:
        pf[:] = p
    else:
        pf[mask] = p
    # Gradient: 2 * cross_convolve(pf, a) in the valid range
    N = npow2(3 * n - 2)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(pf[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:3*n-2]
    grad = 2.0 * conv[n-1:2*n-1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    f = m + math.log(se) / beta
    return f, grad


def softmax(u):
    u = np.asarray(u, dtype=np.float64)
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.ones(u.size) / u.size
    return e / s


def make_v(a):
    n = a.size
    if n <= 1:
        return np.array([]), 0
    ref = int(np.argmax(a))
    aref = max(float(a[ref]), 1e-300)
    la = np.log(np.maximum(a, 1e-300)) - math.log(aref)
    if ref == 0:
        v = la[1:]
    elif ref == n - 1:
        v = la[:-1]
    else:
        v = np.concatenate((la[:ref], la[ref+1:]))
    v = np.where(np.isfinite(v), v, -700.0)
    return v, ref


def from_v(v, ref):
    n = v.size + 1
    u = np.zeros(n)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref+1:] = v[ref:]
    return softmax(u)


def objgrad_v(v, beta, mask=None, ref=0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    if not np.all(np.isfinite(v)):
        return 1e100, np.zeros(v.size)
    ref = int(ref)
    u = np.zeros(n)
    u[ref] = 0.0
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref+1:] = v[ref:]
    a = softmax(u)
    f, ga = lse_obj_grad(a, beta, mask)
    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)
    gv = np.zeros(v.size)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref+1:]
    gv = np.where(np.isfinite(gv), gv, 0.0)
    if not math.isfinite(f):
        f = 1e100
    return f, gv


def run_lbfgs(a0, beta, mask, maxfun=300, ref=None, verbose=False):
    a = norm1(a0)
    n = a.size
    if n <= 1 or not SCIPY_OK:
        return a
    if ref is None:
        _, ref = make_v(a)
    v0, ref = make_v(a)
    if v0.size == 0:
        return a
    try:
        res = scipy.optimize.minimize(
            objgrad_v, v0, args=(beta, mask, ref),
            method='L-BFGS-B', jac=True,
            options={'maxfun': int(maxfun), 'maxiter': int(maxfun+50),
                     'ftol': 1e-14, 'gtol': 1e-9, 'disp': False}
        )
        if res is not None and hasattr(res, 'x'):
            vx = np.asarray(res.x, dtype=np.float64)
            if vx.size == v0.size and np.all(np.isfinite(vx)):
                return norm1(from_v(vx, ref))
    except Exception:
        pass
    return a


def game_dir(f, active, iters=20, seed=0, init_log=None):
    f = norm1(f)
    n = f.size
    sz = 2 * n - 1
    active = np.asarray(active, dtype=int)
    K = active.size
    if K <= 0:
        return np.ones(n) / n
    rng = np.random.default_rng(seed)
    if init_log is None:
        logits = np.zeros(K)
    else:
        logits = np.asarray(init_log, dtype=np.float64).copy()
        if logits.size != K:
            logits = np.zeros(K)
        logits = np.where(np.isfinite(logits), logits, 0.0)
        logits -= float(np.max(logits))
    logits += rng.normal(0, 0.05, K)
    
    def norm_loc(loc):
        loc = np.where(np.isfinite(loc), loc, -1000.0)
        loc -= float(np.max(loc))
        w = np.exp(loc)
        s = float(w.sum())
        if s <= 0:
            w = np.ones(K)
            s = float(K)
        return w / s
    
    lam = norm_loc(logits)
    g = np.ones(n) / n
    for _ in range(max(1, int(iters))):
        pf = np.zeros(sz)
        pf[active] = lam
        # pressure = cross_convolve(pf, f)
        N = npow2(3 * n - 2)
        A = np.fft.rfft(f, N)
        Pr = np.fft.rfft(pf[::-1], N)
        conv = np.fft.irfft(A * Pr, N)[:3*n-2]
        pressure = conv[n-1:2*n-1][::-1]
        pressure = np.where(np.isfinite(pressure) & (pressure > 0), pressure, 0.0)
        pmx = float(np.max(pressure))
        pmn = float(np.min(pressure))
        ptp = pmx - pmn
        if ptp > 1e-30:
            sc = (pressure - pmn) / ptp
            w = np.exp(-8.0 * sc)
            w = np.where(np.isfinite(w), w, 0.0)
            w += 1e-12
            g = w / w.sum()
            g = 0.97 * g + 0.03 * (np.ones(n) / n)
            g /= g.sum()
        else:
            g = np.ones(n) / n
        
        h = fconv2(f, g)
        ha = h[active]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmac = float(np.max(ha))
        hmin = float(np.min(ha))
        hptp = hmac - hmin
        if hptp > 1e-30:
            eta = 12.0 / hptp
            logits += eta * (ha - float(np.mean(ha)))
        lam = norm_loc(logits)
    return g


def lp_dir(f, active, tlim=6.0, maxk=800):
    if not SCIPY_OK or not check_hi():
        return None
    f = norm1(f)
    n = f.size
    active = np.asarray(active, dtype=int)
    if active.size == 0 or active.size > maxk:
        if active.size > maxk:
            active = active[:maxk]
        else:
            return None
    K = active.size
    rows_p, cols_p, vals_p = [], [], []
    for r, k in enumerate(active):
        k = int(k)
        s = max(0, k - n + 1)
        e = min(n, k + 1)
        if s >= e:
            continue
        cols = np.arange(s, e, dtype=np.int32)
        rows_p.append(np.full(cols.size, r, dtype=np.int32))
        cols_p.append(cols)
        vals_p.append(f[k - cols].astype(np.float64))
    if not rows_p:
        return None
    rows = np.concatenate(rows_p)
    cols = np.concatenate(cols_p)
    vals = np.concatenate(vals_p)
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    neg = scipy.sparse.csc_matrix(-np.ones((K, 1)))
    A_ub = scipy.sparse.hstack((A, neg), format='csr')
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.ones(1)
    b_ub = np.zeros(K)
    bounds = [(0.0, None)] * n + [(None, None)]
    try:
        res = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                      bounds=bounds, method='HiGHS',
                                      options={'disp': False, 'time_limit': float(tlim)})
    except Exception:
        return None
    if res is None or not getattr(res, 'success', False):
        return None
    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.where(np.isfinite(g) & (g > 0), g, 0.0)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs


def fw_search(a0, budget, active_max=2000, game_it=22, exact_ev=5, exact_t=6.0, verbose=True):
    t0 = time.monotonic()
    tl = max(1.0, float(budget))
    a = norm1(a0)
    n = a.size
    c = fconv(a)
    M = px_max(c)
    best_a = a.copy()
    best_M = M
    active = top_k(c, None, active_max)
    no_imp = 0
    t = 0
    while time.monotonic() - t0 < tl - 5 and no_imp < 80:
        t += 1
        tl_left = tl - (time.monotonic() - t0)
        if tl_left < 20:
            break
        g = None
        if check_hi() and exact_ev > 0 and t % exact_ev == 0 and tl_left > 80:
            g = lp_dir(a, active, exact_t, 800)
        if g is None:
            ilog = None
            if active.size > 0:
                av = c[active]
                av = np.where(np.isfinite(av) & (av > 0), av, 1e-300)
                ilog = np.log(av)
                ilog -= float(np.max(ilog))
            g = game_dir(a, active, game_it, 1000 + t, ilog)
        if g is None:
            break
        cg = fconv(g)
        fg = fconv2(a, g)
        # Line search
        base_als = [max(0.01, 1.0/(t+1)), max(0.02, 2.0/(t+2)), 0.04, 0.08, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0]
        als = sorted(set([float(x) for x in base_als if 0 < x <= 1.0]))
        if not als:
            als = [0.1]
        best_al = 0.0
        best_m = M
        for al in als:
            on = 1.0 - al
            val = (on*on)*c + (2*al*on)*fg + (al*al)*cg
            m = float(np.max(val))
            if math.isfinite(m) and m < best_m:
                best_m = m
                best_al = al
        if best_al > 0 and best_m < M - 1e-16:
            on = 1.0 - best_al
            a_new = on * a + best_al * g
            a_new = np.maximum(a_new, 0)
            ss = float(a_new.sum())
            if ss > 1e-15:
                a_new /= ss
                c = fconv(a_new)
                M = px_max(c)
                a = a_new
                no_imp = 0
                if M < best_M - 1e-15:
                    best_M = M
                    best_a = a.copy()
                active = top_k(c, None, active_max)
                if verbose and (t == 1 or t % 10 == 0):
                    print("  FW t={} proxy={:.6f} a={:.3f}".format(t, 2*n*best_M, best_al))
        else:
            no_imp += 1
            if no_imp % 12 == 0:
                rng = np.random.default_rng(9973 + t)
                pert = rng.normal(0, 0.02, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-12)
                a /= a.sum()
                c = fconv(a)
                M = px_max(c)
                active = top_k(c, None, active_max)
                no_imp = 0
            if verbose and (t == 1 or t % 12 == 0):
                print("  FW stall t={} best_proxy={:.6f}".format(t, 2*n*best_M))
    return best_a


def apply_move(c, a, s, d, tmove):
    c_new = c.copy()
    n = a.size
    c_new[d:d+n] += 2.0 * tmove * a
    c_new[s:s+n] -= 2.0 * tmove * a
    t2 = tmove * tmove
    if 2*d < c_new.size:
        c_new[2*d] += t2
    if 2*s < c_new.size:
        c_new[2*s] += t2
    if s + d < c_new.size:
        c_new[s+d] -= 2.0*t2
    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
    return c_new


def coord_polish(a0, budget, verbose=True):
    t0 = time.monotonic()
    tl = max(1.0, float(budget))
    a = norm1(a0)
    n = a.size
    c = fconv(a)
    M = px_max(c)
    best_a = a.copy()
    best_M = M
    stall = 0
    rnd = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < tl - 3 and stall < 200:
        rnd += 1
        if n <= 1:
            break
        # Compute pressure from top peaks
        tk = min(30, 2*n-1)
        ti = top_k(c, None, tk)
        if ti.size == 0:
            break
        pt = np.zeros(2*n-1)
        pt[ti] = 1.0 / tk
        # pressure
        N = npow2(3*n-2)
        A = np.fft.rfft(a, N)
        Pr = np.fft.rfft(pt[::-1], N)
        conv = np.fft.irfft(A * Pr, N)[:3*n-2]
        pres = conv[n-1:2*n-1][::-1]
        pres = np.where(np.isfinite(pres) & (pres > 0), pres, 0.0)
        
        # Find best (s, d, frac) move
        sc_cnt = min(10, n)
        dc_cnt = min(15, n)
        src_sc = pres * (1.0 + n * a)
        dst_sc = pres + 1e-5 * a
        
        si = np.argpartition(src_sc, min(sc_cnt-1, n-1))[:sc_cnt]
        di = np.argpartition(dst_sc, min(dc_cnt-1, n-1))[:dc_cnt]
        
        # Also add the overall top pressure positions
        all_src = np.argpartition(pres, min(8, n-1))[:8]
        all_dst = np.argpartition(-pres, min(12, n-1))[:12]
        
        si = np.unique(np.concatenate([si, all_src]))[:sc_cnt+8]
        di = np.unique(np.concatenate([di, all_dst]))[:dc_cnt+12]
        
        best_cand = None
        for srr in si:
            s = int(srr)
            if a[s] < 1e-12:
                continue
            for drr in di:
                d = int(drr)
                if d == s:
                    continue
                for frac in (0.003, 0.005, 0.01, 0.02, 0.03, 0.05):
                    tm = min(a[s] * frac, a[s] * 0.99)
                    if tm <= 1e-16:
                        continue
                    cn = apply_move(c, a, s, d, tm)
                    mn = px_max(cn)
                    if mn < M - 1e-18:
                        if best_cand is None or mn < best_cand[0]:
                            best_cand = (mn, s, d, tm)
        
        if best_cand is None:
            # Random exploratory moves
            for _ in range(40):
                s = int(rng.integers(n))
                d = int(rng.integers(n))
                if s == d or a[s] < 1e-12:
                    continue
                frac = float(rng.uniform(0.002, 0.03))
                tm = min(a[s]*frac, a[s]*0.99)
                if tm <= 1e-16:
                    continue
                cn = apply_move(c, a, s, d, tm)
                mn = px_max(cn)
                if mn < M - 1e-18:
                    if best_cand is None or mn < best_cand[0]:
                        best_cand = (mn, s, d, tm)
        
        if best_cand is not None:
            mn, s, d, tm = best_cand
            a[s] -= tm
            a[d] += tm
            np.maximum(a, 0, out=a)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a /= ss
            c = fconv(a)
            M = px_max(c)
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and (rnd <= 5 or rnd % 30 == 0):
                print("  Coord r={} proxy={:.6f} s={} d={}".format(rnd, 2*n*best_M, s, d))
        else:
            stall += 1
            if stall % 10 == 9:
                # Perturb
                pert = rng.normal(0, 0.01, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-12)
                a /= a.sum()
                c = fconv(a)
                M = px_max(c)
                stall = 0
    return best_a


def default_seq(n=5000):
    x = np.linspace(0, 1, int(n))
    return norm1(np.sin(np.pi * x)**2)


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    TOTAL = max(60.0, min(float(budget_s) - 40.0, 1760.0))
    
    def tl():
        return TOTAL - (time.monotonic() - t_start)
    
    def try_upd(ba, bs, new_a, label):
        if new_a is None:
            return ba, bs
        arr = np.asarray(new_a, dtype=np.float64)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        np.maximum(arr, 0, out=arr)
        s = float(arr.sum())
        if arr.size == 0 or s < 0.01:
            return ba, bs
        arr /= s
        sc = safe_eval(arr)
        if math.isfinite(sc) and sc < bs - 1e-15:
            print("  NEW BEST [{}]: {:.8f} n={} t={:.1f}s".format(label, sc, arr.size, time.monotonic()-t_start))
            return arr.copy(), sc
        return ba, bs
    
    print("Starting. Budget: {:.0f}s".format(TOTAL))
    
    # Initialize
    a0 = None
    try:
        pl = height_sequence_1
        p = np.asarray(pl, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        np.clip(p, 0, 1000, out=p)
        if p.ndim != 1:
            p = p.ravel()
        if p.size > 100 and p.sum() > 1e-10:
            a0 = norm1(p)
    except Exception:
        a0 = None
    
    if a0 is None:
        a0 = default_seq(5000)
    
    best_a = a0.copy()
    best_sc = safe_eval(a0)
    print("Init: n={} score={:.8f}".format(a0.size, best_sc))
    
    # Phase 1: L-BFGS multi-temperature
    t1 = time.monotonic()
    if SCIPY_OK and tl() > 400:
        p1_budget = min(tl() * 0.28, 450.0)
        print("\n=== P1: L-BFGS multi-temp ({:.0f}s) ===".format(p1_budget))
        a_work = a0.copy()
        if a_work.size > 9000:
            a_work = interpl(a_work, 8000)
        
        schedules = [
            (15.0, None),
            (40.0, None),
            (100.0, 200),
            (250.0, 100),
            (500.0, 50),
            (1000.0, 30),
            (2000.0, 15),
            (5000.0, 8),
            (10000.0, 5),
        ]
        
        per_budget = p1_budget / len(schedules)
        for beta, tc in schedules:
            if time.monotonic() - t1 > p1_budget - 3:
                break
            ms = None
            if tc is not None:
                cc = fconv(a_work)
                ms = top_mask(cc, None, tc)
            mf = int(max(60, min(300, per_budget * 3)))
            a_new = run_lbfgs(a_work, beta, ms, maxfun=mf)
            a_work = a_new
            best_a, best_sc = try_upd(best_a, best_sc, a_new, "lb_{}".format(beta))
        
        # Sync work
        if best_a.size > 9000:
            a_work = interpl(best_a, 8000)
        else:
            a_work = best_a.copy()
    
    # Phase 2: Frank-Wolfe
    t1 = time.monotonic()
    if tl() > 250:
        p2_budget = min(tl() * 0.18, 300.0)
        print("\n=== P2: Frank-Wolfe ({:.0f}s) ===".format(p2_budget))
        if a_work.size > 9000:
            a_work = interpl(a_work, 8000)
        a_fw = fw_search(a_work, p2_budget, active_max=2000, game_it=24,
                         exact_ev=5, exact_t=6.0, verbose=True)
        best_a, best_sc = try_upd(best_a, best_sc, a_fw, "fw")
        if best_a.size > 9000:
            a_work = interpl(best_a, 8000)
        else:
            a_work = best_a.copy()
    
    # Phase 3: Aggressive coordinate polish
    t1 = time.monotonic()
    if tl() > 150:
        p3_budget = min(tl() * 0.22, 380.0)
        print("\n=== P3: Coord polish ({:.0f}s) ===".format(p3_budget))
        a_cp = coord_polish(a_work, p3_budget, verbose=True)
        best_a, best_sc = try_upd(best_a, best_sc, a_cp, "coord1")
        if best_a.size > 9000:
            a_work = interpl(best_a, 8000)
        else:
            a_work = best_a.copy()
    
    # Phase 4: Second L-BFGS pass with very high beta
    t1 = time.monotonic()
