def fft_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_n = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_n)
    return np.fft.irfft(A * A, fft_n)[:size]


def eval_norm(a):
    """Evaluate normalized sequence (sum given as-is)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if a.sum() < 0.01:
        return float('inf')
    b = fft_autocorr(a)
    max_b = b.max()
    return 2.0 * n * max_b / (a.sum() ** 2)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def simplex_project(z):
    """Project z onto the simplicial simplex {x >= 0, sum x = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1] + 1
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(z - theta, 0.0)


def interp_to_n(a, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum() if a.sum() > 0 else np.ones(old_n) / old_n
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a_norm)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def spectral_init(n, shape='sin2'):
    """Create initial sequence with analytical shape."""
    x = np.linspace(0.0, 1.0, n)
    if shape == 'sin2':
        a = np.sin(np.pi * x) ** 2
    elif shape == 'cos':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif shape == 'gauss':
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif shape == 'tent':
        a = 1.0 - 2.0 * np.abs(x - 0.5)
    else:
        a = np.ones(n)
    return normalize(a)

def lp_fw(a0, time_limit, solver, K_init=300, K_max=5000, add_rate=15, verbose=True):
    """
    LP-Frank-Wolfe: minimize max_k (a*a)_k s.t. a>=0, sum(a)=1.
    At each step, solve LP: min M s.t. (y*g)_k <= M for k in active set, g>=0, sum(g)=1.
    Then move current solution toward g.
    """
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fft_autocorr(y).max()
    t0 = time.monotonic()
    
    # Initialize active set with top peaks
    ac = fft_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac)[::-1][:K_init])
    
    no_improve = 0
    t_count = 0
    
    for it in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 800:
            break
        
        t_count += 1
        ac = fft_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update active set: add violations
        violations = np.where(ac > M_curr - 1e-6)[0]
        top_viol = violations[np.argsort(ac)[::-1]][:add_rate] if len(violations) > 0 else np.array([])
        for v in top_viol:
            active_set.add(int(v))
        
        # Also periodically add random lags for diversity
        if it % 200 == 0:
            for _ in range(5):
                active_set.add(int(np.random.randint(0, 2 * n - 1)))
        
        # Trim active set if too large
        if len(active_set) > K_max:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep = np.argsort(margins)[:K_max]
            active_set = set(int(list(active_set)[i]) for i in keep)
        
        K = len(active_set)
        if K == 0:
            break
        
        # Build LP constraint matrix
        constraint_list = sorted(active_set)
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        # Solve LP: min M s.t. coeffs @ g <= M, g >= 0, sum(g) = 1
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > K_init:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:K_init])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = g_val.sum()
        if gs < 1e-15:
            continue
        g_dir = g_val / gs
        
        # Line search: try several step sizes
        best_alpha = 2.0 / (t_count + 2.0)
        best_trial_M = M_curr
        
        alphas = [best_alpha, best_alpha * 0.5, best_alpha * 1.5, 0.05, 0.1, 0.2, 0.3]
        for alpha in alphas:
            alpha = max(0.01, min(0.8, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial /= st
            mt = fft_autocorr(trial).max()
            if mt < best_trial_M:
                best_trial_M = mt
                best_alpha = alpha
        
        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new /= sy
        y = y_new
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print("  LP-FW it={}: score={:.6f} K={} t={:.1f}s".format(it, score, K, elapsed))
    
    return best_a


def peak_shave(a0, time_limit, verbose=True, seed=77):
    """
    Targeted peak reduction: identify the peak lag, find the contributing pairs,
    and redistribute mass to reduce that specific autocorrelation value.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fft_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = fft_autocorr(a)
        size = 2 * n - 1
        peak_k = int(np.argmax(ac))
        peak_val = ac[peak_k]
        
        if peak_val < best_m - 1e-14:
            best_m = peak_val
            best_a = a.copy()
            stall = 0
        
        # Find contributing pairs for the peak lag
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib = np.array([a[i] * a[peak_k - i] for i in range(js, je + 1)])
        total = contrib.sum()
        
        if total < 1e-15:
            break
        
        # Identify the largest contributors
        sorted_idx = np.argsort(contrib)[::-1]
        # Try to reduce the top contributors by moving mass away
        improved = False
        for si in sorted_idx[:5]:
            i = js + int(si)
            j = peak_k - i
            if not (0 <= i < n and 0 <= j < n):
                continue
            if i == j:
                continue  # Can't reduce a[i]^2 by moving i to i
            
            # Move a small amount from the larger of a[i], a[j] to a random position
            src = i if a[i] >= a[j] else j
            amt = a[src] * 0.02
            
            # Try several destinations
            for _ in range(3):
                dst = int(rng.integers(n))
                if dst == src or dst == j:
                    continue
                # Check if moving to dst would create a new peak
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial /= st
                mt = fft_autocorr(a_trial).max()
                if mt < peak_val - 1e-15:
                    a = a_trial
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 50:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 0:
                        a /= st
                        m = fft_autocorr(a).max()
                        if m < best_m:
                            best_m = m
                            best_a = a.copy()
                            stall = 0
                if stall > 300:
                    break
            else:
                # Small random move
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    amt = a[i] * rng.uniform(0.005, 0.02)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 0:
                        a /= st
        
        if verbose and stall % 200 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print("  PeakShave stall={}: score={:.6f} t={:.1f}s".format(stall, score, elapsed))
    
    return best_a

def get_solver():
    global _SOLVER
    if _SOLVER is not None:
        return _SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _SOLVER = s
                print("  Solver: {}".format(s))
                return s
        except Exception:
            continue
    _SOLVER = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _SOLVER


def fft_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_n = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_n)
    return np.fft.irfft(A * A, fft_n)[:size]


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1] + 1
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(z - theta, 0.0)


def interp_to_n(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum() if a.sum() > 0 else np.ones(old_n) / old_n
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a_norm)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def spectral_refine(a, time_limit, alpha_init=0.06, verbose=False):
    n = len(a)
    a = normalize(a)
    best_a = a.copy()
    best_m = fft_autocorr(a).max()
    t0 = time.monotonic()
    fft_n = 1 << (2 * n - 2).bit_length()
    alpha = alpha_init
    rng = np.random.default_rng(42)

    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        A = np.fft.rfft(a, fft_n)
        power = np.abs(A) ** 2
        target = power[1:].mean() if len(power) > 1 else power[0]
        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.0:
                A_new[k] *= max(0.2, np.sqrt(1.0 - alpha * (ratio - 1.0)))
            elif ratio < 0.5:
                A_new[k] *= min(3.0, np.sqrt(1.0 + 0.4 * alpha * (0.5 - ratio)))
        a_new = np.fft.irfft(A_new, fft_n)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new /= s
        m = fft_autocorr(a_new).max()
        if m < best_m - 1e-15:
            best_m = m
            best_a = a_new.copy()
            a = a_new
        elif it % 400 == 0:
            A_cur = np.fft.rfft(a, fft_n)
            pert = rng.standard_normal(len(A_cur)) * 0.04
            pert[0] = 0.0
            A_cur[1:] *= np.exp(1j * pert[1:])
            a_t = np.fft.irfft(A_cur, fft_n)[:n]
            a_t = np.maximum(a_t, 0.0)
            st = a_t.sum()
            if st > 1e-15:
                a_t /= st
                mt = fft_autocorr(a_t).max()
                if mt < best_m:
                    best_m = mt
                    best_a = a_t.copy()
                    a = a_t
        if it % 300 == 0:
            alpha = max(0.003, alpha * 0.97)
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print("  Spec it={}: m={:.8f} t={:.1f}s".format(it, best_m, elapsed))
    return best_a


def exact_subgrad_nesterov(a0, time_limit, verbose=False, seed=42):
    n = len(a0)
    x = normalize(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fft_autocorr(x).max()
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 1.0 / np.sqrt(n)
    rng = np.random.default_rng(seed)
    stall = 0

    for it in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        ac = fft_autocorr(y)
        size = 2 * n - 1
        peak_k = int(np.argmax(ac))
        peak_val = ac[peak_k]
        grad = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n, peak_k + 1)
        for i in range(js, je):
            j = peak_k - i
            if 0 <= j < n:
                grad[i] = y[j]
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            x_new = x
            m_new = peak_val
        else:
            grad_norm = grad / gn
            eta_trial = eta
            found = False
            for _ in range(8):
                z = y - eta_trial * grad_norm
                x_new = simplex_project(z)
                m_new = fft_autocorr(x_new).max()
                if m_new <= peak_val + 1e-12:
                    found = True
                    break
                eta_trial *= 0.5
            if not found:
                x_new = x
                m_new = peak_val
                eta = max(eta_trial, 1e-4)
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.05, 5.0)
        else:
            stall += 1
        x_prev = x
        x = x_new
        t_nest = t_new
        if stall > 400:
            stall = 0
            t_nest = 1.0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            eta = 1.0 / np.sqrt(n)
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print("  SubGrad it={}: score={:.6f} t={:.1f}s".format(it, score, elapsed))
    return best_a


def lp_fw(a0, time_limit, solver, K_init=400, K_max=5000, add_rate=20, verbose=False):
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fft_autocorr(y).max()
    t0 = time.monotonic()
    ac = fft_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac)[::-1][:K_init])
    no_improve = 0
    t_count = 0

    for it in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 1000:
            break
        t_count += 1
        ac = fft_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        thresh = M_curr * 0.999
        viol = np.where(ac > thresh)[0]
        if len(viol) > 0:
            top_v = viol[np.argsort(ac[viol])[::-1]][:add_rate]
            for v in top_v:
                active_set.add(int(v))
        if it % 300 == 0:
            for _ in range(3):
                active_set.add(int(np.random.randint(0, 2 * n - 1)))
        if len(active_set) > K_max:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep = np.argsort(margins)[:K_max]
            lst = list(active_set)
            active_set = set(int(lst[i]) for i in keep)
        K = len(active_set)
        if K == 0:
            break
        constraint_list = sorted(active_set)
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > K_init:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:K_init])
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = g_val.sum()
        if gs < 1e-15:
            continue
        g_dir = g_val / gs
        best_alpha = 2.0 / (t_count + 2.0)
        best_trial_M = M_curr
        alphas = [best_alpha, best_alpha * 0.5, best_alpha * 1.5, 0.05, 0.1, 0.2, 0.3]
        for alpha in alphas:
            alpha = max(0.01, min(0.8, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial /= st
            mt = fft_autocorr(trial).max()
            if mt < best_trial_M:
                best_trial_M = mt
                best_alpha = alpha
        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new /= sy
        y = y_new
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print("  LP-FW it={}: score={:.6f} K={} t={:.1f}s".format(it, score, K, elapsed))
    return best_a


def peak_shave(a0, time_limit, verbose=False, seed=77):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fft_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        ac = fft_autocorr(a)
        peak_k = int(np.argmax(ac))
        peak_val = ac[peak_k]
        if peak_val < best_m - 1e-14:
            best_m = peak_val
            best_a = a.copy()
            stall = 0
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        if js > je:
            break
        contrib = np.array([a[i] * a[peak_k - i] for i in range(js, je + 1)])
        total = contrib.sum()
        if total < 1e-15:
            break
        sorted_idx = np.argsort(contrib)[::-1]
        improved = False
        for si in sorted_idx[:6]:
            i = js + int(si)
            j = peak_k - i
            if not (0 <= i < n and 0 <= j < n and i != j):
                continue
            src = i if a[i] >= a[j] else j
            for frac in [0.01, 0.02, 0.03]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                for _ in range(4):
                    dst = int(rng.integers(n))
                    if dst == src or dst == j:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial /= st
                    mt = fft_autocorr(a_trial).max()
                    if mt < peak_val - 1e-15:
                        a = a_trial
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            stall += 1
            if stall > 30:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    amt = a[i] * rng.uniform(0.01, 0.04)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 0:
                        a /= st
                        m = fft_autocorr(a).max()
                        if m < best_m:
                            best_m = m
                            best_a = a.copy()
                            stall = 0
            if stall > 500:
                break
        if verbose and stall % 300 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print("  Shave stall={}: score={:.6f} t={:.1f}s".format(stall, score, elapsed))
    return best_a


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    a = np.minimum(a, 1000.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
    except Exception:
        return best_seq, best_score
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {:.6f} (n={}, t={:.1f}s)".format(label, sc, len(a), elapsed))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print("Time budget: {:.1f}s".format(time_budget))

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            a_list = [float(x) for x in parent]
            p_score = evaluate_sequence(a_list)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print("Parent: n={}, score={:.6f}".format(len(parent), p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        best_seq = normalize(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
        best_score = 2.0 * len(best_seq) * fft_autocorr(best_seq).max()
        print("Default init: n={}, score={:.6f}".format(len(best_seq), best_score))

    n0 = len(best_seq)
    print("Start: n={}, score={:.6f}".format(n0, best_score))

    # P1: Quick spectral initialization
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Spectral (n={}) ===".format(n0))
    t1 = min(remaining * 0.04, 50.0)
    if t1 > 10:
        a_out = spectral_refine(normalize(best_seq), t1, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec1", t_global)

    # P2: LP-FW primary (most time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P2: LP-FW primary (n={}) ===".format(n_cur))
    t2 = min(remaining * 0.25, 420.0)
    if t2 > 50:
        a_out = lp_fw(normalize(best_seq), t2, solver, K_init=400, K_max=4500, add_rate=20, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # P3: Exact subgradient
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P3: SubGrad (n={}) ===".format(n_cur))
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 25:
        a_out = exact_subgrad_nesterov(normalize(best_seq), t3, verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sg1", t_global)

    # P4: Peak shave
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P4: Shave (n={}) ===".format(n_cur))
    t4 = min(remaining * 0.06, 80.0)
    if t4 > 10:
        a_out = peak_shave(normalize(best_seq), t4, verbose=True, seed=77)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "shave1", t_global)

    # P5: LP-FW round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P5: LP-FW r2 (n={}) ===".format(n_cur))
    t5 = min(remaining * 0.20, 280.0)
    if t5 > 30:
        a_out = lp_fw(normalize(best_seq), t5, solver, K_init=500, K_max=5000, add_rate=25, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp2", t_global)

    # P6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_base = len(best_seq)
    print("\n=== P6: Multi-length (base n={}) ===".format(n_base))
    lengths = []
    for d in [-400, -150, 150, 400, 800]:
        c = n_base + d
        if 1500 <= c <= 12000:
            lengths.append(c)
    lengths = sorted(set(lengths))
    for new_n in lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 70:
            break
        print("  -> n={}".format(new_n))
        a_long = interp_to_n(best_seq, new_n)
        t6 = min(remaining * 0.30, 120.0)
        if t6 > 25 and new_n <= 9000:
            a_out = lp_fw(normalize(a_long), t6, solver, K_init=300, K_max=3000, add_rate=15, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n{}".format(new_n), t_global)
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.30, 80.0)
            if t6b > 15:
                a_out = exact_subgrad_nesterov(normalize(best_seq), t6b, verbose=False, seed=new_n % 9973)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "sg_n{}".format(new_n), t_global)

    # P7: Final LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P7: Final LP (n={}) ===".format(n_cur))
    if remaining > 40:
        t7 = min(remaining * 0.35, 200.0)
        a_out = lp_fw(normalize(best_seq), t7, solver, K_init=500, K_max=5000, add_rate=30, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # P8: Final subgradient
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P8: Final SubGrad (n={}) ===".format(n_cur))
    if remaining > 20:
        t8 = min(remaining * 0.35, 120.0)
        a_out = exact_subgrad_nesterov(normalize(best_seq), t8, verbose=True, seed=1234)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sg_final", t_global)

    # P9: Final shave
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P9: Final Shave (n={}) ===".format(n_cur))
    if remaining > 15:
        t9 = min(remaining * 0.50, 80.0)
        a_out = peak_shave(normalize(best_seq), t9, verbose=True, seed=4321)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "shave_final", t_global)

    # Final validation
    if best_seq is not None:
        a_list = [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
        fs = evaluate_sequence(a_list)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.6f}, n={}, time={:.1f}s".format(best_score, len(best_seq), elapsed))
    else:
        print("ERROR: No valid sequence!")
        n_fb = 3000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
