def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project z onto the probability simplex."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_grad_focused(y, beta, top_k):
    """Compute gradient of smoothed max of autocorrelation focused on top peaks."""
    n = len(y)
    b = fast_autocorr(y)
    size = 2 * n - 1
    if top_k is not None and 0 < top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros(size)
        mask = (b_use >= b_max - 1e-10)
        p[mask] = 1.0 / max(1, mask.sum())
    else:
        p = exp_vals / p_total
    # Gradient: grad[j] = sum_k p[k] * (a[k-j] + a[j-k]) with bounds
    # Use FFT: convolve a[::-1] with p, take appropriate slice
    a_rev = y[::-1]
    conv_len = n + size - 1
    fft_sz = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_sz)
    Pr = np.fft.rfft(p, fft_sz)
    conv = np.fft.irfft(Ar * Pr, fft_sz)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, p, b

    avg_count = 0.0

            if len(dst_candidates) == 0:
                continue
            dst = int(dst_candidates[rng.integers(len(dst_candidates))])

            # Try moving a fraction of mass
            for frac in [0.01, 0.02, 0.03, 0.05]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m * (1 - 1e-8):
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = max(3, n // 200)
            srcs = rng.choice(n, size=n_pert, replace=False)
            dsts = rng.choice(n, size=n_pert, replace=False)
            for s_i, d_i in zip(srcs, dsts):
                if a[s_i] > 1e-10:
                    amt = a[s_i] * float(rng.uniform(0.005, 0.05))
                    a[s_i] -= amt
                    a[d_i] += amt
            a = np.maximum(a, 0.0)
            s_t = a.sum()
            if s_t > 1e-15:
                a = a / s_t
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Peak-shaving ({elapsed:.1f}s): J={J:.6f} stall={stall}")

    return best_a, best_m


def freq_clipping(a0, time_limit, n_iters=500, verbose=True):
    """
    PAPR-inspired frequency domain clipping:
    Iteratively clip autocorrelation peaks and project back.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    clip_threshold = best_m * 1.01
    decay = 0.9995
    ramp_up = 1.0005

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        m_curr = ac.max()

        # Clip the autocorrelation
        ac_clipped = np.minimum(ac, clip_threshold)

        # Convert back to time domain (this is the modified signal)
        # The autocorrelation is the IFFT of |A|^2, so we need to find a' such that a'*a' = ac_clipped
        # This is not straightforward, so instead we use a gradient step
        # based on the difference between ac and ac_clipped
        diff = ac - ac_clipped
        if diff.max() < 1e-12:
            break

        # Gradient of ||ac - ac_clipped||^2 / 2 with respect to a
        # = sum_k (ac[k] - ac_clipped[k]) * dac[dk]/da
        # dac[k]/da[j] = a[k-j] + a[j-k]
        # = conv(a_rev, diff) + conv(a, diff) at appropriate indices
        a_rev = a[::-1]
        diff_sym = diff.copy()
        diff_sym[n:] = diff_sym[n::-1]  # make symmetric for the two terms
        # Actually, let's just use the focused gradient approach
        # The "error" signal is diff, and we want to reduce peaks
        # gradient[j] = sum_k diff[k] * (a[k-j] + a[j-k])
        fft_sz2 = 1 << (n + len(diff) - 1).bit_length()
        Ar = np.fft.rfft(a_rev, fft_sz2)
        Dr = np.fft.rfft(diff, fft_sz2)
        conv1 = np.fft.irfft(Ar * Dr, fft_sz2)
        # Second term: sum_k diff[k] * a[j-k] = conv(diff, a)[j]
        Ad = np.fft.rfft(a, fft_sz2)
        conv2 = np.fft.irfft(Dr * Ad, fft_sz2)

        # Extract the relevant n-element slices
        # For conv1: (a_rev * diff)[j + n - 1] gives sum_k a_rev[k]*diff[j+n-1-k]
        # a_rev[k] = a[n-1-k], so this is sum_k a[n-1-k]*diff[j+n-1-k]
        # Let l = n-1-k, k = n-1-l: sum_l a[l]*diff[j+l]
        # This is the cross-correlation we want for the first term
        grad_part1 = conv1[n - 1:2 * n - 1]
        # For conv2: (diff * a)[j] = sum_k diff[k]*a[j-k]
        grad_part2 = conv2[:n]

        grad = grad_part1 + grad_part2

        # Step
        step_size = 0.5 / (np.max(np.abs(grad)) + 1e-10)
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()

        a = a_new

        # Adapt threshold
        clip_threshold = max(best_m * 1.005, clip_threshold * decay)

        if verbose and it % 100 == 0:
            J = 2.0 * n * best_m
            print(f"    FreqClip it={it}: J={J:.6f}")

    return best_a, best_m


def cyclic_shift_best(a0):
    """Find the best cyclic shift of a0."""
    n = len(a0)
    if n < 2:
        return a0.copy(), fast_autocorr(a0).max()
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    # For efficiency, only try a subset of shifts if n is large
    if n > 5000:
        shifts = list(range(0, n, max(1, n // 200)))
    else:
        shifts = range(n)
    fft_size = 1 << (2 * n - 1).bit_length()
    for s in shifts:
        if s == 0:
            continue
        a_shift = np.roll(a0, -s)
        A = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def generate_initialization(n, method="sin2"):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if method == "sin2":
        a = np.sin(np.pi * x) ** 2
    elif method == "gaussian":
        sigma = 0.25
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    elif method == "raised_cosine":
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif method == "multi_sin":
        a = np.sin(np.pi * x) ** 2 + 0.3 * np.sin(2 * np.pi * x) ** 2 + 0.1 * np.sin(3 * np.pi * x) ** 2
    elif method == "triangular":
        a = np.maximum(0, 1 - 2 * np.abs(x - 0.5))
    elif method == "sech":
        a = 1.0 / np.cosh(4 * (x - 0.5))
    else:
        a = np.sin(np.pi * x) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update the best sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.1f}s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main search function. Tries multiple optimization strategies
    and returns the best sequence found.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    n_parent = 0
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            n_parent = len(parent)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={n_parent}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 6700
        best_seq = generate_initialization(n0, "sin2")
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n0}, score={best_score:.6f}")
        n_parent = n0

    n0 = len(best_seq)

    # === Phase 1: FISTA with proper BB (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: FISTA-BB (n={n0}) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 60:
        a_out, _ = fista_optimized(norm_sum1(best_seq), t1,
                                   beta_start=15.0, beta_end=800.0,
                                   beta_growth=1.003, top_k=15,
                                   restart_every=3000, verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista1", t_global)

    # === Phase 2: Adam-based optimizer (different trajectory) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Adam FISTA (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 40:
        a_out, _ = fista_optimized(norm_sum1(best_seq), t2,
                                   beta_start=30.0, beta_end=1200.0,
                                   beta_growth=1.005, top_k=10,
                                   restart_every=2000, verbose=False,
                                   use_adam=True, seed=123)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "adam1", t_global)

    # === Phase 3: Cyclic shift ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Cyclic shift (n={len(best_seq)}) ===")
    if remaining > 30:
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic1", t_global)

    # === Phase 4: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Peak shaving (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        a_out, _ = peak_shaving(norm_sum1(best_seq), t4, verbose=True, seed=77)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak1", t_global)

    # === Phase 5: Focused FISTA with different top_k ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: FISTA focused (n={len(best_seq)}) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 40:
        a_out, _ = fista_optimized(norm_sum1(best_seq), t5,
                                   beta_start=50.0, beta_end=2000.0,
                                   beta_growth=1.006, top_k=8,
                                   restart_every=2500, verbose=False, seed=555)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_focused", t_global)

    # === Phase 6: Frequency clipping ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Freq clipping (n={n_best}) ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 30:
        a_out, _ = freq_clipping(norm_sum1(best_seq), t6, n_iters=300, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "freqclip1", t_global)

    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Multi-length (base n={n_best}) ===")

    # Try slightly longer sequences
    longer_lengths = [n_best + 300, n_best + 700, n_best + 1500]
    longer_lengths = [l for l in longer_lengths if l <= 12000]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.25, 180.0)
        if t7a > 40:
            a_out, _ = fista_optimized(a_long, t7a,
                                       beta_start=20.0, beta_end=800.0,
                                       beta_growth=1.004, top_k=12,
                                       restart_every=2500, verbose=False, seed=new_n % 10000)
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"fista_n{new_n}", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
            best_seq, best_score = try_update(best_seq, best_score, a_shifted, f"cyc_n{new_n}", t_global)

    # === Phase 8: Different initialization + optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Alt initialization (n={len(best_seq)}) ===")
    if remaining > 100:
        n_alt = len(best_seq)
        for method in ["gaussian", "multi_sin"]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 80:
                break
            a_init = generate_initialization(n_alt, method)
            # Blend with current best
            a_blend = 0.5 * norm_sum1(best_seq) + 0.5 * a_init
            t8 = min(remaining * 0.35, 150.0)
            if t8 > 40:
                a_out, _ = fista_optimized(a_blend, t8,
                                           beta_start=25.0, beta_end=1000.0,
                                           beta_growth=1.004, top_k=12,
                                           restart_every=2500, verbose=False,
                                           seed=hash(method) % 10000)
                best_seq, best_score = try_update(best_seq, best_score, a_out, f"alt_{method}", t_global)

    # === Phase 9: Final peak shaving + FISTA polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final polish (n={n_best}) ===")

    t9a = min(remaining * 0.35, 150.0)
    if t9a > 30:
        a_out, _ = peak_shaving(norm_sum1(best_seq), t9a, verbose=True, seed=222)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_final", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    t9b = min(remaining * 0.35, 150.0)
    if t9b > 30:
        a_out, _ = fista_optimized(norm_sum1(best_seq), t9b,
                                   beta_start=100.0, beta_end=3000.0,
                                   beta_growth=1.01, top_k=6,
                                   restart_every=2000, verbose=False, seed=777)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_final", t_global)

    # === Final: Cyclic shift ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Final: Cyclic shift ===")
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_final", t_global)

    # === Final: Micro FISTA ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Final: Micro FISTA ===")
        t_f = min(remaining * 0.6, 120.0)
        if t_f > 20:
            a_out, _ = fista_optimized(norm_sum1(best_seq), t_f,
                                       beta_start=200.0, beta_end=5000.0,
                                       beta_growth=1.015, top_k=5,
                                       restart_every=1500, verbose=False, seed=333)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_micro", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 6700
        best_seq = generate_initialization(n_fb, "sin2")
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}, n={n_fb}")

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
