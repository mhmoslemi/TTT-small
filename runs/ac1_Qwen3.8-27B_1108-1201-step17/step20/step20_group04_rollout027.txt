"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive multi-beta log-sum-exp gradient descent with Nesterov acceleration
2. Peak-directed mass redistribution (targeted local search on top peaks)

import numpy as np
import time

def fast_autocorr(a):
    """Compute full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def smooth_objective_grad(a, beta, top_k=None):
    """Compute gradient of smoothed objective (1/beta)*log(sum top_k exp(beta*c_k)) w.r.t. a.
    Returns (grad, c, peak_weights)"""
    n = len(a)
    c = fast_autocorr(a)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c_use[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    # grad_j = (2/beta) * sum_k p_k * a[k-j] (with a[x]=0 outside [0,n-1])
    fft_size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A_rev = np.fft.rfft(a[::-1], fft_size)
    Conv = np.fft.irfft(P * A_rev, fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad, p, c

def nesterov_smooth(a0, time_limit, beta_start=50.0, beta_end=3000.0,
                     beta_growth=1.008, eta_base=0.5, top_k_ratio=0.015,
                     restart_every=1500, verbose=True, seed=42):
    """Nesterov accelerated projected gradient on smoothed objective."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, c = smooth_objective_grad(y, beta, top_k)

        # Adaptive step size: based on gradient norm and current peak
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            eta = eta_base
        else:
            eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_m - 1e-15:
            best_m = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            # Structured restart
            noise_scale = 0.03 / (1.0 + 0.05 * (t // restart_every))
            # Combine anti-symmetric, frequency-domain, and random
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            
            # Frequency-domain restart component
            A = np.fft.rfft(best_a)
            phase = np.exp(1j * rng.uniform(0, 2*np.pi, len(A)))
            A_new = np.abs(A) * phase
            a_new = np.fft.irfft(A_new, n * 2)[:n]
            a_new = np.maximum(a_new, 0)
            a_new = a_new / (a_new.sum() + 1e-15)
            
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.3 * anti + 0.3 * (a_new - best_a) + 0.4 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + ", beta=" + str(round(beta, 1)))

    return best_a, best_m


def peak_directed_local_search(a0, time_limit, verbose=True, seed=77):
    """Peak-directed local search: for each top peak, find optimal mass redistribution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 5000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Find top peaks (focus on the most important ones)
        n_top = min(20, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]
        
        # Compute "pressure" map: for each position j, how much does it contribute to top peaks?
        pressure = np.zeros(n)
        for k in top_idx[:10]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            partner = k_int - idx
            pressure[idx] += a[partner]
            pressure[partner] += a[idx]
        
        # For each top peak, try to reduce it
        improved = False
        for k in top_idx[:8]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            
            # Find the pair contributing most to this peak
            for pi in np.argsort(products)[::-1][:3]:
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n:
                    continue
                
                # Determine which element to reduce
                if a[src_i] >= a[partner]:
                    reduce_i = src_i
                else:
                    reduce_i = partner
                
                if a[reduce_i] < 1e-14:
                    continue
                
                # Find best destination: minimize pressure, avoid peak regions
                # Exclude positions that would create new peaks
                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_idx[:10]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2 + 1] = True
                
                # Score each candidate destination
                scores = np.zeros(n)
                for j in range(n):
                    if j == reduce_i:
                        scores[j] = -np.inf
                        continue
                    # Penalty for being in peak regions
                    if in_peak[j]:
                        scores[j] -= 1.0
                    # Penalty proportional to pressure
                    scores[j] -= pressure[j]
                    # Small random component for diversity
                    scores[j] += rng.normal(0, 0.1)
                
                # Try top destinations
                top_dsts = np.argsort(scores)[::-1][:5]
                
                for dst in top_dsts:
                    dst_int = int(dst)
                    if dst_int == reduce_i or scores[dst_int] == -np.inf:
                        continue
                    
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.03]:
                        amt = a[reduce_i] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[reduce_i] -= amt
                        a_trial[dst_int] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random exploration when stuck
            if no_improve > 200:
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                break

        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PkSearch (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve))

    return best_a, best_m


def freq_domain_restart(a, rng):
    """Generate a diverse candidate by randomizing FFT phases of a."""
    n = len(a)
    A = np.fft.rfft(a)
    # Randomize phases but keep magnitudes
    phase = np.exp(1j * rng.uniform(0, 2 * np.pi, len(A)))
    A_new = np.abs(A) * phase
    a_new = np.fft.irfft(A_new, 2 * n)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def multi_beta_optimization(a0, time_limit, verbose=True, seed=99):
    """Run Nesterov with multiple beta values and keep the best."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    beta_configs = [
        (30.0, 1000.0, 1.006, 0.6, 0.02),
        (80.0, 3000.0, 1.008, 0.5, 0.015),
        (200.0, 8000.0, 1.01, 0.4, 0.01),
        (500.0, 15000.0, 1.012, 0.3, 0.005),
    ]
    
    # Run each config for a portion of the time
    per_time = time_limit / len(beta_configs)
    
    for bi, (bs, be, bg, eb, tk) in enumerate(beta_configs):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break
        t_alloc = min(per_time, remaining * 0.4)
        if t_alloc < 20:
            break
        
        a_out, m_out = nesterov_smooth(best_a, t_alloc,
                                        beta_start=bs, beta_end=be,
                                        beta_growth=bg, eta_base=eb,
                                        top_k_ratio=tk, restart_every=1000,
                                        verbose=False, seed=seed + bi)
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
            if verbose:
                J = 2.0 * n * best_m
                print("      MultiBeta config " + str(bi) + ": J=" + repr(round(J, 6)))
    
    return best_a, best_m


def fine_grained_polish(a0, time_limit, verbose=True, seed=123):

        c = fast_autocorr(a)

        peak_positions = np.where(c > threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.argsort(c)[::-1][:10]
        
        improved = False
        for k in peak_positions[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:

                        a_trial[larger_i] -= amt

                        if s_t < 1e-15:

                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Structured nudge when stuck
            if stall % 50 == 0:
                # Try a small perturbation in a structured direction

                period = max(3, n // 10)

                a_try = np.maximum(a_try, 0.0)

                if s_t > 1e-15:
                    a_try = a_try / s_t
                    M_try = fast_autocorr(a_try).max()
                    if M_try < best_m:
                        a = a_try
                        best_m = M_try
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0 and stall > 0:

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def smooth_objective_grad(a, beta, top_k=None):
    """Compute gradient of (1/beta)*log(sum top_k exp(beta*c_k)) w.r.t. a."""
    n = len(a)
    c = fast_autocorr(a)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c_use[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    # grad_j = (2/beta) * sum_k p_k * a[k-j]
    fft_size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A_rev = np.fft.rfft(a[::-1], fft_size)
    Conv = np.fft.irfft(P * A_rev, fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad, p, c


def nesterov_smooth(a0, time_limit, beta_start=50.0, beta_end=3000.0,

                    restart_every=1500, verbose=True, seed=42):
    """Nesterov accelerated projected gradient with adaptive step and structured restarts."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, c = smooth_objective_grad(y, beta, top_k)

        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_m - 1e-15:
            best_m = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.03 / (1.0 + 0.05 * (t // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            A = np.fft.rfft(best_a)
            phase = np.exp(1j * rng.uniform(0, 2 * np.pi, len(A)))
            a_new = np.fft.irfft(np.abs(A) * phase, 2 * n)[:n]
            a_new = np.maximum(a_new, 0)
            a_new = a_new / (a_new.sum() + 1e-15)
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.3 * anti + 0.3 * (a_new - best_a) + 0.4 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def conflict_aware_peak_search(a0, time_limit, verbose=True, seed=77):
    """Peak-directed search using conflict graph: identifies indices shared by multiple top peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    temperature = 1.0

    while time.monotonic() - t0 < time_limit and no_improve < 8000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
            temperature = max(0.1, temperature * 0.98)
        else:
            no_improve += 1
            if no_improve % 500 == 0:
                temperature = min(1.0, temperature * 1.02)

        n_top = min(25, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]

        # Build conflict map: for each index j, count how many top peaks it contributes to
        conflict = np.zeros(n)
        peak_contribution = {}
        for k in top_idx[:12]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            partner = k_int - idx
            products = a[idx] * a[partner]
            peak_contribution[k_int] = (idx, partner, products)
            conflict[idx] += products
            conflict[partner] += products

        # Find the most "conflicted" index (contributes to most peak value)
        # We want to reduce mass at high-conflict positions
        improved = False

        # Strategy 1: Reduce mass at the most conflicted position that belongs to the top peak
        top_peak_k = int(top_idx[0])
        if top_peak_k in peak_contribution:
            idx, partner, products = peak_contribution[top_peak_k]
            # Find the pair with largest product at the top peak
            for pi in np.argsort(products)[::-1][:4]:
                src_i = int(idx[pi])
                par_i = int(partner[pi])
                if par_i < 0 or par_i >= n:
                    continue
                # Choose which to reduce: the one with higher conflict
                if conflict[src_i] >= conflict[par_i]:
                    reduce_i = src_i
                else:
                    reduce_i = par_i

                if a[reduce_i] < 1e-14:
                    continue

                # Score all possible destinations
                # A good destination: low conflict, not in any top-peak region
                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_idx[:12]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2 + 1] = True

                # Compute "safety score" for each destination
                safety = np.zeros(n)
                for j in range(n):
                    if j == reduce_i:
                        safety[j] = -1e10
                        continue
                    safety[j] = -conflict[j] * 0.5
                    if in_peak[j]:
                        safety[j] -= best_m * 2.0
                    # Bonus for being far from all peak regions
                    safety[j] += rng.normal(0, 0.01)

                top_dsts = np.argsort(safety)[::-1][:8]

                for dst in top_dsts:
                    dst_int = int(dst)
                    if safety[dst_int] < -1e9:
                        continue
                    # Try a range of fractions (larger than before for more aggressive moves)
                    for frac in [0.01, 0.02, 0.05, 0.1, 0.15, 0.25]:
                        amt = a[reduce_i] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[reduce_i] -= amt
                        a_trial[dst_int] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()

                        # Simulated annealing acceptance
                        if M_trial < best_m:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                        elif rng.random() < np.exp(-(M_trial - best_m) / (best_m * temperature * 0.01)):
                            # Accept worse move (annealing)
                            a = a_trial
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Strategy 2: If no improvement, try random pair moves
        if not improved:
            for _ in range(30):
                if no_improve % 500 < 250:
                    # Guided random: pick from high-conflict sources
                    src_candidates = np.argsort(conflict)[::-1][:200]
                    src_i = int(src_candidates[int(rng.integers(len(src_candidates)))])
                else:
                    src_i = int(rng.integers(n))
                dst_i = int(rng.integers(n))
                if src_i == dst_i or a[src_i] < 1e-12:
                    continue
                frac = rng.uniform(0.01, 0.2)
                amt = a[src_i] * frac
                a_trial = a.copy()
                a_trial[src_i] -= amt
                a_trial[dst_i] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
                elif rng.random() < np.exp(-(M_trial - best_m) / (best_m * temperature * 0.01)):
                    a = a_trial
                    improved = True
                    break

        if verbose and no_improve % 3000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    ConflictPK (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve) + " temp=" + str(round(temperature, 3)))

    return best_a, best_m


def freq_restart(a, rng, strength=0.5):
    """Generate diverse candidate via FFT phase randomization, blended with original."""
    n = len(a)
    A = np.fft.rfft(a)
    phase = np.exp(1j * rng.uniform(0, 2 * np.pi, len(A)))
    A_new = np.abs(A) * phase
    a_new = np.fft.irfft(A_new, 2 * n)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        # Try with partial phase randomization
        partial = rng.random(len(A)) < 0.5
        phase2 = np.ones(len(A), dtype=complex)
        phase2[partial] = np.exp(1j * rng.uniform(0, 2 * np.pi, partial.sum()))
        A_new2 = np.abs(A) * phase2
        a_new = np.fft.irfft(A_new2, 2 * n)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            return a.copy()
        a_new = a_new / s
    else:
        a_new = a_new / s
    # Blend with original for stability
    return (1 - strength) * a + strength * a_new


def multi_scale_optimize(a0, time_limit, verbose=True, seed=99):
    """Run multiple Nesterov configs with different scales and return best."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    configs = [
        (30.0, 800.0, 1.006, 0.6, 0.02, 1200),
        (80.0, 2500.0, 1.009, 0.5, 0.015, 1000),
        (200.0, 6000.0, 1.011, 0.4, 0.01, 800),
        (500.0, 12000.0, 1.014, 0.3, 0.006, 600),
        (100.0, 5000.0, 1.01, 0.45, 0.012, 900),
    ]

    per_time = time_limit / len(configs)

    for ci, (bs, be, bg, eb, tk, re) in enumerate(configs):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break
        t_alloc = min(per_time * 1.2, remaining * 0.35)
        if t_alloc < 20:
            break

        a_out, m_out = nesterov_smooth(best_a, t_alloc,
                                       beta_start=bs, beta_end=be,
                                       beta_growth=bg, eta_base=eb,
                                       top_k_ratio=tk, restart_every=re,
                                       verbose=False, seed=seed + ci)
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
            if verbose:
                J = 2.0 * n * best_m
                print("      MultiScale cfg" + str(ci) + ": J=" + repr(round(J, 6)))

    return best_a, best_m


def fine_polish(a0, time_limit, verbose=True, seed=123):
    """Ultra-fine coordinate descent for final convergence."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 400:
        c = fast_autocorr(a)
        threshold = np.percentile(c, 97)
        peak_pos = np.where(c > threshold)[0]
        if len(peak_pos) == 0:
            peak_pos = np.argsort(c)[::-1][:10]

        improved = False
        for k in peak_pos[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]

            for pi in np.argsort(products)[::-1][:2]:
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n or partner == src_i:
                    continue
                larger_i = src_i if a[src_i] >= a[partner] else partner
                if a[larger_i] < 1e-14:
                    continue

                for frac in [0.0005, 0.001, 0.002, 0.003]:
                    amt = a[larger_i] * frac
                    if amt < 1e-18:
                        continue
                    for _ in range(8):
                        dst = int(rng.integers(n))
                        if dst == larger_i:
                            continue
                        a_trial = a.copy()
                        a_trial[larger_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 40 == 0:
                eps = 0.003 / (1.0 + stall / 40.0)
                period = max(3, n // 8)
                direction = np.sin(2 * np.pi * np.arange(n) / period) * np.cos(np.pi * np.arange(n) / n)
                a_try = a + eps * direction * a
                a_try = np.maximum(a_try, 0.0)
                s_t = a_try.sum()
                if s_t > 1e-15:
                    a_try = a_try / s_t
                    M_try = fast_autocorr(a_try).max()
                    if M_try < best_m:
                        a = a_try
                        best_m = M_try
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 150 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Polish (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def propose_candidate(budget_s=1800.0):

        p = np.clip(p, 0.0, 1000.0)

            parent = p.copy()
            p_score = evaluate_safe(parent)

                best_seq = norm_sum1(parent)
                best_score = p_score

        pass
