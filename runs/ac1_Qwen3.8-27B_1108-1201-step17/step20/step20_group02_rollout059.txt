def fast_convolve(a):
    """Fast autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def score_fast(a):

    return 2.0 * n * fast_convolve(a).max() / (s * s)


def normalize(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def init_ramp(n, p=1.0):

def init_poly2(n, p, q):

def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs

def nesterov_prox(a0, time_limit, verbose=False):
    """Nesterov-accelerated proximal gradient using exact subgradient."""
    n = len(a0)
    x = normalize(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_m = fast_convolve(x).max()
    best_score = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    mu = 0.7
    eta = 0.08
    eta_min = 0.001
    eta_decay = 0.9995
    stall = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        if stall > 30000:
            break
        iter_count += 1

        # Nesterov lookahead
        y = x + mu * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Subgradient of max conv
        ac = fast_convolve(y)
        k_star = int(np.argmax(ac))

        g = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        idx = np.arange(i_start, i_end)
        g[idx] = 2.0 * y[k_star - idx]

        # Add small random perturbation to avoid cycling
        if iter_count % 500 == 0:
            g += 0.005 * eta * y * rng.standard_normal(n)

        # Proximal step: x_new = proj_simplex(y - eta * g)
        v = y - eta * g
        x_new = proj_simplex(v)

        x_prev = x
        x = x_new
        eta = max(eta_min, eta * eta_decay)

        M_new = fast_convolve(x).max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_x = x.copy()
            best_score = 2.0 * n * best_m
            stall = 0
        else:
            stall += 1

        if stall > 5000 and stall % 5000 == 0:
            # Reheat
            noise_scale = 0.02 * (stall // 5000)
            x_p = best_x + noise_scale * rng.standard_normal(n) * best_x
            x_p = np.maximum(x_p, 1e-12)
            x = normalize(x_p)
            x_prev = x.copy()
            eta = max(eta, 0.03)

        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print("    NesterovPG t=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(best_score))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    NesterovPG done (" + str(round(elapsed, 1)) + "s): J=" + repr(best_score) + " iters=" + str(iter_count))
    return best_x, best_score


def peak_shave(a0, time_limit, verbose=False):
    """Targeted mass redistribution to reduce peak convolution."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(1234)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 2000:
        ac = fast_convolve(a)
        M_curr = ac.max()
        k_star = int(np.argmax(ac))

        # Find contributors to the peak
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        idx = np.arange(i_start, i_end)
        products = a[idx] * a[k_star - idx]

        if products.max() < 1e-15:
            stall += 1
            continue

        # Source: position with largest product contribution
        src_local = int(idx[np.argmax(products)])

        # Find safe destinations (low contribution to all near-peaks)
        top_peaks = np.argsort(ac)[::-1][:min(20, len(ac))]
        pressure = np.zeros(n)
        for kp in top_peaks:
            kp = int(kp)
            s2 = max(0, kp - n + 1)
            e2 = min(n, kp + 1)
            if s2 < e2:
                idx2 = np.arange(s2, e2)
                pressure[idx2] += a[kp - idx2]

        # Candidates: low pressure positions
        candidates = np.where(pressure < np.percentile(pressure, 30))[0]
        if len(candidates) < 10:
            candidates = np.argsort(pressure)[:n // 4]

        best_improve = 0
        best_dst = -1
        best_amt = 0

        for dst in candidates[:50]:
            if dst == src_local:
                continue
            for frac in [0.01, 0.02, 0.05, 0.1]:
                amt = a[src_local] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src_local] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                improve = M_curr - M_trial
                if improve > best_improve:
                    best_improve = improve
                    best_dst = int(dst)
                    best_amt = amt

        if best_improve > 1e-15 and best_dst >= 0:
            a[src_local] -= best_amt
            a[best_dst] += best_amt
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            a = a / s_a
            best_m = fast_convolve(a).max()
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            if stall % 200 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_a = a.sum()
                        a = a / s_a
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 1000 == 0:
            elapsed = time.monotonic() - t0
            print("    PeakShave (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + " stall=" + str(stall))

    return best_a, best_m

                        best_action = (src, int(dst), amt)

            if best_action is not None:
                s_, d_, amt = best_action
                a[s_] -= amt
                a[d_] += amt
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                a = a / s_a
                best_m = fast_convolve(a).max()
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1
            if stall % 500 == 0:
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.003, 0.015)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_a = a.sum()
                        a = a / s_a
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 1000 == 0:
            elapsed = time.monotonic() - t0
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + " stall=" + str(stall))

    return best_a, best_m


def cmaes_global(a0, time_limit, n_bumps=12, seed=42, verbose=False):
    """CMA-ES on Gaussian bump parameterization."""
    n = len(a0)
    x = np.linspace(0.0, 1.0, n)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    best_a = normalize(a0)
    best_val = score_fast(best_a)

    params = np.zeros(3 * n_bumps)
    centers = np.linspace(0.1, 0.9, n_bumps)
    for j in range(n_bumps):
        params[3 * j] = 1.0
        params[3 * j + 1] = centers[j]
        params[3 * j + 2] = rng.uniform(0.05, 0.15)

    dim = 3 * n_bumps
    pop_size = max(6 + int(3 * np.log(dim)), 16)
    sigma = 0.2
    mean = params.copy()
    evals_done = 0
    max_evals = 5000

    def bump_to_seq(p):
        a = np.zeros(n)
        for j in range(n_bumps):
            w = max(0.01, p[3 * j])
            c = np.clip(p[3 * j + 1], 0.02, 0.98)
            s = max(0.02, p[3 * j + 2])
            a += w * np.exp(-((x - c) ** 2) / (2.0 * s * s))
        return normalize(a)

    def bump_eval(p):
        a = bump_to_seq(p)
        return score_fast(a)

    while evals_done < max_evals and time.monotonic() - t0 < time_limit:
        samples = mean + sigma * rng.standard_normal((pop_size, dim))
        fitness = np.zeros(pop_size)
        for i in range(pop_size):
            if time.monotonic() - t0 > time_limit:
                break
            fitness[i] = bump_eval(samples[i])
            evals_done += 1
            if fitness[i] < best_val:
                best_val = fitness[i]
                best_a = bump_to_seq(samples[i]).copy()
                if verbose:
                    print("    CMA-ES new best: J=" + repr(best_val))

        lam = pop_size // 2
        top_idx = np.argsort(fitness)[:lam]
        w_norm = np.log(lam + 0.5) - np.log(np.arange(1, lam + 1))
        w_norm = w_norm / w_norm.sum()
        mean = np.zeros(dim)
        for wi, idx in zip(w_norm, top_idx):
            mean += wi * samples[idx]

        sigma *= 0.98
        if sigma < 0.005:
            sigma = 0.05
        if evals_done % (pop_size * 30) == 0 and evals_done > pop_size * 10:
            sigma = min(sigma * 1.5, 0.5)

    if verbose:
        elapsed = time.monotonic() - t0
        print("    CMA-ES done (" + str(round(elapsed, 1)) + "s): J=" + repr(best_val) + " evals=" + str(evals_done))
    return best_a, best_val


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def get_initial_sequence(t_global):
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
                return p_norm, sc
    except Exception:
        pass
    n = 8000
    best_seq = init_ramp(n, 1.0)
    best_score = evaluate_safe(best_seq)
    print("Default init: n=" + str(n) + ", score=" + repr(best_score))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = get_solver()
    best_seq, best_score = get_initial_sequence(t_global)

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: Try diverse initializations ===
    print("\n=== Phase 1: Diverse initialization ===")
    n_parent = len(best_seq)
    init_candidates = []

    # Parent
    init_candidates.append(('parent', best_seq, 'init_parent'))

    # Ramps with various exponents
    for p_val in [0.5, 0.8, 1.0, 1.3, 1.5, 2.0]:
        a_cand = init_ramp(n_parent, p_val)
        sc = evaluate_safe(a_cand)
        if np.isfinite(sc):
            init_candidates.append(('ramp_p' + str(p_val), a_cand, 'init_ramp'))

    # Two-parameter polynomials
    for p_val in [0.5, 1.0, 1.5]:
        for q_val in [0.3, 0.5, 0.8, 1.0]:
            a_cand = init_poly2(n_parent, p_val, q_val)
            sc = evaluate_safe(a_cand)
            if np.isfinite(sc):
                init_candidates.append(('poly_' + str(p_val) + '_' + str(q_val), a_cand, 'init_poly'))

    # Sine-modified
    for alpha in [1.0, 2.0, 3.0, 5.0]:
        a_cand = init_sine_mod(n_parent, alpha)
        sc = evaluate_safe(a_cand)
        if np.isfinite(sc):
            init_candidates.append(('sine_' + str(alpha), a_cand, 'init_sine'))

    # Find best initialization
    best_init = best_seq
    best_init_score = best_score
    for name, seq, _ in init_candidates:
        sc = evaluate_safe(seq)
        if np.isfinite(sc) and sc < best_init_score:
            best_init_score = sc
            best_init = seq
            print("  Better init: " + name + " score=" + repr(sc))

    best_seq = best_init
    best_score = best_init_score
    print("  Best init score: " + repr(best_score))

    # === Phase 2: Nesterov proximal gradient ===
    print("\n=== Phase 2: Nesterov Proximal Gradient (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining_time() * 0.10, 150.0)
    if t2 > 20:
        a_out, _ = nesterov_prox(best_seq, t2, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nesterov_n" + str(len(best_seq)), t_global)

    # === Phase 3: LP-FW first pass ===
    print("\n=== Phase 3: LP-FW (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining_time() * 0.20, 300.0)
    if t3 > 40 and len(best_seq) <= 12000:
        a_out, _ = lp_fw(best_seq, t3, solver, initial_K=1000, max_K=6000, add_per_iter=80, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1", t_global)

    # === Phase 4: Peak shaving ===
    print("\n=== Phase 4: Peak Shaving (n=" + str(len(best_seq)) + ") ===")
    t4 = min(remaining_time() * 0.08, 120.0)
    if t4 > 15:
        a_out, _ = peak_shave(best_seq, t4, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakshave", t_global)

    # === Phase 5: CMA-ES global exploration ===
    print("\n=== Phase 5: CMA-ES Global (n=" + str(len(best_seq)) + ") ===")
    t5 = min(remaining_time() * 0.08, 120.0)
    if t5 > 25:
        a_out, _ = cmaes_global(best_seq, t5, n_bumps=14, seed=777, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "cmaes1", t_global)

    # === Phase 6: Multi-n exploration ===
    print("\n=== Phase 6: Multi-n exploration ===")
    n_best = len(best_seq)
    try_ns = sorted(set([n_best - 1000, n_best + 1000, n_best + 2000,
                         n_best - 2000, n_best + 3000,
                         max(3000, n_best // 2), min(20000, n_best * 2)]))
    try_ns = [ln for ln in try_ns if 2000 <= ln <= 15000]
    try_ns.sort(key=lambda ln: abs(ln - n_best))

    for new_n in try_ns[:4]:
        rt = remaining_time()
        if rt < 100:
            break
        print("  Multi-n: n=" + str(new_n) + " (remaining=" + str(round(rt, 1)) + "s)")
        a_interp = interp_seq(best_seq, new_n)

        t_sub = min(rt * 0.15, 120.0)
        if t_sub > 20 and new_n <= 12000:
            a_out, _ = lp_fw(a_interp, t_sub, solver, initial_K=600, max_K=4000, add_per_iter=50, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "multin_" + str(new_n), t_global)

        rt = remaining_time()
        if rt > 40 and len(best_seq) <= 12000:
            t_sub2 = min(rt * 0.06, 50.0)
            if t_sub2 > 10:
                a_out, _ = nesterov_prox(best_seq, t_sub2, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "nesterov_multin", t_global)

    # === Phase 7: LP-FW second pass (longer) ===
    print("\n=== Phase 7: LP-FW second pass (n=" + str(len(best_seq)) + ") ===")
    t7 = min(remaining_time() * 0.22, 330.0)
    if t7 > 40 and len(best_seq) <= 12000:
        a_out, _ = lp_fw(best_seq, t7, solver, initial_K=1200, max_K=7000, add_per_iter=100, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2", t_global)

    # === Phase 8: Nesterov again on best ===
    print("\n=== Phase 8: Nesterov PG (n=" + str(len(best_seq)) + ") ===")
    t8 = min(remaining_time() * 0.10, 150.0)
    if t8 > 20:
        a_out, _ = nesterov_prox(best_seq, t8, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nesterov2", t_global)

    # === Phase 9: Perturbation + local optimization ===
    print("\n=== Phase 9: Perturbation search ===")
    t9 = min(remaining_time() * 0.12, 180.0)
    if t9 > 30:
        rng_pert = np.random.default_rng(2024)
        n_cur = len(best_seq)
        t9_start = time.monotonic()
        n_pert = 0
        while time.monotonic() - t9_start < t9 and n_pert < 20:
            n_pert += 1
            # Different perturbation scales
            scales = [0.01, 0.03, 0.05, 0.1, 0.15]
            sc_pert = scales[(n_pert - 1) % len(scales)]
            perturbed = best_seq + sc_pert * best_seq * rng_pert.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0.0)
            sp = perturbed.sum()
            if sp > 1e-10:
                perturbed = perturbed / sp
            # Quick local optimization
            t_local = min(t9 - (time.monotonic() - t9_start), 8.0)
            if t_local > 5:
                a_out, _ = lp_fw(perturbed, t_local, solver, initial_K=400, max_K=2000, add_per_iter=30, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "pert" + str(n_pert), t_global)

    # === Phase 10: Final LP-FW polish ===
    print("\n=== Phase 10: Final LP-FW ===")
    t10 = min(remaining_time() * 0.12, 150.0)
    if t10 > 30 and len(best_seq) <= 12000:
        a_out, _ = lp_fw(best_seq, t10, solver, initial_K=1500, max_K=8000, add_per_iter=120, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # === Final coordinate polish ===
    rt = remaining_time()
    if rt > 15:
        a_out, _ = coord_descent_polish(best_seq, min(rt * 0.5, 60.0), rng_seed=31415, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 8000
        best_seq = init_ramp(n_fb, 1.0)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
