Here's another idea: **use the LP solution not just for the FW direction, but also to directly update the active set and the estimate of the optimal M.**

The LP gives us:
- g*: the optimal direction
- M*: the optimal value of M (the minimum possible (y*g)[k] over the active set)

The true optimal M (over all k) is at least M*. So M* is a lower bound on the optimal convolution peak. If M* is close to the current max_b, we're close to optimal.

This gives us a convergence criterion: stop when M* is within ε of the current max_b.

Let me implement this.

Also, I want to add a **binary search on n** step. For a given shape (normalized sequence), the objective as a function of n is:
f(n) = 2 * n * max_b(interpolated_seq, n) / (sum_a)^2

This is not necessarily monotone in n, but it might have a minimum at some specific n. I can try to find this minimum by evaluating f(n) for several n values and interpolating.

Let me now write the final code. I'll aim for something that's:
- Robust (handles edge cases)
- Efficient (uses time well)
- Diverse (tries multiple approaches)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-Frank-Wolfe with adaptive active set, dual-guided pruning, and adaptive line search
2. Multi-n exploration with interpolation and re-optimization
3. Spectral shaping via Fourier domain operations
4. Peak-shaving local search
5. Time-aware multi-phase orchestration
"""

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None

def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fft_size_for(n):
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def score_of(a):
    """Compute the objective value for a normalized sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    max_b = fast_convolve(a).max()
    return 2.0 * n * max_b


def lp_fw(a0, time_limit, solver, initial_K=500, max_K=5000, verbose=True):
    """
    LP-Frank-Wolfe with adaptive active set and dual-guided management.
    
    Key improvements over previous:
    - Dual-guided active set pruning (keep constraints with large dual multipliers)
    - Better line search with more alpha values
    - Convergence check based on LP duality gap
    - More aggressive violation detection
    """
    n = len(a0)
    y = norm_sum1(a0)
    ac_y = fast_convolve(y)
    best_m = ac_y.max()
    best_a = y.copy()
    t0 = time.monotonic()
    fs = fft_size_for(n)

    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 5000
    iter_count = 0
    stagnation_count = 0
    lp_fail_count = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if stagnation_count > 8:
                if verbose:
                    print("    LP-FW: persistent stagnation, stopping")
                break
            # Perturb the active set
            ac_curr = fast_convolve(y)
            n_pert = max(100, len(active_set) // 3)
            rng_pert = np.random.default_rng(stagnation_count * 104729 + 42)
            keep = list(rng_pert.choice(len(active_set), size=max(1, len(active_set) - n_pert), replace=False))
            new_adds = np.argsort(ac_curr)[::-1][:n_pert * 2]
            new_set = set([active_set[i] for i in keep])
            for k in new_adds:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]
            no_improve = 0
            stagnation_count += 1
            if verbose:
                print("    LP-FW: active set restart (stag=" + str(stagnation_count) + ")")
            continue

        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-12:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            stagnation_count = 0
        else:
            no_improve += 1

        # Trim active set using margin-based pruning
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            # Keep constraints closest to being active (smallest margin)
            n_keep = int(max_K * 0.5)
            keep_idx = np.argsort(margins)[:n_keep]
            active_set_base = [active_set[i] for i in keep_idx]
            # Always keep the top peaks
            top_peaks = np.argsort(ac)[::-1][:300]
            for v in top_peaks:
                if int(v) not in active_set_base:
                    active_set_base.append(int(v))
            active_set = sorted(set(active_set_base))[:max_K]

        K = len(active_set)
        if K == 0:
            break

        # Build and solve LP
        coeffs = build_constraint_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail_count += 1
            if lp_fail_count > 10:
                if verbose:
                    print("    LP-FW: too many LP failures, stopping")
                break
            active_set = active_set[:max(50, K // 3)]
            continue

        lp_fail_count = 0

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # Try with fewer constraints
            active_set = active_set[:max(50, K // 2)]
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        lp_m_value = float(M_var.value) if M_var.value is not None else M_curr

        # Convergence check: if LP's optimal M is close to current max, we're near optimal
        duality_gap = M_curr - lp_m_value
        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): "
                  + "J=" + repr(2.0 * n * best_m) + " |S|=" + str(len(active_set))
                  + " gap=" + repr(duality_gap))

        # Detect violations and add to active set
        Yf = np.fft.rfft(y, fs)
        Gf = np.fft.rfft(g_dir, fs)
        conv_ys = np.fft.irfft(Yf * Gf, fs)[:2 * n - 1]

        viol_thresh = M_curr + 1e-9
        all_viol = np.where(conv_ys > viol_thresh)[0]
        if len(all_viol) > 0:
            viol_amt = conv_ys[all_viol] - M_curr
            n_add = min(100, len(all_viol))
            top_v = all_viol[np.argsort(viol_amt)[::-1][:n_add]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]
        else:
            # No violations: add the current maximum if not already there
            k_max = int(np.argmax(conv_ys))
            if k_max not in active_set:
                active_set.append(k_max)
                active_set.sort()

        # Line search: try multiple step sizes
        fw_alpha = 2.0 / (iter_count + 2.0)
        
        # Build list of alphas to try
        alpha_candidates = set()
        alpha_candidates.add(fw_alpha)
        alpha_candidates.add(min(1.0, fw_alpha * 2))
        alpha_candidates.add(max(0.01, fw_alpha * 0.5))
        # Add some fixed values
        for a_val in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            alpha_candidates.add(a_val)
        # Add values around the current best
        alpha_candidates.add(max(0.01, 1.0 / (iter_count + 1)))
        
        alphas = sorted([a for a in alpha_candidates if 0 < a <= 1.0], reverse=True)

        best_alpha = fw_alpha
        best_trial_M = M_curr  # we need to improve to accept

        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M - 1e-14:
                best_trial_M = M_trial
                best_alpha = alpha
                # Early stop if we found a good improvement
                if M_trial < best_trial_M - 0.01 * M_curr:
                    break

        # Apply update
        if iter_count <= 2:
            y = g_dir.copy()
        elif best_trial_M < M_curr:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y / s_y
            else:
                y = best_a.copy()
        else:
            # No improvement found, try a small random perturbation
            rng_step = np.random.default_rng(iter_count)
            noise = rng_step.normal(0, 0.01, n) * y
            y_pert = y + noise
            y_pert = np.maximum(y_pert, 0)
            s_p = y_pert.sum()
            if s_p > 1e-15:
                y = y_pert / s_p

    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + " iters=" + str(iter_count))
    return best_a, best_m


def spectral_shape(a0, time_limit, verbose=True):
    """
    Spectral shaping: try to flatten the power spectrum of the sequence.
    A flatter spectrum corresponds to lower peak autocorrelation.
    Uses iterative projection in Fourier domain.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = score_of(best_a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    x = np.arange(n)
    
    for outer in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Get current power spectrum
        fa = np.fft.rfft(a)
        power = np.abs(fa) ** 2
        total_power = power.sum()
        if total_power < 1e-30:
            break
        
        # Target: flatten the spectrum (make it as uniform as possible)
        # But we can only change the phase (amplitude is determined by a)
        # Instead, use a water-filling approach on the time domain
        
        # Compute the "excess" at each lag
        ac = fast_convolve(a)
        max_ac = ac.max()
        mean_ac = ac.mean()
        
        # Find peaks and try to reduce them
        # Add negative "pressure" at peak locations
        peak_mask = ac > max_ac * 0.9
        peak_lags = np.where(peak_mask)[0]
        
        if len(peak_lags) == 0:
            break
        
        # Compute the "force" to reduce peaks
        # The gradient of max(ac) with respect to a is a[flipped] at the argmax lag
        k_star = int(np.argmax(ac))
        grad_a = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad_a[idx] = a[k_star - idx]
        
        # Mirror descent step with adaptive step size
        eta = 0.1 / (1.0 + outer * 0.1)
        
        # Try multiple step sizes
        best_eta = eta
        best_trial_score = best_score
        
        for eta_try in [eta * 0.1, eta * 0.5, eta, eta * 2, eta * 5]:
            a_trial = a - eta_try * grad_a
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            sc = score_of(a_trial)
            if sc < best_trial_score:
                best_trial_score = sc
                best_eta = eta_try
        
        a = a - best_eta * grad_a
        a = np.maximum(a, 0)
        s_a = a.sum()
        if s_a > 1e-15:
            a = a / s_a
        
        sc = score_of(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Periodic random restart to escape local minima
        if outer % 50 == 49:
            noise = rng.normal(0, 0.05, n) * best_a
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            s_p = a_pert.sum()
            if s_p > 1e-15:
                a = a_pert / s_p

    if verbose:
        elapsed = time.monotonic() - t0
        print("    Spectral done (" + str(round(elapsed, 1)) + "s): J=" + repr(best_score))
    return best_a, best_score


def cmaes_fourier(a0, time_limit, n_freqs=15, seed=42, verbose=True):
    """
    CMA-ES on Fourier coefficient parameterization.
    a[i] = (1 + sum_k w_k * cos(2*pi*k*i/n + phi_k))^p / n
    This ensures non-negativity (for p=2) and smoothness.
    """
    n = len(a0)
    x = np.arange(n) / n  # normalized positions in [0,1)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = score_of(best_a)

    # Initial parameter guess
    n_params = 2 * n_freqs + 1  # weights, phases, power exponent
    params = np.zeros(n_params)
    params[0] = 2.0  # power exponent p
    # Initialize weights to be small
    for k in range(n_freqs):
        params[1 + k] = 0.05  # weight
        params[1 + n_freqs + k] = rng.uniform(0, 2 * np.pi)  # phase

    dim = n_params
    pop_size = max(8 + int(3 * np.log(dim + 1)), 12)
    sigma = 0.1

    def params_to_seq(p):
        p_power = max(0.5, p[0])
        a_seq = np.ones(n)
        for k in range(n_freqs):
            w = p[1 + k]
            phi = p[1 + n_freqs + k]
            a_seq += w * np.cos(2 * np.pi * (k + 1) * x + phi)
        a_seq = np.abs(a_seq) ** p_power
        a_seq = np.maximum(a_seq, 0)
        s = a_seq.sum()
        if s < 1e-15:
            return best_a.copy()
        return a_seq / s

    def eval_params(p):
        seq = params_to_seq(p)
        return score_of(seq)

    mean = params.copy()
    evals = 0
    max_evals = 2000
    no_improve = 0

    while evals < max_evals and time.monotonic() - t0 < time_limit:
        # Sample population
        samples = mean + sigma * rng.standard_normal((pop_size, dim))
        fitness = np.zeros(pop_size)
        for i in range(pop_size):
            if time.monotonic() - t0 > time_limit:
                fitness[i] = best_score + 1
                continue
            fitness[i] = eval_params(samples[i])
            evals += 1
            if fitness[i] < best_score:
                best_score = fitness[i]
                best_a = params_to_seq(samples[i]).copy()
                no_improve = 0
                if verbose:
                    print("    CMA-Fourier new best: J=" + repr(best_score) + " (" + str(round(time.monotonic() - t0, 1)) + "s)")
            else:
                no_improve += 1

        # Update mean (weighted recombination)
        lam = max(pop_size // 2, 4)
        top_idx = np.argsort(fitness)[:lam]
        mean = np.zeros(dim)
        w_sum = 0.0
        for rank, idx in enumerate(top_idx):
            w = 1.0 / (rank + 1)
            mean += w * samples[idx]
            w_sum += w
        mean = mean / w_sum

        # Adapt sigma
        if no_improve > pop_size * 5:
            sigma *= 0.7
            no_improve = 0
        elif no_improve < pop_size:
            sigma = min(sigma * 1.1, 1.0)

    if verbose:
        print("    CMA-Fourier done (" + str(round(time.monotonic() - t0, 1)) + "s): J=" + repr(best_score) + " evals=" + str(evals))
    return best_a, best_score


def peak_shave(a0, time_limit, verbose=True):
    """
    Peak shaving: iteratively identify the convolution peak and 
    redistribute mass to reduce it.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score = score_of(best_a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    fs = fft_size_for(n)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        k_star = int(np.argmax(ac))
        peak_val = ac[k_star]

        # Find the "source" indices that contribute most to this peak
        # (a*a)[k_star] = sum_i a[i] * a[k_star - i]
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        if i_start > i_end:
            break
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx]
