def self_convolve(a):
    """Full self-convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho_val = rho_indices[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Adaptive top-k: increase over time
        progress = min(1.0, t / 5000)
        top_k = int(top_k_start + (top_k_end - top_k_start) * progress)
        
        # Nesterov momentum
        momentum = (t - 1) / (t + 2)  # standard FISTA
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y / y_sum  # Keep y on simplex for gradient evaluation
        else:
            y = a
        
        # Compute gradient at y
        b_y = self_convolve(y)
        grad = compute_smoothed_gradient(y, b_y, top_k)
        
        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        eta = min(1.0 / (grad_norm * 0.5 + 1e-10), 0.1)
        
        # Gradient step + projection
        a_new = simplex_project(y - eta * grad)
        
        # Check improvement
        b_new = self_convolve(a_new)
        m_new = b_new.max()
        
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a_prev = a
        a = a_new
        
        # Restart if stalled
        if stall > 300:
            stall = 0
            noise = 0.01 / (1.0 + stall * 0.1)
            pert = rng.normal(0, noise, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r
            a_prev = a_r.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m  # assumes sum = 1
            print(f"    FISTA t={t} ({elapsed:.1f}s): J={J:.6f}, top_k={top_k}")
    
    return best_a, best_m

def peak_shaving_fast(a0, time_limit, top_k=20, verbose=True):
    """
    Fast peak shaving using incremental convolution updates.
    """
    n = len(a0)
    a = simplex_project(a0.copy())
    best_a = a.copy()
    b = self_convolve(a)
    best_m = b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        # Find top peaks
        peak_idx = np.argpartition(b, -top_k)[-top_k:]
        peak_idx = peak_idx[np.argsort(b[peak_idx])[::-1]]
        
        improved = False
        for k_star in peak_idx[:10]:
            k_int = int(k_star)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            # Contribution of each position to this peak
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            # Find the position with max contribution
            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])
            
            # Find a good destination: position where a[k_int - j] is minimum
            # and j is not near src
            a_partner = a[k_int - idx]  # a[k_int - i] for i in idx
            # We want to move mass from src to a position j where a[k_int - j] is small
            # and j is far from the peak interaction
            a_vals = a[idx].copy()
            a_vals[src_local] = -1  # exclude src
            # Also exclude positions that are "dangerous" for other peaks
            dst_local = int(np.argmin(a_vals * a_partner))
            if a_vals[dst_local] <= 0:
                # Find next best
                a_vals[dst_local] = -1
                dst_local = int(np.argmin(a_vals * a_partner))
            if a_vals[dst_local] <= 0:
                continue
            dst = int(idx[dst_local])
            if dst == src:
                continue
            
            # Try moving mass
            found = False
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3]:
                amt = a[src] * frac
                if amt < 1e-14:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-12:
                    continue
                a_trial = a_trial / s_t
                b_trial = self_convolve(a_trial)
                m_trial = b_trial.max()
                if m_trial < best_m - 1e-14:
                    a = a_trial
                    b = b_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            
            if found:
                continue  # Recompute peaks after a successful move
            # If no move worked for this peak, try a random perturbation
            if stall > 50:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    amt = a[src] * 0.05
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    b = self_convolve(a)
                    m = b.max()
                    if m < best_m:
                        best_m = m
                        best_a = a.copy()
                        stall = 0
                        improved = True
        
        if not improved:
            stall += 1
            if stall % 30 == 0:
                # Random restart with increasing noise
                noise = min(0.05, 0.005 * (stall // 30))
                pert = rng.normal(0, noise, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-10)
                a_r = a_r / a_r.sum()
                b_r = self_convolve(a_r)
                if b_r.max() < best_m:
                    best_m = b_r.max()
                    best_a = a_r.copy()
                a = best_a.copy()
                b = self_convolve(a)
        
        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def interpolate_to_length(a, new_n):
    """Interpolate sequence to a new length, preserving sum."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s_new
