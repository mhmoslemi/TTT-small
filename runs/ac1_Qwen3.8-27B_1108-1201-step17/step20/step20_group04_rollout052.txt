def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    """Compute gradient of (1/beta)*log(sum top_k exp(beta*c)) w.r.t. a."""
    n = len(y)
    c = fast_autocorr(y)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(y, fft_size)
    P = np.fft.rfft(p, fft_size)
    a_rev = y[::-1]
    Conv = np.fft.irfft(P * np.fft.rfft(a_rev, fft_size), fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad, p, c

                    if s_t < 1e-15:

                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:

            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    UFP (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search combining Nesterov, gradient local search, pressure surgery, and ultra-fine polish."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        x = np.linspace(0, 1, n_base)
        a_init = (1.0 - x) ** 1.5 + 0.1
        best_seq = norm_sum1(a_init)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(n_base) + ", score=" + repr(best_score))

    n0 = len(best_seq)
    print("Starting: n=" + str(n0) + ", score=" + repr(best_score))

    # Phase 1: Nesterov with moderate beta (broad exploration)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Nesterov moderate (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.20, 340.0)
    if t1 > 60:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t1,
                                beta_start=15.0, beta_end=2000.0,
                                beta_growth=1.004, eta_base=0.5,
                                top_k_ratio=0.02, restart_every=1200,
                                verbose=True, seed=1)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 2: Pressure-guided surgery
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: Pressure Surgery (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining * 0.12, 200.0)
    if t2 > 40:
        a_out, _ = pressure_guided_surgery(norm_sum1(best_seq), t2, verbose=True, seed=11)
        best_seq, best_score = try_update(best_seq, best_score, "ps1", t_global) if False else try_update(best_seq, best_score, a_out, "ps1", t_global)

    # Phase 3: Nesterov with high beta (focused optimization)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: Nesterov high-beta (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining * 0.20, 340.0)
    if t3 > 60:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t3,
                                beta_start=100.0, beta_end=6000.0,
                                beta_growth=1.008, eta_base=0.4,
                                top_k_ratio=0.01, restart_every=1000,
                                verbose=True, seed=2)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)

    # Phase 4: Gradient local search
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 4: Gradient Local Search (n=" + str(len(best_seq)) + ") ===")
    t4 = min(remaining * 0.10, 170.0)
    if t4 > 30:
        a_out, _ = gradient_local_search(norm_sum1(best_seq), t4, verbose=True, seed=33)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "gls1", t_global)

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-length (current n=" + str(n_best) + ") ===")
    lengths_to_try = []
    # Try slightly longer
    lengths_to_try.append(min(n_best + 800, 8500))
    # Try a bit shorter
    if n_best > 5000:
        lengths_to_try.append(max(5500, n_best - 1000))
    # Deduplicate and filter
    lengths_to_try = sorted(set(l for l in lengths_to_try if 4000 <= l <= 10000 and l != n_best))

    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("  Trying n=" + str(new_n) + " (remaining=" + str(round(remaining, 1)) + "s)")
        a_long = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.40, 250.0)
        if t5a > 50:
            a_out, _ = nesterov_opt(a_long, t5a,
                                    beta_start=30.0, beta_end=3000.0,
                                    beta_growth=1.005, eta_base=0.5,
                                    top_k_ratio=0.015, restart_every=1200,
                                    verbose=True, seed=5)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ml_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80:
            t5b = min(remaining * 0.35, 120.0)
            if t5b > 30:
                a_try = interpolate_sequence(best_seq, new_n)
                a_out, _ = pressure_guided_surgery(a_try, t5b, verbose=False, seed=33)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "mlps_n" + str(new_n), t_global)

    # Phase 6: Nesterov final refinement on best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Nesterov refine (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.15, 250.0)
    if t6 > 60:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t6,
                                beta_start=200.0, beta_end=10000.0,
                                beta_growth=1.010, eta_base=0.3,
                                top_k_ratio=0.005, restart_every=800,
                                verbose=True, seed=7)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest3", t_global)

    # Phase 7: Pressure surgery on best
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 7: Pressure Surgery 2 (n=" + str(len(best_seq)) + ") ===")
    t7 = min(remaining * 0.12, 180.0)
    if t7 > 30:
        a_out, _ = pressure_guided_surgery(norm_sum1(best_seq), t7, verbose=True, seed=44)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps2", t_global)

    # Phase 8: Ultra-fine polish
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 8: Ultra-fine polish (n=" + str(len(best_seq)) + ") ===")
    t8 = min(remaining * 0.25, 300.0)
    if t8 > 30:
        a_out, _ = ultra_fine_polish(norm_sum1(best_seq), t8, verbose=True, seed=55)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ufp1", t_global)

    # Phase 9: Final gradient local search
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 9: Final GLS (n=" + str(len(best_seq)) + ") ===")
    t9 = min(remaining * 0.15, 150.0)
    if t9 > 20:
        a_out, _ = gradient_local_search(norm_sum1(best_seq), t9, verbose=False, seed=66)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "gls2", t_global)

    # Phase 10: Final ultra-fine polish
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 10: Final UFP (n=" + str(len(best_seq)) + ") ===")
    t10 = min(remaining * 0.40, 200.0)
    if t10 > 20:
        a_out, _ = ultra_fine_polish(norm_sum1(best_seq), t10, verbose=True, seed=77)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ufp2", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1((1.0 - x) ** 1.5 + 0.1)
        best_score = evaluate_safe(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)

    return [float(x) for x in best_seq]
