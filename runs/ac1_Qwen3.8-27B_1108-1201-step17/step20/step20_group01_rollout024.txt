def smoothed_nesterov(a0, time_limit, T0=5.0, T_max=200.0, lr0=0.1, verbose=True):
    n = len(a0)
    a = normalize(a0)
    a_prev = a.copy()
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    
    T = T0
    lr = lr0
    momentum = 0.9
    t0 = time.monotonic()
    no_improve = 0
    stall_limit = 5000
    
    S = 1 << (3 * n - 2).bit_length()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        # Nesterov: extrapolate
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = a.copy()
        
        # Compute smoothed gradient at y
        c = fast_autocorr(y)
        p = softmax_weights(c, T)
        q_rev = p[::-1]
        
        A_fft = np.fft.rfft(y, S)
        Q_fft = np.fft.rfft(q_rev, S)
        result = np.fft.irfft(A_fft * Q_fft, S)
        g = 2.0 * result[2*n-2 : n-2 : -1]
        
        # Gradient step
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn > 1e-15:
            a_new = a_new / sn
        else:
            a_new = a.copy()
        
        # Evaluate
        obj_new = 2.0 * n * fast_autocorr(a_new).max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            # Slightly increase momentum and T
            momentum = min(0.99, momentum * 1.001)
        else:
            no_improve += 1
            # Decrease lr if stuck
            if no_improve > 100:
                lr *= 0.95
                if no_improve > 500:
                    T *= 0.8  # Reduce temperature to smooth out
                    momentum = 0.8
                    # Restart from best
                    a = best_a.copy()
                    a_prev = a.copy()
                    lr = lr0
                    momentum = 0.9
                    no_improve = 0
        
        a_prev = a
        a = a_new
        
        # Periodically increase temperature
        if t % 1000 == 0:
            T = min(T_max, T * 1.05)
        
        if verbose and t % 500 == 0:
            print("  SmoothNest t={} T={:.1f} lr={:.4f} J={:.7f} no_imp={}".format(
                t, T, lr, best_obj, no_improve))
    
    return best_a, best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. FFT-accelerated smoothed subgradient with Nesterov acceleration
2. LP-based Frank-Wolfe with adaptive active set
3. Hybrid LP+gradient iterations (LP every K gradient steps)
4. Coordinate peeling for local refinement
5. Multi-restart strategy with adaptive hyperparameters

# Global solver cache
_SOLVER = None

def get_solver():
    global _SOLVER
    if _SOLVER is not None:
        return _SOLVER
    for s in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except:

    _SOLVER = cp.ECOS

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def softmax_w(c, T):
    cm = c.max()
    e = T * (c - cm)
    e = np.clip(e, -50, 0)
    w = np.exp(e)
    w /= w.sum()
    return w

def smooth_grad(a, T):
    """Compute gradient of smoothed max autocorrelation. Returns (gradient, autocorr, weights)."""
    n = len(a)
    c = fast_autocorr(a)
    p = softmax_w(c, T)
    q_rev = p[::-1]
    S = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, S)
    Q = np.fft.rfft(q_rev, S)
    res = np.fft.irfft(A * Q, S)
    g = 2.0 * res[2*n-2 : n-2 : -1]
    return g, c, p

def objective(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):

def nesterov_smooth(a0, tl, T0=10.0, Tmax=500.0, lr0=0.5, seed=42, verbose=True):
    """Nesterov-accelerated smoothed gradient descent."""
    n = len(a0)
    a = normalize(a0)
    a_prev = a.copy()
    best_a = a.copy()
    best_obj = objective(a)
    T = T0
    lr = lr0
    mu = 0.9  # momentum
    t0 = time.monotonic()
    no_imp = 0
    S = 1 << (3 * n - 2).bit_length()
    rng = np.random.default_rng(seed)
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > tl:
            break
        if no_imp > 10000:
            break
        
        # Nesterov extrapolation
        y = a + mu * (a - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y /= sy
        else:
            y = a.copy()
        
        # Smoothed gradient at y
        g, c, p = smooth_grad(y, T)
        
        # Gradient step
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn > 1e-15:
            a_new /= sn
        else:
            a_new = a.copy()
        
        obj_new = objective(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_imp = 0
            # Increase T gradually
            T = min(Tmax, T * 1.002)
        else:
            no_imp += 1
            if no_imp > 200:
                lr *= 0.9
                if no_imp > 1000:
                    # Restart with perturbation
                    pert = 0.01 * rng.standard_normal(n) * best_a
                    a = best_a + pert
                    a = np.maximum(a, 0)
                    a /= a.sum()
                    a_prev = a.copy()
                    lr = lr0
                    mu = 0.9
                    T = max(T0, T * 0.5)
                    no_imp = 0
        
        a_prev = a
        a = a_new
        
        if verbose and t % 2000 == 0:
            print("  Nest t={} T={:.1f} lr={:.4f} J={:.8f} no_imp={}".format(
                t, T, lr, best_obj, no_imp))
    
    return best_a, best_obj

def lp_fw(a0, tl, solver, init_K=200, max_K=5000, verbose=True):
    """LP-based Frank-Wolfe with adaptive active set."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_obj = objective(y)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Initialize active set with top-K lags
    ac = fast_autocorr(y)
    M_curr = ac.max()
    active_set = set(int(k) for k in np.argsort(ac)[::-1][:init_K])
    no_imp = 0
    stall_limit = 3000
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > tl:
            break
        if no_imp > stall_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        cur_obj = 2.0 * n * M_curr
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Update active set
        if t % 20 == 0 or t == 1:
            target = min(max_K, init_K + t * 3)
            top_lags = np.argsort(ac)[::-1]
            for k in top_lags[:target]:
                active_set.add(int(k))
            if len(active_set) > max_K:
                clist = sorted(active_set)
                margins = np.array([M_curr - ac[k] for k in clist])
                keep = np.argsort(margins)[:max_K]
                active_set = set(int(clist[i]) for i in keep)
        active_set.add(int(np.argmax(ac)))
        
        clist = sorted(active_set)
        K = len(clist)
        
        # Build LP
        try:
            coeffs = np.zeros((K, n), dtype=np.float64)
            for jj in range(K):
                ki = clist[jj]
                i_start = max(0, ki - n + 1)
                i_end = min(n, ki + 1)
                if i_start < i_end:
                    idx = np.arange(i_start, i_end)
                    coeffs[jj, idx] = y[ki - idx]
            
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            prob.solve(solver=solver, verbose=False)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                raise Exception("LP fail: " + str(prob.status))
        except Exception as e:
            if len(active_set) > init_K // 2:
                clist = sorted(active_set)
                active_set = set(int(clist[i]) for i in np.argsort([M_curr - ac[kii] for kii in clist])[:init_K])
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = g_val.sum()
        if gs < 1e-15:
            continue
        g_dir = g_val / gs
        
        # Check violations
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:size]
        viol = np.where(conv_yg > M_curr + 1e-10)[0]
        if len(viol) > 0:
            va = conv_yg[viol] - M_curr
            top_v = viol[np.argsort(va)[::-1][:30]]
            for k in top_v:
                active_set.add(int(k))
        
        # Line search: try multiple alphas
        best_alpha = 2.0 / (t + 2.0)
        best_mt = 1e10
        for alpha in [best_alpha, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            trial = np.maximum((1 - alpha) * y + alpha * g_dir, 0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            Mt = fast_autocorr(trial / ts).max()
            if Mt < best_mt:
                best_mt = Mt
                best_alpha = alpha
        
        # Bisection refinement
        lo = max(0.0, best_alpha - 0.1)
        hi = min(1.0, best_alpha + 0.1)
        for _ in range(5):
            m1 = lo + (hi - lo) / 3
            m2 = hi - (hi - lo) / 3
            t1 = np.maximum((1 - m1) * y + m1 * g_dir, 0)
            t2 = np.maximum((1 - m2) * y + m2 * g_dir, 0)
            s1, s2 = t1.sum(), t2.sum()
            mm1 = fast_autocorr(t1/s1).max() if s1 > 1e-15 else 1e10
            mm2 = fast_autocorr(t2/s2).max() if s2 > 1e-15 else 1e10
            if mm1 < mm2:
                hi = m2
            else:
                lo = m1
        best_alpha = (lo + hi) / 2
        
        y = np.maximum((1 - best_alpha) * y + best_alpha * g_dir, 0)
        sy = y.sum()
        if sy > 1e-15:
            y /= sy
        
        if verbose and t % 100 == 0:
            print("  LP-FW t={} ({:.1f}s) J={:.8f} |S|={}".format(
                t, time.monotonic()-t0, best_obj, len(active_set)))
    
    return best_a, best_obj

def coord_peel(a0, tl, n_targets=15, seed=42, verbose=True):
    """Coordinate peeling for local refinement."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_obj = objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 5000
    fracs = [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05]
    
    while time.monotonic() - t0 < tl and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(n_targets, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > tl:
                break
            ki = int(k)
            ws = max(0, ki)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-25:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-15:
                continue
            
            # Find quiet destinations
            # Compute pressure on each position from top lags
            pr = np.zeros(n, dtype=np.float64)
            for kk in top_k[:8]:
                kk = int(kk)
                ws2 = max(0, kk)
                we2 = min(n - 1, kk)
                if ws2 <= we2:
                    idx2 = np.arange(ws2, we2 + 1)
                    pr[idx2] += a[kk - idx2]
            pr[src] = 1e10
            quiet = np.argsort(pr)[:50]
            
            for dst in quiet:
                dst = int(dst)
                if dst == src:
                    continue
                found = False
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-18:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t /= st
                    Mt = objective(a_t)
                    if Mt < best_obj - 1e-16:
                        a = a_t
                        best_obj = Mt
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break
            
            if not improved and stall > 100:
                # Random kick
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    st = a.sum()
                    if st > 1e-15:
                        a /= st
                        Mn = objective(a)
                        if Mn < best_obj:
                            best_obj = Mn
                            best_a = a.copy()
                            stall = 0
                            improved = True
                break
        
        if not improved:
            stall += 1
            if stall > 500:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    st = a.sum()
                    if st > 1e-15:
                        a /= st
                        Mn = objective(a)
                        if Mn < best_obj:
                            best_obj = Mn
                            best_a = a.copy()
                            stall = 0
        
        if verbose and stall > 0 and stall % 1000 == 0:
            print("  Peel ({:.1f}s) J={:.8f} stall={}".format(time.monotonic()-t0, best_obj, stall))
    
    return best_a, best_obj
