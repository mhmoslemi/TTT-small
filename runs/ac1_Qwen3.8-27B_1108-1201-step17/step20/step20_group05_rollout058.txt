"""
Multi-stage optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines LSE-smoothed L-BFGS, Frank-Wolfe with adaptive active set (MW-based),
direct subgradient steps, coordinate polish, and length variation with restarts.

import numpy as np
import time
import math

try:
    import scipy.optimize
    SCIPY_OK = True
except Exception:
    SCIPY_OK = False


def next_pow2(x):
    x = int(max(1, x))
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm1(a):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    np.maximum(a, 0.0, out=a)
    a = np.where(np.isfinite(a), a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return np.maximum(np.where(np.isfinite(c), c, 0.0), 0.0)


def fft_cross(a, b):
    n = max(a.size, b.size)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    return np.maximum(np.where(np.isfinite(c), c, 0.0), 0.0)


def cross_corr_p_a(p, a):
    """Compute grad-like: (p * a)_i = sum_j p[i+j] * a[j] for i=0..n-1"""
    p = np.asarray(p, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    total = (p.size) + n - 1
    N = next_pow2(total)
    P = np.fft.rfft(p, N)
    Af = np.fft.rfft(a[::-1], N)
    corr = np.fft.irfft(P * Af, N)
    out = corr[n - 1:n - 1 + n]
    return np.where(np.isfinite(out), out, 0.0)

    c = fft_conv(a)
    s = float(a.sum())
    if s < 1e-30:
        return float('inf')
    return 2.0 * n * float(c.max()) / (s * s)


def lse_obj_grad(a, beta):
    n = a.size
    c = fft_conv(a)
    size = c.size
    m = float(c.max())
    shifted = beta * (c - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0 or not math.isfinite(se):
        return m, np.zeros(n, dtype=np.float64)
    p = e / se
    f = m + math.log(se) / beta
    grad = 2.0 * cross_corr_p_a(p, a)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return f, grad


def softmax_u(u):
    u = np.asarray(u, dtype=np.float64)
    n = u.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(u)):

    um = float(u.max())
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / n
    return e / s

def run_lbfgs_v(v0, beta, ref, maxfun=300, reg_u=1e-10):

        return v0
    v0 = np.asarray(v0, dtype=np.float64)

    def _fv(v):
        f, g = obj_v(v, beta, ref)
        f += 0.5 * reg_u * float(np.dot(v, v))
        g = g + reg_u * v
        return f, g

    try:

            _fv, v0, jac=True, method='L-BFGS-B',
            options={'maxfun': int(maxfun), 'maxiter': int(maxfun + 20),

def lbfgs_phase(a0, betas, evals_per_beta=200):
    a = norm1(a0)
    v, ref = make_v(a)
    for beta in betas:
        v = run_lbfgs_v(v, float(beta), ref, maxfun=evals_per_beta)
        a_new = from_v(v, ref)
        a = norm1(a_new)
        v, ref = make_v(a)
    return a


def fw_mw_direction(f, active_idx, mw_iters=40, eta=8.0, diff_weight=0.02):
    f = norm1(f)
    n = f.size
    size = 2 * n - 1
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K == 0:
        return np.ones(n, dtype=np.float64) / n

    logits_p = np.zeros(K, dtype=np.float64)
    logits_g = np.zeros(n, dtype=np.float64)

    av = f[active_idx] if active_idx[0] < f.size else np.zeros(K)
    av = np.where(np.isfinite(av) & (av > 0), av, 1e-30)
    logits_p = np.log(av) - float(np.max(np.log(av)))
    logits_p = np.where(np.isfinite(logits_p), logits_p, 0.0)

    for _ in range(mw_iters):
        p = np.exp(logits_p - float(logits_p.max()))
        ps = float(p.sum())
        if ps <= 0:
            p = np.ones(K) / K
        else:
            p = p / ps

        pf = np.zeros(size, dtype=np.float64)
        pf[active_idx] = p

        pressure = cross_corr_p_a(pf, f)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0), pressure, 0.0)

        pmx = float(pressure.max())
        pmn = float(pressure.min())
        ptp = pmx - pmn
        if ptp > 1e-30:
            scaled = (pressure - pmn) / ptp
            logits_g += eta * (-scaled)

        logits_g = np.where(np.isfinite(logits_g), logits_g, 0.0)
        logits_g -= float(logits_g.max())
        w = np.exp(logits_g)
        ws = float(w.sum())
        if ws <= 0:
            g = np.ones(n) / n
        else:
            g = w / ws
            g = (1.0 - diff_weight) * g + diff_weight * (np.ones(n) / n)
            g = g / float(g.sum())

        fg = fft_cross(f, g)
        ha = fg[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hm = float(ha.mean())
        htp = float(ha.max()) - float(ha.min())
        if htp > 1e-30:
            logits_p += eta * ((ha - hm) / htp)

        logits_p = np.where(np.isfinite(logits_p), logits_p, 0.0)

    return g


def line_search_fw(a, g, c, cg, fg, M, alphas=None):
    if alphas is None:
        alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
    best_alpha = 0.0
    best_m = M
    for alpha in alphas:
        one = 1.0 - alpha
        val = (one * one) * c + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
        m_val = float(val.max())
        if m_val < best_m:
            best_m = m_val
            best_alpha = alpha
    if best_alpha <= 0:
        return a.copy(), M
    new_a = (1.0 - best_alpha) * a + best_alpha * g
    new_a = np.maximum(new_a, 0.0)
    s = float(new_a.sum())
    if s < 1e-30:
        return a.copy(), M
    return new_a / s, best_m


def frank_wolfe(a0, budget_s, active_max=3000, mw_iters=35, verbose=False):
    t0 = time.monotonic()
    a = norm1(a0)
    n = a.size
    c = fft_conv(a)
    M = float(c.max())
    best_a = a.copy()
    best_M = M
    active = np.argsort(c)[::-1][:min(active_max, c.size)]
    no_improve = 0
    t = 0

    while time.monotonic() - t0 < budget_s - 3 and no_improve < 120:
        t += 1
        time_left = budget_s - (time.monotonic() - t0)
        if time_left < 10:
            break

        g = fw_mw_direction(a, active, mw_iters=mw_iters, eta=8.0, diff_weight=0.02)
        if g is None:
            break

        cg = fft_conv(g)
        fg = fft_cross(a, g)
        new_a, new_M = line_search_fw(a, g, c, cg, fg, M)

        if new_M < M - 1e-15:
            alpha = 0.1
            one = 1.0 - alpha
            c = (one * one) * c + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
            c = np.maximum(c, 0.0)
            a = new_a
            M = new_M
            no_improve = 0
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
            active = np.argsort(c)[::-1][:min(active_max, c.size)]
            if verbose and (t <= 3 or t % 20 == 0):
                print("  FW t=%d M=%.8f score=%.6f" % (t, M, 2.0 * n * M))
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                rng = np.random.default_rng(1234 + t)
                pert = rng.normal(0, 0.02, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                c = fft_conv(a)
                M = float(c.max())
                active = np.argsort(c)[::-1][:min(active_max, c.size)]
                no_improve = 0

    return best_a


def subgradient_step(a, c, K, lr=None):
    n = a.size
    K_idx = np.argsort(c)[::-1][:K]
    pf = np.zeros(c.size, dtype=np.float64)
    pf[K_idx] = 1.0 / K
    s = 2.0 * cross_corr_p_a(pf, a)
    s = np.where(np.isfinite(s), s, 0.0)
    if lr is None:
        smax = float(s.max())
        if smax < 1e-30:
            return a
        lr = 0.5 / smax
        lr = min(lr, 0.1)
    new_a = a - lr * s
    new_a = np.maximum(new_a, 0.0)
    ns = float(new_a.sum())
    if ns < 1e-30:
        return a
    return new_a / ns


def coord_polish(a0, budget_s, verbose=False):
    t0 = time.monotonic()
    a = norm1(a0)
    c = fft_conv(a)
    M = float(c.max())
    best_a = a.copy()
    best_M = M
    n = a.size
    stall = 0
    rnd = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < budget_s - 2 and stall < 80:
        rnd += 1
        top_k = min(20, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        pf = np.zeros(c.size, dtype=np.float64)
        pf[top_idx] = 1.0 / top_k
        pressure = cross_corr_p_a(pf, a)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0), pressure, 0.0)

        best_move = None
        best_move_M = M

        src_score = pressure * a
        top_src = np.argpartition(src_score, min(6, n - 1))[:min(6, n)]
        top_dst = np.argpartition(-pressure, min(8, n - 1))[:min(8, n)]

        for s_i in top_src:
            s_i = int(s_i)
            if a[s_i] < 1e-12:
                continue
            for d_i in top_dst:
                d_i = int(d_i)
                if d_i == s_i:
                    continue
                for frac in (0.01, 0.02, 0.025, 0.03, 0.05):
                    tmove = a[s_i] * frac
                    if tmove <= 1e-16:
                        continue
                    tmove = min(tmove, a[s_i] * 0.95)
                    c_new = c.copy()
                    d = d_i
                    s = s_i
                    c_new[d:d + n] += 2.0 * tmove * a
                    c_new[s:s + n] -= 2.0 * tmove * a
                    if 2 * d < c_new.size:
                        c_new[2 * d] += tmove * tmove
                    if 2 * s < c_new.size:
                        c_new[2 * s] += tmove * tmove
                    if s + d < c_new.size:
                        c_new[s + d] -= 2.0 * tmove * tmove
                    c_new = np.maximum(c_new, 0.0)
                    m_new = float(c_new.max())
                    if m_new < best_move_M - 1e-15:
                        best_move_M = m_new
                        best_move = (s, d, tmove)

        if best_move is not None:
            s, d, tmove = best_move
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            c = fft_conv(a)
            M = float(c.max())
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and (rnd <= 3 or rnd % 30 == 0):
                print("  Coord rnd=%d M=%.8f" % (rnd, M))
        else:
            stall += 1
            if stall % 10 == 0:
                for _ in range(50):
                    si = int(rng.integers(n))
                    di = int(rng.integers(n))
                    if si == di or a[si] < 1e-12:
                        continue
                    tmove = a[si] * float(rng.uniform(0.005, 0.03))
                    tmove = min(tmove, a[si] * 0.9)
                    c_new = c.copy()
                    c_new[di:di + n] += 2.0 * tmove * a
                    c_new[si:si + n] -= 2.0 * tmove * a
                    if 2 * di < c_new.size:
                        c_new[2 * di] += tmove * tmove
                    if 2 * si < c_new.size:
                        c_new[2 * si] += tmove * tmove
                    if si + di < c_new.size:
                        c_new[si + di] -= 2.0 * tmove * tmove
                    c_new = np.maximum(c_new, 0.0)
                    m_new = float(c_new.max())
                    if m_new < M - 1e-14:
                        a[si] -= tmove
                        a[di] += tmove
                        a = np.maximum(a, 0.0) / a.sum()
                        c = fft_conv(a)
                        M = float(c.max())
                        if M < best_M - 1e-14:
                            best_M = M
                            best_a = a.copy()
                        break

    return best_a


def interpolate_seq(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    new_n = int(max(2, new_n))
    if a.size == new_n:
        return norm1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm1(y)


def default_seq(n=5000):
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.sin(np.pi * x) ** 2
    return norm1(a)


def try_update(best_seq, best_score, a, label, t0):
    if a is None:
        return best_seq, best_score
    try:
        arr = norm1(a)
        sc = true_score(arr)
        if math.isfinite(sc) and sc < best_score - 1e-10:
            best_seq = arr.copy()
            best_score = float(sc)
            print("  NEW BEST [%s]: %.8f n=%d t=%.1fs" % (
                label, sc, arr.size, time.monotonic() - t0))
    except Exception:
        pass
    return best_seq, best_score


def full_optimize(a0, time_budget, verbose=True):
    """Run the full pipeline on a sequence."""
    a = norm1(a0)
    n = a.size

    # Phase A: LSE with increasing beta
    t_start = time.monotonic()
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed

    if remaining > 200:
        betas = [5, 10, 20, 50, 100, 200]
        tick = time.monotonic()
        a = lbfgs_phase(a, betas, evals_per_beta=150)
        if verbose:
            print("    LSE done in %.1fs, score=%.6f" % (
                time.monotonic() - tick, true_score(a)))

    # Phase B: Frank-Wolfe
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed
    if remaining > 100:
        fw_budget = min(remaining * 0.45, 400.0)
        tick = time.monotonic()
        a_fw = frank_wolfe(a, fw_budget, active_max=2500, mw_iters=35, verbose=verbose)
        if true_score(a_fw) < true_score(a):
            a = a_fw
        if verbose:
            print("    FW done in %.1fs, score=%.6f" % (
                time.monotonic() - tick, true_score(a)))

    # Phase C: Coordinate polish
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed
    if remaining > 60:
        cp_budget = min(remaining * 0.35, 200.0)
        tick = time.monotonic()
        a_cp = coord_polish(a, cp_budget, verbose=False)
        if true_score(a_cp) < true_score(a):
            a = a_cp
        if verbose:
            print("    Coord done in %.1fs, score=%.6f" % (
                time.monotonic() - tick, true_score(a)))

    # Phase D: More FW with wider active set
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed
    if remaining > 80:
        fw2_budget = min(remaining * 0.4, 150.0)
        tick = time.monotonic()
        a_fw2 = frank_wolfe(a, fw2_budget, active_max=4000, mw_iters=50, verbose=False)
        if true_score(a_fw2) < true_score(a):
            a = a_fw2
        if verbose:
            print("    FW2 done in %.1fs, score=%.6f" % (
                time.monotonic() - tick, true_score(a)))

    return a


def propose_candidate(budget_s=1800.0):
    t0 = time.monotonic()
    total_budget = max(60.0, min(float(budget_s) - 60.0, 1740.0))

    print("Start: budget=%.0fs" % total_budget)

    # Get initial sequence
    seq = None
    try:
        parent = height_sequence_1
        p = np.asarray(parent, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        np.clip(p, 0.0, 1000.0, out=p)
        if p.ndim != 1:
            p = p.ravel()
        if p.size > 0 and float(p.sum()) > 1e-10:
            seq = norm1(p)
    except Exception:
        seq = None

    if seq is None:
        seq = default_seq(5000)

    best_seq = seq.copy()
    best_score = true_score(best_seq)
    print("Initial: n=%d score=%.8f" % (best_seq.size, best_score))
    ev_best = safe_eval(best_seq)
    if math.isfinite(ev_best) and ev_best < best_score:
        best_score = ev_best

    # Working length: cap at 10000 for speed
    work_len = min(best_seq.size, 9000)
    if best_seq.size > work_len:
        work_seq = interpolate_seq(best_seq, work_len)
    else:
        work_seq = best_seq.copy()

    work_score = true_score(work_seq)
    if work_score < best_score:
        best_seq = work_seq.copy()
        best_score = work_score
        print("Compressed work: score=%.8f" % best_score)

    # Strategy 1: Full optimization on the work sequence
    time_left = total_budget - (time.monotonic() - t0)
    if time_left > 300:
        print("\n=== Strategy 1: Full optimization (n=%d) ===" % work_seq.size)
        t1 = time.monotonic()
        a1 = full_optimize(work_seq, min(time_left * 0.55, 900.0), verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a1, "S1", t0)

    # Strategy 2: Try a different length
    time_left = total_budget - (time.monotonic() - t0)
    if time_left > 250:
        print("\n=== Strategy 2: Length variation ===")
        for target_n in [work_seq.size + 1000, max(3000, work_seq.size - 1000)]:
            time_left = total_budget - (time.monotonic() - t0)
            if time_left < 150:
                break
            target_n = int(max(2000, min(10000, target_n)))
            a0 = interpolate_seq(best_seq, target_n)
            t2 = time.monotonic()
            a2 = full_optimize(a0, min(time_left * 0.4, 250.0), verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a2, "S2_n%d" % target_n, t0)

    # Strategy 3: Perturbation restart
    time_left = total_budget - (time.monotonic() - t0)
    if time_left > 200:
        print("\n=== Strategy 3: Perturbation restart ===")
        rng = np.random.default_rng(777)
        for trial in range(3):
            time_left = total_budget - (time.monotonic() - t0)
            if time_left < 120:
                break
            noise = rng.normal(0, 0.05, best_seq.size) * best_seq
            a_pert = best_seq + noise
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / a_pert.sum()
            t3 = time.monotonic()
            a3 = full_optimize(a_pert, min(time_left * 0.3, 180.0), verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a3, "S3_t%d" % trial, t0)

    # Strategy 4: Subgradient + polish
    time_left = total_budget - (time.monotonic() - t0)
    if time_left > 100:
        print("\n=== Strategy 4: Subgradient + polish ===")
        a_sg = best_seq.copy()
        c = fft_conv(a_sg)
        M = float(c.max())
        tick = time.monotonic()
        for it in range(200):
            if time.monotonic() - tick > time_left * 0.3 or time.monotonic() - t0 > total_budget - 30:
                break
            K = min(50, c.size)
            a_new = subgradient_step(a_sg, c, K, lr=0.03)
            c_new = fft_conv(a_new)
            M_new = float(c_new.max())
            if M_new < M - 1e-15:
                a_sg = a_new
                c = c_new
                M = M_new
            else:
                # Try smaller step
                a_new2 = subgradient_step(a_sg, c, K, lr=0.01)
                c_new2 = fft_conv(a_new2)
                M_new2 = float(c_new2.max())
                if M_new2 < M - 1e-15:
                    a_sg = a_new2
                    c = c_new2
                    M = M_new2
        if verbose_score := true_score(a_sg) < best_score:
            best_seq = a_sg.copy()
            best_score = verbose_score
            print("  SG improved: %.8f" % best_score)

        # Final polish
        time_left = total_budget - (time.monotonic() - t0)
        if time_left > 50:
            a_fin = coord_polish(best_seq, min(time_left * 0.7, 150.0), verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_fin, "final_polish", t0)

    # Final: try a few LSE refinements at high beta
    time_left = total_budget - (time.monotonic() - t0)
    if SCIPY_OK and time_left > 60:
        print("\n=== Final LSE refinement ===")
        v, ref = make_v(best_seq)
        for beta in [500, 1000]:
            time_left = total_budget - (time.monotonic() - t0)
            if time_left < 30:
                break
            v = run_lbfgs_v(v, beta, ref, maxfun=100, reg_u=1e-10)
            a_new = from_v(v, ref)
            best_seq, best_score = try_update(best_seq, best_score, a_new, "final_lb_%d" % beta, t0)
            v, ref = make_v(best_seq)

    # Validate and return
    final_arr = best_seq.copy()
    final_arr = np.where(np.isfinite(final_arr), final_arr, 0.0)
    np.clip(final_arr, 0.0, 1000.0, out=final_arr)
    s = float(final_arr.sum())
    if s < 0.01:
        final_arr = default_seq(5000)
    else:
        final_arr = final_arr / s

    final_score = true_score(final_arr)
    ev = safe_eval(final_arr)
    if math.isfinite(ev):
        final_score = min(final_score, ev)

    elapsed = time.monotonic() - t0
    print("\nFINAL: score=%.8f n=%d elapsed=%.1fs" % (final_score, final_arr.size, elapsed))

    return [float(x) for x in final_arr]
