    # where N >= len(a) + len(p) - 1
    
    # For our case: p has length 2n-1, a has length n.
    # N >= (2n-1) + n - 1 = 3n - 2
    # r has indices from -(n-1) to (2n-2), but we only need r[0] to r[n-1] (since m in [0,n-1]).
    # Actually r[i] = sum_m p[m+i]*a[m] for m in [0, n-1], so we need p[m+i] for m+i in [i, i+n-1].
    # For i in [0, n-1] (our grad indices), we need p[i] to p[i+n-1], which is within [0, 2n-2]. ✓
    
    N_fft = 1 << (3 * n - 2 - 1).bit_length()
    Pa = np.fft.rfft(p, N_fft)
    Aa = np.fft.rfft(a, N_fft)
    r = np.fft.irfft(np.conj(Aa) * Pa, N_fft)
    # r[i] = sum_m a[m]*p[m+i] for the full range
    # We want grad[i] = 2 * r[i] for i in [0, n-1]
    grad = 2.0 * r[:n]
    return grad

"""
Minimizes 2*n*max(autocorr(a))/(sum(a))^2 using:
1. Softmax-parametrized Adam optimization with log-sum-exp smooth minimax objective
2. Adaptive beta annealing for gradient sharpness  
3. Coordinate descent refinement targeting top autocorrelation peaks
4. Multi-length and multi-start exploration
5. Peak-targeted mass redistribution
"""
import numpy as np
import time

def fast_autocorr(a):
    """FFT-based autocorrelation (full convolution of a with itself)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Wrapper around evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize a to be non-negative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    """Project z onto the probability simplex."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def smooth_max_and_weights(b, beta, top_k=None):
    """Compute smooth max and softmax weights of b."""
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.full_like(b, -1e10)
        b_use[idx] = b[idx]
    else:
        b_use = b.copy()
    b_max = b_use.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -700, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        # All underflowed, just use argmax
        p = np.zeros_like(b)
        p[b.argmax()] = 1.0
        return b_max, p
    p = exp_vals / total
    log_sum_exp = np.log(total)
    smooth_val = b_max + log_sum_exp / beta
    return smooth_val, p

def gradient_smooth_max(a, p):
    """
    Gradient of smooth max of self-autocorrelation w.r.t. a.
    grad[i] = 2 * sum_k p[k] * a[k-i] (where k-i is in [0, n-1])
    Computed via FFT cross-correlation.
    """
    n = len(a)
    size_p = len(p)  # 2n-1
    # Need FFT size >= n + size_p - 1
    needed = n + size_p - 1
    fft_size = 1 << (needed - 1).bit_length()
    Pa = np.fft.rfft(p, fft_size)
    Aa = np.fft.rfft(a, fft_size)
    # cross_corr(a, p)[i] = sum_m a[m]*p[m+i]
    r = np.fft.irfft(np.conj(Aa) * Pa, fft_size)
    grad = 2.0 * r[:n]
    return grad

def adam_simplex_optimize(a0, time_limit, beta_start=30.0, beta_end=2000.0,
                          beta_growth=1.003, lr=0.1, top_k_frac=0.05,
                          verbose=True):
    """
    Adam optimization in softmax w-space to minimize smooth max autocorrelation.
    a = softmax(w), gradient computed in chain rule.
    """
    n = len(a0)
    a = norm_sum1(a0)
    # Initialize w: log(a) with floor for zeros
    a_safe = np.maximum(a, 1e-12)
    w = np.log(a_safe / a_safe.sum())  # log(a) - log(sum) ≈ log(a) for sum~1
    # Actually, just use log(a) as init, softmax will normalize
    w = np.log(np.maximum(a, 1e-15))
    w = w - w.mean()  # center for numerical stability
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    stall = 0
    max_stall = 3000
    
    top_k = max(50, int(n * top_k_frac))
    
    beta = beta_start
    current_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute smooth max and gradient
        b = fast_autocorr(a)
        smooth_val, p = smooth_max_and_weights(b, beta, top_k)
        grad_a = gradient_smooth_max(a, p)
        
        # Chain rule: dF/dw = a * (grad_a - dot(a, grad_a))
        dot_ag = float(np.dot(a, grad_a))
        grad_w = a * (grad_a - dot_ag)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * grad_w ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Update w
        w = w - current_lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic re-centering for numerical stability
        if t % 100 == 0:
            w = w - w.mean()
        w = np.clip(w, -50, 50)  # prevent overflow
        
        # Compute new a from softmax
        w_shifted = w - w.max()
        exp_w = np.exp(w_shifted)
        a_new = exp_w / exp_w.sum()
        
        # Evaluate
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Adaptive beta and learning rate
        if t % 20 == 0:
            beta = min(beta * beta_growth, beta_end)
            # Reduce lr as we progress
            progress = min(1.0, t / (time_limit * 50))  # rough
            current_lr = lr * max(0.1, 1.0 - 0.3 * progress)
        
        # Stall-based restart
        if stall > max_stall:
            stall = 0
            max_stall = min(max_stall * 1.3, 10000)
            noise = rng.normal(0, 0.01, n)
            w = w + noise
            w = w - w.mean()
            if verbose:
                elapsed = time.monotonic() - t0
                score = 2.0 * n * best_m
                print("    Adam restart t={} ({}s): score={}".format(
                    t, round(elapsed,1), repr(score)))
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print("    Adam t={} ({}s): score={} beta={}".format(
                t, round(elapsed,1), repr(score), round(beta,1)))
    
    return best_a, best_m

def coord_descent_refine(a0, time_limit, top_k=30, verbose=True):
    """Coordinate descent: move small mass from high-pressure to low-pressure positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_indices = np.argsort(b)[::-1][:top_k]
        
        # Compute pressure
        pressure = np.zeros(n)
        for k_int in peak_indices[:15]:
            ki = int(k_int)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                pressure[idx] += a[ki - idx]
        
        # Find source: highest pressure position with significant mass
        src_score = pressure * a
        src_candidates = np.argsort(src_score)[::-1][:20]
        
        # Find destination: low pressure
        dst_candidates = np.argsort(pressure)[:30]
        
        improved = False
        for src in src_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-10:
                continue
            for dst in dst_candidates:
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.08, 0.15]:
                    amt = a[src] * frac
                    trial = a.copy()
                    trial[src] -= amt
                    trial[dst] += amt
                    trial = np.maximum(trial, 0)
                    s = trial.sum()
                    if s < 1e-15:
                        continue
                    trial = trial / s
                    m_trial = fast_autocorr(trial).max()
                    if m_trial < best_m - 1e-15:
                        a = trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                stall = 0
                break
        
        if not improved:
            stall += 1
            if stall % 100 == 0:
                # Random perturbation
                noise = rng.normal(0, 0.005, n) * best_a
                a = best_a + noise
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
            
            if stall > 2000:
                break
        
        if verbose and (t0 and (time.monotonic() - t0) > 0) and int((time.monotonic() - t0) * 2) % 2 == 0 and verbose:
            pass  # suppress excessive printing
    
    return best_a, best_m

def parametric_init(n, rng):
    """Generate diverse parametric initializations."""
    x = np.linspace(0.0, 1.0, n)
    
    families = []
    
    # Power law pairs
    for alpha in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]:
        for beta_p in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]:
            families.append((x + 0.01)**alpha * (1.0 - x + 0.01)**beta_p)
    
    # Gaussian single
    for center in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sigma in [0.1, 0.15, 0.2, 0.3]:
            families.append(np.exp(-((x - center)**2) / (2*sigma**2)))
    
    # Two Gaussians
    for c1, c2 in [(0.3, 0.7), (0.25, 0.75), (0.35, 0.65)]:
        for s in [0.08, 0.12, 0.15]:
            families.append(np.exp(-((x-c1)**2)/(2*s**2)) + np.exp(-((x-c2)**2)/(2*s**2)))
    
    # Wave-like
    for freq in [5, 10, 20, 50]:
        families.append(0.5 + 0.5*np.cos(2*np.pi*freq*x))
    
    # Sigmoid-like
    for c in [0.3, 0.5, 0.7]:
        for k in [10, 30, 100]:
            families.append(1.0 / (1.0 + np.exp(-k*(x-c))))
    
    # Random smooth
    for _ in range(10):
        freq = rng.uniform(3, 30)
        phase = rng.uniform(0, 2*np.pi)
        families.append(0.5 + 0.3*np.sin(2*np.pi*freq*x + phase) + 0.2*np.cos(4*np.pi*freq*x))
        families[families[-1] < 0] = 0
    
    # Pure random
    for _ in range(5):
        p = rng.uniform(0.5, 3.0)
        families.append(rng.random(n) ** p + 1e-6)
    
    return families

def evaluate_score(a):
    """Compute the objective score for a normalized sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s ** 2)
