def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_constraint_matrix(y, constraint_list, n):

        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs

            for v in violations[:200]:
                if int(v) not in active_set_base:
                    active_set_base.append(int(v))
            active_set = sorted(set(active_set_base))

        K = len(active_set)
        if K == 0:
            break

        # Build and solve LP
        coeffs = build_constraint_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = sorted(active_set[:max(200, K // 4)])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            active_set = sorted(active_set[:max(200, K // 4)])
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Dual-guided constraint addition:
        # Compute y*g_dir convolution and find violations
        fft_size = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]

        # Add top violations
        all_viol = np.where(conv_yg > M_curr * 0.999)[0]
        if len(all_viol) > 0:
            viol_amt = conv_yg[all_viol] - M_curr
            n_add = min(add_per_iter, len(all_viol))
            top_v = all_viol[np.argsort(viol_amt)[::-1][:n_add]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        # Backtracking line search
        best_alpha = 1.0 / (iter_count + 3.0)
        best_trial_M = ac.max()

        # Try a range of alphas from large to small
        if iter_count <= 3:
            alphas = [1.0, 0.8, 0.6, 0.4, 0.2]
        else:
            base = min(1.0, 3.0 / (iter_count + 1.0))
            alphas = [base, base * 0.7, base * 0.5, base * 0.3,
                      base * 0.1, max(0.01, 1.0 / (iter_count + 2.0))]

        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 2:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) +
                  "s): J=" + repr(2.0 * n * best_m) + ", |S|=" + str(len(active_set)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) +
              ", iters=" + str(iter_count))
    return best_a, best_m


def gradient_polish(a0, time_limit, verbose=False):
    """Gradient-based polish using subgradient of max convolution."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_convolve(y).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    stall = 0
    max_stall = 2000
    x = y.copy()
    vel = np.zeros(n)
    momentum = 0.7
    t_iter = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        t_iter += 1
        eta = 0.03 / (np.sqrt(t_iter) + 1.0)

        # Nesterov lookahead
        look = x + momentum * vel
        look = np.maximum(look, 1e-12)
        s_look = look.sum()
        if s_look < 1e-12:
            look = x.copy()
        else:
            look = look / s_look

        # Subgradient
        ac = fast_convolve(look)
        k_max = int(np.argmax(ac))

        grad = np.zeros(n)
        i_start = max(0, k_max - n + 1)
        i_end = min(n, k_max + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] = look[k_max - idx]

        # Add small noise for exploration
        noise_scale = 0.005 / (1.0 + t_iter / 200.0)
        grad += noise_scale * rng.standard_normal(n) * look

        # Mirror descent (softmax update)
        log_a = np.log(np.maximum(x, 1e-30)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = x.copy()
        else:
            a_new = a_new / s_new

        vel = momentum * vel + (a_new - x)
        x = a_new

        M_new = fast_convolve(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 500 and stall % 500 == 0:
            noise = rng.normal(0, 0.02, n) * best_a
            x_p = best_a + noise
            x_p = np.maximum(x_p, 1e-10)
            sp = x_p.sum()
            if sp > 1e-12:
                x = x_p / sp
                vel = np.zeros(n)

        if verbose and t_iter % 3000 == 0:
            elapsed = time.monotonic() - t0
            print("    Grad t=" + str(t_iter) + " (" + str(round(elapsed, 1)) +
                  "s): J=" + repr(2.0 * n * best_m))

    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=77, verbose=False):
    """Targeted coordinate descent: move mass from peak contributors to quiet areas."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        top_k = np.argsort(ac)[::-1][:min(20, len(ac))]

        improved = False
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination (low convolution impact)
            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            candidates = np.where(scores < scores.min() + 0.1 * (scores.max() - scores.min()) + 1e-10)[0]
            n_cand = len(candidates)
            if n_cand < 5:
                candidates = np.where(scores == scores.min())[0]
            if n_cand < 5:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])

            if dst == src:
                continue

            found = False
            for frac in [0.003, 0.007, 0.015, 0.03, 0.06]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0:
                for _ in range(15):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) +
              ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def get_initial_sequence(t_global):
    """Get the best starting sequence from parent or fallback."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
                return p_norm, sc
    except Exception:
        pass

    # Fallback: sine-squared
    n = 5000
    x = np.linspace(0, 1, n)
    best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
    best_score = evaluate_safe(best_seq)
    print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = get_solver()
    best_seq, best_score = get_initial_sequence(t_global)
    n0 = len(best_seq)

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: Aggressive LP-FW on parent (35% of time) ===
    print("\n=== Phase 1: LP-FW initial (n=" + str(n0) + ") ===")
    t1 = min(remaining_time() * 0.35, 500.0)
    if t1 > 40 and n0 <= 12000:
        a_out, _ = lp_fw_improved(norm_sum1(best_seq), t1, solver,
                                  initial_K=1500, max_K=8000,
                                  add_per_iter=100, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out,
                                          "lpfw1_n" + str(n0), t_global)

    # === Phase 2: Multi-n exploration (30% of time, split among candidates) ===
    print("\n=== Phase 2: Multi-n exploration ===")
    n_best = len(best_seq)
    # Try nearby n values
    candidate_ns = []
    for delta in [500, -500, 1000, -1000, 1500, -1500, 2000]:
        candidate_ns.append(n_best + delta)
    # Filter valid range
    candidate_ns = sorted(set([x for x in candidate_ns if 3000 <= x <= 15000]))
    # Sort by proximity to best
    candidate_ns.sort(key=lambda x: abs(x - n_best))

    n_tried = 0
    for new_n in candidate_ns[:5]:
        rt = remaining_time()
        if rt < 100:
            break
        n_tried += 1
        t_alloc = min(rt * 0.15, 200.0)
        if t_alloc < 30:
            continue
        print("\n  Multi-n: n=" + str(new_n) + " (t_alloc=" + str(round(t_alloc, 1)) + "s)")
        a_long = interpolate_seq(best_seq, new_n)

        a_out, _ = lp_fw_improved(a_long, t_alloc, solver,
                                  initial_K=800, max_K=5000,
                                  add_per_iter=60, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out,
                                          "lpfw_n" + str(new_n), t_global)

    # === Phase 3: LP-FW on best found (20% of time) ===
    print("\n=== Phase 3: LP-FW polish (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining_time() * 0.20, 300.0)
    if t3 > 40 and len(best_seq) <= 12000:
        a_out, _ = lp_fw_improved(norm_sum1(best_seq), t3, solver,
                                  initial_K=2000, max_K=8000,
                                  add_per_iter=120, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out,
                                          "lpfw3_n" + str(len(best_seq)), t_global)

    # === Phase 4: Gradient polish (10% of time) ===
    print("\n=== Phase 4: Gradient polish (n=" + str(len(best_seq)) + ") ===")
    t4 = min(remaining_time() * 0.10, 150.0)
    if t4 > 20:
        a_out, _ = gradient_polish(norm_sum1(best_seq), t4, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out,
                                          "grad_n" + str(len(best_seq)), t_global)

    # === Phase 5: LP-FW second pass on best (10% of time) ===
    print("\n=== Phase 5: LP-FW second (n=" + str(len(best_seq)) + ") ===")
    t5 = min(remaining_time() * 0.10, 150.0)
    if t5 > 30 and len(best_seq) <= 12000:
        a_out, _ = lp_fw_improved(norm_sum1(best_seq), t5, solver,
                                  initial_K=2500, max_K=8000,
                                  add_per_iter=150, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out,
                                          "lpfw5_n" + str(len(best_seq)), t_global)

    # === Phase 6: Coordinate polish (5% of time) ===
    print("\n=== Phase 6: Coordinate polish (n=" + str(len(best_seq)) + ") ===")
    t6 = min(remaining_time() * 0.05, 80.0)
    if t6 > 15:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t6, rng_seed=42, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out,
                                          "coord_n" + str(len(best_seq)), t_global)

    # === Final: One more LP-FW if time remains ===
    rt = remaining_time()
    if rt > 30 and len(best_seq) <= 12000:
        print("\n=== Final LP-FW polish ===")
        a_out, _ = lp_fw_improved(norm_sum1(best_seq), min(rt * 0.7, 120.0), solver,
                                  initial_K=3000, max_K=8000,
                                  add_per_iter=150, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out,
                                          "lpfw_final", t_global)

    # Final coordinate polish
    rt = remaining_time()
    if rt > 10:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), min(rt * 0.8, 40.0),
                                     rng_seed=2024, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out,
                                          "coord_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) +
              ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
