1. Enhanced LP-Frank-Wolfe with away steps and aggressive active set management
2. Log-domain mirror descent with adaptive step size
3. Peak shaving local search
4. Multi-start with diverse initializations
5. n-adaptation with interpolation
6. Dual-informed subgradient methods
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar

_MODULE_SOLVER = None

def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw_away(a0, time_limit, solver, initial_K=500, max_K=4000, add_per_iter=50, max_vertices=20, verbose=True):
    """LP-Frank-Wolfe with Away Steps (FW-A) for improved convergence."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac_y = fast_convolve(y)
    best_m = ac_y.max()
    t0 = time.monotonic()

    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 500
    iter_count = 0
    stagnation_count = 0
    
    # For FW-A: maintain a set of vertices
    vertices = [y.copy()]  # list of arrays
    vertex_weights = [1.0]
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if stagnation_count > 4:
                if verbose:
                    print("    LP-FW-A: persistent stagnation, stopping")
                break
            # Restart with perturbed active set
            ac_curr = fast_convolve(y)
            n_pert = max(50, len(active_set) // 5)
            rng = np.random.default_rng(stagnation_count * 1000 + 42)
            keep = list(rng.choice(len(active_set), size=len(active_set) - n_pert, replace=False))
            new_adds = np.argsort(ac_curr)[::-1][:n_pert * 2]
            new_set = set([active_set[i] for i in keep])
            for k in new_adds:
                new_set.add(int(k))
            active_set = sorted(new_set)
            no_improve = 0
            stagnation_count += 1
            # Also reset vertices
            vertices = [y.copy()]
            continue

        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            stagnation_count = 0
        else:
            no_improve += 1

        # Trim active set
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep_idx = np.argsort(margins)[:int(max_K * 0.7)]
            active_set_base = [active_set[i] for i in keep_idx]
            violations = np.where(ac > M_curr - 1e-8)[0]
            for v in violations[:100]:
                if int(v) not in active_set_base:
                    active_set_base.append(int(v))
            active_set = sorted(set(active_set_base))

        K = len(active_set)
        if K == 0:
            break
        if K > max_K:
            active_set = active_set[:max_K]
            K = max_K

        coeffs = build_constraint_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(100, K // 3)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations and add to active set
        fft_size = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]

        all_viol = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(all_viol) > 0:
            viol_amt = conv_ys[all_viol] - M_curr
            top_v = all_viol[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        # FW-A: Try both "towards" and "away" steps
        # Standard FW step: move towards g_dir
        # Away step: move away from a vertex v in the convex hull
        
        best_alpha = max(0.5, 1.0 / (iter_count + 2.0))
        best_trial_M = M_curr
        best_direction = g_dir
        best_is_away = False
        best_away_vertex = None
        
        # Try standard FW step
        for alpha in [1.0, 0.9, 0.7, 0.5, 0.3, 0.15, 0.05]:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
                best_direction = g_dir
                best_is_away = False

        # Try away steps (only if we have multiple vertices)
        if len(vertices) > 1 and iter_count > 50:
            # Try away from the "worst" vertex (the one that's farthest from g_dir)
            for vi in range(min(len(vertices), max_vertices)):
                v = vertices[vi]
                # Away step: y_new = y + alpha * (y - v), then normalize
                for alpha in [0.1, 0.3, 0.5, 0.7]:
                    trial = y + alpha * (y - v)
                    trial = np.maximum(trial, 0.0)
                    s_t = trial.sum()
                    if s_t < 1e-15:
                        continue
                    trial = trial / s_t
                    M_trial = fast_convolve(trial).max()
                    if M_trial < best_trial_M:
                        best_trial_M = M_trial
                        best_alpha = alpha
                        best_direction = y - v  # away direction
                        best_is_away = True
                        best_away_vertex = vi
                        break

        # Apply the best step
        if best_is_away and best_away_vertex is not None:
            y_new = y + best_alpha * (y - vertices[best_away_vertex])
            y_new = np.maximum(y_new, 0.0)
            s_new = y_new.sum()
            if s_new > 1e-15:
                y = y_new / s_new
            # Remove the away vertex from the set
            vertices.pop(best_away_vertex)
            vertices.append(y.copy())
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y / s_y
            # Update vertices
            if len(vertices) < max_vertices:
                vertices.append(g_dir.copy())
            else:
                # Replace the vertex farthest from current y
                dists = [np.linalg.norm(y - v) for v in vertices]
                worst = int(np.argmax(dists))
                if best_trial_M < M_curr:
                    vertices[worst] = g_dir.copy()
        
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW-A iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + " |S|=" + str(len(active_set)) + " verts=" + str(len(vertices)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW-A done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + " iters=" + str(iter_count))
    return best_a, best_m


def log_mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent in log-domain for smooth optimization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    # Work with log(a) parameterization
    eps = 1e-10
    b = np.log(a + eps)
    
    eta = 0.1
    stall = 0
    max_stall = 2000
    iter_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break
        
        iter_count += 1
        # Adaptive learning rate
        eta_t = eta / (1.0 + 0.001 * t)
        
        # Compute current a from b
        a_curr = np.exp(b)
        a_curr = a_curr / a_curr.sum()
        
        # Compute convolution and find peak
        c = fast_convolve(a_curr)
        k_peak = int(np.argmax(c))
        
        # Subgradient of max_k conv(a,a) w.r.t. a
        grad_a = np.zeros(n)
        i_start = max(0, k_peak - n + 1)
        i_end = min(n, k_peak + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad_a[idx] = a_curr[k_peak - idx]
        
        # Also add contributions from top-K peaks for better conditioning
        top_K = 20
        top_idx = np.argsort(c)[::-1][:top_K]
        for ki in top_idx[1:min(top_K, len(top_idx))]:
            ki_int = int(ki)
            weight = c[ki_int] / c[k_peak] * 0.5  # weight by relative peak height
            i_s = max(0, ki_int - n + 1)
            i_e = min(n, ki_int + 1)
            if i_s < i_e:
                idx2 = np.arange(i_s, i_e)
                grad_a[idx2] += weight * a_curr[ki_int - idx2]
        
        # Gradient w.r.t. b: da/db = a * (1 - a) for softmax-like, but here it's just a
        # Actually a = exp(b)/sum(exp(b)), so da/db_i = a_i(1-a_i) for the log-sum-exp
        # But we're using a simpler parameterization: just scale b so that sum(exp(b))=1
        # Gradient of obj w.r.t. b_i = grad_a_i * da_i/db_i = grad_a_i * a_i (approximately, 
        # since we're using the simple exp parameterization)
        grad_b = grad_a * a_curr
        
        # Update b
        b = b - eta_t * grad_b
        
        # Renormalize b so that sum(exp(b)) = 1 (shift b by -log(sum(exp(b))))
        max_b = b.max()
        log_sum = max_b + np.log(np.sum(np.exp(b - max_b)))
        b = b - log_sum
        
        # Get new a
        a_new = np.exp(b)
        a_new = a_new / a_new.sum()
        
        obj_new = 2.0 * n * fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            eta = min(eta * 1.05, 1.0)
        else:
            stall += 1
            if stall % 500 == 0:
                eta = eta * 0.7
                # Random perturbation
                b += rng.normal(0, 0.01, n)
                max_b2 = b.max()
                log_sum2 = max_b2 + np.log(np.sum(np.exp(b - max_b2)))
                b = b - log_sum2
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print("    LogMD t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): obj=" + repr(best_obj))
    
    if verbose:
        elapsed = time.monotonic() - t0
        print("    LogMD done (" + str(round(elapsed, 1)) + "s): obj=" + repr(best_obj) + " iters=" + str(iter_count))
    return best_a, best_obj / (2.0 * n)  # return (a, max_conv)


def peak_shaving(a0, time_limit, verbose=False):
    """Targeted peak shaving: identify top peaks and reduce them by mass transfer."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    stall = 0
    max_stall = 3000
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_convolve(a)
        # Get top peaks
        n_peaks = min(30, len(c))
        top_peaks = np.argsort(c)[::-1][:n_peaks]
        
        improved = False
        for k in top_peaks[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            # Find the dominant contributing pair for this peak
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Source: the pair element with max contribution
            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])
            partner = k_int - src
            
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find destination: a position that minimally affects top peaks
            # Score each position by how much it would affect top peaks if we added mass there
            scores = np.zeros(n)
            for k2 in top_peaks[:15]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
            
            # Prefer positions with low score
            # Also avoid the source
            scores[src] = np.inf
            # Also avoid positions that would create new peaks
            candidates = np.argsort(scores)[:n // 2]
            candidates = candidates[scores[candidates] < np.inf]
            
            if len(candidates) == 0:
                candidates = np.arange(n)
                candidates = candidates[candidates != src]
            
            # Try transferring small amounts
            best_frac = None
            best_trial_m = best_m
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                for dst in candidates[:5]:
                    dst_int = int(dst)
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst_int] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_convolve(a_trial).max()
                    if M_trial < best_trial_m:
                        best_trial_m = M_trial
                        best_frac = (src, dst_int, amt)
            
            if best_frac is not None:
                src_f, dst_f, amt_f = best_frac
                a[src_f] -= amt_f
                a[dst_f] += amt_f
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                a = a / s_a
                best_m = best_trial_m
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            # Try random perturbations
            found = False
            for _ in range(20):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                a = a / s_a
                M_new = fast_convolve(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
                else:
                    # Undo
                    a[i] += amt
                    a[j] -= amt
                    a = a / a.sum()
            
            if not improved:
                stall += 1
                if stall % 1000 == 0:
                    # Larger perturbation
                    noise = rng.normal(0, 0.02, n)
                    a_p = best_a + noise
                    a_p = np.maximum(a_p, 0.0)
                    sp = a_p.sum()
                    if sp > 1e-12:
                        a = a_p / sp
    
    if verbose:
        elapsed = time.monotonic() - t0
        print("    PeakShave done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))
    return best_a, best_m


def diverse_initializations(n, t_global, budget):
    """Generate diverse initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    candidates = []
    
    # Sine-based
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        c = norm_sum1(np.sin(np.pi * x) ** p)
        candidates.append(c)
    
    # Cosine-based
    for p in [0.5, 1.0, 2.0]:
        c = norm_sum1(np.cos(np.pi * (x - 0.5)) ** p + 0.1)
        candidates.append(c)
    
    # Linear ramp
    c = norm_sum1(np.concatenate([np.linspace(0.1, 2.0, n//2), np.linspace(2.0, 0.1, n - n//2)]))
    candidates.append(c)
    
    # Two-peaked
    c1 = norm_sum1(np.exp(-((x - 0.3) ** 2) / (2 * 0.1 ** 2)))
    c2 = norm_sum1(np.exp(-((x - 0.7) ** 2) / (2 * 0.1 ** 2)))
    candidates.append(norm_sum1(0.5 * c1 + 0.5 * c2))
    
    # Random smooth
    rng = np.random.default_rng(999)
    for _ in range(3):
        raw = rng.uniform(0.5, 1.5, n)
        # Smooth
        from scipy.ndimage import gaussian_filter1d
        raw = gaussian_filter1d(raw, sigma=10)
        candidates.append(norm_sum1(raw))
    
    return candidates


def quick_optimize(a, t_limit, solver):
    """Quick LP-FW optimization for ranking candidates."""
    return lp_fw_away(a, t_limit, solver, initial_K=200, max_K=1500, 
                      add_per_iter=30, max_vertices=10, verbose=False)


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def get_initial_sequence(t_global):
    """Get the best starting sequence."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
                return p_norm, sc
    except Exception:
        pass

    n = 5000
    x = np.linspace(0, 1, n)
    best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
    best_score = evaluate_safe(best_seq)
    print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = get_solver()
    best_seq, best_score = get_initial_sequence(t_global)
    n0 = len(best_seq)

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: LP-FW-A initial ===
    print("\n=== Phase 1: LP-FW-A initial (n=" + str(n0) + ") ===")
    t1 = min(remaining_time() * 0.15, 250.0)
    if t1 > 40 and n0 <= 9000:
        a_out, _ = lp_fw_away(norm_sum1(best_seq), t1, solver,
                              initial_K=500, max_K=4000, add_per_iter=50,
                              max_vertices=20, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfwa1_n" + str(n0), t_global)

    # === Phase 2: Log-domain mirror descent ===
    print("\n=== Phase 2: LogMD (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining_time() * 0.10, 150.0)
