def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_constraint_matrix(a, active_lags, n):
    """Build the constraint matrix C where C[k, i] = a[|k-i|] appropriately."""
    K = len(active_lags)
    C = np.zeros((K, n), dtype=np.float64)
    for jj in range(K):
        ki = int(active_lags[jj])
        # b[ki] = sum_i a[i] * a[ki-i] for valid i
        # Valid i: 0 <= i < n and 0 <= ki-i < n
        # So: max(0, ki-n+1) <= i <= min(n-1, ki)
        i_lo = max(0, ki - n + 1)
        i_hi = min(n - 1, ki)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            C[jj, idx] = a[ki - idx]
    return C

def solve_lp_fw(y, active_lags, solver):
    """Solve the LP: min M s.t. sum(g)=1, C@g <= M, g>=0. Returns g and M."""
    n = len(y)
    K = len(active_lags)
    if K == 0:
        return np.ones(n) / n, 1.0
    
    C = build_constraint_matrix(y, active_lags, n)
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1, C @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return None, None
    g_dir = g_val / g_sum
    M_val = float(M_var.value)
    return g_dir, M_val

    prev_dir = None
    momentum = np.zeros(n, dtype=np.float64)
    beta = 0.5  # momentum parameter
    
    # Active set
    ac = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac)[::-1][:init_K])
    
    no_improve = 0
    stall_limit = 3000
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        cur_obj = 2.0 * n * M_curr
        
        if M_curr < best_M - 1e-14:
            best_M = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update active set
        if t % 20 == 0 or t == 1:
            top_lags = np.argsort(ac)[::-1]
            target_K = min(max_K, init_K + t * 3)
            for k in top_lags[:target_K]:
                active_set.add(int(k))
            if len(active_set) > max_K:
                clist = sorted(active_set)
                margins = np.array([M_curr - ac[k] for k in clist])
                keep = np.argsort(margins)[:max_K]
                active_set = set(int(clist[i]) for i in keep)
        
        # Always include the current peak
        active_set.add(int(np.argmax(ac)))
        
        clist = sorted(active_set)
        
        try:
            g_dir, M_lp = solve_lp_fw(y, clist, solver)
            if g_dir is None:
                raise Exception("LP failed")
        except Exception:
            if len(active_set) > init_K:
                clist = sorted(active_set)
                active_set = set(int(clist[i]) for i in np.argsort([M_curr - ac[k] for k in clist])[:max(init_K, len(clist)//2)])
            continue
        
        # Momentum update
        if prev_dir is not None:
            momentum = beta * momentum + (1 - beta) * (g_dir - prev_dir)
        else:
            momentum = 0
        prev_dir = g_dir.copy()
        
        # Combined direction
        direction = g_dir + momentum
        # Ensure non-negative (project)
        direction = np.maximum(direction, 0.0)
        dsum = direction.sum()
        if dsum < 1e-15:
            direction = g_dir
            dsum = 1.0
        direction = direction / dsum
        
        # Line search with backtracking
        best_alpha = 2.0 / (t + 2.0)
        best_trial = 1e10
        
        # Try several alphas
        alphas = [2.0/(t+2), 1.0/(t+1), 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * direction
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            Mt = fast_autocorr(trial / ts).max()
            if Mt < best_trial:
                best_trial = Mt
                best_alpha = alpha
        
        # Apply update
        if t <= 1:
            y = direction.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * direction
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    MFW t={:5d} ({:.1f}s) J={:.7f} |S|={}".format(t, elapsed, 2 * n * best_M, len(active_set)))
    
    elapsed = time.monotonic() - t0
    if verbose:
        print("    MFW done ({:.1f}s) J={:.7f} |S|={}".format(elapsed, 2 * n * best_M, len(active_set)))
    return best_a, 2.0 * n * best_M

def adaptive_mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent with adaptive learning rate and all-lag weighting."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Adam state
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    # Adaptive temperature
    base_temp = 10.0
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 5000:
            break
        
        ac = fast_autocorr(x)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr
        
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Weight all lags by their value (log-weighted for numerical stability)
        # Use top-K with soft weighting
        K = min(200, len(ac))
        idx = np.argsort(ac)[::-1][:K]
        top_vals = ac[idx]
        
        # Temperature-based weighting
        temp = base_temp + 5.0 * np.sqrt(t)
        w = np.exp(temp * (top_vals - top_vals[0]))
        w = w / (w.sum() + 1e-30)
        
        # Compute gradient using FFT
        # grad[i] = sum_k w[k] * (contribution of a[i] to b[k])
        # For b[k] = sum_j a[j]*a[k-j], db[k]/da[i] = a[k-i] + a[i-k] (where valid)
        # Actually for the weighted sum: grad[i] = sum_k w[k] * (a[k-i] if 0<=k-i<n else 0) + (a[i-k] if 0<=i-k<n else 0)
        # But since b[k] = sum_j a[j]*a[k-j], db[k]/da[i] = a[k-i] (when 0<=k-i<n) + a[i-k] (when 0<=i-k<n)
        # Wait, b[k] = sum_j a[j] * a[k-j]. db[k]/da[i] = a[k-i] (if 0<=k-i<n) + ... 
        # Actually: b[k] = sum_j a[j] * a[k-j]
        # db[k]/da[i] = a[k-i] (from the j=i term in the first factor: da[i]/da[i] * a[k-i]) 
        #            + a[i-k] (from the (k-j)=i term, i.e., j=k-i: a[k-i] * da[k-(k-i)]/da[i] = a[k-i]*1)
        # Wait no. Let me be more careful.
        # b[k] = sum_{j=0}^{n-1} a[j] * a[k-j] (where a[k-j] is 0 if k-j < 0 or k-j >= n)
        # db[k]/da[i] = a[k-i] (if 0 <= k-i < n) + a[i-k] (if 0 <= i-k < n)
        # But wait, when j = i: term is a[i] * a[k-i], derivative w.r.t. a[i] gives a[k-i]
        # When k-j = i (i.e., j = k-i): term is a[k-i] * a[i], derivative w.r.t. a[i] gives a[k-i]
        # So db[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i] when both are valid?
        # No wait: the two cases are when i is the "first factor" index and when i is the "second factor" index.
        # Case 1: j = i, so contribution is a[i] * a[k-i], derivative = a[k-i] (valid if 0<=k-i<n)
        # Case 2: k-j = i, so j = k-i, contribution is a[k-i] * a[i], derivative = a[k-i] (valid if 0<=k-i<n and 0<=k-i<n)
        # These are the same condition! So if 0 <= k-i < n, both cases contribute, giving 2*a[k-i].
        # But wait, that's only if i != k-i, i.e., i != k/2. If i = k/2 (k even), then the two cases are the same term, and the derivative is a[k-i] = a[i].
        
        # Hmm, let me just use the FFT approach which handles this automatically.
        # The gradient of sum_k w[k] * b[k] w.r.t. a is the cross-correlation of w with a.
        # Specifically, if we define W as a vector of length 2n-1 with W[k] = w[k],
        # then grad = cross-correlation of W and a, which can be computed via FFT.
        
        # Actually, let's use a simpler approach:
        # grad = C^T @ w where C is the constraint matrix (but transposed)
        # For efficiency, use FFT:
        # Define W_padded of length >= 2n-1 with W_padded[k] = w[k] for k in active set, 0 otherwise
        # Then grad[i] = sum_k W_padded[k] * a[|k-i|]... 
        # This is the cross-correlation of W_padded and a.
        
        # FFT approach for cross-correlation:
        # cross_corr(W, a)[i] = sum_k W[k] * conj(a[k-i]) (for real: sum_k W[k] * a[k-i])
        # Using FFT: cross_corr = ifft(fft(W, N) * conj(fft(a, N))) -- but need to be careful with alignment
        
        # Simpler: just compute it directly for the top-K lags
        # Since K is at most 200 and n can be 12000, direct computation is O(K*n) = 2.4M per step
        # That's fine.
        
        grad = np.zeros(n, dtype=np.float64)
        for ii in range(K):
            ki = int(idx[ii])
            wi = w[ii]
            if wi < 1e-15:
                continue
            # db[ki]/da[i] for each i
            # = a[ki-i] if 0<=ki-i<n, plus a[i-ki] if 0<=i-ki<n
            # But as discussed, for non-negative a, this simplifies.
            # Let's just compute: for each valid i, the contribution to b[ki] from a[i] is a[ki-i] (if valid)
            # And the "other" contribution is a[i-ki] (if valid)
            # Together: if 0 <= ki-i < n: add a[ki-i]
            #           if 0 <= i-ki < n: add a[i-ki]
            # But these are the same: let d = ki - i. If 0 <= d < n, add a[d]. 
            # And let d' = i - ki = -d. If 0 <= d' < n (i.e., -n < d <= 0, i.e., d in (-n, 0]), add a[d'].
            # So the total contribution for coordinate i to b[ki] is:
            # a[ki-i] (if 0 <= ki-i < n) + a[i-ki] (if 0 <= i-ki < n)
            # = a[|ki-i|] * (indicator that ki-i is in [0,n)) + a[|i-ki|] * (indicator that i-ki is in [0,n))
            
            # More simply: the indices i that contribute to b[ki] are those with 0 <= i < n and 0 <= ki-i < n
            # i.e., max(0, ki-n+1) <= i <= min(n-1, ki)
            i_lo = max(0, ki - n + 1)
            i_hi = min(n - 1, ki)
            if i_lo <= i_hi:
                indices = np.arange(i_lo, i_hi + 1)
                # contribution of a[i] to b[ki] is a[ki-i]
                # so grad[i] += w[ki] * a[ki-i]
                # But also, the "symmetric" contribution: b[ki] includes a[i]*a[ki-i]
                # and the derivative of this w.r.t. a[i] is a[ki-i]
                # The derivative w.r.t. a[ki-i] is a[i]
                # So for coordinate j = ki-i, grad[j] += w[ki] * a[i]
                # This means: for each pair (i, ki-i) in the sum, both get a contribution.
                # So grad[i] += w[ki] * a[ki-i] for valid i, AND grad[ki-i] += w[ki] * a[i] for valid i.
                # But these are the same set of updates when we iterate over all valid i!
                # So: grad[i] += w[ki] * a[ki-i] for all valid i is sufficient.
                
                grad[indices] += wi * a[ki - indices]
        
        # Hmm wait, I think I'm overcomplicating this. Let me just use the FFT cross-correlation.
        # The autocorrelation b = conv(a, a). The "cross-correlation" of w with a (where w has length 2n-1)
        # gives the gradient.
        
        # Actually, let me just use the simple formula:
        # grad[i] = sum_k w[k] * (a[k-i] if 0<=k-i<n else 0)
        # This is the cross-correlation of w and a.
        # Using FFT: cross_corr = ifft(fft(w_padded) * fft(a_padded))[n-1:]
        # where w_padded has length >= 3n-2
        
        # Let me just do the direct computation since K is small.
        # The code above already does this. Let me continue.
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Mirror descent: work in log-space
        lr = 0.1 / (1.0 + 0.01 * t**0.5)
        log_x = np.log(x + 1e-30)
        log_x -= lr * m_hat / (np.sqrt(v_hat) + eps)
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        x_new = np.maximum(x_new, 1e-25)
        x_new = x_new / x_new.sum()
        x = x_new
        
        # Periodic restart
        if stall > 2000 and stall % 500 == 0:
            noise_scale = 0.01 * stall / 2000.0
            noise = noise_scale * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x = x_r / x_r.sum()
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            stall = 0
        
        if verbose and t % 200 == 0:
            print("    MD t={:5d} ({:.1f}s) J={:.7f} stall={}".format(t, time.monotonic() - t0, best_val, stall))
    
    return best_a, best_val

def peak_targeting_descent(a0, time_limit, verbose=True):
    """Directly target reducing the autocorrelation peak."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    stall = 0
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 3000:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_M - 1e-14:
            best_M = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find the peak lag
        k_peak = int(np.argmax(ac))
        
        # Compute "blame" for each coordinate: how much a[i] contributes to b[k_peak]
        i_lo = max(0, k_peak - n + 1)
        i_hi = min(n - 1, k_peak)
        indices = np.arange(i_lo, i_hi + 1)
        blame = np.zeros(n, dtype=np.float64)
        blame[indices] = a[indices] * a[k_peak - indices]
        
        # Find coordinates with highest blame
        top_blame = np.argsort(blame)[::-1][:50]
        
        # Find "quiet" positions (low blame for all top lags)
        top_lags = np.argsort(ac)[::-1][:10]
        pressure = np.zeros(n, dtype=np.float64)
        for ki in top_lags:
            ki = int(ki)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            if lo <= hi:
                idx = np.arange(lo, hi + 1)
                pressure[idx] += a[idx] * a[ki - idx]
        
        # Quiet positions are those with low pressure
        quiet = np.argsort(pressure)[:20]
        
        # Try moving mass from high-blame to quiet positions
        improved = False
        for src in top_blame[:10]:
            src = int(src)
            if blame[src] < 1e-20:
                continue
            for dst in quiet:
                dst = int(dst)
                if dst == src:
                    continue
                # Try several fractions
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-20:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < M_curr - 1e-15:
                        a = a_t
                        M_curr = Mt
                        if Mt < best_M:
                            best_M = Mt
                            best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation to escape
            if stall > 200:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_M:
                            best_M = Mn
                            best_a = a.copy()
                            stall = 0
        
        if verbose and t % 500 == 0:
            print("    PT t={:5d} ({:.1f}s) J={:.7f} stall={}".format(t, time.monotonic() - t0, 2 * n * best_M, stall))
    
    return best_a, 2.0 * n * best_M
