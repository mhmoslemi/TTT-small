Let me also think about the mathematical structure. The objective is:
`f(a) = 2n * max_k (a*a)[k] / (sum a)^2`

At the optimum, multiple convolution coefficients should be "active" (at the maximum). The KKT conditions would tell us the structure.

For the coordinate descent step: if we move a small amount δ from position s to position d, the change in c[k] is:
`Δc[k] = δ*(a[k-d] - a[k-s])` (approximately, for small δ)

plus the second-order term `δ^2` at specific positions.

For the max to decrease, we need all currently-active k to decrease:
`δ*(a[k-d] - a[k-s]) < 0` for all active k.

This means `a[k-d] < a[k-s]` for all active k, or the "pressure" from active peaks should be higher at s than at d.

This is exactly what the coordinate_polish function does with the pressure calculation.

    gv = np.zeros(len(v))
    if ref > 0:
        gv[:ref] = grad_u[:ref]
    if ref < n - 1:
        gv[ref:] = grad_u[ref + 1:]
    gv = np.where(np.isfinite(gv), gv, 0.0)
    return f, gv

def _lbfegs(a0, beta, mask, maxfun=400, reg=1e-12):
    a = _norm1(a0)
    n = len(a)
    if n <= 1:
        return a
    v0, ref = _make_v(a)
    if maxfun <= 0:
        return a
    t0 = time.monotonic()
    try:
        res = scipy.optimize.minimize(
            _objgrad_v, v0,
            args=(beta, ref, mask, reg),
            method="L-BFGS-B", jac=True,
            options={"maxfun": maxfun, "maxiter": maxfun + 20,
                     "ftol": 1e-12, "gtol": 1e-7, "disp": False}
        )
        if hasattr(res, "x"):
            vx = np.asarray(res.x, dtype=np.float64)
            if len(vx) == len(v0) and np.all(np.isfinite(vx)):
                a_out = _norm1(_from_v(vx, ref))
                return a_out
    except Exception:
        pass
    return a

def _active_top(c, mask, k):
    c = np.asarray(c, dtype=np.float64)
    sz = len(c)
    k = int(min(k, sz))
    if k <= 0:
        return np.array([], dtype=int)
    m = c.copy()
    if mask is not None:
        m[~mask] = -np.inf
    return np.argsort(m)[::-1][:k]

def _cross_pressure(pf, a):
    """Compute pressure[i] = sum_k pf[k] * a[k-i] for valid k-i."""
    n = len(a)
    size = len(pf)
    total = size + n - 1
    Nf = _next_pow2(total)
    Ara = np.fft.rfft(a, Nf)
    Pr = np.fft.rfft(pf[::-1], Nf)
    conv = np.fft.irfft(Ara * Pr, Nf)[:total]
    idx = size - 1 - np.arange(n)
    out = conv[idx]
    return np.where(np.isfinite(out) & (out > 0), out, 0.0)

def _game_direction(f, active_idx, iters=25, seed=0):
    f = _norm1(f)
    n = len(f)
    size = 2 * n - 1
    active_idx = np.asarray(active_idx, dtype=int)
    K = len(active_idx)
    if K == 0:
        return np.ones(n) / n
    rng = np.random.default_rng(seed)
    logits = rng.normal(0.0, 0.1, K)
    for it in range(iters):
        pf = np.zeros(size)
        pf[active_idx] = _softmax(logits)
        pressure = _cross_pressure(pf, f)
        pmx = float(pressure.max())
        pmn = float(pressure.min())
        ptp = pmx - pmn
        if ptp < 1e-30:
            g = np.ones(n) / n
        else:
            scaled = (pressure - pmn) / ptp
            w = np.exp(-6.0 * scaled)
            w = np.where(np.isfinite(w) & (w > 0), w, 0.0)
            w += 1e-12
            g = w / float(w.sum())
            g = 0.95 * g + 0.05 * (np.ones(n) / n)
            g /= float(g.sum())
        h = _fft_cross(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmac_ = float(ha.max())
        hmin = float(ha.min())
        hptp = hmac_ - hmin
        if hptp > 1e-30:
            eta = 10.0 / hptp
            logits += eta * (ha - float(ha.mean()))
    return g

def _lp_direction(f, active_idx, time_limit=5.0):
    f = _norm1(f)
    n = len(f)
    active_idx = np.asarray(active_idx, dtype=int)
    K = len(active_idx)
    if K == 0 or K > 2000:
        return None
    row_parts, col_parts, val_parts = [], [], []
    for r, k in enumerate(active_idx):
        k = int(k)
        s = max(0, k - n + 1)
        e = min(n, k + 1)
        if s >= e:
            continue
        cols = np.arange(s, e, dtype=np.int32)
        row_parts.append(np.full(len(cols), r, dtype=np.int32))
        col_parts.append(cols)
        val_parts.append(f[k - cols].astype(np.float64))
    if not row_parts:
        return None
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    neg_one = scipy.sparse.csc_matrix(np.ones((K, 1)), dtype=np.float64)
    A_ub = scipy.sparse.hstack((A, neg_one), format="csr")
    b_ub = np.zeros(K)
    A_eq = np.ones((1, n + 1))
    b_eq = np.ones(1)
    bounds = [(0.0, None)] * n + [(None, None)]
    try:
        res = scipy.optimize.linprog(
            c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
            bounds=bounds, method="HiGHS",
            options={"disp": False, "time_limit": float(time_limit)}
        )
        if res is not None and getattr(res, "success", False):
            g = np.asarray(res.x[:n], dtype=np.float64)
            g = np.where(np.isfinite(g) & (g > 0), g, 0.0)
            gs = float(g.sum())
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None

def _fw_search(a0, budget, mask, active_max=2000, game_iters=20, verbose=True):
    t0 = time.monotonic()
    a = _norm1(a0)
    n = len(a)
    c = _fft_conv(a)
    M = float(c.max()) if mask is None else float(c[mask].max())
    best_a = a.copy()
    best_M = M
    active = _active_top(c, mask, active_max)
    stall = 0
    t = 0
    while time.monotonic() - t0 < budget - 3 and stall < 100:
        t += 1
        time_left = budget - (time.monotonic() - t0)
        if time_left < 15:
            break
        g = None
        if t % 6 == 0 and time_left > 80 and len(active) <= 1000:
            g = _lp_direction(a, active, min(6.0, time_left * 0.3))
        if g is None:
            init_log = None
            if len(active) > 0:
                av = c[active]
                av = np.where(av > 1e-300, av, 1e-300)
                init_log = np.log(av)
                init_log -= float(init_log.max())
            g = _game_direction(a, active, iters=game_iters, seed=7919 * t + 13, init_logits=init_log) if hasattr(_game_direction, '__code__') else _game_direction(a, active, iters=game_iters, seed=7919 * t + 13)
        if g is None:
            break
        cg = _fft_conv(g)
        fg = _fft_cross(a, g)
        best_alpha = 0.0
        best_m = M
        for alpha in [0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0]:
            one = 1.0 - alpha
            val = one * one * c + 2.0 * alpha * one * fg + alpha * alpha * cg
            if mask is None:
                m = float(val.max())
            else:
                m = float(val[mask].max())
            if math.isfinite(m) and m < best_m:
                best_m = m
                best_alpha = alpha
        if best_alpha > 0 and best_m < M - 1e-16:
            one = 1.0 - best_alpha
            c = one * one * c + 2.0 * best_alpha * one * fg + best_alpha * best_alpha * cg
            c = np.where(np.isfinite(c) & (c > 0), c, 0.0)
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            s = float(a.sum())
            if s < 1e-15:
                break
            a = a / s
            M = best_m
            stall = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            active = _active_top(c, mask, active_max)
            if verbose and (t <= 2 or t % 10 == 0):
                print("  FW t={} M={:.8f} alpha={:.3f} active={}".format(t, best_M, best_alpha, len(active)))
        else:
            stall += 1
            if stall % 10 == 9:
                rng = np.random.default_rng(4242 + t)
                pert = rng.normal(0.0, 0.02, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-12)
                a /= float(a.sum())
                c = _fft_conv(a)
                M = float(c.max()) if mask is None else float(c[mask].max())
                active = _active_top(c, mask, active_max)
                stall = 0
    return best_a

def _apply_move(c, a, s, d, tmove):
    c_new = c.copy()
    c_new[d:d + len(a)] += 2.0 * tmove * a
    c_new[s:s + len(a)] -= 2.0 * tmove * a
    t2 = tmove * tmove
    if 2 * d < len(c_new):
        c_new[2 * d] += t2
    if 2 * s < len(c_new):
        c_new[2 * s] += t2
    if s + d < len(c_new):
        c_new[s + d] -= 2.0 * t2
    np.maximum(c_new, 0.0, out=c_new)
    np.nan_to_num(c_new, copy=False, nan=0.0, posinf=0.0, neginf=0.0)
    return c_new

def _coord_polish(a0, budget, mask, verbose=True):
    t0 = time.monotonic()
    a = _norm1(a0)
    n = len(a)
    c = _fft_conv(a)
    M = float(c.max()) if mask is None else float(c[mask].max())
    best_a = a.copy()
    best_M = M
    stall = 0
    rnd = 0
    rng = np.random.default_rng(12345)
    while time.monotonic() - t0 < budget - 2 and stall < 200:
        rnd += 1
        if n <= 1:
            break
        top_k = min(50, 2 * n - 1)
        top_idx = _active_top(c, mask, top_k)
        if len(top_idx) == 0:
            break
        p_top = np.zeros(2 * n - 1)
        p_top[top_idx] = 1.0 / len(top_idx)
        pressure = _cross_pressure(p_top, a)
        src_count = min(10, n)
        dst_count = min(15, n)
        src_score = pressure + n * a
        src_cands = np.argpartition(-src_score, min(src_count, n - 1))[:src_count]
        dst_score = -pressure + 0.01 * a
        dst_cands = np.argpartition(-dst_score, min(dst_count, n - 1))[:dst_count]
        best_cand = None
        for srr in src_cands:
            s = int(srr)
            if a[s] < 1e-12:
                continue
            for drr in dst_cands:
                d = int(drr)
                if d == s:
                    continue
                for frac in (0.01, 0.03, 0.07, 0.15):
                    tmove = a[s] * frac
                    if tmove < 1e-16:
                        continue
                    tmove = min(tmove, a[s] * 0.99)
                    c_new = _apply_move(c, a, s, d, tmove)
                    m_new = float(c_new.max()) if mask is None else float(c_new[mask].max())
                    if m_new < M - 1e-18:
                        if best_cand is None or m_new < best_cand[0]:
                            best_cand = (m_new, s, d, tmove)
        if best_cand is None:
            for _ in range(30):
                s = int(rng.integers(n))
                d = int(rng.integers(n))
                if s == d or a[s] < 1e-12:
                    continue
                tmove = a[s] * float(rng.uniform(0.005, 0.05))
                if tmove < 1e-16:
                    continue
                c_new = _apply_move(c, a, s, d, tmove)
                m_new = float(c_new.max()) if mask is None else float(c_new[mask].max())
                if m_new < M - 1e-18:
                    if best_cand is None or m_new < best_cand[0]:
                        best_cand = (m_new, s, d, tmove)
        if best_cand is not None:
            m_new, s, d, tmove = best_cand
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = _fft_conv(a)
            M = float(c.max()) if mask is None else float(c[mask].max())
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and (rnd <= 5 or rnd % 50 == 0):
                print("  Coord r={} M={:.8f}".format(rnd, best_M))
        else:
            stall += 1
            if stall % 15 == 14:
                for _ in range(50):
                    s = int(rng.integers(n))
                    d = int(rng.integers(n))
                    if s == d or a[s] < 1e-12:
                        continue
                    tmove = a[s] * float(rng.uniform(0.002, 0.02))
                    if tmove < 1e-16:
                        continue
                    c_new = _apply_move(c, a, s, d, tmove)
                    m_new = float(c_new.max()) if mask is None else float(c_new[mask].max())
                    if m_new < M - 1e-18:
                        a = best_a.copy()
                        a[s] -= tmove
                        a[d] += tmove
                        a = np.maximum(a, 0.0)
                        a /= float(a.sum())
                        c = _fft_conv(a)
                        M = float(c.max()) if mask is None else float(c[mask].max())
                        stall = 0
                        break
    return best_a

def _interp(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if len(a) == 0:
        return np.ones(max(1, int(new_n)))
    new_n = int(max(1, new_n))
    if len(a) == new_n:
        return _norm1(a)
    old_x = np.linspace(0.0, 1.0, len(a))
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return _norm1(y)

def _default_seq(n=5000):
    x = np.linspace(0.0, 1.0, int(n))
    return _norm1(np.sin(np.pi * x) ** 2)

def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    total_budget = max(120.0, min(float(budget_s) - 60.0, 1740.0))
    print("Start: budget={:.1f}s".format(total_budget))

    # Get parent
    seq_best = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        np.clip(p, 0.0, 1000.0, out=p)
        if len(p) > 0 and p.sum() > 1e-10:
            seq_best = _norm1(p)
    except Exception:
        pass
    if seq_best is None:
        seq_best = _default_seq(5000)
    best_score = _eval(seq_best)
    print("Parent: n={} score={}".format(len(seq_best), best_score))

    # Working sequence (compress if very long)
    work_n = min(len(seq_best), 8000)
    work_seq = _interp(seq_best, work_n) if len(seq_best) > work_n else seq_best.copy()
    ws = _eval(work_seq)
    if math.isfinite(ws) and ws < best_score:
        seq_best = work_seq.copy()
        best_score = ws

    def _try_update(a, label):
        nonlocal seq_best, best_score
        if a is None:
            return
        try:
            arr = np.asarray(a, dtype=np.float64)
            arr = np.where(np.isfinite(arr), arr, 0.0)
            np.maximum(arr, 0.0, out=arr)
            s = float(arr.sum())
            if len(arr) == 0 or s < 0.01:
                return
            arr = arr / s
            sc = _eval(arr)
            if math.isfinite(sc) and sc < best_score - 1e-14:
                seq_best = arr.copy()
                best_score = sc
                print("  NEW BEST [{}]: {:.10f} n={} t={:.1f}s".format(
                    label, sc, len(arr), time.monotonic() - t_start))
        except Exception:
            pass

    def _refresh_work():
        nonlocal work_seq
        target_n = min(len(seq_best), work_n) if len(seq_best) <= work_n else work_n
        work_seq = _interp(seq_best, target_n)

    _refresh_work()

    # === Phase 1: L-BFGS with increasing beta ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 300:
        phase_budget = min(time_left * 0.35, 600.0)
        print("\n=== Phase 1: L-BFGS ({:.0f}s) ===".format(phase_budget))
        n_w = len(work_seq)
        v, ref = _make_v(work_seq)
        # Estimate cost
        t_est = time.monotonic()
        try:
            _objgrad_v(v, 30.0, ref)
            cost = max(0.02, time.monotonic() - t_est)
        except Exception:
            cost = 0.3
        print("  Cost estimate: {:.3f}s/eval".format(cost))
        bethas = [10.0, 30.0, 60.0, 120.0, 250.0, 500.0, 1000.0, 2000.0]
        cur_a = work_seq.copy()
        bi = 0
        while bi < len(bethas) and time.monotonic() - t_start < total_budget - 200:
            beta = bethas[bi]
            # For higher beta, restrict to active peaks
            if beta >= 120:
                c_cur = _fft_conv(cur_a)
                top_k = max(50, (2 * n_w - 1) // int(beta / 100 + 1))
                top_idx = _active_top(c_cur, None, top_k)
                msk = np.zeros(2 * n_w - 1, dtype=bool)
                msk[top_idx] = True
            else:
                msk = None
            maxfun = max(40, min(500, int(phase_budget * 0.12 / cost)))
            a_new = _lbfegs(cur_a, beta, msk, maxfun=maxfun, reg=1e-12)
            _try_update(a_new, "lb_{}".format(beta))
            if time.monotonic() - t_start < total_budget - 200:
                # Also try without mask for smooth bethas
                if beta <= 60:
                    a_new2 = _lbfegs(cur_a, beta, None, maxfun=maxfun, reg=1e-12)
                    _try_update(a_new2, "lb_nb_{}".format(beta))
            _refresh_work()
            cur_a = work_seq.copy()
            bi += 1

    # === Phase 2: Frank-Wolfe ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 200:
        phase_budget = min(time_left * 0.28, 480.0)
        print("\n=== Phase 2: Frank-Wolfe ({:.0f}s) ===".format(phase_budget))
        a_fw = _fw_search(work_seq, phase_budget, None,
                          active_max=2000, game_iters=22, verbose=True)
        _try_update(a_fw, "fw")
        _refresh_work()

    # === Phase 3: Coordinate polish ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 150:
        phase_budget = min(time_left * 0.25, 350.0)
        print("\n=== Phase 3: Coord Polish ({:.0f}s) ===".format(phase_budget))
        a_cp = _coord_polish(work_seq, phase_budget, None, verbose=True)
        _try_update(a_cp, "coord")
        _refresh_work()

    # === Phase 4: Length exploration ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 250:
        print("\n=== Phase 4: Length probes ===")
        base_len = len(seq_best) if len(seq_best) <= 8000 else len(work_seq)
        lengths = []
        for dl in [200, -200, 500, -500, 1000, 300, -300]:
            target = base_len + dl
            if 2000 <= target <= 8000 and target not in lengths:
                lengths.append(target)
        for target in lengths[:5]:
            time_left = total_budget - (time.monotonic() - t_start)
            if time_left < 100:
                break
            print("  Trying n={}".format(target))
            a0 = _interp(seq_best, target) if len(seq_best) <= 8000 else _interp(work_seq, target)
            # Quick L-BFGS
            if time_left > 150:
                v_t, ref_t = _make_v(a0)
                t_est2 = time.monotonic()
                try:
                    _objgrad_v(v_t, 50.0, ref_t)
                    cost2 = max(0.03, time.monotonic() - t_est2)
                except Exception:
                    cost2 = 0.5
                mf = max(30, min(200, int(time_left * 0.4 / cost2)))
                a_new = _lbfegs(a0, 80.0, None, maxfun=mf)
                _try_update(a_new, "len_{}".format(target))
            # Quick FW
            time_left = total_budget - (time.monotonic() - t_start)
            if time_left > 80:
                a_fw2 = _fw_search(a0, min(time_left * 0.5, 80.0), None,
                                   active_max=1200, game_iters=15, verbose=False)
                _try_update(a_fw2, "lenfw_{}".format(target))
            _refresh_work()

    # === Phase 5: Final intensive coord polish ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 80:
        phase_budget = min(time_left * 0.5, 150.0)
        print("\n=== Phase 5: Final Coord ({:.0f}s) ===".format(phase_budget))
        _refresh_work()
        a_cp2 = _coord_polish(work_seq, phase_budget, None, verbose=False)
        _try_update(a_cp2, "final_coord")

    # === Phase 6: Final L-BFGS polish on best ===
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 60:
        print("\n=== Phase 6: Final L-BFGS ===")
        _refresh_work()
        c_fin = _fft_conv(work_seq)
        top_k = min(300, len(c_fin))
        top_idx = _active_top(c_fin, None, top_k)
        msk = np.zeros(len(c_fin), dtype=bool)
        msk[top_idx] = True
        v_f, ref_f = _make_v(work_seq)
        t_est3 = time.monotonic()
        try:
            _objgrad_v(v_f, 500.0, ref_f, msk)
            cost3 = max(0.03, time.monotonic() - t_est3)
        except Exception:
            cost3 = 0.5
        mf_fin = max(30, min(300, int(time_left * 0.6 / cost3)))
        a_fin = _lbfegs(work_seq, 500.0, msk, maxfun=mf_fin)
        _try_update(a_fin, "final_lb")
        # Also try without mask
        time_left = total_budget - (time.monotonic() - t_start)
        if time_left > 40:
            a_fin2 = _lbfegs(work_seq, 300.0, None, maxfun=max(20, int(time_left * 0.5 / cost3)))
            _try_update(a_fin2, "final_lb_nomask")

    # Final validation
    _refresh_work()
    a_cp3 = _coord_polish(work_seq, min(20.0, max(5.0, total_budget - (time.monotonic() - t_start))), None, verbose=False)
    _try_update(a_cp3, "very_final")

    final = seq_best
    final = np.asarray(final, dtype=np.float64)
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if len(final) == 0 or s < 0.01:
        final = _default_seq(5000)
    else:
        final = final / s

    fv = _eval(final)
    elapsed = time.monotonic() - t_start
    print("\nFINAL: score={} n={} elapsed={:.1f}s".format(fv, len(final), elapsed))
    return [float(x) for x in final]
