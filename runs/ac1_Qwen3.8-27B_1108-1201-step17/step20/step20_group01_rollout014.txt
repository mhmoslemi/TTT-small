"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max annealing with FFT-accelerated gradient in softmax parameterization
2. Projected gradient descent with backtracking line search
3. Structured pair-move local search
4. Multi-length exploration with interpolation

import numpy as np
import time
from scipy.optimize import minimize

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def compute_autocorr(a):
    return fast_autocorr(a)

def boltzmann_weights(b, T):
    b_scaled = b / T
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    return exp_b / exp_b.sum()

def softmax_grad_a(a, w, n):
    """Compute grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m] using FFT."""
    a_flip = a[::-1].copy()
    req_size = n + (2 * n - 1)
    fft_size = 1 << (req_size - 1).bit_length()
    A = np.fft.rfft(a_flip, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    return 2.0 * conv_result[n - 1 : 2 * n - 1]

def softmax_obj_grad_z(z, T):
    """Objective and gradient in z-space (a = softmax(z))."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    n = len(a)
    b = fast_autocorr(a)
    w = boltzmann_weights(b, T)
    grad_a = softmax_grad_a(a, w, n)
    obj = T * (np.log(np.sum(np.exp(b / T - (b / T).max())) + 1e-300))
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)
    return obj, grad_z

def anneal_softmax(a0, time_limit, T_init=None, T_min=0.005, decay=0.88, 
                   max_iter=100, lr=1.0, verbose=True):
    """Gradient descent in softmax parameterization with temperature annealing."""
    n = len(a0)
    a = normalize(a0).copy()
    # Initialize z from a
    z = np.log(a + 1e-30)
    z -= z.max()
    
    best_val = 2.0 * n * fast_autocorr(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    if T_init is None:
        b = fast_autocorr(a)
        T_init = max(0.01, (b.max() - np.median(b)) * 0.3)
    
    T = T_init
    total_iters = 0
    
    while T > T_min and time.monotonic() - t0 < time_limit:
        # Run a few iterations at this T
        for it in range(max_iter):
            if time.monotonic() - t0 > time_limit:
                break
            total_iters += 1
            
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_autocorr(a)
            w = boltzmann_weights(b, T)
            grad_a = softmax_grad_a(a, w, n)
            mean_grad = np.dot(a, grad_a)
            grad_z = a * (grad_a - mean_grad)
            
            # Line search
            lr_try = lr
            a_old = a.copy()
            obj_old = T * np.log(np.sum(np.exp(b / T - (b/T).max())) + 1e-300)
            
            found = False
            for ls in range(10):

                a_try = np.exp(z_try - z_try.max())

                obj_try = T * np.log(np.sum(np.exp(b_try / T - (b_try/T).max())) + 1e-300)
                
                if obj_try < obj_old - 1e-12:
                    z = z_try
                    a = a_try
                    found = True
                    break
                lr_try *= 0.5
            
            if not found:
                break
            
            # Update best
            b_curr = fast_autocorr(a)
            val_curr = 2.0 * n * b_curr.max()
            if val_curr < best_val - 1e-12:
                best_val = val_curr
                best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            b_now = fast_autocorr(a)
            print("    Anneal T={:.5f} ({:.1f}s) iter={} peak={:.7f} best={:.7f}".format(
                T, elapsed, total_iters, 2.0*n*b_now.max(), best_val))

    return best_a, best_val

def pair_move_search(a0, time_limit, n_top=20, verbose=True):
    """Structured pair-move local search."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    max_stall = 5000
    size = 2 * n - 1

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_autocorr(a)
        M = b.max()
        
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find top lags
        top_lags = np.argsort(b)[::-1][:n_top]
        
        # For each top lag, find good pair moves
        improved = False
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            
            # Pressure: how much each position contributes to b[ki]
            # b[ki] = sum_i a[i] * a[ki-i] for valid i
            # Contribution of a[j] to b[ki]: a[ki-j] (if valid)
            # Moving from i to j: change in b[ki] = -a[ki-i]*eps + a[ki-j]*eps
            # = eps * (a[ki-j] - a[ki-i])
            # We want a[ki-j] < a[ki-i], i.e., j has lower "pressure"
            
            # Compute pressure for all positions
            pressure = np.zeros(n)
            for j in range(n):
                # a[j] appears in b[ki] at positions where i=j or ki-i=j
                # i.e., i=j (gives a[ki-j]) and i=ki-j (gives a[j])
                # Total contribution of a[j] to b[ki]: a[j]*a[ki-j] (if valid) + a[j]*a[j-ki] (if valid)
                # But for the gradient: db[ki]/da[j] = 2*a[ki-j] (if 0<=ki-j<n)
                if 0 <= ki - j < n:
                    pressure[j] = a[ki - j]
                if 0 <= j - ki < n:
                    pressure[j] += a[j - ki]
            
            # We want to move from high-pressure to low-pressure
            # But also need to check other lags don't increase
            src_candidates = np.argsort(pressure)[::-1][:50]  # top 50 by pressure
            dst_candidates = np.argsort(pressure)[:50]  # bottom 50 by pressure
            
            for si in src_candidates[:20]:
                if time.monotonic() - t0 > time_limit:
                    break
                si = int(si)
                if a[si] < 1e-12:
                    continue
                
                for di in dst_candidates[:20]:
                    di = int(di)
                    if di == si:
                        continue
                    if a[di] > 0.3:  # don't move to already-large positions
                        continue
                    
                    # Try moving a small amount
                    for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                        amt = a[si] * frac
                        a_try = a.copy()
                        a_try[si] -= amt
                        a_try[di] += amt
                        a_try = np.maximum(a_try, 0.0)
                        s = a_try.sum()
                        if s < 1e-15:
                            continue
                        a_try = a_try / s
                        b_try = fast_autocorr(a_try)
                        M_try = b_try.max()
                        if M_try < best_m - 1e-14:
                            a = a_try
                            best_m = M_try
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            
            if not improved and stall > 200:
                # Random perturbation to escape
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.005, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    b_new = fast_autocorr(a)
                    if b_new.max() < best_m:
                        best_m = b_new.max()
                        best_a = a.copy()
                        stall = 0
            if improved:
                break
        
        if not improved:
            stall += 1
        
        if verbose and stall % 500 == 0 and stall > 500:
            print("    PairMove ({:.1f}s) M={:.7f} stall={}".format(
                time.monotonic() - t0, 2.0 * n * best_m, stall))
    
    return best_a, 2.0 * n * best_m

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss
