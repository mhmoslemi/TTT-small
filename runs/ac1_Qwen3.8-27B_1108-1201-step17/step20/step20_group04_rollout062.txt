def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_corr(a, b):
    """Compute full linear convolution of a and b using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1

    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        d_val -= d_val.mean()
        d_val = np.maximum(d_val, -a)
        return d_val, t_var.value if t_var.value is not None else c.max()
    
    return None


def lp_optimize(a0, time_limit, solver, verbose=True, seed=42, 
                k_min=10, k_max=100, k_growth=1.01, max_iters=None):
    """
    LP-based optimization with adaptive active set and analytic line search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    k_active = k_min
    
    if max_iters is None:
        max_iters = 100000
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_autocorr(a)
        c_max = c.max()
        
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive active set size
        # Start small, grow as we flatten
        k_active = min(k_max, int(k_min * (k_growth ** (stall / 10.0))))
        
        # Select active set: top k_active peaks
        top_idx = np.argsort(c)[::-1][:k_active]
        
        # Try to get LP direction
        result = lp_direction(a, top_idx, solver)
        
        if result is None:
            # Fallback: random perturbation
            d = rng.normal(0, 1, n)
            d -= d.mean()
            d = d / max(1, np.max(np.abs(d)))
            alpha = 0.01
        else:
            d, t_lp = result
            # Line search
            alpha = analytic_line_search(a, d, c, c_max, n)
            if alpha < 1e-8:
                # Try a smaller step
                alpha = 0.01
                feas_mask = d < 0
                if feas_mask.any():
                    alpha_max_f = np.min(a[feas_mask] / (-d[feas_mask]))
                    alpha = min(alpha, alpha_max_f * 0.5)
        
        # Apply update
        a_new = a + alpha * d
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        a = a_new
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP t=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + 
                  repr(round(J, 6)) + ", k=" + str(k_active) + ", stall=" + str(stall))
        
        if stall > 500:
            # Restart with perturbation
            stall = 0
            k_active = k_min
            noise = rng.normal(0, 0.01, n) * best_a
            a_r = best_a + noise
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r
    
    return best_a, best_m


def nesterov_opt(a0, time_limit, beta_start=50.0, beta_end=3000.0,
                 beta_growth=1.006, eta_base=0.5, top_k_ratio=0.02,
                 restart_every=2000, verbose=True, seed=42):
    """Nesterov accelerated projected gradient with log-sum-exp smoothing."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        c = fast_autocorr(y)
        if top_k < len(c):
            idx = np.argsort(c)[::-1][:top_k]
            c_use = np.zeros(len(c))
            c_use[idx] = c[idx]
            c_max = c[idx].max()
        else:
            c_use = c
            c_max = c.max()
        shifted = beta * (c_use - c_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        total = exp_vals.sum()
        if total < 1e-300:
            p = np.zeros_like(c_use)
            p[np.argmax(c)] = 1.0
        else:
            p = exp_vals / total

        # Gradient via FFT
        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(y, fft_size)
        P = np.fft.rfft(p, fft_size)
        a_rev = y[::-1]
        Conv = np.fft.irfft(P * np.fft.rfft(a_rev, fft_size), fft_size)
        grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]

        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.03 / (1.0 + 0.05 * (t // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + ", beta=" + str(round(beta, 1)))

    return best_a, best_max_b


def peak_surgery(a0, time_limit, verbose=True, seed=77):
    """Targeted local search for peak reduction."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 5000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_top = min(30, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]
        improved = False

        for k in top_idx[:15]:
            if improved:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            if products.max() < 1e-20:
                continue
            src_i = int(idx[np.argmax(products)])
            partner = k_int - src_i
            if partner < 0 or partner >= n:
                continue
            if a[src_i] >= a[partner]:
                reduce_i = src_i
            else:
                reduce_i = partner
            if a[reduce_i] < 1e-14:
                continue
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[reduce_i] * frac
                if amt < 1e-15:
                    continue
                for _ in range(5):
                    dst = int(rng.integers(n))
                    if dst == reduce_i:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_i] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and no_improve > 500:
            for _ in range(10):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break

        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSurgery (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve))

    return best_a, best_m


def multiplicative_update(a0, time_limit, beta=100.0, n_top=20, verbose=True, seed=13):
    """
    Multiplicative (mirror) descent on the simplex.
    Update: a_new[i] = a[i] * exp(-eta * grad[i])
    This is more natural for the simplex constraint.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    eta = 0.1
    beta = 100.0

    for t in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Compute softmax weights on top peaks
        top_idx = np.argsort(c)[::-1][:n_top]
        c_use = np.zeros(2 * n - 1)
        c_use[top_idx] = c[top_idx]
        shifted = beta * (c_use - c_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        total = exp_vals.sum()
        if total < 1e-300:
            p = np.zeros(2 * n - 1)
            p[np.argmax(c)] = 1.0
        else:
            p = exp_vals / total

        # Gradient: (2/beta) * cross_correlation(p, a) [n:n-1+n]
        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        P = np.fft.rfft(p, fft_size)
        a_rev = a[::-1]
        Conv = np.fft.irfft(P * np.fft.rfft(a_rev, fft_size), fft_size)
        grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]

        # Multiplicative update
        log_step = -eta * grad
        log_step = np.clip(log_step, -2, 2)  # prevent overflow
        a_new = a * np.exp(log_step)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        b_new = fast_autocorr(a_new)
        max_new = b_new.max()
        if max_new < best_m:
            a = a_new
        else:
            # Try smaller step
            for eta_mult in [0.5, 0.1, 0.01]:
                a_try = a * np.exp(eta_mult * log_step)
                a_try = np.maximum(a_try, 1e-15)
                a_try = a_try / a_try.sum()
                if fast_autocorr(a_try).max() < best_m:
                    a = a_try
                    break

        if t % 200 == 0:
            beta = min(beta * 1.05, 5000)
            eta = max(eta * 0.98, 0.001)

        if stall > 3000:
            stall = 0
            noise = rng.normal(0, 0.02, n) * best_a
            a_r = best_a + noise
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r
            beta = max(beta * 0.5, 20)
            eta = 0.1

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    MultUpd t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + ", beta=" + str(round(beta, 1)))

    return best_a, best_m


def pairwise_cd(a0, time_limit, verbose=True, seed=123):
    """Fine-grained pairwise coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        c = fast_autocorr(a)
        threshold = np.percentile(c, 95)
        peak_positions = np.where(c > threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.argsort(c)[::-1][:20]
        improved = False

        for k in peak_positions[:15]:
            if improved and (time.monotonic() - t0 > time_limit):
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            top_pairs = np.argsort(products)[::-1][:3]
            for pi in top_pairs:
                if improved:
                    break
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n or partner == src_i:
                    continue
                if a[src_i] < 1e-14 and a[partner] < 1e-14:
                    continue
                larger_i = src_i if a[src_i] >= a[partner] else partner
                if a[larger_i] < 1e-14:
                    continue
                for frac in [0.001, 0.003, 0.005]:
                    if improved:
                        break
                    amt = a[larger_i] * frac
                    if amt < 1e-18:
                        continue
                    for _ in range(3):
                        dst = int(rng.integers(n))
                        if dst == larger_i:
                            continue
                        a_trial = a.copy()
                        a_trial[larger_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break

        if not improved:
            stall += 1
            if stall % 100 == 0:
                anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
                eps = 0.01 / (1.0 + stall / 100.0)
                a_try = a + eps * anti * a
                a_try = np.maximum(a_try, 0.0)
                s_t = a_try.sum()
                if s_t > 1e-15:
                    a_try = a_try / s_t
                    if fast_autocorr(a_try).max() < best_m:
                        a = a_try
                        best_m = fast_autocorr(a).max()
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PairCD (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main search combining LP-based direction finding, Nesterov gradient,
    multiplicative updates, peak surgery, and pairwise CD.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        x = np.linspace(0, 1, n_base)
        a_init = (1.0 - x) ** 1.5 + 0.1
        best_seq = norm_sum1(a_init)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(n_base) + ", score=" + repr(best_score))

    n0 = len(best_seq)
    print("Starting: n=" + str(n0) + ", score=" + repr(best_score))

    solver = get_solver()

    # Phase 1: LP-based optimization (primary method)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: LP Optimization (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 60 and solver is not None:
        a_out, _ = lp_optimize(norm_sum1(best_seq), t1, solver,
                                verbose=True, seed=1)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 2: Nesterov gradient (refine)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 60:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t2,
                                beta_start=100.0, beta_end=5000.0,
                                beta_growth=1.008, eta_base=0.4,
                                top_k_ratio=0.01, restart_every=1500,
                                verbose=True, seed=2)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 3: LP optimization round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP Round 2 (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 60 and solver is not None:
        a_out, _ = lp_optimize(norm_sum1(best_seq), t3, solver,
                                verbose=True, seed=3)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp2", t_global)

    # Phase 4: Multiplicative update
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Multiplicative Update (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 60:
        a_out, _ = multiplicative_update(norm_sum1(best_seq), t4,
                                          beta=100.0, n_top=20,
                                          verbose=True, seed=4)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "mult1", t_global)

    # Phase 5: Peak surgery
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Peak Surgery (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.10, 180.0)
    if t5 > 40:
        a_out, _ = peak_surgery(norm_sum1(best_seq), t5, verbose=True, seed=5)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps1", t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-length (n=" + str(n_best) + ") ===")
    lengths_to_try = [n_best + 500, n_best + 1000, max(3000, n_best - 500)]
    lengths_to_try = [l for l in lengths_to_try if 2000 <= l <= 10000]
    
    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print("  Trying n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_ml_n" + str(new_n), t_global)
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t6b = min(remaining * 0.3, 120.0)
            if t6b > 30:
                a_try = interpolate_sequence(best_seq, new_n)
                a_out, _ = nesterov_opt(a_try, t6b,
                                        beta_start=50.0, beta_end=3000.0,
                                        beta_growth=1.008, eta_base=0.5,

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based direction finding (scipy HiGHS) with adaptive active sets
2. Analytic quadratic line search for optimal step sizes
3. Nesterov smoothed gradient as complementary approach
4. Multiplicative (mirror) descent on simplex
5. Peak surgery and pairwise CD for local polishing
6. Multi-length exploration with interpolation
Key insight: LP gives exact optimal direction for linearized max-conv problem,
while line search exploits the quadratic structure of conv(a+alpha*d, a+alpha*d).
"""

import numpy as np
import time
import math


_module_solver = None

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_corr(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

    try:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lp_direction_scipy(a, active_set, solver_backend='highs'):
    """
    Solve LP using scipy.optimize.linprog to find direction d minimizing
    max_{k in active_set} c_k(a+d) (linearized).
    
    LP:
    minimize t
    s.t. 2*(a*d)[k] + c[k] <= t  for k in active_set
         sum(d) = 0
         d >= -a
    """
    from scipy.optimize import linprog
    
    n = len(a)
    c = fast_autocorr(a)
    n_act = len(active_set)
    
    if n_act == 0:
        return None, None
    
    # Variables: x = [d_0, ..., d_{n-1}, t]
    # Objective: minimize t => c_obj = [0,...,0,1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each k in active_set: 2 * sum_i a[k-i]*d_i - t <= -c[k]
    # Matrix row: [2*a[k-i_end : k-i_start+1] embedded in d part, -1 in t part]
    A_ub = np.zeros((n_act, n + 1))
    b_ub = np.zeros(n_act)
    
    for idx, k in enumerate(active_set):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        coeff = a[k_int - i_end : k_int - i_start + 1]
        A_ub[idx, i_start:i_end + 1] = 2.0 * coeff
        A_ub[idx, -1] = -1.0
        b_ub[idx] = -c[k_int]
    
    # Also: d >= -a => -d <= a => [-I, 0] @ x <= a
    neg_I = -np.eye(n)
    A_ub2 = np.hstack([neg_I, np.zeros((n, 1))])
    b_ub2 = a.copy()
    
    A_all = np.vstack([A_ub, A_ub2])
    b_all = np.concatenate([b_ub, b_ub2])
    
    # Equality: sum(d) = 0 => [1,...,1, 0] @ x = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: d[i] >= -a[i], t >= -inf (but in practice t > 0)
    bounds = [(-a[i], None) for i in range(n)] + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_all, b_ub=b_all, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method=solver_backend)
    except Exception as e:
        return None, None
    
    if res.status == 0 and res.x is not None:
        d_val = res.x[:n]
        t_val = res.x[n]
        # Clean up numerical issues
        d_val = np.maximum(d_val, -a + 1e-12)
        d_val -= d_val.mean()
        d_val = np.maximum(d_val, -a)
        return d_val, t_val
    
    return None, None


def lp_direction_cvxpy(a, active_set, solver):
    """Solve LP using cvxpy as fallback."""
    import cvxpy as cp
    
    n = len(a)
    c = fast_autocorr(a)
    
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(d_var) == 0,
        d_var >= -a
    ]
    
    for k in active_set:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        coeff = a[k_int - i_end : k_int - i_start + 1]
        expr = coeff @ d_var[i_start:i_end + 1]
        constraints.append(2 * expr + c[k_int] <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if d_var.value is not None:
        d_val = np.asarray(d_var.value, dtype=np.float64)
        d_val = np.maximum(d_val, -a + 1e-12)
        d_val -= d_val.mean()
        d_val = np.maximum(d_val, -a)
        t_val = t_var.value if t_var.value is not None else c.max()
        return d_val, t_val
    
    return None, None


def analytic_line_search(a, d, c, c_max, n, n_grid=80):
    """
    Find optimal step size alpha for a + alpha*d.
    Uses the fact that conv(a+alpha*d, a+alpha*d)[k] is quadratic in alpha.
    """
    # Compute cross terms
    ad = cross_corr(a, d)
    dd = fast_autocorr(d)
    
    # Feasibility: a + alpha*d >= 0
    alpha_max_feas = float('inf')
    neg_mask = d < -1e-15
    if neg_mask.any():
        alpha_max_feas = np.min(a[neg_mask] / (-d[neg_mask]))
    
    alpha_max_feas = max(alpha_max_feas * 0.999, 0.0)
    if alpha_max_feas <= 1e-15:
        return 0.0
    
    # Coarse grid search
    alphas = np.linspace(0, alpha_max_feas, n_grid)
    best_alpha = 0.0
    best_obj = c_max
    
    for alpha in alphas:
        c_new = c + 2.0 * alpha * ad + alpha * alpha * dd
        obj = c_new.max()
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha
    
    # Local refinement
    if best_alpha > 0:
        step = alpha_max_feas / n_grid
        for _ in range(40):
            candidates = [
                best_alpha - step,
                best_alpha - step * 0.33,
                best_alpha,
                best_alpha + step * 0.33,
                best_alpha + step,
            ]
            candidates = [max(0.0, min(ca, alpha_max_feas)) for ca in candidates]
            best_local = best_alpha
            best_local_obj = best_obj
            for ca in candidates:
                c_new = c + 2.0 * ca * ad + ca * ca * dd
                obj = c_new.max()
                if obj < best_local_obj:
                    best_local_obj = obj
                    best_local = ca
            if abs(best_local - best_alpha) < 1e-12:
                break
            best_alpha = best_local
            best_obj = best_local_obj
            step *= 0.3
    
    return best_alpha


def lp_optimize(a0, time_limit, verbose=True, seed=42,
                k_min=5, k_max=80, use_scipy=True):
    """
    LP-based optimization with adaptive active set and line search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    k_active = k_min
    cvx_solver = get_solver() if not use_scipy else None
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        c_max = c.max()
        
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive active set
        k_active = min(k_max, k_min + stall // 15)
        top_idx = np.argsort(c)[::-1][:k_active]
        active_set = [int(k) for k in top_idx]
        
        # Get direction
        d = None
        if use_scipy:
            result = lp_direction_scipy(a, active_set)
            if result[0] is not None:
                d = result[0]
        if d is None and cvx_solver is not None:
            result = lp_direction_cvxpy(a, active_set, cvx_solver)
            if result[0] is not None:
                d = result[0]
        
        if d is None:
            # Fallback: gradient direction
            c_use = np.zeros(len(c))
            c_use[top_idx] = c[top_idx]
            c_max_use = c_use.max()
            beta = 50.0
            shifted = beta * (c_use - c_max_use)
            shifted = np.clip(shifted, -300, 300)
            exp_vals = np.exp(shifted)
            total = exp_vals.sum()
            if total < 1e-300:
                p = np.zeros_like(c_use)
                p[np.argmax(c)] = 1.0
            else:
                p = exp_vals / total
            fft_size = 1 << (2 * n - 1).bit_length()
            A = np.fft.rfft(a, fft_size)
            P = np.fft.rfft(p, fft_size)
            a_rev = a[::-1]
            Conv = np.fft.irfft(P * np.fft.rfft(a_rev, fft_size), fft_size)
            grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
            d = -grad
            d -= d.mean()
            d = np.maximum(d, -a)
        
        # Line search
        alpha = analytic_line_search(a, d, c, c_max, n)
        
        if alpha < 1e-10:
            # Try a small fixed step

            neg_mask = d < 0

                alpha_f = np.min(a[neg_mask] / (-d[neg_mask]))
                alpha = min(alpha, alpha_f * 0.5)
        
        # Apply update
        a_new = a + alpha * d
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        a = a_new
        iteration += 1
        
        if verbose and iteration % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP it=" + str(iteration) + " (" + str(round(elapsed, 1)) + "s): J=" +

        if stall > 800:
            stall = 0
            k_active = k_min
            noise_scale = 0.02 / (1.0 + 0.1 * (iteration // 800))
            noise = rng.normal(0, noise_scale, n) * best_a
            a_r = best_a + noise
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r
    
    return best_a, best_m


def nesterov_opt(a0, time_limit, beta_start=50.0, beta_end=3000.0,
                 beta_growth=1.006, eta_base=0.5, top_k_ratio=0.02,
                 restart_every=2000, verbose=True, seed=42):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        c = fast_autocorr(y)
        if top_k < len(c):
            idx = np.argsort(c)[::-1][:top_k]
            c_use = np.zeros(len(c))
            c_use[idx] = c[idx]
            c_max = c[idx].max()
        else:
            c_use = c
            c_max = c.max()
        shifted = beta * (c_use - c_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        total = exp_vals.sum()
        if total < 1e-300:
            p = np.zeros_like(c_use)
            p[np.argmax(c)] = 1.0
        else:
            p = exp_vals / total

        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(y, fft_size)
        P = np.fft.rfft(p, fft_size)
        a_rev = y[::-1]
        Conv = np.fft.irfft(P * np.fft.rfft(a_rev, fft_size), fft_size)
        grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]

        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.03 / (1.0 + 0.05 * (t // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + ", beta=" + str(round(beta, 1)))

    return best_a, best_max_b


def multiplicative_update(a0, time_limit, beta=100.0, n_top=20, verbose=True, seed=13):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    eta = 0.05

    for t in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        top_idx = np.argsort(c)[::-1][:n_top]
        c_use = np.zeros(2 * n - 1)
        c_use[top_idx] = c[top_idx]
        shifted = beta * (c_use - c_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        total = exp_vals.sum()
        if total < 1e-300:
            p = np.zeros(2 * n - 1)
            p[np.argmax(c)] = 1.0
        else:
            p = exp_vals / total

        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        P = np.fft.rfft(p, fft_size)
        a_rev = a[::-1]
        Conv = np.fft.irfft(P * np.fft.rfft(a_rev, fft_size), fft_size)
        grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]

        log_step = -eta * grad
        log_step = np.clip(log_step, -1.5, 1.5)
        a_new = a * np.exp(log_step)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        b_new = fast_autocorr(a_new)
        max_new = b_new.max()
        if max_new < best_m:
            a = a_new
        else:
            for eta_mult in [0.5, 0.1, 0.01]:
                a_try = a * np.exp(eta_mult * log_step)
                a_try = np.maximum(a_try, 1e-15)
                a_try = a_try / a_try.sum()
                if fast_autocorr(a_try).max() < best_m:
                    a = a_try
                    break

        if t % 200 == 0:
            beta = min(beta * 1.04, 5000)
            eta = max(eta * 0.97, 0.005)

        if stall > 3000:
            stall = 0
            noise = rng.normal(0, 0.02, n) * best_a
            a_r = best_a + noise
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r
            beta = max(beta * 0.5, 20)
            eta = 0.05

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    MultUpd t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + ", beta=" + str(round(beta, 1)))

    return best_a, best_m


def peak_surgery(a0, time_limit, verbose=True, seed=77):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 5000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_top = min(30, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]
        improved = False

        for k in top_idx[:15]:
            if improved:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            if products.max() < 1e-20:
                continue
            src_i = int(idx[np.argmax(products)])
            partner = k_int - src_i
            if partner < 0 or partner >= n:
                continue
            if a[src_i] >= a[partner]:
                reduce_i = src_i
            else:
                reduce_i = partner
            if a[reduce_i] < 1e-14:
                continue
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[reduce_i] * frac
                if amt < 1e-15:
                    continue
                for _ in range(5):
                    dst = int(rng.integers(n))
                    if dst == reduce_i:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_i] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and no_improve > 500:
            for _ in range(10):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break

        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSurgery (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve))

    return best_a, best_m


def pairwise_cd(a0, time_limit, verbose=True, seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        c = fast_autocorr(a)
        threshold = np.percentile(c, 95)
        peak_positions = np.where(c > threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.argsort(c)[::-1][:20]
        improved = False

        for k in peak_positions[:15]:
            if improved and (time.monotonic() - t0 > time_limit):
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            top_pairs = np.argsort(products)[::-1][:3]
            for pi in top_pairs:
                if improved:
                    break
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n or partner == src_i:
                    continue
                if a[src_i] < 1e-14 and a[partner] < 1e-14:
                    continue
                larger_i = src_i if a[src_i] >= a[partner] else partner
                if a[larger_i] < 1e-14:
                    continue
                for frac in [0.001, 0.003, 0.005]:
                    if improved:
                        break
                    amt = a[larger_i] * frac
                    if amt < 1e-18:
                        continue
                    for _ in range(3):
                        dst = int(rng.integers(n))
                        if dst == larger_i:
                            continue
                        a_trial = a.copy()
                        a_trial[larger_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break

        if not improved:
            stall += 1
            if stall % 100 == 0:
                anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
                eps = 0.01 / (1.0 + stall / 100.0)
                a_try = a + eps * anti * a
                a_try = np.maximum(a_try, 0.0)
                s_t = a_try.sum()
                if s_t > 1e-15:
                    a_try = a_try / s_t
                    if fast_autocorr(a_try).max() < best_m:
                        a = a_try
                        best_m = fast_autocorr(a).max()
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PairCD (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main optimization combining LP-based direction finding (primary),
    Nesterov gradient, multiplicative updates, peak surgery, and pairwise CD.
    Uses scipy HiGHS for fast LP solves with analytic line search.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        x = np.linspace(0, 1, n_base)
        a_init = (1.0 - x) ** 1.5 + 0.1
        best_seq = norm_sum1(a_init)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(n_base) + ", score=" + repr(best_score))

    n0 = len(best_seq)
    print("Starting: n=" + str(n0) + ", score=" + repr(best_score))

    # Phase 1: LP-based optimization (primary)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: LP Optimization (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.22, 380.0)
    if t1 > 60:
        a_out, _ = lp_optimize(norm_sum1(best_seq), t1, verbose=True, seed=1)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 2: Nesterov gradient
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 60:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t2,
                                beta_start=100.0, beta_end=5000.0,
                                beta_growth=1.008, eta_base=0.4,
                                top_k_ratio=0.01, restart_every=1500,
                                verbose=True, seed=2)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 3: LP round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP Round 2 (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.18, 300.0)
    if t3 > 60:
        a_out, _ = lp_optimize(norm_sum1(best_seq), t3, verbose=True, seed=3)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp2", t_global)

    # Phase 4: Multiplicative update
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Multiplicative (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.10, 180.0)
    if t4 > 40:
        a_out, _ = multiplicative_update(norm_sum1(best_seq), t4,
                                          beta=100.0, n_top=20,
                                          verbose=True, seed=4)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "mult1", t_global)

    # Phase 5: Peak surgery
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Peak Surgery (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.08, 150.0)
    if t5 > 30:
        a_out, _ = peak_surgery(norm_sum1(best_seq), t5, verbose=True, seed=5)
