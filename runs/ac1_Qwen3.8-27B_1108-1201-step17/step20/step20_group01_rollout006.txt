    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def lse_grad(a, conv, tau, n_top):
    """
    Gradient of log-sum-exp(conv, tau) w.r.t. a using top-K sparsification.
    grad[i] = sum_j a[j] * w[i+j] where w = softmax(conv/tau) (top-K only)
    Computed via: grad = (a_rev * w)[n-1 : 2n-1]
    """
    n = len(a)
    k_sel = min(n_top, len(conv))
    top_idx = np.argpartition(conv, -k_sel)[-k_sel:]
    top_vals = conv[top_idx]
    shifted = (top_vals - top_vals.max()) / tau
    w_top = np.exp(shifted)
    w_sum = w_top.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w_top /= w_sum
    w_full = np.zeros(2 * n - 1)
    w_full[top_idx] = w_top
    a_rev = a[::-1]
    conv_awn = fast_conv_pair(a_rev, w_full)
    return conv_awn[n - 1: 2 * n - 1]

def mirror_descent_lse(a0, time_limit, tau_init=8.0, tau_min=0.08, n_top=300,
                       lr_base=0.8, verbose=True):
    """
    Nesterov-accelerated mirror descent (exponentiated gradient) on
    annealed log-sum-exp surrogate of max(conv(a,a)).
    """
    n = len(a0)
    a = normalize(a0).copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    beta = 0.75
    stall = 0
    stall_limit = 6000

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > stall_limit:
            break

        # Anneal tau
        prog = min(1.0, t / 8000.0)
        tau = tau_init * (tau_min / tau_init) ** (prog ** 0.7)

        # Nesterov extrapolation
        if t > 2:
            z = a + beta * (a - a_prev)
            z = np.maximum(z, 1e-30)
            zs = z.sum()
            if zs > 1e-15:
                z = z / zs
            else:
                z = a.copy()
        else:
            z = a.copy()

        conv_z = fast_conv(z)
        grad = lse_grad(z, conv_z, tau, n_top)

        # Adaptive learning rate
        g_norm = np.abs(grad).max()
        if g_norm > 1e-15:
            lr = lr_base / (1.0 + 0.003 * t) / g_norm
        else:
            lr = 0.01 / sqrt_t if (sqrt_t := math.sqrt(t)) > 0 else 0.01
        lr = min(lr, 0.5)

        # Mirror descent update
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-25)
        ss = a_new.sum()
        if ss < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / ss

        a_prev = a.copy()
        a = a_new

        conv_a = fast_conv(a)
        cur_m = conv_a.max()
        if cur_m < best_m - 1e-14:
            best_m = cur_m
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Periodic restart with noise
        if stall > 4000 and stall % 2000 == 0:
            noise_scale = 0.03 * stall / stall_limit
            noise = noise_scale * np.random.default_rng(t).standard_normal(n)
            a_r = best_a + noise
            a_r = np.maximum(a_r, 1e-10)
            a = a_r / a_r.sum()
            a_prev = a.copy()
            stall = 0

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print("    MD t={:6d} ({:.1f}s) tau={:.4f} J={:.7f} stall={}".format(
                t, elapsed, tau, 2 * n * cur_m, stall))

    return best_a, 2.0 * n * best_m


def nesterov_subgrad(a0, time_limit, n_top=200, lr0=0.3, verbose=True):
    """
    Nesterov-style subgradient method on the exact max (not smoothed).
    Gradient is the convolution of a with a spike at the argmax lag.
    """
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 8000:
            break

        conv_a = fast_conv(a)
        M = conv_a.max()
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Subgradient: average over all argmax lags (or top-K for smoothness)
        k_sel = min(n_top, len(conv_a))
        top_k = np.argpartition(conv_a, -k_sel)[-k_sel:]
        # Weight by how close to max
        margins = M - conv_a[top_k]
        w = np.exp(-margins * 20.0)
        w /= w.sum()

        # Build spike weights
        spike = np.zeros(2 * n - 1)
        for i, ki in enumerate(top_k):
            spike[int(ki)] = w[i]

        # grad[i] = sum_j a[j] * spike[i+j] = (a_rev * spike)[n-1:2n-1]
        a_rev = a[::-1]
        ga = fast_conv_pair(a_rev, spike)
        grad = ga[n - 1: 2 * n - 1]

        lr = lr0 / (1.0 + 0.002 * t)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-25)
        a_new = a_new / a_new.sum()
        a = a_new

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print("    SG t={:6d} ({:.1f}s) J={:.7f} stall={}".format(
                t, elapsed, 2 * n * a.max() * 0 + 2 * n * fast_conv(a).max(), stall))

    return best_a, 2.0 * n * best_m


def lp_relaxation_dir(a, active_lags, n):
    """
    Solve a small LP to find the Frank-Wolfe direction:
    max sum(g) s.t. sum(g)=1, g>=0, conv(a,g)[k] <= 0 for k in active_lags
    Uses scipy linprog for speed.
    """
    from scipy.optimize import linprog
    K = len(active_lags)
    if K == 0:
        return None

    # Build constraint matrix: rows = active_lags, cols = n
    A_ub = np.zeros((K, n))
    for jj, ki in enumerate(active_lags):
        ki = int(ki)
        ws = max(0, ki - n + 1)
        we = min(n - 1, ki)
        if ws <= we:
            idx = np.arange(ws, we + 1)
            A_ub[jj, idx] = a[ki - idx]

    c = -np.ones(n)  # maximize sum(g) = minimize -sum(g)
    A_eq = np.ones((1, n))
    b_eq = [1.0]
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(K), A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success is not True and res.status != 0:
        return None
    g = np.maximum(res.x, 0.0)
    gs = g.sum()
    if gs < 1e-12:
        return None
    return g / gs


def lp_fw_step(a, time_per_iter, solver_type='highs', K_init=150, K_max=2000):
    """One LP-FW step: find direction, line search, update."""
    n = len(a)
    conv_a = fast_conv(a)
    M = conv_a.max()
    top_lags = np.argsort(conv_a)[::-1][:K_init]

    from scipy.optimize import linprog
    A_ub = np.zeros((len(top_lags), n))
    for jj, ki in enumerate(top_lags):
        ki = int(ki)
        ws = max(0, ki - n + 1)
        we = min(n - 1, ki)
        if ws <= we:
            idx = np.arange(ws, we + 1)
            A_ub[jj, idx] = a[ki - idx]

    c = -np.ones(n)
    A_eq = np.ones((1, n))
    b_eq = [1.0]
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(len(top_lags)),
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res is None or (res.success is not True and res.status != 0):
        return a, conv_a.max()

    g = np.maximum(res.x, 0.0)
    gs = g.sum()
    if gs < 1e-12:
        return a, conv_a.max()
    g = g / gs

    # Line search
    best_alpha = 0.5
    best_m = conv_a.max()
    for alpha in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
        trial = (1 - alpha) * a + alpha * g
        trial = np.maximum(trial, 0.0)
        ts = trial.sum()
        if ts < 1e-15:
            continue
        Mt = fast_conv(trial / ts).max()
        if Mt < best_m - 1e-15:
            best_m = Mt
            best_alpha = alpha

    a_new = (1 - best_alpha) * a + best_alpha * g
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss > 1e-15:
        a_new = a_new / ss
    else:
        a_new = a.copy()
    return a_new, fast_conv(a_new).max()


def lp_fw_loop(a0, time_limit, K_max=1500, verbose=True):
    """LP-FW loop with growing active set."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    active = set(int(k) for k in np.argsort(fast_conv(a))[::-1][:200])
    stall = 0
    from scipy.optimize import linprog

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 5000:
            break

        conv_a = fast_conv(a)
        M = conv_a.max()
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Update active set
        if t % 50 == 0 or t == 1:
            target = min(K_max, 200 + t * 3)
            top = np.argsort(conv_a)[::-1][:target]
            for k in top:
                active.add(int(k))
            if len(active) > K_max:
                clist = sorted(active)
                diffs = np.array([M - conv_a[k] for k in clist])
                keep = np.argsort(diffs)[:K_max]
                active = set(int(clist[i]) for i in keep)

        active.add(int(np.argmax(conv_a)))
        clist = sorted(active)
        K = min(len(clist), 500)  # Cap for speed
        clist_use = clist[:K]

        A_ub = np.zeros((K, n))
        for jj, ki in enumerate(clist_use):
            ki = int(ki)
            ws = max(0, ki - n + 1)
            we = min(n - 1, ki)
            if ws <= we:
                idx = np.arange(ws, we + 1)
                A_ub[jj, idx] = a[ki - idx]

        c_obj = -np.ones(n)
        A_eq = np.ones((1, n))
        b_eq = [1.0]
        bnd = [(0, None)] * n
        b_ub = np.zeros(K)

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bnd, method='highs', options={'presolve': True})
        if res is None or (res.success is not True and res.status != 0):
            active = set(int(k) for k in np.argsort(conv_a)[::-1][:200])
            continue

        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs < 1e-12:
            continue
        g = g / gs

        # Check violations
        conv_ag = fast_conv_pair(a, g)
        viol = np.where(conv_ag > M + 1e-10)[0]
        if len(viol) > 0:
            viol_amt = conv_ag[viol] - M
            top_v = viol[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                active.add(int(k))

        # Line search
        best_alpha = 0.5
        best_trial = M
        for alpha in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
            trial = (1 - alpha) * a + alpha * g
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            Mt = fast_conv(trial / ts).max()
            if Mt < best_trial - 1e-15:
                best_trial = Mt
                best_alpha = alpha

        a = (1 - best_alpha) * a + best_alpha * g
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss > 1e-15:
            a = a / ss

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP t={:5d} ({:.1f}s) J={:.7f} |S|={}".format(
                t, elapsed, 2 * n * fast_conv(a).max(), len(active)))

    return best_a, 2.0 * n * best_m


def coord_polish(a0, time_limit, n_targets=10, seed=42, verbose=True):
    """Coordinate-wise mass redistribution."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    fracs = [0.002, 0.005, 0.01, 0.02, 0.05]

    while time.monotonic() - t0 < time_limit and stall < 5000:
        conv_a = fast_conv(a)
        top_k = np.argsort(conv_a)[::-1][:n_targets]
        improved = False

        for k in top_k:
            ki = int(k)
            ws = max(0, ki - n + 1)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-25:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-18:
                continue

            # Find quiet positions (low total pressure)
            pr = np.zeros(n)
            for kk in top_k[:6]:
                ki2 = int(kk)
                ws2 = max(0, ki2 - n + 1)
                we2 = min(n - 1, ki2)
                if ws2 <= we2:
                    idx2 = np.arange(ws2, we2 + 1)
                    pr[idx2] += a[idx2] * a[ki2 - idx2]
            pr[src] = 1e10
            quiet = np.argsort(pr)[:40]

            for dst in quiet:
                dst = int(dst)
                if dst == src:
                    continue
                found = False
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-20:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_conv(a_t).max()
                    if Mt < best_m - 1e-16:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break

            if not improved and stall > 200:
                break

        if not improved:
            stall += 1
            if stall > 500:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_conv(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 1000 == 0 and stall > 0:
            print("    Peel ({:.1f}s) J={:.7f} stall={}".format(
                time.monotonic() - t0, 2 * n * best_m, stall))

    return best_a, 2.0 * n * best_m


def interpolate_seq(f, new_n):
    """Resample sequence to new length via linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def make_init(n, shape='sin2', rng_seed=0):
    """Create an initial sequence of given shape."""
    x = np.linspace(0, 1, n)
    if shape == 'sin2':
        a = np.sin(np.pi * x) ** 2
    elif shape == 'gauss':
        a = np.exp(-8 * (x - 0.5) ** 2)
    elif shape == 'cos':
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif shape == 'tri':
        a = 1 - np.abs(2 * x - 1)
    elif shape == 'flat':
        a = np.ones(n)
    elif shape == 'asym':
        a = np.exp(-4 * (x - 0.4) ** 2)
    elif shape == 'bump':
        a = (x + 0.3) * (1 - x + 0.3) ** 2
        a = np.maximum(a, 0)
    elif shape == 'rand':
        rng = np.random.default_rng(rng_seed)
        a = np.abs(rng.standard_normal(n))
    else:
        a = np.sin(np.pi * x) ** 2
    return normalize(a)


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent: n={} score={}".format(len(p), repr(best_score)))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_init = 10000
        best_seq = make_init(n_init, 'sin2')
        best_score = eval_safe(best_seq)
        print("Default init n={} score={}".format(n_init, repr(best_score)))

    n0 = len(best_seq)

    # === Phase 1: Multi-start exploration (200s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.12, 200.0)
    if t1 > 20:
        print("\n=== P1: Multi-start (n={}) ===".format(n0))
        t_sub = t1 / 6.0
        inits = [
            ('sin2', 0), ('gauss', 1), ('cos', 2),
            ('asym', 3), ('bump', 4), ('rand', 5)
        ]
        for shape, seed in inits:
            if time.monotonic() - t_global > time_budget:
                break
            a_init = make_init(n0, shape, seed)
            a_out, v_out = mirror_descent_lse(a_init, t_sub, tau_init=10.0,
                                              tau_min=0.1, n_top=200, lr_base=0.5,
                                              verbose=False)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  P1 {}({}) -> {}".format(shape, seed, repr(best_score)))

    # === Phase 2: Main Nesterov mirror descent (500s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P2: MD-LSE main (n={}) ===".format(n_cur))
    t2 = min(remaining * 0.28, 500.0)
    if t2 > 30:
        a_out, v_out = mirror_descent_lse(normalize(best_seq), t2,
                                          tau_init=5.0, tau_min=0.05,
                                          n_top=400, lr_base=1.0, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P2 -> {}".format(repr(best_score)))

    # === Phase 3: Subgradient refinement (200s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P3: Subgrad (n={}) ===".format(n_cur))
    t3 = min(remaining * 0.13, 200.0)
    if t3 > 20:
        a_out, v_out = nesterov_subgrad(normalize(best_seq), t3,
                                        n_top=100, lr0=0.2, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P3 -> {}".format(repr(best_score)))

    # === Phase 4: LP-FW refinement (300s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P4: LP-FW (n={}) ===".format(n_cur))
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 30:
        a_out, v_out = lp_fw_loop(normalize(best_seq), t4, K_max=1200, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P4 -> {}".format(repr(best_score)))

    # === Phase 5: Try larger n (250s) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = min(len(best_seq) + 3000, 18000)
    if n_long != len(best_seq) and remaining > 100:
        print("\n=== P5: Long n={} ===".format(n_long))
        t5 = min(remaining * 0.14, 250.0)
        if t5 > 30:
            a_long = interpolate_seq(best_seq, n_long)
            a_out, v_out = mirror_descent_lse(a_long, t5 * 0.5,
                                              tau_init=5.0, tau_min=0.05,
                                              n_top=500, lr_base=0.6, verbose=False)
            a_out2, v_out2 = coord_polish(a_out, t5 * 0.5, n_targets=10,
                                          seed=77, verbose=False)
            sc1 = eval_safe(a_out)
            sc2 = eval_safe(a_out2)
            best_sc5 = min(sc1, sc2)
            if np.isfinite(best_sc5) and best_sc5 < best_score:
