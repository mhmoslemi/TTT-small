def softmax_values(vals, beta):
    vals = np.asarray(vals, dtype=np.float64)
    if vals.size == 0: return np.array([], dtype=np.float64)
    if beta <= 0:
        return np.full(vals.shape, 1.0/vals.size, dtype=np.float64)
    m = float(np.max(vals))
    if not np.isfinite(m):
        return np.full(vals.shape, 1.0/vals.size)

    s = float(np.sum(e))

    return e / s

```python
"""
Search for nonnegative coefficient sequences minimizing

    2 * n * max(convolve(a, a)) / sum(a)^2

using FFT-accelerated mirror-descent directions, approximate line searches,
aggressive coordinate peeling, and length interpolation.

The method:
1. Start from the supplied parent construction if available.
2. Use softmax-weighted top-lag gradients to propose nonnegative directions.
3. For each direction, perform a cheap approximate line search using
   the top convolution lags, then verify the best few steps by exact FFT
   objective evaluation.
4. Occasionally solve a smaller minimax game to produce LP-like directions.
5. Polish with coordinate moves and occasional annealed random swaps.
6. Try interpolated longer sequences when time remains.
"""

import numpy as np
import time

_ALPHAS = np.array(
    [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.18,
     0.25, 0.35, 0.5, 0.7, 0.9, 1.0],
    dtype=np.float64,
)


def normalize_positive(x):
    a = np.asarray(x, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_autocorr_fixed(a, fs):
    aa = np.fft.rfft(a, fs)
    out = np.fft.irfft(aa * aa, fs)
    out = out[: 2 * len(a) - 1]
    np.maximum(out, 0.0, out=out)
    return out


def fft_convolve_fixed(a, b, fs):
    out = np.fft.irfft(np.fft.rfft(a, fs) * np.fft.rfft(b, fs), fs)
    out = out[: len(a) + len(b) - 1]
    np.maximum(out, 0.0, out=out)
    return out


def score_with_fft(a, fs):
    if a.size <= 0:
        return float("inf")
    ac = fast_autocorr_fixed(a, fs)
    if ac.size == 0:
        return float("inf")
    m = float(np.max(ac))
    if not np.isfinite(m):
        return float("inf")
    return 2.0 * float(len(a)) * m


def softmax_values(vals, beta):
    vals = np.asarray(vals, dtype=np.float64)
    if vals.size == 0:
        return np.array([], dtype=np.float64)
    if beta <= 0.0:
        return np.full(vals.shape, 1.0 / float(vals.size), dtype=np.float64)
    m = float(np.max(vals))
    if not np.isfinite(m):
        return np.full(vals.shape, 1.0 / float(vals.size), dtype=np.float64)
    e = np.exp(beta * (vals - m))
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(vals.shape, 1.0 / float(vals.size), dtype=np.float64)
    return e / s


def cross_corr_lambda_a(lam, a, a_fft_corr, fs_corr):
    """
    Given lam of length 2n-1 and a of length n, compute
        t_i = sum_{j=0}^{n-1} a_j * lam_{i+j},  i=0..n-1
    via FFT.
    """
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = len(lam)
    r = lam[::-1].copy()
    out = np.fft.irfft(a_fft_corr * np.fft.rfft(r, fs_corr), fs_corr)
    idx = np.arange(L - 1, L - n - 1, -1, dtype=np.intp)
    res = out[idx]
    res = np.maximum(res, 0.0)
    return res


def interpolate_sequence(x, new_n):
    x = normalize_positive(x)
    old_n = len(x)
    if new_n <= 1:
        return np.ones(int(max(1, new_n)), dtype=np.float64)
    new_n = int(new_n)
    if old_n == new_n:
        return x.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, x)
    y = np.maximum(y, 0.0)
    s = float(np.sum(y))
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64)
    return y / s


def blend_candidates(a, g, alpha):
    alpha = float(max(0.0, min(1.0, alpha)))
    c = (1.0 - alpha) * a + alpha * g
    np.maximum(c, 0.0, out=c)
    s = float(np.sum(c))
    if s <= 1e-15:
        return np.ones_like(c, dtype=np.float64)
    return c / s


def approximate_line_search(a, g, active_idx, ac, fs, cur_score, max_eval):
    """
    For direction g, cheaply choose promising alpha values using active
    lags plus top lags of cross-convolution and self-convolution of g.
    Then full-evaluate a small set of alpha values.
    """
    n = len(a)
    L = len(ac)
    cross = fft_convolve_fixed(a, g, fs)
    gg = fast_autocorr_fixed(g, fs)

    idx = np.asarray(active_idx, dtype=np.intp).ravel()
    if idx.size == 0:
        idx = np.arange(L, dtype=np.intp)

    if idx.size < max_eval and L > 200:
        extra = max(300, min(1600, idx.size // 3))
        top_cross = np.argsort(cross)[::-1][:extra]
        top_gg = np.argsort(gg)[::-1][:extra]
        idx = np.unique(np.concatenate((idx, top_cross, top_gg)))

    if idx.size > max_eval:
        sc = ac[idx] + cross[idx] + gg[idx]
        idx = idx[np.argsort(sc)[::-1][:max_eval]]

    if idx.size == 0:
        return a.copy(), cur_score

    alpha = _ALPHAS
    a1 = alpha[:, None]
    om1 = 1.0 - a1

    AE = ac[idx][None, :]
    CE = cross[idx][None, :]
    GE = gg[idx][None, :]

    vals = om1 * om1 * AE + 2.0 * om1 * a1 * CE + a1 * a1 * GE
    maxes = np.max(vals, axis=1)
    maxes = np.where(np.isfinite(maxes), maxes, np.inf)
    order = np.argsort(maxes)

    test_alphas = []
    top_count = min(5, int(order.size))
    for oi in range(top_count):
        test_alphas.append(float(alpha[order[oi]]))
    for av in [0.25, 0.5, 1.0]:
        test_alphas.append(av)

    unique_alphas = []
    for av in test_alphas:
        if av <= 0.0 or av > 1.0:
            continue
        dup = False
        for uv in unique_alphas:
            if abs(av - uv) < 1e-6:
                dup = True
                break
        if not dup:
            unique_alphas.append(av)

    best_cand = a.copy()
    best_val = cur_score
    for av in unique_alphas:
        cand = blend_candidates(a, g, av)
        val = score_with_fft(cand, fs)
        if np.isfinite(val) and val < best_val:
            best_val = val
            best_cand = cand
    return best_cand, best_val


def game_direction(a, active_idx, ac, a_fft_corr, fs_corr, fs_conv, uniform,
                   max_iter=15, start_time=None, hard_deadline=None):
    """
    Approximate the LP minimax direction:
        minimize_g max_k conv(a, g)_k
    using multiplicative-weights / fictitious-play style iterations.
    """
    n = len(a)
    L = 2 * n - 1
    g = a.copy()
    if start_time is None:
        start_time = time.monotonic()

    for it in range(max_iter):
        if hard_deadline is not None and time.monotonic() >= hard_deadline:
            return g

        s = fft_convolve_fixed(a, g, fs_conv)
        score_all = s + ac

        K = max(1, int(active_idx.size))
        top_new = np.argsort(score_all)[::-1][:K]
        active = np.unique(np.concatenate((np.asarray(active_idx, dtype=np.intp), top_new)))
        if active.size > 2 * K:
            sc = score_all[active]
            active = active[np.argsort(sc)[::-1][:K]]

        active_idx = active
        K = max(1, len(active_idx))
        s_act = s[active_idx]
        M_s = float(np.max(s_act))
        if M_s <= 1e-30 or not np.isfinite(M_s):
            M_s = 1e-30

        prog = float(it) / float(max(1, max_iter))
        beta = (1.0 + 6.0 * prog) / M_s
        lam_act = softmax_values(s_act, beta)
        lam = np.zeros(L, dtype=np.float64)
        lam[active_idx] = lam_act

        t = cross_corr_lambda_a(lam, a, a_fft_corr, fs_corr)
        tmax = float(np.max(t))
        if not np.isfinite(tmax) or tmax <= 1e-30:
            break
        t = t / tmax
        tmean = float(np.mean(t))

        eta = 1.0 + 10.0 * prog
        log_g = -eta * (t - tmean)
        log_g -= float(np.max(log_g))
        g = np.exp(log_g)
        np.maximum(g, 1e-30, out=g)
        gs = float(np.sum(g))
        if gs <= 1e-30:
            g = uniform.copy()
        else:
            g = g / gs

        g = 0.90 * g + 0.10 * uniform
        g = np.maximum(g, 0.0)
        gs = float(np.sum(g))
        if gs > 0.0:
            g = g / gs

    return g


def local_search(a0, time_limit, verbose=False, seed=123, K_initial=None, K_max=None):
    """
    Main fast local search loop.
    """
    start = time.monotonic()
    a = normalize_positive(a0)
    n = len(a)
    if n < 2:
        return a, float(score_with_fft(a, 1 << 1))

    L = 2 * n - 1
    fs_conv = next_pow2(L)
    fs_corr = next_pow2(3 * n - 2)

    best = a.copy()
    best_score = score_with_fft(a, fs_conv)
    if not np.isfinite(best_score):
        return a, float("inf")

    if time_limit <= 0.0:
        return best, best_score

    if K_initial is None:
        K_initial = min(L, max(2000, int(0.20 * n)))
    if K_max is None:
        K_max = min(L, max(8000, int(0.60 * n)))
    K_step = max(5, n // 1000)
    max_eval = min(15000, max(4000, int(0.50 * n)))

    uniform = np.ones(n, dtype=np.float64) / float(n)
    rng = np.random.default_rng(seed)

    beta_list = [0.0, 2.0, 6.0, 16.0]
    eta_list = [2.0, 6.0, 16.0]

    stall = 0
    round_no = 0

    while time.monotonic() - start < time_limit:
        round_no += 1

        ac = fast_autocorr_fixed(a, fs_conv)
        cur_score = 2.0 * n * float(np.max(ac))

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            stall = 0

        if not np.isfinite(cur_score) or cur_score > best_score * 1.002 + 1e-9:
            a = best.copy()
            stall = 0
            continue

        K = min(K_max, K_initial + round_no * K_step)
        active_idx = np.argsort(ac)[::-1][:K]

        a_fft_corr = np.fft.rfft(a, fs_corr)

        best_cand = None
        best_cand_score = cur_score

        # Direction toward uniform, as a stabilizer / exploratory direction.
        cand, val = approximate_line_search(
            a, uniform, active_idx, ac, fs_conv, cur_score, max_eval
        )
        if np.isfinite(val) and val < best_cand_score:
            best_cand_score = val
            best_cand = cand

        active_vals = ac[active_idx]
        Mval = float(np.max(active_vals))
        if not np.isfinite(Mval) or Mval <= 1e-30:
            Mval = 1e-30

        for beta in beta_list:
            if time.monotonic() - start > time_limit:
                break

            if beta <= 0.0:
                lam_act = np.full(int(active_idx.size), 1.0 / float(max(1, active_idx.size)), dtype=np.float64)
            else:
                lam_act = softmax_values(active_vals, beta / Mval)

            lam = np.zeros(L, dtype=np.float64)
            lam[active_idx] = lam_act

            t = cross_corr_lambda_a(lam, a, a_fft_corr, fs_corr)
            tmax = float(np.max(t))
            if not np.isfinite(tmax) or tmax <= 1e-30:
                continue
            t_scaled = t / tmax
            tmean = float(np.mean(t_scaled))

            for eta in eta_list:
                if time.monotonic() - start > time_limit:
                    break

                log_g = -float(eta) * (t_scaled - tmean)
                log_g -= float(np.max(log_g))
                g = np.exp(log_g)
                np.maximum(g, 1e-30, out=g)
                gs = float(np.sum(g))
                if gs <= 1e-30:
                    continue
                g = g / gs

                g = 0.93 * g + 0.07 * uniform
                g = np.maximum(g, 0.0)
                gs = float(np.sum(g))
                if gs > 0.0:
                    g = g / gs

                cand, val = approximate_line_search(
                    a, g, active_idx, ac, fs_conv, cur_score, max_eval
                )
                if np.isfinite(val) and val < best_cand_score:
                    best_cand_score = val
                    best_cand = cand

        if round_no % 8 == 0 and time.monotonic() - start < time_limit - 5.0:
            g_game = game_direction(
                a,
                active_idx,
                ac,
                a_fft_corr,
                fs_corr,
                fs_conv,
                uniform,
                max_iter=15,
                start_time=start,
                hard_deadline=start + time_limit - 2.0,
            )
            if g_game is not None:
                cand, val = approximate_line_search(
                    a, g_game, active_idx, ac, fs_conv, cur_score, max_eval
                )
                if np.isfinite(val) and val < best_cand_score:
                    best_cand_score = val
                    best_cand = cand

        if best_cand is not None and best_cand_score < cur_score - 1e-12:
            a = best_cand
            if best_cand_score < best_score:
                best_score = best_cand_score
                best = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 25:
                scale = 0.04 + 0.01 * min(5, stall // 10)
                mul = np.exp(scale * rng.standard_normal(n))
                a = normalize_positive(a * mul)
                stall = 0
            elif stall > 10 and round_no % 5 == 0:
                a = normalize_positive(0.90 * a + 0.10 * uniform)

        if verbose and round_no % 20 == 0:
            print(
                "  local n={} round={} t={:.1f}s best={:.7f} cur={:.7f} stall={}".format(
                    n, round_no, time.monotonic() - start,
                    best_score, cur_score, stall
                )
            )

    return best, best_score


def coordinate_peel(a0, time_limit, seed=7, verbose=False, n_targets=24, max_pairs=600):
    """
    Coordinate hill climbing / annealing on top autocorrelation lags.
    """
    start = time.monotonic()
    a = normalize_positive(a0)
    n = len(a)
    if n < 2:
        return a, 2.0

    fs = next_pow2(2 * n - 1)
    best = a.copy()
    best_score = score_with_fft(a, fs)
    if not np.isfinite(best_score):
        return a, float("inf")

    if time_limit <= 0.0:
        return best, best_score

    rng = np.random.default_rng(seed)
    fracs = np.array([0.02, 0.05, 0.10, 0.18, 0.30, 0.50], dtype=np.float64)

    stall = 0

    while time.monotonic() - start < time_limit:
        ac = fast_autocorr_fixed(a, fs)
        cur_score = 2.0 * n * float(np.max(ac))

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            stall = 0

        top_k = np.argsort(ac)[::-1][:n_targets]

        pressure = np.zeros(n, dtype=np.float64)
        for kk in top_k:
            ki = int(kk)
            i0 = max(0, ki - n + 1)
            i1 = min(n - 1, ki)
            if i0 <= i1:
                idx = np.arange(i0, i1 + 1, dtype=np.intp)
                pressure[idx] += a[ki - idx]

        src_score = pressure * (a + 1e-30)
        src = np.argsort(src_score)[::-1][:50]
        dst = np.argsort(pressure)[:100]

        improved = False
        attempted = 0

        for si in src:
            if time.monotonic() - start > time_limit:
                break
            si = int(si)
            if a[si] <= 1e-15:
                continue

            for di in dst:
                if attempted >= max_pairs:
                    break
                attempted += 1
                di = int(di)
                if di == si:
                    continue

                for frac in fracs:
                    amt = a[si] * float(frac)
                    if amt <= 1e-15:
                        break

                    cand = a.copy()
                    cand[si] -= amt
                    cand[di] += amt
                    cand = np.maximum(cand, 0.0)
                    s = float(np.sum(cand))
                    if s <= 1e-15:
                        continue
                    cand = cand / s

                    val = score_with_fft(cand, fs)
                    if np.isfinite(val):
                        if val < best_score:
                            best_score = val
                            best = cand.copy()
                        if val < cur_score - 1e-13:
                            a = cand
                            cur_score = val
                            improved = True
                            stall = 0
                            break

                if improved:
                    break

            if improved:
                break

        if improved:
            continue

        stall += 1

        accepted = False
        for _ in range(10):
            if time.monotonic() - start > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            frac = float(rng.uniform(0.01, 0.10))
            amt = a[i] * frac
            if amt <= 1e-15:
                continue

            cand = a.copy()
            cand[i] -= amt
            cand[j] += amt
            cand = np.maximum(cand, 0.0)
            s = float(np.sum(cand))
            if s <= 1e-15:
                continue
            cand = cand / s

            val = score_with_fft(cand, fs)
            if not np.isfinite(val):
                continue

            if val < best_score:
                best_score = val
                best = cand.copy()

            if val < cur_score - 1e-13 or (val < cur_score + 1e-7 and rng.random() < 0.10):
                a = cand
                cur_score = val
                accepted = True
                stall = max(0, stall - 1)
                break

        if not accepted and stall > 1000:
            mul = np.exp(0.05 * rng.standard_normal(n))
            a = normalize_positive(a * mul)
            stall = 0

        if verbose and stall > 0 and stall % 200 == 0:
            print(
                "  peel n={} t={:.1f}s best={:.7f} cur={:.7f} stall={}".format(
                    n, time.monotonic() - start, best_score, cur_score, stall
                )
            )

    return best, best_score


def propose_candidate(budget_s=1800.0):
    t0 = time.monotonic()
    total_budget = float(budget_s)
    hard_deadline = t0 + max(60.0, min(total_budget, 1800.0) - 45.0)

    best = None
    best_score = float("inf")

    # Load parent construction if provided.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1:
            parent = parent.ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        parent = normalize_positive(parent)
        if parent.size >= 2:
            fs = next_pow2(2 * parent.size - 1)
            sc = score_with_fft(parent, fs)
            if np.isfinite(sc):
                best = parent
                best_score = sc
                print("Parent: n={} fft_score={:.7f}".format(parent.size, sc))
    except Exception as exc:
        print("Parent unavailable: {}".format(exc))

    if best is None:
        n_default = 12000
        x = np.linspace(0.0, 1.0, n_default)
        base = np.sin(np.pi * x) ** 2
        base = base + 0.10 * (1.0 + np.cos(6.0 * np.pi * x))
        base = np.maximum(base, 1e-6)
        best = normalize_positive(base)
        fs = next_pow2(2 * best.size - 1)
        best_score = score_with_fft(best, fs)
        print("Default init: n={} fft_score={:.7f}".format(best.size, best_score))

    def remaining():
        return max(0.0, hard_deadline - time.monotonic())

    # Phase 1: main local search on the best current sequence.
    t1 = min(remaining() * 0.52, 850.0)
    if t1 > 45.0:
        print("\n=== Phase 1: main local search (n={:.0f}, {:.0f}s) ===".format(len(best), t1))
        a1, s1 = local_search(best, t1, verbose=True, seed=101)
        if np.isfinite(s1) and s1 < best_score:
            best = a1
            best_score = s1
            print("Phase 1 improved: {:.7f}".format(best_score))
        else:
            print("Phase 1 no improvement; best remains {:.7f}".format(best_score))

    # Phase 2: coordinate polish.
    t2 = min(remaining() * 0.20, 260.0)
    if t2 > 30.0:
        print("\n=== Phase 2: coordinate peel (n={:.0f}, {:.0f}s) ===".format(len(best), t2))
        a2, s2 = coordinate_peel(best, t2, verbose=True, seed=202)
        if np.isfinite(s2) and s2 < best_score:
            best = a2
            best_score = s2
            print("Phase 2 improved: {:.7f}".format(best_score))
        else:
            print("Phase 2 no improvement; best remains {:.7f}".format(best_score))

    # Phase 3: try longer interpolated lengths.
    n0 = len(best)
    trials = []
    if n0 < 18000:
        trials.append(min(18000, n0 + 3000))
    if n0 < 24000:
        trials.append(min(24000, int(1.5 * n0) + 1000))
    if n0 < 30000:
        trials.append(min(30000, n0 + 5000))

    # Unique increasing trial lengths.
    unique_trials = []
    for nt in trials:
        nt = int(nt)
        if nt <= n0 + 300:
            continue
        dup = False
        for ut in unique_trials:
            if abs(ut - nt) <= 200:
                dup = True
                break
        if not dup:
            unique_trials.append(nt)

    for idx_nt, nt in enumerate(unique_trials):
        rem = remaining()
        if rem < 150:
            break

        tri = min(rem * 0.45, 300.0)
        if tri < 60.0:
            break

        print("\n=== Phase 3.{:d}: longer length n={} ({:.0f}s) ===".format(idx_nt + 1, nt, tri))
        ini = interpolate_sequence(best, nt)
        a3, s3 = local_search(ini, tri, verbose=False, seed=303 + idx_nt * 17)
        cand = a3
        cand_score = s3

        rem = remaining()
        if rem > 70.0:
            tri2 = min(rem * 0.40, 170.0)
            a4, s4 = coordinate_peel(cand, tri2, verbose=False, seed=404 + idx_nt * 31)
            if np.isfinite(s4) and s4 < cand_score:
                cand = a4
                cand_score = s4

        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score
            print("Longer length improved: n={} score={:.7f}".format(len(best), best_score))
        else:
            print("Longer length no improvement: n={} score={:.7f}".format(nt, cand_score))

    # Phase 4: final polish on the retained best sequence.
    t4 = min(remaining() * 0.50, 300.0)
    if t4 > 35.0:
        print("\n=== Phase 4: final peel (n={:.0f}, {:.0f}s) ===".format(len(best), t4))
        a5, s5 = coordinate_peel(best, t4, verbose=True, seed=505)
        if np.isfinite(s5) and s5 < best_score:
            best = a5
            best_score = s5
            print("Phase 4 improved: {:.7f}".format(best_score))

    t5 = min(remaining() * 0.95, 220.0)
    if t5 > 30.0:
        print("\n=== Phase 5: final local search (n={:.0f}, {:.0f}s) ===".format(len(best), t5))
        a6, s6 = local_search(best, t5, verbose=False, seed=606)
        if np.isfinite(s6) and s6 < best_score:
            best = a6
            best_score = s6
            print("Phase 5 improved: {:.7f}".format(best_score))

    # Final sanitization: nonnegative, finite, sum far from zero, values bounded.
    out = np.asarray(best, dtype=np.float64).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    out = normalize_positive(out)

    # If somehow extremely short, ensure evaluator is happy.
    if out.size < 2:
        out = np.ones(1000, dtype=np.float64) / 1000.0

    final_fft_score = score_with_fft(out, next_pow2(2 * out.size - 1))
    print("\nFINAL: n={} fft_score={:.7f} elapsed={:.1f}s".format(out.size, final_fft_score, time.monotonic() - t0))

    return [float(x) for x in out]
```
