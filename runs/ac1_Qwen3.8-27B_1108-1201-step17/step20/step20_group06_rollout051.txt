def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * np.conjugate(A), fft_size)[:size]


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def norm1(a):

    s = a.sum()
    if s < 1e-15:

def smooth_max_grad(y, beta, top_k=None):
    n = len(y)
    b = fast_autocorr(y)
    size = 2 * n - 1
    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    if b_max <= 0:
        return np.zeros(n), b
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s
    fft_size = 1 << (n + size - 2).bit_length()
    Yr = np.fft.rfft(y[::-1], fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Yr * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1]
    return grad, b

            stall += 1

            continue

def lp_fw(a_init, time_limit, solver, max_K=2000, verbose=True):
    """LP-Frank-Wolfe heuristic for non-convex peak reduction."""
    n = len(a_init)
    a = norm1(a_init)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()

    max_no_improve = 500

    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_M - 1e-15:
            best_M = M_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if len(active) > max_K:
            margins = np.array([M_curr - ac[k] for k in active])
            keep = np.argsort(margins)[:max_K - 500]
            active = set(int(list(active)[i]) for i in keep)

        K = len(active)
        if K < 10:
            ac = fast_autocorr(a)
            active = set(int(k) for k in np.argsort(ac)[::-1][:K_init])
            K = len(active)

        constraint_list = sorted(active)
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = a[k - idx]

        g = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g) == 1, coeffs @ g <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Add violated constraints
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(a, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:25]]
            for k in top_v:
                active.add(int(k))

        # Line search
        best_alpha = 2.0 / (it + 2.0)
        best_trial = M_curr
        for alpha in [best_alpha, best_alpha * 0.5, best_alpha * 2, 0.03, 0.05, 0.1, 0.15]:
            alpha = max(0.01, min(0.4, alpha))
            trial = (1.0 - alpha) * a + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            Mt = fast_autocorr(trial).max()
            if Mt < best_trial:
                best_trial = Mt
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a / s

        if verbose and it % 30 == 0:
            J = 2.0 * n * best_M
            print(f"    LP-FW it={it} ({time.monotonic()-t0:.1f}s): J={J:.6f} |S|={len(active)}")

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

                best_seq = norm1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6000
        x = np.linspace(0, 1, n_base)
        best_seq = norm1(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Start: n={n0}, J={best_score:.6f}")

    # Phase 1: Nesterov (primary optimizer, ~30% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== P1: Nesterov (n={n_cur}) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 50:
        a_out, _ = nesterov_opt(norm1(best_seq), t1,
                                beta_start=15.0, beta_end=1500.0,
                                beta_growth=1.004, eta_init=0.5,
                                top_k_ratio=0.035, restart_every=2000,
                                verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 2: Peak shaving (~15%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== P2: PeakShave (n={n_cur}) ===")
    t2 = min(remaining * 0.12, 180.0)
    if t2 > 30:
        a_out, _ = peak_shave(norm1(best_seq), t2, verbose=True, seed=77)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps1", t_global)

    # Phase 3: LP-FW (~20%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== P3: LP-FW (n={n_cur}) ===")
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 40 and n_cur <= 9000:
        a_out, _ = lp_fw(norm1(best_seq), t3, solver, max_K=2500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 4: Nesterov round 2 with different params (~15%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== P4: Nesterov r2 (n={n_cur}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 40:
        a_out, _ = nesterov_opt(norm1(best_seq), t4,
                                beta_start=40.0, beta_end=3000.0,
                                beta_growth=1.006, eta_init=0.3,
                                top_k_ratio=0.02, restart_every=1500,
                                verbose=True, seed=999)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)

    # Phase 5: Spectral refinement (~10%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== P5: Spectral (n={n_cur}) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 25:
        a_out, _ = spectral_refine(norm1(best_seq), t5, verbose=True, seed=888)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec1", t_global)

    # Phase 6: Multi-length exploration (~15%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== P6: Multi-length (base n={n_cur}) ===")

    # Try slightly longer
    longer = min(n_cur + 500, 9000)
    if longer > n_cur and remaining > 80:
        print(f"  -> n={longer}")
        a_long = interpolate_seq(best_seq, longer)
        t6a = min(remaining * 0.35, 200.0)
        if t6a > 40:
            a_out, _ = nesterov_opt(a_long, t6a,
                                    beta_start=20.0, beta_end=1000.0,
                                    beta_growth=1.004, eta_init=0.4,
                                    top_k_ratio=0.03, restart_every=1500,
                                    verbose=True, seed=longer % 7919)
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"nest_n{longer}", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            a_out, _ = peak_shave(norm1(best_seq), min(remaining * 0.4, 100.0),
                                  verbose=True, seed=longer % 3163)
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"ps_n{longer}", t_global)

    # Phase 7: Final polish - Nesterov + peak shave alternating (~15%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== P7: Final polish (n={n_cur}) ===")
    if remaining > 40:
        t7a = min(remaining * 0.35, 150.0)
        if t7a > 20:
            a_out, _ = nesterov_opt(norm1(best_seq), t7a,
                                    beta_start=100.0, beta_end=5000.0,
                                    beta_growth=1.01, eta_init=0.2,
                                    top_k_ratio=0.01, restart_every=800,
                                    verbose=True, seed=555)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest3", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 25:
            t7b = min(remaining * 0.40, 120.0)
            if t7b > 15:
                a_out, _ = peak_shave(norm1(best_seq), t7b, verbose=True, seed=654)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "ps2", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20 and n_cur <= 8000:
            t7c = min(remaining * 0.40, 100.0)
            if t7c > 15:
                a_out, _ = lp_fw(norm1(best_seq), t7c, solver, max_K=2000, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: J={best_score:.6f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm1(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
