import numpy as np

_MODULE_SOLVER = None

def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    return _MODULE_SOLVER

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_grad(y, beta, top_k=None):
    """Compute smoothed gradient of max autocorrelation."""
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and 0 < top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b_use)
        eq = (b_max == b_use).sum()
        p[b_max == b_use] = 1.0 / max(1, eq)
    else:
        p = exp_vals / p_total
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1] / beta
    return grad, p, b

def fista_opt(a0, time_limit, beta_start=20.0, beta_end=800.0,
              beta_growth=1.003, top_k=15, restart_every=3000,
              verbose=True):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = float(fast_autocorr(x).max())
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(123)
    avg_x = x.copy()
    avg_count = 0.0
    s_prev = None
    y_prev = None
    bb_step = 1.0 / (n ** 0.5)

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, b = compute_grad(y, beta, top_k)

        # Fixed BB step (was buggy in previous code)
        if s_prev is not None and y_prev is not None:
            sy = y - y_prev
            gy = grad - s_prev
            sy_gy = sy.dot(gy)
            gy_gy = gy.dot(gy)
            if gy_gy > 1e-20 and sy_gy > 0:
                bb_candidate = sy_gy / gy_gy
                bb_step = 0.7 * bb_step + 0.3 * bb_candidate
                bb_step = max(1e-6, min(bb_step, 5.0 / beta))
            else:
                bb_step = max(bb_step * 0.8, 1e-6)
            s_prev = grad.copy()
            y_prev = y.copy()

        eta = min(bb_step, 0.2)
        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        avg_count += 1.0
        avg_x = avg_x + (x_new - avg_x) / avg_count

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 100 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.01 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 5.0)
            s_prev = None
            y_prev = None
            bb_step = 1.0 / (n ** 0.5)
            if verbose:
                elapsed = time.monotonic() - t0

        J_best = 2.0 * n * best_max_b
        if J_avg < J_best:
            return avg_x, float(fast_autocorr(avg_x).max())
    return best_a, best_max_b

def peak_scan_search(a0, time_limit, verbose=True):
    """Directly target the peak autocorrelation using coordinate moves."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        peak_idx = int(np.argmax(ac))
        
        # Find the coordinates contributing most to the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_idx - idx]
        contrib_sum = contrib.sum()
        
        # Find the top contributing coordinates
        top_contrib = np.argsort(contrib)[::-1][:20]
        src_candidates = idx[top_contrib]
        
        # Find "quiet" coordinates (not contributing to any top peak)
        top_peaks = np.argsort(ac)[::-1][:20]
        in_peak = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            s2 = max(0, int(k2) - n + 1)
            e2 = min(n - 1, int(k2))
            in_peak[s2:e2 + 1] = True
        
        quiet = np.where(~in_peak)[0]
        if len(quiet) < 5:
            quiet = np.arange(n)
        
        improved = False
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            
            src = int(src_candidates[rng.integers(len(src_candidates))])
            dst = int(quiet[rng.integers(len(quiet))])
            if dst == src:
                continue
            if a[src] < 1e-12:
                continue
            
            # Try to reduce the contribution at the peak
            # Move some mass from src to dst
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = float(fast_autocorr(a_trial).max())
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.002, 0.02))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = float(fast_autocorr(a).max())
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakSearch ({elapsed:.1f}s): J={J:.6f} stall={stall}")
    
    return best_a, best_m

            constraint_list = sorted(active_set)

        for jj, k in enumerate(constraint_list):

            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

            continue

            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check for violations and add to active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))
        
        # Backtracking line search
        default_alpha = 2.0 / (t + 2.0)
        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in [max(0.01, default_alpha), max(0.01, default_alpha * 0.5),
                      max(0.01, default_alpha * 2.0), 1.0 / (t + 1)]:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = norm_sum1(trial)
            M_trial = float(fast_autocorr(trial).max())
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y = norm_sum1(y_new)
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f} |S|={len(active_set)}")
    
    return best_a, best_m

def cyclic_shift_best(a0):
    n = len(a0)
    best_m = float(fast_autocorr(a0).max())
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()

        AS = np.fft.rfft(a_shift, fft_size)

            best_m = m
            best_shift = s

Actually, we can be smarter. The autocorrelation of a cyclically shifted sequence is just a cyclically shifted version of the original autocorrelation... no, that's not quite right. The convolution changes because the sequence wraps around.

Actually, for a cyclic shift by s positions:
a_shift[i] = a[(i+s) % n]

The convolution b_shift[k] = sum_j a_shift[j] * a_shift[(j+k) % n] = sum_j a[(j+s) % n] * a[(j+k+s) % n]

Let m = (j+s) % n: b_shift[k] = sum_m a[m] * a[(m+k) % n]

This is the CIRCULAR autocorrelation, not the linear autocorrelation. So the cyclic shift doesn't help with the linear convolution in general.

Wait, but the convolution in the evaluator is `np.convolve(sequence, sequence)` which is LINEAR convolution (full mode). So cyclic shifting changes the linear convolution.

For the linear convolution, shifting the sequence by s:
(conv(a_shift, a_shift))[k] = sum_j a_shift[j] * a_shift[k-j]
= sum_j a[(j+s) mod n] * a[(k-j+s) mod n] (with appropriate boundary handling)

This is NOT the same as the original convolution, so cyclic shift can help. But it's expensive to enumerate all shifts.

For n=6700, doing 6700 FFTs is feasible but might take a while. Each FFT is O(n log n) ≈ 6700 * 13 ≈ 87K ops. 6700 * 87K ≈ 580M. At 1GHz single core, that's about 0.6 seconds. So it's fine.

But actually, we can do better. The linear convolution of a shifted sequence can be computed from the original FFT by applying a phase factor:

If A = FFT(a), then FFT(a_shifted) = A * exp(-2*pi*i*k*s/N) where N is the FFT size.

So b_shifted = IFFT(|A * exp(-2*pi*i*k*s/N)|^2) = IFFT(|A|^2) = b

Wait, that's the same! Because |exp(-2*pi*i*k*s/N)|^2 = 1. So the power spectrum is the same, and the autocorrelation (in the circular sense) is the same.

But the LINEAR convolution is different from the circular convolution! The linear convolution has edge effects that depend on the shift.

So cyclic shift CAN change the linear convolution. But we can't use the simple FFT trick. We'd need to actually compute the linear convolution for each shift.

For n=6700, 6700 linear convolutions using FFT would take about 6700 * 2 * O(13400 log 13400) ≈ 6700 * 2 * 13400 * 14 ≈ 2.5 billion ops. That might take a few seconds.

Actually, for the linear convolution, the effect of a shift is:
- The "interior" part of the convolution (lags k where 0 <= k <= n-1 and the overlap is full) is the same regardless of shift
- Only the "edge" lags (k near 0 or k near 2n-2) change

So the max of the linear convolution is determined by either:
- The interior lags (which are the same for all shifts)
- The edge lags (which depend on the shift)

If the max is in the interior, cyclic shift doesn't help. If the max is at the edge, we can try to reduce the edge peaks by shifting.
