                   cooling_rate=0.999, momentum=0.9, top_k=10, verbose=True):
    n = len(a0)

    v = np.zeros(n)  # momentum

    eta = eta_start
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m:
            best_m = c_max
            best_a = a.copy()
        
        # Compute subgradient using top-k peaks
        if top_k is not None and top_k < len(c):
            idx = np.argsort(c)[::-1][:top_k]
            w = c[idx] - c[idx].min() + 0.1
            w = w / w.sum()
        else:

            w = [1.0]
        
        # Subgradient: sum over top peaks of w_k * 2*a[k-j]
        # grad_j = 2 * sum_k w_k * a[k_j - j] (with bounds)
        grad = np.zeros(n)

            # a[k-j] for j in [max(0, k-n+1), min(n-1, k)]

            j_end = min(n - 1, k)
            grad[j_start:j_end+1] += wk * 2.0 * a[k - np.arange(j_start, j_end+1)]
        
        # Heavy ball
        v = momentum * v + (1 - momentum) * grad
        v = v / max(1.0, v.max())  # normalize

        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        eta = max(eta_end, eta * cooling_rate)
        it += 1

            J = 2.0 * n * best_m

    """Find b that maximizes sum(b) subject to b>=0 and (f*b)_k <= M for top K peaks."""
    import cvxpy as cp
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    
    # Compute convolution f * b (symbolically)
    # (f*b)_k = sum_j f[k-j] * b_j (with f out of range = 0)
    # For the top K peaks:
    c = np.convolve(f, f, mode='full')
    top_idx = np.argsort(c)[::-1][:K]
    
    constraints = []
    for k in top_idx:
        k = int(k)
        # (f*b)_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[k-j] * b_j
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        coeffs = f[k - np.arange(j_start, j_end + 1)]
        constraints.append(np.dot(coeffs, b[j_start:j_end+1]) <= M)
    
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    
    if solver is not None:
        prob.solve(solver=solver, verbose=False)
    else:
        prob.solve(verbose=False)
    
    if prob.status in ('optimal', 'optimal_inaccurate'):
        return np.asarray(b.value).flatten()
    return None

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)
