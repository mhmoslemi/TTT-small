import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def subgradient_nesterov(a0, time_limit, verbose=True, seed=42):

    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    t_nest = 1.0
    gamma = 0.4  # base step size
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        y = y / s_y
        
        # Compute autocorrelation and find max

        c_max = c.max()
        k_max = int(np.argmax(c))

        idx = np.arange(n)

        valid = (partner >= 0) & (partner < n)
        grad[valid] = 2.0 * y[partner[valid]]
        
        # Step: normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        
        # Adaptive step: ensure we don't overshoot
        eta = gamma / (t * g_norm)
        eta = min(eta, 0.05)
        
        z = y - eta * grad
        x_new = simplex_project(z)
        
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-12:
            best_max_b = max_new
            best_a = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        if no_improve > 2000:
            # Perturb and restart
            t_nest = 1.0
            noise = 0.02 * rng.normal(0, 1, n) * best_a

            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            no_improve = 0

def simulated_annealing(a0, time_limit, verbose=True, seed=99):
    """Simulated annealing with diverse moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_max_b = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    T_start = 0.001 / n  # initial temperature (in terms of change in max_B)
    T_end = 1e-10 / n
    cooling = 0.9995
    
    no_improve = 0
    max_no_improve = 5000
    
    T = T_start
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if T < T_end:
            break
        
        # Choose move type
        move_type = rng.integers(0, 4)
        a_trial = a.copy()
        
        if move_type == 0:
            # Single pair: move from i to j
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            if a[i] < 1e-15:
                continue
            frac = rng.uniform(0.001, 0.1)
            amt = a[i] * frac
            a_trial[i] -= amt
            a_trial[j] += amt
        elif move_type == 1:
            # Anti-symmetric nudge
            i = int(rng.integers(n))
            j = n - 1 - i
            if i == j:
                j = (i + 1) % n
            amt = a[i] * rng.uniform(0.001, 0.05)
            a_trial[i] -= amt
            a_trial[j] += amt
        elif move_type == 2:
            # Block shift
            width = int(rng.integers(3, max(4, n // 20)))
            start = int(rng.integers(0, n - width))
            shift = int(rng.integers(-5, 6))
            if abs(shift) < width:
                for m in range(width):
                    dst = start + m + shift
                    if 0 <= dst < n and dst != start + m:
                        a_trial[start + m] -= a_trial[start + m] * 0.1
                        a_trial[dst] += a_trial[start + m] * 0.1
        else:
            # Gaussian perturbation on a small fraction
            n_perturb = int(n * 0.01)
            idx = rng.integers(0, n, n_perturb)
            for i in idx:
                if a[i] > 1e-15:
                    amt = a[i] * rng.normal(0, 0.02)
                    if amt > 0:
                        j = int(rng.integers(n))
                        a[i] -= amt
                        a[j] += amt
                    else:
                        j = int(rng.integers(n))
                        a[i] -= amt  # amt is negative, so this adds
                        a[j] += amt  # hmm this is wrong
        
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t
        
        max_trial = fast_autocorr(a_trial).max()
        delta = max_trial - best_max_b
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_trial
            if max_trial < best_max_b:
                best_max_b = max_trial
                best_a = a_trial.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        T *= cooling
        
        if no_improve > max_no_improve:
            # Restart with perturbation
            x_r = best_a + 0.03 * rng.normal(0, 1, n) * best_a
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            a = x_r
            T = T_start * 0.5
            no_improve = 0
