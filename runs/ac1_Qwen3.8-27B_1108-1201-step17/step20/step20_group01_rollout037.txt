```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for nonnegative a.

Strategy:
- Start from the supplied parent if available, otherwise a U-shaped smooth seed.
- Use log-domain / mirror descent with a log-sum-exp smoothing of the max-conv.
- Use an exact two-coordinate local search. Candidate convolution updates are
  applied directly in O(n) per trial instead of recomputing full FFTs.
- Periodically interpolate to nearby lengths to escape length-specific basins.
- Final answer is the best sequence found, normalized to sum 1.
"""

import numpy as np
import time
import math


def _norm1(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0:
        return np.ones(1, dtype=np.float64)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(x.size, dtype=np.float64) / x.size
    return x / s


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _autocorr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(1, dtype=np.float64)
    size = 2 * n - 1
    fs = _next_pow2(size)
    A = np.fft.rfft(a, fs)
    c = np.fft.irfft(A * A, fs)[:size]
    c = np.where(c > 0.0, c, 0.0)
    return c


def _score_unit(a):
    """Assumes sum(a) == 1 and returns 2*n*max(converage)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    return float(2.0 * a.size * float(np.max(_autocorr(a))))


def _score_value(a):
    """Scale-invariant objective, matches evaluate_sequence for valid inputs."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return float("inf")
    return float(2.0 * a.size * float(np.max(_autocorr(a))) / (s * s))


def _logvec(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    z = np.log(np.maximum(x, 1e-300))
    z -= float(z.max())
    return z


def _weights_from_ac(ac, beta):
    ac = np.asarray(ac, dtype=np.float64).ravel()
    L = ac.size
    if beta <= 1e-12:
        p = np.ones(L, dtype=np.float64)
    else:
        m = float(ac.max())
        x = beta * (ac - m)
        p = np.exp(x)
        s = p.sum()
        if not np.isfinite(s) or s <= 0:
            p = np.ones(L, dtype=np.float64)
        else:
            p = p / s
    return p


def _grad_half(a, w):
    """
    Half-gradient of sum_k w_k * conv(a,a)_k with respect to a.
    For weighted c, full gradient is 2 * this.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    if n == 1:
        return np.array([a[0] * w[0]], dtype=np.float64)
    L = 2 * n - 1
    if w.size != L:
        w = np.interp(np.arange(L), np.linspace(0, 1, w.size), w)
    q = w[::-1]
    req = 3 * n - 2
    fs = _next_pow2(req)
    A = np.fft.rfft(a, fs)
    W = np.fft.rfft(q, fs)
    c = np.fft.irfft(A * W, fs)
    return np.asarray(c[2 * n - 2:n - 2:-1], dtype=np.float64)


def _mirror_step(a, z, g, step0, current_score, mults):
    n = a.size
    g = np.asarray(g, dtype=np.float64)
    if n == 0 or not np.all(np.isfinite(g)):
        return a, z, current_score, 0.0
    gscale = math.sqrt(float(np.mean(g * g)))
    if not np.isfinite(gscale) or gscale <= 1e-15:
        return a, z, current_score, 0.0

    base = step0 / gscale
    if base > 0.8:
        base = 0.8
    if base <= 0.0:
        return a, z, current_score, 0.0

    best_a = a
    best_z = z
    best_s = current_score
    best_mult = 0.0

    for mult in mults:
        eta = base * float(mult)
        if eta <= 0.0 or not np.isfinite(eta):
            continue
        zt = z - eta * g
        if not np.all(np.isfinite(zt)):
            continue
        zt = zt - float(zt.max())
        zt = np.maximum(zt, -700.0)
        at = np.exp(zt)
        ss = float(at.sum())
        if not np.isfinite(ss) or ss <= 1e-300:
            continue
        at = at / ss
        sc = _score_unit(at)
        if sc < best_s - 1e-15:
            best_a = at
            best_z = zt
            best_s = sc
            best_mult = float(mult)

    return best_a, best_z, best_s, best_mult


def smooth_mirror_refine(a0, time_limit, verbose=True):
    a = _norm1(a0)
    n = a.size
    if time_limit <= 0.0:
        return a, _score_unit(a)

    t0 = time.monotonic()
    z = _logvec(a)
    best_a = a.copy()
    best_s = _score_unit(a)

    beta = 12.0
    step0 = 0.35
    stall = 0
    it = 0
    rng = np.random.default_rng(999)
    mults = (1.0, 0.5, 0.25, 0.1, 0.05)

    while True:
        if time.monotonic() - t0 >= time_limit:
            break

        ac = _autocorr(a)
        s = 2.0 * n * float(ac.max())
        if s < best_s - 1e-14:
            best_s = s
            best_a = a.copy()
        else:
            stall += 1

        if stall > 50:
            beta = min(500.0, beta * 1.04)

        if stall > 900:
            noise_scale = 0.01 + 0.005 * min(5.0, stall / 150.0)
            try:
                a = best_a.copy() + noise_scale * rng.standard_normal(n)
            except Exception:
                a = best_a.copy()
            a = np.maximum(a, 1e-300)
            ss = float(a.sum())
            if ss > 0:
                a = a / ss
            z = _logvec(a)
            stall = 0
            beta = 20.0
            step0 = 0.35

        w = _weights_from_ac(ac, beta)
        h = _grad_half(a, w)
        g = 4.0 * n * h
        g = g - float(g.mean())

        a_new, z_new, s_new, used_mult = _mirror_step(a, z, g, step0, s, mults)

        if s_new < s - 1e-14:
            a = a_new
            z = z_new
            step0 = min(1.2, step0 * 1.03)
        else:
            step0 = max(0.02, step0 * 0.75)
            if used_mult == 0.0:
                step0 = max(0.01, step0 * 0.5)

        it += 1
        if verbose and it % 250 == 0:
            print("    smooth_mirror it={:5d} time={:6.1f}s beta={:6.1f} step={:.3f} cur={:.7f} best={:.7f}"
                  .format(it, time.monotonic() - t0, beta, step0, s, best_s))

    best_a = _norm1(best_a)
    best_s = _score_unit(best_a)
    return best_a, best_s


def interpolate_seq(a, new_n):
    a = _norm1(a)
    old_n = a.size
    new_n = int(max(2, new_n))
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    b = np.interp(new_x, old_x, a)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s <= 1e-300:
        return np.ones(new_n, dtype=np.float64) / new_n
    return b / s


def _pressure_top(a, ac, top_lags, max_lags=24):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    pr = np.zeros(n, dtype=np.float64)
    top = top_lags[:max_lags]
    for kk in top:
        k = int(kk)
        ws = max(0, k - n + 1)
        we = min(n - 1, k)
        if ws <= we:
            idx = np.arange(ws, we + 1)
            pr[ws:we + 1] += a[idx] * a[k - idx]
    return pr


def _move_max(a, ac, src, dst, frac):
    """Exact new max convolution coefficient after moving frac of a[src] to a[dst]."""
    if frac <= 0.0:
        return float(ac.max())
    if frac > 0.999:
        frac = 0.999
    d = float(a[src]) * frac
    nac = ac.copy()
    n = a.size
    L = ac.size

    e = min(n, L - dst)
    if e > 0:
        nac[dst:dst + e] += 2.0 * d * a[:e]

    e = min(n, L - src)
    if e > 0:
        nac[src:src + e] -= 2.0 * d * a[:e]

    dd = d * d
    k = 2 * dst
    if 0 <= k < L:
        nac[k] += dd
    k = 2 * src
    if 0 <= k < L:
        nac[k] += dd
    k = src + dst
    if 0 <= k < L:
        nac[k] -= 2.0 * dd

    return float(nac.max())


def _move_both(a, ac, src, dst, frac):
    if frac > 0.999:
        frac = 0.999
    d = float(a[src]) * frac
    b = a.copy()
    b[src] -= d
    b[dst] += d

    nac = ac.copy()
    n = a.size
    L = ac.size

    e = min(n, L - dst)
    if e > 0:
        nac[dst:dst + e] += 2.0 * d * a[:e]

    e = min(n, L - src)
    if e > 0:
        nac[src:src + e] -= 2.0 * d * a[:e]

    dd = d * d
    k = 2 * dst
    if 0 <= k < L:
        nac[k] += dd
    k = 2 * src
    if 0 <= k < L:
        nac[k] += dd
    k = src + dst
    if 0 <= k < L:
        nac[k] -= 2.0 * dd

    return b, nac


def coord_pair_refine(a0, time_limit, seed=11, verbose=True):
    a = _norm1(a0)
    n = a.size
    if time_limit <= 0.0:
        return a, _score_unit(a)

    t0 = time.monotonic()
    ac = _autocorr(a)
    s = 2.0 * n * float(ac.max())
    best_a = a.copy()
    best_s = s

    rng = np.random.default_rng(seed)
    beta = 12.0
    max_pairs = 24
    stall = 0
    recompute_counter = 0
    it = 0

    while True:
        if time.monotonic() - t0 >= time_limit:
            break

        recompute_counter += 1
        if recompute_counter >= 60:
            a = _norm1(a)
            ac = _autocorr(a)
            s = 2.0 * n * float(ac.max())
            if s < best_s - 1e-14:
                best_s = s
                best_a = a.copy()
            recompute_counter = 0

        if s < best_s - 1e-14:
            best_s = s
            best_a = a.copy()

        try:
            w = _weights_from_ac(ac, beta)
            h = _grad_half(a, w)
        except Exception:
            w = _weights_from_ac(ac, 10.0)
            h = _grad_half(a, w)

        try:
            top_lags = np.argsort(ac)[::-1][:32]
            pr = _pressure_top(a, ac, top_lags, 24)
        except Exception:
            top_lags = np.argsort(ac)[::-1][:24]
            pr = _pressure_top(a, ac, top_lags, 16)

        hi_h = np.argsort(h)[-36:]
        lo_h = np.argsort(h)[:36:]
        hi_p = np.argsort(pr)[-24:]
        lo_p = np.argsort(pr)[:24:]

        src_set = np.unique(np.concatenate((hi_h, hi_p)))
        dst_set = np.unique(np.concatenate((lo_h, lo_p)))

        src_list = []
        for i in src_set:
            i = int(i)
            if i < n and a[i] > 1e-12:
                src_list.append(i)
        dst_list = []
        for i in dst_set:
            i = int(i)
            if i < n:
                dst_list.append(i)

        if not src_list:
            for i in range(n):
                if a[i] > 1e-10:
                    src_list.append(i)
                    if len(src_list) >= 20:
                        break
        if not dst_list:
            dst_list = list(range(n))

        pairs = []
        for src in src_list:
            for dst in dst_list:
                if src == dst:
                    continue
                pot = float(h[src] - h[dst] + 0.25 * pr[src])
                pairs.append((pot, src, dst))

        if not pairs:
            for _ in range(16):
                src = int(rng.integers(n))
                dst = int(rng.integers(n))
                if src != dst and a[src] > 1e-12:
                    pairs.append((0.0, src, dst))

        if pairs:
            pairs.sort(reverse=True)

        improved = False
        limit = min(len(pairs), max_pairs)
        for rank in range(limit):
            pot, src, dst = pairs[rank]
            src = int(src)
            dst = int(dst)
            if src == dst or src < 0 or dst < 0 or src >= n or dst >= n:
                continue
            if a[src] <= 1e-15:
                continue

            if pot > 0.0:
                fracs = (0.02, 0.05, 0.10, 0.20, 0.35, 0.55)
            else:
                fracs = (0.005, 0.01, 0.02, 0.05, 0.10, 0.20)

            best_cand_s = s
            best_frac = 0.0

            for frac in fracs:
                mc = _move_max(a, ac, src, dst, float(frac))
                sc = 2.0 * n * mc
                if sc < best_cand_s - 1e-14:
                    best_cand_s = sc
                    best_frac = float(frac)

            # Small local refinement for the best few promising pairs.
            if best_frac > 0.0 and rank < 4:
                lo = max(0.0005, best_frac * 0.5)
                hi = min(0.999, best_frac * 2.0)
                if hi > lo + 1e-6:
                    for _ in range(3):
                        m1 = lo + (hi - lo) * (1.0 / 3.0)
                        m2 = hi - (hi - lo) * (1.0 / 3.0)
                        s1 = 2.0 * n * _move_max(a, ac, src, dst, m1)
                        s2 = 2.0 * n * _move_max(a, ac, src, dst, m2)
                        if s1 < s2:
                            hi = m2
                        else:
                            lo = m1
                    mid = 0.5 * (lo + hi)
                    smid = 2.0 * n * _move_max(a, ac, src, dst, mid)
                    if smid < best_cand_s - 1e-15:
                        best_cand_s = smid
                        best_frac = mid

            if best_frac > 0.0 and best_cand_s < s - 1e-14:
                b, nac = _move_both(a, ac, src, dst, best_frac)
                a = b
                ac = nac
                s = best_cand_s
                improved = True
                stall = 0
                if s < best_s - 1e-14:
                    best_s = s
                    best_a = a.copy()
                break

        if not improved:
            stall += 1
            if stall % 50 == 0:
                beta = min(800.0, beta * 1.1)
                if max_pairs < 60:
                    max_pairs += 2

            if stall > 180:
                noise = 0.004 + 0.005 * min(4.0, stall / 250.0)
                try:
                    a = best_a.copy() + noise * rng.standard_normal(n)
                except Exception:
                    a = best_a.copy()
                a = np.maximum(a, 1e-300)
                ss = float(a.sum())
                if ss > 0:
                    a = a / ss
                ac = _autocorr(a)
                s = 2.0 * n * float(ac.max())
                if s < best_s - 1e-14:
                    best_s = s
                    best_a = a.copy()
                beta = min(50.0, beta)
                max_pairs = 24
                stall = 0
                recompute_counter = 0

        it += 1
        if verbose and it % 250 == 0:
            print("    coord_pair it={:5d} time={:6.1f}s beta={:6.1f} pairs={:3d} cur={:.7f} best={:.7f}"
                  .format(it, time.monotonic() - t0, beta, max_pairs, s, best_s))

    best_a = _norm1(best_a)
    best_s = _score_value(best_a)
    return best_a, best_s


def random_start(n, seed=7):
    n = int(max(1000, n))
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.8 * np.cos(np.pi * x)
    a += 0.03 * rng.standard_normal(n)
    a += 0.02 * np.sin(6.0 * np.pi * x)
    a = np.maximum(a, 1e-6)
    return _norm1(a)


def _initial_sequence():
    parent = None
    try:
        p = globals().get("height_sequence_1", None)
        if p is not None:
            arr = np.asarray(p, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 50 and np.isfinite(arr).all() and float(arr.sum()) > 1e-12:
                parent = arr
    except Exception:
        parent = None

    if parent is None:
        return random_start(12000, 7)

    a = _norm1(parent)
    n = a.size
    if n < 6000:
        a = interpolate_seq(a, 10000)
    elif n > 20000:
        a = interpolate_seq(a, 16000)

    # Try a symmetrization once; keep the better of original and symmetric.
    try:
        b = _norm1(0.5 * (a + a[::-1]))
        sb = _score_value(b)
        sa = _score_value(a)
        if sb < sa:
            a = b
    except Exception:
        pass

    return a


def propose_candidate(budget_s=1800.0):
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1800.0
    if not np.isfinite(budget) or budget <= 60.0:
        budget = max(60.0, float(budget_s) if np.isfinite(budget_s) else 1800.0)

    t_global = time.monotonic()
    total = max(30.0, budget - 45.0)

    best_seq = _initial_sequence()
    best_score = _score_value(best_seq)
    n0 = best_seq.size
    print("INIT n={} score={}".format(n0, repr(best_score)))

    elapsed = time.monotonic() - t_global
    limit = min(220.0, total * 0.16)
    if total - elapsed > limit and limit > 30.0:
        seq, sc = smooth_mirror_refine(best_seq, limit, verbose=True)
        sc = _score_value(seq)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq
            print("  after smooth mirror: {}".format(repr(best_score)))

    # Length interpolation exploration.
    elapsed = time.monotonic() - t_global
    explore_budget = min(280.0, total * 0.18)
    used_explore = 0.0
    n_explore = len(best_seq)
    trial_factors = (1.20, 1.45, 1.70, 0.80)
    tried = [n_explore]

    for fac in trial_factors:
        n_try = int(n_explore * fac)
        n_try = int(min(20000, max(6000, n_try)))
        if n_try in tried:
            continue
        tried.append(n_try)

        elapsed = time.monotonic() - t_global
        remain = total - elapsed
        t_try = min(80.0, explore_budget - used_explore)
        if remain < max(90.0, t_try + 120.0) or t_try <= 25.0:
            break

        try:
            a_try = interpolate_seq(best_seq, n_try)
        except Exception:
            a_try = random_start(n_try, 123 + n_try)

        t_start = time.monotonic()
        seq, sc = smooth_mirror_refine(a_try, t_try, verbose=False)
        used_explore += time.monotonic() - t_start
        sc = _score_value(seq)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq
            print("  explore n={} improved: {}".format(n_try, repr(best_score)))
        else:
            print("  explore n={} score={} (no improvement)".format(n_try, repr(sc)))

    elapsed = time.monotonic() - t_global
    limit = min(800.0, (total - elapsed) * 0.45)
    if limit > 60.0 and total - elapsed > limit + 120.0:
        seq, sc = coord_pair_refine(best_seq, limit, seed=42, verbose=True)
        sc = _score_value(seq)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq
            print("  after main coord: {}".format(repr(best_score)))

    elapsed = time.monotonic() - t_global
    limit = min(220.0, (total - elapsed) * 0.22)
    if limit > 30.0 and total - elapsed > limit + 80.0:
        seq, sc = smooth_mirror_refine(best_seq, limit, verbose=True)
        sc = _score_value(seq)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq
            print("  after mid smooth: {}".format(repr(best_score)))

    elapsed = time.monotonic() - t_global
    limit = total - elapsed - 25.0
    if limit > 25.0:
        seq, sc = coord_pair_refine(best_seq, limit, seed=777, verbose=True)
        sc = _score_value(seq)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq
            print("  after final coord: {}".format(repr(best_score)))

    # Final sanity checks and normalization.
    best_seq = _norm1(best_seq)
    fscore = _score_value(best_seq)
    if not np.isfinite(fscore) or fscore > best_score + 1e-12:
        best_score = fscore

    # Optional external verification if evaluation function is available.
    try:
        ext = float(evaluate_sequence([float(x) for x in best_seq]))  # noqa: F821
        if np.isfinite(ext):
            best_score = ext
            print("EXTERNAL VERIFY score={}".format(repr(ext)))
    except Exception:
        pass

    print("FINAL n={} score={:.10f} elapsed={:.1f}s".format(
        len(best_seq), best_score, time.monotonic() - t_global))

    out = []
    for x in best_seq:
        out.append(float(x))
    return out
```
