For `k ≠ 2j`: `c_new[k] = c_old[k] + 2*delta*a[k-j]` (linear in delta)
For `k = 2j`: `c_new[k] = c_old[k] + 2*a[j]*delta + delta^2` (quadratic in delta)

We want to minimize `max_k c_new[k]` over `delta` (subject to `a[j] + delta >= 0`, i.e., `delta >= -a[j]`).

This is minimizing the max of a set of linear functions (and one quadratic function). The max of linear functions is convex, and adding a convex quadratic keeps it convex. So we can use binary search or subgradient methods.

Actually, for the 1D problem of minimizing `max_i (L_i + m_i * delta)` where `L_i` and `m_i` are constants, the minimum is achieved where the "active constraints" are balanced.

Let me implement a simple version: for each coordinate `j`, try a few values of `delta` (e.g., using a golden section search over the range `[-a[j], a[j]]`) and pick the best.

This might be slow for large `n`, but we can:
- Only do it for a subset of coordinates (e.g., the ones that contribute most to the peaks)
- Use a coarse-to-fine approach
- Limit the number of iterations

Let me now write the final code:

```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.
Uses: FISTA with adaptive softmax smoothing, coordinate descent refinement,
multi-start with diverse initializations, and cyclic shift optimization.
"""

def fast_autocorr(a):
    """Compute self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto the probability simplex."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def smoothed_max_grad(y, beta, top_k=None):
    """Compute gradient of smoothed max of self-convolution.
    
    Uses softmax approximation: f(a) ≈ (1/beta) * log(sum(exp(beta * c[k])))
    Gradient: (Q * p) where p[k] = exp(beta*c[k]) / sum(exp(beta*c[k]))
    and Q is the "convolution operator" (linear part).
    """
    n = len(y)
    c = fast_autocorr(y)
    
    # Softmax weights
    c_shifted = beta * (c - c.max())
    c_shifted = np.clip(c_shifted, -500, 500)
    p = np.exp(c_shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(c)
        p[c == c.max()] = 1.0 / (c == c.max()).sum()
    else:
        p = p / p_sum
    
    # Gradient: df/da[j] = sum_k p[k] * dc[k]/da[j]
    # dc[k]/da[j] = a[k-j] (if valid) + a[k-j] (if valid, for the other term)
    # Actually: c[k] = sum_i a[i]*a[k-i]
    # dc[k]/da[j] = a[k-j] (from i=j term) + a[k-j] (from k-i=j term) if j != k-j
    #             = 2*a[j] (from the i=j=k-j term) if j = k-j
    # More carefully: dc[k]/da[j] = sum over terms where j appears
    # = a[k-j] (if 0<=k-j<n) + a[k-j] (if 0<=k-j<n and k-j != j)
    # Wait, let me just use the convolution formula directly.
    # c = convolve(a, a), so c[k] = sum_i a[i]*a[k-i]
    # dc/da[j] = d/d a[j] sum_i a[i]*a[k-i]
    # The term a[j] appears in c[k] when i=j (giving a[k-j]) and when k-i=j (giving a[i]=a[k-j])
    # So dc[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j] if j != k-j
    #                  = 2*a[j] if j = k-j (but we only count it once since a[j]*a[j] = a[j]^2, d/d a[j] = 2*a[j])
    # Actually: if j = k-j, i.e., k = 2j, then the term is a[j]^2, derivative is 2*a[j].
    # If j != k-j, the term appears twice: a[j]*a[k-j] and a[k-j]*a[j], derivative is 2*a[k-j].
    # In both cases: dc[k]/da[j] = 2*a[k-j] (where a[k-j] = a[j] when k=2j).
    
    # So the gradient is: grad[j] = sum_k p[k] * 2 * a[k-j] (for valid k-j)
    # This is 2 * (p * a_reversed) convolved, evaluated at position j.
    # More precisely: grad = 2 * convolve(p, a) evaluated at positions [0, n-1]
    # where convolve(p, a)[j] = sum_k p[k] * a[j-k] ... hmm, let me think again.
    
    # grad[j] = 2 * sum_k p[k] * a[k-j] (where 0 <= k-j < n)
    # Let m = k-j, so k = j+m:
    # grad[j] = 2 * sum_m p[j+m] * a[m] (where 0 <= j+m < 2n-1 and 0 <= m < n)
    # = 2 * (p * a)[j] where * denotes cross-correlation... 
    
    # Actually, let's just compute it as a convolution:
    # grad[j] = 2 * sum_{k=0}^{2n-2} p[k] * a[k-j] * [0 <= k-j < n]
    # Let l = k-j: grad[j] = 2 * sum_{l} p[j+l] * a[l] * [0 <= j+l < 2n-1 and 0 <= l < n]
    # = 2 * sum_{l=0}^{n-1} p[j+l] * a[l] * [j+l < 2n-1]
    # For j in [0, n-1] and l in [0, n-1]: j+l is in [0, 2n-2], always valid.
    # So grad[j] = 2 * sum_{l=0}^{n-1} p[j+l] * a[l]
    
    # This is the cross-correlation of p and a_reversed, or equivalently:
    # grad = 2 * (a * p_shifted) where p_shifted is appropriate
    
    # Simplest: use FFT
    # grad[j] = 2 * sum_{l=0}^{n-1} a[l] * p[j+l]
    # This is the "full convolution" of a and p, evaluated at specific positions.
    # conv(a, p)[j] = sum_l a[l] * p[j-l] ... that's not quite right.
    # Let me define: h[j] = sum_l a[l] * p[j+l]
    # This is the cross-correlation: h = correlate(p, a) (with appropriate indexing)
    # Or: h[j] = sum_l a[l] * p[j+l] = (a * p_rev)[j + n - 1] where p_rev = p[n-1:...:0]
    # Hmm, this is getting confusing. Let me just use a direct FFT approach.
    
    # Alternative: use the fact that grad = 2 * (Q^T * p) where Q is the linear operator
    # mapping a to c = conv(a,a) ... but Q is not linear (it's quadratic).
    
    # Let me just compute it directly using FFT:
    # grad[j] = 2 * sum_{l=0}^{n-1} a[l] * p[j+l] for j = 0, ..., n-1
    # Define b = a, and we want sum_l b[l] * p[j+l] for j = 0, ..., n-1
    # This is the cross-correlation of b and p: (b ⋆ p)[j] = sum_l b[l] * p[l+j]
    # Which equals conv(b, p_rev)[n-1+j] where p_rev = p[::-1]
    # Let me verify: conv(b, p_rev)[k] = sum_i b[i] * p_rev[k-i] = sum_i b[i] * p[n-1-(k-i)]
    # For k = n-1+j: sum_i b[i] * p[n-1-(n-1+j-i)] = sum_i b[i] * p[i-j]
    # That gives sum_i b[i]*p[i-j], not sum_l b[l]*p[l+j].
    
    # Let me try differently. Define q = p reversed and shifted.
    # We want: h[j] = sum_{l=0}^{n-1} a[l] * p[l+j]
    # This is the "full" cross-correlation, which can be computed as:
    # h = ifft(fft(a) * conj(fft(p))) ... but p has length 2n-1.
    
    # Actually, the simplest approach: 
    # h[j] = sum_l a[l] * p[l+j]
    # Let's define a_padded of length 2n-1 with a in the first n positions.
    # Then h[j] for j=0,...,n-1 is the correlation of a and p.
    # Using FFT: h_full = ifft(fft(a_padded, 4n) * conj(fft(p, 4n)))
    # Then h[j] = h_full[j] for j = 0, ..., n-1
    
    # Actually even simpler: just use np.correlate or manual FFT.
    
    # Let me use the FFT approach:
    # h[j] = sum_{l=0}^{n-1} a[l] * p[l+j] (j = 0, ..., n-1)
    # = (a ⋆ p)[j] (cross-correlation)
    # Using FFT: h = irfft(rfft(a, N) * conj(rfft(p, N)))[:n] where N >= 3n-2
    
    N = 1 << (3 * n).bit_length()
    A_fft = np.fft.rfft(y, N)
    P_fft = np.fft.rfft(p, N)
    h = np.fft.irfft(A_fft * np.conj(P_fft), N)[:n]
    
    grad = 2.0 * h  # Wait, I need to double-check the factor of 2.
    
    # Let me re-derive: f(a) ≈ (1/beta) * log(sum_k exp(beta * c[k]))
    # df/da[j] = (1/beta) * sum_k exp(beta*c[k]) * beta * dc[k]/da[j] / sum_k exp(beta*c[k])
    # = sum_k p[k] * dc[k]/da[j]
    # = sum_k p[k] * 2*a[k-j] (for valid k-j, as derived above)
    # = 2 * sum_{l=0}^{n-1} p[j+l] * a[l]
    # = 2 * h[j]
    
    return 2.0 * h


def fista_optimize(a0, time_limit, beta_start=20.0, beta_end=1000.0,
                   beta_growth=1.004, restart_every=3000, verbose=True):
    """FISTA with adaptive step size and Nesterov acceleration."""
    n = len(a0)
    x = normalize(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(42)
    
    # Adaptive step size (Barzilai-Borwein)
    y_prev = None
    grad_prev = None
    step = 1.0 / np.sqrt(n)
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute gradient of smoothed max
        grad = smoothed_max_grad(y, beta)
        
        # Barzilai-Borwein step size update
        if y_prev is not None and grad_prev is not None:
            sy = y - y_prev
            gy = grad - grad_prev
            sy_gy = sy.dot(gy)
            gy_gy = gy.dot(gy) + 1e-20
            if sy_gy > 1e-10:
                bb_step = sy_gy / gy_gy
                step = max(1e-6, min(bb_step, 2.0 / (beta * 10)))
            else:
                step = max(step * 0.9, 1e-6)
        y_prev = y.copy()
        grad_prev = grad.copy()
        
        # Gradient step + projection
        z = y - step * grad
        x_new = simplex_project(z)
        
        # Evaluate
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x.copy()
        x = x_new
        
        # Update beta
        t_nest = t_new
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        # Restart
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.02 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise_scale, n)
            x_r = best_a * (1.0 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 10.0)
            y_prev = None
            grad_prev = None
            step = 1.0 / np.sqrt(n)
            
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    FISTA restart t={} ({}s): J={}".format(t, round(elapsed, 1), round(J, 6)))
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    FISTA t={} ({}s): J={} beta={}".format(t, round(elapsed, 1), round(J, 6), round(beta, 1)))
    
    return best_a, best_max_b


def coordinate_descent(a0, time_limit, n_sweeps=5, verbose=True):
    """Coordinate descent: for each coordinate, find the best value via search."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Process coordinates in random order
        order = rng.permutation(n)
        improved = False
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try a few values for a[j]
            current_val = a[j]
            if current_val < 1e-15:
                # Try setting it to small positive values
                candidates = [1e-6, 1e-5, 1e-4, current_val]
            else:
                # Try scaling up and down
                candidates = [current_val * f for f in [0.5, 0.8, 1.2, 1.5, 2.0]]
                candidates.append(0.0)
            
            for new_val in candidates:
                a_trial = a.copy()
                a_trial[j] = max(0.0, new_val)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
        
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    CoordDesc sweep {} ({}s): J={}".format(sweep, round(elapsed, 1), round(J, 6)))
    
    return best_a, best_m


def cyclic_shift_best(a0):
    """Try all cyclic shifts and return the best one."""
    n = len(a0)
    a = normalize(a0)
    best_m = fast_autocorr(a).max()
    best_shift = 0
    best_a = a.copy()
    
    # For efficiency, only try shifts in steps
    step = max(1, n // 200)
    for s in range(0, n, step):
        if s == 0:
            continue
        a_shift = np.roll(a, -s)
        m = fast_autocorr(a_shift).max()
        if m < best_m:
            best_m = m
            best_shift = s
            best_a = a_shift.copy()
    
    # Fine-tune around the best shift
    if best_shift > 0:
        for s in range(max(0, best_shift - 2*step), min(n, best_shift + 2*step)):
            a_shift = np.roll(a, -s)
            m = fast_autocorr(a_shift).max()
            if m < best_m:
                best_m = m
                best_shift = s
                best_a = a_shift.copy()
    
    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=True):
    """Targeted peak reduction: identify peak positions and modify contributing elements."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_autocorr(a)
        peak_idx = c.argmax()
        peak_val = c[peak_idx]
        
        if peak_val < best_m * 1.001:
            # No peak above best
            continue
        
        # Find the pair (i, k-i) that contributes most to c[peak_idx]
        js = max(0, int(peak_idx) - n + 1)
        je = min(n - 1, int(peak_idx))
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[int(peak_idx) - idx]
        
        if contrib.max() < 1e-20:
            continue
        
        src = int(idx[np.argmax(contrib)])
        
        # Find a good destination: a position that doesn't contribute much to any peak
        # Try moving to a random position
        for attempt in range(10):
            dst = int(rng.integers(n))
            if dst == src:
                continue
            frac = float(rng.choice([0.01, 0.02, 0.05, 0.1]))
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m - 1e-14:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                break
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print("    PeakShave ({}s): J={}".format(round(elapsed, 1), round(J, 6)))
    
    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update the best sequence if the candidate is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {} (n={}, t={})".format(label, round(sc, 6), len(a), round(elapsed, 1)))
    return best_seq, best_score
```

```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.
Strategy: FISTA with adaptive softmax-smoothed max, targeted peak shaving,
coordinate descent on contributing elements, multi-length exploration,
and cyclic shift optimization.

import numpy as np
import time


def fast_conv(a):
    """Compute self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the global evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_grad(y, beta, top_k=None):
    """Compute gradient of smoothed max of self-convolution.
    
    Uses: f(a) ≈ (1/beta) * log(sum(exp(beta * c[k])))
    Gradient: grad[j] = 2 * sum_k p[k] * a[k-j] (for valid indices)
    where p[k] = exp(beta*(c[k]-c_max)) / sum(exp(beta*(c-c_max)))
    """
    n = len(y)
    c = fast_conv(y)
    
    if top_k is not None and top_k < len(c) and top_k > 0:
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    
    c_shifted = beta * (c_use - c_max)
    c_shifted = np.clip(c_shifted, -400, 400)
    exp_vals = np.exp(c_shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(c_use)
        num_max = (c_use == c_max).sum()
        p[c_use == c_max] = 1.0 / max(1, num_max)
    else:
        p = exp_vals / p_total
    
    # grad[j] = 2 * sum_{l=0}^{n-1} a[l] * p[j+l]
    # This is cross-correlation, computed via FFT
    N = 1 << (3 * n).bit_length()
    A_fft = np.fft.rfft(y, N)
    P_fft = np.fft.rfft(p, N)
    h = np.fft.irfft(A_fft * np.conj(P_fft), N)[:n]
    
    return 2.0 * h


def fista_optimize(a0, time_limit, beta_start=20.0, beta_end=1000.0,
                   beta_growth=1.004, top_k=20, restart_every=3000,
                   verbose=True, seed=42):
    """FISTA with Nesterov acceleration, BB step size, and restarts."""
    n = len(a0)
    x = normalize(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_conv(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    step = 1.0 / np.sqrt(n)
    y_prev = None
    grad_prev = None
    avg_x = x.copy()
    avg_count = 0.0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        grad = compute_grad(y, beta, top_k)
        
        # Barzilai-Borwein step size
        if y_prev is not None and grad_prev is not None:
            sy = y - y_prev
            gy = grad - grad_prev
            sy_gy = sy.dot(gy)
            gy_gy = gy.dot(gy) + 1e-20
            if sy_gy > 1e-10:
                bb = sy_gy / gy_gy
                step = max(1e-7, min(bb, 5.0 / beta))

        grad_prev = grad.copy()

        x_new = simplex_project(z)
        
        b_new = fast_conv(x_new)
        max_new = b_new.max()

        avg_x = avg_x + (x_new - avg_x) / avg_count

            best_a = x_new.copy()
            stall = 0

            stall += 1

        x = x_new
        t_nest = t_new

            x = x_r

            beta = max(beta * 0.7, 10.0)

            grad_prev = None
            step = 1.0 / np.sqrt(n)
            avg_count = 0.0
            avg_x = x_r.copy()
            
            if verbose:
                elapsed = time.monotonic() - t0

def peak_shaving(a0, time_limit, verbose=True, seed=555):
    """Targeted peak reduction by modifying elements contributing to peaks."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall_count = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall_count < max_stall:
        c = fast_conv(a)
        # Get top peaks
        top_peaks = np.argsort(c)[::-1][:20]
        stall_count += 1
        improved = False
        
        for peak_k in top_peaks[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            pk = int(peak_k)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[pk - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find top contributors
            top_contrib = np.argsort(contrib)[::-1][:5]
            
            for tc in top_contrib:
                src = int(idx[tc])
                # Identify "safe" destinations (not contributing to top peaks)
                in_peak = np.zeros(n, dtype=bool)
                for pk2 in top_peaks[:10]:
                    pk2i = int(pk2)
                    s2 = max(0, pk2i - n + 1)
                    e2 = min(n - 1, pk2i)
                    if s2 <= e2:
                        in_peak[s2:e2 + 1] = True
                
                # Try moving to non-peak positions
                candidates = np.where(~in_peak)[0]
                if len(candidates) < 5:
                    candidates = np.arange(n)
                
                for _ in range(3):
                    dst = int(candidates[rng.integers(len(candidates))])
                    if dst == src:
                        continue
                    frac = float(rng.choice([0.01, 0.02, 0.03, 0.05, 0.08]))
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] = a[src] - amt
                    a_trial[dst] = a_trial[dst] + amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_conv(a_trial).max()
                    if m_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
        
        if improved:
            stall_count = 0
        
        if verbose and stall_count % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakShave ({}s): J={} stall={}".format(round(elapsed, 1), round(J, 6), stall_count))
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print("    PeakShave done ({}s): J={}".format(round(elapsed, 1), round(J, 6)))
    
    return best_a, best_m

def cyclic_shift_best(a0):
    """Try cyclic shifts and return the best one."""
    n = len(a0)
    a = normalize(a0)
    best_m = fast_conv(a).max()
    best_a = a.copy()
    best_shift = 0
    
    # Coarse search
    step = max(1, n // 300)
    for s in range(step, n, step):
        a_shift = np.roll(a, -s)
        m = fast_conv(a_shift).max()
        if m < best_m:
            best_m = m
            best_shift = s
            best_a = a_shift.copy()
    
    # Fine search around best
    lo = max(0, best_shift - 3 * step)
    hi = min(n - 1, best_shift + 3 * step)
    for s in range(lo, hi + 1):
        if s == 0:
            a_shift = a.copy()
        else:
            a_shift = np.roll(a, -s)
        m = fast_conv(a_shift).max()
        if m < best_m:
            best_m = m
            best_shift = s
            best_a = a_shift.copy()
    
    return best_a, best_m


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum() if f.sum() > 1e-15 else np.ones(old_n) / old_n
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / s

    s = a.sum()
    if s < 0.01:

        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {:.6f} (n={}, t={:.1f}s)".format(label, sc, len(a), elapsed))
    return best_seq, best_score

        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista2", t_global)
    
    # === Phase 4: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 4: Peak shaving ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 30:
        a_out, _ = peak_shaving(best_seq, t4, verbose=True, seed=3)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak1", t_global)
    
    # === Phase 5: Scale up ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    new_n = min(n_best + 3000, 14000)
    if remaining > 250 and new_n > n_best:
        print("\n=== Phase 5: Scale to n={} ===".format(new_n))
        a_long = interpolate_seq(best_seq, new_n)
        
        t5 = min(remaining * 0.25, 300.0)
        if t5 > 60:
            a_out, _ = fista_optimize(a_long, t5,
                                       beta_start=15.0, beta_end=500.0,
                                       beta_growth=1.003, top_k=15,
                                       restart_every=2500, verbose=True, seed=4)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "scale", t_global)
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            a_shifted, _ = cyclic_shift_best(best_seq)
            best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic2", t_global)
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t5b = min(remaining * 0.3, 150.0)
            if t5b > 30:
                a_out, _ = peak_shaving(best_seq, t5b, verbose=True, seed=5)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "peak2", t_global)
    
    # === Phase 6: Final FISTA polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 6: Final FISA polish ===")
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 40:
        a_out, _ = fista_optimize(best_seq, t6,
                                   beta_start=100.0, beta_end=2500.0,
                                   beta_growth=1.01, top_k=8,
                                   restart_every=1500, verbose=True, seed=6)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "final_fista", t_global)
    
    # === Phase 7: Local refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.15, 120.0)
    if t7 > 20:
        a_out, _ = local_refine(best_seq, t7, verbose=True, seed=7)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "local", t_global)
    
    # === Final: Cyclic shift + evaluation ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== Final: Cyclic shift ===")
        a_shifted, _ = cyclic_shift_best(best_seq)
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_final", t_global)
    
    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.6f}, n={}, time={:.1f}s".format(best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: Fallback")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
    
    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
```
