    rng = np.random.default_rng(42)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        b = fast_autocorr(a)
        m_curr = b.max()
        if m_curr < best_m - 1e-14:
            best_m = m_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Top 15 peaks
        top_peaks = np.argsort(b)[::-1][:15]

        # Compute influence map via FFT
        # influence[j] = sum_k p_k * a_{k-j} where p_k are weights on top peaks
        peak_weights = np.zeros(2 * n - 1)
        for k in top_peaks[:10]:
            peak_weights[int(k)] = 1.0

        N = 4 * n
        A_f = np.fft.rfft(a, N)
        W_f = np.fft.rfft(peak_weights[::-1], N)
        conv = np.fft.irfft(A_f * W_f, N)
        influence = conv[n - 1:2 * n - 1][::-1]

        # Find the most contributing pair for the top peak
        k_star = int(top_peaks[0])
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[k_star - idx]
        max_contrib = contrib.max()

        if max_contrib < 1e-20:
            stall += 10
            continue

        src = int(idx[np.argmax(contrib)])
        partner = k_star - src
        if partner < 0 or partner >= n:
            partner = -1

        # Destination: minimize influence, avoid src and partner
        inf_mask = influence.copy()
        inf_mask[src] = np.max(influence) + 10.0
        if 0 <= partner < n:
            inf_mask[partner] = np.max(influence) + 5.0

        # Also penalize positions heavily in top peaks
        for k in top_peaks[:8]:
            k_i = int(k)
            for j in range(max(0, k_i - n + 1), min(n, k_i + 1)):
                if a[j] * a[k_i - j] > 0.3 * m_curr:
                    inf_mask[j] = np.max(influence) + 3.0

        dst = int(np.argmin(inf_mask))

        # Try moving mass
        improved = False
        for frac in [0.02, 0.05, 0.08, 0.12, 0.2, 0.35]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m - 1e-14:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            # Try several random destinations
            for _ in range(5):
                dst_r = int(rng.integers(n))
                if dst_r == src or a[src] < 1e-14:
                    continue
                frac = rng.uniform(0.01, 0.1)
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst_r] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / max(a_trial.sum(), 1e-15)
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Periodic perturbation
            if stall % 80 == 0:
                noise = 0.02 * min(1.0, stall / 200.0)
                pert = rng.normal(0, noise, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-10)
                a_r = a_r / a_r.sum()
                m_r = fast_autocorr(a_r).max()
                if m_r < best_m:
                    a = a_r
                    best_m = m_r
                    best_a = a_r.copy()
                    stall = 0
                else:
                    a = a_r if m_r < best_m + 0.02 else best_a.copy()

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave ({elapsed:.1f}s): max_b={best_m:.10f}, stall={stall}")

    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent: optimize each coordinate with local search."""
    n = len(a0)
    a = norm_nonneg(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 2000:
        b = fast_autocorr(a)
        m_curr = b.max()
        if m_curr < best_m - 1e-14:
            best_m = m_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Pick a coordinate to optimize
        # Priority: coordinates with high values that contribute to peaks
        top_peaks = np.argsort(b)[::-1][:10]
        contrib_score = np.zeros(n)
        for k in top_peaks[:5]:
            k_i = int(k)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            idx = np.arange(i_start, i_end + 1)
            contrib_score[idx] += a[idx] * a[k_i - idx]

        # Pick top-scoring coordinate with significant mass
        candidates = np.where(contrib_score > 0.1 * contrib_score.max())[0]
        if len(candidates) == 0:
            candidates = np.where(a > a.mean())[0]
        if len(candidates) == 0:
            candidates = np.arange(n)

        j = int(candidates[rng.integers(len(candidates))])
        if a[j] < 1e-12:
            continue

        # Try different values for a[j]
        # Current: a[j]. Try reducing and redistributing.
        a_orig = a[j]
        best_trial_m = best_m
        best_trial_a = None

        for frac in [0.1, 0.25, 0.4, 0.6, 0.8]:
            reduce_amt = a_orig * frac
            if reduce_amt < 1e-15:
                continue
            # Distribute to position with lowest influence
            # Quick influence: sum over top peaks of a[k-j]
            a_trial = a.copy()
            a_trial[j] -= reduce_amt
            # Put mass at position with min contribution to peaks
            min_score = np.inf
            dst = 0
            for j2 in range(0, n, max(1, n // 200)):
                score = 0.0
                for k in top_peaks[:5]:
                    k_i = int(k)
                    p2 = k_i - j2
                    if 0 <= p2 < n:
                        score += a_trial[p2]
                if score < min_score:
                    min_score = score
                    dst = j2
            a_trial[dst] += reduce_amt
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / max(a_trial.sum(), 1e-15)
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_trial_a = a_trial.copy()

        if best_trial_a is not None and best_trial_m < best_m:
            a = best_trial_a
            best_m = best_trial_m
            best_a = a.copy()
            stall = 0
        else:
            # Also try increasing a[j] (taking from others)
            src_candidates = np.where(a > a.mean())[0]
            src_candidates = src_candidates[src_candidates != j]
            if len(src_candidates) > 0:
                src = int(src_candidates[rng.integers(len(src_candidates))])
                for frac in [0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / max(a_trial.sum(), 1e-15)
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        stall = 0
                        break

    return best_a, best_m


def multiplicative_descent(a0, time_limit, beta=50.0, verbose=True):
    """Multiplicative updates (EM-style) for non-negativity-preserving optimization."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    rng = np.random.default_rng(555)

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        grad, b, b_max, p = compute_smoothed_grad(a, beta, top_k=100)

        # Multiplicative update: a_i *= exp(-eta * grad_i)
        eta = 1.0 / (beta + 1.0)
        eta = min(eta, 0.1)
        log_mult = np.clip(-eta * grad, -2.0, 2.0)
        mult = np.exp(log_mult)
        a_new = a * mult
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            a = a_new
        else:
            stall += 1
            # Backtracking: halve eta
            if stall % 10 == 0:
                eta *= 0.5
            a = a_new if m_new < best_m + 0.01 else best_a.copy()

        if stall > 500:
            noise = 0.01 * min(1.0, (stall - 500) / 500.0)
            pert = rng.normal(0, noise, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            stall = 0

        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MultDesc iter={iter_count} ({elapsed:.1f}s): max_b={best_m:.10f}")

    return best_a, best_m


def try_length(a0, new_n, time_left, label):
    """Interpolate to new length and do quick optimization."""
    old_n = len(a0)
    if old_n == new_n:
        a = a0.copy()
    else:
        s = a0.sum()
        if s < 1e-15:
            a = np.ones(new_n) / new_n
        else:
            a_norm = a0 / s
            old_x = np.linspace(0.0, 1.0, old_n)
            new_x = np.linspace(0.0, 1.0, new_n)
            a = np.interp(new_x, old_x, a_norm)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    # Quick optimization
    a_out, _ = smart_peak_shave(a, min(time_left * 0.4, 60.0), verbose=False)
    a_out, _ = accelerated_descent(a_out, min(time_left * 0.5, 80.0), beta_start=30.0, beta_end=300.0, verbose=False)
    return a_out


def propose_candidate(budget_s=1800.0):
    """Main search: multi-phase optimization combining gradient, peak shaving, and coordinate descent."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.1f}s)")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            try_update(parent, "parent")
            print(f"Parent: n={len(parent)}, score={best_score:.6f}")
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        x = np.linspace(0, 1, n_base)
        a_raw = (x + 0.01) * (1.0 - x + 0.01)
        best_seq = a_raw / a_raw.sum()
        try_update(best_seq, "default_init")

    n0 = len(best_seq)

    # Phase 1: Quick parametric improvements
    print("\n=== Phase 1: Quick parametric sweep ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.02, 30.0)
    if t1 > 10:
        x = np.linspace(0, 1, n0)
        candidates = [
            (x + 0.01) ** 0.5 * (1.0 - x + 0.01) ** 1.5,
            (x + 0.01) ** 1.5 * (1.0 - x + 0.01) ** 0.5,
            np.exp(-((x - 0.3) ** 2) / 0.05) + np.exp(-((x - 0.7) ** 2) / 0.05),
            np.exp(-((x - 0.35) ** 2) / 0.03) + 0.8 * np.exp(-((x - 0.65) ** 2) / 0.03),
            1.0 + 0.5 * np.sin(2 * np.pi * x),
            1.0 + 0.3 * np.cos(4 * np.pi * x),
        ]
        for c in candidates:
            c = np.maximum(c, 0.0)
            cs = c.sum()
            if cs > 0:
                try_update(c / cs, f"param_n{n0}")

    # Phase 2: Smart peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Smart peak shaving (n={n0}) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_out, _ = smart_peak_shave(best_seq, t2, verbose=True)
        try_update(a_out, f"ps1_n{n0}")

    # Phase 3: Accelerated gradient descent
    remaining = time_budget - (time.monotonic() - t_global)
    nm = len(best_seq)
    print(f"\n=== Phase 3: Accelerated descent (n={nm}) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 50:
        a_out, _ = accelerated_descent(best_seq, t3, beta_start=15.0, beta_end=800.0,
                                       beta_growth=1.012, verbose=True)
        try_update(a_out, f"accel_n{nm}")

    # Phase 4: Peak shaving round 2
    remaining = time_budget - (time.monotonic() - t_global)
    nm = len(best_seq)
    print(f"\n=== Phase 4: Peak shaving 2 (n={nm}) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 20:
        a_out, _ = smart_peak_shave(best_seq, t4, verbose=True)
        try_update(a_out, f"ps2_n{nm}")

    # Phase 5: Multiplicative descent
    remaining = time_budget - (time.monotonic() - t_global)
    nm = len(best_seq)
    print(f"\n=== Phase 5: Multiplicative descent (n={nm}) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20:
        a_out, _ = multiplicative_descent(best_seq, t5, beta=80.0, verbose=True)
        try_update(a_out, f"mult_n{nm}")

    # Phase 6: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    nm = len(best_seq)
    print(f"\n=== Phase 6: Coordinate descent (n={nm}) ===")
    t6 = min(remaining * 0.12, 180.0)
    if t6 > 20:
        a_out, _ = coordinate_descent(best_seq, t6, verbose=True)
        try_update(a_out, f"coord_n{nm}")

    # Phase 7: Second accelerated pass with higher beta
    remaining = time_budget - (time.monotonic() - t_global)
    nm = len(best_seq)
    if remaining > 100:
        print(f"\n=== Phase 7: High-beta descent (n={nm}) ===")
        t7 = min(remaining * 0.2, 200.0)
        if t7 > 30:
            a_out, _ = accelerated_descent(best_seq, t7, beta_start=100.0, beta_end=2000.0,
                                           beta_growth=1.005, verbose=True)
            try_update(a_out, f"highbeta_n{nm}")

    # Phase 8: Try nearby lengths
    remaining = time_budget - (time.monotonic() - t_global)
    nm = len(best_seq)
    if remaining > 200:
        print(f"\n=== Phase 8: Multi-length exploration ===")
        nearby = []
        for dn in [200, -200, 500, -500, 800]:
            nl = nm + dn
            if 5000 < nl <= 8500 and nl not in nearby:
                nearby.append(nl)
        for new_n in nearby[:3]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 120:
                break
            t_tl = min(remaining * 0.3, 150.0)
            a_long = try_length(best_seq, new_n, t_tl, f"n{new_n}")
            try_update(a_long, f"len_{new_n}")

    # Phase 9: Final combined polish
    remaining = time_budget - (time.monotonic() - t_global)
    nm = len(best_seq)
    if remaining > 80:
        print(f"\n=== Phase 9: Final polish (n={nm}) ===")
        t9a = min(remaining * 0.3, 100.0)
        if t9a > 15:
            a_out, _ = smart_peak_shave(best_seq, t9a, verbose=False)
            try_update(a_out, "final_ps")

        remaining = time_budget - (time.monotonic() - t_global)
        t9b = min(remaining * 0.4, 120.0)
        if t9b > 20:
            a_out, _ = accelerated_descent(best_seq, t9b, beta_start=200.0, beta_end=5000.0,
                                           beta_growth=1.003, verbose=True)
            try_update(a_out, "final_accel")

        remaining = time_budget - (time.monotonic() - t_global)
        t9c = min(remaining * 0.3, 80.0)
        if t9c > 10:
            a_out, _ = smart_peak_shave(best_seq, t9c, verbose=False)
            try_update(a_out, "final_ps2")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        a_raw = (x + 0.01) * (1.0 - x + 0.01)
        best_seq = a_raw / a_raw.sum()
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
