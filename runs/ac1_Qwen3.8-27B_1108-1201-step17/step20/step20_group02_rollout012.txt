            coeffs[jj, idx] = y[k_int - idx]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Improved LP-Frank-Wolfe with adaptive active set and dual-guided constraint management
2. Smart multi-n exploration with interpolation
3. Peak-aware coordinate descent for final polish
4. Spectral initialization for diverse starting points
5. Aggressive line search with adaptive step sizes
"""

import numpy as np
import time
import cvxpy as cp

_module_solver = None


def get_solver():
    global _module_solver
    if _module_solver is not None:
        return _module_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _module_solver = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _module_solver = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _module_solver


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_coeff_matrix(y, active_set, n):
    K = len(active_set)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(active_set[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=4000, verbose=True):
    """Improved LP-Frank-Wolfe with better active set management."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac_y = fast_convolve(y)
    best_m = ac_y.max()
    t0 = time.monotonic()

    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())
    no_improve = 0
    iter_count = 0
    stagnation_count = 0
    best_alpha_history = []

    while time.monotonic() - t0 < time_limit:
        if no_improve > 400:
            stagnation_count += 1
            if stagnation_count > 6:
                if verbose:
                    print("  LP-FW: stopping (persistent stagnation)")
                break
            # Perturb active set
            ac_curr = fast_convolve(y)
            n_keep = max(50, len(active_set) - len(active_set) // 4)
            rng = np.random.default_rng(stagnation_count * 1000)
            keep = list(rng.choice(len(active_set), size=n_keep, replace=False))
            new_adds = np.argsort(ac_curr)[::-1][:len(active_set) // 4 + 50]
            new_set = set([active_set[i] for i in keep])
            for k in new_adds:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]
            no_improve = 0
            continue

        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0

        # Manage active set: prune non-binding, keep peaks
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep_count = int(max_K * 0.75)
            keep_idx = np.argsort(margins)[:keep_count]
            active_set_base = [active_set[i] for i in keep_idx]
            top_peaks = np.argsort(ac)[::-1][:min(200, max_K // 10)]
            for kp in top_peaks:
                kp_i = int(kp)
                if kp_i not in active_set_base:
                    active_set_base.append(kp_i)
            active_set = sorted(set(active_set_base))
        elif len(active_set) < 50:
            top_add = np.argsort(ac)[::-1][:min(200, max_K - len(active_set))]
            for k in top_add:
                active_set.append(int(k))
            active_set = sorted(set(active_set))

        K = len(active_set)
        if K < 20:
            continue

        coeffs = build_coeff_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(50, K // 3)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check for violated constraints and add them
        fft_size = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:80]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        # Adaptive line search
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_M = M_curr

        # Generate candidate alphas
        recent_mean = np.mean(best_alpha_history[-20:]) if best_alpha_history else 0.3
        alphas = sorted(set([
            1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05,
            best_alpha, max(0.01, recent_mean * 0.5), min(0.9, recent_mean * 2.0)
        ]), reverse=True)

        for alpha in alphas:
            if alpha <= 0 or alpha > 1.0:
                continue
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        best_alpha_history.append(best_alpha)
        if len(best_alpha_history) > 100:
            best_alpha_history = best_alpha_history[-100:]

        # Update
        if iter_count <= 3:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        if verbose and iter_count % 80 == 0:
            elapsed = time.monotonic() - t0
            print("  LP-FW iter={} ({}s): J={}, |S={} alpha={}".format(
                iter_count, round(elapsed, 1), repr(2.0 * n * best_m), len(active_set), round(best_alpha, 3)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("  LP-FW done ({}s): J={}, iters={}".format(
            round(elapsed, 1), repr(2.0 * n * best_m), iter_count))
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=77, verbose=False):
    """Targeted coordinate descent: transfer mass from peak-contributing positions to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(50, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score all positions by their contribution to top peaks
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            # Find lowest-score destination
            candidates = np.where(a < best_m * n * 0.5)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])

            if dst == src:
                continue

            # Try multiple transfer sizes
            found = False
            for frac in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found:
                # Try random small transfer
                for _ in range(3):
                    j = int(rng.integers(n))
                    if j != src and a[src] > 1e-10:
                        frac = rng.uniform(0.002, 0.01)
                        amt = a[src] * frac
                        a[src] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0:
                # Random perturbation to escape local optima
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt

                        s_t = a.sum()
                        if s_t > 1e-15:

        if verbose and stall % 400 == 0:
            elapsed = time.monotonic() - t0

    return best_a, best_m


def spectral_init(n, method='sine'):
    """Generate initial sequences using spectral methods."""
    x = np.linspace(0.0, 1.0, n)
    if method == 'sine':
        a = np.sin(np.pi * x) ** 2
    elif method == 'raised_cosine':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif method == 'hamming':
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * x)
    elif method == 'hann':
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * x)
    elif method == 'tukey':
        alpha = 0.5
        a = np.ones(n)
        m = int(alpha * n / 2)
        for i in range(m):
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * n) - 1)))
            a[n - 1 - i] = a[i]
    else:
        a = np.sin(np.pi * x) ** 2
    return norm_sum1(a)


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [{}]: {} (n={}, t={}s)".format(label, str(sc), len(a), round(elapsed, 1)))
    return best_seq, best_score
