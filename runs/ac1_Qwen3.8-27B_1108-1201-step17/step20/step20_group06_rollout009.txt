    alpha = 0.1  # relaxation parameter
    
    for it in range(1, 1000000):
        if time expired: break

        power = np.abs(H)**2

        ac = fast_autocorr(a)
        peak_lag = int(np.argmax(ac))

        phases = np.exp(2j * np.pi * m * peak_lag / fft_size)
        alignment = np.real(phases)  # positive = aligned with peak, negative = anti-aligned
        
        # Reduce power at aligned frequencies, keep anti-aligned
        for k in range(len(H)):
            if alignment[k] > 0:
                factor = 1 - alpha * alignment[k]

            a = a_new

            # Try smaller step
            alpha *= 0.9
        
        # Slowly decrease alpha
        if it % 100 == 0:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    z = np.asarray(z, dtype=np.float64).copy()
    n = len(z)
    if n == 0:
        return z
    z = np.maximum(z, 0.0)
    s = z.sum()
    if s < 1e-15:
        return np.ones(n) / n
    # Proper simplex projection (sum=1, nonneg)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)


def evaluate_safe(a):
    try:

        return float(evaluate_sequence(a_list))
    except Exception:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_gradient_fft(a, beta, top_k=None):
    """Compute gradient of softmax-smoothed max of conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    b = fast_autocorr(a)
    b_max = b.max()

    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max_use = b[idx].max()
    else:
        b_use = b
        b_max_use = b_max

    shifted = beta * (b_use - b_max_use)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s_exp = exp_vals.sum()
    if s_exp < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s_exp

    # grad[m] = 2 * sum_k p[k] * a[k-m] for valid k-m
    # This is 2 * cross-correlation of p with reversed a
    a_rev = a[::-1]
    conv_len = n + size - 1
    fft_size2 = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size2)
    Pr = np.fft.rfft(p, fft_size2)
    conv = np.fft.irfft(Ar * Pr, fft_size2)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, b


def gs_spectral_v2(a0, time_limit, verbose=True, seed=777):

        # Also consider second peak for robustness
        ac_sorted = np.sort(ac)[::-1]
        if len(ac_sorted) > 1 and ac_sorted[1] > 0.8 * ac_sorted[0]:
            peak_lag2 = int(np.argmax(np.delete(ac, peak_lag)))
            cos_phases2 = np.cos(2.0 * np.pi * m_arr * peak_lag2 / fft_size)

        # Reduce where aligned, slightly boost where anti-aligned

        mask_neg = cos_phases < -0.05
        new_mag[1:][mask_pos[1:]] *= (1.0 - alpha * cos_phases[1:][mask_pos[1:]])
        new_mag[1:][mask_neg[1:]] *= (1.0 + alpha * 0.4 * (-cos_phases[1:][mask_neg[1:]]))

        power = new_mag * new_mag
        p_mean = power[1:].mean() if len(power) > 1 else power[0]

                new_mag[k] *= np.sqrt(1.0 + 0.01 * (0.3 - ratio))

        H[1:][valid[1:]] = H[1:][valid[1:]] / (np.abs(H[1:][valid[1:]]) + 1e-30) * new_mag[1:][valid[1:]]
        H[0] = complex(H[0].real, 0)

        if s_new < 1e-15:

            continue
        a_new = a_new / s_new

        if M_new < best_M - 1e-16:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            alpha = min(alpha * 1.01, 0.15)
        else:
            stall += 1
            if stall > 50:

                            best_a = a_trial.copy()
                            a = a_trial
                    stall = 0

                x_r = np.maximum(x_r, 0)
            else:
                # Mirror blend (break symmetry)
                mirror = best_a[::-1]
                mix = rng.uniform(0.3, 0.7)
                x_r = mix * best_a + (1 - mix) * mirror

            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 15.0)
            eta = eta_init
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest4 restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest4 t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)) + " beta=" + str(int(beta)))

    return best_a, best_max_b


def lp_fw_v3(a0, time_limit, solver, initial_K=500, max_K=4000,
             add_per_iter=30, verbose=True):
    """LP-Frank-Wolfe with large active sets."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:min(initial_K, 2 * n - 1)])

    no_improve = 0
    max_no_improve = 800

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations for active set update
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        for alpha in [best_alpha, best_alpha * 0.5, best_alpha * 2.0,
                      1.0 / (t + 1), 0.05, 0.1, 0.15, 0.25]:
            alpha = max(0.01, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW3 it=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def peak_target_search(a0, time_limit, verbose=True, seed=321):
    """Target the top peaks directly with asymmetric mass moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 3000:
        ac = fast_autocorr(a)
        size = 2 * n - 1
        top_k = np.argsort(ac)[::-1][:min(30, size)]
        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-15:
                continue
            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            if not (0 <= partner < n):
                continue

            # Try many fraction/destination combos
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                best_dest_M = best_m
                best_dest = -1
                for trial_i in range(8):
                    # Asymmetric: prefer moving toward the sparse end
                    if a[0] < a[-1]:
                        dst = int(rng.integers(0, n // 2))
                    else:
                        dst = int(rng.integers(n // 2, n))
                    if dst == src or dst == partner:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_dest_M:
                        best_dest_M = M_trial
                        best_dest = dst
                        best_trial = a_trial.copy()

                if best_dest >= 0 and best_dest_M < best_m - 1e-17:
                    a = best_trial
                    best_m = best_dest_M
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 150:
                # Larger random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakTarget (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score
