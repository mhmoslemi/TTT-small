import numpy as np

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def logsumexp_smoothed_grad(y, beta, top_k=None):
    """Compute smoothed gradient of the log-sum-exp of autocorrelation."""
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and top_k < len(b) and top_k > 0:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b_use)
        mask = (b_max == b_use)
        p[mask] = 1.0 / max(1, mask.sum())
    else:
        p = exp_vals / p_total
    
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, p, b


def fista_adaptive(a0, time_limit, beta_start=10.0, beta_end=5000.0,

    avg_count = 0.0

    max_iter = 50000000

    for t in range(1, max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:

            y = y / s_y

            gy = grad - s_prev

            stall = 0

            x = x_r

    """Polish using exact subgradients of the max convex function."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    b = fast_autocorr(a)
    best_m = b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    step = 0.5 / n

    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if it % 100 == 0 and step < 0.01 / n:
            break
        if it % 200 == 0:
            step *= 0.995

        b = fast_autocorr(a)
        b_max = b.max()
        argmax_idx = np.where(np.abs(b - b_max) < 1e-15)[0]

        if len(argmax_idx) == 0:
            argmax_idx = [int(np.argmax(b))]

        # Compute subgradient for a random argmax
        k_int = int(argmax_idx[rng.integers(len(argmax_idx))])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        sg = np.zeros(n)
        sg[idx] = a[k_int - idx]

        # Also add a small component from all peaks
        top_peaks = np.argsort(b)[::-1][:min(20, len(b))]
        sg_top = np.zeros(n)
        for k2 in top_peaks[:10]:
            k2 = int(k2)
            s2 = max(0, k2 - n + 1)
            e2 = min(n, k2 + 1)
            idx2 = np.arange(s2, e2)
            sg_top[idx2] += a[k2 - idx2]
        sg_total = 0.3 * sg + 0.7 * sg_top / max(1, len(top_peaks[:10]))

        a_trial = a - step * sg_total
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            break
        a_trial = a_trial / s_t
        m_trial = fast_autocorr(a_trial).max()

        if m_trial < best_m - 1e-16:
            best_m = m_trial
            best_a = a_trial.copy()
            a = a_trial
        else:
            # Try smaller step
            a_trial2 = a - 0.3 * step * sg_total
            a_trial2 = np.maximum(a_trial2, 0.0)
            s_t2 = a_trial2.sum()
            if s_t2 > 1e-15:
                a_trial2 = a_trial2 / s_t2
                m2 = fast_autocorr(a_trial2).max()
                if m2 < best_m - 1e-16:
                    best_m = m2
                    best_a = a_trial2.copy()
                    a = a_trial2
                    step *= 0.9

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SubGrad it=" + str(it) + " (" + str(round(elapsed, 1)) +
                  "s): J=" + repr(round(J, 6)) + " step=" + repr(step))

    return best_a, best_m


def coordinate_refine(a0, time_limit, verbose=True, seed=55):
    """Coordinate-wise optimization with line search."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 500

    # Precompute the convolution matrix structure
    # C[k] = sum_i a[i] * a[k-i]
    # dC[k]/da[j] = a[k-j] (when valid) + a[j-k] (when valid, for j != k-j case)
    # Actually dC[k]/da[j] = a[k-j] if 0<=k-j<n else 0, plus a[j-k] if 0<=j-k<n else 0
    # But since C[k] = sum_i a[i]*a[k-i], dC[k]/da[j] = a[k-j] + a[j-k] (counting both terms)
    # Wait: C[k] = sum_i a[i]*a[k-i]. If we differentiate w.r.t. a[j]:
    #   term i=j: a[k-j]
    #   term i=k-j: a[j] (if j != k-j, i.e., k != 2j)
    # So dC[k]/da[j] = a[k-j] + a[j-k] if j != k-j, else 2*a[k-j]
    # But actually in the sum, when i=j, we get a[j]*a[k-j]. When i=k-j, we get a[k-j]*a[j].
    # These are the same term if k-j = j, i.e., k=2j. Otherwise they're different.
    # In the sum over i, each (i, k-i) pair appears once. So dC[k]/da[j] = a[k-j] if 0<=k-j<n.
    # (The other "copy" a[j-k] would correspond to a different k value.)

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        b = fast_autocorr(a)
        b_max = b.max()
        top_k_vals = np.argsort(b)[::-1][:min(50, len(b))]
        
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            if no_improve >= max_no_improve:
                break
            
            # Pick a coordinate to update
            k_peak = int(top_k_vals[rng.integers(len(top_k_vals))])
            i_start = max(0, k_peak - n + 1)
            i_end = min(n, k_peak + 1)
            js = np.arange(i_start, i_end)
            contrib = a[js] * a[k_peak - js]
            
            if contrib.max() < 1e-20:
                continue
            j_idx = int(js[np.argmax(contrib)])
            
            # Try moving mass from high-contributing position to low-contributing
            # Find a target position
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k_vals[:20]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                in_peak[s2:e2] = True
            
            if not in_peak[j_idx]:
                continue
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            
            # Try several target positions and fractions
            best_trial_m = best_m
            best_trial_a = None
            best_frac = 0
            
            for _samp in range(8):
                dst = int(candidates[rng.integers(len(candidates))])
                if dst == j_idx:
                    continue
                frac = float(rng.choice([0.005, 0.01, 0.02, 0.05, 0.1, 0.2]))
                amt = a[j_idx] * frac
                a_trial = a.copy()
                a_trial[j_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_trial_m:
                    best_trial_m = m_trial
                    best_trial_a = a_trial
                    best_frac = frac
            
            if best_trial_a is not None:
                a = best_trial_a
                best_m = best_trial_m
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if verbose and no_improve % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    CoordRefine (" + str(round(elapsed, 1)) +
                  "s): J=" + repr(round(J, 6)) + " no_improve=" + str(no_improve))
        
        # Random perturbation to escape
        if no_improve > 100:
            idx = rng.integers(n)
            if a[idx] > 1e-8:
                amt = a[idx] * 0.05
                a[idx] -= amt
                dst = int(rng.integers(n))
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                no_improve = 0

    return best_a, best_m


def spectral_optimize(a0, time_limit, verbose=True):
    """Optimize in the frequency domain to flatten the spectrum."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        # The autocorrelation peak is related to the L1 norm of the spectrum
        # by max R >= (1/fft_size) * sum |A|^2 = (1/fft_size) * ||A||_2^2
        # To minimize peak, we want to spread the spectrum more uniformly
        
        b = np.fft.irfft(power, fft_size)[:2 * n - 1]
        b_max = b.max()
        
        # Attenuate frequencies that contribute to the peak
        # The peak at lag k gets contribution from all frequencies
        # Approximation: attenuate based on |A|^2 itself (Wiener-Khinchin)
        attenuation = 1.0 - 0.1 * (power / power.sum())
        attenuation = np.maximum(attenuation, 0.1)
        
        A_new = A * attenuation
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        m_new = fast_autocorr(a_new).max()
        
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller attenuation
            attenuation2 = 1.0 - 0.03 * (power / power.sum())
            attenuation2 = np.maximum(attenuation2, 0.5)
            A_new2 = A * attenuation2
            a_new2 = np.fft.irfft(A_new2, fft_size)[:n]
            a_new2 = np.maximum(a_new2, 0.0)
            s_new2 = a_new2.sum()
            if s_new2 > 1e-15:
                a_new2 = a_new2 / s_new2
                m_new2 = fast_autocorr(a_new2).max()
                if m_new2 < best_m:
                    best_m = m_new2
                    best_a = a_new2.copy()
                    a = a_new2
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral it=" + str(it) + " (" + str(round(elapsed, 1)) +
                  "s): J=" + repr(round(J, 6)))
    
    return best_a, best_m


def cyclic_shift_best(a0):
    """Find the best cyclic shift."""
    n = len(a0)
    b0 = fast_autocorr(a0)
    best_m = b0.max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    
    # Use the fact that convolution of a shifted sequence can be computed efficiently
    # For small n, just try all shifts
    if n <= 20000:
        for s in range(1, n):
            a_shift = np.roll(a0, -s)
            b = fast_autocorr(a_shift)
            m = b.max()
            if m < best_m:
                best_m = m
                best_shift = s
    else:
        # Try random shifts
        rng = np.random.default_rng(31337)
        for _ in range(200):
            s = int(rng.integers(1, n))
            a_shift = np.roll(a0, -s)
            b = fast_autocorr(a_shift)
            m = b.max()
            if m < best_m:
                best_m = m
                best_shift = s
    
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + repr(sc) +
              " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def fista_long(a0, time_limit, beta_start=500.0, beta_end=20000.0,
               beta_growth=1.005, restart_every=2000, verbose=True, seed=777):
    """Extended FISTA with very high beta for fine polishing."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    b = fast_autocorr(x)
    best_max_b = b.max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    bb_step = 1.0 / (n ** 0.7)
    max_iter = 10000000

    for t in range(1, max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, b_curr = logsumexp_smoothed_grad(y, beta, top_k=None)

        if t > 1:
            eta = min(bb_step, 1.0 / max(beta, 1.0))
        else:
            eta = 1.0 / max(beta, 1.0)
        
        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        if max_new < best_max_b - 1e-17:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 100 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.005 / (1.0 + 0.01 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.8, 100.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    FISTA-L restart t=" + str(t) + " (" + str(round(elapsed, 1)) +
                      "s): J=" + repr(round(J, 6)))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    FISTA-L t=" + str(t) + " (" + str(round(elapsed, 1)) +
                  "s): J=" + repr(round(J, 6)) + " beta=" + repr(round(beta, 0)))

    return best_a, best_max_b


def population_search(a0, time_limit, pop_size=10, verbose=True, seed=2024):
    """Population-based search with guided mutation."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    
    # Initialize population
    population = [norm_sum1(a0).copy()]
    for i in range(1, pop_size):
        pert = rng.normal(0, 0.02, n) * a0
        cand = a0 + pert
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()
        population.append(cand)
    
    # Evaluate initial population
    scores = []
    for p in population:
        b = fast_autocorr(p)
        scores.append(b.max())
    
    best_idx = int(np.argmin(scores))
    best_a = population[best_idx].copy()
    best_m = scores[best_idx]
    t0 = time.monotonic()
    
    for gen in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Select parent (tournament)
        i1 = int(rng.integers(pop_size))
        i2 = int(rng.integers(pop_size))
        parent = population[i1] if scores[i1] < scores[i2] else population[i2]
        
        # Mutate: guided by autocorrelation
        b = fast_autocorr(parent)
        b_max = b.max()
        top_peaks = np.argsort(b)[::-1][:min(30, len(b))]
        
        child = parent.copy()
        for _ in range(5 + gen // 10):
            k_peak = int(top_peaks[rng.integers(len(top_peaks))])
            i_start = max(0, k_peak - n + 1)
            i_end = min(n, k_peak + 1)
            js = np.arange(i_start, i_end)
            contrib = child[js] * child[k_peak - js]
            if contrib.max() < 1e-20:
                continue
            src = int(js[np.argmax(contrib)])
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:15]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                in_peak[s2:e2] = True
            if in_peak[src]:
                candidates = np.where(~in_peak)[0]
                if len(candidates) > 0:
                    dst = int(candidates[rng.integers(len(candidates))])
                    frac = float(rng.uniform(0.005, 0.1))
                    amt = child[src] * frac
                    child[src] -= amt
                    child[dst] += amt
        
        child = np.maximum(child, 0.0)
        s_c = child.sum()
        if s_c < 1e-15:
            continue
        child = child / s_c
        m_child = fast_autocorr(child).max()
        
        # Replace worst in population
        worst_idx = int(np.argmax(scores))
        if m_child < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = m_child
        
        if m_child < best_m:
            best_m = m_child
            best_a = child.copy()
        
        if verbose and gen % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PopSearch gen=" + str(gen) + " (" + str(round(elapsed, 1)) +
                  "s): J=" + repr(round(J, 6)))
    
    return best_a, best_m


def propose_candidate(budget_s=1800.0):

    print("Time budget: " + repr(time_budget) + "s")

    try:

            best_seq = norm_sum1(p)
            best_score = evaluate_safe(best_seq)
            print("Parent: n=" + str(len(p)) + ", score=" + repr(best_score))
    except Exception:
        pass

        n_base = 6700

        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(n_base) + ", score=" + repr(best_score))

        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista1", t_global)

        best_seq, best_score = try_update(best_seq, best_score, a_out, "subgrad", t_global)

    # Phase 5: Coordinate refinement
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 5: Coordinate refine ===")

        a_long = interpolate_sequence(best_seq, new_n)
        
        t5a = min(remaining * 0.35, 180.0)
        if t5a > 40:
            a_out, _ = fista_adaptive(a_long, t5a,
                                       beta_start=20.0, beta_end=2000.0,
                                       beta_growth=1.005, top_k=25,
                                       restart_every=2000, verbose=False, seed=111)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_n" + str(new_n), t_global)
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t5b = min(remaining * 0.35, 120.0)
            if t5b > 30:
                a_out, _ = fista_long(norm_sum1(best_seq), t5b,
                                       beta_start=500.0, beta_end=10000.0,
                                       beta_growth=1.005, restart_every=1500,
                                       verbose=False, seed=222)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "fistaL_n" + str(new_n), t_global)

    # Phase 8: Final FISTA polish
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 8: Final FISTA polish (n=" + str(len(best_seq)) + ") ===")
    t8 = min(remaining * 0.20, 200.0)
    if t8 > 30:
        a_out, _ = fista_long(norm_sum1(best_seq), t8,
                               beta_start=1000.0, beta_end=50000.0,
                               beta_growth=1.003, restart_every=1500,
                               verbose=True, seed=333)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista8", t_global)

    # Phase 9: Final coordinate + subgradient
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 9: Final refinement ===")
    if remaining > 30:
        t9a = min(remaining * 0.4, 100.0)
        if t9a > 20:
            a_out, _ = subgradient_polish(norm_sum1(best_seq), t9a, verbose=False, seed=444)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "subgrad_f", t_global)
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t9b = min(remaining * 0.5, 80.0)
            if t9b > 15:
                a_out, _ = coordinate_refine(norm_sum1(best_seq), t9b, verbose=False, seed=555)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_f", t_global)

    # Final cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print("\n=== Final: Cyclic shift ===")
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_f", t_global)

    # Verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) +
              ", n=" + str(len(best_seq)) +
              ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
