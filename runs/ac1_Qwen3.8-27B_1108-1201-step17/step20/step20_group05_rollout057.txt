def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)


def lse_a(a, beta, mask=None, reg_a=0.0):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 0.0, np.array([], dtype=np.float64)
    size = 2 * n - 1
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)

    if mask is None:
        cv = c
    else:
        cv = c[mask]

    if cv.size == 0:
        f = 0.0
        grad = np.zeros(n, dtype=np.float64)
    else:
        m = float(np.max(cv))
        shifted = beta * (cv - m)
        shifted = np.clip(shifted, -700.0, 700.0)
        e = np.exp(shifted)
        se = float(e.sum())
        if se <= 0.0 or not math.isfinite(se):
            e = np.zeros(cv.size, dtype=np.float64)
            e[0] = 1.0
            se = 1.0
        p = e / se

        pf = np.zeros(size, dtype=np.float64)
        if mask is None:
            pf[:] = p
        else:
            pf[mask] = p

        Pr = np.fft.rfft(pf[::-1], N)
        conv = np.fft.irfft(A * Pr, N)[:total]
        grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
        grad = np.where(np.isfinite(grad), grad, 0.0)
        f = m + math.log(se) / beta

    if reg_a > 0.0:
        f += 0.5 * reg_a * float(np.dot(a, a))
        grad = grad + reg_a * a

    if not math.isfinite(f):
        f = 1e100
    return f, grad


def softmax_from_u(u):
    u = np.asarray(u, dtype=np.float64)
    n = u.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(u)):
        return np.ones(n, dtype=np.float64) / float(n)

    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s

def objgrad_v(v, beta, mask=None, reg_a=0.0, reg_u=0.0, ref=0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    if n <= 0:
        return 1e100, np.array([], dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return 1e100, np.zeros(v.size, dtype=np.float64)

    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1

    u = np.empty(n, dtype=np.float64)
    u[ref] = 0.0
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]

    a = softmax_from_u(u)
    f, ga = lse_a(a, beta, mask, reg_a)
    if not math.isfinite(f):
        f = 1e100
        ga = np.zeros(n, dtype=np.float64)

    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)

    gv = np.empty(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]

    if reg_u > 0.0:
        f += 0.5 * reg_u * float(np.dot(v, v))
        gv = gv + reg_u * v

    if not np.all(np.isfinite(gv)):
        gv = np.zeros(v.size, dtype=np.float64)
    return f, gv

def line_search_mix(a, g, cf, cg, fg, current_M, t):
    n = a.size
    base_alphas = [
        max(0.01, 1.0 / (t + 1.0)),
        max(0.02, 2.0 / (t + 2.0)),
        0.04, 0.08, 0.12, 0.18, 0.25, 0.35, 0.50, 0.70, 1.0
    ]
    alphas = []
    seen = set()
    for al in base_alphas:
        al = float(al)
        if not math.isfinite(al) or al <= 0.0 or al > 1.0:
            continue
        key = int(round(al * 10000.0))
        if key in seen:
            continue
        seen.add(key)
        alphas.append(al)
    if len(alphas) == 0:
        alphas = [0.1]

    best_alpha = 0.0
    best_m = current_M
    for alpha in alphas:
        one = 1.0 - alpha
        val = (one * one) * cf + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
        m = float(np.max(val))
        if math.isfinite(m) and m < best_m:
            best_m = m
            best_alpha = alpha

    if best_alpha <= 0.0:
        return a.copy(), best_m, 0.0

    new_a = (1.0 - best_alpha) * a + best_alpha * g
    new_a = np.maximum(new_a, 0.0)
    s = float(new_a.sum())
    if s < 1e-15:
        return a.copy(), best_m, 0.0
    new_a = new_a / s
    return new_a, best_m, best_alpha

def lp_direction_hihts(f, active_idx, time_limit=6.0, max_k=800):

        return None
    if active_idx.size > max_k:
        active_idx = active_idx[:max_k]
    K = active_idx.size

    row_parts = []
    col_parts =
    val_parts = []
    for r, k in enumerate(active_idx):
        s = max(0, int(k) - n + 1)
        e = min(n, int(k) + 1)
        if s >= e:
            continue

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0

    neg_one = scipy.sparse.csc_matrix(-np.ones((K, 1)), dtype=np.float64)
    A_ub = scipy.sparse.hstack((A, neg_one), format="csr")
    A_eq = np.ones((1, n + 1), dtype=np.float64)
    A_eq[0, n] = 0.0
    b_eq = np.ones(1, dtype=np.float64)
    b_ub = np.zeros(K, dtype=np.float64)

    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = scipy.optimize.linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,

            options={"disp": False, "time_limit": float(time_limit)}
        )
    except Exception:

    gs = float(g.sum())

        if time_left < 20:
            break

        g = None
        if check_hihts() and exact_every > 0 and (t % exact_every == 0) and time_left > 80:
            g = lp_direction_hihts(a, active, exact_time, max_k=800)

        if g is None:
            init_log = None
            if active.size > 0:
                av = c[active]
                av = np.where(np.isfinite(av) & (av > 0.0), av, 1e-300)
                init_log = np.log(av)
                init_log -= float(np.max(init_log))
            g = game_direction(a, active, mw_iters=game_iters,
                               seed=2000 + t, init_logits=init_log)

        if g is None:
            break

        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        new_a, new_M, alpha = line_search_mix(a, g, c, cg, fg, M, t)

        if new_M < M - 1e-16:
            if alpha > 0:
                one = 1.0 - alpha
                c = (one * one) * c + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
                c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
            a = new_a
            M = new_M
            no_improve = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            active = active_top(c, None, active_max)
            if verbose and (t == 1 or t % 15 == 0):
                print("    FW t=" + str(t) +
                      " score_proxy=" + repr(2.0 * n * best_M) +
                      " alpha=" + repr(alpha))
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                rng = np.random.default_rng(9973 + t)
                sigma = 0.02 + 0.01 * (no_improve // 20)
                pert = rng.normal(0.0, min(sigma, 0.1), n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-12)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = proxy_max(c, None)
                active = active_top(c, None, active_max)
                no_improve = 0
            if verbose and (t == 1 or t % 20 == 0):
                print("    FW stall t=" + str(t) +
                      " best_score_proxy=" + repr(2.0 * n * best_M))

    return best_a


def apply_single_move(c, a, s, d, tmove):
    c_new = c.copy()
    n = a.size
    c_new[d:d + n] += 2.0 * tmove * a
    c_new[s:s + n] -= 2.0 * tmove * a
    t2 = tmove * tmove
    if 2 * d < c_new.size:
        c_new[2 * d] += t2
    if 2 * s < c_new.size:
        c_new[2 * s] += t2
    if s + d < c_new.size:
        c_new[s + d] -= 2.0 * t2
    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
    return c_new


def eval_move_max(c, a, s, d, tmove):
    """Quickly evaluate the max of the convolution after a move, without full copy."""
    n = a.size
    size = 2 * n - 1
    # Compute the change vector
    delta = np.zeros(size, dtype=np.float64)
    delta[d:d + n] += 2.0 * tmove * a
    delta[s:s + n] -= 2.0 * tmove * a
    t2 = tmove * tmove
    if 2 * d < size:
        delta[2 * d] += t2
    if 2 * s < size:
        delta[2 * s] += t2
    if s + d < size:
        delta[s + d] -= 2.0 * t2
    c_new_vals = c + delta
    return float(np.max(c_new_vals))


def optimal_move_search(c, a, s, d, max_iters=45):
    """Find optimal transfer amount from s to d via ternary search."""
    a_s = a[s]
    if a_s < 1e-14:
        return None
    lo = 0.0
    hi = a_s * 0.9999

    # First check that a small move helps
    small_t = a_s * 0.005
    m_small = eval_move_max(c, a, s, d, small_t)
    m_0 = float(np.max(c))
    if m_small >= m_0 - 1e-18:
        return None

    for _ in range(max_iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = eval_move_max(c, a, s, d, m1)
        f2 = eval_move_max(c, a, s, d, m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-12 * a_s:
            break

    t_opt = (lo + hi) / 2.0
    m_opt = eval_move_max(c, a, s, d, t_opt)
    if m_opt < m_0 - 1e-17:
        return t_opt
    return None


def coordinate_polish_optimal(a0, budget, verbose=True):
    """Coordinate descent with optimal move amounts via ternary search."""
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    n = a.size
    stall = 0
    rnd = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit - 3 and stall < 200:
        rnd += 1
        if n <= 1:
            break

        # Compute pressure from top peaks
        top_k = min(30, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        p_top = np.zeros(2 * n - 1, dtype=np.float64)
        p_top[top_idx] = 1.0 / float(top_k)
        pressure = cross_convolve_p_a(p_top, a, 2.0)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)

        # Rank positions: sources are high-pressure, high-value; destinations are high-pressure, low-value
        src_score = pressure * (1.0 + float(n) * a)
        dst_score = pressure + 1e-6 * a

        src_count = min(10, n)
        dst_count = min(15, n)
        src_candidates = np.argpartition(src_score, min(src_count - 1, n - 1))[:src_count]
        dst_candidates = np.argpartition(dst_score, min(dst_count - 1, n - 1))[:dst_count]

        best_t_move = None
        best_m_new = M
        best_s = -1
        best_d = -1

        # Try all (s, d) pairs from candidates with optimal moves
        for srr in src_candidates:
            s = int(srr)
            if a[s] < 1e-10:
                continue
            for drr in dst_candidates:
                d = int(drr)
                if d == s:
                    continue
                # Quick check: will any move help?
                # The derivative of max at t=0 (from the right) is approximately:
                # For the peak positions, the directional derivative is 2*(pressure contribution)
                # If all peak derivatives are positive, no move helps
                t_opt = optimal_move_search(c, a, s, d, max_iters=40)
                if t_opt is not None:
                    m_new = eval_move_max(c, a, s, d, t_opt)
                    if m_new < best_m_new:
                        best_m_new = m_new
                        best_t_move = t_opt
                        best_s = s
                        best_d = d

        # Also try some random pairs
        for _ in range(20):
            s = int(rng.integers(n))
            d = int(rng.integers(n))
            if s == d or a[s] < 1e-10:
                continue
            t_opt = optimal_move_search(c, a, s, d, max_iters=35)
            if t_opt is not None:
                m_new = eval_move_max(c, a, s, d, t_opt)
                if m_new < best_m_new:
                    best_m_new = m_new
                    best_t_move = t_opt
                    best_s = s
                    best_d = d

        if best_t_move is not None and best_s >= 0:
            a[best_s] -= best_t_move
            a[best_d] += best_t_move
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = fft_conv(a)
            M = float(np.max(c))
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and (rnd <= 5 or rnd % 50 == 0):
                print("    CoordOpt rnd=" + str(rnd) +
                      " score_proxy=" + repr(2.0 * n * best_M) +
                      " s=" + str(best_s) + " d=" + str(best_d) +
                      " t=" + repr(best_t_move))
        else:
            stall += 1
            if stall % 15 == 14:
                # Perturb to escape
                sigma = 0.01 + 0.005 * (stall // 15)
                pert = rng.normal(0.0, min(sigma, 0.08), n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-12)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = float(np.max(c))
                stall = 0

    return best_a


def default_sequence(n=5000):
    x = np.linspace(0.0, 1.0, int(n))
    return norm_sum1(np.sin(np.pi * x) ** 2)


def comb_sequence(n, num_peaks, peak_width=1.0, seed=0):
    """Create a comb-like sequence with evenly spaced peaks."""
    rng = np.random.default_rng(seed)
    a = np.zeros(n, dtype=np.float64)
    for i in range(num_peaks):
        center = int((i + 0.5) * n / num_peaks)
        center = max(0, min(n - 1, center))
        a[center] = 1.0
        if peak_width > 1:
            w = int(peak_width)
            for j in range(1, w + 1):
                if center - j >= 0:
                    a[center - j] = 0.5
                if center + j < n:
                    a[center + j] = 0.5
    return norm_sum1(a)


def random_smooth_sequence(n, seed=0):
    """Create a random smooth non-negative sequence."""
    rng = np.random.default_rng(seed)
    # Start with random values and smooth
    a = np.abs(rng.normal(0, 1, n))
    # Apply a few smoothing passes
    for _ in range(3):
        a = 0.5 * a + 0.25 * np.roll(a, 1) + 0.25 * np.roll(a, -1)
    a = np.maximum(a, 0)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    try:
        total_budget = max(60.0, min(float(budget_s) - 90.0, 1710.0))
    except Exception:
        total_budget = 1710.0

    print("Start: budget=" + repr(total_budget) + "s")

    # Phase 0a: Gather candidates
    candidates = []

    # Parent sequence
    try:
        parent_list = height_sequence_1
    except Exception:
        parent_list = None

    if parent_list is not None:
        try:
            p = np.asarray(parent_list, dtype=np.float64)
            p = np.where(np.isfinite(p), p, 0.0)
            np.clip(p, 0.0, 1000.0, out=p)
            if p.ndim != 1:
                p = p.ravel()
            if p.size > 0 and float(p.sum()) > 1e-10:
                candidates.append(norm_sum1(p))
        except Exception:
            pass

    # Add some diverse candidates if we have time
    if len(candidates) < 3:
        n_base = candidates[0].size if candidates else 5000
        n_base = min(n_base, 7000)
        candidates.append(default_sequence(n_base))
        candidates.append(random_smooth_sequence(n_base, seed=123))

    # Evaluate all candidates
    best_score = float("inf")
    best_seq = None
    for i, cand in enumerate(candidates):
        sc = safe_eval(cand)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand.copy()
        print("  Candidate " + str(i) + ": n=" + str(cand.size) +
              " score=" + repr(sc))

    if best_seq is None:
        best_seq = default_sequence(5000)
        best_score = safe_eval(best_seq)

    print("Initial best: n=" + str(best_seq.size) + " score=" + repr(best_score))

    # Work sequence (compress if too long)
    if best_seq.size > 8000:
        work_seq = interpolate_sequence(best_seq, 7000)
    else:
        work_seq = best_seq.copy()

    # Verify first-order accuracy: 2*n*max(conv)/sum^2
    n_w = work_seq.size
    c_w = fft_conv(work_seq)
    M_w = float(np.max(c_w))
    score_proxy = 2.0 * n_w * M_w  # since sum=1
    print("Work proxy score: " + repr(score_proxy) + " n=" + str(n_w))

    # Phase 1: L-BFGS smoothed optimization
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)

        c_w = fft_conv(work_seq)

        lb_cost = max(0.05, time.monotonic() - t_est)
        print("  L-BFGS cost estimate: " + repr(lb_cost))

        a_lb = lbfegs_phase(work_seq, phase_budget, lb_cost)

        if a_lb.size <= 8000:

        n_w = work_seq.size
        c_w = fft_conv(work_seq)
        M_w = float(np.max(c_w))
        print("  After L-BFGS: proxy=" + repr(2.0 * n_w * M_w))

    time_left = total_budget - (t0 - t_start)

            if a_r2.size <= 8000:

    time_left = total_budget - (t0 - t_start)
    if SCIPY_OK and time_left > 60 and best_seq is not None and best_seq.size <= 8000:

                            reg_a=0.0, reg_u=1e-13, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_last, "final_lb", t_start)

    # Final validation
    final_val = safe_eval(best_seq)
    if math.isfinite(final_val):
        if final_val < best_score:
            best_score = final_val
    else:
        best_seq = default_sequence(5000)
        best_score = safe_eval(best_seq)

    arr = np.asarray(best_seq, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if arr.size == 0 or s < 0.01:
        arr = default_sequence(5000)
    else:
        arr = arr / s

    elapsed = time.monotonic() - t_start
    print("\nFINAL: score=" + repr(best_score) +
          " n=" + str(arr.size) +
          " elapsed=" + repr(elapsed) + "s")

    return [float(x) for x in arr]
