import numpy as np
import time
import cvxpy as cp

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_seq(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def J_val(a):
    n = len(a)
    return 2.0 * n * fast_conv(a).max()

def interp_to(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / ss

def smooth_max_ac(a, temp, top_k=200):
    """Return (value, gradient) of smooth max of autocorrelation."""
    ac = fast_conv(a)
    n = len(a)
    ac_max = ac.max()
    shifted = (ac - ac_max) / temp
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 0)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    val = ac_max + temp * np.log(exps.sum())
    
    grad = np.zeros(n, dtype=np.float64)
    top_idx = np.argsort(w)[::-1][:top_k]
    for k in top_idx:
        ki = int(k)
        wi = w[ki]
        if wi < 1e-12:
            continue
        i_s = max(0, ki - n + 1)
        i_e = min(n, ki + 1)
        idx = np.arange(i_s, i_e)
        grad[idx] += 2.0 * wi * a[ki - idx]
    return val, grad

def subgrad_descent(a0, time_limit, temp_init=10.0, temp_final=0.01, verbose=True):
    """Smooth-max subgradient descent."""
    n = len(a0)
    x = norm_seq(a0)
    best_x = x.copy()
    best_val = J_val(x)
    t0 = time.monotonic()
    total_time = max(time_limit, 1.0)
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Decrease temperature
        frac = min(1.0, t / (total_time * 500))  # decrease over first half
        temp = temp_init * (temp_final / temp_init) ** frac
        
        val, grad = smooth_max_ac(x, temp, top_k=200)
        
        # Nesterov-like acceleration
        if t > 1:
            y = x + ((t - 1) / (t + 2)) * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        val_y, grad_y = smooth_max_ac(y, temp, top_k=200)
        
        eta = 1.0 / (np.sqrt(t) + 1.0)
        x_new = y - eta * grad_y
        x_new = np.maximum(x_new, 1e-15)
        x_new = x_new / x_new.sum()
        
        x_prev = x
        x = x_new
        
        v = J_val(x)
        if v < best_val - 1e-12:
            best_val = v
            best_x = x.copy()
        
        if verbose and t % 500 == 0:
            print(f"    SubGrad t={t} temp={temp:.4f} J={best_val:.6f}")
    
    return best_x, best_val

def build_lp_matrix(a, lags, n):
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_e:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[ki - idx]
    return coeffs

def lp_fw(a0, time_limit, solver, init_K=300, max_K=2000, verbose=True):
    """LP-based Frank-Wolfe with backtracking line search."""
    n = len(a0)
    y = norm_seq(a0)
    best_a = y.copy()
    best_m = fast_conv(y).max()
    t0 = time.monotonic()
    
    ac = fast_conv(y)
    active_set = set(int(k) for k in np.argsort(ac)[::-1][:init_K])
    no_improve = 0
    stall_limit = 2000
    
    fft_size = 1 << (2 * n - 2).bit_length()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            # Keep tightest constraints
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)
        
        K = len(clist)
        if K == 0:
            break

        coeffs = build_lp_matrix(y, clist, n)

        M_var = cp.Variable()
