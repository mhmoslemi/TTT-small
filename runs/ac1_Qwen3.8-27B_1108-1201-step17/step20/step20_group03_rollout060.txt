def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_smoothed_grad(a, c, beta, top_k=None):

            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        c_y = fast_convolve(y)
        grad = compute_smoothed_grad(y, c_y, beta, top_k)
        eta = min(1.0 / (beta * 0.5 + 1.0), 2.0 / n)
        z = y - eta * grad
        x_new = simplex_project(z)
        c_new = fast_convolve(x_new)
        max_new = c_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        x_prev = x.copy()
        x = x_new
        t_nest = t_new
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.02 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise_scale, n)
            x_r = best_a + pert * best_a
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 10.0)
            if verbose:
                print(f"    FISTA restart t={t} ({time.monotonic()-t0:.1f}s): J={2.0*n*best_max_b:.6f}")
        if verbose and t % 5000 == 0:
            print(f"    FISTA t={t} ({time.monotonic()-t0:.1f}s): J={2.0*n*best_max_b:.6f} beta={beta:.0f}")
    return best_a, best_max_b
