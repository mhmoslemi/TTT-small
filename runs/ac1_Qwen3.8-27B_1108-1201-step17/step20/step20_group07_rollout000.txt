def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1)) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def interp_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def spectral_shape_init(n, iters=200, seed=42):
    """Iteratively reduce autocorrelation peaks via spectral domain water-filling."""
    rng = np.random.default_rng(seed)
    # Start with a broad triangular shape
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2  # smooth bump
    a = norm_sum1(a)
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    for it in range(iters):
        b = fast_autocorr(a)
        m = b.max()
        if m < best_m:
            best_m = m
            best_a = a.copy()
        
        # Identify top peaks
        top_k = 50
        peak_idx = np.argsort(b)[::-1][:top_k]
        
        # For each top peak, reduce contribution by lowering a at contributing positions
        # and raising a at positions that don't contribute to any top peak
        pressure = np.zeros(n)
        for k_i in peak_idx[:20]:
            ki = int(k_i)
            i_lo = max(0, ki - n + 1)
            i_hi = min(n - 1, ki)
            idx = np.arange(i_lo, i_hi + 1)
            pressure[idx] += a[ki - idx]
        
        # Move mass from high-pressure to low-pressure positions
        src_threshold = np.percentile(pressure, 70)
        dst_threshold = np.percentile(pressure, 20)
        src_mask = pressure > src_threshold
        dst_mask = pressure < dst_threshold
        
        if src_mask.sum() < 10 or dst_mask.sum() < 10:
            break
        
        src_total = a[src_mask].sum()
        n_dst = dst_mask.sum()
        move_amt = src_total * 0.05
        
        # Distribute to destinations proportional to current a
        dst_idx = np.where(dst_mask)[0]
        dst_weights = a[dst_idx] + 1e-10
        dst_weights = dst_weights / dst_weights.sum()
        
        # Remove mass from sources proportionally
        src_idx = np.where(src_mask)[0]
        src_weights = a[src_idx] + 1e-10
        src_weights = src_weights / src_weights.sum()
        
        a[src_idx] -= move_amt * src_weights
        a[dst_idx] += move_amt * dst_weights
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
    
    return best_a, best_m

        x_new = simplex_project(x_new)
        
        # Check improvement
        b_new = fast_autocorr(x_new)
        m_new = b_new.max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        k0 = k0 + 1.0
        
        # Adaptive parameters
        if t % 100 == 0:
            beta = min(beta * 1.02, beta_end)
            momentum = min(0.95, 0.8 + 0.1 * t / 5000.0)
        
        # Restart if stalled
        if stall > 800:
            stall = 0
            k0 = 2.0
            noise = 0.01
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("  SubGrad restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  SubGrad t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " beta=" + str(round(beta, 1)))
    
    return best_a, best_m

def quad_peak_shaving(a0, time_limit, top_k=20, verbose=True):
    """Peak shaving with quadratic prediction of autocorrelation changes."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 3000:
        b = fast_autocorr(a)
        peak_idx = np.argsort(b)[::-1][:top_k]
        
        improved = False
        for k_int in peak_idx[:8]:
            ki = int(k_int)
            i_lo = max(0, ki - n + 1)
            i_hi = min(n - 1, ki)
            idx = np.arange(i_lo, i_hi + 1)
            contrib = a[idx] * a[ki - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find the position in idx with max contribution
            src_i = int(idx[np.argmax(contrib)])
            partner = ki - src_i
            if partner < 0 or partner >= n:
                continue
            
            # Determine which to reduce (the larger one)
            if a[src_i] >= a[partner]:
                reduce_i = src_i
            else:
                reduce_i = partner
            
            if a[reduce_i] < 1e-14:
                continue
            
            # Find destination: minimize predicted autocorrelation increase
            # Predicted change in c_k if we move eps from reduce_i to dst:
            # δc_k = -eps*(a[k-reduce_i] + a[reduce_i] if reduce_i != k-reduce_i) 
            #         + eps*(a[k-dst] + a[dst] if dst != k-dst)
            # We want to minimize the max over all k
            # For the current peak ki:
            # δc_ki = -eps*(2*a[ki-reduce_i] if reduce_i != ki-reduce_i else 2*a[reduce_i])
            #         + eps*(2*a[ki-dst] if dst != ki-dst else 2*a[dst])
            
            # Compute "cost" of each destination: how much it increases the top peaks
            costs = np.zeros(n)
            for k2 in peak_idx[:10]:
                k2i = int(k2)
                i_lo2 = max(0, k2i - n + 1)
                i_hi2 = min(n - 1, k2i)
                # If we add to position dst, c_k2 increases by eps * (a[k2-dst] + (a[dst] if dst != k2-dst))
                # = eps * 2*a[k2-dst] (if valid)
                valid_dst = np.arange(i_lo2, i_hi2 + 1)
                partner_vals = a[k2i - valid_dst]
                costs[valid_dst] += 2.0 * partner_vals * b[k2i] / max(b.max(), 1e-15)
            
            costs[reduce_i] = np.inf
            
            # Also add a small random component for exploration
            costs += rng.uniform(0, 0.1, n) * np.mean(costs[costs < np.inf])
            
            dst_i = int(np.argmin(costs))
            if dst_i == reduce_i:
                continue
            
            # Try different fractions
            found = False
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = a[reduce_i] * frac
                if amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[reduce_i] -= amt
                a_trial[dst_i] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break
            
            if not found and stall > 500:
                # Random move
                j = int(rng.integers(n))
                if j != reduce_i and a[reduce_i] > 1e-12:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[reduce_i] * frac
                    a[reduce_i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        m_new = fast_autocorr(a).max()
                        if m_new < best_m:
                            best_m = m_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if not improved:
            stall += 1
            if stall % 100 == 0 and stall < 3000:
                # Perturbation
                noise_scale = min(0.02, 0.002 * stall / 100)
                pert = rng.normal(0, noise_scale, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-12)
                a_r = a_r / a_r.sum()
                m_r = fast_autocorr(a_r).max()
                if m_r < best_m:
                    best_m = m_r
                    best_a = a_r.copy()
                a = a_r if m_r < best_m + 0.005 else best_a.copy()
            
            if verbose and stall % 500 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("  PeakShave (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " stall=" + str(stall))
    
    return best_a, best_m

def active_set_equalize(a0, time_limit, n_active=5, verbose=True):
    """Try to equalize the top n_active autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    for outer in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        top_idx = np.argsort(b)[::-1][:n_active]
        top_vals = b[top_idx]
        m_curr = b.max()
        
        if m_curr < best_m - 1e-16:
            best_m = m_curr
            best_a = a.copy()
        
        # Check if peaks are already roughly equal
        val_range = top_vals.max() - top_vals.min()
        if val_range < 1e-12 * top_vals.max():
            break
        
        # For each pair of top peaks (k1, k2), try to transfer mass
        # to reduce c_k1 and increase c_k2 (if c_k1 > c_k2)
        for i in range(len(top_idx)):
            for j in range(i + 1, len(top_idx)):
                k1, k2 = int(top_idx[i]), int(top_idx[j])
                if b[k1] < b[k2]:
                    k1, k2 = k2, k1  # ensure b[k1] >= b[k2]
                
                # Find position that contributes to c_k1 but not c_k2
                # Move mass from a position in c_k1's support to one in c_k2's support
                
                i1_lo = max(0, k1 - n + 1)
                i1_hi = min(n - 1, k1)
                i2_lo = max(0, k2 - n + 1)
                i2_hi = min(n - 1, k2)
                
                # Positions in k1's support
                pos1 = set(range(i1_lo, i1_hi + 1))
                # Positions in k2's support  
                pos2 = set(range(i2_lo, i2_hi + 1))
                
                # Source: in pos1 with high contribution to c_k1
                src_candidates = [p for p in pos1 if a[p] > 1e-10]
                if not src_candidates:
                    continue
                # Destination: in pos2 or outside pos1
                dst_candidates = [p for p in pos2 if p not in pos1 and a[p] < 0.5 * max(a)]
                
                if not dst_candidates:
                    # Try any position not in pos1
                    dst_candidates = [p for p in range(n) if p not in pos1 and p != k1]
                
                if not dst_candidates:
                    continue
                
                # Pick best src and dst
                src_best = max(src_candidates, key=lambda p: a[p] * a[k1 - p] if 0 <= k1 - p < n else 0)
                # Pick dst that minimizes increase in other peaks
                dst_costs = []
                for d in dst_candidates[:20]:
                    cost = 0.0
                    for k3 in top_idx:
                        k3i = int(k3)
                        if 0 <= k3i - d < n:
                            cost += a[k3i - d] * b[k3i]
                    dst_costs.append((cost, d))
                dst_best = min(dst_costs, key=lambda x: x[0])[1]
                
                if src_best == dst_best:
                    continue
                
                for frac in [0.01, 0.02, 0.05]:
                    amt = a[src_best] * frac
                    a_trial = a.copy()
                    a_trial[src_best] -= amt
                    a_trial[dst_best] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        break
    
    elapsed = time.monotonic() - t0
    if verbose:
        J = 2.0 * n * best_m
        print("  ActiveSet (" + str(round(elapsed, 1)) + "s): J=" + repr(J))
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP Frank-Wolfe: minimize max autocorrelation via linear programming directions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    b = fast_autocorr(a)
    n_constraints = min(500, n)  # limit constraints for speed
    active_set = sorted(int(k) for k in np.argsort(b)[::-1][:n_constraints])
    no_improve = 0
    t = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 500:
        t += 1
        b = fast_autocorr(a)
        M_curr = b.max()
        
        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update active set
        if t % 10 == 0:
            active_set = sorted(int(k) for k in np.argsort(b)[::-1][:n_constraints])
        
        K = len(active_set)
        if K < 2:
            break
        
        # Build constraint matrix: row k has a[k-i] at column i for valid i
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(active_set):
            ki = int(k)
            i_lo = max(0, ki - n + 1)
            i_hi = min(n - 1, ki)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                coeffs[jj, idx] = a[ki - idx]
        
        # LP: minimize M subject to sum(g) = 1, g >= 0, coeffs @ g <= M
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # FW update with line search
        alpha_candidates = [2.0 / (t + 2.0)]
        for mult in [0.5, 1.5, 0.3]:
            alpha_candidates.append(max(0.01, alpha_candidates[0] * mult))
        
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = float('inf')
        for alpha in alpha_candidates:
            trial = (1.0 - alpha) * a + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        a_new = (1.0 - best_alpha) * a + best_alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        st = a_new.sum()
        if st > 1e-15:
            a_new = a_new / st
        a = a_new
        
        # Check for new violations
        if t % 5 == 0:
            b_full = fast_autocorr(a)
            violations = np.where(b_full > M_curr + 1e-8)[0]
            if len(violations) > 0:
                top_v = violations[np.argsort(b_full[violations])[::-1][:20]]
                for v in top_v:
                    active_set.append(int(v))
                active_set = sorted(set(active_set))
                if len(active_set) > n_constraints:
                    # Keep only top constraints
                    margins = np.array([M_curr - b_full[k] for k in active_set])
                    keep = np.argsort(margins)[:n_constraints]
                    active_set = sorted([int(active_set[i]) for i in keep])
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  LP t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))
    
    return best_a, best_m

def parametric_init(n, seed=42):
    """Generate diverse parametric initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    best_a = None
    best_m = float('inf')
    
    candidates = []
    
    # Power law shapes
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
        for beta_p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
            candidates.append((x + 0.005) ** alpha * (1.0 - x + 0.005) ** beta_p)
    
    # Gaussian mixtures
    for c1 in [0.2, 0.3, 0.35, 0.4, 0.5]:
        for c2 in [0.5, 0.6, 0.65, 0.7, 0.8]:
            for s in [0.05, 0.08, 0.1, 0.12, 0.15]:
                candidates.append(np.exp(-((x - c1)**2) / (2*s*s)) + np.exp(-((x - c2)**2) / (2*s*s)))
    
    # Cosine windows
    for p in [1, 2, 3, 4]:
        candidates.append(np.cos(np.pi * (x - 0.5)) ** (2*p + 1) + 1e-10)
    
    # Random
    for _ in range(20):
        a_raw = rng.random(n) ** rng.uniform(0.3, 3.0)
        candidates.append(a_raw)
    
    for cand in candidates:
        a = norm_sum1(cand)
        m = fast_autocorr(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()
    
    return best_a, best_m

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def propose_candidate(budget_s=1800.0):
    """
    Multi-phase optimization combining spectral shaping, Nesterov subgradient,
    quadratic peak shaving, active-set equalization, and LP Frank-Wolfe.
    Explores multiple sequence lengths for a better global optimum.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 50.0, 1750.0)
    
    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s, solver: " + str(solver))
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None
    
    if best_seq is None:
        n_base = 7000
        x = np.linspace(0, 1, n_base)
        a_raw = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    
    def optimize_sequence(a_init, n_opt, time_alloc, verbose=True):
        """Run the full optimization pipeline on a sequence of given length."""
        a = a_init
        t_start = time.monotonic()
        
        # Step 1: Quick subgradient
        t1 = min(time_alloc * 0.25, 300.0)
        if time.monotonic() - t_start < t1:
            a_out, _ = nesterov_subgradient(a, t1, beta_start=30.0, beta_end=400.0, verbose=verbose)
            try_update(a_out, "sg_n" + str(n_opt))
            a = a_out if a_out is not None else a
        
        # Step 2: Peak shaving
        remaining = time_alloc - (time.monotonic() - t_start)
        t2 = min(remaining * 0.25, 200.0)
        if remaining > 30:
            a_out, _ = quad_peak_shaving(a, t2, top_k=20, verbose=verbose)
            try_update(a_out, "ps_n" + str(n_opt))
            a = a_out if a_out is not None else a
        
        # Step 3: Active set equalization
        remaining = time_alloc - (time.monotonic() - t_start)
        t3 = min(remaining * 0.15, 100.0)
        if remaining > 20:
            a_out, _ = active_set_equalize(a, t3, n_active=5, verbose=verbose)
            try_update(a_out, "aseq_n" + str(n_opt))
            a = a_out if a_out is not None else a
        
        # Step 4: LP Frank-Wolfe
        remaining = time_alloc - (time.monotonic() - t_start)
        t4 = min(remaining * 0.35, 250.0)
        if remaining > 40 and n_opt <= 8000:
            a_out, _ = lp_frank_wolfe(a, t4, solver, verbose=verbose)
            try_update(a_out, "lp_n" + str(n_opt))
            a = a_out if a_out is not None else a
        
        # Step 5: Final peak shaving + subgradient
        remaining = time_alloc - (time.monotonic() - t_start)
        if remaining > 60:
            t5a = min(remaining * 0.3, 80.0)
            a_out, _ = quad_peak_shaving(a, t5a, top_k=15, verbose=False)
            try_update(a_out, "ps2_n" + str(n_opt))
            a = a_out if a_out is not None else a
            
            remaining = time_alloc - (time.monotonic() - t_start)
            t5b = min(remaining * 0.3, 80.0)
            if remaining > 30:
                a_out, _ = nesterov_subgradient(a, t5b, beta_start=50.0, beta_end=600.0, verbose=False)
                try_update(a_out, "sg2_n" + str(n_opt))
        
        return a
    
    # === PHASE 1: Multi-length exploration ===
    print("\n=== Phase 1: Multi-length exploration ===")
    n_parent = len(best_seq)
    n_values = [n_parent]
    for dn in [-300, -200, -100, 100, 200, 300, 400, 500, 700, 1000]:
        nl = n_parent + dn
        if 5500 <= nl <= 8500 and nl not in n_values:
            n_values.append(nl)
    
    # Sort by priority: parent first, then nearby
    n_values.sort(key=lambda x: abs(x - n_parent))
    
    # Phase 1a: Quick parametric init for each n
    remaining_budget = time_budget - (time.monotonic() - t_global)
    n_per_trial = max(3, len(n_values) // 2)
    time_per_init = min(remaining_budget * 0.05 / len(n_values[:6]), 30.0)
    
    inits = {}
    for nl in n_values[:6]:
        remaining_budget = time_budget - (time.monotonic() - t_global)
        if remaining_budget < 100:
            break
        
        if nl == n_parent:
            a_init = best_seq.copy()
        else:
            a_interp = interp_to_length(best_seq, nl)
            a_param, _ = parametric
