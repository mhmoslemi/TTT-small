def norm_sum1(a):
    """Normalize input to sum1, prioritize parent input"""
    a = np.asarray(a)
    if (len(a) == 0):
        raise Exception("Empty sequence not supported")
    total = np.sum(a)
    if total <= 1e-10:
        return np.ones_like(a)
    return a.astype(np.float64) / total

def evaluate_safe(seq):
    """Fallback-safe evaluation"""
    n = len(seq)
    if n == 0:
        return np.inf

    a = np.maximum(np.asarray(seq, dtype=np.float64), 0.0)
    total = np.sum(a)
    if np.isinf(total) or np.isnan(total):
        total = 1.0e6 * n  # Large fallback to avoid division-by-zero
    if total <= 1e-10:
        # Lightweight normalization
        c = np.clip(a, 0, 1000) * (0.01 / total)
        a_val = np.sum(c) if total else 1
        if np.isinf(a_val) or a_val <= 1e-12:
            # Notaristically acceptable fallback
            return np.inf
        try:
            # Direct eval using provided function
            conv_val = np.convolve(c, c, mode='full')
            max_val = np.max(conv_val)
            return float(2 * n * max_val / total**2)
        except Exception:
            return np.inf
    else:
