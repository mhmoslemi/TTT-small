```
n=z.size
if n==0 return z
u=np.sort(z)[::-1]
cssv=np.cumsum(u)
cond=u*np.arange(1,n+1)>(cssv-1)
rho=np.nonzero(cond)[0]
if rho.size==0: return np.ones(n)/n
rho=rho[-1]
theta=(cssv[rho]-1)/(rho+1)
return np.maximum(z-theta,0)
```

def linear_convolve(a):

    if n == 0: return np.array([0.0])
    if n == 1: return np.array([float(a[0])**2])

"""
Searches for a non-negative sequence minimizing
    2 * n * max(convolve(a, a)) / (sum(a))**2
by exploiting normalization invariance (sum(a)=1) and using:
  - exact fast convolution via FFT,
  - a fast 1D exact move update for swap/spike moves,
  - active-set informed steepest exchanges (softmax / top-lag weights),
  - bounded local polish with adaptive peak weighting,
  - cyclic shift sampling, length interpolation, and noise restarts.
"""

import numpy as np
import time
import math


def normalize_simplex(a):
    a = np.ravel(np.asarray(a, dtype=np.float64))
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def linear_convolve(a):
    a = np.ravel(np.asarray(a, dtype=np.float64))
    n = int(a.size)
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    L = 2 * n - 1
    if L == 1:
        return np.array([float(a[0]) * float(a[0])], dtype=np.float64)
    size = 1 << (L - 1).bit_length()
    A = np.fft.rfft(a, size=size)
    c = np.fft.irfft(A * A, size=size)[:L]
    return np.maximum(c, 0.0)


def cross_corr_weights(a, p):
    """
    Returns H_j = sum_k p_k * a_{k-j}, which is half the gradient
    of sum_k p_k * (a*a)_k with respect to a_j.
    """
    a = np.ravel(np.asarray(a, dtype=np.float64))
    p = np.ravel(np.asarray(p, dtype=np.float64))
    n = int(a.size)
    L = 2 * n - 1
    if n == 0:
        return np.array([], dtype=np.float64)
    need = n + L - 1
    size = 1 << (need - 1).bit_length()
    Ar = np.fft.rfft(a[::-1], size=size)
    Pr = np.fft.rfft(p, size=size)
    conv = np.fft.irfft(Ar * Pr, size=size)
    H = conv[n - 1:2 * n - 1]
    return np.maximum(H, 0.0)


def make_weights(c, beta):
    c = np.ravel(np.asarray(c, dtype=np.float64))
    L = int(c.size)
    if L == 0:
        return np.array([], dtype=np.float64)
    if beta <= 0.0:
        return np.ones(L, dtype=np.float64) / float(L)
    M = float(c.max())
    if M <= 0.0:
        return np.ones(L, dtype=np.float64) / float(L)
    v = np.minimum(c - M, 0.0)
    e = np.exp(beta * v)
    e += 1e-6
    s = float(e.sum())
    if (not math.isfinite(s)) or s <= 0.0:
        return np.ones(L, dtype=np.float64) / float(L)
    return e / s


def make_weights_top(c, top_k=8):
    c = np.ravel(np.asarray(c, dtype=np.float64))
    L = int(c.size)
    if L == 0:
        return np.array([], dtype=np.float64)
    p = np.full(L, 1e-6, dtype=np.float64)
    kk = max(1, min(int(top_k), L))
    idx = np.argsort(c)[::-1][:kk]
    p[idx] = 1.0
    p /= p.sum()
    return p


def spike_move(c, a, t, alpha):
    """
    Exact update for A_new = (1-alpha) A + alpha z^t.
    """
    a = np.ravel(np.asarray(a, dtype=np.float64))
    c = np.ravel(np.asarray(c, dtype=np.float64))
    n = int(a.size)
    L = 2 * n - 1
    c_new = c.copy()
    om = 1.0 - alpha
    if om != 1.0:
        c_new *= (om * om)
    factor = 2.0 * alpha * om
    if factor != 0.0:
        c_new[t:t + n] += factor * a
    if 2 * t < L:
        c_new[2 * t] += alpha * alpha
    return np.maximum(c_new, 0.0)


def exchange_move(c, a, s, t, delta):
    """
    Exact update for A_new = A + delta (z^t - z^s).
    """
    a = np.ravel(np.asarray(a, dtype=np.float64))
    c = np.ravel(np.asarray(c, dtype=np.float64))
    n = int(a.size)
    L = 2 * n - 1
    c_new = c.copy()
    if s != t and delta != 0.0:
        c_new[t:t + n] += 2.0 * delta * a
        c_new[s:s + n] -= 2.0 * delta * a
    if 2 * t < L:
        c_new[2 * t] += delta * delta
    if s + t < L:
        c_new[s + t] -= 2.0 * delta * delta
    if 2 * s < L:
        c_new[2 * s] += delta * delta
    return np.maximum(c_new, 0.0)


def interpolate_sequence(f, new_n):
    f = np.ravel(np.asarray(f, dtype=np.float64))
    old_n = int(f.size)
    new_n = int(max(1, new_n))
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    if old_n == new_n:
        return normalize_simplex(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    g = np.interp(new_x, old_x, f)
    g = np.maximum(g, 0.0)
    return normalize_simplex(g)


def noise_restart(a, strength=0.25, seed=1):
    a = normalize_simplex(a)
    n = int(a.size)
    rng = np.random.default_rng(seed)
    z1 = rng.standard_normal(n)
    z2 = rng.standard_normal(n)
    b = a * (1.0 + 0.5 * strength * z1) + (strength / float(n)) * z2
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s <= 1e-300:
        return a.copy()
    return b / s

def update_best(best_a, best_score, a, M, eval_a, eval_score):
    a = normalize_simplex(a)
    score = 2.0 * int(a.size) * float(M)
    new_best_a = best_a
    new_best_score = best_score
    if math.isfinite(score) and score < best_score:
        new_best_a = a.copy()
        new_best_score = score

    new_eval_a = eval_a
    new_eval_score = eval_score
    ev = evaluate_safe(a)
    if math.isfinite(ev) and ev < new_eval_score:

def best_cyclic_shift_sample(a, budget_s=30.0, max_trials=60):
    a = normalize_simplex(a)
    n = int(a.size)
    if n <= 1:
        m = float(linear_convolve(a).max())
        return a.copy(), m

    best_a = a.copy()
    c = linear_convolve(a)
    best_m = float(c.max())
    t0 = time.monotonic()

    tried = set()
    candidates = []

    min_idx = int(np.argmin(a))
    for off in (-2, -1, 0, 1, 2):
        sh = (min_idx + off) % n
        if sh not in tried:
            tried.add(sh)
            candidates.append(sh)

    rng = np.random.default_rng(424242)
    for _ in range(max(10, max_trials * 8)):
        if len(candidates) >= max_trials:
            break
        sh = int(rng.integers(n))
        if sh not in tried:
            tried.add(sh)
            candidates.append(sh)

    for sh in candidates:
        if time.monotonic() - t0 > budget_s:
            break
        b = np.roll(a, -sh)
        mb = float(linear_convolve(b).max())
        if mb < best_m - 1e-18:
            best_m = mb
            best_a = b.copy()

    return best_a, best_m


def local_polish(a, time_budget, verbose=True, beta_start=5000.0, max_no_improve=600):
    a = normalize_simplex(a)
    n = int(a.size)
    if n <= 1 or time_budget <= 0.0:
        a = normalize_simplex(a)
        c = linear_convolve(a)
        return a.copy(), float(c.max())

    c = linear_convolve(a)
    M = float(c.max())

    best_a = a.copy()
    best_c = c.copy()
    best_M = M

    t0 = time.monotonic()
    beta = float(beta_start)
    no_imp = 0
    it = 0
    moves = 0
    since_recompute = 0
    improve_tol = 1e-15

    fracs = (0.99, 0.80, 0.60, 0.40, 0.22, 0.08, 0.02)
    spike_alphas = (0.002, 0.01, 0.03, 0.10, 0.25)

    while time.monotonic() - t0 < time_budget:
        it += 1
        if no_imp > max_no_improve:
            break

        if since_recompute >= 80:
            a_true = normalize_simplex(a)
            c_true = linear_convolve(a_true)
            M_true = float(c_true.max())
            if M_true > best_M + 1e-12:
                a = best_a.copy()
                c = best_c.copy()
                M = best_M
            else:
                a = a_true
                c = c_true
                M = M_true
                if M < best_M - 1e-18:
                    best_M = M
                    best_a = a.copy()
                    best_c = c.copy()
            since_recompute = 0

        if (it % 4 == 0):
            p = make_weights_top(c, top_k=8)
        else:
            p = make_weights(c, beta)

        H = cross_corr_weights(a, p)
        H_order = np.argsort(H)

        improved = False

        # Spike / Frank-Wolfe-style coordinate move.
        lowH = H_order[:15]
        best_spike_M = M
        best_spike_c = None
        best_spike_t = -1
        best_spike_alpha = 0.0
        for t in lowH:

            for alpha in spike_alphas:
                c_new = spike_move(c, a, ti, alpha)

        if best_spike_c is not None and best_spike_M < M - improve_tol:

            a = om * a
            a[best_spike_t] += best_spike_alpha
            a = normalize_simplex(a)
            c = best_spike_c
            M = best_spike_M

            improved = True
            no_imp = 0
            moves += 1
            since_recompute += 1
            if M < best_M - 1e-18:
                best_M = M
                best_a = a.copy()
                best_c = c.copy()

            if verbose and (moves % 500 == 0):
                el = time.monotonic() - t0
                print("        local spike move %d, t=%.1f, score=%.9f" % (moves, el, 2.0 * n * M))
            continue

        # Two-exchange moves guided by H-values.
        H_min = float(H.min())
        source_score = a * np.maximum(H - H_min, 0.0)
        source_idx = np.argsort(source_score)[::-1][:25]
        dest_idx = H_order[:25]

        first_pairs = []
        for s in source_idx:
            si = int(s)
            as_ = float(a[si])
            if as_ <= 1e-14:
                continue
            for t in dest_idx:
                ti = int(t)
                if si == ti:
                    continue
                if H[ti] < H[si]:
                    score = (float(H[ti]) - float(H[si])) * as_
                    first_pairs.append((score, si, ti))

        first_pairs.sort()

        eval_pairs = []
        seen = set()
        pair_limit = 60
        first_limit = 35

        for item in first_pairs:
            if len(eval_pairs) >= first_limit:
                break
            _, si, ti = item
            key = (si, ti)
            if key in seen:
                continue
            seen.add(key)
            eval_pairs.append(key)

        # Add pairs originating from currently large convolution coefficients.
        if len(eval_pairs) < pair_limit:
            top_lags = np.argsort(c)[::-1][:10]
            for kk in top_lags:
                if len(eval_pairs) >= pair_limit:
                    break
                k = int(kk)
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                if lo <= hi:
                    idxs = np.arange(lo, hi + 1, dtype=int)
                    prods = a[idxs] * a[k - idxs]
                    order = np.argsort(prods)[::-1][:5]
                    for u in order:
                        i = int(idxs[u])
                        j = int(k - i)
                        if i == j:
                            continue
                        key1 = (i, j)
                        key2 = (j, i)
                        if key1 not in seen and len(eval_pairs) < pair_limit:
                            seen.add(key1)
                            eval_pairs.append(key1)
                        if key2 not in seen and len(eval_pairs) < pair_limit:

                            eval_pairs.append(key2)

        best_move_c = None
        best_move_s = 0

            a[best_move_s] -= best_move_delta
            a[best_move_t] += best_move_delta
            a = np.maximum(a, 0.0)
            a = normalize_simplex(a)
            c = best_move_c
            M = best_move_M

            improved = True
            no_imp = 0
            moves += 1
            since_recompute += 1
            if M < best_M - 1e-18:
                best_M = M
                best_a = a.copy()
                best_c = c.copy()

            if verbose and (moves % 500 == 0):
                el = time.monotonic() - t0
                print("        local exchange move %d, t=%.1f, score=%.9f" % (moves, el, 2.0 * n * M))
            continue

        no_imp += 1
        if no_imp % 15 == 0:
            if beta < 25000.0:
                beta = min(beta * 1.7, 25000.0)
            else:

        if verbose and (it % 500 == 0):

        if no_imp > 80:

                mt = float(c_new.max())

                    a[ti] += delta
                    a = np.maximum(a, 0.0)
                    a = normalize_simplex(a)
                    c = c_new
                    M = mt
                    improved = True
                    no_imp = 0
                    moves += 1
                    since_recompute += 1
                    done = True

                    break
            if done:

    a_best = normalize_simplex(best_a)
    c_best = linear_convolve(a_best)

    a_cur = normalize_simplex(a)

    M_cur = float(c_cur.max())
    if M_cur < M_best:
        a_best = a_cur
        M_best = M_cur

    return a_best, M_best

        remaining = total_budget - (time.monotonic() - t_start)
        if remaining < 200.0:
            break
        t_len = min(total_budget * 0.15, remaining - 80.0)
        a_len = interpolate_sequence(best_a, nn)
        a2, M2 = local_polish(a_len, t_len, verbose=True, beta_start=3500.0, max_no_improve=400)
        best_a, best_score, best_eval_a, best_eval_score = update_best(
            best_a, best_score, a2, M2, best_eval_a, best_eval_score
        )
        print("after length %d internal_score=%.10f eval_score=%.10f" %
              (int(best_a.size), best_score, best_eval_score))

    # One noise restart.
    remaining = total_budget - (time.monotonic() - t_start)
    if remaining > 220.0:
        t_restart = min(total_budget * 0.10, remaining - 80.0)
        a_nr = noise_restart(best_a, strength=0.30, seed=98765)
        a2, M2 = local_polish(a_nr, t_restart, verbose=True, beta_start=3000.0, max_no_improve=350)
        best_a, best_score, best_eval_a, best_eval_score = update_best(
            best_a, best_score, a2, M2, best_eval_a, best_eval_score
        )
        print("after restart internal_score=%.10f eval_score=%.10f" %
              (best_score, best_eval_score))

    # Final local polish.
    remaining = total_budget - (time.monotonic() - t_start)
    if remaining > 90.0:
        t_final = min(total_budget * 0.10, remaining - 45.0)
        a2, M2 = local_polish(best_a, t_final, verbose=True, beta_start=8000.0, max_no_improve=300)
        best_a, best_score, best_eval_a, best_eval_score = update_best(
            best_a, best_score, a2, M2, best_eval_a, best_eval_score
        )

    # Final small cyclic-shift polish.
    remaining = total_budget - (time.monotonic() - t_start)
    if remaining > 45.0:
        a_shift, M_shift = best_cyclic_shift_sample(best_a, budget_s=min(25.0, remaining - 20.0), max_trials=40)
        best_a, best_score, best_eval_a, best_eval_score = update_best(
            best_a, best_score, a_shift, M_shift, best_eval_a, best_eval_score
        )

    # Prefer the sequence with the lowest externally evaluated score if available.
    if math.isfinite(best_eval_score) and best_eval_score < best_score:
        chosen = best_eval_a
    else:
        chosen = best_a

    chosen = normalize_simplex(chosen)
    final_internal = 2.0 * int(chosen.size) * float(linear_convolve(chosen).max())
    final_eval = evaluate_safe(chosen)

    print("final n=%d internal_score=%.10f eval_score=%.10f" %
          (int(chosen.size), final_internal, final_eval))

    arr = np.ravel(np.asarray(chosen, dtype=np.float64))
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 1e-300:
        arr = np.ones(1000, dtype=np.float64)
    else:
        arr = arr / s
    arr = np.clip(arr, 0.0, 1000.0)
    return [float(x) for x in arr]
