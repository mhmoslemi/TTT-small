def nesterov_mult_weight(a0, time_limit, beta_init=100.0, beta_min=5.0, 
                          momentum=0.9, eta_init=1.0, eta_decay=0.9999,
                          restart_every=20000, noise_scale=0.01, verbose=True):
    """Nesterov-accelerated multiplicative weight descent with full FFT gradient."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    eta = eta_init
    mom = momentum
    beta = beta_init
    v = np.zeros(n, dtype=np.float64)  # momentum buffer
    stall = 0
    max_stall = 50000
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break
        
        # Compute autocorrelation
        ac = fast_autocorr(a)
        curr_val = 2.0 * n * ac.max()
        
        if curr_val < best_val - 1e-12:
            best_val = curr_val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive beta: start high (focus on top peaks), decrease over time
        beta = max(beta_min, beta_init * (0.99995 ** t))
        
        # Compute full gradient
        grad = full_gradient(a, ac, beta=beta)
        
        # Momentum update
        v = mom * v + (1.0 - mom) * grad
        # Decay momentum slightly over time
        mom = momentum * (0.99999 ** t)
        
        # Adaptive step size
        eta = eta_init * (eta_decay ** t)
        
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30)) - eta * v
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / s
        
        a = a_new
        
        # Periodic restart with noise
        if stall > restart_every and stall % (restart_every // 4) == 0:
            noise = noise_scale * rng.standard_normal(n) * best_a
            a_r = best_a + noise
            a_r = np.maximum(a_r, 1e-12)
            a = a_r / a_r.sum()
            v = np.zeros(n, dtype=np.float64)
            stall = 0
            noise_scale *= 1.5  # Increase noise on restart
            if noise_scale > 0.1:
                noise_scale = 0.1
        
        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print("    NMW t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(best_val) + " eta=" + repr(eta) + " beta=" + repr(beta))
    
    return best_a, best_val

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def compute_gradient(a, ac, beta):
    """Compute full gradient of weighted max autocorrelation.
    
    grad[j] = 2 * sum_k w[k] * a[k-j] where w = softmax(beta * (ac - ac_max))
    
    Uses the identity: grad = 2 * convolve(w, a[::-1])[n-1 : n-1+n]
    """
    n = len(a)
    size = 2 * n - 1
    ac_max = ac.max()
    # Softmax weights
    exponents = beta * (ac - ac_max)
    # Clip to avoid underflow
    exponents = np.maximum(exponents, -50.0)
    w = np.exp(exponents)
    w_sum = w.sum()
    if w_sum < 1e-30:
        # Fallback: just use the top peak
        w = np.zeros(size)
        w[int(np.argmax(ac))] = 1.0
    else:
        w = w / w_sum
    
    # grad[j] = 2 * sum_k w[k] * a[k-j] = 2 * (w * a_reversed)[n-1+j]
    a_rev = a[::-1]
    conv = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv[n-1 : n-1+n]
    return grad

def nesterov_mult_weight(a0, time_limit, beta_init=50.0, beta_end=5.0,

    max_stall = 30000

    # Pre-compute the shift for gradient
    # We'll use np.convolve which is O(n log n) via FFT for large arrays
    
    for t in range(1, 100000000):

        # Compute autocorrelation and objective

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_matrix(a, lags, n):
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs
