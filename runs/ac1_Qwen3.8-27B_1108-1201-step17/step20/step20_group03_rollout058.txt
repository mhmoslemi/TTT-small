def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_smoothed_grad(y, beta, top_k):
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and top_k < len(b) and top_k > 0:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.full(len(b), -1e30)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b.copy()
        b_max = b.max()
    shifted = (b_use - b_max) / beta
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b_use)
        mask = (b_use == b_max)
        p[mask] = 1.0 / max(1, mask.sum())
    else:
        p = exp_vals / p_total
    # Gradient: weighted sum of gradients of b[k] for each k
    # grad of b[k] w.r.t. a[j] = a[k-j] if 0<=k-j<n else 0 (times 2)
    # So gradient vector = 2 * sum_k p[k] * a_reversed_shifted[k]
    # a_reversed_shifted[k][j] = a[k-j] if 0<=k-j<n else 0
    # This is equivalent to: for each j, grad[j] = 2 * sum_k p[k] * a[k-j] (if valid)
    # Which is the convolution of p with a_reversed, evaluated at specific positions
    
    # More efficiently: grad[j] = 2 * sum_k p[k] * a[k-j]
    # Let a_rev[i] = a[n-1-i], then a[k-j] = a_rev[n-1-k+j]
    # grad[j] = 2 * sum_k p[k] * a_rev[n-1-k+j]
    # = 2 * (p * a_rev)[n-1+j]  (cross-correlation)
    # Actually, let's just use FFT
    
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    # We need sum_k p[k] * a[k-j] for j = 0, ..., n-1
    # a[k-j] is valid when 0 <= k-j < n, i.e., j <= k and k-j < n
    # a[k-j] = a_rev[n-1-(k-j)] = a_rev[n-1-k+j]
    # So grad[j] = 2 * sum_k p[k] * a_rev[n-1-k+j]
    # Let m = n-1+j, then grad[j] = 2 * sum_k p[k] * a_rev[m-k]
    # This is (p * a_rev)[m] = (p * a_rev)[n-1+j]
    conv = np.fft.irfft(Ar * Pr, fft_size)
    # conv[m] = sum_k a_rev[k] * Pr... no wait
    # np.convolve(p, a_rev)[m] = sum_k p[k] * a_rev[m-k]
    # So we want conv[n-1+j] for j = 0,...,n-1
    # conv range: 0 to n + len(p) - 2
    # n-1+j range: n-1 to 2n-2
    # We need len(p) >= n for this to work... 
    # Actually len(p) = 2n-1 (same as b), so conv_len = n + 2n-1 - 1 = 3n-2
    # conv index range: 0 to 3n-3
    # We need indices n-1 to 2n-2, which are valid.
    
    grad = conv[n - 1:2 * n - 1]
    return grad * 2, p, b

def fista_adaptive(a0, time_limit, beta_start=20.0, beta_end=1000.0,
                   top_k_start=5, top_k_end=40, 
                   restart_every=3000, verbose=True, seed=123):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        # Adaptive top_k: start small, grow with iterations
        top_k = min(top_k_end, max(top_k_start, int(top_k_start + (top_k_end - top_k_start) * t / (2 * restart_every))))
        
        grad, p, b = compute_smoothed_grad(y, beta, top_k)
        
        # Simple step size: eta = 1/beta ( Lipschitz constant of the smoothed gradient)
        eta = 1.0 / beta
        # Clip to avoid too large steps
        eta = min(eta, 2.0 / (beta + 1.0))
        
        z = y - eta * grad
        x_new = simplex_project(z)
        
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        
        if max_new < best_max_b - 1e-14:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
            # Grow beta (harder to find new peaks)
            beta = min(beta * 1.005, beta_end)
        else:
            stall += 1
            # If stalled a bit, grow beta faster
            if stall > 100:
                beta = min(beta * 1.02, beta_end)
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            # Restart from best with perturbation
            noise_scale = 0.02 / (1.0 + 0.3 * (t / restart_every))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, beta_start * 0.5)
            
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    FISTA restart t={} ({}s): J={}".format(t, round(elapsed,1), round(J, 6)))
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    FISTA t={} ({}s): J={} beta={} top_k={}".format(t, round(elapsed,1), round(J,6), round(beta,1), top_k))
    
    return best_a, best_max_b


def peak_target_lp(a0, time_limit, solver, n_peaks=5, verbose=True):
    """LP that targets the top n_peaks peaks specifically."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
        
        # Get top peaks
        top_idx = np.argsort(ac)[::-1][:n_peaks]
        
        # Build constraint matrix
        K = len(top_idx)
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        # LP: minimize M subject to sum(g)=1, g>=0, coeffs@g <= M
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            break
        g_dir = g_val / g_sum
        
        # Line search
        best_alpha = 0.1
        best_trial_M = fast_autocorr((1-0.1)*y + 0.1*g_dir).max()
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            break
        y = y_new / s_y
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakLP it={} ({}s): J={}".format(t, round(elapsed,1), round(J,6)))
    
    return best_a, best_m


def two_exchange_search(a0, time_limit, verbose=True):
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:20]
        improved = False
        
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(top_peaks[rng.integers(len(top_peaks))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            
            # Find destination that minimizes impact on peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src or a[src] < 1e-12:
                continue
            
            frac = float(rng.choice([0.002, 0.005, 0.01, 0.02, 0.05]))
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-14:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                ac = fast_autocorr(a)
                top_peaks = np.argsort(ac)[::-1][:20]
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.001, 0.01))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
    
    return best_a, best_m


def cyclic_shift_best(a0):
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a0, fft_size)
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_initializations(n, seed=42):

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FISTA with adaptive top-k smoothing and adaptive restart
2. Multiple sequence lengths with interpolation
3. Targeted peak-reduction LP (small active set)
4. 2-exchange local search
5. Cyclic shift enumeration
6. Spectral post-processing

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None


def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    return _MODULE_SOLVER


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_smoothed_grad(y, beta, top_k, rng=None):
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and top_k < len(b) and top_k > 0:
        if rng is not None and rng.random() < 0.15 and top_k * 3 < len(b):
            pool = np.argsort(b)[::-1][:top_k * 3]
            idx = rng.choice(pool, size=top_k, replace=False)
        else:
            idx = np.argsort(b)[::-1][:top_k]
        b_use = np.full(len(b), -1e30)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b.copy()
        b_max = b.max()
    shifted = (b_use - b_max) / beta
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b_use)
        mask = (b_use == b_max)
        p[mask] = 1.0 / max(1, mask.sum())
    else:
        p = exp_vals / p_total
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1]
    return grad * 2, p, b

    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        top_k = min(top_k_end, max(top_k_start,
                                   int(top_k_start + (top_k_end - top_k_start) * min(1.0, t / (2.0 * restart_every)))))

        grad, p, b = compute_smoothed_grad(y, beta, top_k, rng)

        eta = 1.0 / beta
        eta = min(eta, 2.0 / (beta + 1.0))

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        if max_new < best_max_b - 1e-14:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
            beta = min(beta * 1.004, beta_end)
        else:
            stall += 1
            if stall > 200:
                beta = min(beta * 1.015, beta_end)

        x_prev = x
        x = x_new
        t_nest = t_new

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.015 / (1.0 + 0.2 * (t / restart_every))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, beta_start * 0.5)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    FISTA restart t={} ({}s): J={}".format(t, round(elapsed, 1), round(J, 6)))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    FISTA t={} ({}s): J={} beta={} top_k={}".format(t, round(elapsed, 1), round(J, 6), round(beta, 1), top_k))

    return best_a, best_max_b


def peak_target_lp(a0, time_limit, solver, n_peaks=5, verbose=True):
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()

        top_idx = np.argsort(ac)[::-1][:n_peaks]
        K = len(top_idx)
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            break
        g_dir = g_val / g_sum

        best_alpha = 0.1
        best_trial_M = fast_autocorr((1 - 0.1) * y + 0.1 * g_dir).max()
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8]:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y
