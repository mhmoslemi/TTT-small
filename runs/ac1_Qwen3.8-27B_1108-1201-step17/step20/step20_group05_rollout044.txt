def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def fft_conv2(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:

    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)

def objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 for normalized a."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    c = fft_conv(a)
    if c.size == 0:
        return float("inf")
    m = float(np.max(c))
    return 2.0 * n * m


def lse_objective(a, beta):
    """Log-sum-exp smoothed objective and gradient."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    c = fft_conv(a)
    m = float(np.max(c))
    shifted = beta * (c - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0:
        return m, np.zeros(n)
    p = e / se  # probability distribution over convolution indices
    
    # Gradient: d/dx_i [lse(conv(x,x))] = 2 * sum_k p_k * x_{k-i} (with appropriate bounds)
    # This is 2 * (p reversed) * x, truncated
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    f = m + math.log(se) / beta
    return f, grad


def softmax_from_u(u):
    u = np.asarray(u, dtype=np.float64)
    n = u.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(u)):
        return np.ones(n, dtype=np.float64) / float(n)
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def make_v(a):

def run_lbfegs(a0, beta, maxfun=300):
    """Run L-BFGS-B optimization."""
    a = norm_sum1(a0)
    n = a.size
    if n <= 1 or not SCIPY_OK or maxfun <= 0:
        return a

    try:

    except Exception:

    return a

            return np.ones(K, dtype=np.float64) / float(K)

    g = np.ones(n, dtype=np.float64) / float(n)
    
    for it in range(max(1, int(mw_iters))):
        pf = np.zeros(size, dtype=np.float64)

            g = 0.98 * g + 0.02 * (np.ones(n, dtype=np.float64) / float(n))

    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K <= 0 or K > 5000:
        return None

    # Build constraint matrix: for each active index k, 
    # constraint is sum_i g_i * f[k-i] <= 1 (where f[j]=0 outside [0,n-1])
    # Minimize max => min t subject to A@g <= t, sum(g)=1, g>=0
    # After normalizing sum(g)=1, minimize t subject to A@g <= t, g>=0
    
    row_parts = []
    col_parts = []
    val_parts = []
    for r, k in enumerate(active_idx):
        k = int(k)
        s = max(0, k - n + 1)
        e = min(n, k + 1)
        if s >= e:
            continue
        cols = np.arange(s, e, dtype=np.int32)
        # f[k - i] for i in [s, e)
        vals = f[np.clip(k - cols, 0, n - 1)]
        # Only include where k-c is in [0, n-1]
        valid = (k - cols >= 0) & (k - cols < n)
        cols = cols[valid]
        vals = vals[valid]
        if cols.size == 0:
            continue
        row_parts.append(np.full(cols.size, r, dtype=np.int32))
        col_parts.append(cols)
        val_parts.append(vals.astype(np.float64))

    if not row_parts:
        return None

    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))

    # LP: min t
    # s.t. A@g - t <= 0  (K constraints)
    #      sum(g) = 1
    #      g >= 0, t free
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0  # minimize t

    neg_one = scipy.sparse.csc_matrix(-np.ones((K, 1)), dtype=np.float64)
    A_ub = scipy.sparse.hstack((A, neg_one), format="csr")
    b_ub = np.zeros(K, dtype=np.float64)

    A_eq = np.ones((1, n + 1), dtype=np.float64)
    A_eq[0, n] = 0.0
    b_eq = np.ones(1, dtype=np.float64)

    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = scipy.optimize.linprog(
            c_obj, A_ub=A_ub, b_ub=b_ub,
            A_eq=A_eq, b_eq=b_eq, bounds=bounds,
            method="HiGHS", options={"disp": False, "time_limit": float(time_limit)}
        )
        if res is not None and getattr(res, "success", False):
            g = np.asarray(res.x[:n], dtype=np.float64)
            g = np.where(np.isfinite(g) & (g > 0.0), g, 0.0)
            gs = float(g.sum())
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def line_search(a, g, c, cg, fg, current_M):
    """Line search for FW step."""
    base_alphas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
    best_alpha = 0.0
    best_m = current_M
    for alpha in base_alphas:
        one = 1.0 - alpha
        val = (one * one) * c + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
        m = float(np.max(val))
        if math.isfinite(m) and m < best_m:
            best_m = m
            best_alpha = alpha
    if best_alpha <= 0:
        return a.copy(), best_m, 0.0
    new_a = (1.0 - best_alpha) * a + best_alpha * g
    new_a = np.maximum(new_a, 0.0)
    s = float(new_a.sum())
    if s < 1e-15:
        return a.copy(), best_m, 0.0
    new_a = new_a / s
    return new_a, best_m, best_alpha


def frank_wolfe(a0, budget, active_max=1000, game_iters=25,
                 exact_every=5, exact_time=8.0, verbose=True):
    """Frank-Wolfe optimization with exact LP direction."""
    t0 = time.monotonic()
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    active = np.argsort(c)[::-1][:active_max]
    no_improve = 0
    t = 0
    
    while time.monotonic() - t0 < budget - 3 and no_improve < 60:
        t += 1
        time_left = budget - (time.monotonic() - t0)
        if time_left < 20:
            break
        
        # Get direction
        g = None
        if exact_every > 0 and (t % exact_every == 0) and time_left > 80:
            # Use exact LP with smaller active set
            lp_active = np.argsort(c)[::-1][:min(300, active_max)]
            g = lp_direction(a, lp_active, exact_time)
        
        if g is None:
            active = np.argsort(c)[::-1][:active_max]
            init_log = None
            if active.size > 0:
                av = c[active]
                av = np.where(np.isfinite(av) & (av > 0.0), av, 1e-300)
                init_log = np.log(av)
                init_log -= float(np.max(init_log))
            g = game_direction(a, active, mw_iters=game_iters,
                             seed=1000 + t, init_logits=init_log)
        
        if g is None:
            break
        
        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        new_a, new_M, alpha = line_search(a, g, c, cg, fg, M)
        
        if new_M < M - 1e-15:
            a = new_a
            M = new_M
            c = fft_conv(a)
            no_improve = 0
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
            if verbose and (t <= 3 or t % 10 == 0):
                print("    FW t=%d M=%.10f obj=%.6f alpha=%.3f" % (t, M, 2*n*M, alpha))
        else:
            no_improve += 1
            if no_improve % 10 == 9:
                # Perturb to escape
                rng = np.random.default_rng(9999 + t)
                pert = rng.normal(0.0, 0.02, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-12)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = float(np.max(c))
                no_improve = 0
    
    return best_a


def coordinate_descent(a0, budget, verbose=True):
    """Coordinate descent with rank-1 convolution updates."""
    t0 = time.monotonic()
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    stall = 0
    rng = np.random.default_rng(42)
    rnd = 0
    
    while time.monotonic() - t0 < budget - 2 and stall < 200:
        rnd += 1
        if n <= 1:
            break
        
        # Find top peaks
        top_k = min(30, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        
        # Compute pressure
        p_top = np.zeros(2 * n - 1, dtype=np.float64)
        p_top[top_idx] = 1.0 / float(top_idx.size)
        
        total = 3 * n - 2
        N = next_pow2(total)
        A = np.fft.rfft(a, N)
        Pr = np.fft.rfft(p_top[::-1], N)
        conv = np.fft.irfft(A * Pr, N)[:total]
        pressure = 2.0 * conv[n - 1:2 * n - 1][::-1]
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)
        
        # Try moves
        src_count = min(6, n)
        dst_count = min(10, n)
        src_score = pressure * (1.0 + n * a)
        src_cands = np.argpartition(src_score, min(src_count - 1, n - 1))[:src_count]
        dst_cands = np.argpartition(pressure, min(dst_count - 1, n - 1))[:dst_count]
        
        best_cand = None
        for srr in src_cands:
            s = int(srr)
            if a[s] < 1e-12:
                continue
            for drr in dst_cands:
                d = int(drr)
                if d == s:
                    continue
                for frac in (0.003, 0.007, 0.015, 0.03, 0.06):
                    tmove = a[s] * frac
                    if tmove <= 1e-16:
                        continue
                    # Rank-1 update
                    c_new = c.copy()
                    c_new[d:d + n] += 2.0 * tmove * a
                    c_new[s:s + n] -= 2.0 * tmove * a
                    if 2 * d < c_new.size:
                        c_new[2 * d] += tmove * tmove
                    if 2 * s < c_new.size:
                        c_new[2 * s] += tmove * tmove
                    if s + d < c_new.size:
                        c_new[s + d] -= 2.0 * tmove * tmove
                    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
                    m_new = float(np.max(c_new))
                    if m_new < M - 1e-16:
                        if best_cand is None or m_new < best_cand[0]:
                            best_cand = (m_new, s, d, tmove)
        
        if best_cand is None:
            # Random exploration
            for _ in range(30):
                s = int(rng.integers(n))
                d = int(rng.integers(n))
                if
