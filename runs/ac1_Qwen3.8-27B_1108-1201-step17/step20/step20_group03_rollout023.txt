def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

        a_rev = y[::-1]
        conv_len = n + len(p) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        Ar = np.fft.rfft(a_rev, fft_size)
        Pr = np.fft.rfft(p, fft_size)
        conv = np.fft.irfft(Ar * Pr, fft_size)
        grad = conv[n - 1:2 * n - 1]

        eta = min(bb_step / max(beta * 0.1, 1e-10), 0.1)
        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.02 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 10.0)
            bb_step = 1.0 / max(1.0, np.sqrt(n))
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    FISTA restart (t={}s): J={}".format(round(elapsed, 1), round(J, 6)))

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    FISTA t={} ({}s): J={}".format(t, round(elapsed, 1), round(J, 6)))

    return best_a, best_max_b


def lp_fw_optimize(a0, time_limit, solver, initial_K=500, max_K=3000,
                   add_per_iter=30, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:min(initial_K, 2 * n - 1)])
    no_improve = 0
    max_no_improve = 600
    consecutive_lp_fail = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break
        if consecutive_lp_fail > 10:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix efficiently
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            consecutive_lp_fail += 1
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            consecutive_lp_fail += 1
            continue

        consecutive_lp_fail = 0
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violated constraints to add
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_yg > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_yg[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5),
                  max(0.02, default_alpha * 2.0), 1.0 / (t + 1), 0.1, 0.2, 0.3]
        alphas = list(set([max(0.01, min(1.0, a)) for a in alphas]))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        # Periodically refresh active set
        if t % 200 == 0:
            ac_now = fast_autocorr(y)
            top_now = np.argsort(ac_now)[::-1][:initial_K]
            active_set = set(int(k) for k in top_now)

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter={} ({}s): J={} |S|={}".format(
                t, round(elapsed, 1), round(J, 6), len(active_set)))

    return best_a, best_m


def peak_pair_redistribute(a0, time_limit, verbose=True):
    """Targeted redistribution: find top contributing pairs to the peak and move mass away."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    max_stall = 100

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:20]
        improved = False

        for _ in range(80):
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(top_peaks[rng.integers(min(len(top_peaks), 10))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-25:
                continue
            src = int(idx[np.argmax(contrib)])

            # Find a good destination: position that minimizes contribution to all top peaks
            in_peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak_mask[s2:e2 + 1] = True

            # Prefer positions not in any peak region
            safe_positions = np.where(~in_peak_mask)[0]
            if len(safe_positions) < 5:
                safe_positions = np.arange(n)

            # Try several random destinations
            best_trial = None
            best_trial_m = best_m
            for _ in range(8):
                dst = int(safe_positions[rng.integers(len(safe_positions))])
                if dst == src:
                    continue
                if a[src] < 1e-15:
                    continue
                frac = float(rng.choice([0.005, 0.01, 0.02, 0.05, 0.1, 0.15]))
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_trial_m - 1e-14:
                    best_trial = a_trial
                    best_trial_m = M_trial

            if best_trial is not None:
                a = best_trial
                best_m = best_trial_m
                best_a = a.copy()
                improved = True
                stall = 0
                ac = fast_autocorr(a)
                top_peaks = np.argsort(ac)[::-1][:20]

        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = float(rng.uniform(0.005, 0.05))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakRedist ({}s): J={} stall={}".format(round(elapsed, 1), round(J, 6), stall))

    return best_a, best_m


def cyclic_shift_best(a0):
    n = len(a0)
    if n < 2:
        return a0.copy(), fast_autocorr(a0).max()
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()

    # Only try a subset of shifts for large n
    if n > 3000:
        shifts = list(range(0, n, max(1, n // 100)))
        shifts.append(0)
    else:
        shifts = list(range(n))

    for s in shifts:
        if s == 0:
            continue
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s

    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {} (n={}, t={}s)".format(label, round(sc, 6), len(a), round(elapsed, 1)))
    return best_seq, best_score
