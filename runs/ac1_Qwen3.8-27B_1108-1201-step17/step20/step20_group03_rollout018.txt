Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FISTA with exact subgradient (top-K active set) and Nesterov averaging
2. Spectral reshaping (iterative amplitude modification)
3. LP-Frank-Wolfe with adaptive active set
4. Cyclic shift optimization
5. Multi-length exploration with interpolation
6. Random restarts with structured initializations

import numpy as np
import time

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def subgradient_topk(a, K=20):
    """Compute subgradient of max(conv(a,a)_k) using top-K active set."""
    n = len(a)
    b = fast_autocorr(a)
    # Find top K indices
    top_idx = np.argsort(b)[::-1][:K]
    b_max = b[top_idx[0]]
    
    # Compute gradient as uniform average over active set
    # grad of b[k] w.r.t. a[i] is a[k-i] for valid i
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            grad[i_start:i_end+1] += a[k - np.arange(i_start, i_end+1)]
    grad /= len(top_idx)
    return grad, b_max, top_idx

def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def fista_exact(a0, time_limit, K=20, restart_every=3000, verbose=True):
    """FISTA with exact subgradient over top-K active set."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(42)
    
    # Backtracking initial step size
    L = 2.0 / n  # Lipschitz constant estimate
    eta = 1.0 / L
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute subgradient
        grad, b_max, top_idx = subgradient_topk(y, K)
        
        # Backtracking line search (simplified: adaptive step)
        # Use a fixed step with occasional adjustment
        z = y - eta * grad
        x_new = simplex_project(z)
        
        # Evaluate
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        # Adaptive step size
        if t % 200 == 0:
            # Try to increase or decrease step
            if stall > 100:
                eta *= 1.1
            elif best_max_b < fast_autocorr(x).max() * 0.99:
                eta = max(eta * 0.9, 1e-4)
        
        # Restart
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.01 / (1.0 + 0.02 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            eta = 2.0 / n
            if verbose:

                J = 2.0 * n * best_max_b
                print("    FISTA restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
        
        if verbose and t % 5000 == 0:

            print("    FISTA t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
    
    return best_a, best_max_b

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def spectral_equalize(a0, time_limit, alpha=0.3, n_iters=None, verbose=True):
    """Iterative spectral equalization: reduce power at frequencies contributing to autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    if n_iters is None:
        n_iters = 500
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        S = np.abs(A) ** 2
        S_max = S.max()
        if S_max < 1e-30:
            break
        
        # Compute circular autocorrelation from spectrum
        c_circ = np.fft.irfft(S, fft_size)
        
        # Get the relevant lags (linear autocorrelation lags 0 to n-1)
        # For circular autocorrelation of length fft_size:
        # lag 0 is at index 0, lag k is at index k (for k < fft_size/2)
        # The linear autocorrelation at lag k (k>=0) is approximately c_circ[k] for small k
        # and c_circ[fft_size - k] for negative k
        # We care about the range that maps to our convolution
        # Conv position k corresponds to lag (k - n + 1) in the standard sense
        # Let's just work with the circular autocorrelation peaks
        
        # Find the top peaks in the circular autocorrelation (excluding DC)
        c_use = c_circ.copy()
        top_k = min(30, n // 2)
        peak_lags = np.argsort(np.maximum(c_circ[:n], c_circ[fft_size-n:fft_size]))[::-1][:top_k]
        
        # Actually, let's work directly with the linear autocorrelation
        b = fast_autocorr(a)
        top_idx = np.argsort(b)[::-1][:top_k]
        
        # For each peak lag k in the convolution, the corresponding "time lag" is (k - (n-1))
        # The contribution to c[k] in freq domain: c[k] = sum_f S[f] * phase
        # To reduce c[k], reduce S at frequencies where the phasor aligns with k
        
        # Multiplicative update: S_new[f] = S[f] * (1 - alpha * weighted_sum_of_cosines)
        freqs = np.arange(len(S))
        weight = np.zeros(len(S))
        for k in top_idx:
            lag = int(k) - (n - 1)
            # cos(2*pi*f*lag/fft_size) tells us the phase alignment
            weight += np.cos(2.0 * np.pi * freqs * lag / fft_size)
        weight /= len(top_idx)
        
        # Scale weight
        w_max = np.abs(weight).max()
        if w_max < 1e-10:
            break
        weight = weight / w_max
        
        # Update spectrum
        S_new = S * np.exp(-alpha * weight)
        S_new = np.maximum(S_new, S * 0.01)  # Don't zero out any frequency completely
        
        # Compute new A (preserve phase)
        phase = np.angle(A)
        A_new = np.sqrt(S_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 100 == 0:
            J = 2.0 * n * best_m
            print("    SpectralEq it=" + str(it) + ": J=" + repr(round(J, 6)) + " currM=" + repr(round(m_new, 8)))
    
    return best_a, best_m

def fista_smoothed(a0, time_limit, beta_start=20.0, beta_end=1000.0,
                   beta_growth=1.004, top_k=20, restart_every=3000,
                   verbose=True, seed=42):
    """FISTA with softmax-smoothed max objective."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    bb_step = 1.0 / np.sqrt(n)
    s_prev = None
    y_prev = None
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        # Compute smoothed gradient
        b = fast_autocorr(y)
        if top_k is not None and top_k < len(b) and top_k > 0:
            idx = np.argsort(b)[::-1][:top_k]
            b_use = np.zeros(len(b))
            b_use[idx] = b[idx]
            b_max = b[idx].max()
        else:
            b_use = b
            b_max = b.max()
        
        shifted = beta * (b_use - b_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        p_total = exp_vals.sum()
        if p_total < 1e-300:
            p = np.zeros_like(b_use)
            mask = (b_max == b_use)
            p[mask] = 1.0 / max(1, mask.sum())
        else:
            p = exp_vals / p_total
        
        # Gradient: conv(p, reversed(y))[n-1:2n-1]
        a_rev = y[::-1]
        conv_len = n + len(p) - 1
        fft_size_g = 1 << (conv_len - 1).bit_length()
        Ar = np.fft.rfft(a_rev, fft_size_g)
        Pr = np.fft.rfft(p, fft_size_g)
        conv = np.fft.irfft(Ar * Pr, fft_size_g)
        grad = conv[n - 1:2 * n - 1] / beta
        
        # Barzilai-Borwein step size
        if s_prev is not None and y_prev is not None:
            sy = y - y_prev
            gy = grad - s_prev
            sy_gy = sy.dot(gy)
            gy_gy = gy.dot(gy)
            if gy_gy > 1e-20:
                bb_candidate = sy_gy / gy_gy
                if 1e-8 < bb_candidate < 10.0:
                    bb_step = 0.7 * bb_step + 0.3 * bb_candidate
                    bb_step = max(1e-6, min(bb_step, 5.0 / beta))
                else:
                    bb_step = max(bb_step * 0.85, 1e-6)
            s_prev = grad.copy()
            y_prev = y.copy()
        
        eta = min(bb_step, 0.15)
        z = y - eta * grad
        x_new = simplex_project(z)
        
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        if t % 80 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.02 / (1.0 + 0.03 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 10.0)
            s_prev = None
            y_prev = None
            bb_step = 1.0 / np.sqrt(n)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    FISTA restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    FISTA t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + repr(round(beta, 1)))
    
    return best_a, best_max_b

def lp_fw(a0, time_limit, solver, initial_K=400, max_K=3000, add_per_iter=30, verbose=True):
    """LP-Frank-Wolfe with adaptive active set."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 600
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K // 2])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check violations
        size = 2 * n - 1
        fft_size_v = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size_v)
        Gf = np.fft.rfft(g_dir, fft_size_v)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size_v)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))
        
        # Line search on alpha
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5),
                  max(0.02, default_alpha * 1.5), 1.0 / (t + 1), default_alpha * 2.0]
        alphas = [max(0.01, min(1.0, a)) for a in alphas]
        alphas = list(set(alphas))
        
        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y
        
        y = y_new
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))
    
    return best_a, best_m

def two_exchange_search(a0, time_limit, verbose=True, seed=999):
    """2-exchange local search."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:40]
        improved = False
        
        for _ in range(80):
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(top_peaks[rng.integers(len(top_peaks))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:20]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src:
                continue
            if a[src] < 1e-12:
                continue
            
            frac = float(rng.choice([0.003, 0.008, 0.015, 0.03, 0.06]))
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-14:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                ac = fast_autocorr(a)
                top_peaks = np.argsort(ac)[::-1][:40]
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.002, 0.04))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if verbose and stall % 150 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    2-exchange (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))
    
    return best_a, best_m

def cyclic_shift_best(a0):
    """Find best cyclic shift."""
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    
    # Coarse-to-fine: try every 7th shift first, then refine
    step = max(1, n // 100)
    best_coarse_shift = 0
    best_coarse_m = best_m
    for s in range(0, n, step):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_coarse_m:
            best_coarse_m = m
            best_coarse_shift = s
    
    # Fine search around best coarse
    lo = max(0, best_coarse_shift - step)
    hi = min(n, best_coarse_shift + step + 1)
    for s in range(lo, hi):
        if s == 0:
            continue
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_initialization(n, kind='raised_cosine'):
    """Create diverse initializations."""
    x = np.linspace(0, 1, n)
    if kind == 'raised_cosine':
        a = np.sin(np.pi * x) ** 2
    elif kind == 'sqrt_sine':
        a = np.sin(np.pi * x)
    elif kind == 'uniform':
        a = np.ones(n)
    elif kind == 'bimodal':
        a = np.exp(-50 * (x - 0.25) ** 2) + np.exp(-50 * (x - 0.75) ** 2)
    elif kind == 'alternating_cos':
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x * 3)
        a = np.maximum(a, 0.01)
    elif kind == 'triangular':
        a = 1.0 - np.abs(2 * x - 1)
    elif kind == 'sinc_like':
        a = np.sinc(3 * (x - 0.5)) ** 2
        a = np.maximum(a, 0.001)
    else:
        a = np.ones(n)
    return norm_sum1(a)

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score
