- `dL/dM = 1 - sum_k μ_k = 0` → `sum_k μ_k = 1`
- `dL/da_i = -sum_k μ_k * (d(a*a)_k/da_i) - ν_i + λ = 0`
- `μ_k >= 0, μ_k * ((a*a)_k - M) = 0`
- `ν_i >= 0, ν_i * a_i = 0`
- `a_i >= 0, (a*a)_k <= M`

So for `a_i > 0` (ν_i = 0): `sum_k μ_k * (d(a*a)_k/da_i) = λ`

This means: the weighted sum of subgradient components (weighted by the dual variables μ_k) should be constant over all active `i`.

If only one constraint is active (say `k*`), then: `μ_{k*} = 1`, and `d(a*a)_{k*}/da_i = λ` for all `i` with `a_i > 0`.

This means: `2 * a_{k* - i} = λ` for all `i` with `a_i > 0` (within the valid range). So `a_{k* - i}` is constant for all `i` where `a_i > 0`. This means the sequence has a "plateau" in the region that contributes to the peak.

If multiple constraints are active, it's more complex but similar.

This gives us a polishing algorithm:
1. Find the active constraints (where `(a*a)_k` is close to max).
2. Compute the "effective subgradient" as a convex combination of the active subgradients.
3. Adjust `a_i` to equalize the effective subgradient over active `i`.

    """Solve: min M s.t. sum(g)=1, g>=0, (y*g)_k <= M for k in active_set"""
    K = len(active_set)
    # Variables: g_0, ..., g_{n-1}, M (total n+1 variables)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        A_ub[jj, idx] = y[k - idx]
        A_ub[jj, n] = -1.0  # -M

def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def solve_lp(y, active_ks, n):
    """Solve LP: min M s.t. sum(g)=1, g>=0, (y*g)_k <= M for k in active_ks"""
    K = len(active_ks)
    if K == 0:
        return np.ones(n) / n
    
    rows = []
    cols = []
    vals = []
    for jj, k in enumerate(active_ks):
        ki = int(k)
        i_lo = max(0, ki - n + 1)
        i_hi = min(n - 1, ki)
        # Vectorized:
        idx = np.arange(i_lo, i_hi + 1)
        rows.extend([jj] * len(idx))
        cols.extend(idx.tolist())
        vals.extend(y[ki - idx].tolist())
        rows.append(jj)
        cols.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(K, n + 1))
    b_ub = np.zeros(K)
    
    # Equality: sum(g) = 1
    eq_data = np.ones(n)
    A_eq = csr_matrix((eq_data, (np.zeros(n, dtype=int), np.arange(n))), shape=(1, n + 1))
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    bounds = [(0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8,
                               'primal_feasibility_tolerance': 1e-8})
    except Exception:
        return None
    
    if res.status != 0:
        return None
    
    g = np.maximum(res.x[:n], 0.0)
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs

def golden_line_search(y, g_dir, curr_M, max_evals=25):
    """Golden-section search for alpha in [0,1] minimizing max_conv((1-alpha)*y + alpha*g_dir)"""
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618
    
    def obj(alpha):
        trial = (1.0 - alpha) * y + alpha * g_dir
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-15:
            return curr_M
        trial = trial / s
        return fast_convolve(trial).max()
    
    a_lo = 0.0
    a_hi = 1.0
    c_ = a_hi - gr * (a_hi - a_lo)
    d_ = a_lo + gr * (a_hi - a_lo)
    fc = obj(c_)
    fd = obj(d_)
    
    for _ in range(max_evals - 2):
        if fd < fc:
            a_lo = c_
            c_ = d_
            fc = fd
            a_hi = a_hi  # keep
            d_ = a_lo + gr * (a_hi - a_lo)
            fd = obj(d_)
        else:
            a_hi = d_
            d_ = c_
            fd = fc
            c_ = a_hi - gr * (a_hi - a_lo)
            fc = obj(c_)
    
    return (a_lo + a_hi) / 2.0
