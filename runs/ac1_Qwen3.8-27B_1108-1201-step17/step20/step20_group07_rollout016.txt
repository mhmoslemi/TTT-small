import time

def fast_autocorr(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    """Project z onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def cross_convolve(a, g, n):
    """Compute (a*g)[k] for k = 0, ..., 2n-2 using FFT."""
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    Af = np.fft.rfft(a, fft_size)
    Gf = np.fft.rfft(g, fft_size)
    return np.fft.irfft(Af * Gf, fft_size)[:size]

def subgradient_with_linesearch(a, time_limit, top_k=50, verbose=True):
    n = len(a)
    a = normalize(a)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        b = fast_autocorr(a)
        # Get top k lags
        top_idx = np.argsort(b)[::-1][:top_k]
        b_max = b[top_idx].max()
        
        # Compute subgradient: for each top lag k, grad_k[j] = a[k-j] (if valid)
        # Weighted by proximity to max
        grad = np.zeros(n)
        for k_i in top_idx:
            k_int = int(k_i)
            # Weight: how close to max
            w = max(0, 1 - (b_max - b[k_int]) / (b_max + 1e-15))
            # Gradient contribution
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                for j in range(i_start, i_end + 1):
                    grad[j] += w * a[k_int - j]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Try different step sizes
        best_t = 0
        best_m_trial = best_m
        for t in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3]:
            a_trial = a - t * grad / grad_norm
            a_trial = simplex_project(a_trial)
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_m_trial:
                best_m_trial = m_trial
                best_t = t
        
        if best_t > 0:
            a = a - best_t * grad / grad_norm
            a = simplex_project(a)
            m_new = fast_autocorr(a).max()
            if m_new < best_m:
                best_m = m_new
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_a, best_m

def fast_autocorr(a):
    """Compute full self-convolution using FFT. a must be non-negative."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project z onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)


def cross_convolve(a, g, n):
    """Compute (a*g)[k] for k = 0, ..., 2n-2 using FFT."""
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    Af = np.fft.rfft(a, fft_size)
    Gf = np.fft.rfft(g, fft_size)
    return np.fft.irfft(Af * Gf, fft_size)[:size]


def subgradient_with_linesearch(a0, time_limit, top_k=30, verbose=True):
    """Subgradient method with exact line search on the projected simplex."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    b = fast_autocorr(a)
    best_m = b.max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 1000
    rng = np.random.default_rng(2024)
    t = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        t += 1
        b = fast_autocorr(a)
        b_max = b.max()
        top_idx = np.argsort(b)[::-1][:top_k]

        # Compute weighted subgradient
        grad = np.zeros(n)
        b_range = b[top_idx].max() - b[top_idx].min() + 1e-30
        for k_i in top_idx:

            w = 1.0 + (b[k_int] - b[top_idx].min()) / b_range
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)

                grad[j_start:j_end + 1] += w * a[k_int - j_end:k_int - j_start + 1][::-1]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            break

        # Line search: try several steps
        best_alpha = 0.0
        best_m_trial = best_m
        base_step = min(0.5, 2.0 / (t + 10))
        alphas = [base_step * f for f in [0.1, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]]

                best_alpha = alpha

            a = simplex_project(a)
            m_new = fast_autocorr(a).max()
            if m_new < best_m - 1e-16:

def peak_chasing(a0, time_limit, verbose=True):
    """Directly target and reduce the top autocorrelation peaks by moving mass."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_autocorr(a)
        top_n = min(30, len(b))
        peak_indices = np.argsort(b)[::-1][:top_n]
        b_max = b[peak_indices[0]]

        # Compute pressure map: which positions contribute most to peaks
        pressure = np.zeros(n)
        for k_i in peak_indices[:15]:
            k_int = int(k_i)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                contrib = a[idx] * a[k_int - idx]
                pressure[idx] += contrib * (b[k_int] / b_max)

        improved = False

        # For each top peak, try to reduce it
        for k_i in peak_indices[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k_i)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start > j_end:
                continue
            idx = np.arange(j_start, j_end + 1)
            contrib = a[idx] * a[k_int - idx]
            max_c = contrib.max()
            if max_c < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            if partner < 0 or partner >= n:
                continue

            # Decide which of src/partner to reduce
            if a[src] >= a[partner]:
                reduce_idx = src
            else:
                reduce_idx = partner

            if a[reduce_idx] < 1e-14:
                continue

            # Find safe destination: low pressure, not involved in peaks
            safe_mask = pressure < np.percentile(pressure, 40)
            safe_mask[reduce_idx] = False
            if safe_mask.sum() < 10:
                safe_mask = pressure < np.percentile(pressure, 70)
                safe_mask[reduce_idx] = False
            if safe_mask.sum() < 5:
                continue

            safe_pressure = pressure.copy()
            safe_pressure[~safe_mask] = np.max(pressure) + 1.0
            dst = int(np.argmin(safe_pressure))

            # Try to move mass
            found = False
            for frac in [0.01, 0.03, 0.06, 0.1, 0.15, 0.25]:
                amt = a[reduce_idx] * frac
                if amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and improved:
                pass  # Continue trying other peaks

        if not improved:
            stall += 1
            # Random exploration when stuck
            if stall % 30 == 0:
                noise_scale = min(0.04, 0.005 * (stall // 30))
                pert = rng.normal(0, noise_scale, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-14)
                a_r = a_r / a_r.sum()
                m_r = fast_autocorr(a_r).max()
                if m_r < best_m:
                    best_m = m_r
                    best_a = a_r.copy()
                a = a_r
                stall = 0

            if verbose and stall % 300 == 0:
                elapsed = time.monotonic() - t0
                print("    PeakChase: J={:.6f} stall={}".format(2.0 * n * best_m, stall))

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """Improved LP-Frank-Wolfe with dynamic active set and exact line search."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    b = fast_autocorr(a)
    best_m = b.max()
    t0 = time.monotonic()
    t = 0
    no_improve = 0
    max_no_improve = 500

    # Initial active set: top 300 lags
    active_set = list(np.argsort(b)[::-1][:300])
    max_active = 3500

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        t += 1
        b = fast_autocorr(a)
        M_curr = b.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Build constraint matrix for active set
        active_set = sorted(active_set)
        K = len(active_set)
        if K == 0:
            break

        # Trim active set if too large: keep most binding
        if K > max_active:
            margins = np.array([M_curr - b[k] for k in active_set])
            keep = np.argsort(margins)[:max_active]
            active_set = [int(active_set[i]) for i in keep]
            active_set = sorted(active_set)
            K = len(active_set)

        # Build dense constraint matrix (K x n)
        A_mat = np.zeros((K, n))
        for jj, k in enumerate(active_set):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                A_mat[jj, j_start:j_end + 1] = a[k_int - j_end:k_int - j_start + 1][::-1]

        # Solve LP: min M s.t. A*g <= M, g >= 0, sum(g) = 1
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A_mat @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations: lags where (a*g)[k] > M_curr
        conv_ag = cross_convolve(a, g_dir, n)
        violations = np.where(conv_ag > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ag[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:min(50, len(violations))]]
            for k in top_v:
                k_int = int(k)
                if k_int not in active_set:
                    active_set.append(k_int)

        # Exact line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, 0.5 * default_alpha, default_alpha * 1.5,
                  0.1, 0.2, 0.3, 0.5]
        alphas = [min(max(a_val, 0.01), 0.99) for a_val in alphas]
        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * a + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        st = a.sum()
        if st > 1e-15:
            a = a / st

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t={} ({:.1f}s): J={:.6f} |S|={}".format(t, elapsed, 2.0 * n * best_m, K))

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate sequence f to a new length new_n."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_sweep(n, time_limit, best_init, best_score):
    """Try parametric families to find a good initial shape."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    best_a = best_init.copy()
    best_val = best_score

    # Raised cosine and power-law shapes
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        for beta_p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            if time.monotonic() - t0 > time_limit:
                break
            a_raw = (x + 0.005) ** alpha * (1.0 - x + 0.005) ** beta_p
            a = normalize(a_raw)
            m = fast_autocorr(a).max()
            if m < best_val:
                best_val = m
                best_a = a.copy()

    # Gaussian
    for center in [0.3, 0.4, 0.5, 0.6]:
        for sigma in [0.1, 0.15, 0.2, 0.25, 0.3]:
            a_raw = np.exp(-((x - center) ** 2) / (2 * sigma ** 2))
            a = normalize(a_raw)
            m = fast_autocorr(a).max()
            if m < best_val:
                best_val = m
                best_a = a.copy()

    # Bimodal
    for c1 in [0.25, 0.35]:
        for c2 in [0.65, 0.75]:
            for s in [0.08, 0.12, 0.15]:
                a_raw = np.exp(-((x - c1) ** 2) / (2 * s ** 2)) + np.exp(-((x - c2) ** 2) / (2 * s ** 2))
                a = normalize(a_raw)
                m = fast_autocorr(a).max()
                if m < best_val:
                    best_val = m
                    best_a = a.copy()

    # Random starts
    rng = np.random.default_rng(42)
    for _ in range(20):
        if time.monotonic() - t0 > time_limit:
            break
        a = normalize(rng.random(n) ** rng.uniform(0.3, 5.0))
        m = fast_autocorr(a).max()
        if m < best_val:
            best_val = m
            best_a = a.copy()

    return best_a, best_val


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a_arr = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a_arr.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a_arr / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {:.6f} (n={}, t={:.1f}s)".format(label, sc, len(a_arr), elapsed))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search: multi-phase optimization."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print("Time budget: {:.1f}s, solver: {}".format(time_budget, solver))

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = normalize(parent)
                best_score = p_score
                print("Parent loaded: n={}, score={:.6f}".format(len(parent), p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        x = np.linspace(0, 1, n_base)
        a_raw = (x + 0.005) ** 1.0 * (1.0 - x + 0.005) ** 1.0
        best_seq = normalize(a_raw)
        best_score = evaluate_safe(best_seq)
        print("Default init: n={}, score={:.6f}".format(len(best_seq), best_score))

    n0 = len(best_seq)

    # Phase 1: Parametric sweep
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Parametric sweep ===")
    t1 = min(remaining * 0.02, 30.0)
    if t1 > 10:
        a_param, _ = parametric_sweep(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "sweep", t_global)

    # Phase 2: LP-Frank-Wolfe (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: LP-Frank-Wolfe (n={}) ===".format(n_best))
    t2 = min(remaining * 0.30, 400.0)
    if t2 > 60:
        a_out, _ = lp_frank_wolfe(best_seq, t2, solver, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw", t_global)

    # Phase 3: Subgradient with line search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Subgradient (n={}) ===".format(n_best))
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 40:
        a_out, _ = subgradient_with_linesearch(best_seq, t3, top_k=40, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "subgrad", t_global)

    # Phase 4: Peak chasing
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Peak chasing (n={}) ===".format(n_best))
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        a_out, _ = peak_chasing(best_seq, t4, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakchase", t_global)

    # Phase 5: Second LP-FW pass (refined)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: LP-FW round 2 (n={}) ===".format(n_best))
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 40:
        a_out, _ = lp_frank_wolfe(best_seq, t5, solver, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2", t_global)

    # Phase 6: Multi-n exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    nearby_lengths = []
    for dn in [-500, -300, -100, 100, 300, 500, 800]:
        nl = n_best + dn
        if 5000 < nl <= 9000:
            nearby_lengths.append(nl)

    print("\n=== Phase 6: Multi-n exploration ===")
    for new_n in nearby_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("  Exploring n={}...".format(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick LP-FW
        t6a = min(remaining * 0.30, 120.0)
        if t6a > 30:
            a_out, _ = lp_frank_wolfe(a_long, t6a, solver, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n{}".format(new_n), t_global)

        # Quick peak chase
        remaining = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining * 0.25, 80.0)
        if t6b > 20:
            a_out, _ = peak_chasing(best_seq, t6b, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "pk_n{}".format(new_n), t_global)

    # Phase 7: Final polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final polish (n={}) ===".format(n_best))

    # Subgradient fine-tune
    t7a = min(remaining * 0.30, 150.0)
    if t7a > 30:
        a_out, _ = subgradient_with_linesearch(best_seq, t7a, top_k=20, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "final_sg", t_global)

    # Peak chase fine-tune
    remaining = time_budget - (time.monotonic() - t_global)
    t7b = min(remaining * 0.30, 120.0)
    if t7b > 20:
        a_out, _ = peak_chasing(best_seq, t7b, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "final_pk", t_global)

    # Final LP burst
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and len(best_seq) <= 8500:
        print("  Final LP burst...")
        t7c = min(remaining * 0.60, 150.0)
        if t7c > 30:
            a_out, _ = lp_frank_wolfe(best_seq, t7c, solver, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_lp", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.6f}, n={}, time={:.1f}s".format(best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence! Using fallback.")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        a_raw = (x + 0.005) ** 1.0 * (1.0 - x + 0.005) ** 1.0
        best_seq = normalize(a_raw)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
