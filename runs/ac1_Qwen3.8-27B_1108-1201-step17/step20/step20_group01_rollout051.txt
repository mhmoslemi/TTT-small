This is a weighted average of the gradients, with weights proportional to `exp(tau * b[k])`. For large tau, this concentrates on the k with the largest b[k].

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_matrix(a, active_list, n):
    K = len(active_list)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj in range(K):
        ki = int(active_list[jj])
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[ki - idx]
    return coeffs

        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr

        if cur_val < best_val - 1e-13:
            best_val = cur_val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive temperature: start moderate, increase if stalled
        k_sel = min(n_top, len(ac))
        idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = ac[idx]
        base_temp = 20.0 + 5.0 * (t / 2000.0)
        if stall > 2000:
            base_temp += 10.0 * (stall / 5000.0)
        w = np.exp(base_temp * (top_vals - top_vals[0]))
        w = w / w.sum()

        # Build full weight vector for FFT gradient
        w_full = np.zeros(size)
        for i in range(k_sel):
            ki = int(idx[i])
            if 0 <= ki < size:
                w_full[ki] = w[i]

        # Compute gradient
        flipped_w = w_full[::-1]
        A_utc = np.fft.rfft(x, fft_size_full)
        W_utc = np.fft.rfft(flipped_w, fft_size_full)
        conv_res = np.fft.irfft(A_utc * W_utc, fft_size_full)
        grad = conv_res[2 * n - 2 : n - 2 : -1]

        # Mirror descent with adaptive LR
        lr = lr0 / (1.0 + 0.003 * t)
        if stall > 5000:
            lr *= 0.5
        log_x = np.log(x + 1e-30) - lr * grad
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        x_new = np.maximum(x_new, 1e-25)
        x_new = x_new / x_new.sum()
        x = x_new

        # Periodic restart with noise
        if stall > 4000 and stall % 1500 == 0:
            noise = 0.015 * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x = x_r / x_r.sum()
            stall = 0

        if verbose and t % 500 == 0:
            print("    GradRef t={} ({:.1f}s) J={:.7f} stall={}".format(
                t, time.monotonic() - t0, best_val, stall))

    return best_a, best_val

def coord_peel(a0, time_limit, n_targets=15, seed=42, verbose=True):
    """Aggressive coordinate peeling: redistribute mass from peak contributors."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 3000
    fracs = [0.0005, 0.001, 0.002, 0.004, 0.007, 0.012, 0.02, 0.035]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(n_targets, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            ws = max(0, ki)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-25:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-15:
                continue

            # Find low-pressure positions
            pr = np.zeros(n, dtype=np.float64)
            for kk in top_k[:10]:
                kki = int(kk)
                ws2 = max(0, kki)
                we2 = min(n - 1, kki)
                if ws2 > we2:
                    continue
                idx2 = np.arange(ws2, we2 + 1)
                pr[idx2] += a[kki - idx2]
            pr[src] = 1e10
            quiet_positions = np.argsort(pr)[:40]

            for dst in quiet_positions:
                dst = int(dst)
                if dst == src:
                    continue
                found = False
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-18:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-16:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break

            if not improved and stall > 200:
                # Random perturbation
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.001, 0.008)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
                            improved = True
                break

        if not improved:
            stall += 1
            if stall > 300:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 800 == 0 and stall > 0:
            print("    Peel ({:.1f}s) J={:.7f} stall={}".format(
                time.monotonic() - t0, 2 * n * best_m, stall))

    return best_a, 2.0 * n * best_m
