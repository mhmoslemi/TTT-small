import numpy as np
import time

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(z):
    """Project onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1) / (rho[-1] + 1.0)
    return np.maximum(z - theta, 0.0)

def evaluate_safe(a):

def smooth_max_grad(b, beta, top_k=None):
    """Compute gradient of smoothed max of b, and the value."""
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros_like(b)
        b_use[idx] = b[idx]
    else:
        b_use = b
    
    b_max = b_use.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        return b_max, np.zeros_like(b)
    p = exp_vals / total
    value = b_max + np.log(total) / beta
    return value, p

    grad = (2.0 / beta) * conv[n-1 : 2*n-1]

def _proj_simplex(z):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cs = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cs - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cs[rho[-1]] - 1.0) / (rho[-1] + 1.0)
    return np.maximum(z - theta, 0.0)
