```python
"""
Search for non-negative step heights minimizing
    2 * n * max(convolve(a,a)) / sum(a)^2.

Main ideas:
1. Work on the simplex sum(a)=1 with FFT-based linear convolution.
2. Primary global/refinement operator: project a desired flatter convolution
   c into time domain by taking a spectral square root:
       A <- (1-beta)*FFT(a) + beta*sqrt(FFT(target_c)).
   Here target_c is a water-filled version of the current convolution,
   capped at a peak level corresponding to a target score.
3. Secondary local optimizer: active-peak coordinate descent.  The exact
   one-coordinate update is evaluated using a small top-peak surrogate, then
   confirmed by an O(n) convolution update.
4. Multi-length homotopy: interpolate the parent/best sequence to several
   lengths and keep the best result.

The code is deliberately self-contained and avoids filesystem/network I/O.
"""

import numpy as np
import time


def pow2_len(m):
    """Smallest power of two >= m."""
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def norm_sum1(a):
    """Clip to nonnegative and normalize sum to 1."""
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) / float(len(a))
    return a / s


def fast_conv(a):
    """Linear convolution a*a using FFT, clipped to nonnegative values."""
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    L = pow2_len(size)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:size]
    return np.maximum(c, 0.0)


def quick_score(a):
    """Fast internal score, equal to evaluate_sequence() up to roundoff."""
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return float("inf")
    s = float(a.sum())
    if s < 0.01 / max(1, len(a)):
        return float("inf")
    n = len(a)
    c = fast_conv(a)
    if c.size == 0:
        return float("inf")
    return float(2.0 * n * float(np.max(c)) / (s * s))


def interp_sequence(a, n_new):
    """Interpolate a nonnegative sequence to a new length via sample CDF."""
    a = norm_sum1(a)
    n = len(a)
    n_new = int(max(1, n_new))
    if n == 0:
        return np.ones(n_new, dtype=np.float64) / float(n_new)
    old_x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    new_x = (np.arange(n_new, dtype=np.float64) + 0.5) / float(n_new)
    v = np.interp(new_x, old_x, a)
    return norm_sum1(v)


def waterfill_target(c, level):
    """
    Build a target convolution with sum 1 and peak approximately <= level.

    It first caps current peaks at level, then redistributes the missing mass
    among entries with remaining capacity without exceeding level.
    """
    t = np.asarray(c, dtype=np.float64).copy()
    if t.ndim != 1:
        t = t.ravel()
    t = np.where(np.isfinite(t), t, 0.0)
    t = np.maximum(t, 0.0)
    s = float(t.sum())
    size = len(t)
    if size == 0:
        return np.array([], dtype=np.float64)
    if s < 1e-15:
        return np.ones(size, dtype=np.float64) / float(size)

    t = t / s
    level = float(level)
    if level <= 0.0:
        level = 1.0 / float(size)

    if level >= float(np.max(t)):
        return t

    above = t > level
    if np.any(above):
        t[above] = level
        deficit = 1.0 - float(t.sum())
        if deficit > 1e-12:
            cap = level - t
            cap = np.maximum(cap, 0.0)
            cap_sum = float(cap.sum())
            if cap_sum > 1e-12:
                t = t + cap * (deficit / cap_sum)
            else:
                t[:] = 1.0 / float(size)

    t = np.maximum(t, 0.0)
    s = float(t.sum())
    if s < 1e-15:
        t[:] = 1.0 / float(size)
        s = 1.0
    else:
        t = t / s

    # One final cleanup if normalization pushed the maximum slightly above cap.
    if float(size) * level >= 1.0 and float(np.max(t)) > level * (1.0 + 1e-8):
        t = np.minimum(t, level)
        s2 = float(t.sum())
        if s2 > 1e-15:
            t = t / s2
    return t


def get_parent():
    """Return normalized global parent sequence if available."""
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        p = np.asarray(seq, dtype=np.float64)
    except Exception:
        return None
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    p = np.minimum(p, 1000.0)
    s = float(p.sum())
    if s < 1e-8:
        return None
    return p / s


def default_sequence(n):
    """Smooth bump fallback if no parent is available."""
    n = int(max(1, n))
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    v = np.sin(np.pi * x) ** 2
    return norm_sum1(v)


def top_indices(arr, k):
    """Indices of the k largest entries."""
    arr = np.asarray(arr, dtype=np.float64)
    n = len(arr)
    if n == 0:
        return np.array([], dtype=int)
    k = int(min(k, n))
    if k <= 0:
        return np.array([], dtype=int)
    return np.argsort(arr)[::-1][:k].astype(int)


def select_coordinates(a, c, max_peaks=25, per_peak=6, high_count=25):
    """Choose promising coordinates from active convolution peaks."""
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = len(a)
    if n == 0:
        return []
    P = top_indices(c, max_peaks)
    coords = set()

    for k in P:
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n - 1, k)
        if i0 > i1:
            continue
        count = i1 - i0 + 1
        if count <= 2 * per_peak:
            for i in range(i0, i1 + 1):
                j = k - i
                coords.add(i)
                coords.add(j)
        else:
            idx = np.arange(i0, i1 + 1, dtype=int)
            contrib = a[idx] * a[k - idx]
            if contrib.size:
                top = np.argsort(contrib)[::-1][:per_peak]
                for p in top:
                    ii = int(idx[p])
                    jj = int(k - ii)
                    coords.add(ii)
                    coords.add(jj)

    high = top_indices(a, high_count)
    for i in high:
        coords.add(int(i))

    out = []
    for i in coords:
        if 0 <= i < n:
            out.append(int(i))
    return out


def candidate_x_values(old, n):
    """Candidate values for a one-coordinate move."""
    size = 2 * int(n) - 1
    avg = 1.0 / float(size)
    scale = max(1e-6, float(np.sqrt(avg)), 1.0 / float(n))

    vals = [0.0]
    for ex in np.linspace(-2.0, 2.0, 13):
        vals.append(scale * (10.0 ** float(ex)))
    vals.extend([1.0 / n, 2.0 / n, 5.0 / n, 10.0 / n])

    old = float(old)
    if old > 1e-12:
        vals.append(old)
        for ex in np.linspace(-2.0, 2.0, 21):
            vals.append(old * (10.0 ** float(ex)))

    arr = np.array(vals, dtype=np.float64)
    arr = arr[np.isfinite(arr)]
    arr = arr[(arr >= 0.0) & (arr <= 1.0)]
    if arr.size == 0:
        arr = np.array([0.0], dtype=np.float64)
    return np.unique(arr)


def spectral_target_refine(a0, time_limit=120.0, min_score=1.10, verbose=True, seed=0):
    """
    Iteratively project flatter target convolutions back to nonnegative time
    domain through a damped spectral square root.
    """
    a = norm_sum1(a0)
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64), float("inf")
    size = 2 * n - 1
    L = pow2_len(size)

    A0 = np.fft.rfft(a, L)
    c = np.fft.irfft(A0 * A0, L)[:size]
    c = np.maximum(c, 0.0)
    M = float(np.max(c)) if c.size else 0.0
    best = a.copy()
    best_score = float(2.0 * n * M)

    if time_limit <= 0.0 or best_score < min_score:
        return best, best_score

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    last_print = t0

    target_score = max(min_score, best_score - 0.02)
    betas = [0.02, 0.05, 0.10, 0.18, 0.30, 0.45, 0.65, 1.00]
    fail = 0

    while time.monotonic() - t0 < time_limit:
        if verbose and time.monotonic() - last_print > 12.0:
            elapsed = time.monotonic() - t0
            print(
                "    spectral: score=" + repr(round(best_score, 6)) +
                " target=" + repr(round(target_score, 6)) +
                " fail=" + str(fail) +
                " t=" + repr(round(elapsed, 1)) + "s"
            )
            last_print = time.monotonic()

        # Recompute current FFT/convolution.
        A0 = np.fft.rfft(a, L)
        c = np.fft.irfft(A0 * A0, L)[:size]
        c = np.maximum(c, 0.0)
        M = float(np.max(c)) if c.size else 0.0
        current_score = float(2.0 * n * M)

        # Make target strictly lower than current score.
        if target_score > current_score - 1e-5:
            target_score = max(min_score, current_score - 0.01)
        target_score = max(target_score, min_score)

        # Desired peak in normalized convolution.
        level = target_score / (2.0 * float(n))
        min_level = (1.0 + 1e-12) / float(size)
        if level < min_level:
            level = min_level

        target = waterfill_target(c, level)
        Ct = np.fft.rfft(target, L)
        Ct[0] = 1.0

        At = np.sqrt(Ct + 0j)

        # Choose branch signs to stay close to current FFT.
        prod = A0 * At
        re = prod.real
        signs = np.where(re >= 0.0, 1.0, -1.0)
        signs = np.where(np.isfinite(signs), signs, 1.0)
        At = At * signs
        At[0] = 1.0

        # Nyquist must be real for a real time sequence.
        if L % 2 == 0:
            mag = float(np.sqrt(abs(float(np.real(Ct[-1])))))
            if mag == 0.0:
                At[-1] = 0.0
            else:
                At[-1] = mag if (float(np.real(A0[-1])) * mag) >= 0.0 else -mag

        improved = False
        for beta in betas:
            if time.monotonic() - t0 >= time_limit:
                break
            Atr = (1.0 - float(beta)) * A0 + float(beta) * At
            Atr[0] = 1.0
            if L % 2 == 0:
                Atr[-1] = float(np.real(Atr[-1]))

            a_raw = np.fft.irfft(Atr, L)[:n]
            a_trial = np.maximum(a_raw, 0.0)
            s = float(a_trial.sum())
            if s < 1e-6:
                continue
            a_trial = a_trial / s

            Af = np.fft.rfft(a_trial, L)
            ctr = np.fft.irfft(Af * Af, L)[:size]
            ctr = np.maximum(ctr, 0.0)
            Mtr = float(np.max(ctr)) if ctr.size else 0.0
            scr = float(2.0 * n * Mtr)

            if scr < best_score - 1e-11:
                a = a_trial
                best = a_trial.copy()
                best_score = scr
                improved = True

        if improved:
            target_score = max(min_score, target_score - 0.01)
            fail = 0
        else:
            fail += 1
            if fail <= 3:
                # Relax the target a little.
                target_score = min(current_score - 1e-4,
                                   max(min_score, target_score + 0.02))
            elif fail <= 6:
                # Small phase perturbation.
                Ap = np.fft.rfft(a, L)
                phase = 0.025 * rng.standard_normal(len(Ap))
                Ap[1:] *= np.exp(1j * phase[1:])
                if L % 2 == 0:
                    Ap[-1] = float(np.real(Ap[-1]))
                ap = np.fft.irfft(Ap, L)[:n]
                ap = np.maximum(ap, 0.0)
                s = float(ap.sum())
                if s > 1e-6:
                    a = ap / s
                target_score = max(min_score, min(current_score - 1e-4,
                                                  target_score + 0.01))
                fail = 0
            elif fail <= 9:
                # Small additive noise in time.
                noise = 0.012 * rng.standard_normal(n)
                at = a + noise
                at = np.maximum(at, 0.0)
                s = float(at.sum())
                if s > 1e-6:
                    a = at / s
                target_score = max(min_score, min(current_score - 1e-4,
                                                  target_score - 0.005))
                fail = 0
            else:
                # Restart around current best with stronger noise.
                noise = 0.035 * rng.standard_normal(n)
                at = best * (1.0 + noise)
                at = np.maximum(at, 0.0)
                s = float(at.sum())
                if s > 1e-6:
                    a = at / s
                else:
                    a = best.copy()
                target_score = max(min_score, current_score - 0.02)
                fail = 0

    c = fast_conv(best)
    M = float(np.max(c)) if c.size else 0.0
    return best, float(2.0 * n * M)


def coordinate_refine(a0, time_limit=45.0, verbose=False, seed=1):
    """
    Local active-peak coordinate descent.

    For coordinate i, c_k(a) depends affinely on x=a_i except k=2i, where it
    is quadratic.  Candidate x values are selected using the current top peaks
    and then verified by an exact O(n) convolution update.
    """
    a = norm_sum1(a0)
    n = len(a)
    if n == 0:
        return a, float("inf")
    size = 2 * n - 1
    c = fast_conv(a)
    M = float(np.max(c)) if c.size else 0.0
    best = a.copy()
    best_score = float(2.0 * n * M)
    cur_score = best_score

    t0 = time.monotonic()
    last_print = t0
    rng = np.random.default_rng(seed)

    P = top_indices(c, 50)
    coords = select_coordinates(a, c)
    perm = rng.permutation(len(coords)).astype(int) if len(coords) else np.array([], dtype=int)
    ci = 0
    recompute_counter = 0
    stale = 0
    improves_since_recompute = 0

    while time.monotonic() - t0 < time_limit:
        if verbose and time.monotonic() - last_print > 12.0:
            print(
                "    coordinate: best=" + repr(round(best_score, 6)) +
                " cur=" + repr(round(cur_score, 6)) +
                " t=" + repr(round(time.monotonic() - t0, 1)) + "s"
            )
            last_print = time.monotonic()

        if ci >= len(perm) or len(perm) == 0:
            coords = select_coordinates(a, c)
            if len(coords) == 0:
                break
            perm = rng.permutation(len(coords)).astype(int)
            ci = 0
            recompute_counter = 0

        i = int(coords[perm[ci]])
        ci += 1
        recompute_counter += 1

        if recompute_counter > 120 or improves_since_recompute > 18:
            P = top_indices(c, 50)
            coords = select_coordinates(a, c)
            perm = rng.permutation(len(coords)).astype(int) if len(coords) else np.array([], dtype=int)
            ci = 0
            recompute_counter = 0
            improves_since_recompute = 0

        if stale > 500:
            P = top_indices(c, 50)
            coords = select_coordinates(a, c)
            perm = rng.permutation(len(coords)).astype(int) if len(coords) else np.array([], dtype=int)
            ci = 0
            stale = 0
            recompute_counter = 0

        if len(P) == 0:
            break

        old = float(a[i])

        # Build surrogate objective over top peaks.
        P = P.astype(int, copy=False)
        base = c[P].astype(np.float64).copy()
        coeff = np.zeros_like(base)
        quad = np.zeros_like(base)

        for pos, k in enumerate(P):
            k = int(k)
            j = k - i
            if 0 <= j < n:
                if k == 2 * i:
                    base[pos] -= old * old
                    coeff[pos] = 0.0
                    quad[pos] = 1.0
                else:
                    base[pos] -= 2.0 * old * float(a[j])
                    coeff[pos] = 2.0 * float(a[j])
                    quad[pos] = 0.0

        xs = candidate_x_values(old, n)
        if xs.size <= 1:
            stale += 1
            continue

        denom_arg = 1.0 - old + xs
        denom = denom_arg * denom_arg
        denom = np.where(denom > 1e-24, denom, 1e-24)

        with np.errstate(divide="ignore", invalid="ignore", over="ignore"):
            vals = base[:, None] + coeff[:, None] * xs[None, :] + quad[:, None] * xs[None, :] * xs[None, :]
            vals = np.maximum(vals, 0.0)
            mmax = np.max(vals, axis=0)
            obj = mmax / denom

        obj = np.where(np.isfinite(obj), obj, np.inf)
        if obj.size == 0:
            stale += 1
            continue

        best_pos = int(np.argmin(obj))
        if not np.isfinite(obj[best_pos]) or obj[best_pos] >= cur_score - 1e-12:
            stale += 1
            continue

        x = float(xs[best_pos])
        if abs(x - old) < 1e-14:
            stale += 1
            continue

        delta = x - old
        s = 1.0 - old + x
        if s <= 1e-14:
            stale += 1
            continue

        # Exact O(n) update of convolution for this coordinate.
        new_c = c.copy()
        k_arr = i + np.arange(n, dtype=int)
        new_c[k_arr] += 2.0 * a * delta
        diag = 2 * i
        if diag < size:
            new_c[diag] += delta * delta

        new_c = new_c / (s * s)
        new_c = np.maximum(new_c, 0.0)
        Mnew = float(np.max(new_c)) if new_c.size else 0.0
        scor = float(2.0 * n * Mnew)

        if scor < cur_score - 1e-12:
            old_a = a.copy()
            old_c = c.copy()
            old_score = cur_score

            a_new = a.copy()
            a_new[i] = x
            a_new = a_new / s
            a_new = np.maximum(a_new, 0.0)
            s2 = float(a_new.sum())
            if s2 <= 1e-12:
                stale += 1
                continue
            a_new = a_new / s2

            c_new = fast_conv(a_new)
            Mnew_check = float(np.max(c_new)) if c_new.size else 0.0
            cur_check = float(2.0 * n * Mnew_check)

            if cur_check < old_score - 1e-14:
                a = a_new
                c = c_new
                cur_score = cur_check
                improves_since_recompute += 1
                stale = 0
                if cur_check < best_score - 1e-12:
                    best = a_new.copy()
                    best_score = cur_check
                    if verbose:
                        print(
                            "      coord improve: best=" + repr(round(best_score, 6)) +
                            " t=" + repr(round(time.monotonic() - t0, 1)) + "s"
                        )
            else:
                stale += 1
        else:
            stale += 1

    c = fast_conv(best)
    M = float(np.max(c)) if c.size else 0.0
    return best, float(2.0 * n * M)


def try_update_best(best_seq, best_score, a, label, t0):
    """Evaluate a candidate and update the global best if it improves it."""
    if a is None:
        return best_seq, best_score
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return best_seq, best_score
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = quick_score(a)
    if np.isfinite(sc) and sc < best_score:
        best_seq = a.copy()
        best_score = float(sc)
        elapsed = time.monotonic() - t0
        print(
            "  NEW BEST [" + label + "] score=" + repr(round(best_score, 6)) +
            " n=" + str(len(a)) + " t=" + repr(round(elapsed, 1)) + "s"
        )
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Entry point invoked by the evaluator.
    Returns a list of nonnegative heights with improved score during budget.
    """
    t0 = time.monotonic()
    budget = float(budget_s)
    rng = np.random.default_rng(123456)

    parent = get_parent()
    if parent is not None:
        base = norm_sum1(parent)
        n0 = len(base)
        print("Parent sequence: n=" + str(n0) + ", score=" + repr(round(quick_score(base), 6)))
    else:
        n0 = 8000
        base = default_sequence(n0)
        print("No parent; using default bump n=" + str(n0))

    best_seq = base.copy()
    best_score = quick_score(base_seq)

    # Limit length so external O(n^2) evaluation remains practical.
    max_n = min(12000, max(6000, int(n0 * 1.5)))

    cand_arrays = []
    cand_names = []

    cand_arrays.append(base.copy())
    cand_names.append("base")

    lengths = []
    for f in [0.85, 1.0, 1.12, 1.28, 1.45]:
        l = int(round(n0 * f))
        if 600 <= l <= max_n and l not in lengths:
            lengths.append(l)
    lengths.sort()

    for l in lengths:
        a = interp_sequence(base, l)
        cand_arrays.append(a)
        cand_names.append("len" + str(len(a)))

    # One modestly randomized candidate for diverse attraction basins.
    rnd = base * (1.0 + 0.05 * rng.standard_normal(len(base)))
    rnd = norm_sum1(np.maximum(rnd, 0.0))
    cand_arrays.append(rnd)
    cand_names.append("rand")

    scores = []
    for a in cand_arrays:
        scores.append(quick_score(a))
    scores = np.asarray(scores, dtype=np.float64)
    order = np.argsort(scores).astype(int)

    for idx in order:
        best_seq, best_score = try_update_best(best_seq, best_score, cand_arrays[idx], cand_names[idx], t0)

    # Main search loop: alternate spectral refine and local coordinate polish.
    max_rounds = 30
    round_no = 0
    while time.monotonic() - t0 < budget - 70.0 and round_no < max_rounds:
        elapsed = time.monotonic() - t0
        remaining = budget - elapsed
        if remaining < 70.0:
            break
        round_no += 1

        if round_no == 1:
            source = best_seq.copy()
        elif round_no <= len(cand_arrays) and len(cand_arrays) > 1:
            # Use a precomputed diversified candidate for early rounds.
            source = cand_arrays[order[round_no - 1]].copy()
        else:
            # Mostly continue from the global best, occasionally resample length.
            if rng.random() < 0.72:
                source = best_seq.copy()
            else:
                lidx = int(rng.integers(len(cand_arrays)))
                target_len = len(cand_arrays[lidx])
                source = interp_sequence(best_seq, target_len).copy()

        t_spec = min(160.0, max(15.0, remaining * 0.55))
        min_score = max(1.05, best_score - 0.25)
        a_spec, _ = spectral_target_refine(
            source,
            time_limit=t_spec,
            min_score=min_score,
            verbose=True,
            seed=1000 + round_no * 17
        )
        best_seq, best_score = try_update_best(best_seq, best_score, a_spec, "spec" + str(round_no), t0)

        elapsed = time.monotonic() - t0
        remaining = budget - elapsed
        t_coord = min(55.0, max(8.0, remaining * 0.35))
        if t_coord > 5.0 and best_seq is not None and budget - elapsed > 65.0:
            a_coord, _ = coordinate_refine(
                best_seq,
                time_limit=t_coord,
                verbose=False,
                seed=5000 + round_no * 23
            )
            best_seq, best_score = try_update_best(best_seq, best_score, a_coord, "coord" + str(round_no), t0)

    # Final polishing pass.
    elapsed = time.monotonic() - t0
    if best_seq is not None and elapsed < budget - 45.0:
        t_final = min(110.0, budget - elapsed - 25.0)
        if t_final > 15.0:
            a_final, _ = spectral_target_refine(
                best_seq,
                time_limit=t_final,
                min_score=max(1.05, best_score - 0.10),
                verbose=True,
                seed=99999
            )
            best_seq, best_score = try_update_best(best_seq, best_score, a_final, "final_spec", t0)

    elapsed = time.monotonic() - t0
    if best_seq is not None and elapsed < budget - 25.0:
        t_polish = min(45.0, budget - elapsed - 15.0)
        if t_polish > 8.0:
            a_polish, _ = coordinate_refine(
                best_seq,
                time_limit=t_polish,
                verbose=False,
                seed=82828
            )
            best_seq, best_score = try_update_best(best_seq, best_score, a_polish, "final_coord", t0)

    # Sanitize final sequence.
    final = np.asarray(best_seq, dtype=np.float64)
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.maximum(final, 0.0)
    final = np.minimum(final, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(max(1000, min(6000, n0)), dtype=np.float64) / 1000.0
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(1000, dtype=np.float64) / 1000.0
            s = 1.0
    else:
        final = final / s

    elapsed = time.monotonic() - t0
    fs = quick_score(final)
    print(
        "FINAL internal score=" + repr(round(fs, 6)) +
        " n=" + str(len(final)) +
        " time=" + repr(round(elapsed, 1)) + "s"
    )

    # Optional direct evaluation only when length and time margin are safe.
    if len(final) <= 9000 and elapsed < budget - 12.0:
        try:
            direct = evaluate_sequence([float(x) for x in final])
            print("Direct evaluate_sequence score=" + repr(round(float(direct), 8)))
        except Exception:
            pass

    return [float(x) for x in final]
```
