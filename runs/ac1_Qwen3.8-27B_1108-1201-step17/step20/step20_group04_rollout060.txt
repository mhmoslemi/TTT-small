    cssv = np.cumsum(u)

    # Build constraint matrix A: A[i, j] = f[peak_idx[i] - j] if 0 <= peak_idx[i] - j < n
    # Shape: (top_k, n)
    A = lil_matrix((top_k, n))
    for i, k in enumerate(peak_idx):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            A[i, j_start:j_end+1] = f[k_int - j_end : k_int - j_start + 1][::-1]

    rhs = c[peak_idx]
    
    try:

        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= rhs]
        objective = cp.Maximize(cp.sum(g))
        prob = cp.Problem(objective, constraints)

        else:

def interpolate_sequence(f, new_n):

    if old_n == new_n:
        return f.copy()

    if s < 1e-15:

    old_x = np.linspace(0.0, 1.0, old_n)

def compute_smoothed_grad(y, beta, top_k=None):
    n = len(y)
    c = fast_autocorr(y)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
    else:
        c_use = c
    c_max = c_use.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0

        p = exp_vals / total

            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()
            m_try = fast_autocorr(a_try).max()
            a = a_try
            if m_try < best_m:
                best_m = m_try
                best_a = a.copy()
            stall = 0

        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-Descent it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def peak_surgery(a0, time_limit, verbose=True, seed=77):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 2000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_top = min(20, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]
        improved = False

        for k in top_idx[:8]:
            if improved:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            if products.max() < 1e-20:
                continue
            src_i = int(idx[np.argmax(products)])
            partner = k_int - src_i
            if partner < 0 or partner >= n:
                continue
            if a[src_i] < 1e-14:
                continue

            reduce_i = src_i if a[src_i] >= a[partner] else partner
            if a[reduce_i] < 1e-14:
                continue

            # Find low-pressure destination
            pressure = np.zeros(n)
            for k2 in top_idx[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
