    """Solve the LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    n = len(f)
    # Constraint: (f*g)_k <= M for k = 0, ..., 2n-2
    # (f*g)_k = sum_{i} f[i] * g[k-i] where 0<=i<n and 0<=k-i<n
    #         = sum_{j} f[k-j] * g[j] where 0<=j<n and 0<=k-j<n
    # So the constraint matrix A has A[k,j] = f[k-j] if 0<=k-j<n, else 0
    # This is a (2n-1) x n matrix with banded structure
    
    # For scipy linprog: min -sum(g) s.t. A@g <= M, g >= 0
    # A[k,j] = f[k-j] if 0 <= k-j <= n-1 else 0
    
    # This matrix is large. Let's use sparse.
    from scipy.sparse import lil_matrix, csc_matrix
    
    A = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        js = np.arange(j_start, j_end+1)
        A[k, js] = f[k - js]
    A = A.tocsr()
    
    c = -np.ones(n)  # minimize -sum(g)
    b = np.full(2*n-1, M)
    
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.status == 0:
        return res.x
    return None

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe (FW-LP) with constraint generation: principled global descent
2. Nesterov acceleration on FW-LP iterates
3. Smoothed max gradient descent for refinement
4. Peak surgery for targeted local improvements
5. Multi-length exploration and interpolation
Key innovation: LP-FW provides descent directions that see the full constraint landscape,
escaping local minima that trap pure gradient methods.

import numpy as np
import time
import scipy.optimize
from scipy.sparse import coo_matrix, csr_matrix


def fast_autocorr(a):
    """Compute full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    """Compute gradient of smoothed max objective w.r.t. a."""
    n = len(y)
    c = fast_autocorr(y)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    fft_size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A_rev = np.fft.rfft(y[::-1], fft_size)
    Conv = np.fft.irfft(P * A_rev, fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad, p, c

        k_int = int(k)
        j_start = max(0, k_int - (n - 1))

        js = np.arange(j_start, j_end + 1)
        row_list.extend([i] * len(js))
        col_list.extend(js.tolist())
        val_list.extend(f[k_int - js].tolist())

    A = coo_matrix((val_list, (row_list, col_list)), shape=(nc, n)).tocsr()
    c_obj = -np.ones(n)
    b = np.full(nc, M)
    bounds = [(0.0, None)] * n

    try:
        res = scipy.optimize.linprog(
            c_obj, A_ub=A, b_ub=b, bounds=bounds,
            method='highs',
            options={'presolve': True, 'time_limit': lp_time_limit, 'dual_feasibility_tolerance': 1e-7}
        )
        if res.status in (0, 2) and res.x is not None:
            return np.asarray(res.x, dtype=np.float64)
    except Exception:
        pass
    return None


def lp_fw_optimize(a0, time_limit, verbose=True, seed=42, max_constraints=1500,
                   nesterov=True, lp_time_limit=10):
    """LP-based Frank-Wolfe with constraint generation and optional Nesterov acceleration."""
    n = len(a0)
    f = norm_sum1(a0)
    best_f = f.copy()
    best_M = fast_autocorr(f).max()
    t0 = time.monotonic()

    f_nest = f.copy()
    f_prev = f.copy()
    t_nest = 1.0
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov extrapolation
        if nesterov and it > 0:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
            theta = (t_nest - 1.0) / t_new
            y = f_nest + theta * (f_nest - f_prev)
            y = np.maximum(y, 0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = f_nest.copy()
            else:
                y = y / s_y
        else:
            y = f_nest.copy()
            t_new = t_nest + 1

        M_y = fast_autocorr(y).max()

        # Determine active constraints (top peaks)
        c_y = fast_autocorr(y)
        n_top = min(max_constraints, 2 * n - 1)
        active_idx = np.argsort(c_y)[::-1][:n_top]

        # Solve LP
        g = lp_fw_single_step(y, M_y, active_idx, lp_time_limit=lp_time_limit)

        if g is None:
            # Fallback: use y itself as direction
            g = y.copy()
        else:
            g = np.maximum(g, 0)
            s_g = g.sum()
            if s_g < 1e-15:
                g = y.copy()
            else:
                g = g / s_g

        # Constraint generation: check if g violates other constraints
        c_fg = fast_autocorr(y)  # placeholder, we need f*g not f*f
        # Actually compute y*g convolution
        fft_size = 1 << (2 * n - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]
        violations = conv_yg - M_y
        if violations.max() > M_y * 0.01:
            # Add most violated constraints
            viol_idx = np.where(violations > M_y * 0.005)[0]
            active_idx = np.unique(np.concatenate([active_idx, viol_idx[:200]]))
            active_idx = active_idx[:max_constraints * 2]
            g2 = lp_fw_single_step(y, M_y, active_idx, lp_time_limit=lp_time_limit)
            if g2 is not None:
                g2 = np.maximum(g2, 0)
                s_g2 = g2.sum()
                if s_g2 > 1e-15:
                    g = g2 / s_g2

        # Line search for step size
        current_obj = fast_autocorr(f_nest).max()
        best_t = 0.0
        best_obj_ls = current_obj
        for t_try in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1]:
            f_trial = (1.0 - t_try) * f_nest + t_try * g
            f_trial = np.maximum(f_trial, 0)
            s_t = f_trial.sum()
            if s_t < 1e-15:
                continue
            f_trial = f_trial / s_t
            obj_try = fast_autocorr(f_trial).max()
            if obj_try < best_obj_ls - 1e-15:
                best_obj_ls = obj_try
                best_t = t_try

        if best_t == 0.0:
            best_t = 0.02
        f_new = (1.0 - best_t) * f_nest + best_t * g
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            f_new = f_nest.copy()
        else:
            f_new = f_new / s_new

        M_new = fast_autocorr(f_new).max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1

        f_prev = f_nest
        f_nest = f_new
        t_nest = t_new
        it += 1

        if stall > 30:
            # Reset Nesterov
            t_nest = 1.0
            f_prev = f_nest.copy()
            stall = 0
            # Small perturbation
            rng = np.random.default_rng(seed + it)
            noise = rng.normal(0, 0.01, n) * f_nest
            f_nest = f_nest + noise
            f_nest = np.maximum(f_nest, 0)
            f_nest = f_nest / f_nest.sum()

        if verbose and (it % 20 == 0 or it < 3):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print("  LP-FW it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" +
                  repr(round(J, 6)) + " M=" + repr(round(best_M, 12)) + " t=" + repr(round(best_t, 3)))

    return best_f, best_M


def nesterov_opt(a0, time_limit, beta_start=30.0, beta_end=2000.0,
                 beta_growth=1.006, eta_base=0.6, top_k_ratio=0.02,
                 restart_every=2000, verbose=True, seed=42):
    """Nesterov accelerated projected gradient on smoothed objective."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, c = compute_smoothed_grad(y, beta, top_k)

        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.03 / (1.0 + 0.05 * (t // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" +
                      repr(round(J, 6)))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" +
                  repr(round(J, 6)) + ", beta=" + str(round(beta, 1)))

    return best_a, best_max_b


def peak_surgery(a0, time_limit, verbose=True, seed=77):
    """Targeted local search: identify top peaks and reduce them."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 5000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_top = min(20, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]

        improved = False
        found_move = False

        for k in top_idx[:10]:
            if found_move:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            if products.max() < 1e-20:
                continue

            src_i = int(idx[np.argmax(products)])
            partner = k_int - src_i
            if partner < 0 or partner >= n:
                continue

            if a[src_i] >= a[partner]:
                reduce_i = src_i
            else:
                reduce_i = partner

            if a[reduce_i] < 1e-14:
                continue

            pressure = np.zeros(n)
            for k2 in top_idx[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_idx[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)
            mask = candidates != reduce_i
            candidates = candidates[mask]

            if len(candidates) == 0:
                continue

            cand_pressure = pressure[candidates]
            n_try = min(4, len(candidates))
            try_dsts = candidates[np.argsort(cand_pressure)[:n_try]]

            for dst in try_dsts:
                dst_int = int(dst)
                for frac in [0.003, 0.005, 0.01, 0.02, 0.03]:
                    amt = a[reduce_i] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_i] -= amt
                    a_trial[dst_int] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found_move = True
                        break
                if found_move:
                    break

        if not improved:
            if no_improve > 300:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                break

        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSurgery (" + str(round(elapsed, 1)) + "s): J=" +
                  repr(round(J, 6)) + " stall=" + str(no_improve))

    return best_a, best_m


def pairwise_cd(a0, time_limit, verbose=True, seed=123):
    """Pairwise coordinate descent for final polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        c = fast_autocorr(a)
        threshold = np.percentile(c, 90)
        peak_positions = np.where(c > threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.argsort(c)[::-1][:15]

        improved = False

        for k in peak_positions[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]

            top_pairs = np.argsort(products)[::-1][:3]
            for pi in top_pairs:
                if improved:
                    break
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n or partner == src_i:
                    continue
                if a[src_i] < 1e-14 and a[partner] < 1e-14:
                    continue

                larger_i = src_i if a[src_i] >= a[partner] else partner
                if a[larger_i] < 1e-14:
                    continue

                for frac in [0.001, 0.002, 0.004, 0.006]:
                    if improved:
                        break
                    amt = a[larger_i] * frac
                    if amt < 1e-18:
                        continue
                    for _ in range(3):
                        dst = int(rng.integers(n))
                        if dst == larger_i:
                            continue
                        a_trial = a.copy()
                        a_trial[larger_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break

        if not improved:
            stall += 1
            if stall % 50 == 0:
                anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
                eps = 0.005 / (1.0 + stall / 50.0)
                a_try = a + eps * anti * a
                a_try = np.maximum(a_try, 0.0)
                s_t = a_try.sum()
                if s_t > 1e-15:
                    a_try = a_try / s_t
                    M_try = fast_autocorr(a_try).max()
                    if M_try < best_m:
                        a = a_try
                        best_m = M_try
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PairCD (" + str(round(elapsed, 1)) + "s): J=" +
                  repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence if new one is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) +
              " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search combining LP-FW, Nesterov gradient, peak surgery, and pairwise CD."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(round(p_score, 6)))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        x = np.linspace(0, 1, n_base)
        a_init = (1.0 - x) ** 1.5 + 0.1
        best_seq = norm_sum1(a_init)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(n_base) + ", score=" + repr(round(best_score, 6)))

    n0 = len(best_seq)
    print("Starting: n=" + str(n0) + ", score=" + repr(round(best_score, 6)))

    # === Phase 1: LP-FW at current n ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: LP-FW (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.22, 380.0)
    if t1 > 60:
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t1,
                                  verbose=True, seed=1,
                                  max_constraints=1200, nesterov=True,
                                  lp_time_limit=8)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1", t_global)

    # === Phase 2: Nesterov gradient refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov gradient (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 40:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t2,
                                beta_start=50.0, beta_end=3000.0,
                                beta_growth=1.008, eta_base=0.5,
                                top_k_ratio=0.015, restart_every=1500,
                                verbose=True, seed=1)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # === Phase 3: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP-FW round 2 (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.18, 300.0)
    if t3 > 60:
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t3,
                                  verbose=True, seed=99,
                                  max_constraints=1000, nesterov=True,
                                  lp_time_limit=7)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2", t_global)

    # === Phase 4: Peak surgery ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Peak Surgery (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 40:
        a_out, _ = peak_surgery(norm_sum1(best_seq), t4, verbose=True, seed=11)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps1", t_global)

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-length exploration ===")
    lengths_to_try = [n_best + 1000, n_best + 2000, max(3000, n_best - 1500)]
    lengths_to_try = [l for l in lengths_to_try if l <= 12000 and l >= 2000 and l != n_best]

    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("  Trying n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        # LP-FW at new length
        t5a = min(remaining * 0.30, 180.0)
        if t5a > 50:
            a_out, _ = lp_fw_optimize(a_long, t5a,
                                      verbose=False, seed=new_n % 1000,
                                      max_constraints=800, nesterov=True,
                                      lp_time_limit=6)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ml_nf" + str(new_n), t_global)

        # Gradient at new length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80:
            t5b = min(remaining * 0.25, 120.0)
            if t5b > 30:
                a_try = interpolate_sequence(best_seq, new_n)
                a_out, _ = nesterov_opt(a_try, t5b,
                                        beta_start=40.0, beta_end=2500.0,
                                        beta_growth=1.007, eta_base=0.5,
                                        top_k_ratio=0.015, restart_every=1200,
                                        verbose=False, seed=3)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "ml_ng" + str(new_n), t_global)

    # === Phase 6: Nesterov gradient at best n ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Nesterov refine (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.15, 220.0)
    if t6 > 40:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t6,
                                beta_start=100.0, beta_end=5000.0,
                                beta_growth=1.01, eta_base=0.4,
                                top_k_ratio=0.01, restart_every=1200,
                                verbose=True, seed=2)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)

    # === Phase 7: Pairwise CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Pairwise CD (n=" + str(n_best) + ") ===")
    t7 = min(remaining * 0.15, 200.0)
    if t7 > 30:
        a_out, _ = pairwise_cd(norm_sum1(best_seq), t7, verbose=True, seed=55)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd1", t_global)

    # === Phase 8: Final LP-FW burst ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final LP-FW (n=" + str(n_best) + ") ===")
    t8 = min(remaining * 0.15, 200.0)
    if t8 > 50:
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t8,
                                  verbose=True, seed=200,
                                  max_constraints=2000, nesterov=False,
                                  lp_time_limit=10)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # === Phase 9: Ultimate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 9: Ultimate polish (n=" + str(n_best) + ") ===")
    if remaining > 60:
        t9a = min(remaining * 0.3, 100.0)
        if t9a > 20:
            a_out, _ = peak_surgery(norm_sum1(best_seq), t9a, verbose=False, seed=44)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ps_final", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t9b = min(remaining * 0.3, 80.0)
            if t9b > 20:
                a_out, _ = nesterov_opt(norm_sum1(best_seq), t9b,
                                        beta_start=200.0, beta_end=8000.0,
                                        beta_growth=1.012, eta_base=0.3,
                                        top_k_ratio=0.005, restart_every=800,
                                        verbose=False, seed=4)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t9c = min(remaining * 0.4, 80.0)
            if t9c > 15:
                a_out, _ = pairwise_cd(norm_sum1(best_seq), t9c, verbose=False, seed=77)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd_final", t_global)

    # === Phase 10: One more LP-FW if time allows ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_best = len(best_seq)
        print("\n=== Phase 10: Last LP-FW (n=" + str(n_best) + ") ===")
        t10 = min(remaining * 0.5, 120.0)
        if t10 > 40:
            a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t10,
                                      verbose=False, seed=500,
                                      max_constraints=1500, nesterov=True,
                                      lp_time_limit=8)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_last", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) +
              ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1((1.0 - x) ** 1.5 + 0.1)
        best_score = evaluate_safe(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)

    return [float(x) for x in best_seq]
