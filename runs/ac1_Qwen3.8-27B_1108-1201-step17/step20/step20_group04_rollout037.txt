def lp_direction(f, M, solver=None):
    """Solve the LP to find the best direction g."""
    import cvxpy as cp
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    # Convolution constraints: (f*g)[k] <= M for all k
    constraints = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        # (f*g)[k] = sum_{i in idx} f[i] * g[k-i]
        # k-i ranges from k-i_end to k-i_start
        g_idx = k - idx  # These are the indices into g
        constraints.append(f[idx] @ g[g_idx] <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=solver)
    if prob.status in ('optimal', 'optimal_inaccurate'):
        return g.value
    return None

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based direction finding (solving the exact dual of the max-reduction problem)
2. Careful line search with golden section
3. Fallback Nesterov with log-sum-exp smoothing

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based direction finding with truncated constraints
2. Golden section line search for optimal step
3. Nesterov accelerated gradient with top-k smoothing (fallback)
4. Multi-length exploration with interpolation
5. Peak-targeted local refinement
6. Randomized restarts with structured perturbations
"""

import numpy as np
import time

def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_cross_corr(f, g):
    """Compute full linear cross-correlation f*g (convolution) using FFT."""
    n = max(len(f), len(g))
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(np.asarray(f, dtype=np.float64), fft_size)
    G = np.fft.rfft(np.asarray(g, dtype=np.float64), fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize a to have sum 1 and be non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    """Project onto probability simplex."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, maintaining sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def objective_from_max_b(a):
    """Compute objective value from the sequence (assumes sum=1)."""
    n = len(a)
    c = fast_autocorr(a)
    return 2.0 * n * c.max()

def lp_direction(f, M, n_constraints=300, solver=None, verbose=False):
    """
    Solve the LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0
    Using only the top n_constraints constraints.
    Returns normalized direction g (sum=1) or None.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    n = len(f)
    c = fast_autocorr(f)
    
    # Select which constraints to include
    if n_constraints < 2 * n - 1:
        top_idx = np.argsort(c)[::-1][:n_constraints]
    else:
        top_idx = np.arange(2 * n - 1)
    
    g_var = cp.Variable(n, nonneg=True)
    constraints = []
    
    for k in top_idx:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        g_idx = k - idx  # indices into g
        # (f*g)[k] = sum_{i} f[i] * g[k-i]
        coeff = f[idx]
        # Skip if all coefficients are zero
        if coeff.max() < 1e-15:
            continue
        constraints.append(coeff @ g_var[g_idx] <= M)
    
    if not constraints:
        return None
    
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    try:
        if solver is None:
            # Try to find a good solver
            prob.solve(verbose=False)
        else:
            prob.solve(solver=solver, verbose=False)
        
        if prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.asarray(g_var.value, dtype=np.float64)
            g_val = np.maximum(g_val, 0.0)
            s = g_val.sum()
            if s > 1e-10:
                g_norm = g_val / s
                return g_norm
    except Exception as e:
        if verbose:
            print("  LP error: " + str(e))
    
    return None

def line_search(a, d, M, n_trials=20):
    """
    Find optimal t in [0, 1] that minimizes max of autocorrelation
    of (1-t)*a + t*d.
    Uses golden section search.
    """
    def f_of_t(t):
        a_new = (1 - t) * a + t * d
        c = fast_autocorr(a_new)
        return c.max()
    
    # Golden section search on [0, 1]
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ~0.618
    
    a_lo, a_hi = 0.0, 1.0
    t1 = a_hi - gr * (a_hi - a_lo)
    t2 = a_lo + gr * (a_hi - a_lo)
    f1 = f_of_t(t1)
    f2 = f_of_t(t2)
    
    for _ in range(n_trials):
        if f1 < f2:
            a_hi = t2
            t2 = t1
            f2 = f1
            t1 = a_hi - gr * (a_hi - a_lo)
            f1 = f_of_t(t1)
        else:
            a_lo = t1
            t1 = t2
            f1 = f2
            t2 = a_lo + gr * (a_hi - a_lo)
            f2 = f_of_t(t2)
    
    t_opt = (a_lo + a_hi) / 2
    return t_opt, f_of_t(t_opt)

def lp_iteration(a, n_constraints=300, solver=None, max_ls_trials=15, verbose=False):
    """
    One iteration of LP-based optimization.
    Returns updated a and the new max autocorrelation.
    """
    c = fast_autocorr(a)
    M = c.max()
    
    d = lp_direction(a, M, n_constraints=n_constraints, solver=solver, verbose=verbose)
    if d is None:
        return a, M
    
    # Line search
    t_opt, M_new = line_search(a, d, M, n_trials=max_ls_trials)
    
    if M_new >= M:
        # Try smaller steps
        for t in [0.1, 0.05, 0.01, 0.005]:
            a_try = (1 - t) * a + t * d
            M_try = fast_autocorr(a_try).max()
            if M_try < M:
                return a_try, M_try
        return a, M
    
    a_new = (1 - t_opt) * a + t_opt * d
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new, M_new

def compute_smoothed_grad(y, beta, top_k=None):
    """Compute gradient of smoothed max autocorrelation."""
    n = len(y)
    c = fast_autocorr(y)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(y, fft_size)
    P = np.fft.rfft(p, fft_size)
    a_rev = y[::-1]
    Conv = np.fft.irfft(P * np.fft.rfft(a_rev, fft_size), fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad, p, c

Actually, let me reconsider the LP formulation. I want to find a direction d (with sum(d) = 1, d >= 0) such that when I move from a to (1-t)*a + t*d, the max autocorrelation decreases.

The change in autocorrelation at position k is:
  new_c[k] - c[k] = ((1-t)*a + t*d)*((1-t)*a + t*d))[k] - (a*a)[k]
  = (1-t)^2 c_a[k] + 2t(1-t)(a*d)[k] + t^2 c_d[k] - c_a[k]
  = -2t(1-t) c_a[k] + 2t(1-t)(a*d)[k] + t^2 c_d[k]
  = 2t(1-t)((a*d)[k] - c_a[k]) + t^2 c_d[k]

For small t, this is approximately:
  2t((a*d)[k] - c_a[k])

We want this to be negative for all k where c_a[k] is near the maximum. So we need:
  (a*d)[k] < c_a[k] for all peak positions k

The LP constraint (f*g)[k] <= M (where f=a, M=max(c_a)) ensures:
  (a*d)[k] <= M = max(c_a)

But we actually need (a*d)[k] < c_a[k] for the peaks to decrease. The constraint (a*d)[k] <= M is weaker - it allows (a*d)[k] = M even at positions where c_a[k] < M.

Wait, let me re-read the paper description: "maximizing sum_j b_j subject to b_j >= 0 and ||f*g_0||_inf <= M"

So the constraint is ||f*g||_inf <= M, meaning (f*g)[k] <= M for ALL k, not just the peak positions. This is a stronger constraint than what I was thinking.

With this constraint, for peak positions k* (where c_a[k*] = M):
  (a*d)[k*] <= M = c_a[k*]
  
So the change at k* is: 2t((a*d)[k*] - c_a[k*]) <= 0. Good, the peaks don't increase.

For non-peak positions k (where c_a[k] < M):
  (a*d)[k] <= M, but c_a[k] < M, so (a*d)[k] - c_a[k] could be positive or negative.

import numpy as np

def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:

def norm_sum1(a):
    """Normalize to non-negative with sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto probability simplex {x>=0, sum(x)=1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum_u = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, maintaining sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    """Compute gradient of (1/beta)*log(sum exp(beta*c_k)) w.r.t. y."""
    n = len(y)
    c = fast_autocorr(y)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total

    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(y, fft_size)
    P = np.fft.rfft(p, fft_size)
    a_rev = y[::-1]
    Conv = np.fft.irfft(P * np.fft.rfft(a_rev, fft_size), fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad, p, c

        x = x_new

        if t % 50 == 0:

def lp_direction(f, M, n_constraints=500, solver=None, verbose=False):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0
    Using only top n_constraints constraints for efficiency.
    Returns normalized direction g (sum=1) or None.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None

    n = len(f)
    c = fast_autocorr(f)

    if n_constraints < 2 * n - 1:
        top_idx = np.argsort(c)[::-1][:n_constraints]
    else:
        top_idx = np.arange(2 * n - 1)

    g_var = cp.Variable(n, nonneg=True)
    constraints = []

    for k in top_idx:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        g_idx = k - idx
        coeff = f[idx]
        if coeff.max() < 1e-15:
            continue
        constraints.append(coeff @ g_var[g_idx] <= M)

    if not constraints:
        return None

    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    try:
        if solver is not None:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)

        if prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.asarray(g_var.value, dtype=np.float64)
            g_val = np.maximum(g_val, 0.0)
            s = g_val.sum()
            if s > 1e-10:
                return g_val / s
    except Exception as e:
        if verbose:
            print("  LP solve error: " + str(e))

    return None

        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_top = min(20, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]
        improved = False
        found_move = False

        for k in top_idx[:12]:
            if found_move:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            if products.max() < 1e-20:
                continue

            src_i = int(idx[np.argmax(products)])
            partner = k_int - src_i
            if partner < 0 or partner >= n:
                continue

            if a[src_i] >= a[partner]:
                reduce_i = src_i
            else:
                reduce_i = partner

            if a[reduce_i] < 1e-14:
                continue

            pressure = np.zeros(n)
            for k2 in top_idx[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_idx[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 100:
                candidates = np.arange(n)
            mask = candidates != reduce_i
            candidates = candidates[mask]
            if len(candidates) == 0:
                continue

            n_try = min(10, len(candidates))
            try_dsts = candidates[np.argsort(pressure[candidates])[:n_try]]

            for dst in try_dsts:
                dst_int = int(dst)
                for frac in [0.003, 0.007, 0.015, 0.03, 0.05]:
                    amt = a[reduce_i] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_i] -= amt
                    a_trial[dst_int] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found_move = True
                        break
                if found_move:
                    break

        if not improved:
            if no_improve > 300:
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.04)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                break

        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSurg (" + str(round(elapsed, 1)) + "s): J=" +
                  repr(round(J, 6)) + " stall=" + str(no_improve))

    return best_a, best_m


def pairwise_cd(a0, time_limit, verbose=True, seed=123):
    """Fine-grained pairwise coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        c = fast_autocorr(a)
        threshold = np.percentile(c, 93)
        peak_positions = np.where(c > threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.argsort(c)[::-1][:20]

        improved = False

        for k in peak_positions[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            top_pairs = np.argsort(products)[::-1][:3]

            for pi in top_pairs:
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n or partner == src_i:
                    continue
                if a[src_i] < 1e-14 and a[partner] < 1e-14:
                    continue

                larger_i = src_i if a[src_i] >= a[partner] else partner
                if a[larger_i] < 1e-14:
                    continue

                for frac in [0.001, 0.003, 0.005]:
                    amt = a[larger_i] * frac
                    if amt < 1e-18:
                        continue
                    for _ in range(5):
                        dst = int(rng.integers(n))
                        if dst == larger_i:
                            continue
                        a_trial = a.copy()
                        a_trial[larger_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            stall += 1

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PairCD (" + str(round(elapsed, 1)) + "s): J=" +
                  repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence if new candidate is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) +
              " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def find_best_solver():
    """Find the best available solver for LPs."""
    try:
        import cvxpy as cp
        best_s = None
        best_t = float('inf')
        x = cp.Variable(20, nonneg=True)
        A = np.random.rand(50, 20)
        prob_tpl = cp.Problem(cp.Maximize(cp.sum(x)), [A @ x <= 1])

        for s in [cp.GUROBI, cp.MOSEK, cp.GLPK, cp.CBC, cp.PDLP,
                  cp.XPRESS, cp.SCIP, cp.ECOS, cp.SCS]:
            try:
                start = time.monotonic()
                prob = cp.Problem(cp.Maximize(cp.sum(x)), [A @ x <= 1])
                prob.solve(solver=s, verbose=False)
                elapsed = time.monotonic() - start
                if prob.status in ('optimal', 'optimal_inaccurate') and elapsed < best_t:
                    best_t = elapsed
                    best_s = s
            except Exception:
                continue
        if best_s is not None:
            print("  Best LP solver: " + str(best_s) + " (test time=" + str(round(best_t, 3)) + "s)")
        return best_s
    except ImportError:
        return None


def propose_candidate(budget_s=1800.0):
    """
    Main search function combining LP-based direction finding,
    Nesterov optimization, peak surgery, and multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        x = np.linspace(0, 1, n_base)
        a_init = (1.0 - x) ** 1.5 + 0.1
        best_seq = norm_sum1(a_init)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(n_base) + ", score=" + repr(best_score))

    n0 = len(best_seq)
    print("Starting: n=" + str(n0) + ", score=" + repr(best_score))

    # Find best LP solver
    solver = find_best_solver()

    # Phase 1: LP-based optimization (the main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 1: LP Optimization (n=" + str(n_best) + ") ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 60:
        a_out, _ = lp_optimize(norm_sum1(best_seq), t1, solver=solver,
                               n_constraints=500, verbose=True, seed=1)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 2: Nesterov refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 40:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t2,
                                beta_start=80.0, beta_end=4000.0,
                                beta_growth=1.008, eta_base=0.4,
                                top_k_ratio=0.01, restart_every=1500,
                                verbose=True, seed=2)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 3: LP optimization round 2 (with different constraint count)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP Round 2 (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 40:
        a_out, _ = lp_optimize(norm_sum1(best_seq), t3, solver=solver,
                               n_constraints=800, verbose=True, seed=3)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp2", t_global)

    # Phase 4: Peak surgery
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Peak Surgery (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 30:
        a_out, _ = peak_surgery(norm_sum1(best_seq), t4, verbose=True, seed=11)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps1", t_global)

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-length exploration ===")
    lengths_to_try = [n_best + 400, n_best + 1200, max(4000, n_best - 800)]
    lengths_to_try = [l for l in lengths_to_try if 3000 <= l <= 12000]
    # Remove duplicates
    seen = set()
    unique_lengths = []
    for l in lengths_to_try:
        if l not in seen and l != n_best:
            seen.add(l)
            unique_lengths.append(l)
    lengths_to_try = unique_lengths

    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("  Trying n=" + str(new_n) + " (remaining=" + str(round(remaining, 0)) + "s)")
        a_long = interpolate_sequence(best_seq, new_n)

        # LP on new length
        t5a = min(remaining * 0.35, 180.0)
        if t5a > 30:
            a_out, _ = lp_optimize(a_long, t5a, solver=solver,
                                   n_constraints=500, verbose=False, seed=5)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(new_n), t_global)

        # Nesterov on new length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t5b = min(remaining * 0.30, 120.0)
            if t5b > 30:
                a_try = interpolate_sequence(best_seq, new_n)
                a_out, _ = nesterov_opt(a_try, t5b,
                                        beta_start=60.0, beta_end=3000.0,
                                        beta_growth=1.007, eta_base=0.45,
                                        top_k_ratio=0.012, restart_every=1200,
                                        verbose=False, seed=6)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(new_n), t_global)

    # Phase 6: Final LP polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Final LP (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 30:
        a_out, _ = lp_optimize(norm_sum1(best_seq), t6, solver=solver,
                               n_constraints=1000, verbose=True, seed=7)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Phase 7: Nesterov final
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final Nesterov (n=" + str(n_best) + ") ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 30:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t7,
                                beta_start=200.0, beta_end=8000.0,
                                beta_growth=1.01, eta_base=0.25,
                                top_k_ratio=0.005, restart_every=1000,
                                verbose=True, seed=8)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

    # Phase 8: Peak surgery final
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final Peak Surgery (n=" + str(n_best) + ") ===")
    t8 = min(remaining * 0.20, 120.0)
    if t8 > 20:
        a_out, _ = peak_surgery(norm_sum1(best_seq), t8, verbose=False, seed=9)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps_final", t_global)

    # Phase 9: Pairwise CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 9: Pairwise CD (n=" + str(n_best) + ") ===")
    t9 = min(remaining * 0.30, 100.0)
    if t9 > 15:
        a_out, _ = pairwise_cd(norm_sum1(best_seq), t9, verbose=False, seed=10)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd_final", t_global)

    # Phase 10: One more LP if time allows
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_best = len(best_seq)
        print("\n=== Phase 10: Ultimate LP (n=" + str(n_best) + ") ===")
        t10 = min(remaining * 0.70, 150.0)
        a_out, _ = lp_optimize(norm_sum1(best_seq), t10, solver=solver,
                               n_constraints=1000, verbose=True, seed=11)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_ultimate", t_global)

        # Final pair
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            n_best = len(best_seq)
            t11 = min(remaining * 0.60, 50.0)
            a_out, _ = peak_surgery(norm_sum1(best_seq), t11, verbose=False, seed=12)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ps_ultimate", t_global)

    # Final validation
    if best_seq is not None:
        best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
        best_seq = np.minimum(best_seq, 1000.0)
        best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) +
              ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence! Using fallback.")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1((1.0 - x) ** 1.5 + 0.1)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in best_seq]
