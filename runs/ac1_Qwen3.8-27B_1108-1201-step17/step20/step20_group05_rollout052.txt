def _next_pow2(x):
    x = int(max(1, x))
    return 1 << (x - 1).bit_length()


def _norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = _next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def _fft_conv2(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if b.size != n:
        b = np.resize(b, n)
    size = 2 * n - 1
    N = _next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def _cross_convolve_p_a(p, a, factor=1.0):
    p = np.asarray(p, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    total = 3 * n - 2
    if p.size != 2 * n - 1:
        p = np.resize(p, 2 * n - 1)
    N = _next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    out = conv[n - 1:2 * n - 1][::-1]
    out = np.where(np.isfinite(out), out, 0.0)
    return factor * out


def _proxy_max(c, mask=None):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0
    if mask is None:
        vals = c
    else:
        vals = c[mask]
    if vals.size == 0:
        return 0.0
    vals = vals[np.isfinite(vals)]
    if vals.size == 0:
        return 0.0
    return float(np.max(vals))


def _active_top(c, mask, max_k):
    c = np.asarray(c, dtype=np.float64)
    size = c.size
    max_k = int(min(max_k, size))
    if max_k <= 0:
        return np.array([], dtype=int)
    vals = c.copy()
    if mask is not None:
        vals[~mask] = -np.inf
    return np.argsort(vals)[::-1][:max_k]

    if top_count <= 0:
        return base_mask
    if top_count >= size:
        return base_mask
    c = _fft_conv(a)
    top = _active_top(c, base_mask, top_count)
    if top.size == 0:
        return base_mask
    m = np.zeros(size, dtype=bool)
    m[top] = True
    if base_mask is not None:
        m[~base_mask] = False
    if not np.any(m):
        return base_mask
    return m


def _safe_eval(a):
    try:
        arr = np.asarray(a, dtype=np.float64)
        if arr.size == 0:
            return float("inf")
        arr = np.clip(arr, 0.0, 1000.0)
        if not np.all(np.isfinite(arr)):
            return float("inf")
        s = float(arr.sum())
        if s < 0.01:
            return float("inf")
        lst = [float(x) for x in arr]
        val = evaluate_sequence(lst)
        if val is None:
            return float("inf")
        val = float(val)
        if not math.isfinite(val):
            return float("inf")
        return val
    except Exception:
        return float("inf")


def _try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    try:
        arr = np.asarray(a, dtype=np.float64)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        np.maximum(arr, 0.0, out=arr)
        s = float(arr.sum())
        if arr.size == 0 or s < 0.01:
            return best_seq, best_score
        arr = arr / s
        sc = _safe_eval(arr)
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_seq = arr.copy()
            best_score = float(sc)
            print("  NEW BEST [" + label + "]: " + repr(sc) +
                  " n=" + str(arr.size) +
                  " t=" + repr(time.monotonic() - t_global))
    except Exception:
        pass
    return best_seq, best_score


def _interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return _norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return _norm_sum1(y)


def _softmax_from_u(u):
    u = np.asarray(u, dtype=np.float64)
    n = u.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(u)):
        return np.ones(n, dtype=np.float64) / float(n)
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def _make_v(a, ref=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 1:
        return np.array([], dtype=np.float64), 0
    if ref is None:
        if np.any(a > 0.0):
            ref = int(np.argmax(a))
        else:
            ref = 0
    ref = int(ref)
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.where(np.isfinite(v), v, -700.0)
    return v, ref


def _from_v(v, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    return _softmax_from_u(u)


def _lse_objgrad(a, beta, mask, reg_a=0.0):
    n = a.size
    if n == 0:
        return 0.0, np.array([], dtype=np.float64)
    size = 2 * n - 1
    total = 3 * n - 2
    N = _next_pow2(total)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)

    if mask is None:
        cv = c
    else:
        cv = c[mask]

    if cv.size == 0:
        f = 0.0
        grad = np.zeros(n, dtype=np.float64)
    else:
        m = float(np.max(cv))
        shifted = beta * (cv - m)
        shifted = np.clip(shifted, -700.0, 700.0)
        e = np.exp(shifted)
        se = float(e.sum())
        if se <= 0.0 or not math.isfinite(se):
            e = np.zeros(cv.size, dtype=np.float64)
            e[0] = 1.0
            se = 1.0
        p = e / se

        pf = np.zeros(size, dtype=np.float64)
        if mask is None:
            pf[:] = p
        else:
            pf[mask] = p

        Pr = np.fft.rfft(pf[::-1], N)
        conv = np.fft.irfft(A * Pr, N)[:total]
        grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
        grad = np.where(np.isfinite(grad), grad, 0.0)
        f = m + math.log(se) / beta

    if reg_a > 0.0:
        f += 0.5 * reg_a * float(np.dot(a, a))
        grad = grad + reg_a * a

    if not math.isfinite(f):
        f = 1e100
    return f, grad


def _objgrad_v(v, beta, mask, reg_a=0.0, reg_u=0.0, ref=0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    if n <= 0:
        return 1e100, np.array([], dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return 1e100, np.zeros(v.size, dtype=np.float64)

    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1

    u = np.empty(n, dtype=np.float64)
    u[ref] = 0.0
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]

    a = _softmax_from_u(u)
    f, ga = _lse_objgrad(a, beta, mask, reg_a)
    if not math.isfinite(f):
        f = 1e100
        ga = np.zeros(n, dtype=np.float64)

    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)

    gv = np.empty(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]

    if reg_u > 0.0:
        f += 0.5 * reg_u * float(np.dot(v, v))
        gv = gv + reg_u * v

    if not np.all(np.isfinite(gv)):
        gv = np.zeros(v.size, dtype=np.float64)
    return f, gv


def _run_lbfegs(a0, beta, mask, maxfun=500, reg_u=1e-12, verbose=False):
    a = _norm_sum1(a0)
    n = a.size
    if n <= 1 or not SCIPY_OK or maxfun <= 0:
        return a
    v0, ref = _make_v(a)
    t0 = time.monotonic()
    options = {
        "maxfun": int(maxfun),
        "maxiter": int(maxfun + 30),
        "ftol": 1e-12,
        "gtol": 1e-8,
        "disp": False
    }
    res = None
    try:
        res = scipy.optimize.minimize(
            _objgrad_v, v0,
            args=(beta, mask, 0.0, reg_u, ref),
            method="L-BFGS-B", jac=True, options=options
        )
    except Exception:
        res = None
    if res is not None and hasattr(res, "x"):
        vx = np.asarray(res.x, dtype=np.float64)
        if vx.size == v0.size and np.all(np.isfinite(vx)):
            a_out = _from_v(vx, ref)
            a_out = _norm_sum1(a_out)
            if verbose:
                print("    L-BFGS beta=" + repr(beta) + " time=" + repr(time.monotonic() - t0))
            return a_out
    return a


def _game_direction(f, active_idx, mw_iters=25, seed=0):
    f = _norm_sum1(f)
    n = f.size
    size = 2 * n - 1
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K <= 0:
        return np.ones(n, dtype=np.float64) / float(n)

    rng = np.random.default_rng(seed)
    logits = rng.normal(0.0, 0.05, K)

    def _norm_logits(loc):
        loc = np.where(np.isfinite(loc), loc, -1000.0)
        loc = loc - float(np.max(loc))
        w = np.exp(loc)
        s = float(w.sum())
        if s <= 0.0:
            w = np.ones(K, dtype=np.float64)
            s = float(K)
        return w / s

    lambda_w = _norm_logits(logits)
    for _ in range(max(1, int(mw_iters))):
        pf = np.zeros(size, dtype=np.float64)
        pf[active_idx] = lambda_w
        pressure = _cross_convolve_p_a(pf, f, 1.0)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)
        pmx = float(np.max(pressure))
        pmn = float(np.min(pressure))
        ptp = pmx - pmn
        if ptp <= 1e-30:
            g = np.ones(n, dtype=np.float64) / float(n)
        else:
            scaled = (pressure - pmn) / ptp
            w = np.exp(-10.0 * scaled)
            w = np.where(np.isfinite(w), w, 0.0)
            w += 1e-12
            g = w / float(w.sum())
            g = 0.95 * g + 0.05 * (np.ones(n, dtype=np.float64) / float(n))
            g = g / float(g.sum())
        h = _fft_conv2(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmac_ = float(np.max(ha))
        hmin = float(np.min(ha))
        hptp = hmac_ - hmin
        if hptp > 1e-30:
            eta = 15.0 / hptp
            logits += eta * (ha - float(np.mean(ha)))
        lambda_w = _norm_logits(logits)
    return g


def _line_search_mix(a, g, cf, cg, fg, mask, current_M):
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    alphas = [0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9, 1.0]
    best_alpha = 0.0
    best_m = current_M
    for alpha in alphas:
        one = 1.0 - alpha
        val = (one * one) * cf + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
        if mask is None:
            m = float(np.max(val))
        else:
            m = float(np.max(val[mask]))
        if math.isfinite(m) and m < best_m:
            best_m = m
            best_alpha = alpha
    if best_alpha <= 0.0:
        return a.copy(), best_m, 0.0
    new_a = (1.0 - best_alpha) * a + best_alpha * g
    new_a = np.maximum(new_a, 0.0)
    s = float(new_a.sum())
    if s < 1e-15:
        return a.copy(), best_m, 0.0
    new_a = new_a / s
    return new_a, best_m, best_alpha
