def lp_direction(f, active_idx, time_limit=5.0):
    """Solve min_g max_{k in active} (f*g)_k using linear programming."""
    if not SCIPY_OK:
        return None

    # Variable: g[0..n-1], t

            rows.append(r)
            cols.append(i)
            vals.append(f[i])  # f[i] * g[k-i], and k-i ranges from k-s to k-e+1
        # Wait, I need to be careful about indexing
        # (f*g)[k] = sum_{i} f[i] * g[k-i]
        # For fixed k, the terms are f[i] * g[k-i] where i ranges over valid values
        # If we let j = k-i, then i = k-j, and j ranges over [max(0, k-n+1), min(n-1, k)]
        # So (f*g)[k] = sum_j f[k-j] * g[j] for j in valid range
        # The coefficient of g[j] in constraint k is f[k-j]
        pass
    
    # Let me redo this more carefully
    rows = []
    cols = []
    vals = []
    for r, k in enumerate(active_idx):
        # (f*g)[k] = sum_j f[k-j] * g[j] where j in [max(0, k-n+1), min(n-1, k)]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):

            cols.append(j)

    A_full = scipy.sparse.hstack([A_sparse, neg_col], format='csr')
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # Minimize t
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    try:

            method='HiGHS', options={'disp': False, 'time_limit': time_limit}

        if res.success:
            g = res.x[:n]
            g = np.maximum(g, 0)
            gs = g.sum()

def frank_wolfe(a0, time_budget, use_lp=True, lp_every=5, verbose=True):
    """Frank-Wolfe with adaptive oracle."""
    t0 = time.monotonic()
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = c.max()
    best_a = a.copy()
    best_M = M
    
    # Active set: top positions
    active_count = min(500, 2 * n - 1)
    active = np.argsort(c)[::-1][:active_count]
    
    no_improve = 0
    t = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_budget - 3 and no_improve < 100:
        t += 1
        time_left = time_budget - (time.monotonic() - t0)
        if time_left < 10:
            break
        
        # Get direction
        g = None
        if use_lp and (t % lp_every == 0) and time_left > 60:
            g = lp_direction(a, active, time_limit=min(5.0, time_left * 0.2))
        
        if g is None:
            g = game_direction(a, active, iters=min(40, max(15, int(time_left / 2))), seed=1000 + t)
        
        if g is None:
            break
        
        # Line search
        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        
        best_alpha = 0.0
        best_m = M
        for alpha in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            val = (1-alpha)**2 * c + 2*alpha*(1-alpha) * fg + alpha**2 * cg
            m = val.max()
            if m < best_m:
                best_m = m
                best_alpha = alpha
        
        if best_alpha > 0 and best_m < M - 1e-16:
            a = (1 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0)
            a /= a.sum()
            c = fft_conv(a)
            M = c.max()
            no_improve = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            active = np.argsort(c)[::-1][:active_count]
            if verbose and (t == 1 or t % 10 == 0):
                print(f"    FW t={t} M={M:.8f} score={2*n*M:.8f} alpha={best_alpha:.4f}")
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                # Perturb
                pert = rng.normal(0, 0.02, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-12)
                a /= a.sum()
                c = fft_conv(a)
                M = c.max()
                active = np.argsort(c)[::-1][:active_count]
                no_improve = 0
    
    return best_a

def coordinate_polish(a0, time_budget, verbose=True):
    """Coordinate descent with rank-1 convolution update."""
    t0 = time.monotonic()
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = c.max()
    best_a = a.copy()
    best_M = M
    stall = 0
    rng = np.random.default_rng(123)
    rnd = 0
    
    while time.monotonic() - t0 < time_budget - 2 and stall < 80:
        rnd += 1
        top_k = min(20, 2*n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        
        # Compute gradient of max at top positions (averaged)
        p = np.zeros(2*n - 1)
        p[top_idx] = 1.0 / top_k
        total = 3*n - 2
        N = next_pow2(total)
        A = np.fft.rfft(a, N)
        Pr = np.fft.rfft(p[::-1], N)
        conv = np.fft.irfft(A * Pr, N)[:total]
        grad = 2.0 * conv[n-1:2*n-1][::-1]
        
        # Find best single coordinate move
        best_cand = None
        # Try moving mass from high-grad to low-grad positions
        src_candidates = np.argsort(grad)[::-1][:min(10, n)]
        dst_candidates = np.argsort(grad)[:min(15, n)]
        
        for s in src_candidates:
            if a[s] < 1e-12:
                continue
            for d in dst_candidates:
                if d == s:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    tmove = a[s] * frac
                    if tmove <= 1e-16:
                        continue
                    # Rank-1 update of convolution
                    c_new = c.copy()
                    # Moving t from s to d:
                    # (new_a) = a - t*e_s + t*e_d
                    # (new_a * new_a) = (a*a) - t*(a*e_s + e_s*a) + t^2*e_s*e_s + t*(a*e_d + e_d*a) - t^2*(e_s*e_d + e_d*e_s) ... 
                    # This is getting complex. Let me just recompute for small n.
                    # Actually for the rank-1 update:
                    # delta_c = -2*t*(a shifted by s) + 2*t*(a shifted by d) + t^2*(delta at 2s + delta at 2d - 2*delta at s+d)
                    
                    # The change in (a*a) when we move t from position s to position d:
                    # new_a = a, but a[s] -= t, a[d] += t
                    # new_conv = old_conv - t*(e_s * a + a * e_s - e_s*e_s) + t*(e_d * a + a * e_d - e_d*e_d) + cross terms
                    # More precisely:
                    # new_conv[k] = old_conv[k] + t * (a[k-d] - a[k-s]) * 2 (approximately, for the linear terms)
                    # + t^2 * (delta[k-2d] - 2*delta[k-(s+d)] + delta[k-2s])
                    
                    # Let me just do it directly:
                    a[s] -= tmove
                    a[d] += tmove
                    c_new = fft_conv(a)
                    m_new = c_new.max()
                    a[s] += tmove
                    a[d] -= tmove
                    
                    if m_new < M - 1e-17:
                        if best_cand is None or m_new < best_cand[0]:
                            best_cand = (m_new, s, d, tmove)
        
        if best_cand is not None:
            m_new, s, d, tmove = best_cand
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0)
            a /= a.sum()
            c = fft_conv(a)
            M = c.max()
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Try random moves
            found = False
            for _ in range(50):
                s = rng.integers(n)
                d = rng.integers(n)
                if s == d or a[s] < 1e-12:
                    continue
                frac = rng.uniform(0.005, 0.05)
                tmove = a[s] * frac
                a[s] -= tmove
                a[d] += tmove
                c_new = fft_conv(a)
                m_new = c_new.max()
                a[s] += tmove
                a[d] -= tmove
                if m_new < M - 1e-17:
                    a[s] -= tmove
                    a[d] += tmove
                    a = np.maximum(a, 0)
                    a /= a.sum()
                    c = fft_conv(a)
                    M = c.max()
                    if M < best_M:
                        best_M = M
                        best_a = a.copy()
                    stall = 0
                    found = True
                    break
            if not found:
                stall += 5
    
    return best_a

def sine_init(n, freq=1, phase=0):

def gauss_init(n, center=0.5, width=0.3):

def next_pow2(x):
    x = int(max(1, x))
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    n = arr.size
    if n == 0:
        return np.ones(1)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    np.maximum(arr, 0.0, out=arr)
    s = arr.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return arr / s


def fft_conv(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return np.maximum(c, 0.0)


def fft_conv2(a, b):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    return np.maximum(c, 0.0)

def lse_value_and_grad(a, beta):
    c = fft_conv(a)
    m = c.max()
    shifted = np.clip(beta * (c - m), -700.0, 700.0)
    e = np.exp(shifted)
    s = e.sum()
    if s <= 0 or not math.isfinite(s):
        return 1e100, np.zeros_like(a)
    f = m + math.log(s) / beta
    p = e / s
    n = a.size

    A = np.fft.rfft(a, N)

    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)

def fw_subgradient_direction(a, active_idx):
    """Standard FW direction: vertex minimizing the linearized objective."""
    n = a.size
    h = np.zeros(n)
    for k in active_idx:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            h[j_start:j_end + 1] += a[k - np.arange(j_start, j_end + 1)]
    j_star = int(np.argmin(h))
    return np.zeros(n) | (1.0 << j_star) if False else _make_vertex(j_star, n)


def _make_vertex(idx, n):
    v = np.zeros(n)
    v[idx] = 1.0
    return v


def fw_lp_direction(f, active_idx, time_limit=4.0):
    """Solve min_g max_{k in active} (f*g)_k via LP."""
    if not SCIPY_OK:
        return None
    n = f.size
    K = len(active_idx)
    if K == 0:
        return None
    if K > 1500:
        active_idx = active_idx[:1500]
        K = 1500

    rows = []
    cols = []
    vals = []
    for r, k in enumerate(active_idx):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(r)
            cols.append(j)
            vals.append(f[k - j])

    if not rows:
        return None

    A_sp = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))
    neg_t = scipy.sparse.csc_matrix(np.full((K, 1), -1.0))
    A_full = scipy.sparse.hstack([A_sp, neg_t], format='csr')

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    b_ub = np.zeros(K)
    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = scipy.optimize.linprog(
            c_obj, A_ub=A_full, b_ub=b_ub,
            A_eq=A_eq, b_eq=b_eq, bounds=bounds,
            method='HiGHS', options={'disp': False, 'time_limit': float(time_limit)}
        )
        if res.success:
            g = res.x[:n]
            g = np.maximum(g, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def fw_game_direction(f, active_idx, iters=30, seed=0):
    """Approximate LP direction via multiplicative weights game."""
    n = f.size
    K = len(active_idx)
    if K == 0:
        return np.ones(n) / n
    rng = np.random.default_rng(seed)
    lam = np.ones(K) / K
    g = np.ones(n) / n

    for _ in range(iters):
        # Compute gradient: h_j = sum_k lam[k] * f[k-j]
        h = np.zeros(n)
        for idx, k in enumerate(active_idx):
            w = lam[idx]
            if w < 1e-12:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                h[j_start:j_end + 1] += w * f[k - np.arange(j_start, j_end + 1)]

        # Best response: concentrate where h is smallest
        h_min = h.min()
        h_max = h.max()
        if h_max - h_min > 1e-15:
            scaled = (h - h_min) / (h_max - h_min)
            g_new = np.exp(-12.0 * scaled)
            g_new += 1e-10
            g_new /= g_new.sum()
        else:
            g_new = np.ones(n) / n

        # Update lambda
        fg = fft_conv2(f, g_new)
        fg_act = fg[active_idx]
        hptp = fg_act.max() - fg_act.min()
        if hptp > 1e-12:
            logits = np.log(np.maximum(lam, 1e-300)) + (12.0 / hptp) * fg_act
            logits -= logits.max()
            lam = np.exp(logits)
            lam /= lam.sum()

        g = 0.85 * g_new + 0.15 * g
        g = np.maximum(g, 0.0)
        g /= g.sum()

    return g


def frank_wolfe(a0, time_budget, lp_every=5, game_iters=35, verbose=True):

        if g is None:

        # Line search
        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        best_alpha = 0.0
        best_m = M
        # Golden section + a few fixed points
        alphas = [0.01, 0.03, 0.05, 0.08, 0.12, 0.17, 0.25, 0.35, 0.5, 0.65, 0.8, 0.95]

            a = np.maximum(a, 0.0)
            a /= a.sum()

            no_improve = 0
            if M < best_M - 1e-15:

            active = np.argsort(c)[::-1][:active_count]

                    c = c_new

def rank1_conv_update(c, a, s, d, tmove):
    """Update convolution when moving tmove from position s to d."""
    n = a.size
    c_new = c.copy()
    # Linear terms: -tmove * (e_s * a + a * e_s) + tmove * (e_d * a + a * e_d)
    # (e_s * a)[k] = a[k-s] for valid k; shift of a by s
    # Contribution: c_new[k] += 2*tmove * (a[k-d] - a[k-s]) for valid indices
    for k in range(n):
        idx_s = k + s
        idx_d = k + d
        if idx_s < c_new.size:
            c_new[idx_s] -= 2.0 * tmove * a[k]
        if idx_d < c_new.size:
            c_new[idx_d] += 2.0 * tmove * a[k]
    # Quadratic terms: tmove^2 * (delta_{2d} - 2*delta_{s+d} + delta_{2s})
    t2 = tmove * tmove
    if 2 * d < c_new.size:
        c_new[2 * d] += t2
    if 2 * s < c_new.size:
        c_new[2 * s] += t2
    if s + d < c_new.size:
        c_new[s + d] -= 2.0 * t2
    return np.maximum(c_new, 0.0)

                best_a = a.copy()
            stall = 0
            if verbose and (rnd <= 3 or rnd % 20 == 0):
                print(f"    Coord rnd={rnd} proxy={2*n*best_M:.8f}")
        else:
            stall += 1
            if stall % 20 == 19:
                # Bigger perturbation
                pert = rng.normal(0, 0.03, n) * best_a
                a_try = best_a + pert
                a_try = np.maximum(a_try, 1e-12)
                a_try /= a_try.sum()
                c_try = fft_conv(a_try)
                if c_try.max() < best_M * 1.02:
                    a = a_try
                    c = c_try
                    M = c.max()
                    stall = 0

    return best_a


def sine_init(n, freq=1.0, phase=0.0):
    x = np.linspace(0.0, 1.0, n)
    a = np.sin(np.pi * freq * x + phase) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def gauss_init(n, center=0.5, width=0.25):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    return norm_sum1(a)


def two_peak_init(n):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-15.0 * (x - 0.3) ** 2) + np.exp(-15.0 * (x - 0.7) ** 2)
    return norm_sum1(a)


def power_sine_init(n, power=1.0):
    x = np.linspace(0.0, 1.0, n)
    a = np.sin(np.pi * x) ** (2.0 * power)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def random_smooth_init(n, seed=0):
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    kw = np.ones(max(5, n // 500))
    kw /= kw.sum()
    a = np.convolve(a, kw, mode='same')
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def cosine_bell_init(n, moments=3):
    """Product of cosines for a bell-shaped function."""
    x = np.linspace(0.0, 1.0, n)
    t = (x - 0.5) * 2.0  # [-1, 1]
    a = np.ones(n)
    for m in range(1, moments + 1):
        a *= np.cos(np.pi * t / 2.0) ** m
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, int(new_n))
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)
