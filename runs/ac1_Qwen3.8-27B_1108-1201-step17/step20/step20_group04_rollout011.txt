def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_smoothed_grad(y, beta, top_k=None):
    n = len(y)
    c = fast_autocorr(y)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(y, fft_size)
    P = np.fft.rfft(p, fft_size)
    a_rev = y[::-1]
    Conv = np.fft.irfft(P * np.fft.rfft(a_rev, fft_size), fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad, p, c

   change in all autocorrelation values via s[k] and q[k] vectors, then find
   the optimal transfer amount analytically. No FFT needed in inner loop.
2. Multi-peak candidate generation: target top contributing pairs at several
   peak positions, with intelligent destination selection.
3. Nesterov smoothed gradient for initial coarse optimization.
4. Simulated annealing escape with structured perturbations.
5. Multi-length exploration via interpolation.
Key improvement: exact O(n) autocorrelation update per move (no FFT in loop),
enabling 100x more iterations than the previous approach.
"""

import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    n = len(y)
    c = fast_autocorr(y)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    fft_size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    a_rev = y[::-1]
    Conv = np.fft.irfft(P * np.fft.rfft(a_rev, fft_size), fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad, p, c


def nesterov_opt(a0, time_limit, beta_start=30.0, beta_end=3000.0,
                 beta_growth=1.006, eta_base=0.6, top_k_ratio=0.02,
                 restart_every=1500, verbose=True, seed=42):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        grad, p, c = compute_smoothed_grad(y, beta, top_k)
        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)
        z = y - eta * grad
        x_new = simplex_project(z)
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        x_prev = x
        x = x_new
        t_nest = t_new
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.03 / (1.0 + 0.05 * (t // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
    return best_a, best_max_b


def exact_local_search(a0, time_limit, verbose=True, seed=42):
    """Main local search with exact O(n) autocorrelation updates."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    c = fast_autocorr(a)
    best_a = a.copy()
    best_max = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 50000
    n_cand = max(40, min(150, n // 50))

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        n_peaks = min(12, len(c))
        peak_idx = np.argsort(c)[::-1][:n_peaks]
        c_max = c.max()

        best_move = None
        best_new_max = c_max + 1.0

        # Generate candidates from top peaks
        candidates = set()
        for K in peak_idx[:8]:
            K_int = int(K)
            i_start = max(0, K_int - n + 1)
            i_end = min(n - 1, K_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[K_int - idx]
            top_i = idx[np.argsort(products)[::-1][:6]]
            for ii in top_i:
                i_int = int(ii)
                if a[i_int] < 1e-12:
                    continue
                # Find safe destinations: low a[j] and low contributions to other peaks
                peak_contrib = np.zeros(n)
                for K2 in peak_idx[:6]:
                    K2i = int(K2)
                    jc = max(0, K2i - n + 1)
                    je = min(n - 1, K2i)
                    if jc <= je:
                        peak_contrib[jc:je + 1] += a[K2i - np.arange(jc, je + 1)]
                # Score: prefer low a[j] + low peak_contrib[j]
                score = a + 0.5 * peak_contrib
                # Get a few of the lowest-score positions, excluding i
                score[i_int] = np.inf
                # Also exclude positions too close to i (to avoid trivial back-and-forth)
                for ddx in range(-3, 4):
                    jj = i_int + ddx
                    if 0 <= jj < n:
                        score[jj] = np.inf
                n_try = min(10, n // 50)
                low_pos = np.argpartition(score, n_try)[:n_try]
                for jj in low_pos:
                    jj_int = int(jj)
                    if jj_int != i_int:
                        candidates.add((i_int, jj_int))
        # Random candidates
        for _ in range(10):
            ii = int(rng.integers(n))
            jj = int(rng.integers(n))
            if ii != jj and a[ii] > 1e-10:
                candidates.add((ii, jj))
        # Cap candidates
        if len(candidates) > n_cand:
            candidates = list(candidates)[:n_cand]

        # Evaluate each candidate
        for (i, j) in candidates:
            if a[i] < 1e-12:
                continue
            delta_max = a[i]
            if delta_max < 1e-10:
                continue
            # s[k] = a[k-j] - a[k-i] + a[j] - a[i]
            s = np.zeros(2 * n - 1)
            s[j:j + n] += a
            s[i:i + n] -= a
            s += a[j] - a[i]
            # q[k]
            q = np.zeros(2 * n - 1)
            qi = 2 * i
            qj = 2 * j
            kip = i + j
            if qi < 2 * n - 1:
                q[qi] += 1.0
            if qj < 2 * n - 1:
                q[qj] += 1.0
            if kip < 2 * n - 1:
                q[kip] -= 2.0
            # Try several delta values
            frs = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.6, 0.8]
            # Also try the "bottleneck" delta
            pos_s = s > 1e-15
            if pos_s.any():
                bottleneck = (c_max - c) / s
                bottleneck = bottleneck[pos_s]
                bottleneck = bottleneck[bottleneck > 1e-12]
                if len(bottleneck) > 0:
                    b_delta = min(bottleneck.min(), delta_max * 0.9)
                    if b_delta > 1e-10:
                        frs.append(b_delta / delta_max)
            for frac in frs:
                delta = min(delta_max * frac, delta_max * 0.95)
                if delta < 1e-12:
                    continue
                c_new = c + delta * s + delta * delta * q
                new_max = c_new.max()
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_move = (i, j, delta, s.copy(), q.copy())

        if best_move is not None:
            i, j, delta, s_vec, q_vec = best_move
            a[i] -= delta
            a[j] += delta
            c = c + delta * s_vec + delta * delta * q_vec
            actual_max = c.max()
            if actual_max < best_max - 1e-15:
                best_max = actual_max
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Perturbation to escape local minimum
            if no_improve > 300:
                stuck_scale = min(0.15, 0.02 * (no_improve // 300))
                for _ in range(30):
                    ii = int(rng.integers(n))
                    jj = int(rng.integers(n))
                    if ii != jj and a[ii] > 1e-8:
                        delta = a[ii] * rng.uniform(0.02, stuck_scale)
                        a[ii] -= delta
                        a[jj] += delta
                        a = np.maximum(a, 0.0)
                        c = fast_autocorr(a)
                        if c.max() < best_max:
                            best_max = c.max()
                            best_a = a.copy()
                            no_improve = 0
                            break
                if no_improve > 3000:
                    # Larger structured perturbation
                    anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
                    eps = 0.05 / (1.0 + no_improve / 2000.0)
                    a_p = a + eps * anti * a
                    a_p = np.maximum(a_p, 0.0)
                    s_p = a_p.sum()
                    if s_p > 1e-15:
                        a_p = a_p / s_p
                        c_p = fast_autocorr(a_p)
                        if c_p.max() < best_max:
                            a = a_p
                            c = c_p
                            best_max = c.max()
                            best_a = a.copy()
                            no_improve = 0

        if verbose and (no_improve % 10000 == 0 or no_improve == 0):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max
            print("    ExactLS (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " no_imp=" + str(no_improve))

    return best_a, best_max


def fine_grained_search(a0, time_limit, verbose=True, seed=77):
    """Very fine local search for final polish - tiny moves."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    c = fast_autocorr(a)
    best_a = a.copy()
    best_max = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 10000:
        c_max = c.max()
        peak_pos = np.argmax(c)
        best_delta_improve = 1e-15
        best_move = None

        # Focus on the single max peak and its neighbors
        for K in [peak_pos, peak_pos - 1, peak_pos + 1,
                  (2 * n - 2) - peak_pos, (2 * n - 2) - peak_pos - 1]:
            K = max(0, min(2 * n - 2, int(K)))
            K_int = int(K)
            i_start = max(0, K_int - n + 1)
            i_end = min(n - 1, K_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[K_int - idx]
            top_i = idx[np.argsort(products)[::-1][:4]]
            for ii in top_i:
                i_int = int(ii)
                if a[i_int] < 1e-12:
                    continue
                # Try nearby destinations (within n//8)
                center = int(rng.integers(n))
                for ddx in range(-2, 3):
                    jj = (center + ddx) % n
                    if jj == i_int:
                        continue
                    j_int = jj
                    if a[j_int] > 0.3 * a[i_int]:
                        continue
                    # Exact delta
                    s = np.zeros(2 * n - 1)
                    s[j_int:j_int + n] += a
                    s[i_int:i_int + n] -= a
                    s += a[j_int] - a[i_int]
                    q = np.zeros(2 * n - 1)
                    qi = 2 * i_int
                    qj = 2 * j_int
                    kip = i_int + j_int
                    if qi < 2 * n - 1:
                        q[qi] += 1.0
                    if qj < 2 * n - 1:
                        q[qj] += 1.0
                    if kip < 2 * n - 1:
                        q[kip] -= 2.0
                    for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                        delta = a[i_int] * frac
                        if delta < 1e-14:
                            continue
                        c_new = c + delta * s + delta * delta * q
                        new_max = c_new.max()
                        if new_max < best_max - 1e-15:
                            best_delta_improve = new_max
                            best_move = (i_int, j_int, delta, s.copy(), q.copy())
                            break

        if best_move is not None:
            i, j, delta, s_vec, q_vec = best_move
            a[i] -= delta
            a[j] += delta
            c = c + delta * s_vec + delta * delta * q_vec
            if c.max() < best_max - 1e-15:
                best_max = c.max()
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            if no_improve > 500:
                break

        if verbose and no_improve % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max
            print("    FineLS (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " no_imp=" + str(no_improve))

    return best_a, best_max


def sa_search(a0, time_limit, verbose=True, seed=99):
    """Simulated annealing with exact delta evaluation."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    c = fast_autocorr(a)
    best_a = a.copy()
    best_max = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    T_start = c.max() * 0.001
    T_end = c.max() * 1e-8
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        T = T_start * (T_end / T_start) ** elapsed_frac

        # Generate a random move
        for _ in range(50):
            ii = int(rng.integers(n))
            jj = int(rng.integers(n))
            if ii == jj or a[ii] < 1e-10:
                continue
            delta = a[ii] * rng.uniform(0.005, 0.1)
            # Compute new max
            s = np.zeros(2 * n - 1)
            s[jj:jj + n] += a
            s[ii:ii + n] -= a
            s += a[jj] - a[ii]
            q = np.zeros(2 * n - 1)
            qi = 2 * ii
            qj = 2 * jj
            kip = ii + jj
            if qi < 2 * n - 1:
                q[qi] += 1.0
            if qj < 2 * n - 1:
                q[qj] += 1.0
            if kip < 2 * n - 1:
                q[kip] -= 2.0
            c_new = c + delta * s + delta * delta * q
            new_max = c_new.max()
            dE = new_max - c.max()

            if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-20)):
                a[ii] -= delta
                a[jj] += delta
                a = np.maximum(a, 0.0)
                c = c_new
                if new_max < best_max - 1e-15:
                    best_max = new_max
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                break
            else:
                no_improve += 1

        if no_improve > 50000:
            break

        if verbose and no_improve % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max
            print("    SA (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_max


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        x = np.linspace(0, 1, n_base)
        a_init = (1.0 - x) ** 1.5 + 0.1
        best_seq = norm_sum1(a_init)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(n_base) + ", score=" + repr(best_score))

    n0 = len(best_seq)
    print("Starting: n=" + str(n0) + ", score=" + repr(best_score))

    # Phase 1: Quick Nesterov to refine parent (short)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Nesterov (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.10, 150.0)
    if t1 > 30:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t1,
                                beta_start=50.0, beta_end=3000.0,
                                beta_growth=1.008, eta_base=0.5,
                                top_k_ratio=0.015, restart_every=1000,
                                verbose=True, seed=1)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 2: Main exact local search (the workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Exact Local Search (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.35, 600.0)
    if t2 > 60:
        a_out, _ = exact_local_search(norm_sum1(best_seq), t2, verbose=True, seed=10)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "els1", t_global)

    # Phase 3: SA escape
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: SA Escape (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 30:
        a_out, _ = sa_search(norm_sum1(best_seq), t3, verbose=True, seed=11)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sa1", t_global)

    # Phase 4: More exact local search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Exact Local Search 2 (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 60:
        a_out, _ = exact_local_search(norm_sum1(best_seq), t4, verbose=True, seed=20)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "els2", t_global)

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-Length (base n=" + str(n_best) + ") ===")
    lengths_to_try = []
    for dl in [500, -500, 1000, -1000, 2000]:
        nn = n_best + dl
        if 2000 <= nn <= 12000:
            lengths_to_try.append(nn)
    lengths_to_try = list(dict.fromkeys(lengths_to_try))  # deduplicate preserving order

    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("  Trying n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)
        t5 = min(remaining * 0.30, 150.0)
        if t5 > 30:
            a_out, _ = exact_local_search(a_long, t5, verbose=False, seed=30)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ml_n" + str(new_n), t_global)

    # Phase 6: Fine-grained search for final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Fine-Grained (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        a_out, _ = fine_grained_search(norm_sum1(best_seq), t6, verbose=True, seed=55)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fine1", t_global)

    # Phase 7: Another round of exact local search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Exact LS 3 (n=" + str(n_best) + ") ===")
    t7 = min(remaining * 0.25, 200.0)
    if t7 > 30:
        a_out, _ = exact_local_search(norm_sum1(best_seq), t7, verbose=True, seed=60)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "els3", t_global)

    # Phase 8: Final fine-grained
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final Fine (n=" + str(n_best) + ") ===")
    t8 = min(remaining * 0.30, 150.0)
    if t8 > 20:
        a_out, _ = fine_grained_search(norm_sum1(best_seq), t8, verbose=True, seed=77)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fine_final", t_global)

    # Phase 9: Last exact LS burst
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50:
        n_best = len(best_seq)
        print("\n=== Phase 9: Final Exact LS (n=" + str(n_best) + ") ===")
        t9 = min(remaining * 0.70, 100.0)
        if t9 > 20:
            a_out, _ = exact_local_search(norm_sum1(best_seq), t9, verbose=False, seed=88)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "els_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1((1.0 - x) ** 1.5 + 0.1)
        best_score = evaluate_safe(best_seq)

    # Ensure all values valid
    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)

    return [float(x) for x in best_seq]
