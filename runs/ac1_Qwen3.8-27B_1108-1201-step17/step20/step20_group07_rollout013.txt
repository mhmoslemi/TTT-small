def project_simplex(v):
    n=len(v)

    cssv=np.cumsum(u)
    rho=np.nonzero(u*np.arange(1,n+1) > cssv-1)[0]

def project_simplex(v):
    v=np.asarray(v,dtype=np.float64).ravel()
    n=v.size
    if n==0: return v
    if not np.isfinite(v).all(): v=np.nan_to_num(v)
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if rho.size == 0:
        return np.ones(n)/n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    out = v - theta
    out[out<0]=0.0
    # due to numerical, renormalize
    s=out.sum()
    if s <= 0: return np.ones(n)/n
    return out/s
```

```python
"""
Near-optimal search for non-negative step-function heights minimizing
    2 * n * max(convolve(a, a)) / sum(a)^2.

Main ideas:
  - Work with probability vectors (sum = 1, nonnegative), so scale is irrelevant.
  - Fast FFT-based linear convolution, gradients of weighted coefficient sums,
    Euclidean projected gradient with adaptive line search and momentum.
  - Fourier-domain "flatten the square" iterations: gently steer the spectrum
    of A^2 toward a flatter coefficient profile, then project back to a
    nonnegative length-n sequence.
  - Targeted coordinate moves driven by active coefficients (pressure = sum of
    active gradients) to polish the true max coefficient.
  - Multi-length exploration by interpolating the current best to nearby lengths.
"""

import numpy as np
import time


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_pos(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def self_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def internal_score(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        return float("inf")
    c = self_conv(p)
    return float(2.0 * p.size * c.max())


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = int(v.size)
    if n == 0:
        return v
    if not np.isfinite(v).all():
        v = np.nan_to_num(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if rho.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    r = int(rho[-1])
    theta = (cssv[r] - 1.0) / float(r + 1)
    out = v - theta
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return out / s


def correlation_lag(p, w):
    """
    h_i = sum_j p_j w_{i+j}, for i=0..n-1.
    If w are normalized weights on convolution coefficients,
    then d(sum_k w_k c_k)/d p_i = 2 h_i.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = int(p.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.array([], dtype=np.float64)
    p_rev = p[::-1]
    L = next_pow2(n + m - 1)
    Pf = np.fft.rfft(p_rev, L)
    Wf = np.fft.rfft(w, L)
    conv = np.fft.irfft(Pf * Wf, L)
    return conv[n - 1:2 * n - 1]


def make_top_weights(c, K, spread):
    c = np.asarray(c, dtype=np.float64).ravel()
    m = int(c.size)
    if m == 0:
        return np.array([], dtype=np.float64)
    K = int(max(1, min(K, m)))
    spread = float(max(1e-9, spread))

    if K >= m:
        vmax = float(c.max())
        vmin = float(c.min())
        scale = max(vmax - vmin, abs(vmax) * 1e-12, 1e-18)
        z = -spread * (vmax - c) / scale
        z = np.clip(z, -100.0, 0.0)
        w = np.exp(z)
    else:
        idx = np.argpartition(c, -K)[-K:]
        vals = c[idx]
        vmax = float(vals.max())
        vmin = float(vals.min())
        scale = max(vmax - vmin, abs(vmax) * 1e-12, 1e-18)
        zvals = -spread * (vmax - vals) / scale
        zvals = np.clip(zvals, -100.0, 0.0)
        wt = np.exp(zvals)
        s = float(wt.sum())
        if s <= 0.0 or not np.isfinite(s):
            wt = np.ones_like(vals, dtype=np.float64) / float(vals.size)
        else:
            wt /= s
        w = np.zeros(m, dtype=np.float64)
        w[idx] = wt

    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(m, dtype=np.float64) / float(m)
    return w / s


def spectral_shave(p0, time_limit, peak_ratio=0.18, valley_ratio=0.04,
                   blend=0.09, verbose=True):
    """
    Fourier-domain iteration:
      - compute C = A*A (linear convolution via zero-padded circular conv)
      - build a flatter target C_target by pulling peaks down and valleys up
      - mix the spectrum of C with the spectrum of C_target
      - take a phase-preserving square root in the frequency domain
      - project back to a nonnegative length-n sequence
    """
    t0 = time.monotonic()
    p = norm_pos(p0)
    n = int(p.size)
    Lm = 2 * n - 1
    mean = 1.0 / float(Lm)
    L = next_pow2(2 * n)

    best = p.copy()
    best_m = float(self_conv(p).max())
    best_score = float(2.0 * n * best_m)

    rng = np.random.default_rng(777)
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        A = np.zeros(L, dtype=np.float64)
        A[:n] = p
        Af = np.fft.rfft(A, L)
        B = Af * Af
        c = np.fft.irfft(B, L)[:Lm]
        c = np.maximum(c, 0.0)
        m = float(c.max())

        if m < best_m - 1e-18:
            best_m = m
            best_score = float(2.0 * n * best_m)
            best = p.copy()
            stall = 0
        else:
            stall += 1

        Cc = c.copy()
        Kp = max(32, int(Lm * peak_ratio))
        Kv = max(16, int(Lm * valley_ratio))

        if Kp < Lm:
            ipk = np.argpartition(Cc, -Kp)[-Kp:]
        else:
            ipk = np.arange(Lm)

        if Kv < Lm:
            ivl = np.argpartition(Cc, Kv)[:Kv]
        else:
            ivl = np.array([], dtype=int)

        Cnew = Cc.copy()
        over_ratio = max(0.0, m / mean - 1.0)
        peak_strength = min(0.9, 0.20 + 1.20 * over_ratio)
        valley_strength = min(0.6, 0.10 + 0.30 * over_ratio)

        Cnew[ipk] = Cc[ipk] + peak_strength * (mean - Cc[ipk])
        if ivl.size > 0:
            Cnew[ivl] = Cc[ivl] + valley_strength * (mean - Cc[ivl])

        Cnew = np.maximum(Cnew, 0.0)
        s = float(Cnew.sum())
        if s <= 0.0 or not np.isfinite(s):
            Cnew = Cc.copy()
            s = float(Cnew.sum())
        if s > 0.0 and np.isfinite(s):
            Cnew /= s

        Tp = np.zeros(L, dtype=np.float64)
        Tp[:Lm] = Cnew
        Bt = np.fft.rfft(Tp, L)

        Bnew = (1.0 - blend) * B + blend * Bt
        Bnew = np.where(np.isfinite(Bnew), Bnew, B)

        Afnew = np.empty_like(Af, dtype=np.complex128)
        absB = np.abs(B)
        maxB = float(absB.max()) if absB.size else 1.0
        if maxB <= 0.0 or not np.isfinite(maxB):
            maxB = 1.0
        mask = absB > (1e-12 * maxB)

        if np.any(mask):
            ratio = Bnew[mask] / B[mask]
            Afnew[mask] = Af[mask] * np.sqrt(ratio)
        if np.any(~mask):
            Afnew[~mask] = np.sqrt(Bnew[~mask] + 1j * 0.0)

        if Afnew.size > 0:
            Afnew[0] = complex(abs(float(np.real(Afnew[0]))), 0.0)

        Anew = np.fft.irfft(Afnew, L)
        pnew = Anew[:n].copy()
        pnew = np.maximum(pnew, 0.0)
        ss = float(pnew.sum())

        if ss < 1e-12 or not np.isfinite(ss):
            pnew = norm_pos(best + (0.02 * rng.normal(0.0, 1.0, n)) * best)
        else:
            pnew = pnew / ss

        mnew = float(self_conv(pnew).max())
        if mnew < best_m - 1e-18:
            best_m = mnew
            best_score = float(2.0 * n * best_m)
            best = pnew.copy()
            stall = 0

        if mnew <= m * 1.03 and mnew < best_m * 1.08:
            p = pnew
        else:
            if mnew > m * 1.20:
                blend = max(0.015, blend * 0.72)
            p = p

        it += 1
        if stall > 4000:
            p = norm_pos(best + (0.02 * rng.normal(0.0, 1.0, n)) * best)
            stall = 0

        if verbose and it % 500 == 0:
            el = time.monotonic() - t0
            print("    spectral it={} t={:.1f}s best={:.8f} cur={:.8f}".format(
                it, el, best_score, float(2.0 * n * m)))

    return best, best_score


def smooth_gradient_optimize(p0, time_limit,
                             spread_start=5.0, spread_end=45.0,
                             top_frac_start=0.05, top_frac_end=0.01,
                             momentum=0.60, verbose=True):
    """
    Projected gradient descent / heavy-ball on smooth approximations of the max
    coefficient. Weights are softmax-like over the current top coefficients.
    """
    t0 = time.monotonic()
    p = norm_pos(p0)
    n = int(p.size)
    Lm = 2 * n - 1

    best = p.copy()
    best_m = float(self_conv(p).max())
    best_score = float(2.0 * n * best_m)

    v = np.zeros(n, dtype=np.float64)
    alpha_factor = 0.22
    rng = np.random.default_rng(1234)
    it = 0
    stall = 0

    factors = [0.25, 0.5, 1.0, 2.0, 4.0]

    while time.monotonic() - t0 < time_limit:
        it += 1
        frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1e-9))
        spread = spread_start + (spread_end - spread_start) * frac
        top_frac = top_frac_start + (top_frac_end - top_frac_start) * frac
        K = max(32, int(Lm * top_frac))
        mom = min(0.85, momentum + 0.25 * frac)

        c = self_conv(p)
        m = float(c.max())

        if m < best_m - 1e-18:
            best_m = m
            best_score = float(2.0 * n * best_m)
            best = p.copy()
            stall = 0
        else:
            stall += 1

        w = make_top_weights(c, K, spread)
        sobj = float(np.dot(w, c))

        h = correlation_lag(p, w)
        grad = 2.0 * h
        grad -= grad.mean()

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-16:
            if stall > 100:
                p = norm_pos(best + (0.01 / (1.0 + stall / 1000.0)) *
                              rng.normal(0.0, 1.0, n) * best)
                v = np.zeros(n, dtype=np.float64)
                stall = 50
            continue

        v = mom * v + grad
        dirv = -v
        dnrm = float(np.linalg.norm(dirv))
        if not np.isfinite(dnrm) or dnrm < 1e-18:
            v = np.zeros(n, dtype=np.float64)
            continue
        dirv /= dnrm

        p_scale = float(np.sqrt(np.dot(p, p)))
        base = alpha_factor * max(p_scale, 1e-12)

        best_trial = p
        best_m_trial = m
        best_s_trial = sobj
        best_factor = 1.0

        trial_factors = factors
        if stall > 250:
            trial_factors = factors + [8.0]

        for fac in trial_factors:
            alpha = min(0.20, max(0.0, base * float(fac)))
            if alpha <= 0.0:
                continue
            tr = project_simplex(p + alpha * dirv)
            c2 = self_conv(tr)
            m2 = float(c2.max())
            s2 = float(np.dot(w, c2))

            if m2 < best_m_trial - 1e-18 or (
                    abs(m2 - best_m_trial) <= 1e-18 and s2 < best_s_trial - 1e-18):
                best_m_trial = m2
                best_s_trial = s2
                best_trial = tr
                best_factor = float(fac)

        accepted = False
        if best_m_trial < m - 1e-18:
            p = best_trial
            m = best_m_trial
            sobj = best_s_trial
            accepted = True
            if best_factor > 2.0:
                alpha_factor = min(1.6, alpha_factor * 1.12)
            elif best_factor < 0.5:
                alpha_factor = max(0.03, alpha_factor * 0.85)
            stall = 0
            if m < best_m - 1e-18:
                best_m = m
                best_score = float(2.0 * n * best_m)
                best = p.copy()
        elif stall > 300 and best_s_trial < sobj - 1e-15 and best_m_trial < best_m * 1.04:
            p = best_trial
            m = best_m_trial
            sobj = best_s_trial
            accepted = True
            alpha_factor = max(0.03, alpha_factor * 0.88)
            stall = 100
            if m < best_m - 1e-18:
                best_m = m
                best_score = float(2.0 * n * best_m)
                best = p.copy()

        if not accepted:
            stall += 1
            if stall > 400:
                noise = 0.012 / (1.0 + stall / 1200.0)
                p = norm_pos(best + noise * rng.normal(0.0, 1.0, n) * best)
                v = np.zeros(n, dtype=np.float64)
                alpha_factor = 0.18
                stall = 150

        if verbose and it % 250 == 0:
            el = time.monotonic() - t0
            print("    grad it={} t={:.1f}s best={:.8f} cur={:.8f} K={}".format(
                it, el, best_score, float(2.0 * n * m), K))

    return best, best_score


def coordinate_polish(p0, time_limit, top_k=14, max_trials=36, verbose=True):
    """
    Direct local moves from high-pressure positions to low-pressure positions,
    validated by exact FFT convolution scores.
    """
    t0 = time.monotonic()
    p = norm_pos(p0)
    n = int(p.size)
    Lm = 2 * n - 1

    best = p.copy()
    best_m = float(self_conv(p).max())
    best_score = float(2.0 * n * best_m)

    rng = np.random.default_rng(20240)
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = self_conv(p)
        m = float(c.max())

        if m < best_m - 1e-18:
            best_m = m
            best_score = float(2.0 * n * best_m)
            best = p.copy()
            stall = 0
        else:
            stall += 1

        K = max(4, min(top_k, Lm))
        top = np.argpartition(c, -K)[-K:]

        pressure = np.zeros(n, dtype=np.float64)
        for ki0 in top:
            ki = int(ki0)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            if s <= e:
                arr = np.arange(s, e + 1)
                pressure[arr] += 2.0 * p[ki - arr]

        src_list = []
        seen = set()

        for ki0 in top[:8]:
            ki = int(ki0)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            if s > e:
                continue
            arr = np.arange(s, e + 1)
            prod = p[arr] * p[ki - arr]
            if prod.size == 0 or float(prod.max()) < 1e-300:
                continue
            kth = min(4, prod.size)
            if kth < prod.size:
                order = np.argpartition(prod, prod.size - kth)[-kth:]
            else:
                order = np.arange(prod.size)
            for li0 in order:
                li = int(li0)
                i = int(arr[li])
                j = ki - i
                if 0 <= j < n:
                    src = i if p[i] >= p[j] else j
                    if src not in seen:
                        seen.add(src)
                        src_list.append(src)

        if pressure.size > 0:
            hi_pressure = np.argsort(-pressure)[:8]
            for pos0 in hi_pressure:
                pos = int(pos0)
                if pos not in seen and pressure[pos] > 0.0:
                    seen.add(pos)
                    src_list.append(pos)

        low_pressure = np.argsort(pressure)[:40]
        dst_list = []
        for pos0 in low_pressure:
            pos = int(pos0)
            if pos not in seen and len(dst_list) < 8:
                dst_list.append(pos)
        if len(dst_list) < 4:
            dst_list = [int(pos0) for pos0 in low_pressure[:8]]

        improved = False
        trials = 0

        for src in src_list[:8]:
            if p[src] <= 1e-16:
                continue
            for dst in dst_list[:8]:
                if dst == src:
                    continue
                if p[dst] <= 0.0 and p[src] <= 0.0:
                    continue
                for frac in [0.04, 0.12, 0.30]:
                    trials += 1
                    amt = p[src] * frac
                    if amt <= 0.0:
                        continue
                    p2 = p.copy()
                    p2[src] -= amt
                    p2[dst] += amt
                    m2 = float(self_conv(p2).max())

                    if m2 < best_m - 1e-18:
                        p = p2
                        best_m = m2
                        best_score = float(2.0 * n * best_m)
                        best = p2.copy()
                        m = m2
                        improved = True
                        break

                    if m2 < m - 1e-18:
                        p = p2
                        m = m2
                        improved = True
                        break

                    if trials >= max_trials:
                        break
                if improved or trials >= max_trials:
                    break
            if improved or trials >= max_trials:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 120:
                hi = np.argsort(-pressure).min(20, pressure.size)
                lo = np.argsort(pressure).min(20, pressure.size)
                for _ in range(18):
                    if len(hi) == 0 or len(lo) == 0:
                        break
                    src = int(hi[rng.integers(len(hi))])
                    dst = int(lo[rng.integers(len(lo))])
                    if src == dst:
                        continue
                    frac = float(rng.uniform(0.01, 0.20))
                    amt = p[src] * frac
                    if amt <= 0.0:
                        continue
                    p2 = p.copy()
                    p2[src] -= amt
                    p2[dst] += amt
                    m2 = float(self_conv(p2).max())
                    if m2 < m - 1e-18:
                        p = p2
                        m = m2
                        if m < best_m - 1e-18:
                            best_m = m
                            best_score = float(2.0 * n * best_m)
                            best = p2.copy()
                        stall = 25
                        improved = True
                        break

            if stall > 400:
                p = norm_pos(best + (0.012 / (1.0 + stall / 2000.0)) *
                              rng.normal(0.0, 1.0, n) * best)
                stall = 150

        if verbose and it % 80 == 0:
            el = time.monotonic() - t0
            print("    coord it={} t={:.1f}s best={:.8f} cur={:.8f}".format(
                it, el, best_score, float(2.0 * n * m)))

    return best, best_score


def interpolate_sequence(p, new_n):
    p = norm_pos(p)
    n = int(p.size)
    new_n = int(new_n)
    if new_n <= 0:
        return p.copy()
    if new_n == n:
        return p.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    q = np.interp(x_new, x_old, p)
    return norm_pos(q)


def fourier_random_start(n, scale=0.55, seed=0):
    n = int(max(2, n))
    rng = np.random.default_rng(seed)
    i = np.arange(n, dtype=np.float64)
    base = 1.0 / float(n)
    a = np.full(n, base, dtype=np.float64)

    denom = max(1, n - 1)
    a += base * scale * rng.uniform(-1.0, 1.0) * np.sin(np.pi * i / float(denom))
    a += base * scale * rng.uniform(-0.5, 0.5) * np.cos(np.pi * i / float(denom))

    for r in range(2, 14):
        ang = 2.0 * np.pi * float(r) * i / float(n)
        a += base * scale * rng.uniform(-0.45, 0.45) * np.cos(ang)
        a += base * scale * rng.uniform(-0.45, 0.45) * np.sin(ang)

    return norm_pos(a)


def mix_search(a, b, n_grid=81):
    a = norm_pos(a)
    b = norm_pos(b)
    n = int(a.size)
    if b.size != n:
        b = interpolate_sequence(b, n)

    best = a.copy()
    best_s = internal_score(best)
    best_alpha = 0.0

    for t in range(1, n_grid):
        alpha = float(t) / float(n_grid)
        p = norm_pos((1.0 - alpha) * a + alpha * b)
        s = internal_score(p)
        if s < best_s:
            best_s = s
            best = p.copy()
            best_alpha = alpha

    if 0.0 < best_alpha < 1.0:
        lo = max(0.0, best_alpha - 0.06)
        hi = min(1.0, best_alpha + 0.06)
        for t in range(1, 41):
            alpha = lo + (hi - lo) * float(t) / float(40)
            p = norm_pos((1.0 - alpha) * a + alpha * b)
            s = internal_score(p)
            if s < best_s:
                best_s = s
                best = p.copy()
                best_alpha = alpha

    return best, best_s


def try_length_change(best, new_n, time_s, verbose=True):
    best = norm_pos(best)
    new_n = int(new_n)
    if new_n <= 0 or time_s <= 0:
        return best, internal_score(best)

    a = interpolate_sequence(best, new_n)
    a = norm_pos(a)
    cur = a

    t0 = time.monotonic()
    rem = time_s - (time.monotonic() - t0)
    if rem > 20:
        a1, s1 = spectral_shave(cur, min(rem * 0.45, 80.0), verbose=False)
        if s1 < internal_score(cur):
            cur = norm_pos(a1)

    rem = time_s - (time.monotonic() - t0)
    if rem > 20:
        a2, s2 = smooth_gradient_optimize(
            cur, rem,
            spread_start=10.0, spread_end=70.0,
            top_frac_start=0.03, top_frac_end=0.005,
            momentum=0.65, verbose=verbose
        )
        if s2 < internal_score(cur):
            cur = norm_pos(a2)

    cur = norm_pos(cur)
    return cur, internal_score(cur)


def base_fallback():
    n = 6700
    x = np.linspace(0.0, 1.0, n)
    a = (x + 1e-3) * (1.0 - x + 1e-3)
    return norm_pos(a)


def propose_candidate(budget_s=1800.0):
    """
    Entry point. Returns a list of non-negative floats.
    """
    t_global = time.monotonic()
    try:
        total = float(budget_s)
    except Exception:
        total = 1800.0
    total = min(total, 1800.0) - 45.0
    if total < 30.0:
        total = max(10.0, min(float(budget_s), 1800.0) - 3.0)

    print("time budget: {:.1f}s".format(total))

    best = None
    best_score = float("inf")

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.where(np.isfinite(parent), parent, 0.0)
        parent = np.maximum(parent, 0.0)
        if parent.size > 0 and float(parent.sum()) > 1e-12:
            best = norm_pos(parent)
            best_score = internal_score(best)
            print("parent: n={} score={:.8f}".format(best.size, best_score))
    except Exception:
        best = None

    if best is None:
        # If parent unavailable, try a few robust parametric/Fourier starts.
        cands = []
        for n_try in [6500, 6700, 7000]:
            x = np.linspace(0.0, 1.0, n_try)
            a = (x + 1e-3) ** 1.1 * (1.0 - x + 1e-3) ** 0.9
            cands.append(norm_pos(a))
            cands.append(fourier_random_start(n_try, scale=0.55, seed=n_try))
        for cnd in cands:
            s = internal_score(cnd)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best = cnd.copy()
        if best is None:
            best = base_fallback()
            best_score = internal_score(best)
        print("initial fallback: n={} score={:.8f}".format(best.size, best_score))

    def elapsed():
        return time.monotonic() - t_global

    def remaining():
        return max(0.0, total - elapsed())

    def update_best(out, sc, label):
        nonlocal best, best_score
        if out is None or not np.isfinite(sc):
            return
        out = norm_pos(out)
        sc = internal_score(out)
        if np.isfinite(sc) and sc < best_score - 1e-13:
            best = out.copy()
            best_score = float(sc)
            print("  *** NEW BEST [{}]: {:.8f} n={} t={:.1f}s".format(
                label, best_score, best.size, elapsed()))

    # Quick mixture if there is already a separate fresh candidate useful?
    # We generate one diverse candidate only if current best is not near target.
    if remaining() > 180.0 and best_score > 1.52:
        n0 = int(best.size)
        for seed, nn in [(101, n0), (202, min(int(n0 * 1.1), 12000))]:
            if remaining() < 90.0:
                break
            a = fourier_random_start(nn, scale=0.6, seed=seed)
            a, _ = spectral_shave(a, min(remaining() * 0.3, 60.0), verbose=False)
            a, _ = smooth_gradient_optimize(
                a, min(remaining() * 0.35, 60.0),
                spread_start=8.0, spread_end=50.0,
                top_frac_start=0.04, top_frac_end=0.008,
                momentum=0.6, verbose=False
            )
            update_best(a, internal_score(a), "fresh_{}".format(nn))
            if remaining() < 60.0:
                break

    # Main schedule.  Each phase starts from current best and can update it.
    phase_list = [
        ("spectral1", 0.10, 150.0, None),
        ("smooth1", 0.30, 540.0, None),
        ("coord1", 0.08, 150.0, None),
        ("smooth2", 0.20, 360.0, None),
        ("mix_question", 0.0, 0.0, None),
        ("spectral2", 0.06, 90.0, None),
        ("length1", 0.10, 180.0, None),
        ("final", 0.30, 420.0, None),
    ]

    last_diverse = None
    last_diverse_score = float("inf")

    for name, frac, cap, _ in phase_list:
        rem = remaining()
        if rem < 20.0:
            break
        phase_time = min(rem * frac, cap)
        if phase_time < 15.0:
            continue

        if name == "mix_question":
            # If a diverse candidate is available, mix it with current best.
            if last_diverse is not None and last_diverse.size == best.size:
                mixa, mixs = mix_search(best, last_diverse, n_grid=61)
                update_best(mixa, mixs, "mix")
            continue

        out = None
        sc = float("inf")

        if name == "spectral1":
            out, sc = spectral_shave(best, phase_time,
                                     peak_ratio=0.18, valley_ratio=0.04,
                                     blend=0.09, verbose=True)
        elif name == "smooth1":
            out, sc = smooth_gradient_optimize(
                best, phase_time,
                spread_start=4.0, spread_end=35.0,
                top_frac_start=0.05, top_frac_end=0.01,
                momentum=0.58, verbose=True
            )
        elif name == "coord1":
            out, sc = coordinate_polish(best, phase_time,
                                        top_k=14, max_trials=36, verbose=True)
        elif name == "smooth2":
            out, sc = smooth_gradient_optimize(
                best, phase_time,
                spread_start=18.0, spread_end=75.0,
                top_frac_start=0.02, top_frac_end=0.004,
                momentum=0.68, verbose=True
            )
        elif name == "spectral2":
            out, sc = spectral_shave(best, phase_time,
                                     peak_ratio=0.12, valley_ratio=0.03,
                                     blend=0.05, verbose=True)
        elif name == "length1":
            n_best = int(best.size)
            candidates = []
            for mult in [0.92, 1.05, 1.15, 1.30]:
                nn = int(round(n_best * mult))
                if 2500 <= nn <= 13000:
                    candidates.append(nn)
            nn_fixed = min(12000, n_best + 2500)
            if nn_fixed > n_best:
                candidates.append(nn_fixed)

            # unique and filter
            uniq = []
            seen = set()
            for nn in candidates:
                if nn not in seen and nn != n_best:
                    seen.add(nn)
                    uniq.append(nn)

            if len(uniq) > 2:
                uniq = uniq[:2]

            for idx, nn in enumerate(uniq):
                rem = remaining()
                if rem < 60.0:
                    break
                tt = min(rem * 0.55, 160.0)
                if tt < 30.0:
                    break
                oa, sa = try_length_change(best, nn, tt, verbose=False)
                update_best(oa, sa, "len_{}".format(nn))
                if sa < best_score + 0.02 and oa is not None:
                    last_diverse = oa.copy()
                    last_diverse_score = sa

                # Immediately mix with current best, because a slightly worse
                # different basin can combine helpfully.
                if oa is not None and oa.size == best.size:
                    mixa, mixs = mix_search(best, oa, n_grid=41)
                    update_best(mixa, mixs, "mix_len_{}".format(nn))

        elif name == "final":
            out, sc = smooth_gradient_optimize(
                best, phase_time,
                spread_start=40.0, spread_end=160.0,
                top_frac_start=0.008, top_frac_end=0.002,
                momentum=0.75, verbose=True
            )

        else:
            continue

        if out is not None and np.isfinite(sc):
            update_best(out, sc, name)
            if out.size == best.size and abs(sc - best_score) < max(1e-12, 0.005 * best_score):
                last_diverse = out.copy()
                last_diverse_score = sc

        # If a diverse candidate exists and phase didn't consume all time, try mix.
        if last_diverse is not None and last_diverse.size == best.size and remaining() > 40.0:
            mixa, mixs = mix_search(best, last_diverse, n_grid=41)
            update_best(mixa, mixs, "mix_after_" + name)

    # Use any remaining time for a high-beta polish on the current best.
    rem = remaining()
    if rem > 35.0:
        out, sc = smooth_gradient_optimize(
            best, rem - 5.0,
            spread_start=80.0, spread_end=300.0,
            top_frac_start=0.005, top_frac_end=0.0015,
            momentum=0.80, verbose=True
        )
        update_best(out, sc, "final_leftover_smooth")

    # One last coordinate polish if time remains.
    rem = remaining()
    if rem > 40.0:
        out, sc = coordinate_polish(best, rem - 5.0, top_k=10, max_trials=30,
                                    verbose=True)
        update_best(out, sc, "final_leftover_coord")

    # Normalize, clip tiny negatives, and validate against evaluator if available.
    final = np.asarray(best, dtype=np.float64).ravel()
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.maximum(final, 0.0)
    s = float(final.sum())
    if s > 0.0:
        final = final / s
    else:
        final = base_fallback()

    final_list = [float(x) for x in final]

    final_score = internal_score(final)
    try:
        eval_score = evaluate_sequence(final_list)
        if np.isfinite(eval_score):
            final_score = float(eval_score)
            print("FINAL evaluator score: {:.8f} n={}".format(final_score, len(final_list)))
        else:
            print("FINAL evaluator invalid, internal score: {:.8f}".format(final_score))
    except Exception:
        print("FINAL internal score only: {:.8f} n={}".format(final_score, len(final_list)))

    if not np.isfinite(final_score):
        fallback = base_fallback()
        final_list = [float(x) for x in fallback]
        final_score = internal_score(fallback)
        try:
            es = evaluate_sequence(final_list)
            if np.isfinite(es):
                final_score = float(es)
        except Exception:
            pass

    return final_list
