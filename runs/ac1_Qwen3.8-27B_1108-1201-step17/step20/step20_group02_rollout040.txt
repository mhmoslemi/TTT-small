def norm_sym(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0: return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size > 1:
        arr = 0.5 * (arr + arr[::-1])
    s = float(arr.sum())
    if not np.isfinite(s) or s < 1e-15:
        arr = np.ones_like(arr, dtype=np.float64)
        if arr.size == 0: return None
        s = float(arr.size)
    return arr / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for nonnegative sequences.

Approach:
- Always project onto the symmetric simplex; this never increases the objective.
- Start from the provided parent, then try mild smoothing / phase averaging.
- Use a dual-smoothed multiplicative mirror method where the dual weights
  are softmax of the current convolution peaks.
- Add a Newton-inspired correction using the structured Hessian operator
  P(x)_i = n * sum_j p_{i+j} x_j, applied by FFT.
- Use a coordinated exact two-pair mass-transfer polish, with linear
  estimates to select transfers and FFT verification for acceptance.
- Explore nearby sequence lengths by interpolation and short refinement runs.
"""

import numpy as np
import time

_MODULE_N_IDX = {}


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _idx_array(n):
    key = int(n)
    if key not in _MODULE_N_IDX:
        _MODULE_N_IDX[key] = (2 * key - 2) - np.arange(key, dtype=np.int64)
    return _MODULE_N_IDX[key]


def norm_sym(a):
    """Normalize to sum 1, clip negatives, and enforce palindromic symmetry."""
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size > 1:
        arr = 0.5 * (arr + arr[::-1])
    s = float(arr.sum())
    if not np.isfinite(s) or s < 1e-15:
        arr = np.ones_like(arr, dtype=np.float64)
        s = float(arr.size)
    return arr / s


def conv_fft(a):
    """Full linear convolution a*a plus the FFT of a with a large power-of-two size."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    size = 2 * n - 1
    N = _next_pow2(3 * n - 2)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.maximum(c, 0.0)
    return c, A, N


def conv_only(a):
    """Full linear convolution a*a."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    size = 2 * n - 1
    N = _next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return np.maximum(c, 0.0)


def cross_from_a(a, p, A, N, n):
    """
    G_i = sum_j a_j p_{i+j}, for i=0..n-1.
    Uses a precomputed A = fft(a, N) and p length 2n-1.
    """
    P = np.fft.rfft(p, N)
    y = np.fft.irfft(A * P, N)
    idx = _idx_array(n)
    out = y[idx]
    return np.maximum(out.real, 0.0)


def conv_on_a(a, b, A, N, n):
    """Linear convolution a*b, using precomputed A."""
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def interp_seq(a, new_n):

def smooth_candidates(a):

def matvec_shift(x, p_hat, N, n):

            no_improve += 1
            continue

        a_safe = a + 1e-12
        pos_alpha = 1e18
        neg = d < -1e-15
        if np.any(neg):
            ratios = a_safe[neg] / (-d[neg])
            if ratios.size:
                pos_alpha = float(np.min(ratios))
        cap = 5.0 / (dnorm + 1e-12)
        max_alpha = min(1.0, 0.99 * pos_alpha, cap)
        if not np.isfinite(max_alpha) or max_alpha <= 1e-12:
            no_improve += 1
            continue

        best_trial_M = M
        best_trial_a = None
        for frac in (0.1, 0.2, 0.35, 0.5, 0.7, 1.0):
            alpha = frac * max_alpha
            trial = a + alpha * d
            trial = np.maximum(trial, 0.0)
            trial = norm_sym(trial)
            if trial is None:
                continue
            m2 = fast_score(trial)
            if m2 < best_trial_M - 1e-15:
                best_trial_M = m2
                best_trial_a = trial

        if best_trial_a is not None and best_trial_M < M - 1e-15:
            a = best_trial_a
            M = best_trial_M
            no_improve = 0
        else:
            no_improve += 1
            beta = max(30.0, beta * 0.95)

    return best_a


def coordinate_transfer(a0, time_limit, rng_seed=9, top_r=36, top_d=36, max_pairs=70):
    a = norm_sym(a0)
    if a is None or time_limit <= 0:
        return a
    a = norm_sym(a + 1e-12)
    n = int(a.size)
    if n < 4:
        return a
    L = 2 * n - 1
    best_a = norm_sym(a)
    best_M = fast_score(best_a)
    rng = np.random.default_rng(int(rng_seed))
    no_improve = 0
    t0 = time.monotonic()
    fracs = (0.02, 0.05, 0.10, 0.20, 0.35)

    while remaining_time(t0, time_limit) > 0.75:
        a = norm_sym(a + 1e-12)
        c, A, N = conv_fft(a)
        M = float(c.max())
        if M < best_M - 1e-18:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        if not np.isfinite(M) or M <= 1e-300:
            break

        if no_improve > 450:
            pert = norm_sym(a * np.exp(0.035 * rng.standard_normal(n)))
            a = norm_sym(0.8 * best_a + 0.2 * pert)
            a = norm_sym(a + 1e-12)
            no_improve = 0
            continue

        Kp = max(120, min(900, n // 6))
        if Kp >= L:
            idx = np.arange(L)
        else:
            idx = np.argpartition(c, -Kp)[-Kp:]
        active = c[idx]
        vals = 35.0 * (active / max(M, 1e-300) - 1.0)
        vals = np.maximum(vals, -40.0)
        pa = np.exp(vals)
        s = float(pa.sum())
        if s <= 0:
            pa = np.ones_like(pa, dtype=np.float64)
            pa /= pa.sum()
        else:
            pa = pa / s
        p = np.zeros(L, dtype=np.float64)
        p[idx] = pa

        G = cross_from_a(a, p, A, N, n)
        gr = G + G[::-1]
        src_all = (a + a[::-1]) * gr

        half = n // 2
        if half <= 0:
            break
        top_r = max(5, min(int(top_r), half))
        if top_r >= half:
            src_list = np.arange(half)
        else:
            src_list = np.argsort(src_all[:half])[-top_r:]

        top_d = max(5, min(int(top_d), n))
        dst_list = np.argsort(gr)[:top_d]

        pairs = []
        for i0 in src_list:
            i = int(i0)
            ri = n - 1 - i
            base = min(float(a[i]), float(a[ri]))
            if base <= 1e-13:
                continue
            gsrc = float(G[i] + G[ri])
            for j0 in dst_list:
                j = int(j0)
                rj = n - 1 - j
                if j == i or j == ri or rj == i or rj == ri:
                    continue
                gdst = float(G[j] + G[rj])
                profit = gsrc - gdst
                if profit <= 1e-18:
                    continue
                pairs.append((profit * base, i, j))

        no_candidate = len(pairs) == 0
        if no_candidate:
            no_improve += 1
            continue

        pairs.sort(reverse=True)
        est_candidates = []

        for _, i, j in pairs[:int(max_pairs)]:
            ri = n - 1 - i
            rj = n - 1 - j
            base = min(float(a[i]), float(a[ri]))
            if base <= 1e-13:
                continue
            B = np.zeros(L, dtype=np.float64)
            B[j:j + n] += a
            B[i:i + n] -= a
            B[rj:rj + n] += a
            B[ri:ri + n] -= a
            for frac in fracs:
                delta = base * float(frac)
                arr = c + (2.0 * delta) * B
                m_est = float(arr.max())
                if m_est < M - 1e-16:
                    est_candidates.append((m_est, i, j, delta))

        if not est_candidates:
            no_improve += 1
            continue

        est_candidates.sort()
        accepted = False
        for m_est, i, j, delta in est_candidates[:6]:
            if m_est >= M - 1e-14:
                break
            ri = n - 1 - i
            rj = n - 1 - j
            delta = min(float(delta), min(float(a[i]), float(a[ri])))
            if delta <= 1e-15:
                continue
            trial = a.copy()
            trial[i] -= delta
            trial[ri] -= delta
            trial[j] += delta
            trial[rj] += delta
            trial = np.maximum(trial, 0.0)
            trial = 0.5 * (trial + trial[::-1])
            ss = float(trial.sum())
            if ss <= 1e-15:
                continue
            trial = trial / ss
            m2 = float(conv_only(trial).max())
            if m2 < M - 1e-15:
                a = trial
                M = m2
                if M < best_M - 1e-18:
                    best_M = M
                    best_a = a.copy()
                    no_improve = 0
                accepted = True
                break

        if not accepted:
            no_improve += 1

    return best_a


def fw_multiplicative(a0, time_limit, rng_seed=13):
    a = norm_sym(a0)
    if a is None or time_limit <= 0:
        return a
    a = norm_sym(a + 1e-12)
    n = int(a.size)
    L = 2 * n - 1
    best_a = a.copy()
    best_M = fast_score(a)
    rng = np.random.default_rng(int(rng_seed))
    active_frac = 0.05
    beta_v = 45.0
    no_outer = 0
    t0 = time.monotonic()

    while remaining_time(t0, time_limit) > 1.0:
        c, A, N = conv_fft(a)
        M = float(c.max())
        if M < best_M - 1e-18:
            best_M = M
            best_a = a.copy()
            no_outer = 0
        if not np.isfinite(M) or M <= 1e-300:
            break

        improved = False
        Ky = max(60, min(700, int(n * active_frac)))

        for bt in range(3):
            if remaining_time(t0, time_limit) <= 0.5:
                break
            if bt == 0:
                b = best_a.copy()
            elif bt == 1:
                b = norm_sym(np.full(n, 1.0, dtype=np.float64))
            else:
                b = norm_sym(best_a + 0.05 * rng.standard_normal(n))
            if b is None:
                continue

            curr_y = conv_on_a(a, b, A, N, n)
            maxy = float(curr_y.max())
            inner_no = 0
            inner_best_b = b
            inner_best_y = curr_y

            for _ in range(24):
                if remaining_time(t0, time_limit) <= 0.25:
                    break
                if maxy <= M - max(1e-12, 1e-9 * M):
                    break

                if Ky >= L:
                    idxy = np.arange(L)
                else:
                    idxy = np.argpartition(curr_y, -Ky)[-Ky:]
                my = float(curr_y.max())
                if my <= 1e-300:
                    break
                vals = beta_v * (curr_y[idxy] / my - 1.0)
                vals = np.maximum(vals, -40.0)
                pa = np.exp(vals)
                s = float(pa.sum())
                if s <= 0:
                    pa = np.ones_like(pa, dtype=np.float64)
                    pa /= pa.sum()
                else:
                    pa = pa / s
                v = np.zeros(L, dtype=np.float64)
                v[idxy] = pa

                H = cross_from_a(a, v, A, N, n) * float(n)
                H = np.maximum(H, 0.0)
                H = H - float(H.min())
                hmax = float(H.max())
                if hmax <= 1e-12:
                    break
                H = H / hmax
                logb = np.log(b + 1e-30)

                changed = False
                best_my = maxy
                for eta in (0.10, 0.30, 0.80):
                    bb = np.exp(logb - eta * H)
                    bb = np.maximum(bb, 0.0)
                    bb = 0.5 * (bb + bb[::-1])
                    ss = float(bb.sum())
                    if ss <= 0:
                        continue
                    bb = bb / ss
                    bb = bb + 1e-12
                    bb = bb / float(bb.sum())
                    yb = conv_on_a(a, bb, A, N, n)
                    mb = float(yb.max())
                    if mb < best_my - 1e-15:
                        best_my = mb
                        inner_best_b = bb
                        inner_best_y = yb
                        changed = True

                if changed:
                    b = inner_best_b
                    curr_y = inner_best_y
                    maxy = best_my
                    inner_no = 0
                else:
                    inner_no += 1
                    if inner_no > 12:
                        beta_v = min(350.0, beta_v * 1.25)
                        b = norm_sym(np.maximum(b, 1e-30) * np.exp(0.05 * rng.standard_normal(n)))
                        curr_y = conv_on_a(a, b, A, N, n)
                        maxy = float(curr_y.max())
                        inner_no = 0

            if maxy < M - 1e-14:
                for alpha in (0.08, 0.2, 0.4, 0.7, 1.0):
                    trial = (1.0 - alpha) * a + alpha * b
                    trial = norm_sym(trial)
                    if trial is None:
                        continue
                    m2 = fast_score(trial)
                    if m2 < M - 1e-15:
                        a = trial
                        M = m2
                        if M < best_M - 1e-18:
                            best_M = M
                            best_a = a.copy()
                        improved = True
                        break

            if improved:
                break

        if not improved:
            no_outer += 1
            if no_outer % 4 == 0:
                active_frac = min(0.18, active_frac * 1.25)
                beta_v = 25.0
                pert = norm_sym(best_a * np.exp(0.06 * rng.standard_normal(n)))
                a = norm_sym(0.82 * best_a + 0.18 * pert)
                a = norm_sym(a + 1e-12)
                no_outer = 0

    return best_a


def clean_return_arr(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(5000, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        a = np.ones_like(a, dtype=np.float64)
        s = float(a.size)
    return [float(x) for x in a]


def propose_candidate(budget_s=1800.0):
    t0 = time.monotonic()
    total = min(float(budget_s) - 90.0, 1710.0)
    if total < 120.0:
        total = max(60.0, float(budget_s) - 30.0)
    print("Total search budget: " + repr(total) + "s")

    best_seq, best_score = get_initial_best()
    print("Initial best: " + repr(best_score) + " n=" + str(len(best_seq)))

    best_seq = local_transforms(best_seq)
    if best_seq is not None:
        sc = fast_score(best_seq)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            print("After local transforms: " + repr(best_score))

    # Phase 1: main mirror minimax
    rem = remaining_time(t0, total)
    t1 = min(rem * 0.30, 520.0)
    if t1 > 40:
        seq = mirror_minimax(best_seq, t1, rng_seed=11, k_frac=0.05, beta0=60.0)
        best_seq, best_score = try_update(best_seq, best_score, seq, "mirror1", t0)
        best_seq = local_transforms(best_seq)
        if best_seq is not None:
            sc = fast_score(best_seq)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc

    # Phase 2: coordinate transfer polish
    rem = remaining_time(t0, total)
    t2 = min(rem * 0.13, 180.0)
    if t2 > 35 and len(best_seq) <= 12000:
        seq = coordinate_transfer(best_seq, t2, rng_seed=19, top_r=36, top_d=36, max_pairs=70)
        best_seq, best_score = try_update(best_seq, best_score, seq, "coord1", t0)
        best_seq = local_transforms(best_seq)
        if best_seq is not None:
            sc = fast_score(best_seq)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc

    # Phase 3: linearized direction search
    rem = remaining_time(t0, total)
    t3 = min(rem * 0.10, 120.0)
    if t3 > 30 and len(best_seq) <= 12000:
        seq = fw_multiplicative(best_seq, t3, rng_seed=31)
        best_seq, best_score = try_update(best_seq, best_score, seq, "fw", t0)
        best_seq = local_transforms(best_seq)
        if best_seq is not None:
            sc = fast_score(best_seq)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc

    # Phase 4: second mirror pass with different focus
    rem = remaining_time(t0, total)
    t4 = min(rem * 0.20, 300.0)
    if t4 > 40:
        seq = mirror_minimax(best_seq, t4, rng_seed=53, k_frac=0.08, beta0=40.0)
        best_seq, best_score = try_update(best_seq, best_score, seq, "mirror2", t0)

    # Phase 5: Newton-style curvature correction
    rem = remaining_time(t0, total)
    t5 = min(rem * 0.08, 100.0)
    if t5 > 25 and len(best_seq) <= 12000:
        seq = newton_refine(best_seq, t5, rng_seed=71, k_frac=0.07, beta0=70.0)
        best_seq, best_score = try_update(best_seq, best_score, seq, "newton", t0)
        best_seq = local_transforms(best_seq)
        if best_seq is not None:
            sc = fast_score(best_seq)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc

    # Phase 6: nearby sequence lengths
    rem = remaining_time(t0, total)
    if rem > 360 and len(best_seq) <= 12000:
        base_n = int(len(best_seq))
        raw_ns = [int(0.85 * base_n), int(1.15 * base_n), int(0.72 * base_n), int(1.28 * base_n)]
        ns = []
        for nn in raw_ns:
            if 3500 <= nn <= 10000:
                dup = False
                for oo in ns:
                    if abs(nn - oo) < 250:
                        dup = True
                        break
                if not dup:
                    ns.append(nn)
        tuples = []
        for nn in ns:
            tuples.append((abs(nn - base_n), nn))
        tuples.sort()
        tried = 0
        for _, nn in tuples:
            if tried >= 2:
                break
            tried += 1
            rem = remaining_time(t0, total)
            if rem < 180:
                break
            seq = interp_seq(best_seq, nn)
            if seq is None:
                continue
            print("Multi-n candidate: n=" + str(nn))
            tm = min(rem * 0.18, 150.0)
            if tm > 30:
                seq2 = mirror_minimax(seq, tm, rng_seed=100 + nn, k_frac=0.06, beta0=70.0)
                best_seq, best_score = try_update(best_seq, best_score, seq2, "mirror_n" + str(nn), t0)

            rem = remaining_time(t0, total)
            tc = min(rem * 0.07, 70.0)
            if tc > 15 and len(best_seq) <= 12000:
                seq2 = coordinate_transfer(best_seq, tc, rng_seed=200 + nn, top_r=32, top_d=32, max_pairs=55)
                best_seq, best_score = try_update(best_seq, best_score, seq2, "coord_n" + str(nn), t0)

    # Phase 7: final polish
    rem = remaining_time(t0, total)
    best_seq = local_transforms(best_seq)
    if best_seq is not None:
        sc = fast_score(best_seq)
        if np.isfinite(sc) and sc < best_score:

    t6 = min(rem * 0.22, 160.0)
    if t6 > 20 and len(best_seq) <= 12000:
        seq = coordinate_transfer(best_seq, t6, rng_seed=999, top_r=44, top_d=44, max_pairs=90)

    rem = remaining_time(t0, total)
    t7 = min(rem * 0.25, 180.0)
    if t7 > 20:
        seq = mirror_minimax(best_seq, t7, rng_seed=1234, k_frac=0.09, beta0=35.0)
        best_seq, best_score = try_update(best_seq, best_score, seq, "mirror_final", t0)

    rem = remaining_time(t0, total)

    if t8 > 10 and len(best_seq) <= 12000:
        seq = coordinate_transfer(best_seq, t8, rng_seed=2025, top_r=50, top_d=50, max_pairs=110)
        best_seq, best_score = try_update(best_seq, best_score, seq, "coord_final2", t0)

    best_seq = norm_sym(best_seq)
    if best_seq is None:
        best_seq = make_sine(5000)

    fs = safe_score_arr(best_seq)
    if np.isfinite(fs):
        print("FINAL safe score: " + repr(fs) + " n=" + str(len(best_seq)))
    else:
        fs = fast_score(best_seq)
        print("FINAL fast score: " + repr(fs) + " n=" + str(len(best_seq)))

    out = clean_return_arr(best_seq)
    elapsed = time.monotonic() - t0
    print("Returned after " + str(round(elapsed, 1)) + "s, score estimate=" + repr(fs))
    return out
