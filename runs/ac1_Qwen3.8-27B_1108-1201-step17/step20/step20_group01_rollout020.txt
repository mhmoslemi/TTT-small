def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def spectral_flatten(a0, time_limit, alpha0=0.7, n_iter=500, verbose=True):
    """
    Iterative Amplitude Adjustment in the frequency domain.
    Compresses the dynamic range of the power spectrum to flatten the autocorrelation.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    alpha = alpha0
    stall = 0
    max_stall = 100
    
    for t in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break
        
        # Compute DFT
        A = np.fft.rfft(a, n=fft_size)
        power = np.abs(A) ** 2
        
        # Compress: new_power = power^alpha
        # Use log-space for numerical stability
        log_power = np.log(power + 1e-30)
        new_log_power = alpha * log_power
        # Renormalize so that total power is preserved
        new_log_power -= new_log_power.sum() / len(new_log_power) * (1 - alpha)
        new_power = np.exp(new_log_power)
        
        # Reconstruct A_new with same phase
        phase = A / (np.abs(A) + 1e-30)
        A_new = np.sqrt(new_power) * phase
        
        # Inverse DFT
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Evaluate
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Try to decrease alpha (more aggressive compression)
            if t > 50:
                alpha = max(0.3, alpha - 0.01)
        else:
            a = a  # keep current
            stall += 1
            # Relax alpha
            alpha = min(0.95, alpha + 0.02)

            print("    Spectral t={} ({:.1f}s) J={:.7f} alpha={:.3f} stall={}".format(
                t, time.monotonic() - t0, 2*n*best_m, alpha, stall))
    
    return best_a, 2.0 * n * best_m

            for k in top_v:
                active_set.add(int(k))
        
        # Golden section line search
        golden = (np.sqrt(5) - 1) / 2  # ~0.618
        lo = 0.0
        hi = 1.0
        c = hi - golden * (hi - lo)
        d = lo + golden * (hi - lo)
        
        def line_obj(alpha):
            trial = np.maximum((1 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                return 1e10
            return fast_autocorr(trial / ts).max()
        
        fc = line_obj(c)
        fd = line_obj(d)
        
        for _ in range(20):
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - golden * (hi - lo)
                fc = line_obj(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + golden * (hi - lo)
                fd = line_obj(d)
        
        best_alpha = (lo + hi) / 2.0
        
        # Apply update
        if t <= 3:
            y = g_dir.copy()
        else:
            y = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW2 t={:5d} ({:.1f}s) J={:.7f} |S|={}".format(t, elapsed, 2*n*best_m, len(active_set)))
    
    return best_a, 2.0 * n * best_m

def grad_descent_fft(a0, time_limit, n_top=200, verbose=True):
    """
    FFT-based gradient descent on the log-scaled variables.
    Uses momentum and adaptive learning rate.
    """
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Momentum
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros(n)
    vhat = np.zeros(n)
    lr = 0.02
    
    stall = 0
    rng = np.random.default_rng(123)
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 5000:
            break
        
        # Compute autocorrelation and find top lags
        ac = fast_autocorr(x)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr
        
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Select top-K lags with soft weighting
        k_sel = min(n_top, len(ac))
        idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = ac[idx]
        
        # Softmax weighting
        temp = 10.0 + 5.0 * (t / 1000.0)
        w = np.exp(temp * (top_vals - top_vals[0]))
        w = w / w.sum()
        
        # Compute gradient via FFT
        # grad_i = sum_k w_k * 2 * x[k-i] (approximately)
        # More precisely, d/dx_i [sum_k w_k * (x*x)[k]] = sum_k w_k * (x[k-i] + x[k-i] if k != 2i)
        # = 2 * sum_k w_k * x[max(0, min(n-1, k-i))] (with boundary handling)
        
        # FFT-based: grad = 2 * conv(w_padded, x) appropriately sliced
        w_padded = np.zeros(2 * n - 1)
        for i in range(len(idx)):
            ki = int(idx[i])
            w_padded[ki] = 2.0 * w[i]
        
        # conv(w_padded, x)[n-1-i] gives sum_k w_padded[k] * x[k - (n-1-i)]
        # Actually, let me use the direct formula:
        # d/dx_i [(x*x)[k]] = x[k-i] + x[k-i] (if both valid) or x[i] (if k=2i)
        # So grad_i = sum_k w_k * (x[k-i] + x[k-i]) for valid k
        
        # Using cross-correlation: grad = 2 * ifft(fft(w_padded) * conj(fft(x_padded)))
        # Simpler: grad is the cross-correlation of w_padded with x
        W = np.fft.rfft(w_padded, fft_size)
        X = np.fft.rfft(x, fft_size)
        # cross-correlation: ifft(conj(W) * X) or ifft(W * conj(X))
        # (x*w)[i] = sum_k x[k] w[k-i] = ifft(FFT(x) * FFT(w))[i]... 
        # Actually let me just do it directly for the gradient:
        # grad[i] = sum_k w_k * d/dx_i [(x*x)[k]]
        # d/dx_i [(x*x)[k]] = x[k-i] (if 0<=k-i<n) + x[k-i] (if 0<=k-i<n, same term counted twice unless k=2i)
        # Actually (x*x)[k] = sum_j x[j]*x[k-j], so d/dx_i = x[k-i] + x[k-i] = 2*x[k-i] if 0<=k-i<n
        # but if k-i == i (i.e., k=2i), then d/dx_i x[i]*x[i] = 2*x[i], so still 2*x[i]
        # Wait, (x*x)[k] = sum_j x[j]*x[k-j]. d/dx_i: the term with j=i is x[i]*x[k-i], and the term with k-j=i (j=k-i) is x[k-i]*x[i].
        # So if i != k-i (i.e., k != 2i), we get 2*x[i]*x[k-i] / x[i] = 2*x[k-i].
        # If k = 2i, we get d/dx_i (x[i]^2) = 2*x[i] = 2*x[k-i] (since k-i=i).
        # So in all cases, d/dx_i [(x*x)[k]] = 2*x[k-i] if 0 <= k-i < n, else 0.
        
        # So grad[i] = 2 * sum_{k: 0<=k-i<n} w_k * x[k-i] = 2 * sum_j w_{i+j} * x[j] (j = k-i)
        # = 2 * (w_padded * x)[i] where * is cross-correlation
        # = 2 * ifft(FFT(w_padded) * conj(FFT(x))) ... hmm this is getting complicated.
        # Let me just use the direct convolution approach.
        
        # Actually, the simplest way: 
        # grad[i] = 2 * sum_{j=0}^{n-1} w_padded[i+j] * x[j] (where w_padded[i+j] is 0 if i+j >= 2n-1)
        # This is a cross-correlation of w_padded with x.
        # cross_corr[i] = sum_j w_padded[i+j] * x[j]
        # = ifft(FFT(w_padded) * conj(FFT(x))) ... no
        # Let me just compute it as: flip w_padded, then convolve with x
        # conv(flip(w_padded), x)[i + len(w_padded) - 1] = sum_j flip(w_padded)[i+j-1] * x[j]
        # Hmm, this is getting messy. Let me just use the formula from the previous code.
        
        # From the previous code's fft_gradient:
        # flipped_w = w_padded[::-1]
        # conv_result = ifft(FFT(a) * FFT(flipped_w))
        # grad = conv_result[offset]
        
        flipped_w = w_padded[::-1]
        req_size = 3 * n - 2
        fft_size_g = 1 << (req_size - 1).bit_length()
        A = np.fft.rfft(x, fft_size_g)
        Wf = np.fft.rfft(flipped_w, fft_size_g)
        conv_result = np.fft.irfft(A * Wf, fft_size_g)
        grad = conv_result[2 * n - 2 : n - 2 : -1]
        # grad should have length n... let me check
        # conv_result has length fft_size_g
        # We want elements from index (2n-2) down to (n-2), step -1
        # That's (2n-2) - (n-2) + 1 = n elements. Good.
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        vhat = beta2 * vhat + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = vhat / (1 - beta2 ** (t + 1))
        
        x_new = x - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        x_new = np.maximum(x_new, 1e-25)
        x_new = x_new / x_new.sum()
        x = x_new
        
        # Periodic restart
        if stall > 2000 and stall % 500 == 0:
            noise = 0.03 * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x = x_r / x_r.sum()
            m = np.zeros(n)
            vhat = np.zeros(n)
            stall = 0
        
        if verbose and t % 500 == 0:
            print("    Grad t={:5d} ({:.1f}s) J={:.7f} stall={}".format(t, time.monotonic() - t0, best_val, stall))
    
    return best_a, best_val

def coord_optimize(a0, time_limit, n_targets=15, verbose=True):
    """
    Coordinate-wise optimization: for each coordinate, find the best value
    that minimizes the objective, given all other coordinates fixed.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Precompute: for coordinate i, changing a[i] to v affects:
    # conv[k] for k in [i, i+2n-2-i] (approximately)
    # Specifically, conv[k] = old_conv[k] + v*(a[k-i] + a[k-i] if k!=2i) - a[i]*(a[k-i] + a[k-i] if k!=2i)
    # = old_conv[k] + 2*(v - a[i])*a[k-i] (for valid k-i)
    # Plus: if k = 2i, conv[2i] gets v^2 - a[i]^2 instead of 2*(v-a[i])*a[i]
    
    # For efficiency, I'll do a "pass" over all coordinates, and for each,
    # do a 1D search for the optimal value.
    
    full_passes = 0
    max_passes = 200
    
    while time.monotonic() - t0 < time_limit and full_passes < max_passes:
        improved = False
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, 2.0 * n * best_m
            
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            # The contribution of a[i] to conv[k]:
            # conv[k] includes a[i] * a[k-i] (and a[k-i] * a[i] if k-i != i)
            # So if we change a[i] from old_val to new_val, the change in conv[k] is:
            # (new_val - old_val) * (a[k-i] + a[k-i] if k-i != i else new_val - old_val ... )
            # This is getting complicated. Let me use a simpler approach:
            # Just try a few candidate values for a[i] and pick the best.
            
            # Actually, for a fixed i, conv[k] as a function of a[i] is:
            # conv[k] = (other terms not involving a[i]) + a[i]*a[k-i] + a[k-i]*a[i] (if k-i != i and both valid)
            # = (other terms) + a[i] * (a[k-i] + a[k-i] if k-i != i, or a[i] if k-i == i)
            # Wait, (x*x)[k] = sum_j x[j]*x[k-j]. The terms involving x[i] are:
            # j=i: x[i]*x[k-i]
            # j=k-i: x[k-i]*x[i] (same as above if k-i == i)
            # So if k != 2i: contribution = 2*x[i]*x[k-i]
            # If k == 2i: contribution = x[i]^2
            
            # So conv[k] = base_k + a[i]*c_k where:
            # c_k = 2*a[k-i] if 0 <= k-i < n and k != 2i
            # c_k = 2*a[i] if k == 2i (but this depends on a[i] itself!)
            # Actually for k = 2i: conv[2i] includes a[i]^2, so c_{2i} = 2*a[i] (derivative)
            # and conv[2i] = base_{2i} + a[i]^2
            
            # For the purpose of 1D optimization, I'll just evaluate at a few points.
            
            # Try: move mass from i to a "quiet" position
            # Or: just try scaling a[i] by a few factors
            
            # For speed, I'll only do this for coordinates that significantly contribute to peaks
            peak_lags = np.argsort(ac)[::-1][:n_targets]
            
            # Check if coordinate i contributes to any peak
            contributes = False
            for ki in peak_lags:
                ki = int(ki)
                j = ki - i
                if 0 <= j < n and a[j] > 1e-10:
                    contributes = True
                    break
            if not contributes:
                continue
            
            # Try a few multipliers
            for mult in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
                a_test = a.copy()
                mass_moved = a[i] * (1 - mult)
                a_test[i] *= mult
                # Put the mass somewhere - for now, distribute uniformly or to a quiet spot
                if mass_moved > 0:
                    # Find quietest position
                    # (simplified: just add to uniform)
                    a_test += mass_moved / n
                a_test = np.maximum(a_test, 0)
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test = a_test / st
                Mt = fast_autocorr(a_test).max()
                if Mt < best_m - 1e-14:
                    best_m = Mt
                    best_a = a_test.copy()
                    a = a_test
                    improved = True
                    break
        
        full_passes += 1
        
        if verbose and full_passes % 20 == 0:
            print("    Coord pass={} ({:.1f}s) J={:.7f}".format(full_passes, time.monotonic() - t0, 2*n*best_m))
        
        if not improved and full_passes > 10:
            break
    
    return best_a, 2.0 * n * best_m
