import numpy as np
import time
import math

try:
    import scipy.optimize
    SCIPY_OK = True
except Exception:
    SCIPY_OK = False

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s

def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c

def fft_conv2(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c

def lse_obj_grad(a, beta, top_k=0):
    """Compute log-sum-exp smoothed objective and its gradient."""
    n = a.size
    size = 2 * n - 1
    c = fft_conv(a)
    if top_k > 0 and top_k < size:
        idx = np.argsort(c)[::-1][:top_k]
        cv = c[idx]
    else:
        cv = c
    m = float(np.max(cv))
    shifted = beta * (cv - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    f = m + math.log(se) / beta
    
    p = e / se
    # Gradient: d/da of lse = 2 * cross_conv(p, a)
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    if top_k > 0 and top_k < size:
        pf = np.zeros(size, dtype=np.float64)
        pf[idx] = p
    else:
        pf = p
    Pr = np.fft.rfft(pf[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    grad = 2.0 * conv[n-1:2*n-1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return f, grad

def softmax_params(a):
    """Convert a to log-space parameters for L-BFGS."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    a_safe = np.maximum(a, 1e-300)
    loga = np.log(a_safe)
    return loga

def from_log_params(loga):
    """Convert log-space parameters back to a non-negative sequence."""
    loga = np.asarray(loga, dtype=np.float64)
    max_log = float(np.max(loga))
    a = np.exp(loga - max_log)
    s = float(a.sum())
    if s < 1e-30:
        return np.ones(loga.size, dtype=np.float64) / loga.size
    return a / s

def obj_grad_log(loga, beta, top_k=0):
    """Objective and gradient in log-space."""
    a = from_log_params(loga)
    f_a, g_a = lse_obj_grad(a, beta, top_k)
    # Chain rule: d loga = a * (g_a - a . g_a) / sum(a)  ... actually
    # a = softmax(loga) / sum(softmax(loga)) ... 
    # Actually from_log_params does: exp(loga - max) / sum
    # da/dloga[i] = a[i] - a[i]*a[j] for j != i ... 
    # More precisely: if a = exp(v)/sum(exp(v)) where v = loga - max(loga),
    # then da/dv[i] = a[i](delta[i] - a[i])
    # And dv/dloga = I (since max is a constant shift for gradient purposes)
    # Wait, max(loga) depends on loga, so this is tricky.
    # Actually, for softmax, the gradient w.r.t. the logits is a*(g - a.g)
    # Let's just use: grad_loga = a * (g_a - dot(a, g_a))
    # This works because a = softmax(loga) up to a scale factor that doesn't matter
    # due to the normalization.
    mean_g = float(np.dot(a, g_a))
    grad_log = a * (g_a - mean_g)
    return f_a, grad_log

def run_lbfegs_log(a0, beta, top_k, maxfun=300):
    """Run L-BFGS in log-space."""
    if not SCIPY_OK:
        return a0
    loga0 = softmax_params(a0)
    try:
        res = scipy.optimize.minimize(
            lambda v: obj_grad_log(v, beta, top_k),
            loga0,
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max(10, maxfun), 'ftol': 1e-14, 'gtol': 1e-8, 'disp': False}
        )
        if res.success or hasattr(res, 'x'):
            a_out = from_log_params(res.x)
            return norm_sum1(a_out)
    except Exception:
        pass
    return a0

def coordinate_descent(a, budget_s, top_k=20):
    """Aggressive coordinate descent targeting top convolution peaks."""
    t0 = time.monotonic()
    a = norm_sum1(a.copy())
    n = a.size
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    rng = np.random.default_rng(42)
    stall = 0
    
    while time.monotonic() - t0 < budget_s - 1:
        if stall > 50:
            break
        
        # Focus on top peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        top_arr = np.zeros(2*n-1, dtype=np.float64)
        top_arr[top_idx] = 1.0 / float(top_k)
        
        # Compute "pressure" on each coordinate
        total = 3*n - 2
        N = next_pow2(total)
        A = np.fft.rfft(a, N)
        Pr = np.fft.rfft(top_arr[::-1], N)
        conv = np.fft.irfft(A * Pr, N)[:total]
        pressure = 2.0 * conv[n-1:2*n-1][::-1]
        pressure = np.where(np.isfinite(pressure), pressure, 0.0)
        
        # Find best single-coordinate transfer
        # Move from high-pressure + high-value to high-pressure + low-value
        best_improve = 1e-18
        best_move = None
        
        src_sorted = np.argsort(pressure * a)[::-1][:min(10, n)]
        dst_sorted = np.argsort(pressure)[::-1][:min(15, n)]
        
        for s in src_sorted:
            s = int(s)
            if a[s] < 1e-15:
                continue
            for d in dst_sorted:
                d = int(d)
                if d == s:
                    continue
                for frac in (0.01, 0.02, 0.05, 0.1):
                    mv = a[s] * frac
                    # Apply rank-1 update
                    c_new = c.copy()
                    c_new[d:d+n] += 2.0 * mv * a
                    c_new[s:s+n] -= 2.0 * mv * a
                    if 2*d < c_new.size:
                        c_new[2*d] += mv*mv
                    if 2*s < c_new.size:
                        c_new[2*s] += mv*mv
                    if s+d < c_new.size:
                        c_new[s+d] -= 2.0*mv*mv
                    c_new = np.maximum(c_new, 0.0)
                    m_new = float(np.max(c_new))
                    if m_new < M - best_improve:
                        best_improve = M - m_new
                        best_move = (s, d, mv, m_new, c_new.copy())
        
        if best_move is not None:
            s, d, mv, m_new, c_new = best_move
            a[s] -= mv
            a[d] += mv
            a = np.maximum(a, 0.0)
            a = a / float(a.sum())
            c = fft_conv(a)  # recompute for accuracy
            M = float(np.max(c))
            if M < best_M:
                best_M = M
                best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Perturb
            if stall > 5:
                a = (1.0 - 0.02) * best_a + 0.02 * rng.dirichlet(np.ones(n))
                a = norm_sum1(a)
                c = fft_conv(a)
                M = float(np.max(c))
                stall = 0
    
    return best_a

def game_direction(f, top_idx, iters=20):
    """Compute Frank-Wolfe direction via multiplicative weights."""
    f = norm_sum1(f)
    n = f.size
    size = 2*n - 1
    K = len(top_idx)
    if K == 0:
        return np.ones(n, dtype=np.float64) / n
    
    logits = np.zeros(K, dtype=np.float64)
    top_idx = np.asarray(top_idx, dtype=int)
    
    for _ in range(iters):
        pf = np.zeros(size, dtype=np.float64)
        pf[top_idx] = np.exp(logits - float(np.max(logits)))
        pf[top_idx] /= float(pf[top_idx].sum())
        
        # Pressure on f
        total = 3*n - 2
        N = next_pow2(total)
        A = np.fft.rfft(f, N)
        Pr = np.fft.rfft(pf[::-1], N)
        conv = np.fft.irfft(A * Pr, N)[:total]
        pressure = 2.0 * conv[n-1:2*n-1][::-1]
        pressure = np.where(np.isfinite(pressure), pressure, 0.0)
        
        pmx, pmn = float(np.max(pressure)), float(np.min(pressure))
        if pmx - pmn < 1e-30:
            g = np.ones(n, dtype=np.float64) / n
        else:
            scaled = (pressure - pmn) / (pmx - pmn)
            w = np.exp(-5.0 * scaled)
            w = np.where(np.isfinite(w), w, 0.0)
            w += 1e-15
            g = w / float(w.sum())
            g = 0.95 * g + 0.05 * (np.ones(n) / n)
        
        # Update logits
        fg = fft_conv2(f, g)
        fa = fg[top_idx]
        fa = np.where(np.isfinite(fa), fa, 0.0)
        if float(np.max(fa) - np.min(fa)) > 1e-30:
            logits += 10.0 / (float(np.max(fa) - np.min(fa))) * (fa - float(np.mean(fa)))
    
    return g

def frank_wolfe(a0, budget_s, top_k=200):
    """Frank-Wolfe with game-based directions."""
    t0 = time.monotonic()
    a = norm_sum1(a0.copy())
    n = a.size
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    stall = 0
    t = 0
    
    while time.monotonic() - t0 < budget_s - 2:
        t += 1
        top_idx = np.argsort(c)[::-1][:top_k]
        g = game_direction(a, top_idx, iters=15)
        
        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        
        # Line search
        best_alpha = 0.0
        best_m = M
        for alpha in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
            val = (1-alpha)**2 * c + 2*alpha*(1-alpha)*fg + alpha**2 * cg
            m = float(np.max(np.maximum(val, 0)))
            if m < best_m:
                best_m = m
                best_alpha = alpha
        
        if best_alpha > 0:
            a = (1 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0)
            a = a / float(a.sum())
            c = fft_conv(a)
            M = float(np.max(c))
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        if stall > 30:
            rng = np.random.default_rng(100 + t)
            a = best_a + 0.01 * rng.normal(0, 1, n) * best_a
            a = np.maximum(a, 1e-15)
            a = a / float(a.sum())
            c = fft_conv(a)
            M = float(np.max(c))
            stall = 0
    
    return best_a

def make_window(n, kind='sine2'):
    x = np.linspace(0, 1, n)
    if kind == 'sine2':
        a = np.sin(np.pi * x)**2
    elif kind == 'cosine':
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.5))
    elif kind == 'blackman':
        a = 0.42 - 0.5*np.cos(2*np.pi*x) + 0.08*np.cos(4*np.pi*x)
    elif kind == 'hamming':
        a = 0.54 + 0.46*np.cos(2*np.pi*x)
    elif kind == 'tri':
        a = np.maximum(0, 1 - 2*abs(x - 0.5))
    elif kind == 'raised':
        a = (1 + np.cos(2*np.pi*x)) / 2
    else:
        a = np.sin(np.pi * x)**2
    a = np.maximum(a, 0)
    return norm_sum1(a)

def interpolate_seq(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    new_n = int(max(1, new_n))
    old_x = np.linspace(0, 1, a.size)
    new_x = np.linspace(0, 1, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def fft_conv2(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c

def lse_obj_grad(a, beta, top_k=0):
    """Log-sum-exp smoothed objective and gradient (in a-space, sum-normalized)."""
    n = a.size
    size = 2 * n - 1
    c = fft_conv(a)
    if top_k > 0 and top_k < size:
        idx = np.argsort(c)[::-1][:top_k]
        cv = c[idx]
    else:
        idx = None
        cv = c
    m = float(np.max(cv))
    shifted = beta * (cv - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    f = m + math.log(se) / beta
    p = e / se
    pf = np.zeros(size, dtype=np.float64)
    if idx is None:
        pf[:] = p
    else:
        pf[idx] = p
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(pf[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return f, grad


def to_log(a):
    a = np.asarray(a, dtype=np.float64)
    return np.log(np.maximum(a, 1e-300))


def from_log(loga):
    max_l = float(np.max(loga))
    a = np.exp(loga - max_l)
    s = float(a.sum())
    if s < 1e-30:
        return np.ones(loga.size, dtype=np.float64) / loga.size
    return a / s


def obj_grad_log(loga, beta, top_k=0):
    a = from_log(loga)
    f, ga = lse_obj_grad(a, beta, top_k)
    mean_g = float(np.dot(a, ga))
    grad_log = a * (ga - mean_g)
    grad_log = np.where(np.isfinite(grad_log), grad_log, 0.0)
    return f, grad_log


def run_lbfegs(a0, beta, top_k, maxfun=200):
    if not SCIPY_OK:
        return a0
    log0 = to_log(a0)
    try:
        res = scipy.optimize.minimize(
            lambda v: obj_grad_log(v, beta, top_k),
            log0,
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max(20, maxfun), 'ftol': 1e-14, 'gtol': 1e-9, 'disp': False}
        )
        if hasattr(res, 'x') and res.x.size == log0.size:
            a_out = norm_sum1(from_log(res.x))
            return a_out
    except Exception:
        pass
    return a0


def game_direction(a, top_idx, iters=18):
    """FW direction via multiplicative-weights zero-sum game."""
    a = norm_sum1(a)
    n = a.size
    size = 2 * n - 1
    K = len(top_idx)
    if K == 0:
        return np.ones(n, dtype=np.float64) / n
    top_idx = np.asarray(top_idx, dtype=int)
    logits = np.zeros(K, dtype=np.float64)
    for _ in range(iters):
        pf = np.zeros(size, dtype=np.float64)
        lm = float(np.max(logits))
        w = np.exp(logits - lm)
        pf[top_idx] = w / float(w.sum())
        total = 3 * n - 2
        N = next_pow2(total)
        A = np.fft.rfft(a, N)
        Pr = np.fft.rfft(pf[::-1], N)
        conv = np.fft.irfft(A * Pr, N)[:total]
        pressure = 2.0 * conv[n - 1:2 * n - 1][::-1]
        pressure = np.where(np.isfinite(pressure), pressure, 0.0)
        pmx, pmn = float(np.max(pressure)), float(np.min(pressure))
        if pmx - pmn < 1e-30:
            g = np.ones(n, dtype=np.float64) / n
        else:
            sc = (pressure - pmn) / (pmx - pmn)
            w2 = np.exp(-6.0 * sc)
            w2 = np.where(np.isfinite(w2), w2, 0.0)
            w2 += 1e-15
            g = w2 / float(w2.sum())
            g = 0.96 * g + 0.04 * (np.ones(n) / n)
        fg = fft_conv2(a, g)
        fa = fg[top_idx]
        fa = np.where(np.isfinite(fa), fa, 0.0)
        rng_ = float(np.max(fa) - np.min(fa))
        if rng_ > 1e-30:
            logits += (12.0 / rng_) * (fa - float(np.mean(fa)))
    return g

        fg = fft_conv2(a, g)
        best_alpha = 0.0
        best_m = M

def apply_rank1(c, a, s, d, mv):
    n = a.size
    c2 = c.copy()
    c2[d:d + n] += 2.0 * mv * a
    c2[s:s + n] -= 2.0 * mv * a
    if 2 * d < c2.size:
        c2[2 * d] += mv * mv
    if 2 * s < c2.size:
        c2[2 * s] += mv * mv
    if s + d < c2.size:
        c2[s + d] -= 2.0 * mv * mv
    c2 = np.where(np.isfinite(c2), c2, 0.0)
    return c2


def coord_descent(a0, budget_s, top_k=12):

def make_window(n, kind='sine2'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'sine2':
        a = np.sin(np.pi * x) ** 2
    elif kind == 'cosine':
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (x - 0.5))
    elif kind == 'blackman':
        a = 0.42 - 0.5 * np.cos(2.0 * np.pi * x) + 0.08 * np.cos(4.0 * np.pi * x)
    elif kind == 'hamming':
        a = 0.54 + 0.46 * np.cos(2.0 * np.pi * x)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'sine4':
        a = np.sin(np.pi * x) ** 4
    else:
        a = np.sin(np.pi * x) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_seq(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)

    else:
        work_n = best_a.size
    work_a = interpolate_seq(best_a, work_n) if best_a.size != work_n else best_a.copy()

    # Phase 0: Multi-start initialization
    print("\n=== Phase 0: Multi-start ===")
    t0 = time.monotonic()
    candidates = []
    if parent_a is not None:
        candidates.append(work_a.copy())
    for kind in ['sine2', 'sine4', 'cosine', 'blackman', 'hamming', 'raised_cos']:
        candidates.append(make_window(work_n, kind))

    best_ps = proxy_score(work_a)
    best_ws = work_a.copy()
    for ci, cnd in enumerate(candidates):
        el = time.monotonic() - t_start
        if el > BUDGET * 0.12:
            break
        if SCIPY_OK:
            a_opt = run_lbfegs(cnd, 40, 0, maxfun=80)
            ps = proxy_score(a_opt)
            if ps < best_ps - 1e-8:
                best_ps = ps
                best_ws = a_opt
                print("  Start " + str(ci) + " proxy=" + str(ps))
    work_a = best_ws

    # Phase 1: L-BFGS progressive beta
    print("\n=== Phase 1: L-BFGS ===")
    t0 = time.monotonic()
    p1_budget = min(BUDGET * 0.42 - (t0 - t_start), 700.0)
    current = work_a.copy()
    cur_ps = proxy_score(current)
    b1_best = current.copy()
    b1_best_ps = cur_ps

    beta_schedule = [15, 30, 60, 120, 250, 500, 1000, 2000]
    top_schedule = [0, 0, 500, 200, 80, 40, 20, 10]

    for bi, (beta, tk) in enumerate(zip(beta_schedule, top_schedule)):
        el = time.monotonic() - t_start
        if el > BUDGET * 0.42 - 20:
            break
        rem = BUDGET * 0.42 - el
        if rem < 25:
            break
        mf = int(min(350, max(60, rem * 0.6 / max(1.0, bi + 1))))
        a_new = run_lbfegs(current, beta, tk, maxfun=mf)
        ps = proxy_score(a_new)
        if ps < b1_best_ps - 1e-10:
            b1_best_ps = ps
            b1_best = a_new
        current = a_new
        print("  beta=" + str(beta) + " tk=" + str(tk) + " proxy=" + str(ps) +
              " best=" + str(b1_best_ps))

    work_a = b1_best
    sc = safe_eval(work_a)
    if math.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = work_a.copy()
        print("  Phase1 actual: " + str(best_score))

    # Phase 2: Frank-Wolfe
    print("\n=== Phase 2: Frank-Wolfe ===")
    t0 = time.monotonic()
    p2_budget = min(BUDGET * 0.62 - (t0 - t_start), 340.0)
    if p2_budget > 40:
        a_fw = frank_wolfe(best_a, p2_budget, top_k=250)
        sc = safe_eval(a_fw)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_fw.copy()
            print("  FW: " + str(best_score))
        else:
            ps_fw = proxy_score(a_fw)
            ps_best = proxy_score(best_a)
            if ps_fw < ps_best - 1e-10:
                sc2 = safe_eval(a_fw)
                if math.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_a = a_fw.copy()

    # Phase 3: Coordinate descent
    print("\n=== Phase 3: Coord descent ===")
    t0 = time.monotonic()
    p3_budget = min(BUDGET * 0.82 - (t0 - t_start), 280.0)
    if p3_budget > 30:
        a_cd = coord_descent(best_a, p3_budget, top_k=10)
        sc = safe_eval(a_cd)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_cd.copy()
            print("  CD: " + str(best_score))

    # Phase 4: Second L-BFGS on best
    print("\n=== Phase 4: Final L-BFGS ===")
    t0 = time.monotonic()
    p4_budget = BUDGET * 0.94 - (t0 - t_start)
    if SCIPY_OK and p4_budget > 40:
        cur = norm_sum1(best_a)
        a_f1 = run_lbfegs(cur, 500, 5, maxfun=250)
        sc = safe_eval(a_f1)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_f1.copy()
            print("  L-BFGS500: " + str(best_score))
        a_f2 = run_lbfegs(norm_sum1(best_a), 1500, 3, maxfun=150)
        sc = safe_eval(a_f2)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_f2.copy()
            print("  L-BFGS1500: " + str(best_score))

    # Phase 5: Final CD
    print("\n=== Phase 5: Final CD ===")
    t0 = time.monotonic()
    p5_budget = BUDGET * 0.98 - (t0 - t_start)
    if p5_budget > 20:
        a_cd2 = coord_descent(best_a, p5_budget, top_k=6)
        sc = safe_eval(a_cd2)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_cd2.copy()
            print("  CD final: " + str(best_score))

    # Final check
    fv = safe_eval(best_a)
    if math.isfinite(fv):
        best_score = fv

    arr = np.asarray(best_a, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if arr.size == 0 or s < 0.01:
        arr = np.asarray(make_window(5000, 'sine2'), dtype=np.float64)
    else:
        arr = arr / s

    el = time.monotonic() - t_start
    print("\nFINAL: score=" + str(best_score) + " n=" + str(arr.size) + " t=" + str(el))
    return [float(x) for x in arr]
