def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

    if s < 1e-15:

    """Build A matrix where A[k,j] = y[k-j] if 0<=k-j<n else 0."""
    K = len(active_lags)
    # Use sparse representation for efficiency
    rows = []
    cols = []
    vals = []

        k = int(k)

            rows.extend([ii] * len(js))
            cols.extend(js.tolist())
            vals.extend(y[k - js].tolist())
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(K, n))

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe with scipy HiGHS (fast solver, no cvxpy overhead)
2. Adaptive active set with periodic pruning
3. Quadratic-aware line search exploiting conv((1-a)y+ag, (1-a)y+ag) structure

import numpy as np

import scipy.optimize as so
from scipy.sparse import csr_matrix

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize_arr(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_calc_ss(f_interp, ss)

def f_calc_ss(f_interp, ss):
    return f_interp / ss

def build_lp_sparse(y, active_lags, n):
    """Build sparse A matrix: A[k,j] = y[k-j] if 0<=k-j<n else 0."""
    K = len(active_lags)
    row_data = []
    col_data = []
    val_data = []
    for ii, k in enumerate(active_lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            row_data.extend([ii] * len(js))
            col_data.extend(js.tolist())
            val_data.extend(y[k - js].tolist())
    A = csr_matrix((val_data, (row_data, col_data)), shape=(K, n))
    return A

def solve_lp_fw_step(y, active_lags, n):
    """Solve the LP: min M s.t. sum(g)=1, g>=0, A@g <= M."""
    K = len(active_lags)
    if K == 0:
        return None, 0.0
    
    A = build_lp_sparse(y, active_lags, n)
    
    # Full constraint matrix: [A, -1] @ [g; M] <= 0
    # A is K x n, -1 column is K x 1
    neg_one = csr_matrix(
        np.ones((K, 1)), shape=(K, 1)
    )
    A_full = np.hstack([A.toarray(), -np.ones((K, 1))])
    
    # Objective: minimize M (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g >= 0 (first n vars), M free
    bounds = [(0, None)] * n + [(None, None)]
    
    b_ub = np.zeros(K)
    
    try:
        res = so.linprog(c, A_ub=A_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            gs = g.sum()
            if gs > 1e-15:
                g = g / gs
            M_val = res.x[-1]
            return g, M_val
        elif res.status == 3:
            g = np.maximum(res.x[:n], 0.0)
            gs = g.sum()
            if gs > 1e-15:
                g = g / gs
            M_val = res.x[-1]
            return g, M_val
    except Exception:
        pass
    return None, 0.0

def lp_fw_alphase(a0, time_limit, init_K=100, max_K=3000, verbose=True):

        M_curr = ac_y.max()
        if M_curr < best_m - 1e-14:

            no_improve += 1
        
        # Periodically refresh active set
        if t_iter % 20 == 1:

            for k in top_idx[:target_K]:

            # Prune: keep only constraints that are "tight" (within 50% of max)
            if len(active_set) > max_K:
                clist = sorted(active_set)
                margins = np.array([M_curr - ac_y[k] for k in clist])
                keep_idx = np.argsort(margins)[:max_K]
                active_set = set(int(clist[i]) for i in keep_idx)
        
        active_set.add(int(np.argmax(ac_y)))
        clist = sorted(active_set)
        K = len(clist)

        if g_dir is None:
            # LP failed, reduce active set
            if K > init_K // 2:
                clist = sorted(active_set)
                margins = np.array([M_curr - ac_y[k] for k in clist])
                keep_idx = np.argsort(margins)[:init_K]
                active_set = set(int(clist[i]) for i in keep_idx)
            continue
        
        # Check for violations to grow active set
        fft_size = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)

        violations = np.where(conv_ys > M_lp + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_lp
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                active_set.add(int(k))
        
        # Line search: minimize max_k [(1-a)^2*cy_k + 2a(1-a)*cg_k + a^2*ch_k]
        ac_gg = fast_autocorr(g_dir)
        # conv(y, g) - we approximate using the FFT
        ac_yg = conv_ys[:2*n-1]
        
        best_alpha = 2.0 / (t_iter + 2.0)
        best_trial_M = 1e10
        
        # Try candidate alphas
        for alpha in [best_alpha, 0.3, 0.5, 0.7, 0.9, 1.0]:
            trial = (1 - alpha) * y + alpha * g_dir
            ts = trial.sum()
            if ts < 1e-15:
                continue
            # Compute max of (1-a)^2*ac_y + 2a(1-a)*ac_yg + a^2*ac_gg
            M_trial = ((1-alpha)**2 * ac_y + 2*alpha*(1-alpha) * ac_yg + alpha**2 * ac_gg).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        # Bisection refinement
        lo = max(0.0, best_alpha - 0.2)
        hi = min(1.0, best_alpha + 0.2)
        for _ in range(8):
            mid1 = lo + (hi - lo) / 3.0
            mid2 = hi - (hi - lo) / 3.0
            M1 = ((1-mid1)**2 * ac_y + 2*mid1*(1-mid1) * ac_yg + mid1**2 * ac_gg).max()
            M2 = ((1-mid2)**2 * ac_y + 2*mid2*(1-mid2) * ac_yg + mid2**2 * ac_gg).max()
            if M1 < M2:
                hi = mid2
            else:
                lo = mid1
        best_alpha = (lo + hi) / 2.0
        
        # Update
        if t_iter <= 2:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy > 1e-15:

            print("  LP-FW t={:4d} ({:.1f}s) J={:.7f} |S|={}".format(
                t_iter, elapsed, 2 * n * best_m, len(active_set)))
    
    elapsed = time.monotonic() - t0
    if verbose:
        print("  LP-FW done ({:.1f}s) J={:.7f} iters={}".format(
            elapsed, 2 * n * best_m, t_iter))
    return best_a, 2.0 * n * best_m

def grad_refine_alphase(a0, time_limit, n_top=100, lr0=0.05, verbose=True):

            break
        if stall > 6000:
            break
        
        ac = fast_autocorr(x)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr
        
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Select top lags with temperature
        k_sel = min(n_top, len(ac))
        idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = ac[idx]
        temp = 20.0 + 15.0 * (t / 500.0)
        w = np.exp(temp * (top_vals - top_vals[0]))
        w = w / w.sum()
        
        # Compute gradient using FFT
        size = 2 * n - 1
        fft_size = 1 << (3 * n - 3).bit_length()
        w_padded = np.zeros(2 * n - 1)
        for i in range(len(idx)):
            ki = int(idx[i])
            if 0 <= ki < 2 * n - 1:
                w_padded[ki] = w[i]
        flipped_w = w_padded[::-1]
        A = np.fft.rfft(x, fft_size)
        W = np.fft.rfft(flipped_w, fft_size)
        conv_result = np.fft.irfft(A * W, fft_size)
        grad = conv_result[2 * n - 2 : n - 2 : -1]
        
        if grad.max() < 1e-15:
            break
        
        # Mirror descent update
        lr = lr0 / (1.0 + 0.003 * t)
        log_x = np.log(x + 1e-30) - lr * grad
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        x_new = np.maximum(x_new, 1e-25)
        x = x_new / x_new.sum()
        
        # Restart with noise if stalled

            noise_scale = 0.03 * (stall / 2000.0)

            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x = x_r / x_r.sum()
            stall = 0
        
        if verbose and t % 300 == 0:
            print("  Grad t={} ({:.1f}s) J={:.7f} stall={}".format(
                t, time.monotonic() - t0, best_val, stall))
    
    return best_a, best_val
