    """Safely evaluate the objective."""
    try:
        arr = np.asarray(a, dtype=np.float64)
        if arr.size == 0:
            return float('inf')
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if s < 0.01:
            return float('inf')
        lst = [float(x) for x in arr]
        val = evaluate_sequence(lst)
        if val is None:
            return float('inf')
        val = float(val)
        if not math.isfinite(val):
            return float('inf')
        return val
    except Exception:
        return float('inf')

def local_score(a):
    """Fast local score (proxy for evaluate_sequence)."""
    n = a.size
    c = fft_conv(a)
    s = float(a.sum())
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * float(np.max(c)) / (s * s)

def cross_pressure(peaks, a):
    """
    Compute pressure[i] = 2 * sum_{k in peaks} a[k-i] for valid indices.
    peaks: array of peak positions in the convolution (length 2n-1)
    a: sequence (length n)
    Returns: array of length n
    """
    n = a.size
    size = 2 * n - 1
    pressure = np.zeros(n, dtype=np.float64)
    # For each peak position k, a[k-i] contributes to pressure[i] for valid i
    # k-i in [0, n-1] means i in [k-(n-1), k] intersected with [0, n-1]
    # pressure[i] = 2 * sum_{k in peaks} a[max(0, min(n-1, k-i))]
    # Actually, a[k-i] is only defined when 0 <= k-i <= n-1
    # So for a given k, i ranges from max(0, k-(n-1)) to min(n-1, k)
    # and a[k-i] is the value at index k-i
    
    # More efficient: use FFT
    # pressure = 2 * cross_correlate(peak_indicator, a)
    # where peak_indicator has 1 at peak positions
    
    peak_vec = np.zeros(size, dtype=np.float64)
    peak_vec[peaks] = 1.0
    
    # We want pressure[i] = 2 * sum_{k} peak_vec[k] * a[k-i]
    # This is a cross-correlation
    # In convolution form: pressure = 2 * conv(peak_vec_rev, a) at certain offsets
    
    # Actually, let's think of it differently.
    # c[k] = sum_j a[j] * a[k-j]
    # dc[k]/da[i] = 2 * a[k-i] (when valid)
    # So grad_M[i] = 2 * sum_{k in peaks} a[k-i]
    
    # Using FFT: create a signal p of length 2n-1 with p[k] = 1 if k in peaks
    # Then grad_M = 2 * (p correlated with a), shifted appropriately
    
    # Let's just do it directly for small peak counts
    if peaks.size < 50:
        for k in peaks:
            k = int(k)
            # i ranges from max(0, k-n+1) to min(n-1, k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                # a[k-i] for i in [i_start, i_end]
                # k-i ranges from k-i_start to k-i_end, which is [min(n-1,k), max(0,k-n+1)]
                idx = k - np.arange(i_start, i_end + 1)
                pressure[i_start:i_end+1] += 2.0 * a[idx]
    else:
        # FFT approach
        # pressure[i] = 2 * sum_k p[k] * a[k-i]
        # Let's define q[j] = a[j] for j in [0,n-1], 0 otherwise
        # pressure[i] = 2 * sum_k p[k] * q[k-i]
        # This is the cross-correlation of p and q, evaluated at lag i
        # cross_corr_pq[i] = sum_k p[k] * q[k-i]
        # In FFT: cross_corr = ifft(fft(p) * conj(fft(q)))... 
        # Actually, cross-correlation in 1D:
        # (p ⋆ q)[i] = sum_k p[k] * q[k-i]
        # This equals the convolution of p with q reversed: (p * q_rev)[n-1+i]... 
        # Let me just use the direct approach with FFT
        
        # padding
        total = 3 * n - 2  # to cover all needed indices
        N = next_pow2(total)
        P = np.fft.rfft(p if False else peak_vec, N)  # need to be careful
        # Actually, let me use a different approach
        # pressure[i] = 2 * sum_{k in peaks} a[k-i]
        # = 2 * (convolve(peak_vec, a))[i + n - 1]... 
        # Hmm, let me think again.
        # conv(peak_vec, a)[m] = sum_k peak_vec[k] * a[m-k]
        # We want sum_k peak_vec[k] * a[k-i] = conv(a, peak_vec)[i]... 
        # conv(a, peak_vec)[i] = sum_j a[j] * peak_vec[i-j]
        # Let j = k-i: sum_k a[k-i] * peak_vec[i-(k-i)] = sum_k a[k-i] * peak_vec[2i-k]
        # That's not what we want either.
        
        # Let me just use: pressure[i] = 2 * (a * peak_vec_rev)[n-1+i]
        # where * denotes convolution
        # (a * peak_vec_rev)[m] = sum_j a[j] * peak_vec_rev[m-j]
        # peak_vec_rev[j] = peak_vec[n*2-2-j] (reversed, length 2n-1)
        # Hmm, this is getting complicated. Let me just do the direct sum for reasonable peak counts.
        
        # For larger peak sets, use the fact that:
        # pressure = 2 * ifft(fft(a) * conj(fft(peak_vec_padded)))... 
        # Actually, the standard cross-correlation:
        # (a ⋆ b)[i] = sum_k a[k+i] * conj(b[k]) = sum_k a[k] * conj(b[k-i])
        # We want: sum_k peak_vec[k] * a[k-i]
        # Let b = peak_vec. Then sum_k b[k] * a[k-i] = (b ⋆ a)[i] (if we define properly)
        # (b ⋆ a)[i] = sum_k b[k] * a[k+i]... no.
        
        # Standard definition: (f ⋆ g)[n] = sum_k f[k] * conj(g[k-n])
        # So (peak_vec ⋆ a)[i] = sum_k peak_vec[k] * a[k-i] (real, so no conjugate)
        # This is what we want!
        # Cross-correlation via FFT: (f ⋆ g) = ifft(conj(fft(f)) * fft(g))... 
        # Wait: (f ⋆ g)[n] = sum_k f[k+n] * conj(g[k])
        # = sum_k conj(g[k]) * f[k+n]
        # = (conj(g) * f)[n] (convolution)
        # So fft(f ⋆ g) = conj(fft(g)) * fft(f)
        
        # We want (peak_vec ⋆ a)[i] = sum_k peak_vec[k] * a[k-i]
        # This equals (a ⋆ peak_vec)[-i]... hmm, getting confused.
        
        # Let me just do: 
        # sum_k peak_vec[k] * a[k-i]
        # Define h[i] = sum_k peak_vec[k] * a[k-i]
        # = sum_k peak_vec[k] * a[(k-i)]
        # Let j = k-i: sum_j peak_vec[j+i] * a[j]
        # = (peak_shifted ⋆ a)[0] where peak_shifted[j] = peak_vec[j+i]
        # = sum_j peak_vec[j+i] * a[j]
        # In FFT: this is the inner product of peak_vec[i:] (shifted) with a
        # = Re(ifft(fft(peak_vec) * conj(fft(a)))[i])... 
        
        # I'll just pad and use direct convolution
        N = next_pow2(4 * n)
        Pc = np.fft.rfft(peak_vec, N)
        Ac = np.fft.rfft(a, N)
        # cross-corr: ifft(conj(Pc) * Ac) gives sum_k peak_vec[k-i] * a[k]... 
        # Let me verify: ifft(conj(fft(p)) * fft(a))[n] = sum_k p[k-n] * a[k]
        # We want sum_k p[k] * a[k-i] = sum_j p[j+i] * a[j] (sub j=k-i)
        # = ifft(conj(fft(p)) * fft(a))[i]... let me check
        # ifft(conj(P) * A)[n] = sum_k p[k-n] * a[k]
        # We want sum_k p[k] * a[k-i]. Let m = k-i: sum_{m+i} p[m+i] * a[m]
        # = sum_m p[m+i] * a[m]
        # vs ifft: sum_k p[k-n] * a[k]. Set k = m: sum_m p[m-n] * a[m].
        # So ifft(conj(P)*A)[n] = sum_m p[m-n] * a[m]
        # We want sum_m p[m+i] * a[m] = ifft(conj(P)*A)[-i]... but negative index wraps around in periodic FFT.
        
        # This is getting too complicated. Let me just use the direct approach for any reasonable size.
        for k in peaks:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                idx = k - np.arange(i_start, i_end + 1)
                pressure[i_start:i_end+1] += 2.0 * a[idx]
    
    pressure = np.where(np.isfinite(pressure) & (pressure > 0), pressure, 0.0)
    return pressure

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def normalize_seq(a):
    a = np.asarray(a, dtype=np.float64).copy()
