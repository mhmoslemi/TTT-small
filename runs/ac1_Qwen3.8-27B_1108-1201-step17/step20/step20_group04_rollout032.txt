def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    z = np.asarray(z, dtype=np.float64)
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_sharp_grad(y, top_idx):
    """Compute gradient of max autocorrelation using sharp top-k peak selection.
    grad[j] = sum_{k in top_idx} y[k-j] (with y[x]=0 outside [0,n-1])
    Uses FFT for O(n log n) computation.
    """
    n = len(y)
    # Create sparse peak indicator vector of length 2n-1
    p = np.zeros(2 * n - 1)
    for k in top_idx:
        p[int(k)] = 1.0
    # grad = conv(p, y_flipped)[n-1 : 2n-1]
    y_flipped = y[::-1]
    fft_size = 1 << ((2 * n - 1) + n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    YF = np.fft.rfft(y_flipped, fft_size)
    conv = np.fft.irfft(P * YF, fft_size)
    return conv[n - 1:2 * n - 1]


def compute_weighted_grad(y, top_idx, weights):
    """Compute weighted gradient for multiple peaks."""
    n = len(y)
    p = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):

    y_flipped = y[::-1]
    fft_size = 1 << ((2 * n - 1) + n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    YF = np.fft.rfft(y_flipped, fft_size)
    conv = np.fft.irfft(P * YF, fft_size)
    return conv[n - 1:2 * n - 1]

def peak_shaving(a0, time_limit, verbose=True, seed=77):
    """Directed mass redistribution to shave top convolution peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 5000

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Find top peaks
        n_top = min(10, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]

        improved = False
        found_move = False

        for k in top_idx[:8]:
            if found_move:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            prod_max = products.max()
            if prod_max < 1e-20:
                continue

            # Top 5 contributing pairs
            top_pairs = np.argsort(products)[::-1][:5]
            for pi in top_pairs:
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n:
                    continue

                # Compute pressure map (which positions are "hot")
                pressure = np.zeros(n)
                for k2 in top_idx[:6]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        pressure[idx2] += a[k2i - idx2]
                        pressure[k2i - idx2] += a[idx2]

                # Safe positions: low pressure
                safe_mask = pressure < np.percentile(pressure, 25)
                safe_positions = np.where(safe_mask)[0]
                # Exclude source positions
                safe_positions = safe_positions[(safe_positions != src_i) & (safe_positions != partner)]
                if len(safe_positions) < 10:
                    safe_positions = np.argsort(pressure)[:100]
                    safe_positions = safe_positions[(safe_positions != src_i) & (safe_positions != partner)]

                if len(safe_positions) == 0:
                    continue

                # Try moving mass from the larger contributor
                if a[src_i] >= a[partner]:
                    reduce_i = src_i
                else:
                    reduce_i = partner
                if a[reduce_i] < 1e-14:
                    continue

                # Try different fractions and destinations
                for frac in [0.003, 0.005, 0.01, 0.02, 0.03]:
                    amt = a[reduce_i] * frac
                    if amt < 1e-15:
                        continue
                    # Try top 3 safest positions
                    n_try = min(3, len(safe_positions))
                    dsts = safe_positions[np.argsort(pressure[safe_positions])[:n_try]]
                    for dst in dsts:
                        dst_int = int(dst)
                        a_trial = a.copy()
                        a_trial[reduce_i] -= amt
                        a_trial[dst_int] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            found_move = True
                            break
                    if found_move:
                        break
                if found_move:
                    break

        if not improved:
            # Random exploration when stuck
            if no_improve % 200 == 0:
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                break

        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  PeakShave ({elapsed:.1f}s): J={J:.6f} stall={no_improve}")

    return best_a, best_m


def pairwise_refine(a0, time_limit, verbose=True, seed=123):

                        if s_t < 1e-15:

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    """Try updating the best sequence if the candidate is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a_norm)}, t={elapsed:.1f}s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search function combining sharp gradient optimization, peak shaving,
    and multi-length exploration."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 7000
        x = np.linspace(0, 1, n_base)
        a_init = (1.0 - x) ** 1.2 + 0.05
        best_seq = norm_sum1(a_init)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.8f}")

    # Phase 1: Sharp Nesterov (high focus on top peaks)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Sharp Nesterov (n={n0}) ===")
    t1 = min(remaining * 0.20, 350.0)
    if t1 > 50:
        a_out, _ = nesterov_sharp(norm_sum1(best_seq), t1,
                                  top_k_start=10, top_k_end=2,
                                  eta_base=0.5, verbose=True, seed=1)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sharp1", t_global)

    # Phase 2: Peak shaving (directed mass redistribution)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Peak Shaving (n={n_best}) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 40:
        a_out, _ = peak_shaving(norm_sum1(best_seq), t2, verbose=True, seed=11)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "shave1", t_global)

    # Phase 3: Weighted Nesterov (harmonic peak weights)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Weighted Nesterov (n={n_best}) ===")
    t3 = min(remaining * 0.18, 300.0)
    if t3 > 50:
        a_out, _ = weighted_nesterov(norm_sum1(best_seq), t3, verbose=True, seed=2)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "wtd1", t_global)

    # Phase 4: Peak shaving again
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak Shaving 2 (n={n_best}) ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 40:
        a_out, _ = peak_shaving(norm_sum1(best_seq), t4, verbose=True, seed=22)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "shave2", t_global)

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    lengths_to_try = [n_best + 800, n_best + 2000, max(4000, n_best - 1500)]
    lengths_to_try = [l for l in lengths_to_try if 3000 <= l <= 12000]
    # Remove duplicates
    seen = set()
    unique_lengths = []
    for l in lengths_to_try:
        if l not in seen and l != n_best:
            seen.add(l)
            unique_lengths.append(l)

    for new_n in unique_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print(f"  Exploring n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.35, 180.0)
        if t5a > 40:
            a_out, _ = nesterov_sharp(a_long, t5a,
                                      top_k_start=8, top_k_end=2,
                                      eta_base=0.5, verbose=True, seed=3)
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"ml_{new_n}", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80:
            t5b = min(remaining * 0.30, 100.0)
            if t5b > 30:
                a_try = interpolate_sequence(best_seq, new_n)
                a_out, _ = peak_shaving(a_try, t5b, verbose=False, seed=33)
                best_seq, best_score = try_update(best_seq, best_score, a_out, f"mlps_{new_n}", t_global)

    # Phase 6: Pairwise refinement (final polish on best length)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Pairwise Refine (n={n_best}) ===")
    t6 = min(remaining * 0.20, 250.0)
    if t6 > 30:
        a_out, _ = pairwise_refine(norm_sum1(best_seq), t6, verbose=True, seed=55)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pair1", t_global)

    # Phase 7: Final sharp Nesterov with very high focus
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Sharp Nesterov (n={n_best}) ===")
    if remaining > 50:
        t7 = min(remaining * 0.25, 180.0)
        if t7 > 30:
            a_out, _ = nesterov_sharp(norm_sum1(best_seq), t7,
                                      top_k_start=5, top_k_end=1,
                                      eta_base=0.3, verbose=True, seed=4)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "sharp_final", t_global)

    # Phase 8: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Peak Shaving (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.35, 150.0)
        if t8 > 20:
            a_out, _ = peak_shaving(norm_sum1(best_seq), t8, verbose=True, seed=66)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "shave_final", t_global)

    # Phase 9: Very fine pairwise refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Fine Pairwise (n={n_best}) ===")
    if remaining > 25:
        t9 = min(remaining * 0.50, 120.0)
        if t9 > 15:
            a_out, _ = pairwise_refine(norm_sum1(best_seq), t9, verbose=True, seed=77)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "pair_final", t_global)

    # Phase 10: One more round if time allows
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_best = len(best_seq)
        print(f"\n=== Phase 10: Final Round (n={n_best}) ===")
        t10a = min(remaining * 0.35, 100.0)
        if t10a > 25:
            a_out, _ = peak_shaving(norm_sum1(best_seq), t10a, verbose=False, seed=44)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ps_ult", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t10b = min(remaining * 0.50, 80.0)
            if t10b > 20:
                a_out, _ = nesterov_sharp(norm_sum1(best_seq), t10b,
                                          top_k_start=3, top_k_end=1,
                                          eta_base=0.2, verbose=True, seed=5)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_ult", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t10c = min(remaining * 0.60, 60.0)
            if t10c > 10:
                a_out, _ = pairwise_refine(norm_sum1(best_seq), t10c, verbose=False, seed=88)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "pair_ult", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 7000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1((1.0 - x) ** 1.2 + 0.05)
        best_score = evaluate_safe(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)
    # Ensure sum > 0.01
    if best_seq.sum() < 0.01:
        best_seq = best_seq / best_seq.sum() * 1.0 if best_seq.sum() > 0 else np.ones(1000) / 1000

    return [float(x) for x in best_seq]
