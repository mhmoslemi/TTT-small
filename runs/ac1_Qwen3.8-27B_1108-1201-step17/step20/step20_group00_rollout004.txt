    fi = np.maximum(fi, 0)
    s = fi.sum()
    if s < 1e-15:

        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yl = np.fft.irfft(Yf * Gf, fft_size)[:size]
        mc = conv_yl.max()
        
        viol = np.where(conv_yl > mc + 1e-10)[0]
        if len(viol) > 0:
            top_v = viol[np.argsort(conv_yl[viol])[::-1][:50]]
            for k in top_v:
                active.add(int(k))
        
        # Line search
        best_alpha = 0.5
        best_trial_m = best_m
        for alpha in [0.1, 0.25, 0.5, 0.75, 1.0]:
            trial = (1 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            tm = fast_autocorr(trial).max()
            if tm < best_trial_m:
                best_trial_m = tm
                best_alpha = alpha
        
        y = (1 - best_alpha) * y + best_alpha * g_dir
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        mc2 = fast_autocorr(y).max()
        if mc2 < best_m - 1e-15:
            best_m = mc2
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and t % 30 == 0:
            print(f"  LPFW-adv t={t} ({time.monotonic()-t0:.1f}s) J={2*n*best_m:.6f} |S|={len(active)}")
    
    print(f"  LPFW-adv done ({time.monotonic()-t0:.1f}s) J={2*n*best_m:.6f}")
    return best_a, 2.0 * n * best_m

def subgrad_polish(a0, time_limit, verbose=True):
    """Subgradient descent with Nesterov averaging on the simplex."""
    n = len(a0)
    x = normalize(a0)
    x_avg = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    
    Y = x.copy()  # Nesterov point
    x_prev = x.copy()
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(Y)
        mc = ac.max()
        
        # Find top peaks for subgradient
        top_k = min(20, len(ac))
        idx = np.argsort(ac)[::-1][:top_k]
        
        # Compute subgradient: weighted sum of shifted versions of Y
        # Weight: softmax of (ac[idx] - mc) / temperature
        temp = 0.01 * mc
        w = np.exp((ac[idx] - mc) / temp)
        w /= w.sum()
        
        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            idx_arr = np.arange(i_s, i_e)
            grad[idx_arr] += 2.0 * wi * Y[ki - idx_arr]
        
        # Step size
        eta = 1.0 / np.sqrt(t + 1)
        
        # Multiplicative update (stay on simplex naturally)
        x_new = x * np.exp(-eta * grad)
        x_new = np.maximum(x_new, 0)
        sn = x_new.sum()
        if sn < 1e-15:
            x_new = np.ones(n) / n
        else:
            x_new = x_new / sn
        
        # Nesterov averaging
        x_avg = (t / (t + 1)) * x_avg + (1 / (t + 1)) * x_new
        
        mc_avg = fast_autocorr(x_avg).max()
        if mc_avg < best_m - 1e-15:
            best_m = mc_avg
            best_a = x_avg.copy()
        
        x_prev = x
        x = x_new
        Y = x + (t / (t + 2.0)) * (x - x_prev)
        Y = np.maximum(Y, 1e-15)
        Y = Y / Y.sum()
        
        if verbose and t % 500 == 0:
            print(f"  Subgrad t={t} ({time.monotonic()-t0:.1f}s) J={2*n*best_m:.6f}")
    
    return best_a, 2.0 * n * best_m

def coord_polish(a0, time_limit, seed=42, verbose=True):
    """Coordinate descent: move small amounts of mass to reduce peaks."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        mc = ac.max()
        # Top peaks
        n_top = min(15, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            # Top contributing sources
            top_srcs = idx[np.argsort(contrib)[::-1][:5]]
            for src in top_srcs:
                src = int(src)
                if a[src] < 1e-12:
                    continue
                
                # Score destinations (where adding mass would least increase peaks)
                scores = np.zeros(n)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                scores[src] = np.inf
                
                candidates = np.argsort(scores)[:20]
                for dst in candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.005, 0.01, 0.02, 0.05]:
                        amt = a[src] * frac
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-16:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.001, 0.005)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                Mn = fast_autocorr(a).max()
                if Mn < best_m:
                    best_m = Mn
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 200 == 0:
            print(f"  Coord ({time.monotonic()-t0:.1f}s) J={2*n*best_m:.6f} stall={stall}")
    
    return best_a, 2.0 * n * best_m

def diverse_init(n, parent):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []
    
    # Sin^2
    candidates.append(normalize(np.sin(np.pi * x) ** 2))
    # Cos windows
    for f in [1, 2, 3, 4, 6, 8]:
        candidates.append(normalize(np.maximum(1 + np.cos(2*np.pi*f*x), 0.01)))
    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for sg in [0.1, 0.15, 0.25]:
            candidates.append(normalize(np.exp(-((x-c)**2)/(2*sg*sg))))
    # Power law
    for p in [0.5, 1, 2, 3]:
        candidates.append(normalize((i_arr + 1) ** p * (n - i_arr) ** p + 1e-10))
    # Tukey
    for alpha in [0.1, 0.2, 0.3, 0.5]:
        cands = np.ones(n)
        m = int(alpha * n)
        if m > 0:
            taper = 0.5 * (1 + np.cos(np.pi * np.arange(m) / m))
            cands[:m] = taper
            cands[-m:] = taper[::-1]
        candidates.append(normalize(cands))
    
    # Blend with parent
    if parent is not None:
        p = normalize(parent)
        for w in [0.3, 0.5, 0.7]:
            for c in candidates[:5]:
                candidates.append(normalize(w * p + (1-w) * c))
    
    best_a = candidates[0]
    best_v = obj_val(best_a)
    for c in candidates:
        v = obj_val(c)
        if v < best_v:
            best_v = v
            best_a = c.copy()
    return best_a, best_v
