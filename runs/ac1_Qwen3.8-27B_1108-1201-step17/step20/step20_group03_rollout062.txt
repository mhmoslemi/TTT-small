def compute_gradient(a, beta, top_k=None):

        c_use[idx] = c[idx]

**Spectral Shaping:**
The idea is to modify the power spectrum `|A(f)|^2` to be more uniform (flat), which would make the autocorrelation more uniform.

1. Compute `A = FFT(a)`.
2. Compute `|A|^2` (power spectrum).
3. Modify the amplitude: `|A_new| = |A| * (1 - alpha * (|A|^2 - mean(|A|^2)) / max(|A|^2))` or some other flattening function.
4. Keep the phase: `A_new = |A_new| * exp(i * phase(A))`.
5. Inverse FFT: `a_new = IFFT(A_new)`.
6. Project onto non-negative: `a_new = max(a_new, 0)`.
7. Normalize: `a_new = a_new / sum(a_new)`.

The issue is that step 6 (projecting onto non-negative) can significantly change the spectrum, undoing the flattening. This is the fundamental difficulty of the problem: the non-negativity constraint and the flat-spectrum constraint are in tension.

A better approach might be to use "alternating projections" (POCS - Projection Onto Convex Sets):
- Set 1: Non-negative signals with sum = 1 (simplex)
- Set 2: Signals with flat power spectrum (or bounded autocorrelation peak)

Alternating projections between these two sets can converge to a signal that approximately satisfies both constraints.

But Set 2 is not convex (the set of signals with bounded autocorrelation peak is the intersection of half-spaces in the autocorrelation domain, which is convex in the autocorrelation domain but not in the signal domain). So POCS might not converge.

def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project z onto the probability simplex (non-negative, sum=1)."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_grad_fft(a, beta, top_k=None):
    """
    Compute gradient of f_beta(a) = (1/beta)*log(sum_k exp(beta*c[k]))
    where c = full autocorrelation of a.
    grad[i] = sum_k p[k] * dc[k]/da[i] = 2 * sum_j a[j] * p[i+j]
    Returns (grad, c, p).
    """
    n = len(a)
    c = fast_autocorr(a)

    if top_k is not None and 0 < top_k < len(c):
        c_sorted = np.sort(c)[::-1]
        c_thresh = c_sorted[top_k - 1]
        c_use = np.where(c >= c_thresh, c, c_thresh)
    else:
        c_use = c

    c_max = c_use.max()
    shifted = beta * (c_use - c_max)
    exp_vals = np.exp(np.clip(shifted, -500, 500))
    p_sum = exp_vals.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(c_use)
        max_idx = (c_use == c_max)
        p[max_idx] = 1.0 / max(1, max_idx.sum())
    else:
        p = exp_vals / p_sum

    # grad[i] = 2 * sum_j a[j] * p[i+j]
    # = 2 * convolve(a, p_rev)[2n-2-i]
    p_rev = p[::-1]
    fft_size = 1 << (n + len(p) - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p_rev, fft_size)
    conv = np.fft.irfft(A * P, fft_size)
    indices = (2 * n - 2) - np.arange(n)
    grad = 2.0 * conv[indices]

    return grad, c, p


def fista_smoothed_max(a0, time_limit, beta_start=3.0, beta_end=300.0,

def spectral_flatten_iterations(a0, time_limit, n_iters=500, verbose=True):
    """
    Iterative spectral flattening: modify power spectrum to reduce autocorrelation peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    fft_size = 1 << (n - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        power = mag ** 2
        p_max = power.max()
        if p_max < 1e-30:
            break

        # Flatten the power spectrum
        alpha = 0.2 + 0.3 * (it / max(1, n_iters))
        mag_new = mag * np.sqrt(1.0 - alpha * (power / p_max))
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        m_new = float(fast_autocorr(a_new).max())
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        a = a_new

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral it={it} ({elapsed:.1f}s): J={J:.6f}")

def frank_wolfe_pairwise(a0, time_limit, n_pairs=200, verbose=True):
    """
    Frank-Wolfe / pairwise algorithm: at each step, transfer mass from
    a "bad" coordinate (high gradient) to a "good" coordinate (low gradient).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    for it in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        grad, c, p = compute_grad_fft(a, 50.0, top_k=20)
        g_max_idx = int(np.argmax(grad))
        g_min_idx = int(np.argmin(grad))

        if g_max_idx == g_min_idx:
            break
        if a[g_max_idx] < 1e-12:
            break

        # Try multiple transfer fractions
        best_frac = 0.0
        best_trial_m = best_m
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.4, 0.7]:
            amt = a[g_max_idx] * frac
            if amt > a[g_max_idx]:
                continue
            a_trial = a.copy()
            a_trial[g_max_idx] -= amt
            a_trial[g_min_idx] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                continue
            a_trial = a_trial / s_t
            m_trial = float(fast_autocorr(a_trial).max())
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_frac = frac

        if best_frac > 0:
            amt = a[g_max_idx] * best_frac
            a[g_max_idx] -= amt
            a[g_min_idx] += amt
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 1e-10:
                a = a / s_a

        if best_trial_m < best_m - 1e-14:
            best_m = best_trial_m
            best_a = a.copy()
        else:
            # Random perturbation
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 != i2 and a[i1] > 1e-10:
                frac = float(rng.uniform(0.005, 0.1))

                if s_a > 1e-10:

                    m_chk = float(fast_autocorr(a).max())

                        best_a = a.copy()

        if verbose and it % 2000 == 0:

            J = 2.0 * n * best_m
            print(f"    FW it={it} ({elapsed:.1f}s): J={J:.6f}")

                a[dst] += amt
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a > 1e-10:
                    a = a / s_a
                    best_m = best_m_trial
                    best_a = a.copy()
                    improved = True
                    c = fast_autocorr(a)
                    top_peaks = np.argsort(c)[::-1][:20]

        if not improved:
            stall += 1
            # Random kicks
            for _ in range(5):
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                if i1 != i2 and a[i1] > 1e-10:
                    frac = float(rng.uniform(0.003, 0.05))
                    amt = a[i1] * frac
                    a[i1] -= amt
                    a[i2] += amt
                    a = np.maximum(a, 0.0)
                    s_a = a.sum()
                    if s_a > 1e-10:
                        a = a / s_a
                        m_chk = float(fast_autocorr(a).max())
                        if m_chk < best_m:
                            best_m = m_chk
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    2Ex ({elapsed:.1f}s): J={J:.6f} stall={stall}")

    return best_a, best_m


def cyclic_shift_best(a0):
    """Find the best cyclic shift."""
    n = len(a0)
    best_m = float(fast_autocorr(a0).max())
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()

    for s in range(1, min(n, 200)):
        a_s = np.roll(a0, -s)
        A = np.fft.rfft(a_s, fft_size)
        b = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
        m = float(b.max())
        if m < best_m:
            best_m = m
            best_shift = s

    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.1f}s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Time budget: {time_budget:.1f}s")

    best_seq = None
    best_score = float('inf')

    # Initialize from parent or default
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        x = np.linspace(0, 1, n_base, endpoint=False)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Phase 1: Main FISTA optimization
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: FISTA smoothed max (n={n0}) ===")
    t1 = min(remaining * 0.40, 650.0)
    if t1 > 60:
        top_k_sched = [5, 10, 20, 50, 100, 200, None]
        a_out, _ = fista_smoothed_max(norm_sum1(best_seq), t1,
                                      beta_start=2.0, beta_end=200.0,
                                      beta_growth=1.004, top_k_schedule=top_k_sched,
                                      restart_every=2500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista1", t_global)

    # Phase 2: Spectral flattening
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Spectral flattening (n={len(best_seq)}) ===")
    if remaining > 60:
        t2 = min(remaining * 0.10, 120.0)
        a_out, _ = spectral_flatten_iterations(norm_sum1(best_seq), t2,
                                                n_iters=300, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spectral1", t_global)

    # Phase 3: Frank-Wolfe / pairwise
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Frank-Wolfe pairwise (n={len(best_seq)}) ===")
    if remaining > 50:
        t3 = min(remaining * 0.12, 180.0)
        a_out, _ = frank_wolfe_pairwise(norm_sum1(best_seq), t3,
                                        n_pairs=200, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fw1", t_global)

    # Phase 4: 2-exchange local search
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: 2-exchange (n={len(best_seq)}) ===")
    if remaining > 40:
        t4 = min(remaining * 0.12, 150.0)
        a_out, _ = two_exchange_refined(norm_sum1(best_seq), t4, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "2ex1", t_global)

    # Phase 5: Cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 5: Cyclic shift ===")
        a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic1", t_global)

    # Phase 6: Second FISTA with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: FISTA v2 (n={n_best}) ===")
    if remaining > 50:
        t6 = min(remaining * 0.15, 200.0)

                                      beta_start=10.0, beta_end=500.0,
                                      beta_growth=1.005, top_k_schedule=top_k_sched2,
                                      restart_every=2000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista2", t_global)

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    candidate_lengths = [max(2000, n_best - 1000), n_best + 1000, n_best + 2000, n_best + 4000]
    candidate_lengths = [l for l in candidate_lengths if 2000 <= l <= 12000]

    for new_n in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        if abs(new_n - n_best) < 500:
            continue
        print(f"\n=== Phase 7: Multi-length n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.25, 150.0)

                                          beta_start=5.0, beta_end=300.0,

                                          restart_every=2000, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, f"fista_n{new_n}", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
            best_seq, best_score = try_update(best_seq, best_score, a_shifted, f"cyclic_n{new_n}", t_global)

    # Phase 8: Final 2-exchange polish

        a_out, _ = two_exchange_refined(norm_sum1(best_seq), t8, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "2ex_final", t_global)

    # Phase 9: Final spectral + FISTA micro-polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 9: Final polish ===")
        t9 = min(remaining * 0.30, 100.0)
        a_out, _ = spectral_flatten_iterations(norm_sum1(best_seq), t9,
                                                n_iters=200, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spectral_final", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t9b = min(remaining * 0.40, 80.0)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:

        x = np.linspace(0, 1, n_fb, endpoint=False)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
