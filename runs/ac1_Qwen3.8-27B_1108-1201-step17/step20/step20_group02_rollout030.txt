The problem is: `min f(a) = max_k c_k(a)` s.t. `a >= 0, sum a = 1`.

"""

import numpy as np
import time
import cvxpy as cp

# Module-level solver cache
_SOLVER = None

def get_solver():
    global _SOLVER
    if _SOLVER is not None:
        return _SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _SOLVER = s
                print("  Using solver: " + str(s))
                return s
        except Exception:
            continue
    _SOLVER = cp.ECOS
    print("  Fallback solver: ECOS")
    return _SOLVER

def fft_conv(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_fast(a):
    """Fast evaluation: 2*n*max(conv(a,a))/(sum(a))^2"""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fft_conv(a)
    return float(2.0 * len(a) * c.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def interp_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / s

def build_lp_matrix(y, active_set, n):
    """Build the constraint matrix for the LP.
    Row k (for active constraint at position k) has entries:
    coeffs[k, j] = y[k-j] (with y[m]=0 for m<0 or m>=n)
    """
    K = len(active_set)
    coeffs = np.zeros((K, n))
    for ii in range(K):
        k = int(active_set[ii])
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[ii, idx] = y[k - idx]
    return coeffs

def lp_frank_wolfe(a0, time_limit, solver, initial_K=200, max_K=2000, 
                    add_per_iter=30, verbose=True):
    """LP-Frank-Wolfe method with adaptive active set."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    ac = fft_conv(y)
    best_m = ac.max()
    t0 = time.monotonic()
    
    # Initialize active set with top-K convolution values
    active_set = sorted(np.argsort(ac)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 500
    iter_count = 0
    restart_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Check for stagnation
        if no_improve > max_no_improve:
            restart_count += 1
            if restart_count > 5:
                if verbose:
                    print("    LP-FW: too many restarts, stopping")
                break
            # Perturb and restart active set
            rng = np.random.default_rng(restart_count * 12345)
            ac_curr = fft_conv(y)
            n_keep = max(50, len(active_set) // 3)
            # Keep some constraints, add new random ones
            keep = list(rng.choice(len(active_set), size=n_keep, replace=False))
            new_set = set(active_set[i] for i in keep)
            # Add top violations
            top_v = np.argsort(ac_curr)[::-1][:n_keep]
            for k in top_v:
                new_set.add(int(k))
            # Add some random ones
            rand_ks = rng.integers(0, 2 * n - 1, size=n_keep // 2)
            for k in rand_ks:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]
            no_improve = 0
            continue
        
        iter_count += 1
        ac = fft_conv(y)
        M_curr = float(ac.max())
        
        if M_curr < best_m - 1e-12:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Trim active set
        if len(active_set) > max_K:
            margins = np.array([M_curr - float(ac[k]) for k in active_set])
            # Keep tightest constraints (smallest margin) + some random
            n_keep = int(max_K * 0.8)
            tight_idx = np.argsort(margins)[:n_keep]
            active_set_base = [active_set[i] for i in tight_idx]
            # Always keep current argmax
            k_max = int(np.argmax(ac))
            if k_max not in active_set_base:
                active_set_base.append(k_max)
            active_set = sorted(set(active_set_base))
        
        K = len(active_set)
        if K == 0:
            break
        
        # Build and solve LP
        coeffs = build_lp_matrix(y, active_set, n)
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Reduce active set and retry
            active_set = active_set[:max(50, K // 2)]
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-12:
            continue
        g_dir = g_val / g_sum
        
        # Add violated constraints to active set
        conv_ys = np.convolve(y, g_dir, mode='full')
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]
        
        # Line search on the true objective
        best_alpha = 1.0 / (iter_count + 2.0)
        best_trial_M = M_curr
        
        # Try a range of alphas
        alphas = [1.0, 0.9, 0.7, 0.5, 0.3, 0.2, 0.1, 
                  max(0.05, 2.0/(iter_count+2)), max(0.02, 1.0/(iter_count+1))]
        
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-12:
                continue
            trial = trial / s_t
            M_trial = fft_conv(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        if iter_count <= 3:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-12:
                y = best_a.copy()
            else:
                y = y / s_y
        
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter={} ({:.1f}s): J={:.6f}, |S|={}".format(
                iter_count, elapsed, 2.0 * n * best_m, len(active_set)))
    
    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done ({:.1f}s): J={:.6f}, iters={}".format(
            elapsed, 2.0 * n * best_m, iter_count))
    return best_a, best_m

def mirror_descent_polish(a0, time_limit, verbose=True):
    """Mirror descent (entropic) method on the true non-convex objective."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    ac = fft_conv(y)
    best_m = float(ac.max())
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 2000
    lr = 0.05
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break
        
        # Compute subgradient at current point
        ac = fft_conv(y)
        k_star = int(np.argmax(ac))
        M_curr = float(ac[k_star])
        
        # Subgradient of c_{k_star} at y: grad_i = 2 * y_{k_star - i}
        grad = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] = 2.0 * y[k_star - idx]
        
        # Mirror descent update: a_new = exp(-lr * grad) / sum(exp(-lr * grad))
        # With normalization constraint
        scaled_lr = lr / (norm_sum_max(y) + 1e-10)
        log_a = -scaled_lr * grad
        log_a -= log_a.max()  # for numerical stability
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s < 1e-12:
            a_new = y.copy()
        else:
            a_new = a_new / s
        
        # Occasionally mix with random direction
        if t % 500 == 0:
            noise = rng.standard_normal(n) * 0.01
            a_new = a_new + noise
            a_new = np.maximum(a_new, 1e-10)
            a_new = a_new / a_new.sum()
        
        y = a_new
        M_new = float(fft_conv(y).max())
        
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = y.copy()
            stall = 0
            if verbose and (t % 5000 == 0 or M_new < best_m + 1e-6):
                print("    MD new best at t={}: J={:.6f}".format(t, 2.0 * n * best_m))
        else:
            stall += 1
        
        # Adaptive learning rate
        if stall % 500 == 0:
            lr *= 0.9
            if lr < 0.001:
                lr = 0.05  # reset
    
    if verbose:
        elapsed = time.monotonic() - t0
        print("    MD done ({:.1f}s): J={:.6f}".format(elapsed, 2.0 * n * best_m))
    return best_a, best_m

def norm_sum_max(a):
    return float(np.max(a))

def random_init(n, seed):
    """Generate a random initialization."""
    rng = np.random.default_rng(seed)
    # Mix of different random structures
    choice = seed % 5
    if choice == 0:
        a = rng.exponential(1.0, n)
    elif choice == 1:
        # Two bumps
        centers = rng.uniform(0.2, 0.8, 2)
        sigmas = rng.uniform(0.05, 0.15, 2)
        x = np.linspace(0, 1, n)
        a = sum(np.exp(-((x - c)**2) / (2*s*s)) for c, s in zip(centers, sigmas))
    elif choice == 2:
        # Sine-based
        freq = rng.uniform(1, 5)
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(freq * np.pi * x)) + 0.1
    elif choice == 3:
        # Linear ramp
        a = np.linspace(0.1, 1.0, n)
    else:
        # Uniform + noise
        a = np.ones(n) + rng.normal(0, 0.3, n)
        a = np.maximum(a, 0.01)
    return normalize(a)

def subgradient_method(a0, time_limit, verbose=True):
    """Simple subgradient method with Polyak step size."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = float(fft_conv(y).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    
    lower_bound = 1.0 + 1.0 / (2.0 * n)  # theoretical lower bound (rough)
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fft_conv(y)
        k_star = int(np.argmax(ac))
        M_curr = float(ac[k_star])
        M_best = float(fft_conv(best_a).max())
        
        # Polyak step size
        gap = M_curr - lower_bound
        if gap < 1e-10:
            break
        step = gap / (2.0 * M_curr + 1e-10)
        step = min(step, 0.5)
        
        # Subgradient
        grad = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] = 2.0 * y[k_star - idx]
        
        # Projected gradient step
        y_new = y - step * grad
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-12:
            y_new = y.copy()
        else:
            y_new = y_new / s
        
        # Averaging
        if t % 100 == 0:
            y_avg = normalize(y + rng.standard_normal(n) * 0.001)
            y = 0.9 * y_new + 0.1 * y_avg
        else:
            y = y_new
        
        M_new = float(fft_conv(y).max())
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = y.copy()
            if verbose and t % 10000 == 0:
                print("    SG new best at t={}: J={:.6f}".format(t, 2.0 * n * best_m))
    
    if verbose:
        elapsed = time.monotonic() - t0
        print("    SG done ({:.1f}s): J={:.6f}".format(elapsed, 2.0 * n * best_m))
    return best_a, best_m

def interp_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()

    return f_new / s

def build_lp_matrix(y, active_set, n):
    K = len(active_set)

        if i_start < i_end:

            coeffs[ii, idx] = y[k - idx]
    return coeffs
