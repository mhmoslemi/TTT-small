def next_pow2(x):
    x = int(max(1, x))
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def make_symmetric(a):
    """Make a sequence into a palindrome."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 1:
        return a.copy()
    half = n // 2
    result = np.empty(n, dtype=np.float64)
    result[:half] = a[:half]

        result[half:] = a[:half][::-1]
    else:
        result[half] = a[half] if half < n else 0.0
        result[half+1:] = a[:half][::-1]
    return result
