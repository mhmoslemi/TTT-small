"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FISTA with full softmax smoothing and adaptive step size
2. Mirror descent (exponential weights) on the simplex
3. Systematic peak-shaving local search
4. Multi-length exploration with spectral upsampling
5. Cyclic shift optimization
6. Multi-start with diverse initializations
"""

import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def fista_smooth(a0, time_limit, beta_start=20.0, beta_end=2000.0,
                 beta_growth=1.004, restart_every=3000, verbose=True):
    """FISTA with full softmax smoothing on simplex constraint sum(a)=1, a>=0."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_obj = fast_autocorr(x).max()
    t0 = time.monotonic()
    t = 0
    t_nest = 1.0
    beta = beta_start
    stall = 0
    rng = np.random.default_rng(42)
    bb_step = 1.0 / np.sqrt(n)
    s_prev = None
    y_prev = None
    fft_size = 1 << (2 * n - 1).bit_length()

    while t < 50000000:
        if time.monotonic() - t0 > time_limit:
            break
        t += 1

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute smoothed gradient
        b = fast_autocorr(y)
        b_shift = beta * b
        b_shift -= b_shift.max()  # numerical stability
        b_shift = np.clip(b_shift, -500, 500)
        exp_b = np.exp(b_shift)
        p = exp_b / exp_b.sum()

        # Gradient: g[j] = sum_k p[k] * (y[k-j] + y[j-k]) for valid indices
        # = conv(p, y)[j] + cross_corr_term
        # Using FFT:
        Yf = np.fft.rfft(y, fft_size)
        Pf = np.fft.rfft(p, fft_size)
        # conv(p, y)[j] = sum_k p[k]*y[j-k]: this is ifft(Pf*Yf)
        conv_py = np.fft.irfft(Pf * Yf, fft_size)[:2 * n - 1]
        # sum_k p[k]*y[k-j]: 
        # Let y_rev = y[::-1], then y[k-j] = y_rev[n-1-k+j]
        # sum_k p[k]*y[k-j] = sum_k p[k]*y_rev[n-1+j-k] = conv(p, y_rev)[n-1+j]
        y_rev = y[::-1]
        Yrv = np.fft.rfft(y_rev, fft_size)
        conv_pyr = np.fft.irfft(Pf * Yrv, fft_size)[:2 * n - 1]
        
        grad = conv_py[:n] + conv_pyr[n-1:n+n-1]
        
        # Barzilai-Borwein step size update
        if s_prev is not None and y_prev is not None:
            sy_diff = y - y_prev
            g_diff = grad - s_prev
            num = sy_diff.dot(g_diff)
            den = g_diff.dot(g_diff)
            if den > 1e-20:
                bb_cand = num / den
                if 1e-8 < bb_cand < 10.0:
                    bb_step = max(1e-6, min(bb_cand, 2.0 / beta))
                else:
                    bb_step = max(bb_step * 0.85, 1e-6)
            s_prev = grad.copy()
            y_prev = y.copy()

        # Projected gradient step (simplex projection)
        eta = min(bb_step, 0.5 / beta + 1.0 / np.sqrt(n))
        z = y - eta * grad
        # Project to simplex {w: w>=0, sum(w)=1}
        w = np.sort(z)[::-1]
        cssv = np.cumsum(w)
        cond = w * np.arange(1, n + 1) > (cssv - 1)
        rho = np.nonzero(cond)[0]
        if len(rho) > 0:
            rho = rho[-1]
            theta_proj = (cssv[rho] - 1) / (rho + 1.0)
            x_new = np.maximum(z - theta_proj, 0.0)
        else:
            x_new = np.ones(n) / n

        obj_new = fast_autocorr(x_new).max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new

        if t % 100 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.01 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 10.0)
            s_prev = None
            y_prev = None
            bb_step = 1.0 / np.sqrt(n)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA t={t} ({elapsed:.1f}s): obj={best_obj:.8f} score={2*n*best_obj:.6f} beta={beta:.1f}")

    return best_a, best_obj


def mirror_descent_opt(a0, time_limit, verbose=True):
    """Mirror descent (exponential weights) on simplex with softmax-smoothed gradient."""
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_obj = fast_autocorr(x).max()
    t0 = time.monotonic()
    t = 0
    fft_size = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        t += 1
        beta = min(20.0 * (1.003 ** t), 3000.0)
        eta = 1.0 / (np.sqrt(2.0 * t) * max(1.0, beta))

        b = fast_autocorr(x)
        b_shift = beta * b
        b_shift -= b_shift.max()
        b_shift = np.clip(b_shift, -500, 500)
        p = np.exp(b_shift)
        p = p / p.sum()

        # Gradient
        Xf = np.fft.rfft(x, fft_size)
        Pf = np.fft.rfft(p, fft_size)
        conv_px = np.fft.irfft(Pf * Xf, fft_size)[:2 * n - 1]
        x_rev = x[::-1]
        Xrv = np.fft.rfft(x_rev, fft_size)
        conv_pxr = np.fft.irfft(Pf * Xrv, fft_size)[:2 * n - 1]
        grad = conv_px[:n] + conv_pxr[n-1:n+n-1]

        # Mirror descent: x_new[j] ∝ x[j] * exp(-eta * grad[j])
        exponent = -eta * grad
        exponent = np.clip(exponent, -300, 300)
        x_new = x * np.exp(exponent)
        x_new = np.maximum(x_new, 1e-30)
        x_new = x_new / x_new.sum()

        obj_new = fast_autocorr(x_new).max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = x_new.copy()

        x = x_new

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={t} ({elapsed:.1f}s): obj={best_obj:.8f} score={2*n*best_obj:.6f} beta={beta:.1f}")

    return best_a, best_obj


def systematic_local_search(a0, time_limit, verbose=True):
    """Systematic peak-shaving: for each top peak, move mass from contributing pairs to valleys."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_obj = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    rng = np.random.default_rng(999)

    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        b = fast_autocorr(a)
        obj_current = b.max()

        if obj_current < best_obj - 1e-15:
            best_obj = obj_current
            best_a = a.copy()

        # Get top 20 peak lags
        top_k = np.argsort(b)[::-1][:20]

        improved = False
        for k_int in top_k:
            k_int = int(k_int)
            # Compute contributions: a[i]*a[k_int-i] for valid i
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue

            # Find the pair with maximum contribution
            max_pos = int(np.argmax(contrib))
            i_max = int(idx[max_pos])
            j_max = int(k_int - i_max)
            
            # Find a "valley" position (low b nearby)
            # Move mass from the larger of a[i_max], a[j_max] to a valley position
            if a[i_max] >= a[j_max]:
                src = i_max
                dst_candidate = j_max  # balance within the pair
            else:
                src = j_max
                dst_candidate = i_max

            # Also try moving to a globally low position
            low_positions = np.argsort(b)[n // 2:]  # lower half indices (away from peaks)
            # Pick a position that's not in the peak's support
            in_peak = np.zeros(n, dtype=bool)
            in_peak[i_start:i_end + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) > 0:
                # Pick from low-valley candidates
                low_cand = candidates[np.argsort([b[c] for c in candidates[:500] if c in set(candidates)])]
                # Just pick randomly from non-peak positions
                dst = int(rng.choice(candidates))
            else:
                dst = dst_candidate

            # Try multiple fractions
            best_frac = 0
            best_trial_obj = obj_current
            for frac in [0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if a[src] - amt < -1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                obj_trial = fast_autocorr(a_trial).max()
                if obj_trial < best_trial_obj:
                    best_trial_obj = obj_trial
                    best_frac = frac

            if best_frac > 0:
                amt = a[src] * best_frac
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                improved = True

        if not improved and iteration % 500 == 0:
            # Add small random perturbation to escape
            for _ in range(10):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    obj_new = fast_autocorr(a).max()
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a.copy()

        if verbose and iteration % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LocalSearch it={iteration} ({elapsed:.1f}s): obj={best_obj:.8f} score={2*n*best_obj:.6f}")

    return best_a, best_obj


def cyclic_shift_best(a0):
    """Find best cyclic shift of a0."""
    n = len(a0)
    best_obj = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        A = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_obj:
            best_obj = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_obj
    return a0.copy(), best_obj


def spectral_upsample(a0, new_n):
    """Upsample sequence using spectral method (better than linear interpolation)."""
    old_n = len(a0)
    if old_n == new_n:
        return a0.copy()
    a = norm_sum1(a0)
    # Zero-pad in frequency domain
    old_fft_size = 1 << (2 * old_n - 1).bit_length()
    new_fft_size = 1 << (2 * new_n - 1).bit_length()
    A_old = np.fft.rfft(a, old_fft_size)
    # Interpolate the spectrum
    n_freq_old = len(A_old)
    n_freq_new = new_fft_size // 2 + 1
    A_new = np.zeros(n_freq_new, dtype=np.complex128)
    # Map old frequencies to new
    for k in range(n_freq_old):
        # Old freq k corresponds to new freq k * new_fft_size // old_fft_size
        new_k = k * new_fft_size // old_fft_size
        if new_k < n_freq_new:
            A_new[new_k] = A_old[k] * (new_n / old_n)  # scale to maintain sum
    a_new = np.fft.irfft(A_new, new_fft_size)[:new_n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
