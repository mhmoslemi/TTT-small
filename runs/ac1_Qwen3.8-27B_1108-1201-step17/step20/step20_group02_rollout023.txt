                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _SOLVER = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _SOLVER


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2500, add_per_iter=40, verbose=True):
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac_y = fast_convolve(y)
    best_m = ac_y.max()
    t0 = time.monotonic()

    active_set = list(np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 500
    iter_count = 0
    stagnation_count = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if stagnation_count > 5:
                if verbose:
                    print("    LP-FW: persistent stagnation, stopping")
                break
            ac_curr = fast_convolve(y)
            n_pert = max(30, len(active_set) // 4)
            keep_idx = rng.choice(len(active_set), size=len(active_set) - n_pert, replace=False)
            new_adds = np.argsort(ac_curr)[::-1][:n_pert * 3]
            new_set = set(active_set[i] for i in keep_idx)
            for k in new_adds:
                new_set.add(int(k))
            active_set = sorted(new_set)[:]
            no_improve = 0
            stagnation_count += 1
            continue

        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            stagnation_count = 0
        else:
            no_improve += 1

        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep_idx = np.argsort(margins)[:int(max_K * 0.6)]
            active_set_base = [active_set[i] for i in keep_idx]
            violations = np.where(ac > M_curr - 1e-9)[0]
            for v in violations[:80]:
                if int(v) not in active_set_base:
                    active_set_base.append(int(v))
            active_set = sorted(set(active_set_base))

        K = len(active_set)
        if K == 0:
            break
        if K > max_K:
            active_set = active_set[:max_K]
            K = max_K

        coeffs = build_constraint_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(50, K // 4)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        fft_size = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]

        all_viol = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(all_viol) > 0:
            viol_amt = conv_ys[all_viol] - M_curr
            top_v = all_viol[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        # Adaptive step size
        ac_g = fast_convolve(g_dir)
        M_g = ac_g.max()
        # FW gap estimate
        fw_gap = M_curr - (coeffs @ g_dir).min() if K > 0 else M_curr

        best_alpha = 1.0 / (iter_count + 3.0)
        best_trial_M = M_curr

        if iter_count <= 3:
            alphas_to_try = [0.5, 0.3, 0.2, 0.1, 0.05]
        elif iter_count <= 20:
            alphas_to_try = [0.3, 0.2, 0.15, 0.1, 0.05, 0.03]
        else:
            base = 2.0 / (iter_count + 5.0)
            alphas_to_try = [min(0.5, base * 2), base, base * 0.5, base * 0.25, 0.02]

        for alpha in alphas_to_try:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 2:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", |S|=" + str(len(active_set)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", iters=" + str(iter_count))
    return best_a, best_m


def smoothed_subgrad(a0, time_limit, rng_seed=7, verbose=False):
    """Smoothed subgradient with momentum and adaptive temperature."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_convolve(y).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = y.copy()
    vel = np.zeros(n)
    momentum = 0.7
    eta0 = 0.03
    temperature = 50.0  # for log-sum-exp
    stall = 0
    max_stall = 2000
    iter_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break

        iter_count += 1
        eta = eta0 / (np.sqrt(t) + 1.0)

        # Nesterov lookahead
        y_look = x + momentum * vel
        y_look = np.maximum(y_look, 1e-12)
        s_y = y_look.sum()
        if s_y < 1e-12:
            y_look = x.copy()
        else:
            y_look = y_look / s_y

        # Compute convolution and softmax-weighted subgradient
        ac = fast_convolve(y_look)
        # Softmax weights over top values
        ac_clipped = np.clip(ac, 0, None)
        scaled = temperature * ac_clipped
        max_s = scaled.max()
        exp_s = np.exp(scaled - max_s)
        weights = exp_s / exp_s.sum()

        # Weighted subgradient: sum_k w_k * grad_k where grad_k = y_look shifted by k
        grad = np.zeros(n)
        # Only use top contributing lags for efficiency
        top_lags = np.argsort(weights)[::-1][:min(20, n)]
        for k in top_lags:
            k_int = int(k)
            w_k = weights[k_int]
            if w_k < 1e-10:
                continue
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                grad[idx] += w_k * y_look[k_int - idx]

        # Add small perturbation
        perturb_scale = 0.005 / (1.0 + t / 1000.0)
        grad += perturb_scale * rng.standard_normal(n) * y_look

        # Mirror descent (multiplicative update)
        log_a = np.log(np.maximum(y_look, 1e-30)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y_look.copy()
        else:
            a_new = a_new / s_new

        vel = momentum * vel + (a_new - x)
        x = a_new

        M_new = fast_convolve(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
            # Decrease temperature as we get closer to optimum
            if temperature > 20:
                temperature *= 0.999
        else:
            stall += 1

        # Periodic restart
        if stall > 1000 and stall % 1000 == 0:
            noise = rng.normal(0, 0.02, n) * best_a
            x_p = best_a + noise
            x_p = np.maximum(x_p, 1e-10)
            sp = x_p.sum()
            if sp > 1e-12:
                x = x_p / sp
                vel = np.zeros(n)
            temperature = max(20.0, temperature * 1.5)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print("    SubGrad t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    SubGrad done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=77, verbose=False):
    """Peak-aware coordinate descent with smart mass transfers."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 800
    step_scales = [0.001, 0.003, 0.007, 0.015, 0.03, 0.06]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(25, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-impact destination
            low_region = np.where(ac < ac.mean())[0]
            if len(low_region) < n // 4:
                low_region = np.arange(0, n // 2)

            candidates = low_region[rng.choice(len(low_region), size=min(20, len(low_region)), replace=False)]

            found = False
            for dst in candidates[:10]:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in step_scales:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_convolve(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break

            if not found:
                # Try random transfer
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 200 == 0:
                for _ in range(15):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.003, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 400 == 0:
            elapsed = time.monotonic() - t0
            print("    Coord (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + ", stall=" + str(stall))

    return best_a, best_m


def cmaes_explore(a0, time_limit, n_bumps=6, seed=42, verbose=False):
    """CMA-ES on Gaussian bump parameterization for global exploration."""
    n = len(a0)
    x = np.linspace(0.0, 1.0, n)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    best_a = norm_sum1(a0)
    best_val = 2.0 * n * fast_convolve(best_a).max()

    # Initialize with diverse bump configurations
    candidates = []
    for trial in range(5):
        params = np.zeros(3 * n_bumps)
        centers = rng.uniform(0.1, 0.9, n_bumps)
        sigmas = rng.uniform(0.03, 0.15, n_bumps)
        weights = rng.uniform(0.5, 2.0, n_bumps)
        for j in range(n_bumps):
            params[3 * j] = weights[j]
            params[3 * j + 1] = centers[j]
            params[3 * j + 2] = sigmas[j]
        candidates.append(params)

    dim = 3 * n_bumps
    pop_size = max(4 + int(3 * np.log(dim)), 12)
    best_iter = 0
    evals_done = 0
    max_evals = 2000

    def bump_to_seq(p):
        a = np.zeros(n)
        for j in range(n_bumps):
            w = max(0.01, p[3 * j])
            c = np.clip(p[3 * j + 1], 0.02, 0.98)
            s = max(0.01, p[3 * j + 2])
            a += w * np.exp(-((x - c) ** 2) / (2.0 * s * s))
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-10:
            return best_a.copy()
        return a / s_a

    def bump_eval(p):
        a = bump_to_seq(p)
        return 2.0 * n * fast_convolve(a).max()

    # Multi-start CMA-ES
    for trial_params in candidates[:3]:
        if time.monotonic() - t0 > time_limit:
            break
        mean = trial_params.copy()
        sigma = 0.1

        while evals_done < max_evals and time.monotonic() - t0 < time_limit:
            samples = mean + sigma * rng.standard_normal((pop_size, dim))
            fitness = np.zeros(pop_size)
            for i in range(pop_size):
                if time.monotonic() - t0 > time_limit:
                    break
                fitness[i] = bump_eval(samples[i])
                evals_done += 1
                if fitness[i] < best_val:
                    best_val = fitness[i]
                    best_a = bump_to_seq(samples[i]).copy()
                    if verbose:
                        elapsed = time.monotonic() - t0
                        print("    CMA-ES new best: J=" + repr(best_val) + " (" + str(round(elapsed, 1)) + "s)")

            lam = max(1, pop_size // 2)
            top_idx = np.argsort(fitness)[:lam]
            new_mean = np.zeros(dim)
            for rank, idx in enumerate(top_idx):
                new_mean += (1.0 / (rank + 1)) * samples[idx]
            new_mean /= (lam * (lam + 1) / 2)
            mean = new_mean
            sigma *= 0.95
            if sigma < 0.005:
                break

    if verbose:
        elapsed = time.monotonic() - t0
        print("    CMA-ES done (" + str(round(elapsed, 1)) + "s): J=" + repr(best_val) + ", evals=" + str(evals_done))
    return best_a, best_val


def get_initial_sequence(t_global):
    """Get the best starting sequence."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
                return p_norm, sc
    except Exception:
        pass

    n = 5000
    x = np.linspace(0, 1, n)
    best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
    best_score = evaluate_safe(best_seq)
    print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))
    return best_seq, best_score


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = get_solver()
    best_seq, best_score = get_initial_sequence(t_global)
    n0 = len(best_seq)

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: Quick LP-FW to establish baseline ===
    print("\n=== Phase 1: LP-FW initial (n=" + str(n0) + ") ===")
    t1 = min(remaining_time() * 0.20, 300.0)
    if t1 > 40 and n0 <= 9000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t1, solver,
                         initial_K=300, max_K=2000, add_per_iter=40, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1", t_global)

    # === Phase 2: Smoothed subgradient ===
    print("\n=== Phase 2: Smoothed SubGrad (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining_time() * 0.10, 150.0)
    if t2 > 30:
        a_out, _ = smoothed_subgrad(norm_sum1(best_seq), t2, rng_seed=11, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "subgrad", t_global)

    # === Phase 3: LP-FW second pass (more aggressive) ===
    print("\n=== Phase 3: LP-FW second (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining_time() * 0.20, 300.0)
    if t3 > 40 and len(best_seq) <= 9000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t3, solver,
                         initial_K=400, max_K=2500, add_per_iter=50, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2", t_global)

    # === Phase 4: CMA-ES global search (different basin) ===
    print("\n=== Phase 4: CMA-ES (n=" + str(len(best_seq)) + ") ===")
    t4 = min(remaining_time() * 0.08, 120.0)
    if t4 > 25:
        a_out, _ = cmaes_explore(norm_sum1(best_seq), t4, n_bumps=8, seed=123, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "cmaes", t_global)

    # === Phase 5: Multi-n exploration ===
    print("\n=== Phase 5: Multi-n ===")
    n_best = len(best_seq)
    # Try a range of n values around the current best
    n_candidates = sorted(set([
        n_best - 500, n_best - 1000, n_best + 500, n_best + 1000,
        max(4000, n_best - 2000), min(10000, n_best + 2000)
    ]))
    n_candidates = [l for l in n_candidates if 3000 <= l <= 10000]
    n_candidates.sort(key=lambda l: abs(l - n_best))

    for new_n in n_candidates[:4]:
        rt = remaining_time()
        if rt < 150:
            break
        if new_n == n_best:
            continue
        print("\n  Multi-n: n=" + str(new_n) + " (remaining=" + str(round(rt, 1)) + "s)")
        a_long = interpolate_seq(best_seq, new_n)

        t5 = min(rt * 0.15, 150.0)
        if t5 > 30 and new_n <= 9000:
            a_out, _ = lp_fw(a_long, t5, solver,
                             initial_K=300, max_K=2000, add_per_iter=40, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

        rt = remaining_time()
        if rt > 60 and len(best_seq) <= 9000:
            t5b = min(rt * 0.08, 80.0)
            if t5b > 20:
                a_out, _ = smoothed_subgrad(norm_sum1(best_seq), t5b, rng_seed=200 + new_n)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "subgrad_n" + str(new_n), t_global)

    # === Phase 6: LP-FW third pass (longest, most constraints) ===
    print("\n=== Phase 6: LP-FW long (n=" + str(len(best_seq)) + ") ===")
    t6 = min(remaining_time() * 0.20, 300.0)
    if t6 > 40 and len(best_seq) <= 9000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t6, solver,
                         initial_K=500, max_K=3000, add_per_iter=60, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw3", t_global)

    # === Phase 7: Coordinate polish ===
    print("\n=== Phase 7: Coordinate polish (n=" + str(len(best_seq)) + ") ===")
    t7 = min(remaining_time() * 0.10, 150.0)
    if t7 > 20:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t7, rng_seed=99, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord", t_global)

    # === Phase 8: Final LP-FW on best ===
    print("\n=== Phase 8: Final LP-FW (n=" + str(len(best_seq)) + ") ===")
    t8 = min(remaining_time() * 0.15, 200.0)
    if t8 > 30 and len(best_seq) <= 9000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t8, solver,
                         initial_K=600, max_K=3500, add_per_iter=80, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # === Phase 9: Final coordinate polish ===
    rt = remaining_time()
    if rt > 30:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), min(rt * 0.6, 100.0), rng_seed=2024, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
