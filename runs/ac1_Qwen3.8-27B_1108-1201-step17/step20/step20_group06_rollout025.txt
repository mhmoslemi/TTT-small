def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    n = len(y)
    b = fast_autocorr(y)
    size = 2 * n - 1
    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    if b_max <= 0:
        return np.zeros(n), b
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s_exp = exp_vals.sum()
    if s_exp < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s_exp
    a_rev = y[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, b


def nesterov_v4(a0, time_limit, beta_start=15.0, beta_end=3000.0,
                eta_init=0.5, top_k_ratio=0.02, restart_every=1200,
                verbose=True, seed=42):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    eta = eta_init
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        it_count += 1
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, b = compute_smoothed_grad(y, beta, top_k)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            x_new = x.copy()
            max_new = fast_autocorr(x_new).max()
        else:
            eta_trial = eta
            found = False
            y_max = fast_autocorr(y).max()
            for _ in range(12):
                z = y - eta_trial * grad
                x_new = simplex_project(z)
                max_new = fast_autocorr(x_new).max()
                if max_new <= y_max + 1e-15:
                    found = True
                    break
                eta_trial *= 0.5
            if not found:
                x_new = x.copy()
                max_new = fast_autocorr(x_new).max()
                eta = max(eta_trial, 0.005)

        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.03, 3.0)
            adaptive_top_k = max(5, int(n * top_k_ratio * (0.5 + 0.5 * eta / 3.0)))
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new
        adaptive_top_k = min(size if 'size' in dir() else 2 * n - 1, adaptive_top_k)

        if it_count % 40 == 0:
            beta = min(beta * 1.015, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = rng.integers(7)
            if strategy == 0:
                noise_level = 0.015 + 0.02 * rng.random()
                pert = rng.normal(0, noise_level, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                x_r = 0.85 * best_a + 0.15 * rng.dirichlet(np.ones(n) * 0.15)
            elif strategy == 2:
                scale = np.exp(0.04 * rng.standard_normal(n))
                x_r = best_a * scale
            elif strategy == 3:
                mask = rng.random(n) > 0.5
                x_r = best_a.copy()
                x_r[mask] *= 0.5 + 0.3 * rng.random()
                x_r[~mask] *= 1.2 + 0.3 * rng.random()
            elif strategy == 4:
                shift = int(rng.integers(1, max(2, n // 20)))
                x_r = np.roll(best_a, shift)
            elif strategy == 5:
                fft_size = 1 << (2 * n - 2).bit_length()
                A = np.fft.rfft(best_a, fft_size)
                mag = np.abs(A).copy()
                target = mag[1:].mean() if len(mag) > 1 else mag[0]
                for k in range(1, len(mag)):
                    if mag[k] > target * 1.5:
                        mag[k] = target * 1.5
                phase = np.angle(A)
                A_new = mag * np.exp(1j * phase)
                x_r = np.fft.irfft(A_new, fft_size)[:n]
            else:
                blend = rng.uniform(0.7, 0.9)
                x_r = blend * best_a + (1 - blend) * rng.dirichlet(np.ones(n))

            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 8.0)
            eta = eta_init
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest4 restart (t=%.1fs): J=%.6f beta=%.0f" % (elapsed, J, beta))

        if verbose and it_count % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest4 it=%d (t=%.1fs): J=%.6f beta=%.0f" % (it_count, elapsed, J, beta))

    return best_a, best_max_b


def spectral_shaping_v3(a0, time_limit, verbose=True, seed=777):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()

    alpha = 0.03
    flatness_target = 1.0
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        target = power[1:].mean() if len(power) > 1 else power[0]
        if target < 1e-30:
            continue

        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / target
            if ratio > flatness_target:
                red = 1.0 - alpha * (ratio - flatness_target)
                A_new[k] *= max(0.2, np.sqrt(red))
            elif ratio < 0.3:
                inc = 1.0 + alpha * 0.2 * (0.3 - ratio)
                A_new[k] *= min(3.0, np.sqrt(inc))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-17:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 300 == 0:
            A2 = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A2)) * 0.03
            A2[1:] *= np.exp(1j * phase_pert[1:])
            a_t = np.fft.irfft(A2, fft_size)[:n]
            a_t = np.maximum(a_t, 0.0)
            s_t = a_t.sum()
            if s_t > 1e-15:
                a_t = a_t / s_t
                m_t = fast_autocorr(a_t).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_t.copy()
                    a = a_t

        if it % 400 == 0:
            alpha = max(0.003, alpha * 0.96)
            flatness_target = max(1.0, flatness_target * 1.01)

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral3 it=%d (t=%.1fs): J=%.6f" % (it, elapsed, J))

    return best_a, best_m


def peak_cd(a0, time_limit, verbose=True, seed=42):
    """Peak-specific coordinate descent: directly target the max autocorrelation lag."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 3000:
        ac = fast_autocorr(a)
        size = 2 * n - 1
        top_peaks = np.argsort(ac)[::-1][:min(8, size)]
        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            partners = k_int - idx
            contrib = a[idx] * a[partners]
            total = contrib.sum()
            if total < 1e-15:
                continue

            # Find the pair with maximum contribution
            max_i_local = np.argmax(contrib)
            src_i = int(idx[max_i_local])
            src_j = int(partners[max_i_local])

            # Try to reduce a[src_i]*a[src_j] by moving mass
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                if time.monotonic() - t0 > time_limit:
                    break
                amt = a[src_i] * frac
                if amt < 1e-15:
                    continue

                # Try several random destinations
                for _ in range(8):
                    dst = int(rng.integers(n))
                    if dst == src_i or dst == src_j:
                        continue
                    # Check if dst is "far" from src_j (to avoid creating new peaks)
                    dist_j = abs(dst - src_j)
                    if dist_j < n // 4:
                        continue

                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-18:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                # Random exploration
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 500 == 0 and stall > 100:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakCD (t=%.1fs): J=%.6f stall=%d" % (elapsed, J, stall))

    return best_a, best_m


def lp_fw_v3(a0, time_limit, solver, initial_K=300, max_K=2000,
             add_per_iter=20, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 800

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        for alpha in [best_alpha, best_alpha * 0.5, best_alpha * 2.0,
                      1.0 / (t + 1), 0.03, 0.05, 0.1, 0.15]:
            alpha = max(0.005, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW3 it=%d (t=%.1fs): J=%.6f |S|=%d" % (t, elapsed, J, len(active_set)))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [%s]: %.6f (n=%d, t=%.1fs)" % (label, sc, len(a), elapsed))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Time budget: %.1fs" % time_budget)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=%d, score=%.6f" % (len(parent), p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6500
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=%d, score=%.6f" % (len(best_seq), best_score))

    n0 = len(best_seq)
    print("Start: n=%d, J=%.6f" % (n0, best_score))

    # Phase 1: Quick spectral refinement at current length
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Spectral (n=%d) ===" % n0)
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 15:
        a_out, _ = spectral_shaping_v3(norm_sum1(best_seq), t1, verbose=True, seed=777)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec1", t_global)

    # Phase 2: Nesterov (primary)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P2: Nesterov (n=%d) ===" % n_best)
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 40:
        a_out, _ = nesterov_v4(norm_sum1(best_seq), t2,
                               beta_start=15.0, beta_end=2000.0,
                               eta_init=0.5, top_k_ratio=0.02,
                               restart_every=1200, verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 3: Peak CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P3: PeakCD (n=%d) ===" % n_best)
    t3 = min(remaining * 0.08, 100.0)
    if t3 > 15:
        a_out, _ = peak_cd(norm_sum1(best_seq), t3, verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd1", t_global)

    # Phase 4: LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P4: LP-FW (n=%d) ===" % n_best)
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 30 and n_best <= 8000:
        a_out, _ = lp_fw_v3(norm_sum1(best_seq), t4, solver,
                            initial_K=350, max_K=2000,
                            add_per_iter=25, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P5: Multi-length (base n=%d) ===" % n_best)

    candidate_lengths = []
    for delta in [-500, -300, 300, 500, 800, 1200]:
        cand = n_best + delta
        if 3000 <= cand <= 10000 and cand != n_best:
            candidate_lengths.append(cand)
    # Remove duplicates, keep order
    seen = set()
    unique_lengths = []
    for cl in candidate_lengths:
        if cl not in seen:
            seen.add(cl)
            unique_lengths.append(cl)

    for new_n in unique_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print("  -> n=%d" % new_n)
        a_long = interpolate_sequence(best_seq, new_n)

        # Nesterov at new length
        t6a = min(remaining * 0.30, 160.0)
        if t6a > 30:
            a_out, _ = nesterov_v4(a_long, t6a,
                                   beta_start=20.0, beta_end=1500.0,
                                   eta_init=0.4, top_k_ratio=0.02,
                                   restart_every=1000, verbose=True,
                                   seed=new_n % 7919)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n%d" % new_n, t_global)

        # Quick spectral
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t6b = min(remaining * 0.20, 80.0)
            if t6b > 15:
                a_out, _ = spectral_shaping_v3(norm_sum1(best_seq), t6b, verbose=False,
                                               seed=new_n % 3163)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "spec_n%d" % new_n, t_global)

    # Phase 6: Second Nesterov round with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P6: Nesterov r2 (n=%d) ===" % n_best)
    if remaining > 60:
        t7 = min(remaining * 0.20, 250.0)
        if t7 > 30:
            a_out, _ = nesterov_v4(norm_sum1(best_seq), t7,
                                   beta_start=30.0, beta_end=3000.0,
                                   eta_init=0.3, top_k_ratio=0.01,
                                   restart_every=800, verbose=True, seed=999)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)

    # Phase 7: Peak CD round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P7: PeakCD r2 (n=%d) ===" % n_best)
    if remaining > 30:
        t8 = min(remaining * 0.10, 100.0)
        if t8 > 15:
            a_out, _ = peak_cd(norm_sum1(best_seq), t8, verbose=True, seed=654)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd2", t_global)

    # Phase 8: Final spectral + Nesterov polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P8: Final polish (n=%d) ===" % n_best)
    if remaining > 50:
        t9a = min(remaining * 0.20, 100.0)
        if t9a > 15:
            a_out, _ = spectral_shaping_v3(norm_sum1(best_seq), t9a, verbose=True, seed=888)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "spec2", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t9b = min(remaining * 0.35, 200.0)
            if t9b > 20:
                a_out, _ = nesterov_v4(norm_sum1(best_seq), t9b,
                                       beta_start=60.0, beta_end=5000.0,
                                       eta_init=0.2, top_k_ratio=0.005,
                                       restart_every=600, verbose=True, seed=555)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "nest3", t_global)

    # Phase 9: Final Peak CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P9: Final PeakCD (n=%d) ===" % n_best)
    if remaining > 20:
        t10 = min(remaining * 0.30, 120.0)
        if t10 > 10:
            a_out, _ = peak_cd(norm_sum1(best_seq), t10, verbose=True, seed=789)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd3", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: J=%.6f, n=%d, time=%.1fs" % (best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
