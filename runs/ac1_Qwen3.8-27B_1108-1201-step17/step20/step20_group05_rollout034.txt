def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def fft_conv2(a, b):

def cross_convolve_p_a(p, a, factor=1.0):

def proxy_target_penalty(c, mask=None):

def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    new_n = int(max(1, int(new_n)))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)

        m = float(np.max(cv))
        shifted = beta * (cv - m)
        shifted = np.clip(shifted, -700.0, 700.0)
        e = np.exp(shifted)
        se = float(e.sum())
        if se <= 0.0 or not math.isfinite(se):
            e = np.zeros(cv.size, dtype=np.float64)
            e[0] = 1.0
            se = 1.0
        p = e / se

        pf = np.zeros(size, dtype=np.float64)
        if mask is None:
            pf[:] = p
        else:
            pf[mask] = p

        Pr = np.fft.rfft(pf[::-1], N)
        conv = np.fft.irfft(A * Pr, N)[:total]
        grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
        grad = np.where(np.isfinite(grad), grad, 0.0)
        f = m + math.log(se) / beta

    if reg_a > 0.0:
        f += 0.5 * reg_a * float(np.dot(a, a))
        grad = grad + reg_a * a

    if not math.isfinite(f):
        f = 1e100
    return f, grad


def softmax_from_u(u):
    u = np.asarray(u, dtype=np.float64)
    n = u.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(u)):
        return np.ones(n, dtype=np.float64) / float(n)
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def make_v(a, ref=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 1:
        return np.array([], dtype=np.float64), 0
    if ref is None:
        if np.any(a > 0.0):
            ref = int(np.argmax(a))
        else:
            ref = 0
    ref = int(ref)
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.where(np.isfinite(v), v, -700.0)
    return v, ref


def from_v(v, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    return softmax_from_u(u)


def objgrad_v(v, beta, mask=None, reg_a=0.0, reg_u=0.0, ref=0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    if n <= 0:
        return 1e100, np.array([], dtype=np.float642)
    if not np.all(np.isfinite(v)):
        return 1e100, np.zeros(v.size, dtype=np.float64)

    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1

    u = np.empty(n, dtype=np.float64)
    u[ref] = 0.0
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]

    a = softmax_from_u(u)
    f, ga = lse_a(a, beta, mask, reg_a)
    if not math.isfinite(f):
        f = 1e100
        ga = np.zeros(n, dtype=np.float64)

    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)

    gv = np.empty(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]

    if reg_u > 0.0:
        f += 0.5 * reg_u * float(np.dot(v, v))
        gv = gv + reg_u * v

    if not np.all(np.isfinite(gv)):
        gv = np.zeros(v.size, dtype=np.float64)
    return f, gv


def run_lbfegs(a0, beta, mask, maxfun=500, reg_a=0.0, reg_u=1e-12, verbose=False):
    a = norm_sum1(a0)
    n = a.size
    if n <= 1:
        return a, np.array([], dtype=np.float64), 0

    v0, ref = make_v(a)
    if not SCIPY_OK or maxfun <= 0:
        return a, v0, ref

    t0 = time.monotonic()
    options = {
        "maxfun": int(maxfun),
        "maxiter": int(maxfun + 30),
        "ftol": 1e-12,
        "gtol": 1e-7,
        "disp": False
    }
    res = None
    try:
        res = scipy.optimize.minimize(
            objgrad_v,
            v0,
            args=(beta, mask, reg_a, reg_u, ref),
            method="L-BFGS-B",
            jac=True,
            options=options
        )
    except Exception:
        res = None

    if res is not None and hasattr(res, "x"):
        vx = np.asarray(res.x, dtype=np.float64)
        if vx.size == v0.size and np.all(np.isfinite(vx)):
            a_out = from_v(vx, ref)
            a_out = norm_sum1(a_out)
            v_out, ref_out = make_v(a_out)
            return a_out, v_out, ref_out

    return a, v0, ref


def lbfegs_phase(a_init, base_mask, budget, cost):
    a = norm_sum1(a_init)
    n = a.size
    size = 2 * n - 1
    best_a = a.copy()
    best_M = proxy_max(fft_conv(a), base_mask)
    current_a = a.copy()

    t1 = int(min(1000, max(50, size // 10)))
    t2 = int(min(600, max(40, size // 20)))
    t3 = int(min(400, max(30, size // 30)))
    t4 = int(max(12, size // 100))
    t5 = int(max(8, size // 160))

    schedules = [
        (8.0, None),
        (25.0, None),
        (60.0, t1),
        (100.0, t2),
        (200.0, t3),
        (500.0, t4),
        (800.0, t5)
    ]

    total_eval_budget = int(budget / max(2.5 * cost, 0.03))
    per_run = int(max(60, min(600, max(1, total_eval_budget // max(1, len(schedules))))))

    t0 = time.monotonic()
    for beta, top_count in schedules:
        elapsed = time.monotonic() - t0
        if elapsed > budget - 5:
            break
        if top_count is None:
            msk = None
        else:
            msk = make_top_mask(current_a, top_count, base_mask)

        a_new, v_new, ref_new = run_lbfegs(
            current_a, beta, msk, maxfun=per_run,
            reg_a=0.0, reg_u=1e-12
        )
        c_new = fft_conv(a_new)
        m_new = proxy_max(c_new, base_mask)

        if math.isfinite(m_new):
            if m_new <= best_M * (1.0 + 1e-4) + 1e-18:
                current_a = a_new
        else:
            current_a = best_a

        if math.isfinite(m_new) and m_new < best_M - 1e-15:
            best_M = m_new
            best_a = a_new.copy()

    return best_a


def line_search_mix(a, g, cf, cg, fg, mask, current_M, t):
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    n = a.size

    base_alphas = [
        max(0.005, 0.5 / (t + 1.0)),
        max(0.01, 1.0 / (t + 1.0)),
        max(0.02, 2.0 / (t + 2.0)),
        0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.50, 0.70, 0.90, 1.0
    ]
    alphas = []
    seen = set()
    for al in base_alphas:
        al = float(al)
        if not math.isfinite(al) or al <= 0.0 or al > 1.0:
            continue
        key = int(round(al * 10000.0))
        if key in seen:
            continue
        seen.add(key)
        alphas.append(al)
    if len(alphas) == 0:
        alphas = [0.1]

    best_alpha = 0.0
    best_m = current_M
    for alpha in alphas:
        one = 1.0 - alpha
        val = (one * one) * cf + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
        if mask is None:
            m = float(np.max(val))
        else:
            m = float(np.max(val[mask]))
        if math.isfinite(m) and m < best_m:
            best_m = m
            best_alpha = alpha

    if best_alpha <= 0.0:
        return a.copy(), best_m, 0.0

    new_a = (1.0 - best_alpha) * a + best_alpha * g
    new_a = np.maximum(new_a, 0.0)
    s = float(new_a.sum())
    if s < 1e-15:
        return a.copy(), best_m, 0.0
    new_a = new_a / s
    return new_a, best_m, best_alpha


def game_direction(f, active_idx, mw_iters=30, seed=0, init_logits=None):
    f = norm_sum1(f)
    n = f.size
    size = 2 * n - 1
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K <= 0:
        return np.ones(n, dtype=np.float64) / float(n)

    rng = np.random.default_rng(seed)
    if init_logits is None:
        logits = np.zeros(K, dtype=np.float64)
    else:
        logits = np.asarray(init_logits, dtype=np.float64).copy()
        if logits.size != K:
            logits = np.zeros(K, dtype=np.float64)
        logits = np.where(np.isfinite(logits), logits, 0.0)
        logmx = float(np.max(logits))
        if math.isfinite(logmx):
            logits = logits - logmx
    logits += rng.normal(0.0, 0.03, K)

    def normalize_logits_to_lambda(loc):
        loc = np.where(np.isfinite(loc), loc, -1000.0)
        loc = loc - float(np.max(loc))
        w = np.exp(loc)
        s = float(w.sum())
        if s <= 0.0:
            w = np.ones(K, dtype=np.float64)
            s = float(K)
        return w / s

    lambda_w = normalize_logits_to_lambda(logits)
    g = None
    for it in range(max(1, int(mw_iters))):
        pf = np.zeros(size, dtype=np.float64)
        pf[active_idx] = lambda_w

        pressure = cross_convolve_p_a(pf, f, 1.0)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)

        pmx = float(np.max(pressure))
        pmn = float(np.min(pressure))
        ptp = pmx - pmn
        if ptp <= 1e-30:
            g = np.ones(n, dtype=np.float64) / float(n)
        else:
            scaled = (pressure - pmn) / ptp
            w = np.exp(-10.0 * scaled)
            w = np.where(np.isfinite(w), w, 0.0)
            w += 1e-12
            g = w / float(w.sum())
            g = 0.96 * g + 0.04 * (np.ones(n, dtype=np.float64) / float(n))
            g = g / float(g.sum())

        h = fft_conv2(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmac_ = float(np.max(ha))
        hmin = float(np.min(ha))
        hptp = hmac_ - hmin
        if hptp > 1e-30:
            eta = 15.0 / hptp
            eta = min(eta, 50.0)
            logits += eta * (ha - float(np.mean(ha)))
        lambda_w = normalize_logits_to_lambda(logits)

    return g


def lp_direction_h
ighs(f, active_idx, time_limit=6.0, max_k=800):
    if not SCIPY_OK:
        return None
    if not hig
hs_available():
        return None

    f = norm_sum1(f)
    n = f.size
    active_idx = np.asarray(active_idx, dtype=int)
    if active_idx.size == 0:
        return None
    if active_idx.size > max_k:
        active_idx = active_idx[:max_k]
    K = active_idx.size

    row_parts = []
    col_parts = []
    val_parts = []
    for r, k in enumerate(active_idx):
        s = max(0, int(k) - n + 1)
        e = min(n, int(k) + 1)
        if s >= e:
            continue
        cols = np.arange(s, e, dtype=np.int32)
        row_parts.append(np.full(cols.size, r, dtype=np.int32))
        col_parts.append(cols)
        # (a*g)[k] = sum_i a[i] * g[k-i], so coefficient of g[j] is a[k-j]
        # j ranges from max(0, k-n+1) to min(k, n-1)
        # a[k-j] where j is the column index
        vals_row = f[int(k) - cols].astype(np.float64)
        val_parts.append(vals_row)

    if not row_parts:
        return None

    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0

    neg_one = scipy.sparse.csc_matrix(-np.ones((K, 1)), dtype=np.float64)
    A_ub = scipy.sparse.hstack((A, neg_one), format="csr")
    A_eq = np.ones((1, n + 1), dtype=np.float64)
    A_eq[0, n] = 0.0
    b_eq = np.ones(1, dtype=np.float64)
    b_ub = np.zeros(K, dtype=np.float64)

    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = scipy.optimize.linprog(
            c_obj, A_ub=A_ub, b_ub=b_ub,
            A_eq=A_eq, b_eq=b_eq, bounds=bounds,
            method="HiGHS",
            options={"disp": False, "time_limit": float(time_limit)}
        )
    except Exception:
        return None

    if res is None or not getattr(res, "success", False):
        return None

    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.where(np.isfinite(g) & (g > 0.0), g, 0.0)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs


def frank_wolfe_search(a0, budget, mask, active_max=2500, game_iters=28,
                       exact_every=6, exact_time=6.0, verbose=False):
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = proxy_max(c, mask)
    best_a = a.copy()
    best_M = M
    active = active_top(c, mask, active_max)

    no_improve = 0
    t = 0
    while time.monotonic() - t0 < time_limit - 5 and no_improve < 100:
        t += 1
        time_left = time_limit - (time.monotonic() - t0)
        if time_left < 20:
            break

        g = None
        if hig
hs_available() and exact_every > 0 and (t % exact_every == 0) and time_left > 80:
            g = lp_direction_h
ighs(a, active, exact_time, max_k=800)

        if g is None:
            init_log = None
            if active.size > 0:
                av = c[active]
                av = np.where(np.isfinite(av) & (av > 0.0), av, 1e-300)
                init_log = np.log(av)
                init_log -= float(np.max(init_log))
            g = game_direction(a, active, mw_iters=game_iters,
                               seed=1000 + t, init_logits=init_log)

        if g is None:
            break

        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        new_a, new_M, alpha = line_search_mix(a, g, c, cg, fg, mask, M, t)

        if new_M < M - 1e-16:
            if alpha > 0:
                one = 1.0 - alpha
                c = (one * one) * c + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
                c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
            a = new_a
            M = new_M
            no_improve = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            active = active_top(c, mask, active_max)
        else:
            no_improve += 1
            if no_improve % 12 == 0:
                rng = np.random.default_rng(9973 + t)
                # lopsided perturbation
                pert = rng.normal(0.0, 0.02, n)
                # add a small ramp to bias towards lopsided
                ramp = np.linspace(-0.01, 0.01, n)
                pert += ramp * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-12)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = proxy_max(c, mask)
                active = active_top(c, mask, active_max)
                no_improve = 0

    return best_a


def apply_single_move(c, a, s, d, tmove):
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = a.size
    c_new = c.copy()
    # Moving tmove from position s to position d:
    # a_new[s] = a[s] - tmove, a_new[d] = a[d] + tmove (for s != d)
    # delta_a = -tmove at s, +tmove at d
    # delta_conv = delta_a * a + a * delta_a
    # At position k: delta_conv[k] = -tmove*a[k-s] + tmove*a[k-d] (with valid index handling)
    # More precisely: delta_conv[k] = sum_i delta_a[i]*a[k-i] + sum_i a[i]*delta_a[k-i]
    # = -tmove*a[k-s] + tmove*a[k-d] + (-tmove*a[s-(k)]) + tmove*a[d-(k)]  (with range checks)
    # For our purposes with the FFT-based conv definition:
    # conv[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]
    # delta at position s: -tmove, at position d: +tmove
    # delta_conv[k] = -tmove * a[k-s] (if 0<=k-s<n) + tmove * a[k-d] (if 0<=k-d<n)
    #                + (-tmove) * a[s-k] (if 0<=s-k<n) + tmove * a[d-k] (if 0<=d-k<n)
    # But since a is real: a[k-s] term and a[s-k] term are related.
    # Actually conv[k] = sum_i a[i]*a[k-i], so:
    # delta_conv[k] = -tmove * a[k-s] (valid if 0<=k-s<n) 
    #                + tmove * a[k-d] (valid if 0<=k-d<n)
    #                + (-tmove) * a[s] * [s in range AND k-s in range]  -- wait this is getting confusing.
    # Let me just use the simple update:
    # delta_a has -tmove at s and +tmove at d.
    # delta_conv = conv(delta_a, a) + conv(a, delta_a) + conv(delta_a, delta_a)
    # conv(delta_a, a)[k] = delta_a[0]*a[k] if k in range + ... = -tmove*a[k-s] + tmove*a[k-d]
    #   where a[k-s] is defined as 0 if k-s out of [0,n-1]
    # conv(a, delta_a)[k] = a[0]*delta_a[k] + ... = -tmove*a[k-s] + tmove*a[k-d]  (same!)
    # conv(delta_a, delta_a)[k] = -tmove*delta_a[k-s] + tmove*delta_a[k-d]
    #   = (if k-s==s: -tmove*(-tmove)=t^2) + (if k-s==d: -tmove*tmove=-t^2)
    #   + (if k-d==s: tmove*(-tmove)=-t^2) + (if k-d==d: tmove*tmove=t^2)
    #   So: +t^2 at k=2s, -t^2 at k=s+d, -t^2 at k=s+d, +t^2 at k=2d
    #   = +t^2 at k=2s, -2*t^2 at k=s+d, +t^2 at k=2d

    # Simpler: just do the rank-1 update
    # delta_conv[k] = 2*tmove*(a[k-d] - a[k-s]) for valid k
    # where a[x] = 0 if x not in [0, n-1]
    
    # Position k: contribution from delta_a at index s is -tmove * a[k-s] (if 0<=k-s<n)
    #              contribution from delta_a at index d is +tmove * a[k-d] (if 0<=k-d<n)
    # And by symmetry (since delta_conv = conv(da,a)+conv(a,da) and conv is the same):
    # Actually for self-conv: conv[k] = sum_i a[i]a[k-i]
    # delta_conv[k] = sum_i da[i]a[k-i] + sum_i a[i]da[k-i]
    # = da[s]*a[k-s] + da[d]*a[k-d] + a[s]*da[k-s] + a[d]*da[k-d]
    # Wait no, that's not right either. Let me just do it directly.
    # da[i] = -tmove if i==s, +tmove if i==d, 0 otherwise.
    # sum_i da[i]*a[k-i] = -tmove*a[k-s] + tmove*a[k-d]  (with a[x]=0 if x out of range)
    # sum_i a[i]*da[k-i]: da[k-i] is nonzero when k-i==s (i=k-s) or k-i==d (i=k-d)
    # = a[k-s]*da[s] + a[k-d]*da[d] = a[k-s]*(-tmove) + a[k-d]*(tmove)  (same!)
    # sum_i da[i]*da[k-i] = da[s]*da[k-s] + da[d]*da[k-d]
    # = (-tmove)*da[k-s] + (tmove)*da[k-d]
    # da[k-s] = -tmove if k-s==s (k==2s), +tmove if k-s==d (k==s+d), 0 otherwise
    # da[k-d] = -tmove if k-d==s (k==s+d), +tmove if k-d==d (k==2d), 0 otherwise
    # So: (-tmove)*(-tmove) at k=2s = +t^2
    #     (-tmove)*(+tmove) at k=s+d = -t^2
    #     (+tmove)*(-tmove) at k=s+d = -t^2
    #     (+tmove)*(+tmove) at k=2d = +t^2
    # Total second-order: +t^2 at k=2s, -2t^2 at k=s+d, +t^2 at k=2d

    t2 = tmove * tmove
    # First-order update: 2*tmove*(a[k-d] - a[k-s]) for k where indices are valid
    # a[k-s] is valid for k in [s, s+n-1]
    # a[k-d] is valid for k in [d, d+n-1]
    
    lo_s = s
    hi_s = s + n - 1
    lo_d = d
    hi_d = d + n - 1
    
    # For k in [lo_s, hi_s]: subtract 2*tmove*a[k-s]
    # For k in [lo_d, hi_d]: add 2*tmove*a[k-d]
    
    # We can do this as:
    # c_new[k] = c[k] + 2*tmove*(a[k-d] - a[k-s]) + second_order
    
    # Efficient approach: use the fact that the first-order term is a shifted version of a
    # c_new[lo_s:hi_s+1] -= 2*tmove * a[0:n]  (this is the -tmove*a[k-s] part, where k-s goes 0..n-1)
    # c_new[lo_d:hi_d+1] += 2*tmove * a[0:n]  (this is the +tmove*a[k-d] part)
    
    lo_s = max(0, s)
    hi_s = min(c_new.size - 1, s + n - 1)
    lo_d = max(0, d)
    hi_d = min(c_new.size - 1, d + n - 1)
    
    s_start = lo_s - s  # start index in a for the subtraction
    s_end = hi_s - s + 1
    d_start = lo_d - d
    d_end = hi_d - d + 1
    
    c_new[lo_s:hi_s+1] -= 2.0 * tmove * a[s_start:s_end]
    c_new[lo_d:hi_d+1] += 2.0 * tmove * a[d_start:d_end]
    
    # Second order
    if 2 * d < c_new.size:
        c_new[2 * d] += t2
    if 2 * s < c_new.size:
        c_new[2 * s] += t2
    if s + d < c_new.size:
        c_new[s + d] -= 2.0 * t2
    
    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
    return c_new


def apply_pair_move(c, a, s1, d1, s2, d2, t1, t2):
    """Apply two simultaneous moves: t1 from s1 to d1, t2 from s2 to d2."""
    c_new = apply_single_move(c, a, s1, d1, t1)
    c_new = apply_single_move(c_new, a, s2, d2, t2)
    return c_new


def coordinate_polish(a0, budget, mask, verbose=False):
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    c = fft_conv(a)
    M = proxy_max(c, mask)
    best_a = a.copy()
    best_M = M
    n = a.size
    stall = 0
    rnd = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit - 3 and stall < 150:
        rnd += 1
        if n <= 1:
            break

        top_k = min(50, 2 * n - 1)
        if top_k <= 0:
            break
        top_idx = active_top(c, mask, top_k)
        if top_idx.size == 0:
            break

        p_top = np.zeros(2 * n - 1, dtype=np.float64)
        p_top[top_idx] = 1.0 / float(top_idx.size)
        pressure = cross_convolve_p_a(p_top, a, 2.0)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)

        src_count = min(10, n)
        dst_count = min(15, n)

        src_score = pressure * (1.0 + float(n) * a)
        src_candidates = np.argpartition(src_score, min(src_count - 1, n - 1))[:src_count]
        dst_score = pressure + 1e-5 * a
        dst_candidates = np.argpartition(dst_score, min(dst_count - 1, n - 1))[:dst_count]

        improved_any = False
        best_cand = None

        # Single moves
        for srr in src_candidates:
            s = int(srr)
            if a[s] < 1e-12:
                continue
            for drr in dst_candidates:
                d = int(drr)
                if d == s:
                    continue
                for frac in (0.003, 0.008, 0.015, 0.03):
                    tmove = a[s] * frac
                    if tmove <= 1e-16:
                        continue
                    tmove = min(tmove, a[s] * 0.95)
                    c_new = apply_single_move(c, a, s, d, tmove)
                    m_new = proxy_max(c_new, mask)
                    if m_new < M - 1e-18:
                        cand = (m_new, [(s, d, tmove)])
                        if best_cand is None or cand[0] < best_cand[0]:
                            best_cand = cand
                            improved_any = True

        # Pair moves (new!)
        if not improved_any or rnd % 3 == 0:
            for s1r in src_candidates[:5]:
                s1 = int(s1r)
                if a[s1] < 1e-12:
                    continue
                for d1r in dst_candidates[:5]:
                    d1 = int(d1r)
                    if d1 == s1:
                        continue
                    for s2r in src_candidates[:5]:
                        s2 = int(s2r)
                        if s2 == s1 or s2 == d1 or a[s2] < 1e-12:
                            continue
                        for d2r in dst_candidates[:5]:
                            d2 = int(d2r)
                            if d2 in (s1, d1, s2):
                                continue
                            t1 = min(a[s1], a[s2]) * 0.01
                            t2 = min(a[s1], a[s2]) * 0.01
                            if t1 <= 1e-16:
                                continue
                            c_new = apply_pair_move(c, a, s1, d1, s2, d2, t1, t2)
                            m_new = proxy_max(c_new, mask)
                            if m_new < M - 1e-18:
                                cand = (m_new, [(s1, d1, t1), (s2, d2, t2)])
                                if best_cand is None or cand[0] < best_cand[0]:
                                    best_cand = cand
                                    improved_any = True

        if best_cand is not None:
            m_new, moves = best_cand
            for (s, d, tmove) in moves:
                a[s] -= tmove
                a[d] += tmove
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = fft_conv(a)
            M = proxy_max(c, mask)
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Random exploration with lopsided bias
            if stall % 8 == 7:
                for _ in range(40):
                    s = int(rng.integers(n))
                    # Bias destination towards the opposite end for lopsidedness
                    if s < n // 2:
                        d = int(rng.integers(n // 2, n))
                    else:
                        d = int(rng.integers(0, n // 2))
                    if s == d or a[s] < 1e-12:
                        continue
                    frac = float(rng.uniform(0.003, 0.04))
                    tmove = a[s] * frac
                    tmove = min(tmove, a[s] * 0.95)
                    if tmove <= 1e-16:
                        continue
                    c_new = apply_single_move(c, a, s, d, tmove)
                    m_new = proxy_max(c_new, mask)
                    if m_new < M - 1e-18:
                        a[s] -= tmove
                        a[d] += tmove
                        a = np.maximum(a, 0.0)
                        ss = float(a.sum())
                        if ss < 1e-15:
                            break
                        a = a / ss
                        c = fft_conv(a)
                        M = proxy_max(c, mask)
                        if M < best_M:
                            best_M = M
                            best_a = a.copy()
                        stall = 0
                        break

    return best_a
