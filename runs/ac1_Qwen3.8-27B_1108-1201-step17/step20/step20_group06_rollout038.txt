import numpy as np

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def truncated_binomial(n):
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return a

def compute_smoothed_grad(a, b, beta, top_k=None):
    n = len(a)
    size = 2 * n - 1
    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
    else:
        b_use = b
    b_max = b.max()
    if b_max <= 0:
        return np.zeros(n)
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s
    q = a[::-1]
    fft_s = 1 << (n + size - 1).bit_length()
    Pf = np.fft.rfft(p, fft_s)
    Qf = np.fft.rfft(q, fft_s)
    conv = np.fft.irfft(Pf * Qf, fft_s)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad

def peak_equalize(a0, time_limit, verbose=True):
    """Targeted peak equalization: transfer mass to reduce top peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    rng = np.random.default_rng(999)

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        M = b.max()
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()

        # Find top peaks
        top_k_count = min(15, size)
        peak_idxs = np.argsort(b)[::-1][:top_k_count]

        # For the highest peak, find the contributing pair and try to reduce it
        k_max = int(peak_idxs[0])
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        if js > je:
            continue

        idx_range = np.arange(js, je + 1)
        partners = k_max - idx_range
        contrib = a[idx_range] * a[partners]
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            continue

        # Try reducing the largest contributor
        best_improvement = 0
        best_move = None

        # Strategy: move mass from the peak contributor to a random location
        top_contrib_idx = np.argsort(contrib)[::-1][:5]
        for ci in top_contrib_idx:
            src = int(idx_range[ci])
            partner = int(partners[ci])
            if src == partner:
                continue
            # Try moving a small fraction from src to a random destination
            for frac in [0.005, 0.01, 0.02, 0.03]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                # Pick destination that doesn't create new peaks
                for _ in range(3):
                    dst = int(rng.integers(n))
                    if dst == src or dst == partner:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < M - best_improvement:
                        best_improvement = M - M_trial
                        best_move = (src, dst, amt)

        if best_move is not None:
            src, dst, amt = best_move
            a[src] -= amt
            a[dst] += amt
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a
        else:
            # No improvement found, try random perturbation
            if it % 100 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_a = a.sum()
                    if s_a > 1e-15:
                        a = a / s_a

        if verbose and it % 500 == 0:
            J = 2.0 * n * best_m
            print("    PeakEq it=" + str(it) + " J=" + repr(round(J, 6)) + " t=" + str(round(time.monotonic() - t0, 1)) + "s")

    return best_a, best_m

def interpolate_sequence(f, new_n):

    if old_n == new_n:

    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global, verbose=True):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        if verbose:
            elapsed = time.monotonic() - t_global
            print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

            parent = p.copy()
