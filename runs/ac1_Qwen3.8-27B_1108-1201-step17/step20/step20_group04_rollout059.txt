def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)

        p = exp_vals / total
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(y, fft_size)
    P = np.fft.rfft(p, fft_size)
    a_rev = y[::-1]
    Conv = np.fft.irfft(P * np.fft.rfft(a_rev, fft_size), fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad, p, c


def lp_direction(f, M, top_k_indices, solver=None, timeout=30):
    """
    Solve LP: max sum(b) s.t. b>=0, conv(f,b)[k]<=M for k in top_k_indices.
    Returns normalized direction b/sum(b) or None on failure.
    """
    try:
        import cvxpy as cp
    except Exception:
        return None

    n = len(f)
    if solver is None:
        solver = get_solver()

    b = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(b))
    constraints = []

    for k in top_k_indices:
        k = int(k)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        # conv(f,b)[k] = sum_{j=j_start}^{j_end} f[k-j] * b[j]
        js = np.arange(j_start, j_end + 1)
        coeffs = f[k - js]
        # Only include if any coefficient is positive
        if coeffs.max() <= 0:
            continue
        constraints.append(coeffs @ b[j_start:j_end + 1] <= M)

    if not constraints:
        return None

    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None

    if b.value is None:
        return None
    b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-12:
        return None
    return b_val / s


def lp_guided_opt(f0, time_limit, K=40, solver=None, verbose=True, seed=42):
    """
    LP-guided optimization: at each step, solve LP for best direction
    that keeps conv(f, b) <= M on top K peaks, then take a step.
    """
    n = len(f0)
    f = norm_sum1(f0)
    best_f = f.copy()
    best_M = fast_autocorr(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        c = fast_autocorr(f)
        c_max = c.max()

        if c_max < best_M - 1e-15:
            best_M = c_max
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

        if stall > 500:
            # Restart with perturbation
            noise = 0.02 * rng.normal(0, 1, n) * best_f
            f = (best_f + noise)
            f = np.maximum(f, 1e-10)
            f = f / f.sum()
            stall = 0
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_M
                print("    LP restart (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
            continue

        # Find top K peaks
        top_k_indices = np.argsort(c)[::-1][:K]
        M = c.max()

        # Solve LP
        g = lp_direction(f, M, top_k_indices, solver=solver)
        if g is None:
            if verbose and iteration % 50 == 0:
                print("    LP failed at iteration " + str(iteration))
            # Fallback: use gradient direction
            grad, _, _ = compute_smoothed_grad(f, 100.0, top_k=k_top_k_indices := min(K, 20))
            g = simplex_project(-grad)
            g = g / g.sum()

        # Line search: try several step sizes
        best_t = 0.0
        best_trial_M = c_max
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
            f_trial = (1 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-12:
                continue
            f_trial = f_trial / s
            M_trial = fast_autocorr(f_trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_t = t

        if best_t == 0.0:
            stall += 10
            continue

        f = (1 - best_t) * f + best_t * g
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-12:
            f = best_f.copy()
        else:
            f = f / s

        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print("    LP iter=" + str(iteration) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " M=" + repr(round(c_max, 8)))

    return best_f, best_M


def nesterov_opt(a0, time_limit, beta_start=50.0, beta_end=5000.0,
                 beta_growth=1.008, eta_base=0.5, top_k_ratio=0.015,
                 restart_every=1500, verbose=True, seed=42):
    """Nesterov accelerated projected gradient on log-sum-exp smoothed objective."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, c = compute_smoothed_grad(y, beta, top_k)

        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.03 / (1.0 + 0.05 * (t // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + str(round(beta, 1)))

    return best_a, best_max_b


def peak_surgery(a0, time_limit, verbose=True, seed=77):
    """Improved peak surgery with pressure-field guided destination selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 5000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_top = min(40, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]

        improved = False
        found_move = False

        for k in top_idx[:20]:
            if found_move or time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            if products.max() < 1e-20:
                continue

            top_pairs = np.argsort(products)[::-1][:5]
            for pi in top_pairs:
                if found_move:
                    break
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n:
                    continue
                if a[src_i] < 1e-14 and a[partner] < 1e-14:
                    continue

                reduce_i = src_i if a[src_i] >= a[partner] else partner
                if a[reduce_i] < 1e-14:
                    continue

                # Compute pressure from top peaks
                pressure = np.zeros(n)
                for k2 in top_idx[:15]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        pressure[idx2] += a[k2i - idx2]
                        pressure[k2i - idx2] += a[idx2]

                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_idx[:10]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2 + 1] = True

                candidates = np.where(~in_peak)[0]
                if len(candidates) < 100:
                    candidates = np.arange(n)
                mask = candidates != reduce_i
                candidates = candidates[mask]
                if len(candidates) == 0:
                    continue

                cand_pressure = pressure[candidates]
                n_try = min(8, len(candidates))
                try_dsts = candidates[np.argsort(cand_pressure)[:n_try]]

                for dst in try_dsts:
                    if found_move:
                        break
                    dst_int = int(dst)
                    for frac in [0.003, 0.008, 0.015, 0.03, 0.05]:
                        amt = a[reduce_i] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[reduce_i] -= amt
                        a_trial[dst_int] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            found_move = True
                            break

        if not improved:
            if no_improve > 500:
                for _ in range(15):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                break

        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSurgery (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve))

    return best_a, best_m


def pairwise_cd(a0, time_limit, verbose=True, seed=123):
    """Fine-grained pairwise coordinate descent for final polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 800:
        c = fast_autocorr(a)
        threshold = np.percentile(c, 95)
        peak_positions = np.where(c > threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.argsort(c)[::-1][:20]

        improved = False

        for k in peak_positions[:25]:
            if improved and (time.monotonic() - t0 > time_limit):
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]

            top_pairs = np.argsort(products)[::-1][:4]
            for pi in top_pairs:
                if improved:
                    break
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n or partner == src_i:
                    continue
                if a[src_i] < 1e-14 and a[partner] < 1e-14:
                    continue

                larger_i = src_i if a[src_i] >= a[partner] else partner
                if a[larger_i] < 1e-14:
                    continue

                for frac in [0.001, 0.002, 0.004, 0.008]:
                    if improved:
                        break
                    amt = a[larger_i] * frac
                    if amt < 1e-18:
                        continue
                    for _ in range(5):
                        dst = int(rng.integers(n))
                        if dst == larger_i:
                            continue
                        a_trial = a.copy()
                        a_trial[larger_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break

        if not improved:
            stall += 1
            if stall % 150 == 0:
                anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
                eps = 0.008 / (1.0 + stall / 100.0)
                a_try = a + eps * anti * a
                a_try = np.maximum(a_try, 0.0)
                s_t = a_try.sum()
                if s_t > 1e-15:
                    a_try = a_try / s_t
                    M_try = fast_autocorr(a_try).max()
                    if M_try < best_m:
                        a = a_try
                        best_m = M_try
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 300 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PairCD (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 7000
        x = np.linspace(0, 1, n_base)
        a_init = (1.0 - x) ** 1.5 + 0.1
        best_seq = norm_sum1(a_init)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(n_base) + ", score=" + repr(best_score))

    n0 = len(best_seq)
    print("Starting: n=" + str(n0) + ", score=" + repr(best_score))

    # Phase 1: LP-guided optimization (main novel contribution)
    remaining = time_budget - (time.monotonic() - t_global)

        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp2", t_global)

    # Phase 4: Peak surgery
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Peak Surgery (n=" + str(n_best) + ") ===")

    # Phase 5: Multi-length exploration

            break

    # Phase 7: Pairwise CD for final polish
    remaining = time_budget - (time.monotonic() - t_global)

        best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd_final", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60 and solver is not None:
        n_best = len(best_seq)
        print("\n=== Phase 8: Final LP (n=" + str(n_best) + ") ===")
        t8 = min(remaining * 0.50, 150.0)
        if t8 > 20:
            a_out, _ = lp_guided_opt(norm_sum1(best_seq), t8, K=25, solver=solver, verbose=True, seed=9)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_ultimate", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1((1.0 - x) ** 1.5 + 0.1)
        best_score = evaluate_safe(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)

    return [float(x) for x in best_seq]
