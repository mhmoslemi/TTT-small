def fast_autocorr(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interp_seq(f, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

                    break

                break

def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

    conv = np.fft.irfft(Af * Pf, fft_size)
    grad = conv[n - 1:2 * n - 1]

    return grad, b

def peak_shaving(a0, time_limit, beta_start=50.0, beta_end=3000.0,
                 beta_growth=1.004, eta_init=0.4, top_k_ratio=0.015,
                 verbose=True, seed=42):
    """
    Proximal gradient on simplex with softmax-smoothed max autocorrelation.
    Nesterov momentum + backtracking line search + diverse restarts.
    """
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_mom = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(5, int(n * top_k_ratio))
    eta = eta_init

    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        theta = (t_mom - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, b = compute_grad(y, beta, top_k)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            x_new = x
            m_new = fast_autocorr(x_new).max()
        else:
            eta_trial = eta
            y_max = fast_autocorr(y).max()
            found = False
            for _ in range(8):
                z = y - eta_trial * grad
                x_new = simplex_project(z)
                m_new = fast_autocorr(x_new).max()
                if m_new <= y_max + 1e-14:
                    found = True
                    break
                eta_trial *= 0.5
            if not found:
                x_new = x
                m_new = fast_autocorr(x).max()
                eta = max(eta_trial * 0.5, 1e-7)

        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.03, 2.0)
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_mom = t_new

        if t % 30 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > 600:
            stall = 0
            t_mom = 1.0
            strat = rng.integers(7)
            if strat == 0:
                pert = rng.normal(0, 0.012, n) * best_a
                x_r = best_a + pert
            elif strat == 1:
                # Smooth then blend
                k_smooth = max(3, n // 300)
                kernel = np.ones(k_smooth) / k_smooth
                smooth_a = np.convolve(best_a, kernel, mode='same')
                x_r = 0.8 * best_a + 0.2 * smooth_a
            elif strat == 2:
                x_r = best_a * np.exp(0.015 * rng.standard_normal(n))
            elif strat == 3:
                shift = int(rng.integers(1, max(2, n // 30)))
                x_r = np.roll(best_a, shift)
            elif strat == 4:
                mask = rng.random(n) > 0.8
                x_r = best_a.copy()
                x_r[mask] *= 0.5
                x_r[~mask] *= 1.05
            elif strat == 5:
                x_r = 0.85 * best_a + 0.15 * rng.dirichlet(np.ones(n) * 0.1)
            else:
                # Local Gaussian bump
                center = int(rng.integers(n))
                width = int(rng.integers(10, max(11, n // 50)))
                xs = np.arange(n)
                bump = np.exp(-((xs - center) / width) ** 2)
                x_r = 0.9 * best_a + 0.1 * bump
                x_r = np.maximum(x_r, 0)

            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.4, 15.0)
            eta = eta_init
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    PS restart t=%d (%.1fs): J=%.6f b=%.0f" % (t, elapsed, J, beta))

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PS t=%d (%.1fs): J=%.6f b=%.0f" % (t, elapsed, J, beta))

    return best_a, best_m

def spectral_equalize(a0, time_limit, alpha_start=0.05, verbose=True, seed=123):
    """
    Iterative spectral equalization: flatten power spectrum to reduce
    autocorrelation peaks. Projects back to non-negative simplex.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    alpha = alpha_start

    for it in range(1, 800000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Geometric mean of non-DC power as target
        log_p = np.log(np.maximum(power[1:], 1e-30))
        target = np.exp(np.mean(log_p))
        if not np.isfinite(target) or target < 1e-30:
            target = power[1:].mean() + 1e-10

        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.0:
                red = 1.0 - alpha * min(ratio - 1.0, 3.0)
                A_new[k] *= max(0.15, np.sqrt(red))
            elif ratio < 0.2:
                inc = 1.0 + alpha * 0.4 * (0.2 - ratio)
                A_new[k] *= min(1.8, np.sqrt(inc))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            alpha = max(0.005, alpha * 0.9)
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 300 == 0:
            # Phase perturbation
            A_cur = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A_cur)) * 0.02
            A_cur[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A_cur, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial

        if it % 800 == 0:
            alpha = max(0.003, alpha * 0.97)

        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SpecEq it=%d (%.1fs): J=%.6f" % (it, elapsed, J))

    return best_a, best_m

def coordinate_descent(a0, time_limit, verbose=True, seed=789):
    """
    Targeted coordinate descent: for each peak lag, move mass from
    high-contributing pairs to reduce the peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 8000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        size = 2 * n - 1
        top_idx = np.argsort(ac)[::-1][:min(8, size)]
        improved = False

        for k in top_idx:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            if not (0 <= partner < n):
                continue

            for frac in [0.003, 0.008, 0.015, 0.025, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                for attempt in range(12):
                    dst = int(rng.integers(n))
                    if dst == src or dst == partner:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 150:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 2000 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    CD (%.1fs): J=%.6f stall=%d" % (elapsed, J, stall))

    return best_a, best_m

def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2000,
          add_per_iter=20, verbose=True):
    """LP-Frank-Wolfe with warm start and adaptive line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 400

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for kk in top_v:
                active_set.add(int(kk))

        # Line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        for alpha in [best_alpha, best_alpha * 0.5, best_alpha * 2.0, 0.05, 0.1, 0.2, 0.3]:
            alpha = max(0.01, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW it=%d (%.1fs): J=%.6f |S|=%d" % (t, elapsed, J, len(active_set)))

    return best_a, best_m

def multi_start_init(n, parent=None, seed=42):
    """Generate diverse initial sequences for multi-start."""
    rng = np.random.default_rng(seed)
    candidates = []
    x = np.linspace(0, 1, n)

    if parent is not None and len(parent) == n:
        candidates.append(norm_sum1(parent))

    # Sine squared
    candidates.append(norm_sum1(np.sin(np.pi * x) ** 2))
    # Raised cosine
    candidates.append(norm_sum1(0.5 * (1 + np.cos(2 * np.pi * x))))
    # Hann^2
    candidates.append(norm_sum1((0.5 * (1 + np.cos(2 * np.pi * x))) ** 2))
    # Random Dirichlet
    candidates.append(rng.dirichlet(np.ones(n) * 0.3))
    # Chirp magnitude * envelope
    chirp_env = np.sin(np.pi * x) * (1.0 + 0.3 * np.cos(4 * np.pi * x))
    candidates.append(norm_sum1(np.maximum(chirp_env, 0)))
    # Two bumps
    b1 = np.exp(-((x - 0.35) / 0.15) ** 2)
    b2 = np.exp(-((x - 0.65) / 0.15) ** 2)
    candidates.append(norm_sum1(b1 + b2))
    # Flat + noise
    candidates.append(norm_sum1(np.ones(n) + 0.05 * rng.standard_normal(n)))
    # Triangular
    candidates.append(norm_sum1(np.maximum(1 - 2 * np.abs(x - 0.5), 0)))

    return candidates

def try_update(best_seq, best_score, a, label, t_global):
    """Try to update the best sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [%s]: %.6f (n=%d, t=%.1fs)" % (label, sc, len(a), elapsed))
    return best_seq, best_score

def propose_candidate(budget_s=1800.0):
    """
    Multi-phase optimization:
    P1: Multi-start initialization
    P2: Coarse-to-fine (optimize at n/4 -> n/2 -> n)
    P3: Peak shaving (main optimizer, long run)
    P4: Spectral equalization
    P5: Coordinate descent polish
    P6: LP-Frank-Wolfe
    P7: Multi-length exploration
    P8: Final peak shaving + polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1755.0)
    solver = get_solver()
    print("Time budget: %.1fs" % time_budget)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(norm_sum1(parent))
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=%d, score=%.6f" % (len(parent), p_score))
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=%d, score=%.6f" % (len(best_seq), best_score))

    n0 = len(best_seq)
    print("Start: n=%d, J=%.6f" % (n0, best_score))

    # Phase 1: Multi-start at coarser scale
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Multi-start coarse ===")
    if remaining > 60:
        n_coarse = max(500, n0 // 4)
        t1 = min(remaining * 0.04, 60.0)
        candidates = multi_start_init(n_coarse, seed=42)
        best_cand = None
        best_cand_m = float('inf')
        for ci, cand in enumerate(candidates):
            m_c = fast_autocorr(cand).max()
            if m_c < best_cand_m:
                best_cand_m = m_c
                best_cand = cand
        if best_cand is not None:
            # Quick peak shaving at coarse scale
            t1_sub = t1 * 0.6
            if t1_sub > 10:
                a_out, _ = peak_shaving(best_cand, t1_sub, beta_start=30,
                                        beta_end=500, eta_init=0.5,
                                        top_k_ratio=0.03, verbose=False, seed=1)
                a_interp = interp_seq(a_out, n0)
                best_seq, best_score = try_update(best_seq, best_score, a_interp, "coarse", t_global)

    # Phase 2: Peak shaving at base n (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P2: Peak shaving main (n=%d) ===" % n_best)
    t2 = min(remaining * 0.28, 450.0)
    if t2 > 60:
        a_out, _ = peak_shaving(norm_sum1(best_seq), t2,
                                beta_start=40.0, beta_end=2000.0,
                                beta_growth=1.004, eta_init=0.4,
                                top_k_ratio=0.015, verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "PS1", t_global)

    # Phase 3: Spectral equalization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P3: Spectral equalize (n=%d) ===" % n_best)
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 30:
        a_out, _ = spectral_equalize(norm_sum1(best_seq), t3,
                                     alpha_start=0.05, verbose=True, seed=77)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "SE", t_global)

    # Phase 4: Peak shaving round 2 (different params)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P4: Peak shaving r2 (n=%d) ===" % n_best)
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 40:
        a_out, _ = peak_shaving(norm_sum1(best_seq), t4,
                                beta_start=100.0, beta_end=5000.0,
                                beta_growth=1.005, eta_init=0.2,
                                top_k_ratio=0.01, verbose=True, seed=99)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "PS2", t_global)

    # Phase 5: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P5: Coord descent (n=%d) ===" % n_best)
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 20:
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t5, verbose=True, seed=321)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "CD", t_global)

    # Phase 6: LP-Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P6: LP-FW (n=%d) ===" % n_best)
    if n_best <= 10000:
        t6 = min(remaining * 0.12, 180.0)
        if t6 > 30:
            a_out, _ = lp_fw(norm_sum1(best_seq), t6, solver,
                             initial_K=350, max_K=2500, add_per_iter=25, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "LP1", t_global)

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P7: Multi-length (base n=%d) ===" % n_best)

    # Try slightly different lengths
    length_candidates = []
    for delta in [-200, -100, 100, 200, 400, 700]:
        new_n = n_best + delta
        if 1000 <= new_n <= 12000:
            length_candidates.append(new_n)
    # Deduplicate and sort by distance from n_best
    length_candidates = sorted(set(length_candidates), key=lambda x: abs(x - n_best))

    for new_n in length_candidates[:5]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print("  -> n=%d" % new_n)
        a_long = interp_seq(best_seq, new_n)

        t7a = min(remaining * 0.30, 120.0)
        if t7a > 20:
            a_out, _ = peak_shaving(a_long, t7a, beta_start=50,
                                    beta_end=1500, eta_init=0.3,
                                    top_k_ratio=0.02, verbose=False,
                                    seed=new_n % 7919)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "n%d" % new_n, t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 9000:
            t7b = min(remaining * 0.30, 80.0)
            if t7b > 15:
                a_long2 = interp_seq(best_seq, new_n)
                a_out, _ = spectral_equalize(a_long2, t7b, alpha_start=0.04,
                                             verbose=False, seed=new_n % 3163)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "SE_n%d" % new_n, t_global)

    # Phase 8: Final peak shaving + coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P8: Final polish (n=%d) ===" % n_best)
    if remaining > 40:
        t8a = min(remaining * 0.40, 200.0)
        if t8a > 20:
            a_out, _ = peak_shaving(norm_sum1(best_seq), t8a,
                                    beta_start=200.0, beta_end=8000.0,
                                    beta_growth=1.006, eta_init=0.1,
                                    top_k_ratio=0.008, verbose=True, seed=555)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "PS_final", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8b = min(remaining * 0.35, 100.0)
            if t8b > 15:
                a_out, _ = coordinate_descent(norm_sum1(best_seq), t8b, verbose=False, seed=654)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "CD_final", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20 and n_best <= 9000:
            t8c = min(remaining * 0.40, 100.0)
            if t8c > 15:
                a_out, _ = lp_fw(norm_sum1(best_seq), t8c, solver,
                                 initial_K=400, max_K=3000, add_per_iter=30, verbose=True)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "LP_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: J=%.6f, n=%d, time=%.1fs" % (best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
