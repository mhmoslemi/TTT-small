def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto probability simplex."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_to_n(f, new_n):
    """Resample a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_pressure(a, top_k=15):
    """Compute pressure on each position from top autocorrelation peaks."""
    n = len(a)
    b = fast_autocorr(a)
    peak_indices = np.argsort(b)[::-1][:top_k]
    pressure = np.zeros(n)
    for k_int in peak_indices:
        k_i = int(k_int)
        i_start = max(0, k_i - n + 1)
        i_end = min(n - 1, k_i)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            pressure[idx] += a[k_i - idx]
    return pressure, peak_indices

def proximal_gradient_smoothing(a0, time_limit, beta_start=10.0, beta_end=500.0,
                                 momentum=0.85, verbose=True):
    """Proximal gradient on log-sum-exp smoothed max autocorrelation."""
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    v = np.zeros(n)
    stall = 0
    rng = np.random.default_rng(42)
    top_k = max(20, int(n * 0.03))
    t_iter = 0

    while time.monotonic() - t0 < time_limit:
        t_iter += 1
        b = fast_autocorr(x)
        idx_top = np.argsort(b)[::-1][:top_k]
        b_max = b[idx_top].max()

        # Softmax weights for the top peaks
        shifted = beta * (b[idx_top] - b_max)
        exp_vals = np.exp(np.clip(shifted, -300, 0))
        p = exp_vals / exp_vals.sum()

        # Build full-length weight vector
        w = np.zeros(2 * n - 1)
        w[idx_top] = p

        # Gradient of smoothed max: sum_k w[k] * grad(b[k])/grad(b_max)
        # grad of b[k] = sum_i a[i]*a[k-i] w.r.t a[j] = a[k-j] + a[j] (adjusted for bounds)
        # Use FFT: grad = 2 * (w-reversed convolved with a)
        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(x, fft_size)
        W = np.fft.rfft(w, fft_size)
        # The gradient of sum_k w[k]*b[k] = sum_k w[k]*sum_i a[i]*a[k-i]
        # = sum_i a[i] * sum_k w[k]*a[k-i] + (symmetric term)
        # In FFT: gradient ~ 2 * ifft(A * W) truncated
        grad = np.fft.irfft(A * W, fft_size)[:n]
        # Actually need to be more careful. Let's use:
        # d/dx[j] sum_k w[k] * sum_i x[i]*x[k-i]
        # = sum_k w[k] * (x[k-j] + x[j]) for valid indices
        # This is 2 * (w-reversed autocorrelated with x) evaluated at j
        # Simpler: grad = 2 * ifft(fft(x) * fft(w_padded))[:n]
        grad = 2.0 * grad

        # Scale
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Step size
        eta = 1.0 / (beta * (t_iter + 1))
        eta = min(eta * 2.0, 0.03)
        eta = max(eta, 1e-7)

        # Heavy-ball momentum
        v = momentum * v - eta * grad
        x_new = x + v
        x_new = simplex_project(x_new)

        # Evaluate
        b_new = fast_autocorr(x_new)
        m_new = b_new.max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = x_new.copy()
            x = x_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Backtrack: go back to best and add noise
                x = best_a.copy() + rng.normal(0, 0.01, n) * best_a
                x = np.maximum(x, 1e-12)
                x = x / x.sum()
                v = np.zeros(n)
                stall = 0
                beta = max(5.0, beta * 0.5)

        # Adapt beta

            beta = min(beta * 1.02, beta_end)
            momentum = min(0.92, 0.85 + 0.0001 * t_iter)

        if verbose and t_iter % 500 == 0:

            J = 2.0 * n * best_m
            print("    ProxGrad t=" + str(t_iter) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    return best_a, best_m


def frank_wolfe_refine(a0, time_limit, solver, verbose=True):
    """Frank-Wolfe / conditional gradient for pollution-aware refinement."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    t_iter = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 500:
        t_iter += 1
        b = fast_autocorr(a)
        M_curr = b.max()
        active_set = set(int(k) for k in np.argsort(b)[::-1][:600])

        constraint_list = sorted(active_set)
        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix: for each k in active_set, coeff[i] = a[k-i] for valid i
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = a[k_int - idx]

        # Solve LP: minimize M s.t. sum(g)=1, g>=0, coeffs@g <= M
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, cp.multiply(coeffs, g_var).sum(axis=1) <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            no_improve += 1
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 1
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            no_improve += 1
            continue
        g_dir = g_val / g_sum

        # Line search
        alphas = [2.0 / (t_iter + 2.0), 1.0 / (t_iter + 1), 0.1, 0.05]
        best_alpha = 2.0 / (t_iter + 2.0)
        best_trial = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * a + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial:
                best_trial = M_trial
                best_alpha = alpha

        a_new = (1.0 - best_alpha) * a + best_alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        st = a_new.sum()
        if st > 1e-15:
            a_new = a_new / st
        a = a_new

        M_new = fast_autocorr(a).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and t_iter % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    FW iter=" + str(t_iter) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    return best_a, best_m


def simulated_annealing_search(a0, time_limit, verbose=True):
    """Simulated annealing with autocorrelation-aware moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    T = 0.01  # initial temperature
    T_min = 1e-6
    cooling = 0.999
    t_iter = 0

    while time.monotonic() - t0 < time_limit:
        t_iter += 1

        # Pick a random move: transfer mass between two positions
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue

        # Pick amount to transfer
        amt = a[i] * rng.uniform(0.001, 0.05)
        if amt < 1e-15 or a[i] < amt:
            amt = a[i] * 0.01
        if amt < 1e-15:
            continue

        a_trial = a.copy()
        a_trial[i] -= amt
        a_trial[j] += amt
        a_trial = np.maximum(a_trial, 0.0)
        st = a_trial.sum()
        if st < 1e-15:
            continue
        a_trial = a_trial / st

        m_trial = fast_autocorr(a_trial).max()
        delta = m_trial - a_m

        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            a = a_trial
            a_m = m_trial
            if m_trial < best_m:
                best_m = m_trial
                best_a = a.copy()

        T = max(T * cooling, T_min)

        # Periodic restart from best
        if t_iter % 500 == 0:
            a = best_a.copy() + rng.normal(0, 0.005, n) * best_a
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            a_m = fast_autocorr(a).max()

        if verbose and t_iter % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SA t=" + str(t_iter) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " T=" + repr(T))

    return best_a, best_m


def multi_scale_refinement(a0, time_limit, verbose=True):
    """Refine at multiple scales: coarse to fine."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Coarse: optimize in blocks of 8
    for block_size in [16, 8, 4, 2, 1]:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        n_blocks = (n + block_size - 1) // block_size
        rng = np.random.default_rng(42 + block_size)

        # Repeat passes
        for _ in range(3):
            if time.monotonic() - t0 > time_limit - 5:
                break
            for bi in range(n_blocks):
                if time.monotonic() - t0 > time_limit - 5:
                    break
                s = bi * block_size
                e = min((bi + 1) * block_size, n)
                block = a[s:e].copy()
                block_sum = block.sum()
                if block_sum < 1e-15:
                    continue

                # Try redistributing within block to minimize local contribution to peaks
                b = fast_autocorr(a)
                # Find which peaks this block contributes to most
                top_peaks = np.argsort(b)[::-1][:10]
                block_contrib = 0.0
                for k_i in top_peaks:
                    k_int = int(k_i)
                    ks = max(s, k_int - n + 1)
                    ke = min(e, k_int + 1)
                    if ks < ke:
                        idx = np.arange(ks, ke)
                        block_contrib += (a[idx] * a[k_int - idx]).sum()

                # Flattening: move mass from high to low within block
                if block.max() > block.mean() * 1.5:
                    flat = np.full(e - s, block_sum / (e - s))
                    a_trial = a.copy()
                    # Blend
                    blend = 0.3
                    a_trial[s:e] = (1 - blend) * block + blend * flat
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st > 1e-15:
                        a_trial = a_trial / st
                        m_trial = fast_autocorr(a_trial).max()
                        if m_trial < best_m - 1e-15:
                            a = a_trial
                            best_m = m_trial
                            best_a = a.copy()

    return best_a, best_m


def parametric_init(n, rng):
    """Generate diverse parametric initializations."""
    x = np.linspace(0.0, 1.0, n)
    candidates = []

    # Power law
    a = (x + 0.01) ** rng.uniform(0.2, 3.0) * (1.0 - x + 0.01) ** rng.uniform(0.2, 3.0)
    candidates.append(a)

    # Gaussian mixture
    c1, c2 = rng.uniform(0.1, 0.4), rng.uniform(0.6, 0.9)
    s1, s2 = rng.uniform(0.03, 0.2), rng.uniform(0.03, 0.2)
    w = rng.uniform(0.5, 1.5)
    a = np.exp(-((x - c1) ** 2) / (2 * s1 ** 2)) + w * np.exp(-((x - c2) ** 2) / (2 * s2 ** 2))
    candidates.append(a)

    # Polynomial
    p = rng.choice([2, 3, 4, 5, 7])
    a = 1.0 + x ** p
    candidates.append(a)

    # Random
    a = rng.random(n) ** rng.uniform(0.3, 4.0)
    candidates.append(a)

    # Triangular (peaked in middle)
    a = 1.0 - np.abs(2 * x - 1)
    candidates.append(a)

    # Cosine bump
    a = np.cos(np.pi * (x - 0.5)) ** 2
    candidates.append(a)

    best_a = None
    best_m = float('inf')
    for cand in candidates:
        a_norm = norm_sum1(cand)
        m = fast_autocorr(a_norm).max()
        if m < best_m:
            best_m = m
            best_a = a_norm

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global, verbose=True):
    """Try to update best with a new candidate."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        if verbose:
            elapsed = time.monotonic() - t_global
            print("  *** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search procedure with multi-phase optimization."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_solver()

    print("Time budget: " + repr(time_budget) + "s, solver: " + str(solver))

    best_seq = None
    best_score = float('inf')
    rng = np.random.default_rng(2024)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        pass

    if best_seq is None:
        # Generate initial
        n_base = 6700
        a_init, _ = parametric_init(n_base, rng)
        best_seq = a_init
        best_score = evaluate_safe(best_seq)
        print("Initial: n=" + str(n_base) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # === Phase 1: Quick parametric sweep at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t1 = min(remaining * 0.02, 30.0)
        print("\n=== Phase 1: Parametric sweep (n=" + str(n0) + ") ===")
        for _ in range(5):
            a_init, m = parametric_init(n0, rng)
            best_seq, best_score = try_update(best_seq, best_score, a_init, "param_n" + str(n0), t_global)
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break

    # === Phase 2: Peak shaving (round 1) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        t2 = min(remaining * 0.12, 180.0)
        print("\n=== Phase 2: Peak shaving (n=" + str(n_best) + ") ===")
        a_out, _ = peak_shaving_v2(best_seq, t2, top_k=20, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps1_n" + str(n_best), t_global)

    # === Phase 3: Proximal gradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 100:
        t3 = min(remaining * 0.25, 350.0)
        print("\n=== Phase 3: Proximal gradient (n=" + str(n_best) + ") ===")
        a_out, _ = proximal_gradient_smoothing(best_seq, t3, beta_start=8.0, beta_end=400.0,
                                                momentum=0.85, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pg_n" + str(n_best), t_global)

    # === Phase 4: Peak shaving (round 2) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 50:
        t4 = min(remaining * 0.10, 150.0)
        print("\n=== Phase 4: Peak shaving 2 (n=" + str(n_best) + ") ===")
        a_out, _ = peak_shaving_v2(best_seq, t4, top_k=15, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps2_n" + str(n_best), t_global)

    # === Phase 5: Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 80 and n_best <= 8000:
        t5 = min(remaining * 0.15, 250.0)
        print("\n=== Phase 5: Frank-Wolfe (n=" + str(n_best) + ") ===")
        a_out, _ = frank_wolfe_refine(best_seq, t5, solver, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fw_n" + str(n_best), t_global)

    # === Phase 6: Multi-scale refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40:
        t6 = min(remaining * 0.08, 100.0)
        print("\n=== Phase 6: Multi-scale (n=" + str(n_best) + ") ===")
        a_out, _ = multi_scale_refinement(best_seq, t6, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ms_n" + str(n_best), t_global)

    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    candidates_n = []
    for dn in [-600, -400, -200, 100, 300, 500, 700, 1000]:
        nl = n_best + dn
        if 4000 <= nl <= 9000:
            candidates_n.append(nl)

    for new_n in candidates_n[:5]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("\n=== Phase 7: n=" + str(new_n) + " ===")
        a_long = interpolate_to_n(best_seq, new_n)

        # Quick peak shaving
        t7a = min(remaining * 0.2, 80.0)
        if t7a > 20:
            a_out, _ = peak_shaving_v2(a_long, t7a, top_k=15, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ps_n" + str(new_n), t_global)

        # Quick proximal gradient
        remaining = time_budget - (time.monotonic() - t_global)
        t7b = min(remaining * 0.3, 120.0)
        if t7b > 30:
            a_out, _ = proximal_gradient_smoothing(best_seq, t7b, beta_start=15.0, beta_end=300.0,
                                                    momentum=0.88, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "pg_n" + str(new_n), t_global)

    # === Phase 8: Simulated annealing (short burst) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 80:
        t8 = min(remaining * 0.1, 100.0)
        print("\n=== Phase 8: Simulated annealing (n=" + str(n_best) + ") ===")
        a_out, _ = simulated_annealing_search(best_seq, t8, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sa_n" + str(n_best), t_global)

    # === Phase 9: Final polish cycle ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print("\n=== Phase 9: Final polish (n=" + str(n_best) + ") ===")
        # Peak shave
        t9a = min(remaining * 0.3, 100.0)
        if t9a > 15:
            a_out, _ = peak_shaving_v2(best_seq, t9a, top_k=12, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_ps", t_global)

        # Proximal gradient
        remaining = time_budget - (time.monotonic() - t_global)
        t9b = min(remaining * 0.3, 80.0)
        if t9b > 20:
            a_out, _ = proximal_gradient_smoothing(best_seq, t9b, beta_start=30.0, beta_end=500.0,
                                                    momentum=0.9, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_pg", t_global)

        # FW
        remaining = time_budget - (time.monotonic() - t_global)
        t9c = min(remaining * 0.3, 80.0)
        if t9c > 20 and n_best <= 8000:
            a_out, _ = frank_wolfe_refine(best_seq, t9c, solver, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_fw", t_global)

    # === Phase 10: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t10 = min(remaining * 0.5, 60.0)
        a_out, _ = peak_shaving_v2(best_seq, t10, top_k=10, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "final_ps2", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        a_raw = (x + 0.01) ** 1.0 * (1.0 - x + 0.01) ** 1.0
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
