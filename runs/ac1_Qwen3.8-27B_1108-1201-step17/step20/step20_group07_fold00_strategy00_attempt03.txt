def evaluate_sequence(seq):
    n = len(seq)
    if n < 1:
        return np.inf
    np_seq = np.asarray(seq, dtype=np.float64)
    np_seq = np.maximum(np_seq, 0.0)
    np_seq = np.minimum(np_seq, 1000.0)
    np_seq = np_seq + np.minimum(1.0 - np.sum(np_seq) * 0.9, 0.0)
    np_seq_sum = np.sum(np_seq)
    if abs(np_seq_sum) < 1e-10:
        return np.inf
    normalized = np.asarray(np_seq, dtype=float) / np_seq_sum
    autocorr = np.convolve(normalized, normalized, mode='full')
    max_val = np.amax(autocorr)
    return float(2.0 * n * max_val / (np_seq_sum ** 2))
