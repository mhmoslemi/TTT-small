        elif start_i == 1:
            a_init = make_initial(best_n, variant=0)
        elif start_i == 2:
            a_init = make_initial(best_n, variant=1)
        elif start_i == 3:
            a_init = make_initial(best_n, variant=5)
        else:
            a_init = make_initial(best_n, variant=2)

        if parent is not None and start_i < 2:
            a_init = 0.5 * normalize(best_n_seq) + 0.5 * a_init
            a_init = normalize(a_init)

        a_out, sc = lp_fw(a_init, t_per_start, solver,
                          initial_K=400, max_K=3500, verbose=(start_i == 0))
        try_update(a_out, "lfw_s{}_n{}".format(start_i, best_n))

    # Phase 3: Try a few other n values with LP-FW
    print("\n=== Phase 3: Other n values ===")
    rt = time_budget - (time.monotonic() - t_global)
    other_ns = [nn for nn in n_candidates if nn != best_n and 4000 <= nn <= 11000]
    other_ns.sort(key=lambda x: abs(x - best_n))

    for nn in other_ns[:4]:
        rt = time_budget - (time.monotonic() - t_global)
        if rt < 120:
            break
        a_init = interp_to_n(best_n_seq, nn)
        t_lp = min(rt * 0.15, 250)
        if t_lp < 40:
            continue
        a_out, sc = lp_fw(a_init, t_lp, solver,
                          initial_K=350, max_K=3000, verbose=True)
        try_update(a_out, "lfw_n{}".format(nn))

    # Phase 4: Long LP-FW on current best
    print("\n=== Phase 4: Long LP-FW polish ===")
    rt = time_budget - (time.monotonic() - t_global)
    if rt > 100 and best_seq is not None:
        n_best = len(best_seq)
        t_lp = min(rt * 0.30, 400)
        if t_lp > 60:
            a_out, sc = lp_fw(best_seq, t_lp, solver,
                              initial_K=500, max_K=4000, verbose=True)
            try_update(a_out, "lfw_long_n{}".format(n_best))

    # Phase 5: Coordinate polish
    print("\n=== Phase 5: Coordinate polish ===")
    rt = time_budget - (time.monotonic() - t_global)
    if rt > 60 and best_seq is not None:
        t_cp = min(rt * 0.20, 250)
        if t_cp > 30:
            a_out, sc = coord_polish(best_seq, t_cp, verbose=True)
            try_update(a_out, "coord_n{}".format(len(best_seq)))

    # Phase 6: Final LP-FW
    print("\n=== Phase 6: Final LP-FW ===")
    rt = time_budget - (time.monotonic() - t_global)
    if rt > 80 and best_seq is not None:
        n_best = len(best_seq)
        t_lp = min(rt * 0.35, 350)
        if t_lp > 40:
            a_out, sc = lp_fw(best_seq, t_lp, solver,
                              initial_K=600, max_K=4500, verbose=True)
            try_update(a_out, "lfw_final_n{}".format(n_best))

    # Phase 7: Final coordinate polish
    rt = time_budget - (time.monotonic() - t_global)
    if rt > 30 and best_seq is not None:
        a_out, sc = coord_polish(best_seq, min(rt * 0.6, 100), verbose=False)
        try_update(a_out, "coord_final")

    # Final validation
    if best_seq is None:
        best_seq = make_initial(6000, variant=0)
        best_score = eval_score(best_seq)

    fs = eval_score(best_seq)
    if np.isfinite(fs):
        best_score = fs

    print("\nFINAL: score={:.8f}, n={}, time={:.0f}s".format(
        best_score, len(best_seq), time.monotonic() - t_global))

    return [float(x) for x in best_seq]
