def logsumexp_gradient_step(a, beta):
    n = len(a)
    # Compute auto-convolution
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    
    # Compute weights (log-sum-exp trick)
    c_max = c.max()
    w = np.exp(beta * (c - c_max))
    Z = w.sum()
    u = w / Z  # normalized weights
    
    # Compute gradient: grad[j] = (1/beta) * sum_m a[m] * u[m+j]
    # This is (a ⋆ u)[j] / beta
    # Using FFT: IFFT(conj(FFT(a)) * FFT(u))[:n]
    fft_size2 = 1 << (3*n - 2).bit_length()  # need at least 2n-1 + n - 1 = 3n-2
    A2 = np.fft.rfft(a, fft_size2)
    U2 = np.fft.rfft(u, fft_size2)
    cross = np.fft.irfft(np.conj(A2) * U2, fft_size2)
    grad = cross[:n] / beta
    
    return grad, c.max(), Z
